initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h

    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    swarm_size = 30
    max_iter = 150
    perturbation_decay = 0.95
    initial_perturbation = 0.2
    refinement_steps = 10
    n_refinement = 3
    n_sub_swarm = 5
    initial_temperature = 1.0
    cooling_rate = 0.995
    min_temperature = 1e-6
    local_search_steps = 10

    # Initialize a swarm of candidate solutions
    swarm = [h_norm.copy() for _ in range(swarm_size)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Function to evolve swarm with a balance of exploration and exploitation
    def evolve_swarm(swarm, best_solution, best_c5, initial_perturbation, perturbation_decay, max_iter, n_points, dx, compute_overlap, integral_constraint, initial_temperature, cooling_rate, min_temperature, local_search_steps):
        current_temperature = initial_temperature
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Global exploration phase with random perturbation
            for i in range(swarm_size):
                perturbation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
                candidate = np.clip(swarm[i] + perturbation, 0, 1)

                candidate = candidate * (n_points / 2) / sum(candidate)
                # Local refinement with simulated annealing
                current_c5 = compute_overlap(candidate)
                for _ in range(local_search_steps):
                    perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
                    new_candidate = np.clip(candidate + perturbation, 0, 1)
                    new_candidate = new_candidate * (n_points / 2) / sum(new_candidate)
                    new_c5 = compute_overlap(new_candidate)
                    delta = new_c5 - current_c5
                    if delta < 0 or np.random.rand() < np.exp(-delta / current_temperature):
                        candidate = new_candidate
                        current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_solution = candidate
            
            # Local refinement phase with smaller perturbations
            for i in range(swarm_size):
                for _ in range(n_refinement):
                    perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
                    candidate = np.clip(swarm[i] + perturbation, 0, 1)
                    # Ensure integral constraint is satisfied
                    candidate = candidate * (n_points / 2) / sum(candidate)
                    # Local refinement with simulated annealing
                    current_c5 = compute_overlap(candidate)
                    for _ in range(local_search_steps):
                        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
                        new_candidate = np.clip(candidate + perturbation, 0, 1)
                        new_candidate = new_candidate * (n_points / 2) / sum(new_candidate)
                        new_c5 = compute_overlap(new_candidate)
                        delta = new_c5 - current_c5

                            candidate = new_candidate
                            current_c5 = new_c5
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = candidate
            
            # Gradually reduce perturbation range and temperature
            initial_perturbation *= perturbation_decay
            current_temperature *= cooling_rate
            if current_temperature < min_temperature:
                current_temperature = min_temperature
        
        return best_solution, best_c5
    
    # Run the optimization
    best_solution, best_c5 = evolve_swarm(swarm, best_solution, best_c5, initial_perturbation, perturbation_decay, max_iter, n_points, dx, compute_overlap, integral_constraint, initial_temperature, cooling_rate, min_temperature, local_search_steps)
    
    return (best_solution.tolist(), best_c5, n_points)
