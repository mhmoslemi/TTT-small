        start_time = time.time()
        
        # Create evenly spaced points
        x = np.linspace(0, 2, n_points)

        # Function to compute C5
        def compute_c5(h):
            h = np.clip(h, 0, 1)
            h = np.concatenate([h, [0] * (n_points - 1)])
            corr = correlate(h, 1 - h, mode="full")
            c5 = np.max(corr * dx)
            return c5
        
        # Objective function for optimization
        def objective(h):
            return compute_c5(h)
        
        # Constraints: sum(h) == n_points / 2
        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        
        # Bounds: 0 <= h[i] <= 1
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Initial guess
        if n_points == n_initial:
            h0 = h_initial
        else:
            h0 = np.random.uniform(0, 1, n_points)
        
        # Optimization
        result = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        
        # Check if within time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Update best solution
        best_h = result.x
        best_c5 = compute_c5(best_h)
        print(f"n_points={n_points}, C5={best_c5}")
