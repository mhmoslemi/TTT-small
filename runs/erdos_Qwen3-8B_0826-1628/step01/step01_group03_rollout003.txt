import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    # Normalize to ensure the sum constraint is satisfied
    if sum_h != n_points / 2:
        initial_h = initial_h * (n_points / 2) / sum_h
    
    # Function to compute C5 value
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = np.concatenate([h, [0] * (n_points - 1)])
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Objective function for optimization
    def objective(h):
        return compute_c5(h)
    
    # Constraints: sum of h must be n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    h_initial = initial_h.copy()
    
    # Optimization
    start_time = time.time()
    result = opt.minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    # Check elapsed time
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        # If time is up, return the initial solution
        return (initial_h, compute_c5(initial_h), n_points)
    
    # Get the best solution
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n_points)
