import numpy as np

                             0.5, 0.5])

def evaluate_erdos_solution(h_values, dx):
    n_points = len(h_values)
    h = np.array(h_values)
    # Compute the overlap integrals for all lags
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integrals correspond to the first n_points + 1 values of the correlation
    overlap_integrals = corr[:n_points + 1] * dx
    # We need to consider the maximum of these integrals
    c5_bound = np.max(overlap_integrals)
    return c5_bound

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Define the number of points (adjust based on computational budget)
    n_points = 100  # Good balance between resolution and computation
    
    # Define the dx
    dx = 2.0 / n_points
    
    # Initialize h with a simple uniform distribution
    h_initial = np.full(n_points, 0.5)
    # Normalize to ensure the integral constraint
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Now perform a gradient descent optimization
    # Objective function: minimize the maximum overlap integral
    def objective(h_vec):
        h = h_vec.copy()
        # Ensure h stays within [0,1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integrals
        corr = np.correlate(h, 1 - h, mode="full")
        overlap_integrals = corr[:n_points + 1] * dx
        # Return the maximum overlap integral
        return np.max(overlap_integrals)
    
    # Define bounds for h
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization settings
    max_iter = 100
    tol = 1e-5
    
    # Optimization
    start_time = time.time()
    result = minimize(
        objective,
        x0=h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        tol=tol,
        options={'maxiter': max_iter, 'ftol': 1e-8, 'gtol': 1e-8, 'maxcor': 100}
    )
    
    # Ensure the solution stays within bounds
    h_values = np.clip(result.x, 0, 1)
    
    # Compute the final c5 bound
    c5_bound = evaluate_erdos_solution(h_values, dx)
    
    # Keep track of time and return
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        # If we've used up the budget, return the best solution found so far
        return h_values, c5_bound, n_points
    
    return h_values, c5_bound, n_points
