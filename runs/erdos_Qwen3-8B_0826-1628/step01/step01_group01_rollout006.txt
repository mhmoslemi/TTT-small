import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Predefined initial solution
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        shifted = np.zeros_like(h)
        for k in range(1, len(h)):
            shifted[k:] = h[:len(h) - k]
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Function to evaluate the objective for optimization
    def objective(h):
        return compute_c5(h)
    
    # Function to compute the constraint sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization setup
    def optimize_h(h_initial):
        h = cp.Variable(n_points)
        obj = cp.Minimize(objective(h))
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
        prob = cp.Problem(obj, constraints)
        start_time = time.time()
        result = prob.solve(solver=cp.ECOS, verbose=False)
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            return h_initial, compute_c5(h_initial)
        return h.value, compute_c5(h.value)
    
    # Start with the initial solution
    best_h, best_c5 = optimize_h(initial_h)
    
    # Try a few perturbations from the initial solution to avoid local optima
    for _ in range(5):
        perturbed_h = np.array(best_h)
        perturbed_h[np.random.choice(n_points, size=5, replace=False)] = np.random.uniform(0, 1, size=5)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_c5 = compute_c5(perturbed_h)
        if perturbed_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = perturbed_c5
    
    # Fine-tune the solution using local optimization
    def optimize_local(h):
        def objective_func(x):
            return compute_c5(x)
        result = opt.minimize(objective_func, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        return result.x, compute_c5(result.x)
    
    best_h, best_c5 = optimize_local(best_h)
    
    # Final fine-tuning with a few iterations of local optimization
    for _ in range(3):
        best_h, best_c5 = optimize_local(best_h)
    
    return best_h.tolist(), best_c5, n_points
