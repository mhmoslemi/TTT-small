import numpy as np

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros((n_points, n_points))
    for i in range(n_points):
        shifted_h[i, :] = h
    h_shifted = np.roll(shifted_h, shift=-i, axis=1) if i > 0 else shifted_h
    overlap = np.zeros(n_points)
    for k in range(n_points):
        overlap[k] = np.dot(h, 1 - h_shifted[k, :])
    c5 = np.max(overlap) * dx
    return c5

def initial_h_values(n_points=42):
    return np.array([0.0] * n_points)

def run(seed=42, budget_s=100, **kwargs):

    def update_best(h_vals):
        nonlocal best_c5, best_h
        c5 = evaluate_erdos_solution(h_vals)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_vals[:]
    
    # Try different numbers of points
    for n_points in [42, 64, 80, 100, 120, 150]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Initial estimate
        h = initial_h_values(n_points)
        h = h * (n_points / 2)  # Adjust to ensure sum(h) == n_points / 2

        update_best(h)
        
        # Try random perturbations
        for _ in range(10):

            h = h * (n_points / 2)  # Adjust to ensure sum(h) == n_points / 2

            h = h.copy()

            h = np.clip(h, 0, 1)
            for it in range(100):
                if time.time() - start_time > budget_s - 1:
                    break
                # Compute the C5 value
                c5 = evaluate_erdos_solution(h)

                grad = np.zeros(n_points)
                for k in range(n_points):
                    for i in range(n_points):
                        # Compute the derivative of the overlap for each k
                        # This is a simplified approximation
                        h_plus = h.copy()
                        h_plus[i] += 1e-4
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2)
                        h_plus = np.clip(h_plus, 0, 1)

                        h_minus = h_minus * (n_points / 2)

        gradient_descent()
