import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * 2.0 / len(h)
initial_h_values = np.random.rand(42) * 2 - 1  # Placeholder for real initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set initial parameters
    n_points = 100  # Start with 100 points, can be increased if needed
    dx = 2.0 / n_points
    h = np.random.rand(n_points)  # Initial guess
    
    # Set up the optimization problem
    h_var = cp.Variable(n_points, nonneg=True)
    constraints = [
        cp.sum(h_var) == n_points / 2,
        0 <= h_var,
        h_var <= 1
    ]
    
    # Define the objective function
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    time_used = 0
    
    for _ in range(100):  # Max 100 iterations
        if time_used >= budget_s - 1:
            break
        
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Create optimization problem with current h as initial guess
        objective = cp.Minimize(objective_function(h))
        prob = cp.Problem(objective, constraints)
        try:
            result = prob.solve(solver=cp.MOSEK, verbose=False)
            if result is not None and result == 0:
                h = h_var.value
            time_used = time.time() - start_time
        except:
            pass  # Skip if solver fails
        
        # Random perturbation to escape local minima
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, n_points
