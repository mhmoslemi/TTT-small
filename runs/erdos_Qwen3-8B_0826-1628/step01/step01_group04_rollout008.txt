import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for efficiency
    dx = 2.0 / n_points
    
    # Initial guess from provided starting point
    h_initial = initial_h_values.copy()
    h_initial = h_initial[:n_points]  # Ensure it fits the n_points
    
    # Normalize to satisfy the integral constraint
    sum_h = np.sum(h_initial)
    h_initial = h_initial * (n_points / 2) / sum_h
    
    # Define the objective function to minimize the maximum overlap integral
    def objective(h):
        h = np.array(h).reshape(n_points)
        # Ensure h is in [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        overlap = np.correlate(h, 1 - h, mode="full")
        # Integrate over the interval
        c5 = np.max(overlap * dx)
        return c5
    
    # Define the constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization setup
    h = cp.Variable(n_points)
    objective_func = cp.Minimize(objective(h))
    constraints = [constraint(h) == 0, 0 <= h, h <= 1]
    
    # Run optimization
    prob = cp.Problem(objective_func, constraints)
    start_time = time.time()
    
    try:
        result = prob.solve(solver=cp.SCS, max_iter=1000, time_limit=budget_s - (time.time() - start_time))
        best_h = h.value
        best_c5 = objective(best_h)
    except:
        best_h = h_initial
        best_c5 = objective(best_h)
    
    # Check if we can improve with more points
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        dx = 2.0 / n_points
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        h_initial = np.clip(h_initial, 0, 1)
        sum_h = np.sum(h_initial)
        h_initial = h_initial * (n_points / 2) / sum_h
        
        # Optimization with more points
        h = cp.Variable(n_points)
        objective_func = cp.Minimize(objective(h))
        constraints = [constraint(h) == 0, 0 <= h, h <= 1]
        
        prob = cp.Problem(objective_func, constraints)
        try:
            result = prob.solve(solver=cp.SCS, max_iter=1000, time_limit=budget_s - (time.time() - start_time))
            best_h = h.value
            best_c5 = objective(best_h)
        except:
            best_h = h_initial
            best_c5 = objective(best_h)
    
    # Final check for even more points if time allows
    if time.time() - start_time < budget_s - 10:
        n_points = 300
        dx = 2.0 / n_points
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        h_initial = np.clip(h_initial, 0, 1)
        sum_h = np.sum(h_initial)
        h_initial = h_initial * (n_points / 2) / sum_h
        
        # Optimization with even more points
        h = cp.Variable(n_points)
        objective_func = cp.Minimize(objective(h))
        constraints = [constraint(h) == 0, 0 <= h, h <= 1]
        
        prob = cp.Problem(objective_func, constraints)
        try:
            result = prob.solve(solver=cp.SCS, max_iter=1000, time_limit=budget_s - (time.time() - start_time))
            best_h = h.value
            best_c5 = objective(best_h)
        except:
            best_h = h_initial
            best_c5 = objective(best_h)
    
    # Return the best solution found
    return (best_h, best_c5, n_points)
