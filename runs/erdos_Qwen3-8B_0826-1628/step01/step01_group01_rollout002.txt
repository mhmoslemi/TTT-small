def run(seed=42, budget_s=100, **kwargs):
    random.seed(seed)
    start_time = time.time()
    best_h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    n_points = len(best_h)
    
    # Genetic algorithm parameters
    pop_size = 30
    generations = 100
    mutation_rate = 0.1
    elite_size = 5
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Generate population
        population = []
        for _ in range(pop_size):
            h = best_h.copy()
            for i in range(n_points):
                if random.random() < mutation_rate:
                    h[i] = np.clip(random.uniform(0, 1), 0, 1)
            population.append(h)
        
        # Evaluate population
        scores = []
        for h in population:
            # Ensure sum constraint
            sum_h = np.sum(h)
            if sum_h == 0:
                score = float('inf')
            elif abs(sum_h - n_points / 2) > 1e-6:
                score = float('inf')
            else:
                score = evaluate_erdos_solution(h)
            scores.append(score)
        
        # Select elites
        elite_indices = np.argsort(scores)[:elite_size]
        elites = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elites[:]
        while len(new_population) < pop_size:
            # Select parents
            p1 = random.choice(elite_indices)
            p2 = random.choice(elite_indices)
            h1 = population[p1]
            h2 = population[p2]

            crossover_point = random.randint(0, n_points - 1)
            child = h1[:crossover_point] + h2[crossover_point:]
            
            # Mutate
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] = np.clip(random.uniform(0, 1), 0, 1)
            
            # Ensure sum constraint
            sum_h = np.sum(child)

                child = best_h.copy()
            elif abs(sum_h - n_points / 2) > 1e-6:
                child = best_h.copy()
            else:
                child = np.clip(child, 0, 1)
            
            new_population.append(child)
        
        # Update best solution
        for h in new_population:
            sum_h = np.sum(h)
            if sum_h == 0:
                score = float('inf')

                score = float('inf')

            return float('inf')

            return float('inf')
        else:
            return evaluate_erdos_solution(h)

    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
