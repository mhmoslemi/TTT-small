```python
import numpy as np
import scipy.optimize as optimize
import math
import time

def run(seed=42, budget_s=100, **kwargs):

    def evaluate_c5(h_values):
        n_points = len(h_values)
        dx = 2.0 / n_points
        h = np.array(h_values)
        shifted = np.zeros(2 * n_points)
        shifted[n_points:] = 1 - h
        corr = np.correlate(h, shifted, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def initial_h_values(n_points):
        return np.repeat([0.5, 0.5], n_points // 2 + 1)[:n_points]

    def optimize_h(n_points, seed):
        h_initial = initial_h_values(n_points)
        np.random.seed(seed)
        h_initial += np.random.uniform(-0.1, 0.1, size=n_points)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        result = optimize.minimize(
            lambda h: evaluate_c5(h),
            h_initial,
            bounds=[(0, 1) for _ in range(n_points)],
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        return result.x

    def local_search(h, n_points, max_iter=100):
        for _ in range(max_iter):
            idx = np.random.randint(n_points)
            h_new = h.copy()
            h_new[idx] = np.clip(h_new[idx] + np.random.uniform(-0.1, 0.1), 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            if evaluate_c5(h_new) < evaluate_c5(h):
                h = h_new
        return h

    def simulate(n_points, seed):
        h = optimize_h(n_points, seed)
        h = local_search(h, n_points, max_iter=50)
        return h, evaluate_c5(h)

    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0

    for n_points in [50, 75, 100, 125, 150, 175, 200]:
        if time.time() - start_time > budget_s:
            break
        h, c5 = simulate(n_points, seed)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
            best_n_points = n_points

    return best_h, best_c5, best_n_points
```
