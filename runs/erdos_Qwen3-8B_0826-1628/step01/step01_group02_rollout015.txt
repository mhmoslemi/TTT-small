    def optimize_h(h_init):

        bounds = [(0.0, 1.0) for _ in range(n)]
        
        # Objective function (to minimize C5)
        def objective(x):
            return evaluate_c5(x)
        
        # Constraints: sum(x) == n / 2
        def constraint_sum(x):
            return np.sum(x) - n / 2
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        
        # Optimization
        result = opt.minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP')
        if result.success:
