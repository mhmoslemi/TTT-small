import numpy as np
import scipy.optimize as opt
import time
import copy

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros(2 * n_points)
    shifted_h[:n_points] = h
    shifted_h[n_points:] = h
    shifted_1_minus_h = 1 - shifted_h
    correlate = np.correlate(h, shifted_1_minus_h, mode="full")
    c5 = np.max(correlate) * dx
    return c5

def initial_h_values():
    # Initial guess with 47 samples (as per the problem statement)
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points (faster to optimize)
    
    # Initial guess
    h_initial = initial_h_values()
    h_initial = h_initial[:n_points]  # Trim to n_points
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)  # Normalize
    
    # Function to evaluate C5
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)  # Ensure values are within [0, 1]
        dx = 2.0 / len(h)
        shifted_h = np.zeros(2 * len(h))
        shifted_h[:len(h)] = h
        shifted_h[len(h):] = h
        shifted_1_minus_h = 1 - shifted_h
        correlate = np.correlate(h, shifted_1_minus_h, mode="full")
        c5 = np.max(correlate) * dx
        return c5

    # Function to compute the constraint sum(h) == n_points / 2
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2

    # Optimization setup
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    result = opt.minimize(objective, x0=h_initial, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
    best_h = result.x
    best_c5 = result.fun

    # Early stopping if time is up
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (best_h.tolist(), best_c5, n_points)

    # Try increasing the number of points to improve the bound
    for n_points in [200, 300, 400, 500]:
        # Initialize with previous solution
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)  # Normalize

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        
        # Optimizer settings
        result = opt.minimize(objective, x0=h_initial, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
            elapsed_time = time.time() - start_time
            if elapsed_time >= budget_s:
                break

    return (best_h.tolist(), best_c5, n_points)
