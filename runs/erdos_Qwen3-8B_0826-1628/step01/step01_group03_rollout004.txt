```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial parameters
    n_points = 100
    h = np.copy(initial_h_values[:n_points])
    h = h / np.sum(h) * (n_points / 2)
    h = np.clip(h, 0, 1)
    
    # Function to evaluate C5
    def evaluate(h):
        return evaluate_erdos_solution(h)
    
    # Function to optimize h
    def optimize_h(h, max_iter=200, tolerance=1e-5):
        for _ in range(max_iter):
            # Compute current C5
            current_c5 = evaluate(h)
            if current_c5 < tolerance:
                break
            
            # Use gradient descent with bounds
            x = cp.Variable(n_points)
            objective = cp.Minimize(current_c5)
            constraints = [
                cp.sum(x) == n_points / 2,
                0 <= x,
                x <= 1
            ]
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.SCS, eps=1e-6)
            
            # Update h
            h = np.clip(x.value, 0, 1)
            if np.isnan(h).any():
                break
        
        return h
    
    # Optimize the initial h
    h = optimize_h(h)
    
    # Refine with more points
    n_points = 200
    h = np.repeat(h, 2)
    h = h[:n_points]
    h = h / np.sum(h) * (n_points / 2)
    h = np.clip(h, 0, 1)
    
    # Fine-tune with more iterations
    h = optimize_h(h)
    
    # Final evaluation
    final_c5 = evaluate(h)
    elapsed_time = time.time() - start_time
    
    return (h, final_c5, n_points)
```
