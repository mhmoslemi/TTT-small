import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for speed

    # Function to evaluate the C5 value for a given h
    def evaluate_c5(h, dx):
        h = np.clip(h, 0, 1)
        h_extended = np.zeros(2 * n_points)
        h_extended[:n_points] = h
        h_extended[n_points:] = h
        corr = np.correlate(h_extended, 1 - h, mode="full")
        corr = corr[n_points - 1: 2 * n_points - 1]
        c5 = np.max(corr) * dx
        return c5

    # Function to optimize the h values
    def optimize_h(h, dx, n_points):
        # Objective: minimize the maximum overlap
        def objective(x):
            x = np.clip(x, 0, 1)
            x = x.reshape(n_points)
            c5 = evaluate_c5(x, dx)
            return c5

        # Constraints: sum(h) == n_points / 2
        def constraint(x):
            return np.sum(x) - n_points / 2

        # Initial guess
        x0 = h.copy()
        x0 = np.clip(x0, 0, 1)
        x0 = x0.reshape(n_points)

        # Optimization
        res = optimize.minimize(
            objective,
            x0,
            method='SLSQP',
            constraints={'type': 'eq', 'fun': constraint},
            bounds=[(0, 1) for _ in range(n_points)],
            options={'ftol': 1e-6, 'maxiter': 100}
        )

        return res.x

    # Start with initial guess
    h = initial_h_values
    h = h[:n_points]
    dx = 2.0 / n_points
    if np.sum(h) * dx != 1:
        h = np.clip(h, 0, 1)
        h = h / (np.sum(h) * dx) * (n_points / 2)

    # Start optimization
    for _ in range(10):  # Run multiple iterations for improvement
        h = optimize_h(h, dx, n_points)
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
        # Try increasing the number of points for better resolution
        n_points += 50
        dx = 2.0 / n_points
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        h = np.clip(h, 0, 1)
        if np.sum(h) * dx != 1:
            h = h / (np.sum(h) * dx) * (n_points / 2)

    # Final evaluation
    best_h = np.clip(best_h, 0, 1)
    dx = 2.0 / len(best_h)
    best_c5 = evaluate_c5(best_h, dx)

    # Ensure the best_h has the correct number of points
    best_h = best_h[:n_points]
    dx = 2.0 / n_points
    best_h = np.clip(best_h, 0, 1)
    if np.sum(best_h) * dx != 1:
        best_h = best_h / (np.sum(best_h) * dx) * (n_points / 2)

    return (best_h, best_c5, n_points)
