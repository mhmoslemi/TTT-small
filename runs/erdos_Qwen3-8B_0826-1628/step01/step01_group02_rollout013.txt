import numpy as np

from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    
    # Evaluate current C5
    def compute_c5(h):
        dx = 2.0 / len(h)
        shifted = np.zeros_like(h)
        for k in range(1, len(h)):
            shifted[k:] = h[:-k]
        result = correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    best_n = n_points
    
    # Try reducing n_points to improve performance
    for n in range(n_points - 1, 30, -1):
        if n < 30:
            break
        # Create a grid for h
        x = np.linspace(0, 2, n)
        h = np.zeros(n)
        # Simple initial guess: distribute the integral evenly
        h = np.repeat(1.0 / n, n)
        # Optimize h to minimize max overlap
        # Define variables
        h_var = cp.Variable(n, non_negative=True, boolean=False)
        constraints = [
            cp.sum(h_var) == n / 2,
            0 <= h_var,
            h_var <= 1
        ]
        # Objective: minimize the maximum overlap
        # Compute the overlap for all shifts
        def objective_fn(h_var):
            dx = 2.0 / n
            overlaps = []
            for k in range(1, n):
                shifted = np.zeros(n)
                shifted[k:] = h_var[:-k]
                overlap = np.sum(h_var * (1 - shifted)) * dx
                overlaps.append(overlap)
            return cp.Max(overlaps)
        prob = cp.Problem(cp.Minimize(objective_fn(h_var)), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iter=100, eps=1e-4)
            if prob.status == 'optimal':
                h = h_var.value
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    best_n = n
        except:
            pass
        
        # Check time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, best_n)
