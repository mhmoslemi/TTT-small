def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define parameters
    n_points = 100
    h = np.random.rand(n_points)
    h = (h * 2 - 1) * 0.5 + 0.5
    h /= np.mean(h)
    
    # Optimization loop
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h)
        
        # Define objective function for optimization
        def objective(x):
            h_new = x.copy()
            h_new = np.clip(h_new, 0, 1)
            return evaluate_erdos_solution(h_new)
        
        # Define constraints
        def constraints(x):
            return np.sum(x) - n_points / 2
        
        # Use L-BFGS-B optimizer
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraints}
        )
        
        # Update h with the best solution found
        h = result.x
        
        # Print progress
        if _ % 5 == 0:
            print(f"Iteration {_}: C5 = {evaluate_erdos_solution(h):.6f}")
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(h)
    print(f"Final C5 = {final_c5:.6f}")
    
    return h, final_c5, n_points
