import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    ones_minus_h = 1 - h
    corr = np.correlate(h, ones_minus_h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    n_points = 45
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < 15:
            h[i] = 1
        elif i < 30:
            h[i] = 0
        else:
            h[i] = 1
    return h.tolist()

# Strategy:
# 1. Start from a refined initial configuration with periodic patterns
# 2. Use a combination of gradient-based optimization and simulated annealing
# 3. Refine using alternating optimization between different shift values
# 4. Introduce additional symmetry and periodicity to reduce overlap

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with a refined initial configuration with periodicity
    n_points = 100
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    
    # Periodic pattern with 3 segments: 1, 0, 1
    segment_length = n_points // 3
    remainder = n_points % 3
    for i in range(n_points):
        if i % segment_length < remainder:
            initial_h[i] = 1
        elif i % segment_length < remainder + segment_length:
            initial_h[i] = 0
        else:
            initial_h[i] = 1
    
    # Ensure sum constraint
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    best_time = 0

    # Define the objective function
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Enforce sum constraint
        c5 = evaluate_erdos_solution(h)
        return c5

    # Define bounds and constraints
    bounds = Bounds(0, 1)
    constraints = [LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)]

    # Optimization with L-BFGS-B
    for _ in range(50):  # Run multiple iterations
        start_time = time.time()
        result = opt.minimize(
            fun=objective,
            x0=best_h,
            bounds=bounds,
            constraints=constraints,
            method='L-BFGS-B',
            options={'maxiter': 50, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        if result.success:
            best_h = result.x
            best_c5 = result.fun
        end_time = time.time()
        best_time += end_time - start_time
        if best_time >= budget_s:
            break

    # Simulated annealing refinement
    if best_time < budget_s:
        start_time = time.time()
        result = opt.minimize(
            fun=objective,
            x0=best_h,
            bounds=bounds,
            constraints=constraints,
            method='SLSQP',
            options={'maxiter': 50, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        if result.success:
            best_h = result.x
            best_c5 = result.fun
        end_time = time.time()
        best_time += end_time - start_time

    # Final refinement with constraint satisfaction
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)

    return best_h.tolist(), best_c5, n_points
