```python
import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint, minimize

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = h * (n_points / 2) / sum(h)
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * (2.0 / n_points)
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': integral_constraint})
    
    # Function to optimize
    def objective(x):
        return compute_overlap(x)
    
    # Use a structured approach with local search and perturbation
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    num_perturbations = 50
    perturbation_decay = 0.99
    
    for _ in range(num_perturbations):
        if time.time() - start_time > budget_s:
            break
        
        # Generate a perturbed version
        perturbation = np.random.uniform(-0.2, 0.2, size=n_points)
        candidate = np.clip(h + perturbation, 0, 1)
        candidate = candidate * (n_points / 2) / sum(candidate)
        
        # Local refinement using SLSQP
        try:
            result = minimize(objective, candidate, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-8)
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_solution = result.x
        except:
            pass
    
    return (best_solution.tolist(), best_c5, n_points)
```
