import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    n_particles = 50
    max_iter = 100
    inertia_weight = 0.8
    cognitive_coeff = 1.5
    social_coeff = 1.5
    refinement_steps = 3
    
    # Initialize particle swarm
    particles = np.random.uniform(0, 1, (n_particles, n_points))
    particles = particles * (n_points / 2) / np.sum(particles, axis=1, keepdims=True)
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    personal_best_positions = particles.copy()
    personal_best_values = np.array([compute_overlap(p) for p in particles])
    global_best_position = particles[np.argmin(personal_best_values)].copy()
    global_best_value = np.min(personal_best_values)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Update particle velocities and positions
        r1 = np.random.rand(n_particles, n_points)
        r2 = np.random.rand(n_particles, n_points)
        velocities = inertia_weight * velocities + cognitive_coeff * r1 * (personal_best_positions - particles) + social_coeff * r2 * (global_best_position - particles)
        particles += velocities
        
        # Clip values to [0, 1] and normalize
        particles = np.clip(particles, 0, 1)
        sum_p = np.sum(particles, axis=1, keepdims=True)
        particles = particles * (n_points / 2) / sum_p
        
        # Evaluate fitness
        current_values = np.array([compute_overlap(p) for p in particles])
        best_indices = np.argmin(current_values)
        
        # Update personal bests
        for i in range(n_particles):
            if current_values[i] < personal_best_values[i]:
                personal_best_positions[i] = particles[i].copy()
                personal_best_values[i] = current_values[i]
        
        # Update global best
        if np.min(current_values) < global_best_value:
            global_best_value = np.min(current_values)
            global_best_position = particles[best_indices].copy()
    
    # Refinement with SLSQP
    best_solution = global_best_position
    best_c5 = global_best_value
    
    for _ in range(refinement_steps):
        if time.time() - start_time > budget_s:
            break
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': integral_constraint})
        result = optimize.minimize(lambda x: compute_overlap(x), best_solution, bounds=bounds, constraints=cons, method='SLSQP')
        
        if result.success:
            best_solution = result.x
            best_c5 = result.fun
    
    return (best_solution.tolist(), best_c5, n_points)
