import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    total = sum(h)
    h_norm = h * (n_points / 2) / total

    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * (2.0 / n_points)
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    num_iterations = 100
    num_candidates = 50
    perturbation_decay = 0.99
    alpha = 0.5  # Learning rate for local refinement
    min_perturbation = 0.01
    max_perturbation = 0.2

    # Initialize a swarm of candidate solutions
    swarm = [h_norm.copy() for _ in range(num_candidates)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Use a more structured perturbation strategy
    for iteration in range(num_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Perform a global perturbation on each candidate in the swarm
        for i in range(num_candidates):
            # Vary perturbation size based on iteration
            perturbation = np.random.uniform(-max_perturbation, max_perturbation, size=n_points)
            perturbation = perturbation * (1 - iteration / num_iterations)
            perturbation = np.clip(perturbation, -max_perturbation, max_perturbation)
            
            candidate = np.clip(swarm[i] + perturbation, 0, 1)
            candidate = candidate * (n_points / 2) / sum(candidate)
            
            # Local refinement using SLSQP
            bounds = [(0, 1) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': integral_constraint})
            
            # Use a custom objective function that also tracks the overlap
            def objective(x):
                return compute_overlap(x)
            
            try:
                result = optimize.minimize(objective, candidate, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-8)
                if result.success:
                    swarm[i] = result.x
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = result.x
            except:
                pass
    
    return (best_solution.tolist(), best_c5, n_points)
