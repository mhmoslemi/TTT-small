import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    n_particles = 20
    max_iter = 150
    inertia = 0.8
    cognitive = 1.2
    social = 1.2
    personal_best = [h_norm.copy() for _ in range(n_particles)]
    personal_best_values = [compute_overlap(p) for p in personal_best]
    global_best = h_norm.copy()
    global_best_value = compute_overlap(global_best)
    
    # Initialize velocities
    velocities = np.random.uniform(-0.1, 0.1, size=(n_particles, n_points))
    
    # PSO optimization
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        for i in range(n_particles):
            # Update personal best
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = inertia * velocities[i] + cognitive * r1 * (personal_best[i] - h_norm) + social * r2 * (global_best - h_norm)
            h_candidate = np.clip(h_norm + velocities[i], 0, 1)
            
            # Enforce integral constraint
            h_candidate = h_candidate * (n_points / 2) / sum(h_candidate)
            
            # Evaluate candidate
            current_value = compute_overlap(h_candidate)
            
            # Update personal best if better
            if current_value < personal_best_values[i]:
                personal_best[i] = h_candidate.copy()
                personal_best_values[i] = current_value
                
                # Update global best
                if current_value < global_best_value:
                    global_best = h_candidate.copy()
                    global_best_value = current_value
    
    # After PSO, fine-tune with local optimization
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': integral_constraint})
    result = optimize.minimize(lambda x: compute_overlap(x), global_best, bounds=bounds, constraints=cons, method='SLSQP')
    
    if result.success:
        final_h = result.x
        final_c5 = result.fun
    else:
        final_h = global_best
        final_c5 = global_best_value
    
    return (final_h.tolist(), final_c5, n_points)
