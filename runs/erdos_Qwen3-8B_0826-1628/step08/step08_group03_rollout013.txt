import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint, NonlinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    swarm_size = 50
    max_iter = 100
    perturbation_decay = 0.98
    initial_perturbation = 0.25
    refinement_steps = 5
    refinement_perturbation = 0.05
    
    # Initialize a swarm of candidate solutions
    swarm = [h_norm.copy() for _ in range(swarm_size)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    best_n_points = n_points
    
    # Define bounds and constraints for optimization
    bounds = Bounds(0, 1)
    integral_cons = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Perform a global perturbation on each candidate in the swarm
        for i in range(swarm_size):
            perturbation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
            candidate = np.clip(swarm[i] + perturbation, 0, 1)
            candidate = candidate * (n_points / 2) / np.sum(candidate)
            
            # Local refinement using SLSQP
            result = optimize.minimize(
                lambda x: compute_overlap(x),
                candidate,
                bounds=bounds,
                constraints=integral_cons,
                method='SLSQP',
                options={'ftol': 1e-8, 'maxiter': 100}
            )
            
            # Update the swarm member if the optimization succeeded
            if result.success:
                swarm[i] = result.x
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_solution = result.x
                if current_c5 < 0.380876:
                    best_n_points = n_points
    
        # Gradually reduce perturbation range
        initial_perturbation *= perturbation_decay
    
    # Refinement phase with more precise local optimization
    if best_c5 < 0.380876:
        # Use higher resolution for refinement
        n_points_refinement = 100
        dx_refinement = 2.0 / n_points_refinement
        initial_h_refinement = np.zeros(n_points_refinement)
        initial_h_refinement[:int(n_points_refinement / 2)] = 1.0
        initial_h_refinement[int(n_points_refinement / 2):] = 0.0
        initial_h_refinement = initial_h_refinement * (n_points_refinement / 2) / np.sum(initial_h_refinement)
        
        # Refine the solution
        def compute_overlap_refinement(h_values):
            if len(h_values) != n_points_refinement:
                return float('inf')
            corr = np.correlate(h_values, 1 - h_values, mode="full")
            return np.max(corr) * dx_refinement
        
        # Optimization parameters for refinement
        swarm_size_refinement = 20
        max_iter_refinement = 50
        refinement_perturbation = 0.05
        
        # Initialize a swarm of candidate solutions for refinement
        swarm_refinement = [initial_h_refinement.copy() for _ in range(swarm_size_refinement)]
        best_solution_refinement = initial_h_refinement.copy()
        best_c5_refinement = compute_overlap_refinement(best_solution_refinement)
        
        for iteration in range(max_iter_refinement):
            if time.time() - start_time > budget_s:
                break
            
            # Perform a global perturbation on each candidate in the swarm
            for i in range(swarm_size_refinement):
                perturbation = np.random.uniform(-refinement_perturbation, refinement_perturbation, size=n_points_refinement)
                candidate = np.clip(swarm_refinement[i] + perturbation, 0, 1)
                candidate = candidate * (n_points_refinement / 2) / np.sum(candidate)
                
                # Local refinement using SLSQP
                result = optimize.minimize(
                    lambda x: compute_overlap_refinement(x),
                    candidate,
                    bounds=Bounds(0, 1),
                    constraints=LinearConstraint(np.ones(n_points_refinement), [n_points_refinement / 2], [n_points_refinement / 2]),
                    method='SLSQP',
                    options={'ftol': 1e-8, 'maxiter': 100}
                )
                
                # Update the swarm member if the optimization succeeded
                if result.success:
                    swarm_refinement[i] = result.x
                    current_c5 = result.fun
                    if current_c5 < best_c5_refinement:
                        best_c5_refinement = current_c5
                        best_solution_refinement = result.x
                        if current_c5 < 0.380876:
                            best_n_points = n_points_refinement
        
        # Update the best solution and C5 if refinement improves it
        if best_c5_refinement < best_c5:
            best_c5 = best_c5_refinement
            best_solution = best_solution_refinement
    
    return (best_solution.tolist(), best_c5, best_n_points)
