import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': integral_constraint})
    
    # Optimization parameters
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_fraction = 0.2
    initial_perturbation = 0.2
    
    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * (n_points / 2) / sum(h) for h in population]
    best_solution = population[0]
    best_c5 = compute_overlap(best_solution)
    
    # Evolutionary algorithm
    for generation in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [compute_overlap(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_solution = population[best_idx].copy()
        
        # Selection: elitism + tournament
        elite_size = int(elite_fraction * population_size)
        elites = population[:elite_size]
        tournament_indices = np.random.choice(range(population_size), size=population_size, replace=False)
        tournament = [population[i] for i in tournament_indices]
        parents = []
        for _ in range(population_size - elite_size):
            p1, p2 = np.random.choice(range(len(tournament)), size=2, replace=False)
            if np.random.rand() < crossover_rate:
                # Crossover
                child = np.zeros(n_points)
                split = np.random.randint(1, n_points - 1)
                child[:split] = tournament[p1][:split]
                child[split:] = tournament[p2][split:]
            else:
                # Random selection
                child = tournament[np.random.choice([p1, p2])]
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
                child = np.clip(child + mutation, 0, 1)
            
            # Ensure integral constraint
            child = child * (n_points / 2) / sum(child)
            parents.append(child)
        
        # Create new population
        population = elites + parents
    
    return (best_solution.tolist(), best_c5, n_points)
