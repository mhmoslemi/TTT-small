import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial configuration
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Define the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': integral_constraint}]
    
    # Optimization parameters
    max_iter = 100
    refine_steps = 10
    refine_decay = 0.9
    perturbation = 0.2
    
    # Start with the initial solution
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Generate candidate solutions via perturbation
        candidate = np.clip(best_solution + np.random.uniform(-perturbation, perturbation, size=n_points), 0, 1)
        candidate = candidate * (n_points / 2) / np.sum(candidate)
        
        # Use SLSQP with constraints
        result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_solution = result.x
    
    # Refinement phase with smaller perturbations
    for _ in range(refine_steps):
        if time.time() - start_time > budget_s:
            break
        
        # Generate candidate solutions via perturbation
        candidate = np.clip(best_solution + np.random.uniform(-0.05, 0.05, size=n_points), 0, 1)
        candidate = candidate * (n_points / 2) / np.sum(candidate)
        
        # Use SLSQP with constraints
        result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_solution = result.x
    
    return (best_solution.tolist(), best_c5, n_points)
