import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with initial_h_values but scale to satisfy the integral constraint
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    def compute_overlap(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Define the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    mutation_decay = 0.95
    elite_size = 2
    refine_steps = 5
    refine_decay = 0.9
    
    # Initialize a population of candidate solutions
    population = [h.copy() for _ in range(population_size)]
    population = [np.clip(x, 0, 1) for x in population]
    population = [x * (n_points / 2) / np.sum(x) for x in population]
    
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Function to evolve the population with a balance of exploration and exploitation
    def evolve_population(population, best_solution, best_c5, mutation_rate, mutation_decay, generations, n_points, dx, compute_overlap, integral_constraint):
        for gen in range(generations):
            if time.time() - start_time > budget_s:
                break
            
            # Evaluate fitness for all individuals
            fitness = [compute_overlap(ind) for ind in population]
            
            # Select elite individuals
            elite_indices = np.argsort(fitness)[:elite_size]
            elite_population = [population[i] for i in elite_indices]
            
            # Create new population
            new_population = elite_population.copy()
            
            # Generate offspring
            for _ in range(population_size - elite_size):
                # Select parents
                parent1 = random.choice(population)
                parent2 = random.choice(population)
                
                # Crossover
                child = np.zeros(n_points)
                for i in range(n_points):
                    if random.random() < 0.5:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
                
                # Mutation
                mutation_strength = mutation_rate * (1 - gen / generations)
                for i in range(n_points):
                    if random.random() < mutation_strength:
                        child[i] += np.random.uniform(-0.1, 0.1)
                        child[i] = np.clip(child[i], 0, 1)
                
                # Apply the integral constraint
                child = child * (n_points / 2) / np.sum(child)
                
                # Add to new population
                new_population.append(child)
            
            # Replace population with new population
            population = new_population
            population = [np.clip(x, 0, 1) for x in population]
            population = [x * (n_points / 2) / np.sum(x) for x in population]
            
            # Update best solution
            for ind in population:
                current_c5 = compute_overlap(ind)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_solution = ind.copy()
            
            # Refine best solution with local optimization
            if best_c5 < 0.381:
                bounds = [(0, 1) for _ in range(n_points)]
                cons = [{'type': 'eq', 'fun': integral_constraint}]
                result = optimize.minimize(lambda x: compute_overlap(x), best_solution, bounds=bounds, constraints=cons, method='SLSQP')
                if result.success:
                    best_solution = result.x
                    best_c5 = result.fun
            
            # Decrease mutation rate
            mutation_rate *= mutation_decay
        
        return best_solution, best_c5
    
    # Run the optimization
    best_solution, best_c5 = evolve_population(population, best_solution, best_c5, mutation_rate, mutation_decay, generations, n_points, dx, compute_overlap, integral_constraint)
    
    return (best_solution.tolist(), best_c5, n_points)
