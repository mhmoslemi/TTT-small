import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Define the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    num_trials = 50
    max_iter = 100
    perturbation_decay = 0.95
    initial_perturbation = 0.2
    refine_steps = 10
    refine_decay = 0.9
    
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Function to evolve swarm with a balance of exploration and exploitation
    def evolve_swarm(swarm, best_solution, best_c5, initial_perturbation, perturbation_decay, max_iter, n_points, dx, compute_overlap, integral_constraint):
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Generate new candidates
            for i in range(len(swarm)):
                # Random perturbation
                perturbation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
                candidate = np.clip(swarm[i] + perturbation, 0, 1)
                
                # Apply the integral constraint
                candidate = candidate * (n_points / 2) / np.sum(candidate)
                
                # Apply constraint and optimize
                bounds = [(0, 1) for _ in range(n_points)]
                cons = [{'type': 'eq', 'fun': integral_constraint}]
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                
                # Update swarm
                if result.success:
                    swarm[i] = result.x
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = result.x
            
            initial_perturbation *= perturbation_decay
        
        return best_solution, best_c5
    
    # Run multiple trials
    swarm = [h.copy() for _ in range(num_trials)]
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    
    for trial in range(num_trials):
        if time.time() - start_time > budget_s:
            break
        
        # Evolve swarm
        best_solution_trial, best_c5_trial = evolve_swarm(
            swarm, best_solution, best_c5, initial_perturbation, perturbation_decay, max_iter, n_points, dx, compute_overlap, integral_constraint
        )
        
        # Refine with smaller perturbations
        for _ in range(refine_steps):
            perturbation = np.random.uniform(-0.01, 0.01, size=n_points)
            refined = np.clip(best_solution_trial + perturbation, 0, 1)
            refined = refined * (n_points / 2) / np.sum(refined)
            result = optimize.minimize(lambda x: compute_overlap(x), refined, bounds=bounds, constraints=cons, method='SLSQP')
            if result.success:
                best_solution_trial = result.x
                best_c5_trial = result.fun
        
        # Update best solution
        if best_c5_trial < best_c5:
            best_c5 = best_c5_trial
            best_solution = best_solution_trial
    
    return (best_solution.tolist(), best_c5, n_points)
