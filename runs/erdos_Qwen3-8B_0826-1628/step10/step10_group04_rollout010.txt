import numpy as np
import scipy.optimize as optimize
import time
import random
import math

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    
    # Ensure sum(h) * dx = 1
    sum_h = sum(h)
    h = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Define the bounds for each sample
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization parameters
    max_iter = 200
    refine_steps = 10
    refine_decay = 0.95
    perturb_range = 0.3
    perturb_decay = 0.99
    
    # Start with the initial solution
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Perform a gradient-based optimization with SLSQP
    result = optimize.minimize(lambda x: compute_overlap(x), best_solution, bounds=bounds, constraints={'type': 'eq', 'fun': integral_constraint}, method='SLSQP')
    if result.success:
        best_solution = result.x
        best_c5 = result.fun
    
    # Perform local refinement with random walk and SLSQP
    for _ in range(refine_steps):
        if time.time() - start_time > budget_s:
            break
        # Random perturbation
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        candidate = np.clip(best_solution + perturbation, 0, 1)
        candidate = candidate * (n_points / 2) / sum(candidate)
        
        # Local refinement
        result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints={'type': 'eq', 'fun': integral_constraint}, method='SLSQP')
        if result.success and result.fun < best_c5:
            best_solution = result.x
            best_c5 = result.fun
    
    # Perform a multi-step gradient-based optimization
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Generate a perturbed version
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        candidate = np.clip(best_solution + perturbation, 0, 1)
        candidate = candidate * (n_points / 2) / sum(candidate)
        
        # Local refinement
        result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints={'type': 'eq', 'fun': integral_constraint}, method='SLSQP')
        if result.success and result.fun < best_c5:
            best_solution = result.x
            best_c5 = result.fun
    
    return (best_solution.tolist(), best_c5, n_points)
