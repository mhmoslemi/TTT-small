import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the provided initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Define the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    population_size = 50
    generations = 100
    mutation_rate = 0.15
    elite_fraction = 0.1
    refine_steps = 10
    refine_decay = 0.9
    
    # Initialize a population of candidate solutions
    population = [h.copy() for _ in range(population_size)]
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Function to evolve population with a balance of exploration and exploitation
    def evolve_population(population, best_solution, best_c5, mutation_rate, elite_fraction, generations, n_points, dx, compute_overlap, integral_constraint):
        for generation in range(generations):
            if time.time() - start_time > budget_s:
                break
            
            # Evaluate fitness of current population
            fitness = [compute_overlap(ind) for ind in population]
            # Find the best individual
            best_index = np.argmin(fitness)
            best_c5 = fitness[best_index]
            best_solution = population[best_index].copy()
            
            # Select elite individuals
            elite_size = int(elite_fraction * population_size)
            elite_indices = np.argsort(fitness)[:elite_size]
            elite_population = [population[i] for i in elite_indices]
            
            # Create new population
            new_population = elite_population.copy()
            while len(new_population) < population_size:
                # Select two parents
                parent1 = random.choice(elite_population)
                parent2 = random.choice(elite_population)
                
                # Crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
                
                # Mutation
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] = np.random.uniform(0, 1)
                
                # Ensure integral constraint
                child = child * (n_points / 2) / np.sum(child)
                
                # Local refinement with SLSQP
                bounds = [(0, 1) for _ in range(n_points)]
                cons = [{'type': 'eq', 'fun': integral_constraint}]
                result = optimize.minimize(lambda x: compute_overlap(x), child, bounds=bounds, constraints=cons, method='SLSQP')
                
                if result.success:
                    new_population.append(result.x)
                else:
                    new_population.append(child)
            
            population = new_population
    
        return best_solution, best_c5
    
    # Run the optimization
    best_solution, best_c5 = evolve_population(population, best_solution, best_c5, mutation_rate, elite_fraction, generations, n_points, dx, compute_overlap, integral_constraint)
    
    return (best_solution.tolist(), best_c5, n_points)
