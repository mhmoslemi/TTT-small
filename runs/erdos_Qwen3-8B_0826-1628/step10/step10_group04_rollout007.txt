import numpy as np
import scipy.optimize as optimize
import time
import random
import math

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    num_steps = 200
    perturb_range = 0.2
    refine_steps = 5
    refine_decay = 0.95
    max_refine = 50
    initial_perturb = 0.3
    decay_rate = 0.99
    
    # Function to generate a random perturbation
    def generate_perturbation(perturb_range):
        return np.random.uniform(-perturb_range, perturb_range, size=n_points)
    
    # Function to refine a solution with local optimization
    def refine_solution(candidate, max_refine, refine_decay, dx, compute_overlap, integral_constraint):
        current = candidate.copy()
        for _ in range(max_refine):
            perturbation = generate_perturbation(perturb_range * refine_decay)
            candidate = np.clip(current + perturbation, 0, 1)
            candidate = candidate * (n_points / 2) / sum(candidate)
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': integral_constraint})
            result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
            
            if result.success:
                current = result.x
                if result.fun < compute_overlap(current):
                    current = result.x
            else:
                break
        
        return current
    
    # Function to perform simulated annealing
    def simulated_annealing(initial, max_steps, initial_perturb, decay_rate, dx, compute_overlap, integral_constraint):
        current = initial.copy()
        current_overlap = compute_overlap(current)
        best_solution = current.copy()
        best_overlap = current_overlap
        perturb_range = initial_perturb
        temp = 1.0
        
        for step in range(max_steps):
            if time.time() - start_time > budget_s:
                break
            
            # Generate a new candidate by perturbing the current solution
            perturbation = generate_perturbation(perturb_range)
            candidate = np.clip(current + perturbation, 0, 1)
            candidate = candidate * (n_points / 2) / sum(candidate)
            
            # Compute the overlap of the new candidate
            candidate_overlap = compute_overlap(candidate)
            
            # Accept the new solution with a probability based on the temperature
            if candidate_overlap < best_overlap:
                current = candidate.copy()
                current_overlap = candidate_overlap
                best_solution = current.copy()
                best_overlap = current_overlap
            else:
                # Calculate the probability of accepting a worse solution
                prob = np.exp((current_overlap - candidate_overlap) / temp)
                if np.random.random() < prob:
                    current = candidate.copy()
                    current_overlap = candidate_overlap
            
            # Decrease the temperature
            temp *= decay_rate
            perturb_range *= decay_rate
        
        return best_solution, best_overlap
    
    # Run the simulated annealing algorithm
    best_solution, best_c5 = simulated_annealing(h, num_steps, initial_perturb, decay_rate, dx, compute_overlap, integral_constraint)
    
    # Further refine the best solution
    best_solution = refine_solution(best_solution, max_refine, refine_decay, dx, compute_overlap, integral_constraint)
    
    return (best_solution.tolist(), best_c5, n_points)
