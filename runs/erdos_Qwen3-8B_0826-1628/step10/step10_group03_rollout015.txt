import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the provided initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Define the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    population_size = 20
    max_generations = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    refine_steps = 3
    refine_decay = 0.9
    initial_perturbation = 0.15
    
    # Initialize population
    population = [h.copy() for _ in range(population_size)]
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Evaluate initial population
    for indiv in population:
        c5 = compute_overlap(indiv)
        if c5 < best_c5:
            best_c5 = c5
            best_solution = indiv.copy()
    
    # Evolutionary algorithm
    for generation in range(max_generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = np.array([compute_overlap(indiv) for indiv in population])
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_solution = population[best_idx].copy()
        
        # Select parents
        parents = np.random.choice(population_size, size=population_size, p=fitness / np.sum(fitness))
        parents = [population[i] for i in parents]
        
        # Crossover
        new_population = []
        for i in range(0, population_size, 2):
            if i + 1 < population_size:
                parent1 = parents[i]
                parent2 = parents[i + 1]
                child1 = np.zeros_like(parent1)
                child2 = np.zeros_like(parent2)
                for j in range(n_points):
                    if np.random.rand() < crossover_rate:
                        child1[j] = parent1[j]
                        child2[j] = parent2[j]
                    else:
                        child1[j] = parent2[j]
                        child2[j] = parent1[j]
                new_population.append(child1)
                new_population.append(child2)
            else:
                new_population.append(parents[i])
        
        # Mutation
        for i in range(population_size):
            indiv = new_population[i]
            perturbation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
            mutated = np.clip(indiv + perturbation, 0, 1)
            mutated = mutated * (n_points / 2) / np.sum(mutated)
            new_population[i] = mutated
        
        # Replace population
        population = new_population
        
        # Refinement with local optimization
        for i in range(population_size):
            indiv = population[i].copy()
            for _ in range(refine_steps):
                perturbation = np.random.uniform(-0.01, 0.01, size=n_points)
                candidate = np.clip(indiv + perturbation, 0, 1)
                candidate = candidate * (n_points / 2) / np.sum(candidate)
                bounds = [(0, 1) for _ in range(n_points)]
                cons = [{'type': 'eq', 'fun': integral_constraint}]
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                if result.success:
                    indiv = result.x
            population[i] = indiv
        
        # Update best solution
        for indiv in population:
            c5 = compute_overlap(indiv)
            if c5 < best_c5:
                best_c5 = c5
                best_solution = indiv.copy()
    
    return (best_solution.tolist(), best_c5, n_points)
