import numpy as np
import scipy.optimize as optimize
import time
import random
import matplotlib.pyplot as plt

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the provided initial solution
    n_points = len(initial_h_values)
    
    # Function to compute overlap
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * (2.0 / n_points)
    
    # Define the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Initialize the initial solution
    h = initial_h_values.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimization parameters
    max_iter = 200
    refine_steps = 5
    refine_decay = 0.95
    
    # Function to perform local refinement
    def refine_solution(candidate):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': integral_constraint}]
        result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
        return result.x if result.success else candidate
    
    # Function to generate perturbed solutions
    def perturb_solution(candidate, perturbation_range):
        perturbation = np.random.uniform(-perturbation_range, perturbation_range, size=n_points)
        perturbed = np.clip(candidate + perturbation, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        return perturbed
    
    # Function to evaluate the solution
    def evaluate_solution(candidate):
        return compute_overlap(candidate)
    
    # Main optimization loop
    best_solution = h.copy()
    best_c5 = evaluate_solution(best_solution)
    perturbation_range = 0.15
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Generate a new candidate solution
        candidate = perturb_solution(best_solution, perturbation_range)
        candidate = refine_solution(candidate)
        current_c5 = evaluate_solution(candidate)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_solution = candidate
        
        # Refine the best solution further
        for _ in range(refine_steps):
            perturbed = perturb_solution(best_solution, perturbation_range * refine_decay)
            refined = refine_solution(perturbed)
            current_c5 = evaluate_solution(refined)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_solution = refined
        
        perturbation_range *= 0.95
    
    return (best_solution.tolist(), best_c5, n_points)
