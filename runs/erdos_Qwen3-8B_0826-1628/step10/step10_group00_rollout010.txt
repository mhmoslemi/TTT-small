import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial solution with 36 points
    n_points = len(initial_h_values)
    h = initial_h_values.copy().astype(float)
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Define the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Function to evolve the solution using a combination of strategies
    def evolve_solution(h, n_points, dx, compute_overlap, integral_constraint, budget_s, start_time):
        best_solution = h.copy()
        best_c5 = compute_overlap(best_solution)
        refine_steps = 5
        refine_decay = 0.95
        max_refine_attempts = 100
        
        for refine_attempt in range(max_refine_attempts):
            if time.time() - start_time > budget_s:
                break
            
            # Local refinement using SLSQP with small perturbations
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': integral_constraint}]
            
            # Initial perturbation
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            candidate = np.clip(h + perturbation, 0, 1)
            candidate = candidate * (n_points / 2) / np.sum(candidate)
            
            # Use SLSQP to optimize the candidate
            result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
            
            if result.success:
                candidate = result.x
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_solution = candidate
                h = candidate
            
            # Further refinement with smaller perturbations
            for _ in range(refine_steps):
                perturbation = np.random.uniform(-0.01, 0.01, size=n_points)
                candidate = np.clip(h + perturbation, 0, 1)
                candidate = candidate * (n_points / 2) / np.sum(candidate)
                
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                
                if result.success:
                    candidate = result.x
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = candidate
                    h = candidate
            
            # Check if the solution has stabilized
            if np.allclose(h, best_solution, atol=1e-6):
                break
        
        return best_solution, best_c5
    
    # Run the refinement process
    best_solution, best_c5 = evolve_solution(h, n_points, dx, compute_overlap, integral_constraint, budget_s, start_time)
    
    return (best_solution.tolist(), best_c5, n_points)
