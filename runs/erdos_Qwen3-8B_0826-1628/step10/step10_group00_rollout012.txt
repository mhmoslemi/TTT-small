import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial solution with n_points = 36 (the provided initial_h_values)
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Define the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 150
    refine_steps = 10
    refine_decay = 0.99
    swarm_size = 30
    initial_perturbation = 0.2
    perturbation_decay = 0.95
    local_steps = 30
    local_decay = 0.98
    
    # Initialize a swarm of candidate solutions
    swarm = [h.copy() for _ in range(swarm_size)]
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Function to evolve swarm with a balance of exploration and exploitation
    def evolve_swarm(swarm, best_solution, best_c5, initial_perturbation, perturbation_decay, max_iter, n_points, dx, compute_overlap, integral_constraint):
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Phase 1: Global exploration with large perturbations
            for i in range(swarm_size):
                # Generate a random perturbation
                perturbation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
                candidate = np.clip(swarm[i] + perturbation, 0, 1)
                
                # Apply the integral constraint
                candidate = candidate * (n_points / 2) / np.sum(candidate)
                
                # Use SLSQP with constraints
                bounds = [(0, 1) for _ in range(n_points)]
                cons = [{'type': 'eq', 'fun': integral_constraint}]
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                
                # Update the swarm if the optimization succeeded
                if result.success:
                    swarm[i] = result.x
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = result.x
            
            # Reduce perturbation range and do local refinement
            initial_perturbation *= perturbation_decay
            
            # Phase 2: Local refinement with smaller perturbations
            for i in range(swarm_size):
                perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
                candidate = np.clip(swarm[i] + perturbation, 0, 1)
                
                # Apply the integral constraint
                candidate = candidate * (n_points / 2) / np.sum(candidate)
                
                # Use SLSQP with constraints
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                
                if result.success:
                    swarm[i] = result.x
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = result.x
            
            # Phase 3: Multi-step refinement with even smaller perturbations
            for i in range(swarm_size):
                candidate = swarm[i].copy()
                for _ in range(refine_steps):
                    perturbation = np.random.uniform(-0.01, 0.01, size=n_points)
                    candidate = np.clip(candidate + perturbation, 0, 1)
                    
                    # Apply the integral constraint
                    candidate = candidate * (n_points / 2) / np.sum(candidate)
                    
                    # Use SLSQP with constraints
                    result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                    
                    if result.success:
                        candidate = result.x
                        current_c5 = result.fun
                        if current_c5 < best_c5:
                            best_c5 = current_c5
                            best_solution = candidate
                swarm[i] = candidate
        
        return best_solution, best_c5
    
    # Additional phase: Local search with small perturbations around the best solution
    for _ in range(local_steps):
        if time.time() - start_time > budget_s:
            break
        
        # Generate perturbations
        perturbation = np.random.uniform(-0.01, 0.01, size=n_points)
        candidate = np.clip(best_solution + perturbation, 0, 1)
        
        # Apply the integral constraint
        candidate = candidate * (n_points / 2) / np.sum(candidate)
        
        # Use SLSQP with constraints
        result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_solution = result.x
    
    return (best_solution.tolist(), best_c5, n_points)
