import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial solution
    initial_n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = h * (initial_n_points / 2) / np.sum(h)
    
    # If the initial solution is already good, we can use it directly
    c5_initial = evaluate_erdos_solution(h)
    if c5_initial <= 0.380876:
        return (h.tolist(), c5_initial, initial_n_points)
    
    # Try using different starting points: a few different constructions
    # 1. Uniform distribution
    h_uniform = np.full(initial_n_points, 0.5)
    c5_uniform = evaluate_erdos_solution(h_uniform)
    
    # 2. Alternating 0 and 1
    h_alternating = np.array([0.0 if i % 2 == 0 else 1.0 for i in range(initial_n_points)])
    h_alternating = h_alternating * (initial_n_points / 2) / np.sum(h_alternating)
    c5_alternating = evaluate_erdos_solution(h_alternating)
    
    # 3. A simple periodic function
    h_periodic = np.sin(np.linspace(0, np.pi, initial_n_points)) * 0.5 + 0.5
    h_periodic = h_periodic * (initial_n_points / 2) / np.sum(h_periodic)
    c5_periodic = evaluate_erdos_solution(h_periodic)
    
    # Choose the best starting point
    best_h = h
    best_c5 = c5_initial
    if c5_uniform < best_c5:
        best_h = h_uniform
        best_c5 = c5_uniform
    if c5_alternating < best_c5:
        best_h = h_alternating
        best_c5 = c5_alternating
    if c5_periodic < best_c5:
        best_h = h_periodic
        best_c5 = c5_periodic
    
    h_initial = best_h
    n_points = initial_n_points
    dx = 2.0 / n_points
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Define the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 100
    refine_steps = 10
    refine_decay = 0.9
    
    # Initialize candidate solution
    h = h_initial.copy()
    
    # Function to refine a single solution with local perturbations
    def refine_solution(h, n_points, dx, compute_overlap, integral_constraint):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': integral_constraint}]
        for _ in range(refine_steps):
            perturbation = np.random.uniform(-0.01, 0.01, size=n_points)
            candidate = np.clip(h + perturbation, 0, 1)
            candidate = candidate * (n_points / 2) / np.sum(candidate)
            result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
            if result.success:
                h = result.x
        return h
    
    # Iterative refinement with perturbations
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Random perturbation
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        candidate = np.clip(h + perturbation, 0, 1)
        candidate = candidate * (n_points / 2) / np.sum(candidate)
        # Refine the candidate
        refined_h = refine_solution(candidate, n_points, dx, compute_overlap, integral_constraint)
        # Evaluate the refined solution
        c5_refined = compute_overlap(refined_h)
        # Update best solution
        if c5_refined < compute_overlap(h):
            h = refined_h
    
    return (h.tolist(), compute_overlap(h), n_points)
