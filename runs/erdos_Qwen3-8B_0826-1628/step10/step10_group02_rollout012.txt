import numpy as np
import scipy.optimize as optimize
import time
import random
from typing import List, Tuple

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    n_particles = 30
    max_iter = 100
    refine_steps = 10
    refine_decay = 0.95
    perturb_range = 0.3
    perturb_decay = 0.99
    local_search_steps = 5
    
    # Initialize particle swarm
    swarm = [np.array(h.copy()) for _ in range(n_particles)]
    best_position = h.copy()
    best_c5 = compute_overlap(best_position)
    
    # Particle swarm optimization parameters
    inertia = 0.8
    cognitive = 1.2
    social = 1.2
    
    # Function to evolve swarm
    def evolve_swarm(swarm, best_position, best_c5, perturb_range, perturb_decay, max_iter, n_points, dx, compute_overlap, integral_constraint):
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Initialize personal best and global best
            personal_best = [np.array(p.copy()) for p in swarm]
            personal_best_values = [compute_overlap(p) for p in swarm]
            
            # Update particles
            for i in range(n_particles):
                # Randomly generate velocities
                r1, r2 = np.random.rand(), np.random.rand()
                velocity = (inertia * swarm[i] + 
                          cognitive * r1 * (personal_best[i] - swarm[i]) + 
                          social * r2 * (best_position - swarm[i]))
                
                # Update particle position
                new_position = swarm[i] + velocity
                new_position = np.clip(new_position, 0, 1)
                new_position = new_position * (n_points / 2) / np.sum(new_position)
                
                # Compute overlap for the new position
                new_c5 = compute_overlap(new_position)
                
                # Update personal best
                if new_c5 < personal_best_values[i]:
                    personal_best[i] = new_position.copy()
                    personal_best_values[i] = new_c5
                
                # Update swarm member
                swarm[i] = new_position
            
            # Update global best
            for i in range(n_particles):
                if personal_best_values[i] < best_c5:
                    best_position = personal_best[i].copy()
                    best_c5 = personal_best_values[i]
            
            # Reduce perturbation range
            perturb_range *= perturb_decay
            
            # Perform local refinement with SLSQP
            for i in range(n_particles):
                # Generate a refined candidate
                candidate = swarm[i].copy()
                for _ in range(local_search_steps):
                    # Generate random perturbation
                    perturbation = np.random.uniform(-0.01, 0.01, size=n_points)
                    candidate = np.clip(candidate + perturbation, 0, 1)
                    candidate = candidate * (n_points / 2) / np.sum(candidate)
                    
                    # Local refinement with SLSQP
                    bounds = [(0, 1) for _ in range(n_points)]
                    cons = ({'type': 'eq', 'fun': integral_constraint})
                    result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                    
                    if result.success:
                        candidate = result.x
                        current_c5 = result.fun
                        if current_c5 < best_c5:
                            best_position = candidate.copy()
                            best_c5 = current_c5
                swarm[i] = candidate
        
        return best_position, best_c5
    
    # Run the optimization
    best_position, best_c5 = evolve_swarm(swarm, best_position, best_c5, perturb_range, perturb_decay, max_iter, n_points, dx, compute_overlap, integral_constraint)
    
    return (best_position.tolist(), best_c5, n_points)
