import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint
from scipy.optimize import Bounds, LinearConstraint, minimize, Bounds

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Adjust number of points for better performance
    n_points = 64  # Use 64 points for better resolution without being too slow
    dx = 2.0 / n_points
    
    # Initialize with a better starting point: alternating high/low values
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            h[i] = 0.5
        else:
            h[i] = 0.5
    # Normalize the integral to 1
    h_norm = h * (n_points / 2) / np.sum(h)
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Define bounds and constraints for optimization
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    cons = LinearConstraint([[1.0]*n_points], [n_points/2], [n_points/2])
    
    # Optimization parameters
    max_iter = 100
    temp_start = 1.0
    temp_end = 0.01
    perturbation_factor = 1.0
    
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Compute current temperature
        temp = temp_start * (temp_end / temp_start) ** (iteration / max_iter)
        
        # Apply random perturbation to the current best solution
        perturbation = np.random.uniform(-perturbation_factor, perturbation_factor, size=n_points)
        candidate = np.clip(best_solution + perturbation, 0, 1)
        
        # Perform local optimization with constraints
        result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
        
        # Accept the solution with probability based on temperature
        if result.success:
            current_c5 = result.fun
            delta = current_c5 - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                best_solution = result.x
                if current_c5 < best_c5:
                    best_c5 = current_c5
    
    return (best_solution.tolist(), best_c5, n_points)
