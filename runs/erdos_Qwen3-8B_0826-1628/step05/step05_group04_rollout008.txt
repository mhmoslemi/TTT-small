import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, LinearConstraint
import random
from deap import base, creator, tools, algorithms

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    ones_minus_h = 1 - h
    corr = np.correlate(h, ones_minus_h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    n_points = 60
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < 15:
            h[i] = 1
        elif i < 30:
            h[i] = 0
        else:
            h[i] = 1
    return h.tolist()

def run(seed=42, budget_s=100, **kwargs):

    creator.create("Individual", list, fitness=creator.FitnessMin)

    # Initialize the toolbox
    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0, 1)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=60)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)

    # Define evaluation function
    def eval_func(individual):
        h = np.array(individual)
        # Normalize to satisfy the constraint
        h_norm = h * (60 / 2) / np.sum(h)
        c5 = evaluate_erdos_solution(h_norm.tolist())
        return c5,

    toolbox.register("evaluate", eval_func)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)

    # Start with an initial population based on initial_h_values
    initial_population = [np.array(initial_h_values()).tolist() for _ in range(50)]

        bounds = Bounds(0, 1)

            constraints=constraints,
            method='SLSQP',

        )
