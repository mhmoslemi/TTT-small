import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    
    # Start with the initial solution
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    total = sum(h)
    h_norm = h * (n_points / 2) / total
    
    # Function to compute C5 for a given h_values
    def compute_c5(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization strategy: Hybrid approach with random restarts and gradient-based refinement
    max_restarts = 10
    max_refinements = 100
    best_c5 = float('inf')
    best_h = h_norm.copy()
    
    for restart in range(max_restarts):
        if time.time() - start_time > budget_s:
            break
        
        # Generate a random initial perturbation
        perturbation = np.random.uniform(-0.25, 0.25, size=n_points)
        h_perturbed = np.clip(h_norm + perturbation, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / sum(h_perturbed)
        
        # Local refinement using SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': integral_constraint})
        
        # Warm start with a single gradient descent step
        def objective(x):
            return compute_c5(x)
        
        result = optimize.minimize(objective, h_perturbed, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
        
        # If the optimization was successful and we found a better solution
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                
                # Further refine the solution with additional local optimizations
                for refine in range(max_refinements):
                    if time.time() - start_time > budget_s:
                        break
                    result = optimize.minimize(objective, best_h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-8)
                    if result.success and result.fun < best_c5:
                        best_c5 = result.fun
                        best_h = result.x
    
    return (best_h.tolist(), best_c5, n_points)
