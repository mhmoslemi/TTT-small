import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Initial setup
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    total = np.sum(h)
    h = h * (n_points / 2) / total
    
    def compute_overlap(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    def objective(h_values):
        return compute_overlap(h_values)
    
    # Optimization parameters
    max_iter = 50
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Initial local optimization
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': integral_constraint})
    result = minimize(objective, h, bounds=bounds, constraints=cons, method='SLSQP')
    
    if result.success:
        best_c5 = result.fun
        best_h = result.x
    
    # Perform several rounds of guided optimization
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Generate a smart perturbation around the current best solution
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = np.clip(best_h + perturbation, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Local optimization
        result = minimize(objective, h_perturbed, bounds=bounds, constraints=cons, method='SLSQP')
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return (best_h.tolist(), best_c5, n_points)
