import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Initialize from the provided starting point
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    # Normalize to satisfy the integral constraint
    h_norm = h * (n_points / 2) / sum(h)
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Define a more sophisticated optimization strategy
    # 1. Start with a few global perturbations
    # 2. Perform local refinement with SLSQP
    # 3. Explore multiple starting points with random mutations
    
    best_h = h_norm.copy()
    best_c5 = compute_overlap(best_h)
    
    # Define a function to mutate the h values with random noise
    def mutate_h(h_values, mutation_strength):
        h_mutated = h_values.copy()
        noise = np.random.normal(0, mutation_strength, size=n_points)
        h_mutated = np.clip(h_mutated + noise, 0, 1)
        h_mutated = h_mutated * (n_points / 2) / sum(h_mutated)
        return h_mutated
    
    # Define a function to perform local optimization
    def local_optimize(h_start):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': integral_constraint})
        result = optimize.minimize(lambda x: compute_overlap(x), h_start, bounds=bounds, constraints=cons, method='SLSQP')
        if result.success:
            return result.x
        else:
            return h_start
    
    # Run the optimization
    for _ in range(30):  # Number of iterations
        if time.time() - start_time > budget_s:
            break
        # Mutate with a small strength
        h_mutated = mutate_h(best_h, mutation_strength=0.05)
        # Local refine
        h_refined = local_optimize(h_mutated)
        # Update best if improvement
        c5_refined = compute_overlap(h_refined)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    
    # Perform a few additional refinements with larger mutations
    for _ in range(10):  # Number of iterations
        if time.time() - start_time > budget_s:
            break
        # Mutate with larger strength
        h_mutated = mutate_h(best_h, mutation_strength=0.1)
        # Local refine
        h_refined = local_optimize(h_mutated)
        # Update best if improvement
        c5_refined = compute_overlap(h_refined)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    
    return (best_h.tolist(), best_c5, n_points)
