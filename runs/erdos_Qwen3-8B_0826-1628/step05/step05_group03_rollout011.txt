import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Start with the initial solution
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    total = sum(h)
    h_norm = h * (n_points / 2) / total
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization: Use a combination of global search and local refinement
    max_iter = 100
    best_c5 = compute_overlap(h_norm)
    best_h = h_norm.copy()
    
    # Try different starting points to avoid local optima
    for start in range(3):
        if start == 0:
            h_start = h_norm.copy()
        elif start == 1:
            h_start = np.random.uniform(0, 1, n_points)
            h_start = h_start * (n_points / 2) / sum(h_start)
        else:
            h_start = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
            h_start = h_start * (n_points / 2) / sum(h_start)
        
        # Use a hybrid optimization approach
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': integral_constraint})
        
        # Use a global optimization method first
        result_global = optimize.differential_evolution(
            lambda x: compute_overlap(x),
            bounds,
            constraints=cons,
            popsize=15,
            tolerance=1e-4,
            maxiter=50,
            seed=seed + start
        )
        
        # Refine the solution locally using SLSQP
        result_local = optimize.minimize(
            lambda x: compute_overlap(x),
            result_global.x,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'maxiter': 50}
        )
        
        # Update the best solution
        if result_local.success:
            current_c5 = result_local.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_local.x.copy()
        
        # Early stopping if time is up
        if time.time() - start_time > budget_s:
            break
    
    return (best_h.tolist(), best_c5, n_points)
