import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = len(initial_h_values)
    initial_h = initial_h_values.copy()
    
    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    def integral_constraint(h):
        return np.sum(h) - n_points / 2
    
    def objective(h):
        return compute_overlap(h)
    
    def bounded_objective(h):
        return compute_overlap(h)
    
    def optimize_solution(h_initial):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': integral_constraint})
        result = optimize.minimize(bounded_objective, h_initial, bounds=bounds, constraints=cons, method='SLSQP')
        if result.success:
            return result.x, result.fun
        return h_initial, compute_overlap(h_initial)
    
    best_h = initial_h.copy()
    best_c5 = compute_overlap(best_h)
    
    # Try to improve the initial solution using local optimization
    best_h, best_c5 = optimize_solution(best_h)
    
    # Try a few perturbations and local optimizations
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        # Random perturbation
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        candidate = np.clip(best_h + perturbation, 0, 1)
        candidate = candidate * (n_points / 2) / np.sum(candidate)
        candidate_h, candidate_c5 = optimize_solution(candidate)
        if candidate_c5 < best_c5:
            best_h, best_c5 = candidate_h, candidate_c5
    
    return (best_h.tolist(), best_c5, n_points)
