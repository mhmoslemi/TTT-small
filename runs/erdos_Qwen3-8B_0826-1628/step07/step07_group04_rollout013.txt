import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, LinearConstraint, NonlinearConstraint

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h_initial = initial_h_values.copy()
    dx = 2.0 / n_points
    total = sum(h_initial)
    h_norm = h_initial * (n_points / 2) / total
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    max_iter = 120
    perturbation_decay = 0.995
    perturbation_scale = 0.3
    alpha = 0.8  # Learning rate for local refinement
    n_perturbations = 30
    
    # Initialize a swarm of candidate solutions
    swarm = [h_norm.copy() for _ in range(n_perturbations)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Perform a global perturbation on each candidate in the swarm
        for i in range(n_perturbations):
            perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
            candidate = np.clip(swarm[i] + perturbation, 0, 1)
            candidate = candidate * (n_points / 2) / sum(candidate)
            
            # Local refinement using SLSQP with tighter bounds and constraints
            bounds = Bounds([0.0] * n_points, [1.0] * n_points)
            cons = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
            
            result = optimize.minimize(lambda x: compute_overlap(x), candidate, 
                                       bounds=bounds, constraints=cons, method='SLSQP')
            
            # Update the swarm member if the optimization succeeded
            if result.success:
                swarm[i] = result.x
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_solution = result.x
    
    return (best_solution.tolist(), best_c5, n_points)
