import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Start with a different initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = h * (n_points / 2) / sum(h)
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * (2.0 / n_points)
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization: Use a hybrid approach with global perturbations and local refinements
    max_iter = 50
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Global perturbation
        perturbation = np.random.uniform(-0.2, 0.2, size=n_points)
        h_perturbed = np.clip(h + perturbation, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / sum(h_perturbed)
        
        # Local refinement using SLSQP with more aggressive constraints
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': integral_constraint})
        # Add bounds to prevent large fluctuations
        result = optimize.minimize(lambda x: compute_overlap(x), h_perturbed, bounds=bounds, constraints=cons, method='SLSQP')
        
        # Update the solution if improvement is found
        if result.success and result.fun < compute_overlap(h):
            h = result.x
    
    # Try a different initialization using a more structured pattern
    n_points = len(initial_h_values)
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            h[i] = 1.0
        elif i % 3 == 1:
            h[i] = 0.5
        else:
            h[i] = 0.0
    h = h * (n_points / 2) / sum(h)
    
    # Local refinement using SLSQP with more aggressive constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': integral_constraint})
    result = optimize.minimize(lambda x: compute_overlap(x), h, bounds=bounds, constraints=cons, method='SLSQP')
    
    # Update the solution if improvement is found
    if result.success and result.fun < compute_overlap(h):
        h = result.x
    
    best_c5 = compute_overlap(h)
    return (h.tolist(), best_c5, n_points)
