import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    h_norm = h * (n_points / 2) / sum(h)
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    n_particles = 60
    max_iter = 150
    inertia_weight = 0.8
    cognitive_coeff = 1.5
    social_coeff = 1.5
    perturbation_scale = 0.3
    local_refinement = True
    
    # Initialize particles
    particles = np.random.uniform(0, 1, (n_particles, n_points))
    particles = particles * (n_points / 2) / np.sum(particles, axis=1, keepdims=True)
    velocities = np.random.uniform(-perturbation_scale, perturbation_scale, (n_particles, n_points))
    personal_best = particles.copy()
    personal_best_values = np.array([compute_overlap(p) for p in particles])
    global_best = particles[np.argmin(personal_best_values)]
    global_best_value = np.min(personal_best_values)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        for i in range(n_particles):
            # Compute personal best
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            cognitive = cognitive_coeff * r1 * (personal_best[i] - particles[i])
            social = social_coeff * r2 * (global_best - particles[i])
            velocities[i] = inertia_weight * velocities[i] + cognitive + social
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_value = compute_overlap(particles[i])
            
            # Update personal best
            if current_value < personal_best_values[i]:
                personal_best[i] = particles[i].copy()
                personal_best_values[i] = current_value
            
            # Update global best
            if current_value < global_best_value:
                global_best = particles[i].copy()
                global_best_value = current_value
        
        # Local refinement using SLSQP on best solution
        if local_refinement and time.time() - start_time < budget_s - 10:
            bounds = [(0, 1) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': integral_constraint})
            result = optimize.minimize(lambda x: compute_overlap(x), global_best, bounds=bounds, constraints=cons, method='SLSQP')
            if result.success:
                global_best = result.x
                global_best_value = result.fun
    
    return (global_best.tolist(), global_best_value, n_points)
