import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start from the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    total = sum(h)
    h_norm = h * (n_points / 2) / total
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * (2.0 / n_points)
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    population_size = 30
    max_iter = 100
    mutation_rate = 0.1
    crossover_rate = 0.7
    alpha = 0.5  # Learning rate for local refinement
    
    # Initialize a population of candidate solutions
    population = [h_norm.copy() for _ in range(population_size)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Generate new solutions via genetic operations
        new_population = []
        for _ in range(population_size):
            # Select two parents
            parent1 = random.choice(population)
            parent2 = random.choice(population)
            
            # Crossover
            child = np.copy(parent1)
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child[crossover_point:] = parent2[crossover_point:]
            
            # Mutation
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < 0.5:
                        child[i] += np.random.uniform(-0.2, 0.2)
                    else:
                        child[i] -= np.random.uniform(-0.2, 0.2)
            
            # Normalize the child to satisfy the integral constraint
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / sum(child)
            
            new_population.append(child)
        
        # Evaluate the new population
        for candidate in new_population:
            # Local refinement using SLSQP
            bounds = [(0, 1) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': integral_constraint})
            try:
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                if result.success:
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_solution = result.x
            except:
                pass
        
        # Update population
        population = new_population
    
    return (best_solution.tolist(), best_c5, n_points)
