import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    total = sum(h)
    h_norm = h * (n_points / 2) / total

    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * (2.0 / n_points)
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Parameters
    num_iterations = 100
    num_sub_iterations = 50
    population_size = 50
    mutation_rate = 0.1
    learning_rate = 0.95
    elite_ratio = 0.2

    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    for i in range(population_size):
        population[i] = population[i] * (n_points / 2) / np.sum(population[i])
    
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    for iteration in range(num_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate population
        pop_fitness = [compute_overlap(sol) for sol in population]
        pop_fitness = np.array(pop_fitness)
        
        # Find elite individuals
        elite_indices = np.argsort(pop_fitness)[:int(elite_ratio * population_size)]
        elites = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = []
        for _ in range(population_size):
            # Select two parents
            parent1, parent2 = random.sample(elites, 2)
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] = np.clip(child[i] + np.random.uniform(-0.1, 0.1), 0, 1)
            
            # Normalize to meet integral constraint
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        # Replace population
        population = new_population
    
    # Evaluate best solution
    best_solution = np.min(population, key=compute_overlap)
    best_c5 = compute_overlap(best_solution)
    
    return (best_solution.tolist(), best_c5, n_points)
