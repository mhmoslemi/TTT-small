import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    pop_size = 50
    max_iter = 150
    mutation_rate = 0.15
    crossover_rate = 0.8
    elite_size = 5
    mutation_decay = 0.98
    
    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
    population = [h * (n_points / 2) / sum(h) for h in population]
    population = [np.clip(h, 0, 1) for h in population]
    
    # Evaluate initial population
    fitness = [compute_overlap(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_c5 = fitness[best_idx]
    
    # Evolutionary algorithm
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate all individuals
        fitness = [compute_overlap(h) for h in population]
        # Sort population based on fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Select elites
        elite_population = sorted_population[:elite_size]
        elite_fitness = sorted_fitness[:elite_size]
        
        # Generate new population
        new_population = []
        for _ in range(pop_size - elite_size):
            # Select parents
            parent1 = sorted_population[np.random.randint(elite_size, pop_size)]
            parent2 = sorted_population[np.random.randint(elite_size, pop_size)]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                child = np.zeros(n_points)
                for i in range(n_points):
                    if np.random.rand() < 0.5:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
            else:
                child = np.random.uniform(0, 1, n_points)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation_strength = np.random.uniform(0, mutation_rate)
                child += mutation_strength * np.random.randn(n_points)
                child = np.clip(child, 0, 1)
            
            # Ensure integral constraint
            child = child * (n_points / 2) / sum(child)
            new_population.append(child)
        
        # Combine elite and new population
        population = elite_population + new_population
        population = [np.clip(h, 0, 1) for h in population]
        
        # Update best solution
        for i in range(len(population)):
            if fitness[i] < best_c5:
                best_c5 = fitness[i]
                best_h = population[i]
    
    return (best_h.tolist(), best_c5, n_points)
