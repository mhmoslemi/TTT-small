```python
import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_sum = np.sum(initial_h_values)
    initial_h = initial_h_values.copy()
    initial_h_norm = initial_h * (n_points / 2) / initial_sum
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Start with the initial solution
    best_solution = initial_h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Optimization parameters
    swarm_size = 10
    max_iter = 150
    perturbation_decay = 0.99
    initial_perturbation = 0.15
    
    # Initialize a swarm of candidate solutions
    swarm = [best_solution.copy() for _ in range(swarm_size)]
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Perform a global perturbation on each candidate in the swarm
        for i in range(swarm_size):
            perturbation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
            candidate = np.clip(swarm[i] + perturbation, 0, 1)
            
            # Local refinement using SLSQP
            bounds = [(0, 1) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': integral_constraint})
            result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
            
            # Update the swarm member if the optimization succeeded
            if result.success:
                swarm[i] = result.x
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_solution = result.x
        
        # Gradually reduce perturbation range
        initial_perturbation *= perturbation_decay
    
    return (best_solution.tolist(), best_c5, n_points)
```
