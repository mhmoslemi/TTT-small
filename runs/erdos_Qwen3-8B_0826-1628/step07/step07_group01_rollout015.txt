import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    swarm_size = 30
    max_iter = 100
    perturbation_decay = 0.95
    initial_perturbation = 0.15
    local_refinement_steps = 20
    local_refinement_decay = 0.9
    mutation_rate = 0.3
    crossover_rate = 0.7
    
    # Initialize a swarm of candidate solutions
    swarm = [h_norm.copy() for _ in range(swarm_size)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    # Function to evolve swarm with a balance of exploration and exploitation
    def evolve_swarm(swarm, best_solution, best_c5, initial_perturbation, perturbation_decay, max_iter, n_points, dx, compute_overlap, integral_constraint):
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            
            # Mutation and crossover for exploration
            for i in range(swarm_size):
                # Mutation
                perturbation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
                candidate = np.clip(swarm[i] + perturbation, 0, 1)
                candidate = candidate * (n_points / 2) / sum(candidate)
                
                # Crossover
                j = np.random.randint(0, swarm_size)
                crossover_mask = np.random.rand(n_points) < crossover_rate
                crossover_candidate = np.where(crossover_mask, candidate, swarm[j])
                crossover_candidate = crossover_candidate * (n_points / 2) / sum(crossover_candidate)
                
                # Local refinement with SLSQP
                bounds = [(0, 1) for _ in range(n_points)]
                cons = ({'type': 'eq', 'fun': integral_constraint})
                
                # Refine crossover candidate
                result = optimize.minimize(lambda x: compute_overlap(x), crossover_candidate, bounds=bounds, constraints=cons, method='SLSQP')
                if result.success:
                    if result.fun < best_c5:
                        best_c5 = result.fun
                        best_solution = result.x
                        swarm[i] = result.x
                
                # Refine original candidate
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                if result.success:
                    if result.fun < best_c5:
                        best_c5 = result.fun
                        best_solution = result.x
                        swarm[i] = result.x
            
            # Gradually reduce perturbation range and local refinement
            initial_perturbation *= perturbation_decay
            # Introduce some local refinement with random perturbation
            for i in range(swarm_size):
                perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
                candidate = np.clip(swarm[i] + perturbation, 0, 1)
                candidate = candidate * (n_points / 2) / sum(candidate)
                result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                
                if result.success:
                    if result.fun < best_c5:
                        best_c5 = result.fun
                        best_solution = result.x
                        swarm[i] = result.x
            
        return best_solution, best_c5
    
    # Run the optimization
    best_solution, best_c5 = evolve_swarm(swarm, best_solution, best_c5, initial_perturbation, perturbation_decay, max_iter, n_points, dx, compute_overlap, integral_constraint)
    
    return (best_solution.tolist(), best_c5, n_points)
