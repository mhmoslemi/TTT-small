import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    total = sum(h)
    h_norm = h * (n_points / 2) / total
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Define the objective function for optimization
    def objective_function(x):
        return compute_overlap(x)
    
    # Define the bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization: Use a hybrid approach with a combination of global search and local refinement
    max_iter = 20
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Generate a random perturbation
        perturbation = np.random.uniform(-0.15, 0.15, size=n_points)
        h_perturbed = np.clip(h_norm + perturbation, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / sum(h_perturbed)
        
        # Local refinement using SLSQP
        cons = ({'type': 'eq', 'fun': integral_constraint})
        result = optimize.minimize(objective_function, h_perturbed, bounds=bounds, constraints=cons, method='SLSQP')
        
        # Update the solution if improvement is found
        if result.success and result.fun < compute_overlap(h_norm):
            h_norm = result.x
    
    # Try a different initialization with a larger perturbation
    perturbation = np.random.uniform(-0.2, 0.2, size=n_points)
    h_perturbed = np.clip(h_norm + perturbation, 0, 1)
    h_perturbed = h_perturbed * (n_points / 2) / sum(h_perturbed)
    
    # Local refinement using SLSQP
    cons = ({'type': 'eq', 'fun': integral_constraint})
    result = optimize.minimize(objective_function, h_perturbed, bounds=bounds, constraints=cons, method='SLSQP')
    
    # Update the solution if improvement is found
    if result.success and result.fun < compute_overlap(h_norm):
        h_norm = result.x
    
    best_c5 = compute_overlap(h_norm)
    return (h_norm.tolist(), best_c5, n_points)
