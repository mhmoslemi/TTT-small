import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    num_particles = 50
    max_iter = 100
    inertia_weight = 0.8
    cognitive_coeff = 1.5
    social_coeff = 1.5
    perturbation_decay = 0.98
    n_points = len(initial_h_values)
    initial_perturbation = 0.3
    
    # Initialize swarm
    swarm = [np.random.uniform(0, 1, n_points) for _ in range(num_particles)]
    personal_best = [s.copy() for s in swarm]
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(num_particles)]
    global_best = np.copy(swarm[0])
    best_c5 = compute_overlap(global_best)
    
    # Function to evaluate particle
    def evaluate_particle(particle):
        cons = ({'type': 'eq', 'fun': lambda x: sum(x) - (n_points / 2)})
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(lambda x: compute_overlap(x), particle, bounds=bounds, constraints=cons, method='SLSQP')
        return result.x, result.fun
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        for i in range(num_particles):
            # Update personal best
            candidate = swarm[i].copy()
            if compute_overlap(candidate) < best_c5:
                personal_best[i] = candidate
                best_c5 = compute_overlap(candidate)
                if best_c5 < compute_overlap(global_best):
                    global_best = candidate.copy()
            
            # Update velocity and position
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = inertia_weight * velocities[i] + cognitive_coeff * r1 * (personal_best[i] - swarm[i]) + social_coeff * r2 * (global_best - swarm[i])
            swarm[i] = np.clip(swarm[i] + velocities[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / sum(swarm[i])
        
        # Gradual perturbation and refinement
        initial_perturbation *= perturbation_decay
        for i in range(num_particles):
            perturbation = np.random.uniform(-initial_perturbation, initial_perturbation, size=n_points)
            candidate = np.clip(swarm[i] + perturbation, 0, 1)
            candidate = candidate * (n_points / 2) / sum(candidate)
            candidate, _ = evaluate_particle(candidate)
            if compute_overlap(candidate) < best_c5:
                personal_best[i] = candidate.copy()
                best_c5 = compute_overlap(candidate)
                if best_c5 < compute_overlap(global_best):
                    global_best = candidate.copy()
    
    return (global_best.tolist(), best_c5, n_points)
