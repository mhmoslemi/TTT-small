import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    h_norm = h * (n_points / 2) / sum_h
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    n_particles = 50
    max_iter = 200
    inertia = 0.8
    cognitive_coeff = 1.2
    social_coeff = 1.2
    perturbation_decay = 0.99
    
    # Initialize particle swarm
    particles = np.random.uniform(0, 1, (n_particles, n_points))
    particles = particles * (n_points / 2) / np.sum(particles, axis=1, keepdims=True)
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    personal_best_positions = particles.copy()
    personal_best_values = np.array([compute_overlap(p) for p in particles])
    global_best_position = particles[np.argmin(personal_best_values)]
    global_best_value = np.min(personal_best_values)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        for i in range(n_particles):
            # Update velocity and position
            r1, r2 = np.random.rand(2, n_points)
            velocities[i] = inertia * velocities[i] + cognitive_coeff * r1 * (personal_best_positions[i] - particles[i]) + social_coeff * r2 * (global_best_position - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate overlap
            current_value = compute_overlap(particles[i])
            
            # Update personal best
            if current_value < personal_best_values[i]:
                personal_best_values[i] = current_value
                personal_best_positions[i] = particles[i].copy()
        
        # Update global best
        current_global_value = np.min(personal_best_values)
        if current_global_value < global_best_value:
            global_best_value = current_global_value
            global_best_position = personal_best_positions[np.argmin(personal_best_values)].copy()
        
        # Gradually reduce perturbation range
        perturbation_decay *= 0.999
    
    return (global_best_position.tolist(), global_best_value, n_points)
