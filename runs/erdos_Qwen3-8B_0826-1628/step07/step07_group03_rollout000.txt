import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    total = sum(h)
    h_norm = h * (n_points / 2) / total
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * (2.0 / n_points)
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    swarm_size = 50
    max_iter = 150
    perturbation_decay = 0.995
    alpha = 0.5  # Learning rate for local refinement
    local_refinement_steps = 20
    
    # Initialize a swarm of candidate solutions
    swarm = [h_norm.copy() for _ in range(swarm_size)]
    best_solution = h_norm.copy()
    best_c5 = compute_overlap(best_solution)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Perform a global perturbation on each candidate in the swarm
        for i in range(swarm_size):
            # Adaptive perturbation based on iteration
            if iteration < 20:
                perturbation = np.random.uniform(-0.4, 0.4, size=n_points)
            elif iteration < 80:
                perturbation = np.random.uniform(-0.2, 0.2, size=n_points)
            else:
                perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            
            candidate = np.clip(swarm[i] + perturbation, 0, 1)
            candidate = candidate * (n_points / 2) / sum(candidate)
            
            # Local refinement using SLSQP
            bounds = [(0, 1) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': integral_constraint})
            try:
                for _ in range(local_refinement_steps):
                    result = optimize.minimize(lambda x: compute_overlap(x), candidate, bounds=bounds, constraints=cons, method='SLSQP')
                    if result.success:
                        candidate = result.x
                    else:
                        break
                if compute_overlap(candidate) < best_c5:
                    best_c5 = compute_overlap(candidate)
                    best_solution = candidate
                swarm[i] = candidate
            except:
                pass
    
    return (best_solution.tolist(), best_c5, n_points)
