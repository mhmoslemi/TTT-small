import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    # Normalize to satisfy the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * (2.0 / n_points)
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - (n_points / 2)
    
    # Optimization parameters
    population_size = 50
    generations = 50
    mutation_rate = 0.1
    crossover_rate = 0.7
    elite_size = 5
    
    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    for i in range(population_size):
        population[i] = population[i] * (n_points / 2) / np.sum(population[i])
    
    best_solution = h.copy()
    best_c5 = compute_overlap(best_solution)
    
    for generation in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = np.array([compute_overlap(ind) for ind in population])
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_solution = population[best_idx].copy()
        
        # Select parents
        parents = np.random.choice(population_size, size=population_size, p=fitness / np.sum(fitness))
        parents = [population[i] for i in parents]
        
        # Crossover
        offspring = []
        for i in range(population_size // 2):
            parent1 = parents[2 * i]
            parent2 = parents[2 * i + 1]
            child = np.zeros(n_points)
            for j in range(n_points):
                if np.random.rand() < crossover_rate:
                    child[j] = parent1[j]
                else:
                    child[j] = parent2[j]
            # Apply mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[j] = np.random.uniform(0, 1)
            # Normalize
            child = child * (n_points / 2) / np.sum(child)
            offspring.append(child)
        
        # Replace population
        population = np.array(population)
        offspring = np.array(offspring)
        population = np.vstack((population, offspring))
        population = np.unique(population, axis=0)
        population = population[:population_size]
    
    return (best_solution.tolist(), best_c5, n_points)
