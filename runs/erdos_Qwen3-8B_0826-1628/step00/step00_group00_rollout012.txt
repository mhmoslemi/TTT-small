import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    # Set a random seed for reproducibility
    np.random.seed(seed)
    
    # Define the number of points for the function h
    n_points = 45  # Start with the initial point count (from initial_h_values)
    
    # Get initial h values from the provided function
    h_initial = initial_h_values()
    h_initial = np.array(h_initial)
    
    # Check if the initial h meets the constraints
    if not np.isclose(np.sum(h_initial) * 2.0 / n_points, 1.0):
        raise ValueError("Initial h values do not satisfy the integral constraint.")
    
    # Define the function to evaluate C5
    def evaluate_c5(h):
        h = h.reshape(-1)
        h = np.pad(h, (0, 1), mode='constant')  # Pad to make convolution work
        conv = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(conv) * (2.0 / n_points)
        return max_overlap
    
    # Define the objective function for optimization
    def objective(x):
        return evaluate_c5(x)
    
    # Define the constraints for the optimization problem
    def constraints(x):
        # Constraint: sum of h * dx = 1 (equivalently, sum(h) == n_points / 2)
        sum_h = np.sum(x)
        return [np.isclose(sum_h, n_points / 2)]
    
    # Define the bounds for each element of h
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Perform a local optimization starting from the initial h values
    start_time = time.time()
    result = optimize.minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        jac=None,
        options={'ftol': 1e-9, 'maxiter': 1000}
    )
    
    # Check if the optimization completed within the budget
    if time.time() - start_time > budget_s:
        print("Warning: Optimization exceeded budget time.")
        # Return the initial solution if time is up
        return (h_initial, evaluate_c5(h_initial), n_points)
    
    # Extract the optimized h values
    optimized_h = result.x
    c5_bound = result.fun
    print(f"Optimized C5: {c5_bound}")
    
    return (optimized_h, c5_bound, n_points)
