import numpy as np
import scipy.optimize as optimize
import time

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = np.array(initial_h, dtype=float)
    
    def evaluate_c5(h_values):
        h = np.array(h_values)
        h = np.concatenate([h, h[:n_points//2]])  # Pad for full correlation
        shifted_h = np.roll(h, shift=1, axis=0)
        overlap = np.correlate(h, 1 - shifted_h, mode="full")[n_points-1:2*n_points-1]
        overlap *= dx
        return np.max(overlap)
    
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        return evaluate_c5(h)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Optimization
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    start_time = time.time()
    result = optimize.minimize(
        objective, 
        h, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        print("Budget exceeded, returning initial solution")
        return (initial_h, evaluate_c5(initial_h), n_points)
    
    best_h = result.x
    best_c5 = evaluate_c5(best_h)
    
    return (best_h, best_c5, n_points)
