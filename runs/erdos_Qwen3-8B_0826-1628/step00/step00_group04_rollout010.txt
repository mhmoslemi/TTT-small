import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.signal import correlate
from functools import partial

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(42) * 2 - 1  # Placeholder; actual initial values are pre-defined

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    n_points = len(best_h)
    dx = 2.0 / n_points

    # Function to evaluate C5
    def evaluate(h):
        h = np.clip(h, 0, 1)
        # Ensure integral constraint is satisfied
        total_area = np.sum(h) * dx
        if not np.isclose(total_area, 1.0):
            return float('inf')  # Penalize violation of the constraint
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Function to optimize
    def optimize_h(initial_h, max_iter=100):
        result = minimize(
            evaluate,
            initial_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(len(initial_h))],
            options={'maxiter': max_iter, 'disp': False}
        )
        if result.success:
            return result.x
        else:
            return initial_h

    # Try different starting points and refine
    for _ in range(3):
        # Try with initial guess
        h = optimize_h(best_h.copy(), max_iter=50)
        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Try with random perturbation
        perturbation = np.random.normal(0, 0.05, size=len(h))
        h = optimize_h(h + perturbation, max_iter=50)
        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Early stopping if budget is exhausted
        if time.time() - start_time > budget_s - 1:
            break

    # Final check
    best_h = np.clip(best_h, 0, 1)
    final_c5 = evaluate(best_h)
    return best_h, final_c5, len(best_h)
