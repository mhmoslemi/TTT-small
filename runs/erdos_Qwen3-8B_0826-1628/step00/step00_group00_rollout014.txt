import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Set up parameters
    n_points = 45  # Start with the existing 45-point solution
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    def objective(h):
        # Calculate correlation and find max overlap
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def update_h(h, t):
        # Simple heuristic: move points to reduce overlap
        # Find indices where current h is 0 or 1 and consider swapping
        # This is a basic heuristic to guide the search
        h = h.copy()
        n = len(h)
        for i in range(n):
            if h[i] < 0.5:
                # Try increasing h[i]
                if np.sum(h) < n / 2:
                    h[i] += 0.05
            elif h[i] > 0.5:
                # Try decreasing h[i]
                if np.sum(h) > n / 2:
                    h[i] -= 0.05
        # Ensure h stays within [0,1] and sum is exactly n/2
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - n / 2) > 1e-6:
            h = h / total * (n / 2)
        return h
    
    # Local search optimization
    best_h = h
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        h = update_h(h, 0)
        c5 = objective(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    # Try a small-scale optimization using CVXPY
    # Convert h to a vector and attempt an optimization problem
    h_var = cp.Variable(n_points)
    constraints = [cp.sum(h_var) == n_points / 2, 0 <= h_var, h_var <= 1]
    # Objective is to minimize the maximum overlap integral
    # This is a nonlinear optimization problem
    # We approximate by minimizing the maximum of the correlations
    # For simplicity, we use a heuristic approach here
    # For real optimization, a more sophisticated approach is needed
    # In this case, we use a small-scale optimization
    h_opt = best_h
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        h_opt = update_h(h_opt, 0)
        c5 = objective(h_opt)
        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5
    
    # Final evaluation
    final_c5 = best_c5
    return best_h, final_c5, n_points
