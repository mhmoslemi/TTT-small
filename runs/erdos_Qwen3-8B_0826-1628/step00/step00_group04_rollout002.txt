import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.array([0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333,
                             0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333,
                             0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333,
                             0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333,
                             0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333])

def run(seed=42, budget_s=100, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 100  # Adjust based on time budget and performance
    # Start with initial guess
    h = initial_h_values[:n_points].copy()
    # Normalize to meet the integral constraint
    h /= (np.sum(h) / (n_points / 2))
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Use genetic algorithm with early stopping
    start_time = time.time()
    
    def fitness_function(x):
        nonlocal best_c5
        h = x.copy()
        h = np.clip(h, 0, 1)
        # Ensure the integral constraint is satisfied
        h /= (np.sum(h) / (n_points / 2))
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    # Genetic algorithm parameters
    pop_size = 20
    generations = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = np.random.rand(pop_size, n_points)
    population = np.clip(population, 0, 1)
    population /= (np.sum(population, axis=1)[:, np.newaxis] / (n_points / 2))
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = np.array([fitness_function(ind) for ind in population])
        
        # Select parents
        parents = population[np.argsort(fitness)[:int(pop_size / 2)]]
        
        # Generate offspring
        offspring = []
        for _ in range(pop_size - len(parents)):
            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = np.random.rand(n_points)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, 0.05, n_points)
                child += mutation
                child = np.clip(child, 0, 1)
            
            # Normalize
            child /= (np.sum(child) / (n_points / 2))
            offspring.append(child)
        
        population = np.vstack([parents, offspring])
    
    # Check best solution
    best_h = best_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
