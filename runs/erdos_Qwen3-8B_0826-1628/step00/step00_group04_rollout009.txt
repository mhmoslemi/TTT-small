import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp
from functools import partial

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.concatenate([np.zeros(n_points), h[:-1]], axis=0)
    corr = signal.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for good balance
    dx = 2.0 / n_points
    h_initial = initial_h_values()
    h_initial = h_initial[:n_points] * (n_points / 2)  # Adjust to satisfy the integral constraint
    h_initial = np.clip(h_initial, 0, 1)

    best_c5 = evaluate_erdos_solution(h_initial)
    best_h = h_initial.copy()

    # Start with a few iterations of gradient descent
    for _ in range(5):
        # Define objective function
        def obj_func(h_flat):
            h = h_flat.reshape(n_points)
            # Enforce the integral constraint
            if np.abs(np.sum(h) - n_points / 2) > 1e-6:
                return float('inf')
            # Compute the overlap integral
            shifted = np.concatenate([np.zeros(n_points), h[:-1]], axis=0)
            corr = signal.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return c5

        # Gradient descent step
        h = best_h.copy()
        h = np.clip(h, 0, 1)
        h = h.reshape(n_points)
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus.reshape(n_points)
            grad[i] = (obj_func(h_plus) - obj_func(h)) / 1e-5
        h -= 0.01 * grad
        h = np.clip(h, 0, 1)
        h = h.reshape(n_points)
        current_c5 = obj_func(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    # Try increasing n_points if time allows
    remaining_time = budget_s - time.time() + start_time
    if remaining_time > 5:
        n_points = 200
        dx = 2.0 / n_points
        h_initial = initial_h_values()
        h_initial = h_initial[:n_points] * (n_points / 2)
        h_initial = np.clip(h_initial, 0, 1)
        best_c5 = evaluate_erdos_solution(h_initial)
        best_h = h_initial.copy()

        for _ in range(5):
            h = best_h.copy()
            h = np.clip(h, 0, 1)
            h = h.reshape(n_points)
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus.reshape(n_points)
                grad[i] = (obj_func(h_plus) - obj_func(h)) / 1e-5
            h -= 0.01 * grad
            h = np.clip(h, 0, 1)
            h = h.reshape(n_points)
            current_c5 = obj_func(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points

start_time = time.time()
run()
