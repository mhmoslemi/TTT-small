import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import sys
import cvxpy as cp

# Pre-imported functions
# evaluate_erdos_solution is not used here
# initial_h_values is available as a starting point

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Start with a reasonable number of points
    n_points = 100
    max_n_points = 500  # Limit to avoid exceeding budget
    
    # Initial guess from the provided starting point
    if hasattr(initial_h_values, '__iter__') and len(initial_h_values) > 0:
        initial_h = np.array(initial_h_values)
        if len(initial_h) < n_points:
            initial_h = np.concatenate([initial_h, np.zeros(n_points - len(initial_h))])
        initial_h = initial_h[:n_points]
    else:
        # Fallback to uniform distribution
        initial_h = np.full(n_points, 0.5)
    
    # Scale to meet the integral constraint
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    # Optimization loop
    for n_points in range(100, max_n_points + 1, 50):
        dx = 2.0 / n_points
        h = initial_h[:n_points].copy()
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        
        # Use a simple optimization method to refine the solution
        def objective_func(x):
            h_vals = x.reshape(n_points)
            h_vals = np.clip(h_vals, 0, 1)
            integral = np.sum(h_vals) * dx
            if np.abs(integral - 1) > 1e-6:
                return 1e6  # Penalize for not satisfying the constraint
            
            # Compute the overlap integral
            corr = signal.correlate(h_vals, 1 - h_vals, mode='full')
            max_corr = np.max(corr)
            c5 = max_corr * dx
            return c5
        
        # Use L-BFGS-B to minimize the overlap integral
        try:
            result = optimize.minimize(
                objective_func,
                h,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                options={'maxiter': 100, 'ftol': 1e-6}
            )
            h_opt = result.x
            c5 = objective_func(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
        except:
            pass
        
        # Check for time budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final check with the best solution found
    n_points = len(best_h)
    dx = 2.0 / n_points
    h_vals = best_h
    h_vals = np.clip(h_vals, 0, 1)
    integral = np.sum(h_vals) * dx
    corr = signal.correlate(h_vals, 1 - h_vals, mode='full')
    max_corr = np.max(corr)
    final_c5 = max_corr * dx
    
    return (best_h.tolist(), final_c5, n_points)
