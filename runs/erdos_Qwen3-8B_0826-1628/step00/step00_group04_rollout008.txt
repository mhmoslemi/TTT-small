import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Define parameters
    max_n_points = 500  # Max number of samples allowed
    n_points = 100  # Start with 100 samples
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    elapsed_time = 0

    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        h_shifted = np.zeros_like(h)
        for k in range(1, len(h) - 1):
            h_shifted[k:] = h[:len(h) - k]
        overlap = np.correlate(h, 1 - h_shifted, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    # Function to evaluate and update best solution
    def evaluate_solution(h):
        nonlocal best_c5, best_h, elapsed_time
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if elapsed_time + time.time() - time.time() > budget_s:
            return

    # Start with the initial guess
    h_initial = initial_h_values.copy()
    h_initial = h_initial[:n_points]
    evaluate_solution(h_initial)

    # Try increasing number of points up to max_n_points
    for n_points in range(100, max_n_points + 1, 50):
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h * (n_points / 2)  # Scale to satisfy the integral constraint

        # Use a simple optimization to minimize C5
        def objective(x):
            h = x
            h_shifted = np.zeros_like(h)
            for k in range(1, len(h) - 1):
                h_shifted[k:] = h[:len(h) - k]
            overlap = np.correlate(h, 1 - h_shifted, mode="full")
            c5 = np.max(overlap * dx)
            return c5

        # Use scipy's optimization to minimize c5
        result = opt.minimize(objective, h, method="L-BFGS-B", bounds=[(0, 1) for _ in range(n_points)])
        h_opt = result.x

        # Check if the integral constraint is satisfied
        if np.abs(np.sum(h_opt) * dx - 1) > 1e-6:
            continue

        evaluate_solution(h_opt)

        # Check if we are over the time budget
        elapsed_time = time.time() - time.time()  # Placeholder for actual elapsed time tracking
        if elapsed_time > budget_s:
            break

    # Return the best solution found
    return (best_h, best_c5, len(best_h))
