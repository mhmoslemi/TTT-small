```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for speed
    max_n_points = 200  # Max points to try within budget

    # Starting from initial_h_values
    initial_h = initial_h_values
    initial_h = np.array(initial_h[:n_points])  # Truncate to n_points
    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)  # Ensure sum is n_points / 2

    # Function to compute overlap integral
    def compute_c5(h):
        h = np.array(h)
        shift = np.arange(len(h))
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / len(h))
        return c5

    # Function to optimize h with CVXPY
    def optimize_h(h_prev):
        n = len(h_prev)
        h = cp.Variable(n)
        constraints = [
            cp.sum(h) == n / 2,
            0 <= h,
            h <= 1
        ]
        obj_func = cp.Maximize(cp.max(cp.sum(h * (1 - h[shift])) for shift in range(n)))
        prob = cp.Problem(obj_func, constraints)
        result = prob.solve(solver='ECOS', abstol=1e-8, reltol=1e-8)
        if result is None:
            return h_prev
        return h.value

    # Try increasing n_points within budget
    for n_points in range(100, max_n_points + 1, 50):
        if time.time() - start_time > budget_s:
            break
        h = initial_h_values[:n_points]
        h = np.array(h)
        h = h / np.sum(h) * (n_points / 2)
        h = optimize_h(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h

    # Try perturbing and refining the solution
    if best_h is not None:
        perturbations = np.random.normal(0, 0.05, size=len(best_h))
        h_perturb = np.clip(best_h + perturbations, 0, 1)
        h_perturb = h_perturb / np.sum(h_perturb) * (len(h_perturb) / 2)
        current_c5 = compute_c5(h_perturb)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_perturb

    # Final refinement
    h_final = best_h
    h_final = optimize_h(h_final)
    final_c5 = compute_c5(h_final)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h_final

    return (best_h, best_c5, len(best_h))
```
