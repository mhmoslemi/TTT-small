def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    overlap = np.correlate(h, 1.0 - h, mode="full") * dx
    c5 = np.max(overlap)
    return c5

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start from the initial construction
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    
    # Check if we can reduce the number of points
    for n in range(n_points, 10, -1):
        if n % 2 == 0:
            initial_h = initial_h[:n]
            n_points = n
            break
    
    # If we couldn't reduce, use the initial number
    if n_points == len(initial_h_values):
        n_points = len(initial_h_values)
    
    # Define the function to evaluate the overlap
    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        overlap = np.correlate(h, 1.0 - h, mode="full")
        dx = 2.0 / len(h)
        c5 = np.max(overlap * dx)
        return c5
    
    # Define the constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define the bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    h_initial = initial_h.copy()
    
    # Start the optimization
    start_time = time.time()
    
    # Try different optimization methods
    for method in ['L-BFGS-B', 'SLSQP', 'TNC']:
        try:
            # Create a function to minimize
            def objective(h):
                return compute_c5(h)
            
            # Set up constraints
            constraints = [{'type': 'eq', 'fun': constraint_sum}]
            
            # Perform optimization
            result = optimize.minimize(
                objective,
                h_initial,
                method=method,
                bounds=bounds,
                constraints=constraints,
                tol=1e-8,
                options={'maxiter': 1000, 'disp': False}
            )
            
            if result.success:
                h_best = result.x
                c5_best = compute_c5(h_best)
                return (h_best.tolist(), c5_best, n_points)
            
            # If not successful, try next method
            h_initial = result.x
        except:
            # If optimization fails, try next method
            continue
        
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # If no method succeeded, return the initial construction
    return (initial_h.tolist(), evaluate_erdos_solution(initial_h), n_points)
