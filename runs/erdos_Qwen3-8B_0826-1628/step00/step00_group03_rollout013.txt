import scipy.optimize as optimize
import time

# Pre-imported functions

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 100  # We'll try different values within budget
    dx = 2.0 / n_points
    max_iter = 100  # Number of iterations for optimization
    tolerance = 1e-5  # Convergence tolerance
    n_trials = 5  # Number of trials to run for robustness

    # Initial guess
    initial_h = initial_h_values.copy()
    initial_h = initial_h[:n_points]  # Ensure it fits the n_points
    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)  # Normalize to satisfy the constraint
    
    # Function to compute overlap integral
    def compute_c5(h):
        h = np.array(h)
        # Normalize to ensure the integral is 1
        h = h / np.sum(h) * (n_points / 2)  # Ensure sum(h) * dx = 1
        # Compute the cross-correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Compute the max overlap
        c5 = np.max(corr) * dx
        return c5

    # Function to find the optimal h using optimization
    def optimize_h(initial_h):
        h = cp.Variable(n_points)
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,  # Ensure sum(h) * dx = 1
            0 <= h,
            h <= 1
        ]
        # Objective: minimize the maximum overlap
        # Approximate the maximum using a convex relaxation
        # We'll use a grid search over shifted h(x + k) for k in a range
        
        # Sample k values to check
        k_values = np.arange(-n_points + 1, n_points)
        k_values = k_values[np.abs(k_values) < n_points // 2]  # Avoid wrap-around
        
        # Objective: find the maximum over k of the integral
        # We'll use a convex relaxation by finding the maximum over a set of k's
        c5 = 0.0
        for k in k_values:
            # Compute the shifted h
            shifted_h = np.roll(h.value, k)
            # Compute the overlap integral
            overlap = np.sum(h.value * (1 - shifted_h)) * dx
            c5 = max(c5, overlap)
        
        # We'll minimize the maximum overlap
        objective = cp.Minimize(c5)
        
        # Solve the optimization problem
        prob = cp.Problem(objective, constraints)

            return h.value
        else:
            return initial_h
