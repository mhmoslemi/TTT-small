import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from functools import partial

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Example initial values (should be predefined)

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Set up parameters
    n_points = 100  # Start with 100 samples (can be increased later if needed)
    dx = 2.0 / n_points
    max_iterations = 100
    improvement_threshold = 1e-5
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Function to compute C₅ for a given h
    def compute_c5(h):
        return evaluate_erdos_solution(h)
    
    # Initial guess
    h = initial_h_values[:n_points].copy()
    h = h / np.sum(h) * (n_points / 2)  # Ensure the integral is 1
    h = np.clip(h, 0, 1)
    
    for _ in range(max_iterations):
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Optimization step
        # We use a simple gradient descent approach to minimize C₅
        # We compute the derivative of C₅ with respect to h
        
        # Compute the overlap integral at each point
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        # Find the maximum overlap value (C₅)
        max_overlap = np.max(overlap)
        
        # Compute derivative of C₅ with respect to h
        # This is complex; instead, we approximate it with finite differences
        # For each point, we adjust h slightly and recompute C₅
        
        # Create a perturbation matrix for finite differences
        perturbation = 1e-4
        gradient = np.zeros_like(h)
        for i in range(n_points):
            h_perturb = h.copy()
            h_perturb[i] += perturbation
            h_perturb = np.clip(h_perturb, 0, 1)
            c5_perturb = compute_c5(h_perturb)
            gradient[i] = (c5_perturb - current_c5) / perturbation
        
        # Update h in the direction of minimizing C₅
        # We also add a small regularization term to avoid getting stuck in local minima
        h -= gradient * 0.01
        
        # Clip h to [0, 1] and ensure integral constraint
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Return the best found solution
    return (best_h, best_c5, n_points)
