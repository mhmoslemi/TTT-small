def run(seed=42, budget_s=100, **kwargs):

    def evaluate(h):
        if len(h) < 2:

        return np.max(corr) * (2.0 / len(h))

    def optimize_step_function(h_init, n_points, max_iter=100, tol=1e-5):

            h_var = cp.Variable(n_points)

                cp.sum(h_var) == n_points / 2,
                0 <= h_var,
                h_var <= 1
            ]

            corr = np.correlate(h_var.value, 1 - h_var.value, mode="full")
            c5 = cp.Max(cp.multiply(corr, (2.0 / n_points)))
            
            # Solve the optimization problem
            prob = cp.Problem(cp.Minimize(c5), constraints)

                pass  # Skip if optimization fails

            if np.abs(current_c5 - best_c5) < tol:
                break
