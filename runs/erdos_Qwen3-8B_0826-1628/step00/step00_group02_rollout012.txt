def run(seed=42, budget_s=100, **kwargs):

    h_values = np.copy(initial_h_values)
    
    # Define the function to compute the overlap integral
    def evaluate_overlap(h):
        h = np.array(h)
        h = np.concatenate((h, h[0:1]))  # Pad for full convolution
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap * (2.0 / n_points))  # dx = 2 / n_points

    # Objective function for optimization
    def objective(h):
        return evaluate_overlap(h)

    # Constraints: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Bounds: 0 <= h[i] <= 1
    bounds = Bounds(0, 1)

    # Initial optimization
    result = minimize(
        objective,
        h_values,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'ftol': 1e-8, 'maxiter': 50}
    )

    best_h = result.x
    best_c5 = evaluate_overlap(best_h)
    
    # Refinement using local search
    for _ in range(5):
        # Perturb the solution slightly
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = np.clip(best_h + perturb, 0, 1)
        perturbed_h = np.array(perturbed_h, dtype=np.float64)
        # Enforce sum constraint
        sum_perturbed = np.sum(perturbed_h)
        if sum_perturbed != n_points / 2:
            scaling = (n_points / 2) / sum_perturbed
            perturbed_h *= scaling
        # Evaluate
        current_c5 = evaluate_overlap(perturbed_h)
        if current_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = current_c5

    # Final evaluation
    final_c5 = evaluate_overlap(best_h)
    return (best_h.tolist(), final_c5, n_points)
