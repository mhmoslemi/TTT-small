import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    h_shifted = np.zeros((n_points, n_points))
    for i in range(n_points):
        h_shifted[i, :] = h
    corr = np.zeros(n_points)
    for k in range(n_points):
        corr[k] = np.sum(h * (1 - h_shifted[k, :])) * dx
    return np.max(corr)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5])  # 45 samples

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    n_points = 100  # Start with 100 samples

    # Initial setup
    h_initial = initial_h_values()
    h_initial = np.resize(h_initial, n_points)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    h_current = h_initial.copy()

    # Evaluate initial solution
    c5_initial = evaluate_erdos_solution(h_current)
    best_c5 = c5_initial
    best_h = h_current.copy()

    # Check if we can reduce the number of points
    for n in range(50, 40, -1):
        if n < 40:
            break
        h_resize = np.resize(h_current, n)
        h_resize = np.clip(h_resize, 0, 1)
        h_resize = h_resize * (n / 2) / np.sum(h_resize)
        c5 = evaluate_erdos_solution(h_resize)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_resize.copy()
            n_points = n
            break

    # Main optimization loop
    for iteration in range(100):
        if time.time() - start_time > budget_s:
            break
        h_current = best_h.copy()
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)

        # Use gradient descent with momentum
        learning_rate = 0.1
        momentum = 0.9
        gradient = np.zeros(n_points)
        for k in range(n_points):
            shift = np.roll(h_current, k)
            overlap = np.sum(h_current * (1 - shift)) * (2.0 / n_points)
            gradient[k] = overlap
        gradient = gradient - np.mean(gradient) * np.ones(n_points)

        # Update h_current
        h_current -= learning_rate * gradient
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)

        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_current)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_current.copy()
            print(f"Iteration {iteration}: C5 = {best_c5:.6f}")

        # Add random perturbation to escape local minima
        noise = np.random.normal(0, 0.05, n_points)
        h_current = np.clip(h_current + noise, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)

        # Evaluate perturbed solution
        c5_perturbed = evaluate_erdos_solution(h_current)
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_current.copy()
            print(f"Perturbed: C5 = {best_c5:.6f}")

    # Check smaller n_points
    for n in range(n_points - 10, 20, -1):
        if n < 20:
            break
        h_resize = np.resize(best_h, n)
        h_resize = np.clip(h_resize, 0, 1)
        h_resize = h_resize * (n / 2) / np.sum(h_resize)
        c5 = evaluate_erdos_solution(h_resize)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_resize.copy()
            n_points = n
            print(f"Reduced to {n} points: C5 = {best_c5:.6f}")

    # Final check with small perturbations
    noise = np.random.normal(0, 0.05, n_points)
    h_final = np.clip(best_h + noise, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    c5_final = evaluate_erdos_solution(h_final)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final.copy()
        print(f"Final perturbation: C5 = {best_c5:.6f}")

    return best_h, best_c5, n_points
