import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import cvxpy as cp
import matplotlib.pyplot as plt

from scipy.optimize import Bounds
from scipy.optimize import minimize

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")  # Assuming initial_h_values.npy exists

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters for the optimization
    n_points = 100  # Start with a smaller number of points for speed
    max_iterations = 100
    tolerance = 1e-5
    best_c5 = float('inf')
    best_h = None
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = h.astype(float)
        corr = signal.correlate(h, 1 - h, mode="full")
        return np.max(corr) * (2.0 / len(h))
    
    # Function to check constraints
    def constraint(h):
        return np.sum(h) - (n_points / 2)
    
    # Optimization using scipy.optimize
    def optimize_h(h0):
        nonlocal best_c5, best_h, start_time
        bounds = Bounds(lb=0, ub=1)
        result = minimize(
            fun=compute_c5,
            x0=h0,
            method='L-BFGS-B',
            bounds=bounds,
            tol=tolerance,
            options={'maxiter': max_iterations}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            if time.time() - start_time > budget_s:
                return best_h, best_c5
        return best_h, best_c5
    
    # Initial guess from the provided initial_h_values
    initial_h = initial_h_values[:n_points]
    best_h, best_c5 = optimize_h(initial_h)
    
    # Try with a slightly larger number of points if time permits
    if time.time() - start_time < budget_s - 10:
        n_points = 150
        initial_h = np.random.rand(n_points)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        best_h, best_c5 = optimize_h(initial_h)
    
    # Try a few more random initial guesses if time permits
    if time.time() - start_time < budget_s - 20:
        for _ in range(3):
            initial_h = np.random.rand(n_points)
            initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
            current_h, current_c5 = optimize_h(initial_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    
    return (best_h, best_c5, n_points)
