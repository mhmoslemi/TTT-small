import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_h = np.sum(h_initial)
    if not np.isclose(sum_h, n_points / 2):
        h_initial = (h_initial / sum_h) * (n_points / 2)
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        h = np.minimum(np.maximum(h, 0), 1)
        h = np.concatenate([h, np.zeros_like(h)])
        conv = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(conv * dx)
        return max_overlap
    
    # Optimization: minimize C5 by adjusting h
    def objective(x):
        return compute_c5(x)
    
    # Optimization bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    max_iter = 100
    tol = 1e-6
    result = optimize.minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        tol=tol,
        options={'maxiter': max_iter, 'disp': False}
    )
    
    # Evaluate the best solution found
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Check if we can refine further with a smaller n_points
    # Try with a smaller number of points (e.g., n_points = 256)
    if time.time() - start_time < budget_s * 0.5:
        n_points_new = 256
        dx_new = 2.0 / n_points_new
        h_new = np.zeros(n_points_new)
        h_new[:n_points_new//2] = 1.0
        h_new[n_points_new//2:] = 0.0
        # Perturb slightly to avoid local minima
        perturbation = np.random.normal(0, 0.05, n_points_new)
        h_new = np.clip(h_new + perturbation, 0, 1)
        h_new = (h_new * n_points_new / 2.0) / np.sum(h_new) * n_points_new / 2.0
        best_c5_new = compute_c5(h_new)
        if best_c5_new < best_c5:
            best_h = h_new
            best_c5 = best_c5_new
    
    # Final result
    elapsed_time = time.time() - start_time
    return (best_h, best_c5, len(best_h))
