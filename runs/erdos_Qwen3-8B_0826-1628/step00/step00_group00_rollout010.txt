```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 45  # Starting with the initial h_values
    current_h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(current_h)
    best_h = current_h.copy()
    
    # Define the function to evaluate the overlap integral
    def evaluate_overlap(h):
        h = np.array(h)
        h = np.concatenate([h, h[:n_points-1]]),  # Extend to full correlation
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    # Define the optimization problem
    def optimize_h(h_values):
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,  # Ensure integral constraint
            0 <= h,  # h(x) ∈ [0, 1]
            h <= 1
        ]
        obj = cp.Minimize(evaluate_overlap(h))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver='SCIPY', options={'maxiter': 1000})
            if prob.success:
                return h.value
            else:
                return h_values
        except:
            return h_values
    
    # Optimization loop with early stopping
    start_time = time.time()
    remaining_time = budget_s
    while remaining_time > 0:
        # Perform a local optimization step
        current_h = optimize_h(current_h)
        c5 = evaluate_overlap(current_h)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = current_h.copy()
        
        # Adjust n_points for faster optimization
        if n_points < 200 and remaining_time > 5:
            n_points += 50
        else:
            n_points = 200
        
        # Check if we're within budget
        elapsed_time = time.time() - start_time
        remaining_time = budget_s - elapsed_time
        if remaining_time < 5:
            break
    
    # Return the best solution found
    return best_h, best_c5, n_points
```
