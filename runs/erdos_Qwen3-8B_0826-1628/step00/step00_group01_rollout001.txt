import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial h values
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        n_points = len(h)
        dx = 2.0 / n_points
        h = np.concatenate([h, np.zeros(n_points)])
        shifted = np.zeros(2 * n_points)
        for i in range(n_points):
            shifted[i] = 1.0 - h[i]
        conv = np.correlate(h, shifted, mode="full")
        return np.max(conv) * dx
    
    # Function to evaluate and return the best solution
    def evaluate_solution(h_values):
        c5 = compute_c5(h_values)
        return c5
    
    # Optimization function: hill climbing with random perturbations
    def optimize_h(h, num_iterations=1000, perturbation=0.05, budget_s=100):
        best_h = h.copy()
        best_c5 = evaluate_solution(best_h)
        elapsed = 0
        
        for _ in range(num_iterations):
            if elapsed >= budget_s:
                break
            # Random perturbation
            perturbed_h = h.copy()
            for i in range(len(perturbed_h)):
                if np.random.rand() < 0.1:
                    perturbed_h[i] = np.clip(perturbed_h[i] + np.random.uniform(-perturbation, perturbation), 0, 1)
            
            # Ensure the integral constraint is satisfied
            total = np.sum(perturbed_h)
            scale = (n_points / 2) / total if total != 0 else 1.0
            perturbed_h = np.clip(perturbed_h * scale, 0, 1)
            
            # Evaluate
            current_c5 = evaluate_solution(perturbed_h)
            if current_c5 < best_c5:
                best_h = perturbed_h.copy()
                best_c5 = current_c5
                elapsed = time.time() - start_time
            else:
                elapsed = time.time() - start_time
        
        return best_h, best_c5
    
    # Initial optimization
    h_optimized, c5_optimized = optimize_h(h, num_iterations=1000, perturbation=0.05, budget_s=budget_s)
    
    # Final evaluation
    final_c5 = evaluate_solution(h_optimized)
    return h_optimized, final_c5, len(h_optimized)
