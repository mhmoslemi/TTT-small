import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    n_points = 100  # Start with 100 points for faster computation
    dx = 2.0 / n_points
    
    # Initial guess from provided starting values, if available
    if hasattr(initial_h_values, "__len__") and len(initial_h_values) == n_points:
        h_values = initial_h_values[:]
    else:
        # Random initialization within [0, 1]
        h_values = np.random.rand(n_points)
        # Normalize to satisfy ∫h(x)dx = 1
        h_values = h_values * (n_points / 2.0) / np.sum(h_values)
    
    best_h = h_values.copy()
    best_c5 = np.inf
    
    # Function to evaluate the overlap integral
    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        h_shifted = np.zeros(2 * n_points)
        h_shifted[:n_points] = h
        shifted_h = np.zeros(2 * n_points)
        shifted_h[n_points:] = h
        correlation = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlation) * dx
        return c5
    
    # Function to compute and return the overlap integral
    def compute_c5(h):
        return evaluate_c5(h)
    
    # Optimization loop
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate the current solution
        current_c5 = compute_c5(h_values)
        
        # Update best solution if current is better
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_values.copy()
        
        # Perturb the current solution to explore the neighborhood
        perturbation = np.random.normal(0, 0.05, n_points)
        h_values = np.clip(h_values + perturbation, 0, 1)
        
        # Ensure ∫h(x) dx = 1
        h_values = h_values * (n_points / 2.0) / np.sum(h_values)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return best_h, final_c5, n_points
