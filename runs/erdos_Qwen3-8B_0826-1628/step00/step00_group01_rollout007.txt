def run(seed=42, budget_s=100, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)

    # Initial construction from the provided initial_h_values
    initial_h = initial_h_values
    n_points_initial = len(initial_h)
    n_points = n_points_initial  # Start with the initial number of points

    # Objective: minimize the maximum overlap integral C5
    # Strategy:
    # 1. Start with the initial construction and refine it.
    # 2. Use coordinate descent to optimize h, iteratively minimizing the max overlap.
    # 3. Use a small number of points to stay within the time budget.

    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    start_time = time.time()
    elapsed = 0

    # Try a few steps of refinement
    for iteration in range(5):
        # Convert h to a numpy array
        h = np.array(best_h, dtype=np.float64)
        n_points = len(h)

        # Compute overlap integrals for all shifts
        # Use convolution with mode='full' and scale by dx
        dx = 2.0 / n_points
        overlap = signal.correlate(h, 1 - h, mode='full') * dx

        # Compute max overlap
        current_c5 = np.max(overlap)

        # Check if we can improve
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # If we've already improved enough, break early
        if current_c5 < 0.3808:
            break

        # Try to refine h by coordinate descent
        for i in range(n_points):
            # Try different values of h[i] and choose the one that minimizes the maximum overlap
            # We consider values of h[i] in [0, 1] and choose the best one

            # Create a temporary array
            temp_h = h.copy()
            for val in [0.0, 0.5, 1.0, 0.25, 0.75, 0.1, 0.9, 0.3, 0.7, 0.4, 0.6]:
                temp_h[i] = val
                # Check if the integral constraint is satisfied
                total_integral = np.sum(temp_h) * dx
                if abs(total_integral - 1.0) > 1e-6:
                    continue

                # Compute overlap
                overlap_temp = signal.correlate(temp_h, 1 - temp_h, mode='full') * dx
                c5_temp = np.max(overlap_temp)

                # If this is better, update
                if c5_temp < current_c5:
                    h = temp_h.copy()
                    current_c5 = c5_temp

            # Update best_h if needed
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

        # Update the elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    # Return the best solution found
    return best_h, best_c5, n_points
