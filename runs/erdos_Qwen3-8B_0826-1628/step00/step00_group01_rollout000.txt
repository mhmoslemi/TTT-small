```python
import numpy as np
import scipy.optimize as opt
import time

# Pre-imported functions and initial solution
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points for faster evaluation
    max_iter = 100  # Max iterations of optimization
    max_time = budget_s  # Time budget in seconds
    
    # Initial h_values from provided starting point
    h_values = initial_h_values.copy()
    
    # Ensure the initial h_values satisfy the integral constraint
    h_values = h_values / np.sum(h_values) * (n_points / 2)
    
    # Function to compute the C5 value for a given h_values
    def compute_c5(h_values):
        h = np.array(h_values)
        h_corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(h_corr) * (2.0 / len(h))
        return c5
    
    # Function to optimize h_values with constraints
    def optimize_h(h_values, n_points, max_iter, max_time):
        h = np.array(h_values, dtype=np.float64)
        n = len(h)
        
        # Objective function: minimize the maximum overlap
        def objective(x):
            x = np.clip(x, 0, 1)
            h = x
            h_corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(h_corr) * (2.0 / n)
            return c5
        
        # Constraint: sum(h) = n_points / 2
        def constraint(x):
            return np.sum(x) - n_points / 2
        
        cons = [{'type': 'eq', 'fun': constraint}]
        bounds = [(0, 1) for _ in range(n)]
        
        # Optimization algorithm
        result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': max_iter})
        
        # Return the optimized h_values
        return result.x
    
    # Main optimization loop
    best_h = h_values.copy()
    best_c5 = compute_c5(h_values)
    
    for _ in range(5):  # Try multiple starting points
        # Adjust n_points for faster computation
        n_points = min(200, int(1000 * (1 - _ / 5)))
        h_values = optimize_h(h_values, n_points, max_iter, max_time)
        
        # Evaluate current C5
        current_c5 = compute_c5(h_values)
        if current_c5 < best_c5:
            best_h = h_values.copy()
            best_c5 = current_c5
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
```
