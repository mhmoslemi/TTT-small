import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp

# Pre-imported functions
# evaluate_erdos_solution is pre-imported
# initial_h_values is a pre-imported initial construction

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = None
    n_points = 100  # Start with 100 points for speed
    max_points = 200  # Maximum points to try within budget
    
    # Precompute dx
    dx = 2.0 / n_points
    
    # Function to compute C5 value given h
    def compute_c5(h):
        h = np.array(h)
        h_shifted = np.roll(h, shift=1, axis=0)
        h_shifted = h_shifted[:len(h)-1]  # Discard last element for overlap
        overlap = h * (1 - h_shifted)
        c5 = np.max(overlap) * dx
        return c5
    
    # Initial guess from initial_h_values
    initial_h = initial_h_values.copy()
    initial_h = initial_h[:n_points]  # Truncate to n_points if needed
    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)  # Normalize to satisfy the constraint
    initial_h = np.clip(initial_h, 0, 1)
    
    # Define optimization variables and constraints
    h = cp.Variable(n_points, lb=0, ub=1)
    constraint = cp.sum(h) == n_points / 2
    
    # Try increasing n_points up to max_points within budget
    for n_points in range(100, max_points + 1, 20):
        dx = 2.0 / n_points
        
        # Define the objective: minimize the maximum overlap integral
        # We use a convex relaxation by minimizing the maximum of the overlaps
        # This is a simplified approach; more advanced methods could use iterative optimization
        
        # Evaluate the current solution
        current_h = initial_h[:n_points]
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h / np.sum(current_h) * (n_points / 2)
        current_c5 = compute_c5(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        
        # Try optimization with CVXPY
        try:
            prob = cp.Problem(cp.Minimize(cp.Max([h[i] * (1 - h[(i+1)%n_points]) * dx for i in range(n_points)])),
                              [constraint])
            prob.solve(solver=cp.SCS, eps=1e-4, max_iter=100, time_limit=budget_s - (time.time() - start_time))
            
            if prob.status == 'OPTIMAL':
                h_values = h.value
                c5 = compute_c5(h_values)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_values
        except:
            pass
        
        # If we're out of time, break
        if time.time() - start_time > budget_s - 1:
            break
    
    # Return the best solution found
    return best_h, best_c5, n_points
