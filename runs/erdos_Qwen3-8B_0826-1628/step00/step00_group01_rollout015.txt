import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions and initial solution
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 47  # Start with the initial n_points

    # Function to evaluate the overlap integral
    def evaluate_overlap(h_values):
        h = np.array(h_values)
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Function to generate a random perturbation of the current solution
    def perturb_solution(h_values, perturb_strength=0.1):
        h = np.array(h_values)
        perturb = np.random.uniform(-perturb_strength, perturb_strength, len(h))
        new_h = np.clip(h + perturb, 0, 1)
        return new_h.tolist()

    # Function to optimize the solution using simulated annealing
    def optimize_solution(initial_h, n_points, budget_s):
        h = initial_h.copy()
        current_c5 = evaluate_overlap(h)
        best_h = h
        best_c5 = current_c5
        best_n_points = n_points

        # Try increasing the number of points if time allows
        for n in range(n_points + 1, min(n_points + 10, 100)):
            new_h = np.zeros(n)
            new_h[:len(h)] = h
            new_h[len(h):] = np.random.uniform(0, 1, n - len(h))
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n / 2) / np.sum(new_h)
            new_c5 = evaluate_overlap(new_h)
            if new_c5 < best_c5:
                best_h = new_h.tolist()
                best_c5 = new_c5
                best_n_points = n
            if time.time() - start_time > budget_s - 1:
                break

        # Perform simulated annealing on best_h
        h = np.array(best_h)
        current_c5 = best_c5
        temp = 100.0
        cooling_rate = 0.995
        iterations = 0
        max_iterations = 1000

        while temp > 1e-5 and iterations < max_iterations and time.time() - start_time < budget_s:
            new_h = perturb_solution(h, perturb_strength=0.05)
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (len(new_h) / 2) / np.sum(new_h)
            new_c5 = evaluate_overlap(new_h)

            # Acceptance probability
            if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / temp):
                h = new_h
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.tolist()
            temp *= cooling_rate
            iterations += 1

        return best_h, best_c5, best_n_points

    # Initial optimization
    initial_h = initial_h_values.copy()
    h = initial_h
    current_c5 = evaluate_overlap(h)
    best_h = h
    best_c5 = current_c5

    # Use simulated annealing to improve the solution
    h, best_c5, n_points = optimize_solution(h, len(h), budget_s)

    # If time allows, try increasing the number of points for better accuracy
    if time.time() - start_time < budget_s - 1:
        for n in range(len(h) + 1, min(len(h) + 10, 100)):
            new_h = np.zeros(n)
            new_h[:len(h)] = h
            new_h[len(h):] = np.random.uniform(0, 1, n - len(h))
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n / 2) / np.sum(new_h)
            new_c5 = evaluate_overlap(new_h)
            if new_c5 < best_c5:
                best_h = new_h.tolist()
                best_c5 = new_c5
            if time.time() - start_time > budget_s - 1:
                break

    # Final check
    if time.time() - start_time > budget_s - 1:
        pass  # We have already used the time budget

    # Ensure the final h is in the correct format
    best_h = np.array(best_h).tolist()
    return best_h, best_c5, len(best_h)
