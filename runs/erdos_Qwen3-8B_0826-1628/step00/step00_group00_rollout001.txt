```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Determine n_points: start with 45, try increasing if needed
    n_points = 45
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    initial_h = np.resize(initial_h, n_points)  # Ensure it's the correct length
    
    # Initial check
    h_values = initial_h.copy()
    c5_bound = evaluate_erdos_solution(h_values)
    best_h = h_values.copy()
    best_c5 = c5_bound
    best_n_points = n_points
    
    # Optimization settings
    max_iter = 30
    iter_count = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        
        # Define decision variable for optimization
        h = cp.Variable(n_points, bounds=(0, 1))
        
        # Objective: minimize the maximum overlap with all shifts
        # We use a convex relaxation by introducing an auxiliary variable
        tau = cp.Variable()
        constraints = []
        for k in range(n_points):
            # Compute the shift k, adjust to be within the range [0, n_points)
            shift = k % n_points
            h_shifted = cp.reshape(h[shift:], (n_points, 1))  # Shifted version
            h_ones = cp.reshape(np.ones(n_points), (1, n_points))
            overlap = cp.multiply(h, 1 - h_shifted)  # Element-wise product
            overlap = cp.sum(overlap)
            constraints.append(tau >= overlap)
        
        # Constraint on the integral
        constraints.append(cp.sum(h) == n_points / 2)
        
        # Objective is to minimize tau
        problem = cp.Problem(cp.Minimize(tau), constraints)
        
        # Solve the problem
        try:
            problem.solve(solver=cp.MOSEK, verbose=False)
            if problem.status != 'optimal':
                print("Warning: Optimization did not converge.")
                continue
            
            # Extract solution
            h_values = h.value
            c5_bound = tau.value
            print(f"Iteration {iter_count}: C₅ = {c5_bound}")
            
            # Update best solution
            if c5_bound < best_c5:
                best_h = h_values.copy()
                best_c5 = c5_bound
                best_n_points = n_points
            
        except Exception as e:
            print(f"Error during optimization: {e}")
            continue
        
        # Try increasing n_points if needed
        if iter_count == 1 and c5_bound > 0.38:
            n_points = 60
            dx = 2.0 / n_points
            # Resize initial_h to new n_points
            if len(initial_h) < n_points:
                initial_h = np.resize(initial_h, n_points)
            else:
                initial_h = initial_h[:n_points]
            h_values = initial_h.copy()
            print(f"Switching to n_points = {n_points}")
    
    # Return best solution found
    return best_h, best_c5, best_n_points
```
