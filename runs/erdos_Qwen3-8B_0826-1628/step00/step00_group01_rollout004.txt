import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    # Initial parameters
    n_points_initial = len(initial_h_values)
    dx_initial = 2.0 / n_points_initial
    h_initial = np.array(initial_h_values)
    h_initial = h_initial / np.sum(h_initial) * (n_points_initial / 2)  # Normalize to satisfy the integral constraint

    # Try with a smaller number of points for faster optimization
    n_points = min(n_points_initial, 100)  # Use up to 100 points for efficiency
    dx = 2.0 / n_points

    # Initial guess with some randomness to escape local optima
    np.random.seed(seed)
    h = h_initial[:n_points].copy()
    h += np.random.normal(0, 0.1, n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)  # Normalize to meet the integral constraint

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)

    start_time = time.time()
    elapsed_time = 0

    # Use L-BFGS-B optimization
    def objective(x):
        nonlocal elapsed_time
        if elapsed_time >= budget_s - 1:
            return best_c5
        elapsed_time = time.time() - start_time

        h = x.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = evaluate_erdos_solution(h, dx)
        return c5

    def constraint(x):
        nonlocal elapsed_time
        if elapsed_time >= budget_s - 1:
            return 0
        elapsed_time = time.time() - start_time
        return np.sum(x) - (n_points / 2)

    # Set up bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]

    # Optimization options
    options = {
        'maxiter': 100,
        'ftol': 1e-8,
        'disp': False
    }

    # Run optimization
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options=options
    )

    # Update best solution
    best_h = result.x.copy()
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)

    return best_h.tolist(), best_c5, n_points
