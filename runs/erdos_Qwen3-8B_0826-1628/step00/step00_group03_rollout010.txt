    def evaluate_h(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) != n_points / 2:
            return float('inf')
        corr = signal.correlate(h, 1 - h, mode='full') * dx
        c5 = np.max(corr)
        return c5
    
    # Function to optimize h
    def optimize_h(h):
        # Use a simple gradient descent with momentum
        h = h.copy()
        learning_rate = 0.1
        momentum = 0.8
        prev_grad = np.zeros_like(h)
        for _ in range(max_iter):
            # Compute gradient (approximate)
            eps = 1e-4
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                c5_plus = evaluate_h(h_plus)
                h_minus = h.copy()
                h_minus[i] -= eps
                c5_minus = evaluate_h(h_minus)
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            # Update
            h = h - learning_rate * grad - momentum * prev_grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            # Compute current c5
            current_c5 = evaluate_h(h)
            # Check for convergence
            if abs(current_c5 - best_c5) < tol:
                break
            prev_grad = grad
            best_c5 = current_c5
            best_h = h.copy()
            # Check time
            elapsed = time.time() - start_time
            if elapsed > budget_s - 1:
                break
        return best_h, best_c5
    
    # Try several random initializations
    for _ in range(5):
        # Start from a random initial guess
        h = np.random.rand(n_points) * 2 - 1
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h, c5 = optimize_h(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Check time
        elapsed = time.time() - start_time

    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = evaluate_h(best_h)
