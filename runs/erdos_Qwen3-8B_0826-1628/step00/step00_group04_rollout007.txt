    def evaluate_c5(h):
        h = np.array(h, dtype=np.float64)

        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result)
        return c5 * dx

    # Optimization loop
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    elapsed_time = 0.0

    # Simple gradient descent-like optimization
    for _ in range(100):
        if elapsed_time >= budget_s:
            break
        start_time = time.time()

        # Objective function: maximize the overlap integral
        # We use a convex relaxation for optimization
        h_var = cp.Variable(n_points, low=0.0, up=1.0)
        objective = cp.Maximize(cp.sum(cp.multiply(h_var, 1 - cp.roll(h_var, 1, axis=0))) * dx)
        constraints = [cp.sum(h_var) * dx == 1.0]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.MOSEK, eps=1e-6)

        if prob.status != 'OPTIMAL':
            break

        h_new = np.array(h_var.value)
        current_c5 = evaluate_c5(h_new)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()

        elapsed_time += time.time() - start_time
