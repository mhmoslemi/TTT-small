```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
import matplotlib.pyplot as plt

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution as eval_erdos
from evaluate_erdos_solution import initial_h_values as initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Define parameters
    n_points = 100  # Use 100 samples for faster evaluation
    dx = 2.0 / n_points
    target_sum = n_points / 2  # To satisfy ∫₀² h(x) dx = 1

    # Initial guess from provided initial_h_values
    h = np.array(initial_h_values[:n_points], dtype=np.float64)
    h = h[:n_points]  # Ensure correct length
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)  # Normalize to meet integral constraint

    best_h = h.copy()
    best_c5 = eval_erdos(best_h, dx)

    # Function to evaluate the overlap integral
    def evaluate_overlap(h_values, dx):
        h = h_values.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - target_sum) > 1e-6:
            return np.inf  # Penalize for violating integral constraint

        h = h.reshape(-1, 1)
        shifted_h = np.roll(h, shift=1, axis=0)
        overlap = np.sum(h * (1 - shifted_h)) * dx
        return overlap

    # Define objective function for optimization
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - target_sum) > 1e-6:
            return np.inf  # Penalize for violating integral constraint
        h = h.reshape(-1, 1)
        shifted_h = np.roll(h, shift=1, axis=0)
        overlap = np.sum(h * (1 - shifted_h)) * dx
        return overlap

    # Define optimization problem using CVXPY
    def optimize_h(h_initial):
        x = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(x * (1 - cp.roll(x, 1)))) * dx
        constraints = [0 <= x, x <= 1, cp.sum(x) == target_sum]
        prob = cp.Problem(objective, constraints)
        result = prob.solve(solver=cp.CVXPY_MI, eps=1e-6)
        if result is not None:
            h_opt = np.clip(x.value, 0, 1)
            return h_opt
        return h_initial

    # Local search with adaptive steps
    def local_search(h_initial, steps=100):
        h = h_initial.copy()
        for _ in range(steps):
            # Perturb the function slightly
            perturbation = np.random.normal(0, 0.05, size=n_points)
            h_perturbed = h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
            current_c5 = evaluate_overlap(h_perturbed, dx)
            if current_c5 < evaluate_overlap(h, dx):
                h = h_perturbed
        return h

    # Adaptive optimization
    for _ in range(10):
        h = optimize_h(h)
        h = local_search(h)
        current_c5 = evaluate_overlap(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        if time.time() - start_time > budget_s - 1:
            break

    # Final evaluation
    final_c5 = evaluate_overlap(best_h, dx)
    return (best_h, final_c5, n_points)
```
