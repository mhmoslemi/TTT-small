```python
import numpy as np
import scipy.optimize as opt
import time
import multiprocessing

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 42  # Starting with 42 samples
    h = initial_h_values.copy()
    
    # Define the overlap integral function
    def compute_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        h_extended = np.concatenate([h, np.zeros_like(h)])
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Define the objective function
    def objective(h_vec):
        return compute_c5(h_vec)
    
    # Define the constraint: sum(h) == n_points / 2
    def constraint(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint}]
    
    # Try different optimization methods
    for method in ['SLSQP', 'L-BFGS-B', 'Nelder-Mead', 'TNC']:
        try:
            result = opt.minimize(
                objective,
                h,
                method=method,
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-8, 'maxiter': 100}
            )
            if result.success:
                h_best = result.x
                c5_best = compute_c5(h_best)
                print(f"Found solution with C5 = {c5_best} using {method}")
                break
        except:
            continue
    
    # Check if we need to increase the number of points
    if time.time() - start_time < budget_s - 10:
        n_points = 60
        h = np.zeros(n_points)
        # Distribute the sum evenly
        h[:int(n_points / 2)] = 1.0
        h[int(n_points / 2):] = 0.0
        # Refine with optimization
        dx = 2.0 / n_points
        def objective_refined(h_vec):
            h = np.array(h_vec)
            dx = 2.0 / len(h)
            h_extended = np.concatenate([h, np.zeros_like(h)])
            corr = np.correlate(h, 1 - h, mode='full')
            max_overlap = np.max(corr) * dx
            return max_overlap
        
        def constraint_refined(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint_refined}]
        
        try:
            result = opt.minimize(
                objective_refined,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-8, 'maxiter': 100}
            )
            if result.success:
                h_best = result.x
                c5_best = compute_c5(h_best)
                print(f"Found refined solution with C5 = {c5_best}")
        except:
            pass
    
    # Final check with initial solution
    if 'h_best' not in locals():
        h_best = initial_h_values
        c5_best = compute_c5(h_best)
    
    # Return the best solution
    elapsed = time.time() - start_time
    print(f"Total time used: {elapsed:.2f} seconds")
    print(f"Best C5 found: {c5_best}")
    
    return h_best, c5_best, n_points
```
