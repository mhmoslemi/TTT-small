    # Function to evaluate C₅ for a given h
    def evaluate_c5(h):
        h = np.array(h)
        h = np.concatenate([h, h])  # To handle wrap-around in convolution
        conv_result = np.correlate(h, 1 - h, mode="full")
        # We are interested in the part corresponding to k in [0, 2]
        # Since convolution is symmetric, we take the middle part
        c5 = np.max(conv_result[n_points - 1 : 2 * n_points - 1]) * dx
        return c5

    # Optimization problem: minimize c5
    def objective_function(h_values):
        # Convert to numpy array
        h = np.array(h_values)
        # Check if the sum is approximately 1
        if np.abs(np.sum(h) * dx - 1) > 1e-6:
            return float('inf')
        # Compute C₅
        return evaluate_c5(h)

    # Set dx
    dx = 2.0 / n_points

    # Initial guess
    initial_guess = h_initial.copy()

    # Use scipy.optimize to minimize the C₅ value
    # Use the L-BFGS-B method with box constraints
    result = optimize.minimize(
        objective_function,
        initial_guess,
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        method='L-BFGS-B',
        options={'disp': False, 'maxiter': 100}
    )

    # Extract the best solution
    best_h_values = result.x
