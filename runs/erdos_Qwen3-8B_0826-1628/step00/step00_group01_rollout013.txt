import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import sys
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    # Set random seed
    np.random.seed(seed)
    start_time = time.time()
    
    # Define n_points and dx
    n_points = 47  # Start with the initial n_points
    dx = 2.0 / n_points
    
    # Initial guess: use initial_h_values
    h = np.array(initial_h_values, dtype=np.float64)
    h = h[:n_points]  # Ensure it's the correct length
    
    # Ensure the integral constraint is satisfied
    sum_h = np.sum(h) * dx
    if not np.isclose(sum_h, 1.0):
        h = h / sum_h * 1.0
    
    def objective(h_vec):
        h = h_vec.reshape(-1)
        # Ensure values are within [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        one_minus_h = 1.0 - h
        corr = signal.correlate(h, one_minus_h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization: Use a local optimization method to refine the initial guess
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 100, 'ftol': 1e-7, 'gtol': 1e-7}
    )
    
    # Get the best solution found
    best_h = result.x
    
    # Ensure integral constraint
    sum_h = np.sum(best_h) * dx
    if not np.isclose(sum_h, 1.0):
        best_h = best_h / sum_h * 1.0
    
    # Evaluate C5
    c5_bound = evaluate_erdos_solution(best_h)
    
    # If we have time, try to increase n_points
    if time.time() - start_time < budget_s - 5:
        n_points = 53
        dx = 2.0 / n_points
        # Initial guess based on previous solution
        h = np.linspace(0, 2, n_points)
        h = best_h * (n_points / 47)
        h = np.clip(h, 0, 1)
        # Ensure integral constraint
        sum_h = np.sum(h) * dx
        if not np.isclose(sum_h, 1.0):
            h = h / sum_h * 1.0
        
        def objective_new(h_vec):
            h = h_vec.reshape(-1)
            # Ensure values are within [0, 1]
            h = np.clip(h, 0, 1)
            # Compute the overlap integral
            one_minus_h = 1.0 - h
            corr = signal.correlate(h, one_minus_h, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        result_new = optimize.minimize(
            objective_new,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-7, 'gtol': 1e-7}
        )
        
        best_h = result_new.x
        # Ensure integral constraint
        sum_h = np.sum(best_h) * dx
        if not np.isclose(sum_h, 1.0):
            best_h = best_h / sum_h * 1.0
        
        c5_bound = evaluate_erdos_solution(best_h)
    
    # Return the result
    return (best_h.tolist(), c5_bound, n_points)
