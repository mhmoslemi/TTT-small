```python
import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess
    n_points = 100  # Starting with 100 points (adjustable if needed)
    h = np.array(initial_h_values[:n_points], dtype=float)
    h = h / np.sum(h) * (n_points / 2)  # Ensure the integral is 1
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        h = np.pad(h, (0, 1), 'constant')  # Pad for full convolution
        result = np.correlate(h, 1 - h, mode='full') * (2.0 / n_points)
        return np.max(result)
    
    # Function to minimize C5
    def objective(x):
        h = x.copy()
        # Ensure h has correct length
        if len(h) != n_points:
            h = np.resize(h, n_points)
        # Ensure integral constraint
        integral = np.sum(h) * (2.0 / n_points)
        # Add small epsilon to avoid 0 for division
        eps = 1e-8
        penalty = np.abs(integral - 1) * 1e6
        # Compute C5
        c5 = compute_c5(h)
        return c5 + penalty
    
    # Optimizer settings
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = opt.minimize(
        objective,
        h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    # Track best result
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Try with a higher resolution if time allows
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        best_h = np.resize(best_h, n_points)
        best_h = best_h / np.sum(best_h) * (n_points / 2)
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.minimize(
            objective,
            best_h,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        best_h = result.x
        best_c5 = compute_c5(best_h)
    
    # Final check to ensure integral constraint
    final_integral = np.sum(best_h) * (2.0 / n_points)
    if np.abs(final_integral - 1) > 1e-6:
        best_h = best_h / final_integral * 1
        best_c5 = compute_c5(best_h)
    
    # Return result
    return (best_h, best_c5, n_points)
```
