import numpy as np
import scipy.optimize as optimize
import time
from scipy.interpolate import interp1d
from scipy.signal import correlate
from scipy.sparse import diags
from scipy.optimize import Bounds
from functools import partial

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100  # Start with n=100 for speed
    dx = 2.0 / n_points
    
    # Initialize h with a uniform distribution
    h = np.ones(n_points) * 0.5
    sum_h = np.sum(h)
    if sum_h * dx != 1:
        h = np.ones(n_points) * (n_points / 2) / n_points
    
    # Function to calculate C5 for a given h
    def compute_c5(h):
        h_interp = interp1d(np.linspace(0, 2, n_points), h, kind='linear', fill_value='extrapolate')
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Function to compute the gradient of C5 with respect to h
    def compute_gradient(h):
        h_interp = interp1d(np.linspace(0, 2, n_points), h, kind='linear', fill_value='extrapolate')
        corr = correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        max_corr = corr[max_index]
        
        # Compute the derivative of the max_corr with respect to h
        # This is an approximation using finite differences
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-6
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - max_corr) / 1e-6
        
        return grad
    
    # Function to compute the gradient of the constraint
    def constraint_gradient(h):
        return np.ones_like(h) * dx  # Derivative of sum(h) * dx = 1 is dx
    
    # Optimization objective
    def objective(h):
        return compute_c5(h)
    
    # Constraints
    def constraint(h):
        return np.sum(h) * dx - 1
    
    # Define bounds for h
    bounds = Bounds(0, 1)
    
    # Use L-BFGS-B optimizer with gradient
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        jac=compute_gradient,
        bounds=bounds,
        options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # If we have remaining time, try with more points
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        n_points = 200
        dx = 2.0 / n_points
        h = np.ones(n_points) * (n_points / 2) / n_points
        # Reoptimize with more points
        result = optimize.minimize(
            objective,
            h,
            method='L-BFGS-B',
            jac=compute_gradient,
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        best_h = result.x
        best_c5 = compute_c5(best_h)
    
    # Return the best solution
    return (best_h, best_c5, n_points)
