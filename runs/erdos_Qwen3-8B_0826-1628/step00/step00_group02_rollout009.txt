```python
import numpy as np
import scipy.optimize as opt
import time
import functools

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial solution
    h_values = initial_h_values.copy()
    n_points = len(h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(h_values)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        h_values = (h_values / sum_h) * (n_points / 2)
    
    def evaluate_c5(h):
        h = np.array(h, dtype=np.float64)
        h = np.concatenate([h, np.zeros_like(h)])
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.min(h) >= 0 and np.max(h) <= 1
    
    # Initial optimization using L-BFGS-B
    result = opt.minimize(
        objective,
        h_values,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        jac=None,
        tol=1e-6,
        options={'maxiter': 100, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = evaluate_c5(best_h)
    
    # Refine with local search
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        # Perturb the solution
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        # Ensure the sum constraint is satisfied
        sum_perturbed = np.sum(perturbed_h)
        if np.abs(sum_perturbed - n_points / 2) > 1e-6:
            scaling = (n_points / 2) / sum_perturbed
            perturbed_h *= scaling
        # Evaluate
        current_c5 = evaluate_c5(perturbed_h)
        if current_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = current_c5
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, n_points
```
