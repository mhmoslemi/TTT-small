import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Define parameters
    max_points = 1000  # Maximum number of samples allowed
    n_points = 100  # Start with a smaller number of samples
    dx = 2.0 / n_points
    max_iter = 100  # Maximum number of optimization iterations
    tolerance = 1e-5  # Optimization tolerance
    best_c5 = float('inf')
    best_h = None
    best_n_points = n_points
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        h_expanded = np.concatenate((h, np.zeros_like(h)))
        conv = correlate(h_expanded, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    # Initial guess
    h = initial_h_values.copy()
    h = np.array(h[:n_points])
    h = h / (np.sum(h) * dx) * (n_points / 2)
    
    # Optimize
    for _ in range(max_iter):
        start_time = time.time()
        # Define optimization variables
        x = cp.Variable(n_points, lower_bound=0, upper_bound=1)
        # Constraint: total integral is 1
        constraint = cp.sum(x) * dx == 1
        # Objective: minimize the maximum overlap integral
        # We use a convex optimization approach
        # We approximate the max by using an epsilon-constraint approach
        epsilon = 1e-3
        obj = cp.Minimize(epsilon * cp.sum(cp.square(x - x * x)) + cp.sum(cp.square(x - cp.min(x))))
        
        # Solve the problem
        prob = cp.Problem(obj, [constraint])
        try:
            prob.solve(solver=cp.ECOS, eps=tolerance)
            if prob.status == 'optimal':
                h_new = x.value
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
                    best_n_points = n_points
                # Optionally try increasing n_points if time allows
                # Uncomment the following lines to try increasing n_points
                # if time.time() - start_time < budget_s * 0.5:
                #     n_points = min(n_points * 2, max_points)
                #     dx = 2.0 / n_points
                #     h = h_new
                #     continue
            else:
                print(f"Optimization did not converge, status: {prob.status}")
        except:
            print("Solver failed, trying with a different approach")
            # Fallback to a grid search or other method if needed
            # Random perturbation
            h = np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1)
    
    # Final evaluation
    h = np.clip(best_h, 0, 1)
    h = h / (np.sum(h) * dx) * (best_n_points / 2)
    c5 = compute_c5(h)
    
    # Return the best solution found
    return (h.tolist(), c5, best_n_points)
