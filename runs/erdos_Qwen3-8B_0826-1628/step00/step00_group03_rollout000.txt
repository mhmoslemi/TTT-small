import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

# Strategy:
# 1. Define a function to compute the C5 value for a given h
# 2. Use a local optimization approach (e.g., gradient descent or Nelder-Mead) to minimize C5
# 3. Use the initial_h_values as a starting point
# 4. Perform multiple restarts with different initializations to avoid local minima
# 5. Ensure that the solution remains within the constraints (sum h = n_points / 2, 0 ≤ h ≤ 1)

# Define the function to compute the C5 value
def compute_c5(h, n_points):
    dx = 2.0 / n_points
    h = np.array(h)
    # Compute the correlation
    corr = signal.correlate(h, 1 - h, mode="full")
    # Normalize by dx
    c5_values = corr * dx
    c5 = np.max(c5_values)
    return c5

# Define the optimization function
def optimize_erdos(n_points, initial_h, seed=42, max_iter=100, tolerance=1e-5):
    np.random.seed(seed)
    h = np.array(initial_h)
    # Normalize h to satisfy the sum constraint
    h_norm = h / np.sum(h) * (n_points / 2)
    # Optimize using Nelder-Mead
    result = opt.minimize(
        lambda x: compute_c5(x, n_points),
        x0=h_norm,
        method='Nelder-Mead',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': max_iter, 'tol': tolerance}
    )
    best_h = result.x
    best_c5 = result.fun
    return best_h, best_c5

# Main run function
def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    # Choose a number of points (less than 1k for efficiency)
    n_points = 500
    # Get initial samples
    initial_h = initial_h_values
    # Check if the initial_h has the correct length
    if len(initial_h) != n_points:
        # If not, pad or truncate to match n_points
        initial_h = initial_h[:n_points] + [0.5] * (n_points - len(initial_h))
        initial_h = initial_h[:n_points]
    # Normalize initial_h to satisfy the sum constraint
    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    # Optimize
    best_h, best_c5 = optimize_erdos(n_points, initial_h, seed=seed)
    # Ensure the solution satisfies the constraints
    # Check the sum constraint
    sum_h = np.sum(best_h)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        best_h = best_h / sum_h * (n_points / 2)
    # Clip values to [0, 1]
    best_h = np.clip(best_h, 0, 1)
    # Compute the final C5 value
    final_c5 = compute_c5(best_h, n_points)
    elapsed_time = time.time() - start_time
    # Output the result
    return best_h, final_c5, n_points
