```python
import numpy as np
import scipy.optimize
import scipy.signal
import time
import copy
import cvxpy as cp
import matplotlib.pyplot as plt

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * 2.0 / len(h)
initial_h_values = np.load("initial_h_values.npy")  # Placeholder for initial construction

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    # Start with the initial construction
    n_points = len(initial_h_values)
    h = copy.deepcopy(initial_h_values)
    c5 = evaluate_erdos_solution(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h
        best_n = n_points
    
    # Try smaller n_points for faster optimization
    for n_points in range(50, 1000, 10):  # Try from 50 to 1000
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate initial guess with uniform distribution
        h = np.random.rand(n_points)
        h = h * 2.0 / np.sum(h)  # Ensure sum equals n_points / 2
        
        # Define optimization variables
        x = cp.Variable(n_points)
        constraints = [
            cp.sum(x) == n_points / 2,
            0 <= x,
            x <= 1
        ]
        
        # Evaluate the overlap integral for all shifts
        def objective(x):
            h_vals = x.value
            corr = scipy.signal.correlate(h_vals, 1 - h_vals, mode='full')
            c5_value = np.max(corr) * 2.0 / n_points
            return c5_value
        
        # Use a gradient-based optimizer
        try:
            prob = cp.Problem(cp.Minimize(objective(x)), constraints)
            result = prob.solve(solver=cp.SCS, verbose=False)
            if result is not None and result < best_c5:
                best_c5 = result
                best_h = x.value
                best_n = n_points
        except:
            pass
        
        # Try a greedy optimization
        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
            # Randomly select a point to adjust
            idx = np.random.randint(n_points)
            h[idx] = np.random.rand() * 2.0 / np.sum(h)  # Ensure sum is maintained
            h = np.clip(h, 0, 1)
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
                best_n = n_points
    
    result = (best_h, best_c5, best_n)
    print(f"Found best C5 bound: {result[1]}")
    return result
```
