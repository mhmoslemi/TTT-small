```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * 2.0 / len(h)
initial_h_values = np.load("initial_h_values.npy")  # Example initial solution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial setup
    n_points_initial = len(initial_h_values)
    dx = 2.0 / n_points_initial
    h_initial = initial_h_values.copy()
    
    # Evaluate initial solution
    c5_initial = evaluate_erdos_solution(h_initial)
    best_h = h_initial
    best_c5 = c5_initial
    
    # Try reducing points to below 1k for faster computation
    for n_points in [500, 400, 300, 200]:
        if n_points < 10:
            continue
        # Generate uniform grid
        x = np.linspace(0, 2, n_points)
        # Initial guess: evenly distribute the 1s over the interval
        h = np.zeros(n_points)
        h[np.round(x * n_points / 2).astype(int)] = 1
        
        # Define variables
        h_var = cp.Variable(n_points, nonneg=True, bounded=(0, 1))
        
        # Objective: minimize the maximum overlap integral over all shifts
        # We use a convex relaxation by minimizing the maximum value using epigraph form
        t = cp.Variable()
        constraints = [cp.sum(h_var) == n_points / 2]
        
        for k in range(-n_points + 1, n_points):
            # Compute the overlap integral for shift k
            # Wrap around the array using modulo
            h_shifted = np.roll(h_var.value, k) if k >= 0 else np.roll(h_var.value, k + n_points)
            overlap = cp.sum(h_var * (1 - h_shifted))
            constraints.append(t >= overlap)
        
        # Problem: minimize t
        prob = cp.Problem(cp.Minimize(t), constraints)
        
        # Solve
        try:
            result = prob.solve(solver="SCIPY", max_iter=1000, eps=1e-6, time_limit=budget_s - (time.time() - start_time))
            if result is not None:
                h_opt = h_var.value
                c5_opt = t.value
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt
        except:
            pass
    
    # Return best solution
    return best_h, best_c5, len(best_h)
```
