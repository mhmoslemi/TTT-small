```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for speed and efficiency
    h_values = initial_h_values(n_points)
    dx = 2.0 / n_points
    sum_h = np.sum(h_values)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        h_values = (h_values * (n_points / 2) / sum_h) if sum_h != 0 else np.ones(n_points) * (n_points / 2) / n_points

    def objective_function(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            h = h * (n_points / 2) / sum_h
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        corr = signal.correlate(h, one_minus_h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def evaluate_solution(h):
        return objective_function(h)

    def optimize_h(h_values, n_points, budget_s=100):
        start_time = time.time()
        best_c5 = float('inf')
        best_h = h_values.copy()

        # Use a simple gradient descent approach for initial optimization
        learning_rate = 0.01
        max_iterations = 1000
        for _ in range(max_iterations):
            current_c5 = evaluate_solution(h_values)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_values.copy()
            # Simple gradient descent step
            h_values += learning_rate * np.random.randn(n_points)
            h_values = np.clip(h_values, 0, 1)
            sum_h = np.sum(h_values)
            if np.abs(sum_h - n_points / 2) > 1e-6:
                h_values = h_values * (n_points / 2) / sum_h
            h_values = np.clip(h_values, 0, 1)
            if time.time() - start_time > budget_s - 1:
                break

        # Use a more advanced optimization if time allows
        if time.time() - start_time < budget_s - 5:
            variables = cp.Variable(n_points)
            constraints = [
                cp.sum(variables) == n_points / 2,
                0 <= variables,
                variables <= 1
            ]
            obj = cp.Minimize(cp.Max(signal.correlate(variables.value, 1 - variables.value, mode="full") * dx))
            problem = cp.Problem(obj, constraints)
            try:
                problem.solve(solver=cp.SCS, max_iter=100)
                h_values = variables.value
            except:
                pass

        return h_values, best_c5

    h_values, best_c5 = optimize_h(h_values, n_points, budget_s)
    return h_values, best_c5, n_points
```
