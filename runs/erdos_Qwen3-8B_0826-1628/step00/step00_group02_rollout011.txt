import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Initial construction parameters
    n_points_initial = len(initial_h_values)
    dx_initial = 2.0 / n_points_initial
    
    # Start with the initial construction
    h_values = np.array(initial_h_values, dtype=np.float64)
    
    # Define a function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        n = len(h)
        dx = 2.0 / n
        # Compute the convolution
        corr = np.correlate(h, 1 - h, mode="full")
        # Only consider the relevant part of the convolution
        max_corr = np.max(corr[n-1:2*n-1])
        return max_corr * dx
    
    # Define a function to perform gradient-based optimization
    def optimize_h(h_initial, max_iter=100, tol=1e-4):
        h = cp.Variable(len(h_initial))
        constraints = [0 <= h, h <= 1, cp.sum(h) == len(h_initial) / 2]
        obj = cp.Minimize(cp.max(cp.multiply(h, 1 - h)))
        problem = cp.Problem(obj, constraints)
        result = problem.solve(solver=cp.SCS, eps=tol, max_iter=max_iter)
        if result is None:
            return h_initial
        return h.value
    
    # Optimization loop
    best_c5 = compute_c5(h_values)
    best_h = h_values.copy()
    
    start_time = time.time()
    
    # Perform gradient descent-based optimization
    h_optimized = optimize_h(h_values, max_iter=50, tol=1e-3)
    current_c5 = compute_c5(h_optimized)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_optimized
    
    # Perform local search with perturbations
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the solution slightly
        perturbation = np.random.uniform(-0.1, 0.1, size=len(best_h))
        h_perturbed = np.clip(best_h + perturbation, 0, 1)
        # Enforce the constraint sum(h) = n / 2
        h_perturbed = h_perturbed * (len(h_perturbed) / 2) / np.sum(h_perturbed)
        current_c5 = compute_c5(h_perturbed)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_perturbed
    
    # Final check with gradient descent
    h_optimized = optimize_h(best_h, max_iter=50, tol=1e-3)
    current_c5 = compute_c5(h_optimized)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_optimized
    
    return (best_h, best_c5, len(best_h))
