```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from functools import partial

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.random.rand(45) * 2 - 1  # Placeholder for initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 45  # Start with 45 samples for efficiency

    # Try different numbers of points, up to a maximum of 1000
    for n_points in [45, 60, 75, 90, 105, 120, 135, 150, 165, 180, 200, 225, 250, 275, 300, 325, 350, 375, 400, 425, 450, 475, 500, 525, 550, 575, 600, 625, 650, 675, 700, 725, 750, 775, 800, 825, 850, 875, 900, 925, 950, 975, 1000]:
        if time.time() - start_time > budget_s:
            break
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)

        # Initial guess: use random values from initial_h_values scaled to match the constraint
        h_initial = initial_h_values.copy()
        h_initial = h_initial[:n_points] * (n_points / 2) / np.sum(h_initial[:n_points])
        h_initial = np.clip(h_initial, 0, 1)

        # Define the optimization problem
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]

        # Define the cost function to minimize the maximum overlap
        def objective(h):
            h_vals = h.value
            corr = np.correlate(h_vals, 1 - h_vals, mode="full")
            return np.max(corr) * dx

        # Use a local optimization method to refine the solution
        result = optimize.minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)]
        )

        h_opt = result.x
        c5 = evaluate_erdos_solution(h_opt)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
            print(f"New best C5: {best_c5} with {n_points} points")

    return (best_h, best_c5, len(best_h))
```
