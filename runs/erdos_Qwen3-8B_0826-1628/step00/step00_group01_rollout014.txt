```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial n_points
    n_points = 100  # Start with 100 points for efficiency
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = h[:n_points]  # Ensure h has the correct length
        
    # Normalize to meet the integral constraint
    h = h / np.sum(h) * (n_points / 2)
    
    def compute_c5(h_values):
        h = h_values.copy()
        h = h / np.sum(h) * (n_points / 2)
        corr = signal.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Function to update h and compute C5
    def update_h(h_values):
        h = h_values.copy()
        h = h / np.sum(h) * (n_points / 2)
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization
    def objective(x):
        return update_h(x)
    
    # Initial guess
    x0 = h
    
    # Bounds: h[i] must be in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    max_iter = 50  # Limit iterations to stay within budget
    tol = 1e-6
    
    # Run optimization
    result = opt.minimize(
        objective,
        x0,
        bounds=bounds,
        method='L-BFGS-B',
        tol=tol,
        options={'maxiter': max_iter}
    )
    
    # Best solution found so far
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Check if we can try more points for better accuracy
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        dx = 2.0 / n_points
        h = best_h.copy()
        h = h / np.sum(h) * (n_points / 2)
        
        def compute_c5_new(h_values):
            h = h_values.copy()
            h = h / np.sum(h) * (n_points / 2)
            corr = signal.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        def update_h_new(h_values):
            h = h_values.copy()
            h = h / np.sum(h) * (n_points / 2)
            corr = signal.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        def objective_new(x):
            return update_h_new(x)
        
        x0_new = h
        bounds_new = [(0.0, 1.0) for _ in range(n_points)]
        
        result_new = opt.minimize(
            objective_new,
            x0_new,
            bounds=bounds_new,
            method='L-BFGS-B',
            tol=tol,
            options={'maxiter': max_iter}
        )
        
        best_h = result_new.x
        best_c5 = compute_c5_new(best_h)
    
    # Final result
    final_h = best_h
    final_c5 = best_c5
    final_n_points = len(final_h)
    
    # Ensure final_h has exactly n_points samples
    final_h = final_h[:final_n_points]
    
    return final_h, final_c5, final_n_points
```
