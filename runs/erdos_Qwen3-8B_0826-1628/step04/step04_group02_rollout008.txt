import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    total = sum(h)
    h_norm = h * (n_points / 2) / total
    
    def compute_overlap(h_values):
        if len(h_values) != n_points:
            return float('inf')
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return max(corr) * dx
    
    # Function to enforce the integral constraint
    def integral_constraint(h_values):
        return sum(h_values) - (n_points / 2)
    
    # Optimization: Use an ensemble of local optima
    max_iter = 50
    best_h = h_norm.copy()
    best_c5 = compute_overlap(best_h)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Generate a random perturbation
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = np.clip(h_norm + perturbation, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / sum(h_perturbed)
        
        # Local refinement using SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': integral_constraint})
        result = optimize.minimize(lambda x: compute_overlap(x), h_perturbed, bounds=bounds, constraints=cons, method='SLSQP')
        
        # Update the best solution
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return (best_h.tolist(), best_c5, n_points)
