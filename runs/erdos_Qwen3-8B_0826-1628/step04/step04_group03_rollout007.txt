import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    ones_minus_h = 1 - h
    corr = np.correlate(h, ones_minus_h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    n_points = 60
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < 15:
            h[i] = 1
        elif i < 30:
            h[i] = 0.5
        else:
            h[i] = 1
    return h.tolist()

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    best_c5 = 1.0
    best_h = None
    best_n = 0
    
    # Try different n_points to find optimal resolution
    n_points_list = [60, 80, 100]  # Start with the given 60, then refine with more points
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        # Initial construction inspired by the initial_h_values function
        h_initial = np.zeros(n_points)
        for i in range(n_points):
            if i < int(n_points / 3):
                h_initial[i] = 1.0
            elif i < 2 * int(n_points / 3):
                h_initial[i] = 0.5
            else:
                h_initial[i] = 1.0
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Define objective function
        def objective(h):
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            c5 = evaluate_erdos_solution(h)
            return c5
        
        # Define bounds and constraints
        bounds = Bounds(0, 1)
        constraints = [LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)]
        
        # Optimization with hybrid methods
        start_time = time.time()
        # Use L-BFGS-B for fast convergence
        result = opt.minimize(
            fun=objective,
            x0=h_initial,
            bounds=bounds,
            constraints=constraints,
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                best_n = n_points
        
        # Additional refinement with SLSQP
        if time.time() - start_time < budget_s:
            result = opt.minimize(
                fun=objective,
                x0=result.x,
                bounds=bounds,
                constraints=constraints,
                method='SLSQP',
                options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
                    best_n = n_points
        
        # Final refinement with differential evolution for global exploration
        if time.time() - start_time < budget_s:
            result = opt.differential_evolution(
                fun=objective,
                bounds=[(0, 1) for _ in range(n_points)],
                constraints=constraints,
                popsize=15,
                maxiter=50,
                mutation=(0.5, 1),
                recombination='blend',
                strategy='best1bin',
                tol=1e-6,
                seed=seed
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
                    best_n = n_points
    
    # Ensure constraints are satisfied before returning
    best_h = best_h * (best_n / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    
    return best_h.tolist(), best_c5, best_n
