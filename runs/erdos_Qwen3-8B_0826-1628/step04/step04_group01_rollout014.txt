```python
import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, LinearConstraint
import random

def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    ones_minus_h = 1 - h
    corr = np.correlate(h, ones_minus_h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    n_points = 60
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < 15:
            h[i] = 1
        elif i < 30:
            h[i] = 0
        else:
            h[i] = 1
    return h.tolist()

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points_list = [60, 70, 80, 90, 100]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 100

    # Start with the initial solution
    initial_h = initial_h_values()
    initial_h = np.array(initial_h)
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        h = np.resize(initial_h, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        def objective(h_vec):
            h = np.clip(h_vec, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            c5 = evaluate_erdos_solution(h)
            return c5
        
        bounds = Bounds(0, 1)
        constraints = [LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)]
        
        start_time = time.time()
        result = opt.minimize(
            fun=objective,
            x0=h,
            bounds=bounds,
            constraints=constraints,
            method='SLSQP',
            options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 200}
        )
        
        if result.success:
            current_h = result.x
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
                best_n_points = n_points
        
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    # Refine the best solution
    if best_h is not None:
        best_h = best_h * (best_n_points / 2) / np.sum(best_h)
        best_h = np.clip(best_h, 0, 1)
    
    return best_h.tolist(), best_c5, best_n_points
```
