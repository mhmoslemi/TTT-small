import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 120
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h[i] = 1.0
            else:
                h[i] = 0.0
        h = np.random.permutation(h)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        cross_corr = np.correlate(x, 1 - x, mode='full')
        return np.max(cross_corr) * dx
    
    def norm(x):
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        return x
    
    class PSO:
        def __init__(self, n_particles=50, max_iter=100):
            self.n_particles = n_particles
            self.max_iter = max_iter
            self.pbest = []
            self.gbest = []
            self.velocities = []
            self.pbest_error = []
            self.gbest_error = np.inf
            self.best_position = np.zeros(n_points)
        
        def initialize(self, initial_x):
            self.particles = [initial_x.copy() for _ in range(self.n_particles)]
            self.velocities = [np.random.rand(n_points) * 0.1 for _ in range(self.n_particles)]
            self.pbest = [p.copy() for p in self.particles]
            self.pbest_error = [objective(p) for p in self.particles]
            self.gbest = self.particles[np.argmin(self.pbest_error)]
            self.best_position = self.gbest.copy()
            self.gbest_error = np.min(self.pbest_error)
        
        def update(self):
            for i in range(self.n_particles):
                r1 = np.random.rand(n_points)
                r2 = np.random.rand(n_points)
                self.velocities[i] = 0.8 * self.velocities[i] + 1.5 * r1 * (self.pbest[i] - self.particles[i]) + 1.5 * r2 * (self.best_position - self.particles[i])
                self.particles[i] = self.particles[i] + self.velocities[i]
                self.particles[i] = norm(self.particles[i])
                current_error = objective(self.particles[i])
                if current_error < self.pbest_error[i]:
                    self.pbest[i] = self.particles[i].copy()
                    self.pbest_error[i] = current_error
            best_index = np.argmin(self.pbest_error)
            if self.pbest_error[best_index] < self.gbest_error:
                self.gbest = self.pbest[best_index].copy()
                self.gbest_error = self.pbest_error[best_index]
                self.best_position = self.gbest.copy()
        
        def run(self):
            start_time = time.time()
            self.initialize(h)
            for _ in range(self.max_iter):
                self.update()
                if time.time() - start_time > budget_s - 1:
                    break
            return self.best_position, self.gbest_error
    
    pso = PSO(n_particles=50, max_iter=100)
    best_h, best_c5 = pso.run()
    best_h = norm(best_h)
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
