import numpy as np
import scipy.optimize as opt
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h = initial_h_values
    if len(initial_h) != n_points:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.full_like(h, 0.5)
        return h_clipped * (n_points / 2) / total

    def evaluate_candidate(h):
        h = project(h)
        c5 = compute_c5(h)
        return c5

    def get_random_candidate():
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=n_points // 2, replace=False)
        h[indices] = 1
        return h

    def optimize_with_l_bfgs(h, maxiter=100):
        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.minimize(
            lambda x: evaluate_candidate(x),
            h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'ftol': 1e-9, 'maxiter': maxiter}
        )
        return result.x

    def optimize_with_slsqp(h, maxiter=100):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2, 'jac': lambda x: np.ones_like(x)}
        result = opt.minimize(
            lambda x: evaluate_candidate(x),
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': maxiter}
        )
        return result.x

    def bayesian_optimization(n_iter=200):
        X = []
        y = []
        for _ in range(n_iter):
            if len(X) < 20:
                h = get_random_candidate()
                X.append(h)
                y.append(evaluate_candidate(h))
            else:
                kernel = RBF(length_scale=1.0, length_scale_bounds=(1e-2, 1e2))
                gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=10)
                gp.fit(np.array(X), np.array(y))
                next_h = gp.kernel_.length_scale * np.random.rand(n_points)
                next_h = project(next_h)
                X.append(next_h)
                y.append(evaluate_candidate(next_h))
            time_used = time.time() - time_start
            if time_used > time_remaining - 100:
                break
        X = np.array(X)
        y = np.array(y)
        return X, y

    def find_top_candidates(X, y, top_k=10):
        indices = np.argsort(y)[:top_k]
        return X[indices]

    X, y = bayesian_optimization(n_iter=200)
    top_candidates = find_top_candidates(X, y)
    for candidate in top_candidates:
        h = optimize_with_l_bfgs(candidate, maxiter=100)
        c5 = evaluate_candidate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    for _ in range(5):
        if time.time() - time_start > time_remaining - 100:
            break
        h = best_h + np.random.uniform(-0.05, 0.05, size=n_points)
        h = project(h)
        c5 = evaluate_candidate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    final_h = project(best_h)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, n_points
