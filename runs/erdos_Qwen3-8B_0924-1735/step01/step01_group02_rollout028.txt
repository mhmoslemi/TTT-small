import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        if np.abs(np.sum(h) - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    def objective(h):
        return compute_c5(h)
    
    def project(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.full_like(h, 0.5)
        return h_clipped * (n_points / 2) / total
    
    def perturb(h, sigma):
        noise = np.random.normal(0, sigma, size=n_points)
        h = h + noise
        return project(h)
    
    def optimize_population(pop, max_iter=100, sigma=0.25):
        for _ in range(max_iter):
            for i in range(len(pop)):
                h = pop[i]
                h_new = perturb(h, sigma)
                c5_new = compute_c5(h_new)
                if c5_new < compute_c5(h):
                    pop[i] = h_new
            sigma *= 0.99
        return pop
    
    def refine_h(h):
        def surrogate(x):
            h_clipped = np.clip(x, 0, 1)
            total = np.sum(h_clipped)
            if np.abs(total - n_points / 2) > 1e-6:
                return np.inf
            corr = np.correlate(h_clipped, 1 - h_clipped, mode='full') * dx
            return -np.log(np.sum(np.exp(100 * (corr - 1e-6))))
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2, 'jac': lambda x: np.ones_like(x)}
        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.minimize(surrogate, h, method='L-BFGS-B', bounds=bounds, constraints=cons)
        return result.x
    
    pop_size = 50
    pop = [project(initial_h + np.random.normal(0, 0.25, n_points)) for _ in range(pop_size)]
    pop = optimize_population(pop, max_iter=100, sigma=0.25)
    candidates = [refine_h(h) for h in pop]
    
    for h in candidates:
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    h = best_h
    h = np.clip(h, 0, 1)
    if np.abs(np.sum(h) - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / np.sum(h)
    best_h = h
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
