import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    h_initial = np.zeros(n_points)
    initial_h_values = np.random.permutation(h_initial)
    initial_h_values[:n_points//2] = 1.0
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    initial_h_values = np.clip(initial_h_values, 0, 1)

    def objective(h):
        h_clip = np.clip(h, 0, 1)
        h_norm = h_clip * (n_points / 2) / np.sum(h_clip)
        cross_corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    def constraint(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = initial_h_values.copy()

    def evolve_population(pop, F=0.8, CR=0.9):
        n = len(pop)
        new_pop = []
        for i in range(n):
            a, b, c = np.random.choice(n, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            trial = np.clip(pop[i] * (1 - CR) + mutant * CR, 0, 1)
            trial = trial * (n_points / 2) / np.sum(trial)
            trial = np.clip(trial, 0, 1)
            if np.abs(objective(trial) - objective(pop[i])) < 1e-6:
                new_pop.append(trial)
            else:
                new_pop.append(pop[i])
        return new_pop

    population = [initial_guess.copy() for _ in range(20)]
    best_h = initial_guess.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        population = evolve_population(population)
        for h in population:
            c5 = objective(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    def grad_descent(h, max_iter=100, tol=1e-6):
        h = h.copy()
        for _ in range(max_iter):
            h_norm = h * (n_points / 2) / np.sum(h)
            cross_corr = np.correlate(h_norm, 1 - h_norm, mode='full')
            c5 = np.max(cross_corr) * dx
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-6
                h_plus_norm = h_plus * (n_points / 2) / np.sum(h_plus)
                cross_corr_plus = np.correlate(h_plus_norm, 1 - h_plus_norm, mode='full')
                c5_plus = np.max(cross_corr_plus) * dx
                grad[i] = (c5_plus - c5) / 1e-6
            h = h - 0.1 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            if np.linalg.norm(grad) < tol:
                break
        return h

    best_h = grad_descent(best_h)
    best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
