import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    if 'initial_h_values' in locals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                initial_h[i] = 1.0
            else:
                initial_h[i] = 0.0
        initial_h = np.random.permutation(initial_h)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        initial_h = np.clip(initial_h, 0, 1)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5
    
    def constraint(params):
        return np.sum(params) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = initial_h.copy()
    
    def de_search(pop_size=20, mut_fac=0.8, cross_prob=0.9, max_iter=300):
        pop = np.random.rand(pop_size, n_points)
        pop = pop * (n_points / 2) / np.sum(pop, axis=1, keepdims=True)
        pop = np.clip(pop, 0, 1)
        best = pop.copy()
        best_val = np.inf
        for _ in range(max_iter):
            for i in range(pop_size):
                a, b, c = np.random.choice(pop_size, 3, replace=False)
                mutant = pop[a] + mut_fac * (pop[b] - pop[c])
                mutant = np.clip(mutant, 0, 1)
                mutant = mutant * (n_points / 2) / np.sum(mutant)
                mutant = np.clip(mutant, 0, 1)
                trial = np.zeros_like(pop[i])
                for j in range(n_points):
                    if np.random.rand() < cross_prob:
                        trial[j] = mutant[j]
                    else:
                        trial[j] = pop[i][j]
                trial = trial * (n_points / 2) / np.sum(trial)
                trial = np.clip(trial, 0, 1)
                trial_val = evaluate(trial)
                if trial_val < evaluate(pop[i]):
                    pop[i] = trial
            current_best = np.min([evaluate(ind) for ind in pop])
            if current_best < best_val:
                best = pop[np.argmin([evaluate(ind) for ind in pop])]
                best_val = current_best
            if best_val < 1e-6:
                break
        return best, best_val
    
    start_time = time.time()
    best_h, best_c5 = de_search()
    elapsed_time = time.time() - start_time
    
    if elapsed_time >= budget_s:
        return (best_h, best_c5, n_points)
    
    def objective(params):
        return evaluate(params)
    
    def grad_objective(params):
        eps = 1e-6
        grad = np.zeros_like(params)
        for i in range(n_points):
            params_plus = params.copy()
            params_plus[i] += eps
            params_plus = params_plus * (n_points / 2) / np.sum(params_plus)
            params_plus = np.clip(params_plus, 0, 1)
            params_minus = params.copy()
            params_minus[i] -= eps
            params_minus = params_minus * (n_points / 2) / np.sum(params_minus)
            params_minus = np.clip(params_minus, 0, 1)
            grad[i] = (evaluate(params_plus) - evaluate(params_minus)) / (2 * eps)
        return grad
    
    def grad_constr(params):
        return np.array([1.0] * n_points)
    
    result = opt.minimize(
        fun=objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint, 'jac': grad_constr}],
        jac=grad_objective,
        tol=1e-6,
        options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-8}
    )
    
    final_h = result.x
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_h = np.clip(final_h, 0, 1)
    final_c5 = evaluate(final_h)
    
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (final_h, final_c5, n_points)
    
    return (final_h, final_c5, n_points)
