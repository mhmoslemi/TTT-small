import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        n_points = 100
        initial_h = np.zeros(n_points)
        block_width = 2.0 / 10
        for i in range(n_points):
            if i < block_width * 5:
                initial_h[i] = 1.0
            elif i < block_width * 10:
                initial_h[i] = 0.0
            elif i < block_width * 15:
                initial_h[i] = 1.0
            elif i < block_width * 20:
                initial_h[i] = 0.0
            else:
                initial_h[i] = 1.0
        initial_h = np.random.permutation(initial_h)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        initial_h = np.clip(initial_h, 0, 1)

    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        cross_corr = np.correlate(x, 1 - x, mode='full')
        return np.max(cross_corr) * dx

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    class PSO:
        def __init__(self, n_particles=50, n_iter=100):
            self.n_particles = n_particles
            self.n_iter = n_iter
            self.particles = np.random.rand(n_particles, n_points)
            self.particles = self.particles * (n_points / 2) / np.sum(self.particles, axis=1, keepdims=True)
            self.particles = np.clip(self.particles, 0, 1)
            self.pbest = self.particles.copy()
            self.pbest_cost = np.array([objective(p) for p in self.particles])
            self.gbest = self.pbest[self.pbest_cost.argmin()]
            self.gbest_cost = self.pbest_cost.min()
            self.velocities = np.random.rand(n_particles, n_points) * 0.01

        def update(self):
            for i in range(self.n_particles):
                r1, r2 = np.random.rand(2)
                self.velocities[i] = 0.8 * self.velocities[i] + 1.5 * r1 * (self.pbest[i] - self.particles[i]) + 1.5 * r2 * (self.gbest - self.particles[i])
                self.particles[i] = self.particles[i] + self.velocities[i]
                self.particles[i] = np.clip(self.particles[i], 0, 1)
                self.particles[i] = self.particles[i] * (n_points / 2) / np.sum(self.particles[i])
                cost = objective(self.particles[i])
                if cost < self.pbest_cost[i]:
                    self.pbest[i] = self.particles[i].copy()
                    self.pbest_cost[i] = cost
            self.gbest = self.pbest[self.pbest_cost.argmin()]
            self.gbest_cost = self.pbest_cost.min()

    start_time = time.time()
    pso = PSO(n_particles=50, n_iter=100)
    for _ in range(pso.n_iter):
        if time.time() - start_time > budget_s:
            break
        pso.update()
    best_h = pso.gbest
    best_c5 = pso.gbest_cost

    return (best_h, best_c5, n_points)
