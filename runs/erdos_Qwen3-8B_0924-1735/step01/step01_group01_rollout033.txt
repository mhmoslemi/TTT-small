import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h_initial[i] = 1.0
            else:
                h_initial[i] = 0.0
        h_initial = np.random.permutation(h_initial)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        h_initial = np.clip(h_initial, 0, 1)

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        return np.max(cross_corr) * dx

    def objective(params):
        return evaluate(params)

    def constraint(params):
        return np.sum(params) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = h_initial.copy()

    def de_search():
        pop_size = 20
        pop = np.random.uniform(0, 1, (pop_size, n_points))
        pop = pop * (n_points / 2) / np.sum(pop, axis=1, keepdims=True)
        pop = np.clip(pop, 0, 1)
        best = pop[0]
        best_val = evaluate(best)
        for _ in range(300):
            for i in range(pop_size):
                a, b, c = np.random.choice(pop_size, 3, replace=False)
                mutant = pop[a] + 0.8 * (pop[b] - pop[c])
                mutant = np.clip(mutant, 0, 1)
                mutant = mutant * (n_points / 2) / np.sum(mutant)
                mutant = np.clip(mutant, 0, 1)
                cross = np.random.rand(n_points) < 0.9
                trial = np.where(cross, mutant, pop[i])
                trial = trial * (n_points / 2) / np.sum(trial)
                trial = np.clip(trial, 0, 1)
                trial_val = evaluate(trial)
                if trial_val < best_val:
                    best = trial
                    best_val = trial_val
            if np.abs(best_val - evaluate(best)) < 1e-6:
                break
        return best, best_val

    start_time = time.time()
    h_opt, c5_opt = de_search()
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (h_opt, c5_opt, n_points)

    def grad_descent(h):
        for _ in range(100):
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-4
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = h.copy()
                h_minus[i] -= 1e-4
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                h_minus = np.clip(h_minus, 0, 1)
                val_plus = evaluate(h_plus)
                val_minus = evaluate(h_minus)
                grad[i] = (val_plus - val_minus) / (2e-4)
            h = h - 0.1 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            if np.abs(evaluate(h) - c5_opt) < 1e-6:
                break
        return h

    h_opt = grad_descent(h_opt)
    c5_opt = evaluate(h_opt)

    return (h_opt, c5_opt, n_points)
