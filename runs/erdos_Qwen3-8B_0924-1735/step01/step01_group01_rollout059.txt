import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        m_blocks = 10
        block_size = 2.0 / m_blocks
        for i in range(m_blocks):
            start = i * block_size
            end = (i + 1) * block_size
            idx = (start <= np.linspace(0, 2, n_points)) * (np.linspace(0, 2, n_points) <= end)
            h[idx] = 1.0
        h = np.random.permutation(h)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)

    def evaluate(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - n_points / 2) > 1e-6:
            h = h * (n_points / 2) / total
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    def objective(params):
        return evaluate(params)

    def constraint(params):
        h = params.copy()
        total = np.sum(h)
        if abs(total - n_points / 2) > 1e-6:
            h = h * (n_points / 2) / total
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = h.copy()

    start_time = time.time()
    max_iter = 100
    particles = 50
    inertia = 0.8
    c1, c2 = 1.5, 1.5
    velocities = np.random.rand(particles, n_points) * 0.1
    personal_bests = np.array([initial_guess.copy() for _ in range(particles)])
    global_best = initial_guess.copy()
    best_value = evaluate(global_best)

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for i in range(particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = inertia * velocities[i] + c1 * r1 * (personal_bests[i] - initial_guess) + c2 * r2 * (global_best - initial_guess)
            new_pos = initial_guess + velocities[i]
            new_pos = np.clip(new_pos, 0, 1)
            total = np.sum(new_pos)
            if abs(total - n_points / 2) > 1e-6:
                new_pos = new_pos * (n_points / 2) / total
            new_value = evaluate(new_pos)
            if new_value < evaluate(personal_bests[i]):
                personal_bests[i] = new_pos.copy()
                if new_value < best_value:
                    global_best = new_pos.copy()
                    best_value = new_value

    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (initial_guess, evaluate(initial_guess), n_points)
    return (global_best, best_value, n_points)
