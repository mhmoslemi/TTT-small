```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 120
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0:
                h[i] = 1.0
            elif i % 3 == 0:
                h[i] = 0.5
        h = h / np.sum(h) * n_points / 2
    dx = 2.0 / n_points
    h = np.clip(h, 0, 1)
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def objective(h):
        return compute_c5(h)

    def surrogate_objective(h):
        if np.sum(h) < n_points / 2 - 1e-6 or np.sum(h) > n_points / 2 + 1e-6:
            return np.inf
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.log(np.sum(np.exp(100 * (corr - 1e-6))))

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    cons = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)}
    bounds = [(0, 1) for _ in range(n_points)]

    def perturb(h, sigma):
        noise = np.random.normal(0, sigma, size=n_points)
        return np.clip(h + noise, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    for _ in range(50):
        if time.time() - time_start > budget_s - 100:
            break
        h = perturb(best_h, 0.15)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    result = opt.minimize(
        surrogate_objective,
        best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 100}
    )

    best_h = result.x
    best_c5 = compute_c5(best_h)

    for _ in range(10):
        if time.time() - time_start > budget_s - 100:
            break
        h = perturb(best_h, 0.075)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    result = opt.minimize(
        objective,
        best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 100}
    )

    best_h = result.x
    best_c5 = compute_c5(best_h)

    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)

    best_h = np.clip(best_h, 0, 1)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_points
```
