import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    if 'initial_h_values' in kwargs:
        initial_h_values = kwargs['initial_h_values']
    else:
        initial_h_values = np.zeros(n_points)
        block_size = n_points // 5
        for i in range(5):
            start = i * block_size
            end = (i + 1) * block_size
            initial_h_values[start:end] = 1.0
        initial_h_values = np.random.permutation(initial_h_values)
        initial_h_values /= np.sum(initial_h_values)
        initial_h_values *= n_points / 2
        initial_h_values = np.clip(initial_h_values, 0, 1)

    def objective(params):
        h = params.reshape(-1)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    def constraint(params):
        h = params.reshape(-1)
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = initial_h_values.copy()
    initial_guess = initial_guess.reshape(-1)

    def differential_evolution(population_size=20, max_iter=300, tol=1e-6):
        population = np.random.uniform(0, 1, (population_size, n_points))
        population = np.clip(population, 0, 1)
        population = population * (n_points / 2) / np.sum(population, axis=1, keepdims=True)
        population = np.clip(population, 0, 1)
        best = population.copy()
        best_c5 = np.inf
        for _ in range(max_iter):
            for i in range(population_size):
                a, b, c = np.random.choice(population_size, 3, replace=False)
                mutant = population[a] + 0.8 * (population[b] - population[c])
                mutant = np.clip(mutant, 0, 1)
                mutant = mutant * (n_points / 2) / np.sum(mutant)
                mutant = np.clip(mutant, 0, 1)
                trial = np.copy(population[i])
                for j in range(n_points):
                    if np.random.rand() < 0.9:
                        trial[j] = mutant[j]
                trial = np.clip(trial, 0, 1)
                trial = trial * (n_points / 2) / np.sum(trial)
                trial = np.clip(trial, 0, 1)
                c5 = objective(trial)
                if c5 < objective(population[i]):
                    population[i] = trial
            current_best = population[np.argmin([objective(p) for p in population])]
            current_c5 = objective(current_best)
            if current_c5 < best_c5:
                best = current_best
                best_c5 = current_c5
            if abs(best_c5 - objective(best)) < tol:
                break
        return best, best_c5

    start_time = time.time()
    best_h, best_c5 = differential_evolution()
    elapsed_time = time.time() - start_time

    if elapsed_time >= budget_s:
        return (best_h, best_c5, n_points)

    def gradient_descent(h_start, max_iter=100, tol=1e-6):
        h = h_start.copy()
        for _ in range(max_iter):
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            cross_corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(cross_corr) * dx
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-6
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_plus = np.clip(h_plus, 0, 1)
                cross_corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(cross_corr_plus) * dx
                grad[i] = (c5_plus - c5) / 1e-6
            h -= 0.01 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            if np.linalg.norm(grad) < tol:
                break
        return h

    best_h = gradient_descent(best_h)
    best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
