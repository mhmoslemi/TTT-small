import numpy as np
import scipy.optimize as opt
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    def project(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.full_like(h, 0.5)
        return h_clipped * (n_points / 2) / total
    
    def objective(h):
        return compute_c5(h)
    
    def surrogate_objective(h):
        if np.sum(h) < n_points / 2 - 1e-6 or np.sum(h) > n_points / 2 + 1e-6:
            return np.inf
        h_clipped = np.clip(h, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode="full") * dx
        return np.max(corr)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2, 'jac': lambda h: np.ones_like(h)}
    
    def ucb_acquisition(X, gp, beta=1.2):
        mean, std = gp.predict(X, return_std=True)
        return mean - beta * std
    
    def optimize_with_bo(n_evals=200):
        X = np.zeros((n_evals, n_points))
        y = np.zeros(n_evals)
        X[0] = project(initial_h)
        y[0] = compute_c5(X[0])
        gp = GaussianProcessRegressor(kernel=RBF(), alpha=1e-5)
        for i in range(1, n_evals):
            if time.time() - time_start > budget_s - 60:
                break
            X_next = np.random.uniform(0, 1, (1, n_points))
            X_next = project(X_next)
            X[i] = X_next
            y[i] = compute_c5(X_next)
            gp.fit(X[:i+1], y[:i+1])
        X_sorted = X[np.argsort(-y)]
        return X_sorted[:10]
    
    candidates = optimize_with_bo()
    for cand in candidates:
        if time.time() - time_start > budget_s - 200:
            break
        result = opt.minimize(
            objective,
            cand,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        h = result.x
        c5_value = compute_c5(h)
        if c5_value < best_c5:
            best_c5 = c5_value
            best_h = h.copy()
    
    h = best_h
    h = project(h)
    c5_value = compute_c5(h)
    if c5_value < best_c5:
        best_c5 = c5_value
        best_h = h.copy()
    
    h = best_h
    h = np.clip(h, 0, 1)
    if np.abs(np.sum(h) - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / np.sum(h)
    c5_value = compute_c5(h)
    if c5_value < best_c5:
        best_c5 = c5_value
        best_h = h.copy()
    
    h_final = best_h
    h_final = np.clip(h_final, 0, 1)
    if np.abs(np.sum(h_final) - n_points / 2) > 1e-6:
        h_final = h_final * (n_points / 2) / np.sum(h_final)
    final_c5 = compute_c5(h_final)
    
    return h_final, final_c5, n_points
