import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if hasattr(globals(), 'initial_h_values') and len(initial_h_values) == n_points:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    h = h / np.sum(h) * n_points / 2
    dx = 2.0 / n_points
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        if np.abs(np.sum(h) - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    bounds = [(0, 1) for _ in range(n_points)]

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    cons = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)}

    def simulate_annealing(h0, max_iter=1000, T0=300, T_min=0.1):
        T = T0
        h = h0.copy()
        best_h = h.copy()
        best_c5 = compute_c5(h)
        for t in range(max_iter):
            if time.time() - time_start > budget_s - 100:
                break
            noise = np.random.normal(0, 0.25, size=n_points)
            h_new = np.clip(h + noise, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            delta = c5_new - best_c5
            if delta < 0 or np.random.random() < np.exp(-delta / T):
                h = h_new.copy()
            T *= 0.99
        return best_h, best_c5

    def local_refinement(h0, max_iter=500):
        result = opt.minimize(
            objective,
            h0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': max_iter}
        )
        return result.x, compute_c5(result.x)

    h, best_c5 = simulate_annealing(h)
    for _ in range(5):
        if time.time() - time_start > budget_s - 100:
            break
        h, _ = local_refinement(h)
        h = np.clip(h, 0, 1)
        if np.abs(np.sum(h) - n_points / 2) > 1e-6:
            h = h * (n_points / 2) / np.sum(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    h = best_h
    h = np.clip(h, 0, 1)
    if np.abs(np.sum(h) - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / np.sum(h)
    c5_final = compute_c5(h)
    return h, c5_final, n_points
