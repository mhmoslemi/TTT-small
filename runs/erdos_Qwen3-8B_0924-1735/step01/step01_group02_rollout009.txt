import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]

    def perturb(h, sigma):
        return np.clip(h + np.random.normal(0, sigma, n_points), 0, 1)

    def feasibility_project(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return 0.5 * np.ones(n_points)
        return h_clipped * (n_points / 2) / sum_clipped

    def objective(h):
        return compute_c5(h)

    def surrogate_objective(h):
        h_clipped = np.clip(h, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.inf
        corr = np.correlate(h_clipped, 1 - h_clipped, mode="full") * dx
        return np.log(np.sum(np.exp(100 * (corr - 1e-6))))

    cons = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)}

    def simulated_annealing():
        nonlocal best_h, best_c5
        T = 300
        t = 0
        max_iterations = 1000
        candidates = []
        while t < max_iterations and time.time() - time_start < time_remaining - 100:
            h_perturbed = perturb(h, 0.25 if t < 200 else 0.1)
            h_perturbed = feasibility_project(h_perturbed)
            c5_perturbed = compute_c5(h_perturbed)
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed.copy()
            delta_c5 = c5_perturbed - best_c5
            if delta_c5 < 0:
                h = h_perturbed
            else:
                if np.random.rand() < np.exp(-delta_c5 / T):
                    h = h_perturbed
            t += 1
            T *= 0.99
            if t % 10 == 0:
                candidates.append(h.copy())
        return candidates

    def refine_candidates(candidates):
        nonlocal best_h, best_c5
        for c in candidates[:100]:
            if time.time() - time_start > time_remaining - 100:
                break
            result = opt.minimize(
                surrogate_objective,
                c,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 100}
            )
            h_refined = result.x
            c5_refined = compute_c5(h_refined)
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined.copy()

    candidates = simulated_annealing()
    refine_candidates(candidates)

    best_h = feasibility_project(best_h)
    c5_value = compute_c5(best_h)
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    c5_value = compute_c5(best_h)
    return best_h, c5_value, n_points
