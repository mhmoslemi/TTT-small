import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 120
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        initial_pop = np.array(initial_h_values, dtype=float)
        initial_pop = initial_pop.reshape(-1, 1)
    else:
        initial_pop = np.random.rand(n_points, 30)
        initial_pop *= (n_points / 2) / np.sum(initial_pop, axis=1, keepdims=True)
    
    def evaluate_individual(h):
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    best_h = initial_pop.mean(axis=1)
    best_c5 = evaluate_individual(best_h)
    
    elapsed = 0
    generations = 100
    pop_size = 30
    mutation_rate = 0.1
    
    for gen in range(generations):
        if elapsed >= budget_s:
            break
        
        fitness = np.array([evaluate_individual(p) for p in initial_pop])
        elite_indices = np.argsort(fitness)[:int(pop_size * 0.2)]
        elite = initial_pop[elite_indices]
        
        new_pop = elite
        for _ in range(pop_size - int(pop_size * 0.2)):
            parent1 = np.random.choice(elite)
            parent2 = np.random.choice(elite)
            child = np.mean([parent1, parent2], axis=0)
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] += np.random.uniform(-0.1, 0.1)
            child = np.clip(child, 0, 1)
            child *= (n_points / 2) / np.sum(child)
            new_pop = np.vstack([new_pop, child])
        
        initial_pop = new_pop
        current_fitness = fitness[np.argsort(fitness)[:int(pop_size * 0.2)]]
        if np.min(current_fitness) < best_c5:
            best_c5 = np.min(current_fitness)
            best_h = initial_pop[np.argmin(current_fitness)]
        elapsed += 1
    
    best_h = best_h
    best_h = np.clip(best_h, 0, 1)
    best_h *= (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    c5_bound = evaluate_individual(best_h)
    
    return best_h, c5_bound, n_points
