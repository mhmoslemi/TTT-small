import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds
from scipy.optimize import minimize
from scipy.optimize import Bounds
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45  # Starting with the provided initial_h_values
    initial_h = initial_h_values
    if len(initial_h) != n_points:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.full_like(h, 0.5)
        return h_clipped * (n_points / 2) / total

    def objective(h):
        return compute_c5(h)

    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2, 'jac': lambda x: np.ones_like(x)}

    bounds = [(0, 1) for _ in range(n_points)]

    def eval_func(h):
        h_proj = project(h)
        return compute_c5(h_proj)

    X = np.random.rand(20, n_points)
    X = project(X)
    y = np.array([compute_c5(x) for x in X])
    kernel = C(1.0, (0.1, 100.0)) * RBF(1.0, (0.1, 10.0))
    gp = GaussianProcessRegressor(kernel=kernel, alpha=0.1, n_restarts_optimizer=10)

    def acquisition(X, gp, y_max):
        mu, sigma = gp.predict(X, return_std=True)
        mu = np.clip(mu, 1e-3, 1e3)
        imp = y_max - mu
        beta = 1.2
        acq = imp * np.exp(-beta * sigma ** 2)
        return acq

    n_evals = min(200, int(budget_s / 5))
    X_acq = np.random.rand(n_evals, n_points)
    X_acq = project(X_acq)
    y_acq = np.array([compute_c5(x) for x in X_acq])
    gp.fit(X_acq, y_acq)
    best_y = y_acq.min()

    for _ in range(n_evals):
        if time.time() - time_start > time_remaining - 100:
            break
        acq = acquisition(X_acq, gp, best_y)
        next_x = X_acq[np.argmin(acq)]
        next_x = project(next_x)
        next_x += np.random.normal(0, 0.05, size=n_points)
        next_x = project(next_x)
        y_new = compute_c5(next_x)
        X_acq = np.vstack((X_acq, next_x))
        y_acq = np.append(y_acq, y_new)
        gp.fit(X_acq, y_acq)
        best_y = y_acq.min()

    candidates = X_acq[y_acq == best_y]
    for candidate in candidates[:10]:
        if time.time() - time_start > time_remaining - 100:
            break
        result = minimize(
            objective,
            candidate,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        h_candidate = result.x
        c5_value = compute_c5(project(h_candidate))
        if c5_value < best_c5:
            best_c5 = c5_value
            best_h = h_candidate.copy()

    for _ in range(5):
        if time.time() - time_start > time_remaining - 100:
            break
        noise = np.random.uniform(-0.05, 0.05, size=n_points)
        h = best_h + noise
        h = project(h)
        c5_value = compute_c5(h)
        if c5_value < best_c5:
            best_c5 = c5_value
            best_h = h.copy()

    h = project(best_h)
    c5_value = compute_c5(h)
    if np.abs(np.sum(h) - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / np.sum(h)
    c5_value = compute_c5(h)
    return h, c5_value, n_points
