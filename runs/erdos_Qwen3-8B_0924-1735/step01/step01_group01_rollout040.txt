import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        initial_h_values = np.array(globals()['initial_h_values'])
    else:
        initial_h_values = np.zeros(n_points)
        for i in range(n_points):
            if i < 60:
                initial_h_values[i] = 1.0
            else:
                initial_h_values[i] = 0.0
        initial_h_values = np.random.permutation(initial_h_values)
        initial_h_values /= np.sum(initial_h_values)
        initial_h_values *= n_points / 2
        initial_h_values = np.clip(initial_h_values, 0, 1)

    def objective(params):
        h = params.reshape(-1)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    def constraint(params):
        h = params.reshape(-1)
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = initial_h_values.copy()
    initial_guess = initial_guess.reshape(-1)

    def de_search(pop, generations, tol):
        F = 0.8
        CR = 0.9
        for _ in range(generations):
            mutant, target = pop[np.random.choice(len(pop), 3, replace=False)], pop[np.random.choice(len(pop), 1, replace=False)][0]
            mutant = mutant + F * (pop[np.random.choice(len(pop), 3, replace=False)] - pop[np.random.choice(len(pop), 3, replace=False)])
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * (n_points / 2) / np.sum(mutant)
            mutant = np.clip(mutant, 0, 1)
            cross_over = np.zeros_like(target)
            for i in range(n_points):
                if np.random.rand() < CR or mutant[i] > target[i]:
                    cross_over[i] = mutant[i]
                else:
                    cross_over[i] = target[i]
            cross_over = cross_over * (n_points / 2) / np.sum(cross_over)
            cross_over = np.clip(cross_over, 0, 1)
            fx = objective(cross_over)
            target_fx = objective(target)
            if fx < target_fx:
                pop[0] = cross_over
            if np.abs(fx - target_fx) < tol:
                break
        return pop[0]

    def grad_descent(h, max_iter=100, tol=1e-6):
        for _ in range(max_iter):
            h_vals = h.reshape(-1)
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
            h_vals = np.clip(h_vals, 0, 1)
            cross_corr = np.correlate(h_vals, 1 - h_vals, mode='full')
            c5 = np.max(cross_corr) * dx
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_vals[i] += 1e-5
                h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
                h_vals = np.clip(h_vals, 0, 1)
                cross_corr = np.correlate(h_vals, 1 - h_vals, mode='full')
                c5_plus = np.max(cross_corr) * dx
                h_vals[i] -= 2e-5
                h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
                h_vals = np.clip(h_vals, 0, 1)
                cross_corr = np.correlate(h_vals, 1 - h_vals, mode='full')
                c5_minus = np.max(cross_corr) * dx
                grad[i] = (c5_plus - c5_minus) / (2e-5)
            grad = grad * (n_points / 2) / np.sum(grad)
            grad = np.clip(grad, -1, 1)
            step = -grad * 0.1
            h_vals = h_vals + step
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
            h_vals = np.clip(h_vals, 0, 1)
            if np.linalg.norm(grad) < tol:
                break
            h = h_vals.reshape(-1)
        return h

    start_time = time.time()
    population = [initial_guess.copy() for _ in range(20)]
    for _ in range(200):
        population = [de_search(population, 10, 1e-6) for _ in range(20)]
        population = np.unique(population, axis=0)
        if len(population) < 10:
            break
        population = population[:10]
    best = population[0]
    best = grad_descent(best)
    best = np.clip(best, 0, 1)
    best = best * (n_points / 2) / np.sum(best)
    best = np.clip(best, 0, 1)
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (initial_h_values, objective(initial_h_values), n_points)
    c5_opt = objective(best)
    return (best, c5_opt, n_points)
