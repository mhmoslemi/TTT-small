import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    if 'initial_h_values' in locals():
        h = initial_h_values.copy()
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < 50:
                h[i] = 1.0
            else:
                h[i] = 0.0
        h = np.random.permutation(h)
        h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)

    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    start_time = time.time()
    result = opt.minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        tol=1e-6,
        options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-8}
    )

    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h, objective(h), n_points)

    h_opt = np.clip(result.x, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    h_opt = np.clip(h_opt, 0, 1)
    c5_opt = objective(h_opt)

    return (h_opt, c5_opt, n_points)
