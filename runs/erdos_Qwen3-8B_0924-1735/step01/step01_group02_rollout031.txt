import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h = initial_h_values
    if len(initial_h) != n_points:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.full_like(h, 0.5)
        return h_clipped * (n_points / 2) / total

    def objective(h):
        return compute_c5(h)

    def surrogate_objective(h):
        if np.sum(h) < n_points / 2 - 1e-6 or np.sum(h) > n_points / 2 + 1e-6:
            return np.inf
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2, 'jac': lambda h: np.ones_like(h)}
    bounds = [(0, 1) for _ in range(n_points)]

    def bayesian_optimization():
        nonlocal best_h, best_c5
        candidates = []
        evaluations = {}
        for _ in range(200):
            if time.time() - time_start > budget_s - 60:
                break
            h_candidate = project(np.random.uniform(0, 1, n_points))
            c5_val = compute_c5(h_candidate)
            candidates.append(h_candidate)
            evaluations[tuple(h_candidate)] = c5_val
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_candidate.copy()
        for h_candidate in candidates:
            if tuple(h_candidate) not in evaluations:
                c5_val = compute_c5(h_candidate)
                evaluations[tuple(h_candidate)] = c5_val
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_candidate.copy()

    def local_refinement():
        nonlocal best_h, best_c5
        for _ in range(10):
            if time.time() - time_start > budget_s - 100:
                break
            h_candidate = best_h + np.random.uniform(-0.05, 0.05, n_points)
            h_candidate = project(h_candidate)
            c5_val = compute_c5(h_candidate)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_candidate.copy()

    def gradient_refinement():
        nonlocal best_h, best_c5
        result = opt.minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        h_candidate = result.x
        c5_val = compute_c5(h_candidate)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_candidate.copy()

    try:
        bayesian_optimization()
        local_refinement()
        gradient_refinement()
    except:
        pass

    best_h = project(best_h)
    best_h = np.clip(best_h, 0, 1)
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_points
