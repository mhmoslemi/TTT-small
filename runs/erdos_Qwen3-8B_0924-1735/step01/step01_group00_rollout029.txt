import numpy as np
import scipy.optimize as opt
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 120
    dx = 2.0 / n_points

    # Use provided initial_h_values if available
    if 'initial_h_values' in globals():
        initial_pop = np.array(initial_h_values)
        initial_pop = initial_pop.astype(float)
        initial_pop = initial_pop * (n_points / 2) / np.sum(initial_pop)
        initial_pop = np.clip(initial_pop, 0, 1)
    else:
        initial_pop = np.random.rand(n_points, 30)
        initial_pop = initial_pop * (n_points / 2) / np.sum(initial_pop, axis=1, keepdims=True)
        initial_pop = np.clip(initial_pop, 0, 1)

    population_size = 30
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8

    best_h = initial_pop.mean(axis=0)
    best_c5 = float('inf')

    def evaluate_individual(h):
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx

    for gen in range(generations):
        elapsed = gen * (budget_s // generations)
        if elapsed > budget_s:
            break

        # Evaluate population
        fitness = np.array([evaluate_individual(ind) for ind in initial_pop])
        best_idx = np.argmin(fitness)
        current_c5 = fitness[best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = initial_pop[best_idx].copy()

        # Select parents
        parents = initial_pop[np.argsort(fitness)[:int(population_size * 0.5)]]

        # Crossover
        offspring = []
        for i in range(int(population_size * 0.5)):
            for j in range(i + 1, int(population_size * 0.5)):
                if np.random.rand() < crossover_rate:
                    parent1 = parents[i]
                    parent2 = parents[j]
                    alpha = np.random.uniform(-1, 1)
                    child = 0.5 * (parent1 + parent2) + alpha * (parent1 - parent2)
                    child = np.clip(child, 0, 1)
                    child = child * (n_points / 2) / np.sum(child)
                    offspring.append(child)

        # Mutate
        for ind in initial_pop:
            if np.random.rand() < mutation_rate:
                mut = np.random.uniform(-0.1, 0.1, size=n_points)
                ind += mut
                ind = np.clip(ind, 0, 1)
                ind = ind * (n_points / 2) / np.sum(ind)

        # Combine and select
        initial_pop = np.vstack([parents, offspring])
        initial_pop = np.vstack([initial_pop, np.random.rand(n_points, population_size - len(initial_pop))])
        initial_pop = initial_pop[np.random.choice(len(initial_pop), size=population_size, replace=False)]

    # Final evaluation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = evaluate_individual(best_h)

    return best_h, best_c5, n_points
