import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        num_blocks = 10
        block_size = n_points // num_blocks
        for i in range(num_blocks):
            if i % 2 == 0:
                h[i * block_size:(i + 1) * block_size] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    def objective(params):
        h = params.reshape(-1)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5
    
    def constraint(params):
        h = params.reshape(-1)
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = h.copy()
    initial_guess = initial_guess.reshape(-1)
    
    start_time = time.time()
    
    class PSOParticle:
        def __init__(self, pos, vel, pbest, pbest_cost):
            self.pos = pos
            self.vel = vel
            self.pbest = pbest
            self.pbest_cost = pbest_cost
    
    class PSOSwarm:
        def __init__(self, n_particles, n_dimensions, inertia, c1, c2):
            self.n_particles = n_particles
            self.n_dimensions = n_dimensions
            self.inertia = inertia
            self.c1 = c1
            self.c2 = c2
            self.particles = []
            self.gbest = np.zeros(n_dimensions)
            self.gbest_cost = float('inf')
            
        def init_particles(self, initial_guess):
            for _ in range(self.n_particles):
                pos = initial_guess + 0.1 * np.random.rand(self.n_dimensions) - 0.05
                pos = np.clip(pos, 0, 1)
                pos = pos * (n_points / 2) / np.sum(pos)
                pos = np.clip(pos, 0, 1)
                vel = 0.01 * np.random.rand(self.n_dimensions)
                pbest = pos.copy()
                pbest_cost = objective(pos)
                self.particles.append(PSOParticle(pos, vel, pbest, pbest_cost))
            
            best_cost = float('inf')
            best_pos = np.zeros(self.n_dimensions)
            for p in self.particles:
                if p.pbest_cost < best_cost:
                    best_cost = p.pbest_cost
                    best_pos = p.pbest.copy()
            self.gbest = best_pos
            self.gbest_cost = best_cost
    
        def update(self, n_iterations):
            for _ in range(n_iterations):
                for p in self.particles:
                    r1, r2 = np.random.rand(2)
                    p.vel = self.inertia * p.vel + self.c1 * r1 * (p.pbest - p.pos) + self.c2 * r2 * (self.gbest - p.pos)
                    p.pos = p.pos + p.vel
                    p.pos = np.clip(p.pos, 0, 1)
                    p.pos = p.pos * (n_points / 2) / np.sum(p.pos)
                    p.pos = np.clip(p.pos, 0, 1)
                    p_cost = objective(p.pos)
                    if p_cost < p.pbest_cost:
                        p.pbest = p.pos.copy()
                        p.pbest_cost = p_cost
                        if p.pbest_cost < self.gbest_cost:
                            self.gbest = p.pbest.copy()
                            self.gbest_cost = p.pbest_cost
                if self.gbest_cost < 1e-8:
                    break
    
    swarm = PSOSwarm(50, n_points, 0.8, 1.5, 1.5)
    swarm.init_particles(initial_guess)
    swarm.update(100)
    
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (swarm.gbest, swarm.gbest_cost, n_points)
    
    h_opt = swarm.gbest.copy()
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    h_opt = np.clip(h_opt, 0, 1)
    c5_opt = objective(h_opt)
    
    return (h_opt, c5_opt, n_points)
