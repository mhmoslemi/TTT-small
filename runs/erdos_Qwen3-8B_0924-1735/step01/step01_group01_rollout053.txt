import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
        initial_h = np.random.permutation(initial_h)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        initial_h = np.clip(initial_h, 0, 1)
    else:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                initial_h[i] = 1.0
            elif i % 3 == 1:
                initial_h[i] = 0.5
            else:
                initial_h[i] = 0.0
        initial_h = np.random.permutation(initial_h)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        initial_h = np.clip(initial_h, 0, 1)

    def objective(params):
        h = params.reshape(-1)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    def constraint(params):
        h = params.reshape(-1)
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = initial_h.copy()
    initial_guess = initial_guess.reshape(-1)

    start_time = time.time()
    result = opt.differential_evolution(
        fun=objective,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        strategy='best1bin',
        mutation=(0.5, 1),
        crossover=0.9,
        popsize=20,
        tol=1e-6,
        callback=lambda x: None,
        maxiter=100,
        disp=False,
        polish=True,
        vectorized=True,
        mutation_func=None,
        crossover_func=None,
        seed=seed
    )

    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (initial_h, objective(initial_h), n_points)

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
