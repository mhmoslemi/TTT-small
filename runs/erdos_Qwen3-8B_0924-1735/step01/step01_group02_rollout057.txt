import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    h = h / np.sum(h) * n_points / 2
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]

    def objective(h):
        return compute_c5(h)

    def surrogate_objective(h):
        if np.sum(h) < n_points / 2 - 1e-6 or np.sum(h) > n_points / 2 + 1e-6:
            return np.inf
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.log(np.sum(np.exp(100 * (corr - 1e-6))))

    def feasibility_project(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return 0.5 * np.ones_like(h)
        return h * (n_points / 2) / sum_h

    def perturb(h, sigma):
        noise = np.random.normal(0, sigma, h.shape)
        return feasibility_project(h + noise)

    def simulated_annealing():
        nonlocal best_h, best_c5
        T = 300
        t = 0
        max_iter = 1000
        candidates = []
        while t < max_iter and time.time() - time_start < time_remaining - 100:
            h_candidate = perturb(best_h, 0.25 if t < 500 else 0.05)
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
                candidates.append(h_candidate)
            delta_c5 = c5_candidate - best_c5
            if delta_c5 < 0:
                accept = 1.0
            else:
                accept = np.exp(-delta_c5 / T)
            if np.random.rand() < accept:
                best_h = h_candidate
                best_c5 = c5_candidate
                candidates.append(h_candidate)
            T *= 0.95
            t += 1

    def local_refinement():
        nonlocal best_h, best_c5
        for _ in range(100):
            if time.time() - time_start > time_remaining - 100:
                break
            h_candidate = best_h.copy()
            h_candidate += np.random.normal(0, 0.05, h_candidate.shape)
            h_candidate = feasibility_project(h_candidate)
            result = opt.minimize(
                surrogate_objective,
                h_candidate,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)},
                options={'ftol': 1e-9, 'maxiter': 100}
            )
            h_candidate = result.x
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate

    simulated_annealing()
    local_refinement()

    h = best_h
    h = feasibility_project(h)
    c5_value = compute_c5(h)
    if c5_value < best_c5:
        best_c5 = c5_value
        best_h = h

    return best_h, best_c5, n_points
