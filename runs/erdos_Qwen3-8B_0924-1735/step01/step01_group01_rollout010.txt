import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    if 'initial_h_values' in kwargs and kwargs['initial_h_values'] is not None:
        initial_h_values = kwargs['initial_h_values'].copy()
    else:
        initial_h_values = np.zeros(n_points)
        for i in range(n_points):
            if i < 60:
                initial_h_values[i] = 1.0
            else:
                initial_h_values[i] = 0.0
        initial_h_values = np.random.permutation(initial_h_values)
        initial_h_values /= np.sum(initial_h_values)
        initial_h_values *= n_points / 2
        initial_h_values = np.clip(initial_h_values, 0, 1)

    def objective(params):
        h = params.reshape(-1)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    def constraint(params):
        h = params.reshape(-1)
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = initial_h_values.copy()
    initial_guess = initial_guess.reshape(-1)

    def de_objective(params):
        return objective(params)

    def de_constraint(params):
        return constraint(params)

    def de_bounds():
        return bounds

    start_time = time.time()
    result = opt.differential_evolution(
        de_objective,
        bounds=de_bounds(),
        args=(),
        strategy='best1bin',
        popsize=20,
        mutation=(0.8, 1.0),
        recombination=0.9,
        tol=1e-6,
        maxiter=100,
        constraints=[de_constraint],
        disp=False,
        vectorized=True
    )

    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s or result.success:
        h_opt = np.clip(result.x, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        h_opt = np.clip(h_opt, 0, 1)
        c5_opt = objective(h_opt)
        return (h_opt, c5_opt, n_points)

    h_opt = np.clip(result.x, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    h_opt = np.clip(h_opt, 0, 1)
    c5_opt = objective(h_opt)
    return (h_opt, c5_opt, n_points)
