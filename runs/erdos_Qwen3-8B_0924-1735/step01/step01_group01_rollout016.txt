import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    
    if 'initial_h_values' in locals():
        initial_h_values = initial_h_values.copy()
        initial_h_values = np.random.permutation(initial_h_values)
        initial_h_values /= np.sum(initial_h_values)
        initial_h_values *= n_points / 2
        initial_h_values = np.clip(initial_h_values, 0, 1)
    else:
        block_size = n_points // 10
        initial_h_values = np.zeros(n_points)
        for i in range(10):
            initial_h_values[i * block_size : (i + 1) * block_size] = 1.0
        initial_h_values = np.random.permutation(initial_h_values)
        initial_h_values /= np.sum(initial_h_values)
        initial_h_values *= n_points / 2
        initial_h_values = np.clip(initial_h_values, 0, 1)
    
    def objective(params):
        h = params.reshape(-1)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5
    
    def constraint(params):
        h = params.reshape(-1)
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = initial_h_values.copy()
    initial_guess = initial_guess.reshape(-1)
    
    def pso_run():
        P = 50
        max_iter = 100
        omega = 0.8
        c1, c2 = 1.5, 1.5
        particles = np.random.rand(P, n_points)
        particles = particles * 2.0 - 1.0
        particles = np.clip(particles, 0, 1)
        particles = particles * (n_points / 2) / np.sum(particles, axis=1, keepdims=True)
        particles = np.clip(particles, 0, 1)
        velocities = np.random.rand(P, n_points) * 0.01
        pbest = particles.copy()
        pbest_vals = np.array([objective(p) for p in pbest])
        gbest = pbest[np.argmin(pbest_vals)]
        gbest_val = np.min(pbest_vals)
        best_h = gbest.copy()
        best_c5 = gbest_val
        start_time = time.time()
        for i in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            for j in range(P):
                r1, r2 = np.random.rand(2)
                velocities[j] = omega * velocities[j] + c1 * r1 * (pbest[j] - particles[j]) + c2 * r2 * (gbest - particles[j])
                particles[j] += velocities[j]
                particles[j] = np.clip(particles[j], 0, 1)
                particles[j] = particles[j] * (n_points / 2) / np.sum(particles[j])
                particles[j] = np.clip(particles[j], 0, 1)
                current_val = objective(particles[j])
                if current_val < pbest_vals[j]:
                    pbest[j] = particles[j].copy()
                    pbest_vals[j] = current_val
            current_gbest = pbest[np.argmin(pbest_vals)]
            current_gbest_val = np.min(pbest_vals)
            if np.abs(current_gbest_val - gbest_val) < 1e-8:
                break
            gbest = current_gbest
            gbest_val = current_gbest_val
            best_h = gbest.copy()
            best_c5 = gbest_val
        return best_h, best_c5
    
    start_time = time.time()
    h_opt, c5_opt = pso_run()
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (initial_h_values, objective(initial_h_values), n_points)
    
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    h_opt = np.clip(h_opt, 0, 1)
    c5_opt = objective(h_opt)
    
    return (h_opt, c5_opt, n_points)
