import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    h = h / np.sum(h) * n_points / 2
    dx = 2.0 / n_points
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        if np.abs(np.sum(h) - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def feasibility_proj(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return 0.5 * np.ones_like(h)
        return h * (n_points / 2) / total

    def surrogate(h):
        h_proj = feasibility_proj(h)
        corr = np.correlate(h_proj, 1 - h_proj, mode="full") * dx
        return -np.log(np.sum(np.exp(100 * (corr - 1e-6))))

    def objective(h):
        return compute_c5(h)

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2, 'jac': lambda h: np.ones_like(h)}

    best_h = h.copy()
    best_c5 = compute_c5(h)
    time_start = time.time()
    time_remaining = budget_s

    def perturb(h, sigma):
        return feasibility_proj(h + np.random.normal(0, sigma, n_points))

    def annealing_step(h, temp, sigma, best_c5):
        h_perturbed = perturb(h, sigma)
        c5_perturbed = compute_c5(h_perturbed)
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed.copy()
        delta_c5 = c5_perturbed - best_c5
        if delta_c5 < 0:
            return h_perturbed, best_c5, best_h
        prob = np.exp(-delta_c5 / temp)
        if np.random.rand() < prob:
            return h_perturbed, best_c5, best_h
        return h, best_c5, best_h

    current_h = h.copy()
    temp = 300
    sigma = 0.25
    for _ in range(1000):
        if time.time() - time_start > time_remaining - 100:
            break
        current_h, best_c5, best_h = annealing_step(current_h, temp, sigma, best_c5)
        temp *= 0.95
        sigma *= 0.98

    def refine_candidate(h):
        result = opt.minimize(
            surrogate,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        h_refined = result.x
        return h_refined

    top_candidates = [best_h]
    for _ in range(20):
        if time.time() - time_start > time_remaining - 100:
            break
        h_candidate = feasibility_proj(np.random.normal(best_h, 0.1, n_points))
        top_candidates.append(h_candidate)
    top_candidates = np.array(top_candidates)
    for i in range(len(top_candidates)):
        h_candidate = top_candidates[i]
        h_refined = refine_candidate(h_candidate)
        c5_refined = compute_c5(h_refined)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()

    final_h = feasibility_proj(best_h)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, n_points
