import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    h = h / np.sum(h) * n_points / 2
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def objective(h):
        return compute_c5(h)

    def feasibility(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        return h

    def perturb(h, sigma):
        return feasibility(h + np.random.normal(0, sigma, size=n_points))

    def simulated_annealing():
        nonlocal best_h, best_c5
        T = 300.0
        max_steps = 1000
        for step in range(max_steps):
            if time.time() - time_start > time_remaining - 100:
                break
            noise = np.random.normal(0, 0.25, size=n_points)
            h_candidate = perturb(best_h, 0.25)
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
            delta = c5_candidate - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                best_h = h_candidate.copy()
            T *= 0.97

    def local_refinement():
        nonlocal best_h, best_c5
        for _ in range(20):
            if time.time() - time_start > time_remaining - 100:
                break
            result = opt.minimize(
                lambda x: compute_c5(x),
                best_h,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2},
                options={'ftol': 1e-9, 'maxiter': 200}
            )
            h_candidate = result.x
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()

    simulated_annealing()
    local_refinement()

    h = best_h
    h = np.clip(h, 0, 1)
    if np.abs(np.sum(h) - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / np.sum(h)
    c5_value = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_value
    return best_h, best_c5, n_points
