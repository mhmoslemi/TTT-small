import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        if np.abs(np.sum(h) - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    cons = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)}
    bounds = [(0, 1) for _ in range(n_points)]

    def simulated_annealing():
        nonlocal best_h, best_c5
        T = 300.0
        cooling_rate = 0.995
        max_iter = 1000
        for t in range(max_iter):
            if time.time() - time_start > time_remaining - 100:
                break
            noise = np.random.normal(0, 0.2, n_points)
            h_prime = np.clip(h + noise, 0, 1)
            sum_hp = np.sum(h_prime)
            if np.abs(sum_hp - n_points / 2) > 1e-6:
                h_prime = h_prime * (n_points / 2) / sum_hp
            c5_h = compute_c5(h_prime)
            if c5_h < best_c5:
                best_c5 = c5_h
                best_h = h_prime.copy()
            delta_c5 = c5_h - best_c5
            if delta_c5 < 0:
                h = h_prime.copy()
            else:
                if np.random.rand() < np.exp(-delta_c5 / T):
                    h = h_prime.copy()
            T *= cooling_rate

    def local_refinement():
        nonlocal best_h, best_c5
        for _ in range(5):
            if time.time() - time_start > time_remaining - 100:
                break
            result = opt.minimize(
                objective,
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 200}
            )
            h = result.x
            c5_value = compute_c5(h)
            if c5_value < best_c5:
                best_c5 = c5_value
                best_h = h.copy()

    def perturb_and_refine():
        nonlocal best_h, best_c5
        for _ in range(5):
            if time.time() - time_start > time_remaining - 100:
                break
            noise = np.random.uniform(-0.1, 0.1, n_points)
            h = best_h + noise
            h = np.clip(h, 0, 1)
            if np.abs(np.sum(h) - n_points / 2) < 1e-6:
                c5_value = compute_c5(h)
                if c5_value < best_c5:
                    best_c5 = c5_value
                    best_h = h.copy()

    simulated_annealing()
    local_refinement()
    perturb_and_refine()

    h = best_h
    h = np.clip(h, 0, 1)
    if np.abs(np.sum(h) - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / np.sum(h)
    c5_value = compute_c5(h)
    if c5_value < best_c5:
        best_c5 = c5_value
        best_h = h.copy()

    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    c5_value = compute_c5(best_h)
    return best_h, c5_value, n_points
