import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    if 'initial_h_values' in locals():
        h = initial_h_values.copy()
    else:
        block_size = n_points // 5
        h[:block_size] = 1.0
        h[block_size:2*block_size] = 0.0
        h[2*block_size:3*block_size] = 1.0
        h[3*block_size:4*block_size] = 0.0
        h[4*block_size:] = 1.0
        h = np.random.permutation(h)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)

    def objective(params):
        h = params.reshape(-1)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    def constraint(params):
        return np.sum(params) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = h.copy()
    initial_guess = initial_guess.reshape(-1)

    start_time = time.time()
    result = opt.minimize(
        fun=objective,
        x0=initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        tol=1e-6,
        options={'maxiter': 200, 'ftol': 1e-6, 'eps': 1e-8}
    )

    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (initial_guess, objective(initial_guess), n_points)

    h_opt = np.clip(result.x, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    h_opt = np.clip(h_opt, 0, 1)
    c5_opt = objective(h_opt)

    return (h_opt, c5_opt, n_points)
