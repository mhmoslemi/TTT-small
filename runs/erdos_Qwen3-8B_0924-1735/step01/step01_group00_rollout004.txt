def run(seed=42, budget_s=600, **kwargs):

    n_points = 120

    initial_h_values = np.zeros(n_points)
    initial_h_values[::3] = 1.0
    initial_h_values[1::4] = 0.5
    initial_h_values[2::5] = 0.75
    initial_h_values = initial_h_values.astype(float)

    initial_h_values *= n_points / 2
    h_values = initial_h_values.copy()
    
    def compute_c5(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    def evaluate_individual(h):
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.uniform(-0.1, 0.1)
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / np.sum(h)
        return h
    
    def crossover(parent1, parent2):
        alpha = np.random.uniform(-1, 1)
        child = 0.5 * (parent1 + parent2) + alpha * (parent1 - parent2)
        return child
    
    def select_parents(population, fitnesses):
        indices = np.argsort(fitnesses)
        best_indices = indices[:int(len(population) * 0.2)]
        return [population[i] for i in best_indices]
    
    def evolve_population(population, fitnesses, mutation_rate=0.1):
        parents = select_parents(population, fitnesses)
        offspring = []
        for i in range(len(parents) // 2):
            p1 = np.random.choice(parents)
            p2 = np.random.choice(parents)
            child = crossover(p1, p2)
            child = mutate(child, mutation_rate)
            offspring.append(child)

    population = [np.clip(h, 0, 1) for h in population]

    best_h = None

        fitnesses = [evaluate_individual(h) for h in population]
        current_best = np.min(fitnesses)

            best_h = population[np.argmin(fitnesses)]
