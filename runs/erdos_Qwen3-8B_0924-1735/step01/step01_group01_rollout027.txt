import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    if 'initial_h_values' in globals():
        initial_h_values = np.array(initial_h_values)
    else:
        for i in range(n_points):
            if i < n_points // 2:
                initial_h_values[i] = 1.0
            else:
                initial_h_values[i] = 0.0
        initial_h_values = np.random.permutation(initial_h_values)
        initial_h_values = np.clip(initial_h_values, 0, 1)
        initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        cross_corr = np.correlate(h, 1 - h, mode='full')
        return np.max(cross_corr) * dx

    def objective(params):
        return evaluate(params)

    def constraint(params):
        h = params.reshape(-1)
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = initial_h_values.copy()
    initial_guess = initial_guess.reshape(-1)

    class DE:
        def __init__(self, n_points):
            self.n = n_points
            self.pop_size = 20
            self.F = 0.8
            self.CR = 0.9
            self.max_iter = 300
            self.tol = 1e-6

        def run(self, initial_guess):
            pop = np.random.rand(self.pop_size, self.n) * 2 - 1
            pop = pop * (n_points / 2) / np.sum(pop, axis=1, keepdims=True)
            pop = np.clip(pop, 0, 1)
            pop = pop * (n_points / 2) / np.sum(pop, axis=1, keepdims=True)
            best = pop[0]
            best_fitness = evaluate(best)
            for _ in range(self.max_iter):
                for i in range(self.pop_size):
                    idxs = np.random.choice(self.pop_size, 3, replace=False)
                    mutant = pop[idxs[0]] + self.F * (pop[idxs[1]] - pop[idxs[2]])
                    mutant = np.clip(mutant, 0, 1)
                    mutant = mutant * (n_points / 2) / np.sum(mutant)
                    mutant = np.clip(mutant, 0, 1)
                    mutant = mutant * (n_points / 2) / np.sum(mutant)
                    trial = np.random.rand(self.n)
                    trial = mutant * trial + pop[i] * (1 - trial)
                    trial = np.clip(trial, 0, 1)
                    trial = trial * (n_points / 2) / np.sum(trial)
                    trial = np.clip(trial, 0, 1)
                    trial = trial * (n_points / 2) / np.sum(trial)
                    fit = evaluate(trial)
                    if fit < evaluate(pop[i]):
                        pop[i] = trial
                curr_best = pop[np.argmin([evaluate(h) for h in pop])]
                curr_fitness = evaluate(curr_best)
                if abs(curr_fitness - best_fitness) < self.tol:
                    break
                best_fitness = curr_fitness
                best = curr_best
            return best

    de_solver = DE(n_points)
    best_h = de_solver.run(initial_guess)

    def gradient_descent(h, max_steps=100, tol=1e-6):
        h = h.copy()
        for _ in range(max_steps):
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-6
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_minus = h.copy()
                h_minus[i] -= 1e-6
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                grad[i] = (evaluate(h_plus) - evaluate(h_minus)) / (2 * 1e-6)
            h = h - 0.01 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            if np.sqrt(np.sum(grad**2)) < tol:
                break
        return h

    best_h = gradient_descent(best_h)
    best_c5 = evaluate(best_h)

    elapsed_time = time.time() - time.time()
    if elapsed_time >= budget_s:
        return (initial_h_values, evaluate(initial_h_values), n_points)

    h_opt = best_h
    return (h_opt, best_c5, n_points)
