import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():
        h0 = initial_h_values.copy()
    else:
        h0 = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h0[i] = 1.0
            else:
                h0[i] = 0.0
        h0 = np.random.permutation(h0)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        h0 = np.clip(h0, 0, 1)

    def objective(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    def constraint(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    n_particles = 50
    max_iter = 100
    w = 0.8
    c1 = 1.5
    c2 = 1.5
    inertia = 0.9

    particles = np.random.uniform(low=0, high=1, size=(n_particles, n_points))
    particles = particles * (n_points / 2) / np.sum(particles, axis=1, keepdims=True)
    particles = np.clip(particles, 0, 1)

    velocities = np.random.uniform(low=-0.1, high=0.1, size=(n_particles, n_points))

    pbest_positions = particles.copy()
    pbest_values = np.array([objective(p) for p in particles])
    gbest_position = particles[np.argmin(pbest_values)]
    gbest_value = np.min(pbest_values)

    start_time = time.time()

    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        for j in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[j] = w * velocities[j] + c1 * r1 * (pbest_positions[j] - particles[j]) + c2 * r2 * (gbest_position - particles[j])
            particles[j] += velocities[j]
            particles[j] = np.clip(particles[j], 0, 1)
            particles[j] = particles[j] * (n_points / 2) / np.sum(particles[j])
            particles[j] = np.clip(particles[j], 0, 1)
            current_value = objective(particles[j])
            if current_value < pbest_values[j]:
                pbest_positions[j] = particles[j].copy()
                pbest_values[j] = current_value
        current_gbest = np.min(pbest_values)
        if current_gbest < gbest_value:
            gbest_position = pbest_positions[np.argmin(pbest_values)]
            gbest_value = current_gbest

    h_opt = gbest_position
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    h_opt = np.clip(h_opt, 0, 1)
    c5_opt = objective(h_opt)

    return (h_opt, c5_opt, n_points)
