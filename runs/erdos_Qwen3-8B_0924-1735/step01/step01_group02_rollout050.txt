import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    h = h / np.sum(h) * n_points / 2
    dx = 2.0 / n_points
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s
    max_iter = 1000
    temp = 300.0
    cooling_rate = 0.95
    best_candidates = [h.copy()]
    noise_scale = 0.25
    noise_decay = 0.9995
    top_candidates = 100

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    cons = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)}
    bounds = [(0, 1) for _ in range(n_points)]

    for _ in range(max_iter):
        if time.time() - time_start > time_remaining - 100:
            break
        h_candidate = best_h.copy()
        noise = np.random.normal(0, noise_scale, n_points)
        h_candidate += noise
        h_candidate = np.clip(h_candidate, 0, 1)
        sum_h_candidate = np.sum(h_candidate)
        if np.abs(sum_h_candidate - n_points / 2) > 1e-6:
            h_candidate = h_candidate * (n_points / 2) / sum_h_candidate
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
        if np.random.rand() < np.exp(-(c5_candidate - best_c5) / temp):
            best_h = h_candidate.copy()
        temp *= cooling_rate
        noise_scale *= noise_decay

    def surrogate_objective(h):
        if np.sum(h) < n_points / 2 - 1e-6 or np.sum(h) > n_points / 2 + 1e-6:
            return np.inf
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.log(np.sum(np.exp(100 * (corr - 1e-6))))

    for h_candidate in best_candidates:
        if time.time() - time_start > time_remaining - 150:
            break
        result = opt.minimize(
            surrogate_objective,
            h_candidate,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        h_refined = result.x
        c5_refined = compute_c5(h_refined)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()

    for _ in range(5):
        if time.time() - time_start > time_remaining - 100:
            break
        noise = np.random.uniform(-0.05, 0.05, n_points)
        h_candidate = best_h + noise
        h_candidate = np.clip(h_candidate, 0, 1)
        sum_h_candidate = np.sum(h_candidate)
        if np.abs(sum_h_candidate - n_points / 2) > 1e-6:
            h_candidate = h_candidate * (n_points / 2) / sum_h_candidate
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()

    for _ in range(5):
        if time.time() - time_start > time_remaining - 100:
            break
        noise = np.random.normal(0, 0.05, n_points)
        h_candidate = best_h + noise
        h_candidate = np.clip(h_candidate, 0, 1)
        sum_h_candidate = np.sum(h_candidate)
        if np.abs(sum_h_candidate - n_points / 2) > 1e-6:
            h_candidate = h_candidate * (n_points / 2) / sum_h_candidate
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()

    best_h = np.clip(best_h, 0, 1)
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
