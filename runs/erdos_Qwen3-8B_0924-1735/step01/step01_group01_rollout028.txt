import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points

    # Use provided initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        # Default structured initialization: alternating blocks of 1 and 0
        m_blocks = 10
        block_size = 2.0 / m_blocks
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            if int(i / block_size) % 2 == 0:
                initial_h[i] = 1.0
        # Normalize to ensure ∑h = n_points / 2
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        initial_h = np.clip(initial_h, 0, 1)

    def objective(params):
        h = params.reshape(-1)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        # Normalize to ensure ∑h = n_points / 2
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    # Particle Swarm Optimization setup
    P = 50  # Population size
    omega = 0.8
    c1 = 1.5
    c2 = 1.5
    max_iter = 100

    # Initialize particles
    particles = np.random.rand(P, n_points)
    particles = initial_h.copy() + (np.random.rand(P, n_points) - 0.5) * 0.2
    particles = np.clip(particles, 0, 1)
    particles = particles * (n_points / 2) / np.sum(particles, axis=1, keepdims=True)
    particles = np.clip(particles, 0, 1)

    # Personal best and global best
    pbest = particles.copy()
    pbest_vals = np.array([objective(p) for p in particles])
    gbest = particles[np.argmin(pbest_vals)]
    gbest_val = np.min(pbest_vals)

    start_time = time.time()
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Update velocities and positions
        r1 = np.random.rand(P, n_points)
        r2 = np.random.rand(P, n_points)
        vel = omega * np.random.rand(P, n_points) + c1 * r1 * (pbest - particles) + c2 * r2 * (gbest[np.newaxis, :] - particles)
        particles += vel
        # Clip and normalize
        particles = np.clip(particles, 0, 1)
        particles = particles * (n_points / 2) / np.sum(particles, axis=1, keepdims=True)
        particles = np.clip(particles, 0, 1)
        # Evaluate and update pbest and gbest
        current_vals = np.array([objective(p) for p in particles])
        improve = current_vals < pbest_vals
        pbest[improve] = particles[improve]
        pbest_vals[improve] = current_vals[improve]
        gbest = particles[np.argmin(pbest_vals)]
        gbest_val = np.min(pbest_vals)

    # Final evaluation
    best_h = gbest
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = objective(best_h)

    return (best_h, best_c5, n_points)
