import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.where(np.random.rand(n_points) < 0.5, 1.0, 0.0)
        h = h * (n_points / 2) / np.sum(h)
    
    h = h * (1 + 0.1 * np.cos(2 * np.pi * np.arange(n_points) / n_points))
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    best_h = h
    best_c5 = np.inf
    dx = 2.0 / n_points
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)
    
    def project(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.full_like(h, 0.5)
        return h_clipped * (n_points / 2) / total
    
    def objective(h):
        return compute_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)}
    
    time_start = time.time()
    time_remaining = budget_s
    
    def l_bfgs_refine(h):
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'ftol': 1e-9, 'maxiter': 100})
        return result.x
    
    def random_perturb(h):
        noise = np.random.uniform(-0.05, 0.05, size=n_points)
        h = h + noise
        h = project(h)
        return h
    
    for _ in range(20):
        if time.time() - time_start > time_remaining - 100:
            break
        h = random_perturb(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    if time.time() - time_start < time_remaining - 100:
        h = l_bfgs_refine(best_h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    h = project(best_h)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h
    
    return best_h, best_c5, n_points
