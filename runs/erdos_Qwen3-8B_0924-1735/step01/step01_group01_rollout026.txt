import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 240
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i < 120:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.0
    initial_h_values = np.random.permutation(initial_h_values)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    total = np.sum(initial_h_values)
    initial_h_values = initial_h_values * (n_points / 2) / total
    initial_h_values = np.clip(initial_h_values, 0, 1)

    def objective(h_values):
        h_values = np.clip(h_values, 0, 1)
        total = np.sum(h_values)
        h_values = h_values * (n_points / 2) / total
        h_values = np.clip(h_values, 0, 1)
        cross_corr = np.correlate(h_values, 1 - h_values, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def stage1():
        from scipy.optimize import differential_evolution
        def objective_de(h):
            return objective(h)
        result = differential_evolution(
            objective_de,
            bounds,
            strategy='best1bin',
            popsize=20,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-6,
            maxiter=300,
            polish=True,
            disp=False,
            vectorized=True
        )
        return result.x

    start_time = time.time()
    h = stage1()
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (initial_h_values, objective(initial_h_values), n_points)

    def stage2(h):
        def grad_func(h):
            eps = 1e-6
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                total = np.sum(h_plus)
                h_plus = h_plus * (n_points / 2) / total
                h_plus = np.clip(h_plus, 0, 1)
                c5_plus = objective(h_plus)
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                total = np.sum(h_minus)
                h_minus = h_minus * (n_points / 2) / total
                h_minus = np.clip(h_minus, 0, 1)
                c5_minus = objective(h_minus)
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            return grad

        def objective_grad(h):
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            h = h * (n_points / 2) / total
            h = np.clip(h, 0, 1)
            cross_corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(cross_corr) * dx
            return c5

        result = opt.minimize(
            objective_grad,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            jac=grad_func,
            tol=1e-6,
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-5}
        )
        return result.x

    h = stage2(h)
    return (h, objective(h), n_points)
