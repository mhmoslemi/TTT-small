import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h = initial_h_values
    if len(initial_h) != n_points:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.full_like(h, 0.5)
        return h_clipped * (n_points / 2) / total

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)}

    # Bayesian Optimization with UCB acquisition
    def ucb_acquisition(X, Y, X_test, beta=1.2):
        from sklearn.gaussian_process import GaussianProcessRegressor
        from sklearn.gaussian_process.kernels import RBF
        kernel = RBF(length_scale=1.0)
        gpr = GaussianProcessRegressor(kernel=kernel, alpha=1e-10, n_restarts_optimizer=10)
        gpr.fit(X, Y)
        mu, sigma = gpr.predict(X_test, return_std=True)
        return mu - beta * sigma

    def random_init(num_samples=20):
        samples = []
        for _ in range(num_samples):
            h = np.random.rand(n_points)
            h = project(h)
            samples.append(h)
        return np.array(samples)

    def bayesian_optimization(max_iter=200):
        nonlocal best_h, best_c5
        X = random_init(20)
        Y = np.array([compute_c5(x) for x in X])
        X = X.reshape(-1, n_points)
        best_idx = np.argmin(Y)
        best_h = X[best_idx].copy()
        best_c5 = Y[best_idx]
        for _ in range(max_iter):
            if time.time() - time_start > time_remaining - 100:
                break
            X_test = np.random.rand(10, n_points)
            X_test = np.vstack([X, X_test])
            X_test = project(X_test)
            Y_test = np.array([compute_c5(x) for x in X_test])
            Y = np.concatenate([Y, Y_test])
            X = X_test
            acq = ucb_acquisition(X, Y, X_test)
            best_acq_idx = np.argmin(acq)
            best_h = X[best_acq_idx].copy()
            best_c5 = Y[best_acq_idx]
            if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
                best_h = project(best_h)

    bayesian_optimization(200)

    # Local refinement
    def local_refinement(candidates, max_iter=100):
        nonlocal best_h, best_c5
        for h in candidates:
            if time.time() - time_start > time_remaining - 100:
                break
            h = project(h)
            result = opt.minimize(
                objective,
                h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': max_iter}
            )
            c5 = compute_c5(result.x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = result.x.copy()

    local_refinement([best_h], max_iter=100)

    # Final refinement
    if time.time() - time_start < time_remaining - 100:
        h = best_h.copy()
        noise = np.random.uniform(-0.05, 0.05, size=n_points)
        h = project(h + noise)
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()

    best_h = project(best_h)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_points
