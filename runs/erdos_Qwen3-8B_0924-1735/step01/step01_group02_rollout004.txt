import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60  # Slightly larger than initial for better resolution
    initial_h = initial_h_values
    if len(initial_h) != n_points:
        # Construct initial h as 1s at spaced positions with some density
        h = np.zeros(n_points)
        spacing = n_points // 4
        h[::spacing] = 1.0
        h[spacing::2*spacing] = 0.5
        h[::3*spacing] = 1.0
        initial_h = h / np.sum(h) * n_points / 2
    initial_h = np.clip(initial_h, 0, 1)
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    bounds = [(0, 1) for _ in range(n_points)]

    def projection(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return 0.5 * np.ones_like(h)
        return h * (n_points / 2) / np.sum(h)

    def perturb(h, sigma):
        return projection(h + np.random.normal(0, sigma, size=n_points))

    # Simulated Annealing Setup
    T = 300.0
    cooling_rate = 0.995
    max_iterations = 1000
    sigma_start = 0.25
    sigma_end = 0.05
    best_h = projection(h)
    best_c5 = compute_c5(best_h)
    top_candidates = [best_h.copy()]

    for t in range(max_iterations):
        if time.time() - time_start > time_remaining - 100:
            break
        sigma = sigma_start * (sigma_end / sigma_start) ** (t / max_iterations)
        h_candidate = perturb(best_h, sigma)
        c5 = compute_c5(h_candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()
            top_candidates.append(h_candidate.copy())
        # Simple acceptance based on improvement
        if c5 < best_c5:
            best_h = h_candidate.copy()
        # Cool down
        T *= cooling_rate

    # Select top candidates for local refinement
    top_candidates = np.array(top_candidates)
    top_candidates = [projection(h) for h in top_candidates]
    top_candidates = np.array([h for h in top_candidates if np.abs(np.sum(h) - n_points / 2) < 1e-6])

    # Refine top candidates
    for h_candidate in top_candidates[:100]:
        if time.time() - time_start > time_remaining - 100:
            break
        def surrogate_objective(x):
            h = projection(x)
            corr = np.correlate(h, 1 - h, mode="full") * dx
            return np.log(np.sum(np.exp(100 * (corr - 1e-6))))
        result = opt.minimize(
            surrogate_objective,
            h_candidate,
            method='L-BFGS-B',
            bounds=bounds,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        refined_h = result.x
        c5 = compute_c5(refined_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = refined_h.copy()

    # Final check and return
    best_h = projection(best_h)
    best_h = np.clip(best_h, 0, 1)
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    c5_value = compute_c5(best_h)
    return best_h, c5_value, n_points
