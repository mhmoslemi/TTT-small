import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 120
    initial_h = initial_h_values
    if len(initial_h) != n_points:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    dx = 2.0 / n_points
    best_h = initial_h.copy()
    best_c5 = np.inf
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]

    # Simulated Annealing
    def simulated_annealing():
        nonlocal best_h, best_c5
        T = 300
        max_iter = 1000
        for t in range(max_iter):
            if time.time() - time_start > time_remaining - 100:
                break
            noise = np.random.normal(0, 0.15, size=n_points)
            h = best_h + noise
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if abs(sum_h - n_points / 2) > 1e-6:
                h = h * (n_points / 2) / sum_h
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            delta = c5 - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                best_h = h.copy()
            T *= 0.99

    # Local Refinement
    def local_refinement():
        nonlocal best_h, best_c5
        for _ in range(100):
            if time.time() - time_start > time_remaining - 100:
                break
            result = opt.minimize(
                objective,
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)},
                options={'ftol': 1e-9, 'maxiter': 100}
            )
            h = result.x
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if abs(sum_h - n_points / 2) > 1e-6:
                h = h * (n_points / 2) / sum_h
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Random restarts
    for _ in range(5):
        if time.time() - time_start > time_remaining - 100:
            break
        noise = np.random.uniform(-0.2, 0.2, size=n_points)
        h = best_h + noise
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if abs(sum_h - n_points / 2) > 1e-6:
            h = h * (n_points / 2) / sum_h
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Run the optimization
    simulated_annealing()
    local_refinement()

    # Final checks and return
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        best_h = best_h * (n_points / 2) / sum_h
    c5_value = compute_c5(best_h)
    return best_h, c5_value, n_points
