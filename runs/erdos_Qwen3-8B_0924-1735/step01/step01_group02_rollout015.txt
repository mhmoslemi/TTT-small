import numpy as np
import scipy.optimize as opt
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.full_like(h, 0.5)
        return h_clipped * (n_points / 2) / total

    def evaluate(h):
        return compute_c5(project(h))

    def surrogate(h):
        return np.log(np.sum(np.exp(100 * (evaluate(h) - 1e-6))))

    def objective(h):
        return evaluate(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)}

    # Bayesian optimization setup
    kernel = ConstantKernel(1.0, (1e-3, 1e3)) * RBF(1.0, (0.1, 10.0))
    gp_model = GaussianProcessRegressor(kernel=kernel, alpha=1e-5, n_restarts_optimizer=10)
    X = np.random.rand(20, n_points)
    y = np.array([evaluate(project(x)) for x in X])
    gp_model.fit(X, y)

    evaluations = []
    for i in range(200):
        if time.time() - time_start > time_remaining - 100:
            break
        # UCB acquisition function
        mu, sigma = gp_model.predict(X, return_std=True)
        ucb = mu + 1.2 * sigma
        best_idx = np.argmax(ucb)
        best_x = X[best_idx].copy()
        evaluations.append((best_x, evaluate(project(best_x))))
        # Add new point
        new_x = np.random.rand(n_points)
        new_y = evaluate(project(new_x))
        X = np.vstack((X, new_x))
        y = np.append(y, new_y)
        gp_model.fit(X, y)

    # Refine top candidates
    top_candidates = sorted(evaluations, key=lambda x: x[1])[:10]
    for i in range(len(top_candidates)):
        if time.time() - time_start > time_remaining - 100:
            break
        x, _ = top_candidates[i]
        result = opt.minimize(
            objective,
            x,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        h_candidate = project(result.x)
        c5_value = evaluate(h_candidate)
        if c5_value < best_c5:
            best_c5 = c5_value
            best_h = h_candidate.copy()

    # Final check
    best_h = project(best_h)
    c5_value = evaluate(best_h)
    return best_h, c5_value, n_points
