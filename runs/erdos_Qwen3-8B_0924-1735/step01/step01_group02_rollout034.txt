import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)}

    def run_simulated_annealing():
        nonlocal best_c5, best_h
        T = 300.0
        t = 0
        max_iter = 1000
        best_h = h.copy()
        best_c5 = compute_c5(best_h)
        for _ in range(max_iter):
            if time.time() - time_start > time_remaining - 100:
                break
            t += 1
            std_dev = max(0.05, 0.25 * np.exp(-t / 100))
            noise = np.random.normal(0, std_dev, size=n_points)
            h_candidate = h + noise
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
            delta = c5_candidate - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h = h_candidate.copy()
            T *= 0.99

    def run_refinement():
        nonlocal best_c5, best_h
        for _ in range(100):
            if time.time() - time_start > time_remaining - 100:
                break
            result = opt.minimize(
                objective,
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 100}
            )
            h_candidate = result.x
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()

    run_simulated_annealing()
    run_refinement()

    h = best_h
    h = np.clip(h, 0, 1)
    if np.abs(np.sum(h) - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / np.sum(h)
    c5_value = compute_c5(h)
    if c5_value < best_c5:
        best_c5 = c5_value
        best_h = h.copy()

    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    c5_value = compute_c5(best_h)
    return best_h, c5_value, n_points
