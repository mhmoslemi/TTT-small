import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 50
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.full_like(h, 0.5)
        return h_clipped * (n_points / 2) / total

    def perturb(h, sigma):
        return np.clip(h + np.random.normal(0, sigma, size=n_points), 0, 1)

    def optimize(h, sigma=0.25, max_iter=100):
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum, 'jac': lambda x: np.ones_like(x)},
            options={'ftol': 1e-9, 'maxiter': max_iter}
        )
        return result.x

    def random_candidates(n=5):
        candidates = []
        for _ in range(n):
            h = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    h[i] = 1
            h = project(h)
            candidates.append(h)
        return candidates

    def evaluate_candidates(candidates):
        results = []
        for h in candidates:
            results.append((compute_c5(h), h))
        results.sort()
        return results

    if time.time() - time_start > time_remaining - 100:
        pass
    else:
        candidates = random_candidates(10)
        candidates = [optimize(h) for h in candidates]
        candidates = [project(h) for h in candidates]
        candidates = evaluate_candidates(candidates)
        best_h = candidates[0][1]
        best_c5 = candidates[0][0]

    for _ in range(5):
        if time.time() - time_start > time_remaining - 100:
            break
        h = perturb(best_h, 0.1)
        h = project(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    h = project(best_h)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h

    h = project(best_h)
    if np.abs(np.sum(h) - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / np.sum(h)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h

    h = np.clip(best_h, 0, 1)
    return h, best_c5, n_points
