import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h_values = initial_h_values.copy() if 'initial_h_values' in locals() else np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h_values[i] = 1.0
        else:
            h_values[i] = 0.0
    h_values = np.random.permutation(h_values)
    h_values /= np.sum(h_values)
    h_values *= n_points / 2
    h_values = np.clip(h_values, 0, 1)

    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        cross_corr = np.correlate(x, 1 - x, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = h_values.copy()
    initial_guess = initial_guess.reshape(-1)
    start_time = time.time()

    result = opt.minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        tol=1e-6,
        options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-8}
    )

    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (h_values, objective(h_values), n_points)

    h_opt = np.clip(result.x, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    h_opt = np.clip(h_opt, 0, 1)
    c5_opt = objective(h_opt)

    return (h_opt, c5_opt, n_points)
