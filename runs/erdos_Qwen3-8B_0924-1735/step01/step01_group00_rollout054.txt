import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 120
    dx = 2.0 / n_points

    # Use the provided initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h /= np.sum(initial_h) * dx
        initial_h *= n_points / 2

    # Rescale to meet the integral constraint
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    h_values = initial_h.copy()

    def compute_c5(h):
        h = np.array(h, dtype=float)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr

    def objective(x):
        return compute_c5(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    initial_guess = h_values.copy()

    # Optimize with more iterations and tighter tolerances
    result = opt.minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 150, 'ftol': 1e-9, 'eps': 1e-9, 'disp': False}
    )

    h_values = result.x
    h_values = np.clip(h_values, 0, 1)
    h_values = h_values * (n_points / 2) / np.sum(h_values)
    h_values = np.clip(h_values, 0, 1)

    c5_bound = compute_c5(h_values)
    return h_values, c5_bound, n_points
