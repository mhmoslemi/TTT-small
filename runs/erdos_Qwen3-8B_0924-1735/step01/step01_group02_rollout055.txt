import numpy as np
import scipy.optimize as opt
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h = initial_h_values
    if len(initial_h) != n_points:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.full_like(h, 0.5)
        return h_clipped * (n_points / 2) / total

    def surrogate_objective(h):
        h = project(h)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def objective(h):
        h = project(h)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    bounds = [(0, 1) for _ in range(n_points)]

    def generate_initial_points(n):
        X = np.random.uniform(0, 1, (n, n_points))
        X = np.clip(X, 0, 1)
        X = X * (n_points / 2) / np.sum(X, axis=1)[:, None]
        return X

    X_initial = generate_initial_points(20)
    Y_initial = np.array([compute_c5(x) for x in X_initial])
    X_initial = np.array(X_initial)

    kernel = RBF(length_scale=1.0, length_scale_bounds=(1e-5, 1e5)) * C(1.0, (1e-5, 1e5))
    gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=10, alpha=1e-10)
    gp.fit(X_initial, Y_initial)

    X_candidate = X_initial
    Y_candidate = Y_initial
    X_evaluated = X_initial
    Y_evaluated = Y_initial
    X_remaining = X_candidate
    Y_remaining = Y_candidate
    time_used = time.time() - time_start
    time_left = budget_s - time_used

    for i in range(200):
        if time_used > time_remaining - 50:
            break
        if len(X_remaining) == 0:
            break
        if i < 10:
            X_next = generate_initial_points(10)
        else:
            mu, sigma = gp.predict(X_remaining, return_std=True)
            acq = mu - 1.2 * sigma
            idx = np.argmax(acq)
            X_next = [X_remaining[idx]]
        X_next = np.array(X_next)
        X_next = np.clip(X_next, 0, 1)
        X_next = X_next * (n_points / 2) / np.sum(X_next, axis=1)[:, None]
        Y_next = np.array([compute_c5(x) for x in X_next])
        X_candidate = np.vstack((X_candidate, X_next))
        Y_candidate = np.concatenate((Y_candidate, Y_next))
        gp.fit(X_candidate, Y_candidate)
        X_remaining = X_candidate
        Y_remaining = Y_candidate
        time_used = time.time() - time_start

    X_best = X_candidate
    Y_best = Y_candidate
    indices = np.argsort(Y_best)
    X_top = X_candidate[indices[:10]]
    Y_top = Y_candidate[indices[:10]]

    for i in range(len(X_top)):
        if time_used > time_remaining - 100:
            break
        result = opt.minimize(
            objective,
            X_top[i],
            method='L-BFGS-B',
            bounds=bounds,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        h_refined = result.x
        h_refined = project(h_refined)
        c5_value = compute_c5(h_refined)
        if c5_value < best_c5:
            best_c5 = c5_value
            best_h = h_refined.copy()

    for _ in range(5):
        if time_used > time_remaining - 100:
            break
        noise = np.random.normal(0, 0.05, size=n_points)
        h = best_h + noise
        h = project(h)
        c5_value = compute_c5(h)
        if c5_value < best_c5:
            best_c5 = c5_value
            best_h = h.copy()

    h = best_h
    h = project(h)
    c5_value = compute_c5(h)
    if np.abs(np.sum(h) - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / np.sum(h)
    c5_value = compute_c5(h)
    return h, c5_value, n_points
