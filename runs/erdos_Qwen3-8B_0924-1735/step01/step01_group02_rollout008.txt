import numpy as np
import scipy.optimize as opt
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h = initial_h_values
    if len(initial_h) != n_points:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0

    # Cosine-based initialization with perturbation
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    initial_h = initial_h * (1 + 0.1 * np.cos(2 * np.pi * np.arange(n_points) / n_points))

    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.full_like(h, 0.5)
        return h_clipped * (n_points / 2) / total

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)}

    # Bayesian optimization parameters
    n_evaluations = 200
    n_refine = 10
    kernel = C(1.0, (1e-2, 1e2)) * RBF(1.0, (1e-2, 1e2))
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-5, n_restarts_optimizer=10)
    X = np.array([np.arange(n_points) / n_points]).T
    Y = np.array([compute_c5(project(initial_h))])

    def acquisition(X, gp, X_candidate):
        mean, std = gp.predict(X_candidate, return_std=True)
        return mean + 1.2 * std

    X_candidate = np.random.uniform(0, 1, (n_evaluations, n_points))
    X_candidate = project(X_candidate)
    Y_candidate = np.array([compute_c5(x) for x in X_candidate])
    X = np.vstack([X, X_candidate])
    Y = np.vstack([Y, Y_candidate])

    for i in range(n_evaluations - len(X_candidate)):
        if time.time() - time_start > time_remaining - 100:
            break
        if len(X) > n_evaluations:
            break
        if len(X) < n_evaluations:
            new_x = np.random.uniform(0, 1, n_points)
            new_x = project(new_x)
            X = np.vstack([X, new_x])
            Y = np.vstack([Y, compute_c5(new_x)])
            gp.fit(X, Y.flatten())

    X_best = X[np.argmin(Y), :]
    best_h = X_best.copy()
    best_c5 = np.min(Y)

    for _ in range(n_refine):
        if time.time() - time_start > time_remaining - 100:
            break
        result = opt.minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        h = result.x
        c5_value = compute_c5(h)
        if c5_value < best_c5:
            best_c5 = c5_value
            best_h = h.copy()

    h = best_h
    h = project(h)
    c5_value = compute_c5(h)
    if c5_value < best_c5:
        best_c5 = c5_value
        best_h = h.copy()

    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    c5_value = compute_c5(best_h)

    return best_h, c5_value, n_points
