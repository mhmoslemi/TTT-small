import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h = initial_h_values
    if len(initial_h) != n_points:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    h = initial_h * (n_points / 2) / np.sum(initial_h)
    dx = 2.0 / n_points
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.full_like(h, 0.5)
        return h_clipped * (n_points / 2) / total

    def objective(h):
        return compute_c5(h)

    def surrogate_objective(h):
        h_proj = project(h)
        return compute_c5(h_proj)

    def grad_objective(h):
        h_proj = project(h)
        corr = np.correlate(h_proj, 1 - h_proj, mode="full")
        grad = np.zeros_like(h)
        for i in range(n_points):
            for k in range(2 * n_points - 1):
                if k < n_points:
                    grad[i] += corr[k] * (1 - h_proj[i]) * (1 if k == i else 0)
                else:
                    j = k - n_points
                    grad[i] += corr[k] * h_proj[j] * (1 if j == i else 0)
        return grad

    bounds = Bounds(0, 1)
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2, 'jac': lambda h: np.ones_like(h)}

    for _ in range(100):
        if time.time() - time_start > budget_s - 100:
            break
        noise = np.random.uniform(-0.1, 0.1, size=n_points)
        h = best_h + noise
        h = project(h)
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()

    result = opt.minimize(
        surrogate_objective,
        best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        jac=grad_objective,
        options={'ftol': 1e-9, 'maxiter': 200}
    )

    if result.success:
        best_h = result.x
        best_c5 = compute_c5(best_h)

    for _ in range(20):
        if time.time() - time_start > budget_s - 100:
            break
        noise = np.random.uniform(-0.05, 0.05, size=n_points)
        h = best_h + noise
        h = project(h)
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()

    best_h = project(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
