import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h_initial = np.zeros(n_points)
    block_count = 10
    block_width = 20
    for i in range(block_count):
        start = i * block_width
        end = start + block_width
        if i % 2 == 0:
            h_initial[start:end] = 1.0
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    h_initial = np.clip(h_initial, 0, 1)

    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = h_initial.copy()

    def de_objective(x):
        return objective(x)

    def de_constraint(x):
        return constraint(x)

    start_time = time.time()
    result = opt.differential_evolution(
        de_objective,
        bounds,
        constraints=[de_constraint],
        popsize=20,
        mutation=(0.8, 1.0),
        recombination=0.9,
        tol=1e-6,
        maxiter=100,
        polish=True,
        disp=False
    )

    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (initial_guess, objective(initial_guess), n_points)

    h_opt = result.x
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    h_opt = np.clip(h_opt, 0, 1)
    c5_opt = objective(h_opt)

    return (h_opt, c5_opt, n_points)
