import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i < 75:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.0
    initial_h_values = np.random.permutation(initial_h_values)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values /= np.sum(initial_h_values)
    initial_h_values *= n_points / 2
    initial_h_values = np.clip(initial_h_values, 0, 1)

    def objective(params):
        h = params.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective_de(params):
        h = params.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    def evaluate_de_pop(population):
        return [objective_de(p) for p in population]

    def mutate(parent, bounds, F=0.8):
        idxs = np.random.choice(len(bounds), 3, replace=False)
        mutant = parent[idxs] * F + np.random.rand(3) * (bounds[idxs[0]][1] - bounds[idxs[0]][0])
        mutant = np.clip(mutant, bounds[idxs[0]][0], bounds[idxs[0]][1])
        return mutant

    def crossover(parent, mutant, CR=0.9):
        child = np.copy(parent)
        idx = np.random.rand(len(child)) < CR
        child[idx] = mutant[idx]
        return child

    def de_run(population_size=20, generations=300):
        population = np.random.rand(population_size, n_points)
        population = np.clip(population, 0, 1)
        population = population * (n_points / 2) / np.sum(population, axis=1, keepdims=True)
        population = np.clip(population, 0, 1)

        best_fitness = float('inf')
        best_solution = population[0]
        for gen in range(generations):
            if time.time() - start_time > budget_s:
                return best_solution
            fitness = evaluate_de_pop(population)
            for i in range(population_size):
                parent = population[i]
                mutant = mutate(parent, bounds)
                child = crossover(parent, mutant)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                child = np.clip(child, 0, 1)
                child_fitness = objective_de(child)
                if child_fitness < fitness[i]:
                    population[i] = child
                    if child_fitness < best_fitness:
                        best_fitness = child_fitness
                        best_solution = child
            if np.abs(best_fitness - float('inf')) < 1e-6:
                break
        return best_solution

    start_time = time.time()
    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)
    if 'initial_h_values' in locals():
        best_h = initial_h_values
        best_c5 = objective(best_h)

    population_size = 20
    generations = 300
    h_de = de_run(population_size, generations)
    h_de = np.clip(h_de, 0, 1)
    h_de = h_de * (n_points / 2) / np.sum(h_de)
    h_de = np.clip(h_de, 0, 1)
    c5_de = objective(h_de)

    if c5_de < best_c5:
        best_h = h_de
        best_c5 = c5_de

    if time.time() - start_time < budget_s:
        def grad_descent(h, max_iter=100):
            h = h.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            for _ in range(max_iter):
                h_orig = h.copy()
                h = h * (n_points / 2) / np.sum(h)
                h = np.clip(h, 0, 1)
                cross_corr = np.correlate(h, 1 - h, mode='full')
                c5 = np.max(cross_corr) * dx
                grad = np.zeros(n_points)
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += 1e-5
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    h_plus = np.clip(h_plus, 0, 1)
                    cross_corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(cross_corr_plus) * dx
                    grad[i] = (c5_plus - c5) / 1e-5
                grad = grad * (n_points / 2) / np.sum(h)
                h = h - 0.1 * grad
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                h = np.clip(h, 0, 1)
                if np.sum(np.abs(h - h_orig)) < 1e-6:
                    break
            return h

        h_grad = grad_descent(best_h)
        h_grad = np.clip(h_grad, 0, 1)
        h_grad = h_grad * (n_points / 2) / np.sum(h_grad)
        h_grad = np.clip(h_grad, 0, 1)
        c5_grad = objective(h_grad)
        if c5_grad < best_c5:
            best_h = h_grad
            best_c5 = c5_grad

    return (best_h, best_c5, n_points)
