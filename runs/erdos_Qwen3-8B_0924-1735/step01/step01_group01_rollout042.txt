import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 120
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i < 60:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.0
    initial_h_values = np.random.permutation(initial_h_values)
    initial_h_values /= np.sum(initial_h_values)
    initial_h_values *= n_points / 2
    initial_h_values = np.clip(initial_h_values, 0, 1)

    def evaluate_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    def normalize_h(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    class Particle:
        def __init__(self, position):
            self.position = position
            self.velocity = np.random.uniform(-0.1, 0.1, size=position.shape)
            self.best_position = position.copy()
            self.best_value = evaluate_h(position)

    class Swarm:
        def __init__(self, initial_h, n_particles=50):
            self.particles = [Particle(initial_h + np.random.uniform(-0.1, 0.1, size=initial_h.shape)) for _ in range(n_particles)]
            self.gbest = np.min([p.best_value for p in self.particles])
            self.gbest_position = np.array([p.best_position for p in self.particles])[np.argmin([p.best_value for p in self.particles])]

        def update(self, omega=0.8, c1=1.5, c2=1.5, max_iter=100, time_budget=600):
            start_time = time.time()
            for _ in range(max_iter):
                if time.time() - start_time > time_budget:
                    break
                for p in self.particles:
                    r1, r2 = np.random.rand(), np.random.rand()
                    p.velocity = omega * p.velocity + c1 * r1 * (p.best_position - p.position) + c2 * r2 * (self.gbest_position - p.position)
                    p.position += p.velocity
                    p.position = normalize_h(p.position)
                    current_value = evaluate_h(p.position)
                    if current_value < p.best_value:
                        p.best_value = current_value
                        p.best_position = p.position.copy()
                self.gbest = np.min([p.best_value for p in self.particles])
                self.gbest_position = np.array([p.best_position for p in self.particles])[np.argmin([p.best_value for p in self.particles])]
            return self.gbest_position, self.gbest

    best_h = initial_h_values.copy()
    best_c5 = evaluate_h(best_h)
    elapsed_time = 0

    if 'initial_h_values' in locals():
        initial_guess = normalize_h(initial_h_values)
        swarm = Swarm(initial_guess)
        best_h, best_c5 = swarm.update(time_budget=budget_s)
        best_c5 = evaluate_h(best_h)

    return (best_h, best_c5, n_points)
