import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.zeros(n_points)
        block_size = n_points // 5
        for i in range(5):
            start = i * block_size
            end = (i + 1) * block_size
            initial_h[start:end] = 1.0
        initial_h = np.random.permutation(initial_h)
        initial_h /= np.sum(initial_h)
        initial_h *= n_points / 2
        initial_h = np.clip(initial_h, 0, 1)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    def de_objective(params):
        return objective(params)
    
    def de_constraint(params):
        return constraint(params)
    
    def gradient_descent(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5
    
    def grad_descent_update(params):
        h = params.copy()
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-6
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            h_plus = h_plus * (n_points / 2) / h_plus_sum
            cross_corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(cross_corr_plus) * dx
            grad[i] = (c5_plus - c5) / 1e-6
        
        h_new = h - 0.01 * grad
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        h_new = h_new * (n_points / 2) / h_new_sum
        h_new = np.clip(h_new, 0, 1)
        return h_new
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5
    
    start_time = time.time()
    
    # Differential Evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    pop = np.array([np.random.uniform(0, 1, n_points) for _ in range(20)])
    pop = np.clip(pop, 0, 1)
    for i in range(20):
        pop[i] = np.clip(pop[i] * (n_points / 2) / np.sum(pop[i]), 0, 1)
    
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        for j in range(20):
            a, b, c = np.random.choice(20, 3, replace=False)
            mutant = pop[a] + 0.8 * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0, 1)
            mutant_sum = np.sum(mutant)
            mutant = mutant * (n_points / 2) / mutant_sum
            mutant = np.clip(mutant, 0, 1)
            
            cross_over = np.random.rand(n_points) < 0.9
            trial = np.where(cross_over, mutant, pop[j])
            trial_sum = np.sum(trial)
            trial = trial * (n_points / 2) / trial_sum
            trial = np.clip(trial, 0, 1)
            
            c5_pop = evaluate(pop[j])
            c5_trial = evaluate(trial)
            if c5_trial < c5_pop:
                pop[j] = trial
    
    best_h = pop[np.argmin([evaluate(h) for h in pop])]
    best_c5 = evaluate(best_h)
    
    # Gradient descent refinement
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        best_h = grad_descent_update(best_h)
        best_c5 = evaluate(best_h)
    
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    best_h = best_h * (n_points / 2) / best_h_sum
    best_h = np.clip(best_h, 0, 1)
    final_c5 = evaluate(best_h)
    
    return (best_h, final_c5, n_points)
