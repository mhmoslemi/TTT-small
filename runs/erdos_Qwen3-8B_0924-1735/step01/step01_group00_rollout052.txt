import numpy as np
import scipy.optimize as opt
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 120
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    initial_h_values[::3] = 1.0
    initial_h_values[1::4] = 0.5
    initial_h_values[2::5] = 0.75
    initial_h_values = initial_h_values.astype(float)
    initial_h_values /= np.sum(initial_h_values) * dx
    initial_h_values *= n_points / 2
    h_values = initial_h_values.copy()

    def compute_c5(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr

    def evaluate_individual(h):
        return compute_c5(h)

    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.uniform(-0.1, 0.1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        return h

    def crossover(parent1, parent2):
        alpha = np.random.uniform(-1, 1)
        child = 0.5 * (parent1 + parent2) + alpha * (parent1 - parent2)
        child = np.clip(child, 0, 1)
        child = child * (n_points / 2) / np.sum(child)
        child = np.clip(child, 0, 1)
        return child

    population_size = 30
    population = np.random.rand(population_size, n_points)
    population = population * (n_points / 2) / np.sum(population, axis=1, keepdims=True)
    population = np.clip(population, 0, 1)
    best_solution = population[0].copy()
    best_c5 = evaluate_individual(best_solution)
    start_time = time.time()

    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([evaluate_individual(ind) for ind in population])
        best_idx = np.argmin(fitness)
        best_solution = population[best_idx].copy()
        best_c5 = fitness[best_idx]

        if np.random.rand() < 0.8:
            selected = np.random.choice(population_size, size=2, replace=False)
            parent1, parent2 = population[selected[0]], population[selected[1]]
            child = crossover(parent1, parent2)
            population = np.vstack((population, child))
            population = np.delete(population, np.random.choice(population_size), axis=0)
        else:
            for i in range(population_size):
                population[i] = mutate(population[i])

    h_values = best_solution
    h_values = np.clip(h_values, 0, 1)
    h_values = h_values * (n_points / 2) / np.sum(h_values)
    h_values = np.clip(h_values, 0, 1)
    c5_bound = evaluate_individual(h_values)
    return h_values, c5_bound, n_points
