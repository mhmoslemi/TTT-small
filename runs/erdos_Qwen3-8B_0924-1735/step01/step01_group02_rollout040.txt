import numpy as np
import scipy.optimize as opt
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def evaluate(h):
        return compute_c5(h)

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.full_like(h, 0.5)
        return h_clipped * (n_points / 2) / total

    def create_initial_guess():
        h = initial_h.copy()
        perturbation = 0.1 * np.cos(2 * np.pi * np.arange(n_points) / n_points)
        h = h * (1 + perturbation)
        h = project(h)
        return h

    h = create_initial_guess()

    X = []
    Y = []
    for _ in range(20):
        if time.time() - time_start > time_remaining - 100:
            break
        h_candidate = h + np.random.uniform(-0.1, 0.1, size=n_points)
        h_candidate = project(h_candidate)
        c5_value = compute_c5(h_candidate)
        X.append(h_candidate)
        Y.append(c5_value)

    kernel = ConstantKernel(1.0, (1e-2, 1e2)) * RBF(1.0, (1e-2, 1e2))
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-5, n_restarts_optimizer=10)

    for _ in range(150):
        if time.time() - time_start > time_remaining - 100:
            break
        try:
            gp.fit(np.array(X), np.array(Y))
        except:
            break
        next_x = gp.ramdom_sample()
        next_x = project(next_x)
        next_c5 = compute_c5(next_x)
        X.append(next_x)
        Y.append(next_c5)

    candidates = np.array(X)
    values = np.array(Y)

    indices = np.argsort(values)
    top_candidates = candidates[indices[:10]]
    top_values = values[indices[:10]]

    for i in range(10):
        if time.time() - time_start > time_remaining - 100:
            break
        h_refined = top_candidates[i].copy()
        result = opt.minimize(
            lambda x: compute_c5(x),
            h_refined,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2},
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        h_refined = project(result.x)
        c5_value = compute_c5(h_refined)
        if c5_value < best_c5:
            best_c5 = c5_value
            best_h = h_refined.copy()

    for _ in range(5):
        if time.time() - time_start > time_remaining - 100:
            break
        noise = np.random.uniform(-0.05, 0.05, size=n_points)
        h_candidate = best_h + noise
        h_candidate = project(h_candidate)
        c5_value = compute_c5(h_candidate)
        if c5_value < best_c5:
            best_c5 = c5_value
            best_h = h_candidate.copy()

    best_h = project(best_h)
    c5_value = compute_c5(best_h)

    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    c5_value = compute_c5(best_h)
    return best_h, c5_value, n_points
