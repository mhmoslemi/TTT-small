import numpy as np
import scipy.optimize as opt
import random
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 120
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    initial_h_values[::3] = 1.0
    initial_h_values[1::4] = 0.5
    initial_h_values[2::5] = 0.75
    initial_h_values = initial_h_values.astype(float)
    initial_h_values /= np.sum(initial_h_values) * dx
    initial_h_values *= n_points / 2
    h_values = initial_h_values.copy()
    
    def evaluate_individual(h):
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    def evaluate_erdos_solution(h):
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    def rescale(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total
    
    def mutate(h):
        h = h.copy()
        mut_rate = 0.1
        for i in range(n_points):
            if np.random.rand() < mut_rate:
                h[i] += np.random.uniform(-0.1, 0.1)
        h = rescale(h)
        return h
    
    def crossover(parent1, parent2):
        alpha = np.random.uniform(-1, 1)
        child = 0.5 * (parent1 + parent2) + alpha * (parent1 - parent2)
        return rescale(child)
    
    def select_parents(population, fitnesses):
        parents = []
        for _ in range(len(population)):
            idx = np.random.choice(len(population), p=fitnesses / np.sum(fitnesses))
            parents.append(population[idx])
        return parents
    
    population_size = 30
    population = np.random.rand(population_size, n_points)
    population = np.array([rescale(p) for p in population])
    best_h = population[0]
    best_c5 = evaluate_individual(best_h)
    elapsed = 0
    
    for gen in range(100):
        if elapsed >= budget_s:
            break
        fitnesses = np.array([evaluate_individual(p) for p in population])
        parents = select_parents(population, fitnesses)
        offspring = []
        for i in range(population_size // 2):
            p1, p2 = random.sample(parents, 2)
            child = crossover(p1, p2)
            offspring.append(child)
            child = mutate(child)
            offspring.append(child)
        population = np.array(offspring)
        current_best = population[np.argmin(fitnesses)]
        current_c5 = evaluate_individual(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
        elapsed += 1
    
    h_values = rescale(best_h)
    c5_bound = evaluate_erdos_solution(h_values)
    return h_values, c5_bound, n_points
