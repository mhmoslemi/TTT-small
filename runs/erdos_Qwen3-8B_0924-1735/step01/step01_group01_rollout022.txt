import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy() if 'initial_h_values' in locals() else np.zeros(n_points)
    
    # Structured initialization
    m = 10
    block_size = 2.0 / m
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        pos = i * dx
        block_idx = int(np.floor(pos / block_size))
        if block_idx % 2 == 0:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    # PSO parameters
    num_particles = 50
    max_iter = 100
    inertia = 0.8
    c1, c2 = 1.5, 1.5
    velocities = np.random.rand(num_particles, n_points) * 0.1
    particles = np.random.rand(num_particles, n_points) * 0.5 + 0.5
    particles = normalize(particles)
    p_best = particles.copy()
    p_best_fitness = np.array([objective(p) for p in particles])
    g_best = particles[np.argmin(p_best_fitness)]
    g_best_fitness = np.min(p_best_fitness)
    
    start_time = time.time()
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = inertia * velocities[i] + c1 * r1 * (p_best[i] - particles[i]) + c2 * r2 * (g_best - particles[i])
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize(particles[i])
        
        current_fitness = np.array([objective(p) for p in particles])
        better_indices = current_fitness < p_best_fitness
        p_best[better_indices] = particles[better_indices]
        p_best_fitness[better_indices] = current_fitness[better_indices]
        
        if np.min(current_fitness) < g_best_fitness:
            g_best = particles[np.argmin(current_fitness)]
            g_best_fitness = np.min(current_fitness)
    
    h_opt = g_best
    h_opt = normalize(h_opt)
    c5_opt = objective(h_opt)
    
    return (h_opt, c5_opt, n_points)
