import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h = initial_h_values
    if len(initial_h) != n_points:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.full_like(h, 0.5)
        return h_clipped * (n_points / 2) / total

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)}

    def perturb(h, sigma):
        return np.clip(h + np.random.normal(0, sigma, size=n_points), 0, 1)

    h = project(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    for _ in range(20):
        if time.time() - time_start > budget_s - 100:
            break
        h = project(np.random.uniform(0, 1, size=n_points))
        h = project(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    for _ in range(10):
        if time.time() - time_start > budget_s - 100:
            break
        h = project(best_h + np.random.normal(0, 0.05, size=n_points))
        h = project(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    def optimize(h):
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        return result.x

    h = optimize(project(best_h))
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for _ in range(5):
        if time.time() - time_start > budget_s - 100:
            break
        h = project(best_h + np.random.normal(0, 0.05, size=n_points))
        h = project(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    best_h = project(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
