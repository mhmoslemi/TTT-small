import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i < 60:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.0
    initial_h_values = np.random.permutation(initial_h_values)
    initial_h_values /= np.sum(initial_h_values)
    initial_h_values *= n_points / 2
    initial_h_values = np.clip(initial_h_values, 0, 1)

    def objective(params):
        h = params.reshape(-1)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    def constraint(params):
        h = params.reshape(-1)
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = initial_h_values.copy()
    initial_guess = initial_guess.reshape(-1)

    start_time = time.time()

    def objective_de(params):
        h = params.reshape(-1)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    def constraint_de(params):
        h = params.reshape(-1)
        return np.sum(h) - n_points / 2

    def de_search():
        pop_size = 20
        pop = np.random.rand(pop_size, n_points)
        pop = np.clip(pop, 0, 1)
        pop = pop * (n_points / 2) / np.sum(pop, axis=1, keepdims=True)
        pop = np.clip(pop, 0, 1)
        best = pop[0]
        best_c5 = objective_de(pop[0])
        for _ in range(300):
            for i in range(pop_size):
                a, b, c = np.random.choice(pop_size, 3, replace=False)
                mutant = pop[a] + 0.8 * (pop[b] - pop[c])
                mutant = np.clip(mutant, 0, 1)
                mutant = mutant * (n_points / 2) / np.sum(mutant)
                mutant = np.clip(mutant, 0, 1)
                trial = np.zeros_like(pop[i])
                for j in range(n_points):
                    if np.random.rand() < 0.9:
                        trial[j] = mutant[j]
                    else:
                        trial[j] = pop[i][j]
                trial = np.clip(trial, 0, 1)
                trial = trial * (n_points / 2) / np.sum(trial)
                trial = np.clip(trial, 0, 1)
                trial_c5 = objective_de(trial)
                if trial_c5 < objective_de(pop[i]):
                    pop[i] = trial
            current_best = pop[np.argmin([objective_de(p) for p in pop])]
            current_c5 = objective_de(current_best)
            if current_c5 < best_c5:
                best = current_best
                best_c5 = current_c5
            if time.time() - start_time >= budget_s:
                return best
        return best

    best_h = de_search()
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = objective_de(best_h)
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (best_h, best_c5, n_points)
    
    def objective_grad(params):
        h = params.reshape(-1)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    def gradient(params):
        h = params.reshape(-1)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        grad = np.zeros_like(h)
        for i in range(n_points):
            h[i] += 1e-6
            c5p = objective_grad(h)
            grad[i] = (c5p - c5) / 1e-6
            h[i] -= 1e-6
        return grad

    def constraint_grad(params):
        h = params.reshape(-1)
        return np.sum(h) - n_points / 2

    result = opt.minimize(
        fun=objective_grad,
        x0=best_h.flatten(),
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_grad}],
        tol=1e-6,
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    h_opt = np.clip(result.x, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    h_opt = np.clip(h_opt, 0, 1)
    c5_opt = objective_grad(h_opt)

    return (h_opt, c5_opt, n_points)
