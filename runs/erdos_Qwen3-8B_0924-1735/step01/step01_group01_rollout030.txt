import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    if 'initial_h_values' in locals():
        initial_h_values = np.copy(initial_h_values)
    else:
        initial_h_values = np.zeros(n_points)
        for i in range(n_points):
            if i < 60:
                initial_h_values[i] = 1.0
            else:
                initial_h_values[i] = 0.0
        initial_h_values = np.random.permutation(initial_h_values)
        initial_h_values /= np.sum(initial_h_values)
        initial_h_values *= n_points / 2
        initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = np.copy(initial_h_values)

    def objective(params):
        h = params.reshape(-1)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    def constraint(params):
        h = params.reshape(-1)
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = initial_h_values.copy()
    initial_guess = initial_guess.reshape(-1)

    def differential_evolution(pop_size=20, max_iter=100, tol=1e-6):
        pop = np.random.uniform(0, 1, (pop_size, n_points))
        pop = pop * (n_points / 2) / np.sum(pop, axis=1, keepdims=True)
        pop = np.clip(pop, 0, 1)
        best = pop[np.argmin([objective(p) for p in pop])]
        best_val = objective(best)
        best_idx = 0
        for _ in range(max_iter):
            for i in range(pop_size):
                a, b, c = np.random.choice(pop_size, 3, replace=False)
                mutant = pop[a] + 0.8 * (pop[b] - pop[c])
                mutant = np.clip(mutant, 0, 1)
                mutant = mutant * (n_points / 2) / np.sum(mutant)
                mutant = np.clip(mutant, 0, 1)
                trial = pop[i]
                if np.random.rand() < 0.9:
                    trial = mutant
                trial_val = objective(trial)
                if trial_val < best_val:
                    best = trial
                    best_val = trial_val
                    best_idx = i
            if np.abs(best_val - objective(best)) < tol:
                break
        return best, best_val

    start_time = time.time()
    if budget_s > 300:
        result = differential_evolution(pop_size=30, max_iter=100, tol=1e-6)
    else:
        result = differential_evolution(pop_size=15, max_iter=50, tol=1e-6)
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (initial_h_values, objective(initial_h_values), n_points)
    h_opt = np.clip(result[0], 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    h_opt = np.clip(h_opt, 0, 1)
    c5_opt = result[1]
    return (h_opt, c5_opt, n_points)
