import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 120
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    initial_h_values[::3] = 1.0
    initial_h_values[1::4] = 0.5
    initial_h_values[2::5] = 0.75
    initial_h_values = initial_h_values.astype(float)

        h = np.array(h)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    def evaluate_individual(h):
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    def mutate(h):
        h = h.copy()
        mut_rate = 0.1
        for i in range(n_points):
            if np.random.rand() < mut_rate:
                h[i] += np.random.uniform(-0.1, 0.1)
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / np.sum(h)
        return h
    
    def crossover(parent1, parent2):
        alpha = np.random.uniform(-1, 1)
        child = 0.5 * (parent1 + parent2) + alpha * (parent1 - parent2)
        return child
    
    def create_population(size):
        pop = np.random.rand(size, n_points)
        pop *= (n_points / 2) / np.sum(pop, axis=1, keepdims=True)

    pop = create_population(pop_size)
    best_h = pop[0]
    best_c5 = evaluate_individual(best_h)

        if time.time() - start_time > budget_s:

        # Evaluate fitness
        fitness = np.array([evaluate_individual(ind) for ind in pop])
        top_idx = np.argsort(fitness)[:pop_size // 2]
        selected = pop[top_idx]
        
        # Generate new population
        new_pop = []
        for _ in range(pop_size // 2):
            parent1, parent2 = random.choices(selected)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_pop.append(child)
        
        pop = np.vstack([selected, new_pop])

            if c5 < best_c5:
