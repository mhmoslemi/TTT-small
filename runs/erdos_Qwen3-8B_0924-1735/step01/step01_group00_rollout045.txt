import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 120
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    initial_h_values[::3] = 1.0
    initial_h_values[1::4] = 0.5
    initial_h_values[2::5] = 0.75
    initial_h_values = initial_h_values.astype(float)
    initial_h_values /= np.sum(initial_h_values) * dx
    initial_h_values *= n_points / 2
    h_values = initial_h_values.copy()

    def compute_c5(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr

    def evaluate_individual(h):
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx

    population_size = 30
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8

    population = np.random.rand(population_size, n_points)
    population *= (n_points / 2) / np.sum(population, axis=1, keepdims=True)
    population = np.clip(population, 0, 1)

    best_h = population[0]
    best_c5 = evaluate_individual(best_h)

    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitness = np.array([evaluate_individual(h) for h in population])
        idxs = np.argsort(fitness)
        elites = population[idxs[:int(population_size * 0.2)]]

        new_population = elites.copy()
        for _ in range(population_size - len(elites)):
            p1, p2 = np.random.choice(population_size, 2, replace=False)
            parent1, parent2 = population[p1], population[p2]
            if np.random.rand() < crossover_rate:
                alpha = np.random.uniform(-1, 1)
                child = 0.5 * (parent1 + parent2) + alpha * (parent1 - parent2)
            else:
                child = np.random.rand(n_points)
            child *= (n_points / 2) / np.sum(child)
            child = np.clip(child, 0, 1)
            new_population = np.vstack((new_population, child))

        population = new_population
        population = np.clip(population, 0, 1)
        population *= (n_points / 2) / np.sum(population, axis=1, keepdims=True)

        for i in range(population_size):
            if np.random.rand() < mutation_rate:
                idxs = np.random.choice(n_points, 5, replace=False)
                population[i, idxs] += np.random.uniform(-0.1, 0.1)
                population[i, idxs] = np.clip(population[i, idxs], 0, 1)
                population[i] *= (n_points / 2) / np.sum(population[i])

        fitness = np.array([evaluate_individual(h) for h in population])
        idxs = np.argsort(fitness)
        best_h = population[idxs[0]]
        best_c5 = fitness[idxs[0]]

    best_h = np.clip(best_h, 0, 1)
    best_h *= (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    c5_bound = evaluate_individual(best_h)
    return best_h, c5_bound, n_points
