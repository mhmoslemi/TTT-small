import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45  # Starting with the provided initial_h_values
    initial_h = initial_h_values
    if len(initial_h) != n_points:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    cons = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)}
    bounds = [(0, 1) for _ in range(n_points)]

    # Simulated Annealing parameters
    T = 300
    max_iter = 1000
    cooling_rate = 0.95
    noise_std = 0.25

    # Main SA loop
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for t in range(max_iter):
        if time.time() - time_start > budget_s - 100:
            break
        noise = np.random.normal(0, noise_std, n_points)
        h_noise = h + noise
        h_noise = np.clip(h_noise, 0, 1)
        h_noise = h_noise * (n_points / 2) / np.sum(h_noise)
        c5 = compute_c5(h_noise)
        delta = c5 - best_c5
        if c5 < best_c5 or np.random.rand() < np.exp(-delta / T):
            best_h = h_noise.copy()
            best_c5 = c5
        T *= cooling_rate
        noise_std = max(0.05, noise_std * 0.95)

    # Local refinement with L-BFGS-B
    result = opt.minimize(
        objective,
        best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 100}
    )
    h = result.x
    c5_value = compute_c5(h)
    if c5_value < best_c5:
        best_c5 = c5_value
        best_h = h.copy()

    # Random restarts
    for _ in range(5):
        if time.time() - time_start > budget_s - 100:
            break
        noise = np.random.uniform(-0.1, 0.1, n_points)
        h = best_h + noise
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5_value = compute_c5(h)
        if c5_value < best_c5:
            best_c5 = c5_value
            best_h = h.copy()

    # Final check and projection
    best_h = np.clip(best_h, 0, 1)
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_points
