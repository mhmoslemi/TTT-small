import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 50
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.random.rand(n_points)
        h = np.where(np.random.rand(n_points) < 0.5, 1.0, 0.0)
        h = h / np.sum(h) * n_points / 2
    h = h * (1 + 0.1 * np.cos(2 * np.pi * np.arange(n_points) / n_points))
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.full_like(h, 0.5)
        return h_clipped * (n_points / 2) / total

    def objective(h):
        return compute_c5(h)

    def surrogate_objective(h):
        if np.sum(h) < n_points / 2 - 1e-6 or np.sum(h) > n_points / 2 + 1e-6:
            return np.inf
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.log(np.sum(np.exp(100 * (corr - 1e-6))))

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)}

    def bayesian_optimization():
        from sklearn.gaussian_process import GaussianProcessRegressor
        from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C
        import warnings
        warnings.filterwarnings("ignore")
        X = np.zeros((1, n_points))
        y = np.array([compute_c5(best_h)])
        gp = GaussianProcessRegressor(kernel=RBF(length_scale=1.0, length_scale_bounds=(1e-2, 1e2)), alpha=1e-5)
        for _ in range(200):
            if time.time() - time_start > time_remaining - 100:
                break
            if len(X) > 1:
                gp.fit(X, y)
            else:
                gp.kernel_ = RBF(length_scale=1.0, length_scale_bounds=(1e-2, 1e2))
            mu, sigma = gp.predict(X, return_std=True)
            acq = mu - 1.2 * sigma
            best_idx = np.argmax(acq)
            best_x = X[best_idx]
            h_candidate = best_x.copy()
            for i in range(n_points):
                h_candidate[i] += np.random.normal(0, 0.05)
                h_candidate[i] = np.clip(h_candidate[i], 0, 1)
            h_candidate = project(h_candidate)
            c5 = compute_c5(h_candidate)
            X = np.vstack((X, h_candidate))
            y = np.append(y, c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate.copy()

    bayesian_optimization()

    def local_refinement():
        for _ in range(10):
            if time.time() - time_start > time_remaining - 100:
                break
            h_candidate = best_h.copy()
            noise = np.random.uniform(-0.05, 0.05, size=n_points)
            h_candidate += noise
            h_candidate = project(h_candidate)
            result = opt.minimize(
                objective,
                h_candidate,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 100}
            )
            h_candidate = result.x
            c5 = compute_c5(h_candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate.copy()

    local_refinement()

    h = best_h
    h = project(h)
    c5_value = compute_c5(h)
    if np.abs(np.sum(h) - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / np.sum(h)
    c5_value = compute_c5(h)
    return h, c5_value, n_points
