import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h = initial_h_values
    if len(initial_h) != n_points:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def objective(h):
        return compute_c5(h)

    def surrogate_objective(h):
        if np.sum(h) < n_points / 2 - 1e-6 or np.sum(h) > n_points / 2 + 1e-6:
            return np.inf
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.log(np.sum(np.exp(100 * (corr - 1e-6))))

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def feasibility_projection(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return 0.5 * np.ones_like(h)
        return h * (n_points / 2) / sum_h

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)}

    def simulated_annealing():
        nonlocal best_c5, best_h
        T = 300
        cooling_rate = 0.99
        max_iterations = 1000
        noise_sigma = 0.25
        top_candidates = []
        for t in range(max_iterations):
            if time.time() - time_start > time_remaining - 100:
                break
            noise = np.random.normal(0, noise_sigma, n_points)
            h_candidate = feasibility_projection(h + noise)
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
                top_candidates.append(h_candidate)
                if len(top_candidates) > 100:
                    top_candidates.pop()
            if np.random.rand() < np.exp(-(c5_candidate - best_c5) / T):
                h = h_candidate.copy()
            T *= cooling_rate
            noise_sigma *= 0.99
        return top_candidates

    top_candidates = simulated_annealing()
    if not top_candidates:
        return best_h, best_c5, n_points

    def refine_candidate(h_candidate):
        result = opt.minimize(
            surrogate_objective,
            h_candidate,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        h_refined = result.x
        c5_refined = compute_c5(h_refined)
        return h_refined, c5_refined

    refined_candidates = [refine_candidate(h_candidate) for h_candidate in top_candidates]
    refined_candidates.sort(key=lambda x: x[1])
    best_refined, best_refined_c5 = refined_candidates[0]

    if best_refined_c5 < best_c5:
        best_c5 = best_refined_c5
        best_h = best_refined.copy()

    for _ in range(5):
        if time.time() - time_start > time_remaining - 100:
            break
        noise = np.random.normal(0, 0.05, n_points)
        h_candidate = feasibility_projection(best_h + noise)
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()

    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = feasibility_projection(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
