import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    initial_h_values = np.random.permutation(np.zeros(n_points))
    initial_h_values[:n_points//2] = 1.0
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)

    def objective(params):
        h = params.reshape(-1)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    def constraint(params):
        h = params.reshape(-1)
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = initial_h_values.copy()
    initial_guess = initial_guess.reshape(-1)

    start_time = time.time()

    def de_optimizer():
        from scipy.optimize import differential_evolution
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = differential_evolution(
            fun=objective,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            strategy='cosine',
            popsize=20,
            tol=1e-6,
            mutation=(0.5, 1.0),
            recombination=0.9,
            maxiter=80,
            disp=False,
            polish=True,
            seed=seed
        )
        return result.x

    result = de_optimizer()

    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (initial_h_values, objective(initial_h_values), n_points)

    h_opt = result
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    h_opt = np.clip(h_opt, 0, 1)
    c5_opt = objective(h_opt)

    return (h_opt, c5_opt, n_points)
