import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    if 'initial_h_values' in globals():
        initial_h_values = np.array(initial_h_values)
        initial_h_values = np.clip(initial_h_values, 0, 1)
        initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
        initial_h_values = np.clip(initial_h_values, 0, 1)
    else:
        initial_h_values = np.zeros(n_points)
        for i in range(n_points // 2):
            initial_h_values[i] = 1.0
        for i in range(n_points // 2, n_points):
            initial_h_values[i] = 0.0
        np.random.shuffle(initial_h_values)
        initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
        initial_h_values = np.clip(initial_h_values, 0, 1)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    def constraint(h):
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def pso_run():
        pop_size = 50
        max_iter = 100
        inertia = 0.8
        c1, c2 = 1.5, 1.5
        pos = np.random.rand(pop_size, n_points)
        vel = np.random.rand(pop_size, n_points) * 0.1
        pbest_pos = pos.copy()
        pbest_cost = np.inf * np.ones(pop_size)
        gbest_pos = pos[0].copy()
        gbest_cost = np.inf

        for _ in range(max_iter):
            for i in range(pop_size):
                r1, r2 = np.random.rand(2)
                vel[i] = inertia * vel[i] + c1 * r1 * (pbest_pos[i] - pos[i]) + c2 * r2 * (gbest_pos - pos[i])
                pos[i] = np.clip(pos[i] + vel[i], 0, 1)
                pos[i] = pos[i] * (n_points / 2) / np.sum(pos[i])
                pos[i] = np.clip(pos[i], 0, 1)
                cost = objective(pos[i])
                
                if cost < pbest_cost[i]:
                    pbest_pos[i] = pos[i].copy()
                    pbest_cost[i] = cost
            current_gbest = np.min(pbest_cost)
            if current_gbest < gbest_cost:
                gbest_pos = pbest_pos[np.argmin(pbest_cost)].copy()
                gbest_cost = current_gbest
            if gbest_cost < 1e-6:
                break
        
        return gbest_pos
    
    start_time = time.time()
    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)
    elapsed = 0
    
    while elapsed < budget_s - 1:
        try:
            candidate = pso_run()
            candidate_c5 = objective(candidate)
            if candidate_c5 < best_c5:
                best_h = candidate.copy()
                best_c5 = candidate_c5
            elapsed = time.time() - start_time
        except:
            break
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
