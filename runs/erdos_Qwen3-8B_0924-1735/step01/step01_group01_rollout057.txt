import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.0
    initial_h_values = np.random.permutation(initial_h_values)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)

    def objective(params):
        h = params.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = initial_h_values.copy()
    initial_guess = initial_guess.reshape(-1)

    def pso_update(x, v, pbest, gbest, omega, c1, c2, r1, r2):
        v_new = omega * v + c1 * r1 * (pbest - x) + c2 * r2 * (gbest - x)
        x_new = x + v_new
        x_new = np.clip(x_new, 0, 1)
        x_new = x_new * (n_points / 2) / np.sum(x_new)
        return x_new, v_new

    n_particles = 50
    max_iter = 300
    omega = 0.8
    c1, c2 = 1.5, 1.5
    particles = np.random.rand(n_particles, n_points)
    particles = particles * (n_points / 2) / np.sum(particles, axis=1, keepdims=True)
    particles = np.clip(particles, 0, 1)
    velocities = np.random.rand(n_particles, n_points) * 0.1
    pbest = particles.copy()
    pbest_vals = np.array([objective(p) for p in pbest])
    gbest = pbest[np.argmin(pbest_vals)]
    gbest_val = np.min(pbest_vals)
    start_time = time.time()
    for _ in range(max_iter):
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = omega * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            curr_val = objective(particles[i])
            if curr_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = curr_val
        if np.min(pbest_vals) < gbest_val:
            gbest = pbest[np.argmin(pbest_vals)]
            gbest_val = np.min(pbest_vals)
        if time.time() - start_time > budget_s:
            break
    h_opt = gbest
    c5_opt = gbest_val
    return (h_opt, c5_opt, n_points)
