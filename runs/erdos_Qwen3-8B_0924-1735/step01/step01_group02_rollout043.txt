import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * 2.0 / n_points
        return np.max(corr)

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]

    def perturb(h, sigma):
        return np.clip(h + np.random.normal(0, sigma, size=n_points), 0, 1)

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.full_like(h, 0.5)
        return h_clipped * (n_points / 2) / total

    def optimize_with_pso():
        from scipy.optimize import minimize
        candidates = [project(perturb(initial_h, 0.25)) for _ in range(50)]
        for i in range(100):
            for j in range(len(candidates)):
                h_j = candidates[j]
                c5_j = compute_c5(h_j)
                if c5_j < compute_c5(candidates[0]):
                    candidates[0] = h_j
            for j in range(len(candidates)):
                h_j = candidates[j]
                h_j = project(h_j + np.random.normal(0, 0.1, size=n_points))
                candidates[j] = h_j
            if time.time() - time_start > time_remaining - 100:
                break
        return candidates

    candidates = optimize_with_pso()
    candidates = [project(c) for c in candidates]
    candidates = [c for c in candidates if compute_c5(c) < np.inf]
    if not candidates:
        return initial_h, compute_c5(initial_h), n_points

    for c in candidates:
        if compute_c5(c) < best_c5:
            best_c5 = compute_c5(c)
            best_h = c

    result = opt.minimize(
        objective,
        best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum, 'jac': lambda x: np.ones_like(x)},
        options={'ftol': 1e-9, 'maxiter': 100}
    )
    if result.success:
        h = result.x
        c5_value = compute_c5(h)
        if c5_value < best_c5:
            best_c5 = c5_value
            best_h = h.copy()

    for _ in range(5):
        if time.time() - time_start > time_remaining - 100:
            break
        noise = np.random.uniform(-0.05, 0.05, size=n_points)
        h = best_h + noise
        h = np.clip(h, 0, 1)
        if np.abs(np.sum(h) - n_points / 2) < 1e-6:
            c5_value = compute_c5(h)
            if c5_value < best_c5:
                best_c5 = c5_value
                best_h = h.copy()

    h = best_h
    h = np.clip(h, 0, 1)
    if np.abs(np.sum(h) - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / np.sum(h)
    c5_value = compute_c5(h)
    if c5_value < best_c5:
        best_c5 = c5_value
        best_h = h.copy()

    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    c5_value = compute_c5(best_h)
    return best_h, c5_value, n_points
