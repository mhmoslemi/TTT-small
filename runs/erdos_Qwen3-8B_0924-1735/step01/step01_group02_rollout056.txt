import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)}

    def simulated_annealing():
        nonlocal best_c5, best_h
        T = 300
        cooling_rate = 0.995
        max_iter = 1000
        for t in range(max_iter):
            if time.time() - time_start > time_remaining - 100:
                break
            noise = np.random.normal(0, 0.25, n_points)
            h_new = h + noise
            h_new = np.clip(h_new, 0, 1)
            sum_h_new = np.sum(h_new)
            if np.abs(sum_h_new - n_points / 2) > 1e-6:
                h_new = h_new * (n_points / 2) / sum_h_new
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            delta = c5_new - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h = h_new.copy()
            T *= cooling_rate

    def local_refinement():
        nonlocal best_c5, best_h
        for _ in range(100):
            if time.time() - time_start > time_remaining - 100:
                break
            result = opt.minimize(
                lambda x: compute_c5(x),
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 100}
            )
            if result.success:
                h = result.x
                c5_value = compute_c5(h)
                if c5_value < best_c5:
                    best_c5 = c5_value
                    best_h = h.copy()
            else:
                h = best_h.copy()
                if np.abs(np.sum(h) - n_points / 2) > 1e-6:
                    h = h * (n_points / 2) / np.sum(h)
                c5_value = compute_c5(h)
                if c5_value < best_c5:
                    best_c5 = c5_value
                    best_h = h.copy()

    simulated_annealing()
    local_refinement()

    h = best_h
    h = np.clip(h, 0, 1)
    if np.abs(np.sum(h) - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / np.sum(h)
    c5_value = compute_c5(h)
    if c5_value < best_c5:
        best_c5 = c5_value
        best_h = h.copy()

    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    c5_value = compute_c5(best_h)
    return best_h, c5_value, n_points
