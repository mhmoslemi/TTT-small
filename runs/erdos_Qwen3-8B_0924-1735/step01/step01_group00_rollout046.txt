import numpy as np
import scipy.optimize as opt
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 120
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    initial_h_values[::3] = 1.0
    initial_h_values[1::4] = 0.5
    initial_h_values[2::5] = 0.75
    initial_h_values = initial_h_values.astype(float)
    initial_h_values /= np.sum(initial_h_values) * dx
    initial_h_values *= n_points / 2
    h_values = initial_h_values.copy()
    
    def compute_c5(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    def evaluate_individual(h):
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    def mutate(h):
        h = h.copy()
        mut_rate = 0.1
        for i in range(n_points):
            if np.random.rand() < mut_rate:
                h[i] += np.random.uniform(-0.1, 0.1)
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / np.sum(h)
        return h
    
    def crossover(parent1, parent2):
        alpha = np.random.uniform(-1, 1)
        child = 0.5 * (parent1 + parent2) + alpha * (parent1 - parent2)
        return child
    
    def select_parents(population, fitnesses):
        indices = np.argsort(fitnesses)
        return population[indices[:2]]
    
    def evolve(population, fitnesses, n_points):
        new_population = []
        for _ in range(len(population)):
            parent1, parent2 = select_parents(population, fitnesses)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        return new_population
    
    population_size = 30
    population = [np.random.rand(n_points) * (n_points / 2) / np.sum(np.random.rand(n_points)) for _ in range(population_size)]
    best_h = population[0]
    best_c5 = evaluate_individual(best_h)

    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        fitnesses = [evaluate_individual(ind) for ind in population]
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_c5 = fitnesses[best_idx]
            best_h = population[best_idx]
        population = evolve(population, fitnesses, n_points)
    
    h_values = best_h
    h_values = np.clip(h_values, 0, 1)
    h_values = h_values * (n_points / 2) / np.sum(h_values)
    h_values = np.clip(h_values, 0, 1)
    c5_bound = compute_c5(h_values)
    return h_values, c5_bound, n_points
