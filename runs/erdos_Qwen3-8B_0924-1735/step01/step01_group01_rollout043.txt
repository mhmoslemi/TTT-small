import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        initial_h = np.copy(initial_h_values)
    else:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                initial_h[i] = 1.0
        initial_h = np.random.permutation(initial_h)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        initial_h = np.clip(initial_h, 0, 1)

    def objective(params):
        h = np.clip(params, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    def constraint(params):
        return np.sum(np.clip(params, 0, 1)) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = initial_h.copy()

    start_time = time.time()
    result = opt.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-6,
        maxiter=100,
        disp=False,
        polish=True,
        constraints=[{'type': 'eq', 'fun': constraint}]
    )
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        h_opt = initial_guess
    else:
        h_opt = np.clip(result.x, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        h_opt = np.clip(h_opt, 0, 1)

    c5_opt = objective(h_opt)
    return (h_opt, c5_opt, n_points)
