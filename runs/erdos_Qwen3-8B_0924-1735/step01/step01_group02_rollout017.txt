import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 60  # Larger n_points for better resolution
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)}

    # Simulated Annealing with adaptive perturbation
    T = 300.0
    max_iter = 1000
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for t in range(max_iter):
        if time.time() - time_start > time_remaining - 100:
            break
        if t < 500:
            sigma = 0.25 * (1 - t / 500)
        else:
            sigma = 0.05
        noise = np.random.normal(0, sigma, n_points)
        h_new = np.clip(h + noise, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new) if np.sum(h_new) > 0 else h_new
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        if np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            h = h_new.copy()
        T *= 0.99

    # Local refinement with L-BFGS-B
    if time.time() - time_start < time_remaining - 200:
        result = opt.minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        h = result.x
        c5_value = compute_c5(h)
        if c5_value < best_c5:
            best_c5 = c5_value
            best_h = h.copy()

    # Check for valid solution and finalize
    h = np.clip(best_h, 0, 1)
    sum_h = np.sum(h)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / sum_h
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
