import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    if 'initial_h_values' in globals():
        initial_h_values = np.copy(initial_h_values)
    else:
        # Structured block pattern
        m = 10
        block_width = 2.0 / m
        for i in range(n_points):
            if i * dx < block_width * (i % 2 + 1):
                initial_h_values[i] = 1.0
        initial_h_values = np.random.permutation(initial_h_values)
        initial_h_values /= np.sum(initial_h_values)
        initial_h_values *= n_points / 2
        initial_h_values = np.clip(initial_h_values, 0, 1)

    def objective(params):
        h = params.reshape(-1)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def pso_search():
        P = 50
        max_iter = 100
        inertia = 0.8
        c1, c2 = 1.5, 1.5
        particles = np.random.rand(P, n_points)
        particles = np.clip(particles, 0, 1)
        velocities = np.random.rand(P, n_points) * 0.1
        pbest = np.copy(particles)
        pbest_vals = np.array([objective(p) for p in pbest])
        gbest = pbest[np.argmin(pbest_vals)]
        gbest_val = np.min(pbest_vals)

        for _ in range(max_iter):
            for i in range(P):
                r1, r2 = np.random.rand(2)
                velocities[i] = inertia * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
                particles[i] += velocities[i]
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                particles[i] = np.clip(particles[i], 0, 1)
                current_val = objective(particles[i])
                if current_val < pbest_vals[i]:
                    pbest[i] = np.copy(particles[i])
                    pbest_vals[i] = current_val
            gbest = pbest[np.argmin(pbest_vals)]
            gbest_val = np.min(pbest_vals)
            if gbest_val < 1e-10:
                break
        return gbest, gbest_val

    start_time = time.time()
    h_opt, c5_opt = pso_search()
    elapsed_time = time.time() - start_time

    if elapsed_time >= budget_s:
        return (initial_h_values, objective(initial_h_values), n_points)

    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    h_opt = np.clip(h_opt, 0, 1)
    c5_opt = objective(h_opt)

    return (h_opt, c5_opt, n_points)
