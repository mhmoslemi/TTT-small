import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    if 'initial_h_values' in locals():
        initial_h_values = np.random.permutation(initial_h_values)
        initial_h_values /= np.sum(initial_h_values)
        initial_h_values *= n_points / 2
        initial_h_values = np.clip(initial_h_values, 0, 1)
    else:
        for i in range(n_points):
            if i < n_points // 2:
                initial_h_values[i] = 1.0
            else:
                initial_h_values[i] = 0.0
        initial_h_values = np.random.permutation(initial_h_values)
        initial_h_values /= np.sum(initial_h_values)
        initial_h_values *= n_points / 2
        initial_h_values = np.clip(initial_h_values, 0, 1)

    def objective(params):
        h = params.reshape(-1)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    def constraint(params):
        h = params.reshape(-1)
        return np.sum(h) - n_points / 2

    def pso(n_particles=50, max_iter=100, omega=0.8, c1=1.5, c2=1.5):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        pos = np.random.uniform(0, 1, (n_particles, n_points))
        vel = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
        pbest_pos = pos.copy()
        pbest_val = np.array([objective(p) for p in pos])
        gbest_pos = pbest_pos[np.argmin(pbest_val)]
        gbest_val = np.min(pbest_val)

        for _ in range(max_iter):
            for i in range(n_particles):
                r1, r2 = np.random.rand(2)
                vel[i] = omega * vel[i] + c1 * r1 * (pbest_pos[i] - pos[i]) + c2 * r2 * (gbest_pos - pos[i])
                pos[i] = np.clip(pos[i] + vel[i], 0, 1)
                pos[i] = pos[i] * (n_points / 2) / np.sum(pos[i])
                pos[i] = np.clip(pos[i], 0, 1)
                current_val = objective(pos[i])
                if current_val < pbest_val[i]:
                    pbest_pos[i] = pos[i].copy()
                    pbest_val[i] = current_val
            gbest_val = np.min(pbest_val)
            gbest_pos = pbest_pos[np.argmin(pbest_val)]
            if gbest_val < 1e-8:
                break
        return gbest_pos, gbest_val

    start_time = time.time()
    if 'initial_h_values' in locals():
        h_opt = initial_h_values.copy()
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        h_opt = np.clip(h_opt, 0, 1)
        c5_opt = objective(h_opt)
    else:
        h_opt, c5_opt = pso()

    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (initial_h_values, objective(initial_h_values), n_points)

    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    h_opt = np.clip(h_opt, 0, 1)
    c5_opt = objective(h_opt)

    return (h_opt, c5_opt, n_points)
