import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                h[i] = 1
            elif i % 3 == 1:
                h[i] = 0.5
            else:
                h[i] = 0
    h = h / np.sum(h) * n_points / 2
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]

    def perturb(h, std):
        return np.clip(h + np.random.normal(0, std, size=n_points), 0, 1)

    def objective(h):
        return compute_c5(h)

    def surrogate_objective(h):
        if abs(np.sum(h) - n_points / 2) > 1e-6:
            return np.inf
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return -np.log(np.sum(np.exp(100 * (corr - 1e-6))))

    cons = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)}

    # Simulated annealing
    T = 300
    best_h_sa = h.copy()
    best_c5_sa = compute_c5(best_h_sa)
    for t in range(1000):
        if time.time() - time_start > time_remaining - 100:
            break
        std = max(0.05, 0.25 * (1 - t / 1000))
        h_new = perturb(best_h_sa, std)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5_sa:
            best_h_sa = h_new
            best_c5_sa = c5_new
        delta = c5_new - best_c5_sa
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h_sa = h_new
        T *= 0.99

    # Local refinement
    result = opt.minimize(
        surrogate_objective,
        best_h_sa,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 200}
    )
    h_refined = result.x
    c5_refined = compute_c5(h_refined)
    if c5_refined < best_c5_sa:
        best_h_sa = h_refined
        best_c5_sa = c5_refined

    # Random restarts
    for _ in range(5):
        if time.time() - time_start > time_remaining - 100:
            break
        h_restart = perturb(best_h_sa, 0.1)
        c5_restart = compute_c5(h_restart)
        if c5_restart < best_c5_sa:
            best_h_sa = h_restart
            best_c5_sa = c5_restart

    # Final check
    best_h_sa = np.clip(best_h_sa, 0, 1)
    if abs(np.sum(best_h_sa) - n_points / 2) > 1e-6:
        best_h_sa = best_h_sa * (n_points / 2) / np.sum(best_h_sa)
    best_c5_sa = compute_c5(best_h_sa)

    return best_h_sa, best_c5_sa, n_points
