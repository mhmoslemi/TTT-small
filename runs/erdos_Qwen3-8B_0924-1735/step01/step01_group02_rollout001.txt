import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 50  # Start with 50 points for better trade-off
    initial_h = initial_h_values
    if len(initial_h) != n_points:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 5 == 0 else 0.5 if i % 5 == 1 else 0
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.full_like(h, 0.5)
        return h_clipped * (n_points / 2) / total

    def objective(h):
        return compute_c5(h)

    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2, 'jac': lambda h: np.ones_like(h)}
    bounds = [(0, 1) for _ in range(n_points)]

    # Evaluate initial solution
    h = project(h)
    c5_value = compute_c5(h)
    if c5_value < best_c5:
        best_c5 = c5_value
        best_h = h.copy()

    # Bayesian optimization with random sampling
    def surr_func(h):
        return compute_c5(h)

    n_evals = min(200, int(budget_s * 0.6))
    samples = np.random.uniform(0, 1, (n_evals, n_points))
    samples = np.apply_along_axis(project, 1, samples)
    evaluations = np.array([surr_func(s) for s in samples])
    best_idx = np.argmin(evaluations)
    best_h = samples[best_idx].copy()
    best_c5 = evaluations[best_idx]

    # Local refinement
    for _ in range(10):
        if time.time() - time_start > time_remaining - 100:
            break
        h = best_h.copy()
        h = np.clip(h, 0, 1)
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        h = result.x
        h = project(h)
        c5_value = compute_c5(h)
        if c5_value < best_c5:
            best_c5 = c5_value
            best_h = h.copy()

    # Final adjustment
    h = best_h.copy()
    h = np.clip(h, 0, 1)
    if np.abs(np.sum(h) - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / np.sum(h)
    c5_value = compute_c5(h)
    if c5_value < best_c5:
        best_c5 = c5_value
        best_h = h.copy()

    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    c5_value = compute_c5(best_h)
    return best_h, c5_value, n_points
