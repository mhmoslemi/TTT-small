import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds
from scipy.optimize import minimize
from scipy.stats import norm
import matplotlib.pyplot as plt
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45  # Starting with the provided initial_h_values
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        initial_h = initial_h_values
    else:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    # Scale initial_h to satisfy the constraint
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return np.full_like(h, 0.5)
        return h_clipped * (n_points / 2) / total

    def objective(h):
        return compute_c5(h)

    def surrogate_objective(h):
        if np.sum(h) < n_points / 2 - 1e-6 or np.sum(h) > n_points / 2 + 1e-6:
            return np.inf
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def bayesian_optimization():
        nonlocal best_c5, best_h
        n_evals = 200
        X = []
        y = []
        bounds = np.array([[0, 1] for _ in range(n_points)])
        kernel = RBF(length_scale=1.0, length_scale_bounds=(1e-5, 1e5)) * C(constant_value=1.0)
        gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-5, n_restarts_optimizer=10)
        for _ in range(n_evals):
            if time.time() - time_start > time_remaining - 50:
                break
            if len(X) < 10:
                random_h = np.random.uniform(0, 1, n_points)
                random_h = project(random_h)
                X.append(random_h)
                y_val = compute_c5(random_h)
                y.append(y_val)
                continue
            gp.fit(X, y)
            mu, sigma = gp.predict(bounds, return_std=True)
            acq = mu - 1.2 * sigma
            idx = np.argmin(acq)
            candidate = bounds[idx]
            candidate = project(candidate)
            X.append(candidate)
            y_val = compute_c5(candidate)
            y.append(y_val)
            if y_val < best_c5:
                best_c5 = y_val
                best_h = candidate.copy()
        if len(X) > 0:
            X = np.array(X)
            y = np.array(y)
            gp.fit(X, y)

    def local_refinement():
        nonlocal best_c5, best_h
        for _ in range(10):
            if time.time() - time_start > time_remaining - 100:
                break
            h_candidate = best_h + np.random.uniform(-0.05, 0.05, n_points)
            h_candidate = project(h_candidate)
            c5_val = compute_c5(h_candidate)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_candidate.copy()

    bayesian_optimization()
    local_refinement()

    h = best_h
    h = project(h)
    c5_value = compute_c5(h)
    if c5_value < best_c5:
        best_c5 = c5_value
        best_h = h.copy()

    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = project(best_h)
    best_h = np.clip(best_h, 0, 1)
    c5_value = compute_c5(best_h)
    return best_h, c5_value, n_points
