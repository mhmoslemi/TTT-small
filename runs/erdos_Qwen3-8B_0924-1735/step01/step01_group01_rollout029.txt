import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 120
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.zeros(n_points)
        block_size = 2.0 / 10
        for i in range(n_points):
            if i * dx < block_size * 5:
                initial_h[i] = 1.0
            else:
                initial_h[i] = 0.0
        initial_h = np.random.permutation(initial_h)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        initial_h = np.clip(initial_h, 0, 1)
    
    def objective(params):
        h = params.reshape(-1)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    def constraint(params):
        h = params.reshape(-1)
        return np.sum(h) - n_points / 2

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    initial_guess = initial_h.copy()
    initial_guess = initial_guess.reshape(-1)

    def pso_run():
        import pyswarms as ps
        options = {'c1': 1.5, 'c2': 1.5, 'w': 0.8, 'p': 50}
        bounds = (np.zeros(n_points), np.ones(n_points))
        
        def fitness(x):
            x = np.clip(x, 0, 1)
            x = x * (n_points / 2) / np.sum(x)
            cross_corr = np.correlate(x, 1 - x, mode='full')
            return np.max(cross_corr) * dx
        
        optimizer = ps.pso(fitness, bounds=bounds, options=options, n_iterations=100, verbose=False)
        best_pos = optimizer.cost_function[1]
        return best_pos

    start_time = time.time()
    h_opt = initial_h.copy()
    best_c5 = objective(initial_h)
    elapsed = 0

    try:
        h_opt = pso_run()
        best_c5 = objective(h_opt)
        elapsed = time.time() - start_time
    except:
        pass

    if elapsed >= budget_s:
        return (initial_h, best_c5, n_points)

    return (h_opt, best_c5, n_points)
