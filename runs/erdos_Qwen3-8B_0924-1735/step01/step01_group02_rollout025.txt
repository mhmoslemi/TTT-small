import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        if len(h) != n_points:
            h = np.zeros(n_points)
            for i in range(n_points):
                h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    h = h / np.sum(h) * n_points / 2
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    time_start = time.time()
    time_remaining = budget_s

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    cons = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)}
    bounds = [(0, 1) for _ in range(n_points)]

    def annealing_step(h, T):
        noise = np.random.normal(0, 0.1, size=n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            h = h * (n_points / 2) / sum_h
        h = np.clip(h, 0, 1)
        c5_new = compute_c5(h)
        c5_current = compute_c5(best_h)
        if c5_new < best_c5:
            best_h[:] = h[:]
            best_c5 = c5_new
        else:
            delta = c5_new - c5_current
            if np.random.rand() < np.exp(-delta / T):
                best_h[:] = h[:]
                best_c5 = c5_new

    T = 300
    for _ in range(1000):
        if time.time() - time_start > time_remaining - 100:
            break
        annealing_step(best_h, T)
        T *= 0.95

    def refine_solution(h):
        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        h_refined = result.x
        h_refined = np.clip(h_refined, 0, 1)
        sum_h = np.sum(h_refined)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            h_refined = h_refined * (n_points / 2) / sum_h
        h_refined = np.clip(h_refined, 0, 1)
        c5_refined = compute_c5(h_refined)
        return h_refined, c5_refined

    for _ in range(5):
        if time.time() - time_start > time_remaining - 100:
            break
        h_candidate = best_h + np.random.normal(0, 0.05, size=n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h_candidate)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            h_candidate = h_candidate * (n_points / 2) / sum_h
        h_candidate = np.clip(h_candidate, 0, 1)
        h_refined, c5_refined = refine_solution(h_candidate)
        if c5_refined < best_c5:
            best_h[:] = h_refined[:]
            best_c5 = c5_refined

    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        best_h = best_h * (n_points / 2) / sum_h
    best_h = np.clip(best_h, 0, 1)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
