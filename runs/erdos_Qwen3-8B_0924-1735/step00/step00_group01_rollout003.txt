import numpy as np
import scipy.optimize as optimize
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else None
    n_points = 512
    dx = 2.0 / n_points
    
    if initial_h_values is not None:
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 16 == 0:
                h[i] = 1.0
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        c5 = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(c5)
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = optimize.minimize(objective, h.flatten(), method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    h_best = result.x.reshape(n_points)
    
    def evaluate_c5(h):
        c5 = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(c5)
    
    c5_bound = evaluate_c5(h_best)
    
    return (h_best, c5_bound, n_points)
