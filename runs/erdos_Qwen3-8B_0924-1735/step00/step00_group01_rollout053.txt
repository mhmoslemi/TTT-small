import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    import numpy as np
    import scipy.signal
    import time

    # Choose n_points based on the current trend and available time
    n_points = 1024
    n = n_points
    dx = 2.0 / n

    # Initialize h
    initial_h_values = np.array([0.5] * n)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n)
        h = (h * 2 - 1) * 0.5 + 0.5
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
    h = np.clip(h, 0, 1)

    # Prepare the lags
    lags = np.arange(-n + 1, n)
    max_lag = n - 1

    # Optimization parameters
    learning_rate = 0.05
    max_iterations = 1000
    early_stop_threshold = 1e-6
    decay_rate = 0.98
    time_limit = budget_s

    best_c5 = np.inf
    best_h = h.copy()

    start_time = time.time()

    for t in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > time_limit:
            break

        # Compute overlap integrals for all lags
        overlaps = []
        for k in lags:
            if k >= 0:
                shifted = np.roll(h, k)
            else:
                shifted = np.roll(h, k + n)
            overlap = np.sum(h * (1 - shifted)) * dx
            overlaps.append(overlap)

        c5 = np.max(overlaps)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Early stop if no significant improvement
        if t > 0 and abs(c5 - best_c5) < early_stop_threshold:
            break

        # Find the worst lag
        k_max_idx = np.argmax(overlaps)
        k_max = lags[k_max_idx]

        # Compute gradient for the worst lag
        if k_max >= 0:
            shifted = np.roll(h, k_max)
        else:
            shifted = np.roll(h, k_max + n)
        gradient = np.zeros(n)
        for i in range(n):
            if k_max >= 0:
                j = (i + k_max) % n
            else:
                j = (i + k_max + n) % n
            gradient[i] = (1 - shifted[i]) * dx if k_max >= 0 else 0
            if i >= abs(k_max):
                gradient[i] -= h[i - abs(k_max)] * dx

        # Update h
        h = h - learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)

        learning_rate *= decay_rate

    # Final evaluation
    final_c5 = np.inf
    for k in lags:
        if k >= 0:
            shifted = np.roll(best_h, k)
        else:
            shifted = np.roll(best_h, k + n)
        overlap = np.sum(best_h * (1 - shifted)) * dx
        final_c5 = np.min([final_c5, overlap])

    return (best_h, final_c5, n)
