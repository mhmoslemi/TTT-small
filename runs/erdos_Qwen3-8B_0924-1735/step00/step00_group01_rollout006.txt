import numpy as np
import numpy.random as npr
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = 512
    np.random.seed(seed)
    dx = 2.0 / n_points
    initial_h = None

    # Check if initial_h_values is available
    if 'initial_h_values' in globals():
        initial_h = globals()['initial_h_values']
        # Normalize initial_h to [0,1] and scale to meet the sum constraint
        initial_h = (initial_h - np.min(initial_h)) / (np.max(initial_h) - np.min(initial_h))
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    else:
        # Generate a random initial guess
        initial_h = np.random.rand(n_points)
        initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h / np.sum(initial_h) * (n_points / 2)

    population = [initial_h.copy()]
    for _ in range(49):
        noise = np.random.normal(0, 0.1, n_points)
        temp = initial_h + noise
        temp = np.clip(temp, 0, 1)
        temp = temp / np.sum(temp) * (n_points / 2)
        temp = np.clip(temp, 0, 1)
        temp = temp / np.sum(temp) * (n_points / 2)
        population.append(temp)

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h, dx)

    start_time = time.time()
    generation = 0
    mutation_std = 0.02

    while time.time() - start_time < budget_s:
        if generation > 0:
            # Evaluate all candidates
            scores = [compute_c5(h, dx) for h in population]
            # Select top 20 candidates for breeding
            indices = np.argsort(scores)[:20]
            selected = [population[i] for i in indices]
            # Generate offspring
            offspring = []
            for _ in range(10):
                if len(selected) >= 2:
                    parent1, parent2 = np.random.choice(selected, 2, replace=False)
                    child = np.random.rand(n_points)
                    for i in range(n_points):
                        if np.random.rand() < 0.5:
                            child[i] = parent1[i]
                        else:
                            child[i] = parent2[i]
                else:
                    parent = selected[0]
                    child = parent.copy()
                # Mutation
                noise = np.random.normal(0, mutation_std, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = child / np.sum(child) * (n_points / 2)
                child = np.clip(child, 0, 1)
                child = child / np.sum(child) * (n_points / 2)
                offspring.append(child)
            # Replace population
            population = selected + offspring
            # Update best
            for h in population:
                c5 = compute_c5(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            # Check for stagnation
            if generation > 100 and np.abs(best_c5 - compute_c5(best_h, dx)) < 1e-4:
                mutation_std *= 0.9
        generation += 1

    # Final evaluation
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    h = h.astype(float)
    h_pad = np.zeros(2 * len(h) - 1)
    h_pad[:len(h)] = h
    h_pad[len(h):] = 1 - h
    corr = np.correlate(h_pad, h_pad, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr
