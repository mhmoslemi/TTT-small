import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Start with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = np.roll(h, 1)
        h = h / (np.sum(h) / (n_points / 2))
    
    def objective(x):
        h_new = np.clip(x, 0, 1)
        h_new = h_new / (np.sum(h_new) / (n_points / 2))
        corr = np.correlate(h_new, 1 - h_new, mode='full') * dx
        return np.max(corr)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    start_time = time.time()
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        return (h, np.max(np.correlate(h, 1 - h, mode='full') * dx), n_points)
    
    h_opt = np.clip(result.x, 0, 1)
    h_opt = h_opt / (np.sum(h_opt) / (n_points / 2))
    
    c5_bound = np.max(np.correlate(h_opt, 1 - h_opt, mode='full') * dx)
    
    return (h_opt, c5_bound, n_points)
