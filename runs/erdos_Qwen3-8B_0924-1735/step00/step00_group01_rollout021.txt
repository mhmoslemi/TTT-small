import time

def run(seed=42, budget_s=600, **kwargs):

    # Default parameters
    n_points = 512
    max_iter = 1000
    learning_rate = 0.05
    decay_rate = 0.95
    early_stop_threshold = 1e-6
    n_restarts = 3
    
    best_h = None
    best_c5 = float('inf')
    
    def generate_initial_h(n_points):
        h = np.random.uniform(0, 1, n_points)
        h -= h.mean()
        h = np.clip(h, 0, 1)
        h /= h.sum()
        h *= n_points / 2
        return h
    
    def compute_overlap(h, n_points):
        dx = 2.0 / n_points
        overlap = np.zeros(2 * n_points - 1)
        for k in range(1, 2 * n_points - 1):
            h_shifted = np.roll(h, k)
            overlap[k] = np.sum(h * (1 - h_shifted)) * dx
        return overlap
    
    def compute_gradient(h, lag_idx, n_points):
        dx = 2.0 / n_points
        grad = np.zeros(n_points)
        for i in range(n_points):
            k = lag_idx
            i_k = (i + k) % n_points
            grad[i] += (1 - h[i_k]) * dx
            i_k_neg = (i - k) % n_points
            grad[i] -= h[i_k_neg] * dx
        return grad
    
    def update_h(h, grad, learning_rate):
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / h_new.sum() * (n_points / 2)
