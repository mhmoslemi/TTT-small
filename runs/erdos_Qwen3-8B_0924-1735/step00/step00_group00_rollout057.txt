def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = h.copy()
    max_iter = 400
    learning_rate = 0.05
    rho = 0.5
    lambda_c = 0.0
    last_improvement = 0
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        if iteration > 100 and last_improvement > 20:
            break
        lags = np.random.choice(range(1, n_points), size=10, replace=False)
        current_max = -1
        selected_k = 0
        for k in lags:
            idx = (np.arange(n_points) + k) % n_points
            overlap = np.dot(h, 1 - h[idx]) * dx
            if overlap > current_max:
                current_max = overlap
                selected_k = k
        k = selected_k
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] += (1 - h[j]) * dx
            j = (i - k) % n_points
            grad[i] -= h[j] * dx
        h += learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h != 0:
            h *= (n_points / 2) / sum_h
        error = np.sum(h) - n_points / 2
        lambda_c += rho * error
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
    
    h_values = best_h
    c5_bound = best_c5
    return h_values, c5_bound, n_points
