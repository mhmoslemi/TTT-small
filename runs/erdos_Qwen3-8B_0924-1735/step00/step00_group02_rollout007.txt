import numpy as np
import random
import time
import scipy.optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 100  # Start with 100 samples for balance between speed and accuracy
    dx = 2.0 / n_points
    if initial_h_values is not None and len(initial_h_values) == n_points:
        h = np.array(initial_h_values, dtype=np.float64)
        h /= np.sum(h)
        h *= n_points / 2
    else:
        h = np.random.rand(n_points)
        h /= np.sum(h)
        h *= n_points / 2
    best_c5 = np.inf
    best_h = h.copy()

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h /= np.sum(h)
        h *= n_points / 2
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population = [h.copy() for _ in range(50)]
    best_h = np.array(population[0], dtype=np.float64)
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = np.array([evaluate(ind) for ind in population])
        # Select elites
        elite_idx = np.argsort(fitness)[:int(len(population)*0.2)]
        elites = np.array([population[i] for i in elite_idx])
        # Crossover
        new_pop = []
        for _ in range(len(population)):
            p1 = random.choice(elites)
            p2 = random.choice(population)
            split = np.random.randint(1, n_points - 1)
            child = np.concatenate([p1[:split], p2[split:]])
            new_pop.append(child)
        # Mutation
        for i in range(len(new_pop)):
            if np.random.rand() < 0.3:
                idxs = np.random.choice(n_points, 3, replace=False)
                for idx in idxs:
                    new_pop[i][idx] += np.random.laplace(0, 0.1)
                    new_pop[i][idx] = np.clip(new_pop[i][idx], 0, 1)
        # Normalize
        for i in range(len(new_pop)):
            total = np.sum(new_pop[i])
            new_pop[i] = np.array(new_pop[i], dtype=np.float64)
            new_pop[i] /= total
            new_pop[i] *= n_points / 2
        # Update population
        population = new_pop
        # Update best
        for ind in population:
            c5 = evaluate(ind)
            if c5 < best_c5:
                best_c5 = c5
                best_h = ind.copy()
        # Early stopping
        if best_c5 < 0.3807:
            break

    # Final validation
    h_final = np.clip(best_h, 0, 1)
    h_final /= np.sum(h_final)
    h_final *= n_points / 2
    final_c5 = evaluate(h_final)
    return h_final, final_c5, n_points
