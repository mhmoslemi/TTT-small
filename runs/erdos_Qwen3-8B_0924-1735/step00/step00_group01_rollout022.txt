```python
import numpy as np
import scipy.signal
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 512
    initial_h_values = np.random.rand(n_points)
    initial_h_values = (initial_h_values * 2 - 1) * 0.5 + 0.5
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    h = initial_h_values.copy()
    
    def evaluate(h):
        dx = 2.0 / n_points
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def gradient(h, k_max):
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] += (1 - h[i + k_max]) * dx
            if i - k_max >= 0:
                grad[i] -= h[i - k_max] * dx
        return grad
    
    best_c5 = evaluate(h)
    best_h = h.copy()
    
    max_iterations = 500
    learning_rate = 0.05
    start_time = time.time()
    
    for t in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        k_max = np.argmax(scipy.signal.correlate(h, 1 - h, mode='full'))
        grad = gradient(h, k_max)
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    h = best_h
    dx = 2.0 / n_points
    corr = scipy.signal.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return h, final_c5, n_points
```
