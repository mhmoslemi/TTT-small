import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Ensure initial_h_values is available
    if 'initial_h_values' not in globals():
        n_points = 45
        h = np.zeros(n_points)
        h[0:5] = 1.0
        h[10:15] = 1.0
        h[20:25] = 1.0
        h[30:35] = 1.0
        h[40:45] = 1.0
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale_factor = (n_points / 2) / sum_h
        h *= scale_factor
        initial_h_values = h
    else:
        initial_h_values = globals()['initial_h_values']
    
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    scale_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h *= scale_factor
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = np.inf
    
    start_time = time.time()
    
    def compute_q(h):
        n = len(h)
        q = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            idx = np.roll(np.arange(n), k)
            q[k] = np.dot(h, h[idx]) * dx
        return q
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        
        q = compute_q(h)
        min_idx = np.argmin(q)
        min_q = q[min_idx]
        k = min_idx - n_points + 1
        
        grad = np.zeros(n)
        for i in range(n):
            if 0 <= (i + k) < n:
                grad[i] += h[(i + k)] * dx
            if 0 <= (i - k) < n:
                grad[i] += h[(i - k)] * dx
        
        learning_rate = 0.05 * (1 / (1 + _ / 100))
        h += learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h *= scale_factor
        
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    h = best_h
    sum_h = np.sum(h)
    scale_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h *= scale_factor
    h = np.clip(h, 0, 1)
    
    # Final validation
    if np.any(h < 0) or np.any(h > 1):
        h = initial_h_values.copy()
        sum_h = np.sum(h)
        scale_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h *= scale_factor
        h = np.clip(h, 0, 1)
    
    final_c5 = compute_c5(h)
    
    return h, final_c5, n_points
