import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Choosing a manageable number for optimization
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.zeros(n_points)
        # Construct a basic initial configuration with 50% of the interval occupied
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        # Shuffle to break periodicity
        np.random.shuffle(h)
    # Ensure that the integral condition is satisfied
    total = np.sum(h)
    if total != n_points / 2:
        h = h / total * (n_points / 2)
    h = np.clip(h, 0, 1)
    
    def objective(x):
        nonlocal h
        h = x.copy()
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if np.abs(total - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)
    
    def constraint(x):
        nonlocal h
        h = x.copy()
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        return total - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    start_time = time.time()
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8},
        tol=1e-8
    )
    
    if time.time() - start_time > budget_s - 5:
        return (h, objective(h), n_points)
    
    final_h = result.x
    final_h = np.clip(final_h, 0, 1)
    total = np.sum(final_h)
    if np.abs(total - n_points / 2) > 1e-6:
        final_h = final_h / total * (n_points / 2)
    final_h = np.clip(final_h, 0, 1)
    c5_bound = objective(final_h)
    
    return (final_h, c5_bound, n_points)
