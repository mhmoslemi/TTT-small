import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    scaling_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h *= scaling_factor
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    max_iterations = 1000
    learning_rate = 0.05
    early_stop_count = 0
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        q = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            idx = np.roll(np.arange(n_points), k)
            q[k] = np.dot(h, h[idx]) * dx
        min_idx = np.argmin(q)
        min_q = q[min_idx]
        k = min_idx - n_points + 1
        grad = np.zeros(n_points)
        for i in range(n_points):
            if (i + k) % n_points < n_points:
                grad[i] += h[(i + k) % n_points] * dx
            if (i - k) % n_points < n_points:
                grad[i] += h[(i - k) % n_points] * dx
        h += learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h *= scale_factor
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            early_stop_count = 0
        else:
            early_stop_count += 1
        if early_stop_count >= 20:
            break
    
    if best_c5 < np.inf:
        h_values = best_h
    else:
        h_values = initial_h_values.copy()
        h_values = np.clip(h_values, 0, 1)
        sum_h = np.sum(h_values)
        scale_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h_values *= scale_factor
        h_values = np.clip(h_values, 0, 1)
        best_c5 = np.max(np.correlate(h_values, 1 - h_values, mode='full')) * dx
    return (h_values, best_c5, n_points)
