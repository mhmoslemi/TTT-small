```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial configuration (uses initial_h_values if provided)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 50
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if i % 3 == 0 else 0.5
        h = np.concatenate([h, np.array([0.5]*5)])
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Normalize to satisfy the integral constraint
    h = h / (np.sum(h) / (n_points / 2))
    
    def objective(x):
        h_flat = x.reshape(n_points)
        corr = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    start_time = time.time()
    result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5)
    elapsed_time = time.time() - start_time
    
    if elapsed_time >= budget_s:
        print("Time limit exceeded, returning initial configuration")
        return (h, objective(h), n_points)
    
    if result.success:
        h_opt = result.x
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt / (np.sum(h_opt) / (n_points / 2))
        c5_bound = objective(h_opt)
        return (h_opt, c5_bound, n_points)
    else:
        print("Optimization failed, returning initial configuration")
        return (h, objective(h), n_points)
```
