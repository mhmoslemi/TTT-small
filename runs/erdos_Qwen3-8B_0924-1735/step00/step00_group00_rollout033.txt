import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Assume initial values pre-loaded
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = np.inf
    max_iterations = 400
    learning_rate = 0.05
    rho = 0.5
    lambda_c = 0.0
    early_stop = 100
    iter_no_improve = 0

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        # Select critical lags for evaluation
        k_indices = np.random.choice(2 * n_points - 1, size=10, replace=False)
        current_max = 0.0
        selected_k = 0

        for k in k_indices:
            idx = (np.arange(n_points) + k) % n_points
            overlap = np.dot(h, 1 - h[idx]) * dx
            if overlap > current_max:
                current_max = overlap
                selected_k = k

        # Compute gradient for selected lag
        k = selected_k
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = (i + k) % n_points
            grad[i] += (1 - h[i_plus_k]) * dx
            i_minus_k = (i - k) % n_points
            grad[i] -= h[i_minus_k] * dx

        # Gradient descent update
        h += learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)

        # Constraint enforcement and scaling
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h *= scale

        # Update Lagrange multiplier
        error = sum_h - n_points / 2
        lambda_c += rho * error

        # Check for improvement
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
            iter_no_improve = 0
        else:
            iter_no_improve += 1
            if iter_no_improve > early_stop:
                break

    # Validate final solution
    if np.sum(best_h) < n_points / 2 - 1e-6:
        best_h = initial_h_values
        best_c5 = np.inf
    elif np.sum(best_h) > n_points / 2 + 1e-6:
        best_h = initial_h_values
        best_c5 = np.inf

    return best_h, best_c5, n_points

start_time = time.time()
