import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Assuming initial_h_values is provided
    n_points = 512
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = (h - np.min(h)) / (np.max(h) - np.min(h))
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    
    population_size = 50
    population = [h.copy() for _ in range(population_size)]
    best_h = h
    best_c5 = float('inf')
    mutation_std = 0.02
    generation = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s and generation < 1000:
        # Evaluate population
        evaluations = []
        for h_candidate in population:
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate / np.sum(h_candidate) * (n_points / 2)
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate / np.sum(h_candidate) * (n_points / 2)
            corr = np.correlate(h_candidate, 1 - h_candidate, mode="full")
            c5 = np.max(corr) * dx
            evaluations.append((c5, h_candidate))
        
        # Sort by C5
        evaluations.sort()
        best_c5, best_h = evaluations[0]
        
        # Update best solution
        if best_c5 < best_c5:
            best_c5 = best_c5
            best_h = best_h
        
        # Select top 20 candidates for breeding
        selected = evaluations[:20]
        selected_c5s, selected_h = zip(*selected)
        selected_h = list(selected_h)
        
        # Generate offspring
        offspring = []
        for _ in range(10):
            parent1, parent2 = np.random.choice(selected_h, 2, replace=False)
            child = np.copy(parent1)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = parent2[i]
            child = np.clip(child, 0, 1)
            child = child / np.sum(child) * (n_points / 2)
            child = np.clip(child, 0, 1)
            child = child / np.sum(child) * (n_points / 2)
            # Apply mutation
            mutation = np.random.normal(0, mutation_std, n_points)
            child += mutation
            child = np.clip(child, 0, 1)
            child = child / np.sum(child) * (n_points / 2)
            child = np.clip(child, 0, 1)
            child = child / np.sum(child) * (n_points / 2)
            offspring.append(child)
        
        # Replace old population
        population = selected_h + offspring
        population = population[:population_size]
        
        # Check for improvement
        if generation % 10 == 0 and np.abs(best_c5 - evaluations[0][0]) < 1e-4:
            mutation_std *= 0.7
        
        generation += 1
    
    # Final evaluation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    corr = np.correlate(best_h, 1 - best_h, mode="full")
    c5_bound = np.max(corr) * dx
    
    return (best_h, c5_bound, n_points)
