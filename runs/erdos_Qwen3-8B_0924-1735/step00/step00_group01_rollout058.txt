import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 512
    dx = 2.0 / n_points
    
    def validate_and_rescale(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros(n_points)
        h = h / total * (n_points / 2)
        return h
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def generate_initial_population(base_h, n_pop=50):
        pop = [base_h.copy()]
        for _ in range(n_pop - 1):
            noise = np.random.uniform(-0.1, 0.1, n_points)
            candidate = base_h + noise
            candidate = validate_and_rescale(candidate)
            pop.append(candidate)
        return pop
    
    def crossover(parent1, parent2):
        child = np.zeros(n_points)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        child = validate_and_rescale(child)
        return child
    
    def mutate(parent, std=0.02):
        noise = np.random.normal(0, std, n_points)
        child = parent + noise
        child = validate_and_rescale(child)
        return child
    
    def evolve(population, mutation_std=0.02, elite_count=10):
        scores = [compute_c5(h) for h in population]
        best_idx = np.argmin(scores)
        best_h = population[best_idx]
        best_c5 = scores[best_idx]
        
        selected = np.argsort(scores)[:elite_count]
        next_gen = [population[i] for i in selected]
        
        while len(next_gen) < len(population):
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child = crossover(population[parent1], population[parent2])
            next_gen.append(child)
        
        for i in range(len(next_gen) - len(population) + elite_count):
            parent = np.random.choice(selected)
            child = mutate(population[parent], mutation_std)
            next_gen.append(child)
        
        return next_gen, best_c5, best_h
    
    if initial_h_values is not None:
        base_h = validate_and_rescale(initial_h_values)
    else:
        base_h = np.random.uniform(0, 1, n_points)
        base_h = validate_and_rescale(base_h)
    
    population = generate_initial_population(base_h)
    best_h = base_h
    best_c5 = compute_c5(best_h)
    time_used = 0
    
    start_time = time.time()
    for gen in range(1000):
        if time_used > budget_s:
            break
        population, best_c5, best_h = evolve(population, mutation_std=0.02)
        time_used = time.time() - start_time
    
    best_h = validate_and_rescale(best_h)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_points
