import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Check if initial_h_values is available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Construct a starting vector if no initial is provided
        n_points = 45
        h = np.zeros(n_points)
        h[:(n_points//2)] = 1.0
        h = np.roll(h, n_points//4)
        h = np.clip(h, 0.0, 1.0)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    sum_h = np.sum(h)
    if abs(sum_h - target_sum) > 1e-6:
        scale = target_sum / sum_h
        h *= scale
    
    best_h = h.copy()
    best_c5 = np.inf
    learning_rate = 0.05
    max_iterations = 400
    rho = 0.5
    lambda_c = 0.0
    early_stop_counter = 0
    max_early_stop = 20
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Sample critical lags
        k_indices = np.random.choice(2 * n_points - 1, size=10, replace=False)
        current_max = 0.0
        selected_k = 0
        
        # Evaluate selected lags
        for k in k_indices:
            idx = (np.arange(n_points) + k) % n_points
            overlap = np.dot(h, 1.0 - h[idx]) * dx
            if overlap > current_max:
                current_max = overlap
                selected_k = k
        
        # Compute gradient for selected lag
        k = selected_k
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Forward influence
            if (i + k) % n_points < n_points:
                gradient[i] += (1.0 - h[(i + k) % n_points]) * dx
            # Backward influence
            if (i - k) % n_points < n_points:
                gradient[i] -= h[(i - k) % n_points] * dx
        
        # Update h
        h += learning_rate * gradient
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = target_sum / sum_h
            h *= scale
        
        # Update Lagrange multiplier
        error = sum_h - target_sum
        lambda_c += rho * error
        
        # Early stopping
        if iteration > 100 and np.abs(current_max - best_c5) < 1e-6:
            early_stop_counter += 1
            if early_stop_counter >= max_early_stop:
                break
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
    
    # Final validation
    sum_best_h = np.sum(best_h)
    if abs(sum_best_h - target_sum) > 1e-6:
        scale = target_sum / sum_best_h
        best_h *= scale
    
    # Recompute C5 using the original method
    corr = scipy.signal.correlate(best_h, 1.0 - best_h, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (best_h, c5_bound, n_points)
