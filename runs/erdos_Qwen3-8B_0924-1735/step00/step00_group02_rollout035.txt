```python
import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Base parameters
    n_points = 100  # Start with 100 samples for balance between speed and precision
    dx = 2.0 / n_points
    initial_h_values = np.random.rand(n_points) * 2 - 1  # Start with basic random values for testing
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values / np.sum(initial_h_values) * (n_points / 2)
    
    # Population setup
    population_size = 50
    population = [initial_h_values.copy() for _ in range(population_size)]
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    # Genetic algorithm parameters
    mutation_rate = 0.3
    tournament_size = 5
    elitism_ratio = 0.2
    generations = 100
    diversity_injection_interval = 20
    
    for generation in range(generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append((c5, h))
        
        # Sort by fitness
        fitness.sort()
        best_fitness, best_h = fitness[0]
        
        # Update best solution
        if best_fitness < best_c5:
            best_c5 = best_fitness
            best_h = best_h.copy()
        
        # Elitism
        elites = int(elitism_ratio * population_size)
        elite_indices = np.argsort([f[0] for f in fitness])[:elites]
        elites = [fitness[i][1] for i in elite_indices]
        
        # Reproduction
        new_population = elites.copy()
        
        for _ in range(population_size - elites):
            # Tournament selection
            participants = np.random.choice(population, size=tournament_size)
            selected = min(participants, key=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx)
            
            # Crossover (single-point)
            parent1 = selected
            parent2 = np.random.choice(population)
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation_scale = 0.1
                indices = np.random.choice(n_points, size=int(mutation_rate * n_points), replace=False)
                child[indices] += np.random.laplace(0, mutation_scale, size=len(indices))
                child = np.clip(child, 0, 1)
                child = child / np.sum(child) * (n_points / 2)
            
            new_population.append(child)
        
        # Normalize and update population
        population = new_population
        population = [np.clip(h, 0, 1) for h in population]
        population = [h / np.sum(h) * (n_points / 2) for h in population]
        
        # Inject diversity
        if generation % diversity_injection_interval == 0:
            for _ in range(5):
                h = np.random.rand(n_points)
                h = np.clip(h, 0, 1)
                h = h / np.sum(h) * (n_points / 2)
                population.append(h)
        
        # Early stopping
        if best_c5 < 0.3807:
            break
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    
    return (best_h, best_c5, n_points)
```
