import numpy as np
import scipy.optimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 512
    dx = 2.0 / n_points
    if initial_h_values is not None:
        if len(initial_h_values) != n_points:
            initial_h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    def update_h(h, learning_rate):
        corr = np.correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i - max_k) % n_points
            grad[i] -= h[i] * (1 - h[j]) * dx
            j = (i + max_k) % n_points
            grad[i] += (1 - h[i]) * h[j] * dx
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    def perturb_h(h, perturbation_rate):
        for i in range(n_points):
            if np.random.rand() < perturbation_rate:
                h[i] += np.random.uniform(-0.1, 0.1)
                h[i] = np.clip(h[i], 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    learning_rate = 0.05
    iteration_limit = 1000
    iteration = 0
    perturbation_rate = 0.05
    decay_factor = 0.95
    start_time = time.time()
    
    while iteration < iteration_limit and time.time() - start_time < budget_s:
        h = update_h(h, learning_rate)
        h = perturb_h(h, perturbation_rate)
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            learning_rate *= 1.1
        else:
            learning_rate *= decay_factor
        iteration += 1
    
    best_c5 = objective(best_h)
    h = best_h.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    return (h, best_c5, n_points)
