import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Assuming initial_h_values is provided as a .npy file
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_required = n_points / 2
    
    # Scale initial guess to meet sum constraint
    h = initial_h_values / np.sum(initial_h_values) * sum_required
    h = np.clip(h, 0, 1)
    
    def objective_surrogate(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.log(np.sum(np.exp(100 * (corr - 1e-3))))
    
    def objective_exact(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - sum_required
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    start_time = time.time()
    for _ in range(10):
        if time.time() - start_time > budget_s * 0.7:
            break
        result = opt.minimize(objective_surrogate, h, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'maxiter': 100})
        h = result.x
        h = np.clip(h, 0, 1)
        h = h * sum_required / np.sum(h)
        
        # Apply small random perturbations every 100 iterations
        if _ % 10 == 0:
            noise = np.random.normal(0, 1e-3, size=n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * sum_required / np.sum(h)
    
    # Final evaluation
    h = np.clip(h, 0, 1)
    h = h * sum_required / np.sum(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    c5_bound = np.max(corr)
    
    return (h, c5_bound, n_points)
