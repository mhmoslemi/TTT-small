import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = evaluate_erdos_solution()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    target_sum = n_points / 2
    if np.abs(sum_h - target_sum) > 1e-6:
        h = h * (target_sum / sum_h)
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    if np.abs(sum_h - target_sum) > 1e-6:
        h = h * (target_sum / sum_h)
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    best_h = h.copy()
    best_c5 = np.inf
    max_iterations = 200
    learning_rate = 0.05
    rho = 1.0
    lambda_c = 0.0
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        lag_indices = np.random.choice(2 * n_points - 1, size=10, replace=False)
        current_max = 0.0
        selected_k = 0
        for k in lag_indices:
            if k < 0:
                k += n_points
            idx = (np.arange(n_points) + k) % n_points
            overlap = np.dot(h, 1 - h[idx]) * dx
            if overlap > current_max:
                current_max = overlap
                selected_k = k
        k = selected_k
        if k >= n_points:
            k -= n_points
        grad = np.zeros(n_points)
        for i in range(n_points):
            idx = (i + k) % n_points
            grad[i] += (1 - h[idx]) * dx
            idx = (i - k) % n_points
            grad[i] -= h[idx] * dx
        grad = np.clip(grad, -0.5, 0.5)
        h += learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if np.abs(sum_h - target_sum) > 1e-6:
            h = h * (target_sum / sum_h)
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        error = sum_h - target_sum
        lambda_c += rho * error
        if np.abs(error) > 1e-3:
            rho *= 1.5
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        if iteration > 100 and np.abs(current_max - best_c5) < 1e-6:
            break
    
    h = best_h
    sum_h = np.sum(h)
    if np.abs(sum_h - target_sum) > 1e-6:
        h = h * (target_sum / sum_h)
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    c5_bound = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    return (h, c5_bound, n_points)
