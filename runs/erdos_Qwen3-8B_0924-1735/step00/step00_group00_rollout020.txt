        cross_corr = np.correlate(h, 1 - h, mode='full')

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

            break

        k = max_idx - (n_points - 1)
        if k < 0 or k >= n_points:
            continue
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + k < n_points:
                grad[i] -= h[i + k]
            if i - k >= 0:
                grad[i] += (1 - h[i - k])
        h += learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    
    if np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx < best_c5:
