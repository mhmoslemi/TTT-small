import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    scaling_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h *= scaling_factor
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scaling_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h *= scaling_factor
    best_c5 = float('inf')
    best_h = h.copy()
    max_iterations = 1000
    learning_rate = 0.05
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        q = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            idx = np.arange(n_points)
            if k < n_points:
                idx = (idx + k) % n_points
            else:
                idx = (idx - (k - n_points)) % n_points
            q[k] = np.dot(h, h[idx]) * dx
        min_idx = np.argmin(q)
        k = min_idx - n_points + 1
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= (i + k) % n_points < n_points:
                grad[i] += h[(i + k) % n_points] * dx
            if 0 <= (i - k) % n_points < n_points:
                grad[i] += h[(i - k) % n_points] * dx
        h += learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scaling_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h *= scaling_factor
        current_c5 = np.max(scipy.signal.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    h_values = best_h
    c5_bound = best_c5
    return (h_values, c5_bound, n_points)
