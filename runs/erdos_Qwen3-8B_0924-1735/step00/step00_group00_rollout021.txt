            break

        if lag < 0:
            lag += n_points
        elif lag >= n_points:
            lag -= n_points
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if i + lag < n_points:
                gradient[i] += 1 - h[i + lag]
            if i - lag >= 0:
                gradient[i] -= h[i - lag]
        step_size = 0.01
        h += gradient * step_size
        h = np.clip(h, 0, 1)
        h = (h - np.min(h)) / (np.max(h) - np.min(h)) * (n_points / 2)
        h = h / np.sum(h) * (n_points / 2)
