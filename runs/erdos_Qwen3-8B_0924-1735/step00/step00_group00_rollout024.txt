import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Assume initial_h_values is pre-loaded
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    scaling_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h = initial_h_values.copy() * scaling_factor
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.inf
    elapsed = 0
    start_time = time.time()

    def compute_q(h):
        q = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            q_k = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                q_k += h[i] * h[j]
            q[k] = q_k * dx
        return q

    def compute_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    for iteration in range(1000):
        if elapsed > budget_s:
            break
        q = compute_q(h)
        min_idx = np.argmin(q)
        k = min_idx - n_points + 1
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + k < n_points:
                grad[i] += h[(i + k)] * dx
            if 0 <= i - k < n_points:
                grad[i] += h[(i - k)] * dx
        h += 0.05 * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h *= scale_factor
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        elapsed = time.time() - start_time

    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        best_h = best_h * (n_points / 2) / sum_h
        best_h = np.clip(best_h, 0, 1)
    return best_h, best_c5, n_points
