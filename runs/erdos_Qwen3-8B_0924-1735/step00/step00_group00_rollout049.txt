import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Check if initial_h_values is available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Use a simple starting point: evenly spaced 1s and 0s with 1s on even indices
        n_points = 45
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = np.clip(h, 0.0, 1.0)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Validate initial h
    if np.abs(np.sum(h) - target_sum) > 1e-6:
        h = np.clip(h, 0.0, 1.0)
        h = h * (target_sum / np.sum(h))
    
    best_h = h.copy()
    best_c5 = np.inf
    
    # Optimization parameters
    learning_rate = 0.05
    max_iterations = 400
    lag_history = []
    rho = 0.5
    lambda_c = 0.0
    decay_rate = 0.98
    early_stop = 50
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Select a subset of lags for evaluation
        lags = np.unique(np.arange(n_points) - np.arange(n_points))
        lags = np.sort(np.unique(lags))
        selected_lags = np.random.choice(lags, size=10, replace=False)
        
        # Evaluate current max overlap
        current_max = 0.0
        selected_k = 0
        for k in selected_lags:
            if k < 0:
                k += n_points
            idx = (np.arange(n_points) + k) % n_points
            overlap = np.dot(h, 1 - h[idx]) * dx
            if overlap > current_max:
                current_max = overlap
                selected_k = k
        
        # Compute gradient for selected k
        k = selected_k % n_points
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Contribution from h[i] * (1 - h[i + k])
            j = (i + k) % n_points
            grad[i] += (1 - h[j]) * dx
            # Contribution from h[i] * h[i - k] (from 1 - h[i - k])
            j = (i - k) % n_points
            grad[i] -= h[j] * dx
        
        # Update h
        h += learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        
        # Enforce integral constraint
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = (target_sum) / sum_h
            h *= scale
        
        # Update Lagrange multiplier
        error = sum_h - target_sum
        lambda_c += rho * error
        
        # Early stopping if no improvement
        if iteration > early_stop and np.abs(current_max - best_c5) < 1e-6:
            break
        
        # Store best solution
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        
        # Decay learning rate
        if iteration % 50 == 0:
            learning_rate *= decay_rate
    
    # Final validation
    sum_h = np.sum(best_h)
    if np.abs(sum_h - target_sum) > 1e-6:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = best_h * (target_sum / np.sum(best_h))
    
    # Recompute C5 using the exact method
    corr = scipy.signal.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (best_h.tolist(), c5_bound, n_points)
