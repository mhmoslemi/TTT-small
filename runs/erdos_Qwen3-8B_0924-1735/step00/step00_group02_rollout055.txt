import numpy as np
import random
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        h = (h / np.sum(h)) * (n_points / 2)
    else:
        h = np.random.rand(n_points)
        h = (h / np.sum(h)) * (n_points / 2)
    best_h = h.copy()
    best_c5 = np.inf
    
    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def evaluate(x):
        nonlocal best_c5, best_h
        c5 = objective(x)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()
        return c5
    
    def objective_min(x):
        return objective(x)
    
    def update_params(population):
        new_pop = []
        for i in range(0, len(population), 2):
            a, b = population[i], population[i+1]
            n = np.random.randint(0, n_points)
            child = np.concatenate([a[:n], b[n:]])
            mu = np.random.normal(0, 0.1, n_points)
            child += mu
            child = np.clip(child, 0, 1)
            child = (child / np.sum(child)) * (n_points / 2)
            new_pop.append(child)
        return new_pop
    
    def mutation(x):
        mu = np.random.normal(0, 0.1, n_points)
        x += mu
        x = np.clip(x, 0, 1)
        x = (x / np.sum(x)) * (n_points / 2)
        return x
    
    def crossover(p1, p2):
        n = np.random.randint(0, n_points)
        child = np.concatenate([p1[:n], p2[n:]])
        return child
    
    start_time = time.time()
    population = [h.copy() for _ in range(50)]
    for gen in range(100):
        fitness = [evaluate(ind) for ind in population]
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices[:20]]
        new_pop = []
        for _ in range(25):
            a, b = random.sample(population, 2)
            child = crossover(a, b)
            child = mutation(child)
            new_pop.append(child)
        population += new_pop
        if time.time() - start_time > budget_s:
            break
    return best_h, best_c5, n_points
