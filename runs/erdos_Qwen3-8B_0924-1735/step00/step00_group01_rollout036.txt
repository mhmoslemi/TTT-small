import scipy.signal
import time

        corr = scipy.signal.correlate(h, 1 - h, mode='full')

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if c5 < 1e-6:
            break
        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if k_max >= n_points:
                j = k_max - n_points
                grad[i] += (1 - h[(i + j) % n_points]) * dx
            else:
                j = k_max
                grad[i] += (1 - h[(i + j) % n_points]) * dx
            if k_max < n_points:
                j = k_max
                grad[i] -= h[(i - j) % n_points] * dx
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
