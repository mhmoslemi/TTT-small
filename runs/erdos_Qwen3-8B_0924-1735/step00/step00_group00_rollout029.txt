import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Check if initial_h_values is available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Default initial construction if parent construction is not available
        n_points = 45
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if i % 3 == 0 else 0.0
        h = h + np.random.rand(n_points) * 0.5
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    n_points = len(h)
    dx = 2.0 / n_points
    # Ensure the sum is correct
    h = h * (n_points / 2) / np.sum(h)
    # Precompute 1 - h
    one_minus_h = 1.0 - h
    # Objective function
    def objective(h_vec):
        h = np.array(h_vec, dtype=np.float64)
        # Ensure bounds
        h = np.clip(h, 0, 1)
        # Ensure sum constraint
        h = h * (n_points / 2) / np.sum(h)
        # Compute cross-correlation
        corr = np.correlate(h, one_minus_h, mode='full')
        # Compute maximum overlap
        return np.max(corr) * dx
    # Constraints
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = Bounds(0, 1)
    # Optimization
    start_time = time.time()
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 200})
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        return (h, objective(h), n_points)
    else:
        return (result.x, objective(result.x), n_points)
