import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Default n_points
    n_points = 512
    dx = 2.0 / n_points
    
    # Pre-imported initial_h_values
    if 'initial_h_values' in globals():
        h = (initial_h_values - np.min(initial_h_values)) / (np.max(initial_h_values) - np.min(initial_h_values))
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if i % 3 == 0 else 0
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h_norm = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h_norm, 1 - h_norm, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def repair(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    population_size = 50
    pop = [h.copy() for _ in range(population_size)]
    best_h = h.copy()
    best_c5 = evaluate(best_h)
    gen = 0
    mutation_std = 0.02
    elite_count = 10
    max_generations = 1000
    early_stop_threshold = 1e-4
    start_time = time.time()
    
    while gen < max_generations and time.time() - start_time < budget_s:
        # Evaluate population
        scores = [evaluate(ind) for ind in pop]
        best_idx = np.argmin(scores)
        best_c5 = scores[best_idx]
        best_h = pop[best_idx].copy()
        
        # Select top elites
        elite_indices = np.argsort(scores)[:elite_count]
        elites = [pop[i] for i in elite_indices]
        
        # Generate offspring
        offspring = []
        for _ in range(population_size - elite_count):
            # Crossover
            parent1, parent2 = np.random.choice(elite_indices, 2, replace=False)
            child = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = pop[parent1][i]
                else:
                    child[i] = pop[parent2][i]
            # Mutation
            child += np.random.normal(0, mutation_std, n_points)
            child = repair(child)
            offspring.append(child)
        
        # Combine elites and offspring
        pop = elites + offspring
        gen += 1
        
        # Early stopping
        if gen > 2 and np.abs(best_c5 - scores[best_idx]) < early_stop_threshold:
            mutation_std *= 0.8
            if mutation_std < 1e-5:
                break
    
    # Final evaluation
    best_h = repair(best_h)
    best_c5 = evaluate(best_h)
    
    # Final checks
    assert np.all(0 <= best_h <= 1)
    assert np.isclose(np.sum(best_h), n_points / 2, atol=1e-6)
    
    return (best_h, best_c5, n_points)
