import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 512
    dx = 2.0 / n_points
    
    if initial_h_values is not None:
        h = (initial_h_values - np.min(initial_h_values)) / (np.max(initial_h_values) - np.min(initial_h_values))
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    
    population_size = 50
    pop = [h.copy() for _ in range(population_size)]
    best_h = h.copy()
    best_c5 = float('inf')
    
    t_start = time.time()
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if not np.isclose(total, n_points / 2, atol=1e-6):
            h = h / total * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    for gen in range(1000):
        if time.time() - t_start > budget_s:
            break
        
        # Evaluate all candidates
        c5_values = [evaluate(h) for h in pop]
        
        # Track best
        best_idx = np.argmin(c5_values)
        current_c5 = c5_values[best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = pop[best_idx].copy()
        
        # Select top 20 for breeding
        idxs = np.argsort(c5_values)[:20]
        parents = [pop[i] for i in idxs]
        
        # Generate 10 offspring
        offspring = []
        for _ in range(10):
            if len(parents) >= 2:
                # Uniform crossover
                p1, p2 = random.sample(parents, 2)
                child = p1.copy()
                for i in range(n_points):
                    if np.random.rand() < 0.5:
                        child[i] = p2[i]
            else:
                # Random initialization
                child = np.random.rand(n_points)
                child = child / np.sum(child) * (n_points / 2)
                child = np.clip(child, 0, 1)
                child = child / np.sum(child) * (n_points / 2)
            
            # Apply mutation
            mutation = np.random.normal(0, 0.02, n_points)
            child += mutation
            child = np.clip(child, 0, 1)
            total = np.sum(child)
            if not np.isclose(total, n_points / 2, atol=1e-6):
                child = child / total * (n_points / 2)
            offspring.append(child)
        
        # Combine new population
        new_pop = parents + offspring
        np.random.shuffle(new_pop)
        pop = new_pop[:population_size]
    
    # Final evaluation of best
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    if not np.isclose(total, n_points / 2, atol=1e-6):
        best_h = best_h / total * (n_points / 2)
    best_c5 = evaluate(best_h)
    
    return (best_h, best_c5, n_points)
