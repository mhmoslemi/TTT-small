import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Check if initial_h_values is available
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        n_points = 45
        initial_h = np.zeros(n_points)
        offset = n_points // 2
        for i in range(n_points):
            if i < offset:
                initial_h[i] = 1.0
            else:
                initial_h[i] = 0.0
        initial_h = np.concatenate([initial_h[offset:], initial_h[:offset]])
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        initial_h *= (n_points / 2) / sum_h

    # Initialize
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = np.inf
    max_iterations = 400
    learning_rate = 0.05
    rho = 0.5
    lambda_c = 0.0
    early_stop_counter = 0
    start_time = time.time()

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Randomly select 10 lags to evaluate
        lags = np.random.choice(np.arange(1, n_points), size=10, replace=False)
        current_max = 0.0
        selected_k = 0
        for k in lags:
            idx = (np.arange(n_points) + k) % n_points
            overlap = np.dot(h, 1 - h[idx]) * dx
            if overlap > current_max:
                current_max = overlap
                selected_k = k
        # Compute gradient for selected lag
        k = selected_k
        idx = (np.arange(n_points) + k) % n_points
        grad = np.zeros(n_points)
        grad += (1 - h[idx]) * dx
        grad -= h * dx
        # Update h
        h += learning_rate * grad
        # Enforce bounds
        h = np.clip(h, 0.0, 1.0)
        # Enforce integral constraint
        sum_h = np.sum(h)
        if sum_h != 0:
            h *= (n_points / 2) / sum_h
        # Update Lagrange multiplier
        error = sum_h - n_points / 2
        lambda_c += rho * error
        # Early stopping
        if iteration > 100 and np.abs(current_max - best_c5) < 1e-6:
            early_stop_counter += 1
            if early_stop_counter > 20:
                break
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        # Adaptive learning rate and rho
        if iteration % 50 == 0 and iteration > 50:
            learning_rate *= 0.98
            rho *= 1.5
        # Random perturbation
        if iteration % 50 == 0:
            perturb = np.random.normal(0, 0.05, n_points)
            h += perturb
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            if sum_h != 0:
                h *= (n_points / 2) / sum_h

    # Final check
    h_values = best_h.copy()
    sum_h = np.sum(h_values)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        h_values *= (n_points / 2) / sum_h
    c5_bound = np.max(np.correlate(h_values, 1 - h_values, mode='full')) * dx
    return (h_values, c5_bound, n_points)
