    learning_rate = 0.05

        if elapsed > budget_s:

        q = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            idx = np.roll(np.arange(n_points), k)
            q[k] = np.dot(h, h[idx]) * dx
        
        # Find the minimum Q_k
        min_idx = np.argmin(q)
        min_q = q[min_idx]

        k = min_idx - n_points + 1

                grad[i] += h[(i + k)] * dx
            if 0 <= (i - k) < n_points:
                grad[i] += h[(i - k)] * dx
