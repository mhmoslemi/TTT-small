import numpy as np
import random
import time
import scipy.optimize as opt
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.rand(n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Fitness function
    def fitness(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization with CVXPY
    h = cp.Variable(n_points)
    constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]
    obj = cp.Minimize(cp.Max(cp.multiply(h, 1 - cp.roll(h, 1)), axis=0).max())
    prob = cp.Problem(obj, constraints)
    result = prob.solve(solver=cp.ECOS, verbose=False)
    
    best_h = h.value
    best_c5 = fitness(best_h)
    
    # Try with different n_points if time allows
    if time.time() - start_time < budget_s * 0.8:
        n_points = 150
        dx = 2.0 / n_points
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]
        obj = cp.Minimize(cp.Max(cp.multiply(h, 1 - cp.roll(h, 1)), axis=0).max())
        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver=cp.ECOS, verbose=False)
        best_h = h.value
        best_c5 = fitness(best_h)
    
    if time.time() - start_time < budget_s * 0.95:
        n_points = 200
        dx = 2.0 / n_points
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]
        obj = cp.Minimize(cp.Max(cp.multiply(h, 1 - cp.roll(h, 1)), axis=0).max())
        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver=cp.ECOS, verbose=False)
        best_h = h.value
        best_c5 = fitness(best_h)
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = fitness(best_h)
    
    return best_h, best_c5, n_points
