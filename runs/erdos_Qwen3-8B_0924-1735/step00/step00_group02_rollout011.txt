import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h.npy')  # Assume this is preloaded
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    scaled_h = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    h = scaled_h.copy()

    def objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full") * dx
        return np.max(corr)

    def surrogate(x, beta=100, eps=1e-6):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full") * dx
        return np.log(np.sum(np.exp(beta * (corr - eps))))

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    result = opt.minimize(objective, h, bounds=bounds, constraints=constraints, method='L-BFGS-B', options={'maxiter': 200})
    h = result.x

    if np.max(np.correlate(h, 1 - h, mode="full") * dx) > 0.38:
        result = opt.minimize(surrogate, h, bounds=bounds, constraints=constraints, method='L-BFGS-B', options={'maxiter': 200})
        h = result.x

    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    c5_bound = np.max(corr)
    return (h, c5_bound, n_points)
