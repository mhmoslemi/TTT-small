import numpy as np
import scipy.signal
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    # Check if initial_h_values is available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Generate a starting vector for n_points=45
        n_points = 45
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        h[0:15] = 1.0
        h[15:30] = 0.0
        h[30:45] = 1.0
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale_factor = (n_points / 2) / sum_h
        h *= scale_factor
        h = np.clip(h, 0, 1)
        h = np.round(h, 5)

    n_points = len(h)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    scaling_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h = h * scaling_factor
    h = np.clip(h, 0, 1)

    max_iterations = 1000
    learning_rate = 0.05
    best_c5 = np.inf
    best_h = h.copy()
    start_time = time.time()

    for iteration in range(max_iterations):
        # Compute all Q_k values
        q = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            idx = np.roll(np.arange(n_points), k)
            q[k] = np.dot(h, h[idx]) * dx

        # Find the current minimum Q_k
        min_idx = np.argmin(q)
        min_q = q[min_idx]
        if min_q < 1e-7:
            min_q = 1e-7  # Avoid division by zero

        # Compute the gradient for the worst-case lag
        k = min_idx - n_points + 1
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + k < n_points:
                grad[i] += h[(i + k) % n_points] * dx
            if 0 <= i - k < n_points:
                grad[i] += h[(i - k) % n_points] * dx

        # Update the function values
        h = h + learning_rate * grad
        h = np.clip(h, 0, 1)

        # Rescale to maintain the integral constraint
        sum_h = np.sum(h)
        scale_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h = h * scale_factor
        h = np.clip(h, 0, 1)

        # Compute current C5 value
        current_c5 = np.max(scipy.signal.correlate(h, 1 - h, mode='full')) * dx

        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Early stopping checks
        if time.time() - start_time > budget_s - 5:
            break
        if iteration > 500 and np.abs(current_c5 - best_c5) < 1e-6:
            break

        # Adaptive learning rate
        if iteration % 100 == 0 and learning_rate > 1e-4:
            learning_rate *= 0.5

    # Final validation
    h = best_h.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scale_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h = h * scale_factor
    h = np.clip(h, 0, 1)

    # Recompute C5 for accuracy
    final_c5 = np.max(scipy.signal.correlate(h, 1 - h, mode='full')) * dx
    return (h, final_c5, n_points)
