import numpy as np
import scipy.optimize as opt
import copy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Starting with 100 points, can be increased if needed
    
    # Check if initial_h_values is provided (i.e., in a parent construction)
    if 'initial_h_values' in globals():
        h_values = np.array(initial_h_values, dtype=np.float64)
    else:
        # Generate a random initial guess
        h_values = np.random.rand(n_points)
        h_values /= np.sum(h_values) * 2  # Normalize to satisfy sum(h) * dx = 1

    # Ensure the values are in [0, 1] and the sum is correct
    h_values = np.clip(h_values, 0, 1)
    sum_h = np.sum(h_values) * (2.0 / n_points)
    if abs(sum_h - 1.0) > 1e-6:
        h_values = np.random.rand(n_points)
        h_values /= np.sum(h_values) * 2
        h_values = np.clip(h_values, 0, 1)

    def compute_c5(h):
        dx = 2.0 / n_points
        h_shifted = np.roll(h, 1)
        cross_corr = np.correlate(h, 1 - h_shifted, mode='full')
        max_overlap = np.max(cross_corr * dx)
        return max_overlap

    # Define objective function for optimization
    def objective(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h) * (2.0 / n_points)
        if abs(sum_h - 1.0) > 1e-6:
            return np.inf  # Penalize constraint violation
        return compute_c5(h)

    # Define bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) * (2.0 / n_points) - 1.0}]

    # Optimization settings
    max_iter = 100
    tol = 1e-4
    result = opt.minimize(
        objective,
        h_values,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': max_iter, 'ftol': tol}
    )

    # Post-process the result
    h_opt = np.clip(result.x, 0, 1)
    sum_h_opt = np.sum(h_opt) * (2.0 / n_points)
    if abs(sum_h_opt - 1.0) > 1e-6:
        h_opt = np.random.rand(n_points)
        h_opt /= np.sum(h_opt) * 2
        h_opt = np.clip(h_opt, 0, 1)
        c5_bound = compute_c5(h_opt)
    else:
        c5_bound = compute_c5(h_opt)

    return h_opt, c5_bound, n_points
