import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Default n_points if not provided
    n_points = 512
    
    # Initialize h_values
    if 'initial_h_values' in globals():
        h = (initial_h_values - np.min(initial_h_values)) / (np.max(initial_h_values) - np.min(initial_h_values))
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.zeros(n_points)
        # Create a uniform distribution that satisfies the sum constraint
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    
    # Objective function
    def evaluate(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Fitness function
    def fitness(h):
        return -evaluate(h)  # Minimize C5
    
    # Mutation function
    def mutate(h, std):
        h = h + np.random.normal(0, std, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    # Crossover function
    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        child = child / np.sum(child) * (n_points / 2)
        child = np.clip(child, 0, 1)
        child = child / np.sum(child) * (n_points / 2)
        return child
    
    # Initialize population
    pop_size = 50
    population = [h.copy() for _ in range(pop_size)]
    population = [mutate(p, 0.05) for p in population]
    
    # Fitness evaluation
    def evaluate_population(pop):
        return np.array([fitness(p) for p in pop])
    
    # Elitism
    elite_size = 10
    
    # Evolution parameters
    mutation_std = 0.02
    generations = 1000
    generation_count = 0
    best_fitness = -np.inf
    best_h = h.copy()
    improvement_threshold = 1e-4
    improvement_count = 0
    
    while time.time() - start_time < budget_s and generation_count < generations:
        # Evaluate population fitness
        fitnesses = evaluate_population(population)
        # Select top performers
        top_indices = np.argsort(fitnesses)[-elite_size:]
        top_population = [population[i] for i in top_indices]
        # Generate new population
        new_population = top_population.copy()
        for _ in range(pop_size - elite_size):
            # Select two parents
            parent1 = random.choice(top_population)
            parent2 = random.choice(top_population)
            # Crossover
            child = crossover(parent1, parent2)
            # Mutate
            child = mutate(child, mutation_std)
            new_population.append(child)
        # Update population
        population = new_population
        # Track best
        current_fitnesses = evaluate_population(population)
        current_best = np.max(current_fitnesses)
        if current_best > best_fitness:
            best_fitness = current_best
            best_h = population[np.argmax(current_fitnesses)].copy()
            improvement_count = 0
        else:
            improvement_count += 1
        # Early termination
        if improvement_count >= 3:
            mutation_std *= 0.9
            if mutation_std < 1e-6:
                break
        generation_count += 1
    
    # Final evaluation
    final_h = best_h.copy()
    final_h = final_h / np.sum(final_h) * (n_points / 2)
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h / np.sum(final_h) * (n_points / 2)
    
    dx = 2.0 / n_points
    corr = np.correlate(final_h, 1 - final_h, mode="full")
    c5_bound = np.max(corr) * dx
    
    # Final checks
    assert np.all(0 <= final_h <= 1)
    assert np.isclose(np.sum(final_h), n_points / 2, atol=1e-6)
    
    return (final_h, c5_bound, n_points)
