import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial guess
    n_points = 45
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5])
    initial_h_values = np.array(initial_h_values)
    
    # Determine the current h_values
    h_values = initial_h_values.copy()
    
    # Normalize h_values to meet the constraint
    h_values = h_values / np.mean(h_values) * (n_points / 2)
    h_values = np.clip(h_values, 0, 1)
    
    # Calculate the initial C5 value
    dx = 2 / n_points
    cross_corr = np.correlate(h_values, 1 - h_values, mode="full")
    c5_initial = np.max(cross_corr) * dx
    
    # Define the objective function for optimization
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        cross_corr = np.correlate(h, 1 - h, mode="full")
        return np.max(cross_corr) * dx
    
    # Define the constraints
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Bounds for h values
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess for optimization
    x0 = h_values.copy()
    
    # Time tracking
    start_time = time.time()
    
    # Optimization settings
    options = {
        'disp': False,
        'ftol': 1e-8,
        'maxiter': 1000,
        'iprint': -1,
        'eps': 1e-8
    }
    
    # Run the optimization
    result = opt.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        method='SLSQP',
        options=options
    )
    
    # Check if the optimization was successful
    if result.success:
        h_vals = result.x
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
        h_vals = np.clip(h_vals, 0, 1)
        cross_corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        c5_bound = np.max(cross_corr) * dx
        n_points_used = n_points
        return (h_vals, c5_bound, n_points_used)
    else:
        # If optimization fails, use the initial configuration
        cross_corr = np.correlate(h_values, 1 - h_values, mode="full")
        c5_bound = np.max(cross_corr) * dx
        return (h_values, c5_bound, n_points)
