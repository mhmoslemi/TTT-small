import numpy as np
import scipy.optimize as optimize
import time
import copy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = evaluate_erdos_solution() if 'initial_h_values' in globals() else None
    n_points = 100
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.tile(h, n_points // len(h) + 1)[:n_points]
    else:
        h = np.zeros(n_points)
        h[::n_points//5] = 1.0
    h = h * (n_points / 2.0) / np.sum(h)
    
    def objective_func(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def surrogate_func(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        weights = np.exp(100 * (corr - 1e-6))
        log_sum = np.log(np.sum(weights))
        return log_sum * dx
    
    def constraint_func(h):
        return np.sum(h) - n_points / 2.0
    
    def constraint_jac(h):
        return np.ones_like(h)
    
    def optimize_with_penalty(h, max_iter=1000, tol=1e-6):
        result = optimize.minimize(
            fun=surrogate_func,
            x0=h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint_func, 'jac': constraint_jac}],
            options={'maxiter': max_iter, 'ftol': tol}
        )
        h_opt = result.x
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2.0) / np.sum(h_opt)
        return h_opt
    
    start_time = time.time()
    h = optimize_with_penalty(h)
    elapsed = time.time() - start_time
    if elapsed > budget_s * 0.6:
        h = optimize_with_penalty(h, max_iter=500, tol=1e-5)
    
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2.0) / np.sum(h)
    
    result = objective_func(h)
    return h, result, n_points
