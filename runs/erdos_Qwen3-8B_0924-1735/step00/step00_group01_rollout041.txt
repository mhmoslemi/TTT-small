import random
import time

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Adjust n_points to be a multiple of 2 for symmetry
    n_points = 512
    dx = 2.0 / n_points

        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create uniform initial guess
        h = np.full(n_points, 0.5)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    
    population_size = 50
    population = [h.copy() for _ in range(population_size)]
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    
    mutation_std = 0.02
    gen = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s and gen < 1000:
        # Evaluate population
        fitness = []
        for h_candidate in population:
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate / np.sum(h_candidate) * (n_points / 2)
            corr = np.correlate(h_candidate, 1 - h_candidate, mode="full")
            c5 = np.max(corr) * dx
            fitness.append((-c5, h_candidate))  # Negative for minimization
        
        # Sort by fitness
        fitness.sort()
        fitness = [f[1] for f in fitness]
        
        # Select top 20 parents
        parents = fitness[:20]
        
        # Generate new population
        new_population = []
        for _ in range(population_size // 2):
            # Crossover
            parent1 = random.choice(parents)
            parent2 = random.choice(parents)
            child = np.zeros(n_points)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            # Mutation
            noise = np.random.normal(0, mutation_std, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child = child / np.sum(child) * (n_points / 2)
            new_population.append(child)
        
        # Add elites
        new_population += parents[:10]
        
        # Replace old population
        population = new_population
        
        # Update best
        for h_candidate in population:
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate / np.sum(h_candidate) * (n_points / 2)
            corr = np.correlate(h_candidate, 1 - h_candidate, mode="full")
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate.copy()
        
        gen += 1
        
        # Early stopping
        if gen > 2 and np.abs(best_c5 - evaluate_erdos_solution(best_h, n_points)) < 1e-4:
            mutation_std *= 0.7
        
    # Final evaluation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    corr = np.correlate(best_h, 1 - best_h, mode="full")
    c5_bound = np.max(corr) * dx
    return (best_h, c5_bound, n_points)
