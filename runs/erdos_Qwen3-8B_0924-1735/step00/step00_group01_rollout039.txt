import numpy as np
import math
import random
import scipy.signal
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    
    # Default n_points
    n_points = 512
    
    # Start from initial_h_values if available, else construct a simple initial guess
    if initial_h_values is not None:
        # Resample initial_h_values to 512 points via linear interpolation
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    else:
        # Construct a simple initial guess: a step function with two levels
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
    # Ensure sum(h) = n_points / 2
    h = h / np.sum(h) * (n_points / 2)
    # Clip values to [0, 1]
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.inf
    
    # Precompute constants
    dx = 2.0 / n_points
    lags = np.arange(0, 2 * n_points - 1)
    max_overlap = np.inf
    
    # Parameters
    learning_rate = 0.05
    temp = 1.0
    perturb_rate = 0.05
    iteration_limit = 1000
    min_improvement = 1e-5
    decay_factor = 0.95
    time_limit = budget_s
    
    start_time = time.time()
    for _ in range(iteration_limit):
        elapsed = time.time() - start_time
        if elapsed > time_limit:
            break
        
        # Compute overlaps for all lags
        overlap = correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(overlap)
        
        # Update best solution if improvement
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient for the maximum overlap lag
        max_k = np.argmax(overlap)
        grad = np.zeros(n_points)
        
        # Compute gradient for lag max_k
        for i in range(n_points):
            if (i + max_k) % n_points < n_points:
                grad[i] += (1 - h[(i + max_k) % n_points]) / n_points
            if (i - max_k) % n_points < n_points:
                grad[i] -= h[(i - max_k) % n_points] / n_points
        
        # Update h using gradient
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
        # Perturb a fraction of the h values
        if np.random.rand() < perturb_rate:
            indices = np.random.choice(n_points, int(perturb_rate * n_points), replace=False)
            h[indices] += np.random.choice([-0.1, 0.1], size=len(indices))
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
        
        # Check for convergence
        if np.abs(current_c5 - best_c5) < min_improvement:
            learning_rate *= decay_factor
            if learning_rate < 1e-6:
                break
    
    # Final validation
    final_overlap = correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(final_overlap)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    assert np.all(best_h >= 0) and np.all(best_h <= 1), "h values out of bounds"
    assert np.isclose(np.sum(best_h), n_points / 2, atol=1e-6), "Constraint violated"
    
    return (best_h, final_c5, n_points)
