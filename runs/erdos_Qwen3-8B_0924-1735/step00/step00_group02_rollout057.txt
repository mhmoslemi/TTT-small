import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else np.random.rand(45)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_h_values = np.clip(initial_h_values, 0, 1)
    norm = n_points / 2.0
    initial_h_values = (initial_h_values * norm) / np.sum(initial_h_values)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * norm / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def surrogate_objective(h):
        h = np.clip(h, 0, 1)
        h = h * norm / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        beta = 100
        return np.log(np.sum(np.exp(beta * (corr - 1e-6)))) / beta * dx
    
    def constraint_func(h):
        return np.sum(h) - norm
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}
    
    h0 = initial_h_values.copy()
    best_h = h0.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        res = opt.minimize(surrogate_objective, h0, bounds=bounds, constraints=cons, method='L-BFGS-B', tol=1e-8)
        if res.success:
            h = res.x
            h = np.clip(h, 0, 1)
            h = h * norm / np.sum(h)
            c5 = objective(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
            h0 = h.copy()
            np.random.shuffle(h0)
            h0 = h0 * norm / np.sum(h0)
        else:
            break
    
    h = best_h
    h = np.clip(h, 0, 1)
    h = h * norm / np.sum(h)
    c5 = objective(h)
    return h, c5, n_points
