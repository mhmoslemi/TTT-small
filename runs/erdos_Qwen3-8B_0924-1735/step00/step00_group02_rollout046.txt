import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None  # Placeholder for initial construction

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 128  # Fast enough for optimization
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def evaluate_c5(h_values):
        h = np.array(h_values, dtype=float)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5_val = np.max(corr) * dx
        return c5_val
    
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    result = opt.minimize(
        objective,
        h,
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        method='SLSQP',
        options={
            'ftol': 1e-6,
            'gtol': 1e-6,
            'maxiter': 100,
            'disp': False
        }
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        h_values = np.clip(result.x, 0, 1)
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        c5_bound = evaluate_c5(h_values)
        return h_values, c5_bound, n_points
    
    h_values = np.clip(result.x, 0, 1)
    h_values = h_values * (n_points / 2) / np.sum(h_values)
    c5_bound = evaluate_c5(h_values)
    return h_values, c5_bound, n_points

start_time = time.time()
