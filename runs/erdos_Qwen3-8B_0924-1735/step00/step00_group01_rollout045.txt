        h = np.repeat(h, 2)
        h = h[:n_points]
        h = h / np.sum(h) * (n_points / 2)

    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)

    best_h = h.copy()
    best_c5 = np.inf
    learning_rate = 0.05
    iterations = 0
    max_iterations = 1000
    perturbation_rate = 0.05
    min_improvement = 1e-5
    decay_factor = 0.95
    temp = 1.0

    for _ in range(max_iterations):
        if iterations >= max_iterations:
            break
        # Compute overlap
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr)
        current_c5 = max_overlap * dx

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            learning_rate = 0.05
        else:
            # Compute gradient for the lag with maximum overlap
            k_max = np.argmax(corr)
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = (i + k_max) % n_points
                grad[i] -= h[i] * (1 - h[j]) * dx
                j = (i - k_max) % n_points
                grad[i] += (1 - h[i]) * h[j] * dx
            grad = grad / np.linalg.norm(grad)

            # Update h
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)

            # Perturbations
            perturb_mask = np.random.rand(n_points) < perturbation_rate
            h[perturb_mask] += np.random.uniform(-0.1, 0.1, size=np.sum(perturb_mask))
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)

            # Check for improvement
            corr = np.correlate(h, 1 - h, mode='full')
            improved = np.max(corr) * dx < current_c5 - min_improvement
            if improved:
                learning_rate = 0.05
            else:
                learning_rate *= decay_factor

        iterations += 1

    # Final validation
    h = best_h.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    corr = np.correlate(h, 1 - h, mode='full')
    c5_bound = np.max(corr) * dx

    return (h, c5_bound, n_points)
