import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with parent solution or random if not provided
    if 'initial_h_values' in globals():
        h = (initial_h_values - np.min(initial_h_values)) / (np.max(initial_h_values) - np.min(initial_h_values))
    else:
        h = np.random.rand(512)
    h = h / np.sum(h) * (512 / 2)
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (512 / 2)
    
    n_points = 512
    dx = 2.0 / n_points
    
    population_size = 50
    population = [h.copy()]
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate all candidates
        candidates = []
        for candidate in population:
            corr = np.correlate(candidate, 1 - candidate, mode="full")
            c5 = np.max(corr) * dx
            candidates.append((c5, candidate))
        
        # Sort candidates by C5 and select top performers
        candidates.sort()
        best_c5, best_h = candidates[0]
        top_parents = [c[1] for c in candidates[:20]]
        
        # Generate new population
        new_population = []
        for _ in range(10):
            # Uniform crossover
            parent1 = random.choice(top_parents)
            parent2 = random.choice(top_parents)
            child = np.zeros_like(parent1)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            
            # Apply Gaussian mutation
            mutation_std = 0.02
            child += np.random.normal(0, mutation_std, n_points)
            child = np.clip(child, 0, 1)
            child = child / np.sum(child) * (n_points / 2)
            
            new_population.append(child)
        
        # Add elites
        for parent in top_parents[:10]:
            new_population.append(parent)
        
        population = new_population
        
        # Early termination check
        if len(candidates) > 1 and abs(best_c5 - candidates[1][0]) < 1e-4:
            mutation_std *= 0.3
    
    # Final evaluation of best candidate
    best_candidate = population[0]
    corr = np.correlate(best_candidate, 1 - best_candidate, mode="full")
    best_c5 = np.max(corr) * dx
    
    # Final checks
    h_values = best_candidate
    c5_bound = best_c5
    n_points = 512
    
    return (h_values, c5_bound, n_points)
