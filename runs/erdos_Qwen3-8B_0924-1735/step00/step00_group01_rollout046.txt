```python
import numpy as np
import numpy.random as npr
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    npr.seed(seed)
    n_points = 512
    initial_h_values = np.array([0.0]*n_points, dtype=np.float64)
    
    if 'initial_h_values' in kwargs:
        initial_h_values = kwargs['initial_h_values']
        n_points = len(initial_h_values)
        initial_h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    
    h = np.copy(initial_h_values)
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    h = h / total * (n_points / 2)
    dx = 2.0 / n_points
    
    def compute_overlap(h):
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        return max_val * dx
    
    def clip_and_normalize(h):
        h = np.clip(h, 0.0, 1.0)
        total_h = np.sum(h)
        if total_h == 0:
            return np.zeros_like(h)
        h = h / total_h * (n_points / 2)
        return h
    
    best_h = np.copy(h)
    best_c5 = compute_overlap(best_h)
    learning_rate = 0.05
    iteration_limit = 1000
    min_improvement = 1e-5
    decay_factor = 0.95
    perturbation_rate = 0.05
    max_lag = 2 * n_points - 1
    
    for iter in range(iteration_limit):
        start_time = time.time()
        
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        k_max = np.argmax(corr)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            lag = k_max % n_points
            idx = (i - lag) % n_points
            grad[i] = (1 - h[idx]) - h[i]
        
        h -= learning_rate * grad
        h = clip_and_normalize(h)
        
        if np.random.rand() < perturbation_rate:
            indices = np.random.choice(n_points, size=int(n_points * 0.05), replace=False)
            h[indices] += np.random.choice([-0.1, 0.1], size=len(indices))
            h = clip_and_normalize(h)
        
        current_c5 = compute_overlap(h)
        if current_c5 < best_c5 - min_improvement:
            best_c5 = current_c5
            best_h = np.copy(h)
            learning_rate = 0.05
        elif current_c5 >= best_c5:
            learning_rate *= decay_factor
        
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    
    h_values = clip_and_normalize(best_h)
    c5_bound = compute_overlap(h_values)
    return (h_values, c5_bound, n_points)
```
