```python
import numpy as np
import scipy
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    if initial_h_values is None:
        n_points = 45
        h = np.zeros(n_points)
        h[0:23] = 1
        h[23:45] = 0
        initial_h_values = h
    else:
        n_points = len(initial_h_values)
        h = initial_h_values.copy()
    
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    assert abs(sum_h - n_points / 2) < 1e-6, "Initial h does not satisfy integral constraint"
    
    best_c5 = np.inf
    best_h = h.copy()
    
    max_iterations = 400
    learning_rate = 0.05
    rho = 0.5
    lambda_c = 0.0
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        k_indices = np.random.choice(2 * n_points - 1, size=10, replace=False)
        current_max = -np.inf
        selected_k = 0
        for k in k_indices:
            idx = (np.arange(n_points) + k) % n_points
            overlap = np.dot(h, 1 - h[idx]) * dx
            if overlap > current_max:
                current_max = overlap
                selected_k = k
        
        k = selected_k
        gradient = np.zeros(n_points)
        for i in range(n_points):
            pos = (i + k) % n_points
            gradient[i] += (1 - h[pos]) * dx
            neg = (i - k) % n_points
            gradient[i] -= h[neg] * dx
        
        h += learning_rate * gradient
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h *= scale
        
        error = sum_h - n_points / 2
        lambda_c += rho * error
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        
        if iteration > 100 and abs(current_max - best_c5) < 1e-6:
            break
    
    h_values = best_h
    c5_bound = best_c5
    return (h_values, c5_bound, n_points)
```
