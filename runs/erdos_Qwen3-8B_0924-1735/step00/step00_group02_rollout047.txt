import numpy as np
import scipy.optimize as opt
import scipy.signal as sig
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Start with 200 points for balance between resolution and speed
    dx = 2.0 / n_points
    # Initial guess from provided initial_h_values or a default
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        # Default: uniform distribution with 0.5 density
        h = np.full(n_points, 0.5)
    
    # Normalize to sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    def cost_function(h_flat):
        h = h_flat.reshape(n_points)
        # Compute cross-correlation
        corr = sig.correlate(h, 1 - h, mode='full') * dx
        # Return the maximum value as the cost
        return corr.max()
    
    # Define constraints: sum(h) == n_points / 2
    cons = ({'type': 'eq', 'fun': lambda h_flat: np.sum(h_flat.reshape(n_points)) - n_points / 2})
    
    # Bounds: [0, 1] for each element
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Surrogate function for smooth optimization
    def surrogate_cost(h_flat, beta=100, epsilon=1e-6):
        h = h_flat.reshape(n_points)
        corr = sig.correlate(h, 1 - h, mode='full') * dx
        # Softmax approximation to smooth max
        return np.log(np.sum(np.exp(beta * (corr - epsilon)))) / beta
    
    # Optimization setup
    result = opt.minimize(
        fun=surrogate_cost,
        x0=h.flatten(),
        bounds=bounds,
        constraints=cons,
        method='L-BFGS-B',
        options={'maxiter': 1000, 'ftol': 1e-7}
    )
    
    # Extract best solution
    best_h = result.x.reshape(n_points)
    best_h = np.clip(best_h, 0.0, 1.0)
    # Renormalize to satisfy sum constraint
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Compute final C₅ value
    final_corr = sig.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = final_corr.max()
    
    # If time allows, try further refines
    if time.time() < time.time() + budget_s - 200:
        # Trial of gradient descent with random perturbations
        for _ in range(5):
            noise = np.random.normal(0, 1e-3, n_points)
            perturbed = best_h + noise
            perturbed = np.clip(perturbed, 0.0, 1.0)
            perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
            # Compute C5
            corr = sig.correlate(perturbed, 1 - perturbed, mode='full') * dx
            if corr.max() < c5_bound:
                best_h = perturbed
                c5_bound = corr.max()
    
    # Final check and return
    return (best_h, c5_bound, n_points)
