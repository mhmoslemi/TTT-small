import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 512
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.zeros(n_points)
    if 'initial_h_values' in globals():
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    h = initial_h.copy()
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    learning_rate = 0.05
    temp = 1.0
    perturb_rate = 0.05
    iteration_limit = 1000
    start_time = time.time()
    
    def compute_overlap(h):
        nonlocal dx
        corr = np.correlate(h, 1 - h, mode="full")
        return corr * dx
    
    def compute_c5(h):
        return np.max(compute_overlap(h))
    
    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    for iteration in range(iteration_limit):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        corr = compute_overlap(h)
        max_c5 = np.max(corr)
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
        
        if max_c5 < 1e-6:
            break
        
        k_max = np.argmax(corr)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i - k_max) % n_points
            grad[i] -= h[j]
            j = (i + k_max) % n_points
            grad[i] += (1 - h[j])
        
        h = h - learning_rate * grad
        h = clip_and_normalize(h)
        
        if np.random.rand() < perturb_rate:
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    h[i] += 0.1
                else:
                    h[i] -= 0.1
            h = clip_and_normalize(h)
        
        if iteration % 100 == 0:
            learning_rate *= 0.95
            temp *= 0.95
    
    h = clip_and_normalize(h)
    corr = compute_overlap(h)
    c5_bound = np.max(corr)
    
    return (h, c5_bound, n_points)
