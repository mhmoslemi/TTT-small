import scipy.optimize as opt
import time

        shifted = np.zeros_like(h)

            shifted[k:] = h[:-k]
        corr = np.correlate(h, 1 - h, mode='full')

        return np.sum(h) - (n_points / 2)
    
    h0 = initial_h.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    result = opt.minimize(
        objective,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'ftol': 1e-6, 'maxiter': 100}
    )
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (initial_h[:n_points], objective(initial_h), n_points)
    
    h_opt = result.x
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt / np.sum(h_opt) * (n_points / 2)
    
    c5 = objective(h_opt)
    
    return (h_opt, c5, n_points)
