import numpy as np
import scipy.signal
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 512
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        shift = np.roll(h, 1)
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def gradient(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        shift = np.roll(h, 1)
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if k_max - i >= 0 and k_max - i < n_points:
                grad[i] -= (2.0 / n_points) * (1 - h[k_max - i])
            if k_max + i < len(corr) and k_max + i < 2 * n_points:
                grad[i] += (2.0 / n_points) * h[k_max + i]
        grad = grad * (n_points / 2) / np.sum(h)
        grad = np.clip(grad, 0, 1)
        return grad
    
    def update_rule(h_vec, grad, lr):
        h = h_vec.copy()
        h = h - lr * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    for step in range(1000):
        if time.time() - start_time > budget_s:
            break
        grad = gradient(h)
        h = update_rule(h, grad, 0.05)
        
        if np.isnan(np.max(scipy.signal.correlate(h, 1 - h, mode='full') * dx)):
            h = best_h.copy()
            continue
        
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if np.abs(current_c5 - best_c5) < 1e-6:
            break
    
    h = best_h.copy()
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    c5_bound = objective(h)
    
    return (h, c5_bound, n_points)
