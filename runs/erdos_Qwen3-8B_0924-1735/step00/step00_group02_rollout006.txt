```python
import numpy as np
import random
import time
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else None
    n_points = len(initial_h_values) if initial_h_values is not None else 100
    dx = 2.0 / n_points
    population_size = 50
    generations = 100
    mutation_rate = 0.3
    tournament_size = 5
    elitism_ratio = 0.2
    diversity_injection_interval = 20
    mutation_scale = 0.1

    def normalize(h):
        total = np.sum(h)
        return h * (n_points / 2.0) / total if total > 0 else h

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    if initial_h_values is not None:
        population = [np.copy(initial_h_values) + np.random.normal(0, 0.1, n_points) for _ in range(population_size)]
        population = [normalize(p) for p in population]
    else:
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [normalize(p) for p in population]

    best_h = population[0]
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break

        fitness = [evaluate(h) for h in population]
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]

        elites = int(elitism_ratio * population_size)
        elites = sorted_population[:elites]

        new_population = elites.copy()
        while len(new_population) < population_size:
            tournament = np.random.choice(sorted_population, size=tournament_size)
            winner = min(tournament, key=evaluate)
            parent1, parent2 = np.random.choice([winner] * 2, size=2)
            crossover_point = np.random.randint(1, n_points)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child = normalize(child)

            if np.random.rand() < mutation_rate:
                mutate_indices = np.random.choice(n_points, int(mutation_rate * n_points), replace=False)
                for i in mutate_indices:
                    child[i] += np.random.laplace(0, mutation_scale)
                    child[i] = np.clip(child[i], 0, 1)
                child = normalize(child)

            new_population.append(child)

        population = new_population
        current_best = min(population, key=evaluate)
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5

        if gen % diversity_injection_interval == 0:
            for _ in range(5):
                h = np.random.uniform(0, 1, n_points)
                h = normalize(h)
                population.append(h)

    final_h = best_h
    final_c5 = evaluate(final_h)
    return final_h, final_c5, n_points
```
