import numpy as np
import scipy.optimize as opt
import time

initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 90  # Use double the initial size for better refinement
    dx = 2.0 / n_points
    h = np.copy(initial_h_values)
    h = np.resize(h, n_points)  # Resize to n_points, assuming initial is smaller
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    if np.isclose(sum_h, n_points / 2):
        pass
    else:
        h = h / sum_h * (n_points / 2)
    h = np.clip(h, 0.0, 1.0)
    start_time = time.time()
    max_iterations = 200
    step_size = 0.01
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        max_idx = np.argmax(corr)
        # Compute gradient for the max lag
        grad = np.zeros(n_points)
        k = max_idx - n_points + 1  # Convert to lag in the original domain
        for i in range(n_points):
            if k + i < 0 or k + i >= n_points:
                continue
            grad[i] += (1 - h[i + k]) - h[i - k]
        # Update h
        h = h - step_size * grad
        h = np.clip(h, 0.0, 1.0)
        # Ensure sum constraint
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0.0, 1.0)
    # Final evaluation
    final_corr = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return h, final_c5, n_points
