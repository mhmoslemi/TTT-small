import numpy as np
import scipy.optimize as opt
import copy
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h.npy')  # Preloaded initial construction
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = n_points / 2
    h = copy.deepcopy(initial_h_values)
    h = h.astype(np.float64)
    
    # Enforce constraint
    h = h / np.sum(h) * sum_h
    
    def compute_c5(h):
        cor = np.correlate(h, 1 - h, mode='full')
        return np.max(cor) * dx
    
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        return compute_c5(h)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - sum_h
    
    def constraint_bounds(h_vec):
        return np.all((h_vec >= 0) & (h_vec <= 1))
    
    def gradient_descent_step(h, max_corr_index, dx, learning_rate):
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + max_corr_index < n_points:
                grad[i] += (1 - h[i + max_corr_index])  # Contribution from h[j] in 1 - h
            if i - max_corr_index >= 0:
                grad[i] -= h[i - max_corr_index]  # Contribution from h[j] in h
        grad = grad * learning_rate
        return grad
    
    start_time = time.time()
    iter_limit = 200
    learning_rate = 0.01
    grad_norm_threshold = 1e-5
    improvement_threshold = 1e-6
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for iteration in range(iter_limit):
        if time.time() - start_time > budget_s:
            break
        cor = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(cor)
        max_corr_index = np.argmax(cor)
        grad = gradient_descent_step(h, max_corr_index, dx, learning_rate)
        h = np.clip(h + grad, 0, 1)
        h = h / np.sum(h) * sum_h
        
        current_c5 = compute_c5(h)
        if abs(current_c5 - best_c5) < improvement_threshold:
            break
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if np.linalg.norm(grad) < grad_norm_threshold:
            break
    
    final_h = best_h
    final_c5 = compute_c5(final_h)
    
    return (final_h, final_c5, n_points)
