import numpy as np
import random
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Initial setup
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = h / np.sum(h) * (n_points / 2)
    h = np.clip(h, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_c5_full(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def evaluate_objective(h):
        return compute_c5(h)

    def compute_constraints(h):
        return np.sum(h) - n_points / 2

    def evaluate_constraints(h):
        return np.array([compute_constraints(h)])

    start_time = time.time()

    # Initial evaluation
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Genetic algorithm parameters
    num_generations = 100
    population_size = 50
    mutation_rate = 0.3
    tournament_size = 5
    elitism_ratio = 0.2
    mutation_strength = 0.1
    diversity_injection_rate = 0.05
    mutation_distribution = 'laplace'

    # Initialize population
    if initial_h_values.size > 0:
        population = [np.clip(h + np.random.laplace(0, mutation_strength, n_points), 0, 1) for _ in range(population_size)]
        for i in range(population_size):
            population[i] = population[i] / np.sum(population[i]) * (n_points / 2)
        population = [np.clip(p, 0, 1) for p in population]
        population.append(h.copy())
    else:
        population = [np.random.rand(n_points) for _ in range(population_size)]
        for i in range(population_size):
            population[i] = population[i] / np.sum(population[i]) * (n_points / 2)
        population = [np.clip(p, 0, 1) for p in population]

    # Genetic algorithm loop
    for gen in range(num_generations):
        # Evaluate fitness
        fitness = [compute_c5(p) for p in population]
        best_idx = np.argmin(fitness)
        best_c5_current = fitness[best_idx]
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = population[best_idx].copy()

        # Elitism
        elite_idx = np.argsort(fitness)[:int(elitism_ratio * population_size)]
        elites = [population[i] for i in elite_idx]

        # Selection and crossover
        new_population = elites[:]
        while len(new_population) < population_size:
            # Tournament selection
            tournament = np.random.choice(range(population_size), size=tournament_size, replace=False)
            selected = tournament[np.argmin([fitness[i] for i in tournament])]
            p1 = population[selected]
            tournament = np.random.choice(range(population_size), size=tournament_size, replace=False)
            selected = tournament[np.argmin([fitness[i] for i in tournament])]
            p2 = population[selected]
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
            child = child / np.sum(child) * (n_points / 2)
            child = np.clip(child, 0, 1)
            new_population.append(child)

        # Mutation
        for i in range(len(new_population)):
            if np.random.random() < mutation_rate:
                for j in range(n_points):
                    if np.random.random() < mutation_rate:
                        if mutation_distribution == 'laplace':
                            delta = np.random.laplace(0, mutation_strength)
                        else:
                            delta = np.random.normal(0, mutation_strength)
                        new_population[i][j] += delta
                        new_population[i][j] = np.clip(new_population[i][j], 0, 1)
                new_population[i] = new_population[i] / np.sum(new_population[i]) * (n_points / 2)
                new_population[i] = np.clip(new_population[i], 0, 1)

        # Check for diversity injection
        if np.random.random() < diversity_injection_rate:
            new_population.append(np.random.rand(n_points))
            new_population[-1] = new_population[-1] / np.sum(new_population[-1]) * (n_points / 2)
            new_population[-1] = np.clip(new_population[-1], 0, 1)

        population = new_population

        # Time check
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    best_h = np.clip(best_h, 0, 1)
    c5_bound = compute_c5(best_h)

    # Return result
    return (best_h, c5_bound, n_points)
