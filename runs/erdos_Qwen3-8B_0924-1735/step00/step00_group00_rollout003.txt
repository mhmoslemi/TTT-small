import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    scaling_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h *= scaling_factor
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scaling_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h *= scaling_factor
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    max_iterations = 1000
    learning_rate = 0.05
    early_stop = 0
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Compute all Q_k
        q = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            idx = np.roll(np.arange(n_points), k)
            q[k] = np.dot(h, h[idx]) * dx
        
        min_idx = np.argmin(q)
        min_q = q[min_idx]
        k = min_idx - n_points + 1
        
        # Compute gradient for this lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + k < n_points:
                grad[i] += h[(i + k) % n_points] * dx
            if 0 <= i - k < n_points:
                grad[i] += h[(i - k) % n_points] * dx
        
        # Update h
        h += learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scaling_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h *= scaling_factor
        h = np.clip(h, 0, 1)
        
        # Compute current C5
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            early_stop = 0
        else:
            early_stop += 1
        
        # Early stopping
        if early_stop > 20:
            break
    
    # Final validation
    h = best_h.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scaling_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h *= scaling_factor
    h = np.clip(h, 0, 1)
    corr = scipy.signal.correlate(h, 1 - h, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (h, c5_bound, n_points)
