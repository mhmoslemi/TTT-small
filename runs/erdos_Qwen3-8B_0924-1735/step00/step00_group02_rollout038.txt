import numpy as np
import random
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h /= np.sum(h)
    h = np.clip(h, 0, 1)
    
    def evaluate_c5(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        if np.sum(h) != n_points / 2:
            h = h * (n_points / 2) / np.sum(h)
        if np.sum(h) != n_points / 2:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def fitness(h):
        return evaluate_c5(h)
    
    def mutate(h, mutation_rate=0.3, mutation_scale=0.1):
        h = np.array(h, dtype=np.float64)
        n = len(h)
        for i in range(n):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.laplace(0, mutation_scale)
                h[i] = np.clip(h[i], 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        return h
    
    def crossover(parent1, parent2):
        n = len(parent1)
        idx = np.random.randint(1, n - 1)
        child = np.concatenate([parent1[:idx], parent2[idx:]])
        return child
    
    def select_parents(population):
        tournament = np.random.choice(population, size=5, replace=False)
        return tournament[np.argmin([fitness(p) for p in tournament])]
    
    def generate_population(pop_size):
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values, dtype=np.float64)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
        else:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
        return [h for _ in range(pop_size)]
    
    def optimize():
        population = generate_population(50)
        best_h = population[0]
        best_c5 = fitness(best_h)
        start_time = time.time()
        for gen in range(100):
            if time.time() - start_time > budget_s:
                break
            new_population = []
            for _ in range(50):
                parent1 = select_parents(population)
                parent2 = select_parents(population)
                child = crossover(parent1, parent2)
                child = mutate(child)
                new_population.append(child)
            population = new_population
            current_best = min(population, key=fitness)
            current_c5 = fitness(current_best)
            if current_c5 < best_c5:
                best_h = current_best
                best_c5 = current_c5
        return best_h, best_c5
    
    hbest, c5best = optimize()
    hbest = np.clip(hbest, 0, 1)
    hbest = hbest * (n_points / 2) / np.sum(hbest)
    hbest = np.clip(hbest, 0, 1)
    c5best = evaluate_c5(hbest)
    
    return hbest.tolist(), c5best, n_points
