import numpy as np
import scipy.optimize as opt
import time

initial_h_values = None  # Will be set during runtime

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values) if initial_h_values is not None else 45
    dx = 2.0 / n_points
    
    if initial_h_values is None:
        # Default initial construction if no parent provided
        init_h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                init_h[i] = 1.0
        initial_h_values = init_h
    
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = np.inf
    max_iter = 500
    learning_rate = 0.05
    rho = 0.5
    lambda_c = 0.0
    
    def compute_overlap(h_vals):
        h = h_vals.copy()
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.inf
        scale = (n_points / 2) / sum_h
        h *= scale
        overlap = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def update_h(h_vals):
        nonlocal lambda_c, rho, best_c5, best_h
        start_time = time.time()
        h = h_vals.copy()
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h *= scale
        
        # Stochastic selection of lags
        lags = np.random.choice(2 * n_points - 1, size=16, replace=False)
        max_overlap = 0.0
        selected_k = 0
        for k in lags:
            if k < 0:
                idx = (np.arange(n_points) + k) % n_points
                overlap = np.dot(h, 1 - h[idx]) * dx
            else:
                idx = (np.arange(n_points) + k) % n_points
                overlap = np.dot(h, 1 - h[idx]) * dx
            if overlap > max_overlap:
                max_overlap = overlap
                selected_k = k
        
        # Compute gradient for the selected lag
        k = selected_k
        grad = np.zeros(n_points)
        if k < 0:
            for i in range(n_points):
                j = (i + k) % n_points
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        else:
            for i in range(n_points):
                j = (i + k) % n_points
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        
        # Update h with gradient descent
        h += learning_rate * grad
        
        # Enforce bounds and integral constraint
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h *= scale
        
        # Update Lagrange multiplier
        error = sum_h - n_points / 2
        lambda_c += rho * error
        rho += 0.01 if abs(error) > 1e-3 else 0.0
        
        # Early stopping if no improvement
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        elif time.time() - start_time > budget_s:
            return best_h, best_c5
        
        return h
    
    for _ in range(max_iter):
        h = update_h(h)
        if time.time() > budget_s:
            break
    
    # Final validation
    h = best_h
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    if sum_h != 0:
        scale = (n_points / 2) / sum_h
        h *= scale
    final_c5 = compute_overlap(h)
    
    return h.tolist(), final_c5, n_points
