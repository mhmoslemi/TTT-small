import numpy as np
import scipy.signal as signal
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.copy(initial_h_values)
    best_h = h.copy()
    best_c5 = np.inf
    max_iterations = 400
    learning_rate = 0.05
    rho = 0.5
    lambda_c = 0.0
    decay_rate = 0.99
    improvement_threshold = 1e-5
    early_stop = 0
    start_time = time.time()

    for iter in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        k_indices = np.random.choice(2 * n_points - 1, size=15, replace=False)
        current_max = 0.0
        selected_k = 0
        for k in k_indices:
            idx = (np.arange(n_points) + k) % n_points
            overlap = np.dot(h, 1 - h[idx]) * dx
            if overlap > current_max:
                current_max = overlap
                selected_k = k
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        k = selected_k
        idx = (np.arange(n_points) + k) % n_points
        gradient = np.zeros(n_points)
        for i in range(n_points):
            gradient[i] += (1 - h[idx[i]]) * dx
            gradient[i] -= h[(i - k) % n_points] * dx
        h += learning_rate * gradient
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h *= scale
        error = sum_h - n_points / 2
        lambda_c += rho * error
        if abs(error) < 1e-6:
            rho = max(rho / 2, 1e-6)
        else:
            rho *= 1.5
        if abs(current_max - best_c5) < improvement_threshold:
            early_stop += 1
            if early_stop > 20:
                break
        learning_rate *= decay_rate

    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-6:
        scale = (n_points / 2) / sum_h
        h *= scale
    h = np.clip(h, 0.0, 1.0)
    c5 = np.max(signal.correlate(h, 1 - h, mode='full')) * dx
    return h, c5, n_points
