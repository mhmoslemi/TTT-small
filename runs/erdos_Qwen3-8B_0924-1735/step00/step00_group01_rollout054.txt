    np.random.seed(seed)
    start_time = time.time()
    n_points = 512
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():

        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def gradient(h, k):
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i - k) % n_points
            grad[i] -= h[i] * (1 - h[j]) * dx
            j = (i + k) % n_points
            grad[i] += (1 - h[i]) * h[j] * dx
        return grad

    best_h = h.copy()
    best_c5 = objective(best_h)
    iteration = 0
    learning_rate = 0.05
    while time.time() - start_time < budget_s:
        iteration += 1
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        k = np.argmax(corr)
        grad = gradient(h, k)
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        if np.random.rand() < 0.05:
            perturb = np.random.choice(n_points, int(n_points * 0.05), replace=False)
            h[perturb] += np.random.choice([-0.1, 0.1], size=len(perturb))
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if iteration % 100 == 0:
            learning_rate *= 0.995
        if iteration > 1000:
            break
    h_values = best_h
    c5_bound = best_c5
    return (h_values, c5_bound, n_points)
