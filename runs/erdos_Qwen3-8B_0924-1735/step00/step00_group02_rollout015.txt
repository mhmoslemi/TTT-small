import numpy as np
import random
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else None
    n_points = 100
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h_pop = [np.clip(initial_h_values + np.random.normal(0, 0.1, n_points), 0, 1) for _ in range(50)]
        h_pop = [h / np.sum(h) * (n_points / 2) for h in h_pop]
    else:
        h_pop = [np.random.rand(n_points) for _ in range(50)]
        h_pop = [h / np.sum(h) * (n_points / 2) for h in h_pop]

    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = []
        for h in h_pop:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append((c5, h))
        # Sort by fitness
        fitness.sort()
        # Select elites
        elites = fitness[:int(len(fitness) * 0.2)]
        # Generate new population
        new_pop = []
        for _ in range(50 - len(elites)):
            # Tournament selection
            tournament = random.sample(fitness, 5)
            parent = min(tournament, key=lambda x: x[0])[1]
            # Crossover
            crossover_point = np.random.randint(1, n_points)
            child = np.concatenate([parent[:crossover_point], parent[crossover_point:]])
            # Mutation
            if np.random.rand() < 0.3:
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        delta = np.random.laplace(0, 0.1)
                        child[i] += delta
                        child[i] = np.clip(child[i], 0, 1)
            child = child / np.sum(child) * (n_points / 2)
            new_pop.append(child)
        h_pop = [h for c5, h in elites] + new_pop
        # Track best
        for c5, h in fitness:
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
        # Inject diversity
        if gen % 20 == 0:
            for _ in range(5):
                h = np.random.rand(n_points)
                h = h / np.sum(h) * (n_points / 2)
                h_pop.append(h)
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    assert np.isclose(np.sum(best_h), n_points / 2)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx
    return (best_h, c5_bound, n_points)
