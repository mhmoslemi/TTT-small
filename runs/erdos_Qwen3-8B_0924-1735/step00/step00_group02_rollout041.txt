```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 1000
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5 if i % 2 == 0 else 0.5 for i in range(n_points)])
    
    if len(initial_h_values) < n_points:
        initial_h_values = np.tile(initial_h_values, 2)[:n_points]
    
    # Normalize to meet the constraint sum(h) == n_points / 2
    initial_h_values = initial_h_values / np.sum(initial_h_values) * (n_points / 2)
    
    # Objective function: compute max cross-correlation
    def objective(h):
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        c5 = np.max(corr * dx)
        return c5
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for h values
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    h_initial = initial_h_values.copy()
    
    # Optimization settings
    method = 'SLSQP'
    tol = 1e-4
    max_iter = 100
    start_time = time.time()
    
    # Perform the optimization
    result = opt.minimize(
        fun=objective,
        x0=h_initial,
        method=method,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        tol=tol,
        options={'maxiter': max_iter, 'disp': False}
    )
    
    # Check if optimization went over the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 10:
        # Fallback to initial_h_values if time is tight
        h_values = initial_h_values.copy()
        h_values = np.clip(h_values, 0, 1)
        h_values = h_values / np.sum(h_values) * (n_points / 2)
        c5_bound = objective(h_values)
        return h_values, c5_bound, n_points
    
    # Extract the result
    h_values = result.x
    h_values = np.clip(h_values, 0, 1)
    h_values = h_values / np.sum(h_values) * (n_points / 2)
    c5_bound = objective(h_values)
    
    # Ensure that the constraints are satisfied
    if abs(np.sum(h_values) - n_points / 2) > 1e-6:
        h_values = np.clip(h_values, 0, 1)
        h_values = h_values / np.sum(h_values) * (n_points / 2)
        c5_bound = objective(h_values)
    
    return h_values, c5_bound, n_points
```
