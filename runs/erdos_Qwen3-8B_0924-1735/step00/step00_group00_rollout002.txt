import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    scaling_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h = initial_h_values.copy() * scaling_factor
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scaling_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h *= scaling_factor
    h = np.clip(h, 0, 1)
    best_c5 = np.inf
    best_h = h.copy()
    start_time = time.time()
    max_iterations = 1000
    learning_rate = 0.05
    decay_rate = 0.995
    for iteration in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        q = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            idx = np.roll(np.arange(n_points), k)
            q[k] = np.dot(h, h[idx]) * dx
        min_idx = np.argmin(q)
        min_q = q[min_idx]
        k = min_idx - n_points + 1
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= (i + k) < n_points:
                grad[i] += h[(i + k)] * dx
            if 0 <= (i - k) < n_points:
                grad[i] += h[(i - k)] * dx
        h += learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scaling_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h *= scaling_factor
        h = np.clip(h, 0, 1)
        current_c5 = np.max(scipy.signal.correlate(h, 1 - h, mode='full')) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        learning_rate *= decay_rate
    h_values = best_h
    return h_values, best_c5, n_points
