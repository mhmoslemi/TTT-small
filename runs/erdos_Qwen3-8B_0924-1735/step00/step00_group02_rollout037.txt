import numpy as np
import scipy.optimize as opt
import time

# Pre-imported helpers and initial construction
# evaluate_erdos_solution and initial_h_values are available

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 256  # Use 256 samples for balance between speed and precision
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    h = h / np.sum(h) * (n_points / 2)  # Ensure the sum constraint is met

    # Objective function with surrogate for max
    def objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        return np.max(corr) * dx

    # Surrogate objective for smooth optimization
    def surrogate_objective(x):
        h_vals = x.copy()
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        return np.log(np.sum(np.exp(100 * (corr - 1e-6)))) * dx

    # Constraint: sum(h) = n_points / 2
    def constraint(x):
        return np.sum(x) - (n_points / 2)

    cons = {'type': 'eq', 'fun': constraint}

    # Bounds for h[i] in [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization
    start_time = time.time()
    result = opt.minimize(
        surrogate_objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 300, 'ftol': 1e-8, 'eps': 1e-8}
    )
    elapsed_time = time.time() - start_time

    # If time is left, apply random perturbations
    if elapsed_time < budget_s:
        # Perturb h with small noise
        h = result.x
        h += np.random.normal(0, 1e-3, size=n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

        # Re-evaluate with the exact objective
        corr = np.correlate(h, 1 - h, mode="full")
        c5_bound = np.max(corr) * dx
    else:
        h = result.x
        corr = np.correlate(h, 1 - h, mode="full")
        c5_bound = np.max(corr) * dx

    # Final check of constraints
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)

    # Recompute with exact objective
    corr = np.correlate(h, 1 - h, mode="full")
    c5_bound = np.max(corr) * dx

    return h, c5_bound, n_points
