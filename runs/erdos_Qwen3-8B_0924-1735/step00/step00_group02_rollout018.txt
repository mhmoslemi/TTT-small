```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 128
    dx = 2.0 / n_points
    initial_h_values = np.random.rand(n_points)
    initial_h_values = (initial_h_values - 0.5) * 2
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values /= np.sum(initial_h_values) * dx
    initial_h_values *= n_points / 2
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values.astype(float)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h.astype(float)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr * dx)
        return c5
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    h0 = initial_h_values.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    result = optimize.minimize(
        objective,
        h0,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100, 'ftol': 1e-6, 'disp': False}
    )
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        return (initial_h_values, objective(initial_h_values), n_points)
    
    h_opt = np.clip(result.x, 0, 1)
    h_opt = h_opt.astype(float)
    h_opt = h_opt / np.sum(h_opt) * (n_points / 2)
    h_opt = np.clip(h_opt, 0, 1)
    c5_opt = objective(h_opt)
    
    return (h_opt, c5_opt, n_points)
```
