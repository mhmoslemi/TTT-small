        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        max_index = np.argmax(corr)
        
        lag = max_index - n_points + 1
        if lag < 0:
            lag += n_points
        
        h_grad = np.zeros_like(h)
        for i in range(n_points):
            if (i + lag) < n_points:
                h_grad[i] += 1 - h[i + lag]
            if (i - lag) >= 0:
                h_grad[i] -= h[i - lag]
        
        h -= learning_rate * h_grad
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2.0
        h = np.clip(h, 0, 1)
    
    corr = np.correlate(h, 1 - h, mode='full')
    c5_bound = np.max(corr) * dx
