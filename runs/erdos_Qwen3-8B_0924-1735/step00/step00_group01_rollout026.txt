import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    
    n_points = 512
    population_size = 50
    elite_count = 10
    mutation_std = 0.02
    max_generations = 1000
    early_stop_threshold = 1e-4
    time_budget = budget_s
    
    # Validate initial_h_values if provided
    if initial_h_values is not None:
        h = np.copy(initial_h_values)
        h = (h - np.min(h)) / (np.max(h) - np.min(h))
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        best_h = h
    else:
        best_h = np.random.rand(n_points)
        best_h = best_h / np.sum(best_h) * (n_points / 2)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    def evaluate(h):
        dx = 2.0 / n_points
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    def repair(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parent2[i]
        return child
    
    def mutate(h):
        noise = np.random.normal(0, mutation_std, n_points)
        h = h + noise
        h = repair(h)
        return h
    
    population = [np.copy(best_h) for _ in range(population_size)]
    best_c5 = evaluate(best_h)
    best_h = np.copy(best_h)
    generation = 0
    start_time = time.time()
    
    while generation < max_generations and time.time() - start_time < time_budget:
        current_c5 = []
        for h in population:
            c5 = evaluate(h)
            current_c5.append(c5)
        current_c5 = np.array(current_c5)
        idx = np.argsort(current_c5)
        population = np.array(population)[idx]
        current_c5 = current_c5[idx]
        
        # Track best candidate
        if current_c5[0] < best_c5:
            best_c5 = current_c5[0]
            best_h = np.copy(population[0])
        
        # Select elites
        elites = population[:elite_count]
        
        # Generate offspring
        offspring = []
        for _ in range(population_size - elite_count):
            parent1, parent2 = np.random.choice(elites, size=2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child)
            offspring.append(child)
        population = np.vstack([elites, offspring])
        
        # Early stopping check
        if generation > 0 and abs(current_c5[0] - best_c5) < early_stop_threshold:
            mutation_std *= 0.9
        generation += 1
    
    # Final evaluation
    best_h = repair(best_h)
    c5_bound = evaluate(best_h)
    
    # Ensure constraints
    assert np.all(0 <= best_h <= 1)
    assert np.isclose(np.sum(best_h), n_points / 2, atol=1e-6)
    
    return (best_h, c5_bound, n_points)
