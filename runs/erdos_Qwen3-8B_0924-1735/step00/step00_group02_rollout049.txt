            candidate = candidate / np.sum(candidate) * (n_points / 2)

            population.append((c5, candidate))

        population.sort()
        population = np.array(population)

        elite_count = int(population_size * elite_ratio)

        new_population = []

            # Tournament selection
            tournament = np.random.choice(population_size, tournament_size, replace=False)
            best_idx = tournament[np.argmin(population[tournament, 0])]
            parent1 = population[best_idx][1]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Random crossover point
                crossover_point = np.random.randint(1, n_points)
                child = np.concatenate([parent1[:crossover_point], parent1[crossover_point:]])

                # Clone parent
                child = parent1.copy()

            # Normalize
            child = child / np.sum(child) * (n_points / 2)
            c5 = evaluate_erdos_solution(child)
            new_population.append((c5, child))
        
        # Merge elites and new population
        population = np.vstack([elites, new_population])

        current_best = population[0]
