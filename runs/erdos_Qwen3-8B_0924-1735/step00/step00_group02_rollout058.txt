```python
import numpy as np
import scipy.optimize as opt
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100  # chosen for efficiency and performance within budget
    dx = 2.0 / n_points
    initial_h_values = np.random.rand(n_points)
    initial_h_values /= np.sum(initial_h_values)
    initial_h_values *= n_points / 2
    initial_h_values = np.clip(initial_h_values, 0, 1)

    def objective(x):
        h = x.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    result = opt.minimize(objective, initial_h_values, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
    h_values = np.clip(result.x, 0, 1)
    h_values /= np.sum(h_values)
    h_values *= n_points / 2
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
```
