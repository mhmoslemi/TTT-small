import numpy as np
import scipy.optimize as opt
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    return np.max(corr) * dx

def initial_h_values(n_points):
    # This is a placeholder; actual initial values are provided in the parent context
    return np.zeros(n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Step 1: choose n_points
    dx = 2.0 / n_points
    initial_h = initial_h_values(n_points)
    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    h = initial_h.copy()

    # Define the objective as max correlation with smooth surrogate
    def objective(x):
        h_temp = x.copy()
        h_temp = np.clip(h_temp, 0, 1)
        h_temp = h_temp / np.sum(h_temp) * (n_points / 2)
        corr = np.correlate(h_temp, 1 - h_temp, mode="full")
        return np.max(corr) * dx

    # Define the constraints
    def constraint(x):
        return np.sum(x) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()

    # Stage 1: Gradient-based optimization
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    if time.time() - start_time > 0.99 * budget_s:
        return (result.x, evaluate_erdos_solution(result.x, n_points), n_points)

    h_opt = result.x
    c5_opt = evaluate_erdos_solution(h_opt, n_points)

    # Stage 2: Random perturbations and refinement
    for _ in range(10):
        noise = np.random.uniform(-0.01, 0.01, size=n_points)
        h_candidate = np.clip(h_opt + noise, 0, 1)
        h_candidate = h_candidate / np.sum(h_candidate) * (n_points / 2)
        corr = np.correlate(h_candidate, 1 - h_candidate, mode="full")
        c5_candidate = np.max(corr) * dx
        if c5_candidate < c5_opt:
            h_opt = h_candidate
            c5_opt = c5_candidate
        if time.time() - start_time > 0.99 * budget_s:
            return (h_opt, c5_opt, n_points)

    # Stage 3: Fallback to penalty method if needed
    def penalty_objective(x):
        h_temp = x.copy()
        h_temp = np.clip(h_temp, 0, 1)
        h_temp = h_temp / np.sum(h_temp) * (n_points / 2)
        corr = np.correlate(h_temp, 1 - h_temp, mode="full")
        penalty = 1e5 * (np.sum(h_temp) - n_points / 2) ** 2
        return np.max(corr) * dx + penalty

    result_penalty = opt.minimize(penalty_objective, h_opt, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    h_penalty = result_penalty.x
    c5_penalty = evaluate_erdos_solution(h_penalty, n_points)

    if c5_penalty < c5_opt:
        h_opt = h_penalty
        c5_opt = c5_penalty

    # Final check and return
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt / np.sum(h_opt) * (n_points / 2)
    return (h_opt, c5_opt, n_points)
