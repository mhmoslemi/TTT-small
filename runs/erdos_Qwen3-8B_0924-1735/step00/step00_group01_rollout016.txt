import numpy as np
import scipy.optimize as opt
import time
import random

initial_h_values = None  # Assume this is predefined elsewhere

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 512
    if initial_h_values is not None:
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    time_start = time.time()
    
    def compute_overlap(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return conv * dx
    
    for _ in range(1000):
        if time.time() - time_start > budget_s:
            break
        
        conv = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(conv)
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        k = np.argmax(conv)
        if k >= len(conv):
            k = len(conv) - 1
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i - k) % n_points
            grad[i] -= h[j]
            j = (i + k) % n_points
            grad[i] += (1 - h[j])
        
        h = h - 0.05 * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        if np.random.rand() < 0.05:
            indices = np.random.choice(n_points, size=int(n_points * 0.05), replace=False)
            h[indices] += np.random.choice([-0.1, 0.1], size=len(indices))
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    conv_final = np.correlate(best_h, 1 - best_h, mode='full')
    c5_final = np.max(conv_final) * dx
    return (best_h, c5_final, n_points)
