import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.array([0.6222, 0.6778, 0.6889, 0.6556, 0.6000, 0.5778, 0.5556, 0.5556, 0.5667, 0.5556, 0.5556, 0.5556, 0.5556, 0.5667, 0.5778, 0.6000, 0.6111, 0.6222, 0.6222, 0.6222, 0.6556, 0.6556, 0.6667, 0.6667, 0.6667, 0.6667, 0.6667, 0.6667, 0.6667, 0.6667, 0.6667, 0.6667, 0.6667, 0.6667, 0.6667, 0.6667, 0.6889, 0.6889, 0.6889, 0.6889, 0.6889, 0.6889, 0.6889, 0.6889, 0.6889, 0.6889, 0.6889])
    n_points = len(initial_h_values)
    total_sum = n_points / 2
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h_sum = np.sum(h)
    if np.abs(h_sum - total_sum) > 1e-6:
        h = (h / h_sum) * total_sum
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    start_time = time.time()
    max_iter = 200
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_index = np.argmax(corr)
        max_corr_value = corr[max_corr_index]
        if max_corr_value <= best_c5:
            break
        best_c5 = max_corr_value
        k = max_corr_index
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] += (1 - h[j]) - h[(i - k) % n_points]
        step_size = 0.01 / (1 + iter)
        h += step_size * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = (h / h_sum) * total_sum
    final_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    return (h, final_c5, n_points)
