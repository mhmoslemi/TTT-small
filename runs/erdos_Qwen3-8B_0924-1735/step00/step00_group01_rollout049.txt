import numpy as np
import random
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 512
    dx = 2.0 / n_points
    budget_s_remaining = budget_s
    start_time = time.time()
    
    # Parent validation
    if initial_h_values is not None:
        h = initial_h_values.copy()
        h = (h - np.min(h)) / (np.max(h) - np.min(h))
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 if i % 4 == 0 else 0.0
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
    
    def compute_c5(h):
        c = correlate(h, 1 - h, mode="full")
        return np.max(c) * dx
    
    def rescale(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        scaling = (n_points / 2) / total
        h = h * scaling
        return h
    
    def evaluate_candidate(h):
        h = rescale(h)
        c5 = compute_c5(h)
        return c5
    
    # Population initialization
    population_size = 50
    population = [h.copy() for _ in range(population_size)]
    for i in range(population_size):
        noise = np.random.normal(0, 0.1, n_points)
        population[i] += noise
        population[i] = rescale(population[i])
    
    best_h = population[0]
    best_c5 = evaluate_candidate(best_h)
    
    mutation_std = 0.02
    max_generations = 1000
    elite_count = 10
    generation = 0
    
    while generation < max_generations and time.time() - start_time < budget_s_remaining:
        # Evaluate all candidates
        fitness = [evaluate_candidate(h) for h in population]
        
        # Track best candidate
        best_index = np.argmin(fitness)
        current_c5 = fitness[best_index]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[best_index].copy()
        
        # Select top performers for breeding
        sorted_indices = np.argsort(fitness)
        elite_indices = sorted_indices[:elite_count]
        elites = [population[i] for i in elite_indices]
        
        # Generate offspring
        offspring = []
        for _ in range(10):
            parent1, parent2 = np.random.choice(elite_indices, size=2, replace=False)
            child = (population[parent1] + population[parent2]) / 2
            noise = np.random.normal(0, mutation_std, n_points)
            child += noise
            child = rescale(child)
            offspring.append(child)
        
        # Combine population
        new_population = elites + offspring
        np.random.shuffle(new_population)
        population = new_population[:population_size]

        previous_best_c5 = best_c5
        
        generation += 1

    best_h = rescale(best_h)
    best_c5 = compute_c5(best_h)
    
    # Final checks
    assert np.all(0 <= best_h <= 1), "Values out of bounds"
    assert np.isclose(np.sum(best_h), n_points / 2, atol=1e-6), "Sum constraint violated"
