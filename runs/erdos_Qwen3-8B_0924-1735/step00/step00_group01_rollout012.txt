```python
import numpy as np
import random
import time
import scipy.signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 512
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    def validate_and_rescale(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        scale = (n_points / 2) / h_sum if h_sum != 0 else 1.0
        h = h * scale
        return h
    
    def compute_c5(h):
        h_pad = np.pad(h, (0, n_points), 'constant')
        corr = scipy.signal.correlate(h_pad, 1 - h, mode='full')
        corr = corr[:2 * n_points - 1]
        return np.max(corr) * dx
    
    def evolve_population(population, mutation_std, elite_count):
        new_population = []
        for _ in range(elite_count):
            new_population.append(min(population, key=compute_c5))
        for _ in range(len(population) - elite_count):
            parent1, parent2 = random.choices(population, k=2)
            child = np.mean([parent1, parent2], axis=0)
            noise = np.random.normal(0, mutation_std, n_points)
            child += noise
            child = validate_and_rescale(child)
            new_population.append(child)
        return new_population
    
    if initial_h_values is not None:
        h = (initial_h_values - np.min(initial_h_values)) / (np.max(initial_h_values) - np.min(initial_h_values))
        h = validate_and_rescale(h)
        best_h = h
        best_c5 = compute_c5(h)
    
    population = [np.random.uniform(0, 1, n_points) for _ in range(50)]
    population = [validate_and_rescale(h) for h in population]
    population = [min(population, key=compute_c5)]
    
    start_time = time.time()
    mutation_std = 0.02
    for gen in range(1000):
        if time.time() - start_time > budget_s:
            break
        population = evolve_population(population, mutation_std, 10)
        current_c5 = compute_c5(min(population, key=compute_c5))
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = min(population, key=compute_c5)
        if gen % 10 == 0:
            mutation_std *= 0.995
    
    best_h = validate_and_rescale(best_h)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n_points)
```
