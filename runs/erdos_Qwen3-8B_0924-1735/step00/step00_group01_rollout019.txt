import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 512
    population_size = 50
    max_generations = 1000
    mutation_std = 0.02
    elite_count = 10
    early_termination_threshold = 1e-4
    early_termination_generations = 3

    # Initialize population
    def initialize_population(base_h):
        population = []
        for _ in range(population_size):
            h = base_h.copy()
            noise = np.random.uniform(-0.1, 0.1, size=n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            h += np.random.uniform(-0.05, 0.05, size=n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
        return population

    # Evaluate C5 value
    def evaluate_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    # Repair infeasible solutions
    def repair_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Crossover operation
    def crossover(parent1, parent2):
        child = np.random.rand(n_points)
        child = np.where(child < 0.5, parent1, parent2)
        return child

    # Mutation operation
    def mutate(h):
        noise = np.random.normal(0, mutation_std, size=n_points)
        h += noise
        return repair_h(h)

    # Initialize population
    if initial_h_values is not None:
        base_h = initial_h_values / np.max(initial_h_values) if np.max(initial_h_values) > 0 else initial_h_values
        base_h = base_h / np.sum(base_h) * (n_points / 2)
        base_h = np.clip(base_h, 0, 1)
        base_h = base_h / np.sum(base_h) * (n_points / 2)
        population = initialize_population(base_h)
    else:
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        for i in range(population_size):
            population[i] = repair_h(population[i])

    best_h = population[0]
    best_c5 = evaluate_c5(best_h)
    best_time = time.time()

    start_time = time.time()
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break

        # Evaluate all candidates
        fitness = [evaluate_c5(h) for h in population]
        idxs = np.argsort(fitness)
        fitness = np.sort(fitness)
        best_fitness = fitness[0]
        current_fitness = fitness[-1]

        # Check for early termination
        if gen >= 2 and abs(best_fitness - current_fitness) < early_termination_threshold:
            if gen - 2 >= early_termination_generations:
                mutation_std *= 0.7
                print(f"Early termination triggered, reducing mutation std to {mutation_std}")
                continue

        # Track best candidate
        if best_fitness < best_c5:
            best_c5 = best_fitness
            best_h = population[idxs[0]]
            best_time = time.time()

        # Select top elites
        elites = population[idxs[:elite_count]]

        # Generate new population
        new_population = elites.copy()
        for _ in range(population_size - elite_count):
            parent1, parent2 = random.sample(elites, 2)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        population = new_population

    # Final evaluation
    final_h = best_h
    final_c5 = evaluate_c5(final_h)
    final_h = repair_h(final_h)

    # Check constraints
    if not np.all(0 <= final_h <= 1):
        final_h = repair_h(final_h)
    if not np.isclose(np.sum(final_h), n_points / 2, atol=1e-6):
        final_h = repair_h(final_h)

    # Return results
    return (final_h, final_c5, n_points)
