import numpy as np
import numpy.fft as fft
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in globals() else np.random.rand(n_points)
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        scale_factor = (n_points / 2) / sum_h
    else:
        scale_factor = 1.0
    h *= scale_factor
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            scale_factor = (n_points / 2) / sum_h
        else:
            scale_factor = 1.0
        h *= scale_factor
        h = np.clip(h, 0, 1)
        c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        return c5
    
    def objective(h):
        return compute_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = optimize.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    
    if time.time() - start_time > budget_s:
        print("Time budget exceeded, returning initial solution.")
        return (initial_h_values, compute_c5(initial_h_values), n_points)
    
    h_opt = result.x
    h_opt = np.clip(h_opt, 0, 1)
    sum_h = np.sum(h_opt)
    if sum_h != 0:
        scale_factor = (n_points / 2) / sum_h
    else:
        scale_factor = 1.0
    h_opt *= scale_factor
    h_opt = np.clip(h_opt, 0, 1)
    c5_opt = compute_c5(h_opt)
    
    if c5_opt < best_c5:
        best_h = h_opt
        best_c5 = c5_opt
    
    return (best_h, best_c5, n_points)
