import numpy as np
import numpy.random as npr
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    npr.seed(seed)
    # Predefine parameters
    max_gen = 100
    pop_size = 50
    elite_ratio = 0.2
    tournament_size = 5
    mutation_rate = 0.3
    mutation_scale = 0.1
    diversity_freq = 20
    dx = 2.0
    n_points = len(initial_h_values)
    
    # Initialize population
    def generate_individual():
        h = npr.uniform(0, 1, n_points)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if npr.rand() < mutation_rate:
                h[i] += npr.laplace(0, mutation_scale)
        return clip_and_normalize(h)
    
    def crossover(parent1, parent2):
        point = npr.randint(1, n_points - 1)
        child = np.concatenate([parent1[:point], parent2[point:]])
        return clip_and_normalize(child)
    
    def tournament_selection(pop, scores):
        indices = npr.choice(len(pop), tournament_size)
        best_idx = indices[np.argmin(scores[indices])]
        return pop[best_idx]
    
    population = [initial_h_values.copy() if initial_h_values is not None else generate_individual() for _ in range(pop_size)]
    scores = [evaluate(h) for h in population]
    best_h = population[0]
    best_c5 = scores[0]
    
    start_time = time.time()
    
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break
        
        # Sort by score
        sorted_indices = np.argsort(scores)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_scores = [scores[i] for i in sorted_indices]
        
        # Keep elites
        num_elites = int(pop_size * elite_ratio)
        elites = sorted_pop[:num_elites]
        elite_scores = sorted_scores[:num_elites]
        
        # Generate new population
        new_pop = elites.copy()
        for _ in range(pop_size - num_elites):
            parent1 = tournament_selection(sorted_pop, sorted_scores)
            parent2 = tournament_selection(sorted_pop, sorted_scores)
            child = crossover(parent1, parent2)
            if npr.rand() < mutation_rate:
                child = mutate(child)
            new_pop.append(child)
        
        population = new_pop
        scores = [evaluate(h) for h in population]
        
        # Track best
        current_best_idx = np.argmin(scores)
        current_best = population[current_best_idx]
        current_c5 = scores[current_best_idx]
        
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
        
        # Inject diversity if needed
        if gen % diversity_freq == 0 and gen > 0:
            for _ in range(5):
                new_h = generate_individual()
                population.append(new_h)
                scores.append(evaluate(new_h))
        
    # Final validation
    best_h = clip_and_normalize(best_h)
    best_c5 = evaluate(best_h)
    
    # Ensure constraints
    assert np.all(0 <= best_h <= 1), "Values out of bounds"
    assert np.isclose(np.sum(best_h), n_points / 2), "Sum constraint violated"
    
    return best_h, best_c5, n_points
