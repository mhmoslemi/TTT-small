import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Start with initial guess if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 100
        h = np.full(n_points, 0.5)
        h[::2] = 1.0
        h = h * 2 / n_points
    n_points = len(h)
    dx = 2.0 / n_points
    total_sum = n_points / 2
    if np.sum(h) != total_sum:
        h *= total_sum / np.sum(h)
    # Function to compute the max overlap
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)
    # Objective function with smoothing
    def obj_func(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr * dx)
        # Smooth max with exponential
        beta = 100
        epsilon = 1e-8
        smooth_max = np.log(np.sum(np.exp(beta * (corr * dx - epsilon)))) / beta
        return smooth_max
    # Constraints: sum(h) = total_sum and 0 <= h[i] <= 1
    def constraint_sum(h_flat):
        return np.sum(h_flat) - total_sum
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    # Optimization
    start_time = time.time()
    result = optimize.minimize(obj_func, h, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'maxiter': 1000})
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h, compute_c5(h), n_points)
    h_opt = result.x
    # Clip and normalize
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * total_sum / np.sum(h_opt)
    # Fallback to penalty method if optimization stalls
    if result.success and result.fun < 0.38085:
        best_c5 = result.fun
        best_h = h_opt
    else:
        # Use penalty method
        def penalty_obj(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr * dx)
            penalty = 1e5 * (np.sum(h) - total_sum)**2
            return max_corr + penalty
        result_penalty = optimize.minimize(penalty_obj, h_opt, bounds=bounds, constraints=cons, method='SLSQP', options={'maxiter': 500})
        h_opt = result_penalty.x
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * total_sum / np.sum(h_opt)
        best_c5 = compute_c5(h_opt)
        best_h = h_opt
    # Final check
    h_final = best_h
    final_c5 = compute_c5(h_final)
    if final_c5 < 0.38085:
        return (h_final, final_c5, n_points)
    else:
        return (h_final, final_c5, n_points)
