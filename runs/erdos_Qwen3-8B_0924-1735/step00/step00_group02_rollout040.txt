import numpy as np
import random
import time
import copy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else None
    n_points = 100
    dx = 2.0 / n_points
    if initial_h_values is not None:
        h_pop = [copy.copy(initial_h_values) for _ in range(50)]
    else:
        h_pop = [np.random.rand(n_points) for _ in range(50)]
    
    def normalize(h):
        total = np.sum(h)
        if abs(total) < 1e-10:
            return h
        factor = (n_points / 2) / total
        return h * factor

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [compute_c5(normalize(h)) for h in h_pop]
        
        # Select elites
        elite_idx = np.argsort(fitness)[:int(len(h_pop) * 0.2)]
        elites = [h_pop[i] for i in elite_idx]
        current_best = min(elites, key=compute_c5)
        current_best_c5 = compute_c5(current_best)
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best
        
        # Create new population
        new_pop = []
        for _ in range(50):
            # Tournament selection
            candidates = np.random.choice(h_pop, 5, replace=False)
            winner = min(candidates, key=compute_c5)
            new_pop.append(winner)
        
        # Apply mutation
        for i in range(len(new_pop)):
            h = new_pop[i]
            if np.random.rand() < 0.3:
                idx = np.random.choice(n_points)
                delta = np.random.laplace(0, 0.1)
                h[idx] += delta
                h = np.clip(h, 0, 1)
                h = normalize(h)
                new_pop[i] = h
        
        h_pop = new_pop
        
        # Inject diversity if needed
        if gen % 20 == 0 and gen > 0:
            new_pop = [np.random.rand(n_points) for _ in range(5)]
            h_pop = new_pop + h_pop[:45]
    
    best_h = normalize(best_h)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_points
