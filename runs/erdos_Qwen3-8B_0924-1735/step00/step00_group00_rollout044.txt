import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Example path, actual path may vary
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    scaling_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h = initial_h_values * scaling_factor
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.inf
    max_iterations = 1000
    learning_rate = 0.05
    early_stop = 20
    start_time = time.time()

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        # Compute all Q_k
        q = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            idx = np.roll(np.arange(n_points), k)
            q[k] = np.dot(h, h[idx]) * dx

        min_idx = np.argmin(q)
        min_q = q[min_idx]
        k = min_idx - n_points + 1  # Convert to actual lag index

        # Compute gradient for this lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            if (i + k) % n_points < n_points:
                grad[i] += h[(i + k) % n_points] * dx
            if (i - k) % n_points < n_points:
                grad[i] += h[(i - k) % n_points] * dx

        # Update h
        h += learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h *= scale_factor

        # Compute current C₅
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Early stopping
        if iteration > 500 and np.isclose(current_c5, best_c5, atol=1e-6, rtol=1e-6):
            if iteration > 500 + early_stop:
                break

    # Final validation
    h_values = np.clip(best_h, 0, 1)
    sum_h = np.sum(h_values)
    if abs(sum_h - n_points / 2) > 1e-6:
        h_values = best_h.copy()
        sum_h = np.sum(h_values)
        scale_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h_values *= scale_factor
        h_values = np.clip(h_values, 0, 1)
    c5_bound = np.max(np.correlate(h_values, 1 - h_values, mode='full')) * dx
    n_points = len(h_values)
    return h_values, c5_bound, n_points
