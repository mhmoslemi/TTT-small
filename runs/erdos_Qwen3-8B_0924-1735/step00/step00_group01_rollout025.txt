import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 512
    initial_h_values = np.load('initial_h_values.npy') if 'initial_h_values' in kwargs else np.zeros(n_points)
    
    # Upscale initial_h_values if provided
    if 'initial_h_values' in kwargs:
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    else:
        # Default initial construction: evenly spaced points with random values
        h = np.random.rand(n_points)
        h = (h - np.min(h)) / (np.max(h) - np.min(h))
        h = h / np.sum(h) * (n_points / 2)
    
    # Normalize and clip
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')
    learning_rate = 0.05
    iteration_limit = 1000
    perturbation_rate = 0.05
    improvement_threshold = 1e-5
    decay_factor = 0.95
    time_start = time.time()
    
    for iter in range(iteration_limit):
        if time.time() - time_start > budget_s:
            break
        
        # Compute overlap for all lags
        corr = np.correlate(h, 1 - h, mode="full")
        current_max = np.max(corr)
        current_c5 = current_max * dx
        
        # Update best result
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Gradient for the maximum overlap lag
        k_max = np.argmax(corr)
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            if (i + k_max) < n_points:
                grad[i] += (1 - h[(i + k_max)])
            if (i - k_max) >= 0:
                grad[i] -= h[(i - k_max)]
        
        # Update h with gradient descent
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        
        # Add random perturbations
        perturb = np.random.rand(n_points) < perturbation_rate
        h[perturb] += np.random.choice([-0.1, 0.1], size=np.sum(perturb))
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        
        # Check for improvement
        corr = np.correlate(h, 1 - h, mode="full")
        new_max = np.max(corr)
        new_c5 = new_max * dx
        
        if new_c5 < current_c5 - improvement_threshold:
            learning_rate *= 1.1
        elif new_c5 > current_c5 + improvement_threshold:
            learning_rate *= decay_factor
        
        if iter % 100 == 0:
            print(f"Iteration {iter}, Current C5: {new_c5:.6f}")
    
    # Final validation
    h = best_h.copy()
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    h = np.clip(h, 0, 1)
    
    corr = np.correlate(h, 1 - h, mode="full")
    final_c5 = np.max(corr) * dx
    
    # Ensure constraints are satisfied
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    h = np.clip(h, 0, 1)
    
    return (h, final_c5, n_points)
