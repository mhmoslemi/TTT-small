```python
import numpy as np
import scipy.signal
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    
    n_points = 512
    dx = 2.0 / n_points
    
    def validate_and_rescale(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.zeros(n_points)
        h = h / total * (n_points / 2)
        return h
    
    def evaluate_c5(h):
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def generate_initial_population(base_h):
        population = []
        h = base_h.copy()
        h = validate_and_rescale(h)
        population.append(h)
        for _ in range(49):
            noise = np.random.uniform(-0.1, 0.1, n_points)
            h = base_h.copy() + noise
            h = validate_and_rescale(h)
            population.append(h)
        return population
    
    def uniform_crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return validate_and_rescale(child)
    
    def mutate(parent, std=0.02):
        noise = np.random.normal(0, std, n_points)
        child = parent + noise
        return validate_and_rescale(child)
    
    def evolve_population(population, mutation_std, elite_count):
        population = np.array(population)
        scores = np.array([evaluate_c5(h) for h in population])
        best_idx = np.argmin(scores)
        best_h = population[best_idx].copy()
        
        selected = np.argsort(scores)[:elite_count]
        selected = population[selected]
        
        new_population = []
        for _ in range(len(population) - elite_count):
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child = uniform_crossover(parent1, parent2)
            child = mutate(child, mutation_std)
            new_population.append(child)
        new_population.extend(selected)
        return new_population, best_h, scores[best_idx]
    
    def run_evolution():
        if initial_h_values is not None:
            initial_h = validate_and_rescale(initial_h_values)
        else:
            initial_h = np.random.uniform(0, 1, n_points)
            initial_h = validate_and_rescale(initial_h)
        
        population = generate_initial_population(initial_h)
        mutation_std = 0.02
        best_h = initial_h.copy()
        best_c5 = evaluate_c5(best_h)
        best_time = 0
        
        start_time = time.time()
        for gen in range(1000):
            if time.time() - start_time > budget_s - 1:
                break
            population, current_best, current_c5 = evolve_population(population, mutation_std, 20)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_best.copy()
            if abs(best_c5 - current_c5) < 1e-4:
                mutation_std *= 0.7
            best_time = time.time() - start_time
        
        best_h = validate_and_rescale(best_h)
        final_c5 = evaluate_c5(best_h)
        return best_h, final_c5, n_points
    
    h_values, c5_bound, n_points = run_evolution()
    return h_values, c5_bound, n_points
```
