import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
    initial_h_values = initial_h_values[:n_points]
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = h.sum()
    if not np.isclose(sum_h, n_points / 2):
        h = h / sum_h * (n_points / 2)
    start_time = time.time()
    max_iterations = 200
    learning_rate = 0.01
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = corr.max() * dx
        max_corr_index = corr.argmax()
        lag = max_corr_index - (n_points - 1)
        if lag < 0:
            lag += n_points
        if lag >= n_points:
            lag -= n_points
        grad = np.zeros(n_points)
        for i in range(n_points):
            k = (i + lag) % n_points
            if k < n_points:
                grad[i] += 1 - h[k]
            k = (i - lag) % n_points
            if k < n_points:
                grad[i] -= h[k]
        grad = grad / np.abs(grad).max() if np.abs(grad).max() > 0 else np.zeros(n_points)
        h += learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h / h.sum() * (n_points / 2)
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = corr.max() * dx
    return h, c5, n_points
