import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5 if (i % 2 == 0) else 0.0 for i in range(n_points)])
    h = initial_h_values.copy()
    h = h / np.sum(h) * (n_points / 2)

    def objective(x):
        h_vals = x.reshape((n_points,))
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def constraint_sum(x):
        return np.sum(x) - (n_points / 2)

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
    h_opt = result.x
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt / np.sum(h_opt) * (n_points / 2)
    corr = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5_bound = np.max(corr) * dx
    return (h_opt.tolist(), c5_bound, n_points)
