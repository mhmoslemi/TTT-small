    learning_rate = 0.05
    rho = 0.5

        selected_k = 0
        for k in k_indices:
            idx = (np.arange(n_points) + k) % n_points
            overlap = np.dot(h, 1 - h[idx]) * dx
            if overlap > current_max:
                current_max = overlap
                selected_k = k

        # Gradient computation for selected lag
        k = selected_k
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            gradient[i] += (1 - h[j]) * dx
            j = (i - k) % n_points
            gradient[i] -= h[j] * dx

        h += learning_rate * gradient
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h *= scale

        # Update Lagrange multiplier
        error = sum_h - n_points / 2
        lambda_c += rho * error

        # Early stopping if no improvement
        if iteration > early_stop_iterations and np.abs(current_max - best_c5) < improvement_threshold:
            break

        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()

        # Adaptive learning rate and penalty
        if iteration % 50 == 0:
            learning_rate *= decay_rate
            if abs(error) > 1e-3:
                rho *= 1.5

    sum_h = np.sum(best_h)
