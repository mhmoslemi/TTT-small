import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # If initial_h_values exists, use it; otherwise create a random initial guess
    if 'initial_h_values' in globals():
        h = np.copy(initial_h_values)
    else:
        h = np.random.rand(512)
        h = h / np.sum(h) * (512 / 2)
        h = np.clip(h, 0, 1)
    
    n_points = 512
    dx = 2.0 / n_points
    
    # Helper function to compute C5
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if not np.isclose(h_sum, n_points / 2, atol=1e-6):
            h = h / h_sum * (n_points / 2)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h / h_sum * (n_points / 2)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initial evaluation
    best_h = np.copy(h)
    best_c5 = compute_c5(best_h)
    
    # Population size and parameters
    pop_size = 50
    elite_count = 10
    mutation_std = 0.02
    generations = 1000
    early_stop_threshold = 1e-4
    max_time = budget_s
    
    for gen in range(generations):
        if time.time() - start_time > max_time:
            break
        
        # Evaluate population
        population = [np.copy(best_h) for _ in range(pop_size)]
        for i in range(pop_size):
            if i == 0:
                candidate = np.copy(best_h)
            else:
                candidate = np.copy(best_h)
                noise = np.random.normal(0, mutation_std, n_points)
                candidate += noise
                candidate = np.clip(candidate, 0, 1)
                candidate = candidate / np.sum(candidate) * (n_points / 2)
                candidate = np.clip(candidate, 0, 1)
            population[i] = candidate
        
        # Evaluate all candidates
        fitness = []
        for candidate in population:
            c5 = compute_c5(candidate)
            fitness.append((c5, candidate))
        
        # Sort by fitness
        fitness.sort()
        best_fitness, best_candidate = fitness[0]
        
        # Update best_h if improvement is significant
        if best_fitness < best_c5 - 1e-4:
            best_h = np.copy(best_candidate)
            best_c5 = best_fitness
        
        # Select top 20 for breeding
        top_indices = np.argsort(fitness, axis=0)[:20]
        top_candidates = [fitness[i][1] for i in top_indices]
        
        # Generate offspring
        offspring = []
        for _ in range(10):
            parent1 = random.choice(top_candidates)
            parent2 = random.choice(top_candidates)
            child = (parent1 + parent2) / 2
            noise = np.random.normal(0, mutation_std, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child = child / np.sum(child) * (n_points / 2)
            child = np.clip(child, 0, 1)
            offspring.append(child)
        
        # Replace population with new offspring and elites
        population = top_candidates + offspring
        population = population[:pop_size]
        
        # Early stopping
        if gen > 0 and abs(best_c5 - fitness[0][0]) < early_stop_threshold:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    # Validate
    assert np.all(0 <= best_h <= 1), "h values out of bounds"
    assert np.isclose(np.sum(best_h), n_points / 2, atol=1e-6), "Sum constraint not satisfied"
    
    return (best_h, final_c5, n_points)
