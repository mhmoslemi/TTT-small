import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # Fallback: create a simple initial guess with 10 samples
        n_points = 10
        h = np.zeros(n_points)
        h[0] = 1
        h[-1] = 1
        h = np.concatenate([h, np.zeros(n_points - 2)])
        h = np.repeat(h, 2)
        h = h[:n_points]
        h = np.repeat(h, 2)
        h = h[:n_points]
        h = np.array(h, dtype=np.float64)
    
    n_points = len(h)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    scaling_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h *= scaling_factor
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    max_iterations = 1000
    learning_rate = 0.05
    early_stop_patience = 20
    iteration = 0
    last_improvement = 0
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s and iteration < max_iterations:
        q = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            # Compute overlap using circular convolution for lag k
            idx = np.roll(np.arange(n_points), k)
            q[k] = np.dot(h, h[idx]) * dx
        
        min_idx = np.argmin(q)
        min_q = q[min_idx]
        k = min_idx - n_points + 1
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= (i + k) < n_points:
                grad[i] += h[(i + k)] * dx
            if 0 <= (i - k) < n_points:
                grad[i] += h[(i - k)] * dx
        
        h += learning_rate * grad
        h = np.clip(h, 0, 1)
        
        sum_h = np.sum(h)
        scale_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h *= scale_factor
        
        current_c5 = np.max(scipy.signal.correlate(h, 1 - h, mode='full')) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
        
        if last_improvement >= early_stop_patience:
            break
        
        iteration += 1
    
    # Final validation
    h = best_h.copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scale_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h *= scale_factor
    h = np.clip(h, 0, 1)
    current_c5 = np.max(scipy.signal.correlate(h, 1 - h, mode='full')) * dx
    
    return h, current_c5, n_points
