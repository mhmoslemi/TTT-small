import numpy as np
import scipy.signal
import copy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 512
    dx = 2.0 / n_points

    # Initialize h_values
    initial_h_values = np.zeros(n_points)
    initial_h_values[::n_points // 4] = 1
    initial_h_values[::n_points // 4 + n_points // 8] = 0.5
    initial_h_values[::n_points // 4 + n_points // 4] = 1
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    h_values = initial_h_values.copy()

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = scipy.signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(corr)

    def update_h(h, grad, learning_rate):
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new

    best_h = h_values.copy()
    best_c5 = evaluate(best_h)
    best_iter = 0

    max_iterations = 1000
    learning_rate = 0.05
    decay_rate = 0.95
    for t in range(max_iterations):
        # Compute overlaps for all lags
        h_corr = scipy.signal.correlate(h_values, 1 - h_values, mode="full") * dx
        c5 = np.max(h_corr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_values.copy()
            best_iter = t
        if t == best_iter + 100:
            break

        # Find max lag
        k_max = np.argmax(h_corr)
        k = k_max - n_points + 1
        if k < 0:
            k += 2 * n_points
        if k >= 2 * n_points:
            k -= 2 * n_points

        # Compute gradient for k
        grad = np.zeros(n_points)
        for i in range(n_points):
            if (i + k) % n_points != 0:
                grad[i] += (1 - h_values[(i + k) % n_points])
            if (i - k) % n_points != 0:
                grad[i] -= h_values[(i - k) % n_points]

        # Update h_values
        h_values = update_h(h_values, grad, learning_rate)
        learning_rate *= decay_rate

        # Early stopping
        if np.abs(c5 - best_c5) < 1e-6:
            break

    # Final evaluation
    final_c5 = evaluate(best_h)
    return (best_h, final_c5, n_points)
