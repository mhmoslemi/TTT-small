import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 100
        h = np.zeros(n_points)
        # Initialize with a simple pattern
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        h = np.roll(h, int(n_points/4))
    
    n_points = len(h)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    target_sum = n_points / 2
    if abs(sum_h - target_sum) > 1e-6:
        h = h / sum_h * target_sum
    
    # Evaluate initial C5
    corr = np.correlate(h, 1 - h, mode='full')
    c5_initial = np.max(corr) * dx
    
    # Define constraints
    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), [target_sum], [target_sum])
    
    # Objective function
    def objective(x):
        h_current = x
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        return np.max(corr) * dx
    
    # Optimization options
    options = {
        'disp': False,
        'ftol': 1e-8,
        'maxiter': 100,
        'iprint': -1
    }
    
    start_time = time.time()
    
    # Run optimization
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options=options
    )
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 10:
        # If near budget, return the best found so far
        return (h, c5_initial, n_points)
    
    # Postprocess result
    h_opt = result.x
    if np.any(h_opt < 0) or np.any(h_opt > 1):
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt / np.sum(h_opt) * target_sum
    
    # Evaluate optimized solution
    corr_opt = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5_opt = np.max(corr_opt) * dx
    
    return (h_opt, c5_opt, n_points)
