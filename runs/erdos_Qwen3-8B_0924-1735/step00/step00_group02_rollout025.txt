import numpy as np
import scipy.optimize as opt
import warnings

# Pre-defined initial_h_values from parent construction (n=45 samples)
initial_h_values = np.array([...])  # placeholder as actual values are provided at runtime

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45  # Use the same n_points as the initial configuration for consistency
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)  # Ensure values are within bounds
    h_sum = np.sum(h)
    if abs(h_sum - n_points / 2) > 1e-6:
        h = h / h_sum * (n_points / 2)  # Normalize to meet the integral constraint
    
    def objective(x):
        h_new = np.clip(x, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        corr = np.correlate(h_new, 1 - h_new, mode='full')
        return np.max(corr) * dx
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    
    result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6, options={'maxiter': 100})
    
    if result.success and result.fun < 0.38085:
        h_final = result.x
        c5_bound = result.fun
    else:
        h_final = h
        c5_bound = objective(h_final)
    
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final / np.sum(h_final) * (n_points / 2)
    
    return h_final, c5_bound, n_points
