import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h = initial_h_values
    if initial_h is None:
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 1.0 if i % 3 == 0 else 0.0
        initial_h = np.concatenate([initial_h[:15], np.ones(15), initial_h[15:30], np.zeros(15), initial_h[30:45]])
        initial_h = np.clip(initial_h, 0, 1)
        total = np.sum(initial_h)
        initial_h = np.round(initial_h * (n_points / 2) / total, 5)
    initial_h = np.clip(initial_h, 0, 1)
    total = np.sum(initial_h)
    initial_h = np.round(initial_h * (n_points / 2) / total, 5)
    initial_h = np.clip(initial_h, 0, 1)
    h = initial_h.copy()
    dx = 2.0 / n_points
    max_iter = 200
    start_time = time.time()
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        cross_corr = scipy.signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        max_index = np.argmax(cross_corr)
        k = max_index - n_points + 1
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            if k + i < 0:
                val = 1.0
            elif k + i >= n_points:
                val = 0.0
            else:
                val = 1 - h[k + i]
            if i - k < 0:
                val -= 0.0
            elif i - k >= n_points:
                val -= 0.0
            else:
                val -= h[i - k]
            grad[i] = val
        
        # Update h
        learning_rate = 0.01 / (np.sqrt(iter + 1) + 1)
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        h = np.round(h * (n_points / 2) / total, 5)
        h = np.clip(h, 0, 1)
    
    cross_corr = scipy.signal.correlate(h, 1 - h, mode='full')
    c5 = np.max(cross_corr) * dx
    return h, c5, n_points
