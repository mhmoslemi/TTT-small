    h = h / np.sum(h) * (n_points / 2)
    h = np.clip(h, 0, 1)
    
    # Precompute dx
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Function to compute gradient for a specific lag
    def compute_gradient(h, k):
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] += (1 - h[j]) * dx
            j = (i - k) % n_points
            grad[i] -= h[j] * dx
        return grad
    
    # Optimization parameters
    learning_rate = 0.05
    temperature = 1.0
    perturbation_rate = 0.05
    iteration_limit = 1000
    min_improvement = 1e-5
    decay_factor = 0.95
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for iteration in range(iteration_limit):
        if time.time() - start_time > budget_s:
            break
        
        # Compute current C5
        current_c5 = compute_c5(h)
        if current_c5 < best_c5 - min_improvement:
            best_c5 = current_c5
            best_h = h.copy()
            learning_rate = 0.05
        else:
            learning_rate *= decay_factor
        
        # Compute max overlap
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient for k_max
        grad = compute_gradient(h, k_max)
        
        # Update h
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
