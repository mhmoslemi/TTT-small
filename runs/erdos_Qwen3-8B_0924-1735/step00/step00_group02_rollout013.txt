import numpy as np
import numpy.random as npr
import scipy.optimize as sco
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * n_points / 2
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        norm = n_points / 2
        if not np.isclose(h_sum, norm):
            h = h * norm / h_sum
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def mutate(h, mutation_rate=0.3, delta=0.1):
        h = h.copy()
        for i in range(n_points):
            if npr.rand() < mutation_rate:
                h[i] += delta * npr.laplace()
                h[i] = np.clip(h[i], 0, 1)
        h_sum = np.sum(h)
        norm = n_points / 2
        h = h * norm / h_sum
        return h
    
    def crossover(parent1, parent2):
        point = npr.randint(1, n_points)
        child = np.concatenate([parent1[:point], parent2[point:]])
        return child
    
    for gen in range(100):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        population = [h]
        for _ in range(49):
            parent1 = population[npr.choice(len(population))]
            parent2 = population[npr.choice(len(population))]
            child = crossover(parent1, parent2)
            child = mutate(child)
            population.append(child)
        scores = [evaluate(ind) for ind in population]
        sorted_pop = [p for _, p in sorted(zip(scores, population))]
        best_c5 = min(best_c5, min(scores))
        best_h = sorted_pop[0].copy()
        if best_c5 < 0.3807:
            break
        if gen % 20 == 0:
            population = sorted_pop[:int(len(sorted_pop)*0.2)]
            for _ in range(5):
                population.append(mutate(np.random.rand(n_points)))
    
    h = best_h.copy()
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    norm = n_points / 2
    if not np.isclose(h_sum, norm):
        h = h * norm / h_sum
    c5 = evaluate(h)
    return h, c5, n_points
