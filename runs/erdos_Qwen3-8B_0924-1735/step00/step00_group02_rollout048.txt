import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    h = initial_h_values.copy()
    h = h / np.sum(h) * n_points / 2
    dx = 2.0 / n_points
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def surrogate_objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        beta = 100
        max_corr = np.max(corr) * dx
        weights = np.exp(beta * (corr * dx - 1e-6))
        return np.log(np.sum(weights)) / beta
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = optimize.minimize(
        surrogate_objective,
        h.flatten(),
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 1000, 'ftol': 1e-8}
    )
    
    h_best = result.x.copy()
    h_best = np.clip(h_best, 0, 1)
    h_best = h_best * (n_points / 2) / np.sum(h_best)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        try:
            result = optimize.minimize(
                objective,
                h_best.flatten(),
                method='L-BFGS-B',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_sum}],
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            h_best = result.x.copy()
            h_best = np.clip(h_best, 0, 1)
            h_best = h_best * (n_points / 2) / np.sum(h_best)
        except:
            break
    
    h_best = np.clip(h_best, 0, 1)
    h_best = h_best * (n_points / 2) / np.sum(h_best)
    
    c5_bound = np.max(np.correlate(h_best, 1 - h_best, mode="full")) * dx
    
    return (h_best, c5_bound, n_points)
