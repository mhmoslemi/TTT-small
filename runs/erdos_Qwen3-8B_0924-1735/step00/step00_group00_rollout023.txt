import numpy as np
import scipy.optimize as opt
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction - use provided initial_h_values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Fallback to a simple uniform distribution
        n_points = 45
        h = np.full(n_points, 0.5)
    
    n_points = len(h)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    scaling_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h *= scaling_factor
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    scaling_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h *= scaling_factor
    
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    
    max_iterations = 500
    learning_rate = 0.05
    early_stop_iterations = 20
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Compute all Q_k
        q = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            idx = np.roll(np.arange(n_points), k)
            q[k] = np.dot(h, h[idx]) * dx
        
        min_idx = np.argmin(q)
        min_q = q[min_idx]
        
        # Compute gradient for the worst lag
        k = min_idx - n_points + 1
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] += h[j] * dx
            j = (i - k) % n_points
            grad[i] += h[j] * dx
        
        # Update h
        h += learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h *= scale_factor
        
        # Compute current C5
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early stopping
        if iteration > 50 and all(np.abs(current_c5 - best_c5) < 1e-7 for _ in range(early_stop_iterations)):
            break
    
    # Final validation
    h_values = np.clip(best_h, 0, 1)
    sum_h = np.sum(h_values)
    assert np.isclose(sum_h, n_points / 2, atol=1e-6), "Sum of h_values does not match constraint"
    for val in h_values:
        assert 0 <= val <= 1, "Value out of bounds"
    
    # Compute final C5
    corr = correlate(h_values, 1 - h_values, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (h_values, c5_bound, n_points)
