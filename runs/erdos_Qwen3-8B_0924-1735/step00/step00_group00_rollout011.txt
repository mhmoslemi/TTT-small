import numpy as np
import scipy.signal as signal
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Initial construction
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64)
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-6:
        h *= (n_points / 2) / sum_h
    best_h = h.copy()
    best_c5 = np.inf
    max_iterations = 400
    learning_rate = 0.05
    rho = 1.0
    lambda_c = 0.0
    last_improvement = 0
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Randomly select lags to evaluate
        k_indices = np.random.choice(2 * n_points - 1, size=10, replace=False)
        current_max = 0.0
        selected_k = 0
        
        for k in k_indices:
            idx = (np.arange(n_points) + k) % n_points
            overlap = np.dot(h, 1 - h[idx]) * dx
            if overlap > current_max:
                current_max = overlap
                selected_k = k
        
        # Compute gradient for selected lag
        k = selected_k
        grad = np.zeros(n_points)
        for i in range(n_points):
            ni = (i + k) % n_points
            grad[i] += (1 - h[ni]) * dx
            ni = (i - k) % n_points
            grad[i] -= h[ni] * dx
        
        # Update h
        h += learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h *= scale
        
        # Update Lagrange multiplier
        error = sum_h - n_points / 2
        lambda_c += rho * error
        
        # Early stopping if improvement stalls
        if iteration > 100 and abs(current_max - best_c5) < 1e-6:
            last_improvement += 1
            if last_improvement > 20:
                break
        else:
            last_improvement = 0
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        
        # Adjust learning rate and rho
        learning_rate *= 0.995
        if abs(error) > 1e-3:
            rho *= 1.5
    
    # Final validation
    sum_h = np.sum(best_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        best_h *= (n_points / 2) / sum_h
    
    # Compute final C5
    corr = signal.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (best_h.tolist(), c5_bound, n_points)
