import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = evaluate_erdos_solution()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = n_points / 2.0
    h = initial_h_values.copy()
    
    max_iter = 200
    learning_rate = 0.01
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        cross_corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(cross_corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        if c5 < 1e-10:
            break
        
        max_idx = np.argmax(cross_corr)
        lag = max_idx - (n_points - 1)
        
        grad = np.zeros_like(h)
        for i in range(n_points):
            if lag >= 0:
                j = (i + lag) % n_points
                grad[i] += (1 - h[j]) - h[i]
            else:
                j = (i + lag + n_points) % n_points
                grad[i] += (1 - h[j]) - h[i]
        
        h += learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (sum_h / np.sum(h)) if np.sum(h) != 0 else h
    
    h = best_h
    c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
    return h, c5, n_points
