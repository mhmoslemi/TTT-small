```python
import numpy as np
import scipy.signal as signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 512
    initial_h_values = np.random.rand(n_points)
    initial_h_values /= initial_h_values.sum()
    initial_h_values *= n_points / 2
    initial_h_values = np.clip(initial_h_values, 0, 1)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    max_iterations = 1000
    learning_rate = 0.05
    decay_rate = 0.95
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    
    def compute_c5(h):
        h = h.reshape(-1, 1)
        corr = signal.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    for t in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if c5 < 1e-6:
            break
        k_max = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if (k_max + i) < n_points:
                grad[i] += (1 - h[(k_max + i) % n_points]) * dx
            if (k_max - i) >= 0:
                grad[i] -= h[(k_max - i) % n_points] * dx
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        learning_rate *= decay_rate
    
    h = best_h
    c5 = compute_c5(h)
    return h, c5, n_points
```
