import numpy as np
import scipy.signal
import random
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 512
    h = np.zeros(n_points)
    
    if initial_h_values is not None:
        if len(initial_h_values) < n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        else:
            h = initial_h_values[:n_points]
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.zeros(n_points)
        h[:n_points // 2] = 1
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')
    learning_rate = 0.05
    max_iterations = 1000
    perturb_rate = 0.05
    temperature = 1.0
    min_improvement = 1e-5
    decay_factor = 0.95
    iteration = 0
    
    while time.time() - start_time < budget_s and iteration < max_iterations:
        overlap = scipy.signal.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(overlap) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        else:
            if np.random.rand() < 0.1:
                h = best_h.copy()
        
        grads = np.zeros(n_points)
        k_max = np.argmax(overlap)
        k_max = np.clip(k_max, 0, len(overlap) - 1)
        for i in range(n_points):
            k = k_max
            j = (i + k) % n_points
            grads[i] += (1 - h[j])
            j = (i - k) % n_points
            grads[i] -= h[j]
        grads = grads * dx
        h = h - learning_rate * grads
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
        if np.random.rand() < perturb_rate:
            indices = np.random.choice(n_points, int(n_points * perturb_rate), replace=False)
            h[indices] += np.random.choice([-0.1, 0.1], size=len(indices))
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
        
        if np.random.rand() < 0.05:
            h = best_h.copy()
        
        if current_c5 < best_c5 - min_improvement:
            learning_rate *= 1.1
        else:
            learning_rate *= decay_factor
        
        if iteration % 100 == 0 and np.random.rand() < 0.2:
            h = best_h.copy()
        
        if np.random.rand() < 0.01:
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * (n_points / 2)
        
        iteration += 1
    
    overlap = scipy.signal.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(overlap) * dx
    h_values = best_h
    
    return (h_values, final_c5, n_points)
