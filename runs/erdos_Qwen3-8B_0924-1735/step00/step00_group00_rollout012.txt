import numpy as np
import scipy
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
    if len(initial_h_values) == 0:
        n_points = 45
        initial_h_values = np.full(n_points, 0.5)
    else:
        n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    if abs(sum_h - n_points / 2) > 1e-6:
        initial_h_values = np.interp(initial_h_values, [0, 1], [0, 1])
        initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = np.inf
    max_iterations = 400
    learning_rate = 0.05
    learning_rate_decay = 0.98
    rho = 0.5
    rho_increase = 1.5
    lag_history = set()
    start_time = time.time()
    
    for iteration in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        if iteration > 20 and np.abs(best_c5 - np.inf) < 1e-6:
            break
        
        # Sample lags
        lags = np.arange(1, n_points)
        np.random.shuffle(lags)
        selected_lags = lags[:10]
        
        current_max = -np.inf
        selected_k = 0
        
        for k in selected_lags:
            if k not in lag_history:
                idx = (np.arange(n_points) + k) % n_points
                overlap = np.dot(h, 1 - h[idx]) * dx
                if overlap > current_max:
                    current_max = overlap
                    selected_k = k
        if selected_k == 0 or selected_k in lag_history:
            selected_k = np.random.choice(lags)
        
        lag_history.add(selected_k)
        if len(lag_history) > 20:
            lag_history.pop()
        
        k = selected_k
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            gradient[i] += (1 - h[j]) * dx
            j = (i - k) % n_points
            gradient[i] -= h[j] * dx
        
        # Clip gradient and update
        gradient = np.clip(gradient, -0.5, 0.5)
        h += learning_rate * gradient
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h *= scale
        
        # Update Lagrange multiplier
        error = sum_h - n_points / 2
        lambda_c = rho * error
        if abs(error) > 1e-3:
            rho *= rho_increase
        
        # Check for improvement
        idx = (np.arange(n_points) + selected_k) % n_points
        overlap = np.dot(h, 1 - h[idx]) * dx
        if overlap < best_c5:
            best_c5 = overlap
            best_h = h.copy()
        
        # Decay learning rate
        if iteration % 50 == 0 and iteration > 0:
            learning_rate *= learning_rate_decay
    
    # Final validation
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = np.interp(best_h, [0, 1], [0, 1])
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Compute final C5
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (best_h, c5_bound, n_points)
