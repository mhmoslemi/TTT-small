import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 512
    max_iterations = 1000
    learning_rate = 0.05
    decay_rate = 0.95
    threshold = 1e-6
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    # Normalize h
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_overlap(h, n_points):
        dx = 2.0 / n_points
        overlaps = np.zeros(n_points * 2 - 1)
        for k in range(n_points * 2 - 1):
            if k < n_points:
                h_k = h
            else:
                h_k = np.roll(h, -k + n_points)
            overlap = np.dot(h, 1 - h_k) * dx
            overlaps[k] = overlap
        return overlaps
    
    def compute_gradient(h, n_points, k):
        dx = 2.0 / n_points
        if k < n_points:
            h_k = h
        else:
            h_k = np.roll(h, -k + n_points)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + k < n_points:
                grad[i] += (1 - h_k[i + k]) * dx
            if i - k >= 0:
                grad[i] -= h[i - k] * dx
        return grad
    
    best_h = h.copy()
    best_c5 = compute_overlap(h, n_points).max()
    
    for t in range(max_iterations):
        if time.time() - start_time > budget_s - 0.1:
            break
        overlaps = compute_overlap(h, n_points)
        k_max = np.argmax(overlaps)
        grad = compute_gradient(h, n_points, k_max)
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        current_c5 = overlaps[k_max]
        if abs(current_c5 - best_c5) < threshold:
            break
        best_c5 = min(best_c5, current_c5)
    
    # Evaluate final h
    final_c5 = compute_overlap(h, n_points).max()
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (h, best_c5, n_points)
