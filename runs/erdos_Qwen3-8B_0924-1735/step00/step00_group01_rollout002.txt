import numpy as np
import scipy.optimize as opt
import random
import time

initial_h_values = None  # Placeholder for the initial parent construction

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 512
    if initial_h_values is not None:
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    else:
        # Initial construction: evenly spaced values with 0.5 density
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 4 == 0:
                h[i] = 1.0
        h = h / np.sum(h) * (n_points / 2)

    h = np.clip(h, 0.0, 1.0)
    h = h / np.sum(h) * (n_points / 2)

    def compute_c5(h):
        h = np.clip(h, 0.0, 1.0)
        h = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode='full')
        return corr.max() * (2.0 / n_points)

    def objective(x):
        x = np.clip(x, 0.0, 1.0)
        sum_x = np.sum(x)
        x = x / sum_x * (n_points / 2)
        return compute_c5(x)

    def constraints(x):
        return [np.sum(x) - n_points / 2]

    def constraint_func(x):
        return np.array([np.sum(x) - n_points / 2])

    # Use a custom optimization with some simulated annealing
    best_h = h.copy()
    best_c5 = compute_c5(h)
    learning_rate = 0.05
    temp = 1.0
    perturbation_rate = 0.05
    iteration_limit = 1000
    min_improvement = 1e-5
    decay_factor = 0.95

    for iteration in range(iteration_limit):
        if time.time() - start_time > budget_s:
            break
        # Compute current c5
        current_c5 = compute_c5(h)
        # Compute correlation for all lags to find the worst overlap
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        # Compute gradient for the worst lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            if (i - k_max) % n_points >= 0:
                grad[i] -= h[(i - k_max) % n_points]
            if (i + k_max) % n_points >= 0:
                grad[i] += (1 - h[(i + k_max) % n_points])
        # Update with gradient descent
        h -= learning_rate * grad
        # Normalize and clip
        h = np.clip(h, 0.0, 1.0)
        h = h / np.sum(h) * (n_points / 2)
        # Perturb some entries
        for i in range(n_points):
            if np.random.random() < perturbation_rate:
                h[i] += np.random.choice([-0.1, 0.1])
                h[i] = np.clip(h[i], 0.0, 1.0)
        h = h / np.sum(h) * (n_points / 2)
        # Evaluate
        new_c5 = compute_c5(h)
        if new_c5 < current_c5 - min_improvement:
            best_h = h.copy()
            best_c5 = new_c5
            learning_rate *= 1.1
        else:
            learning_rate *= decay_factor
        # Simulated annealing step
        if np.random.random() < temp:
            h = best_h.copy()
        temp *= 0.99

    # Final evaluation
    final_c5 = compute_c5(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    return (best_h, final_c5, n_points)
