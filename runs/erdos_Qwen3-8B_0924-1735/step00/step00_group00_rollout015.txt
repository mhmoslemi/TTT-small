import numpy as np
import scipy.signal
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5], dtype=np.float64)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = h.sum()
    if abs(sum_h - n_points / 2) > 1e-6:
        h = h / sum_h * (n_points / 2)
    start_time = time.time()
    max_iterations = 200
    learning_rate = 0.01
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        cross_corr = scipy.signal.correlate(h, 1 - h, mode='full')
        c5 = cross_corr.max() * dx
        max_idx = cross_corr.argmax()
        lag = max_idx - (n_points - 1)
        if lag < 0:
            lag += n_points
        if lag >= 2 * n_points:
            lag -= 2 * n_points
        gradient = np.zeros_like(h)
        for i in range(n_points):
            k = (i + lag) % n_points
            if k < n_points:
                gradient[i] += (1 - h[k]) - h[i]
            k = (i - lag) % n_points
            if k < n_points:
                gradient[i] -= h[k]
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h / h.sum() * (n_points / 2)
    final_c5 = scipy.signal.correlate(h, 1 - h, mode='full').max() * dx
    return h, final_c5, n_points
