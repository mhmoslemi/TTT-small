import numpy as np
import scipy.signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Assuming preloaded from parent
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        scale = (n_points / 2) / sum_h
        h *= scale
    best_h = h.copy()
    best_c5 = np.inf
    max_iterations = 400
    learning_rate = 0.05
    rho = 0.5
    lambda_c = 0.0
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        k_indices = np.random.choice(2 * n_points - 1, size=10, replace=False)
        current_max = 0.0
        selected_k = 0
        for k in k_indices:
            idx = (np.arange(n_points) + k) % n_points
            overlap = np.dot(h, 1 - h[idx]) * dx
            if overlap > current_max:
                current_max = overlap
                selected_k = k
        
        k = selected_k
        grad = np.zeros(n_points)
        for i in range(n_points):
            idx = (i + k) % n_points
            grad[i] += (1 - h[idx]) * dx
            idx = (i - k) % n_points
            grad[i] -= h[idx] * dx
        
        h += learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h *= scale
        
        error = sum_h - n_points / 2
        lambda_c += rho * error
        
        if iteration % 50 == 0 and rho < 100:
            rho *= 1.5
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        
        if iteration > 100 and np.abs(current_max - best_c5) < 1e-6:
            break
    
    # Final validation
    sum_h = np.sum(best_h)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        scale = (n_points / 2) / sum_h
        best_h *= scale
    
    # Compute C5
    c5_bound = np.max(scipy.signal.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, c5_bound, n_points)
