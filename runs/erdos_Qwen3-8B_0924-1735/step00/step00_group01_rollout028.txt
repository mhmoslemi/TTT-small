import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 512
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h *= n_points / 2
        h = h / np.sum(h) * n_points / 2
        h = np.clip(h, 0, 1)
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        h = np.clip(h, 0, 1)
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    def gradient_descent_step(h, learning_rate):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        h = np.clip(h, 0, 1)
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        k_max = np.argmax(overlaps)
        grad = np.zeros_like(h)
        for i in range(n_points):
            for k in range(n_points):
                if k == k_max:
                    if i + k < n_points:
                        grad[i] += (1 - h[i + k]) * dx
                    if i - k >= 0:
                        grad[i] -= h[i - k] * dx
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * n_points / 2
        h_new = np.clip(h_new, 0, 1)
        return h_new
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    
    for t in range(1000):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        learning_rate = 0.05 * (0.95 ** (t // 100))
        h = gradient_descent_step(h, learning_rate)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if t % 100 == 0:
            print(f"Iteration {t}, C5={best_c5:.6f}")
    
    h = best_h
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * n_points / 2
    h = np.clip(h, 0, 1)
    
    return h, best_c5, n_points
