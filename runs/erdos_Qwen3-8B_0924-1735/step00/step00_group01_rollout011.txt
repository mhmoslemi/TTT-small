import numpy as np
import scipy.optimize as optimize
import random
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 512
    dx = 2.0 / n_points

    if initial_h_values is not None:
        h = np.interp(np.linspace(0, 2, n_points), np.sort(np.unique(initial_h_values)), initial_h_values)
    else:
        h = np.zeros(n_points)
        h[::3] = 1.0
        h[::5] = 0.5
        h[::7] = 0.0
        h = h / np.sum(h) * (n_points / 2)

    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)

    best_h = h.copy()
    best_c5 = np.inf

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2

    cons = {'type': 'eq', 'fun': constraint, 'jac': lambda h_flat: np.ones(n_points) / n_points * (n_points / 2)}

    result = optimize.minimize(objective, h.flatten(), method='SLSQP', constraints=cons, tol=1e-8, options={'maxiter': 200})
    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt / np.sum(h_opt) * (n_points / 2)

    corr = np.correlate(h_opt, 1 - h_opt, mode="full")
    c5 = np.max(corr) * dx
    return h_opt, c5, n_points
