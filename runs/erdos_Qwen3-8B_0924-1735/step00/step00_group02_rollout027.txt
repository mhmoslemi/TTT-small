import numpy as np
import scipy.optimize as optimize
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 256  # start with 256 points for balance between accuracy and speed
    h = np.random.rand(n_points) if initial_h_values is None else initial_h_values.copy()
    h = h / np.sum(h) * n_points / 2
    dx = 2.0 / n_points
    start_time = time.time()

    # Function to compute the objective and constraints
    def objective_func(x):
        nonlocal dx, n_points
        h_vals = x.reshape(-1)
        if np.any(h_vals < 0) or np.any(h_vals > 1):
            return np.inf
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        max_corr = np.max(corr) * dx
        return max_corr

    def constraint_func(x):
        return np.sum(x) - n_points / 2

    # Initial optimization with L-BFGS-B
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = optimize.minimize(
        objective_func,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_func},
        jac=None,
        options={'maxiter': 100, 'ftol': 1e-8}
    )

    # Check if optimization succeeded
    if result.success:
        h_opt = result.x
        # Apply clipping and renormalization
        h_opt = np.clip(h_opt, 0.0, 1.0)
        h_opt = h_opt / np.sum(h_opt) * n_points / 2
        # Recompute the objective
        corr = np.correlate(h_opt, 1 - h_opt, mode="full") * dx
        c5_bound = np.max(corr)
    else:
        # Fallback to penalty method
        def penalty_objective(x):
            h_vals = x.reshape(-1)
            if np.any(h_vals < 0) or np.any(h_vals > 1):
                return np.inf
            corr = np.correlate(h_vals, 1 - h_vals, mode="full") * dx
            max_corr = np.max(corr)
            penalty = 1e5 * (np.sum(h_vals) - n_points / 2) ** 2
            return max_corr + penalty

        result = optimize.minimize(
            penalty_objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            jac=None,
            options={'maxiter': 100, 'ftol': 1e-8}
        )

        # Check if penalty method succeeded
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0.0, 1.0)
            h_opt = h_opt / np.sum(h_opt) * n_points / 2
            corr = np.correlate(h_opt, 1 - h_opt, mode="full") * dx
            c5_bound = np.max(corr)
        else:
            # Fallback to initial guess
            h_opt = h
            corr = np.correlate(h_opt, 1 - h_opt, mode="full") * dx
            c5_bound = np.max(corr)

    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        # Ensure we don't exceed the budget by more than 10 seconds
        print("Early termination due to time constraints")
    return (h_opt, c5_bound, n_points)
