import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Default number of points
    n_points = 512
    
    # Initial population size and parameters
    pop_size = 50
    mutation_std = 0.02
    elite_count = 10
    max_generations = 1000
    early_stop = 1e-4
    crossover_rate = 1.0
    generations = 0
    time_start = time.time()
    
    # Define initial h_values
    if 'initial_h_values' in globals():
        h = (initial_h_values - np.min(initial_h_values)) / (np.max(initial_h_values) - np.min(initial_h_values))
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Generate random initial vector with uniform distribution
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
    
    # Initialize population
    population = [h.copy() for _ in range(pop_size)]
    best_candidate = h.copy()
    best_c5 = float('inf')
    
    def evaluate(h):
        dx = 2.0 / n_points
        h_ones = 1 - h
        corr = np.correlate(h, h_ones, mode="full")
        return np.max(corr) * dx
    
    while generations < max_generations and time.time() - time_start < budget_s:
        # Evaluate all candidates
        fitness = [evaluate(ind) for ind in population]
        
        # Update best candidate
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_candidate = population[best_idx].copy()
        
        # Select top performers
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:elite_count]
        top_parents = [population[i].copy() for i in top_indices]
        
        # Generate new population
        new_population = []
        for _ in range(pop_size - elite_count):
            # Select two parents for crossover
            parent1 = random.choice(top_parents)
            parent2 = random.choice(top_parents)
            
            # Uniform crossover with some probability
            if random.random() < crossover_rate:
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            else:
                # Simple mutation
                child = parent1.copy()
                noise = np.random.normal(0, mutation_std, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = child / np.sum(child) * (n_points / 2)
                child = np.clip(child, 0, 1)
            
            # Rescale to maintain sum constraint
            child = child / np.sum(child) * (n_points / 2)
            child = np.clip(child, 0, 1)
            new_population.append(child)
        
        # Replace old population with new population and elites
        population = top_parents + new_population
        
        # Check for early stopping
        if generations > 10 and abs(best_c5 - fitness[best_idx]) < early_stop:
            break
        
        generations += 1
    
    # Evaluate the best candidate one more time for accuracy
    best_c5 = evaluate(best_candidate)
    best_candidate = np.clip(best_candidate, 0, 1)
    best_candidate = best_candidate / np.sum(best_candidate) * (n_points / 2)
    best_candidate = np.clip(best_candidate, 0, 1)
    
    # Final checks
    assert np.all(0 <= best_candidate <= 1), "Values out of bounds"
    assert np.isclose(np.sum(best_candidate), n_points / 2, atol=1e-6), "Sum constraint violated"
    
    return best_candidate, best_c5, n_points
