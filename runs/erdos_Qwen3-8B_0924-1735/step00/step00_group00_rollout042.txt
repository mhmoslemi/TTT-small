    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    if abs(sum_h - n_points / 2) > 1e-6:
        raise ValueError("Initial h_values do not sum to n/2")
    
    h = initial_h_values.copy()
    max_iterations = 200
    lr = 0.01
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        cross_corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(cross_corr) * dx
        max_k = np.argmax(cross_corr)
        
        grad = np.zeros_like(h)
        for i in range(n_points):
            for k in range(n_points):
                if max_k == k:
                    if i + k < n_points:
                        grad[i] += (1 - h[i + k])
                    if i - k >= 0:
                        grad[i] -= h[i - k]
        
        h -= lr * grad
        h = np.clip(h, 0, 1)
        h = h - np.min(h)
        h = h / np.sum(h) * (n_points / 2)
        
        if np.max(np.abs(h - np.clip(h, 0, 1))) < 1e-6:
            break

    cross_corr = np.correlate(h, 1 - h, mode="full")
    final_c5 = np.max(cross_corr) * dx
    return h, final_c5, n_points
