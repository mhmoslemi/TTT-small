import numpy as np
import scipy.signal
import time

initial_h_values = None  # Assume this is defined elsewhere

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 512
    h = np.zeros(n_points)
    if initial_h_values is not None:
        h[:] = initial_h_values[:n_points]
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    max_iterations = 1000
    learning_rate = 0.05
    decay_rate = 0.95
    early_stop_threshold = 1e-6
    best_c5 = np.inf
    best_h = h.copy()
    start_time = time.time()

    for t in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Compute overlap integrals
        overlap = []
        for k in range(n_points):
            k_wrap = k % n_points
            current = np.sum(h[:n_points - k_wrap] * (1 - h[k_wrap:]))
            overlap.append(current)
        c5 = max(overlap)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Compute gradient for the worst lag
        k_max = np.argmax(overlap)
        k_wrap = k_max % n_points
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_wrap < n_points:
                grad[i] += 1 - h[i + k_wrap]
            if i - k_wrap >= 0:
                grad[i] -= h[i - k_wrap]
        grad = grad * (2.0 / n_points)
        # Update h
        h -= learning_rate * grad
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        # Normalize sum
        sum_h = np.sum(h)
        if sum_h != 0:
            h = h * (n_points / 2) / sum_h
        # Decay learning rate
        learning_rate *= decay_rate
        if np.abs(c5 - best_c5) < early_stop_threshold:
            break

    # Post-processing to ensure constraints
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        h = h * (n_points / 2) / sum_h
    # Re-evaluate
    overlap = []
    for k in range(n_points):
        k_wrap = k % n_points
        current = np.sum(h[:n_points - k_wrap] * (1 - h[k_wrap:]))
        overlap.append(current)
    final_c5 = max(overlap)
    return (h, final_c5, n_points)
