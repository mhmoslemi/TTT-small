    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    # Ensure sum constraint
    sum_h = np.sum(h)
    if not np.isclose(sum_h, n_points / 2):
        h = h / sum_h * (n_points / 2)
    
    start_time = time.time()
    max_iterations = 200  # Max iteration count
    learning_rate = 0.01
    grad_norm_threshold = 1e-5
    improvement_threshold = 1e-5
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Compute cross-correlation
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5_current = np.max(cross_corr) * dx
        
        # Find the worst lag
        max_index = np.argmax(cross_corr)
        lag = max_index - (n_points - 1)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            if lag + i >= 0 and lag + i < n_points:
                grad[i] += (1 - h[lag + i])  # contribution from (1 - h(x + k))
            if i - lag >= 0 and i - lag < n_points:
                grad[i] -= h[i - lag]  # contribution from h(x)
        
        # Update h
        h = h - learning_rate * grad
        
        # Clip values to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Normalize to maintain sum constraint
        sum_h = np.sum(h)
        if not np.isclose(sum_h, n_points / 2):
