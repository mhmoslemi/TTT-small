import numpy as np
import random
import time

initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100  # Start with 100 points for efficiency
    dx = 2.0 / n_points
    
    if initial_h_values is not None and len(initial_h_values) == n_points:
        population = [np.array(initial_h_values, dtype=float)]
    else:
        population = [np.random.uniform(0, 1, n_points) for _ in range(50)]
    
    # Normalize population
    for i in range(len(population)):
        population[i] = np.clip(population[i], 0, 1)
        population[i] = population[i] * (n_points / 2.0) / np.sum(population[i])
    
    best_h = population[0]
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    generation = 0
    max_generations = 100
    mutation_rate = 0.3
    mutation_scale = 0.1
    
    while generation < max_generations and time.time() - start_time < budget_s:
        generation += 1
        
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append((c5, h))
        
        # Sort by fitness
        fitness.sort()
        best_fitness, best_h = fitness[0]
        
        if best_fitness < best_c5:
            best_c5 = best_fitness
            best_h = best_h
        
        # Select elites
        elite_count = int(len(population) * 0.2)
        elites = population[:elite_count]
        
        # Generate new population
        new_population = elites[:]
        while len(new_population) < len(population):
            # Tournament selection
            tournament = random.sample(population, 5)
            winner = min(tournament, key=lambda x: evaluate_erdos_solution(x, dx))
            
            # Crossover
            if random.random() < 1.0:
                p1 = winner
                p2 = random.choice(population)
                idx = np.random.randint(1, n_points - 1)
                child = np.concatenate((p1[:idx], p2[idx:]))
            else:
                child = winner
            
            # Mutation
            if random.random() < mutation_rate:
                for i in range(n_points):
                    if random.random() < 0.1:
                        child[i] += np.random.laplace(0, mutation_scale)
                        child[i] = np.clip(child[i], 0, 1)
            
            # Normalize
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2.0) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
    
    # Final validation
    final_h = np.array(best_h, dtype=float)
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2.0) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (final_h, final_c5, n_points)
