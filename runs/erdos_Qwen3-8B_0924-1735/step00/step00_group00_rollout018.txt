```python
import numpy as np
import scipy.signal
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Assume initial construction is preloaded
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_h = h.copy()
    best_c5 = np.inf
    max_iterations = 400
    learning_rate = 0.05
    rho = 0.5
    lambda_c = 0.0
    early_stop = 0
    start_time = time.time()
    
    # Function to compute C5
    def compute_c5(h):
        h_array = np.array(h)
        corr = scipy.signal.correlate(h_array, 1 - h_array, mode='full')
        return np.max(corr * dx)
    
    # Function to enforce constraints
    def enforce_constraints(h):
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        return h * scale
    
    # Optimization loop
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        # Sample top lags
        lags = np.random.choice(2 * n_points - 1, size=10, replace=False)
        max_overlap = 0.0
        selected_k = 0
        for k in lags:
            idx = (np.arange(n_points) + k) % n_points
            overlap = np.dot(h, 1 - h[idx]) * dx
            if overlap > max_overlap:
                max_overlap = overlap
                selected_k = k
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            if (i + selected_k) % n_points < n_points:
                grad[i] += (1 - h[(i + selected_k) % n_points]) * dx
            if (i - selected_k) % n_points < n_points:
                grad[i] -= h[(i - selected_k) % n_points] * dx
        # Update h
        h += learning_rate * grad
        h = enforce_constraints(h)
        # Update Lagrange multiplier
        sum_h = np.sum(h)
        lambda_c += rho * (sum_h - n_points / 2)
        # Check for early stopping
        if _ > 100 and np.abs(max_overlap - best_c5) < 1e-6:
            early_stop += 1
            if early_stop > 20:
                break
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        # Adjust learning rate
        if _ % 50 == 0:
            learning_rate *= 0.98
        # Adjust rho
        if np.abs(sum_h - n_points / 2) > 1e-3:
            rho *= 1.5
    # Final validation
    best_h = enforce_constraints(best_h)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
```
