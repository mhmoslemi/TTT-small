```python
import numpy as np
import scipy.optimize as opt
import numpy.random as npr
import time

def run(seed=42, budget_s=600, **kwargs):
    npr.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def normalize(h):
        total = np.sum(h)
        return h * (n_points / 2) / total if total != 0 else h
    
    def mutate(h, std):
        h = h.copy()
        indices = np.random.choice(n_points, int(n_points * 0.3), replace=False)
        delta = np.random.laplace(0, std, len(indices))
        h[indices] += delta
        h = np.clip(h, 0, 1)
        return normalize(h)
    
    def crossover(p1, p2):
        idx = np.random.randint(1, n_points - 1)
        return np.concatenate([p1[:idx], p2[idx:]])
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    generations = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s and generations < 100:
        population = [mutate(h, 0.1) for _ in range(50)]
        population.append(best_h)
        population = np.array(population)
        fitness = np.array([evaluate_c5(p) for p in population])
        sorted_indices = np.argsort(fitness)
        population = population[sorted_indices[:50]]
        fitness = fitness[sorted_indices[:50]]
        
        new_population = []
        for _ in range(50):
            p1, p2 = np.random.choice(50, 2, replace=False)
            child = crossover(population[p1], population[p2])
            if np.random.rand() < 0.3:
                child = mutate(child, 0.1)
            new_population.append(child)
        
        population = np.array(new_population)
        fitness = np.array([evaluate_c5(p) for p in population])
        sorted_indices = np.argsort(fitness)
        population = population[sorted_indices[:50]]
        fitness = fitness[sorted_indices[:50]]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        if np.abs(best_c5 - 0.3807) < 1e-5:
            break
        
        generations += 1
    
    best_h = normalize(best_h)
    return best_h, best_c5, n_points
```
