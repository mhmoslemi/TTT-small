import numpy as np
import scipy.signal
import numpy.random as npr
import time

def run(seed=42, budget_s=600, **kwargs):
    npr.seed(seed)
    # Pre-imported initial_h_values (n=45 samples)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    # Enforce integral constraint
    if abs(sum_h - n_points / 2) > 1e-7:
        scale = (n_points / 2) / sum_h
        initial_h = np.clip(initial_h * scale, 0.0, 1.0)
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = np.inf
    max_iterations = 400
    learning_rate = 0.05
    rho = 0.5
    lambda_c = 0.0
    lag_history = []
    elapsed = 0.0
    start_time = time.time()
    
    for iteration in range(max_iterations):
        # Stochastic sampling of lags
        if not lag_history:
            k_indices = np.random.choice(2 * n_points - 1, size=10, replace=False)
        else:
            k_indices = np.random.choice(np.unique(lag_history), size=10, replace=False)
        
        current_max = 0.0
        selected_k = 0
        for k in k_indices:
            if k < 0:
                k += n_points
            elif k >= n_points:
                k -= n_points
            idx = (np.arange(n_points) + k) % n_points
            overlap = np.dot(h, 1 - h[idx]) * dx
            if overlap > current_max:
                current_max = overlap
                selected_k = k
        lag_history.append(selected_k)
        if len(lag_history) > 10:
            lag_history.pop(0)
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        
        # Compute gradient for selected lag
        k = selected_k
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if (i + k) % n_points < n_points:
                gradient[i] += (1 - h[(i + k) % n_points]) * dx
            if (i - k) % n_points < n_points:
                gradient[i] -= h[(i - k) % n_points] * dx
        
        # Gradient descent step
        h += learning_rate * gradient
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = (n_points / 2) / sum_h
            h *= scale
        
        # Update Lagrange multiplier
        error = sum_h - n_points / 2
        lambda_c += rho * error
        
        # Early stopping
        if iteration > 100 and np.abs(current_max - best_c5) < 1e-6:
            break
        
        # Time check
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    
    # Final validation
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-6:
        scale = (n_points / 2) / sum_h
        h = np.clip(h * scale, 0.0, 1.0)
    
    # Recompute C5 using original evaluator for accuracy
    c5 = np.max(scipy.signal.correlate(h, 1 - h, mode='full')) * dx
    return h, c5, n_points
