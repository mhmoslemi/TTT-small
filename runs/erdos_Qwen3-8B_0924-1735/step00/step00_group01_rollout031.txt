import numpy as np
import random
import time
import scipy.optimize as opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 512
    initial_h_values = np.random.rand(n_points) if 'initial_h_values' not in kwargs else kwargs['initial_h_values']
    
    # Upscale initial values if provided
    if 'initial_h_values' in kwargs:
        initial_h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    
    h = np.copy(initial_h_values)
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if not np.isclose(h_sum, n_points / 2, atol=1e-6):
        h = h - np.min(h)
        h = h / np.max(h)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    lags = np.arange(0, 2 * n_points - 1)
    
    best_h = np.copy(h)
    best_c5 = float('inf')
    
    learning_rate = 0.05
    iteration_limit = 1000
    temp = 1.0
    perturbation_rate = 0.05
    decay_factor = 0.95
    min_improvement = 1e-5
    stagnation_threshold = 300
    stagnation_count = 0
    
    for _ in range(iteration_limit):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Compute overlap for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr)
        current_c5 = max_overlap * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.copy(h)
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        if stagnation_count > stagnation_threshold:
            learning_rate *= decay_factor
            temp *= decay_factor
            stagnation_count = 0
        
        if temp > 1e-6:
            # Gradient descent update
            k_max = np.argmax(corr)
            grad = np.zeros(n_points)
            
            for i in range(n_points):
                k = k_max
                idx = (i + k) % n_points
                if idx >= 0 and idx < n_points:
                    grad[i] += (1 - h[idx])
                idx = (i - k) % n_points
                if idx >= 0 and idx < n_points:
                    grad[i] -= h[idx]
            
            grad = grad * (learning_rate / (n_points * dx))
            h = h - grad
            
            # Apply perturbation
            if random.random() < perturbation_rate:
                indices = np.random.choice(n_points, int(perturbation_rate * n_points), replace=False)
                h[indices] += np.random.choice([-0.1, 0.1], size=len(indices))
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        
        # Clip and normalize
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final validation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    max_overlap = np.max(corr)
    c5_bound = max_overlap * dx
    
    # Ensure data is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h, c5_bound, n_points)
