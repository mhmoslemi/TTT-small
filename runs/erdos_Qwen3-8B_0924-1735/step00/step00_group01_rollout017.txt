import numpy as np
import scipy.signal
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    # Predefined initial configuration
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        n_points = 512
        h = np.zeros(n_points)
        h[::4] = 1.0
        h[1::8] = 0.5
        h[2::16] = 0.25
        h[3::32] = 0.125
        h[4::64] = 0.0625
        h = h / np.sum(h) * (n_points / 2)
    
    # Normalize and clip to [0, 1]
    h = np.clip(h, 0.0, 1.0)
    h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    max_iterations = 1000
    learning_rate = 0.05
    perturbation_rate = 0.05
    
    for _ in range(max_iterations):
        # Compute overlap
        corr = scipy.signal.correlate(h, 1.0 - h, mode="full")
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Find worst lag
        lag_idx = np.argmax(corr)
        lag = lag_idx - n_points + 1
        
        # Compute gradients for this lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + lag) % n_points
            grad[i] += (1.0 - h[j]) * dx
            j = (i - lag) % n_points
            grad[i] -= h[j] * dx
        
        # Update h
        h -= learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        h = h / np.sum(h) * (n_points / 2)
        
        # Add random perturbations
        mask = np.random.rand(n_points) < perturbation_rate
        h[mask] += np.random.uniform(-0.1, 0.1, size=np.sum(mask))
        h = np.clip(h, 0.0, 1.0)
        h = h / np.sum(h) * (n_points / 2)
        
        # Check for early stopping
        if time.time() - start_time > budget_s - 5:
            break
    
    # Final computation
    corr = scipy.signal.correlate(best_h, 1.0 - best_h, mode="full")
    final_c5 = np.max(corr) * dx
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    
    # Final validation
    assert np.allclose(np.sum(best_h), n_points / 2, atol=1e-6)
    assert np.all(best_h >= 0.0)
    assert np.all(best_h <= 1.0)
    
    return (best_h, final_c5, n_points)
