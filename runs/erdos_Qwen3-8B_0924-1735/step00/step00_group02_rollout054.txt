import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    initial_h_values = np.random.rand(n_points) * 2 - 1
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)

    dx = 2.0 / n_points
    h = initial_h_values.copy()

    def objective(x):
        h[:,:] = x.reshape(n_points, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}

    start_time = time.time()
    try:
        res = optimize.minimize(objective, h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6, options={'maxiter': 200})
        best_h = res.x
    except:
        best_h = h

    def eval_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        return np.max(corr) * dx

    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)

    c5_bound = eval_c5(final_h)
    return (final_h, c5_bound, n_points)
