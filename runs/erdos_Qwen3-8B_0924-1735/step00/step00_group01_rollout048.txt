```python
import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = kwargs.get('initial_h_values', None)
    n_points = 512
    population_size = 50
    mutation_std = 0.02
    max_generations = 1000
    elite_count = 10
    early_stop_threshold = 1e-4
    max_time = budget_s
    
    # Validate initial_h_values
    if initial_h_values is not None:
        h = (initial_h_values - np.min(initial_h_values)) / (np.max(initial_h_values) - np.min(initial_h_values))
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.full(n_points, 0.5)
    
    def evaluate(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def repair(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.full(n_points, 0.5)
        h = h / total * (n_points / 2)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    def crossover(parent1, parent2):
        child = np.zeros(n_points)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child
    
    def mutate(h, std):
        noise = np.random.normal(0, std, n_points)
        h = h + noise
        return repair(h)
    
    def create_population(base_h):
        population = [base_h.copy()]
        for _ in range(population_size - 1):
            h = base_h.copy()
            noise = np.random.normal(0, 0.1, n_points)
            h += noise
            h = repair(h)
            population.append(h)
        return np.array(population)
    
    population = create_population(h)
    best_h = population[0]
    best_c5 = evaluate(best_h)
    generation = 0
    start_time = time.time()
    
    while generation < max_generations and time.time() - start_time < max_time:
        scores = [evaluate(h) for h in population]
        idxs = np.argsort(scores)
        sorted_population = population[idxs]
        sorted_scores = scores[idxs]
        
        # Select elites
        elites = sorted_population[:elite_count]
        elite_scores = sorted_scores[:elite_count]
        
        # Generate new population
        new_population = []
        for _ in range(population_size - elite_count):
            parent1, parent2 = np.random.choice(elites, size=2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child, mutation_std)
            new_population.append(child)
        
        population = np.vstack([elites, new_population])
        
        # Track best
        current_best_idx = np.argmin(scores)
        current_best_h = population[current_best_idx]
        current_best_c5 = evaluate(current_best_h)
        
        if np.abs(current_best_c5 - best_c5) < early_stop_threshold:
            mutation_std *= 0.85
        
        best_h = current_best_h
        best_c5 = current_best_c5
        generation += 1
    
    best_c5 = evaluate(best_h)
    best_h = repair(best_h)
    
    return (best_h, best_c5, n_points)
```
