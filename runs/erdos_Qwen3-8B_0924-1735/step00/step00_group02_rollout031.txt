import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 120
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    initial_h_values[::3] = 1.0
    initial_h_values[1::4] = 0.5
    initial_h_values[2::5] = 0.75
    initial_h_values = initial_h_values.astype(float)
    initial_h_values /= np.sum(initial_h_values) * dx
    initial_h_values *= n_points / 2
    h_values = initial_h_values.copy()
    
    def compute_c5(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    def objective(x):
        return compute_c5(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = Bounds(0, 1)
    initial_guess = h_values.copy()
    
    result = opt.minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8, 'disp': False}
    )
    
    h_values = result.x
    h_values = np.clip(h_values, 0, 1)
    h_values = h_values * (n_points / 2) / np.sum(h_values)
    h_values = np.clip(h_values, 0, 1)
    
    c5_bound = compute_c5(h_values)
    
    return h_values, c5_bound, n_points
