import numpy as np
import scipy.signal as signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 512
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5 if i % 2 == 0 else 0 for i in range(n_points)])
    initial_h_values = initial_h_values / np.sum(initial_h_values) * (n_points / 2)
    h = initial_h_values.copy()
    
    max_iterations = 500
    learning_rate = 0.1
    decay_rate = 0.995
    threshold = 1e-6
    
    start_time = time.time()
    
    def compute_overlap(h):
        overlap = []
        for k in range(1, len(h) + 1):
            idx = np.arange(len(h) - k)
            val = h[idx] * (1 - h[idx + k])
            overlap.append(np.sum(val) * dx)
        for k in range(1, len(h) + 1):
            idx = np.arange(len(h) - k)
            val = h[idx] * (1 - h[idx + k])
            overlap.append(np.sum(val) * dx)
        return np.max(overlap)
    
    def compute_gradient(h, k):
        grad = np.zeros(len(h))
        for i in range(len(h)):
            if i + k < len(h):
                grad[i] -= h[i] * dx
            if i - k >= 0:
                grad[i] += (1 - h[i]) * dx
        return grad
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    for t in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        c5 = compute_overlap(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        k_max = np.argmax([compute_overlap(h) for k in range(1, len(h) + 1)])
        grad = compute_gradient(h, k_max)
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        learning_rate *= decay_rate
    
    final_h = best_h
    final_c5 = compute_overlap(final_h)
    
    return (final_h, final_c5, n_points)
