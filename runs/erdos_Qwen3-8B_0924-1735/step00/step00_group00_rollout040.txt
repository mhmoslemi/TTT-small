import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # assuming this is provided
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    
    best_h = h.copy()
    best_c5 = np.inf
    
    max_iterations = 400
    learning_rate = 0.05
    decay_rate = 0.98
    rho = 0.5
    lambda_c = 0.0
    tolerance = 1e-6
    early_stop_patience = 20
    iteration = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Sample lags for evaluation
        k_indices = np.random.choice(2 * n_points - 1, size=10, replace=False)
        current_max = 0.0
        selected_k = 0
        for k in k_indices:
            idx = (np.arange(n_points) + k) % n_points
            overlap = np.dot(h, 1 - h[idx]) * dx
            if overlap > current_max:
                current_max = overlap
                selected_k = k
        
        # Compute gradient for the selected lag
        k = selected_k
        gradient = np.zeros(n_points)
        for i in range(n_points):
            pos = (i + k) % n_points
            neg = (i - k) % n_points
            gradient[i] += (1 - h[pos]) * dx
            gradient[i] -= h[neg] * dx
        
        # Update h
        h += learning_rate * gradient
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h *= scale
        
        # Update Lagrange multiplier
        error = sum_h - n_points / 2
        lambda_c += rho * error
        
        # Check for improvement
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
        
        # Early stopping
        if iteration > early_stop_patience and np.abs(current_max - best_c5) < tolerance:
            break
        
        # Decay learning rate
        learning_rate *= decay_rate
        
        iteration += 1
    
    # Final validation
    sum_h = np.sum(best_h)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    best_h *= scale
    best_h = np.clip(best_h, 0.0, 1.0)
    
    # Recompute C5 using original method
    c5_bound = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    return best_h, c5_bound, n_points
