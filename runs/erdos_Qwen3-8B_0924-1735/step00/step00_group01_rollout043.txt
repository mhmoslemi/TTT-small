import numpy as np
import scipy.signal
import time

initial_h_values = None  # Placeholder for parent construction

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 512
    max_iterations = 1000
    learning_rate = 0.05
    decay_rate = 0.95
    threshold = 1e-6
    best_c5 = float('inf')
    best_h = None

    # Function to project h to feasible space
    def project_h(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.zeros_like(h)
        scaling = (n_points / 2) / sum_h
        return h * scaling

    # Function to compute overlap integral for all lags
    def compute_c5(h):
        h_pad = np.pad(h, (0, n_points), mode='constant', constant_values=0)
        overlap = scipy.signal.correlate(h_pad, 1 - h, mode='full')
        dx = 2.0 / n_points
        c5_values = overlap * dx
        return np.max(c5_values)

    # Initialize h
    if initial_h_values is not None:
        h = np.array(initial_h_values)
        h = project_h(h)
    else:
        h = np.random.rand(n_points)
        h = project_h(h)

    start_time = time.time()

    for t in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        # Compute overlap integrals for all lags
        c5_values = compute_c5(h)
        if c5_values < best_c5:
            best_c5 = c5_values
            best_h = h.copy()

        # Find maximum lag
        h_pad = np.pad(h, (0, n_points), mode='constant', constant_values=0)
        overlap = scipy.signal.correlate(h_pad, 1 - h, mode='full')
        dx = 2.0 / n_points
        c5_values = overlap * dx
        k_max = np.argmax(c5_values)

        # Compute gradient for the worst lag
        gradient = np.zeros_like(h)
        h_pad = np.roll(h, -k_max)
        gradient += (1 - h_pad) * dx
        h_pad = np.roll(h, k_max)
        gradient -= h_pad * dx

        # Update h
        h -= learning_rate * gradient
        h = project_h(h)

        # Decay learning rate
        if t % 100 == 0 and t != 0:
            learning_rate *= decay_rate

    # Final evaluation
    final_c5 = compute_c5(h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()

    # Return the best solution found
    return (best_h, best_c5, n_points)
