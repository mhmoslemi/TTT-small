import numpy as np
import scipy.signal
import copy
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with 512 points
    n_points = 512
    dx = 2.0 / n_points
    
    # Initial construction from parent
    if 'initial_h_values' in globals():
        h = np.interp(np.linspace(0, 2, n_points), np.sort(np.unique(np.concatenate([np.linspace(0, 2, 50), np.linspace(0, 2, 50) + 1]))), initial_h_values)
    else:
        # Create a simple initial guess: evenly spaced 1s and 0s
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 4 == 0:
                h[i] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    # Clip and normalize to [0, 1] and correct sum
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    
    # Optimization parameters
    learning_rate = 0.05
    temperature = 1.0
    perturbation_rate = 0.05
    max_iterations = 1000
    improve_threshold = 1e-5
    decay_factor = 0.95
    stagnation_count = 0
    best_h = copy.deepcopy(h)
    best_c5 = np.inf
    
    for _ in range(max_iterations):
        # Compute correlation
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr)
        k_max = np.argmax(corr)
        
        # Compute gradient for k_max
        grad = np.zeros(n_points)
        for i in range(n_points):
            if (i - k_max) % n_points >= 0:
                grad[i] -= h[(i - k_max) % n_points]
            if (i + k_max) % n_points < n_points:
                grad[i] += (1 - h[(i + k_max) % n_points])
        
        # Update h
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
        # Perturbation
        perturb_indices = np.random.choice(n_points, int(n_points * perturbation_rate), replace=False)
        h[perturb_indices] += np.random.choice([-0.1, 0.1], size=len(perturb_indices))
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
        # Compute current c5
        curr_c5 = max_overlap * dx
        
        # Update best solution
        if curr_c5 < best_c5:
            best_h = copy.deepcopy(h)
            best_c5 = curr_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Decay learning rate if stagnant
        if stagnation_count > 300:
            learning_rate *= decay_factor
            stagnation_count = 0
        
        # Check time remaining
        if _ == max_iterations - 1:
            break
        
    # Final check
    h = best_h
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    corr = scipy.signal.correlate(h, 1 - h, mode='full')
    c5_bound = np.max(corr) * dx
    
    return h, c5_bound, n_points
