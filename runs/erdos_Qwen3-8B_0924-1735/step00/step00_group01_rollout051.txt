```python
import numpy as np
import scipy.signal as signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 512
    initial_h_values = np.array([0.5 if i % 2 == 0 else 0 for i in range(n_points)])
    initial_h_values = initial_h_values / np.sum(initial_h_values) * (n_points / 2)

    def compute_overlap(h):
        corr = signal.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        return np.max(corr * dx)

    def project(h):
        h[0] = max(0.0, min(1.0, h[0]))
        h[-1] = max(0.0, min(1.0, h[-1]))
        total = np.sum(h)
        if total <= 0:
            return np.ones_like(h) * (n_points / 2) / n_points
        scale = (n_points / 2) / total
        h = np.clip(h * scale, 0, 1)
        h[0] = max(0.0, min(1.0, h[0]))
        h[-1] = max(0.0, min(1.0, h[-1]))
        return h

    def gradient_step(h, learning_rate):
        corr = signal.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        max_corr = np.max(corr * dx)
        max_idx = np.argmax(corr)
        k = max_idx - n_points + 1
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + k < 0 or i + k >= n_points:
                grad[i] -= h[i]
            else:
                grad[i] += (1 - h[i + k])
        grad = grad * dx
        h_new = h - learning_rate * grad
        h_new = project(h_new)
        return h_new

    h = initial_h_values.copy()
    best_c5 = compute_overlap(h)
    best_h = h.copy()
    start_time = time.time()

    for t in range(1000):
        if time.time() - start_time > budget_s:
            break
        learning_rate = 0.05 * (0.95 ** (t // 100))
        h = gradient_step(h, learning_rate)
        c5 = compute_overlap(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if np.abs(c5 - best_c5) < 1e-6:
            break

    return best_h, best_c5, n_points
```
