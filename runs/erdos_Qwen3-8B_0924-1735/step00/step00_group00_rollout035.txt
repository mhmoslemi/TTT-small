import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.array([0.9429, 0.9429, 0.9429, 0.9429, 0.9429,
                                 0.9429, 0.9429, 0.9429, 0.9429, 0.9429,
                                 0.9429, 0.9429, 0.9429, 0.9429, 0.9429,
                                 0.9429, 0.9429, 0.9429, 0.9429, 0.9429,
                                 0.9429, 0.9429, 0.9429, 0.9429, 0.9429,
                                 0.9429, 0.9429, 0.9429, 0.9429, 0.9429,
                                 0.9429, 0.9429, 0.9429, 0.9429, 0.9429,
                                 0.9429, 0.9429, 0.9429, 0.9429, 0.9429,
                                 0.9429, 0.9429, 0.9429, 0.9429, 0.9429,
                                 0.9429, 0.9429, 0.9429, 0.9429, 0.9429])
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    sum_h = np.sum(h)
    if np.isclose(sum_h, n_points / 2):
        pass
    else:
        h += (n_points / 2 - sum_h) / n_points
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h
    h = np.clip(h, 0, 1)
    start_time = time.time()
    max_iterations = 200
    for _ in range(max_iterations):
        current_time = time.time()
        if current_time - start_time > budget_s:
            break
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        k_max = np.argmax(cross_corr)
        grad = np.zeros_like(h)
        for i in range(n_points):
            if k_max - i >= 0 and k_max - i < n_points:
                grad[i] += (1 - h[k_max - i]) - h[i]
            if k_max + i < n_points:
                grad[i] += h[k_max + i] - (1 - h[i])
        step_size = 0.01
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        sum_h_new = np.sum(h_new)
        h_new = h_new * (n_points / 2) / sum_h_new
        h_new = np.clip(h_new, 0, 1)
        h = h_new
    cross_corr = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(cross_corr) * dx
    return (h, final_c5, n_points)
