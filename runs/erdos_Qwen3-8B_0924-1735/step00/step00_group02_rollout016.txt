import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial configuration if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Automatically generate a starting configuration if no parent exists
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    # Reserve 10% of time for validation and fallback
    remaining_time = budget_s * 0.9
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Normalize h to satisfy integral constraint
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (n_points / 2) / h_sum
    
    # Function to compute C5
    def compute_c5(h):
        h_cor = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(h_cor * dx)
        return c5
    
    # Objective: minimize the maximum overlap
    def objective(x):
        h = x.copy()
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (n_points / 2) / h_sum
        return compute_c5(h)
    
    # Constraint: sum(h) = n_points / 2
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    # Bounds for h values
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    x0 = h.copy()
    
    # Optimize
    start_time = time.time()
    result = opt.minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    # Check if optimization succeeded
    if result.success:
        h_opt = result.x.copy()
        # Ensure h_opt is within bounds and normalized
        h_opt = np.clip(h_opt, 0.0, 1.0)
        h_sum_opt = np.sum(h_opt)
        if h_sum_opt != 0:
            h_opt = h_opt * (n_points / 2) / h_sum_opt
        c5_opt = compute_c5(h_opt)
        return (h_opt.tolist(), c5_opt, n_points)
    
    # Fallback to initial configuration if optimization failed
    h = np.array(initial_h_values, dtype=float)
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (n_points / 2) / h_sum
    c5 = compute_c5(h)
    return (h.tolist(), c5, n_points)
