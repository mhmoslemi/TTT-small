import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    initial_h_values = np.array([0.2, 0.3, 0.4, 0.1, 0.1, 0.2, 0.2, 0.3, 0.3, 0.2, 0.2, 0.2, 0.3, 0.1, 0.2, 0.2, 0.2, 0.3, 0.1, 0.1, 0.2, 0.2, 0.2, 0.2, 0.3, 0.2, 0.2, 0.3, 0.1, 0.2, 0.2, 0.2, 0.2, 0.1, 0.2, 0.2, 0.3, 0.1, 0.2, 0.2, 0.2, 0.2, 0.3, 0.1, 0.2])
    n_points = len(initial_h_values)
    sum_h = np.sum(initial_h_values)
    if abs(sum_h - n_points / 2) > 1e-6:
        initial_h_values = initial_h_values / sum_h * (n_points / 2)
    
    def objective(h, dx):
        h = np.clip(h, 0, 1)
        cross_corr = np.correlate(h, 1 - h, mode='full')
        return np.max(cross_corr) * dx
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    def optimize_h(initial_h, dx, max_iter=200, learning_rate=0.1, decay=0.999):
        h = initial_h.copy()
        best_h = h.copy()
        best_c5 = objective(h, dx)
        for i in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            cross_corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(cross_corr)
            max_k = np.argmax(cross_corr)
            
            # Compute gradient w.r.t. h
            grad = np.zeros_like(h)
            for i in range(n_points):
                for k in range(n_points):
                    if i - k >= 0 and i - k < n_points:
                        grad[i] += (1 - h[k]) - h[i - k]
                    if i + k < n_points:
                        grad[i] += (1 - h[k]) - h[i + k]
            grad /= n_points
            
            # Apply gradient descent
            h -= learning_rate * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Update best solution
            current_c5 = objective(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            learning_rate *= decay
        return best_h, best_c5
    
    dx = 2.0 / n_points
    start_time = time.time()
    optimized_h, c5_bound = optimize_h(initial_h_values, dx)
    return (optimized_h, c5_bound, n_points)
