```python
import numpy as np
import random
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        h = h[:n_points]
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    population = [h.copy() for _ in range(50)]
    best_h = h.copy()
    best_c5 = float('inf')
    generation = 0
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def mutate(h):
        h = h.copy()
        for _ in range(3):
            idx = np.random.randint(len(h))
            h[idx] += np.random.laplace(0, 0.1)
            h = np.clip(h, 0, 1)
        total = np.sum(h)
        h = h / total * (n_points / 2)
        return h
    
    def crossover(p1, p2):
        idx = np.random.randint(len(p1))
        return np.concatenate([p1[:idx], p2[idx:]])
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        if generation % 20 == 0:
            population = [mutate(h) for h in population]
        scores = [evaluate(h) for h in population]
        sorted_idx = np.argsort(scores)
        population = [population[i] for i in sorted_idx[:20]]
        new_population = []
        while len(new_population) < 50:
            p1, p2 = np.random.choice(population, 2, replace=False)
            child = crossover(p1, p2)
            if np.random.rand() < 0.3:
                child = mutate(child)
            new_population.append(child)
        population = new_population
        current_scores = [evaluate(h) for h in population]
        best_idx = np.argmin(current_scores)
        current_best = population[best_idx]
        current_c5 = current_scores[best_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
        generation += 1
        if best_c5 < 0.3808:
            break
    
    h_values = best_h
    c5_bound = evaluate(h_values)
    return (h_values, c5_bound, n_points)
```
