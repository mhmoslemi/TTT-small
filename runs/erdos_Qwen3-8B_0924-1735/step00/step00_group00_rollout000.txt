import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.array([0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333,
                                 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333,
                                 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333,
                                 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333,
                                 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333,
                                 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333,
                                 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333, 0.3333])
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = n_points / 2
    h = initial_h_values.copy()
    h = h / np.sum(h) * sum_h

    start_time = time.time()
    max_iter = 200
    learning_rate = 0.01
    for iter in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = corr * dx
        max_c5 = np.max(c5)
        # Find the maximum lag and its value
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        if k < 0:
            k += n_points
        max_val = corr[max_index]
        # Compute gradient with respect to h[i]
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + k < n_points:
                grad[i] += 1 - h[i + k]
            if i - k >= 0:
                grad[i] -= h[i - k]
        # Update h
        h -= learning_rate * grad
        # Clip and rescale
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * sum_h

    # Final evaluation
    final_corr = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (h, final_c5, n_points)
