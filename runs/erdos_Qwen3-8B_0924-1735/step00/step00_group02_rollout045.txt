import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 256  # Choose a reasonable number of points
    dx = 2.0 / n_points
    
    # Initial guess
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = (h / np.sum(h)) * (n_points / 2)  # Ensure sum constraint
    
    # Ensure h is within [0, 1] and satisfies the sum constraint
    h = np.clip(h, 0, 1)
    h = (h / np.sum(h)) * (n_points / 2)
    
    def objective(x):
        x = x.reshape(n_points)
        x = np.clip(x, 0, 1)
        x = (x / np.sum(x)) * (n_points / 2)
        corr = np.correlate(x, 1 - x, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def surrogate_objective(x):
        x = x.reshape(n_points)
        x = np.clip(x, 0, 1)
        x = (x / np.sum(x)) * (n_points / 2)
        corr = np.correlate(x, 1 - x, mode="full")
        max_corr = np.max(corr) * dx
        beta = 100
        soft_max = np.log(np.sum(np.exp(beta * (corr - 1e-6)))) * dx
        return soft_max
    
    bounds = [(0, 1)] * n_points
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    
    # Main optimization loop
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        result = optimize.minimize(
            surrogate_objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints
        )
        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            h = (h / np.sum(h)) * (n_points / 2)
            c5 = objective(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # Fallback to penalty method
            def penalty_objective(x):
                x = x.reshape(n_points)
                x = np.clip(x, 0, 1)
                x = (x / np.sum(x)) * (n_points / 2)
                corr = np.correlate(x, 1 - x, mode="full")
                max_corr = np.max(corr) * dx
                penalty = 1e5 * (np.sum(x) - n_points / 2)**2
                return max_corr + penalty
            
            result = optimize.minimize(
                penalty_objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints
            )
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = (h / np.sum(h)) * (n_points / 2)
                c5 = objective(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            else:
                break
    
    # Final check and return
    h = best_h
    h = np.clip(h, 0, 1)
    h = (h / np.sum(h)) * (n_points / 2)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return (h, c5, n_points)
