import numpy as np
import numpy.random as npr
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = 512
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.zeros(n_points, dtype=np.float64)
        idxs = np.arange(n_points)
        idxs = np.random.permutation(idxs)[:n_points // 2]
        h[idxs] = 1.0
        h /= np.sum(h) * dx
        h *= dx
    h = h.astype(np.float64)
    
    max_iterations = 1000
    learning_rate = 0.05
    decay_rate = 0.95
    threshold = 1e-6
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    for t in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
        # Compute all lags
        overlap = np.zeros(n_points * 2, dtype=np.float64)
        for k in range(n_points):
            idxs = (np.arange(n_points) + k) % n_points
            overlap[k] = np.dot(h, 1.0 - h[idxs]) * dx
            overlap[k + n_points] = np.dot(h, 1.0 - h[idxs]) * dx
        c5 = np.max(overlap)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the worst lag
        k_max = np.argmax(overlap)
        if k_max >= n_points:
            k_max -= n_points
        # Compute gradient for this lag
        grad = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] += (1.0 - h[i + k_max]) * dx
            if i - k_max >= 0:
                grad[i] -= h[i - k_max] * dx
        # Update h
        h -= learning_rate * grad
        # Clip and normalize
        h = np.clip(h, 0.0, 1.0)
        h = (h * n_points / 2.0) / np.sum(h)
        learning_rate *= decay_rate
    
    # Final evaluation
    overlap = np.zeros(n_points * 2, dtype=np.float64)
    for k in range(n_points):
        idxs = (np.arange(n_points) + k) % n_points
        overlap[k] = np.dot(h, 1.0 - h[idxs]) * dx
        overlap[k + n_points] = np.dot(h, 1.0 - h[idxs]) * dx
    c5 = np.max(overlap)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    return (best_h, c5, n_points)
