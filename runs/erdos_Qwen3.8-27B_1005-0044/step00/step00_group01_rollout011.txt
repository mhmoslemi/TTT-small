def c5_gradient(h, n):

    c5, cross = compute_c5(h, n)
    # Find active constraints (those achieving the max)
    active = np.where(cross >= c5 - 1e-10)[0]
    grad = np.zeros(n)
    for k_idx in active:
        # k_idx corresponds to lag k_idx - (n-1)
        k = k_idx - (n - 1)
        for i in range(n):
            j = i
            # contribution: h[i] * (1-h[i+k])
            # d/dh[j] = (1-h[j+k]) if j=i, minus h[j-k] if i+k=j
            ipk = i + k
            if 0 <= ipk < n:

    # Average over active constraints for a subgradient
    grad /= len(active)

    g = 1.0 - h

    c5 = np.max(cross)

    active_lags = np.where(active_mask)[0]
    
    grad = np.zeros(n)
    for lag_idx in active_lags:
        k = lag_idx - (n - 1)
        # grad_k[j] = (1 - h[j+k]) - h[j-k] with bounds
        # A[j] = 1 - h[j+k] if 0 <= j+k < n
        # B[j] = h[j-k] if 0 <= j-k < n
        A = np.zeros(n)

        grad += A - B

    dx = 2.0 / n
    cross = get_cross(h)
    
    if smooth_beta > 0:
        # Log-sum-exp for smooth approximation
        m = np.max(cross)
        c5 = m + np.log(np.sum(np.exp(smooth_beta * (cross - m)))) / smooth_beta
        weights = np.exp(smooth_beta * (cross - m)) / np.sum(np.exp(smooth_beta * (cross - m)))
    else:

        active = cross >= c5 - 1e-12
        weights = np.zeros(cross.shape)
        weights[active] = 1.0 / np.sum(active)
    
    grad = np.zeros(n)
    for m_idx in range(len(cross)):
        if weights[m_idx] < 1e-15:
            continue
        k = m_idx - (n - 1)
        # grad_k[j] = (1-h[j+k]) - h[j-k] (with bounds)
        A = np.zeros(n)
        B = np.zeros(n)
        if k >= 0:
            A[:n-k] = 1.0 - h[k:]
            B[k:] = h[:n-k]
        else:
            A[-k:] = 1.0 - h[:n+k]
            B[:n+k] = h[-k:]
        grad += weights[m_idx] * (A - B)
    grad *= dx
    return c5, grad

    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)

    dx = 2.0 / n

def c5_smoothed(h, beta):
    n = len(h)
    dx = 2.0 / n
    cross = get_cross(h)
    m = np.max(cross)
    es = np.exp(beta * (cross - m))
    se = np.sum(es)
    c5_val = m + np.log(se) / beta
    weights = es / se
    thresh = 1e-12
    active = np.where(weights > thresh)[0]
    grad = np.zeros(n)
    for idx in active:
        w = weights[idx]
        k = idx - (n - 1)
        if k >= 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n-k]
        else:
            grad[-k:] += w * (1.0 - h[:n+k])
            grad[:n+k] -= w * h[-k:]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_cross(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def c5_smoothed(h, beta):
    n = len(h)
    dx = 2.0 / n
    cross = get_cross(h)
    m = np.max(cross)
    es = np.exp(beta * (cross - m))
    se = np.sum(es)
    c5_val = m + np.log(se) / beta
    weights = es / se
    thresh = 1e-12
    active = np.where(weights > thresh)[0]
    grad = np.zeros(n)
    for idx in active:
        w = weights[idx]
        k = idx - (n - 1)
        if k >= 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n-k]
        else:
            kn = -k
            grad[kn:] += w * (1.0 - h[:n-kn])
            grad[:n-kn] -= w * h[kn:]
    grad *= beta * dx
    return c5_val, grad

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h
