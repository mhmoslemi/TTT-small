                best_c5 = c5_new
                best_h = h_new.copy()
                h_current = h_new.copy()
    
    # Phase 2: Nelder-Mead polish on raw max
    elapsed = time.time() - t_start
    if elapsed < budget_s - 120:
        def obj_max(x):
            return compute_c5(x)
        
        res = minimize(obj_max, h_current, method='Nelder-Mead',
                       options={'maxiter': 20000, 'xatol': 1e-10, 'fatol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = enforce_feasibility(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    # Phase 3: More SLSQP rounds from best
    elapsed = time.time() - t_start
    if elapsed < budget_s - 60:
        for alpha in [200.0, 500.0]:
            if time.time() - t_start > budget_s - 30:
                break
            def obj2(x):
                return smooth_max_obj(x, dx, alpha)
            res = minimize(obj2, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 1000, 'ftol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = enforce_feasibility(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
    
    # Phase 4: Random restarts
    elapsed = time.time() - t_start
    n_restarts = 0
    while time.time() - t_start < budget_s - 20 and n_restarts < 15:
        # Various random initializations
        rtype = n_restarts % 3
        if rtype == 0:
            h_rand = rng.uniform(0, 1, n)
        elif rtype == 1:
            h_rand = np.where(rng.rand(n) < 0.5, 0.0, 1.0)
            h_rand[rng.randint(n)] = 0.5
        else:
            # Perturb current best
            h_rand = best_h + rng.normal(0, 0.1, n)
        
        h_rand = enforce_feasibility(h_rand, n)
        
        def obj_r(x):
            return smooth_max_obj(x, dx, 50.0)
        
        res = minimize(obj_r, h_rand, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = enforce_feasibility(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        
        n_restarts += 1
    
    # Final feasibility check
    best_h = enforce_feasibility(best_h, n)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n)
