```python
def compute_gradient_smooth(h, temperature=1.0):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    # Soft max weights
    c_shifted = c - np.max(c)
    w = np.exp(c_shifted / temperature)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    # Only process significant weights for efficiency
    significant = np.where(w > 1e-6)[0]
    for k in significant:
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += w[k] * (1.0 - h[:n - lag])
            grad[:n - lag] -= w[k] * h[lag:]
        else:
            s = -lag
            grad[:n - s] += w[k] * (1.0 - h[s:])
            grad[s:] -= w[k] * h[:n - s]
    return grad
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def corr_and_grad(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(c))
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        grad[lag:] += (1.0 - h[:n - lag])
        grad[:n - lag] -= h[lag:]
    else:
        s = -lag
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] -= h[:n - s]
    return c, grad

def corr_and_smooth_grad(h, T=1.0):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cs = (c - np.max(c)) / T
    w = np.exp(cs)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-8:
            continue
        lag = k - (n - 1)
        wk = w[k]
        if lag >= 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] -= wk * h[lag:]
        else:
            s = -lag
            grad[:n - s] += wk * (1.0 - h[s:])
            grad[s:] -= wk * h[:n - s]
    return c, grad

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
```
