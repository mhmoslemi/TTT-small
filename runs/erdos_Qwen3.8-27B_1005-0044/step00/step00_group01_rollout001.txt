import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def smooth_objective(h, dx, beta):
    corr = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(corr)
    shifted = (corr - m) * beta
    return m + np.log(np.sum(np.exp(shifted)) / len(corr)) / beta

def project_feasible(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        idx = np.argsort(-room)
        for i in idx:
            add = min(room[i], diff)
            if add > 0:
                h[i] += add
                diff -= add
            if diff <= 1e-12:
                break
    else:
        idx = np.argsort(h)
        for i in idx:
            rem = min(h[i], -diff)
            if rem > 0:
                h[i] -= rem
                diff += rem
            if diff >= -1e-12:
                break
    return h

def local_optimize(h0, dx, n_points, beta=50, maxiter=2000):
    target_sum = n_points / 2.0
    bounds = [(0.0, 1.0)] * n_points
    try:
        result = minimize(
            smooth_objective, h0.copy(), args=(dx, beta),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-13, 'gtol': 1e-9}
        )
    except Exception:
        return h0.copy(), compute_c5(h0, dx)
    h_opt = project_feasible(result.x, target_sum)
    return h_opt, compute_c5(h_opt, dx)
