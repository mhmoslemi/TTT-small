        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def compute_weighted_subgrad(h, top_idx, weights):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for wi, ii in zip(weights, top_idx):
        k = int(ii - (n - 1))
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if hi1 > lo1:
            g[lo1:hi1] += wi * (1.0 - h[lo1 - k:hi1 - k]) * dx
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            g[lo2:hi2] -= wi * h[lo2 + k:hi2 + k] * dx
    return g


def optimize_subgrad(h_init, max_time, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = project_h(h_init.copy(), n)
    lr = 0.05
    best_c5 = c5_value(h)
    best_h = h.copy()
    t_start = time.time()
    no_improve = 0

    for it in range(500000):
        if time.time() - t_start > max_time:
            break
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        n_top = min(15, len(corr))
        top_idx = np.argsort(corr)[-n_top:]
        top_vals = corr[top_idx]
        temp = 8.0 + 4.0 * (it % 500) / 500.0
        w = np.exp(temp * (top_vals - top_vals.max()))
        w = w / w.sum()
        g = compute_weighted_subgrad(h, top_idx, w)
        h = h - lr * g
        h = project_h(h, n)
        c5 = c5_value(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            lr = min(lr * 1.03, 0.5)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                lr = max(lr * 0.85, 0.002)
                no_improve = 0
                if rng.rand() < 0.3:
                    h = best_h + rng.randn(n) * 0.03
                    h = project_h(h, n)
    return best_h, best_c5


def simulate_anneal(h_init, max_time, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    best_h = project_h(h_init.copy(), n)
    best_c5 = c5_value(best_h)
    h = best_h.copy()
    cur_c5 = best_c5
    temp = 0.03
    t_start = time.time()

    for it in range(200000):
        if time.time() - t_start > max_time:
            break
        pert_size = 0.02 + temp
        mask = rng.rand(n) < 0.4
        h_new = h + rng.randn(n) * pert_size * mask
        h_new = project_h(h_new, n)
        c5_new = c5_value(h_new)
        delta = c5_new - cur_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(temp, 1e-8)):
            h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        temp = max(temp * 0.9995, 0.002)
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)

    best_h = project_h(h0.copy(), n0)
    best_c5 = c5_value(best_h)

    # Phase 1: Subgradient on original resolution
    t1 = budget_s * 0.28
    h1, c5_1 = optimize_subgrad(h0, t1, seed)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: Try upsampled versions
    elapsed = time.time() - t0
    remaining = budget_s - 30 - elapsed
    if remaining > 40:
        for mult in [2, 3]:
            if time.time() - t0 > budget_s - 40:
                break
            n_new = n0 * mult
            x0 = np.arange(n0).astype(float) * (2.0 / n0)
            x1 = np.arange(n_new).astype(float) * (2.0 / n_new)
            h_up = np.interp(x1, x0, best_h, right=best_h[-1])
            h_up = project_h(h_up, n_new)
            t_alloc = min(remaining * 0.4, budget_s - 40 - (time.time() - t0))
            if t_alloc > 15:
                h_new, c5_new = optimize_subgrad(h_up, t_alloc, seed + 100 * mult)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
            remaining = budget_s - 30 - (time.time() - t0)
            if remaining < 40:
                break

    # Phase 3: Random restarts at original resolution
    elapsed = time.time() - t0
    remaining = budget_s - 30 - elapsed
    if remaining > 20:
        for trial in range(4):
            if time.time() - t0 > budget_s - 40:
                break
            h_rand = rng.rand(n0)
            h_rand = project_h(h_rand, n0)
            t_alloc = min(25.0, budget_s - 40 - (time.time() - t0))
            if t_alloc > 10:
                h_new, c5_new = optimize_subgrad(h_rand, t_alloc, seed + 200 + trial)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new

    # Phase 4: Annealing on best
    elapsed = time.time() - t0
    remaining = budget_s - 10 - elapsed
    if remaining > 15:
        best_h, best_c5 = simulate_anneal(best_h, remaining * 0.8, seed + 500)

    # Final projection and evaluation
    best_h = project_h(best_h, len(best_h))
    final_c5 = c5_value(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
