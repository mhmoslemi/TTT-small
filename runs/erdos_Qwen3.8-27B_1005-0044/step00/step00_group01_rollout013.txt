        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

    if best_h is None:
        best_h = np.full(47, 0.5)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
