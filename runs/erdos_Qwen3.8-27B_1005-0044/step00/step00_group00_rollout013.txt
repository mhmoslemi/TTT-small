import time
from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_and_weights(v, tau):
    v_max = np.max(v)
    exp_vals = np.exp((v - v_max) / tau)
    w = exp_vals / np.sum(exp_vals)
    val = v_max + tau * np.log(np.sum(exp_vals))
    return val, w

def c5_smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    v = corr * dx
    val, w = smooth_max_and_weights(v, tau)
    
    grad = np.zeros(n)
    b = 1.0 - h
    
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-12:
            continue
        # First term: b[m + k - (n-1)] for valid m
        shift1 = k - (n - 1)
        # m + shift1 in [0, n-1] → m in [-shift1, n-1-shift1]
        lo1 = max(0, -shift1)
        hi1 = min(n - 1, n - 1 - shift1)
        if lo1 <= hi1:
            m_range = np.arange(lo1, hi1 + 1)
            grad[m_range] += wk * b[m_range + shift1]

        shift2 = (n - 1) - k
        # m + shift2 in [0, n-1] → m in [-shift2, n-1-shift2]
        lo2 = max(0, -shift2)
        hi2 = min(n - 1, n - 1 - shift2)
        if lo2 <= hi2:
            m_range = np.arange(lo2, hi2 + 1)
            grad[m_range] -= wk * h[m_range + shift2]
    
    return val, grad

        if budget_s - (time.time() - t0) < 120:
            break
        n_new = n * mult
        if n_new > 1500:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        h_up = np.interp(x_new, x_old, h_best)
        h_up = fix_sum(h_up, n_new / 2.0)
        h_best, c5_best = optimize_local(h_up, 20, 0.05, t0, budget_s)
        n = n_new

    # Phase 4: Fine-tuning with random perturbations
    n = len(h_best)
    sigma = 0.05
    while budget_s - (time.time() - t0) > 30:
        h_pert = h_best + rng.normal(0, sigma, n)
        h_pert = fix_sum(h_pert, n / 2.0)
        h_pert, c5_pert = optimize_local(h_pert, 5, 0.02, t0, budget_s)
        if c5_pert < c5_best:
            h_best = h_pert
            c5_best = c5_pert
        else:
            sigma *= 0.85
        if sigma < 0.001:
            break

    # Final polish with very small tau
    h_best, c5_best = optimize_local(h_best, 10, 0.005, t0, budget_s)

    h_best = fix_sum(h_best, len(h_best) / 2.0)
    c5_best = evaluate_c5(h_best)
    return h_best, c5_best, len(h_best)
