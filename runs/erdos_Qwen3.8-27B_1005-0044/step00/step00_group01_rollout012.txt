def compute_grad_for_shift(h, k):
    """Compute gradient of R(k) w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    j = np.arange(n)
    idx1 = j - k
    valid1 = (idx1 >= 0) & (idx1 < n)
    grad[valid1] += (1.0 - h[idx1[valid1]])
    idx2 = j + k
    valid2 = (idx2 >= 0) & (idx2 < n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode="full")

def get_worst_shift(corr, n):
    max_idx = np.argmax(corr)
    return max_idx - (n - 1)

def grad_for_shift(h, k, n):
    j = np.arange(n)
    grad = np.zeros(n)
    idx1 = j - k
    v1 = (idx1 >= 0) & (idx1 < n)
    grad[v1] += (1.0 - h[idx1[v1]])
    idx2 = j + k
    v2 = (idx2 >= 0) & (idx2 < n)
    grad[v2] -= h[idx2[v2]]
    return grad

def normalize(h, n):
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    target = n / 2.0
    if s < 1e-10:
        h = np.zeros(n)
        h[:23] = 1.0

                break

            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            
            # Decide direction based on gradient
            corr = compute_correlations(h)
            k = get_worst_shift(corr, n)
            grad = grad_for_shift(h, k, n)
            
            # Move in direction that reduces the worst correlation
            di = -np.sign(grad[i]) * rng.uniform(0.01, 0.05)

            # Ensure sum is preserved

Let me think about an even faster approach. Instead of recomputing the full correlation after each small change, I can update it incrementally.

If I change h[i] by delta and h[j] by -delta, how does the correlation change?

R(k) = sum_n h[n+k] * (1-h[n])

The change in R(k) due to changing h[i] by delta and h[j] by -delta:
- h[i] appears as h[n+k] when n+k=i, i.e., n=i-k. Contribution: delta*(1-h[i-k])
- h[i] appears in (1-h[n]) when n=i. Contribution: -delta*h[i+k]
- h[j] appears as h[n+k] when n+k=j, i.e., n=j-k. Contribution: -delta*(1-h[j-k])
- h[j] appears in (1-h[n]) when n=j. Contribution: delta*h[j+k]

(All terms are 0 if the index is out of bounds.)

So ΔR(k) = delta * [(1-h[i-k]) - h[i+k] - (1-h[j-k]) + h[j+k]]
           = delta * [(1-h[i-k]) - h[i+k] - (1-h[j-k]) + h[j+k]]

This can be computed for all k quickly if I have the current h.
