    h = np.clip(h, 0, 1)

    if s == 0:
        h = np.zeros(n)
        h[:int(target_sum)] = 1
        return h
    h = h / s * target_sum
    h = np.clip(h, 0, 1)

    diff = target_sum - h.sum()

        idx = np.argsort(1 - h)

            diff -= add

    return np.max(corr) * dx

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-10:
        return h
    # Adjust
    diff = target_sum - s
    if diff > 0:
        idx = np.argsort(1.0 - h)  # most room first
    else:
        idx = np.argsort(h)  # least value first
    remaining = diff
    for i in idx:
        if remaining > 0:
            add = min(remaining, 1.0 - h[i])
            h[i] += add
            remaining -= add
        elif remaining < 0:
            sub = min(-remaining, h[i])
            h[i] -= sub
            remaining += sub
        if abs(remaining) < 1e-12:
            break
    return h

    x_old = (np.arange(n) + 0.5) * 2.0 / n

import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, s):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = s - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.argsort(1.0 - h)
    else:
        idx = np.argsort(h)
    for i in idx:
        if diff > 0:
            step = min(diff, 1.0 - h[i])
        else:
            step = max(diff, -h[i])
        h[i] += step
        diff -= step
        if abs(diff) < 1e-12:
            break
    return h

def upsample(h, n):
    m = len(h)
    x0 = (np.arange(m) + 0.5) * 2.0 / m
    x1 = (np.arange(n) + 0.5) * 2.0 / n
    return np.interp(x1, x0, h)

def thue_morse(n, s):
    t = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    if t.sum() > s:
        excess = int(t.sum() - s)
        one_idx = np.where(t == 1)[0]
        np.random.shuffle(one_idx)
        t[one_idx[:excess]] = 0
    elif t.sum() < s:
        deficit = int(s - t.sum())
        zero_idx = np.where(t == 0)[0]
        np.random.shuffle(zero_idx)
        t[zero_idx[:deficit]] = 1
    return project(t, s)

def to_binary(h, s):
    n = len(h)
    order = np.argsort(h)[::-1]
    result = np.zeros(n)
    remaining = s
    for i in order:
        if remaining >= 1.0 - 1e-10:
            result[i] = 1.0
            remaining -= 1.0
        elif remaining > 1e-10:
            result[i] = remaining
            remaining = 0.0
            break
    return result
