import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        order = np.argsort(h)
        rem = diff
        for idx in order:
            a = min(rem, 1.0 - h[idx])
            h[idx] += a
            rem -= a
            if rem < 1e-14:
                break
    else:
        order = np.argsort(h)[::-1]
        rem = -diff
        for idx in order:
            a = min(rem, h[idx])
            h[idx] -= a
            rem -= a
            if rem < 1e-14:
                break
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def grad_at_max(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k = int(np.argmax(corr))
    shift = k - (n - 1)
    g = np.zeros(n)
    for j in range(n):
        v = 0.0
        a = j - shift
        b = j + shift
        if 0 <= a < n:
            v += 1.0 - h[a]
        if 0 <= b < n:
            v -= h[b]
        g[j] = v * dx
    return g, float(corr[k] * dx)
