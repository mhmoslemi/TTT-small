```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_capped_simplex(h, target):
    """Project h onto {x : 0 <= x <= 1, sum(x) = target}"""
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def smooth_max(corr, alpha):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def gradient_smooth_max(h, alpha):
    """Compute gradient of smooth_max(corr(h), alpha) * dx w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    w = exps / exps.sum()  # weights for each lag
    # corr has indices 0..2n-2, corresponding to lags -(n-1)..(n-1)
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-15:
            continue
        k = k_idx - (n - 1)  # actual lag
        if k >= 0:
            # corr[k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
            # d/dh[j] for j in [k, n-1]: += (1-h[j-k])
            # d/dh[j] for j in [0, n-1-k]: += -h[j+k]
            n_common = n - k
            grad[k:] += wk * (1 - h[:n_common])
            grad[:n_common] -= wk * h[k:]
        else:
            kp = -k
            # corr[-kp] = sum_{i=kp}^{n-1} h[i-kp]*(1-h[i])
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
```
```python
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_capped_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_val(corr, alpha):
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

def grad_smooth_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    w = exps / exps.sum()
    grad = np.zeros(n)

            kp = -k
```
