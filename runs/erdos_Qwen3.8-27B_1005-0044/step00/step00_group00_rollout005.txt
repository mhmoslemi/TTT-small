```python
from scipy.optimize import linprog

def linearize_and_solve(h_current, target_sum, n):
    # Compute gradients of q_d at h_current
    # q_d = sum_{i valid} h[i]*(1-h[i+d])
    # dq_d/dh[j] = (1-h[j+d]) * I(j+d valid) - h[j-d] * I(j-d valid)
    
    h = h_current
    corr = np.correlate(h, 1.0 - h, mode="full")
    n_corr = len(corr)  # 2n-1
    
    # Build the LP
    # Variables: h[0], ..., h[n-1], t
    # Minimize t
    # Constraints:
    #   For each d: q_d(h) ≤ t  (linearized)
    #   sum h = target_sum
    #   0 ≤ h ≤ 1
    
    # Linearized constraint: q_d(h*) + sum_j (dq_d/dh[j])*(h[j]-h*[j]) ≤ t
    # => sum_j (dq_d/dh[j])*h[j] ≤ t - q_d(h*) + sum_j (dq_d/dh[j])*h*[j]
    # => sum_j (dq_d/dh[j])*h[j] - t ≤ -q_d(h*) + sum_j (dq_d/dh[j])*h*[j]
    
    # Compute all gradients
    # For lag d (corresponding to corr index d + n - 1):
    # dq/dd_h[j] = (1-h[j+d]) if j+d in [0,n-1] else 0
    #             - h[j-d] if j-d in [0,n-1] else 0
    
    c = np.zeros(n + 1)  # objective: minimize t (last variable)
    c[-1] = 1.0
    
    A_ub = np.zeros((n_corr, n + 1))
    b_ub = np.zeros(n_corr)
    
    for d_idx in range(n_corr):
        d = d_idx - (n - 1)  # actual lag
        grad = np.zeros(n)
        for j in range(n):
            if 0 <= j + d <= n - 1:
                grad[j] += (1.0 - h[j + d])
            if 0 <= j - d <= n - 1:
                grad[j] -= h[j - d]
        A_ub[d_idx, :n] = grad
        A_ub[d_idx, n] = -1.0  # -t
        b_ub[d_idx] = -corr[d_idx] + np.dot(grad, h)
    
    # Equality: sum h = target_sum
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    bounds_lp = [(0, 1)] * n + [(None, None)]  # h in [0,1], t free
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp, method='highs')
    
    if result.success:
        return result.x[:n], result.x[n]
    return None, None
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(x, target_sum):
    x = np.clip(x, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(x)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        if not np.any(mask):
            break
        x[mask] += diff / np.sum(mask)
        x = np.clip(x, 0.0, 1.0)
    return x

def lp_refine(h, target_sum, n, max_iters=50):
    """Iteratively linearize and solve LP to improve h."""
    h = h.copy()
    for _ in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        n_corr = len(corr)
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_ub = np.zeros((n_corr, n + 1))
        b_ub = np.zeros(n_corr)
        for d_idx in range(n_corr):

            grad = np.zeros(n)
            for j in range(n):
                if 0 <= j + d <= n - 1:
                    grad[j] += (1.0 - h[j + d])
                if 0 <= j - d <= n - 1:
                    grad[j] -= h[j - d]
            A_ub[d_idx, :n] = grad
            A_ub[d_idx, n] = -1.0
            b_ub[d_idx] = -corr[d_idx] + np.dot(grad, h)
        A_eq = np.zeros((1, n + 1))

        bounds_lp = [(0, 1)] * n + [(None, None)]
        try:

        if c5_new < c5_old:
```
