    return np.max(corr) * dx

def smooth_objective(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr - m))) / alpha

def smooth_gradient(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    # Soft max weights
    weights = np.exp(corr - m)
    weights /= np.sum(weights)
    weights = weights / alpha
    
    # For each lag m_idx (0 to 2n-2), lag = m_idx - (n-1)
    # d(corr[m_idx])/dh[i] = (1-h[i-lag]) * valid - h[i+lag] * valid
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        lag = m_idx - (n - 1)
        if weights[m_idx] < 1e-15:
            continue
        # Contribution from (1-h[i-lag]) for valid i where i-lag in [0,n-1]
        # i ranges from max(0, lag) to min(n-1, n-1+lag)
        i_start = max(0, lag)
        i_end = min(n - 1, n - 1 + lag)
        if i_start <= i_end:
            valid_i = np.arange(i_start, i_end + 1)
            grad[valid_i] += weights[m_idx] * (1.0 - h[valid_i - lag])

        j_end = min(n - 1, n - 1 - lag)
        if j_start <= j_end:

            total_room = np.sum(room)

        else:
            room = h

            if total_room > 1e-10:

    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _obj_and_grad(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1-h, mode="full") * dx
    m = corr.max()
    exp_c = np.exp(corr - m)
    logsum = m + np.log(exp_c.sum())
    obj = logsum / alpha
    w = exp_c / exp_c.sum() / alpha
    grad = np.zeros(n)
    for ki in range(2*n-1):
        if w[ki] < 1e-14: continue

def _binary_best_swap(h, dx):
    """Find the best single swap (i: 1→0, j: 0→1) that minimizes C5"""
    n = len(h)
    corr = np.correlate(h, 1-h, mode="full")
    best_c5 = np.max(corr) * dx
    best_i, best_j = -1, -1
    best_val = best_c5
    
    one_idx = np.where(h==1)[0]
    zero_idx = np.where(h==0)[0]
    
    for i in one_idx:
        # Effect of h[i]: 1→0
        delta_i = np.zeros(2*n-1)
        for ki in range(2*n-1):
            lag = ki - (n-1)
            # First factor: i = n+lag => n = i-lag, valid if n in [max(0,-lag), min(n-1,n-1-lag)]
            if max(0,-lag) <= i-lag <= min(n-1, n-1-lag):
                delta_i[ki] -= (1 - h[i-lag])
            # Second factor: i = n, valid if i in [max(0,-lag), min(n-1,n-1-lag)]
            if max(0,-lag) <= i <= min(n-1, n-1-lag):
                delta_i[ki] += h[i+lag]
        
        new_corr_base = corr + delta_i
        # For each j in zero_idx, effect of h[j]: 0→1
        # delta_j[ki] = (1-h[j-lag])*valid1 - h[j+lag]*valid2
        # Vectorize over j
        for j in zero_idx:

                if max(0,-lag) <= j-lag <= min(n-1, n-1-lag):

            c5 = np.max(new_corr) * dx
            if c5 < best_val:
                best_val = c5
                best_i, best_j = i, j
    
    if best_i >= 0:
        h[best_i] = 0
        h[best_j] = 1
    return h, best_val

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _soft_obj(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.max()
    return m + np.log(np.exp(alpha * (corr - m)).sum()) / alpha

def _soft_grad(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.max()
    exp_c = np.exp(alpha * (corr - m))
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        if p[ki] < 1e-14:
            continue
        lag = ki - (n - 1)
        s1 = max(0, lag)
        e1 = min(n - 1, n - 1 + lag)
        if s1 <= e1:
            grad[s1:e1+1] += p[ki] * (1.0 - h[s1-lag:e1-lag+1])
        s2 = max(0, -lag)
        e2 = min(n - 1, n - 1 - lag)
        if s2 <= e2:
            grad[s2:e2+1] -= p[ki] * h[s2+lag:e2+lag+1]
    grad *= dx
    return grad

    n = len(h)
    return _soft_obj(h, dx, alpha) + lam * (h.sum() - n / 2.0) ** 2

    n = len(h)
    return _soft_grad(h, dx, alpha) + 2.0 * lam * (h.sum() - n / 2.0) * np.ones(n)

    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h)

        h[idx[:target - cur]] = 1
    return h

    corr_len = 2 * n - 1
    # delta_pos[i] = change in corr when h[i] goes 1->0
    # delta_neg[i] = change in corr when h[i] goes 0->1
    delta_pos = np.zeros((n, corr_len))
    delta_neg = np.zeros((n, corr_len))
    
    for ki in range(corr_len):
        lag = ki - (n - 1)
        # For h[i]: 1->0
        # First factor contribution: -(1-h[i-lag]) if i-lag valid as n
        #   valid: max(0,-lag) <= i-lag <= min(n-1, n-1-lag)
        #   => max(lag,0) <= i <= min(n-1+lag, n-1)
        s1 = max(0, lag)
        e1 = min(n - 1, n - 1 + lag)
        if s1 <= e1:
            delta_pos[s1:e1+1, ki] = -(1.0 - h[s1-lag:e1-lag+1])
        # Second factor contribution: +h[i+lag] if i valid as n
        #   valid: max(0,-lag) <= i <= min(n-1, n-1-lag)
        s2 = max(0, -lag)
        e2 = min(n - 1, n - 1 - lag)
        if s2 <= e2:
            delta_pos[s2:e2+1, ki] += h[s2+lag:e2+lag+1]
        
        # For h[i]: 0->1
        # First factor: +(1-h[i-lag]) if valid
        if s1 <= e1:
            delta_neg[s1:e1+1, ki] = (1.0 - h[s1-lag:e1-lag+1])
        # Second factor: -h[i+lag] if valid
        if s2 <= e2:
            delta_neg[s2:e2+1, ki] = -h[s2+lag:e2+lag+1]
    
    return delta_pos, delta_neg

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _soft_obj(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.max()
    return m + np.log(np.exp(alpha * (corr - m)).sum()) / alpha

def _soft_grad(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.max()
    exp_c = np.exp(alpha * (corr - m))
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        if p[ki] < 1e-14:
            continue
        lag = ki - (n - 1)
        s1 = max(0, lag)
        e1 = min(n - 1, n - 1 + lag)
        if s1 <= e1:
            grad[s1:e1+1] += p[ki] * (1.0 - h[s1-lag:e1-lag+1])
        s2 = max(0, -lag)
        e2 = min(n - 1, n - 1 - lag)
        if s2 <= e2:
            grad[s2:e2+1] -= p[ki] * h[s2+lag:e2+lag+1]
    grad *= dx
    return grad

def _obj_pen(h, dx, alpha, lam):
    n = len(h)
    return _soft_obj(h, dx, alpha) + lam * (h.sum() - n / 2.0) ** 2

def _grad_pen(h, dx, alpha, lam):
    n = len(h)
    base = _soft_grad(h, dx, alpha)
    base += 2.0 * lam * (h.sum() - n / 2.0)
    return base

def _upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h)

def _fix_binary(h, n):
    h = np.round(np.clip(h, 0, 1))
    target = n // 2
    cur = int(h.sum())
    if cur > target:
        idx = np.where(h == 1)[0]
        h[idx[:cur - target]] = 0
    elif cur < target:
        idx = np.where(h == 0)[0]
        h[idx[:target - cur]] = 1
    return h

def _precompute_deltas(h, n):
    corr_len = 2 * n - 1
    delta_pos = np.zeros((n, corr_len))
    delta_neg = np.zeros((n, corr_len))
    for ki in range(corr_len):
        lag = ki - (n - 1)
        s1 = max(0, lag)
        e1 = min(n - 1, n - 1 + lag)
        s2 = max(0, -lag)
        e2 = min(n - 1, n - 1 - lag)
        if s1 <= e1:
            delta_pos[s1:e1+1, ki] = -(1.0 - h[s1-lag:e1-lag+1])
            delta_neg[s1:e1+1, ki] = (1.0 - h[s1-lag:e1-lag+1])
        if s2 <= e2:
            delta_pos[s2:e2+1, ki] += h[s2+lag:e2+lag+1]
            delta_neg[s2:e2+1, ki] += -h[s2+lag:e2+lag+1]
    return delta_pos, delta_neg

def _binary_local_search(h, dx, max_passes=50, time_limit=30):
    t0 = time.time()
    n = len(h)
    h = h.copy()
    best = compute_c5(h, dx)
    for _ in range(max_passes):
        if time.time() - t0 > time_limit:
            break
        delta_pos, delta_neg = _precompute_deltas(h, n)
        corr = np.correlate(h, 1.0 - h, mode="full")
        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        improved = False
        best_val = best
        best_i, best_j = -1, -1
        for i in one_idx:
            base = corr + delta_pos[i]
            candidate_c5s = (base[None, :] + delta_neg[zero_idx]).max(axis=1) * dx
            min_idx = candidate_c5s.argmin()
            if candidate_c5s[min_idx] < best_val - 1e-12:
                best_val = candidate_c5s[min_idx]
                best_i = i
                best_j = zero_idx[min_idx]
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            best = best_val
            improved = True
        if not improved:
            break
    return h, best
