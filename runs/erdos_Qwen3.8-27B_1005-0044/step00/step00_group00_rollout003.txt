import numpy as np
from scipy.optimize import brentq
import time

def project_onto_feasible(h, target_sum):
    n = len(h)
    if target_sum <= 1e-12:
        return np.zeros(n)
    if target_sum >= n - 1e-12:
        return np.ones(n)
    
    def f(lam):
        return np.clip(h - lam, 0.0, 1.0).sum() - target_sum
    
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    if f(lo) <= 0:
        return np.clip(h - lo, 0, 1)
    if f(hi) >= 0:
        return np.clip(h - hi, 0, 1)
    
    lam = brentq(f, lo, hi, xtol=1e-14)
    return np.clip(h - lam, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def compute_subgrad(h, corr, temp):
    n = len(h)
    dx = 2.0 / n
    cmax = np.max(corr)
    w = np.exp((corr - cmax) / temp)
    w /= w.sum()
    
    grad = np.zeros(n)
    j = np.arange(n)
    for m in range(len(corr)):
        wm = w[m]
        if wm < 1e-10:
            continue
        k = m - (n - 1)
        idx1 = j + k
        mask1 = (idx1 >= 0) & (idx1 < n)
        if np.any(mask1):
            g1 = np.zeros(n)
            g1[np.where(mask1)[0]] = 1.0 - h[idx1[mask1]]
            grad += wm * g1
        idx2 = j - k
        mask2 = (idx2 >= 0) & (idx2 < n)
        if np.any(mask2):
            g2 = np.zeros(n)
            g2[np.where(mask2)[0]] = h[idx2[mask2]]
            grad -= wm * g2
    
    return dx * grad

def local_search(h, target, n_swaps, rng):
    """Pairwise swap local search"""
    n = len(h)
    c5, _ = evaluate_c5(h)
    for _ in range(n_swaps):
        j = rng.randint(n)
        l = rng.randint(n)
        if j == l:
            continue
        delta = 0.05 + 0.1 * rng.rand()
        # Move delta from j to l
        if h[j] >= delta and h[l] + delta <= 1.0:
            h_old_j, h_old_l = h[j], h[l]
            h[j] -= delta
            h[l] += delta
            c5_new, _ = evaluate_c5(h)
            if c5_new < c5:
                c5 = c5_new
            else:
                h[j], h[l] = h_old_j, h_old_l
    return h, c5

    n = len(h_best)
    target = n / 2.0
    h_best = project_onto_feasible(h_best, target)
    c5_best, _ = evaluate_c5(h_best)
    
    # Phase 1: Subgradient descent
    lr = 0.05
    temp = 0.05
    it_count = 0
    no_improve = 0
    
    while time.time() - t0 < budget_s - 100:
        c5_cur, corr = evaluate_c5(h_best)
        grad = compute_subgrad(h_best, corr, temp)
        
        h_new = h_best - lr * grad
        h_new = project_onto_feasible(h_new, target)
        c5_new, _ = evaluate_c5(h_new)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
            no_improve = 0
        else:
            no_improve += 1
        
        it_count += 1
        
        if no_improve > 50:
            lr *= 0.7
            temp = max(0.01, temp * 0.9)
            no_improve = 0
            if lr < 1e-5:

        # Periodic local search
        if it_count % 100 == 0:
            h_ls = h_best.copy()
            h_ls, c5_ls = local_search(h_ls, target, 200, rng)
            if c5_ls < c5_best:

            h_best = h_ls

        else:
            break
    
    return h_best, c5_best, n

import numpy as np
from scipy.optimize import brentq
import time


def project_onto_feasible(h, target_sum):
    n = len(h)
    if target_sum <= 1e-12:
        return np.zeros(n)
    if target_sum >= n - 1e-12:
        return np.ones(n)

    def f(lam):
        return np.clip(h - lam, 0.0, 1.0).sum() - target_sum

    lo = h.min() - 1.0
    hi = h.max() + 1.0
    if f(lo) <= 0:
        return np.clip(h - lo, 0, 1)
    if f(hi) >= 0:
        return np.clip(h - hi, 0, 1)
    lam = brentq(f, lo, hi, xtol=1e-14)
    return np.clip(h - lam, 0.0, 1.0)


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr


def compute_subgrad(h, corr, temp):
    n = len(h)
    dx = 2.0 / n
    cmax = np.max(corr)
    w = np.exp((corr - cmax) / temp)
    w /= w.sum()

    grad = np.zeros(n)
    j = np.arange(n)
    for m in range(len(corr)):
        wm = w[m]
        if wm < 1e-10:
            continue
        k = m - (n - 1)
        idx1 = j + k
        mask1 = (idx1 >= 0) & (idx1 < n)
        if np.any(mask1):
            g1 = np.zeros(n)
            g1[np.where(mask1)[0]] = 1.0 - h[idx1[mask1]]
            grad += wm * g1
        idx2 = j - k
        mask2 = (idx2 >= 0) & (idx2 < n)
        if np.any(mask2):
            g2 = np.zeros(n)
            g2[np.where(mask2)[0]] = h[idx2[mask2]]
            grad -= wm * g2
    return dx * grad


def local_search(h, target, n_trials, rng, step_size=0.1):

    c5_cur, _ = evaluate_c5(h)
    for _ in range(n_trials):
        j = rng.randint(n)
        l = rng.randint(n)
        if j == l:
            continue
        delta = step_size * (0.3 + 0.7 * rng.rand())
        if rng.rand() < 0.5:
            # move from j to l
            d = min(delta, h[j], 1.0 - h[l])
        else:
            # move from l to j
            d = min(delta, h[l], 1.0 - h[j])
            j, l = l, j
        if d < 1e-10:
            continue
        h[j] -= d
        h[l] += d
        c5_new, _ = evaluate_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            h[j] += d
            h[l] -= d
    return h, c5_cur


def phase_subgradient(h, target, rng, time_limit, start_time):

    c5_best, _ = evaluate_c5(h)
    lr = 0.08
    temp = 0.04

    it = 0

    while time.time() - start_time < time_limit:

        grad = compute_subgrad(h, corr, temp)
        grad_norm = np.linalg.norm(grad)

        c5_new, _ = evaluate_c5(h_new)

            c5_best = c5_new
            no_improve = 0

            h_ls, c5_ls = local_search(h_ls, target, 200, rng)

                c5_best = c5_ls

    h[::2] = 0.75

    return h


def make_two_bumps(n):
    h = np.zeros(n)
    m = max(1, n // 4)
    h[:m] = 1.0
    h[-m:] = 1.0
    return h
