    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)

        return m

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)

            h3_new = project_feasible(res.x, target_sum3)
            c5_new = compute_c5(h3_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h3_new.copy()
                h3 = h3_new.copy()
        except Exception:
            pass

    # Final: ensure feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
