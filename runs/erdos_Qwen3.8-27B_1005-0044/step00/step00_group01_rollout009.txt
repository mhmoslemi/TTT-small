def compute_smooth_gradient(h, dx, temperature=0.05):
    n = len(h)
    overlaps = compute_overlaps(h, dx)
    # Softmax weights
    scaled = overlaps / temperature
    scaled -= np.max(scaled)
    weights = np.exp(scaled)
    weights /= np.sum(weights)
    
    # Compute gradient as weighted sum of individual gradients
    grad = np.zeros(n)
    for k in range(len(overlaps)):
        if weights[k] < 1e-10:
            continue
        lag = k - (n - 1)
        # grad_k[j] = dx * [(1-h[j-lag])*I(0<=j-lag<n) - h[j+lag]*I(0<=j+lag<n)]
        # Vectorized:
        j = np.arange(n)
        # First term: valid when 0 <= j-lag < n
        mask1 = (j - lag >= 0) & (j - lag < n)
        # Second term: valid when 0 <= j+lag < n
        mask2 = (j + lag >= 0) & (j + lag < n)
        
        g = np.zeros(n)
        g[mask1] = (1.0 - h[j[mask1] - lag])
        g[mask2] -= h[j[mask2] + lag]
        g *= dx
