import numpy as np
from scipy.optimize import minimize
import time

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    ov = np.zeros(n - 1)
    for k in range(1, n):
        ov[k - 1] = np.sum(h[:n - k] * (1.0 - h[k:])) * dx
    return ov

def c5_value(h):
    return float(np.max(compute_overlaps(h)))

def softmax_val_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    ov = compute_overlaps(h)
    om = np.max(ov)
    shifted = T * (ov - om)
    max_shift = np.max(shifted)
    exps = np.exp(shifted - max_shift)
    total = np.sum(exps)
    weights = exps / total
    val = om + (max_shift + np.log(total)) / T
    grad = np.zeros(n)
    for idx in range(n - 1):
        w = weights[idx]
        if w < 1e-14:
            continue
        k = idx + 1
        grad[:n - k] += w * dx * (1.0 - h[k:])
        grad[k:] -= w * dx * h[:n - k]
    return val, grad

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    mid = 0.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        hp = np.clip(h + mid, 0.0, 1.0)
        if np.sum(hp) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + mid, 0.0, 1.0)

def try_optimize(h0, T, n, target_sum, time_limit):
    def obj(h):
        return softmax_val_grad(h, T)[0]
    def obj_grad(h):
        return softmax_val_grad(h, T)
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum,
             'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    t_start = time.time()
    try:
        res = minimize(obj, h0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 400, 'ftol': 1e-14,
                                'catol': 1e-10, 'iprint': 0})
    except Exception:
        return None
    if time.time() - t_start > time_limit:
        return None
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_sum(h_new, target_sum)
    return h_new

def coordinate_polish(h, target_sum, max_time):
    n = len(h)
    h = h.copy()
    current = c5_value(h)
    t_end = time.time() + max_time
    step_sizes = [0.1, 0.05, 0.02, 0.01, 0.005]
    improved = True
    while improved and time.time() < t_end:
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            for d in step_sizes:
                for sign in [1, -1]:
                    h_try = h.copy()
                    val = h[i] + sign * d
                    if val < 0 or val > 1:
                        continue
                    h_try[i] = val
                    h_try = project_sum(h_try, target_sum)
                    c = c5_value(h_try)
                    if c < current - 1e-12:
                        h = h_try
                        current = c
                        improved = True
                        break
    return h, current
