            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            if c5_new < c5_cur:
                h0 = h_new
                c5_cur = c5_new
        except Exception:
            pass

        # Pairwise coordinate descent
        if remaining() < 15.0:
            continue
        h_cd = pairwise_cd(h0, n, rng, 5000, remaining)
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        if c5_cd < c5_cur:
            h0 = h_cd
            c5_cur = c5_cd

        # Try random restarts for this n
        n_restarts = 3
        for r in range(n_restarts):
            if remaining() < 40.0:
                break
            h_rand = rng.dirichlet(np.ones(n) * 0.5)
            h_rand = h_rand * float(n)  # scale so sum = n, then clip
            h_rand = np.clip(h_rand, 0.0, 1.0)
            h_rand = project_feasible(h_rand, n)
            if compute_c5(h_rand) > c5_cur + 0.05:
                continue
            try:
                cons3 = [{'type': 'eq', 'fun': sp_sum_eq, 'jac': sp_sum_jac, 'args': (n,)}]
                mret = minimize(
                    compute_c5, h_rand, method='SLSQP',
                    bounds=[(0.0, 1.0)] * n, constraints=cons3,
                    options={'maxiter': 400, 'ftol': 1e-12})
                h_r = project_feasible(mret.x, n)
                c5_r = compute_c5(h_r)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r.copy()
                if c5_r < c5_cur:
                    h0 = h_r
                    c5_cur = c5_r
            except Exception:
                pass

    # Final polish on best solution
    if best_h is not None and remaining() > 10.0:
        bn = len(best_h)
        h_final = pairwise_cd(best_h, bn, rng, 3000, remaining)
        c5_f = compute_c5(h_final)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_final

    if best_h is None:
        n = 47
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
