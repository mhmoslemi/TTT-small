        t = m - (n - 1)
        if t >= 0:
            g[t:] += wm * (1.0 - h[:n-t])
            if n > t:
                g[:n-t] -= wm * h[t:]
        else:
            s = -t
            g[:n-s] += wm * (1.0 - h[s:])
            g[s:] -= wm * h[:n-s]
    return dx * g

def grad_exact(h, dx, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = int(np.argmax(corr))
    t = m - (n - 1)
    g = np.zeros(n)
    if t >= 0:
        g[t:] += (1.0 - h[:n-t])
        if n > t:
            g[:n-t] -= h[t:]
    else:
        s = -t
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    return dx * g
