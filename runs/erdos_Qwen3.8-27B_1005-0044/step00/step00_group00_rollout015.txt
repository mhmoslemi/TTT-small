import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_c5_and_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    umh = 1.0 - h
    corr_raw = np.correlate(h, umh, mode='full')
    g = corr_raw * dx
    g_max = np.max(g)
    exp_v = np.exp(t * (g - g_max))
    w = exp_v / exp_v.sum()
    obj = g_max + np.log(exp_v.sum()) / t
    offsets = np.arange(len(g)) - (n - 1)
    j = np.arange(n)
    grad = np.zeros(n)
    for kk in range(len(g)):
        if w[kk] < 1e-12:
            continue
        o = offsets[kk]
        idx1 = j + o
        m1 = (idx1 >= 0) & (idx1 < n)
        grad[m1] += w[kk] * (1.0 - h[idx1[m1]])
        idx2 = j - o
        m2 = (idx2 >= 0) & (idx2 < n)
        grad[m2] -= w[kk] * h[idx2[m2]]
    return obj, dx * grad

def project_sum(h, target):
    h = h.copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            return h
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            return h
        h = h + (diff / cnt) * mask
    return h

def pobj(h, target, pen, t):
    o, _ = soft_c5_and_grad(h, t)
    return o + pen * (h.sum() - target) ** 2

def pgrad(h, target, pen, t):
    _, g = soft_c5_and_grad(h, t)
    return g + 2.0 * pen * (h.sum() - target)

def exact_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(corr))
    offset = k_max - (n - 1)
    j = np.arange(n)
    grad = np.zeros(n)
    idx1 = j + offset
    m1 = (idx1 >= 0) & (idx1 < n)
    grad[m1] += (1.0 - h[idx1[m1]])
    idx2 = j - offset
    m2 = (idx2 >= 0) & (idx2 < n)
    grad[m2] -= h[idx2[m2]]
    return dx * grad

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(45, 0.5)

    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    best_h = project_sum(h0, target)
    best_c5 = compute_c5(best_h)

    # Phase 1: Anneal soft-max temperature from low to high
    for t_val in [30, 50, 100, 200, 500, 1000, 2000, 5000]:
        if time.time() - t0 > budget_s - 80:

        pen = 1e5
        of = partial(pobj, target=target, pen=pen, t=t_val)
        gf = partial(pgrad, target=target, pen=pen, t=t_val)
        try:
            res = minimize(of, best_h, method='L-BFGS-B', jac=gf,
                          bounds=bounds,
                          options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-14})
            h_new = project_sum(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:

        except Exception:
            pass
