    c = compute_all_c(h)
    # Soft max approximation
    c_max = np.max(c)
    return c_max + np.log(np.sum(np.exp(beta * (c - c_max)))) / beta

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode="full") * dx))

def project_feasible(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-9
        else:
            mask = h > 1e-9
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(x, n, beta):
    x = project_feasible(x, n)
    c = np.correlate(x, 1-x, mode="full") * (2.0 / n)
    cmax = c.max()
    shifted = (c - cmax) * beta
    return cmax + np.log(np.sum(np.exp(shifted)) / len(c)) / beta

def get_upsampled(h_init, n_init, n):
    if n == n_init:
        return h_init.copy()

        if c5 < best_c5:

        h_new[j] = 1.0
        c5_new = compute_c5(h_new)

    best_c5 = compute_c5(h)
    h = h.copy()

        elif c5_new - c5 < 0.003 and rng.random() < 0.1:

        best_c5 = c5
        best_h = h.copy()
    
    # Phase 3: Multiple restarts (30% budget)
    p3_deadline = t0 + budget_s * 0.90
    n_options = [n_init, 48, 50, 64, 80, 100, 128]
    n_options = [x for x in n_options if x >= 2]
    
    for n in n_options:
        if time.time() > p3_deadline:
            break
        # Upsampled start
        h = upsample(h_init, n_init, n)
        h = project_feasible(h, n)
        dl = min(p3_deadline, time.time() + 25)
        h, c5 = random_local_search(h, n, rng, dl)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Binary start (even n)
        if n % 2 == 0 and time.time() < p3_deadline - 15:
            for _ in range(3):
                if time.time() > p3_deadline:
                    break
                pos = rng.choice(n, n // 2, replace=False)
                hb = np.zeros(n)
                hb[pos] = 1.0
                dl = min(p3_deadline, time.time() + 15)
                hb, c5b = binary_local_search(hb, n, rng, dl)
                if c5b < best_c5:
                    best_c5 = c5b
                    best_h = hb.copy()
        
        # Random continuous start
        if time.time() < p3_deadline - 15:
            hr = rng.uniform(0, 1, n)
            hr = project_feasible(hr, n)
            dl = min(p3_deadline, time.time() + 15)
            hr, c5r = random_local_search(hr, n, rng, dl)
            if c5r < best_c5:
                best_c5 = c5r
                best_h = hr.copy()
