    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def soft_max(x, T):
    m = np.max(x)
    if T * (np.max(x) - m) < -700:
        return m
    return m + np.log(np.sum(np.exp(T * (x - m)))) / T

def objective_fn(h, n, target_sum, T, pen_weight):
    terms = np.correlate(np.clip(h, 0, 1), 1 - np.clip(h, 0, 1), mode="full") * (2.0/n)
    sm = soft_max(terms, T)
    pen = pen_weight * (np.sum(h) - target_sum)**2
    return sm + pen

def project_to_sum(h, target_sum):
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if s == 0:
        return h
    # Water-filling projection
    h_proj = h * (target_sum / s)
    h_proj = np.clip(h_proj, 0, 1)
    # Iterate to get exact sum
    for _ in range(100):
        diff = target_sum - np.sum(h_proj)
        if abs(diff) < 1e-10:
            break
        # Adjust free variables
        free = (h_proj > 0) & (h_proj < 1)
        if not np.any(free):
            break
        h_proj[free] += diff / np.sum(free)
        h_proj = np.clip(h_proj, 0, 1)
    return h_proj

def project_to_sum(h, target_sum):
    h = np.clip(h, 0, 1).astype(float).copy()
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(free):
            adj = diff / np.sum(free)
            h[free] = np.clip(h[free] + adj, 0, 1)
        else:
            if diff > 0:
                idx = int(np.argmax(h))
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = int(np.argmin(h))
                h[idx] = max(0.0, h[idx] + diff)
    return h

def to_binary(h, target_count):
    n = len(h)
    indices = np.argsort(h)
    h_bin = np.zeros(n)
    h_bin[indices[n - target_count:]] = 1.0
    return h_bin

def obj_softmax(h, n, target_sum, T, pen_w):
    hc = np.clip(h, 0, 1)
    terms = np.correlate(hc, 1 - hc, mode="full") * (2.0 / n)
    return soft_max(terms, T) + pen_w * (np.sum(h) - target_sum) ** 2

def binary_local_search(h_bin, n, max_iters=5000, t_start=None, deadline=None):
    h = h_bin.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode="full")
        k_star = int(np.argmax(corr))
        shift = k_star - (n - 1)
        improved = False
        if shift >= 0:
            limit = min(n - shift, 200)
            for i in range(limit):
                if h[i] == 1.0 and h[i + shift] == 0.0:
                    h[i] = 0.0
                    h[i + shift] = 1.0
                    c5_new = compute_c5(h)
                    if c5_new < best_c5 - 1e-14:
                        best_c5 = c5_new
                        best_h = h.copy()
                        improved = True
                        break
                    h[i] = 1.0
                    h[i + shift] = 0.0
        else:
            k = -shift
            limit = min(k + 200, n)
            for i in range(k, limit):
                if h[i] == 1.0 and h[i - k] == 0.0:
                    h[i] = 0.0
                    h[i - k] = 1.0
                    c5_new = compute_c5(h)
                    if c5_new < best_c5 - 1e-14:
                        best_c5 = c5_new
                        best_h = h.copy()
                        improved = True
                        break
                    h[i] = 1.0
                    h[i - k] = 0.0
        if not improved:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            found = False
            for _ in range(10):
                i = int(ones[np.random.randint(len(ones))])
                j = int(zeros[np.random.randint(len(zeros))])
                h[i] = 0.0
                h[j] = 1.0
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h.copy()
                    found = True
                    break
                h[i] = 1.0
                h[j] = 0.0
            if not found:
                break
    return best_h, best_c5

def binary_sa_search(h_bin, n, max_iters=3000, deadline=None):
    h = h_bin.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_c5 = best_c5
    T_init = 0.005
    T_fin = 0.0005
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        progress = it / max(1, max_iters)
        T = T_init * (T_fin / T_init) ** progress
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = int(ones[np.random.randint(len(ones))])
        j = int(zeros[np.random.randint(len(zeros))])
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        accept = c5_new < curr_c5 or (np.random.random() < np.exp(-(c5_new - curr_c5) / T))
        if accept:
            curr_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    np.random.seed(seed)
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = np.ones(45) * 0.5

    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    n = len(h0)
    target_sum = n / 2.0

    # Phase 1: Continuous optimization on original n
    bounds = [(0.0, 1.0)] * n
    for T in [30, 80, 200]:
        deadline = t_start + 0.15 * budget_s
        for attempt in range(5):
            if time.time() > deadline:
                break
            if attempt == 0:
                x0 = best_h.copy()
            else:
                x0 = np.clip(best_h + rng.normal(0, 0.05, n), 0, 1)
            res = minimize(obj_softmax, x0, args=(n, target_sum, T, 200.0),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-12})
            h_new = project_to_sum(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 2: Binary local search + SA on original n
    dl2 = t_start + 0.40 * budget_s
    h_bin = to_binary(best_h, int(target_sum))
    h_sol, c5_sol = binary_local_search(h_bin, n, max_iters=10000, deadline=dl2)
    if c5_sol < best_c5:
        best_h = h_sol
        best_c5 = c5_sol

    dl2b = t_start + 0.50 * budget_s
    h_sa, c5_sa = binary_sa_search(best_h, n, max_iters=3000, deadline=dl2b)
    if c5_sa < best_c5:
        best_h = h_sa
        best_c5 = c5_sa

    # Phase 3: Higher resolution
    for n_large in [90, 150, 225]:
        if time.time() > t_start + 0.90 * budget_s:
            break
        x_old = np.arange(n) * (2.0 / n)
        x_new = np.arange(n_large) * (2.0 / n_large)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_to_sum(h_interp, n_large / 2.0)
        bounds_l = [(0.0, 1.0)] * n_large
        res = minimize(obj_softmax, h_interp, args=(n_large, n_large / 2.0, 60, 100.0),
                      method='L-BFGS-B', bounds=bounds_l,
                      options={'maxiter': 150, 'ftol': 1e-12, 'gtol': 1e-8})
        h_l = project_to_sum(res.x, n_large / 2.0)
        c5_l = compute_c5(h_l)
        if c5_l < best_c5:
            best_h = h_l
            best_c5 = c5_l

        dl3 = t_start + 0.90 * budget_s
        h_bin_l = to_binary(h_l, n_large // 2)
        max_iter_l = max(500, 10000 * 45 // n_large)
        h_sol_l, c5_sol_l = binary_local_search(h_bin_l, n_large, max_iters=max_iter_l, deadline=dl3)
        if c5_sol_l < best_c5:
            best_h = h_sol_l
            best_c5 = c5_sol_l

    # Final polish: one more SA round on best solution
    if time.time() < t_start + 0.97 * budget_s:
        dl_final = t_start + 0.97 * budget_s
        n_final = len(best_h)
        h_final_bin = to_binary(best_h, n_final // 2)
        h_fin, c5_fin = binary_sa_search(h_final_bin, n_final, max_iters=2000, deadline=dl_final)
        if c5_fin < best_c5:
            best_h = h_fin
            best_c5 = c5_fin

    c5_check = compute_c5(best_h)
    best_c5 = min(best_c5, c5_check)
    return best_h, best_c5, len(best_h)
