import numpy as np

def project_to_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def gradient_for_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    else:
        t = -s
        g[:n - t] += (1.0 - h[t:])
        g[t:] -= h[:n - t]
    return g

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(corr)
    exp_t = np.exp(T * (corr - max_c))
    w = exp_t / exp_t.sum()
    f = (max_c + np.log(exp_t.sum())) / T
    grad = np.zeros(n)
    ns = len(w)
    for k in range(ns):
        if w[k] < 1e-8:
            continue
        shift = k - (n - 1)
        grad += w[k] * gradient_for_shift(h, shift)
    grad *= dx
    return f, grad
