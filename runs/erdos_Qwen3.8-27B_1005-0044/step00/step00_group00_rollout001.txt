from scipy.optimize import minimize, LinearConstraint
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def obj_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    k_max = np.argmax(corr)
    lag = k_max - (n - 1)
    
    val = corr[k_max] * dx
    grad = np.zeros(n)
    
    for i in range(n):
        if 0 <= i + lag < n:
            grad[i] -= h[i + lag]
        if 0 <= i - lag < n:
            grad[i] += (1 - h[i - lag])
    
    return val, grad * dx

def make_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0, 1)
    s = h.sum()
    if s > 0:
        h = h * (target / s)
    h = np.clip(h, 0, 1)
    # Fine-tune
    diff = target - h.sum()
    if diff > 0:
        # Add to elements that are < 1
        mask = h < 1
        if mask.sum() > 0:
            add = min(diff / mask.sum(), 1 - h[mask]).min() if (1-h[mask]).min() < diff/mask.sum() else diff/mask.sum()
            h[mask] += add
    elif diff < 0:
        mask = h > 0
        if mask.sum() > 0:
            sub = min(-diff / mask.sum(), h[mask].min()) if h[mask].min() < -diff/mask.sum() else -diff/mask.sum()
            h[mask] -= sub
    return h

import numpy as np
from scipy.optimize import minimize, LinearConstraint
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx


def obj_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    k_max = np.argmax(corr)
    val = corr[k_max] * dx
    grad = np.zeros(n)
    for i in range(n):
        if 0 <= i - k_max < n:
            grad[i] += (1.0 - h[i - k_max])
        if 0 <= i + k_max < n:
            grad[i] -= h[i + k_max]
    return val, grad * dx


def make_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-12:
        h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() > 0:
            available = (1.0 - h[mask]).sum() if diff > 0 else h[mask].sum()
            frac = abs(diff) / available
            frac = min(frac, 1.0)
            h[mask] += np.sign(diff) * abs(diff) / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def upsample(h):
    n = len(h)
    h_new = np.zeros(2 * n - 1)
    h_new[0::2] = h
    h_new[1::2] = (h[:-1] + h[1:]) / 2.0
    return h_new


def slsqp_obj(h):
    return obj_and_grad(h)[0]


def slsqp_grad(h):
    return obj_and_grad(h)[1]

def sum_constraint(h, target, n):
    return h.sum() - target


def sum_jac(h, target, n):
    return np.ones(n)
