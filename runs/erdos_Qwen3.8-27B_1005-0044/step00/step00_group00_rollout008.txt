    corr = np.correlate(h, 1-h, mode="full") * dx
    # Smoothed max
    max_val = np.max(corr)
    # Shift for numerical stability
    exp_vals = np.exp(tau * (corr - max_val))
    smooth_max = max_val + np.log(np.sum(exp_vals)) / tau
    return smooth_max

def compute_c5_smooth_grad(h, dx, tau=100):

The minimum of max_k (B_k + C_k * t) on t in [0,1] can be found by:
1. Evaluating at t=0 and t=1
2. Finding the intersection point where the "upper envelope" has a kink

The upper envelope of lines B_k + C_k*t changes at points where two lines cross: B_k + C_k*t = B_m + C_m*t, i.e., t = (B_m - B_k)/(C_k - C_m).

So the algorithm is:
1. Compute B_k, C_k for all k
2. Evaluate f(0) = max_k B_k and f(1) = max_k (B_k + C_k)
3. Find all intersection points t* in (0,1)
4. Evaluate f at each t*
5. Return the t that gives the minimum f

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def smooth_obj_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    mc = np.max(corr)
    e = np.exp(tau * (corr - mc))
    se = np.sum(e)
    val = mc + np.log(se) / tau
    w = e / se
    grad = np.zeros(n)
    active = np.where(w > 1e-7)[0]
    for idx in active:
        k = idx - (n - 1)
        j = np.arange(n)
        jm = j - k
        jp = j + k
        m1 = (jm >= 0) & (jm < n)
        m2 = (jp >= 0) & (jp < n)
        grad += w[idx] * dx * (np.where(m1, 1 - h[jm], 0.0) - np.where(m2, h[jp], 0.0))
    return val, grad

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    for _ in range(200):

        if abs(d) < 1e-12:

        m = (h > 1e-10) & (h < 1 - 1e-10)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        np.clip(h, 0, 1, out=h)
    return h
