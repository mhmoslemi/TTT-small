    n = len(x)
    # Compute autocorrelation via FFT
    f = np.fft.fft(x, 2*n)
    autocorr = np.real(np.fft.ifft(f * np.conj(f)))[:n]

    best_x = x.copy()
    best_cost = compute_c5_binary_fast(x)
    x = x.copy()

    temp = 0.01
    
    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Pick a random 1 and a random 0
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        
        # Make the move
        x[i] = 0
        x[j] = 1
        
        new_cost = compute_c5_binary_fast(x)
        
        delta = new_cost - current_cost
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            current_cost = new_cost
            # Update ones and zeros
            ones = ones[ones != i]
            zeros = zeros[zeros != j]
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
            if new_cost < best_cost:
                best_cost = new_cost
                best_x = x.copy()
        else:
            # Undo
            x[i] = 1
            x[j] = 0
        
        if it % 10000 == 0 and it > 0:
            temp *= 0.99
    
    return best_x, best_cost

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_fast_from_autocorr(x, n, half_n):
    """Compute C5 for binary x using precomputed autocorrelation and prefix sums."""
    # autocorr[k] = sum_{i=0}^{n-1-k} x[i]*x[i+k] for k=0..n-1
    f = np.fft.fft(x, 2*n)

        count_neg = prefix[n-k] - autocorr[k]
        max_corr = max(max_corr, count_pos, count_neg)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

def compute_c5_binary(x):
    """Fast C5 for binary sequence using FFT."""
    n = len(x)
    half_n = n // 2
    f = np.fft.fft(x.astype(float), 2 * n)
    autocorr = np.real(np.fft.ifft(f * np.conj(f)))[:n]
    prefix = np.zeros(n + 1)
    prefix[1:] = np.cumsum(x)
    max_corr = 0.0
    for k in range(1, n):
        cp = (half_n - prefix[k]) - autocorr[k]
        cn = prefix[n - k] - autocorr[k]
        if cp > max_corr:
            max_corr = cp
        if cn > max_corr:
            max_corr = cn
    return max_corr * 2.0 / n

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_binary(x):
    n = len(x)
    half_n = n // 2
    xf = x.astype(float)
    f = np.fft.fft(xf, 2 * n)
    autocorr = np.real(np.fft.ifft(f * np.conj(f)))[:n]
    prefix = np.zeros(n + 1)
    prefix[1:] = np.cumsum(xf)
    max_corr = 0.0
    for k in range(1, n):
        cp = (half_n - prefix[k]) - autocorr[k]
        cn = prefix[n - k] - autocorr[k]
        if cp > max_corr:
            max_corr = cp
        if cn > max_corr:
            max_corr = cn
    return max_corr * 2.0 / n

def binary_sa(x0, n, max_iter, deadline, seed, temp_init=0.003, cool=0.98):
    rng = np.random.RandomState(seed)
    x = x0.copy().astype(float)
    current_cost = compute_c5_binary(x)
    best_x = x.copy()
    best_cost = current_cost
    temp = temp_init
    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    for it in range(max_iter):
        if time.time() > deadline:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        x[i] = 0.0
        x[j] = 1.0
        new_cost = compute_c5_binary(x)
        delta = new_cost - current_cost
        if delta < 0 or (temp > 1e-12 and rng.random() < np.exp(-delta / temp)):
            current_cost = new_cost
            oi = np.where(ones == i)[0][0]
            oj = np.where(zeros == j)[0][0]
            ones[oi] = j
            zeros[oj] = i
            if new_cost < best_cost:
                best_cost = new_cost
                best_x = x.copy()
        else:
            x[i] = 1.0
            x[j] = 0.0
        if it % 10000 == 0 and it > 0:
            temp *= cool
    return best_x, best_cost

def steepest_descent_binary(x0, deadline, max_rounds=200):
    n = len(x0)
    x = x0.copy().astype(float)
    current_cost = compute_c5_binary(x)
    best_x = x.copy()
    best_cost = current_cost
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        ones = np.where(x == 1)[0]
        zeros = np.where(x == 0)[0]
        best_delta = 0.0
        best_swap = None
        for i in ones:
            for j in zeros:
                x[i] = 0.0
                x[j] = 1.0
                c = compute_c5_binary(x)
                x[i] = 1.0
                x[j] = 0.0
                delta = c - current_cost
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)
        if best_swap is None:
            break
        i, j = best_swap
        x[i] = 0.0
        x[j] = 1.0
        current_cost += best_delta
        if current_cost < best_cost:
            best_cost = current_cost
            best_x = x.copy()
    return best_x, best_cost

def project_capped_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def continuous_refine(h0, deadline):
    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    alphas = [50.0, 200.0, 500.0, 2000.0]
    for alpha in alphas:
        h = best_h.copy()
        step = 0.01
        for it in range(5000):
            if time.time() > deadline:
                return best_h, best_c5
            corr = np.correlate(h, 1.0 - h, mode='full')
            m = np.max(corr)
            exps = np.exp(alpha * (corr - m))
            w = exps / exps.sum()
            grad = np.zeros(n)
            for k_idx in range(2 * n - 1):
                wk = w[k_idx]
                if wk < 1e-12:
                    continue
                k = k_idx - (n - 1)
                if k >= 0:
                    nc = n - k
                    grad[k:] += wk * (1.0 - h[:nc])
                    grad[:nc] -= wk * h[k:]
                else:
                    kp = -k
                    nc = n - kp
                    grad[:nc] += wk * (1.0 - h[kp:])
                    grad[kp:] -= wk * h[:nc]
            grad *= (2.0 / n)
            h_new = project_capped_simplex(h - step * grad, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            c5_cur = compute_c5(h)
            if c5_new < c5_cur:
                h = h_new
                step = min(step * 1.03, 0.1)
            else:
                step *= 0.5
                if step < 1e-12:
                    break
    return best_h, best_c5

def make_starts(n, seed, h_init, n_init):
    starts = []
    rng = np.random.RandomState(seed)
    x = np.zeros(n)
    x[rng.choice(n, n // 2, replace=False)] = 1.0
    starts.append(x)
    h_r = np.interp(np.linspace(0, 2, n, endpoint=False),
                    np.linspace(0, 2, n_init, endpoint=False), h_init)
    idx = np.argsort(h_r)[::-1]
    x2 = np.zeros(n)
    x2[idx[:n // 2]] = 1.0
    starts.append(x2)
    x3 = np.zeros(n)
    x3[rng.choice(n, n // 2, replace=False)] = 1.0
    starts.append(x3)
    x_alt = np.zeros(n)
    x_alt[::2] = 1.0
    starts.append(x_alt)
    return starts

    deadline = t0 + budget_s - 5

        results[n] = (n_best_c5, n_best_x)
        if n_best_c5 < best_c5:
