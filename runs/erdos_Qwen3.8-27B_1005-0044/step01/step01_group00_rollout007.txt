import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_capped_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_val(corr, alpha):
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

def smooth_max_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_val(corr, alpha) * dx

def smooth_max_grad(h, alpha):
    # Numerical gradient for simplicity and robustness
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k_idx in range(n_corr):
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            nc = n - k
            grad[k:] += wk * (1.0 - h[:nc])
            grad[:nc] -= wk * h[k:]
        else:
            kp = -k
            nc = n - kp
            grad[:nc] += wk * (1.0 - h[kp:])
            grad[kp:] -= wk * h[:nc]
    return grad * dx

def pgd_refine(h, target, alphas, max_iter, deadline):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    for alpha in alphas:
        h_cur = best_h.copy()
        step = 0.05
        for it in range(max_iter):

                break
            g = smooth_max_grad(h_cur, alpha)
            h_new = project_capped_simplex(h_cur - step * g, target)
            val_new = smooth_max_obj(h_new, alpha)
            val_old = smooth_max_obj(h_cur, alpha)
            if val_new < val_old:
                h_cur = h_new
                step = min(step * 1.1, 0.5)
            else:
                step *= 0.5
                if step < 1e-12:
                    break
            c5 = compute_c5(h_cur)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cur.copy()
    return best_h, best_c5

def sla_search(h, target, n_iter, deadline, seed, temp0=0.03):
    rng = np.random.RandomState(seed)
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    temp = temp0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Random perturbation of a random subset
        p = max(1, n // 10)
        idx = rng.choice(n, size=p, replace=False)
        h_pert = h_cur.copy()
        h_pert[idx] += rng.randn(p) * temp
        h_pert = project_capped_simplex(h_pert, target)
        c5_pert = compute_c5(h_pert)
        if c5_pert <= best_c5:
            h_cur = h_pert
            best_c5 = c5_pert
            best_h = h_cur.copy()
        else:

                h_cur = h_pert
        if it % 200 == 0 and it > 0:
            temp *= 0.9
    return best_h, best_c5

def slsqp_optimize(h, target, deadline, alpha=50.0):
    n = len(h)
    # Use scipy SLSQP
    def obj(x):
        return smooth_max_obj(x, alpha)
    
    def obj_grad(x):
        return smooth_max_grad(x, alpha)
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    bounds = [(0, 1)] * n
    
    x0 = project_capped_simplex(h.copy(), target)
    res = minimize(obj, x0, jac=obj_grad, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': 500, 'ftol': 1e-10})
    if res.success or res.fun < smooth_max_obj(x0, alpha):
        h_res = project_capped_simplex(res.x, target)
        return h_res
    return h

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_capped_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    total = exps.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exps / total
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            nc = n - k
            grad[k:] += wk * (1.0 - h[:nc])
            grad[:nc] -= wk * h[k:]
        else:
            kp = -k
            nc = n - kp
            grad[:nc] += wk * (1.0 - h[kp:])
            grad[kp:] -= wk * h[:nc]
    return grad * dx

def pgd_refine(h, target, alphas, iters_per_alpha, deadline):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for alpha in alphas:
        h_cur = best_h.copy()
        step = 0.05
        stall = 0
        for it in range(iters_per_alpha):
            if time.time() > deadline:
                break
            g = smooth_max_grad(h_cur, alpha)
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-15:
                break
            h_new = project_capped_simplex(h_cur - step * g, target)
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h_cur)
            if c5_new < c5_old:
                h_cur = h_new
                step = min(step * 1.08, 0.6)
                stall = 0
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_cur.copy()
            else:
                step *= 0.6
                stall += 1
                if stall > 30 or step < 1e-10:
                    break
    return best_h, best_c5

def sla_paired(h, target, n_iter, deadline, seed, temp0=0.02):
    rng = np.random.RandomState(seed)
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    cur_c5 = best_c5
    temp = temp0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi, hj = h_cur[i], h_cur[j]
        dmin = max(-hi, hj - 1.0)
        dmax = min(1.0 - hi, hj)
        if dmin >= dmax - 1e-12:
            continue
        # Pick a delta
        d = (dmin + dmax) / 2.0 + rng.randn() * (dmax - dmin) * 0.3
        d = max(dmin + 1e-9, min(dmax - 1e-9, d))
        h_try = h_cur.copy()
        h_try[i] = hi + d
        h_try[j] = hj - d
        c5_try = compute_c5(h_try)
        if c5_try <= cur_c5:
            h_cur = h_try
            cur_c5 = c5_try
            if c5_try < best_c5:

                best_h = h_cur.copy()
        else:
            if temp > 1e-7 and rng.random() < np.exp(-(c5_try - cur_c5) / temp):
                h_cur = h_try
                cur_c5 = c5_try
        if it % 300 == 0 and it > 0:
            temp *= 0.85
    return best_h, best_c5

def greedy_paired_cd(h, target, n_sweeps, deadline):
    """Do full sweeps of paired coordinate descent, always accepting improvements."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i+1, n):
                hi, hj = h_cur[i], h_cur[j]
                dmin = max(-hi, hj - 1.0)
                dmax = min(1.0 - hi, hj)
                if dmin >= dmax - 1e-9:
                    continue
                # Try a few deltas
                best_d = 0.0
                best_val = compute_c5(h_cur)
                for frac in [0.1, 0.3, 0.5, 0.7, 0.9, -0.1, -0.3, -0.5, -0.7, -0.9]:
                    d = dmin + (dmax - dmin) * (frac + 1.0) / 2.0
                    if d <= dmin or d >= dmax:
                        continue
                    h_try = h_cur.copy()
                    h_try[i] = hi + d
                    h_try[j] = hj - d
                    c5_try = compute_c5(h_try)
                    if c5_try < best_val:
                        best_val = c5_try
                        best_d = d
                if best_d != 0.0:

                    h_cur[j] = hj - best_d
                    improved = True
                    if best_val < best_c5:
                        best_c5 = best_val
                        best_h = h_cur.copy()
        if not improved:
            break
    return best_h, best_c5

        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi, hj = h_cur[i], h_cur[j]
        dmin = max(-hi, hj - 1.0)
        dmax = min(1.0 - hi, hj)
        if dmin >= dmax - 1e-12:
            continue
        d = (dmin + dmax) / 2.0 + rng.randn() * (dmax - dmin) * 0.35
        d = max(dmin + 1e-9, min(dmax - 1e-9, d))
        h_try = h_cur.copy()
        h_try[i] = hi + d
        h_try[j] = hj - d
        c5_try = compute_c5(h_try)
        if c5_try <= cur_c5:
            h_cur = h_try
            cur_c5 = c5_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_cur.copy()
        else:
            if temp > 1e-7 and rng.random() < np.exp(-(c5_try - cur_c5) / temp):
                h_cur = h_try
                cur_c5 = c5_try
        if it % 250 == 0 and it > 0:
            temp *= 0.84
    return best_h, best_c5

def greedy_paired_cd(h, target, n_sweeps, deadline):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c5
            for j in range(i + 1, n):
                hi, hj = h_cur[i], h_cur[j]
                dmin = max(-hi, hj - 1.0)
                dmax = min(1.0 - hi, hj)
                if dmin >= dmax - 1e-9:
                    continue
                best_d = 0.0
                best_val = compute_c5(h_cur)
                for t in range(1, 9):
                    d = dmin + (dmax - dmin) * t / 9.0
                    h_try = h_cur.copy()
                    h_try[i] = hi + d
                    h_try[j] = hj - d
                    cv = compute_c5(h_try)
                    if cv < best_val:
                        best_val = cv
                        best_d = d
                if best_d != 0.0:
                    h_cur[i] = hi + best_d
                    h_cur[j] = hj - best_d
                    improved = True
                    if best_val < best_c5:
                        best_c5 = best_val
                        best_h = h_cur.copy()
        if not improved:
            break
    return best_h, best_c5

def make_sinusoidal(n, freq, phase, target):
    x = np.linspace(0, 2, n)
    h = 0.5 + 0.35 * np.sin(2.0 * np.pi * freq * x + phase)
    return project_capped_simplex(h, target)

def make_block(n, frac, target):
    h = np.zeros(n)
    n_hi = int(n * frac)
    h[:n_hi] = 1.0
    return project_capped_simplex(h, target)

def make_random_init(n, seed, target):
    rng = np.random.RandomState(seed)
    h = rng.rand(n)
    return project_capped_simplex(h, target)

def optimize_candidate(h, target, deadline, seed, do_greedy=True):
    h_r, c5_r = pgd_refine(h, target, [10, 25, 50, 100], 800, min(deadline, time.time() + 60))
    h_s, c5_s = sla_paired(h_r, target, 8000, min(deadline, time.time() + 80), seed)
    if c5_s < c5_r:
        h_r, c5_r = h_s, c5_s
    if do_greedy and time.time() < deadline - 5:
        h_g, c5_g = greedy_paired_cd(h_r, target, 3, min(deadline, time.time() + 60))
        if c5_g < c5_r:
            h_r, c5_r = h_g, c5_g
    return h_r, c5_r

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3

    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values, dtype=float)
    else:
        h_start = np.full(50, 0.5)

    n = len(h_start)
    target = n / 2.0
    h_start = project_capped_simplex(h_start, target)

    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize from initial
    t1 = min(deadline, t0 + budget_s * 0.25)
    h_r, c5_r = pgd_refine(h_start, target, [5, 10, 20, 50, 100, 200], 1500, t1)
    if c5_r < best_c5:
        best_c5, best_h = c5_r, h_r
    h_s, c5_s = sla_paired(best_h, target, 15000, t1, seed)
    if c5_s < best_c5:
        best_c5, best_h = c5_s, h_s
    if time.time() < t1:
        h_g, c5_g = greedy_paired_cd(best_h, target, 5, t1)
        if c5_g < best_c5:
            best_c5, best_h = c5_g, h_s if False else (h_g, c5_g)

    # Phase 2: Multi-start with diverse n and patterns
    t2 = min(deadline, t0 + budget_s * 0.85)
    n_list = [40, 50, 60, 70, 80, 47, 35, 55, 65, 75]
    idx = 0
    for n_try in n_list:
        if time.time() > t2 - 30:
            break
        target_try = n_try / 2.0
        inits = []
        for freq in [1, 2, 3]:
            for phase in [0.0, 1.57, 3.14]:
                inits.append(make_sinusoidal(n_try, freq, phase, target_try))
        for frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
            inits.append(make_block(n_try, frac, target_try))
        for s in range(4):
            inits.append(make_random_init(n_try, seed + idx * 100 + s, target_try))
        # Also try the initial if n matches
        if n_try == len(h_start):
            inits.append(h_start.copy())

        for h_init in inits:
            if time.time() > t2 - 30:
                break
            h_opt, c5_opt = optimize_candidate(h_init, target_try, t2 - 30, seed + idx, do_greedy=True)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            idx += 1

    # Phase 3: Final polishing
    if time.time() < deadline - 2:
        n_f = len(best_h)
        target_f = n_f / 2.0
        h_f, c5_f = pgd_refine(best_h, target_f, [100, 200, 500], 2000, deadline - 2)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f
        if time.time() < deadline - 1:
            h_f2, c5_f2 = sla_paired(best_h, target_f, 10000, deadline - 1, seed + 777)
            if c5_f2 < best_c5:
                best_c5, best_h = c5_f2, h_f2
        if time.time() < deadline - 0.5:
            h_f3, c5_f3 = greedy_paired_cd(best_h, target_f, 5, deadline - 0.5)
            if c5_f3 < best_c5:
                best_c5, best_h = c5_f3, h_f3

    return best_h, best_c5, len(best_h)
