```python
        # Numerical gradient (SLSQP will do its own anyway)
        return None

        obj, h, method='SLSQP',
        bounds=[(0.0, 1.0)] * n,

        options={'maxiter': maxiter, 'ftol': 1e-14}

    h_new = fix_sum(res.x, n)
```

```python
def linearized_iteration(h, n, max_iter=100):
    """Iteratively linearize and solve LP."""
    h = h.copy()
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        
        # Compute gradient of g_{lag} at h
        grad = np.zeros(n)
        if lag >= 0:
            for i in range(n - lag):
                grad[i] += (1 - h[i+lag])
                grad[i+lag] += (-h[i])
        else:
            lag2 = -lag
            for i in range(lag2, n):
                grad[i] += (1 - h[i-lag2])
                grad[i-lag2] += (-h[i])
        
        # Solve: min grad · h subject to 0<=h<=1, sum(h)=n/2
        # This is an LP. The optimal solution is:
        # Sort grad values. Set h[m]=1 where grad[m] is smallest, h[m]=0 where largest.
        # The "boundary" value is set to make the sum work out.
        
        sorted_idx = np.argsort(grad)
        h_new = np.zeros(n)

        if n % 2 == 0:
            h_new[sorted_idx[:n//2]] = 1.0
            h_new[sorted_idx[n//2:]] = 0.0
        else:
            h_new[sorted_idx[:(n-1)//2]] = 1.0
            h_new[sorted_idx[n//2:]] = 0.0

        h_new = 0.5 * h + 0.5 * h_new

            best_c5 = c5_new
```

Actually, I just realized something. Let me reconsider the LP approach. 

The linearized problem is:
min grad · h
s.t. 0 ≤ h ≤ 1
     sum(h) = n/2

The optimal solution to this LP is indeed to set h[m] = 1 where grad[m] is smallest and h[m] = 0 where grad[m] is largest (with a possible fractional value at the boundary). This is because the objective is linear and the feasible set is a polytope.

So the LP gives us a vertex of the feasible polytope. Moving to this vertex might overshoot. A better approach is:
1. Compute the LP solution h_lp
2. Take a fraction towards it: h_new = (1-α) * h + α * h_lp
3. Choose α by line search (maximize the decrease in the true objective)

This is essentially a "conditional gradient" (a.k.a. Frank-Wolfe) method! The conditional gradient method for minimizing f(h) over a convex set C is:
1. s = argmin_{s ∈ C} ∇f(h) · s  (the "linear minimization oracle")
2. h_new = h + γ * (s - h)  (or with backtracking)

For our problem, the "linear minimization oracle" is the LP I described. And the step size can be chosen by line search.

This is a well-known method that can handle non-smooth objectives (we use a subgradient for the max). Let me implement this properly:

```python
def frank_wolfe(h, n, max_iter=200, tol=1e-10):
    """Conditional gradient (Frank-Wolfe) method with subgradient."""
    h = h.copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        # Compute current C5 and find active lag
        corr = np.correlate(h, 1.0 - h, mode="full") * (2.0/n)
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        
        # Subgradient at h
        grad = np.zeros(n)
        if lag >= 0:
            for i in range(n - lag):
                grad[i] += (1 - h[i+lag])
                grad[i+lag] += (-h[i])
        else:
            lag2 = -lag
            for i in range(lag2, n):
                grad[i] += (1 - h[i-lag2])
                grad[i-lag2] += (-h[i])
        
        # Linear minimization: min grad·s s.t. 0≤s≤1, sum(s)=n/2
        sorted_idx = np.argsort(grad)
        s = np.zeros(n)
        if n % 2 == 0:
            s[sorted_idx[:n//2]] = 1.0
        else:
            s[sorted_idx[:(n-1)//2]] = 1.0
            s[sorted_idx[n//2:]] = 0.0
            s[sorted_idx[(n-1)//2]] = 0.5
        s = fix_sum(s, n)
        
        # Line search for step size
        best_alpha = 0
        best_val = compute_c5(h)
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            h_try = (1 - alpha) * h + alpha * s
            h_try = fix_sum(h_try, n)
            val = compute_c5(h_try)
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha == 0:
            # Use the standard FW step size: 2/(it+2)
            alpha = 2.0 / (it + 2)
        else:
            alpha = best_alpha
        
        h = (1 - alpha) * h + alpha * s
        h = fix_sum(h, n)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        
        # Convergence check
        if it > 10 and abs(compute_c5(best) - best_c5) < tol:
            break
    
    return best, best_c5
```

This is clean and should work well. The key advantage over plain subgradient is that the step is computed exactly (via LP) rather than using a fixed step size.

However, I'm concerned about the "subgradient" being just for one lag. The true subdifferential of the max function is the convex hull of gradients of all active constraints. Using just one might cause oscillation.

A better approach: use the gradient of the smooth max (log-sum-exp) instead of a single subgradient. This gives a "soft" subgradient that averages over all lags.

Or: cycle through the top few lags.

Let me use a compromise: use the smooth max gradient (which is a convex combination of individual gradients, weighted by softmax).

The gradient of smooth_max(corr, alpha) = corr_max + log(sum(exp(alpha*(corr - corr_max))))/alpha is:
∂/∂h = alpha * exp(alpha*(corr - corr_max)) / sum(exp(alpha*(corr - corr_max))) dotted with ∂corr/∂h

For large alpha, this concentrates on the maximum. For moderate alpha, it's a weighted average.

This is actually what the previous SLSQP approach was doing (implicitly, through the smooth_max objective). The Frank-Wolfe approach with this "soft" subgradient would be:

```python
def frank_wolfe_smooth(h, n, max_iter=200, alpha_sm=50):

        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        corr_max = np.max(corr)

        weights = np.exp(alpha_sm * (corr - corr_max))
        weights /= np.sum(weights)

        grad = np.zeros(n)

                best_alpha = alpha_ls

        if c5 < best_c5:

            best = h.copy()
    
    return best, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - np.sum(h)
    for _ in range(n + 5):
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        per = diff / np.sum(mask)
        if diff > 0:
            h[mask] = np.minimum(1.0, h[mask] + per)
        else:
            h[mask] = np.maximum(0.0, h[mask] + per)
        diff = target - np.sum(h)
    return h

def smooth_max(x, alpha):
    m = np.max(x)
    s = alpha * (x - m)
    s = np.clip(s, -50, 50)
    return m + np.log(np.sum(np.exp(s))) / alpha

def fw_smooth(h, n, max_iter, alpha_sm=50):
    """Frank-Wolfe with smooth max subgradient."""
    h = h.copy()
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        corr_max = np.max(corr)
        weights = np.exp(alpha_sm * (corr - corr_max))
        wsum = np.sum(weights)
        if wsum < 1e-300:
            break
        weights /= wsum
        
        grad = np.zeros(n)
        for k in range(len(corr)):
            w = weights[k]
            if w < 1e-10:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                for i in range(n - lag):
                    grad[i] += w * (1.0 - h[i + lag])
                    grad[i + lag] += w * (-h[i])
            else:
                lag2 = -lag
                for i in range(lag2, n):
                    grad[i] += w * (1.0 - h[i - lag2])
                    grad[i - lag2] += w * (-h[i])
        
        # Linear min oracle
        idx = np.argsort(grad)
        s = np.zeros(n)
        if n % 2 == 0:
            s[idx[:n // 2]] = 1.0
        else:
            s[idx[:(n - 1) // 2]] = 1.0
            s[idx[n // 2:]] = 0.0
            s[idx[(n - 1) // 2]] = 0.5
        s = fix_sum(s, n)
        
        # Line search
        best_a = 2.0 / (it + 2)
        best_v = compute_c5(h)
        for a in [0.05, 0.1, 0.2, 0.4, 0.6, 1.0]:

            v = compute_c5(hv)

        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    
    return best, best_c5

def slsqp_opt(h, n, alpha, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        return smooth_max(np.correlate(x, 1.0 - x, mode="full") * dx, alpha)
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}],

        h_new = fix_sum(res.x, n)
        return h_new, compute_c5(h_new)
    except:
        return h, compute_c5(h)
```

```python
        for rst in range(3):
            if el() > t3:
                break
            hr = fix_sum(rng.uniform(0, 1, n_try), n_try)
            h_fw, c5_fw = fw_smooth(hr, n_try, max_iter=1500, alpha_sm=60)
            if c5_fw < best_c5:
                best_c5 = c5_fw
                best_h = h_fw.copy()
            for alpha in [50, 300]:
                if el() > t3:
                    break
                h_new, c5 = slsqp_opt(h_fw, n_try, alpha, maxiter=300)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                h_fw = h_new
            hb = binary_round(h_fw, n_try)
            if compute_c5(hb) < best_c5:
                best_c5 = compute_c5(hb)
                best_h = hb.copy()

    # Phase 4: Final high-precision polish (8% budget)
    if el() < budget_s * 0.97:
        for alpha in [5000, 15000, 40000]:
            if el() > budget_s * 0.97:
                break
            h_new, c5 = slsqp_opt(best_h, len(best_h), alpha, maxiter=600)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new
        # One more FW pass from polished solution
        if el() < budget_s * 0.95:
            h_fw, c5_fw = fw_smooth(best_h, len(best_h), max_iter=2000, alpha_sm=200)
            if c5_fw < best_c5:
                best_c5 = c5_fw
                best_h = h_fw
            h_new, c5 = slsqp_opt(best_h, len(best_h), 10000, maxiter=400)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new

    final_h = fix_sum(best_h, len(best_h))
    return (final_h, compute_c5(final_h), len(final_h))
```
