import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_max_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = np.argmax(corr)
    d = k_max - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        idx = np.arange(n - d)
        g[idx] += (1.0 - h[idx + d])
        g[idx + d] += (-h[idx])
    else:
        idx = np.arange(-d, n)
        g[idx] += (1.0 - h[idx + d])
        g[idx + d] += (-h[idx])

def grad_smooth_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = np.max(corr)
    weights = np.exp(alpha * (corr - c_max))
    weights /= np.sum(weights)
    # Now compute weighted sum of gradients
    g = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        d = k - (n - 1)
        if d >= 0:
            idx = np.arange(n - d)
            g[idx] += weights[k] * (1.0 - h[idx + d])
            g[idx + d] += -weights[k] * h[idx]
        else:
            idx = np.arange(-d, n)
            g[idx] += weights[k] * (1.0 - h[idx + d])
            g[idx + d] += -weights[k] * h[idx]
    return g

def adam_opt(h0, steps, lr, alpha=100.0, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(h0)
    target = n / 2.0
    x = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    
    for t in range(steps):
        g = grad_smooth_max(x, alpha)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** (t + 1))
        v_hat = v / (1 - beta2 ** (t + 1))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x = project_box_sum(x, target)
    
    return x

def nesterov_opt(h0, steps, lr0, alpha=100.0):
    n = len(h0)
    target = n / 2.0
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    
    for s in range(steps):
        g = grad_smooth_max(y, alpha)
        lr = lr0 / (1 + s * 0.01)
        x_new = project_box_sum(y - lr * g, target)
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        y = x_new + (t - 1) / t_new * (x_new - x)
        y = np.clip(y, 0, 1)
        x = x_new
        t = t_new
    
    return x

def do_slsqp(h_start, n, alpha, maxiter):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
        c_max = np.max(corr)
        s = np.exp(alpha * (corr - c_max))
        return c_max + np.log(np.sum(s)) / alpha

        return grad_smooth_max(x, alpha)
    
    res = minimize(
        obj, grad, h_start,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10}
    )
    h_new = fix_sum(res.x, n)
    return h_new, compute_c5(h_new)

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def smooth_max_val(x, alpha):
    x_max = np.max(x)
    shifted = np.clip(alpha * (x - x_max), -50, 50)
    return x_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = np.max(corr)
    weights = np.exp(np.clip(alpha * (corr - c_max), -50, 50))
    weights /= np.sum(weights)
    
    g = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-12:
            continue
        d = k - (n - 1)
        w = weights[k]
        if d >= 0:
            idx = np.arange(n - d)
            g[idx] += w * (1.0 - h[idx + d])
            g[idx + d] += -w * h[idx]
        else:
            idx = np.arange(-d, n)
            g[idx] += w * (1.0 - h[idx + d])
            g[idx + d] += -w * h[idx]
    return g * dx

def project_box_sum(h, target):
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s < target:
            hi = mid
        else:
            lo = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def fix_sum(h, n):
    return project_box_sum(h, n / 2.0)

def adam_opt(h0, steps, lr, alpha=200.0):
    n = len(h0)
    target = n / 2.0
    x = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    for t in range(steps):
        g = smooth_max_grad(x, alpha)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** (t + 1))
        v_hat = v / (1 - beta2 ** (t + 1))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x = project_box_sum(x, target)
    
    return x

def nesterov_opt(h0, steps, lr0, alpha=200.0):
    n = len(h0)
    target = n / 2.0
    x = h0.copy()
    y = h0.copy()
    t_val = 1.0
    
    for s in range(steps):
        g = smooth_max_grad(y, alpha)
        lr = lr0 / np.sqrt(s + 1)
        x_new = project_box_sum(y - lr * g, target)
        t_new = (1 + np.sqrt(1 + 4 * t_val * t_val)) / 2
        y = x_new + (t_val - 1) / t_new * (x_new - x)
        y = np.clip(y, 0, 1)
        x = x_new
        t_val = t_new
    
    return x

    h_new = fix_sum(res.x, n)

    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr = corr_raw * dx
    c_max = np.max(corr)
    weights = np.exp(np.clip(alpha * (corr - c_max), -50, 50))
    weights /= np.sum(weights)
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-12:
            continue
        d = k - (n - 1)
        w = weights[k]
        # For j where j-d is valid: g[j] += w * (1-h[j-d])
        # j-d >= 0 → j >= d; j-d < n → j < n+d
        j_lo = max(0, d)
        j_hi = min(n, n + d)
        if j_lo < j_hi:
            g[j_lo:j_hi] += w * (1.0 - h[j_lo - d:j_hi - d])
        
        # For j where j+d is valid: g[j] += -w * h[j+d]
        # j+d >= 0 → j >= -d; j+d < n → j < n-d
        j_lo2 = max(0, -d)
        j_hi2 = min(n, n - d)
        if j_lo2 < j_hi2:
            g[j_lo2:j_hi2] += -w * h[j_lo2 + d:j_hi2 + d]
    
    return g * dx
```

This is O(n) per k, and there are 2n-1 k values, so O(n^2) total. Good.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def grad_smooth_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr = corr_raw * dx
    c_max = np.max(corr)
    weights = np.exp(np.clip(alpha * (corr - c_max), -50, 50))
    weights /= np.sum(weights)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-12:
            continue
        d = k - (n - 1)
        w = weights[k]
        j_lo = max(0, d)
        j_hi = min(n, n + d)
        if j_lo < j_hi:
            g[j_lo:j_hi] += w * (1.0 - h[j_lo - d:j_hi - d])
        j_lo2 = max(0, -d)
        j_hi2 = min(n, n - d)
        if j_lo2 < j_hi2:
            g[j_lo2:j_hi2] += -w * h[j_lo2 + d:j_hi2 + d]
    return g * dx

def project_box_sum(h, target):
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s < target:
            hi = mid
        else:
            lo = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def fix_sum(h, n):
    return project_box_sum(h, n / 2.0)

def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / len(h))
    c_max = np.max(corr)
    return c_max + np.log(np.sum(np.exp(np.clip(alpha*(corr-c_max),-50,50))))/alpha

def adam_opt(h0, steps, lr, alpha=200.0, tol=1e-10):
    n = len(h0)
    target = n / 2.0
    x = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    prev_val = smooth_max_obj(x, alpha)
    stuck = 0
    for t in range(steps):
        g = grad_smooth_max(x, alpha)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        x = x - lr * (m / (1-b1**(t+1))) / (np.sqrt(v / (1-b2**(t+1))) + eps)
        x = project_box_sum(x, target)
        if t % 100 == 99:
            val = smooth_max_obj(x, alpha)
            if abs(prev_val - val) < tol:
                stuck += 1
                if stuck > 3:
                    break
            else:
                stuck = 0
            prev_val = val
    return x

def nesterov_opt(h0, steps, lr0, alpha=200.0):
    n = len(h0)
    target = n / 2.0
    x = h0.copy()
    y = h0.copy()
    tv = 1.0
    for s in range(steps):
        g = grad_smooth_max(y, alpha)
        lr = lr0 / np.sqrt(s + 1)
        x_new = project_box_sum(y - lr * g, target)
        tv_new = (1 + np.sqrt(1 + 4 * tv * tv)) / 2
        y = x_new + (tv - 1) / tv_new * (x_new - x)
        y = np.clip(y, 0, 1)
        x = x_new
        tv = tv_new
    return x

def lbfgs_polish(h0, alpha, maxiter=500):
    n = len(h0)
    def obj(x):
        return smooth_max_obj(x, alpha)
    def grad(x):
        return grad_smooth_max(x, alpha)
    res = minimize(obj, grad, h0, method='L-BFGS-B',
                   bounds=[(0,1)]*n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
    h_new = fix_sum(res.x, n)
    return h_new, compute_c5(h_new)

    start_time = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:

    n_init = len(h_init)
    h_init = fix_sum(h_init, n_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    # Phase 1: Adam from initial with various hyperparameters
    if tl() > 60:
        for lr in [0.008, 0.005, 0.003, 0.002, 0.001]:

        for lr0 in [0.05, 0.03, 0.02]:

            h_ne = nesterov_opt(h_init.copy(), 5000, lr0, 200)
            c5 = compute_c5(h_ne)
            if c5 < best_c5:
                best_c5, best_h = c5, h_ne.copy()

    if tl() > 30:
        for alpha in [100, 300, 500, 1000, 2000]:
            if tl() < 15:
                break
            h_p, c5_p = lbfgs_polish(best_h.copy(), alpha, 300)
            if c5_p < best
