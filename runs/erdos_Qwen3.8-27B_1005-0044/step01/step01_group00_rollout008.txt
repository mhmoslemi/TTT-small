import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_capped_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max(corr, alpha):
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

def grad_smooth_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k_idx in range(n_corr):
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            nc = n - k
            grad[:nc] += -wk * h[k:]
            grad[k:] += wk * (1.0 - h[:nc])
        else:
            kp = -k
            nc = n - kp
            grad[:nc] += wk * (1.0 - h[kp:])
            grad[kp:] += -wk * h[:nc]
    return grad * dx

def fista(h0, target, alpha, max_iter, deadline, step0=None):
    n = len(h0)
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best_x = x.copy()
    best_c5 = compute_c5(x)
    if step0 is not None:
        step = step0
    else:
        step = 0.5 / (1.0 + alpha * 0.01)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = grad_smooth_max(y, alpha)
        x_new = project_capped_simplex(y - step * g, target)
        f_old = smooth_max(np.correlate(x, 1.0 - x, mode='full'), alpha) * dx
        f_new = smooth_max(np.correlate(x_new, 1.0 - x_new, mode='full'), alpha) * dx
        if f_new > f_old + 1e-15:
            step *= 0.5
            if step < 1e-14:
                break
            continue
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = np.clip(y, -0.5, 1.5)
        x = x_new
        t = t_new
        step = min(step * 1.06, 5.0)
        c5 = compute_c5(x)
        if c5 < best_c5:
            best_c5 = c5
            best_x = x.copy()
    return best_x, best_c5

def local_search(h0, target, max_iter, deadline, seed, temp0=0.012):
    rng = np.random.RandomState(seed)
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    temp = temp0
    for it in range(max_iter):
        if time.time() > deadline:
            break

        idx = rng.choice(n, size=p, replace=False)
        h_p = h.copy()

        h_p = project_capped_simplex(h_p, target)

            h = h_p

            h = h_p
