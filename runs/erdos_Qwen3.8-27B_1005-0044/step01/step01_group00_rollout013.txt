import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_capped_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):

        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

        else:
            kp = -k
            nc = n - kp
            grad[:nc] += wk * (1.0 - h[kp:])

For coordinate descent: given h with all coordinates fixed except h[i], the objective max_k f_k(h) as a function of h[i] is:
- f_k(h) = a_k + b_k * h[i] + c_k * h[i]^2 (for k=0, c_k < 0; for k≠0, c_k = 0)

Wait, let me be more precise. For a given k:
f_k(h) = Σ_{valid j} h[j]*(1-h[j+k])*dx

The terms involving h[i]:
- If i is in the "first" position: h[i]*(1-h[i+k]) → contributes h[i] - h[i]*h[i+k]
- If i is in the "second" position: h[i-k]*(1-h[i]) → contributes h[i-k] - h[i-k]*h[i]
- For k=0: h[i]*(1-h[i]) → h[i] - h[i]^2

So f_k as a function of h[i] is:
- For k=0: (a_0 - h[i]^2) where a_0 is the sum of all other terms
  Actually: f_0 = Σ_j h[j]*(1-h[j])*dx = (Σ_{j≠i} h[j]*(1-h[j]) + h[i] - h[i]^2)*dx
  = (const + h[i] - h[i]^2)*dx
  This is a concave quadratic in h[i], maximized at h[i] = 0.5.

- For k≠0: f_k = (const + (1-h[i+k])*h[i] - h[i-k]*h[i])*dx (for valid indices)
  = (const + h[i]*((1-h[i+k]) - h[i-k]))*dx
  This is LINEAR in h[i]!

So for k≠0, f_k is linear in h[i], and for k=0, f_0 is concave quadratic in h[i].

The max of these is: max(max_{k≠0} (a_k + b_k*h[i]), a_0 + h[i] - h[i]^2)

Since the linear functions have different slopes, the max is a convex piecewise-linear function (ignoring the quadratic). Adding the concave quadratic makes it more complex, but the maximum is still well-defined.

For coordinate descent: given h[i] = current value, find h[i]' that minimizes max_k f_k(h). Since each f_k is either linear or concave quadratic in h[i], the max is a "roof" function. We can find the minimum by:
1. Evaluating at h[i] = 0 and h[i] = 1 (boundaries)
2. Finding the points where any two f_k cross
3. Evaluating the max at each candidate point

This is O(K^2) per coordinate where K is the number of shifts, so O(n^2) per full pass. For n=47, that's about 2200*47 ≈ 100K operations per pass. Very fast!

Actually, since for k≠0 the functions are linear, the max is determined by the two "upper envelope" lines. The minimum of the max of lines is where two lines cross (or at a boundary). So for each coordinate, we just need to find the crossing point of the two most restrictive linear constraints.

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_capped_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_val(corr, alpha):
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

def grad_smooth_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        wk = w[ki]
        if wk < 1e-13:
            continue
        k = ki - (n - 1)
        if k >= 0:
            nc = n - k
            grad[k:] += wk * (1.0 - h[:nc])
            grad[:nc] -= wk * h[k:]
        else:
            kp = -k
            nc = n - kp
            grad[:nc] += wk * (1.0 - h[kp:])
            grad[kp:] -= wk * h[:nc]
    return grad * dx

def slsqp_solve(h0, target, alpha, max_iter=300):
    n = len(h0)
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max_val(corr, alpha) * dx
    
    def grad(x):
        return grad_smooth_max(x, alpha)
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    bnds = [(0.0, 1.0)] * n
    
    res = minimize(obj, h0, jac=grad, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    h_out = np.clip(res.x, 0, 1)
    h_out = project_capped_simplex(h_out, target)
    return h_out

    alphas = [5, 20, 80, 300, 1000, 3000]

        if time.time() > deadline:
            break
        h = best_h.copy()

        beta = 0.85

        for it in range(5000):
            if time.time() > deadline:
                return best_h, best_c5
            g = grad_smooth_max(h, alpha)
            v = beta * v + g
            h_new = project_capped_simplex(h - step * v, target)
            val_new = smooth_max_val(np.correlate(h_new, 1.0 - h_new, mode='full'), alpha) * (2.0/n)
            if val_new < prev_val:
                prev_val = val_new
                step = min(step * 1.04, 1.0)
            else:
                step *= 0.5
                if step < 1e-10:
                    break
            h = h_new
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5

        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

        c5_p = compute_c5(h_p)
        if c5_p <= best_c5:
            h = h_p
            best_c5 = c5_p
            best_h = h.copy()
        elif rng.random() < np.exp(-(c5_p - best_c5) / max(temp, 1e-10)):
            h = h_p
        if it % 300 == 0:
            temp = max(temp * 0.92, 0.003)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(47, 0.5)

    n0 = len(h0)
    target0 = n0 / 2.0
    h0 = project_capped_simplex(h0, target0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing alpha
    for alpha in [5, 20, 100, 500, 2000]:
        if time.time() > t0 + budget_s * 0.30:
            break
        h_s = slsqp_solve(best_h, target0, alpha, max_iter=500)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s

    # Phase 2: PGD with momentum
    if time.time() < t0 + budget_s * 0.50:
        h_p, c5_p = pgd_optimize(best_h, target0, t0, t0 + budget_s * 0.50, seed)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p

    # Phase 3: Coordinate sweep
    if time.time() < t0 + budget_s * 0.65:
        h_c, c5_c = coord_sweep(best_h, target0, 40, t0 + budget_s * 0.65)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c

    # Phase 4: Simulated annealing
    if time.time() < t0 + budget_s * 0.80:
        h_a, c5_a = anneal_search(best_h, target0, 20000, t0 + budget_s * 0.80, seed + 50)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a

    # Phase 5: SLSQP + coord polish on best
    if time.time() < deadline:
        h_f = slsqp_solve(best_h, target0, 5000, max_iter=500)
        h_f, c5_f = coord_sweep(h_f, target0, 30, deadline)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    # Phase 6: Try complement and reverse as starts
    for transform in [lambda x: 1.0 - x, lambda x: x[::-1]]:
        if time.time() > deadline:
            break
        h_alt = project_capped_simplex(transform(best_h.copy()), target0)
        c5_alt = compute_c5(h_alt)
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt
        elif time.time() < deadline:
            h_alt2 = slsqp_solve(h_alt, target0, 500, max_iter=300)
            h_alt2, c5_alt2 = coord_sweep(h_alt2, target0, 20, deadline)
            if c5_alt2 < best_c5:
                best_c5 = c5_alt2
                best_h = h_alt2

    # Phase 7: Try different n values
    for n_try in [40, 50, 56, 64, 80, 100, 128]:
        if time.time() > deadline:
            break
        rng = np.random.RandomState(seed + n_try)
        for attempt in range(2):
            if time.time() > deadline:
                break
            if attempt == 0:
                h_try = 0.5 + 0.3 * np.sin(2 * np.pi * np.arange(n_try) / n_try * 3)
            else:
                h_try = np.concatenate([np.ones(n_try // 3), np.zeros(n_try - n_try // 3)])
            h_try = project_capped_simplex(h_try + rng.randn(n_try) * 0.3, n_try / 2.0)
            h_try = slsqp_solve(h_try, n_try / 2.0, 100, max_iter=300)
            h_try = project_capped_simplex(h_try, n_try / 2.0)
            if time.time() > deadline:
                break
            h_try, c5_try = coord_sweep(h_try, n_try / 2.0, 15, deadline)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Final coord sweep
    if time.time() < deadline:
        h_fin, c5_fin = coord_sweep(best_h, len(best_h) / 2.0, 20, deadline)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    return best_h, best_c5, len(best_h)
