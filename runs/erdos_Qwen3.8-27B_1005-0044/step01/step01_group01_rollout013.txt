```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def corr_grad(h, k):
    """Gradient of (corr[k] * dx) w.r.t. h"""
    n = len(h)
    s = k - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    idx1 = np.arange(n) - s
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = np.arange(n) + s
    valid2 = (idx2 >= 0) & (idx2 < n)
    grad[valid1] += (1.0 - h[idx1[valid1]])
    grad[valid2] += (-h[idx2[valid2]])
    return grad * dx

def project_simplex(h, n):
    """Project h onto {0<=h_i<=1, sum(h_i) = n/2}"""
    target = n / 2.0
    h = h.copy()
    # Bisection on tau for sum(clip(h-tau, 0, 1)) = target
    lo, hi = -max(h)-0.1, max(h)+0.1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    h = np.clip(h - tau, 0.0, 1.0)
    # Fix tiny residual
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if np.count_nonzero(mask) > 0:
            h[mask] += diff / np.count_nonzero(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def fix_sum(h, n):
    """Fix sum to exactly n/2 while keeping 0<=h<=1"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - np.sum(h)
    for _ in range(200):
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        per = diff / cnt
        if diff > 0:
            h[mask] += np.minimum(per, 1.0 - h[mask])
        else:
            h[mask] += np.maximum(per, -h[mask])
        diff = target - np.sum(h)
    return h

# Global variables for smooth max objective
_g_n = 0
_g_alpha = 50.0

def smooth_max_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full") * (2.0 / _g_n)
    x_max = np.max(corr)
    shifted = _g_alpha * (corr - x_max)
    shifted = np.clip(shifted, -50, 50)
    return x_max + np.log(np.sum(np.exp(shifted))) / _g_alpha

def sum_constraint(x):
    return np.sum(x) - _g_n / 2.0

def slsqp_optimize(h_start, n, alpha, maxiter=300):
    global _g_n, _g_alpha
    _g_n = n
    _g_alpha = alpha
    try:
        res = minimize(
            smooth_max_obj, h_start, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_constraint}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        h_new = fix_sum(np.clip(res.x, 0, 1), n)
        return h_new, compute_c5(h_new)
    except:
        return None, float('inf')

def projected_gradient_optimize(h, n, max_iters=5000, budget_frac=0.3):
    """Projected subgradient method with adaptive step size"""
    start = time.time()
    h = h.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    step = 1.0 / n
    best_step = step
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() - start > budget_frac:
            break
        corr = compute_corr(h)
        val = np.max(corr)
        
        # Soft active set: top m constraints
        m = min(5, len(corr))
        top_k = np.argsort(corr)[-m:]
        # Weights: exponential of (corr[k] - max) / temperature
        temp = 0.005
        weights = np.exp((corr[top_k] - val) / temp)
        weights /= np.sum(weights)
        
        # Combine gradients
        grad = np.zeros(n)
        for w, k in zip(weights, top_k):
            grad += w * corr_grad(h, k)
        
        # Gradient norm for step sizing
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Take a step with backtracking
        h_new = project_simplex(h - step * grad, n)
        new_val = compute_c5(h_new)
        
        if new_val < best_val - 1e-12:
            best_h = h_new.copy()
            best_val = new_val
            no_improve = 0
            best_step = step
        else:
            no_improve += 1
            if no_improve > 20:
                step *= 0.7
                no_improve = 0
            elif no_improve > 5:
                step *= 0.85
        
        h = h_new
    
    return best_h, best_val

def resample(h_old, n_old, n_new):
    """Resample h from n_old points to n_new points"""
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_new = np.interp(x_new, x_old, h_old, period=2.0)  # periodic interpolation
    # Actually, the domain is [0,2] and h is 0 outside, so no periodicity
    # Let me use linear interpolation
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return fix_sum(h_new, n_new)
```

```python
        hr = fix_sum(rng.uniform(0, 1, n_best), n_best)
        h_r, c5_r = pg_optimize(hr, n_best, time_limit=12, rng=rng)
        if c5_r < best_c5:
            best_h = h_r
            best_c5 = c5_r
            n_best = len(best_h)

    for alpha in [500, 2000, 10000, 50000]:
        if time.time() - t0 > budget_s * 0.98:
            break
        res, c5 = slsqp_optimize(best_h, len(best_h), alpha, 500)
        if res is not None and c5 < best_c5:
            best_h = res
            best_c5 = c5

    h2, c5_2 = pg_optimize(best_h, len(best_h), time_limit=10, rng=rng)
    if c5_2 < best_c5:
        best_h = h2
        best_c5 = c5_2

    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, len(best_h))
```
