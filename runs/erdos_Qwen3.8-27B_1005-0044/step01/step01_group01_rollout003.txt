import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr), corr

def corr_and_dw(corr, beta):
    """Compute softmax weights for reweighting."""
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -700, 700)
    e = np.exp(shifted)
    w = e / e.sum()
    return w

def smooth_max_and_grad(h, alpha):
    """Smooth max of correlation values with analytical gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    
    x_max = np.max(corr)
    shifted = alpha * (corr - x_max)
    shifted = np.clip(shifted, -700, 700)
    e = np.exp(shifted)
    w = e / e.sum()
    value = x_max + np.log(e.sum()) / alpha
    
    # Gradient
    grad = np.zeros(n)
    nm1 = n - 1
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-16:
            continue
        m = nm1 - j
        # Term 1: (1-h[p+m]) for p where 0 <= p+m < n
        p_lo1 = max(0, -m)
        p_hi1 = min(n, n - m)
        if p_hi1 > p_lo1:
            grad[p_lo1:p_hi1] += wj * dx * (1.0 - h[p_lo1 + m:p_hi1 + m])
        # Term 2: -h[p-m] for p where 0 <= p-m < n
        p_lo2 = max(0, m)
        p_hi2 = min(n, n + m)
        if p_hi2 > p_lo2:
            grad[p_lo2:p_hi2] -= wj * dx * h[p_lo2 - m:p_hi2 - m]
    
    return value, grad

def reweight_obj_and_grad(h, beta):
    """Reweighted objective: sum w_j * corr[j], with gradient (treating w as fixed)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx

    value = np.dot(w, corr)

        wj = w[j]
        if wj < 1e-16:

        m = nm1 - j
        p_lo1 = max(0, -m)
        p_hi1 = min(n, n - m)
        if p_hi1 > p_lo1:
            grad[p_lo1:p_hi1] += wj * dx * (1.0 - h[p_lo1 + m:p_hi1 + m])

        p_hi2 = min(n, n + m)
        if p_hi2 > p_lo2:
            grad[p_lo2:p_hi2] -= wj * dx * h[p_lo2 - m:p_hi2 - m]
    
    return value, grad

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - np.sum(h)
    for _ in range(100):
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:

            break

        if diff > 0:
            h[mask] += np.minimum(per, 1.0 - h[mask])
        else:
            h[mask] += np.maximum(per, -h[mask])
        diff = target - np.sum(h)
    return h

def optimize_reweight(h_start, n, beta, penalty, maxiter=5000):
    """L-BFGS-B with reweighted objective + sum constraint penalty."""
    def obj(h):
        v, g = reweight_obj_and_grad(h, beta)
        pen = penalty * (np.sum(h) - n / 2.0) ** 2
        gp = 2.0 * penalty * (np.sum(h) - n / 2.0) * np.ones(n)
        return v + pen, g + gp
    
    res = minimize(obj, h_start, jac=True, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
    h_new = fix_sum(res.x, n)
    c5, _ = compute_c5(h_new)
    return h_new, c5

                   options={'maxiter': maxiter, 'ftol': 1e-14})
    h_new = fix_sum(res.x, n)
    c5, _ = compute_c5(h_new)
    return h_new, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    elapsed = lambda: time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = fix_sum(np.array(initial_h_values, dtype=float), len(initial_h_values))
    else:
        best_h = fix_sum(rng.uniform(0, 1, 45), 45)
    n0 = len(best_h)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: Reweighted L-BFGS-B with increasing beta
    for beta in [20, 50, 100, 300, 1000, 3000, 8000]:
        if elapsed() > budget_s * 0.25:
            break
        for _ in range(3):
            if elapsed() > budget_s * 0.25:
                break
            h_new, c5 = opt_lbfgsb(best_h, n0, beta, 50.0, 5000, True)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5

    # Phase 2: Smooth max L-BFGS-B
    for alpha in [50, 200, 800, 3000, 10000]:
        if elapsed() > budget_s * 0.40:
            break
        h_new, c5 = opt_lbfgsb(best_h, n0, alpha, 50.0, 5000, False)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5

    # Phase 3: SLSQP polish
    for alpha in [200, 1000, 5000]:
        if elapsed() > budget_s * 0.52:
            break
        h_new, c5 = slsqp_polish(best_h, n0, alpha)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5

    # Phase 4: Try other n values
    other_ns = [44, 46, 36, 48, 54, 60, 72, 90, 120, 180]
    for n_try in other_ns:
        if elapsed() > budget_s * 0.80:
            break
        x_old = np.arange(n0) * (2.0 / n0)
        x_new = np.arange(n_try) * (2.0 / n_try)
        h_interp = fix_sum(np.interp(x_new, x_old, best_h), n_try)
        c5_i, _ = compute_c5(h_interp)
        if c5_i < best_c5:
            best_h, best_c5 = h_interp, c5_i

        candidates = [h_interp]
        candidates.append(fix_sum(rng.uniform(0, 1, n_try), n_try))
        if n_try % 2 == 0:
            hb = np.zeros(n_try)
            hb[:n_try // 2] = 1.0
            hb = hb[rng.permutation(n_try)]
            candidates.append(fix_sum(hb, n_try))

        for hs in candidates:
            if elapsed() > budget_s * 0.85:
                break
            c5_s, _ = compute_c5(hs)
            if c5_s < best_c5:
                best_h, best_c5 = hs.copy(), c5_s
            for beta in [100, 500, 2000]:
                if elapsed() > budget_s * 0.87:
                    break
                h_new, c5 = opt_lbfgsb(hs, n_try, beta, 50.0, 3000, True)
                if c5 < best_c5:
                    best_h, best_c5 = h_new, c5
                hs = h_new

    # Phase 5: Final high-precision polish
    n_best = len(best_h)
    for alpha in [3000, 10000, 30000]:
        if elapsed() > budget_s * 0.95:
            break
        h_new, c5 = opt_lbfgsb(best_h, n_best, alpha, 200.0, 8000, False)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5

    if elapsed() < budget_s * 0.97:
        h_new, c5 = slsqp_polish(best_h, n_best, 10000, 400)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5

    final_c5, _ = compute_c5(best_h)
    return (best_h, final_c5, len(best_h))
