import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h

def smooth_max_grad(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -80, 80)
    w = np.exp(shifted)
    w /= w.sum()
    val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += wk * (1.0 - h[:n - lag])
            g[:n - lag] -= wk * h[lag:]
        else:
            kp = -lag
            g[:n - kp] += wk * (1.0 - h[kp:])
            g[kp:] -= wk * h[:n - kp]
    return val, g

        return (cm + np.log(np.sum(np.exp(sh))) / alpha) * dx
