    corr = np.correlate(h, 1.0 - h, mode='full')
    val, w = smooth_max_and_weights(corr, alpha)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):

            continue
        k = k_idx - (n - 1)
        if k >= 0:

            grad[k:] += wk * (1.0 - h[:nc])
            grad[:nc] -= wk * h[k:]
        else:
            kp = -k
            nc = n - kp
            grad[:nc] += wk * (1.0 - h[kp:])
            grad[kp:] -= wk * h[:nc]

    best_h = h.copy()
    best_c5 = compute_c5(h)
    for alpha in alphas:
        h = best_h.copy()
        step = 0.05
        prev_val = smooth_max_and_weights(np.correlate(h, 1.0 - h, mode='full'), alpha)[0]
        for it in range(max_iter_per_alpha):
            if time.time() > deadline:
                return best_h, best_c5
            g = grad_smooth_max(h, alpha)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-12:
                break
            # Backtracking
            for ls in range(20):
                h_new = project_capped_simplex(h - step * g, target)
                val_new = smooth_max_and_weights(np.correlate(h_new, 1.0 - h_new, mode='full'), alpha)[0]

                    break
                step *= 0.5

                step = min(step * 1.3, 0.5)
                c5 = compute_c5(h)
                if c5 < best_c5:

                    best_h = h.copy()

                step *= 0.5
                if step < 1e-12:
                    break
