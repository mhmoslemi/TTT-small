from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - np.sum(h)
    for _ in range(3 * n):
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        per = diff / np.sum(mask)
        if diff > 0:
            h[mask] = np.minimum(1.0, h[mask] + per)
        else:
            h[mask] = np.maximum(0.0, h[mask] + per)
        diff = target - np.sum(h)
    return h

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def lag_gradient(h, j, n):
    dx = 2.0 / n
    s = j - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        i1 = np.arange(s, n)
        grad[i1] += dx * (1.0 - h[i1 - s])
        i2 = np.arange(0, n - s)
        grad[i2] -= dx * h[i2 + s]
    else:
        ss = -s
        i1 = np.arange(0, n - ss)
        grad[i1] += dx * (1.0 - h[i1 + ss])
        i2 = np.arange(ss, n)
        grad[i2] -= dx * h[i2 - ss]
    return grad

def smooth_max_val_grad(h, alpha):
    n = len(h)
    lags = compute_lags(h)
    x_max = np.max(lags)
    shifted = np.clip(alpha * (lags - x_max), -80, 80)
    val = x_max + np.log(np.sum(np.exp(shifted))) / alpha
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    grad = np.zeros(n)
    for j in range(len(exp_shifted)):
        w = exp_shifted[j] / total
        if w > 1e-12:
            grad += w * lag_gradient(h, j, n)
    return val, grad

def slsqp_optimize(h0, n, alpha, maxiter=300, ftol=1e-14):
    def obj(x):
        return smooth_max_val_grad(x, alpha)[0]
    def jac(x):
        return smooth_max_val_grad(x, alpha)[1]
    def eq_con(x):
        return np.sum(x) - n / 2.0
    def eq_con_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eq_con, 'jac': eq_con_jac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = fix_sum(np.clip(res.x, 0, 1), n)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0, compute_c5(h0)

def subgrad_descent(h0, n, n_iter=500, init_step=0.1, time_limit=None):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = init_step
    t0 = time.time() if time_limit else None
    
    for it in range(n_iter):
        if time_limit and time.time() - t0 > time_limit:
            break
        lags = compute_lags(h)
        current_max = np.max(lags)
        if current_max < best_c5 - 1e-12:
            best_c5 = current_max
            best_h = h.copy()

        n_active = min(1 + it // 100, 5)
        top_idx = np.argsort(lags)[-n_active:]
        
        grad = np.zeros(n)
        for idx in top_idx:

        else:
            step = min(step * 1.1, 0.5)
    
    return best_h, best_c5

def binary_local_search(n, n_iter=10000, seed=42):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(n_iter):
        # Pick random i with h[i]=0 and j with h[j]=1
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        # Swap
        h[i], h[j] = h[j], h[i]
        c5 = compute_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]  # revert
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - np.sum(h)
    for _ in range(3 * n):
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        per = diff / np.sum(mask)
        if diff > 0:
            h[mask] = np.minimum(1.0, h[mask] + per)
        else:
            h[mask] = np.maximum(0.0, h[mask] + per)
        diff = target - np.sum(h)
    return h

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def lag_gradient(h, j, n):
    dx = 2.0 / n
    s = j - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        i1 = np.arange(s, n)
        grad[i1] += dx * (1.0 - h[i1 - s])
        i2 = np.arange(0, n - s)
        grad[i2] -= dx * h[i2 + s]
    else:
        ss = -s
        i1 = np.arange(0, n - ss)
        grad[i1] += dx * (1.0 - h[i1 + ss])
        i2 = np.arange(ss, n)
        grad[i2] -= dx * h[i2 - ss]
    return grad

def smooth_max_val_grad(h, alpha):
    n = len(h)
    lags = compute_lags(h)
    x_max = np.max(lags)
    shifted = np.clip(alpha * (lags - x_max), -80, 80)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    val = x_max + np.log(total) / alpha
    grad = np.zeros(n)
    for j in range(len(exp_s)):
        w = exp_s[j] / total
        if w > 1e-13:
            grad += w * lag_gradient(h, j, n)
    return val, grad

def slsqp_optimize(h0, n, alpha, maxiter=300, ftol=1e-14):
    def obj(x):
        return smooth_max_val_grad(x, alpha)[0]
    def jac(x):
        return smooth_max_val_grad(x, alpha)[1]
    def eq_con(x):
        return np.sum(x) - n / 2.0
    def eq_con_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eq_con, 'jac': eq_con_jac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = fix_sum(np.clip(res.x, 0, 1), n)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0, compute_c5(h0)

def subgrad_descent(h0, n, n_iter=500, init_step=0.1, time_limit=60.0):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = init_step
    t0 = time.time()
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        lags = compute_lags(h)
        cur_max = np.max(lags)
        if cur_max < best_c5 - 1e-12:
            best_c5 = cur_max
            best_h = h.copy()
        n_act = min(1 + it // 80, 5)
        top_idx = np.argsort(lags)[-n_act:]
        grad = np.zeros(n)
        for idx in top_idx:
            grad += lag_gradient(h, idx, n)
        grad /= n_act
        accepted = False
        for _ in range(15):
            h_new = fix_sum(h - step * grad, n)
            c5_new = compute_c5(h_new)
            if c5_new < cur_max - 1e-12:
                h = h_new
                accepted = True
                break
            step *= 0.5
        if accepted:
            step = min(step * 1.05, 0.5)
        else:
            step = min(step * 2.0, 0.5)
    return best_h, best_c5

def binary_local_search(n, n_iter=20000, seed=42):
    rng = np.random.RandomState(seed)
    idx = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    zeros = list(idx)
    ones = [i for i in range(n) if i not in set(idx)]
    for it in range(n_iter):
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h[i], h[j] = 1.0, 0.0
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            zeros.append(j)
            ones.remove(i)
        else:
            h[i], h[j] = 0.0, 1.0
            zeros.remove(i)
            ones.append(j)
    return best_h, best_c5
