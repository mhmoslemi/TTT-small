    best_c5 = float('inf')
    n_main = 45

    h_starts = []
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        h0 = fix_sum(h0, len(h0))
        h_starts.append(h0)
        n_main = len(h0)

    for _ in range(4):
        hr = rng.uniform(0, 1, n_main)
        hr = fix_sum(hr, n_main)
        h_starts.append(hr)

    if n_main % 2 == 0:
        h_bin = np.zeros(n_main)
        h_bin[:n_main // 2] = 1.0
        h_starts.append(h_bin.copy())

    alphas_p1 = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]

    # Phase 1: SLSQP from initial starts
    for hs in h_starts:
        if time.time() - start_time > budget_s * 0.30:
            break
        h_cur = hs.copy()
        n_cur = len(h_cur)
        c5_cur = compute_c5(h_cur)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        for alpha in alphas_p1:
            if time.time() - start_time > budget_s * 0.30:
                break
            result = do_slsqp(h_cur, n_cur, alpha, 200)
            if result is None:
                break
            h_cur, c5_new = result
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_cur.copy()

    # Phase 2: Subgradient descent + re-optimize
    if best_h is not None and time.time() - start_time < budget_s * 0.50:
        n_best = len(best_h)
        h_sg, c5_sg = do_subgrad(best_h, n_best, 5000, 0.05)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
        for alpha in [200, 1000, 4000, 10000]:
            if time.time() - start_time > budget_s * 0.55:
                break
            result = do_slsqp(best_h, n_best, alpha, 200)
            if result is not None:
                h_new, c5_new = result
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()

    # Phase 3: Perturbation restarts on best solution
    if best_h is not None:
        n_pert = len(best_h)
        for _ in range(5):
            if time.time() - start_time > budget_s * 0.68:
                break
            h_pert = best_h.copy()
            idxs = rng.choice(n_pert, size=min(3, n_pert), replace=False)
            h_pert[idxs] = rng.uniform(0, 1, size=len(idxs))
            h_pert = fix_sum(h_pert, n_pert)
            h_cur = h_pert
            for alpha in [100, 500, 2000]:
                if time.time() - start_time > budget_s * 0.68:
                    break
                result = do_slsqp(h_cur, n_pert, alpha, 150)
                if result is None:
                    break
                h_cur, c5_new = result
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_cur.copy()

    # Phase 4: Try different n values
    n_list = [44, 46, 48, 50, 52, 42, 40, 54, 56, 38, 36, 60, 64]
    for n_try in n_list:
        if time.time() - start_time > budget_s * 0.85:
            break
        starts_n = [fix_sum(rng.uniform(0, 1, n_try), n_try)]
        if n_try % 2 == 0:
            h_bin2 = np.zeros(n_try)
            h_bin2[:n_try // 2] = 1.0
            for _ in range(2):
                rng.shuffle(h_bin2)
                starts_n.append(h_bin2.copy())

        for hs in starts_n:
            if time.time() - start_time > budget_s * 0.85:
                break
            h_cur = hs.copy()
            c5_cur = compute_c5(h_cur)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
            for alpha in [20, 100, 500, 2000, 8000]:
                if time.time() - start_time > budget_s * 0.85:
                    break
                result = do_slsqp(h_cur, n_try, alpha, 150)
                if result is None:
                    break
                h_cur, c5_new = result
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_cur.copy()

    # Phase 5: Final high-alpha polish
    if best_h is not None:
        n_final = len(best_h)
        for alpha in [3000, 8000, 20000]:
            if time.time() - start_time > budget_s * 0.97:
                break
            result = do_slsqp(best_h, n_final, alpha, 400)
            if result is not None:
                h_new, c5_new = result
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new

    if best_h is None:
        best_h = fix_sum(np.full(n_main, 0.5), n_main)

    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, len(best_h))
