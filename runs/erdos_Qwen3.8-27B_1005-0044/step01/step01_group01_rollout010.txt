def compute_c5(h):

    dx = 2.0 / n

    return np.max(corr), corr

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()

        h[mask] = np.clip(h[mask] + per, 0, 1)
        diff = target - np.sum(h)
    return h

def compute_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -80, 80)
    exp_s = np.exp(s)
    weights = exp_s / np.sum(exp_s)
