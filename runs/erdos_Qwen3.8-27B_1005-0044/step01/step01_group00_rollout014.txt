def compute_c5(h):

def project_simplex_binary(h, target):
    """Project to nearest binary-like solution with correct sum."""
    n = len(h)
    sorted_h = np.sort(h)[::-1]
    # Find the threshold
    cumulative = np.cumsum(sorted_h)
    # We want exactly target ones (if target is integer)
    if abs(target - round(target)) < 1e-9:
        target = int(round(target))
        threshold = sorted_h[target - 1] if target > 0 else 0
        result = (h >= threshold).astype(float)
        # Fix if we have the wrong count
        count = result.sum()
        if count > target:
            # Set some to 0
            above = np.where(result == 1)[0]
            # Set the ones with lowest h values to 0
            above_sorted = above[np.argsort(h[above])]
            for i in range(int(count - target)):
                result[above_sorted[i]] = 0

            below_sorted = below[np.argsort(-h[below])]

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def max_overlap_val(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def project_capped_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def binary_best_swap(h, deadline):
    """Steepest descent on binary hypercube via best single swap."""
    n = len(h)
    h = h.copy().astype(float)
    best_h = h.copy()
    best_mo = max_overlap_val(h)
    improved = True
    
    while improved and time.time() < deadline:
        improved = False
        cur_mo = max_overlap_val(h)
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])
        best_delta = 0
        best_pair = None
        
        for a in ones:
            for b in zeros:
                h[a] = 0.0
                h[b] = 1.0
                mo = max_overlap_val(h)
                h[a] = 1.0
                h[b] = 0.0
                if mo < cur_mo + best_delta:  # mo - cur_mo < best_delta
                    best_delta = mo - cur_mo
                    best_pair = (a, b)
        
        if best_pair is not None:
            a, b = best_pair
            h[a] = 0.0
            h[b] = 1.0
            new_mo = max_overlap_val(h)
            if new_mo < best_mo:
                best_mo = new_mo
                best_h = h.copy()
            improved = True
    
    return best_h, best_mo * (2.0 / n)

def binary_sa(h_init, seed, deadline, phase_iters=200000):
    """SA with 1-swap and 2-swap moves."""
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_mo = max_overlap_val(h)
    cur_mo = best_mo
    
    T = 1.5
    T_min = 0.01
    no_improve = 0
    max_no_improve = 30000
    
    for it in range(phase_iters):
        if time.time() > deadline:
            break
        if it % 100 == 0:
            T = max(T * 0.9997, T_min)
            if no_improve > max_no_improve:
                T = min(T * 5.0, 1.0)  # reheat
                no_improve = 0
        
        # Choose move type
        if rng.random() < 0.8:
            # 1-swap: swap a 1 with a 0
            a = rng.randint(n)
            b = rng.randint(n)
            if a == b or h[a] == h[b]:
                continue
            h[a], h[b] = 1.0 - h[a], 1.0 - h[b]
        else:
            # 2-swap: swap two 1s with two 0s
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                continue
            a, b = rng.choice(ones, 2, replace=False)
            c, d = rng.choice(zeros, 2, replace=False)
            h[a] = 0.0; h[b] = 0.0; h[c] = 1.0; h[d] = 1.0
        
        new_mo = max_overlap_val(h)
        
        if new_mo <= best_mo:
            best_mo = new_mo
            best_h = h.copy()
            cur_mo = new_mo
            no_improve = 0
        elif new_mo <= cur_mo:
            cur_mo = new_mo
            no_improve += 1
        else:
            if rng.random() < np.exp(-(new_mo - cur_mo) / T):
                cur_mo = new_mo
                no_improve += 1
            else:
                # Revert
                if rng.random() < 0.8:  # was 1-swap
                    h[a], h[b] = 1.0 - h[a], 1.0 - h[b]
                else:  # was 2-swap
                    h[a] = 1.0; h[b] = 1.0; h[c] = 0.0; h[d] = 0.0
                no_improve += 1
    
    return best_h, best_mo * (2.0 / n)

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        bd = 0
        bp = None
        for a in ones:
            for b in zeros:
                h[a] = 0.0; h[b] = 1.0
                v = max_ov(h)
                h[a] = 1.0; h[b] = 0.0
                if v - cur < bd:
                    bd = v - cur
                    bp = (int(a), int(b))
        if bp is None:
            break
        h[bp[0]] = 0.0; h[bp[1]] = 1.0
        if max_ov(h) < bm:
            bm = max_ov(h)
            bh = h.copy()
    return bh, bm * (2.0 / n)

def sa_bin(h0, seed, dl):
    rng = np.random.RandomState(seed)
    n = len(h0)
    h = h0.copy().astype(float)
    bh = h.copy()
    bm = max_ov(h)
    cm = bm
    T = 1.5
    stk = 0
    for it in range(500000):
        if time.time() > dl:
            break
        if it % 200 == 0:
            T = max(T * 0.9997, 0.01)
            if stk > 40000:
                T = 1.0; stk = 0
        a = rng.randint(n)
        b = rng.randint(n)
        if a == b or h[a] == h[b]:
            continue
        h[a], h[b] = 1.0 - h[a], 1.0 - h[b]
        nm = max_ov(h)
        if nm <= bm:
            bm = nm; bh = h.copy(); cm = nm; stk = 0
        elif nm <= cm:
            cm = nm; stk += 1
        else:
            if rng.random() < np.exp(-(nm - cm) / T):
                cm = nm; stk += 1
            else:
                h[a], h[b] = 1.0 - h[a], 1.0 - h[b]
                stk += 1
    return bh, bm * (2.0 / n)

def gref(h, dl):
    n = len(h)
    tg = n / 2.0
    h = pclip(h.copy(), tg)
    bh = h.copy()
    bc = compute_c5(h)
    for al in [15.0, 60.0, 200.0]:
        st = 0.03
        for it in range(30000):
            if time.time() > dl:
                return bh, bc
            corr = np.correlate(h, 1.0 - h, mode='full')
            m = np.max(corr)
            ex = np.exp(al * (corr - m))
            w = ex / ex.sum()
            g = np.zeros(n)
            for ki in range(2 * n - 1):
                wk = w[ki]
                if wk < 1e-12:
                    continue
                k = ki - (n - 1)
                if k >= 0:
                    nc = n - k
                    g[k:] += wk * (1.0 - h[:nc])
                    g[:nc] -= wk * h[k:]
                else:
                    kp = -k
                    nc = n - kp
                    g[:nc] += wk * (1.0 - h[kp:])
                    g[kp:] -= wk * h[:nc]
            hn = pclip(h - st * g, tg)
            cn = compute_c5(hn)
            if cn < bc:
                bc = cn; bh = hn.copy(); h = hn
                st = min(st * 1.03, 0.2)
            else:
                h = hn
                st *= 0.5
                if st < 1e-10:
                    break
    return bh, bc
