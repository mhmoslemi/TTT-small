def compute_c5(h):

def project_capped_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for k_idx in sorted_idx:
            k = k_idx - (n - 1)
            gk = np.zeros(n)
            if k >= 0:
                nc = n - k
                gk[k:] += (1.0 - h_curr[:nc])
                gk[:nc] -= h_curr[k:]
            else:
                kp = -k
                nc = n - kp
                gk[:nc] += (1.0 - h_curr[kp:])
                gk[kp:] -= h_curr[:nc]
            rhs = t_var - corr[k_idx] + gk @ h_curr
            constraints.append(gk @ h_var <= rhs)
        constraints.append(cp.norm(h_var - h_curr, 2) <= radius)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        solved = False
        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            try:
                prob.solve(solver=solver, max_iters=2000, abstol=1e-9, reltol=1e-9)
                if h_var.value is not None:
                    solved = True
                    break
            except Exception:
                continue
        if not solved:
            break
        h_new = np.clip(h_var.value, 0.0, 1.0)
        h_new = project_capped_simplex(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new
            radius = max(radius * 0.75, 0.02)
        else:
            radius *= 0.6
            if radius < 0.01:
                top_m = min(top_m + 5, 2 * n - 1)
                radius = 0.08
    return best_h, best_c5

def pair_swap_descent(h, target, n_sweeps, deadline, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    deltas = np.array([-0.1, -0.05, -0.025, -0.01, 0.01, 0.025, 0.05, 0.1])
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            j = (i + rng.randint(1, n)) % n
            hi, hj = h[i], h[j]
            for delta in deltas:
                nhi = hi + delta
                nhj = hj - delta
                if nhi < 0 or nhi > 1 or nhj < 0 or nhj > 1:
                    continue
                h[i], h[j] = nhi, nhj
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    best_h = h.copy()
                    hi, hj = nhi, nhj
                else:
                    h[i], h[j] = hi, hj
                    break
    return best_h, best_c5

def simulated_annealing(h, target, n_iter, deadline, seed, init_temp=0.012):
    rng = np.random.RandomState(seed)
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    temp = init_temp
    block = max(2, n // 14)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        idx = rng.choice(n, size=block, replace=False)
        h_pert = h.copy()
        h_pert[idx] += rng.randn(block) * temp
        h_pert = project_capped_simplex(h_pert, target)
        c5_pert = compute_c5(h_pert)
        if c5_pert <= best_c5:
            h = h_pert
            best_c5 = c5_pert
            best_h = h.copy()
        else:
            if rng.random() < np.exp(-(c5_pert - best_c5) / max(temp, 1e-10)):
                h = h_pert
        if it % 250 == 0 and it > 0:
            temp = max(temp * 0.87, 0.0008)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values, dtype=float)
    else:
        h_start = np.full(47, 0.5)
    n = len(h_start)
    target = n / 2.0
    h_start = project_capped_simplex(h_start, target)
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)
    # Phase 1: PGD line search
    d1 = t0 + budget_s * 0.12
    for alpha in [10.0, 30.0, 80.0, 200.0]:
        if time.time() > d1:
            break
        h_r, c5_r = pgd_line_search(best_h, target, alpha, 400, d1)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r
    # Phase 2: CVXPY LP polish
    d2 = t0 + budget_s * 0.40
    if time.time() < d2:
        h_c, c5_c = cvxpy_lp_polish(best_h, target, d2, max_iters=50, init_radius=0.3)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c
    # Phase 3: Pair swap descent
    d3 = t0 + budget_s * 0.55
    if time.time() < d3:
        h_p, c5_p = pair_swap_descent(best_h, target, 150, d3, seed)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p
    # Phase 4: CVXPY LP polish round 2
    d4 = t0 + budget_s * 0.68
    if time.time() < d4:
        h_c2, c5_c2 = cvxpy_lp_polish(best_h, target, d4, max_iters=40, init_radius=0.15)
        if c5_c2 < best_c5:
            best_c5 = c5_c2
            best_h = h_c2
    # Phase 5: Simulated annealing
    d5 = t0 + budget_s * 0.85
    if time.time() < d5:
        h_s, c5_s = simulated_annealing(best_h, target, 20000, d5, seed, init_temp=0.01)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
    # Phase 6: Final pair swap + PGD
    d6 = t0 + budget_s - 5
    if time.time() < d6:
        h_p2, c5_p2 = pair_swap_descent(best_h, target, 80, d6, seed + 111)
        if c5_p2 < best_c5:
            best_c5 = c5_p2
            best_h = h_p2
    d7 = t0 + budget_s - 2
    if time.time() < d7:
        h_f, c5_f = pgd_line_search(best_h, target, 500.0, 200, d7)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    return best_h, best_c5, len(best_h)
