        per = diff / count

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

    h = np.clip(h, 0.0, 1.0).copy()

    for _ in range(n + 2):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14

        per = diff / count
        if diff > 0:
            h[mask] += np.minimum(per, 1.0 - h[mask])
        else:
            h[mask] += np.maximum(per, -h[mask])

    shifted = alpha * (x - x_max)
    shifted = np.clip(shifted, -700, 700)

def subgrad_step(h, n, step_size):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    i_star = np.argmax(corr)
    k = i_star - (n - 1)
    j = np.arange(n)

        v2_mask = (j >= 0) & (j <= n - 1 - k)

        v2_mask = (j >= -k) & (j <= n - 1)

    h_new = h - step_size * dx * grad
    return fix_sum(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    def elapsed():
        return time.time() - start_time

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = fix_sum(np.array(initial_h_values, dtype=float), len(initial_h_values))
    else:
        h0 = fix_sum(rng.uniform(0, 1, 45), 45)
    n_main = len(h0)

    c5_0 = compute_c5(h0)
    best_c5 = c5_0
    best_h = h0.copy()

    # Phase 1: SLSQP alpha chain
    alphas = [3, 10, 30, 100, 300, 1000, 3000, 10000, 30000, 100000]
    for a in alphas:
        if elapsed() > budget_s * 0.30:
            break
        h0, c5 = slsqp_opt(h0, n_main, a, maxiter=500)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h0.copy()

    # Phase 2: Subgradient + re-optimize cycle
    for sg in range(30):
        if elapsed() > budget_s * 0.55:
            break
        h_spec = best_h.copy()
        for ss in [0.02, 0.01, 0.005]:
            h_spec = subgrad_step(h_spec, n_main, ss)
        h_spec, c5 = slsqp_opt(h_spec, n_main, 5000, maxiter=300)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_spec.copy()
        h_spec, c5 = lbfgsb_opt(h_spec, n_main, 10000, pen_lam=1e5, maxiter=1000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_spec.copy()

    # Phase 3: L-BFGS-B polish with high alpha
    for a in [5000, 20000, 100000, 500000]:
        if elapsed() > budget_s * 0.70:
            break
        h0, c5 = lbfgsb_opt(best_h, n_main, a, pen_lam=1e6, maxiter=3000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h0.copy()

    # Phase 4: Shake and re-optimize
    for sh in range(15):
        if elapsed() > budget_s * 0.85:
            break
        h_sh = best_h.copy()
        n_perturb = rng.randint(3, 10)
        idx = rng.choice(n_main, size=n_perturb, replace=False)
        h_sh[idx] += rng.uniform(-0.15, 0.15, size=n_perturb)
        h_sh = fix_sum(h_sh, n_main)
        for a in [500, 5000, 50000]:
            if elapsed() > budget_s * 0.85:
                break
            h_sh, c5 = slsqp_opt(h_sh, n_main, a, maxiter=400)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_sh.copy()

    # Phase 5: Try nearby n values
    for n_try in [n_main - 1, n_main + 1, n_main + 3, n_main - 3]:
        if n_try < 10 or elapsed() > budget_s * 0.93:
            continue
        x_old = np.linspace(0, 2, n_main, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2.0)
        h_interp = fix_sum(h_interp, n_try)
        for a in [100, 1000, 10000]:
            if elapsed() > budget_s * 0.93:
                break
            h_interp, c5 = slsqp_opt(h_interp, n_try, a, maxiter=400)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_interp.copy()

    # Phase 6: Final polish
    for a in [100000, 500000, 2000000]:
        if elapsed() > budget_s * 0.98:
            break
        n_best = len(best_h)
        best_h, c5 = lbfgsb_opt(best_h, n_best, a, pen_lam=1e7, maxiter=5000)
        if c5 < best_c5:
            best_c5 = c5

    # SLSQP final
    n_best = len(best_h)
    if elapsed() < budget_s * 0.99:
        best_h, c5 = slsqp_opt(best_h, n_best, 100000, maxiter=500)
        if c5 < best_c5:
            best_c5 = c5

    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)
    return (best_h, final_c5, len(best_h))
