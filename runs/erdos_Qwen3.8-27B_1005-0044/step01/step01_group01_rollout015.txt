import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_full(h):
    """Returns all correlation values times dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_c5(h):
    return np.max(compute_c5_full(h))

def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - np.sum(h)
    for _ in range(n + 2):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        count = float(np.sum(mask))
        per = diff / count
        if diff > 0:
            h[mask] += np.minimum(per, 1.0 - h[mask])
        else:
            h[mask] += np.maximum(per, -h[mask])
        diff = target - np.sum(h)
    return h

def two_coord_update(h, i, j, n_grid=100):
    """Update h[i] and h[j] maintaining sum, minimizing C5."""
    n = len(h)
    old_i = h[i]
    old_j = h[j]
    
    # Feasible range for d: h[i] = old_i + d, h[j] = old_j - d
    d_min = max(0.0 - old_i, old_j - 1.0)
    d_max = min(1.0 - old_i, old_j)
    
    if d_min >= d_max - 1e-15:
        return h, 0
    
    # Evaluate C5 at grid points
    d_vals = np.linspace(d_min, d_max, n_grid)
    
    # For each d, compute the change in each correlation
    # c_s(d) = c_s(0) + d * slope_s
    # slope_s comes from the four terms involving i and j
    
    # Let's just directly evaluate
    best_d = 0
    best_c5 = compute_c5(h)
    
    for d in d_vals:
        h[i] = old_i + d
        h[j] = old_j - d
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_d = d
    
    h[i] = old_i + best_d
    h[j] = old_j - best_d
    return h, best_c5

    pair_i = np.array([i for i in range(n) for j in range(i + 1, n)])
    pair_j = np.array([j for i in range(n) for j in range(i + 1, n)])
    for sweep in range(n_sweeps):
        if time.time() - start_time > budget * frac:
            break
        perm = rng.permutation(n_pairs)
        n_improved = 0
        for idx in perm:
            c5v, imp = two_coord_update(h, int(pair_i[idx]), int(pair_j[idx]), c5v, n_grid)
            n_improved += imp
        if sweep % 50 == 49:
            c5v = compute_c5_vec(h)
        if n_improved == 0:
            break
    c5v = compute_c5_vec(h)
    return h, float(np.max(c5v))

def slsqp_polish(h, n, alpha, maxiter):
    global _pn, _pa
    _pn = n
    _pa = alpha
    n_full = 2 * n - 1
    def obj(x):
        c5v = compute_c5_vec(x)
        x_max = np.max(c5v)
        shifted = alpha * (c5v - x_max)
        shifted = np.clip(shifted, -50, 50)
        return x_max + np.log(np.sum(np.exp(shifted))) / alpha
    def constraint(x):
        return np.sum(x) - n / 2.0
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = fix_sum(res.x, n)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, float('inf')

_pn = 45
_pa = 50

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        h0 = fix_sum(h0, len(h0))
    else:
        h0 = fix_sum(rng.uniform(0, 1, 45), 45)
    n0 = len(h0)

    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h_cur.copy()

    # Phase 1: Coordinate descent (coarse to fine)
    phases = [(200, 40, 0.25), (300, 60, 0.50), (400, 80, 0.70)]
    for n_sw, ng, fr in phases:
        h_cur, c5_cur = do_sweeps(h_cur, n_sw, ng, rng, start_time, budget_s, fr)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        if time.time() - start_time > budget_s * 0.7:
            break

    # Phase 2: Random restarts with perturbation
    for restart in range(8):
        if time.time() - start_time > budget_s * 0.82:
            break
        scale = 0.02 + 0.01 * restart
        h_r = best_h + rng.normal(0, scale, n0)
        h_r = fix_sum(h_r, n0)
        h_r, c5_r = do_sweeps(h_r, 150, 60, rng, start_time, budget_s, 0.82)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 3: Try other n values
    for n_try in [42, 48, 50, 36, 54]:
        if time.time() - start_time > budget_s * 0.90:
            break
        for h_start in [fix_sum(rng.uniform(0, 1, n_try), n_try),
                        fix_sum(np.concatenate([np.ones(n_try // 2), np.zeros(n_try - n_try // 2)]), n_try)]:
            if time.time() - start_time > budget_s * 0.90:
                break
            h_r, c5_r = do_sweeps(h_start, 100, 50, rng, start_time, budget_s, 0.90)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 4: SLSQP polish + final coord descent
    if best_h is not None and time.time() - start_time < budget_s * 0.97:
        nb = len(best_h)
        for alpha in [200, 1000]:
            if time.time() - start_time > budget_s * 0.97:
                break
            h_p, c5_p = slsqp_polish(best_h, nb, alpha, 200)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p
        h_f, c5_f = do_sweeps(best_h, 200, 100, rng, start_time, budget_s, 0.995)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, len(best_h))
