    active = np.where(corr >= C - tol)[0]
    
    # Choose a subset of constraints to include (active + a few nearby)
    # For efficiency, include top 20 or so
    sorted_idx = np.argsort(corr)[::-1][:20]

    n_vars = n + 1  # h[0..n-1] + t

        g *= (2.0 / n)
        z = project_capped_simplex(y - lr * g, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = z + beta * (z - y)
        y = project_capped_simplex(y, target)
        t = t_new
        c5 = compute_c5(z)
        if c5 < best_c5:
            best_c5 = c5
            best_h = z.copy()
        if it % 200 == 0 and it > 0:
            lr = max(lr * 0.9, 0.005)
    return best_h, best_c5

def jump_and_refine(h, target, deadline, n_jumps=15, jump_size=0.15, inner_iter=2000):
    n = len(h)
    rng = np.random.RandomState(31337)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for j in range(n_jumps):
        if time.time() > deadline:
            break
        h_jump = best_h + rng.randn(n) * jump_size
        h_jump = project_capped_simplex(h_jump, target)
        h_ref, c5_ref = nesterov_subgrad(h_jump, target, deadline, max_iter=inner_iter)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values, dtype=float)
    else:
        h_start = np.full(47, 0.5)
    
    n = len(h_start)
    target = n / 2.0
    h_start = project_capped_simplex(h_start, target)
    
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: SLSQP with increasing alpha
    for alpha in [10.0, 30.0, 80.0, 200.0, 500.0]:
        if time.time() > t0 + budget_s * 0.3:
            break
        h_s, c5_s = slsqp_run(best_h, target, alpha, maxiter=200)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
    
    # Phase 2: Nesterov subgradient
    dl2 = t0 + budget_s * 0.5
    h_n, c5_n = nesterov_subgrad(best_h, target, dl2, max_iter=8000, tol=1e-4)
    if c5_n < best_c5:
        best_c5 = c5_n
        best_h = h_n
    
    # Phase 3: Jump and refine
    dl3 = t0 + budget_s * 0.7
    h_j, c5_j = jump_and_refine(best_h, target, dl3, n_jumps=12, jump_size=0.12, inner_iter=2000)
    if c5_j < best_c5:
        best_c5 = c5_j
        best_h = h_j
    
    # Phase 4: SLSQP polish
    for alpha in [500.0, 1000.0, 2000.0]:
        if time.time() > t0 + budget_s * 0.85:
            break
        h_s, c5_s = slsqp_run(best_h, len(best_h)/2.0, alpha, maxiter=200)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
    
    # Phase 5: Try other resolutions
    dl5 = t0 + budget_s * 0.92
    for n_try in [50, 64, 80]:
        if time.time() > dl5:
            break
        n_cur = len(best_h)
        if n_try == n_cur:
            continue
        x_old = np.linspace(0, 1, n_cur)
        x_new = np.linspace(0, 1, n_try)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project_capped_simplex(h_try, n_try / 2.0)
        h_t, c5_t = slsqp_run(h_try, n_try / 2.0, 200.0, maxiter=3
