import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_capped_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_val(corr, alpha):
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

def pair_golden(h, i, j, alpha):
    """Optimize h[i], h[j] jointly, maintaining h[i]+h[j]."""
    n = len(h)
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-12:
        return h[i], h[j]
    
    dx = 2.0 / n
    h_i = h[i]
    h_j = h[j]
    
    # We'll temporarily modify h and compute the smoothed max
    def obj(a):
        h[i] = a
        h[j] = s - a
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
        return val
    
    a, b = lo, hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = obj(c)
    fd = obj(d)

            d = c
            fd = fc

            fc = obj(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = obj(d)
    
    best_a = (a + b) / 2.0

    """Do sweeps of pair coordinate descent."""
    n = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    pairs = [(i, j) for i in range(n) for j in range(i+1, n)]
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        rng.shuffle(pairs)
        improved = False
        for i, j in pairs:
            if time.time() > deadline:

            old_i, old_j = h[i], h[j]
            pair_golden(h, i, j, alpha)
            # Check if we improved
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-10:
                best_c5 = c5
                best_h = h.copy()
                improved = True
    
    return best_h, best_c5

The smooth max is: (1/alpha) * log(sum exp(alpha * c_m))
where c_m are the correlation values, each of which is affine in h[i] (for fixed other variables, except for the k=0 term which is quadratic).

If we ignore the k=0 term (or for the case where h[i] is at 0 or 1 so the k=0 term vanishes), then each c_m is affine in h[i], and the smooth max is a convex function of h[i]. So golden section should work.

With the k=0 term, c_0 = sum h[j](1-h[j]) includes h[i](1-h[i]) which is a concave quadratic in h[i]. So the smooth max is NOT convex in h[i] due to this term. However, the non-convexity is mild (it's just one concave quadratic term among many), and in practice, golden section should still find a good local minimum.

To be safe, I could also try a few random starting points for the 1D search (not just golden section from the full interval). Or I could use a multi-start approach: evaluate at the current point, at 0, at 1, and at the golden section result, and take the best.

Let me also think about the k=0 term more carefully. c_0 = sum_j h[j](1-h[j]) = sum_j h[j] - sum_j h[j]^2.
For the constraint sum(h) = n/2, we have sum h[j] = n/2 (fixed). So c_0 = n/2 - sum h[j]^2.
To minimize c_0, we want to maximize sum h[j]^2, which means making h as "peaky" as possible (close to binary).

So the k=0 term actually encourages h to be close to binary! This is interesting. It means the optimal solution is likely close to binary (with a few fractional values to satisfy the sum constraint exactly).

import numpy as np
import time
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_capped_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_val(corr, alpha):
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

def pair_golden(h, i, j, alpha):
    n = len(h)
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-12:
        return False
    orig_i, orig_j = h[i], h[j]
    
    def obj(a):
        h[i] = a
        h[j] = s - a
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    
    f_orig = obj(orig_i)
    a, b = lo, hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = obj(c)
    fd = obj(d)
    
    for _ in range(25):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = obj(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = obj(d)
    
    best_a = (a + b) / 2.0
    f_best = obj(best_a)
    if f_best < f_orig:
        h[i] = best_a
        h[j] = s - best_a
        return True
    else:
        h[i] = orig_i
        h[j] = orig_j
        return False

def pair_descent(h, alpha, n_sweeps, seed, deadline):
    n = len(h)
    rng = np.random.RandomState(seed)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        rng.shuffle(pairs)
        for i, j in pairs:
            if time.time() > deadline:
                return best_h, best_c5
            pair_golden(h, i, j, alpha)
    
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def solve_milp(n, timeout=120):
    """Solve binary MILP."""
    if n % 2 != 0:
        return None, None
    half = n // 2
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    dx = 2.0 / n
    
    constraints = [cp.sum(h) == half, t >= 0]
    
    for k in range(1, n):
        m = n - k
        z = cp.Variable(m)
        for i in range(m):
            constraints.append(z[i] <= h[i + k])
            constraints.append(z[i] <= h[i])
            constraints.append(z[i] >= h[i] + h[i + k] - 1)
            constraints.append(z[i] >= 0)
        
        corr_k = cp.sum(h[k:]) - cp.sum(z)
        constraints.append(t >= dx * corr_k)
        
        corr_neg = cp.sum(h[:m]) - cp.sum(z)
        constraints.append(t >= dx * corr_neg)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.CBC, time_limit=timeout)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, time_limit=timeout)
        except Exception:
            return None, None
    
    if h.value is None:
        return None, None
    
    h_val = np.round(np.array(h.value)).astype(int).astype(float)
    # Fix sum if needed
    s = h_val.sum()
    if s > half:
        idx = np.where(h_val == 1)[0]
        h_val[idx[:s-half]] = 0
    elif s < half:
        idx = np.where(h_val == 0)[0]
        h_val[idx[:half-s]] = 1
    
    c5 = compute_c5(h_val)
    return h_val, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Get initial solution
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.array([1.0 if i % 2 == 0 else 0.0 for i in range(48)])
    
    n = len(h_start)
    target = n / 2.0
    h_start = project_capped_simplex(h_start, target)
    
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: MILP for several n values
    milp_deadline = t0 + min(300, budget_s * 0.3)
    for n_try in [48, 50, 52, 54, 56, 60, 64]:
        if time.time() > milp_deadline:
            break
        remaining = max(30, milp_deadline - time.time())
        h_milp, c5_milp = solve_milp(n_try, timeout=remaining)
        if h_milp is not None and c5_milp < best_c5:
            best_c5 = c5_milp
            best_h = h_milp.copy()
            n = n_try
            target = n / 2.0
    
    # Phase 2: Pair coordinate descent from best solution
    pd_deadline = t0 + budget_s * 0.7
    n = len(best_h)
    target = n / 2.0
    for alpha in [10, 30, 80, 200, 500]:
        if time.time() > pd_deadline:
            break
        h_cur, c5_cur = pair_descent(best_h, alpha, 5, seed + alpha, pd_deadline)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
    
    # Phase 3: Random restarts
    rs_deadline = t0 + budget_s * 0.9
    for r in range(30):
        if time.time() > rs_deadline:
            break
        noise_scale = 0.15 - 0.1 * (r / 30.0)
        h_r = best_h + np.random.randn(n) * noise_scale
        h_r = project_capped_simplex(h_r, target)
        h_r, c5_r = pair_descent(h_r, 200, 3, seed + 1000 + r, rs_deadline)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Phase 4: Final polishing
    if time.time() < t0 + budget_s - 10:
        h_f, c5_f = pair_descent(best_h, 1000, 10, seed + 9999, t0 + budget_s - 5)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    
    return best_h, best_c5, len(best_h)

    Change: +h[i-k]
  
  h[j] appears when m = j (if j ≤ n-1-k): term h[j]*(1-h[j+k])
    Before: 0
    After: 1*(1-h[j+k])
    Change: +(1-h[j+k])
  h[j] appears when m+k = j (if j ≥ k): term h[j-k]*(1-h[j])
    Before: h[j-k]*(1-0) = h[j-k]
    After: h[j-k]*(1-1) = 0
    Change: -h[j+k]... wait, h[j-k]*(1-h[j]), before h[j]=0 so 1-h[j]=1, term = h[j-k]. After h[j]=1 so 1-h[j]=0, term = 0. Change = -h[j-k].
  
  So Δc[-k] = -(1-h[i+k]) + h[i-k] + (1-h[j+k]) - h[j-k]  [with bounds checks]

This is O(n) per swap (need to check all shifts k from 1 to n-1). With precomputation, it can be O(1) per shift.

For SA, the energy is C₅ = max_k |c[k]| * dx (actually just max_k c[k] * dx since c[k] ≥ 0 for our definition).

For the SA to work well, I need to efficiently track the maximum. One approach: maintain a max-heap or sorted list of correlation values. When a swap changes Δc[k] for various k, I update the affected values and recompute the max.

Since a swap affects c[k] for all k (from 1 to n-1), that's O(n) updates per swap. Each update is O(1) (just change the value and update the max tracking structure). So each SA step is O(n).

For n=48, that's very fast. We can do millions of SA steps in a few seconds.

def binary_sa(h, n_sweeps=100, t_start=1.0, t_end=0.01, seed=42, deadline=None):
    """Simulated annealing on binary h (n must be even, sum = n/2)."""
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.copy()
    
    # Ensure binary
    h = np.where(h > 0.5, 1.0, 0.0).astype(float)
    # Fix sum
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > n//2:
        for i in ones[:len(ones)-n//2]:
            h[i] = 0
    elif len(ones) < n//2:
        for i in zeros[:n//2-len(ones)]:
            h[i] = 1
    
    # Precompute correlations
    def compute_corrs(h):
        return np.correlate(h, 1.0 - h, mode='full')
    
    corrs = compute_corrs(h)
    dx = 2.0 / n
    best_c5 = float(np.max(corrs) * dx)
    best_h = h.copy()
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    temp = t_start
    cooling = (t_end / t_start) ** (1.0 / (n_sweeps * len(ones) * 2))
    
    for sweep in range(n_sweeps):
        if deadline and time.time() > deadline:
            break
        for _ in range(len(ones) * 2):
            if deadline and time.time() > deadline:
                break
            # Pick random swap
            i_idx = rng.randint(len(ones))
            j_idx = rng.randint(len(zeros))
            i = ones[i_idx]
            j = zeros[j_idx]
            
            # Compute delta in C5
            old_corrs = corrs.copy()  # This is expensive... O(n)
            
            # Apply swap
            h[i] = 0
            h[j] = 1
            new_corrs = compute_corrs(h)  # O(n^2) in numpy
            
            delta = float(np.max(new_corrs) - np.max(old_corrs)) * dx
            
            if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
                # Accept
                ones[i_idx] = j
                zeros[j_idx] = i
                corrs = new_corrs
                c5 = float(np.max(corrs) * dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            else:
                # Reject
                h[i] = 1
                h[j] = 0
            
            temp *= cooling
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_capped_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max(corr, alpha):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def pair_golden(h, i, j, alpha):
    n = len(h)
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-12:
        return
    oi, oj = h[i], h[j]
    
    def obj(a):
        h[i] = a
        h[j] = s - a
        corr = np.correlate(h, 1.0 - h, mode='full')
        return smooth_max(corr, alpha)
    
    f0 = obj(oi)
    a, b = lo, hi
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc, fd = obj(c), obj(d)
    for _ in range(28):
        if fc < fd:
            b, d, fd = d, c, fc
            c = b - gr * (b - a)
            fc = obj(c)
        else:
            a, c, fc = c, d, fd
            d = a + gr * (b - a)
            fd = obj(d)
    best_a = (a + b) / 2.0
    if obj(best_a) < f0:
        h[i], h[j] = best_a, s - best_a
    else:
        h[i], h[j] = oi, oj

def pair_descent(h, alpha, n_sweeps, seed, deadline):
    n = len(h)
    rng = np.random.RandomState(seed)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    bh = h.copy()
    bc5 = compute_c5(h)
    for _ in range(n_sweeps):
        if time.time() > deadline:
            break
        rng.shuffle(pairs)
        for i, j in pairs:
            if time.time() > deadline:
                return bh, bc5
            pair_golden(h, i, j, alpha)
    c5 = compute_c5(h)
    if c5 < bc5:
        bh, bc5 = h.copy(), c5
    return bh, bc5

    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    temp = t0_temp
    total_steps = n_sweeps * len(ones) * 2
    cool_rate = (0.001 / t0_temp) ** (1.0 / total_steps)
    
    for _ in range(n_sweeps):
        if time.time() > deadline:
            break
        for _ in range(len(ones) * 2):
            if time.time() > deadline:
                return best_h, best_c5
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            i, j = ones[ii], zeros[jj]
            
            h[i], h[j] = 0.0, 1.0
            new_corrs = np.correlate(h, 1.0 - h, mode='full')
            delta = float(np.max(new_corrs) - np.max(corrs)) * dx

                if c5 < best_c5:

            else:
                h[i], h[j] = 1.0, 0.0
            temp *= cool_rate
    
    return best_h, best_c5
