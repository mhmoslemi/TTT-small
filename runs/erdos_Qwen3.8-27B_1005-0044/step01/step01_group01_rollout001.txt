import numpy as np
from scipy.optimize import minimize
import time

_n_opt = 45
_alpha_opt = 50

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def smooth_max(x, alpha):
    x_max = np.max(x)
    shifted = alpha * (x - x_max)
    shifted = np.clip(shifted, -80, 80)
    return x_max + np.log(np.sum(np.exp(shifted))) / alpha

def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - np.sum(h)
    for _ in range(n * 3 + 10):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        count = float(np.sum(mask))
        per = diff / count
        if diff > 0:
            h[mask] += np.minimum(per, 1.0 - h[mask])
        else:
            h[mask] += np.maximum(per, -h[mask])
        diff = target - np.sum(h)
    return h

def obj_func(x):
    dx = 2.0 / _n_opt
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    return smooth_max(corr, _alpha_opt)

def sum_constraint(x):
    return np.sum(x) - _n_opt / 2.0

def do_optimize_slsqp(h_start, n, alpha, maxiter=300):
    global _n_opt, _alpha_opt
    _n_opt = n
    _alpha_opt = alpha
    try:
        res = minimize(
            obj_func, h_start.copy(), method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_constraint}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        h_new = np.clip(res.x, 0, 1)
        h_new = fix_sum(h_new, n)
        return (h_new, compute_c5(h_new))
    except Exception:
        return None

def binary_cd(h_bin, max_sweeps=200):
    """Coordinate descent on binary vector with exactly half ones."""
    n = len(h_bin)
    half = n // 2
    h = h_bin.copy()
    best_c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        improved = False
        ones = list(np.where(h > 0.5)[0])
        zeros = list(np.where(h < 0.5)[0])
        
        for i in ones:
            if i not in zeros and len(zeros) > 0:
                h[i] = 0.0
                zeros.append(i)
            for j in zeros:
                h[j] = 1.0
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    improved = True
                    ones.append(j)
                    zeros.remove(j)
                    if i in zeros:
                        zeros.remove(i)
                    if i in ones:
                        ones.remove(i)
                    break
                else:
                    h[j] = 0.0
                    if j in ones:
                        ones.remove(j)
                    if i in zeros:
                        zeros.remove(i)
                    zeros.append(i)
    
    return h, best_c5

def continuous_cd(h, n, sweeps=50, grid_pts=60):
    """Coordinate descent with continuous values."""
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for sweep in range(sweeps):
        sweep_improved = False
        for i in range(n):
            old_val = h[i]
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode="full") * dx
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode="full") * dx
            
            best_t = old_val
            best_val = np.max(c0 + old_val * (c1 - c0))
            
            ts = np.linspace(0, 1, grid_pts)
            for t in ts:
                val = np.max(c0 + t * (c1 - c0))
                if val < best_val - 1e-12:
                    best_val = val
                    best_t = t
            
            # Refine
            if 0.01 < best_t < 0.99:
                ts2 = np.linspace(max(0, best_t - 0.02), min(1, best_t + 0.02), 20)
                for t in ts2:
                    val = np.max(c0 + t * (c1 - c0))
                    if val < best_val - 1e-12:
                        best_val = val
                        best_t = t
            
            h[i] = best_t
        
        h = fix_sum(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            sweep_improved = True
        if not sweep_improved and sweep > 10:
            break
    
    return h, best_c5

def ga_search(n, pop_size=15, generations=25, seed=42):
    """GA for binary vectors."""
    if n % 2 != 0:
        return None, None
    rng = np.random.RandomState(seed)
    half = n // 2
    pop = []
    for _ in range(pop_size):
        h = np.zeros(n)
        idx = rng.choice(n, size=half, replace=False)
        h[idx] = 1.0
        pop.append(h)
    
    best_h, best_c5 = None, float('inf')
    
    for gen in range(generations):
        fits = [compute_c5(h) for h in pop]
        order = np.argsort(fits)
        pop = [pop[i] for i in order]
        fits = [fits[i] for i in order]
        if fits[0] < best_c5:
            best_c5 = fits[0]
            best_h = pop[0].copy()
        
        new_pop = [pop[i].copy() for i in range(min(pop_size // 2, len(pop)))]
        while len(new_pop) < pop_size:
            i = rng.randint(0, max(1, pop_size // 3))
            j = rng.randint(0, max(1, pop_size // 3))
            if i == j:
                j = (j + 1) % max(1, pop_size // 3)
            if i >= len(pop) or j >= len(pop):
                break
            child = 0.7 * pop[i] + 0.3 * pop[j]
            child += rng.normal(0, 0.08, n)
            child = np.clip(child, 0, 1)
            # Round to binary
            sorted_vals = np.sort(child)[::-1]
            threshold = sorted_vals[half - 1]
            child_bin = (child >= threshold).astype(float)
            ones_count = int(np.sum(child_bin))
            if ones_count > half:
                oi = np.where(child_bin == 1)[0]
                to_flip = rng.choice(oi, size=ones_count - half, replace=False)
                child_bin[to_flip] = 0
            elif ones_count < half:
                zi = np.where(child_bin == 0)[0]
                to_flip = rng.choice(zi, size=half - ones_count, replace=False)
                child_bin[to_flip] = 1
            new_pop.append(child_bin)
        pop = new_pop
    
    return best_h, best_c5
