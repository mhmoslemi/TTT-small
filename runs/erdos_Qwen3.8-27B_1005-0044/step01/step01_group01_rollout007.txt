        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        per = diff / float(np.sum(mask))
        if diff > 0:
            h[mask] += np.minimum(per, 1.0 - h[mask])
        else:
            h[mask] += np.maximum(per, -h[mask])
        diff = target - np.sum(h)
    return h

def smooth_max_obj(x):
    dx = 2.0 / _g_n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    alpha = _g_alpha
    x_max = np.max(corr)
    shifted = np.clip(alpha * (corr - x_max), -50, 50)
    return x_max + np.log(np.sum(np.exp(shifted))) / alpha

def sum_con(x):
    return np.sum(x) - _g_n / 2.0

def do_slsqp(h0, n, alpha, maxiter):
    global _g_n, _g_alpha
    _g_n = n
    _g_alpha = alpha
    try:
        res = minimize(smooth_max_obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_con}],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        h_new = fix_sum(res.x, n)
        return h_new, compute_c5(h_new)
    except Exception:
        return None

def targeted_local_search(h, n, n_iter, temp0, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    temp = temp0
    for it in range(n_iter):
        r = rng.random()
        new_h = cur_h.copy()
        if r < 0.35:
            i, j = rng.choice(n, 2, replace=False)
            new_h[i], new_h[j] = cur_h[j], cur_h[i]
        elif r < 0.65:
            i = rng.randint(n)
            j = rng.randint(n - 1)
            if j >= i:
                j += 1
            delta = rng.uniform(-0.15, 0.15)
            new_h[i] = np.clip(new_h[i] + delta, 0, 1)
            new_h[j] = np.clip(new_h[j] - delta, 0, 1)
            new_h = fix_sum(new_h, n)
        elif r < 0.85:
            i = rng.randint(n)
            w = rng.randint(2, max(3, n // 6))
            end = min(i + w, n)
            new_h[i:end] = np.clip(new_h[i:end] + rng.normal(0, 0.02, end - i), 0, 1)
            new_h = fix_sum(new_h, n)
        else:
            # Targeted: find max lag direction and perturb toward alignment
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            max_idx = np.argmax(corr)
            lag = max_idx - (n - 1)
            if lag > 0:
                mask = (cur_h > 0.3) & (np.roll(cur_h, -lag) < 0.7)
            else:
                mask = (cur_h < 0.7) & (np.roll(cur_h, -lag) > 0.3)
            idxs = np.where(mask)[0]
            if len(idxs) > 0:
                i = idxs[rng.randint(len(idxs))]
                j = rng.randint(n - 1)
                if j >= i:
                    j += 1
                if lag > 0:
                    delta = rng.uniform(0.02, 0.1)
                else:
                    delta = rng.uniform(-0.1, -0.02)
                new_h[i] = np.clip(new_h[i] + delta, 0, 1)
                new_h[j] = np.clip(new_h[j] - delta, 0, 1)
                new_h = fix_sum(new_h, n)
        new_c5 = compute_c5(new_h)
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / max(temp, 1e-12)):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        temp *= 0.9995
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    def elapsed():
        return time.time() - t0

    # Initialize from initial_h_values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        h0 = fix_sum(h0, len(h0))
        best_h = h0.copy()
        best_c5 = compute_c5(h0)

    if best_h is None:
        n = 45
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)

    n0 = len(best_h)

    # Phase 1: SLSQP continuation
    h_cur = best_h.copy()
    for alpha in [10, 20, 50, 100, 200, 500, 1000, 3000]:
        if elapsed() > budget_s * 0.12:
            break
        result = do_slsqp(h_cur, n0, alpha, 400)
        if result is not None:
            h_new, c5_new = result
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new

    # Phase 2: Intensive local search
    for restart in range(5):
        if elapsed() > budget_s * 0.35:
            break
        if restart == 0:
            h_ls = best_h.copy()
        else:
            h_ls = fix_sum(best_h + rng.normal(0, 0.02, n0), n0)
        h_ls, c5_ls = targeted_local_search(h_ls, n0, 5000, 0.003, rng)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 3: SLSQP polish after local search
    for alpha in [500, 1000, 3000, 5000]:
        if elapsed() > budget_s * 0.45:
            break
        result = do_slsqp(best_h, n0, alpha, 300)
        if result is not None and result[1] < best_c5:
            best_c5 = result[1]
            best_h = result[0].copy()

    # Phase 4: Try other n values
    n_tries = [44, 46, 50, 55, 60, 80, 100, 120, 160, 200]
    for n_try in n_tries:
        if elapsed() > budget_s * 0.78:
            break
        x_old = np.linspace(0, 2, n0, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, np.concatenate([best_h, best_h[:1]]), period=2)
        h_interp = fix_sum(h_interp, n_try)

        candidates = [h_interp]
        # Structured starts
        x = np.linspace(0, 1, n_try)
        for width in [0.15, 0.25, 0.4]:
            h_b = np.exp(-((x - 0.5) ** 2) / width)
            h_b = fix_sum(h_b, n_try)
            candidates.append(h_b)

        for h_cand in candidates:
            if elapsed() > budget_s * 0.78:
                break
            h_c = h_cand.copy()
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()
            for alpha in [100, 500]:
                if elapsed() > budget_s * 0.78:
                    break
                result = do_slsqp(h_c, n_try, alpha, 200)
                if result is not None:
                    h_c, c5_c = result
                    if c5_c < best_c5:
                        best_c5 = c5_c
                        best_h = h_c.copy()
            # Quick local search on this n
            if n_try <= 100:
                if elapsed() < budget_s * 0.78:
                    h_c, c5_c = targeted_local_search(h_c, n_try, 2000, 0.003, rng)
                    if c5_c < best_c5:
                        best_c5 = c5_c
                        best_h = h_c.copy()

    # Phase 5: Final polish
    n_best = len(best_h)
    for alpha in [1000, 3000, 5000, 10000]:
        if elapsed() > budget_s * 0.95:
            break
        result = do_slsqp(best_h, n_best, alpha, 500)
        if result is not None and result[1] < best_c5:
            best_c5 = result[1]
            best_h = result[0].copy()

    # Final local search polish
    if elapsed() < budget_s * 0.96:
        h_ls, c5_ls = targeted_local_search(best_h, n_best, 3000, 0.001, rng)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, len(best_h))
