import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - np.sum(h)
    for _ in range(100):
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        per = diff / max(1, np.sum(mask))
        if diff > 0:
            h[mask] = np.minimum(1.0, h[mask] + per)
        else:
            h[mask] = np.maximum(0.0, h[mask] + per)
        diff = target - np.sum(h)
    return h

def smooth_max(x, alpha):
    x_max = np.max(x)
    shifted = np.clip(alpha * (x - x_max), -700, 700)
    return x_max + np.log(np.sum(np.exp(shifted))) / alpha

_g_n = 0
_g_alpha = 50.0

def _obj(x):
    dx = 2.0 / _g_n
    corr = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max(corr * dx, _g_alpha)

def _sum_con(x):
    return np.sum(x) - _g_n / 2.0

def slsqp_optimize(h, n, alpha, maxiter=500):
    global _g_n, _g_alpha
    _g_n = n
    _g_alpha = alpha
    try:
        res = minimize(_obj, h, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': _sum_con}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return fix_sum(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def binary_ls(h, n, max_rounds=15):
    h = h.copy()
    best = compute_c5(h)
    for _ in range(max_rounds):
        imp = False
        for j in range(n):
            h[j] = 1.0 - h[j]
            c = compute_c5(h)
            if c < best - 1e-12:
                best = c
                imp = True
            else:
                h[j] = 1.0 - h[j]
        if not imp:
            break
    return h, best

def coord_polish(h, n, sweeps=2):
    h = h.copy()
    c5 = compute_c5(h)
    for _ in range(sweeps):
        imp = False
        for j in range(n):
            for d in (-0.3, -0.15, -0.05, 0.05, 0.15, 0.3):
                h2 = h.copy()
                h2[j] = np.clip(h2[j] + d, 0, 1)
                h2 = fix_sum(h2, n)
                c2 = compute_c5(h2)
                if c2 < c5 - 1e-12:
                    h = h2
                    c5 = c2
                    imp = True
                    break
        if not imp:
            break
    return h
