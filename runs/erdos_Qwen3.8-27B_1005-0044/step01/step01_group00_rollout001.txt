import numpy as np
import time
import scipy.optimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_capped_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_val(corr, alpha):
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

def grad_smooth_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            nc = n - k
            grad[k:] += wk * (1.0 - h[:nc])
            grad[:nc] -= wk * h[k:]
        else:
            kp = -k
            nc = n - kp
            grad[:nc] += wk * (1.0 - h[kp:])
            grad[kp:] -= wk * h[:nc]
    return grad * dx

def pgd_refine(h, target, alphas, max_iter, deadline):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for alpha in alphas:
        h = best_h.copy()
        step = 0.05 / alpha * 10  # scale step with alpha
        prev_val = smooth_max_val(np.correlate(h, 1.0 - h, mode='full'), alpha) * (2.0 / n)
        for it in range(max_iter):
            if time.time() > deadline:
                return best_h, best_c5
            g = grad_smooth_max(h, alpha)
            h_new = project_capped_simplex(h - step * g, target)
            val_new = smooth_max_val(np.correlate(h_new, 1.0 - h_new, mode='full'), alpha) * (2.0 / n)
            if val_new < prev_val:
                h = h_new
                prev_val = val_new
                step = min(step * 1.1, 0.5)
            else:
                step *= 0.5
                if step < 1e-12:
                    break
            c5 = compute_c5(h)
            if c5 < best_c5:

        if c5_s < best_c5:
            best_c5 = c5_s; best_h = h_s

    # Phase 7: Final pair swap polish
    d7 = t0 + budget_s - 5
    h7, c57 = pair_swap_search(best_h, target, 20000, d7, seed + 200)
    if c57 < best_c5:
        best_c5 = c57; best_h = h7

    # Phase 8: Final high-precision PGD
    d8 = t0 + budget_s - 1
    h8, c58 = nesterov_pgd(best_h, target, [5000.0, 10000.0], 500, d8, mu=0.2)
    if c58 < best_c5:
        best_c5 = c58; best_h = h8

    return best_h, best_c5, len(best_h)
