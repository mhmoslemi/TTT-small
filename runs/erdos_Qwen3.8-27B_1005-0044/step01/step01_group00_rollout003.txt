def push_towards_binary(h, target, strength):
    """Push h values towards 0 or 1 while maintaining the sum constraint."""
    # For each h[i] > 0.5, push towards 1
    # For each h[i] < 0.5, push towards 0
    h_new = h.copy()
    above = h_new > 0.5
    below = ~above
    h_new[above] += strength * (1.0 - h_new[above])
    h_new[below] -= strength * h_new[below]
    return project_capped_simplex(h_new, target)

def smooth_max(corr, alpha):

    return float(m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_capped_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_val(corr, alpha):
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

def grad_smooth_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            nc = n - k
            grad[k:] += wk * (1.0 - h[:nc])
            grad[:nc] -= wk * h[k:]
        else:
            kp = -k
            nc = n - kp
            grad[:nc] += wk * (1.0 - h[kp:])
            grad[kp:] -= wk * h[:nc]
    return grad * dx

        d3 = t0 + budget_s * 0.85

        # PGD polish with high alpha
        h_p, c5_p = pgd_refine(best_h, target_b, [200.0, 500.0, 1000.0], 2000, d3, step0=0.01)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_p

        # Directed binary search
        h_db, c5_db = directed_binary_search(best_h, 20000, d3, seed + 500)
        if c5_db < best_c5:
            best_c5, best_h = c5_db, h_db

        # Pair swap search
        h_ps, c5_ps = binary_pair_swap_search(best_h, 30000, d3, seed + 600)
        if c5_ps < best_c5:
            best_c5, best_h = c5_ps, h_ps

    # Phase 4: Final attempts with more diverse starts
    if best_h is not None:
        d4 = deadline_tight
        n_b = len(best_h)
        target_b = n_b / 2.0

        # Try a few more random binary starts at the best n
        for extra in range(3):
            if time.time() > d4 - 15:
                break
            s = seed + 1000 + extra * 13
            h_b = make_binary_start(n_b, s)
            h_d, c5_d = directed_binary_search(h_b, 6000, time.time() + 20, s)
            if c5_d < best_c5:
                best_c5, best_h = c5_d, h_d

        # Final pair swap polishing
        if time.time() < d4 - 5:
            h_fs, c5_fs = binary_pair_swap_search(best_h, 15000, d4, seed + 2000)
            if c5_fs < best_c5:
                best_c5, best_h = c5_fs, h_fs

    # Fallback
    if best_h is None:
        n = 50
        h = make_binary_start(n, seed)
        best_h = h
        best_c5 = compute_c5(h)

    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    best_h = project_capped_simplex(best_h, target_final)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
