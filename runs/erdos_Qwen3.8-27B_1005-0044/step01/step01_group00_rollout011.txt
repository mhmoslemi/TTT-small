```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_capped_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_of_c_at_i(h, i):
    n = len(h)
    grad = np.zeros(2 * n - 1)
    for k in range(n):
        m = n - 1 + k
        val = 0.0
        if i >= k:
            val += (1.0 - h[i - k])
        if i <= n - 1 - k:
            val -= h[i + k]
        grad[m] = val
    for r in range(1, n):
        m = n - 1 - r
        val = 0.0
        if i <= n - 1 - r:
            val += (1.0 - h[i + r])
        if i >= r:
            val -= h[i - r]
        grad[m] = val
    return grad

def smooth_max(corr, alpha):
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

def fista_smooth_max(h, target, alphas, max_iter, t0, deadline):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for alpha in alphas:
        h = best_h.copy()
        h_prev = h.copy()
        t_mom = 1.0
        step = 2.0 / (alpha * n * dx)  # Lipschitz estimate
        
        for it in range(max_iter):
            if time.time() > deadline:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            m = np.max(corr)
            exps = np.exp(alpha * (corr - m))
            w = exps / exps.sum()
            
            # Compute gradient: sum_m w[m] * dc[m]/dh
            # dc[m]/dh[i] for the full correlation
            grad = np.zeros(n)
            # For each m, add w[m] * dc[m]/dh
            # This is expensive if done naively. Let's use the structure.
            # dc[n-1+k]/dh[i] = (1-h[i-k])*[i>=k] - h[i+k]*[i<=n-1-k] for k>=0
            # dc[n-1-r]/dh[i] = (1-h[i+r])*[i<=n-1-r] - h[i-r]*[i>=r] for r>0
            
            # Alternative: compute gradient via the adjoint
            # grad[i] = sum_k w[n-1+k] * ((1-h[i-k])*[i>=k] - h[i+k]*[i<=n-1-k])
            #          + sum_r w[n-1-r] * ((1-h[i+r])*[i<=n-1-r] - h[i-r])*[i>=r])
            
            # This can be computed in O(n²) total using convolutions
            # But for n=47, O(n²) = 2209 is fine
            
            for i in range(n):
                gi = 0.0
                for k in range(n):
                    if i >= k:
                        gi += w[n-1+k] * (1.0 - h[i-k])
                    if i <= n-1-k:
                        gi -= w[n-1+k] * h[i+k]
                for r in range(1, n):
                    if i <= n-1-r:
                        gi += w[n-1-r] * (1.0 - h[i+r])
                    if i >= r:
                        gi -= w[n-1-r] * h[i-r]
                grad[i] = gi
            grad *= dx
            
            # FISTA update
            h_new = project_capped_simplex(h_prev - step * grad, target)
            
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
            h_prev = h_new + (t_mom - 1.0) / t_new * (h_new - h)
            h = h_new
            t_mom = t_new
            
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5
```

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_capped_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_of_c_at_i(h, i):
    n = len(h)
    grad = np.zeros(2 * n - 1)
    for k in range(n):
        m = n - 1 + k
        val = 0.0
        if i >= k:
            val += (1.0 - h[i - k])
        if i + k < n:
            val -= h[i + k]
        grad[m] = val
    for r in range(1, n):
        m = n - 1 - r
        val = 0.0
        if i + r < n:
            val += (1.0 - h[i + r])
        if i >= r:
            val -= h[i - r]
        grad[m] = val
    return grad

def smooth_max_val(corr, alpha):
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

def fista_refine(h, target, alpha, max_iter, t0, deadline, step_scale=1.0):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    h_prev = h.copy()
    t_mom = 1.0
    step = step_scale * 0.5 / (alpha * dx)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        m = np.max(corr)
        exps = np.exp(alpha * (corr - m))
        w = exps / exps.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for i in range(n):
            gi = 0.0
            for k in range(n):
                if i >= k:
                    gi += w[n-1+k] * (1.0 - h_curr[i-k])
                if i + k < n:
                    gi -= w[n-1+k] * h_curr[i+k]
            for r in range(1, n):
                if i + r < n:
                    gi += w[n-1-r] * (1.0 - h_curr[i+r])
                if i >= r:
                    gi -= w[n-1-r] * h_curr[i-r]
            grad[i] = gi * dx
        
        h_new = project_capped_simplex(h_prev - step * grad, target)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        h_prev = h_new + (t_mom - 1.0) / t_new * (h_new - h_curr)
        h_curr = h_new
        t_mom = t_new
        
        c5 = compute_c5(h_curr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_curr.copy()
    
    return best_h, best_c5

def pairwise_cd(h, target, n_sweeps, t0, deadline, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        any_improved = False
        
        for idx in range(n):
            i = order[idx]
            j = (i + 1 + rng.randint(n - 1)) % n
            if i == j:
                j = (i + 1) % n
            s = h[i] + h[j]
            lb = max(0.0, s - 1.0)
            ub = min(1.0, s)
            if ub - lb < 1e-12:
                continue
            
            v0 = h[i]
            c = np.correlate(h, 1.0 - h, mode='full')
            gi = grad_of_c_at_i(h, i)
            gj = grad_of_c_at_i(h, j)
            D = gi - gj
            
            # Solve 1D minimax
            def obj1d(v):
                return np.max(c + (v - v0) * D)
            
            res = minimize_scalar(obj1d, bounds=(lb, ub), method='bounded',
                                  options={'xatol': 1e-10})
            v_opt = res.x
            
            if abs(v_opt - v0) < 1e-10:
                continue
            
            # Try the full step
            h[i] = v_opt
            h[j] = s - v_opt
            c5_new = compute_c5(h)
            
            if c5_new <= best_c5 + 1e-12:
                any_improved = True
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                # Revert and try half step
                h[i] = v0 + 0.5 * (v_opt - v0)
                h[j] = s - h[i]
                c5_half = compute_c5(h)
                if c5_half < best_c5:
                    best_c5 = c5_half
                    best_h = h.copy()
                    any_improved = True
                else:
                    # Revert
                    h[i] = v0
                    h[j] = s - v0
        
        if not any_improved:
            break
    
    return best_h, best_c5

def make_block_init(n, n_blocks, seed):

    block_size = n // n_blocks

        end = start + block_size if b < n_blocks - 1 else n
        h[start:end] = rng.choice([0.0, 1.0])
    return project_capped_simplex(h, n / 2.0)
