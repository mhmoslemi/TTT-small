```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_capped_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_val(corr, alpha):
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            nc = n - k
            grad[k:] += wk * (1.0 - h[:nc])
            grad[:nc] -= wk * h[k:]
        else:
            kp = -k
            nc = n - kp
            grad[:nc] += wk * (1.0 - h[kp:])
            grad[kp:] -= wk * h[:nc]
    return grad * dx

def pgd_refine(h, target, alphas, max_iter, deadline):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for alpha in alphas:
        h = best_h.copy()
        step = 0.02
        stall = 0
        for it in range(max_iter):
            if time.time() > deadline:
                break
            g = smooth_grad(h, alpha)
            h_new = project_capped_simplex(h - step * g, target)
            corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
            val_new = smooth_max_val(corr_new, alpha) * (2.0 / n)
            corr_old = np.correlate(h, 1.0 - h, mode='full')
            val_old = smooth_max_val(corr_old, alpha) * (2.0 / n)
            if val_new < val_old:
                h = h_new
                step = min(step * 1.06, 0.25)
                stall = 0
            else:
                step *= 0.5
                stall += 1
                if step < 1e-10 or stall > 150:
                    break
            c5 = float(np.max(corr_new) * (2.0 / n))
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5

def pair_descent(h, target, deadline, seed=42):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    delta = 0.04
    no_imp = 0
    dvals_coarse = np.concatenate([np.linspace(-0.08, -0.01, 4), np.linspace(0.01, 0.08, 4)])
    dvals_fine = np.linspace(-0.015, 0.015, 7)
    for sweep in range(60000):
        if time.time() > deadline:
            break
        rng.shuffle(pairs)
        improved = False
        for idx in range(len(pairs)):
            if time.time() > deadline:
                break
            i, j = pairs[idx]
            a, b = h[i], h[j]
            best_d = 0.0
            best_local = best_c5
            for d in dvals_coarse:
                na, nb = a + d, b - d
                if na < 0 or na > 1 or nb < 0 or nb > 1:
                    continue
                h[i], h[j] = na, nb
                c5 = compute_c5(h)
                if c5 < best_local:
                    best_local = c5
                    best_d = d
            if best_d != 0.0:
                h[i], h[j] = a + best_d, b - best_d
                for d2 in dvals_fine:
                    dd = best_d + d2
                    na, nb = a + dd, b - dd
                    if na < 0 or na > 1 or nb < 0 or nb > 1:
                        continue
                    h[i], h[j] = na, nb
                    c5 = compute_c5(h)
                    if c5 < best_local:
                        best_local = c5
                improved = True
            if best_local < best_c5 - 1e-15:
                best_c5 = best_local
                best_h = h.copy()
        if not improved:
            no_imp += 1
            if no_imp >= 4:
                delta *= 0.75
                if delta < 0.005:
                    h = project_capped_simplex(h + rng.randn(n) * 0.01, target)
                    delta = 0.04
                    no_imp = 0
        else:
            no_imp = 0
    return best_h, best_c5

def sa_coord(h, target, n_iter, deadline, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    temp = 0.012
    for it in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.choice([1, -1]) * rng.uniform(0.004, 0.04)
        na, nb = h[i] + d, h[j] - d
        if na < 0 or na > 1 or nb < 0 or nb > 1:
            continue
        h[i], h[j] = na, nb
        c5_new = compute_c5(h)
        if c5_new <= best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        else:
            if rng.random() >= np.exp(-(c5_new - best_c5) / max(temp, 1e-10)):
                h[i], h[j] = na - d, nb + d
        if it % 500 == 0 and it > 0:
            temp = max(temp * 0.91, 0.0008)
    return best_h, best_c5
```
