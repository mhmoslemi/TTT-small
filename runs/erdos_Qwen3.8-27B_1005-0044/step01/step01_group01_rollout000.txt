import numpy as np
from scipy.optimize import minimize
import time

_opt_n = 45
_opt_alpha = 50

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def compute_c5_vec(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def smooth_max(x, alpha):
    xm = np.max(x)
    sh = np.clip(alpha * (x - xm), -80, 80)
    return xm + np.log(np.sum(np.exp(sh))) / alpha

def obj_func(x):
    dx = 2.0 / _opt_n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    return smooth_max(corr, _opt_alpha)

def sum_constraint(x):
    return np.sum(x) - _opt_n / 2.0

def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - np.sum(h)
    for _ in range(3 * n + 10):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        count = float(np.sum(mask))
        per = diff / count
        if diff > 0:
            h[mask] += np.minimum(per, 1.0 - h[mask])
        else:
            h[mask] += np.maximum(per, -h[mask])
        diff = target - np.sum(h)
    return h

def project_capped(v, target):
    """Project onto {x: 0<=x_i<=1, sum(x)=target} via bisection on lambda"""
    lo = -v.max() - 2.0
    hi = 1.0 - v.min() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def subgradient_of_active(h, k):
    """Gradient of the k-th correlation (shift k relative to center) w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    idx = np.arange(n)
    if k >= 0:
        # corr[n-1+k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        # d/dh[j]: (1-h[j-k]) for j in [k, n-1]; -h[j+k] for j in [0, n-1-k]
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        kk = -k
        # corr[n-1-kk] = sum_{i=0}^{n-1-kk} h[i]*(1-h[i+kk])
        # d/dh[j]: (1-h[j+kk]) for j in [0, n-1-kk]; -h[j-kk] for j in [kk, n-1]
        g[:n - kk] += (1.0 - h[kk:])
        g[kk:] -= h[:n - kk]
    return g

def do_subgradient(h, n, budget_frac, start_time, budget_s):
    """Momentum subgradient for the active correlation"""
    deadline = start_time + budget_s * budget_frac
    momentum = 0.9
    step_size = 0.15
    vel = np.zeros(n)
    best_local = h.copy()
    best_c5_local = compute_c5(h)
    h_cur = h.copy()
    
    for it in range(20000):
        if time.time() > deadline:
            break
        c5v = compute_c5_vec(h_cur)
        k_active = int(np.argmax(c5v)) - (n - 1)
        g = subgradient_of_active(h_cur, k_active)
        vel = momentum * vel - step_size * g
        h_new = project_capped(h_cur + vel, n / 2.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5_local:
            best_c5_local = c5_new
            best_local = h_new.copy()
        h_cur = h_new
        step_size = max(0.005, step_size * 0.9995)
        if it % 5000 == 0 and it > 0:
            step_size = 0.05
            momentum = 0.85
    return best_local, best_c5_local

def do_slsqp(h, n, alpha, maxiter):
    global _opt_n, _opt_alpha
    _opt_n = n
    _opt_alpha = alpha
    try:
        res = minimize(
            obj_func, h, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': sum_constraint}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        h_new = fix_sum(np.clip(res.x, 0, 1), n)
        c5 = compute_c5(h_new)
        return h_new, c5
    except Exception:
        return None

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')

    # Load initial
    h_main = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_main = np.array(initial_h_values, dtype=float)
        h_main = fix_sum(h_main, len(h_main))
        c5_init = compute_c5(h_main)
        best_h = h_main.copy()
        best_c5 = c5_init

    if h_main is None:
        h_main = np.full(45, 0.5)
        best_h = h_main.copy()
        best_c5 = compute_c5(h_main)

    n_main = len(h_main)

    # Phase 1: Subgradient from initial (30% budget)
    h_sg, c5_sg = do_subgradient(h_main, n_main, 0.30, start_time, budget_s)
    if c5_sg < best_c5:
        best_c5 = c5_sg
        best_h = h_sg

    # Phase 2: SLSQP continuation (30% budget)
    h_cur = best_h.copy()
    n_cur = len(h_cur)
    alphas = [5, 10, 20, 50, 100, 200, 500, 1000]
    for alpha in alphas:
        if time.time() - start_time > budget_s * 0.60:
            break
        result = do_slsqp(h_cur, n_cur, alpha, 400)
        if result is not None:
            h_new, c5_new = result
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new

    # Phase 3: More subgradient from current best (10% budget)
    h_sg2, c5_sg2 = do_subgradient(best_h, len(best_h), 0.10, start_time + (time.time() - start_time), budget_s)
    if c5_sg2 < best_c5:
        best_c5 = c5_sg2
        best_h = h_sg2

    # Phase 4: Try different n values (25% budget)
    n_candidates = [55, 65, 75, 44, 46, 36, 100, 120]
    for n_try in n_candidates:
        if time.time() - start_time > budget_s * 0.90:
            break
        # Start from perturbed version of best
        h_try = np.zeros(n_try)
        # Interpolate best_h to n_try points
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = fix_sum(h_try, n_try)
        
        # Also try binary start
        h_bin = np.zeros(n_try)
        h_bin[:n_try // 2] = 1.0
        h_bin[n_try // 2] = (n_try % 2) * 0.5  # handle odd
        h_bin = fix_sum(np.random.permutation(h_bin), n_try)
        
        h_start = h_try if compute_c5(h_try) < compute_c5(h_bin) else h_bin
        c5_start = compute_c5(h_start)
        if c5_start < best_c5:
            best_c5 = c5_start
            best_h = h_start.copy()

        # SLSQP on this n
        for alpha in [20, 80, 300, 800]:
            if time.time() - start_time > budget_s * 0.92:
                break
            result = do_slsqp(h_start, n_try, alpha, 300)
            if result is not None:
                h_new, c5_new = result
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_start = h_new

    # Phase 5: Final intensive polish (10% budget)
    h_final = best_h.copy()
    n_final = len(h_final)
    for alpha in [500, 1500, 3000, 8000]:
        if time.time() - start_time > budget_s * 0.98:
            break
        result = do_slsqp(h_final, n_final, alpha, 500)
        if result is not None:
            h_new, c5_new = result
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
            h_final = h_new

    # Final validation
    if best_h is None:
        best_h = fix_sum(np.full(45, 0.5), 45)
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, len(best_h))
