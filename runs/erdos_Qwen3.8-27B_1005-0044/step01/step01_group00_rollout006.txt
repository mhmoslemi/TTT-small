```python
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_capped_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def solve_qp(h_init, n, target, weights, solver='ECOS'):
    h = cp.Variable(n)
    exprs = []
    for k_idx in range(2 * n - 1):
        offset = k_idx - (n - 1)
        if offset >= 0:
            nc = n - offset
            exprs.append(cp.sum(h[offset:] * (1 - h[:nc])))
        else:
            kp = -offset
            nc = n - kp
            exprs.append(cp.sum(h[:nc] * (1 - h[kp:])))
    obj = cp.Minimize(cp.sum(cp.multiply(weights, cp.hstack(exprs))))
    constraints = [h >= 0, h <= 1, cp.sum(h) == target]
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=solver)
        if h.value is not None:
            return np.clip(h.value, 0, 1)
    except:
        pass
    return h_init

def irls_qp(h_init, n, target, deadline, max_rounds=50):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break

        # Exponential weights
        alpha = 50.0
        m = np.max(c)
        w = np.exp(alpha * (c - m))
        w = w / w.sum()

        if h_new is None:
            break
        c5_new = compute_c5(h_new)

            h_new = solve_qp(h, n, target, w, solver='ECOS')
```

```python
def smooth_max(corr, alpha):
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha)

def objective_and_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = smooth_max(corr, alpha) * dx
    # gradient
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            nc = n - k
            grad[k:] += wk * (1.0 - h[:nc])
            grad[:nc] -= wk * h[k:]
        else:
            kp = -k
            nc = n - kp
            grad[:nc] += wk * (1.0 - h[kp:])
            grad[kp:] -= wk * h[:nc]
    return val, grad * dx

def scipy_optimize(h_init, n, target, alpha, max_iter=1000):
    # Use trust-constr with equality constraint

    constraint = LinearConstraint(np.ones(n), target, target)

        return v
```

Even though the full QP is non-convex, I can linearize it:
At current h*, the linearization of c_k at h* is:
c_k ≈ c_k(h*) + grad_k(h*) · (h - h*)

So minimizing sum_k w_k * c_k becomes:
min sum_k w_k * (c_k(h*) + grad_k(h*) · (h - h*))
= constant + sum_k w_k * grad_k(h*) · h
= constant + (sum_k w_k * grad_k(h*)) · h

This is a LINEAR program! And LPs can be solved exactly and efficiently.

So the algorithm is:
1. At current h*, compute all c_k and gradients grad_k
2. Set weights w_k (e.g., exponential in c_k)
3. Solve LP: min (sum_k w_k * grad_k) · h subject to 0 <= h <= 1, sum(h) = n/2
4. Take a step towards the LP solution (or use it directly if it improves)
5. Repeat

This is essentially the linearized version of IRLS, and it should work! The LP can be solved with CVXPY or scipy.

```python
def lp_step(h, target, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w = w / w.sum()
    
    # Compute weighted gradient
    grad_w = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            nc = n - k
            grad_w[k:] += wk * (1.0 - h[:nc])
            grad_w[:nc] -= wk * h[k:]
        else:
            kp = -k
            nc = n - kp
            grad_w[:nc] += wk * (1.0 - h[kp:])
            grad_w[kp:] -= wk * h[:nc]
    
    # Solve LP: min grad_w · h subject to 0 <= h <= 1, sum(h) = target
    # This is a transportation problem! The optimal solution is to put
    # mass on the n/2 elements with the smallest grad_w values (set to 1)
    # and the rest to 0. But we need sum = target = n/2.
    # So: set h[i] = 1 for the n/2 indices with smallest grad_w[i],
    # and h[i] = 0 for the rest. If there are ties at the boundary,
    # we might need fractional values.
    
    # Actually, this is the classic "pick the largest n/2" problem.
    # Sort by grad_w ascending, pick top n/2.
    order = np.argsort(grad_w)
    h_lp = np.zeros(n)
    h_lp[order[:int(target)]] = 1.0
    # Handle if target is not integer
    if target != int(target):
        # Fractional assignment
        h_lp[order[:int(target)]] = 1.0
        h_lp[order[int(target)]] = target - int(target)
    
    return h_lp
```

```python
def project_capped_simplex(h, target):

        else:
```

```python
            idx = rng.choice(n, size=p, replace=False)
            h_pert = h.copy()
            h_pert[idx] += rng.randn(p) * temp * 0.5
            h_pert = project_capped_simplex(h_pert, target)
        c5_pert = compute_c5(h_pert)
        if c5_pert <= best_c5 + 1e-12:
            h = h_pert
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h.copy()
        else:
            accept_prob = np.exp(-(c5_pert - best_c5) / max(temp, 1e-10))
            if rng.random() < accept_prob:
                h = h_pert
        if it % 150 == 0 and it > 0:
            temp = max(temp * 0.92, 0.003)
    return best_h, best_c5

def multi_scale_ls(h, target, deadline, seed):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for temp_scale in [0.08, 0.05, 0.03, 0.015, 0.008]:
        if time.time() > deadline:
            break
        h_r, c5_r = local_search(best_h, target, 8000, deadline, seed, base_temp=temp_scale)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r
        seed += 1000
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline_end = t0 + budget_s - 3

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(47, 0.5)

    n = len(h0)
    target = n / 2.0
    h0 = project_capped_simplex(h0, target)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: PGD from initial
    d1 = min(t0 + 60, deadline_end)
    h1, c5_1 = pgd_refine(h0, target, [5, 10, 25, 50, 100, 300, 800], 3000, d1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: LP-mixed starts
    d2 = min(t0 + 140, deadline_end)
    for alpha in [20, 80, 200, 500]:
        if time.time() > d2:
            break
        h_lp = lp_01_solution(best_h, alpha)
        for tf in [0.25, 0.5, 0.75]:
            if time.time() > d2:
                break
            h_mix = (1.0 - tf) * best_h + tf * h_lp
            h_mix = project_capped_simplex(h_mix, target)
            h_r, c5_r = pgd_refine(h_mix, target, [50, 200, 800], 800, d2)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r

    # Phase 3: Random restart at same n
    d3 = min(t0 + 200, deadline_end)
    rng = np.random.RandomState(seed + 7)
    for rs in range(3):
        if time.time() > d3:
            break
        h_rand = rng.rand(n)
        h_rand = project_capped_simplex(h_rand, target)
        h_r, c5_r = pgd_refine(h_rand, target, [10, 50, 200, 500], 1500, d3)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r

    # Phase 4: Intensive multi-scale local search
    d4 = min(t0 + 500, deadline_end)
    h4, c5_4 = multi_scale_ls(best_h, target, d4, seed)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4

    # Phase 5: Try other n values
    d5 = min(t0 + budget_s - 80, deadline_end)
    for n_try in [50, 53, 56, 60, 64, 72, 80, 96]:
        if time.time() > d5:
            break
        t_sub = n_try / 2.0
        # Resample from best solution
        h_try = np.interp(
            np.linspace(0, n - 1, n_try),
            np.linspace(0, n - 1, n),
            best_h
        )
        h_try = project_capped_simplex(h_try, t_sub)
        h_r, c5_r = pgd_refine(h_try, t_sub, [10, 50, 200, 500], 1500, d5)
        # Also try a fresh random start at this n
        h_rand = np.random.RandomState(seed + n_try).rand(n_try)
        h_rand = project_capped_simplex(h_rand, t_sub)
        h_r2, c5_r2 = pgd_refine(h_rand, t_sub, [10, 50, 200], 800, d5)
        c5_cand = min(c5_r, c5_r2)
        h_cand = h_r if c5_r <= c5_r2 else h_r2
        if c5_cand < best_c5 and time.time() < d5 - 5:
            h_ls, c5_ls = local_search(h_cand, t_sub, 5000, d5, seed + n_try)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls
                n = n_try
            elif c5_cand < best_c5:
                best_c5 = c5_cand
                best_h = h_cand
                n = n_try
        elif c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand
            n = n_try

    # Phase 6: Final polishing
    nf = len(best_h)
    tf = nf / 2.0
    h_f, c5_f = pgd_refine(best_h, tf, [500, 1000, 2000, 5000], 3000, deadline_end)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f
    if time.time() < deadline_end - 2:
        hf2, c5f2 = local_search(best_h, tf, 10000, deadline_end, seed + 4242, base_temp=0.01)
        if c5f2 < best_c5:
            best_c5 = c5f2
            best_h = hf2

    return best_h, best_c5, len(best_h)
```
