def project_feasible(x,n):
    x=np.asarray(x,float).reshape(n).copy()
    tgt=0.5*n
    # If extreme, pull into [0,1]
    x=np.clip(x,0.0,1.0)
    # solve shift
    lo=-1.0 - float(np.max(x)) if n else 0.0  # Actually if all x=1, need tau=-1 to zeros; lo=-max(x)-1e-9
    hi=1.0 - float(np.min(x)) + 1e-9
    # But if target between 0 and n, shift range enough: from all 0 at tau=-max(x), all1 at tau=1-min(x)
    lo = -float(np.max(x)) - 1e-9
    hi = 1.0 - float(np.min(x)) + 1e-9
    for _ in range(60):
       mid=0.5*(lo+hi)
       s=float(np.clip(x+mid,0.0,1.0).sum())
       if s < tgt:
          lo=mid
       else:
          hi=mid
    y=np.clip(x+0.5*(lo+hi),0.0,1.0)
    # residual distribute
    for _ in range(10):
       diff=tgt-float(np.sum(y))
       if abs(diff)<1e-12: break
       mask=(y>1e-12)&(y<1.0-1e-12)
       cnt=int(np.sum(mask))
       if cnt:
           step=diff/cnt
           # limit
           if step>0:
              maxstep=float(np.min(1.0-y[mask]))
              if step>maxstep: step=maxstep
           else:
              minstep=float(np.min(y[mask]))
              if step< -minstep: step=-minstep
           y[mask]+=step
       else:
           # adjust a coordinate with enough slack
           if diff>0:
              idx=int(np.argmax(1.0-y)); y[idx]=np.clip(y[idx]+diff,0.0,1.0)
           else:
              idx=int(np.argmin(y)); y[idx]=np.clip(y[idx]+diff,0.0,1.0)

def project_feasible(x, n):
    x = np.asarray(x, dtype=float).reshape(int(n)).copy()
    tgt = 0.5 * float(n)
    x = np.clip(x, 0.0, 1.0)

    if n == 0:
        return x

    mx = float(np.max(x))
    mn = float(np.min(x))
    lo = -mx - 1e-9
    hi = 1.0 - mn + 1e-9

    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x + mid, 0.0, 1.0).sum())
        if s < tgt:
            lo = mid
        else:
            hi = mid

    y = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)

                if step > maxstep:
                    step = maxstep
            else:
                minstep = float(np.min(y[mask]))
                if step < -minstep:
                    step = -minstep
            y[mask] += step
        else:
            if diff > 0.0:
                idx = int(np.argmax(1.0 - y))

            else:
                idx = int(np.argmin(y))

def random_composition(total, k, rng):
    total = int(round(float(total)))
    k = int(round(float(k)))
    if k <= 0 or total < k:
        return None
    arr = np.ones(k, dtype=int)
    rem = total - k
    if rem == 0:
        return arr
    w = rng.rand(k)
    w += 1e-12
    wsum = float(w.sum())
    inc = np.floor(w * float(rem) / wsum).astype(int)
    diff = rem - int(inc.sum())
    while diff > 0:
        idx = int(rng.randint(k))
        inc[idx] += 1
        diff -= 1
    while diff < 0:
        idx = int(rng.randint(k))
        if inc[idx] > 0:
            inc[idx] -= 1
            diff += 1
    arr += inc
    return arr

def grad_weighted(h, w):
    n = int(h.size)
    g = np.zeros(n, dtype=float)
    base = n - 1

    w0 = float(w[base])
    if abs(w0) > 1e-300:
        g += w0 * (1.0 - 2.0 * h)

    for d in range(1, n):
        wp = float(w[base + d])
        wn = float(w[base - d])
        ws = wp + wn
        if abs(ws) <= 1e-300 and abs(wp) <= 1e-300 and abs(wn) <= 1e-300:
            continue
        m = n - d
        if m <= 0:
            continue
        if abs(wp) > 1e-300:
            g[:m] += wp
        if abs(wn) > 1e-300:
            g[d:] += wn
        if abs(ws) > 1e-300:
            g[:m] -= ws * h[d:]
            g[d:] -= ws * h[:m]

    return g


def soft_value_grad(h, beta, give_grad=True):
    h = np.asarray(h, dtype=float).ravel()
    c = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(c))
    beta = max(1.0, float(beta))
    diff = beta * (c - m)
    diff = np.clip(diff, -500.0, 500.0)
    e = np.exp(diff)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros_like(c, dtype=float)
        maxmask = (c == m)
        cnt = int(maxmask.sum())
        if cnt == 0:
            w[base] = 1.0
        else:
            w[maxmask] = 1.0 / float(cnt)
    else:
        w = e / s

    val = (m + float(np.log(s)) / beta) * _G_DX
    if not give_grad:
        return val
    grad = grad_weighted(h, w) * _G_DX
    return val, grad

    return soft_value_grad(h, _G_BETA, True)


def slsqp_polish(h, n, beta_schedule, maxiter):
    global _G_BETA
    set_n(n)
    h = project_feasible(h, n).copy()
    best = compute_c5(h)
    best_h = h.copy()

    bounds = [(0.0, 1.0)] * int(n)
    con = {"type": "eq", "fun": constraint_sum}

    for beta in beta_schedule:
        if time_left() < 20.0:
            break
        _G_BETA = float(beta)
        mi = int(maxiter)
        if n > 384:
            mi = min(mi, 45)
        else:
            mi = min(mi, 90)
        try:
            res = minimize(
                smooth_fn_grad,
                h,
                method="SLSQP",
                bounds=bounds,
                constraints=con,
                jac=True,
                options={"maxiter": mi, "ftol": 1e-12, "iprint": 0}
            )
            h2 = project_feasible(res.x, n)
            c2 = compute_c5(h2)
            if c2 < best - 1e-14:
                best = c2
                best_h = h2.copy()
                h = h2.copy()
        except Exception:
            pass

    best_h = project_feasible(best_h, n)
    best = compute_c5(best_h)
    return best_h, best


def coord_candidates(old, fine=False):
    vals = [old, 0.0, 1.0]
    if fine:
        deltas = (
            1e-4, 2e-4, 5e-4, 1e-3, 2e-3, 5e-3,
            0.01, 0.02, 0.05, 0.1, 0.2, 0.5
        )
    else:
        deltas = (
            5e-4, 1e-3, 2e-3, 5e-3, 0.01, 0.05, 0.1, 0.2, 0.5
        )
    for d in deltas:
        vals.append(old - d)
        vals.append(old + d)
    if 0.02 < old < 0.98:
        vals.append(0.5)

    out = []
    for v in vals:
        try:
            v = float(v)
        except Exception:
            continue
        if not np.isfinite(v):
            continue
        if v < 0.0:
            v = 0.0
        elif v > 1.0:
            v = 1.0
        dup = False
        for u in out:
            if abs(u - v) <= 1e-13:
                dup = True
                break
        if not dup:
            out.append(v)
    return out


def coordinate_descent(h, n, max_sweeps, rng, fine=False):
    n = int(n)
    h = project_feasible(h, n).copy()
    best = compute_c5(h)
    best_h = h.copy()
    max_sweeps = int(max_sweeps)

    stalled = 0
    for sweep in range(max_sweeps):
        if time_left() < 5.0 or call_time_left() < 2.0:
            break

        improved = False
        order = rng.permutation(n)
        for i in order:
            if time_left() < 3.0 or call_time_left() < 1.0:
                break

            base = best_h
            old = float(base[i])
            b_c = best
            best_trial = None

            for t in coord_candidates(old, fine=fine):
                if t == old:
                    continue
                trial = base.copy()
                trial[i] = t
                trial = fast_project(trial, n)
                c = compute_c5(trial)
                if c < b_c - 1e-13:
                    b_c = c
                    best_trial = trial

            if (fine or sweep < 2) and rng.rand() < 0.45:
                j = int(rng.randint(n))
                if j != i:
                    for delta in (5e-4, 1e-3, 5e-3, 1e-2, 5e-2, 1e-1):
                        for sgn in (1.0, -1.0):
                            vi = old + sgn * delta
                            vj = float(base[j]) - sgn * delta
                            if vi < 0.0 or vi > 1.0 or vj < 0.0 or vj > 1.0:
                                continue
                            trial = base.copy()
                            trial[i] = vi
                            trial[j] = vj
                            trial = fast_project(trial, n)
                            c = compute_c5(trial)
                            if c < b_c - 1e-13:
                                b_c = c
                                best_trial = trial

            if best_trial is not None:
                best_h = best_trial
                best = b_c
                improved = True

        if improved:
            stalled = 0
        else:
            stalled += 1
            if stalled >= 2:
                break

    best_h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h)


def cand_key(item):
    return item[0]


def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET, _G_N, _G_TGT, _G_DX, _G_BETA, _G_CALL_END

    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    _G_CALL_END = time.time() + 10.0
    rng = np.random.RandomState(int(seed))

    parent = None
    parent_n = 0
    try:
        parent = np.asarray(initial_h_values, dtype=float).copy()
        parent_n = int(parent.size)
        if parent_n > 0:
            parent = project_feasible(parent, parent_n)
        else:
            parent = None
    except Exception:
        parent = None
        parent_n = 0

    if parent is None:
        parent_n = 256
        parent = make_random_uniform(rng, parent_n)
        parent = project_feasible(parent, parent_n)

    best_h = parent.copy()
    best_n = parent_n
    best_c = compute_c5(best_h)

    candidates = [(best_c, best_n, best_h.copy())]

    # 1. Rapid resolution scan around the parent and broader integer grid.
    nset = set()
    if parent_n > 0:
        for d in range(-28, 29):
            nset.add(parent_n + d)
    for n in range(144, 369):
        nset.add(n)
    for n in (128, 160, 192, 224, 256, 288, 320, 384, 512):
        nset.add(n)

    nl = sorted([n for n in nset if 96 <= n <= 512])
    for n in nl:
        if time_left() < 15.0:
            break
        h = project_feasible(interp_to(parent, parent_n, n), n)
        c = compute_c5(h)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            best_n = n
        candidates.append((c, n, h.copy()))

    candidates.sort(key=cand_key)
    if len(candidates) > 45:
        del candidates[45:]

    # 2. Add structured/random basins at a few promising resolutions.
    top_ns = []
    for c, n, h in candidates:
        if n not in top_ns:
            top_ns.append(n)
        if len(top_ns) >= 10:
            break

    for n in top_ns:
        if time_left() < 20.0:
            break

        if parent is not None and parent_n > 0:
            hb = binarize(interp_to(parent, parent_n, n), n)
            c = compute_c5(hb)
            candidates.append((c, n, hb))
            if c < best_c - 1e-12:
                best_c = c
                best_h = hb.copy()
                best_n = n

        huf = make_random_uniform(rng, n)
        c = compute_c5(huf)
        candidates.append((c, n, huf))

        hf = make_random_fourier(rng, n, 4)
        c = compute_c5(hf)
        candidates.append((c, n, hf))

        ht = make_tanh_sinusoid(rng, n)
        c = compute_c5(ht)
        candidates.append((c, n, ht))

        if n % 2 == 0:
            hr = make_run_binary(n, rng)
            c = compute_c5(hr)
            candidates.append((c, n, hr))

        candidates.sort(key=cand_key)
        if len(candidates) > 60:
            del candidates[60:]

    # 3. Binary local search at selected even resolutions.
    binary_budget = min(230.0, max(0.0, time_left() - 720.0))
    if binary_budget >= 80.0:
        bin_ns = []
        for c, n, h in candidates:
            if n % 2 == 0 and n not in bin_ns:
                bin_ns.append(n)
            if len(bin_ns) >= 4:
                break
        if len(bin_ns) == 0 and best_n % 2 == 0:
            bin_ns.append(best_n)
        if len(bin_ns) == 0 and parent_n % 2 == 0:
            bin_ns.append(parent_n)

        if len(bin_ns) > 0:
            num = min(3, len(bin_ns))
            slot = binary_budget / float(num)
            for n in bin_ns[:num]:
                if time_left() < 80.0:
                    break
                slot = max(20.0, min(slot, time_left() - 60.0))

                h0 = None
                for c, nn, hh in candidates:
                    if nn == n:
                        h0 = hh
                        break
                if h0 is None:
                    h0 = binarize(interp_to(parent, parent_n, n), n)

                _G_CALL_END = time.time() + 0.58 * slot
                h2, c2 = binary_local_search(h0, n, 250000, rng)
                c2 = compute_c5(h2)
                candidates.append((c2, n, h2.copy()))
                if c2 < best_c - 1e-12:
                    best_c = c2
                    best_h = h2.copy()
                    best_n = n

                if time_left() > 90.0:
                    hrun = make_run_binary(n, rng)
                    _G_CALL_END = time.time() + 0.38 * slot
                    h3, c3 = binary_local_search(hrun, n, 180000, rng)
                    c3 = compute_c5(h3)
                    candidates.append((c3, n, h3.copy()))
                    if c3 < best_c - 1e-12:
                        best_c = c3
                        best_h = h3.copy()
                        best_n = n

                candidates.sort(key=cand_key)
                if len(candidates) > 80:
                    del candidates[80:]

    # 4. Coordinate descent polish over the current best candidates.
    coord_budget = min(540.0, max(0.0, time_left() - 240.0))
    if coord_budget >= 60.0:
        candidates.sort(key=cand_key)
        num = min(8, len(candidates))
        slot = coord_budget / float(max(1, num))
        used = 0
        for item in candidates[:num]:
            if time_left() < 60.0:
                break
            if used >= coord_budget - 20.0:
                break
            c, n, h = item
            s = max(10.0, min(slot, time_left() - 50.0, coord_budget - used - 10.0))
            if s < 10.0:
                break
            _G_CALL_END = time.time() + s
            used += s
            h2, c2 = coordinate_descent(h.copy(), n, 18, rng, fine=False)
            c2 = compute_c5(h2)
            candidates.append((c2, n, h2.copy()))
            if c2 < best_c - 1e-12:
                best_c = c2
                best_h = h2.copy()
                best_n = n
            candidates.sort(key=cand_key)
            if len(candidates) > 80:
                del candidates[80:]

        if time_left() > 160.0:
            candidates.sort(key=cand_key)
            num = min(4, len(candidates))
            slot = max(20.0, min(70.0, (time_left() - 80.0) / float(max(1, num))))
            for item in candidates[:num]:
                if time_left() < 100.0:
                    break
                c, n, h = item
                s = max(10.0, min(slot, time_left() - 60.0))
                if s < 10.0:
                    break
                _G_CALL_END = time.time() + s
                h2, c2 = coordinate_descent(h.copy(), n, 25, rng, fine=True)
                c2 = compute_c5(h2)
                candidates.append((c2, n, h2.copy()))
                if c2 < best_c - 1e-12:
                    best_c = c2
                    best_h = h2.copy()
                    best_n = n
                candidates.sort(key=cand_key)
                if len(candidates) > 80:
                    del candidates[80:]

    # 5. SLSQP gradient polish with smooth log-sum-exp approximation.
    if time_left() > 150.0:
        candidates.sort(key=cand_key)
        for item in candidates[:3]:
            if time_left() < 110.0:
                break
            c, n, h = item
            mi = 50 if n > 384 else 80
            h2, c2 = slsqp_polish(h.copy(), n, [12.0, 60.0, 250.0], mi)
            c2 = compute_c5(h2)
            candidates.append((c2, n, h2.copy()))
            if c2 < best_c - 1e-12:
                best_c = c2
                best_h = h2.copy()
                best_n = n

            if time_left() > 40.0:
                _G_CALL_END = time.time() + min(25.0, time_left() - 10.0)
                h3, c3 = coordinate_descent(h2.copy(), n, 12, rng, fine=True)
                c3 = compute_c5(h3)
                if c3 < c2:
                    h2, c2 = h3, c3
                candidates.append((c2, n, h2.copy()))
                if c2 < best_c - 1e-12:
                    best_c = c2
                    best_h = h2.copy()
                    best_n = n

            candidates.sort(key=cand_key)
            if len(candidates) > 100:
                del candidates[100:]

    # 6. Final fine polished local search on the incumbent.
    if time_left() > 30.0:
        _G_CALL_END = time.time() + max(10.0, time_left() - 8.0)
        best_h, best_c = coordinate_descent(best_h.copy(), best_n, 25, rng, fine=True)

    best_h = project_feasible(best_h, best_n)
    final_c = compute_c5(best_h)

    try:
        val = evaluate_erdos_solution(best_h)
        val = float(val)
        if np.isfinite(val):
            final_c = val
    except Exception:
        pass

    return (best_h, final_c, int(best_n))
