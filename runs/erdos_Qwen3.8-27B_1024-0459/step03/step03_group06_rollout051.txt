def block_move(h, n, rng, block_size=3):
    """Move a contiguous block of 1s to a random position."""
    h = h.copy()
    # Find a contiguous block of 1s
    ones = np.where(h == 1.0)[0]
    if len(ones) == 0:
        return h
    
    # Find contiguous runs of 1s
    runs = []
    start = ones[0]
    for k in range(1, len(ones)):
        if ones[k] != ones[k-1] + 1:
            runs.append((start, ones[k-1]))
            start = ones[k]
    runs.append((start, ones[-1]))
    
    # Pick a run of appropriate size
    candidates = [(s, e) for s, e in runs if e-s+1 >= block_size]
    if not candidates:
        return h
    
    s, e = rng.choice(len(candidates))
    s, e = candidates[s]
    bs = e - s + 1
    
    # Find a random position to move the block
    # The block should fit in [0, n-1]
    # And not overlap with existing 1s
    # This is complex, so let's just try random positions

        new_s = rng.randint(0, n - bs + 1)

        # (excluding the original position)

        h[s:e+1] = 0.0

            if h[new_s + k] > 0.5:

                break
        if ok:
