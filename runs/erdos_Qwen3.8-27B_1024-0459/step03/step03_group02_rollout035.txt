    #   As h[j] when j=i: contributes h[i]*(1-h[i+k]) for k=1..n-1-i
    #   As h[j+k] when j+k=i, i.e., j=i-k: contributes h[i-k]*(1-h[i]) for k=1..i

def project_binary(h, n):

def project_feasible(h, n):

        delta = new_c5 - cur_c5

                cur_h[i] = 1.0 - cur_h[i]
            else:
                cur_h[idx] = 1.0 - cur_h[idx]
        
        T *= alpha

def upscale_binary(src_h, src_n, dst_n, seed):
    """Upscale a binary array using probability-based assignment."""
    rng = np.random.RandomState(seed)
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    probs = np.interp(x_new, xp, src_h)
    
    # Assign 1s and 0s based on probabilities, ensuring exactly dst_n/2 ones
    # Use a greedy approach: sort by probability, assign top dst_n/2 to 1
    order = np.argsort(-probs)
    h = np.zeros(dst_n)
    h[order[:dst_n//2]] = 1.0
    
    # Add some randomness to break ties
    # Flip a few random bits near the boundary
    n_flips = dst_n // 20
    boundary_idx = order[dst_n//2 - n_flips: dst_n//2 + n_flips]

    for i in flip_candidates:
        if probs[i] > 0.4 and probs[i] < 0.6:
            h[i] = 1.0 - h[i]
    
    return h

                    break  # Move to next i
                else:
                    best_h[i], best_h[j] = 1.0, 0.0
        
        if not improved:

            if no_improve >= 3:

        else:
            no_improve = 0

That's maybe 2ms per sweep in numpy. We can do 500,000 sweeps per second... that's too optimistic. More like 500 sweeps per second. In 100s, that's 50,000 sweeps. That's more than enough.

Let me be smarter: instead of trying ALL swaps for each i, try a random subset. Or use a "nearest neighbor" approach: for each 1, only try swapping with nearby 0s.

Actually, a better approach: compute the "gradient" of C5 with respect to each position, and use that to guide which swaps to try.

For binary h, the "effect" of moving a 1 from position i to position j is:
- Position i: was contributing h[i]*stuff, now contributes 0
- Position j: was contributing 0*stuff, now contributes 1*stuff

The change in corr[k] for k>0:
- Remove contribution at i: h[i]*(1-h[i+k]) for k=1..n-1-i, and h[i-k]*(1-h[i]) for k=1..i
- Add contribution at j: 1*(1-h[j+k]) for k=1..n-1-j, and h[j-k]*(1-0)=h[j-k] for k=1..j

Net change in corr[k]:
For k where both i+k<n and j+k<n:
  delta_corr[k] = -[(1-h[i+k]) - h[i-k] if i-k>=0] + [(1-h[j+k]) - h[j-k] if j-k>=0]

This is O(n) to compute for a given (i,j) pair. But we can precompute the "potential" at each position.

Let me define for each position p:
  pot[p] = (n-1-p) - sum_{k=1}^{n-1-p} h[p+k] + sum_{k=1}^{p} h[p-k]

This is the "net contribution" of position p being 1 to the total correlation. Then:
  delta_corr_total = pot[j] - pot[i] (approximately, for the sum)

But we care about the MAX, not the sum. So this doesn't directly help.

Let me just use the practical approach: try random swaps, accept if C5 improves. The key is to make enough tries.

For n=160: one C5 eval is ~0.05ms. We can do ~20,000 evals/s. In 100s, that's 2,000,000 swaps.
For n=200: one C5 eval is ~0.1ms. We can do ~10,000 evals/s. In 100s, that's 1,000,000 swaps.

import numpy as np
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        dist = np.abs(h - 0.5)
        order = np.argsort(dist)
        for j in order[:min(n, max(1, int(np.ceil(abs(diff) * 2))))]:
            step = np.sign(diff) * min(abs(diff), 0.5 - abs(h[j] - 0.5) + 0.01)
            h[j] = np.clip(h[j] + step, 0.0, 1.0)
            diff = target - np.sum(h)
            if abs(diff) < 1e-13:
                break
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def make_binary_balanced(h, n, seed):
    """Convert to binary with exactly n/2 ones using probability-based assignment."""
    rng = np.random.RandomState(seed)
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:n // 2]] = 1.0
    return h_bin

def swap_cd(h, n, time_limit, max_sweeps=50000, random_frac=1.0):
    """Swap-based coordinate descent. Preserves count exactly."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    no_improve = 0
    total_swaps = 0
    
    n_ones = np.where(best_h == 1.0)[0]
    n_zeros = np.where(best_h == 0.0)[0]
    
    for sweep in range(max_sweeps):
        if time.time() - _START_TIME > time_limit or remaining_time() < 5:
            break
        
        improved = False
        # Randomize the order of 1-positions to try
        np.random.shuffle(n_ones)
        n_try = max(1, int(len(n_ones) * random_frac))
        
        for i_pos in n_ones[:n_try]:
            i = int(i_pos)
            if best_h[i] != 1.0:
                continue
            # Try swapping with random 0s
            n_zero_try = max(3, len(n_zeros) // 5)
            j_candidates = np.random.choice(len(n_zeros), min(n_zero_try, len(n_zeros)), replace=False)
            
            found = False
            for ji in j_candidates:
                j = int(n_zeros[ji])
                if best_h[j] != 0.0:
                    continue
                best_h[i], best_h[j] = 0.0, 1.0
                c5_new = compute_c5(best_h)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    improved = True
                    total_swaps += 1
                    found = True
                    break
                else:
                    best_h[i], best_h[j] = 1.0, 0.0

                # Update tracking

        if not improved:

def sa_swaps(h, n, T0, T1, alpha, time_limit, batch=1):
    """Simulated annealing using swap moves."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    T = T0
    
    while T > T1 and time.time() - _START_TIME < time_limit:
        if remaining_time() < 5:
            break
        
        i = np.random.randint(n)
        j = np.random.randint(n)
        if cur_h[i] == cur_h[j]:
            T *= alpha
            continue
        
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c5 = compute_c5(cur_h)
        delta = new_c5 - cur_c5
        
        if delta <= 0 or np.random.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        
        T *= alpha
    
    return best_h, best_c5

def thue_morse_n(n):
    """Thue-Morse balanced binary sequence of length n."""
    h = np.zeros(n)
    for i in range(n):
        v, cnt = i, 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt else 0.0
    # Balance
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        one_idx = np.where(h == 1)[0]
        h[one_idx[:n_ones - target]] = 0
    elif n_ones < target:
        zero_idx = np.where(h == 0)[0]
        h[zero_idx[:target - n_ones]] = 1
    return h

    # Add small noise to break symmetry
    rng = np.random.RandomState(seed)
    probs += rng.randn(dst_n) * 0.01
