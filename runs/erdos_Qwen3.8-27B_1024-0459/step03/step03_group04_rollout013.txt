def coord_descent_step(h, n, order):
    """One sweep of coordinate descent with exact 1D solves."""
    dx = 2.0 / n
    for i in order:
        # Compute gradients g_k = d(corr[k])/d(h[i])
        # corr[k] = sum_j h[j]*(1-h[j+k-n+1]) for valid j
        # h[i] appears as:
        #   1st factor: j=i, corr[k] += h[i]*(1-h[i+k-n+1]) if valid
        #   2nd factor: j+k-n+1=i, i.e., j=i-k+n-1, corr[k] += h[j]*(1-h[i]) if valid
        
        g = np.zeros(2*n-1)
        # h[i] as first factor
        for k in range(2*n-1):
            j2 = i + k - (n-1)  # index in (1-h)
            if 0 <= j2 < n:
                g[k] = (1.0 - h[j2])
        # h[i] as second factor (1-h[i])
        for k in range(2*n-1):
            j1 = i - k + (n-1)  # index in h
            if 0 <= j1 < n:
                g[k] -= h[j1]
        
        # Now corr[k](v) = corr[k](current) + g[k]*(v - h[i])
        # = (corr[k] - g[k]*h[i]) + g[k]*v
        # Let a_k = corr[k] - g[k]*h[i], b_k = g[k]
        # Minimize max_k (a_k + b_k * v) for v in [0,1]
        
        corr_vals = np.correlate(h, 1.0 - h, mode='full')
        a = corr_vals - g * h[i]
        b = g  # this is the gradient
        
        # f(v) = max_k (a_k + b_k * v) is convex in v
        # Find minimum on [0,1]
        # Check boundaries and crossing points
        
        candidates = [0.0, 1.0]
        # Find top few a_k + b_k*v at v=0.5 (midpoint) to identify relevant constraints
        mid_vals = a + b * 0.5
        top_idx = np.argsort(mid_vals)[-min(10, 2*n-1):]
        
        for p in range(len(top_idx)):
            for q in range(p+1, len(top_idx)):
                ip, iq = top_idx[p], top_idx[q]
                if b[ip] != b[iq]:
                    v_cross = (a[iq] - a[ip]) / (b[ip] - b[iq])
                    if 0.0 <= v_cross <= 1.0:
                        candidates.append(v_cross)
        
        best_v = h[i]
        best_f = float('inf')
        for v in candidates:
            f_v = np.max(a + b * v)
            if f_v < best_f:
                best_f = f_v
                best_v = v
        
        h[i] = max(0.0, min(1.0, best_v))
    
    # Enforce sum constraint
    h = project_h(h, n)
    return h

def compute_c5_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def project_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final exact adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Find the element closest to 0.5
        mid = np.argmin(np.abs(h - 0.5))
        h[mid] += diff
        h = np.clip(h, 0.0, 1.0)

**Lagrangian coordinate descent**: Optimize max_k corr[k] + λ*(sum(h) - n/2)² where λ is a large penalty. This way the sum constraint is built into the 1D subproblems.

For coordinate i, the 1D problem becomes:
minimize max_k (a_k + b_k*v) + λ * (sum_remainder + v - n/2)²

where sum_remainder = sum(h) - h[i] (sum of all other coordinates, which is fixed during the 1D solve).

This is: minimize max_k (a_k + b_k*v) + λ*(c + v)² where c = sum_remainder - n/2.

The second term is a parabola in v, and the first is piecewise linear. The sum is piecewise quadratic (convex). We can find the minimum by:
- Checking the boundaries v=0, v=1
- For each "seg" where a particular k is active: minimize (a_k + b_k*v) + λ(c+v)²
  This is a quadratic: λv² + (b_k + 2λc)v + (a_k + λc²), minimized at v = -(b_k + 2λc)/(2λ)
  Clipped to [0,1] and the segment's valid range.

This is solvable in O(K) where K is the number of segments (at most 2n-1 but typically much fewer).

Let me implement this. With λ = 10000 or so, the sum constraint will be very tight.

Actually, for the 1D subproblem with the quadratic penalty, the optimal v can be found by:
1. For each k, the function f_k(v) = a_k + b_k*v + λ(c+v)² is a parabola
2. The minimum of the max is found by trying the "crossover" points
3. In practice, we can just evaluate f(v) = max_k(a_k + b_k*v) + λ(c+v)² at many points in [0,1] and take the min

            corr += g * delta

    return h, corr


def slsqp_polish(h, n, t0, budget):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def sf(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        b = 500.0
        exps = np.exp(b * (c - m))
        return (m + np.log(np.sum(exps)) / b) * dx

    def sc(x):
        return np.sum(x) - n / 2.0

    cons = [{'type': 'eq', 'fun': sc}]
    best_local = h.copy()
    best_c5 = compute_c5_corr(h)[0]

    for beta in [500, 2000]:
        if time_left(t0, budget) < 5:
            break
        try:
            def sf_beta(x):
                c = np.correlate(x, 1.0 - x, mode='full')
                m = np.max(c)
                exps = np.exp(beta * (c - m))
                return (m + np.log(np.sum(exps)) / beta) * dx

            res = minimize(sf_beta, best_local, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 400, 'ftol': 1e-18, 'iprint': 0})
            hn = project_h(res.x, n)
            cn, _ = compute_c5_corr(hn)
            if cn < best_c5:
                best_c5 = cn
                best_local = hn.copy()
        except Exception:
            pass
    return best_local, best_c5


def lbfgs_polish(h, n, t0, budget):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def pb(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        b = 300.0
        exps = np.exp(b * (c - m))
        sme = m + np.log(np.sum(exps)) / b
        pen = 5000.0 * (np.sum(x) - n / 2.0) ** 2
        return sme * dx + pen

    best_local = h.copy()
    best_c5 = compute_c5_corr(h)[0]
    try:
        res = minimize(pb, best_local, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 1000, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 100})
        hn = project_h(res.x, n)
        cn, _ = compute_c5_corr(hn)
        if cn < best_c5:
            best_c5 = cn
            best_local = hn.copy()
    except Exception:
        pass
    return best_local, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5, _ = compute_c5_corr(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 160
        h0 = np.zeros(best_n)
        h0[:best_n // 2] = 1.0
        best_h = project_h(h0, best_n)
        best_c5, _ = compute_c5_corr(best_h)
        print("Default: n={}, C5={:.8f}".format(best_n, best_c5))

    # Try multiple resolutions via interpolation
    resolutions = [120, 140, 160, 180, 200, 220, 240, 260, 280, 320]
    for nr in resolutions:
        if time_left(t0, budget) < 20:
            break
        if nr == best_n:
            continue
        ht = interp_h(best_h, best_n, nr)
        ht = project_h(ht, nr)
        ct, _ = compute_c5_corr(ht)
        if ct < best_c5:
            best_c5 = ct
            best_h = ht.copy()
            best_n = nr
            print("  Res n={}: C5={:.8f}".format(nr, ct))

    # Main coordinate descent at best resolution
    n = best_n
    h = best_h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    order = np.arange(n)
    sweeps = 0
    stall = 0
    lam_schedule = [0.0, 0.0, 50.0, 200.0, 1000.0]

    while time_left(t0, budget) > 120:
        if sweeps == 0 or sweeps % 500 == 0:
            order = rng.permutation(n)

        batch = 100
        for _ in range(batch):
            if time_left(t0, budget) < 120:
                break
            lam_idx = min(len(lam_schedule) - 1, sweeps // 300)
            lam = lam_schedule[lam_idx]
            if lam == 0.0:
                h, corr = coord_sweep(h, n, corr, order)
            else:
                h, corr = coord_sweep_lam(h, n, corr, order, lam)
            sweeps += 1

        h = project_h(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = np.max(corr) * (2.0 / n)

        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            best_n = n
            stall = 0
            if sweeps % 5000 == 0:
                print("  CD n={}: sweep={}, C5={:.8f}".format(n, sweeps, best_c5))
        else:
            stall += batch

        if stall > 3000:
            # Perturb to escape
            npert = rng.randint(3, 10)
            for _ in range(npert):
                ii = rng.randint(n)
                jj = rng.randint(n)
                if ii == jj:
                    continue
                d = rng.rand() * 0.08
                if h[ii] >= d and h[jj] <= 1.0 - d:
                    h[ii] -= d
                    h[jj] += d
            h = project_h(h, n)
            corr = np.correlate(h, 1.0 - h, mode='full')
            order = rng.permutation(n)
            stall = 0

    # SLSQP / L-BFGS-B polish
    if time_left(t0, budget) > 30:
        h_pol, c5_pol = lbfgs_polish(best_h, best_n, t0, budget)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            print("  L-BFGS-B: C5={:.8f}".format(best_c5))

    if time_left(t0, budget) > 20:
        h_pol, c5_pol = slsqp_polish(best_h, best_n, t0, budget)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            print("  SLSQP: C5={:.8f}".format(best_c5))

    # Try coordinate descent at neighboring resolutions
    for dn in [-20, -10, 10, 20]:
        if time_left(t0, budget) < 30:
            break
        nt = best_n + dn
        if nt < 100 or nt > 400:
            continue
        ht = interp_h(best_h, best_n, nt)
        ht = project_h(ht, nt)
        ct, _ = compute_c5_corr(ht)
        if ct > best_c5 + 0.002:
            continue
        ct2, corr2 = ct, np.correlate(ht, 1.0 - ht, mode='full')
        ord2 = rng.permutation(nt)
        for _ in range(200):
            if time_left(t0, budget) < 30:
                break
            ht, corr2 = coord_sweep(ht, nt, corr2, ord2)
        ht = project_h(ht, nt)
        c5_t, _ = compute_c5_corr(ht)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = ht.copy()
            best_n = nt
            print("  Res n={}: C5={:.8f}".format(nt, best_c5))

    # Fine perturbation phase
    if time_left(t0, budget) > 30:
        n = best_n
        h = best_h.copy()
        deltas = [0.001, 0.002, 0.003, 0.005, 0.008, 0.01, 0.015, 0.02]
        for piter in range(50000):
            if time_left(t0, budget) < 5:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in deltas:
                if h[i] >= d and h[j] <= 1.0 - d:
                    h_tr = h.copy()
                    h_tr[i] -= d
                    h_tr[j] += d
                    h_tr = project_h(h_tr, n)
                    c5_tr, _ = compute_c5_corr(h_tr)
                    if c5_tr < best_c5:
                        best_c5 = c5_tr
                        best_h = h_tr.copy()
                        h = h_tr.copy()
                        d = None
                        break
                    d = None
                if h[j] >= d and h[i] <= 1.0 - d:
                    h_tr = h.copy()
                    h_tr[j] -= d
                    h_tr[i] += d
                    h_tr = project_h(h_tr, n)
                    c5_tr, _ = compute_c5_corr(h_tr)
                    if c5_tr < best_c5:
                        best_c5 = c5_tr
                        best_h = h_tr.copy()
                        h = h_tr.copy()
                        d = None
                        break

    # Final double SLSQP on best
    if time_left(t0, budget) > 10:
        h_pol, c5_pol = slsqp_polish(best_h, best_n, t0, budget)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    best_h = project_h(best_h, best_n)
    final_c5, _ = compute_c5_corr(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
