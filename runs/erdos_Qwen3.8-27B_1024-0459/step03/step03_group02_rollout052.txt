    """Compute new correlation after changing h[i] from h[i] to t."""
    # Only lags involving i change
    # For each offset off in [-(n-1), n-1]:
    #   k = n-1 + off
    #   If i+off in [0,n): corr[k] += (t - h[i]) * (1 - h[i+off])
    #   If off != 0 and i-off in [0,n): corr[k] += -h[i-off] * (t - h[i])
    #   If off == 0: corr[k] += t*(1-t) - h[i]*(1-h[i])
    
    delta = t - h[i]
    new_corr = corr.copy()  # O(n)
    
    for off in range(-(n-1), n):
        k = n - 1 + off
        if off == 0:
            new_corr[k] += t * (1-t) - h[i] * (1-h[i])
        else:
            change = 0.0
            if 0 <= i + off < n:
                change += delta * (1.0 - h[i+off])
            if 0 <= i - off < n:
                change -= h[i-off] * delta
            if change != 0.0:
                new_corr[k] += change
    
    return new_corr

from scipy.optimize import minimize
import time

_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = np.argmin(h)
            else:
                idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-14:
            for j in range(n):
                if h[j] > 1e-14 and h[j] < 1.0 - 1e-14:
                    h[j] = np.clip(h[j] + diff2, 0.0, 1.0)
                    break
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def _smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def slsqp_refine(h, n, beta, maxiter):
    global _N_WORK, _BETA_WORK
    _N_WORK = n
    _BETA_WORK = beta
    bounds = [(0.0, 1.0)] * n

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()

    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def subgrad_direction(h, n, n_active=5):
    corr = compute_corr(h)
    max_corr = np.max(corr)
    active = np.where(corr >= max_corr - 0.3)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        active = np.argsort(corr)[::-1][:n_active]
    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return None
    return grad / gnorm

def efficient_coord_descent(h, n, max_sweeps=30):
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n

    for sweep in range(max_sweeps):

        interior = np.where((best > 1e-6) & (best < 1.0 - 1e-6))[0]
        if len(interior) == 0:
            break
        order = np.random.permutation(interior)

        for i in order:
            if remaining_time() < 1:
                break
            # Golden section search for optimal h[i]
            # Objective: max_k (base[k] + coeff[k]*t) * dx
            # where for lag with offset off:
            #   if off==0: term is t*(1-t) -> special
            #   if i+off valid: +(t - best[i])*(1-best[i+off])
            #   if i-off valid (off!=0): -best[i-off]*(t-best[i])

            lo, hi = 0.0, 1.0
            t_best = best[i]
            c5_best_local = best_c5

            # Precompute coefficients
            ncorr = 2 * n - 1
            a = np.zeros(ncorr)  # linear coeff
            b = np.zeros(ncorr)  # quadratic coeff (only for off=0)
            for off in range(-(n-1), n):
                k = n - 1 + off
                if off == 0:
                    a[k] += 1.0
                    b[k] += -1.0
                else:
                    if 0 <= i + off < n:
                        a[k] += (1.0 - best[i + off])
                    if 0 <= i - off < n:
                        a[k] += (-best[i - off])

            # base[k] = corr[k] - a[k]*best[i] - b[k]*best[i]^2
            t0 = best[i]
            base = best_corr - a * t0 - b * t0 * t0

            def obj_at_t(t):
                vals = base + a * t + b * t * t
                return np.max(vals) * dx

            # Try several candidates
            candidates = [0.0, 1.0, t0 * 0.5, t0 * 1.5, 1.0 - t0]
            candidates = [max(0.0, min(1.0, c)) for c in candidates]
            candidates.append(t0)

            for t in candidates:
                if abs(t - t0) < 1e-10:
                    continue
                c = obj_at_t(t)
                if c < c5_best_local - 1e-15:
                    c5_best_local = c
                    t_best = t

            # Refine with golden section if improvement found
            if abs(t_best - t0) > 1e-8:
                lo = max(0.0, t0 - 0.3)
                hi = min(1.0, t0 + 0.3)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = obj_at_t(c1)
                f2 = obj_at_t(c2)
                for _ in range(40):
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = obj_at_t(c1)
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = obj_at_t(c2)
                t_best = (lo + hi) / 2.0
                c5_gs = obj_at_t(t_best)
                if c5_gs < c5_best_local:
                    c5_best_local = c5_gs

            if c5_best_local < best_c5 - 1e-14:
                best[i] = t_best
                best_corr = compute_corr(best)
                best_c5 = compute_c5(best)
                improved = True

        if not improved:
            break

    return best, best_c5
