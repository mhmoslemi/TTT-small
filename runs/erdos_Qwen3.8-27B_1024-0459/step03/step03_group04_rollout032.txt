def compute_softmax_weights(corr, beta):
    m = np.max(corr)
    if beta < 1e-6:
        return m / (2.0 * len(corr) / 2.0), np.ones(2 * len(corr) - 1) / (2 * len(corr) - 1)
    shifted = corr - m
    max_shifted = np.max(shifted)
    exps = np.exp(beta * (shifted - max_shifted))
    sum_exps = np.sum(exps)
    weights = exps / sum_exps
    sme = m + max_shifted + np.log(sum_exps) / beta
    return sme, weights

def compute_grad(h, weights, beta):
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-12:
            continue
        tau = k - (n - 1)
        if tau >= 0:
            idx1 = np.arange(n - tau)
            idx2 = idx1 + tau
            grad[idx1] += wk * (1.0 - h[idx2])
            grad[idx2] -= wk * h[idx1]
        else:
            d = -tau
            idx1 = np.arange(d, n)
            idx2 = np.arange(0, n - d)
            grad[idx1] += wk * (1.0 - h[idx2])
            grad[idx2] -= wk * h[idx1]
    return grad / beta

def projected_grad_descent(h, n, beta_start, beta_end, num_iters, t0, budget, verbose=False):
    """Run projected gradient descent with adaptive beta."""
    best_h = h.copy()
    best_c5, _ = compute_c5(h)
    h_cur = h.copy()
    
    for it in range(num_iters):
        if time.time() - t0 > budget - 5:
            break
        beta = beta_start + (beta_end - beta_start) * (it / max(1, num_iters - 1))
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        sme, weights = compute_softmax_weights(corr, beta)
        grad = compute_grad(h_cur, weights, beta)
        
        # Gradient direction (descent)
        step_dir = -grad
        # Constraint-aware: push toward sum = n/2
        sum_err = np.sum(h_cur) - n / 2.0
        step_dir -= sum_err / n
        
        gnorm = np.linalg.norm(step_dir)
        if gnorm < 1e-14:
            break
        step_dir /= gnorm
        
        # Try multiple step sizes
        best_step = 0
        best_obj = best_c5
        for alpha in [0.01, 0.02, 0.05, 0.1, 0.2, 0.005, 0.001, 0.08, 0.03, 0.15]:
            h_try = h_cur + alpha * step_dir
            h_try = project_h(h_try, n)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_obj - 1e-15:
                best_obj = c5_try
                best_step = alpha
                best_h = h_try.copy()
        
        if best_step > 0:
            h_cur = best_h
            if best_c5 - best_obj > 1e-6:
                if verbose:
                    print("    it={}, c5={:.8f}, step={:.4f}".format(it, best_obj, best_step))
                best_c5 = best_obj
        else:
            # No improvement: try smaller steps
            improved = False
            for alpha in [0.001, 0.0005, 0.0002, 0.0001]:
                h_try = h_cur + alpha * step_dir
                h_try = project_h(h_try, n)
                c5_try, _ = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h_cur = h_try
                    improved = True
                    break
            if not improved:
                # Try a random perturbation to escape
                rng_local = np.random.RandomState(it + 7)
                h_try = h_cur + 0.01 * rng_local.randn(n)
                h_try = project_h(h_try, n)
                c5_try, _ = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h_cur = h_try
    
    return best_h, best_c5

def coordinate_descent(h, n, max_iters, t0, budget):
    """Coordinate descent with pair moves preserving sum constraint."""
    best_h = h.copy()
    best_c5, _ = compute_c5(h)
    h_cur = h.copy()
    rng = np.random.RandomState(12345)
    
    deltas = [0.001, 0.002, 0.005, 0.01, 0.02, 0.003, 0.008, 0.015]
    
    for it in range(max_iters):
        if time.time() - t0 > budget - 3:
            break
        # Pick two random coordinates
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        found = False
        for d in deltas:
            if h_cur[i] >= d and h_cur[j] <= 1.0 - d:
                h_try = h_cur.copy()
                h_try[i] -= d
                h_try[j] += d
                c5_try, _ = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h_cur = h_try
                    found = True
                    break
            if h_cur[j] >= d and h_cur[i] <= 1.0 - d:
                h_try = h_cur.copy()
                h_try[j] -= d
                h_try[i] += d
                c5_try, _ = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h_cur = h_try
                    found = True
                    break
        
        if not found:
            # Try triple move
            k = rng.randint(n)
            if k != i and k != j:
                d = 0.005
                if h_cur[i] >= 2*d and h_cur[j] <= 1-d and h_cur[k] <= 1-d:
                    h_try = h_cur.copy()
                    h_try[i] -= 2*d
                    h_try[j] += d
                    h_try[k] += d
                    c5_try, _ = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-15:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h_cur = h_try
    
    return best_h, best_c5

def active_set_optimize(h, n, t0, budget, verbose=False):
    """Focus optimization on the most critical (active) shifts."""
    best_h = h.copy()
    best_c5, corr = compute_c5(h)
    h_cur = h.copy()
    
    for outer in range(40):
        if time.time() - t0 > budget - 10:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        mx = np.max(corr)
        thresh = mx - 0.3
        active_mask = corr >= thresh
        
        # Compute weighted gradient focusing on active set
        # Use softmax on active set only
        active_vals = corr[active_mask]
        b = 200.0
        m_act = np.max(active_vals)
        exps = np.exp(b * (active_vals - m_act))
        w_act = exps / np.sum(exps)
        
        # Build full weight vector (zero for inactive)
        weights_full = np.zeros(2 * n - 1)
        weights_full[active_mask] = w_act
        
        grad = compute_grad(h_cur, weights_full, b)
        step_dir = -grad
        sum_err = np.sum(h_cur) - n / 2.0
        step_dir -= sum_err / n
        
        gnorm = np.linalg.norm(step_dir)
        if gnorm < 1e-14:
            break
        step_dir /= gnorm
        
        improved = False
        for alpha in [0.05, 0.1, 0.2, 0.02, 0.08, 0.15, 0.3, 0.01]:
            h_try = h_cur + alpha * step_dir
            h_try = project_h(h_try, n)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break
        
        if not improved:
            break
    
    return best_h, best_c5

def slsqp_polish(h, n, t0, budget):
    """Use SLSQP as a final polish."""
    best_h = h.copy()
    best_c5, _ = compute_c5(h)
    
    def obj_and_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        beta = 500.0
        shifted = corr - m
        exps = np.exp(beta * shifted)
        sme = m + np.log(np.sum(exps)) / beta
        # Gradient
        weights = exps / np.sum(exps)
        grad = compute_grad(x, weights, beta)
        # Add constraint gradient
        grad += 10.0 * (np.sum(x) - n / 2.0) / n
        return sme * (2.0 / n), grad
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n) / n}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj_and_grad, h, method='SLSQP', jac=True,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 500, 'ftol': 1e-18, 'iprint': 0})
        h_new = project_h(res.x, n)
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    except Exception:
        pass
    
    return best_h, best_c5

def make_binary_balanced(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return h

def make_sine_init(n, rng):
    t = np.arange(n) / n
    f1 = 2 + rng.randint(0, 4)
    f2 = 5 + rng.randint(0, 6)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    a1 = 0.3 + 0.2 * rng.rand()
    a2 = 0.1 + 0.1 * rng.rand()
    h = 0.5 + a1 * np.sin(2 * np.pi * f1 * t + p1) + a2 * np.sin(2 * np.pi * f2 * t + p2)
    return project_h(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5, _ = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 160
        best_h = make_binary_balanced(best_n, rng)
        best_c5, _ = compute_c5(best_h)
    
    # Try multiple resolutions
    resolutions = [best_n]
    # Add nearby resolutions
    for dn in [-20, -10, 10, 20, 40, 60, 80]:
        nt = best_n + dn
        if nt >= 80 and nt <= 500:
            resolutions.append(nt)
    # Add some "good" resolutions
    for nt in [200, 240, 280, 320, 360, 400]:
        if nt not in resolutions:
            resolutions.append(nt)
    resolutions = sorted(set(resolutions))
    
    # Phase 1: Quick evaluation of interpolated candidates at different resolutions
    for n_try in resolutions:
        if time.time() - t0 > budget - 60:
            break
        if n_try == best_n:
            continue
        h_try = interp_h(best_h, best_n, n_try)
        h_try = project_h(h_try, n_try)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try
    
    # Phase 2: Main optimization at best resolution
    n = best_n
    h_work = best_h.copy()
    
    # 2a: Projected gradient descent with increasing beta
    if time.time() - t0 < budget - 30:
        for beta_lo, beta_hi, nits in [(20, 100, 2000), (100, 500, 2000), (500, 2000, 1500), (2000, 10000, 1000)]:
            if time.time() - t0 > budget - 15:
                break
            h_work, c5_new = projected_grad_descent(h_work, n, beta_lo, beta_hi, nits, t0, budget)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_work.copy()
    
    # 2b: Active-set focused optimization
    if time.time() - t0 < budget - 20:
        h_work = active_set_optimize(best_h, n, t0, budget)
        c5_new, _ = compute_c5(h_work)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_work.copy()
    
    # 2c: SLSQP polish
    if time.time() - t0 < budget - 15:
        h_work = slsqp_polish(best_h, n, t0, budget)
        c5_new, _ = compute_c5(h_work)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_work.copy()
    
    # 2d: Coordinate descent
    if time.time() - t0 < budget - 10:
        n_cd = max(2000, int((budget - (time.time() - t0)) * 50))
        h_work, c5_new = coordinate_descent(best_h, n, n_cd, t0, budget)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_work.copy()
    
    # Phase 3: Try other resolutions with full optimization
    remaining = budget - (time.time() - t0)
    if remaining > 60:
        tried_ns = {n}
        for n_try in resolutions:
            if time.time() - t0 > budget - 30:
                break
            if n_try in tried_ns or n_try == n:
                continue
            tried_ns.add(n_try)
            
            h_try = interp_h(best_h, best_n, n_try)
            h_try = project_h(h_try, n_try)
            
            # Quick gradient descent
            h_try, c5_gd = projected_grad_descent(h_try, n_try, 50, 500, 1000, t0, budget)
            if c5_gd < best_c5:
                best_c5 = c5_gd
                best_h = h_try.copy()
                best_n = n_try
            
            # Active set
            if time.time() - t0 < budget - 20:
                h_try2 = active_set_optimize(h_try, n_try, t0, budget)
                c5_as, _ = compute_c5(h_try2)
                if c5_as < best_c5:
                    best_c5 = c5_as
                    best_h = h_try2.copy()
                    best_n = n_try
    
    # Phase 4: Multi-start from perturbed versions
    remaining = budget - (time.time() - t0)
    if remaining > 40:
        n = best_n
        for restart in range(5):
            if time.time() - t0 > budget - 20:
                break
            noise_scale = 0.02 * (1 + restart * 0.5)
            h_pert = best_h + noise_scale * rng.randn(n)
            h_pert = project_h(h_pert, n)
            
            h_pert, c5_p = projected_grad_descent(h_pert, n, 100, 2000, 800, t0, budget)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_pert.copy()
    
    # Phase 5: Final coordinate descent polish
    if time.time() - t0 > 5:
        n = best_n
        max_cd = int((budget - (time.time() - t0) - 3) * 80)
        if max_cd > 100:
            h_cd, c5_cd = coordinate_descent(best_h, n, max_cd, t0, budget)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
    
    # Final projection and evaluation
    best_h = project_h(best_h, best_n)
    final_c5, _ = compute_c5(best_h)
    
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
