            if nt < 100 or nt > 400:
                continue
            ht = interp_h(best_h, best_n, nt)
            ht = project_h(ht, nt)
            ct, _ = compute_c5_corr(ht)
            if ct < best_c5 - 1e-14:
                best_c5 = ct
                best_h = ht.copy()
                best_n = nt

    # Phase 7: One more optimization round at final resolution
    if time_left(t0, budget) > 20:
        n = best_n
        h = best_h.copy()
        bounds = [(0.0, 1.0)] * n
        c5_cur, corr = compute_c5_corr(h)
        for top_m in [10, 20]:
            if time_left(t0, budget) < 10:
                break
            active = find_active(corr, top_m=top_m, gap=0.4)
            for beta in [200, 600]:
                if time_left(t0, budget) < 6:
                    break
                try:
                    def obj_fin(x, a=active, b=beta):
                        return obj_active_lse(x, a, n, b)

                    def con_fin(x):
                        return np.sum(x) - n / 2.0

                    res = minimize(obj_fin, h, method='SLSQP',
                                   bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': con_fin, 'jac': lambda x: np.ones(n)}],
                                   options={'maxiter': 150, 'ftol': 1e-17, 'iprint': 0})
                    h_new = project_h(res.x, n)
                    c5_new, _ = compute_c5_corr(h_new)
                    if c5_new < best_c5 - 1e-14:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h = h_new.copy()
                        c5_cur, corr = compute_c5_corr(h)
                except Exception:
                    pass

    # Final
    best_h = project_h(best_h, best_n)
    final_c5, _ = compute_c5_corr(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
