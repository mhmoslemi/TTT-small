def _coord_descent(h, n, niter, rng):
    best = h.copy()
    bc = _c5(best)
    for _ in range(niter):
        if _tl() < 5:
            break
        nbi = np.where((best > 0.01) & (best < 0.99))[0]
        if len(nbi) < 2:
            break
        imp = False
        rng.shuffle(nbi)
        for idx in nbi[:40]:
            if _tl() < 3:
                break
            old = best[idx]
            for delta in [0.01, -0.01, 0.02, -0.02, 0.005, -0.005, 0.003, -0.003, 0.015, -0.015]:
                nv = old + delta
                if nv < 0.0 or nv > 1.0:
                    continue
                others = nbi[nbi != idx]
                if len(others) == 0:
                    break
                j = others[rng.randint(len(others))]
                h2 = best.copy()
                h2[idx] = nv
                h2[j] += (old - nv)
                h2 = _proj(h2, n)
                c2 = _c5(h2)
                if c2 < bc - 1e-14:
                    best, bc = h2, c2
                    imp = True
                    break
        if not imp:
            break
    return best, bc


def _slsqp_refine(h0, n, beta, maxiter):
    n_global = n
    bv = beta

    def _obj(x):
        c = _corr(x)
        mx = float(np.max(c))
        sh = c - mx
        emx = float(np.max(sh * bv))
        return (mx + float(np.sum(np.exp(sh * bv - emx))) / bv) * (2.0 / n_global)

    def _con(x):
        return x.sum() - 0.5 * n_global

    def _cons_g(x):
        return np.zeros(len(x))
    # constraint gradient is ones, handled implicitly

    try:
        res = minimize(_obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return _proj(np.asarray(res.x, dtype=np.float64), n)
    except Exception:
        return _proj(h0.copy(), n)


def _cons(x):
    return x.sum() - 0.5 * len(x)


def _interp(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    xs = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    xd = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return np.interp(xd, xs, h_src)


def _full_refine(h0, n, rng):
    h = _proj(h0.copy(), n)
    c5 = _c5(h)

    # Phase 1: SLSQP with moderate smoothing
    for beta in [50, 200]:
        if _tl() < 20:
            break
        h2 = _slsqp_refine(h, n, beta, 200)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2

    # Phase 2: gradient descent with adaptive temperature
    h, c5 = _gd_refine(h, n, 500, rng, 25.0, 300.0)

    # Phase 3: binary 1-swap
    h, c5 = _binary1swap(h, n, 50, rng)

    # Phase 4: more gradient descent
    h, c5 = _gd_refine(h, n, 400, rng, 50.0, 500.0)

    # Phase 5: binary 2-swap
    h, c5 = _binary2swap(h, n, 40, rng)

    # Phase 6: fine gradient descent
    h, c5 = _gd_refine(h, n, 300, rng, 100.0, 1000.0)

    # Phase 7: coordinate descent
    h, c5 = _coord_descent(h, n, 150, rng)

    # Phase 8: final SLSQP with high beta
    if _tl() > 15:
        h2 = _slsqp_refine(h, n, 2000, 400)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2
        h, c5 = _gd_refine(h, n, 200, rng, 200.0, 2000.0)

    # Phase 9: final polish
    h, c5 = _binary1swap(h, n, 30, rng)
    h, c5 = _coord_descent(h, n, 100, rng)
    h, c5 = _gd_refine(h, n, 100, rng, 500.0, 3000.0)

    return h, c5


def run(seed=42, budget_s=1300, **kwargs):
    global _t0, _tb
    _t0 = time.time()
    _tb = budget_s
    rng = np.random.RandomState(seed)

    try:
        hpar = np.array(initial_h_values, dtype=np.float64)
        npar = len(hpar)
    except Exception:
        npar = 160
        hpar = np.zeros(npar, dtype=np.float64)
        hpar[:npar // 2] = 1.0

    bh = _proj(hpar, npar)
    bc5 = _c5(bh)
    bn = npar
    print("Init n={} C5={:.8f}".format(bn, bc5))

    # Main optimization at parent resolution
    h, c5 = _full_refine(bh, npar, rng)
    if c5 < bc5:
        bh, bc5, bn = h, c5, npar
        print("Refined n={} C5={:.8f}".format(npar, c5))

    # Try nearby resolutions
    alts = [npar + 2, npar - 2, npar + 4, npar - 4, npar + 6, npar - 6, npar + 8, npar - 8, npar + 10]
    for nn in alts:
        if _tl() < 60:
            break
        if nn < 32 or nn % 2 != 0:
            continue
        hi = _proj(_interp(bh, bn, nn), nn)
        h2, c2 = _full_refine(hi, nn, rng)
        if c2 < bc5:
            bh, bc5, bn = h2, c2, nn
            print("N={} C5={:.8f}".format(nn, c2))

    # Try higher resolution
    for nn in [npar + 20, npar + 40, npar + 60]:
        if _tl() < 90:
            break
        if nn % 2 != 0:
            nn += 1
        hi = _proj(_interp(bh, bn, nn), nn)
        h2, c2 = _full_refine(hi, nn, rng)
        if c2 < bc5:
            bh, bc5, bn = h2, c2, nn
            print("HiRes N={} C5={:.8f}".format(nn, c2))

    # Final intensive polish on best solution
    if _tl() > 30:
        h, c5 = _gd_refine(bh, bn, 400, rng, 50.0, 5000.0)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _binary1swap(bh, bn, 50, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _binary2swap(bh, bn, 30, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _gd_refine(bh, bn, 300, rng, 200.0, 5000.0)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _coord_descent(bh, bn, 200, rng)
        if c5 < bc5:
            bh, bc5 = h, c5

    if _tl() > 15:
        h2 = _slsqp_refine(bh, bn, 5000, 500)
        c2 = _c5(h2)
        if c2 < bc5:
            bh, bc5 = h2, c2
        h, c5 = _gd_refine(bh, bn, 200, rng, 500.0, 10000.0)
        if c5 < bc5:
            bh, bc5 = h, c5

    # Final projection and evaluation
    bh = _proj(bh, bn)
    fc5 = _c5(bh)
    try:
        ec = evaluate_erdos_solution(bh)
        if ec < fc5:
            fc5 = ec
    except Exception:
        pass
    print("FINAL n={} C5={:.8f}".format(bn, fc5))
    return (bh, fc5, bn)
