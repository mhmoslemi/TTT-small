def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val

def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-14:
            g += w[k] * _corr_grad_single(x, n, k)
    return g * dx_val

def _corr_grad_single(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g

def sum_con(x):
    return np.sum(x) - _TGT

def sum_con_jac(x):
    return np.ones(_N)

def slsqp_run(h, n, beta, maxiter=500):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

def lbfgsb_run(h, n, beta, maxiter=2000):
    """L-BFGS-B with sum constraint handled by parameterization."""
    # We'll use the log-sum-exp smoothing but with L-BFGS-B
    # L-BFGS-B handles box constraints natively but not equality constraints
    # So we'll use a penalty for the sum constraint
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    
    def penalty_obj(x):
        base = smooth_obj(x)
        penalty = 1000.0 * (np.sum(x) - n/2.0)**2
        return base + penalty
    
    def penalty_grad(x):
        base_g = smooth_grad(x)
        pen_g = 2000.0 * (np.sum(x) - n/2.0) * np.ones(n)
        return base_g + pen_g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(penalty_obj, h, jac=penalty_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_local_search(h, n, time_limit=30, rng=None):
    """Binary pair-swap local search with active-set focusing."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        ones_idx = np.where(h == 1)[0]
        remove = rng.choice(ones_idx, n_ones - n // 2, replace=False)
        h[remove] = 0.0
    elif n_ones < n // 2:
        zeros_idx = np.where(h == 0)[0]
        add = rng.choice(zeros_idx, n // 2 - n_ones, replace=False)
        h[add] = 1.0
    
    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx
    start = time.time()
    stall = 0
    
    while remaining() > 5 and time.time() - start < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Active set: correlations within 1 of max (for binary, these are integers)
        active_thresh = cur_max - 0.5
        
        found = False
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        # Try random swaps, biased toward promising ones
        n_try = min(1000, n_ones * n_zeros)
        for _ in range(n_try):
            i = ones[rng.randint(n_ones)]
            j = zeros[rng.randint(n_zeros)]
            
            # Quick check: compute new correlation
            h[i] = 0.0
            h[j] = 1.0
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            
            if new_max < cur_max - 1e-12:
                cur_max = new_max
                corr = new_corr
                best_c5 = cur_max * dx
                found = True
                break
            else:
                h[i] = 1.0
                h[j] = 0.0
        
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 20:
                # Perturbation: random swap
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                corr = compute_corr(h)
                cur_max = float(np.max(corr))
                stall = 0
    
    return h, best_c5

def active_set_optimize(h, n, max_iters=200):
    """Focus on equalizing the top correlations."""
    h = h.copy()
    dx_val = 2.0 / n
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        top_idx = np.argmax(corr)
        second_idx = np.argsort(-corr)[1]
        c1, c2 = corr[top_idx], corr[second_idx]
        
        if c1 - c2 < 1e-12:
            break
        
        # Direction: reduce top, increase second
        g1 = _corr_grad_single(h, n, int(top_idx))
        g2 = _corr_grad_single(h, n, int(second_idx))
        d = g2 - g1
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-14:
            break
        d /= dnorm
        
        improved = False
        for step in [0.01, 0.02, 0.05, 0.005, 0.08, 0.015, 0.03]:
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-12:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            break
    
    return h, best_c5
