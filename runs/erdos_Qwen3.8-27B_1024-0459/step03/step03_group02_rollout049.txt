import numpy as np
from scipy.optimize import minimize, nelder
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_N_GLOBAL = 0
_BETA_GLOBAL = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            adj = diff / cnt
            # Check if adjustment would violate bounds
            new_vals = h[mask] + adj
            if np.all(new_vals >= 0) and np.all(new_vals <= 1):
                h[mask] = new_vals
            else:
                # Adjust only in the right direction
                if adj > 0:
                    m2 = h < 1.0 - 1e-12
                    cnt2 = np.sum(m2)
                    if cnt2 > 0:
                        h[m2] += diff / cnt2
                else:
                    m2 = h > 1e-12
                    cnt2 = np.sum(m2)
                    if cnt2 > 0:
                        h[m2] += diff / cnt2
        h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        # Distribute among free coordinates
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
            # Re-correct
            diff2 = target - np.sum(h)
            if abs(diff2) > 1e-14:
                for j in range(n):
                    if h[j] > 0 and h[j] < 1:
                        h[j] += diff2
                        h[j] = np.clip(h[j], 0, 1)
                        break
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def smooth_max_obj(x):
    n = _N_GLOBAL
    beta = _BETA_GLOBAL
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def smooth_max_grad(x):
    n = _N_GLOBAL
    beta = _BETA_GLOBAL
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    # Gradient of sum_k w_k * corr[k] where w_k = softmax(beta*(corr - m))
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-15:
            continue
        offset = k_idx - (n - 1)
        if offset >= 0:
            n_pairs = n - offset
            if n_pairs <= 0:
                continue
            grad[offset:offset+n_pairs] += weights[k_idx] * (1.0 - x[:n_pairs])
            grad[:n_pairs] -= weights[k_idx] * x[offset:offset+n_pairs]
        else:
            abs_off = -offset
            n_pairs = n - abs_off
            if n_pairs <= 0:
                continue
            grad[:n_pairs] += weights[k_idx] * (1.0 - x[abs_off:abs_off+n_pairs])
            grad[abs_off:abs_off+n_pairs] -= weights[k_idx] * x[:n_pairs]
    return grad * dx

def slsqp_refine(h, n, beta, maxiter):
    global _N_GLOBAL, _BETA_GLOBAL
    _N_GLOBAL = n
    _BETA_GLOBAL = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
    try:
        res = minimize(smooth_max_obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def coordinate_descent_sweep(h, n, max_iter=200):
    """Exact coordinate descent exploiting linearity of each lag in each coordinate."""
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for sweep in range(max_iter):
        if remaining_time() < 3:
            break
        improved = False
        # Randomize coordinate order
        order = np.random.permutation(n)
        
        for j in order:
            if remaining_time() < 1:
                break
            v = h[j]
            
            # Compute coefficients for each lag
            corr = compute_corr(h)
            offsets = np.arange(2*n-1) - (n-1)
            
            coeff = np.zeros(2*n-1)
            for k_idx in range(2*n-1):
                off = offsets[k_idx]
                c = 0.0
                if 0 <= j - off < n:
                    c += (1.0 - h[j - off])
                if 0 <= j + off < n:
                    c -= h[j + off]
                coeff[k_idx] = c
            
            # Minimize max_k (corr[k] + coeff[k]*(t-v)) over t in [0,1]
            # This is a convex 1D problem
            # Use bisection on subgradient
            def f_of_t(t):
                return np.max(corr + coeff * (t - v))
            
            # Golden section search
            gr = (np.sqrt(5) - 1) / 2  # golden ratio reciprocal
            lo, hi = 0.0, 1.0
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = f_of_t(c)
            fd = f_of_t(d)
            
            for _ in range(60):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = f_of_t(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = f_of_t(d)
            
            best_t = (lo + hi) / 2
            
            if abs(best_t - v) < 1e-10:
                continue
            
            # Apply change
            h[j] = best_t
            h = project_feasible(h, n)
            c5_new = compute_c5(h)
            
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                improved = True
            else:
                # Revert if not improved (projection might have helped or hurt)
                # Actually keep it if projection made things better for future iterations
                pass  # Keep the change anyway as it might help future coordinates
        
        if not improved:

import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_N_GLOBAL = 0
_BETA_GLOBAL = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            adj = diff / cnt
            h[mask] += adj
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - np.sum(h)
            if abs(diff2) > 1e-14:
                for j in range(n):
                    if h[j] > 0 and h[j] < 1:
                        h[j] = np.clip(h[j] + diff2, 0, 1)
                        break
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)

def smooth_max_obj(x):
    n = _N_GLOBAL
    beta = _BETA_GLOBAL
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def constraint_sum(x):
    return np.sum(x) - _N_GLOBAL / 2.0

def slsqp_refine(h, n, beta, maxiter):
    global _N_GLOBAL, _BETA_GLOBAL
    _N_GLOBAL = n
    _BETA_GLOBAL = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    try:
        res = minimize(smooth_max_obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def coord_descent_single(h, n, j, rng):
    """Optimize coordinate j exactly via 1D convex minimization."""
    v = h[j]
    dx = 2.0 / n
    corr = compute_corr(h)
    offsets = np.arange(2 * n - 1) - (n - 1)
    
    t1_mask = (j - offsets >= 0) & (j - offsets < n)
    t2_mask = (j + offsets >= 0) & (j + offsets < n)
    
    coeff = np.zeros(2 * n - 1)
    idx1 = j - offsets[t1_mask]
    coeff[t1_mask] = 1.0 - h[idx1]
    idx2 = j + offsets[t2_mask]
    coeff[t2_mask] -= h[idx2]
    
    base = corr - coeff * v
    
    # Golden section to minimize max(base + coeff * t) over [0,1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = np.max(base + coeff * c)
    fd = np.max(base + coeff * d)
    
    for _ in range(70):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = np.max(base + coeff * c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = np.max(base + coeff * d)
    
    best_t = (lo + hi) / 2.0
    return best_t

def coord_descent_sweep(h, n, rng, max_sweeps=500):
    """Full coordinate descent with exact 1D optimization per coordinate."""
    best_c5 = compute_c5(h)

            break
        if sweep > 20 and sweep % 50 == 0:
            break  # Diminishing returns
        order = rng.permutation(n)
        improved = False
        
        for j in order:
            if remaining_time() < 2:
                break
            v = h[j]
            best_t = coord_descent_single(h, n, j, rng)

            h[j] = best_t
            h = project_feasible(h, n)
            c5_new = compute_c5(h)

                improved = True
            else:
                h = old_h
        
        if not improved and sweep > 5:
            break
    
    return h, best_c5

def pair_coord_descent(h, n, rng, max_iter=2000):
    """Optimize pairs of coordinates simultaneously (2D convex problem)."""
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    offsets = np.arange(2 * n - 1) - (n - 1)
    
    for it in range(max_iter):
        if remaining_time() < 3:
            break
        j = rng.randint(n)
        m = rng.randint(n)
        if j == m:
            m = (m + 1) % n
        
        vj, vm = h[j], h[m]
        corr = compute_corr(h)
        
        # Coefficients for j and m
        t1j = (j - offsets >= 0) & (j - offsets < n)
        t2j = (j + offsets >= 0) & (j + offsets < n)
        cj = np.zeros(2 * n - 1)
        cj[t1j] = 1.0 - h[j - offsets[t1j]]
        cj[t2j] -= h[j + offsets[t2j]]
        
        t1m = (m - offsets >= 0) & (m - offsets < n)
        t2m = (m + offsets >= 0) & (m + offsets < n)
        cm = np.zeros(2 * n - 1)
        cm[t1m] = 1.0 - h[m - offsets[t1m]]
        cm[t2m] -= h[m + offsets[t2m]]
        
        base = corr - cj * vj - cm * vm
        
        def obj_2d(params):
            tj, tm = params
            return np.max(base + cj * tj + cm * tm)
        
        # Nelder-Mead on 2D
        x0 = np.array([vj, vm])
        best_x = x0.copy()
        best_val = obj_2d(x0)
        
        try:
            res = minimize(obj_2d, x0, method='Nelder-Mead',
                          options={'maxiter': 500, 'xatol': 1e-12, 'fatol': 1e-14,
                                   'adaptive': True})
            if res.fun < best_val - 1e-12:
                best_val = res.fun
                best_x = res.x
        except Exception:
            pass

            continue
        
        old_h = h.copy()

        new_vals = np.array([np.clip(best_x[0], 0, 1), np.clip(best_x[1], 0, 1)])
        h[j] = new_vals[0]
        h[m] = new_vals[1]
        h = project_feasible(h, n)
        c5_new = compute_c5(h)
        
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
    
    return h, best_c5

def binary_round_and_refine(h, n, rng):
    """Round to binary, then fine-tune free coordinates."""
    dx = 2.0 / n
    # Soft rounding: push values toward 0 or 1
    h_bin = h.copy()
    h_bin[h_bin < 0.5] = 0.0
    h_bin[h_bin >= 0.5] = 1.0
    
    # Fix sum: need exactly n/2 ones
    target = n / 2
    ones = np.sum(h_bin == 1)
    
    if ones > target:
        # Need to turn off (ones - target) coordinates
        # Prefer turning off those closest to 0.5
        one_idx = np.where(h_bin == 1)[0]
        sort_key = -h[one_idx]  # closest to 0.5 first (highest "soft" value means least committed)
        order = one_idx[np.argsort(sort_key)]
        need = int(ones - target)
        for idx in order[:need]:
            h_bin[idx] = 0.0
    elif ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        sort_key = h[zero_idx]  # closest to 0.5 first

        need = int(target - ones)

            h_bin[idx] = 1.0

    free = np.zeros(n, dtype=bool)

        if h_final[i] != h_final[(i+1) % n]:
            free[i] = True
            free[(i+1) % n] = True

        free_idx = np.where(free)[0]
        n_free = len(free_idx)

            h_full = h_bin.copy()
            h_full[free_idx] = xf
            h_full = np.clip(h_full, 0, 1)
            # Fix sum
            diff = target - np.sum(h_full)
            if abs(diff) > 1e-14:
                h_full[free_idx] += diff / n_free
                h_full = np.clip(h_full, 0, 1)
            return compute_c5(h_full)
        
        x0 = h_final[free_idx].copy()

            h_final = project_feasible(h_final, n)

            pass
    
    return h_final, compute_c5(h_final)

def subgrad_step(h, n):
    """Compute weighted subgradient from active lags."""
    corr = compute_corr(h)
    max_corr = np.max(corr)
    tol = max_corr * 0.01 + 0.01
    active = np.where(corr >= max_corr - tol)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    
    weights = np.maximum(corr[active] - (max_corr - 2.0 * tol), 0.0)
    if np.sum(weights) < 1e-15:
        weights = np.ones(len(active))
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for w, k_idx in zip(weights, active):
        offset = k_idx - (n - 1)
        if offset >= 0:
            n_pairs = n - offset
            if n_pairs > 0:
                grad[offset:offset+n_pairs] += w * (1.0 - h[:n_pairs])
                grad[:n_pairs] -= w * h[offset:offset+n_pairs]
        else:
            abs_off = -offset
            n_pairs = n - abs_off
            if n_pairs > 0:
                grad[:n_pairs] += w * (1.0 - h[abs_off:abs_off+n_pairs])
                grad[abs_off:abs_off+n_pairs] -= w * h[:n_pairs]
    
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return None
    return grad / gnorm
