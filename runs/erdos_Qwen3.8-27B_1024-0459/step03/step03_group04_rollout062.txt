```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def project_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):

        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def slp_solve(h_star, n, max_iters=500, tol=1e-12, solver='GLOP'):

    best_t = float('inf')
    best_h = h_star.copy()
    
    h_cur = h_star.copy()
    
    for it in range(max_iters):
        cons = list(base_cons)
        linear_coeffs = []
        constants = []
        
        for k in range(2 * n - 1):
            lag = k - (n - 1)
            if lag >= 0:
                idx = np.arange(n - lag)
                idx2 = idx + lag
            else:
                idx = np.arange(-lag, n)
                idx2 = idx + lag
            
            c_i = 1.0 - h_cur[idx2]
            c_j = -h_cur[idx]
            const = np.sum(h_cur[idx] * h_cur[idx2])

            lin[idx] += c_i
            lin[idx2] += c_j

        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, cons)
        
        try:
            prob.solve(solver=solver)
        except Exception:
            try:
                prob.solve(solver='CBC')
            except Exception:
                break
        
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        
        h_new = np.array(h_var.value, dtype=float)
        t_new = float(t_var.value)
```

```python
import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def project_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_project(h, n):
    h = np.array(h, dtype=float)
    idx = np.argsort(-h)  # top n/2 indices
    hb = np.zeros(n)
    hb[idx[:n//2]] = 1.0
    return hb

def slp_optimize(h_init, n, max_iters=500, tol=1e-10, damping=0.5, solver='GLOP'):
    dx = 2.0 / n
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)[0]
    h_cur = h_init.copy()
    
    for it in range(max_iters):
        # Build LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2.0]
        
        for k in range(2*n-1):
            lag = k - (n-1)
            coeff = np.zeros(n)
            const = 0.0
            n_valid = n - abs(lag)
            if lag >= 0:
                for i in range(n_valid):
                    a, b = i+lag, i
                    const += h_cur[a] * h_cur[b]
                    coeff[a] += (1.0 - h_cur[b])
                    coeff[b] += -h_cur[a]
            else:
                m = -lag
                for i in range(m, n):
                    a, b = i+lag, i  # a = i-m, b = i
                    const += h_cur[a] * h_cur[b]
                    coeff[a] += (1.0 - h_cur[b])
                    coeff[b] += -h_cur[a]
            cons.append(const + coeff @ h_var <= t_var / dx)
        
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=solver)
        except Exception:
            try:
                prob.solve(solver='CBC')
            except Exception:
                break
        
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        
        h_lp = np.array(h_var.value, dtype=float)
        t_lp = float(t_var.value)
        
        # Damped update
        h_new = damping * h_lp + (1-damping) * h_cur
        h_new = project_h(h_new, n)
        
        c5_new = compute_c5(h_new)[0]
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        if np.max(np.abs(h_new - h_cur)) < tol:
            break
        h_cur = h_new
        
        # Also check undamped
        h_undamped = project_h(h_lp, n)
        c5_und = compute_c5(h_undamped)[0]
        if c5_und < best_c5:
            best_c5 = c5_und
            best_h = h_undamped.copy()
    
    return best_h, best_c5
```

```python
import numpy as np
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def project_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def precompute_indices(n):
    """Precompute index arrays for all lags."""
    lag_info = []
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        m = abs(lag)
        if lag >= 0:
            idx_a = np.arange(m, n)  # h[i+lag] positions
            idx_b = np.arange(0, n - m)  # h[i] positions
        else:
            idx_a = np.arange(0, n - m)  # h[i+lag] = h[i-m] positions
            idx_b = np.arange(m, n)  # h[i] positions
        lag_info.append((idx_a, idx_b))
    return lag_info

def slp_optimize(h_init, n, max_iters=500, tol=1e-11, damping=0.5, solver='GLOP', t0=None, time_limit=None):
    dx = 2.0 / n
    lag_info = precompute_indices(n)
    
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)[0]
    h_cur = h_init.copy()
    
    for it in range(max_iters):
        if t0 is not None:
            remaining = time_limit - (time.time() - t0) if time_limit else float('inf')
            if remaining < 10:
                break
        
        # Build linearized constraints
        coeffs = np.zeros((2*n-1, n))
        consts = np.zeros(2*n-1)
        
        for k, (idx_a, idx_b) in enumerate(lag_info):
            ha = h_cur[idx_a]
            hb = h_cur[idx_b]
            consts[k] = np.sum(ha * hb)
            coeffs[k, idx_a] += (1.0 - hb)
            coeffs[k, idx_b] += -ha
        
        # Build LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n / 2.0
        ]
        for k in range(2*n-1):
            cons.append(consts[k] + coeffs[k] @ h_var <= t_var / dx)
        
        prob = cp.Problem(cp.Minimize(t_var), cons)
        solved = False
        for sol in [solver, 'CBC', 'GLPK']:
            try:
                prob.solve(solver=sol)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    solved = True
                    break
            except Exception:
                continue
        
        if not solved:
            break
        
        h_lp = np.array(h_var.value, dtype=float)
        t_lp = float(t_var.value)
        
        # Damped update
        h_new = damping * h_lp + (1.0 - damping) * h_cur
        h_new = project_h(h_new, n)
        
        c5_new = compute_c5(h_new)[0]
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Also check undamped
        h_undamped = project_h(h_lp, n)
        c5_und = compute_c5(h_undamped)[0]
        if c5_und < best_c5:
            best_c5 = c5_und
            best_h = h_undamped.copy()
            h_new = h_undamped
        
        if np.max(np.abs(h_new - h_cur)) < tol:
            h_cur = h_new
            break
        h_cur = h_new
    
    return best_h, best_c5
```

```python
    N = 2*n - 1
```

```python
import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def project_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def precompute_indices(n):
    lag_info = []
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        m = abs(lag)
        if lag >= 0:
            idx_a = np.arange(m, n)
            idx_b = np.arange(0, n - m)
        else:
            idx_a = np.arange(0, n - m)
            idx_b = np.arange(m, n)
        lag_info.append((idx_a, idx_b))
    return lag_info
```
