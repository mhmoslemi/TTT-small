def enforce_exact_sum(h, n):
    target = 0.5 * float(n)

    xq = (np.arange(dst_n) + 0.5) * (2.0 / float(dst_n))
    return np.interp(xq, xp, np.asarray(src, dtype=float))


def make_thue_morse(n):
    h = np.zeros(n, dtype=float)
    for i in range(n):
        if bin(i).count("1") % 2 == 1:
            h[i] = 1.0
    return h


def make_continuous_init(n, rng):
    t = (np.arange(n) + 0.5) / float(n)
    ph1 = rng.uniform(0.0, 2.0 * np.pi)
    ph2 = rng.uniform(0.0, 2.0 * np.pi)
    ph3 = rng.uniform(0.0, 2.0 * np.pi)
    f1 = rng.uniform(2.0, 4.0)
    f2 = rng.uniform(5.0, 9.0)
    f3 = rng.uniform(11.0, 17.0)
    a1 = rng.uniform(0.22, 0.38)
    a2 = rng.uniform(0.10, 0.22)
    a3 = rng.uniform(0.05, 0.15)
    h = 0.5 + a1 * np.sin(2.0 * np.pi * f1 * t + ph1)
    h += a2 * np.sin(2.0 * np.pi * f2 * t + ph2)
    h += a3 * np.sin(2.0 * np.pi * f3 * t + ph3)
    return np.clip(h, 0.0, 1.0)


def compute_q_values(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    M = 2 * n - 1
    q = np.empty(M, dtype=float)
    q[0] = c[n - 1]
    if n > 1:
        q[1::2] = c[:n - 1][::-1]
        q[2::2] = c[n:]
    return q


def compute_c5_fast(h):
    n = len(h)
    q = compute_q_values(h)
    return float(np.max(q)) * (2.0 / float(n))


def compute_c5_exact(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * (2.0 / float(n))


def compute_gradient_rows(h):
    n = len(h)
    M = 2 * n - 1
    G = np.zeros((n, M), dtype=float)
    if n > 0:
        G[:, 0] = 1.0 - 2.0 * h
    for s in range(1, n):
        L = n - s
        idxp = 1 + 2 * (s - 1)
        idxn = idxp + 1
        G[:L, idxp] += (1.0 - h[s:s + L])
        G[s:, idxp] += -h[:L]
        G[s:, idxn] += (1.0 - h[:L])
        G[:L, idxn] += -h[s:s + L]
    return G


def lbfgs_fun(x):
    n = _N
    dx = 2.0 / float(n)
    q = compute_q_values(x)
    mx = float(np.max(q))
    z = _BETA * (q - mx)
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        sm = mx
    else:
        sm = mx + float(np.log(s)) / _BETA
    val = sm * dx
    dev = float(np.sum(x) - _TGT)
    val += _PEN * dev * dev
    return float(val)


def lbfgs_grad(x):
    n = _N
    dx = 2.0 / float(n)
    q = compute_q_values(x)
    mx = float(np.max(q))
    z = _BETA * (q - mx)
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        w = np.full_like(q, 1.0 / float(q.size))
    else:
        w = e / s
    grad = np.zeros(n, dtype=float)
    w0 = float(w[0])
    if w0 > 1e-18:
        grad += w0 * (1.0 - 2.0 * x)
    for sh in range(1, n):
        L = n - sh
        idxp = 1 + 2 * (sh - 1)
        idxn = idxp + 1
        wp = float(w[idxp])
        wn = float(w[idxn])
        if wp > 1e-18:
            grad[:L] += wp * (1.0 - x[sh:sh + L])
            grad[sh:] += -wp * x[:L]
        if wn > 1e-18:
            grad[sh:] += wn * (1.0 - x[:L])
            grad[:L] += -wn * x[sh:sh + L]
    grad *= dx
    dev = float(np.sum(x) - _TGT)
    grad += 2.0 * _PEN * dev
    return grad


def lbfgs_phase(h, n, beta, end_time):
    h = project_to_sum_box(h, n)
    c0 = compute_c5_fast(h)
    sec = max(5.0, float(end_time - time.time()))
    if n <= 200:
        maxit = int(min(300, max(50, sec * 8)))
    elif n <= 350:
        maxit = int(min(180, max(40, sec * 5)))
    else:
        maxit = int(min(100, max(30, sec * 3)))
    maxit = max(30, maxit)
    _N = n
    _BETA = float(beta)
    _TGT = 0.5 * float(n)
    _PEN = 1000.0
    bounds = [(0.0, 1.0)] * n
    options = {
        "maxiter": maxit,
        "ftol": 1e-14,
        "gtol": 1e-10,
        "maxls": 50,
        "maxfun": maxit * 30,
        "iprint": 0,
    }
    try:
        res = minimize(lbfgs_fun, h, jac=lbfgs_grad, method="L-BFGS-B",
                       bounds=bounds, options=options)
        h2 = project_to_sum_box(np.asarray(res.x, dtype=float), n)
    except Exception:
        return h, c0
    c2 = compute_c5_fast(h2)
    if np.isfinite(c2):
        return h2, float(c2)
    return h, c0


def pair_value(delta, q, A, C):
    return float(np.max(q + A * delta + C * delta * delta))


def pair_feasible(tau, L, H, a0, b0, a1, b1,
                  a_pos, b_pos, a_neg, b_neg, max_b_const):
    tol = 1e-12
    l = L
    h = H
    if max_b_const > tau + tol:
        return False
    for k in range(a1.size):
        a = float(a1[k])
        b = float(b1[k])
        disc = a * a - 4.0 * (b - tau)
        if disc < -tol:
            return False
        if disc < 0.0:
            disc = 0.0
        sd = math.sqrt(disc)
        r1 = 0.5 * (-a - sd)
        r2 = 0.5 * (-a + sd)
        if r1 > l:
            l = r1
        if r2 < h:
            h = r2
        if l > h + tol:
            return False
    if a_pos.size:
        Rp = (tau - b_pos) / a_pos
        up = float(np.min(Rp))
        if up < h:
            h = up
    if a_neg.size:
        Rn = (tau - b_neg) / a_neg
        lo = float(np.max(Rn))
        if lo > l:
            l = lo
    if l > h + tol:
        return False
    D = a0 * a0 - 8.0 * (tau - b0)
    if D < -1e-10:
        return True
    if D < 0.0:
        D = 0.0
    sd = math.sqrt(D)
    r1 = 0.25 * (a0 - sd)
    r2 = 0.25 * (a0 + sd)
    if l <= r1 + tol:
        if min(h, r1) >= l:
            return True
    if max(l, r2) <= h:
        return True
    return False


def pair_intervals0(tau, L, H, a0, b0, a1, b1,
                    a_pos, b_pos, a_neg, b_neg, max_b_const):
    tol = 1e-12
    pieces = []
    if max_b_const > tau + tol:
        return pieces
    l = L
    h = H
    for k in range(a1.size):
        a = float(a1[k])
        b = float(b1[k])
        disc = a * a - 4.0 * (b - tau)
        if disc < -tol:
            return pieces
        if disc < 0.0:
            disc = 0.0
        sd = math.sqrt(disc)
        r1 = 0.5 * (-a - sd)
        r2 = 0.5 * (-a + sd)
        if r1 > l:
            l = r1
        if r2 < h:
            h = r2
        if l > h + tol:
            return pieces
    if a_pos.size:
        Rp = (tau - b_pos) / a_pos
        up = float(np.min(Rp))
        if up < h:
            h = up
    if a_neg.size:
        Rn = (tau - b_neg) / a_neg
        lo = float(np.max(Rn))
        if lo > l:
            l = lo
    if l > h + tol:
        return pieces
    D = a0 * a0 - 8.0 * (tau - b0)
    if D < -1e-10:
        if l <= h:
            pieces.append([l, h])
        return pieces
    if D < 0.0:
        D = 0.0
    sd = math.sqrt(D)
    r1 = 0.25 * (a0 - sd)
    r2 = 0.25 * (a0 + sd)
    if l <= r1 + tol:
        end = min(h, r1)
        if l <= end + tol:
            pieces.append([l, end])
    start = max(l, r2)
    if start <= h + tol:
        pieces.append([start, h])
    return pieces


def pair_optimize(h, q, G, i, j, n):
    M = len(q)
    hi = float(h[i])
    hj = float(h[j])
    L = max(-hi, hj - 1.0)
    H = min(1.0 - hi, hj)
    cur_max = float(np.max(q))
    if H <= L + 1e-15:
        return 0.0, cur_max

    A = G[i] - G[j]
    C = np.zeros(M, dtype=float)
    C[0] = -2.0
    d = abs(int(i) - int(j))
    idxp = -1
    idxn = -1
    if d > 0:
        idxp = 1 + 2 * (d - 1)
        idxn = idxp + 1
        C[idxp] = 1.0
        C[idxn] = 1.0

    valL = pair_value(L, q, A, C)
    valH = pair_value(H, q, A, C)
    high = max(cur_max, valL, valH) + 1e-12
    if high <= 1e-300:
        high = 1e-12

    zero_mask = np.ones(M, dtype=bool)
    zero_mask[0] = False
    if d > 0:
        zero_mask[idxp] = False
        zero_mask[idxn] = False
    a_zero = A[zero_mask]
    b_zero = q[zero_mask]
    posm = a_zero > 1e-15
    negm = a_zero < -1e-15
    a_pos = a_zero[posm]
    b_pos = b_zero[posm]
    a_neg = a_zero[ne gm] if False else a_zero[negm]
    b_neg = b_zero[negm]
    constm = ~(posm | negm)
    b_const = b_zero[constm]
    if b_const.size:
        max_b_const = float(np.max(b_const))
    else:
        max_b_const = -1e300

    if d > 0:
        a1 = np.array([float(A[idxp]), float(A[idxn])], dtype=float)
        b1 = np.array([float(q[idxp]), float(q[idxn])], dtype=float)
    else:
        a1 = np.empty(0, dtype=float)
        b1 = np.empty(0, dtype=float)
    a0 = float(A[0])
    b0 = float(q[0])

    if not pair_feasible(high, L, H, a0, b0, a1, b1,
                         a_pos, b_pos, a_neg, b_neg, max_b_const):
        ok = False
        for _ in range(30):
            high = high * 1.5 + 1.0
            if pair_feasible(high, L, H, a0, b0, a1, b1,
                             a_pos, b_pos, a_neg, b_neg, max_b_const):
                ok = True
                break
        if not ok:
            return 0.0, cur_max

    lo = 0.0
    hi2 = high
    for _ in range(45):
        mid = 0.5 * (lo + hi2)
        if pair_feasible(mid, L, H, a0, b0, a1, b1,
                         a_pos, b_pos, a_neg, b_neg, max_b_const):
            hi2 = mid
        else:
            lo = mid
        if hi2 - lo <= 1e-12 * max(1.0, hi2):
            break
    tau = hi2

    cand = [L, H, 0.0]
    for k in range(a1.size):
        cand.append(-0.5 * float(a1[k]))
    D = a0 * a0 - 8.0 * (tau - b0)
    if D >= -1e-10:
        if D < 0.0:
            D = 0.0
        sd = math.sqrt(D)
        cand.append(0.25 * (a0 - sd))
        cand.append(0.25 * (a0 + sd))
    pieces = pair_intervals0(tau, L, H, a0, b0, a1, b1,
                             a_pos, b_pos, a_neg, b_neg, max_b_const)
    for p in pieces:
        cand.append(p[0])
        cand.append(p[1])
        cand.append(0.5 * (p[0] + p[1]))

    best_delta = 0.0
    best_f = 1e300
    for dv in cand:
        if not np.isfinite(dv):
            continue
        if dv < L:
            dv = L
        elif dv > H:
            dv = H
        fv = pair_value(dv, q, A, C)
        if fv < best_f:
            best_f = fv
            best_delta = dv
    if best_f > cur_max + 1e-12:
        return 0.0, cur_max
    if abs(best_delta) < 1e-15:
        return 0.0, cur_max
    return float(best_delta), float(best_f)


def generate_candidates(h, q, n, limit, rng):
    active = np.argsort(q)[-12:]
    scores = {}
    top_contrib = 12

    for idx in active:
        idx = int(idx)
        if idx == 0:
            continue
        if idx % 2 == 1:
            s = (idx + 1) // 2
            L = n - s
            if L <= 0:
                continue
            contrib = h[:L] * (1.0 - h[s:s + L])
            if L > top_contrib:
                pos = np.argpartition(-contrib, top_contrib)[:top_contrib]
            else:
                pos = np.arange(L)
            for aa in pos:
                a = int(aa)
                b = a + s
                if a == b:
                    continue
                key = (a, b) if a < b else (b, a)
                sc = float(contrib[aa])
                if sc > 1e-15:
                    scores[key] = scores.get(key, 0.0) + sc
        else:
            s = idx // 2
            L = n - s
            if L <= 0:
                continue
            contrib = h[s:s + L] * (1.0 - h[:L])
            if L > top_contrib:
                pos = np.argpartition(-contrib, top_contrib)[:top_contrib]
            else:
                pos = np.arange(L)
            for aa in pos:
                a = int(aa)
                b = a + s
                if a == b:
                    continue
                key = (a, b) if a < b else (b, a)
                sc = float(contrib[aa])
                if sc > 1e-15:
                    scores[key] = scores.get(key, 0.0) + sc

    order = np.argsort(h)
    lw = 15
    lows = order[:lw]
    highs = order[::-1][:lw]
    for li in lows:
        li = int(li)
        for hi in highs:
            hi = int(hi)
            if li == hi:
                continue
            if h[hi] <= h[li]:
                continue
            key = (li, hi) if li < hi else (hi, li)
            sc = 0.25 * float(min(h[li], 1.0 - h[hi]))
            if sc > 1e-15:
                scores[key] = scores.get(key, 0.0) + sc

    nrand = 40
    for _ in range(nrand):
        a = int(rng.randint(n))
        b = int(rng.randint(n))
        if a == b:
            continue
        key = (a, b) if a < b else (b, a)
        sc = 0.05 * float(abs(h[a] - h[b]))
        if sc > 0.0:
            scores[key] = scores.get(key, 0.0) + sc

    arr = [(sc, k) for k, sc in scores.items()]
    arr.sort(reverse=True)
    return [k for sc, k in arr[:limit]]


def random_pair_polish(h, cur_max, q, G, end_time, rng):
    n = len(h)
    dx = 2.0 / float(n)
    best_h = h.copy()
    best_q = cur_max
    best_c = cur_max * dx
    no_imp = 0
    trial = 0
    lows = None
    highs = None

    while time.time() < end_time:
        if trial % 100 == 0:
            order = np.argsort(h)
            m = min(100, n // 2)
            if m <= 1:
                break
            lows = order[:m]
            highs = order[::-1][:m]

        mode = int(rng.randint(3))
        if mode < 2 and lows is not None:
            a = int(lows[int(rng.randint(lows.size))])
            b = int(highs[int(rng.randint(highs.size))])
        else:
            a = int(rng.randint(n))
            b = int(rng.randint(n))

        if a == b:
            trial += 1
            continue

        delta, local = pair_optimize(h, q, G, a, b, n)
        if local < cur_max - 1e-12:
            nh = h.copy()
            nh[a] += delta
            nh[b] -= delta
            nh = np.clip(nh, 0.0, 1.0)
            nh = enforce_exact_sum(nh, n)
            nq = compute_q_values(nh)
            nm = float(np.max(nq))
            if nm < cur_max - 1e-14:
                h = nh
                q = nq
                G = compute_gradient_rows(h)
                cur_max = nm
                best_h = h.copy()
                best_c = cur_max * dx
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        if no_imp > 300:
            break
        trial += 1

    return h, cur_max, q, G, best_h, best_c


def pair_search(h, end_time, rng=None, cand_limit=0):
    if rng is None:
        rng = np.random.RandomState(987654321)
    n = len(h)
    dx = 2.0 / float(n)
    h = project_to_sum_box(h, n)
    q = compute_q_values(h)
    cur_max = float(np.max(q))
    G = compute_gradient_rows(h)
    best_h = h.copy()
    best_c = cur_max * dx

    if cand_limit == 0:
        if n <= 200:
            cl = 250
        elif n <= 320:
            cl = 180
        else:
            cl = 120
    else:
        cl = cand_limit

    while time.time() < end_time - 20.0:
        cands = generate_candidates(h, q, n, cl, rng)
        results = []
        for pair in cands:
            a, b = pair
            delta, local = pair_optimize(h, q, G, a, b, n)
            if local < cur_max - 1e-13:
                results.append((local, delta, pair))
            if time.time() > end_time - 5.0:
                break

        results.sort()
        improved = False
        for local, delta, pair in results[:5]:
            a, b = pair
            old_h = h.copy()
            h[a] += delta
            h[b] -= delta
            h = np.clip(h, 0.0, 1.0)
            h = enforce_exact_sum(h, n)
            nq = compute_q_values(h)
            nm = float(np.max(nq))
            if nm < cur_max - 1e-14:
                cur_max = nm
                q = nq
                G = compute_gradient_rows(h)
                best_h = h.copy()
                best_c = cur_max * dx
                improved = True
                break
            else:
                h = old_h

        if not improved:
            rand_end = min(end_time - 10.0, time.time() + 90.0)
            if rand_end > time.time():
                h2, cur2, q2, G2, bh2, bc2 = random_pair_polish(
                    h, cur_max, q, G, rand_end, rng)
                if bc2 < best_c - 1e-14:
                    h = h2
                    q = q2
                    G = G2
                    cur_max = cur2
                    best_h = bh2
                    best_c = bc2
                    improved = True

        if not improved:
            break

    return best_h, best_c


def optimize_candidate(h, n, end_time, rng):
    h = project_to_sum_box(h, n)
    c = compute_c5_fast(h)
    best_h = h.copy()
    best_c = c
    c_current = c

    quick = 120.0 if n <= 300 else 60.0
    quick_end = min(end_time - 30.0, time.time() + quick)
    if quick_end > time.time() + 15.0:
        h2, c2 = pair_search(h, quick_end, rng)
        h = h2
        c_current = c2
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2

    if time.time() < end_time - 80.0:
        for beta in (50.0, 250.0, 900.0):
            if time.time() > end_time - 80.0:
                break
            rem = end_time - time.time() - 80.0
            if rem < 15.0:
                break
            btime = max(15.0, min(40.0, 0.5 * rem))
            beta_end = time.time() + btime
            h3, c3 = lbfgs_phase(h, n, beta, beta_end)
            if c3 < best_c:
                best_h = h3.copy()
                best_c = c3
                h = h3
                c_current = c3
            elif c3 < c_current + 0.02:
                h = h3
                c_current = c3

    if time.time() < end_time - 5.0:
        h4, c4 = pair_search(h, end_time, rng)
        if c4 < best_c:
            best_h = h4.copy()
            best_c = c4

    best_h = project_to_sum_box(best_h, n)
    cfinal = compute_c5_fast(best_h)
    if cfinal < best_c:
        best_c = cfinal
    return best_h, float(best_c)


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float("inf")
    best_n = 160

    try:
        parent = np.asarray(initial_h_values, dtype=float)
        n0 = int(len(parent))
        h0 = project_to_sum_box(parent, n0)
        c0 = compute_c5_exact(h0)
        best_h = h0.copy()
        best_c5 = float(c0)
        best_n = n0
    except Exception:
        n0 = 160
        h0 = make_continuous_init(n0, rng)
        h0 = project_to_sum_box(h0, n0)
        c0 = compute_c5_exact(h0)
        best_h = h0.copy()
        best_c5 = float(c0)
        best_n = n0

    base_n = int(best_n)
    raw = [base_n]
    for d in (-40, -32, -24, -16, -8, -4, -2, -1, 1, 2, 4, 8, 16, 24, 32, 48, 64, 96, 128):
        raw.append(base_n + d)
    raw.extend([128, 256, 320, 512])

    ns = []
    for x in raw:
        x = int(x)
        if x < 80 or x > 512:
            continue
        if x not in ns:
            ns.append(x)

    sort_info = [[abs(x - base_n), i] for i, x in enumerate(ns)]
    sort_info.sort()
    ns = [ns[i] for _, i in sort_info[:8]]

    for n in ns:
        if remaining_time() < 60.0:
            break
        if n == best_n:
            alloc = min(750.0, max(120.0, remaining_time() - 300.0))
        else:
            alloc = min(180.0, max(60.0, remaining_time() - 200.0))
        if alloc < 35.0:
            alloc = remaining_time() - 20.0
        if alloc < 30.0:
            break
        end_time = min(_START + _BUDGET - 30.0, time.time() + alloc)
        if n == best_n and best_h is not None:
            hcur = best_h.copy()
        else:
            hcur = interp_to(best_h, best_n, n)
            hcur = project_to_sum_box(hcur, n)

        h2, c2 = optimize_candidate(hcur, n, end_time, rng)
        if c2 < best_c5 - 1e-14:
            best_h = h2.copy()
            best_c5 = float(c2)
            best_n = int(n)

    if best_h is None:
        n = 160
        best_h = make_continuous_init(n, rng)
        best_h = project_to_sum_box(best_h, n)
        best_c5 = compute_c5_exact(best_h)
        best_n = n

    best_h = enforce_exact_sum(best_h, best_n)
    final_c5 = compute_c5_exact(best_h)
    try:
        ev_c5 = evaluate_erdos_solution(best_h)
        if np.isfinite(ev_c5) and float(ev_c5) < final_c5:
            final_c5 = float(ev_c5)
    except Exception:
        pass

    best_h = enforce_exact_sum(best_h, best_n)
    return (best_h, float(final_c5), int(best_n))
