def linearized_step(h0, n, max_offsets=50):
    """Solve LP to find a descent direction for the max-correlation objective."""
    import cvxpy as cp
    
    dx = 2.0 / n
    corr0 = np.correlate(h0, 1-h0, mode='full')
    
    # Compute gradients of C[k] for all k
    # Only use top max_offsets shifts to keep LP small
    top_idx = np.argsort(corr0)[-max_offsets:]
    
    h = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2.0]
    
    for j in top_idx:
        s = j - (n - 1)
        # Compute grad C[k] / h at h0
        g = np.zeros(n)
        if s >= 0:
            for m in range(s, n):
                g[m] += (1.0 - h0[m - s])
            for m in range(0, n - s):
                g[m] -= h0[m + s]
        else:
            sp = -s
            for m in range(0, n - sp):
                g[m] += (1.0 - h0[m + sp])
            for m in range(sp, n):
                g[m] -= h0[m - sp]
        
        # Linearize: C[k](h) ≈ C[k](h0) + g · (h - h0)
        # We want this <= t / dx
        lhs = corr0[j] + g @ (h - h0)
        constraints.append(lhs <= t / dx)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.ECOS)
    except:
        try:
            prob.solve(solver=cp.SCS)  # or whatever is available
        except:
            return None, None
    
    if h.value is None:
        return None, None
    
    return h.value, t.value

        s = j - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            for m in range(s, n):
                g[m] += (1.0 - h0[m - s])
            for m in range(0, n - s):
                g[m] -= h0[m + s]
        else:
            sp = -s
            for m in range(0, n - sp):
                g[m] += (1.0 - h0[m + sp])
            for m in range(sp, n):
                g[m] -= h0[m - sp]

        c_k = corr0[j]
        # g.h <= t/dx - (c_k - g.h0) → g.h - t/dx <= c_k - g.h0
        row = np.concatenate([g, [-1.0 / dx]])
        rhs = c_k - g.dot(h0)
        A_rows.append(row)
        b_rows.append(rhs)

    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)

    # Equality: sum(h) = n/2, and t is free (>= 0)
    A_eq = np.array([[np.ones(n).tolist() + [0.0]]])
    b_eq = np.array([n / 2.0])

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            return h0.copy()
        h_lp = res.x[:n]
    except Exception:
        return h0.copy()

    # Line search
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
        h_trial = (1 - alpha) * h0 + alpha * h_lp
        h_trial = project_feasible(h_trial, n)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()

    return best_h


def guided_swap(h, n, top_k=2, max_rounds=200, rng=None):
    """Guided pairwise swaps targeting active shifts."""
    if rng is None:
        rng = np.random.RandomState(42)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    stall = 0

    for rnd in range(max_rounds):
        if remaining_time() < 3:
            break
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[-top_k:]

        ones = np.where(h_cur > 0.5)[0]
        zeros = np.where(h_cur <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        improved = False
        n_attempts = min(500, len(ones) * len(zeros))
        for _ in range(n_attempts):
            oi = rng.randint(len(ones))
            oz = rng.randint(len(zeros))
            i = ones[oi]
            j = zeros[oz]
            h_cur[i] = 1.0 - h_cur[i]
            h_cur[j] = 1.0 - h_cur[j]
            h_feas = project_feasible(h_cur, n)
            c5 = compute_c5(h_feas)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h_feas.copy()
                h_cur = h_feas.copy()
                ones = np.where(h_cur > 0.5)[0]
                zeros = np.where(h_cur <= 0.5)[0]
                improved = True
                stall = 0
                break
            else:
                h_cur[i] = 1.0 - h_cur[i]
                h_cur[j] = 1.0 - h_cur[j]

        if not improved:
            stall += 1
            if stall > 5:
                break

    return best_h, best_c5


def coordinate_descent(h, n, sweeps=5, rng=None):
    """Try changing each coordinate to several values."""
    if rng is None:
        rng = np.random.RandomState(7)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()

    for _ in range(sweeps):
        if remaining_time() < 4:
            break
        order = rng.permutation(n)
        any_imp = False
        for i in order:
            if remaining_time() < 2:
                break
            old_val = h_cur[i]
            best_local_c5 = best_c5
            best_local_h = h_cur.copy()

            candidates = [0.0, 1.0, 0.5, 0.1, 0.9]
            if 0.1 < old_val < 0.4:
                candidates.append(0.2)
            elif 0.6 < old_val < 0.9:
                candidates.append(0.8)
            candidates.append(max(0.0, old_val - 0.1))
            candidates.append(min(1.0, old_val + 0.1))

            for nv in candidates:
                if abs(nv - old_val) < 1e-8:
                    continue
                h_cur[i] = nv
                h_feas = project_feasible(h_cur, n)
                c5_test = compute_c5(h_feas)
                if c5_test < best_local_c5:
                    best_local_c5 = c5_test
                    best_local_h = h_feas.copy()

            h_cur = best_local_h.copy()
            if best_local_c5 < best_c5:
                best_c5 = best_local_c5
                best_h = h_cur.copy()
                any_imp = True

        if not any_imp:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _N, _BETA, _TGT, _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Phase 0: Initialize
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.7f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 160
        best_h = make_thue_morse(best_n)
        best_c5 = compute_c5(best_h)
        print("No parent. Thue-Morse: n={}, C5={:.7f}".format(best_n, best_c5))

    # Phase 1: L-BFGS-B at parent resolution with multiple betas
    n = best_n
    _N = n
    _TGT = n / 2.0
    bounds = [(0.0, 1.0)] * n
    print("\n=== Phase 1: L-BFGS-B at n={} ===".format(n))

    h_work = best_h.copy()
    for beta_val in [100, 300, 800]:
        if remaining_time() < 30:
            break
        _BETA = float(beta_val)
        try:
            res = minimize(smooth_max_obj, h_work, jac=smooth_max_grad,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 1000, 'ftol': 1e-15,
                                    'gtol': 1e-10, 'iprint': 0,
                                    'maxls': 80, 'maxfun': 10000})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_work = h_new.copy()
                print("  beta={} => C5={:.7f}".format(beta_val, c5_new))
        except Exception as e:
            print("  L-BFGS-B beta={} failed: {}".format(beta_val, str(e)[:50]))

    # Phase 2: Multi-resolution L-BFGS-B
    print("\n=== Phase 2: Multi-resolution ===")
    n_cands = [n, 200, 240, 320]
    n_cands = sorted(set(nn for nn in n_cands if 120 <= nn <= 400 and nn != n))
    n_cands = [n] + n_cands

    for nn in n_cands[1:]:
        if remaining_time() < 60:
            break
        _N = nn
        _TGT = nn / 2.0
        bounds_n = [(0.0, 1.0)] * nn
        print("  n={}".format(nn))

        h_interp = project_feasible(interp_to(best_h, best_n, nn), nn)
        c5_interp = compute_c5(h_interp)

        for beta_val in [200, 500]:
            if remaining_time() < 20:
                break
            _BETA = float(beta_val)
            try:
                res = minimize(smooth_max_obj, h_interp, jac=smooth_max_grad,
                               method='L-BFGS-B', bounds=bounds_n,
                               options={'maxiter': 600, 'ftol': 1e-15,
                                        'gtol': 1e-10, 'iprint': 0,
                                        'maxls': 50, 'maxfun': 6000})
                h_new = project_feasible(res.x, nn)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = nn
                    h_interp = h_new.copy()
                    print("    n={}, beta={} => C5={:.7f}".format(nn, beta_val, c5_new))
            except Exception:
                pass

    # Phase 3: LP cutting-plane refinement
    print("\n=== Phase 3: LP cutting plane (n={}) ===".format(best_n))
    n = best_n
    _N = n
    _TGT = n / 2.0
    h_lp = best_h.copy()

    for lp_iter in range(40):
        if remaining_time() < 15:
            break
        h_try = lp_cutting_plane_step(h_lp, n, top_m=15)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            h_lp = h_try.copy()
            if lp_iter % 5 == 4:
                print("  LP iter {}: C5={:.7f}".format(lp_iter, c5_try))
        else:
            # Try L-BFGS-B from current point
            if remaining_time() < 15:
                break
            _BETA = 500.0
            try:
                res = minimize(smooth_max_obj, h_lp, jac=smooth_max_grad,
                               method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                               options={'maxiter': 200, 'ftol': 1e-15,
                                        'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_lp = h_new.copy()
                    print("  LP+LBFGS iter {}: C5={:.7f}".format(lp_iter, c5_new))
                else:
                    break
            except Exception:
                break

    # Phase 4: Guided pairwise swaps + coordinate descent
    print("\n=== Phase 4: Local search (n={}) ===".format(best_n))
    n = best_n
    h_sw, c5_sw = guided_swap(best_h, n, top_k=2, max_rounds=300, rng=rng)
    if c5_sw < best_c5:
        best_c5 = c5_sw
        best_h = h_sw.copy()
        print("  Swap: C5={:.7f}".format(c5_sw))

    if remaining_time() > 15:
        h_cd, c5_cd = coordinate_descent(best_h, n, sweeps=4, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            print("  CoordDesc: C5={:.7f}".format(c5_cd))

    # Phase 5: Random restarts at current resolution
    print("\n=== Phase 5: Restart + polish (n={}) ===".format(best_n))
    n = best_n
    _N = n
    _TGT = n / 2.0
    bounds = [(0.0, 1.0)] * n

    if remaining_time() > 40:
        # Try interpolating best to higher res and back
        for nn in [240, 200, 320]:
            if remaining_time() < 30:
                break
            if nn == n:
                continue
            _N = nn
            _TGT = nn / 2.0
            bounds_n = [(0.0, 1.0)] * nn
            h_int = project_feasible(interp_to(best_h, best_n, nn), nn)
            for beta_val in [400, 1000]:
                if remaining_time() < 15:
                    break
                _BETA = float(beta_val)
                try:
                    res = minimize(smooth_max_obj, h_int, jac=smooth_max_grad,
                                   method='L-BFGS-B', bounds=bounds_n,
                                   options={'maxiter': 400, 'ftol': 1e-15,
                                            'iprint': 0})
                    h_new = project_feasible(res.x, nn)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        best_n = nn
                        print("  Resample n={}, beta={}: C5={:.7f}".format(nn, beta_val, c5_new))
                except Exception:
                pass
            _N = best_n
            _TGT = best_n / 2.0

    # Phase 6: Final high-beta polish
    print("\n=== Phase 6: Final polish (n={}) ===".format(best_n))
    n = best_n
    _N = n
    _TGT = n / 2.0
    bounds = [(0.0, 1.0)] * n

    for beta_val in [800, 1500, 3000]:
        if remaining_time() < 8:
            break
        _BETA = float(beta_val)
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 800, 'ftol': 1e-16,
                                    'gtol': 1e-12, 'iprint': 0,
                                    'maxls': 100, 'maxfun': 20000})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  Polish beta={}: C5={:.7f}".format(beta_val, c5_new))
        except Exception:
            pass

    # Additional guided swaps if time allows
    if remaining_time() > 10:
        h_sw, c5_sw = guided_swap(best_h, n, top_k=3, max_rounds=200, rng=rng)
        if c5_sw < best_c5:
            best_c5 = c5_sw
            best_h = h_sw.copy()
            print("  Final swap: C5={:.7f}".format(c5_sw))

    # Additional LP step
    if remaining_time() > 8:
        h_lp = lp_cutting_plane_step(best_h, n, top_m=25)
        c5_lp = compute_c5(h_lp)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()
            print("  Final LP: C5={:.7f}".format(c5_lp))

    # Ensure feasibility
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
