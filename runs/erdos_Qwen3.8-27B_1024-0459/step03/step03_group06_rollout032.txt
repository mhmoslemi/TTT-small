def fast_topk_gradient(h, top_k=10):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Get top-k indices
    top_idx = np.argsort(corr)[-top_k:]
    top_corr = corr[top_idx]
    
    # Softmax weights
    beta = 50.0
    m = np.max(top_corr)
    weights = np.exp(beta * (top_corr - m))
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for j, wj in zip(top_idx, weights):
        s = j - (n - 1)
        if s >= 0:
            # dc/dh[m] = (1-h[m-s]) for m in [s, n-1] minus h[m+s] for m in [0, n-1-s]
            if s == 0:
                # c = sum h[i](1-h[i]) = sum h[i] - sum h[i]^2
                # dc/dh[m] = 1 - 2h[m]
                grad += wj * (1.0 - 2.0 * h)
            else:
                # Part 1: (1-h[m-s]) for m in [s, n-1]
                part1 = np.zeros(n)
                part1[s:] = 1.0 - h[:n-s]
                # Part 2: -h[m+s] for m in [0, n-1-s]
                part2 = np.zeros(n)
                part2[:n-s] = -h[s:]
                grad += wj * (part1 + part2)
        else:
            sp = -s
            # dc/dh[m] = (1-h[m+sp]) for m in [0, n-1-sp] minus h[m-sp] for m in [sp, n-1]
            if sp == 0:
                grad += wj * (1.0 - 2.0 * h)
            else:
                part1 = np.zeros(n)
                part1[:n-sp] = 1.0 - h[sp:]
                part2 = np.zeros(n)
                part2[sp:] = -h[:n-sp]
                grad += wj * (part1 + part2)
    
    grad *= dx
    # Add constraint gradient (sum h = n/2)
    grad += 2.0 * 500.0 * (np.sum(h) - n / 2.0)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_G_K = 10
_G_BETA = 50.0
_G_PEN = 500.0
_G_N = 160


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 2.0)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        order = np.argsort(np.abs(h - 0.5))
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h


def obj_func(h):
    n = _G_N
    dx = 2.0 / n
    k = _G_K
    beta = _G_BETA
    pen = _G_PEN
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-k:]
    top_vals = corr[top_idx]
    m = np.max(top_vals)
    sm = m + np.log(np.sum(np.exp(beta * (top_vals - m)))) / beta
    penalty = pen * (np.sum(h) - n / 2.0) ** 2
    return sm * dx + penalty


def obj_grad(h):
    n = _G_N
    dx = 2.0 / n
    k = _G_K
    beta = _G_BETA
    pen = _G_PEN
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-k:]
    top_vals = corr[top_idx]
    m = np.max(top_vals)
    weights = np.exp(beta * (top_vals - m))
    weights /= np.sum(weights)

    grad = np.zeros(n)
    for j, wj in zip(top_idx, weights):
        if wj < 1e-14:
            continue
        s = j - (n - 1)
        if s == 0:
            grad += wj * (1.0 - 2.0 * h)
        elif s > 0:
            part1 = np.zeros(n)
            part1[s:] = 1.0 - h[:n - s]
            part2 = np.zeros(n)
            part2[:n - s] = -h[s:]
            grad += wj * (part1 + part2)
        else:
            p = -s
            part1 = np.zeros(n)
            part1[:n - p] = 1.0 - h[p:]
            part2 = np.zeros(n)
            part2[p:] = -h[:n - p]
            grad += wj * (part1 + part2)

    grad *= dx
    grad += 2.0 * pen * (np.sum(h) - n / 2.0)
    return grad


def init_random(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return project_feasible(h, n)


def init_sin(n, rng, freq=3):
    t = np.arange(n) / float(n)
    h = 0.5 + 0.4 * np.sin(2.0 * np.pi * freq * t + rng.rand() * 6.283)
    return project_feasible(np.clip(h, 0, 1), n)


def init_multisin(n, rng):
    t = np.arange(n) / float(n)
    h = 0.5
    for _ in range(3):
        h += rng.uniform(0.1, 0.2) * np.sin(2.0 * np.pi * rng.randint(2, 8) * t + rng.rand() * 6.283)
    return project_feasible(np.clip(h, 0, 1), n)


def binary_greedy(h, n, max_iters=50000, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    dx = 2.0 / n
    h = np.where(h > 0.5, 1.0, 0.0)
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    target = n // 2
    while len(ones) > target:
        i = rng.randint(len(ones))
        h[ones[i]] = 0.0
        ones[i], zeros.append(ones.pop(i))
        zeros.append(h[ones[i]])
    while len(ones) < target:
        i = rng.randint(len(zeros))
        h[zeros[i]] = 1.0
        zeros[i], ones.append(zeros.pop(i))
        ones.append(h[zeros[i]])

    h = np.where(h > 0.5, 1.0, 0.0)
    ones = list(np.where(h == 1.0)[0].tolist())
    zeros = list(np.where(h == 0.0)[0].tolist())
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    best_c5 = c5
    stall = 0

    for it in range(max_iters):
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h[a] = 0.0
        h[b] = 1.0
        nc5 = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
        if nc5 < c5 - 1e-15:
            c5 = nc5
            ones[ones.index(a)] = b
            zeros[zeros.index(b)] = a
            stall = 0
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[a] = 1.0
            h[b] = 0.0
            stall += 1
            if stall > 5000:
                break
    return best_h, best_c5


def binary_greedy_fast(h, n, max_iters=100000, rng=None):
    if rng is None:

    dx = 2.0 / n
    h = np.where(h > 0.5, 1.0, 0.0).astype(float)

        h[zeros_idx[rng.randint(len(zeros_idx))]] = 1.0
