def project_feasible(y, n):
    y = np.asarray(y, dtype=float).copy()
    if y.size != n: y = y[:n]
    y = np.clip(y, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(50):
        s = float(np.sum(y))
        diff = target - s
        if abs(diff) < 1e-13: break
        if diff > 0:
            mask = y < 1.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                y[:] = 1.0
                diff = target - n
                # impossible if target<=n; if target==n, done
                if abs(diff)<1e-12: break
                # fallback set all target/n? but bounds
                y[:] = target / n
                break
            y[mask] += diff / cnt
        else:
            mask = y > 0.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                y[:] = 0.0
                diff = target
                if abs(diff)<1e-12: break
                y[:] = target / n
                break
            y[mask] += diff / cnt
    y = np.clip(y, 0.0, 1.0)
    # final correction
    diff = target - float(np.sum(y))
    if abs(diff) > 1e-14:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():
            y[mask] += diff / int(np.sum(mask))
        else:
            i = int(np.argmin(np.abs(y - 0.5)))
            y[i] += diff
        y = np.clip(y, 0.0, 1.0)
    return y

def smooth_obj_grad(x):
    n=_CN

    omh=1.0-x
    corr=np.correlate(x, omh, mode='full')
    m=float(np.max(corr))
    y=np.clip(_BETA*(corr-m), -700.0, 0.0)
    e=np.exp(y)
    sume=e.sum()
    if sume <=0: sume=1.0
    f=dx*(m + np.log(sume)/_BETA)
    s=float(np.sum(x))
    diff=s-_TGT
    if _PEN:
       f += _PEN*diff*diff
    w=e/sume
    active=np.where(w>1e-8)[0]
    g=np.zeros(n)

       d=int(idx)-center

           g[:n-d] += wj*omh[d:]

    if _PEN:

    return f, g

def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 0.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)


def project_feasible(y, n):
    y = np.asarray(y, dtype=float).copy()
    if y.size != n:
        if y.size > n:
            y = y[:n].copy()
        else:
            y = np.concatenate([y, np.zeros(max(0, n - y.size))])
    y = np.clip(y, 0.0, 1.0)
    target = float(n) / 2.0

    for _ in range(80):
        s = float(np.sum(y))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            mask = y < 1.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                y[:] = 0.5
                break
            y[mask] += diff / cnt
        else:
            mask = y > 0.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                y[:] = 0.5
                break
            y[mask] += diff / cnt

    y = np.clip(y, 0.0, 1.0)
    diff = target - float(np.sum(y))
    if abs(diff) > 1e-13:

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return np.asarray(src_h, dtype=float).copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    xnew = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(xnew, xp, np.asarray(src_h, dtype=float))


def make_thue_morse(n):
    h = np.zeros(n, dtype=float)
    for i in range(n):
        if bin(i).count('1') & 1:
            h[i] = 1.0
    return h


def make_golden(n, alpha, beta):
    if n <= 0:
        return np.zeros(0)
    t = (np.arange(n) * float(alpha) + float(beta)) % 1.0
    return (t < 0.5).astype(float)

    f1 = float(rng.randint(2, 9))
    f2 = float(rng.randint(3, 13))
    ph1 = float(rng.uniform(0.0, 2.0 * math.pi))
    ph2 = float(rng.uniform(0.0, 2.0 * math.pi))
    h = (0.5
         + 0.22 * np.sin(2.0 * math.pi * f1 * t + ph1)
         + 0.12 * np.sin(2.0 * math.pi * f2 * t + ph2)
         + 0.08 * rng.normal(0.0, 1.0, n))
    return project_feasible(np.clip(h, 0.0, 1.0), n)


def make_start_candidates(h_best, best_n, n, rng):
    out = []
    if n <= 0:
        return out
    if h_best is not None and best_n > 0:
        base = interp_to(h_best, best_n, n)
        base = project_feasible(base, n)
        out.append(base)
        out.append(project_feasible(base + rng.normal(0.0, 0.03, n), n))
        out.append(project_feasible(base + rng.normal(0.0, 0.12, n), n))
    out.append(project_feasible(make_thue_morse(n), n))
    out.append(project_feasible(make_golden(n, 0.6180339887, float(rng.rand())), n))
    out.append(project_feasible(make_golden(n, 0.4142135623, float(rng.rand())), n))
    if n % 2 == 0:
        out.append(project_feasible(make_random_binary(n, rng), n))
    out.append(make_sine(n, rng))
    return out


def make_binary_from_h(h0, n, rng):
    target = n // 2
    if n <= 0:
        return np.zeros(0, dtype=np.int8)
    if h0 is None or len(h0) != n:
        h = np.zeros(n, dtype=np.int8)
        if target > 0:
            perm = rng.permutation(n)
            h[perm[:target]] = 1
        return h

    hf = project_feasible(np.asarray(h0, dtype=float), n)
    h = (hf >= 0.5).astype(np.int8)
    cnt = int(np.sum(h))

    if cnt > target:
        ones = np.where(h == 1)[0]
        order = np.argsort(hf[ones])
        h[ones[order[:cnt - target]]] = 0
    elif cnt < target:
        zeros = np.where(h == 0)[0]
        order = np.argsort(hf[zeros])[::-1]
        h[zeros[order[:target - cnt]]] = 1

    cnt = int(np.sum(h))
    while cnt > target:
        ones = np.where(h == 1)[0]
        if len(ones) == 0:
            break
        h[int(ones[int(rng.randint(len(ones)))])] = 0
        cnt -= 1
    while cnt < target:
        zeros = np.where(h == 0)[0]
        if len(zeros) == 0:
            break
        h[int(zeros[int(rng.randint(len(zeros)))])] = 1
        cnt += 1
    return h


def make_binary_idx(n):
    center = n - 1
    posL = []
    negL = []
    posR = []
    negR = []
    empty = np.empty(0, dtype=np.int32)
    for p in range(n):
        if p > 0:
            dL = np.arange(p, 0, -1, dtype=np.int32)
            posL.append(center + dL)
            negL.append(center - dL)
        else:
            posL.append(empty)
            negL.append(empty)
        rlen = n - 1 - p
        if rlen > 0:
            dR = np.arange(1, rlen + 1, dtype=np.int32)
            posR.append(center + dR)
            negR.append(center - dR)
        else:
            posR.append(empty)
            negR.append(empty)
    return posL, negL, posR, negR


def flip_bit_bin(h, c, p, n, posL, negL, posR, negR):
    x = h[p]
    if p > 0:
        hq = h[:p]
        if x:
            c[posL[p]] += hq
            c[negL[p]] += hq - 1
        else:
            c[posL[p]] -= hq
            c[negL[p]] += 1 - hq
    if p < n - 1:
        hq = h[p + 1:]
        if x:
            c[posR[p]] += hq - 1
            c[negR[p]] += hq
        else:
            c[posR[p]] += 1 - hq
            c[negR[p]] -= hq
    h[p] = 1 - x


def propose_active_p(h, max_idx, n, rng):
    d = max_idx - (n - 1)
    if d > 0:
        lo = n - d
        if lo <= 0:
            return None
        for _ in range(30):
            i = int(rng.randint(lo))
            if h[i] == 1 and h[i + d] == 0:
                return i
    elif d < 0:
        e = -d
        lo = n - e
        if lo <= 0:
            return None
        for _ in range(30):
            i = int(rng.randint(lo))
            if h[i] == 0 and h[i + e] == 1:
                return i + e
    return None


def binary_local_search(n, h_init, rng, max_iters, time_limit):
    if n <= 0:
        return np.zeros(0), np.inf
    target_ones = n // 2
    h = make_binary_from_h(h_init, n, rng)
    c = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode='full').astype(np.int32)
    posL, negL, posR, negR = make_binary_idx(n)

    ones_arr = np.where(h == 1)[0]
    zeros_arr = np.where(h == 0)[0]
    ones_list = ones_arr.tolist()
    zeros_list = zeros_arr.tolist()
    one_id = np.full(n, -1, dtype=np.int32)
    zero_id = np.full(n, -1, dtype=np.int32)
    for i, pos in enumerate(ones_list):
        one_id[int(pos)] = i
    for j, pos in enumerate(zeros_list):
        zero_id[int(pos)] = j

    max_idx = int(np.argmax(c))
    cmax = int(c[max_idx])
    best_cmax = cmax
    best_h = h.copy()

    it = 0
    no_imp = 0
    local_end = time.time() + max(1.0, float(time_limit))
    t0 = 3.0
    t1 = 0.05

    while (remaining_time() > 5.0 and time.time() < local_end and it < max_iters):
        p = None
        i = -1
        j = -1

        if rng.rand() < 0.28:
            p = propose_active_p(h, max_idx, n, rng)
            if p is not None:
                i = int(one_id[int(p)])
                if i < 0:
                    p = None

        if p is None:
            i = int(rng.randint(target_ones))
            j = int(rng.randint(target_ones))
            p = int(ones_list[i])

        if j < 0 or p is None or i < 0:
            j = int(rng.randint(target_ones))
        q = int(zeros_list[j])

        flip_bit_bin(h, c, int(p), n, posL, negL, posR, negR)
        flip_bit_bin(h, c, int(q), n, posL, negL, posR, negR)
        new_max_idx = int(np.argmax(c))
        new_cmax = int(c[new_max_idx])
        delta = new_cmax - cmax

        frac = float(it) / float(max(1, max_iters))
        T = t0 * (t1 / t0) ** frac
        accept = False
        if delta < 0:
            accept = True
        elif delta == 0:
            accept = bool(rng.rand() < 0.10)
        elif delta <= 3 and T > 1e-8:
            accept = bool(rng.rand() < math.exp(-float(delta) / T))

        if accept:
            ones_list[i] = q
            zeros_list[j] = p
            one_id[int(p)] = -1
            one_id[int(q)] = int(i)
            zero_id[int(p)] = int(j)
            zero_id[int(q)] = -1
            cmax = new_cmax
            max_idx = new_max_idx
            if new_cmax < best_cmax:
                best_cmax = new_cmax
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            flip_bit_bin(h, c, int(q), n, posL, negL, posR, negR)
            flip_bit_bin(h, c, int(p), n, posL, negL, posR, negR)
            no_imp += 1

        it += 1
        if no_imp > 350000:
            break

        if it % 50000 == 0:
            c = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode='full').astype(np.int32)
            max_idx = int(np.argmax(c))
            cmax = int(c[max_idx])
            ones_arr = np.where(h == 1)[0]
            zeros_arr = np.where(h == 0)[0]
            ones_list = ones_arr.tolist()
            zeros_list = zeros_arr.tolist()
            if len(ones_list) != target_ones:
                h = make_binary_from_h(h, n, rng)
                c = np.correlate(h.astype(np.int32), (1 - h).astype(np.int32), mode='full').astype(np.int32)
                max_idx = int(np.argmax(c))
                cmax = int(c[max_idx])
                ones_arr = np.where(h == 1)[0]
                zeros_arr = np.where(h == 0)[0]
                ones_list = ones_arr.tolist()
                zeros_list = zeros_arr.tolist()
            one_id = np.full(n, -1, dtype=np.int32)
            zero_id = np.full(n, -1, dtype=np.int32)
            for ii, pos in enumerate(ones_list):
                one_id[int(pos)] = ii
            for jj, pos in enumerate(zeros_list):
                zero_id[int(pos)] = jj

    return best_h, float(best_cmax) * (2.0 / n)


def smooth_obj_grad(x):
    n = _CN
    dx = 2.0 / float(n)
    omh = 1.0 - x
    corr = np.correlate(x, omh, mode='full')
    m = float(np.max(corr))
    y = np.clip(_BETA * (corr - m), -700.0, 0.0)
    e = np.exp(y)
    sume = float(e.sum())
    if sume <= 0.0:
        sume = 1.0
    f = dx * (m + math.log(sume) / float(_BETA))

    s = float(np.sum(x))
    diff = s - _TGT
    if _PEN > 0.0:
        f += _PEN * diff * diff

    w = e / sume
    active = np.where(w > 1e-8)[0]
    g = np.zeros(n, dtype=float)
    center = n - 1

    for idx in active:
        wj = float(w[idx])
        d = int(idx) - center
        if d > 0:
            g[:n - d] += wj * omh[d:]
            g[d:] -= wj * x[:n - d]
        elif d < 0:
            e_ = -d
            g[:n - e_] -= wj * x[e_:]
            g[e_:] += wj * omh[:n - e_]
        else:
            g += wj * (1.0 - 2.0 * x)

    g *= dx
    if _PEN > 0.0:
        g += 2.0 * _PEN * diff
    return f, g


def active_refine(h0, n, iterations=40, beta=250.0):
    global _CN, _TGT, _BETA, _PEN
    if n <= 0:
        return np.asarray(h0, dtype=float).copy(), 0.0
    _CN = n
    _TGT = float(n) / 2.0
    _BETA = float(beta)
    _PEN = 0.0

    h = project_feasible(h0, n)
    c = compute_c5(h)

    for _ in range(int(iterations)):
        if remaining_time() < 6.0:
            break
        try:
            _, g = smooth_obj_grad(h)
        except Exception:
            break
        if not np.all(np.isfinite(g)):
            break
        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-12:
            break

        scale = float(n) / (gnorm + 1e-12)
        improved = False
        for alpha in (0.05, 0.10, 0.02, 0.20, 0.08, 0.01, 0.35, 0.04):
            h2 = project_feasible(h - (alpha * scale) * g, n)
            c2 = compute_c5(h2)
            if c2 < c - 1e-12:
                h = h2
                c = c2
                improved = True
                break
        if not improved:
            break

    return h, c


def cont_polish(h0, n, beta_list, maxfun=3000, pen=500.0, do_active=True):
    global _CN, _TGT, _BETA, _PEN
    if n <= 0:
        return np.asarray(h0, dtype=float).copy(), 0.0
    _CN = n
    _TGT = float(n) / 2.0
    h = project_feasible(h0, n)
    c = compute_c5(h)
    bounds = [(0.0, 1.0)] * n

    for beta in beta_list:
        if remaining_time() < 25.0:
            break
        _BETA = float(beta)
        _PEN = float(pen)
        options = {
            'maxiter': int(max(120, maxfun // 4)),
            'ftol': 1e-12,
            'gtol': 1e-7,
            'maxfun': int(maxfun),
            'iprint': 0,
            'maxls': 20,
            'history': 8,
        }
        try:
            res = minimize(smooth_obj_grad, h, jac=True, method='L-BFGS-B',
                           bounds=bounds, options=options)
            h2 = project_feasible(res.x, n)
            c2 = compute_c5(h2)
            if c2 < c:
                h = h2
                c = c2
        except Exception:
            pass

    if do_active and remaining_time() > 15.0:
        h, c = active_refine(h, n, iterations=35, beta=260.0)

    return h, c


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')
    best_n = 0

    try:
        parent = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        parent = None

    if parent is not None and parent.size > 0:
        parent = project_feasible(parent, parent.size)
        pc = compute_c5(parent)
        best_h = parent.copy()
        best_c = pc
        best_n = int(parent.size)
        print("Initial parent C5={:.7f}".format(pc))

    if best_h is None:
        n0 = 160
        best_h = project_feasible(make_thue_morse(n0), n0)
        best_c = compute_c5(best_h)
        best_n = n0
        print("Fallback Thue-Morse C5={:.7f}".format(best_c))

    n_plan = []
    base_even = best_n if best_n % 2 == 0 else best_n + 1
    for nn in (base_even, 160, 208, 256, 320, 384):
        if nn % 2 == 0 and 128 <= nn <= 448 and nn not in n_plan:
            n_plan.append(nn)
    n_plan.sort()
    n_plan = n_plan[:4]

    print("Continuous resolutions:", n_plan)
    for n in n_plan:
        if remaining_time() < 80.0:
            break
        starts = make_start_candidates(best_h, best_n, n, rng)
        evals = []
        for s in starts:
            s = project_feasible(s, n)
            c = compute_c5(s)
            evals.append((c, s))
        if not evals:
            continue
        vals = [float(c) for c, _ in evals]
        order = np.argsort(vals)

        max_starts = 3 if n <= 224 else 2
        if n == best_n and remaining_time() > 800.0:
            max_starts = 4

        if n <= 192:
            maxfun = 3800
            betas = (28, 80, 220, 550)
        elif n <= 300:
            maxfun = 2800
            betas = (35, 100, 300)
        else:
            maxfun = 2100
            betas = (45, 130, 380)

        for pos in range(min(max_starts, len(order))):
            if remaining_time() < 70.0:
                break
            ei = int(order[pos])
            c0, s0 = evals[ei]
            if pos > 0 and c0 > best_c + 0.012:
                continue
            h2, c2 = cont_polish(s0, n, betas, maxfun=maxfun, pen=500.0, do_active=True)
            if c2 < best_c:
                best_c = c2
                best_h = np.asarray(h2, dtype=float).copy()
                best_n = int(n)
                print("Continuous improvement: n={}, C5={:.7f}".format(best_n, best_c))

    if remaining_time() > 160.0:
        bin_start = time.time()
        bin_budget = min(remaining_time() * 0.45, 450.0)
        b_ns = []
        cand_bs = [best_n if best_n % 2 == 0 else best_n + 1, 192, 256, 320, 384]
        for nn in cand_bs:
            if nn % 2 == 0 and 160 <= nn <= 400 and nn not in b_ns:
                b_ns.append(nn)
        b_ns.sort()
        b_ns = b_ns[:3]
        print("Binary resolutions:", b_ns)

        for n in b_ns:
            if time.time() - bin_start > bin_budget - 50.0:
                break
            starts = []
            if best_h is not None:
                starts.append(project_feasible(interp_to(best_h, best_n, n), n))
            starts.append(make_thue_morse(n))
            starts.append(make_golden(n, 0.6180339887, float(rng.rand())))

            bvals = []
            bstarts = []
            for s in starts:
                hb = make_binary_from_h(s, n, rng)
                cb = float(np.max(np.correlate(hb.astype(np.int32), (1 - hb).astype(np.int32), mode='full'))) * (2.0 / n)
                bvals.append(cb)
                bstarts.append(hb)

            if not bstarts:
                continue
            bord = np.argsort(bvals)
            per_start_limit = min(170.0, bin_budget - (time.time() - bin_start))
            if per_start_limit < 40.0:
                break

            for pos in range(min(2, len(bord))):
                if time.time() - bin_start > bin_budget - 30.0:
                    break
                ei = int(bord[pos])
                rem = bin_budget - (time.time() - bin_start)
                limit = min(220.0, max(35.0, rem))
                hbin, cbin = binary_local_search(n, bstarts[ei], rng,
                                                 max_iters=6000000,
                                                 time_limit=limit)
                if cbin < best_c:
                    best_c = cbin
                    best_h = np.asarray(hbin, dtype=float).copy()
                    best_n = int(n)
                    print("Binary improvement: n={}, C5={:.7f}".format(best_n, best_c))

                if time.time() - bin_start > bin_budget:
                    break
                if cbin <= best_c + 0.015 and remaining_time() > 70.0:
                    h2, c2 = cont_polish(hbin, n, (70, 220, 650),
                                         maxfun=2200, pen=500.0, do_active=True)
                    if c2 < best_c:
                        best_c = c2
                        best_h = np.asarray(h2, dtype=float).copy()
                        best_n = int(n)
                        print("Binary+continuous improvement: n={}, C5={:.7f}".format(best_n, best_c))

    if remaining_time() > 35.0:
        h2, c2 = cont_polish(best_h, best_n, (90, 260, 750),
                             maxfun=4200, pen=500.0, do_active=False)
        if c2 < best_c:
            best_c = c2
            best_h = np.asarray(h2, dtype=float).copy()
            best_n = int(best_h.size)
            print("Final continuous polish: n={}, C5={:.7f}".format(best_n, best_c))

    if remaining_time() > 20.0:
        h2, c2 = active_refine(best_h, best_n, iterations=80, beta=320.0)
        if c2 < best_c:
            best_c = c2
            best_h = np.asarray(h2, dtype=float).copy()
            best_n = int(best_h.size)
            print("Final active polish: n={}, C5={:.7f}".format(best_n, best_c))

    best_h = project_feasible(best_h, best_n)
    final_c = compute_c5(best_h)

    try:
        ev = evaluate_erdos_solution(best_h)
        ev = float(ev)
        if np.isfinite(ev) and ev > 0.0 and ev < final_c:
            final_c = ev
    except Exception:
        pass

    if not np.isfinite(final_c):
        final_c = compute_c5(best_h)

    print("FINAL: n={}, C5={:.7f}".format(best_n, final_c))
    return (best_h, float(final_c), int(best_n))
