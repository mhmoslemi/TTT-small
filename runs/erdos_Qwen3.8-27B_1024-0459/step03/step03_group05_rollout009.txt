            corr_full = corr_full[self.subset]
        m = corr_full.max()
        exps = np.exp(self.beta * (corr_full - m))
        sum_e = np.sum(exps)
        w = exps / sum_e
        
        se = np.sum(x) - self.target
        val = (m + np.log(sum_e) / self.beta) * self.dx
        val += self.pw * se * se
        
        grad = np.zeros(n)
        # Need to map back to full correlation indices
        if self.subset is not None:
            full_indices = self.subset
        else:
            full_indices = np.arange(2 * n - 1)
        
        for ii in range(len(full_indices)):
            wk = w[ii]
            if wk < 1e-14:
                continue
            k = full_indices[ii]
            lag = k - (n - 1)
            i_lo = max(0, -lag)
            i_hi = min(n, n - lag)
            if i_lo >= i_hi:
                continue
            a_lo = i_lo + lag
            a_hi = i_hi + lag
            grad[a_lo:a_hi] += wk * (1.0 - x[i_lo:i_hi])
            grad[i_lo:i_hi] -= wk * x[a_lo:a_hi]
        
        grad *= self.dx
        grad += 2.0 * self.pw * se
        return val, grad
