            grad[mask_a] += wj * (1.0 - x[m_idx[mask_a] - s])
            # Second part: m in [0, n-1-s], contribution -x[m+s]
            mask_b = m_idx <= n - 1 - s
            grad[mask_b] += wj * (-x[m_idx[mask_b] + s])
        else:
            sp = -s
            # C = sum_{i=0}^{n-1-sp} h[i+sp]*(1-h[i]) ... wait
            # For s < 0, let sp = -s > 0
            # C[j] = sum_{n: n+s in [0,n-1]} h[n]*(1-h[n+s])
            # = sum_{n=sp}^{n-1} h[n]*(1-h[n-sp])
            # = sum_{m=0}^{n-1-sp} h[m+sp]*(1-h[m])
            # dC/dh[m] = [m appears in h[m+sp]]*[1-h[m']] + h[m'*sp]*[m appears in (1-h[m'])]*(-1)
            # = [m-sp in [0,n-1-sp]]*(1-h[m-sp]) + [m in [sp, n-1]]*... 
            
            # Let me redo: C = sum_{m=0}^{n-1-sp} h[m+sp]*(1-h[m])
            # dC/dh[p] = sum_m [p=m+sp]*(1-h[m]) + h[m+sp]*(-[p=m])
            # First: p = m+sp => m = p-sp, valid when 0 <= p-sp <= n-1-sp => sp <= p <= n-1
            #   contribution: (1-h[p-sp])
            # Second: p = m, valid when 0 <= p <= n-1-sp
            #   contribution: -h[p+sp]
            
            m_idx = np.arange(n)
            mask_a = m_idx >= sp
            grad[mask_a] += wj * (1.0 - x[m_idx[mask_a] - sp])
            mask_b = m_idx <= n - 1 - sp
            grad[mask_b] += wj * (-x[m_idx[mask_b] + sp])
    
    grad *= dx
    grad += 2.0 * 1000.0 * (np.sum(x) - _TGT)
    return obj, grad

def projected_gradient_step(h, grad, step_size, n):
    """Take a projected gradient step"""
    h_new = h - step_size * grad
    return project_feasible(h_new, n)

def active_set_gradient(h, n, top_k=5):
    """Compute gradient as average of top-k correlation gradients"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    # Use softmax weighting for smooth transition
    top_vals = corr[top_idx]
    m = np.max(top_vals)
    beta = 100.0
    w = np.exp(beta * (top_vals - m))
    w /= np.sum(w)
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, j in zip(w, top_idx):
        s = j - (n - 1)
        m_idx = np.arange(n)
        if s >= 0:
            mask_a = m_idx >= s
            grad[mask_a] += wi * (1.0 - h[m_idx[mask_a] - s])
            mask_b = m_idx <= n - 1 - s
            grad[mask_b] += wi * (-h[m_idx[mask_b] + s])
        else:
            sp = -s
            mask_a = m_idx >= sp
            grad[mask_a] += wi * (1.0 - h[m_idx[mask_a] - sp])
            mask_b = m_idx <= n - 1 - sp
            grad[mask_b] += wi * (-h[m_idx[mask_b] + sp])
    grad *= dx
    return grad

def coordinate_descent(h, n, max_sweeps=50, rng=None):
    """Coordinate descent with exact 1D optimization"""
    h = h.copy()
    dx = 2.0 / n
    target = n / 2.0
    
    for sweep in range(max_sweps):
        improved = False
        order = np.arange(n) if rng is None else rng.permutation(n)
        for i in order:
            # Optimize h[i] while keeping sum constraint
            # h[i] + (target - sum(h) + h[i]) * (adjustment to others) ... 
            # Actually, when we change h[i], we need to adjust the sum.
            # Simplest: change h[i] and then re-project. But that's expensive.
            # Alternative: change h[i] by δ and h[j] by -δ for some j.
            # For simplicity, just try a few values and re-project.
            old_val = h[i]
            best_val = old_val
            best_c5 = compute_c5(project_feasible(h, n))
            
            for candidate in [0.0, 1.0, 0.5, max(0, old_val - 0.3), min(1, old_val + 0.3), 
                             max(0, old_val - 0.1), min(1, old_val + 0.1)]:
                if abs(candidate - old_val) < 1e-10:
                    continue
                h[i] = candidate
                h_proj = project_feasible(h, n)
                c5 = compute_c5(h_proj)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    best_val = candidate
                    h = h_proj.copy()
            # Put back if no improvement
            if best_val != old_val and best_val != h[i]:
                h[i] = old_val
                h = project_feasible(h, n)
            elif h[i] != best_val:
                h[i] = best_val
                h = project_feasible(h, n)
        
        if not improved:
            break
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    return h

def make_pattern(n, freq, phase, amp=0.3):
    t = np.arange(n) / float(n)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * t + phase)
    return np.clip(h, 0.0, 1.0)

def binary_sa(h_init, n, iterations, rng, t_start=0.003, t_end=0.00003):
    """Binary simulated annealing with swap moves"""
    h = np.where(h_init > 0.5, 1.0, 0.0)
    target_ones = n // 2
    num_ones = int(np.sum(h))
    
    if num_ones > target_ones:
        ones_idx = np.where(h == 1.0)[0]
        remove = rng.choice(len(ones_idx), num_ones - target_ones, replace=False)
        h[ones_idx[remove]] = 0.0
    elif num_ones < target_ones:
        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
        h[zeros_idx[add]] = 1.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    
    for it in range(iterations):
        if remaining_time() < 3:
            break
        
        i = rng.randint(len(ones_list))
        j = rng.randint(len(zeros_list))
        oi = ones_list[i]
        oz = zeros_list[j]
        
        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = compute_c5(h)
        
        T = t_start * (t_end / t_start) ** (float(it) / max(float(iterations), 1.0))
        
        if new_c5 < current_c5 or (T > 1e-15 and rng.rand() < np.exp(-(new_c5 - current_c5) / T)):
            current_c5 = new_c5
            ones_list[i] = oz
            zeros_list[j] = oi
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0
    
    return best_h, best_c5
