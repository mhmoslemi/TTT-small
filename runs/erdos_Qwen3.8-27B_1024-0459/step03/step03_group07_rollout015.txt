import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-12
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    diff = target - h.sum()
