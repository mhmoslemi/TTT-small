            a = np.zeros(total_lags)
            
            # Term A: h[j] as first factor, lag in [-j, n-1-j]
            # c[m] += (1 - h[j+lag]) * x
            lo_A = -j
            hi_A = n - 1 - j
            lags_A = np.arange(lo_A, hi_A + 1)
            idx_A = lags_A + (n - 1)
            # Exclude m0
            mask_A = idx_A != m0
            if np.any(mask_A):
                valid_idx_A = idx_A[mask_A]
                valid_lags_A = lags_A[mask_A]
                a[valid_idx_A] += (1.0 - h[j + valid_lags_A])
            
            # Term B: h[j] as second factor (in 1-h[j]), lag in [j-n+1, j]
            # c[m] += h[j-lag] * (1-x) = h[j-lag] - h[j-lag]*x
            lo_B = j - (n - 1)
            hi_B = j
            lags_B = np.arange(lo_B, hi_B + 1)
            idx_B = lags_B + (n - 1)
            mask_B = idx_B != m0
            if np.any(mask_B):
                valid_idx_B = idx_B[mask_B]
                valid_lags_B = lags_B[mask_B]
                a[valid_idx_B] += (-h[j - valid_lags_B])
            
            # Base values: c[m](x) = base[m] + a[m]*x for m != m0
            base = corr - a * old
            # Quadratic at m0
            q = corr[m0] - old * (1.0 - old)
            base[m0] = 0.0  # Will be handled by quadratic
            
            # Vectorized grid evaluation
            xs = np.linspace(0.0, 1.0, grid_size)
            # vals shape: (grid_size, total_lags)
            vals = base[None, :] + xs[:, None] * a[None, :]
            vals[:, m0] = q + xs - xs * xs
            obj = np.max(vals, axis=1)
            
            best_i = np.argmin(obj)
            best_x = xs[best_i]
            
            # Refine with ternary search
            lo_bin = max(0, best_i - 2)
            hi_bin = min(grid_size - 1, best_i + 2)
            lo = xs[lo_bin]
            hi = xs[hi_bin]
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                # Evaluate at m1 and m2
                v1 = np.max(base + m1 * a)
                v1_quad = q + m1 - m1 * m1
                v1 = max(v1, v1_quad)
                v2 = np.max(base + m2 * a)
                v2_quad = q + m2 - m2 * m2
                v2 = max(v2, v2_quad)
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            best_x = (lo + hi) / 2.0
            
            # Update
            if abs(best_x - old) > 1e-12:
                # Incremental correlation update
                corr[a != 0] += a[a != 0] * (best_x - old)
                # Fix m0
                corr[m0] += (best_x - best_x * best_x) - (old - old * old)
                h[j] = best_x
                improved += 1
        
        # End of sweep
        c5 = np.max(corr) * (2.0 / n)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count >= max_stall and sw > 20:
            # Try perturbation
            perturbation = 0.02 * np.random.randn(n)
            h = project_feasible(h + perturbation, n)
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5 = np.max(corr) * (2.0 / n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            stall_count = 0
    
    return best_h, best_c5

def slsqp_polish(h, n, time_limit=None, t0=None):
    """Polish with SLSQP using smooth max objective."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    
    def eq_con(x):
        return np.sum(x) - n / 2.0
    
    def smooth_obj(x, beta):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -400.0, 0.0)
        return (m + np.log(np.sum(np.exp(sh))) / beta) * (2.0 / n)
    
    for beta in [50.0, 200.0, 1000.0, 5000.0]:
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        try:
            res = minimize(lambda x: smooth_obj(x, beta), h, method='SLSQP',
                          bounds=bounds, constraints=[{'type': 'eq', 'fun': eq_con}],
                          options={'maxiter': 200, 'ftol': 1e-14, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    # Top-k polish
    for k in [3, 5]:
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        def obj_topk(x, kk=k):
            corr = np.correlate(x, 1.0 - x, mode='full')
            top = np.sort(corr)[-kk:]
            return np.mean(top) * (2.0 / n)
        try:
            res = minimize(obj_topk, h, method='SLSQP',
                          bounds=bounds, constraints=[{'type': 'eq', 'fun': eq_con}],
                          options={'maxiter': 150, 'ftol': 1e-14, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

def piecewise_init(n, m, rng):
    """Generate a piecewise-constant initialization with m pieces."""
    # Random boundaries
    bounds = np.sort(rng.uniform(0, 2, m - 1))
    bounds = np.concatenate([[0.0], bounds, [2.0]])
    # Random heights
    heights = rng.uniform(0.1, 0.9, m)
    # Create full array
    h = np.zeros(n)
    for i in range(m):
        lo_idx = int(bounds[i] * n / 2.0)
        hi_idx = min(int(bounds[i + 1] * n / 2.0), n)
        h[lo_idx:hi_idx] = heights[i]
    return project_feasible(h, n)

def optimize_piecewise(rng, n, m, budget_s=30):
    """Optimize a piecewise-constant function with m pieces using SLSQP."""
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')
    
    for trial in range(5):
        if time.time() - t0 > budget_s:
            break
        # Random initialization of m-1 boundaries and m heights
        bounds_rand = np.sort(rng.uniform(0.2, 1.8, m - 1))
        heights_rand = rng.uniform(0.2, 0.8, m)
        x0 = np.concatenate([bounds_rand, heights_rand])
        
        def pw_obj(x):
            b = x[:m-1]
            hts = x[m-1:]
            boundaries = np.concatenate([[0.0], b, [2.0]])
            # Ensure valid
            if np.any(np.diff(boundaries) < 0.01):
                return 1.0
            # Enforce sum constraint approximately won't work with SLSQP easily
            # Use penalty
            h_full = np.zeros(n)
            for i in range(m):
                lo_idx = int(np.floor(boundaries[i] * n / 2.0))
                hi_idx = min(int(np.floor(boundaries[i+1] * n / 2.0)), n)
                if hi_idx > lo_idx:
                    h_full[lo_idx:hi_idx] = hts[i]
            h_full = np.clip(h_full, 0, 1)
            # Penalty for sum constraint
            s = np.sum(h_full) * (2.0/n)
            penalty = 100.0 * abs(s - 1.0)
            c5 = compute_c5(h_full)
            return c5 + penalty
        
        bounds_opt = []
        for i in range(m - 1):
            bounds_opt.append((0.05, 1.95))
        for i in range(m):
            bounds_opt.append((0.0, 1.0))
        
        try:
            res = minimize(pw_obj, x0, method='Nelder-Mead',
                          options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-10})
            x_opt = res.x
            b = x_opt[:m-1]
            hts = np.clip(x_opt[m-1:], 0, 1)
            boundaries = np.concatenate([[0.0], np.sort(b), [2.0]])
            h_full = np.zeros(n)
            for i in range(m):
                lo_idx = int(np.floor(boundaries[i] * n / 2.0))
                hi_idx = min(int(np.floor(boundaries[i+1] * n / 2.0)), n)
                if hi_idx > lo_idx:
                    h_full[lo_idx:hi_idx] = hts[i]
            h_full = project_feasible(h_full, n)
            c5 = compute_c5(h_full)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_full.copy()
        except Exception:
            pass
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return max(0.0, budget_s - (time.time() - t0) - 5.0)
    
    # Get parent
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn > 10:
            best_h = ph.copy()
            best_c5 = compute_c5(ph)
            best_n = pn
    except Exception:
        pass
    
    if best_h is None:
        best_n = 160
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)
    
    # Stage 1: Coordinate descent at parent resolution
    if time_left() > 100:
        h_cd, c5_cd = coord_descent_fast(best_h, best_n, max_sweeps=3000,
                                         time_limit=min(200, time_left()), t0=t0)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
            best_n = best_n
    
    # Stage 2: Try piecewise initializations at n=256
    target_n = 256
    if time_left() > 200:
        h_interp = interpolate_h(best_h, best_n, target_n)
        
        # Also try piecewise inits
        pw_inits = [h_interp.copy()]
        for m in [4, 5, 6, 7, 8]:
            if time_left() > 100:
                hp, cp = optimize_piecewise(rng, target_n, m, budget_s=min(30, time_left() * 0.1))
                if hp is not None:
                    pw_inits.append(hp)
        
        # Run coord descent on each init
        for h_init in pw_inits:
            if time_left() < 50:
                break
            h_cd, c5_cd = coord_descent_fast(h_init, target_n, max_sweeps=5000,
                                             time_limit=min(150, time_left() * 0.3), t0=t0)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd
                best_n = target_n
    
    # Stage 3: Go to n=512
    target_n2 = 512
    if time_left() > 200:
        h_interp2 = interpolate_h(best_h, best_n, target_n2)
        h_cd2, c5_cd2 = coord_descent_fast(h_interp2, target_n2, max_sweeps=10000,
                                           time_limit=min(300, time_left() * 0.5), t0=t0)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2
            best_n = target_n2
        
        # Continue at n=512 with the best
        if time_left() > 100:
            h_cd2b, c5_cd2b = coord_descent_fast(h_cd2, target_n2, max_sweeps=10000,
                                                 time_limit=min(200, time_left() * 0.4), t0=t0)
            if c5_cd2b < best_c5:
                best_c5 = c5_cd2b
                best_h = h_cd2b
                best_n = target_n2
    
    # Stage 4: SLSQP polish
    if time_left() > 60:
        h_sl, c5_sl = slsqp_polish(best_h, best_n, time_limit=min(80, time_left() * 0.5), t0=t0)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl
            best_n = best_n
    
    # Stage 5: Final coordinate descent polish
    if time_left() > 50:
        h_final, c5_final = coord_descent_fast(best_h, best_n, max_sweeps=5000,
                                              time_limit=min(100, time_left() * 0.6), t0=t0,
                                              grid_size=300)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    
    # Stage 6: Try n=1024 if time permits
    if time_left() > 300:
        target_n3 = 1024
        h_interp3 = interpolate_h(best_h, best_n, target_n3)
        h_cd3, c5_cd3 = coord_descent_fast(h_interp3, target_n3, max_sweeps=5000,
                                           time_limit=min(200, time_left() * 0.5), t0=t0)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3
            best_n = target_n3
        if time_left() > 60:
            h_cd3b, c5_cd3b = coord_descent_fast(h_cd3, target_n3, max_sweeps=5000,
                                                 time_limit=min(100, time_left() * 0.4), t0=t0)
            if c5_cd3b < best_c5:
                best_c5 = c5_cd3b
                best_h = h_cd3b
                best_n = target_n3
    
    # Final SLSQP if time
    if time_left() > 30:
        h_sl2, c5_sl2 = slsqp_polish(best_h, best_n, time_limit=min(40, time_left() * 0.5), t0=t0)
        if c5_sl2 < best_c5:
            best_c5 = c5_sl2
            best_h = h_sl2
    
    # Final coord descent
    if time_left() > 20:
        h_fin, c5_fin = coord_descent_fast(best_h, best_n, max_sweeps=3000,
                                          time_limit=time_left(), t0=t0, grid_size=400)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    # Validate
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
