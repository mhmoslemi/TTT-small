- In 100 seconds: ~10,000 iterations
- Each iteration tries O(n) swaps, so total swap evaluations: ~3,000,000

That should be plenty for good local search.

But actually, we can be smarter. Instead of trying random swaps, we can:
1. Identify the lags with the highest correlations
2. For each such lag s, identify which positions contribute to it
3. Try swaps that specifically reduce those correlations

This is the "focused" approach from the previous code, but with exact delta computation.

Let me now write the full program. I'll aim for:

Phase 1 (100s): Binary local search from parent
Phase 2 (1000s): Continuous optimization (L-BFGS-B with penalty) at multiple resolutions
Phase 3 (90s): Final polish

    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def binary_local_search(h_bin, n, time_limit=100, rng=None):
    """Fast binary local search with exact delta computation."""
    if rng is None:
        rng = np.random.RandomState(42)
    
    h = h_bin.copy()
    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx_val
    start = time.time()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    stall = 0
    temp = 0.5  # temperature for simulated annealing
    
    while time.time() - start < time_limit and remaining() > 10:
        # Compute delta for a candidate swap
        # Focus on reducing the max correlation
        k_max = int(np.argmax(corr))
        
        improved = False
        n_try = 100
        for _ in range(n_try):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            
            # Compute delta
            # Δ(m) = -(1-h[i-m])*[0≤i-m<n] + (1-h[j-m])*[0≤j-m<n] + h[i+m]*[0≤i+m<n] - h[j+m]*[0≤j+m<n]
            # for m in [-(n-1), n-1]
            # The affected m are those where at least one indicator is non-zero
            
            # For efficiency, compute the new max by checking affected lags
            # Only lags where the correlation changes can affect the max
            
            # The max can only decrease (good) or the new values at affected lags might exceed old max
            # Since we're doing a local search, just check if new max < old max
            
            # Quick check: compute delta at the current max lag
            m_max = k_max - (n - 1)
            delta_max = 0.0
            # term 1: -(1-h[i-m_max]) if 0 <= i-m_max < n
            if 0 <= i - m_max < n:
                delta_max += -(1.0 - h[i - m_max])
            # term 2: +(1-h[j-m_max]) if 0 <= j-m_max < n
            if 0 <= j - m_max < n:
                delta_max += (1.0 - h[j - m_max])
            # term 3: +h[i+m_max] if 0 <= i+m_max < n
            if 0 <= i + m_max < n:
                delta_max += h[i + m_max]
            # term 4: -h[j+m_max] if 0 <= j+m_max < n
            if 0 <= j + m_max < n:
                delta_max -= h[j + m_max]
            
            if delta_max >= 0:
                # The max lag doesn't decrease, skip (not necessarily wrong for SA, but for local search)
                # Actually for SA we might still accept, but let's focus on improvements first
                pass
            else:
                # The max lag decreases. But we need to check if any OTHER lag increases above the new max
                # For simplicity in the fast path, just do the swap and recompute
                # (This is O(n²) but we only do it when the max lag is improving)
                h[i] = 0.0
                h[j] = 1.0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                new_corr = compute_corr(h)
                new_max = float(np.max(new_corr))
                if new_max < cur_max - 1e-12:
                    cur_max = new_max
                    corr = new_corr
                    best_c5 = cur_max * dx_val
                    improved = True
                    k_max = int(np.argmax(corr))
                    break
                else:
                    # Revert
                    h[i] = 1.0
                    h[j] = 0.0
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
        
        if improved:
            stall = 0
            temp = max(0.1, temp * 0.99)
        else:
            stall += 1
            if stall > 50:
                # Simulated annealing kick
                if temp > 0.05:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i] = 0.0
                    h[j] = 1.0
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    corr = compute_corr(h)
                    new_max = float(np.max(corr))
                    # Accept with probability exp(-Δ/temp) where Δ is in units of correlation
                    if new_max > cur_max:
                        prob = np.exp(-(new_max - cur_max) / (temp * n * 0.01))
                        if rng.rand() > prob:
                            # Revert
                            h[i] = 1.0
                            h[j] = 0.0
                            ones = np.where(h == 1)[0]
                            zeros = np.where(h == 0)[0]
                            corr = compute_corr(h)
                            new_max = cur_max
                    cur_max = new_max
                    k_max = int(np.argmax(corr))
                    temp *= 0.95
                    stall = 0
    
    return h, best_c5

Actually, let me reconsider the problem. The previous code got 0.3809. The target is 0.38085. The difference is 0.00005. 

For n=320, the C₅ of a binary solution is M * 2/320 = M/160 for some integer M. The values around 0.38 are:
- 61/160 = 0.38125

So a binary solution at n=320 can give either 0.375 or 0.38125 (or higher). The value 0.3809 is between these, so it must be from a non-binary (fractional) solution.

For a fractional solution, C₅ can take any value. The SLSQP/gradient optimization is what gets us between the binary levels.

So the strategy should be:
1. Find a good binary solution (structure)
2. Optimize the fractional values to reduce C₅ below the binary level

The key question is: can we get C₅ < 0.38085?

Let me think about what the continuous optimum looks like. If the binary solution has max count M=61 at several lags, then by fractional perturbation, we can reduce the max from 61 to something like 60.9 or 60.8, giving C₅ = 60.9/160 = 0.3806 or 60.8/160 = 0.38.

The amount of reduction depends on how "tight" the constraint is. If multiple lags are at the maximum, we can't reduce them all simultaneously without increasing others. The SLSQP smooth approximation handles this by finding the tradeoff.

So the plan is:
1. Get a good binary structure (max count = 61 or lower)
2. Use gradient optimization to push the max below 61

Let me focus on making the gradient optimization more effective. The key improvements over the previous code:

This is smoother and its gradient is:

```python
import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 5.0)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def smooth_val_grad(x, n, beta, penalty=0.0):
    """Compute smooth max objective and gradient, with optional sum penalty."""
    dx_val = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    shifted = corr - m
    cl = np.clip(beta * shifted, -50.0, 50.0)
    ex = np.exp(cl)
    s = float(np.sum(ex))
    val = (m + np.log(s) / beta) * dx_val
    w = ex / s
    
    # Gradient of smooth max w.r.t. x
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        off = k - (n - 1)
        if off >= 0:
            npairs = n - off
            if npairs > 0:
                g[off:off+npairs] += wk * (1.0 - x[:npairs])
                g[:npairs] -= wk * x[off:off+npairs]
        else:
            ao = -off
            npairs = n - ao
            if npairs > 0:
                g[:npairs] += wk * (1.0 - x[ao:ao+npairs])
                g[ao:ao+npairs] -= wk * x[:npairs]
    g *= dx_val
    
    # Add penalty for sum constraint
    if penalty > 0:
        s_diff = np.sum(x) - n / 2.0
        val += penalty * s_diff * s_diff
        g += 2.0 * penalty * s_diff * np.ones(n)
    
    return val, g

def lbfgs_smooth(h, n, beta, maxiter=200, penalty=100.0):
    """Optimize using L-BFGS-B with smooth max + penalty."""
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_val_grad(x, n, beta, penalty)[0]
    def grad(x):
        return smooth_val_grad(x, n, beta, penalty)[1]
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_search_focused(h_bin, n, time_limit=100, rng=None):
    """Focused binary local search."""
    if rng is None:
        rng = np.random.RandomState(42)
    
    h = (h_bin > 0.5).astype(float).copy()
    # Ensure exactly n/2 ones
    n_ones = int(np.round(np.sum(h)))
    if n_ones > n // 2:
        idx = np.where(h == 1)[0]
        to_remove = rng.choice(idx, n_ones - n // 2, replace=False)
        h[to_remove] = 0.0
    elif n_ones < n // 2:
        idx = np.where(h == 0)[0]
        to_add = rng.choice(idx, n // 2 - n_ones, replace=False)
        h[to_add] = 1.0
    
    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx_val
    start = time.time()
    
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    best_h = h.copy()
    best_max = cur_max
    
    stall = 0
    while time.time() - start < time_limit and remaining() > 15:
        k_max = int(np.argmax(corr))
        m_max = k_max - (n - 1)
        
        improved = False
        # Try swaps that reduce the max correlation
        n_try = 200
        for _ in range(n_try):
            if not ones or not zeros:
                break
            # Bias selection toward positions that contribute to the max lag
            if m_max >= 0:
                # Lag m_max: pairs (i, i+m_max) where h[i]=1, h[i+m_max]=0
                # To reduce: set h[i]=0 or h[i+m_max]=1
                candidates_i = [i for i in ones if i + m_max < n and h[i + m_max] == 0]
                if candidates_i and rng.rand() < 0.6:
                    i = candidates_i[rng.randint(len(candidates_i))]
                else:
                    i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            elif m_max < 0:
                ao = -m_max
                candidates_i = [i for i in ones if i - ao >= 0 and h[i - ao] == 0]
                if candidates_i and rng.rand() < 0.6:
                    i = candidates_i[rng.randint(len(candidates_i))]
                else:
                    i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            
            if i == j:
                continue
            
            # Perform swap
            h[i] = 0.0
            h[j] = 1.0
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            
            if new_max < cur_max - 1e-12:
                cur_max = new_max
                corr = new_corr
                best_c5 = cur_max * dx_val
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
                improved = True
                break
            else:
                # Revert
                h[i] = 1.0
                h[j] = 0.0
                ones.remove(j)
                zeros.remove(i)
                ones.append(i)
                zeros.append(j)
        
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 100:
                # Random perturbation (simulated annealing kick)
                if ones and zeros:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i] = 0.0
                    h[j] = 1.0
                    ones.remove(i)
                    zeros.remove(j)
                    ones.append(j)
                    zeros.append(i)
                    corr = compute_corr(h)
                    cur_max = float(np.max(corr))
                stall = 0
    
    if best_max < cur_max:
        h[best_h == 1] = 1.0
        h[best_h == 0] = 0.0
        # Re-fix count
        n_ones = int(np.round(np.sum(h)))
        if n_ones > n // 2:
            idx = np.where(h == 1)[0]
            to_remove = rng.choice(idx, n_ones - n // 2, replace=False)
