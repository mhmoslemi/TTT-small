import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            idx = int(np.argmin(np.abs(h - 0.5)))
            h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def smooth_max_obj(h, n, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    exps = np.exp(shifted)
    return (m + np.log(np.sum(exps)) / beta) * dx


def smooth_max_grad(h, n, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    exps = np.exp(shifted)
    w = exps / np.sum(exps)
    
    val = (m + np.log(np.sum(exps)) / beta) * dx
    grad = np.zeros(n)
    
    n_full = 2 * n - 1
    for k_full in range(n_full):
        wk = w[k_full]
        if wk < 1e-15:
            continue
        k = k_full - (n - 1)
        mk = abs(k)
        if mk == 0:
            grad[:] += wk * (1.0 - 2.0 * h)
        else:
            n_pairs = n - mk
            # For k >= 0: d(c_k)/d(h_j) = (1-h[j+k])*1[j<=n-1-k] - h[j-k]*1[j>=k]
            # For k < 0 (m=|k|): d(c_{-m})/d(h_j) = (1-h[j-m])*1[j>=m] - h[j+m]*1[j<=n-1-m]
            if k >= 0:
                grad[:n_pairs] += wk * (1.0 - h[mk:n])
                grad[mk:] -= wk * h[:n_pairs]
            else:
                grad[:n_pairs] += wk * (-h[mk:n])
                grad[mk:] += wk * (1.0 - h[:n_pairs])
    
    grad *= dx
    return val, grad


def lbfgs_optimize(h, n, beta, maxiter, budget_check):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_max_obj(x, n, dx, beta)
    
    def obj_grad(x):
        return smooth_max_grad(x, n, dx, beta)
    
    res = minimize(obj_grad, h, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12, 'iprint': 0})
    h_new = project_feasible(res.x, n)
    return h_new, compute_c5(h_new)


def simulated_annealing(h, n, iters, budget_check, rng):
    c5_best = compute_c5(h)
    h_best = h.copy()
    h_cur = h.copy()
    
    T_start = 0.05
    T_stop = 0.001
    alpha = (T_stop / T_start) ** (1.0 / max(iters, 1))
    
    for it in range(iters):
        if budget_check() < 3.0:
            break
        T = T_start * (alpha ** it)
        
        # Flip 1-3 random positions
        n_flips = rng.randint(1, 4)
        idxs = rng.choice(n, size=n_flips, replace=False)
        for idx in idxs:
            if h_cur[idx] > 0.5:
                h_cur[idx] = h_cur[idx] - 1.0
            else:
                h_cur[idx] = h_cur[idx] + 1.0
        h_cur = project_feasible(h_cur, n)
        
        c5_cur = compute_c5(h_cur)
        delta = c5_cur - c5_best
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h_cur.copy()
        else:
            h_cur = h_best.copy()
    
    return h_best, c5_best


def coordinate_descent(h, n, budget_check, rng, max_sweeps=20):
    c5_best = compute_c5(h)
    h_best = h.copy()
    
    for sweep in range(max_sweeps):
        if budget_check() < 3.0:

        any_imp = False
        order = rng.permutation(n)

                    c5_best = c5_test
                    h_best = trial.copy()
                    any_imp = True
                    break

            break
