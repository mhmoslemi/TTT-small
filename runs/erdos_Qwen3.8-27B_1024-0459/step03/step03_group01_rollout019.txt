def interp_to(src_h, src_n, dst_n):

    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def make_thue_morse(n):

    T_end = cur_max * 0.001
    best_max = cur_max
    best_h = h.copy()
    start = time.time()
    
    for it in range(max_iters):
        t_frac = it / max_iters
        T = T_start * (T_end / T_start) ** t_frac

            break

            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        h[i] = 0; h[j] = 1

                best_h = h.copy()
        else:
            h[j] = 0; h[i] = 1
    
    return best_h, best_max * dx_val
