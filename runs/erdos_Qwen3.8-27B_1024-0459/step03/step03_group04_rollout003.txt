    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-13:
        return h
    lo = -1.0
    hi = 1.0
    # We want f(tau)=sum clip(h-tau,0,1) = target. f decreasing. If diff>0 need increase sum => decrease tau (more negative). Bisection works regardless.
    for _ in range(60):
       mid=(lo+hi)*0.5
       s=np.clip(h-mid,0.0,1.0).sum()
       if s > target:
          lo=mid
       else:
          hi=mid
    tau=(lo+hi)*0.5

def setup_slq(p,n):
   global _sq_n,_sq_w0,_sq_terms
   _sq_n=n
   _sq_w0=float(p[0]) if n>0 else 0.0
   terms=[]
   for d in range(1,n):
      wn=float(p[d-1])
      wp=float(p[n+d-1])
      if wn > 1e-13 or wp > 1e-13:
         terms.append((d,wn,wp))
   _sq_terms=terms
   global _sq_last_id; _sq_last_id=None; _sq_last_val=None; _sq_last_grad=None

def slq_calc(x):
   n=_sq_n
   g=np.zeros(n)
   L=0.0
   w0=_sq_w0
   if w0 > 1e-15:
      L += w0 * float(np.dot(x, 1.0 - x))
      g += w0 * (1.0 - 2.0 * x)
   for d,wn,wp in _sq_terms:
      Ld=x[d:]
      Ll=x[:n-d]
      if wn > 1e-15:
         L += wn * (float(np.sum(Ld)) - float(np.dot(Ld, Ll)))
         g[d:] += wn
         g[d:] -= wn * Ll
         g[:n-d] -= wn * Ld
      if wp > 1e-15:
         L += wp * (float(np.sum(Ll)) - float(np.dot(Ll, Ld)))
         g[:n-d] += wp
         g[:n-d] -= wp * Ld
         g[d:] -= wp * Ll
   return L,g

def slq_obj(x):
   global _sq_last_id,_sq_last_val,_sq_last_grad
   val,grad=slq_calc(x)
   _sq_last_id=id(x); _sq_last_val=val; _sq_last_grad=grad
   return val
def slq_grad(x):
   global _sq_last_id,_sq_last_val,_sq_last_grad
   if _sq_last_id is not None and _sq_last_id == id(x) and _sq_last_grad is not None:
      return _sq_last_grad
   val,grad=slq_calc(x)
   _sq_last_id=id(x); _sq_last_val=val; _sq_last_grad=grad
   return grad
def slq_con(x): return float(np.sum(x) - _sq_n/2.0)
def slq_con_jac(x): return np.ones(_sq_n, dtype=float)

import numpy as np

def project_box_sum(h, target):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)

    s = float(h.sum())
    diff = target - s
    if abs(diff) < 1e-13:
        return h

    lo = -1.0
    hi = 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(h - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid

    tau = 0.5 * (lo + hi)
    h = np.clip(h - tau, 0.0, 1.0)

    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            if diff > 0:
                cand = np.where(h < 1.0 - 1e-12)[0]
            else:
                cand = np.where(h > 1e-12)[0]
            if len(cand) > 0:
                h[int(cand[0])] = np.clip(h[int(cand[0])] + diff, 0.0, 1.0)
    return h


def compute_vals(h):
    n = len(h)
    g = 1.0 - h
    raw = np.correlate(h, g, mode='full')
    vals = np.empty(2 * n - 1, dtype=float)
    vals[0] = raw[n - 1]
    if n > 1:
        vals[1:n] = raw[n:2 * n - 1]
        vals[n:2 * n - 1] = raw[0:n - 1][::-1]

    small_neg = (vals < 0.0) & (vals > -1e-10)
    if np.any(small_neg):
        vals[small_neg] = 0.0
    return vals


def evaluate_h(h):
    vals = compute_vals(h)
    m = float(np.max(vals))
    if m < 0.0:
        m = 0.0
    return m * (2.0 / float(len(h)))


def project_eval(h, n):
    hp = project_box_sum(h, 0.5 * float(n))
    cp = evaluate_h(hp)
    return hp, cp


def interp_linear(src, src_n, dst_n):
    if src_n == dst_n:
        return project_box_sum(src, 0.5 * float(dst_n))
    xo = (np.arange(src_n, dtype=float) + 0.5) * (2.0 / float(src_n))
    xn = (np.arange(dst_n, dtype=float) + 0.5) * (2.0 / float(dst_n))
    h = np.interp(xn, xo, src)
    return project_box_sum(h, 0.5 * float(dst_n))


def make_interp_candidate(src, src_n, dst_n):

    c1 = evaluate_h(h1)

    hbest = h1
    cbest = c1

    if dst_n > 0 and src_n > 0 and (dst_n % src_n) == 0:
        rep = np.repeat(src, dst_n // src_n)
        rep, c2 = project_eval(rep, dst_n)
        if c2 < cbest:
            hbest = rep
            cbest = c2
    return hbest, cbest


def rank_to_binary(scores, n):
    h = np.zeros(n, dtype=float)
    target = n // 2
    if target <= 0:
        return h
    if target >= n:
        h[:] = 1.0
        return h
    order = np.argsort(scores)
    h[order[-target:]] = 1.0
    return h


def thue_scores(n):
    scores = np.empty(n, dtype=float)
    for i in range(n):
        v = i
        p = 0
        while v:
            p ^= (v & 1)
            v >>= 1
        scores[i] = 1.0 if p else -1.0
    return scores


def pattern_inits(n, rng, limit=6):
    out = []
    if limit <= 0:
        return out

    out.append(rank_to_binary(thue_scores(n), n))
    limit -= 1

    while len(out) < limit:
        out.append(rank_to_binary(rng.rand(n), n))

    if len(out) < limit:
        t = np.arange(n, dtype=float) / float(n)
        freqs = [2, 3, 5, 7, 11]
        phs = [0.0, math.pi / 4.0, math.pi / 2.0]
        for f in freqs:
            for ph in phs:
                if len(out) >= limit:
                    break
                out.append(rank_to_binary(np.sin(2.0 * math.pi * f * t + ph), n))
            if len(out) >= limit:
                break

    if len(out) < limit:
        gr = 0.5 * (1.0 + math.sqrt(5.0))
        out.append(rank_to_binary(np.sin(2.0 * math.pi * gr * np.arange(n, dtype=float)), n))

    return out[:limit]


def binary_ify(h, n):
    if n % 2 != 0:
        return project_box_sum(h, 0.5 * float(n))
    h2 = np.zeros(n, dtype=float)
    target = n // 2
    if target > 0:
        order = np.argsort(h)
        h2[order[-target:]] = 1.0
    return project_box_sum(h2, 0.5 * float(n))

def softmax_weights(vals, beta):
    m = float(np.max(vals))
    z = beta * (vals - m)
    z = np.where(z < -60.0, -60.0, z)
    p = np.exp(z)
    s = float(p.sum())
    if s <= 0.0:
        p = np.full_like(vals, 1.0 / max(1, len(vals)))
        return p
    p /= s
    p[p < 1e-12] = 0.0
    s = float(p.sum())
    if s > 0.0:
        p /= s
    return p


def weighted_gradient(h, p, n):
    g = np.zeros(n, dtype=float)
    w0 = float(p[0])
    if w0 > 1e-15:
        g += w0 * (1.0 - 2.0 * h)

    for d in range(1, n):
        wn = float(p[d - 1])
        wp = float(p[n + d - 1])
        if wn <= 1e-15 and wp <= 1e-15:
            continue

        L = n - d
        hl = h[:L]
        hd = h[d:]

        if wn > 1e-15:
            g[d:] += wn
            g[d:] -= wn * hl
            g[:L] -= wn * hd

        if wp > 1e-15:
            g[:L] += wp
            g[:L] -= wp * hd
            g[d:] -= wp * hl

    return g


def pair_coordinate_pass(h, g, pairw, w0, n, idx, max_passes, deadline, tol):
    half_pw = 0.5 * pairw
    moves = 0

    for _pass in range(max_passes):
        pass_moves = 0
        for i in range(n):
            if (i & 63) == 0 and time.time() > deadline:
                return h, g, moves

            hi = float(h[i])
            if hi <= 1e-14:
                continue

            B = np.minimum(hi, 1.0 - h)
            B[i] = 0.0

            a = g - g[i]
            d = np.abs(idx - i)
            q = 2.0 * w0 - pairw[d]

            fB = a * B - q * B * B
            fS = np.full(n, np.inf, dtype=float)

            mask = np.abs(q) > 1e-12
            ds = np.zeros(n, dtype=float)
            if np.any(mask):
                np.divide(a, 2.0 * q, out=ds, where=mask)
                valid = mask & (ds > 1e-15) & (ds < B - 1e-15)
                if np.any(valid):
                    fS[valid] = (a[valid] * ds[valid]
                                 - q[valid] * ds[valid] * ds[valid])

            fBest = np.minimum(fB, fS)
            fBest = np.minimum(fBest, 0.0)
            fBest[i] = 0.0

            j = int(np.argmin(fBest))
            decr = -float(fBest[j])

            if j == i or decr <= tol:
                continue

            if fB[j] <= fS[j]:
                delta = float(B[j])
            else:
                delta = float(ds[j])

            if delta <= 0.0:
                delta = float(B[j])
            if delta > B[j]:
                delta = float(B[j])
            if delta <= 1e-15:
                continue

            hi = float(h[i])
            hj = float(h[j])
            if delta > hi:
                delta = float(hi)
            if delta > 1.0 - hj:
                delta = float(1.0 - hj)
            if delta <= 1e-15:
                continue

            h[i] = hi - delta
            h[j] = hj + delta

            row_a = half_pw[d]
            row_a[i] = w0
            db = np.abs(idx - j)
            row_b = half_pw[db]
            row_b[j] = w0
            g += -2.0 * delta * (row_b - row_a)

            pass_moves += 1
            moves += 1

        if pass_moves == 0:
            break

    return h, g, moves


def active_coordinate_improve(start_h, n, beta, deadline, best_h, best_c):

    best_h = np.asarray(best_h, dtype=float).copy()

        max_reweights = 60
        max_passes = 45
    elif n <= 512:
        max_reweights = 35
        max_passes = 28
    elif n <= 1024:
        max_reweights = 22
        max_passes = 16
    else:
        max_reweights = 14
        max_passes = 9

    tol = 1e-10 + 1e-13 * float(n)
    prev_best = best_c
    stagn = 0

    for rw in range(max_reweights):
        if time.time() > deadline:
            break

        vals = compute_vals(h)
        tmax = float(np.max(vals))
        c = tmax * (2.0 / float(n))

            pairw[1:] = p[1:n] + p[n:2 * n - 1]

        g = weighted_gradient(h, p, n)

        h, c2 = project_eval(h, n)

def setup_slq(p, n):
    global _sq_n, _sq_w0, _sq_terms
    _sq_n = int(n)
    _sq_w0 = float(p[0]) if n > 0 else 0.0
    terms = []
    for d in range(1, n):
        wn = float(p[d - 1])
        wp = float(p[n + d - 1])
        if wn > 1e-13 or wp > 1e-13:
            terms.append((d, wn, wp))
    _sq_terms = terms


def slq_calc(x):
    n = _sq_n
    g = np.zeros(n, dtype=float)
    val = 0.0
    w0 = _sq_w0

    if w0 > 1e-15:
        val += w0 * float(np.dot(x, 1.0 - x))
        g += w0 * (1.0 - 2.0 * x)

    for d, wn, wp in _sq_terms:
        L = n - d
        xl = x[:L]
        xd = x[d:]

        if wn > 1e-15:
            val += wn * (float(np.sum(xd)) - float(np.dot(xd, xl)))
            g[d:] += wn
            g[d:] -= wn * xl
            g[:L] -= wn * xd

        if wp > 1e-15:
            val += wp * (float(np.sum(xl)) - float(np.dot(xl, xd)))
            g[:L] += wp
            g[:L] -= wp * xd
            g[d:] -= wp * xl

    return val, g


def slq_obj(x):
    v, _ = slq_calc(x)
    return v


def slq_grad(x):
    _, gr = slq_calc(x)
    return gr


def slq_con(x):
    return float(np.sum(x) - 0.5 * float(_sq_n))


def slq_con_jac(x):
    return np.ones(_sq_n, dtype=float)


def slsqp_polish(h0, n, deadline, best_h, best_c):
    h, c = project_eval(h0, n)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    for beta in [400, 1500]:
        if time.time() > deadline:
            break

        vals = compute_vals(h)
        p = softmax_weights(vals, beta)
        setup_slq(p, n)

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': slq_con, 'jac': slq_con_jac}]

        try:
            res = minimize(slq_obj, h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 120, 'ftol': 1e-16, 'iprint': 0})
            hn, cn = project_eval(res.x, n)
            if cn < best_c:
                best_c = cn
                best_h = hn.copy()
                h = hn.copy()
        except Exception:
            pass

    return best_h, best_c

        n_restart = best_n if 200 <= best_n <= 512 else 320
        if n_restart % 2 != 0:
            n_restart += 1

        for restart in range(4):
            if time.time() > t_start + budget - 35.0:
                break
            rng = np.random.RandomState(seed_child + 9001 + restart)
            cands = pattern_inits(n_restart, rng, 3)
            hp = None
            c_init = None
            for hp0 in cands:
                hp0, cp = project_eval(hp0, n_restart)
                if c_init is None or cp < c_init:
                    hp = hp0
                    c_init = float(cp)

            if hp is not None and c_init < best_c5 + 0.05:
                deadline = time.time() + max(20.0, min(70.0, 0.2 * (t_start + budget - time.time())))
                best_h, best_c5 = optimize_fixed(hp, n_restart, deadline, best_h, best_c5, seed_child)
                best_n = int(len(best_h))
            seed_child += 1

    rem = t_start + budget - time.time()
    if rem > 40.0 and best_n <= 800:
        deadline = time.time() + min(rem - 12.0, 220.0)
        best_h, best_c5, _ = active_coordinate_improve(
            best_h, best_n, 3500.0, deadline, best_h, best_c5
        )
        best_n = int(len(best_h))

    if best_n != len(best_h):
        best_n = int(len(best_h))

    final_h, final_c = project_eval(best_h, best_n)
    if final_c < best_c5:
        best_h = final_h.copy()
        best_c5 = float(final_c)
    else:
        final_h = best_h
        final_c = float(best_c5)

    try:
        ec5 = evaluate_erdos_solution(final_h)
        ec5 = float(ec5)
        if ec5 < final_c:
            final_c = ec5
        elif ec5 > final_c + 1e-7:
            final_c = ec5
    except Exception:
        pass

    return (final_h, float(final_c), int(best_n))
