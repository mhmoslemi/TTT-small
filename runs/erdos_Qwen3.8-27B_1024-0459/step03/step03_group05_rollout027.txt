So:
- For m in [0, n-1-k]: contribution from first term: (1-h[m+k])
- For m in [k, n-1]: contribution from second term: -h[m-k]

grad[k:] += -h[:n-k]       # second term

dO/dh[m] = (1-h[m-j]) for m >= j (from h[i] term)
         + (-h[m+j]) for m < n-j (from h[i-j] term)

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(60):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nfree = np.sum(free)
        if nfree > 0:
            h[free] += diff / nfree
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_max_value(h, beta, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sm = m + np.log(np.sum(exps)) / beta
    sum_err = np.sum(h) - target
    pen = 500.0 * sum_err * sum_err
    return sm * dx + pen


def active_gradient(h, top_k):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr_raw)[-top_k:]
    w = np.exp(corr_raw[idx] - corr_raw[idx].max())
    w /= np.sum(w)

    grad = np.zeros(n)
    for wi, ki in zip(w, idx):
        k = int(ki) - (n - 1)
        if k == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif k > 0:
            if k < n:
                grad[:n - k] += wi * (1.0 - h[k:])
                grad[k:] += -wi * h[:n - k]
        else:
            j = -k
            if j < n:
                grad[j:] += wi * (1.0 - h[:n - j])
                grad[:n - j] += -wi * h[j:]
    return grad * dx


def lbfgs_func(x, beta, target):
    return smooth_max_value(x, beta, target)


def nesterov_phase(h, target, max_time, rng):
    n = len(h)
    t0 = time.time()
    v = np.zeros(n)
    lr = 0.005
    momentum = 0.85
    best_local = h.copy()
    best_local_c5 = compute_c5(h)

    for it in range(20000):
        if time.time() - t0 > max_time:
            break
        y = h + momentum * v
        y = np.clip(y, 0.0, 1.0)
        y = project_feasible(y, target)
        g = active_gradient(y, top_k=5)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-16:
            break
        v = momentum * v - lr * g / gnorm * 0.1
        h_new = project_feasible(h + v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_local_c5:
            best_local_c5 = c5_new
            best_local = h_new.copy()
        h = h_new
        if it % 200 == 0 and it > 0:
            lr *= 0.95
            momentum = min(0.95, momentum * 1.005)

    return best_local, best_local_c5


def simulated_annealing(h, target, max_time, rng):
    n = len(h)
    t0 = time.time()
    T = 5e-5
    T_min = 1e-8
    cooling = 0.9997
    best_local = h.copy()
    best_local_c5 = compute_c5(h)
    current_c5 = best_local_c5

    for it in range(50000):
        if time.time() - t0 > max_time:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.choice([-0.02, -0.01, -0.005, 0.005, 0.01, 0.02])
        h_save = h.copy()
        h[i] += delta
        h[j] -= delta
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
        c5_new = compute_c5(h)
        delta_c5 = c5_new - current_c5
        accept = delta_c5 < 0 or (T > T_min and rng.random() < np.exp(-delta_c5 / T))
        if accept:
            current_c5 = c5_new
            if c5_new < best_local_c5:
                best_local_c5 = c5_new
                best_local = h.copy()
        else:
            h = h_save
        T *= cooling

    return best_local, best_local_c5

                print("  Higher res n={}: C5={:.7f}".format(n2, c5_new))

    # Final projection and verification
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
