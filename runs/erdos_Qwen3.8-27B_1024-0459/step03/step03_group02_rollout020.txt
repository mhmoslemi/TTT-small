import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        interior = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.sum(interior) > 0:
            h[interior] += diff / np.sum(interior)
            h = np.clip(h, 0.0, 1.0)
        else:
            idx = np.argmin(np.abs(h - 0.5))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def smooth_max_value_and_weights(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = beta * (corr - m)
    s = np.minimum(s, 50.0)
    ws = np.exp(s)
    ws_sum = ws.sum()
    ws /= ws_sum
    val = (m + np.log(ws_sum) / beta) * dx
    return val, ws, corr

def smooth_max_grad_from_weights(ws, h, n):
    g = np.zeros(n)
    for j in range(2 * n - 1):
        w = ws[j]
        if w < 1e-12:
            continue
        d = j - (n - 1)
        if d >= 0:
            g[:n - d] += w * (1.0 - h[d:n])
            g[d:] -= w * h[:n - d]
        else:
            ad = -d
            g[ad:] += w * (1.0 - h[:n - ad])
            g[:n - ad] -= w * h[ad:]
    return g * (2.0 / n)

def slsqp_optimize(h, n, beta, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = beta * (corr - m)
        s = np.minimum(s, 50.0)
        return (m + np.log(np.sum(np.exp(s))) / beta) * dx
    def obj_jac(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = beta * (corr - m)
        s = np.minimum(s, 50.0)
        ws = np.exp(s)
        ws /= ws.sum()
        g = np.zeros(n)
        for j in range(2 * n - 1):
            w = ws[j]
            if w < 1e-12:
                continue
            d = j - (n - 1)
            if d >= 0:
                g[:n - d] += w * (1.0 - x[d:n])
                g[d:] -= w * x[:n - d]
            else:
                ad = -d
                g[ad:] += w * (1.0 - x[:n - ad])
                g[:n - ad] -= w * x[ad:]
        return g * dx
    def constraint(x):
        return np.sum(x) - n / 2.0
    try:
        res = minimize(obj, h, jac=obj_jac, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': constraint}],
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n

def simulated_anneal(h, n, n_steps=20000):
    h = h.copy()
    best_h = h.copy()

        if dE < 0 or np.random.random() < np.exp(-dE / T):

            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h = saved
        T = max(T_min, T * cool_rate)
    return best_h, best_c5
