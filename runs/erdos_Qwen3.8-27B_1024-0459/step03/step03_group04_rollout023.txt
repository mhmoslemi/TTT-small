def make_random_binary(n, rng):

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return h

    t = np.arange(n) / float(n)
    h = 0.5 + 0.3 * np.sin(2.0 * np.pi * freq * t + phase)
    return h
