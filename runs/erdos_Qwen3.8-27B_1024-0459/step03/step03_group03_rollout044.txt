            g[p:] -= h[:n - p]
    return g

    h, c5 = _pswap(h, n, 80, rng)
    if c5 < best_c5:
        best_h, best_c5 = h.copy(), c5

    h, c5 = _sgloop(h, n, 300, rng)
    if c5 < best_c5:
        best_h, best_c5 = h.copy(), c5

    h, c5 = _coorddescent(h, n, 150, rng)
    if c5 < best_c5:
        best_h, best_c5 = h.copy(), c5

    for beta in [3000, 8000]:
        if _tl() < 10:
            break
        h2 = _slsqp(best_h, n, beta, 400)
        c2 = _c5(h2)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2
            h, c5 = h2, c2
        else:
            h, c5 = best_h, best_c5

    h, c5 = _sgloop(h, n, 250, rng)
    if c5 < best_c5:
        best_h, best_c5 = h.copy(), c5

    h, c5 = _pswap(h, n, 50, rng)
    if c5 < best_c5:
        best_h, best_c5 = h.copy(), c5

    return best_h, best_c5


def _random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)[:n // 2]
    h[idx] = 1.0
    return _proj(h, n)


def _perturb(h, n, rng, scale=0.03):
    h2 = h.copy()
    mask = (~np.isclose(h2, 0, atol=0.01)) & (~np.isclose(h2, 1, atol=0.01))
    if np.sum(mask) > 0:
        h2[mask] += rng.uniform(-scale, scale, size=np.sum(mask))
    return _proj(h2, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _t0, _tb
    _t0 = time.time()
    _tb = budget_s
    rng = np.random.RandomState(seed)

    try:
        hpar = np.array(initial_h_values, dtype=np.float64)
        npar = len(hpar)
    except Exception:
        npar = 160
        hpar = np.zeros(npar, dtype=np.float64)
        hpar[:npar // 2] = 1.0

    bh = _proj(hpar.copy(), npar)
    bc5 = _c5(bh)
    bn = npar
    print("Init: n={} C5={:.8f}".format(bn, bc5))

    # Phase 1: Moderate optimization at parent resolution (35% budget)
    phase1_end = _t0 + 0.35 * budget_s
    while _tl() > (phase1_end - _t0) + 30:
        h, c5 = _slsqp(bh, bn, 200, 200)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _sgloop(bh, bn, 300, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _pswap(bh, bn, 50, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        print("Phase1: n={} C5={:.8f}".format(bn, bc5))
        if _tl() < 60:
            break

    # Phase 2: Try higher resolutions (40% budget)
    phase2_end = _t0 + 0.72 * budget_s
    resolutions = []
    # Generate candidate resolutions
    for dr in [10, 20, 30, 40, -10, -20, 60, 80]:
        nr = bn + dr
        if nr >= 40 and nr <= 400 and nr % 2 == 0:
            resolutions.append(nr)
    # Also try some divisors/multiples
    for nr in [200, 240, 280, 320, 192, 224]:
        if nr >= 40 and nr <= 400 and nr % 2 == 0 and nr not in resolutions:
            resolutions.append(nr)
    resolutions = sorted(set(resolutions), key=lambda x: abs(x - bn))[:8]

    for nr in resolutions:
        if _tl() < (phase2_end - _t0) + 45:
            break
        hi = _proj(_interp_f(bh, bn, nr), nr)
        # Add small perturbation for diversity
        hi = _perturb(hi, nr, rng, 0.01)
        h2, c2 = _full_opt(hi, nr, rng)
        if c2 < bc5:
            bh, bc5, bn = h2, c2, nr
            print("Phase2: n={} C5={:.8f}".format(bn, bc5))

    # Also try a few random binary initializations at current best resolution
    for trial in range(4):
        if _tl() < (phase2_end - _t0) + 60:
            break
        hr = _random_binary(bn, rng)
        h2, c2 = _full_opt(hr, bn, rng)
        if c2 < bc5:
            bh, bc5 = h2, c2
            print("Phase2-Rand{}: C5={:.8f}".format(trial, c5))

    # Phase 3: Intensive polishing at best resolution (25% budget)
    phase3_end = _t0 + 0.95 * budget_s
    while _tl() > (phase3_end - _t0) + 20:
        # SLSQP with increasing beta
        for beta in [2000, 5000, 10000, 20000]:
            if _tl() < (phase3_end - _t0) + 15:
                break
            h2 = _slsqp(bh, bn, beta, 500)
            c2 = _c5(h2)
            if c2 < bc5 - 1e-15:
                bh, bc5 = h2, c2
                print("Phase3-SLSQP beta={}: C5={:.8f}".format(beta, bc5))

        h, c5 = _sgloop(bh, bn, 300, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _pswap(bh, bn, 60, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _sgloop(bh, bn, 200, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _coorddescent(bh, bn, 100, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        print("Phase3: C5={:.8f}".format(bc5))
        if _tl() < 40:
            break

    # Phase 4: Final polish with very high beta SLSQP
    if _tl() > 15:
        for beta in [30000, 50000, 100000]:
            if _tl() < 10:
                break
            h2 = _slsqp(bh, bn, beta, 600)
            c2 = _c5(h2)
            if c2 < bc5 - 1e-15:
                bh, bc5 = h2, c2
                print("Phase4 beta={}: C5={:.8f}".format(beta, bc5))

    if _tl() > 10:
        h, c5 = _sgloop(bh, bn, 150, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _pswap(bh, bn, 30, rng)
        if c5 < bc5:
            bh, bc5 = h, c5

    bh = _proj(bh, bn)
    fc5 = _c5(bh)
    try:
        ec = evaluate_erdos_solution(bh)
        if ec < fc5:
            fc5 = ec
    except Exception:
        pass
    print("FINAL: n={} C5={:.8f}".format(bn, fc5))
    return (bh, fc5, bn)
