def fast_coord_descent(h, n, sweeps=30, time_limit=None, t0=None):

    dx = 2.0 / n
    best = h.copy()
    best_c5 = np.max(np.correlate(h, 1.0-h, mode='full')) * dx
    
    for sw in range(sweeps):
        if time_limit is not None and t0 is not None and (time.time() - t0) > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        for j in range(n):
            if time_limit is not None and t0 is not None and (time.time() - t0) > time_limit:
                break
            old = h[j]
            
            # Affected lag range
            # First factor (h[j] as h[i]): d in [-j, n-1-j]
            # Second factor (h[j] as h[i+d]): d in [j-n+1, j]
            d_lo = max(min(-j, j - n + 1), -(n-1))
            d_hi = min(max(n - 1 - j, j), n-1)
            n_affected = d_hi - d_lo + 1
            
            # Precompute for each affected lag
            ds = np.arange(d_lo, d_hi + 1)
            c_idx = ds + (n - 1)
            c_vals = corr[c_idx]
            
            # For each d, determine first_valid and second_valid
            first_valid = (ds >= -j) & (ds <= n - 1 - j)
            second_valid = (ds >= j - n + 1) & (ds <= j)
            
            rest = c_vals.copy()
            coef = np.zeros(n_affected)
            
            quad_mask = (ds == 0)
            
            # For non-quad lags:
            # If first_valid: r -= old*(1-h[j+d]), a += (1-h[j+d])
            # If second_valid: r -= h[j-d]*(1-old), a += -h[j-d]
            
            fv = first_valid & ~quad_mask
            sv = second_valid & ~quad_mask
            
            # First factor contribution
            if np.any(fv):
                # j + ds[fv] must be in [0, n-1]
                idx_fv = j + ds[fv]
                rest[fv] -= old * (1.0 - h[idx_fv])
                coef[fv] += (1.0 - h[idx_fv])
            
            if np.any(sv):
                # j - ds[sv] must be in [0, n-1]
                idx_sv = j - ds[sv]
                rest[sv] -= h[idx_sv] * (1.0 - old)
                coef[sv] -= h[idx_sv]
            
            # Quad lag (d=0): rest = c_val - old*(1-old), coef marked specially
            if np.any(quad_mask):
                q_idx = quad_mask[0]  # position in the array
                rest[q_idx] = c_vals[q_idx] - old * (1.0 - old)
            # coef[q_idx] stays 0, we handle it separately
            
            # Max unaffected
            full_mask = np.ones(2*n-1, dtype=bool)
            full_mask[c_idx] = False
            max_unaff = np.max(corr[full_mask]) if np.any(full_mask) else 0.0
            
            # 1D optimization
            xs = np.linspace(0.0, 1.0, 32)
            vals = rest[None, :] + coef[None, :] * xs[:, None]
            if np.any(quad_mask):
                q_pos = np.where(quad_mask)[0][0]
                vals[:, q_pos] = rest[q_pos] + xs - xs**2
            fvals = np.maximum(max_unaff, np.max(vals, axis=1))
            best_i = np.argmin(fvals)
            best_x = xs[best_i]
            
            # Ternary refine
            span = 2.0 / 32
            lo = max(0.0, best_x - span)
            hi = min(1.0, best_x + span)
            for _ in range(25):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                vals1 = rest + coef * m1
                vals2 = rest + coef * m2
                if np.any(quad_mask):
                    vals1[q_pos] = rest[q_pos] + m1 - m1**2
                    vals2[q_pos] = rest[q_pos] + m2 - m2**2
                v1 = max(max_unaff, np.max(vals1))
                v2 = max(max_unaff, np.max(vals2))
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            h[j] = (lo + hi) / 2.0
        
        h = project_feasible(h, n)
        c5 = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    
    return best, best_c5

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for i in order:
            amt = min(abs(h[i] if diff < 0 else 1.0 - h[i]), abs(diff))
            h[i] += diff * (amt / abs(diff)) if diff != 0 else 0
            diff -= diff * (amt / abs(diff)) if abs(diff) > 0 else 0
            if abs(diff) < 1e-14:
                break
        h = np.clip(h, 0.0, 1.0)
    return h

    grid_xs = np.linspace(0.0, 1.0, 48)
    
    for sw in range(sweeps):
        if time_limit is not None and t0 is not None and (time.time() - t0) > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        improved = 0
        
        for j in range(n):
            if time_limit is not None and t0 is not None and (time.time() - t0) > time_limit:
                break
            old = h[j]
            
            d_lo = max(min(-j, j - n + 1), -(n-1))
            d_hi = min(max(n - 1 - j, j), n-1)
            n_aff = d_hi - d_lo + 1
            ds = np.arange(d_lo, d_hi + 1)
            c_idx = ds + (n - 1)
            c_vals = corr[c_idx].copy()
            
            first_valid = (ds >= -j) & (ds <= n - 1 - j)
            second_valid = (ds >= j - n + 1) & (ds <= j)
            quad_mask = (ds == 0)
            
            rest = c_vals.copy()
            coef = np.zeros(n_aff)
            
            fv = first_valid & ~quad_mask
            sv = second_valid & ~quad_mask
            
            if np.any(fv):

                rest[fv] -= old * (1.0 - h[idx_fv])
                coef[fv] += (1.0 - h[idx_fv])
            
            if np.any(sv):
                idx_sv = j - ds[sv]
                rest[sv] -= h[idx_sv] * (1.0 - old)

            if np.any(quad_mask):

def topk_subgradient(h, n, iters=500, time_limit=None, t0=None):
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n
    best = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    
    k_phases = [(3, 200), (5, 200), (10, 100)]
    lr = 0.01
    n_total = 2 * n - 1
    
    for k, n_it in k_phases:
        for it in range(n_it):
            if time_limit is not None and t0 is not None and (time.time() - t0) > time_limit:
                return best, best_c5
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_curr = np.max(corr) * dx
            
            # Top-k lags
            top_idx = np.argsort(corr)[-k:]
            
            # Subgradient
            grad = np.zeros(n)
            for lidx in top_idx:
                lag = lidx - (n - 1)
                w = 1.0 / k
                if lag >= 0:
                    # corr[lag] = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
                    # d/dh[j]: (1-h[j+lag]) for j <= n-1-lag; -h[j-lag] for j >= lag
                    # Combined: for j in [0, n-1-lag]: grad += w*(1-h[j+lag])
                    #           for j in [lag, n-1]: grad += w*(-h[j-lag])
                    # But these overlap for j in [lag, n-1-lag]
                    if lag == 0:
                        grad[:] += w * (1.0 - 2.0 * h)
                    else:
                        # First part: j from 0 to n-1-lag
                        grad[0:n-lag] += w * (1.0 - h[lag:n])
                        # Second part: j from lag to n-1
                        grad[lag:n] += w * (-h[0:n-lag])
                else:
                    o = -lag
                    # corr[-o] = sum_{i=o}^{n-1} h[i]*(1-h[i-o])
                    # d/dh[j]: (1-h[j-o]) for j >= o; -h[j+o] for j <= n-1-o
                    if o == 0:
                        pass  # handled above
                    else:
                        grad[o:n] += w * (1.0 - h[0:n-o])
                        grad[0:n-o] += w * (-h[o:n])
            
            # Normalize gradient
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-10:
                h_new = h - lr * grad / gnorm * gnorm
                # Actually: h_new = h - lr * grad
                h_new = h - lr * grad
                h_new = project_feasible(h_new, n)
                c5_new = compute_c5(h_new)
                
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best = h_new.copy()
                    h = h_new
                    lr = min(lr * 1.05, 0.05)
                else:
                    lr *= 0.8
            else:
                lr *= 0.9
    
    return best, best_c5

def interpolate_h(h_src, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_new = np.interp(xn, xp, h_src)
    return project_feasible(h_new, n_dst)

def slsqp_polish(h, n, time_limit=None, t0=None):
    h = project_feasible(h.copy(), n)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        beta = 200.0
        sh = np.clip(beta * (corr - m), -500.0, 0.0)
        return (m + np.log(np.sum(np.exp(sh))) / beta) * (2.0 / n)
    
    def constr(x):
        return np.sum(x) - n / 2.0
    
    best = h.copy()
    best_c5 = compute_c5(h)
    
    for maxiter in [100, 200]:
        if time_limit is not None and t0 is not None and (time.time() - t0) > time_limit:
            break
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': constr}],
                          options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
            h_t = project_feasible(res.x, n)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_c5 = c5_t
                best = h_t.copy()
                h = h_t
        except Exception:
            pass
    
    return best, best_c5

            # Second factor: j in [0, n-1-o], contribution -h[j+o]

def project_feasible(h, n):
    target = n / 2.0

    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-13
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        rem = diff
        for i in order:
            if abs(rem) < 1e-14:
                break
            if rem > 0:
                amt = min(1.0 - h[i], rem)
            else:
                amt = max(-h[i], rem)
            h[i] += amt
            rem -= amt
    h = np.clip(h, 0.0, 1.0)
    return h

            o = -lag
            grad[o:n] += w * (1.0 - h[0:n - o])

            best_c5 = c_p2
            best_h = h_p2.copy()

    # Another CD pass
    if tl() > 25:
        h_p3, c_p3 = fast_cd(best_h, best_n, sweeps=50, time_limit=tl(), t0=t0,
                             grid_res=64, tern_iters=35)
        if c_p3 < best_c5:
            best_c5 = c_p3
            best_h = h_p3.copy()

    # Final subgradient micro-polish
    if tl() > 20:
        h_p4, c_p4 = subgradient_descent(best_h, best_n, iters=300, time_limit=tl(), t0=t0)
        if c_p4 < best_c5:
            best_c5 = c_p4
            best_h = h_p4.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
