    _G_TGT = n / 2.0
    h_curr = h.copy()
    c5_best = compute_c5(h_curr)
    step = step0
    stall = 0
    for it in range(max_iter):
        if time.time() - t0 > time_limit_s:
            break
        corr = compute_corr(h_curr)
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        k = max_idx - (n - 1)
        # Compute gradient of corr at lag k only
        grad = np.zeros(n)
        n_pairs = n - abs(k)
        if k >= 0:
            grad[:n_pairs] += (1.0 - h_curr[k:k + n_pairs])
            if k > 0:
                grad[k:] -= h_curr[:n_pairs]
        else:
            l = -k
            grad[l:] += (1.0 - h_curr[:n_pairs])
            grad[:n_pairs] -= h_curr[l:l + n_pairs]
        # Also add small contribution from lag 0 to keep things balanced
        grad += 0.1 * (1.0 - 2.0 * h_curr)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        improved = False
        for trial_step in [step, step * 0.5, step * 2, step * 0.25]:
            h_trial = h_curr - trial_step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-10:
                c5_best = c5_trial
                h_curr = h_trial
                improved = True
                break
        if improved:
            stall = 0
            if it % 50 == 0:
                step = min(step * 1.1, 0.1)
        else:
            stall += 1
            step *= 0.7
            if step < 1e-6 or stall > 30:
                break
    return h_curr, c5_best


def multi_beta_lbfgsb(h0, n, betas=[50, 200, 500, 1000, 2000, 5000], time_limit_s=300.0):
    """Run L-BFGS-B with increasing beta values."""
    t0 = time.time()
    _G_N = n
    _G_DX = 2.0 / n
    _G_TGT = n / 2.0
    h = h0.copy()
    c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        if time.time() - t0 > time_limit_s:
            break
        _G_BETA = float(beta)
        try:
            res = fmin_l_bfgs_b(penalty_obj_grad, h, fprime=None,
                                pgtol=1e-12, gtol=1e-12,
                                maxiter=500, iprint=0, pgtol1=1e-14)
            # penalty_obj_grad returns (val, grad) for func_and_grad
            # Actually fmin_l_bfgs_b expects func that returns (f, g) if fprime=None
            h_new = project_feasible(res[:], n) if isinstance(res, tuple) else project_feasible(res, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-12:
                c5 = c5_new
                h = h_new
        except Exception:
            try:
                res = minimize(penalty_obj_grad, h, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 500, 'gtol': 1e-12, 'ftol': 1e-14})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < c5 - 1e-12:
                    c5 = c5_new
                    h = h_new
            except Exception:
                pass
    return h, c5


def slate_running_descent(h, n, time_limit_s=180.0):
    """Slate's running descent: alternate between all-lag and active-lag optimization."""
    t0 = time.time()
    _G_N = n
    _G_DX = 2.0 / n
    _G_TGT = n / 2.0
    h_curr = h.copy()
    c5_best = compute_c5(h_curr)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - _G_TGT}]
    for phase in range(20):
        if time.time() - t0 > time_limit_s:
            break
        # Alternate: full smooth max then targeted
        for beta in [200, 500, 1000, 2000]:
            if time.time() - t0 > time_limit_s:
                break
            _G_BETA = float(beta)
            def obj_f(x):
                return smooth_obj_grad(x)[0]
            def obj_g(x):
                return smooth_obj_grad(x)[1]
            try:
                res = minimize(smooth_obj_grad, h_curr, jac=None, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 200, 'gtol': 1e-11, 'ftol': 1e-13})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-12:
                    c5_best = c5_new
                    h_curr = h_new
            except Exception:
                pass
        # Targeted descent on the active lag
        h_curr, c5_t = targeted_gradient_descent(h_curr, n, max_iter=500, step0=0.005, time_limit_s=15.0)
        if c5_t < c5_best:
            c5_best = c5_t
            h_curr = h_curr
    return h_curr, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    def budget_remain():
        return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = make_thue_morse(parent_n)
        parent_h = project_feasible(parent_h, parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("No parent. ThueMorse: n={}, C5={:.8f}".format(parent_n, best_c5))

    # Phase 1: Targeted gradient descent on parent resolution
    if budget_remain() > 100:
        print("\n--- Phase 1: Targeted GD on n={} ---".format(best_n))
        h_td, c5_td = targeted_gradient_descent(best_h, best_n, max_iter=3000, step0=0.01, time_limit_s=80.0)
        if c5_td < best_c5:
            best_c5 = c5_td
            best_h = h_td
            print("  Phase1a: C5={:.8f}".format(best_c5))

    # Phase 2: Multi-beta L-BFGS-B
    if budget_remain() > 200:
        print("\n--- Phase 2: L-BFGS-B multi-beta ---")
        h_lb, c5_lb = multi_beta_lbfgsb(best_h, best_n,
                                         betas=[30, 100, 300, 1000, 3000, 8000],
                                         time_limit_s=min(200, budget_remain() - 50))
        if c5_lb < best_c5:
            best_c5 = c5_lb
            best_h = h_lb
            print("  Phase2: C5={:.8f}".format(best_c5))

    # Phase 3: Slate's running descent (alternating strategies)
    if budget_remain() > 150:
        print("\n--- Phase 3: Running Descent ---")
        h_rd, c5_rd = slate_running_descent(best_h, best_n, time_limit_s=min(150, budget_remain() - 50))
        if c5_rd < best_c5:
            best_c5 = c5_rd
            best_h = h_rd
            print("  Phase3: C5={:.8f}".format(best_c5))

    # Phase 4: Additional targeted GD with smaller steps
    if budget_remain() > 80:
        print("\n--- Phase 4: Fine targeted GD ---")
        h_td2, c5_td2 = targeted_gradient_descent(best_h, best_n, max_iter=5000, step0=0.003, time_limit_s=min(80, budget_remain() - 30))
        if c5_td2 < best_c5:
            best_c5 = c5_td2
            best_h = h_td2
            print("  Phase4: C5={:.8f}".format(best_c5))

    # Phase 5: Binary local search
    if budget_remain() > 60:
        print("\n--- Phase 5: Binary local search ---")
        _G_N = best_n
        _G_DX = 2.0 / best_n
        h_bin = (best_h > 0.5).astype(float)
        # Fix count
        n_ones = int(np.sum(h_bin))
        if n_ones > best_n // 2:
            # Flip excess ones to zeros (smallest ones first)
            one_idxs = np.argsort(h_bin[np.arange(best_n) >= 0.5])[::-1] if n_ones > 0 else np.array([])
            excess = n_ones - best_n // 2
            # Just re-balance by flipping the smallest-valued ones
            ones_mask = h_bin > 0.5
            ones_vals = np.where(ones_mask, best_h, 0)
            flip_to_zero = np.argpartition(ones_vals, -excess)[-excess:] if excess > 0 else np.array([], dtype=int)
            h_bin[flip_to_zero] = 0.0
        elif n_ones < best_n // 2:
            deficit = best_n // 2 - n_ones
            zeros_mask = h_bin < 0.5
            zeros_vals = np.where(zeros_mask, 1.0 - best_h, 0)
            flip_to_one = np.argpartition(zeros_vals, -deficit)[-deficit:] if deficit > 0 else np.array([], dtype=int)
            h_bin[flip_to_one] = 1.0
        c5_bin_init = compute_c5(h_bin)
        print("  Binary init: C5={:.8f}".format(c5_bin_init))
        h_bl, c5_bl = binary_local_search(h_bin, best_n, max_rounds=300, time_limit_s=min(60, budget_remain() - 20))
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl
            print("  Phase5 binary: C5={:.8f}".format(best_c5))

    # Phase 6: SLSQP polish on best continuous solution
    if budget_remain() > 30:
        print("\n--- Phase 6: SLSQP polish ---")
        n = best_n
        _G_N = n
        _G_TGT = n / 2.0
        _G_DX = 2.0 / n
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - _G_TGT}]
        for beta_val in [2000, 5000, 10000]:
            if budget_remain() < 10:
                break
            _G_BETA = float(beta_val)
            def obj_s(x):
                return smooth_obj_grad(x)[0]
            try:
                res = minimize(obj_s, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-15, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h_new
                    print("  SLSQP beta={}: C5={:.8f}".format(beta_val, best_c5))
            except Exception:
                pass

    # Phase 7: Try different resolutions
    if budget_remain() > 60:
        print("\n--- Phase 7: Other resolutions ---")
        for n_try in [200, 256, 320, 384]:
            if budget_remain() < 30:
                break
            if n_try == best_n:
                continue
            _G_N = n_try
            _G_TGT = n_try / 2.0
            _G_DX = 2.0 / n_try
            h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
            c5_interp = compute_c5(h_interp)
            # Quick optimization
            h_opt, c5_opt = targeted_gradient_descent(h_interp, n_try, max_iter=2000, step0=0.008, time_limit_s=min(40, budget_remain() - 20))
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try
                print("  n={}: C5={:.8f}".format(n_try, best_c5))

    # Phase 8: Final fine-tuning with very small steps
    if budget_remain() > 20:
        print("\n--- Phase 8: Final fine-tune ---")
        n = best_n
        _G_N = n
        _G_TGT = n / 2.0
        _G_DX = 2.0 / n
        h_final, c5_final = targeted_gradient_descent(best_h, n, max_iter=3000, step0=0.002, time_limit_s=min(20, budget_remain() - 5))
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
