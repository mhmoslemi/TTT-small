def project_feasible(h, n):
    """Project h to feasible set: 0<=h<=1, sum(h)=n/2"""
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

        diff = target - np.sum(h)

            break

            # Don't push any value outside [0,1]

def project_feasible(h, n):

    target = n / 2.0
    for _ in range(200):

        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-16 and n > 0:
        order = np.argsort(np.abs(h - 0.5))
        for idx_i in order:
            if diff > 0:
                room = 1.0 - h[idx_i]
                add = min(diff, room)
            else:
                room = h[idx_i]
                add = max(diff, -room)
            h[idx_i] += add
            diff -= add
            if abs(diff) < 1e-16:

def slsqp_optimize(h0, n, beta, maxiter=400):
    target = n / 2.0
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    try:

                      constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0.copy(), n)

    for trial_n in trial_ns:
        if remaining_time() < 35:
            break
        h_interp = interp_to(best_h, best_n, trial_n)
        h_interp = project_feasible(h_interp, trial_n)
        c5_interp = compute_c5(h_interp)
        
        if c5_interp < best_c5 + 0.003:
            h_new = slsqp_optimize(h_interp, trial_n, 500, maxiter=250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = trial_n
            if remaining_time() > 15 and c5_interp < best_c5 + 0.001:
                h_new = slsqp_optimize(best_h, best_n, 2000, maxiter=300)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            if remaining_time() > 10 and best_n == trial_n:
                h_new = lbfgsb_optimize(best_h, best_n, 3000, maxiter=2000, penalty=3000.0)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
    
    # Phase 8: If we found a better n, do pair descent on it
    if remaining_time() > 25:
        h_pair, c5_pair = pair_descent_fast(best_h.copy(), best_n, max_sweeps=25)
        if c5_pair < best_c5:
            best_c5 = c5_pair
            best_h = h_pair.copy()
    
    # Phase 9: Final ultra-high-precision polish
    for beta in [10000, 20000, 50000]:
        if remaining_time() < 6:
            break
        h_new = slsqp_optimize(best_h, best_n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 10: One final L-BFGS-B pass
    if remaining_time() > 8:
        h_new = lbfgsb_optimize(best_h, best_n, 10000, maxiter=5000, penalty=10000.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 11: Tiny pair descent at the end
    if remaining_time() > 10:
        h_pair, c5_pair = pair_descent_fast(best_h.copy(), best_n, max_sweeps=15)
        if c5_pair < best_c5:
            best_c5 = c5_pair
            best_h = h_pair.copy()
    
    # Final
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
