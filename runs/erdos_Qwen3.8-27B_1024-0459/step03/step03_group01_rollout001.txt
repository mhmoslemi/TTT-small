```python
                if add <= 0: break

                acc += add
```

```python
def dc_dx_all(c, x, n, W):
    # returns sum_t W[t] * dc/dx(t)
    res = np.zeros(n)
    for t in range(len(c)):
        wv = W[t]
        if wv <= 1e-9:

        if off >= 0:
            L = n - off

            res[off:] -= wv * x[:L]

            res[:L] += wv * (1.0 - x[ao:ao+L])
```

```python
def binarize(h, n, seed):
    rng = np.random.RandomState(seed)
    h = np.array(h, float)
    # round to 0/1 preferring values near .5 to set later
    base = (h >= 0.5).astype(float)
    need = 0.5*n - int(h.sum())
    # if base has wrong count, adjust by flipping the least-confident
    # conf = distance to 0.5
    conf = np.abs(h - 0.5)
    if need > 0:
        idx = np.where(base==0)[0]
        idx = idx[np.argsort(conf[idx])]  # least confident zeros first
        base[idx[:need]] = 1
    elif need < 0:
        idx = np.where(base==1)[0]
        idx = idx[np.argsort(conf[idx])]
        base[idx[: -need]] = 0
    if int(h.sum()) != int(base.sum()):
        # fallback exact
        ones = np.where(base==1)[0]
        if len(ones) > 0.5*n:
            rem = ones[rng.choice(len(ones), len(ones)-0.5*n, replace=False)]
            base[rem]=0
        else:
            zeros=np.where(base==0)[0]
            add=zeros[rng.choice(len(zeros), int(0.5*n)-len(ones), replace=False)]
            base[add]=1
    return base
```

```python
def logsumexp_descent(h, n, steps_total, betas, rng=None):
    hcur = project_feasible(h, n).copy()
    best = compute_c5(hcur)
    best_h = hcur.copy()
    for bet in betas:
        for s in range(steps_total // len(betas)):
            if time_left() < 8: return best_h, best
            c = compute_corr(hcur)
            m = c.max()

            W = e/e.sum()
            g = (2.0/n)*dc_dx_all(c, hcur, n, W)
            # preserve sum: project off mean
            g = g - np.mean(g)
            gn = np.linalg.norm(g)
            if gn < 1e-14: break
            g = g/gn
            # backtracking
            improved = False
            for lr in [0.02,0.01,0.005,0.05]:
                htrial = project_feasible(hcur - lr*g, n)

                hcur = project_feasible(hcur + (rng.rand(n)-0.5)*0.02, n)
    return best_h, best
```

```python
def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(40):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = (1.0 - h)
        else:
            room = h
        # distribute over all with room, proportional capacity
        tot = np.sum(room)
        if tot <= 1e-15:
            break
        frac = diff / tot
        # add frac*room but clip
        h = np.clip(h + (frac*room if diff>0 else -frac*room), 0.0, 1.0)
    # exact residual fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idxs = np.arange(n)
        # push throughout using largest room
        for idx in idxs:
            if diff > 0:
                add = min(1.0 - h[idx], diff)
                if add <= 0: continue
                h[idx] += add; diff -= add
                if abs(diff) < 1e-13: break
            else:
                rem = min(h[idx], -diff)
                if rem <= 0: continue
                h[idx] -= rem; diff += rem
                if abs(diff) < 1e-13: break
    return h
```

```python
def _dc_gather(c, x, n, W):
    res = np.zeros(n)
    # iterate lags; skip tiny weights
    for t in range(len(c)):
        wv = W[t]
        if wv <= 1e-8:
```

```python
def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i; c = 0
        while v:
            c ^= (v & 1); v >>= 1
        if c: h[i] = 1.0
```

```python
def fix_count(h, n, rng):
    h = (h >= 0.5).astype(float)
    want = int(n // 2)
    have = int(h.sum())
    if have != want:
        if have > want:
            idx = np.where(h == 1)[0]
            sel = idx[rng.choice(len(idx), have - want, replace=False)]
            h[sel] = 0.0
        else:
            idx = np.where(h == 0)[0]
            sel = idx[rng.choice(len(idx), want - have, replace=False)]
            h[sel] = 1.0
    return h
```

```python
def upscale(h, n):
    m = len(h)
    if m == n: return h.copy()
    xp = (np.arange(m) + 0.5) * (2.0/m)
    nv = (np.arange(n) + 0.5) * (2.0/n)
    return np.interp(nv, xp, h, left=float(h[0]), right=float(h[-1]))
```

```python
def tabu_binary_search(hinit, n, rng, iters, T0, Tcool, time_limit):
    n = int(n)
    h = project_feasible(hinit, n)
    h = (h >= 0.5).astype(float)
    h = fix_count(h, n, rng)
    corr = compute_corr(h)
    curmax = float(np.max(corr))
    best_c5 = curmax * (2.0/n)
    best_h = h.copy()
    tabu = np.zeros(n, dtype='int64')
    it = 0
    T = float(T0)
    stall = 0
    start = time.time()
    while it < iters and (time.time() - start) < time_limit and time_left() > 5:
        top4 = np.argsort(-corr)[:6]
        d0 = int(top4[0]) - (n - 1)
        e = abs(d0)
        if e == 0: e = 1
        accepted = None
        for _ in range(40):
            i, j = propose_move(h, d0, e, n, rng)
            if i is None: break
            if tabu[i] > (it - 120) or tabu[j] > (it - 120):
                continue
            h[i] = 0.0; h[j] = 1.0
            nc = compute_corr(h)
            nm = float(np.max(nc))
            if nm < curmax - 1e-9:
                corr = nc; curmax = nm
                tabu[i] = it; tabu[j] = it
                cv = curmax * (2.0/n)
                if cv < best_c5:
                    best_c5 = cv; best_h = h.copy()
                accepted = True; stall = 0
                break
            else:
                # SA aspiration / traverse
                if T > 0.05 and nm < best_c5*(n/2) + 0.5*rng.rand():  # allow near-best moves sometimes
                    if rng.rand() < math.exp(-(nm - curmax)/max(T,1e-6)):
                        corr = nc; curmax = nm
                        tabu[i]=it; tabu[j]=it
                        cv=curmax*(2.0/n)
                        if cv < best_c5:
                            best_c5 = cv; best_h = h.copy()
                        accepted=True; stall=0
                        break
                h[i] = 1.0; h[j] = 0.0
        if accepted is None:
            stall += 1
        it += 1
        T = max(1e-3, T * Tcool)
        if stall > 60:
            # 2-swap perturbation
            ones = np.where(h==1)[0]; zeros = np.where(h==0)[0]
            m1 = len(ones); m0 = len(zeros)
            if m1 >= 2 and m0 >= 2:
                a1 = ones[rng.randint(m1)]; b1 = ones[rng.randint(m1)]
                a0 = zeros[rng.randint(m0)]; b0 = zeros[rng.randint(m0)]
                if len({a1,b1,a0,b0})==4:
                    h[a1]=0;h[b1]=0;h[a0]=1;h[b0]=1
                    nc=compute_corr(h); nm=float(np.max(nc))

                        corr=nc;curmax=nm

                        h[a1]=1;h[b1]=1;h[a0]=0;h[b0]=0
```

```python
def propose_move(h, d0, e, n, rng):
    if e >= n: e = 1
    r = rng.rand()
    if r < 0.6:
        # pair-flip at distance e
        if d0 > 0:
            cand = np.where((h[:n-e]==1.0) & (h[e:]==0.0))[0]
            if len(cand):
                a = int(cand[rng.randint(len(cand))])
                return (a, a+e)
        else:
            cand = np.where((h[:n-e]==0.0) & (h[e:]==1.0))[0]
            if len(cand):
                a = int(cand[rng.randint(len(cand))])
                return (a+e, a)
    # random swap biased to locality
    ones = np.where(h==1.0)[0]; zeros = np.where(h==0.0)[0]
    if len(ones)==0 or len(zeros)==0: return (None,None)
    i = int(ones[rng.randint(len(ones))])
    if rng.rand() < 0.5:
        j = int(zeros[rng.randint(len(zeros))])
    else:
        off = rng.randint(-(n//4), n//4+1)
        j = (i+off) % n
        if j == i or h[j] == 1.0:
            j = int(zeros[rng.randint(len(zeros))])
    if j == i:
        j = int(zeros[rng.randint(len(zeros))])
        if j == i: return (None,None)
    return (i, j)
```

```python
def logsumexp_descent(h, n, steps, betas):
    n = int(n)
    hcur = project_feasible(h, n).copy()
    best = compute_c5(hcur); best_h = hcur.copy()
    rng = np.random.RandomState(1)
    per = max(1, steps // len(betas))
    for bet in betas:
        for s in range(per):
            if time_left() < 6: return best_h, float(best)
            c = compute_corr(hcur)
            m = float(np.max(c))
            e = np.exp(np.clip(bet*(c-m), -40, 40))
            W = e/e.sum()
            g = (2.0/n)*_dc_gather(c, hcur, n, W)
            g = g - g.mean()
            gn = np.linalg.norm(g)
            if gn < 1e-15: break
            g /= gn
            moved = False
            for lr in (0.03, 0.01, 0.005, 0.02):
                htrial = project_feasible(hcur - lr*g, n)
                cv = compute_c5(htrial)
                if cv < best - 5e-13:
                    hcur = htrial; best = cv; best_h = hcur.copy(); moved=True; break
            if not moved:
                hcur = project_feasible(hcur + (rng.rand(n)-0.5)*0.03, n)
    return best_h, float(best)
```

```python
def _dc_gather(c, x, n, W):
    res = np.zeros(n)
    for t in range(len(c)):
        wv = W[t]
        if wv <= 1e-8: continue
        off = t - (n-1)
        if off >= 0:
            L = n - off
            res[:L] += wv*(1.0 - x[off:off+L])
            res[off:] -= wv*x[:L]

            ao = -off
            L = n - ao
            res[ao:ao+L] += wv*(1.0 - x[:L])
            res[:L] -= wv*x[ao:ao+L]
    return res
```

```python
def thue_morse(n):
    h=np.zeros(n)
    for i in range(n):
        v=i; c=0
        while v:
            c ^= (v & 1); v >>= 1
        h[i] = 1.0 if c else 0.0
    return h
```

```python
        else:
            room = h
        tot = np.sum(room)
        if tot <= 1e-15:
            break
        frac = diff / tot
        if diff > 0:
            h = np.clip(h + frac * room, 0.0, 1.0)
        else:
            h = np.clip(h + frac * room, 0.0, 1.0)
    diff = target - np.sum(h)

                    h[idx] += add
                    diff -= add

                rem = min(h[idx], -diff)

                    diff += rem

def compute_corr(h):

def _dc_gather(c, x, n, W):
    res = np.zeros(n)
    for t in range(len(c)):
        wv = W[t]
        if wv <= 1e-8:
            continue
        off = t - (n - 1)
        if off >= 0:
            L = n - off
            res[:L] += wv * (1.0 - x[off:off + L])
            res[off:] -= wv * x[:L]
        else:
            ao = -off
            L = n - ao
            res[ao:ao + L] += wv * (1.0 - x[:L])
            res[:L] -= wv * x[ao:ao + L]
    return res

def thue_morse(n):
    h = np.zeros(int(n))
    for i in range(int(n)):
        v = i
        c = 0
        while v:
            c ^= (v & 1)

        h[i] = 1.0 if c else 0.0
    return h
```
