def subgradient_active_set(h, n, top_k=6):

def smooth_max_obj(x, N, TGT, beta):
    dx = 2.0 / N
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    sm = (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx
    penalty = 1000.0 * (np.sum(x) - TGT) ** 2
    return sm + penalty

def smooth_max_grad(x, N, TGT, beta):
    dx = 2.0 / N
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    weights /= np.sum(weights)
    
    grad = np.zeros(N)
    for j in range(2 * N - 1):
        wj = weights[j]
        if wj < 1e-14:
            continue
        s = j - (N - 1)
        if s >= 0:
            for m in range(s, N):
                grad[m] += wj * (1.0 - x[m - s])
            for m in range(0, N - s):
                grad[m] += wj * (-x[m + s])
        else:
            sp = -s
            for m in range(0, N - sp):
                grad[m] += wj * (1.0 - x[m + sp])
            for m in range(sp, N):
                grad[m] += wj * (-x[m - sp])
    
    grad *= dx
    grad += 2.0 * 1000.0 * (np.sum(x) - TGT)
    return grad

def make_costas_like(n):
    """Generate a pseudo-Costas sequence (low autocorrelation)."""
    h = np.zeros(n)
    # Use a simple LFSR-like construction
    state = 1
    for i in range(n):
        h[i] = 1.0 if (bin(state).count('1') % 2) == 0 else 0.0
        # LFSR step (x^16 + x^14 + x^13 + x^11 + 1)
        state = ((state << 1) ^ ((state >> 15) & 1)) & 0xFFFFFFFF
        state ^= ((state >> 1) & 1)  # simple tap
    return h

import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        unsat = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(unsat)
        if cnt > 0:
            h[unsat] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 2.0)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def subgrad_at(h, n, top_k):
    """Compute subgradient using top_k active constraints."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for j in top_idx:
        s = j - (n - 1)
        if s >= 0:
            # C = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
            # d/dh[m]: (1-h[m-s]) for m in [s, n-1], -h[m+s] for m in [0, n-1-s]
            ms = np.arange(s, n)
            grad[ms] += (1.0 - h[ms - s])
            ms2 = np.arange(0, n - s)
            grad[ms2] += (-h[ms2 + s])
        else:
            sp = -s
            ms = np.arange(0, n - sp)
            grad[ms] += (1.0 - h[ms + sp])
            ms2 = np.arange(sp, n)
            grad[ms2] += (-h[ms2 - sp])
    grad *= dx
    return grad, corr

def active_set_opt(h, n, top_k=8, max_iter=2000, lr0=0.03):

    c5 = best_c5

        if gnorm < 1e-14:
            break

        lr = lr0 / (1.0 + 0.005 * it)

                c5 = c5_try

_G_TGT = 0.0

def obj_fn(x):
    n = _G_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _G_BETA
    sm = (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx
    penalty = 500.0 * (np.sum(x) - _G_TGT) ** 2
    return sm + penalty

def obj_grad(x):
    n = _G_N
    dx = 2.0 / n
    beta = _G_BETA
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = weights[j]
        if wj < 1e-14:
            continue
        s = j - (n - 1)
        if s >= 0:
            ms = np.arange(s, n)
            grad[ms] += wj * (1.0 - x[ms - s])
            ms2 = np.arange(0, n - s)
            grad[ms2] += wj * (-x[ms2 + s])
        else:
            sp = -s
            ms = np.arange(0, n - sp)
            grad[ms] += wj * (1.0 - x[ms + sp])
            ms2 = np.arange(sp, n)
            grad[ms2] += wj * (-x[ms2 - sp])
    
    grad *= dx
    grad += 2.0 * 500.0 * (np.sum(x) - _G_TGT)
    return grad
