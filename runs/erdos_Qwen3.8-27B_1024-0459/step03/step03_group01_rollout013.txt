        g[offset:offset+npairs] += (1.0 - h[:npairs])

        g[:npairs] += (1.0 - h[ao:ao+npairs])

    return np.interp(x_new, xp, src_h)


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 320
        parent_h = np.zeros(parent_n)
        for i in range(parent_n):
            c = 0
            v = i
            while v:
                c ^= (v & 1)
                v >>= 1
            parent_h[i] = 1.0 if c == 1 else 0.0
        n_ones = int(np.sum(parent_h))
        if n_ones != parent_n // 2:
            parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Quick SLSQP refinement of parent
    n = best_n
    h = best_h.copy()
    for beta in [500, 2000, 8000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Active set equalization
    if remaining_time() > 10:
        h_eq, c5_eq = active_set_equalize(h, n, max_iters=200, k_top=3)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 2: Binary local search at parent n
    if remaining_time() > 20:
        h_bin = (h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        if n_ones != n // 2:
            if n_ones > n // 2:
                idx = np.where(h_bin == 1)[0]
                h_bin[idx[rng.choice(len(idx), n_ones - n // 2, replace=False)]] = 0
            else:
                idx = np.where(h_bin == 0)[0]
                h_bin[idx[rng.choice(len(idx), n // 2 - n_ones, replace=False)]] = 1
        h_b, c5_b = binary_local_search(h_bin, n, time_limit=min(40, remaining_time() - 15), rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n

        # Polish binary result with continuous refinement
        if remaining_time() > 15:
            h_polish = project_feasible(h_b.astype(float), n)
            for beta in [3000, 10000]:
                if remaining_time() < 8:
                    break
                h_pol = slsqp_smooth(h_polish, n, beta, maxiter=200)
                c5_pol = compute_c5(h_pol)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()
            if remaining_time() > 8:
                h_pol, c5_pol = active_set_equalize(best_h, n, max_iters=150, k_top=4)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()

    # Phase 3: Try nearby n values
    n_candidates = []
    base_n = best_n
    for delta in [-4, -2, 2, 4, 6, 8, 10, 12, 16, 20, 24, 28, 32]:
        nn = base_n + delta
        if 200 <= nn <= 800 and nn % 2 == 0:
            n_candidates.append(nn)
    # Also try scaled versions
    for scale in [1.25, 1.5, 2.0]:
        nn = int(base_n * scale)
        if nn % 2 != 0:
            nn += 1
        if 200 <= nn <= 1000:
            n_candidates.append(nn)
    n_candidates = sorted(set(n_candidates))

    for n_t in n_candidates:
        if remaining_time() < 20:
            break
        if n_t == best_n:
            continue

        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_interp = compute_c5(h_feas)

        # SLSQP refinement
        h_opt = h_feas
        for beta in [1000, 5000]:
            if remaining_time() < 10:
                break
            h_opt = slsqp_smooth(h_opt, n_t, beta, maxiter=250)
        c5_opt = compute_c5(h_opt)

        # Active set
        if remaining_time() > 8:
            h_opt, c5_opt = active_set_equalize(h_opt, n_t, max_iters=150, k_top=3)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

        # Binary search on this n
        if remaining_time() > 25:
            h_bin_t = (h_opt > 0.5).astype(float)
            n_ones_t = int(np.sum(h_bin_t))
            if n_ones_t != n_t // 2:
                if n_ones_t > n_t // 2:
                    idx = np.where(h_bin_t == 1)[0]
                    h_bin_t[idx[rng.choice(len(idx), n_ones_t - n_t // 2, replace=False)]] = 0
                else:
                    idx = np.where(h_bin_t == 0)[0]
                    h_bin_t[idx[rng.choice(len(idx), n_t // 2 - n_ones_t, replace=False)]] = 1
            h_b_t, c5_b_t = binary_local_search(h_bin_t, n_t, time_limit=min(25, remaining_time() - 20), rng=rng)
            if c5_b_t < best_c5:
                best_c5 = c5_b_t
                best_h = h_b_t.copy()
                best_n = n_t

            # Polish
            if remaining_time() > 12:
                h_pol = project_feasible(h_b_t.astype(float), n_t)
                for beta in [5000, 15000]:
                    if remaining_time() < 6:
                        break
                    h_pol = slsqp_smooth(h_pol, n_t, beta, maxiter=200)
                    c5_pol = compute_c5(h_pol)
                    if c5_pol < best_c5:
                        best_c5 = c5_pol
                        best_h = h_pol.copy()
                        best_n = n_t

    # Phase 4: Intensive final refinement on best solution
    n = best_n
    h = best_h.copy()

    # Multiple SLSQP passes with increasing beta
    for beta in [3000, 8000, 15000, 30000]:
        if remaining_time() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Active set with more top-k
    if remaining_time() > 10:
        h_eq, c5_eq = active_set_equalize(h, n, max_iters=400, k_top=5)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Final binary search
    if remaining_time() > 20:
        h_bin_final = (h > 0.5).astype(float)
        n_ones_f = int(np.sum(h_bin_final))
        if n_ones_f != n // 2:
            if n_ones_f > n // 2:
                idx = np.where(h_bin_final == 1)[0]
                h_bin_final[idx[rng.choice(len(idx), n_ones_f - n // 2, replace=False)]] = 0
            else:
                idx = np.where(h_bin_final == 0)[0]
                h_bin_final[idx[rng.choice(len(idx), n // 2 - n_ones_f, replace=False)]] = 1
        h_b_f, c5_b_f = binary_local_search(h_bin_final, n, time_limit=min(30, remaining_time() - 10), rng=rng)
        if c5_b_f < best_c5:
            best_c5 = c5_b_f
            best_h = h_b_f.copy()
            best_n = n

        # Polish binary result
        if remaining_time() > 12:
            h_pol = project_feasible(h_b_f.astype(float), n)
            for beta in [8000, 20000, 50000]:
                if remaining_time() < 6:
                    break
                h_pol = slsqp_smooth(h_pol, n, beta, maxiter=300)
                c5_pol = compute_c5(h_pol)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()
            if remaining_time() > 6:
                h_pol, c5_pol = active_set_equalize(best_h, n, max_iters=200, k_top=5)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()

    # L-BFGS-B final attempt
    if remaining_time() > 15:
        n = best_n
        for beta in [10000, 30000]:
            if remaining_time() < 8:
                break
            h_lb = lbfgs_smooth(best_h, n, beta, maxiter=500)
            c5_lb = compute_c5(h_lb)
            if c5_lb < best_c5:
                best_c5 = c5_lb
                best_h = h_lb.copy()

    # Final validation and projection
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
