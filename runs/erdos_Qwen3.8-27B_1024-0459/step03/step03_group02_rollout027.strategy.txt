def gradient_from_correlation(h, n):
    dx = 2.0/n
    corr = np.correlate(h,1.0-h,mode='full')
    max_corr = np.max(corr)
    # Differentiate wrt coefficient with max lag
    lag_idx = np.argmax(corr) - (n-1)  # account for correlation mode
    if lag_idx < n-1:
        weights = [1.0] + [0]*(n-1+lag_idx)  # pad for edges

    return dx * sum( [weights[i]* ((1-h[i+k])-h[i]) for i,k in enumerate(lag_idx)] ) / max_corr
