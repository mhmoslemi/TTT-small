def compute_deltas_fast(h, n):
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    n_ones = len(ones)
    n_zeros = len(zeros)

    # = (h[i+d_off] - 1) * [0<=i+d_off<n] + h[i-d_off] * [0<=i-d_off<n]
    # 
    # Vectorized: for all ones positions at once
    # i is shape (n_ones,), d_off is shape (n_corr,)
    # i + d_off has shape (n_ones, n_corr)
    i_arr = ones[:, None].astype(float)  # (n_ones, 1)
    d_arr = (np.arange(n_corr) - (n-1))[None, :].astype(float)  # (1, n_corr)
    
    # i + d: (n_ones, n_corr)
    ip = i_arr + d_arr
    im = i_arr - d_arr
    
    # Validity masks
    ip_valid = (ip >= 0) & (ip < n)
    im_valid = (im >= 0) & (im < n)
    
    # Get h values (with padding)
    ip_idx = np.clip(ip.astype(int), 0, n-1)
    im_idx = np.clip(im.astype(int), 0, n-1)
    
    ones_delta = (h[ip_idx] - 1.0) * ip_valid + h[im_idx] * im_valid
    
    # For zeros (flipping 0→1):
    j_arr = zeros[:, None].astype(float)
    jp = j_arr + d_arr
    jm = j_arr - d_arr
    jp_valid = (jp >= 0) & (jp < n)
    jm_valid = (jm >= 0) & (jm < n)
    jp_idx = np.clip(jp.astype(int), 0, n-1)
    jm_idx = np.clip(jm.astype(int), 0, n-1)

    h_bin = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[idx])
        h_bin[idx[order[:n_ones - target]]] = 0.0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[idx])
        h_bin[idx[order[:target - n_ones]]] = 1.0
    return h_bin


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 5.0)


def compute_deltas_fast(h, n):
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    n_ones = len(ones)
    n_zeros = len(zeros)
    n_corr = 2 * n - 1

    i_arr = ones[:, None].astype(float)
    d_arr = (np.arange(n_corr) - (n - 1))[None, :].astype(float)

    ip = i_arr + d_arr
    im = i_arr - d_arr
    ip_valid = (ip >= 0) & (ip < n)
    im_valid = (im >= 0) & (im < n)
    ip_idx = np.clip(ip.astype(int), 0, n - 1)
    im_idx = np.clip(im.astype(int), 0, n - 1)

    ones_delta = (h[ip_idx] - 1.0) * ip_valid + h[im_idx] * im_valid

    j_arr = zeros[:, None].astype(float)
    jp = j_arr + d_arr
    jm = j_arr - d_arr
    jp_valid = (jp >= 0) & (jp < n)
    jm_valid = (jm >= 0) & (jm < n)
    jp_idx = np.clip(jp.astype(int), 0, n - 1)
    jm_idx = np.clip(jm.astype(int), 0, n - 1)

    zeros_delta = (1.0 - h[jp_idx]) * jp_valid - h[jm_idx] * jm_valid

    return ones, zeros, ones_delta, zeros_delta


def find_best_pair(corr, ones, zeros, ones_delta, zeros_delta, cur_max):
    n_ones = len(ones)
    n_zeros = len(zeros)

    best_new_max = cur_max + 1.0
    best_i = -1
    best_j = -1

    batch_size = min(n_ones, 40)
    for batch_start in range(0, n_ones, batch_size):
        batch_end = min(batch_start + batch_size, n_ones)
        combined = corr[None, None, :] + ones_delta[batch_start:batch_end, None, :] + zeros_delta[None, :, :]
        new_maxes = np.max(combined, axis=2)
        flat_min = float(np.min(new_maxes))
        if flat_min < best_new_max - 1e-14:
            idx = np.unravel_index(np.argmin(new_maxes), new_maxes.shape)
            best_new_max = flat_min
            best_i = batch_start + idx[0]
            best_j = idx[1]

    return best_i, best_j, best_new_max


def efficient_binary_ls(h, n, max_iters=500, time_limit=30, rng=None, verbose=False):
    if rng is None:
        rng = np.random.RandomState(42)
    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx_val
    best_h = h.copy()
    start = time.time()
    stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining() < 8:
            break

        ones, zeros, ones_delta, zeros_delta = compute_deltas_fast(h, n)
        n_ones = len(ones)
        n_zeros = len(zeros)

        if n_ones == 0 or n_zeros == 0:
            break

        best_i, best_j, best_new_max = find_best_pair(corr, ones, zeros, ones_delta, zeros_delta, cur_max)

        if best_i >= 0:
            i_pos = int(ones[best_i])
            j_pos = int(zeros[best_j])
            h[i_pos] = 0.0
            h[j_pos] = 1.0
            corr = compute_corr(h)
            cur_max = float(np.max(corr))
            if cur_max * dx_val < best_c5:
                best_c5 = cur_max * dx_val
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            T = max(0.05, 0.4 * (0.93 ** stall))

            if stall > 50:
                break

            n_pert = 1 + (rng.rand() < min(0.6, 0.1 + stall * 0.03))
            for _ in range(n_pert):
                i = ones[rng.randint(n_ones)]
                j = zeros[rng.randint(n_zeros)]
                h[i] = 0.0
                h[j] = 1.0
                ones, zeros = np.where(h == 1.0)[0], np.where(h == 0.0)[0]
                n_ones, n_zeros = len(ones), len(zeros)
                if n_ones == 0 or n_zeros == 0:
                    h = make_binary(h, n, rng)
                    ones, zeros = np.where(h == 1.0)[0], np.where(h == 0.0)[0]
                    n_ones, n_zeros = len(ones), len(zeros)

            if n_pert >= 2:
                h = make_binary(h, n, rng)

            corr = compute_corr(h)
            cur_max = float(np.max(corr))

            delta = (cur_max - best_c5) / dx_val
            if delta < 0 or rng.rand() < np.exp(-delta / T):
                pass
            else:
                h = best_h.copy()
                corr = compute_corr(h)
                cur_max = float(np.max(corr))

    return best_h, best_c5


def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-12:
            g += w[k] * corr_grad_per_shift(x, n, k)
    return g * dx_val


def sum_con(x):
    return np.sum(x) - _TGT


def sum_con_jac(x):
    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=300):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def thue_morse_init(n, rng):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1')
        h[i] = 1.0 if c % 2 == 1 else 0.0
    h = make_binary(h, n, rng)
    return h


def random_init(n, rng):
    idx = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 320
        parent_h = random_init(parent_n, rng)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Intensive binary LS on parent resolution
    n = best_n
    h_bin = make_binary(best_h, n, rng)
    h_bin, c5_bin = efficient_binary_ls(h_bin, n, max_iters=300, time_limit=80, rng=rng)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 2: Explore nearby resolutions
    n_tries = sorted(set([
        n - 6, n - 4, n - 2, n + 2, n + 4, n + 6,
        n + 8, n + 10, n + 12, n + 14, n + 16, n + 20, n + 24
    ]))
    n_tries = [nn for nn in n_tries if 80 <= nn <= 500 and nn % 2 == 0 and nn != n]

    for n_t in n_tries:
        if remaining() < 15:
            break
        xp = (np.arange(best_n) + 0.5) * 2.0 / best_n
        x_new = (np.arange(n_t) + 0.5) * 2.0 / n_t
        h_interp = np.interp(x_new, xp, best_h)
        h_bin_t = make_binary(h_interp, n_t, rng)

        h_t, c5_t = efficient_binary_ls(h_bin_t, n_t, max_iters=120, time_limit=25, rng=rng)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t

    # Phase 3: Alternative initializations (Thue-Morse, random) on best_n
    n = best_n
    for init_seed in [seed + 1, seed + 100, 7, 123]:
        if remaining() < 20:
            break
        h_alt = thue_morse_init(n, np.random.RandomState(init_seed))
        h_alt, c5_alt = efficient_binary_ls(h_alt, n, max_iters=100, time_limit=20, rng=rng)
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt.copy()

    if remaining() > 20:
        h_rand = random_init(n, np.random.RandomState(seed + 999))
        h_rand, c5_rand = efficient_binary_ls(h_rand, n, max_iters=100, time_limit=20, rng=rng)
        if c5_rand < best_c5:
            best_c5 = c5_rand
            best_h = h_rand.copy()

    # Phase 4: Deep binary LS on best solution with more iterations
    n = best_n
    h = best_h.copy()
    h, c5 = efficient_binary_ls(h, n, max_iters=500, time_limit=60, rng=rng)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 5: Continuous SLSQP refinement
    if remaining() > 30:
        n = best_n
        h_cont = best_h.copy()
        for beta in [100, 300, 800, 2000, 5000]:
            if remaining() < 12:
                break
            h_new = slsqp_smooth(h_cont, n, beta, maxiter=250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cont = h_new.copy()

    # Phase 6: Binary LS on SLSQP result
    if remaining() > 15:
        n = best_n
        h_bin2 = make_binary(best_h, n, rng)
        h_bin2, c5_bin2 = efficient_binary_ls(h_bin2, n, max_iters=200, time_limit=int(min(35, remaining() - 8)), rng=rng)
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()

    # Phase 7: Final SLSQP polish
    if remaining() > 12:
        n = best_n
        for beta in [3000, 8000, 15000]:
            if remaining() < 6:
                break
            h_new = slsqp_smooth(best_h, n, beta, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
