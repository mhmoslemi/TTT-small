def _c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / n)


def _proj(h, n):
    h = np.array(h, dtype=np.float64)
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(60):
        d = 0.5 * n - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        k = int(np.sum(m))
        if k > 0:
            h[m] += d / k
        else:
            h[:] = 0.5
        np.clip(h, 0.0, 1.0, out=h)
    return h


def _ggrad_core(h, k):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        p = -s
        g[:n - p] += 1.0 - h[p:]
        g[p:] -= h[:n - p]
    return g


def _sobj(x):
    n = _ng
    c = np.correlate(x, 1.0 - x, mode='full')

        c2 = _c5(h2)

            break

        idx = np.argsort(corr)[::-1][:topK]
        g = np.zeros(n)

            best2, bc2 = _line_search(best, g2, n, lrs)
