        m = n - ao
        if m <= 0: return grad
        grad[:m] += (1.0 - h[ao:ao+m])
        grad[ao:ao+m] -= h[:m]
    return grad

def smooth_max_obj(x, n, beta, active=None):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if active is not None:
        cs = corr[active]
    else:
        cs = corr
    m = np.max(cs)
    sh = np.minimum(beta * (cs - m), 0.0)
    return (m + np.log(np.sum(np.exp(sh))) / beta) * dx

def smooth_max_grad(x, n, beta, active=None):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if active is not None:
        cs = corr[active]
        indices = active
    else:
        cs = corr
        indices = np.arange(len(corr))
    m = np.max(cs)
    sh = np.minimum(beta * (cs - m), 0.0)
    exp_sh = np.exp(sh)
    weights = exp_sh / np.sum(exp_sh)
    grad = np.zeros(n)
    for i, k_idx in enumerate(indices):
        grad += weights[i] * dx * lag_grad(x, n, k_idx)
    return grad

def slsqp(h, n, beta, maxiter, active=None):
    def obj(x):
        return smooth_max_obj(x, n, beta, active)
    def grad(x):
        return smooth_max_grad(x, n, beta, active)
    def cfun(x):
        return np.sum(x) - n / 2.0
    def cjac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def get_active(h, top_k=5, thresh_frac=0.01):
    corr = compute_corr(h)
    mx = np.max(corr)
    active = np.where(corr >= mx - thresh_frac * (mx - np.min(corr) + 1e-15))[0]
    top = np.argsort(corr)[::-1][:top_k]
    return np.union1d(active, top)

def interp_to(src, sn, dn):
    if sn == dn: return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)
    
    best_h, best_c5, best_n = None, float('inf'), 0
    
    try:
        parent = np.array(initial_h_values, dtype=float)
        best_n = len(parent)
        best_h = project_feasible(parent, best_n)
        best_c5 = compute_c5(best_h)
    except:
        best_n = 160
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    
    # Phase 1: SLSQP at parent resolution
    n = best_n
    h = best_h.copy()
    for beta in [100, 300, 1000, 3000, 10000]:
        if remaining_time() < 10: break
        act = get_active(h, 5, 0.01)
        h2 = slsqp(h, n, beta, 400, act)
        c5 = compute_c5(h2)
        if c5 < best_c5:
            best_c5, best_h, h = c5, h2.copy(), h2.copy()
    
    # Phase 2: Subgradient with adaptive step
    h = best_h.copy()
    n = best_n
    for it in range(3000):
        if remaining_time() < 5: break
        corr = compute_corr(h)
        mx = np.max(corr)
        active = np.where(corr >= mx - 0.05)[0]
        grad = np.zeros(n)
        for k in active:
            grad += lag_grad(h, n, k)
        gn = np.linalg.norm(grad)
        if gn < 1e-16: break
        grad /= gn
        # Try multiple step sizes
        improved = False
        for s in [0.05, 0.02, 0.01, 0.005, 0.001, 0.03]:
            h2 = project_feasible(h - s * grad, n)
            c5 = compute_c5(h2)
            if c5 < best_c5:
                best_c5, best_h, h = c5, h2.copy(), h2.copy()
                improved = True
                break
        if not improved and it % 200 == 100 and remaining_time() > 30:
            act = get_active(h, 5, 0.01)
            h2 = slsqp(h, n, 1000, 300, act)
            c5 = compute_c5(h2)
            if c5 < best_c5:
                best_c5, best_h, h = c5, h2.copy(), h2.copy()
    
    # Phase 3: Multi-resolution
    for tn in [best_n-8, best_n-4, best_n+4, best_n+8, best_n+16, 200, 256]:
        if tn <= 0 or remaining_time() < 20: continue
        hi = project_feasible(interp_to(best_h, best_n, tn), tn)
        ci = compute_c5(hi)
        if ci < best_c5 + 0.003:
            for beta in [200, 1000, 5000]:
                if remaining_time() < 8: break
                act = get_active(hi, 5, 0.01)
                h2 = slsqp(hi, tn, beta, 300, act)
                c5 = compute_c5(h2)
                if c5 < best_c5:
                    best_c5, best_h, best_n = c5, h2.copy(), tn
                    hi = h2.copy()
    
    # Phase 4: Random restarts
    for r in range(15):
        if remaining_time() < 20: break
        noise = rng.normal(0, 0.015, best_n)
        hp = project_feasible(best_h + noise, best_n)
        cp = compute_c5(hp)
        if cp < best_c5 + 0.002:
            act = get_active(hp, 5, 0.01)
            h2 = slsqp(hp, best_n, 1000, 300, act)
            c5 = compute_c5(h2)
            if c5 < best_c5:
                best_c5, best_h = c5, h2.copy()
    
    # Phase 5: Final polish
    h = best_h.copy()
    n = best_n
    for beta in [5000, 10000, 20000, 50000]:
        if remaining_time() < 4: break
        act = get_active(h, 3, 0.005)
        h2 = slsqp(h, n, beta, 500, act)
        c5 = compute_c5(h2)
        if c5 < best_c5:
            best_c5, best_h, h = c5, h2.copy(), h2.copy()
    
    # Phase 6: Single-lag micro-optimization
    h = best_h.copy()
    n = best_n
    for it in range(1000):
        if remaining_time() < 2: break
        corr = compute_corr(h)
        tl = int(np.argmax(corr))
        g = lag_grad(h, n, tl)
        gn = np.linalg.norm(g)
        if gn < 1e-16: break
        g /= gn
        improved = False
        for s in [0.001, 0.003, 0.0005, 0.002, 0.0001, 0.005]:
            h2 = project_feasible(h - s * g, n)
            c5 = compute_c5(h2)
            if c5 < best_c5:
                best_c5, best_h, h = c5, h2.copy(), h2.copy()
