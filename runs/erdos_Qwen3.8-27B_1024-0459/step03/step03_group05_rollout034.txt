def gradient_active(h, corr, top_k, temperature, dx):
    """Compute weighted gradient over top-k correlation values."""
    n = len(h)
    # Find top-k indices
    if top_k >= len(corr):

    vals = corr[idx]
    shifted = vals - vals.max()
    w = np.exp(shifted / temperature)

    h = np.asarray(h, dtype=float).copy()

    n = len(h)

        grad[j2] += (-h[j2 - L])
    return grad * dx

    n = len(h)

    else:

    if w_sum < 1e-30:

```python
def project_to_box_sum(h_raw, target_sum):
    """Project h_raw onto {h: 0≤h≤1, sum(h)=target_sum} using bisection on offset."""
    n = len(h_raw)
    # Find λ such that sum(clip(h_raw + λ, 0, 1)) = target_sum
    # f(λ) = sum(clip(h_raw + λ, 0, 1)) is monotonically increasing in λ
    # f(-1) = 0, f(1) = n, so a solution exists if 0 ≤ target_sum ≤ n
    
    lo = -1.0
    hi = 1.0
    # Ensure f(lo) ≤ target ≤ f(hi)
    for _ in range(60):
        mid = (lo + hi) / 2
        val = np.sum(np.clip(h_raw + mid, 0.0, 1.0))
        if np.isclose(val, target_sum, atol=1e-14):
            break
        if val < target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h_raw + (lo + hi) / 2, 0.0, 1.0)
    # Final adjustment for floating point
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-15:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            result[mask] += diff / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)
    return result
```

import numpy as np
from scipy.optimize import minimize
import time


def project_to_box_sum(h_raw, target_sum):
    n = len(h_raw)
    lo = -1.5
    hi = 1.5
    for _ in range(64):
        mid = (lo + hi) / 2.0
        val = np.sum(np.clip(h_raw + mid, 0.0, 1.0))
        if abs(val - target_sum) < 1e-14:
            break
        if val < target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h_raw + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-14:
        mask = (result > 1e-13) & (result < 1.0 - 1e-13)
        cnt = np.sum(mask)
        if cnt > 0:
            result[mask] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def all_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')


def grad_at_lag_idx(h, m_idx, dx):
    n = len(h)
    grad = np.zeros(n)
    k = m_idx - (n - 1)
    if k >= 0:
        j1 = np.arange(k, n)
        grad[j1] += (1.0 - h[j1 - k])
        j2 = np.arange(0, n - k)
        grad[j2] += (-h[j2 + k])
    else:
        L = -k
        j1 = np.arange(0, n - L)
        grad[j1] += (1.0 - h[j1 + L])
        j2 = np.arange(L, n)
        grad[j2] += (-h[j2 - L])
    return grad * dx


def weighted_gradient(h, corr, top_k, temperature, dx):
    n = len(h)
    if top_k >= len(corr):
        idx = np.arange(len(corr))
    else:
        idx = np.argsort(corr)[-top_k:]
    vals = corr[idx]
    vshift = vals - vals.max()
    w = np.exp(vshift / temperature)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(idx))
        w[-1] = 1.0
    else:
        w /= wsum
    grad = np.zeros(n)
    for t in range(len(idx)):
        if w[t] < 1e-14:
            continue
        grad += w[t] * grad_at_lag_idx(h, int(idx[t]), dx)
    return grad


def smoothmax_value(h, beta, dx):
    corr = all_correlations(h)
    m = corr.max()
    shifted = corr - m
    return (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx


def smoothmax_grad(h, beta, dx):
    corr = all_correlations(h)
    m = corr.max()
    shifted = corr - m
    w = np.exp(beta * shifted)
    w /= w.sum()
    n = len(h)
    grad = np.zeros(n)
    for t in range(len(corr)):
        if w[t] < 1e-14:
            continue
        grad += w[t] * grad_at_lag_idx(h, t, dx)
    return grad


def lbfgs_obj_wrapper(x, beta, dx):
    h = np.clip(x, 0.0, 1.0)
    return smoothmax_value(h, beta, dx)
