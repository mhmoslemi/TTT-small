    # Use scipy SLSQP

    def f(lam):
        g = d @ lam
        return 0.5 * g @ g
    def g(lam):
        return d @ (d @ lam)
    cons = [{'type': 'eq', 'fun': lambda l: l.sum()-1}]
    res = sp_min(f, np.ones(k)/k, jac=g, method='SLSQP',
                 bounds=[(0,1)]*k, constraints=cons, options={'maxiter':200})
    lam = np.clip(res.x, 0, 1)
    lam /= lam.sum()
    return d @ lam
