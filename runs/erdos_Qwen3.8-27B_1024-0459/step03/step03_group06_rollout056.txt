def binary_sa(h_init, n, iterations, rng, t_start=0.003, t_end=0.00003):
    h = np.where(h_init > 0.5, 1.0, 0.0)
    target_ones = n // 2
    num_ones = int(np.sum(h))
    if num_ones > target_ones:
        ones_idx = np.where(h == 1.0)[0]
        remove = rng.choice(len(ones_idx), num_ones - target_ones, replace=False)
        h[ones_idx[remove]] = 0.0
    elif num_ones < target_ones:
        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
        h[zeros_idx[add]] = 1.0

    best_c5 = compute_c5(h)
    current_c5 = best_c5
    
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    
    for it in range(iterations):
        if remaining_time() < 3:
            break
        if not ones_list or not zeros_list:
            break
        
        i = rng.randint(len(ones_list))
        j = rng.randint(len(zeros_list))
        oi = ones_list[i]
        oz = zeros_list[j]
        
        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = compute_c5(h)
        
        T = t_start * (t_end / t_start) ** (float(it) / max(float(iterations), 1.))
        
        if new_c5 < current_c5 or (T > 1e-15 and rng.rand() < np.exp(-(new_c5 - current_c5) / min(T, 0.01))):
            current_c5 = new_c5
            ones_list[i] = oz
            zeros_list[j] = oi
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0
    
    return best_h, best_c5

        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = compute_c5(h)
        T = t_start * (t_end / t_start) ** (float(it) / max(float(iterations), 1.))
        if new_c5 < cur_c5 or (T > 1e-16 and rng.rand() < np.exp(-(new_c5 - cur_c5) / min(T, 0.005))):
            cur_c5 = new_c5
            ones_list[i] = oz
            zeros_list[j] = oi
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0
    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def continuous_optimize(h0, n, max_time=30):
    """Run L-BFGS-B with adaptive beta, return best."""
    _N = n
    global _BETA
    bounds = [(0.0, 1.0)] * n
    h_best = h0.copy()
    c5_best = compute_c5(h_best)

    # Anneal beta from low to high
    for beta in [30, 80, 200, 500, 1200, 3000]:
        if remaining_time() < 5:
            break
        _BETA = beta
        try:
            res = minimize(smooth_max_obj, h_best, jac=smooth_max_grad,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 800, 'ftol': 1e-16,
                                    'gtol': 1e-11, 'iprint': 0,
                                    'maxls': 100, 'maxfun': 15000})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                h_best = h_new
                c5_best = c5_new
        except Exception:
            pass
    return h_best, c5_best


def topk_optimize(h0, n, top_k=10):
    """Run top-k focused L-BFGS-B."""
    _N = n
    global _TOP_K
    _TOP_K = top_k
    bounds = [(0.0, 1.0)] * n
    h_best = h0.copy()
    c5_best = compute_c5(h_best)
    try:
        res = minimize(topk_obj, h_best, jac=topk_grad,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 600, 'ftol': 1e-16,
                                'iprint': 0, 'maxls': 80})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    except Exception:
        pass
    return h_best, c5_best


def pairwise_local_search(h, n, max_sweeps=50):
    """Try pairwise exchanges to improve."""
    c5_cur = compute_c5(h)
    improved_total = False
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        order = np.random.permutation(n)
        for ii in order:
            if remaining_time() < 2:
                break
            for jj in order:
                if jj <= ii:
                    continue
                old_i, old_j = h[ii], h[jj]
                if abs(old_i - 0.5) < 1e-6 and abs(old_j - 0.5) < 1e-6:
                    continue
                for delta in [0.05, -0.05, 0.1, -0.1, 0.2, -0.2]:
                    ni = old_i + delta
                    nj = old_j - delta
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[ii], h[jj] = ni, nj
                    c5_test = compute_c5(h)
                    if c5_test < c5_cur - 1e-12:
                        c5_cur = c5_test
                        improved = True
                        improved_total = True
                        break
                else:
                    h[ii], h[jj] = old_i, old_j
                if h[ii] != old_i or h[jj] != old_j:
                    if c5_cur < compute_c5(h):
                        h[ii], h[jj] = old_i, old_j
        if not improved:
            break
    return h, c5_cur, improved_total


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S, _N, _BETA, _TOP_K
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 160
        best_h = project_feasible(make_thue_morse(best_n), best_n)
        best_c5 = compute_c5(best_h)

    print("Init: n={}, C5={:.7f}".format(best_n, best_c5))

    # Phase 1: Multi-resolution initial optimization
    n_candidates = sorted(set([best_n, 160, 200, 240]))
    n_candidates = [nn for nn in n_candidates if 120 <= nn <= 280]

    all_results = []

    for n in n_candidates:
        if remaining_time() < 60:
            break
        _N = n
        print("\n=== Phase1 n={} (time left: {:.0f}s) ===".format(n, remaining_time()))

        # Build initializations
        inits = []
        if best_h is not None:
            h_interp = interp_to(best_h, best_n, n)
            inits.append(project_feasible(h_interp, n))
        inits.append(project_feasible(make_thue_morse(n), n))
        inits.append(project_feasible(make_golden(n), n))
        inits.append(project_feasible(make_vander(n), n))

        # A few random binary inits
        for _ in range(3):
            h_bin = np.zeros(n)
            idx = rng.permutation(n)
            h_bin[idx[:n // 2]] = 1.0
            inits.append(h_bin.copy())

        for h0 in inits:
            if remaining_time() < 15:
                break
            h0 = project_feasible(h0, n)
            c5_0 = compute_c5(h0)
            if c5_0 > best_c5 * 1.2:
                continue

            # Continuous optimization with annealing
            h_opt, c5_opt = continuous_optimize(h0, n)
            if c5_opt < c5_0:
                h0 = h_opt
                c5_0 = c5_opt

            # Top-k refinement
            for tk in [5, 10, 15]:
                if remaining_time() < 8:
                    break
                h_tk, c5_tk = topk_optimize(h0, n, tk)
                if c5_tk < c5_0:
                    h0 = h_tk
                    c5_0 = c5_tk

            # Binary threshold check
            h_bin = np.where(h0 > 0.5, 1.0, 0.0)
            h_bin = project_feasible(h_bin, n)
            c5_bin = compute_c5(h_bin)
            if c5_bin < c5_0:
                h0 = h_bin
                c5_0 = c5_bin

            all_results.append((c5_0, h0.copy(), n))
            if c5_0 < best_c5:
                best_c5 = c5_0
                best_h = h0.copy()
                best_n = n
                print("  Phase1 improved: n={}, C5={:.7f}".format(n, c5_0))

    # Sort results and keep top candidates
    all_results.sort(key=lambda x: x[0])
    print("\nTop candidates: " + ", ".join(["C5={:.6f}@n={}".format(r[0], r[2]) for r in all_results[:5]]))

    # Phase 2: Focused intensive optimization on best candidates
    n_best = best_n
    _N = n_best
    print("\n=== Phase2 Intensive (n={}) ===".format(n_best))

    # Work with top 3 candidates at best_n
    top_cands = [r[1] for r in all_results if r[2] == n_best][:3]
    if not top_cands:
        top_cands = [best_h.copy()]

    for ci, h_cand in enumerate(top_cands):
        if remaining_time() < 40:
            break
        h_work = project_feasible(h_cand, n_best)
        c5_work = compute_c5(h_work)
        print("  Candidate {}: C5={:.7f}".format(ci, c5_work))

        # Extended continuous optimization with very high beta
        _N = n_best
        bounds = [(0.0, 1.0)] * n_best
        for beta in [300, 800, 2000, 5000, 10000]:
            if remaining_time() < 8:
                break
            _BETA = beta
            try:
                res = minimize(smooth_max_obj, h_work, jac=smooth_max_grad,
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 1500, 'ftol': 1e-16,
                                        'gtol': 1e-12, 'iprint': 0,
                                        'maxls': 200, 'maxfun': 30000})
                h_new = project_feasible(res.x, n_best)
                c5_new = compute_c5(h_new)
                if c5_new < c5_work:
                    h_work = h_new
                    c5_work = c5_new
                    print("    beta={}: C5={:.7f}".format(beta, c5_work))
            except Exception:
                pass

        # Top-k with very high effective beta
        for tk in [3, 5, 8, 12]:
            if remaining_time() < 6:
                break
            h_tk, c5_tk = topk_optimize(h_work, n_best, tk)
            if c5_tk < c5_work:
                h_work = h_tk
                c5_work = c5_tk

        # Binary SA from current
        if remaining_time() > 25:
            h_bin, c5_bin = binary_sa_fast(h_work, n_best, 8000, rng,
                                           t_start=0.001, t_end=0.000005)
            if c5_bin < c5_work:
                h_work = h_bin
                c5_work = c5_bin
                print("    SA: C5={:.7f}".format(c5_work))
            # Polish binary
            if remaining_time() > 10:
                h_pol, c5_pol = continuous_optimize(h_work, n_best)
                if c5_pol < c5_work:
                    h_work = h_pol
                    c5_work = c5_pol

        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()
            best_n = n_best

    # Phase 3: Pairwise local search
    if remaining_time() > 20:
        print("\n=== Phase3 Pairwise search ===")
        _N = best_n
        h_ls, c5_ls, imp = pairwise_local_search(best_h.copy(), best_n, max_sweeps=80)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls
            print("  Pairwise: C5={:.7f}".format(best_c5))

    # Phase 4: Another round of SA + polish at finer schedule
    if remaining_time() > 30:
        print("\n=== Phase4 Final SA+polish ===")
        _N = best_n
        # Multiple short SA runs
        for sa_round in range(4):
            if remaining_time() < 10:
                break
            h_sa, c5_sa = binary_sa_fast(best_h, best_n, 5000, rng,
                                         t_start=0.0005, t_end=0.000001)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa
                print("  SA round {}: C5={:.7f}".format(sa_round, best_c5))

            # Quick continuous polish
            if remaining_time() > 8:
                _BETA = 2000
                bounds = [(0.0, 1.0)] * best_n
                try:
                    res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                                   method='L-BFGS-B', bounds=bounds,
                                   options={'maxiter': 500, 'ftol': 1e-16,
                                            'gtol': 1e-12, 'iprint': 0})
                    h_new = project_feasible(res.x, best_n)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new
                        print("  Polish: C5={:.7f}".format(best_c5))
                except Exception:
                    pass

    # Phase 5: Final micro-optimization
    if remaining_time() > 10:
        print("\n=== Phase5 Micro-opt ===")
        _N = best_n
        _BETA = 10000
        bounds = [(0.0, 1.0)] * best_n
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 2000, 'ftol': 1e-17,
                                    'gtol': 1e-13, 'iprint': 0,
                                    'maxls': 300, 'maxfun': 50000})
            h_new = project_feasible(res.x, best_n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                print("  Micro: C5={:.7f}".format(best_c5))
        except Exception:
            pass

        # Final pairwise
        if remaining_time() > 5:
            h_ls, c5_ls, _ = pairwise_local_search(best_h.copy(), best_n, max_sweeps=30)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
