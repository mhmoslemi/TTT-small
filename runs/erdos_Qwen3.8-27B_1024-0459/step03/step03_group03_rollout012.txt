def _exhaustive_swap(h, n):
    bc = _c5(h)
    improved = True
    guard = 0
    while improved and guard < 200 and _tl() > 3:
        improved = False
        guard += 1
        ones = np.where(h > 0.999)[0]
        zeros = np.where(h < 0.001)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_total_gain = 1e-14
        best_move = None
        for i in ones:
            for j in zeros:
                h2 = h.copy()
                h2[i] = 0.0
                h2[j] = 1.0
                c2 = _c5(h2)
                if c2 < bc - 1e-14:
                    gain = bc - c2
                    if gain > best_total_gain:
                        best_total_gain = gain
                        best_move = (i, j)
        if best_move is not None:
            i, j = best_move
            h[i] = 0.0
            h[j] = 1.0
            bc = _c5(h)
            improved = True
    return h, bc


def _anneal_double_swap(h, n, niter, rng):
    bc = _c5(h)
    temp = 0.001
    for it in range(niter):
        if _tl() < 3:
            break
        ones = np.where(h > 0.999)[0]
        zeros = np.where(h < 0.001)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i1 = ones[rng.randint(len(ones))]
        i2 = ones[rng.randint(len(ones))]
        if i1 == i2:
            continue
        j1 = zeros[rng.randint(len(zeros))]
        j2 = zeros[rng.randint(len(zeros))]
        if j1 == j2:
            continue
        if i1 in (j1, j2) or i2 in (j1, j2):
            continue
        h2 = h.copy()
        h2[i1] = 0.0
        h2[i2] = 0.0
        h2[j1] = 1.0
        h2[j2] = 1.0
        c2 = _c5(h2)
        if c2 < bc - 1e-14:
            h = h2
            bc = c2
        else:
            delta = c2 - bc
            if rng.rand() < np.exp(-delta / temp):
                h = h2
                bc = c2
        if it % 200 == 199:
            temp *= 0.95
    return h, bc


def _anneal_single_swap(h, n, niter, rng):
    bc = _c5(h)
    temp = 0.001
    for it in range(niter):
        if _tl() < 3:
            break
        ones = np.where(h > 0.999)[0]
        zeros = np.where(h < 0.001)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h2 = h.copy()
        h2[i] = 0.0
        h2[j] = 1.0
        c2 = _c5(h2)
        if c2 < bc - 1e-14:
            h = h2
            bc = c2
        else:
            delta = c2 - bc
            if rng.rand() < np.exp(-delta / temp):
                h = h2
                bc = c2
        if it % 300 == 299:
            temp *= 0.97
    return h, bc


def _gradient_smooth(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(corr))
    w = np.exp(beta * (corr - mx))
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
        sw = 1.0
    w /= sw
    g = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for k in nz:
        kk = int(k)
        wt = w[kk]
        if kk == 0:
            g[:] += wt * (1.0 - 2.0 * h)
        elif kk < n:
            g[:n - kk] += wt * (1.0 - h[kk:n])
            g[kk:] -= wt * h[:n - kk]
        else:
            p = kk - (n - 1)
            g[p:] += wt * (1.0 - h[:n - p])
            g[:n - p] -= wt * h[p:]
    return g


def _smooth_gd(h, n, nit, rng):
    best = h.copy()
    bc = _c5(best)
    for it in range(nit):
        if _tl() < 3:
            break
        g = _gradient_smooth(best, n, 80.0)
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            g = _gradient_smooth(best, n, 500.0)
            gn = float(np.linalg.norm(g))
            if gn < 1e-15:
                break
        g /= gn
        moved = False
        for a in [0.05, 0.03, 0.015, 0.008, 0.004, 0.02, 0.04, 0.01, 0.002]:
            h2 = _proj(best - a * g, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-15:
                best = h2
                bc = c2
                moved = True
                break
        if not moved:
            corr = np.correlate(best, 1.0 - best, mode='full')
            kmax = int(np.argmax(corr))
            g2 = np.zeros(n)
            if kmax == 0:
                g2[:] = 1.0 - 2.0 * best
            elif kmax < n:
                g2[:n - kmax] += 1.0 - best[kmax:]
                g2[kmax:] -= best[:n - kmax]
            else:
                p = kmax - (n - 1)
                g2[p:] += 1.0 - best[:n - p]
                g2[:n - p] -= best[p:]
            gn2 = float(np.linalg.norm(g2))
            if gn2 > 1e-15:
                g2 /= gn2
                for a in [0.04, 0.02, 0.01, 0.005, 0.008]:
                    h2 = _proj(best - a * g2, n)
                    c2 = _c5(h2)
                    if c2 < bc - 1e-15:
                        best = h2
                        bc = c2
                        moved = True
                        break
    return best, bc


def _slsqp_solve(h, n, beta, maxiter):
    ng = n

    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mx = float(np.max(c))
        sh = c - mx
        return (mx + np.log(np.exp(beta * sh).sum()) / beta) * (2.0 / ng)

    def _con_fun(x):
        return x.sum() - 0.5 * ng

    try:
        r = minimize(_obj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _con_fun}],
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return _proj(np.asarray(r.x, dtype=np.float64), n)
    except Exception:
        return _proj(h.copy(), n)


def _coordinate_descent(h, n, nit, rng):
    bc = _c5(h)
    for _ in range(nit):
        if _tl() < 3:
            break
        improved = False
        nb = np.where((h > 0.01) & (h < 0.99))[0]
        if len(nb) < 2:
            break
        rng.shuffle(nb)
        for idx in nb:
            if _tl() < 2:
                break
            old = h[idx]
            for delta in [0.02, -0.02, 0.01, -0.01, 0.04, -0.04, 0.005, -0.005]:
                nv = old + delta
                if nv < 0.0 or nv > 1.0:
                    continue
                others = np.delete(nb, idx)
                if len(others) == 0:
                    break
                j = others[rng.randint(len(others))]
                h2 = h.copy()
                h2[idx] = nv
                h2[j] += (old - nv)
                h2 = _proj(h2, n)
                c2 = _c5(h2)
                if c2 < bc - 1e-15:
                    h = h2
                    bc = c2
                    improved = True
                    break
    return h, bc


def _optimize_at_n(h_init, n, rng, time_budget):
    t_start = time.time()
    h = _proj(h_init.copy(), n)
    c5 = _c5(h)

    h_bin = _to_binary(h, n)
    h_bin, c5b = _exhaustive_swap(h_bin, n)
    c5b = _c5(h_bin)
    if c5b < c5:
        h = h_bin.copy()
        c5 = c5b

    for _ in range(5):
        if time.time() - t_start > time_budget * 0.5 or _tl() < 30:
            break
        h_bin, c5b = _anneal_double_swap(h_bin, n, 8000, rng)
        h_bin, c5b = _anneal_single_swap(h_bin, n, 5000, rng)
        h_bin, c5b2 = _exhaustive_swap(h_bin, n)
        if c5b2 < c5:
            h = h_bin.copy()
            c5 = c5b2
            h_bin = h_bin.copy()

    if time.time() - t_start < time_budget * 0.7 and _tl() > 20:
        h = _proj(_interp(h_bin, n, n), n)
        for beta in [50, 200, 500]:
            if _tl() < 15:
                break
            h2 = _slsqp_solve(h, n, beta, 200)
            c2 = _c5(h2)
            if c2 < c5:
                h = h2
                c5 = c2
        h, c5 = _smooth_gd(h, n, 300, rng)

    if time.time() - t_start < time_budget * 0.85 and _tl() > 15:
        h_bin2 = _to_binary(h, n)
        h_bin2, c5b2 = _exhaustive_swap(h_bin2, n)
        h_bin2, c5b2 = _anneal_double_swap(h_bin2, n, 5000, rng)
        h_bin2, c5b2 = _exhaustive_swap(h_bin2, n)
        if c5b2 < c5:
            h = h_bin2.copy()
            c5 = c5b2

    if _tl() > 10 and time.time() - t_start < time_budget:
        h2 = _slsqp_solve(h, n, 2000, 300)
        c2 = _c5(h2)
        if c2 < c5:
            h = h2
            c5 = c2
        h, c5 = _smooth_gd(h, n, 200, rng)
        h, c5 = _coordinate_descent(h, n, 100, rng)

    h = _proj(h, n)
    c5 = _c5(h)
    return h, c5


def _multi_start_binary(n, rng, n_starts=5):
    best_h = None
    best_c = 1.0
    for s in range(n_starts):
        h = np.zeros(n)
        positions = np.arange(n)
        rng.shuffle(positions)
        h[positions[:n // 2]] = 1.0
        h, c = _exhaustive_swap(h, n)
        h, c = _anneal_double_swap(h, n, 5000, rng)
        h, c = _exhaustive_swap(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    global _t0, _tb
    _t0 = time.time()
    _tb = budget_s
    rng = np.random.RandomState(seed)

    try:
        hpar = np.array(initial_h_values, dtype=np.float64)
        npar = len(hpar)
    except Exception:
        npar = 160
        hpar = np.zeros(npar)
        hpar[:npar // 2] = 1.0

    bh = _proj(hpar, npar)
    bc5 = _c5(bh)
    bn = npar
    print("Init n={} C5={:.8f}".format(bn, bc5))

    # Phase 1: Optimize at parent resolution
    t_budget1 = 350
    h1, c1 = _optimize_at_n(bh, npar, rng, t_budget1)
    if c1 < bc5:
        bh, bc5, bn = h1, c1, npar
        print("n={} C5={:.8f}".format(npar, bc5))

    # Phase 2: Try nearby resolutions
    alts = []
    for d in [2, -2, 4, -4, 6, -6, 8, 12, 16, 24, 32]:
        nn = npar + d
        if nn >= 32 and nn % 2 == 0 and nn != npar:
            alts.append(nn)
    alts = sorted(set(alts))

    n_alts_budget = 200
    for nn in alts:
        if _tl() < 40:
            break
        t_alloc = min(n_alts_budget, max(30, _tl() // max(1, len(alts))))
        hi = _proj(_interp(bh, bn, nn), nn)
        h2, c2 = _optimize_at_n(hi, nn, rng, t_alloc)
        if c2 < bc5:
            bh, bc5, bn = h2, c2, nn
            print("n={} C5={:.8f}".format(nn, c2))

    # Phase 3: Try a multi-start at best resolution
    if _tl() > 120:
        mb_h, mb_c = _multi_start_binary(bn, rng, n_starts=4)
        if mb_c < bc5:
            h3, c3 = _optimize_at_n(mb_h, bn, rng, min(100, int(_tl())))
            if c3 < bc5:
                bh, bc5 = h3, c3
                print("MultiStart n={} C5={:.8f}".format(bn, c3))

    # Phase 4: Final polish at best resolution
    if _tl() > 30:
        h, c5 = _smooth_gd(bh, bn, 300, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        h_bin_f = _to_binary(bh, bn)
        h_bin_f, cbf = _exhaustive_swap(h_bin_f, bn)
        h_bin_f, cbf = _anneal_double_swap(h_bin_f, bn, 6000, rng)
        h_bin_f, cbf = _exhaustive_swap(h_bin_f, bn)
        if cbf < bc5:
            bh, bc5 = h_bin_f, cbf
            print("FinalBin n={} C5={:.8f}".format(bn, cbf))
        # Fractional polish
        if _tl() > 15:
            h2 = _slsqp_solve(bh, bn, 3000, 400)
            c2 = _c5(h2)
            if c2 < bc5:
                bh, bc5 = h2, c2
            h, c5 = _smooth_gd(bh, bn, 200, rng)
            if c5 < bc5:
                bh, bc5 = h, c5
            h, c5 = _coordinate_descent(bh, bn, 200, rng)
            if c5 < bc5:
                bh, c5 = h, c5
            if c5 < bc5:
                bh, bc5 = h, c5

    # Phase 5: Try one more resolution bump
    if _tl() > 40:
        for nn in [bn + 2, bn - 2, bn + 4]:
            if _tl() < 25:
                break
            if nn < 32 or nn % 2 != 0:
                continue
            hi = _proj(_interp(bh, bn, nn), nn)
            h2, c2 = _optimize_at_n(hi, nn, rng, min(30, int(_tl())))
            if c2 < bc5:
                bh, bc5, bn = h2, c2, nn
                print("Late n={} C5={:.8f}".format(nn, c2))

    # Final evaluation
    bh = _proj(bh, bn)
    fc5 = _c5(bh)
    try:
        ec = evaluate_erdos_solution(bh)
        if ec < fc5:
            fc5 = ec
    except Exception:
        pass
    print("FINAL n={} C5={:.8f}".format(bn, fc5))
    return (bh, fc5, bn)
