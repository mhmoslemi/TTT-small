        return h.copy()
    xs = (np.arange(n_from) + 0.5) * (2.0 / n_from)
    xd = (np.arange(n_to) + 0.5) * (2.0 / n_to)
    return np.interp(xd, xs, h)


def _lag_grad(h, k, n):
    g = np.zeros(n)
    if k == 0:
        return 1.0 - 2.0 * h
    elif k > 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        m = -k
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    return g


def _smooth_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(corr))
    w = np.exp(beta * (corr - mx))
    sw = float(w.sum())
    if sw < 1e-300:
        idx = int(np.argmax(corr))
        w = np.zeros(len(corr))
        w[idx] = 1.0
        sw = 1.0

    nz = np.where(w > thresh)[0]
    g = np.zeros(n)

        g += w[kk] * _lag_grad(h, d, n)

        return x.sum() - 0.5 * n

    h, c5 = _pgd_loop(h, n, 300, rng)
    h, c5 = _coord_descent(h, n, 100, rng)

    if _tl() > time_alloc * 0.15:
        h2 = _slsqp(h, n, 20000, 500)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2
        h, c5 = _pgd_loop(h, n, 200, rng)

    return h, c5


def run(seed=42, budget_s=1300, **kwargs):
    global _t0, _tb
    _t0 = time.time()
    _tb = budget_s
    rng = np.random.RandomState(seed)

    try:
        hpar = np.array(initial_h_values, dtype=np.float64)
        npar = len(hpar)
    except Exception:
        npar = 160
        hpar = np.zeros(npar, dtype=np.float64)
        hpar[:npar // 2] = 1.0

    bh = _proj(hpar.copy(), npar)
    bc5 = _c5(bh)
    bn = npar
    print("Init n={} C5={:.10f}".format(bn, bc5), flush=True)

    # Phase 1: Optimize at parent resolution
    h, c5 = _optimize_at_res(bh, npar, rng, 350)
    if c5 < bc5:
        bh, bc5, bn = h, c5, npar
    print("Phase1 n={} C5={:.10f}".format(bn, bc5), flush=True)

    # Phase 2: Try nearby resolutions
    alts = []
    for delta in [2, -2, 4, -4, 6, -6, 8, -8, 10, -10, 12, 14, 16, 20, 30, 40]:
        nn = npar + delta
        if nn >= 40 and nn % 2 == 0 and nn not in alts and nn not in [npar]:
            alts.append(nn)
    alts = sorted(alts, key=lambda x: abs(x - npar))

    for nn in alts:
        if _tl() < 40:
            break
        time_each = min(60, _tl() // max(1, len(alts)))
        hi = _interp(bh, bn, nn)
        h2, c2 = _optimize_at_res(hi, nn, rng, time_each)
        if c2 < bc5:
            bh, bc5, bn = h2, c2, nn
            print("Alt n={} C5={:.10f}".format(nn, c2), flush=True)

    # Phase 3: Try higher resolutions
    high_res = [npar + 50, npar + 80, npar + 120, npar + 160]
    for nn in high_res:
        if _tl() < 60:
            break
        nn = nn if nn % 2 == 0 else nn + 1
        if nn < 100:
            continue
        hi = _interp(bh, bn, nn)
        h2, c2 = _optimize_at_res(hi, nn, rng, min(80, _tl() // 2))
        if c2 < bc5:
            bh, bc5, bn = h2, c2, nn
            print("High n={} C5={:.10f}".format(nn, c2), flush=True)

    # Phase 4: Random restarts to escape local minima
    for attempt in range(5):
        if _tl() < 40:
            break
        h_noisy = bh + rng.randn(bn) * 0.01
        h_noisy = _proj(h_noisy, bn)
        h2, c2 = _optimize_at_res(h_noisy, bn, rng, 30)
        if c2 < bc5:
            bh, bc5 = h2, c2
            print("Restart{} n={} C5={:.10f}".format(attempt, bn, c2), flush=True)

    # Phase 5: Final high-precision polish
    if _tl() > 15:
        h2 = _slsqp(bh, bn, 50000, 800)
        c2 = _c5(h2)
        if c2 < bc5:
            bh, bc5 = h2, c2
        h, c5 = _pgd_loop(bh, bn, 300, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _swap_loop(bh, bn, 30, rng)
        if c5 < bc5:
            bh, c5 = h, c5
            bc5 = c5
        h, c5 = _coord_descent(bh, bn, 200, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        if _tl() > 10:
            h2 = _slsqp(bh, bn, 100000, 600)
            c2 = _c5(h2)
            if c2 < bc5:
                bh, bc5 = h2, c2
            h, c5 = _pgd_loop(bh, bn, 200, rng)
            if c5 < bc5:
                bh, bc5 = h, c5

    bh = _proj(bh, bn)
    fc5 = _c5(bh)
    try:
        ec = evaluate_erdos_solution(bh)
        if ec is not None and ec < fc5:
            fc5 = ec
    except Exception:
        pass
    print("FINAL n={} C5={:.10f}".format(bn, fc5), flush=True)
    return (bh, fc5, bn)
