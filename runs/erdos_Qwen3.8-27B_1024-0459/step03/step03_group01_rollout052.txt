def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():

def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def to_binary(h, n, rng):
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h_bin == 1)[0]
        to_remove = ones_idx[np.argsort(h[ones_idx])][:n_ones - target]
        h_bin[to_remove] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        to_add = zeros_idx[np.argsort(-h[zeros_idx])][:target - n_ones]
        h_bin[to_add] = 1.0
    return h_bin

def thue_morse(n, rng):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') & 1
        h[i] = 1.0 if c == 1 else 0.0

        if n_ones > target:

        else:

            perm = rng.permutation(len(idx))[:target - n_ones]
            h[idx[perm]] = 1.0

def van_der_corput(n, rng):
    h = np.zeros(n)
    for i in range(1, n + 1):
        f = 0.0
        d = 1.0
        x = i - 1
        while x > 0:
            d /= 2.0
            f += d * (x & 1)
            x >>= 1
        h[i - 1] = 1.0 if f >= 0.5 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            idx = np.where(h == 1)[0]
            perm = rng.permutation(len(idx))[:n_ones - target]

def shift_gradient(h, n, k):
    """Gradient of corr[k] w.r.t. h."""
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - h[:npairs])
        g[:npairs] -= h[off:off + npairs]
    elif off < 0:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - h[ao:ao + npairs])
        g[ao:ao + npairs] -= h[:npairs]
    return g

def subgradient_hmax(h, n, top_k=8):
    """Weighted subgradient of max corr."""
    corr = compute_corr(h)
    # Soft top-k weighting
    m = np.max(corr)
    active = np.where(corr > m - 1e-8)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    elif len(active) > top_k * 3:
        active = np.argsort(-corr)[:top_k]
    g = np.zeros(n)
    for k in active:
        g += shift_gradient(h, n, int(k))
    g /= len(active)
    return g

        if gnorm < 1e-15:
            break
        step = lr / gnorm

        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new.copy()
            lr = min(lr * 1.02, 0.3)
        else:
            lr = max(lr * 0.85, 0.005)
        if it % 200 == 0:
            # Annneal: occasional large step
            h_pert = project_feasible(h + 0.01 * np.random.randn(n), n)
            if compute_c5(h_pert) < best_c5:
                h = h_pert
    return best_h, best_c5

def block_move_search(h, n, time_limit=20, rng=None):
    """Move contiguous blocks of 1s."""
    if rng is None:
        rng = np.random.RandomState(42)
    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx_val
    start = time.time()
    stall = 0
    
    while time.time() - start < time_limit and remaining() > 5:
        if stall > 50:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            corr = compute_corr(h)
            cur_max = float(np.max(corr))
            stall = 0
            continue
        
        ones = np.where(h == 1)[0]
        if len(ones) < 3:
            break
        diffs = np.diff(ones)
        breaks = np.where(diffs > 1)[0]
        block_starts = np.concatenate([[0], breaks + 1])
        block_ends = np.concatenate([breaks, [len(ones) - 1]])
        
        if len(block_starts) == 0:
            break
        
        # Pick blocks of length 2-10
        valid_blocks = [(s, e) for s, e in zip(block_starts, block_ends) 
                        if 2 <= (ones[e] - ones[s] + 1) <= min(12, n // 8)]
        if not valid_blocks:
            stall += 1
            continue
        
        si, ei = valid_blocks[rng.randint(len(valid_blocks))]
        block = ones[si:ei + 1]
        block_len = len(block)
        
        # Find valid target positions
        target = rng.randint(0, n - block_len + 1)
        overlap = 0
        for pos in block:
            if target <= pos < target + block_len:
                overlap += 1
        if 0 < overlap < block_len:
            stall += 1
            continue
        
        target_seg = h[target:target + block_len]
        if overlap == 0:
            if np.any(target_seg == 1):
                stall += 1
                continue
        
        # Perform move
        h[block] = 0.0
        h[target:target + block_len] = 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        
        if new_max < cur_max - 1e-15:
            corr = new_corr
            cur_max = new_max
            best_c5 = cur_max * dx_val
            stall = 0
        else:
            h[target:target + block_len] = 0.0
            h[block] = 1.0
            stall += 1
    
    return h, best_c5

def slsqp_refine(h, n, maxiter=200):
    """Quick SLSQP with softened max."""
    global _N, _BETA, _TGT
    _N = n
    _BETA = 2000.0
    _TGT = n / 2.0
    dx_val = 2.0 / n
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(_BETA * shifted, -50, 50)
        return (m + np.log(np.sum(np.exp(cl))) / _BETA) * dx_val
    
    def obj_grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(_BETA * shifted, -50, 50)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] > 1e-14:
                g += w[k] * shift_gradient(x, n, k)
        return g * dx_val
    
    def csum(x):
        return np.sum(x) - _TGT
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': csum, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

        h_best_bin = best_h.copy() if np.all((best_h == 0) | (best_h == 1)) else to_binary(best_h, best_n, rng)
        c5_check = compute_c5(h_best_bin)
        if c5_check < best_c5:
            best_h, best_c5 = h_best_bin.copy(), c5_check
        
        # Multiple rounds of dense + block + SA
        t_total = min(120, remaining() - 20)
        t_chunk = t_total // 3
        if t_chunk > 15:
            h_d2, c5_d2 = binary_dense_search(best_h, best_n, time_limit=t_chunk, rng=rng)
            if c5_d2 < best_c5:
                best_h, best_c5, best_n = h_d2.copy(), c5_d2, best_n
            if remaining() > 15:
                h_bm3, c5_bm3 = block_move_search(best_h, best_n, time_limit=min(t_chunk, remaining() - 10), rng=rng)
                if c5_bm3 < best_c5:
                    best_h, best_c5, best_n = h_bm3.copy(), c5_bm3, best_n
            if remaining() > 20:
                h_sa2, c5_sa2 = simulated_annealing_binary(best_h, best_n, time_limit=min(30, remaining() - 10), rng=rng)
                if c5_sa2 < best_c5:
                    best_h, best_c5, best_n = h_sa2.copy(), c5_sa2, best_n
    
    # === Phase 6: Continuous polish on best binary solution ===
    if remaining() > 30:
        h_poly = best_h.astype(float).copy()
        h_pol, c5_pol = continuous_refine(h_poly, best_n, time_limit=min(40, remaining() - 15))
        if c5_pol < best_c5:
            best_h, best_c5, best_n = h_pol, c5_pol, best_n
    
    # === Phase 7: Final SLSQP polish ===
    if remaining() > 20:
        h_final = slsqp_refine(best_h, best_n, maxiter=150)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final
    
    # === Final validation ===
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
