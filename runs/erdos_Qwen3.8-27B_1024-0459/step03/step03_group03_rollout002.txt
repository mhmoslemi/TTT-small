def _fix_sum(h, n):
    h = np.asarray(h, dtype=np.float64)
    target = 0.5 * float(n)
    s = float(np.sum(h))
    diff = target - s
    if abs(diff) <= 1e-13:
        return h

    if abs(diff) > 1e-6:
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(m):
            h[m] += diff / float(np.count_nonzero(m))
        else:
            h[:] += diff / float(n)
        np.clip(h, 0.0, 1.0, out=h)
        s = float(np.sum(h))
        diff = target - s

    if abs(diff) > 1e-15:
        if diff > 0:
            m = (h > 1e-10) & (h < 1.0 - 1e-10)
            if np.any(m):
                idx = np.where(m)[0]
                i = int(idx[np.argmax(1.0 - h[idx])])
            else:
                i = int(np.argmax(1.0 - h))
            room = 1.0 - h[i]
            add = min(diff, room)
            h[i] += add
            diff -= add
        else:
            m = (h > 1e-10) & (h < 1.0 - 1e-10)
            if np.any(m):

                i = int(idx[np.argmax(h[idx])])

            diff += sub

def _ggrad(h, k):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n, dtype=np.float64)

    if s >= 0:
        if s == 0:
            g[:] += 1.0 - 2.0 * h
        else:
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
    else:
        p = -s
        g[:n - p] += 1.0 - h[p:n]
        g[p:] -= h[:n - p]
    return g

            w[0] = 1.0

            if c2 < bc - 1e-14:
                best = h2
                bc = c2
                imp = True
                no = 0
                break

        if not imp and oi > oj:
            for d in (0.25, 0.10, 0.05, 0.02):
                if oi - d >= 0.0 and oj + d <= 1.0:
                    h2 = best.copy()
                    h2[i] = oi - d
                    h2[j] = oj + d
                    h2 = _proj(h2, n)
                    c2 = _c5(h2)
                    if c2 < bc - 1e-14:
                        best = h2
                        bc = c2
                        imp = True
                        no = 0
                        break

        if not imp and oj > oi:
            for d in (0.25, 0.10, 0.05, 0.02):
                if oi + d <= 1.0 and oj - d >= 0.0:
                    h2 = best.copy()
                    h2[i] = oi + d
                    h2[j] = oj - d
                    h2 = _proj(h2, n)
                    c2 = _c5(h2)
                    if c2 < bc - 1e-14:
                        best = h2
                        bc = c2
                        imp = True
                        no = 0
                        break

        if not imp:
            no += 1
            if no > 40:
                break

    return best, bc


def _freq_perturb(h, n, m, amp):
    idx = np.arange(n)
    h2 = h + float(amp) * np.cos(2.0 * np.pi * float(m) * idx / float(n))
    return _proj(h2, n)


def _fourier_start(n, M, rng):
    idx = np.arange(n)
    h = np.full(n, 0.5, dtype=np.float64)
    scale = 0.15
    for m in range(1, int(M) + 1):
        if _time_left() < 2.0:
            break
        ac = rng.uniform(-scale, scale)
        sc = rng.uniform(-scale, scale)
        arg = 2.0 * np.pi * float(m) * idx / float(n)
        h += ac * np.cos(arg) + sc * np.sin(arg)
    return _proj(h, n)


def _binary_start(n, rng):
    h = np.zeros(n, dtype=np.float64)
    if n % 2 == 0:
        k = n // 2
        idx = rng.choice(n, size=k, replace=False)
        h[idx] = 1.0
    else:
        k = (n - 1) // 2
        idx = rng.choice(n, size=k, replace=False)
        h[idx] = 1.0
        others = np.setdiff1d(np.arange(n), idx)
        if others.size:
            h[others[0]] = 0.5
    return _proj(h, n)


def _refine(h, n, rng, limit):
    t0 = time.time()
    best = h.copy()
    bc = _c5(best)

    steps = (0.10, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001)
    deltas = (0.02, 0.01, 0.005, 0.05, 0.015, 0.03)

    if n <= 220:
        maxit = 420
    elif n <= 420:
        maxit = 260
    else:
        maxit = 130

    for beta in (10, 40, 160):
        rem = float(limit) - (time.time() - t0)
        if rem < 15.0 or _time_left() < 12.0:
            break
        h2, c2 = _grad_loop(best, n, maxit, beta, 32, steps)
        if c2 < bc:
            best, bc = h2, c2
        h2, c2 = _pair_loop(best, n, 110, rng, deltas)
        if c2 < bc:
            best, bc = h2, c2

    rem = float(limit) - (time.time() - t0)
    if rem > 15.0 and _time_left() > 12.0:
        h2, c2 = _grad_loop(best, n, max(50, maxit // 2), 600, 10, steps)
        if c2 < bc:
            best, bc = h2, c2
        h2, c2 = _pair_loop(best, n, 70, rng, deltas)
        if c2 < bc:
            best, bc = h2, c2

    return best, bc


def _start_key(item):
    return item[0]


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _TB
    _T0 = time.time()
    _TB = float(budget_s)
    rng = np.random.RandomState(int(seed))

    best_h = None
    best_c = 1e100
    best_n = 160

    try:
        bh = _cos_h(160, 1.0 / np.sqrt(6.0))
        bc = _c5(bh)
        best_h = bh
        best_c = bc
        best_n = 160
    except Exception:
        h = np.zeros(160, dtype=np.float64)
        h[:80] = 1.0
        best_h = _proj(h, 160)
        best_c = _c5(best_h)
        best_n = 160

    parent_n = None
    try:
        parent_n = int(len(initial_h_values))
    except Exception:
        parent_n = None

    base_ns = [96, 128, 148, 160, 180, 192, 212, 224, 240, 256, 288, 320, 360, 400, 448, 512]
    ns = set()
    for v in base_ns:
        if v >= 32:
            ns.add(v)

    if parent_n is not None:
        for d in (-32, -24, -16, -12, -8, -6, -4, -2, 0, 2, 4, 6, 8, 12, 16, 24, 32, 48, 64, 96, 128):
            vv = parent_n + d
            if vv % 2 != 0:
                vv -= 1
            if vv >= 32 and vv <= 1200:
                ns.add(vv)

    ns = sorted([v for v in ns if 32 <= v <= 1200])

    analytic_results = []
    bmap = {}

    for n in ns:
        if _time_left() < 4.0:
            break
        try:
            h, c, b = _cos_analytic_best(n, 220)
            if np.isfinite(c):
                analytic_results.append((float(c), int(n), float(b)))
                bmap[int(n)] = float(b)
                if c < best_c:
                    best_h = h
                    best_c = float(c)
                    best_n = int(n)
        except Exception:
            pass

    analytic_results.sort(key=_start_key)

    direct_ns = []
    for _, n, _ in analytic_results:
        if n not in direct_ns:
            direct_ns.append(n)
        if len(direct_ns) >= 5:
            break
    if int(best_n) not in direct_ns:
        direct_ns.append(int(best_n))
    if parent_n is not None:
        pn = int(parent_n) - (int(parent_n) % 2)
        if pn >= 32 and pn not in direct_ns:
            direct_ns.append(pn)

    for n in direct_ns:
        if _time_left() < 8.0:
            break
        try:
            h, c, b = _cos_direct_scan(n, 72, 0.0, 1.0)
            if c < best_c:
                best_h = h
                best_c = float(c)
                best_n = int(n)
            if _time_left() > 12.0:
                lo = max(0.0, b - 0.05)
                hi = min(1.0, b + 0.05)
                h2, c2, b2 = _cos_direct_scan(n, 96, lo, hi)
                if c2 < best_c:
                    best_h = h2
                    best_c = float(c2)
                    best_n = int(n)
        except Exception:
            pass

    try:
        hpar = np.array(initial_h_values, dtype=np.float64)
        npar = int(len(hpar))
        if npar >= 32:
            hpar = _proj(hpar, npar)
            cpar = _c5(hpar)
            if cpar < best_c:
                best_h = hpar
                best_c = float(cpar)
                best_n = npar
    except Exception:
        pass

    if best_h is not None and _time_left() > 60.0:
        lim = min(180.0, max(30.0, _time_left() * 0.25))
        try:
            h2, c2 = _refine(best_h, best_n, rng, lim)
            if c2 < best_c:
                best_h = h2
                best_c = float(c2)
        except Exception:
            pass

    starts = []

    if best_h is not None and _time_left() > 80.0:
        nstart = int(best_n) if int(best_n) <= 512 else 256
        hbase = _interp(best_h, nstart) if nstart != int(best_n) else best_h.copy()
        cb = _c5(hbase)
        if cb < best_c:
            best_h = hbase
            best_c = float(cb)
            best_n = nstart

        for ph in (0.25, -0.25, 0.55, -0.55, 0.85, -0.85):
            hp = _cos_phase_h(nstart, 0.45, ph)
            cp = _c5(hp)
            if cp < best_c + 0.015:
                starts.append((float(cp), hp, int(nstart), 35.0))

        if _time_left() > 35.0:
            for m in (2, 3, 4):
                for amp in (0.05, -0.05, 0.10, -0.10):
                    hp = _freq_perturb(hbase, nstart, m, amp)
                    cp = _c5(hp)
                    if cp < best_c + 0.015:
                        starts.append((float(cp), hp, int(nstart), 35.0))

        for amp in (0.02, 0.05, 0.12):
            hp = _proj(hbase + amp * (rng.rand(nstart) - 0.5), nstart)
            cp = _c5(hp)
            if cp < best_c + 0.02:
                starts.append((float(cp), hp, int(nstart), 30.0))

        if _time_left() > 30.0:
            for _ in range(2):
                hp = _fourier_start(nstart, 6, rng)
                cp = _c5(hp)
                if cp < best_c + 0.03:
                    starts.append((float(cp), hp, int(nstart), 40.0))

        hp = _binary_start(nstart, rng)
        cp = _c5(hp)
        if cp < best_c + 0.05:
            starts.append((float(cp), hp, int(nstart), 40.0))

        for d in (-16, 16, -32, 32):
            n2 = int(best_n) + d
            if 32 <= n2 <= 900:
                hp = _interp(best_h, n2)
                cp = _c5(hp)
                if cp < best_c + 0.015:
                    starts.append((float(cp), hp, int(n2), 30.0))

        try:
            hpar = np.array(initial_h_values, dtype=np.float64)
            npar = int(len(hpar))
            if npar >= 32:
                hpar = _proj(hpar, npar)
                cpar = _c5(hpar)
                if cpar < best_c + 0.03:
                    starts.append((float(cpar), hpar, int(npar), 35.0))
        except Exception:
            pass

    starts.sort(key=_start_key)

    for item in starts[:6]:
        if _time_left() < 25.0:
            break
        cp, hp, sn, s_lim = item
        try:
            hp = _proj(hp, sn)
            cp = _c5(hp)
            if cp < best_c:
                best_h = hp
                best_c = float(cp)
                best_n = int(sn)

            lim_use = min(float(s_lim), _time_left() * 0.22, 50.0)
            if _time_left() > max(12.0, lim_use):
                h2, c2 = _refine(hp, sn, rng, lim_use)
                if c2 < best_c:
                    best_h = h2
                    best_c = float(c2)
        except Exception:
            pass

    if best_h is not None and _time_left() > 40.0:
        lim = min(120.0, max(20.0, _time_left() * 0.45))
        try:
            h2, c2 = _refine(best_h, best_n, rng, lim)
            if c2 < best_c:
                best_h = h2
                best_c = float(c2)
        except Exception:
            pass

    best_h = _proj(best_h, int(best_n))
    best_c = _c5(best_h)

    try:
        ec = evaluate_erdos_solution(best_h)
        ec = float(ec)
        if np.isfinite(ec) and ec > 0.0:
            best_c = min(best_c, ec)
    except Exception:
        pass

    return (best_h, float(best_c), int(best_n))
