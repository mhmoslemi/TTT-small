import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Distribute uniformly among non-saturated variables
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            # All saturated, need to adjust
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            if np.sum(mask) > 0:
                h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        # Find indices to adjust
        if diff > 0:
            candidates = np.where(h < 1.0)[0]
        else:
            candidates = np.where(h > 0.0)[0]
        for idx in candidates:
            adjustment = min(diff if diff > 0 else 1-diff, (1.0-h[idx]) if diff > 0 else h[idx])
            h[idx] += adjustment
            diff -= adjustment
            if abs(diff) < 1e-14:
                break
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 2.0)

def smooth_max(corr, beta):
    """Smooth approximation of max using log-sum-exp"""
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def objective(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max(corr, beta) * dx

def objective_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    weights /= np.sum(weights)
    # Gradient of smooth_max wrt corr
    # Then gradient of corr wrt x
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-10:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            n_pairs = n - offset
            grad[offset:offset+n_pairs] += w * (1.0 - x[:n_pairs])
            grad[:n_pairs] -= w * x[offset:offset+n_pairs]
        else:
            abs_off = -offset
            n_pairs = n - abs_off
            grad[:n_pairs] += w * (1.0 - x[abs_off:abs_off+n_pairs])
            grad[abs_off:abs_off+n_pairs] -= w * x[:n_pairs]
    return grad * dx

def lag_gradient(h, n, offset):
    """Gradient of corr at a specific lag"""
    grad = np.zeros(n)
    if offset >= 0:
        n_pairs = n - offset
        if n_pairs <= 0:
            return grad
        grad[offset:offset+n_pairs] += (1.0 - h[:n_pairs])
        grad[:n_pairs] -= h[offset:offset+n_pairs]
    else:
        abs_off = -offset
        n_pairs = n - abs_off
        if n_pairs <= 0:
            return grad
        grad[:n_pairs] += (1.0 - h[abs_off:abs_off+n_pairs])
        grad[abs_off:abs_off+n_pairs] -= h[:n_pairs]
    return grad

def lbfgs_optimize(h, n, beta, maxiter, time_limit):
    """Optimize using L-BFGS-B"""
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return objective(x, beta)
    
    def grads(x):
        return objective_grad(x, beta)
    
    try:
        res = minimize(obj, h, jac=grads, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10, 'iprint': 0})
        h_new = project_feasible(res.x, n)
    except Exception:
        h_new = h.copy()
    return h_new

- If offset ≥ 0: corr = sum_{i=0}^{n-1-offset} h[i+offset] * (1-h[i])
  This counts pairs where position i+offset has a 1 and position i has a 0.
  So it's #{i: h[i+offset]=1, h[i]=0} = #{(i,j): j=i+offset, h[j]=1, h[i]=0}
  
- If offset < 0: let a = -offset. corr = sum_{i=0}^{n-1-a} h[i] * (1-h[i+a])
  This counts #{i: h[i]=1, h[i+a]=0}

So the correlation at lag |offset| counts the number of "1 followed by 0 after |offset| steps" (for positive offset) or "1 preceded by 0 after |offset| steps" (for negative offset).

In both cases, for a binary sequence, corr at lag d is:
#{i: h[i]=1, h[i+d]=0} (for the valid range)

And this is NOT the same as the autocorrelation. It's more like the "asymmetric correlation" or "cross-correlation" between h and (1-h).

The key constraint is: for ALL lags d (positive and negative, i.e., all k in the full correlation), the count must be ≤ threshold.

For a balanced binary sequence (equal number of 1s and 0s), by symmetry:
#{i: h[i]=1, h[i+d]=0} = #{i: h[i]=0, h[i+d]=1} (in the cyclic case)
