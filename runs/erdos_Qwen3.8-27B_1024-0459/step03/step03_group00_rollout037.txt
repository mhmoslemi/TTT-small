    n = _G_N
    dx = _G_DX
    beta = _G_BETA

    # corr[k] = sum_i x[i] * (1 - x[i + offset_k])
    # d(corr[k])/dx[i] = (1 - x[i + offset_k]) - x[i - offset_k] (with appropriate bounds)

            n_pairs = n - offset
            if n_pairs <= 0:

def make_thue_morse(n):
    h = np.zeros(n)

        cnt = 0
        v = i
        while v:
            cnt ^= (v & 1)
            v >>= 1

        else:

def make_van_der_corput(n):
    """van der Corput sequence based initialization."""
    h = np.zeros(n)
    for i in range(n):
        # van der Corput in base 2
        v = 0.0
        d = 0.5
        k = i + 1
        while k > 0:
            v += d * (k % 2)
            k //= 2
            d *= 0.5
        h[i] = 1.0 if v < 0.5 else 0.0
    ones = np.sum(h == 1.0)
    if ones != n // 2:
        idx_ones = np.where(h == 1.0)[0]
        idx_zeros = np.where(h == 0.0)[0]
        if ones > n // 2:
            h[idx_ones[ones - n//2:]] = 0.0
        else:
            h[idx_zeros[:n//2 - int(ones)]] = 1.0
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def _smooth_max_val(corr):

    beta = _G_BETA
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    return m + np.log(np.sum(exps)) / beta

    m = np.max(corr)
    beta = _G_BETA
    shifted = corr - m

    return exps / np.sum(exps)

def obj_grad(x):
    n = _G_N
    corr = np.correlate(x, 1.0 - x, mode='full')
    weights = _smooth_max_weights(corr)
    dx = _G_DX
    grad = np.zeros(n)
    n_corr = len(corr)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-14:
            continue
        offset = k_idx - (n - 1)
        abs_off = abs(offset)
        n_pairs = n - abs_off
        if n_pairs <= 0:
            continue
        if offset >= 0:
            grad[offset:] += w * (1.0 - x[:n_pairs])
            grad[:n_pairs] -= w * x[offset:]
        else:
            grad[:n_pairs] += w * (1.0 - x[abs_off:])
            grad[abs_off:] -= w * x[:n_pairs]
    grad *= dx
    cv = np.sum(x) - _G_TGT
    grad += 2.0 * _G_PEN * cv * np.ones(n)
    return grad


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = 0
        v = i
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    ones = int(np.sum(h == 1.0))
    if ones != n // 2:
        if ones > n // 2:
            idx = np.where(h == 1.0)[0]
            h[idx[ones - n // 2:]] = 0.0
        else:
            idx = np.where(h == 0.0)[0]
            h[idx[:n // 2 - ones]] = 1.0
    return h


def make_van_der_corput(n):
    h = np.zeros(n)

        v = 0.0
        d = 0.5
        k = i + 1

            v += d * (k % 2)
            k //= 2
            d *= 0.5
        h[i] = 1.0 if v < 0.5 else 0.0
    ones = int(np.sum(h == 1.0))
    if ones != n // 2:
        if ones > n // 2:
            idx = np.where(h == 1.0)[0]
            h[idx[ones - n // 2:]] = 0.0
        else:
            idx = np.where(h == 0.0)[0]
            h[idx[:n // 2 - ones]] = 1.0
    return h

def projected_subgradient(h, n, max_time, top_k=5):
    global _G_N, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    h_best = h.copy()
    c5_best = compute_c5(h)
    deadline = time.time() + max_time
    step_sizes = [0.01, 0.02, 0.05, 0.1, 0.2, 0.03, 0.08]

    for it in range(500):
        if time.time() > deadline - 3:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode='full')
        n_corr = len(corr)
        top_idx = np.argsort(corr)[-top_k:]
        top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]

        grad = np.zeros(n)
        for k_idx in top_idx:
            offset = int(k_idx) - (n - 1)
            abs_off = abs(offset)
            n_pairs = n - abs_off
            if n_pairs <= 0:
                continue
            if offset >= 0:
                grad[offset:] += (1.0 - h_best[:n_pairs])
                grad[:n_pairs] -= h_best[offset:]
            else:
                grad[:n_pairs] += (1.0 - h_best[abs_off:])
                grad[abs_off:] -= h_best[:n_pairs]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm

        improved = False
        for step in step_sizes:
            h_trial = h_best - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-12:
                c5_best = c5_trial
                h_best = h_trial.copy()
                improved = True
                break
        if not improved and it > 50:
            break
    return h_best, c5_best

        print("  After swaps: n={}, C5={:.8f}".format(best_n, best_c5))

    # Phase 6: Local perturbation refinement
    if remaining_time() > 20.0:
        print("\n--- Phase 6: Local Perturbation ---")
        max_pert_time = min(remaining_time() * 0.3, 120.0)
        best_h, best_c5 = local_perturb_refine(best_h, best_n, max_pert_time, rng)
        best_n = len(best_h)
        print("  After perturbation: n={}, C5={:.8f}".format(best_n, best_c5))

    # Phase 7: Final polish
    if remaining_time() > 15.0:
        print("\n--- Phase 7: Final Polish ---")
        global _G_N, _G_TGT, _G_DX, _G_BETA, _G_PEN
        _G_N = best_n
        _G_TGT = best_n / 2.0
        _G_DX = 2.0 / best_n
        _G_BETA = 10000.0
        _G_PEN = 20.0
        try:
            res = minimize(obj_func, best_h, method='L-BFGS-B', jac=obj_grad,
                           bounds=[(0.0, 1.0)] * best_n,
                           options={'maxiter': 500, 'gtol': 1e-12,
                                     'ftol': 1e-16, 'maxfun': 30000})
            h_new = project_feasible(res.x, best_n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass
        print("  After final polish: n={}, C5={:.8f}".format(best_n, best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
