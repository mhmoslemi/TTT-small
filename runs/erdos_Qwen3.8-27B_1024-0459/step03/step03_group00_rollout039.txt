def project_onto_simplex_box(h, target_sum):
    """Project h onto {x: 0 <= x[i] <= 1, sum(x) = target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    # Find interior points
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    cnt = np.sum(interior)
    if cnt > 0:
        h[interior] += diff / cnt
    else:
        h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

    if k > 0:
        # c_k = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        # d/dh[m] = (1-h[m-k])*1[m>=k] - h[m+k]*1[m<=n-1-k]
        # Middle region: k <= m <= n-1-k
        mid_s = k
        mid_e = n - 1 - k
        if mid_e >= mid_s:
            grad[mid_s:mid_e + 1] += (1.0 - h[mid_s:mid_e + 1 - k])
            grad[mid_s:mid_e + 1] -= h[mid_s + k:mid_e + 1 + k]
        # Upper: m in [n-k+1, n-1] (only first term, if n-k+1 <= n-1)
        u_s = n - k + 1
        if u_s <= n - 1 and u_s > mid_e:
            grad[u_s:] += (1.0 - h[u_s - k:])
        # Wait, let me redo. For m > n-1-k (i.e., m >= n-k), only first term applies (m >= k is satisfied)
        # First term: (1-h[m-k]) for m in [k, n-1]
        # Second term: -h[m+k] for m in [0, n-1-k]
        # So for m in [max(k, n-k+1), n-1]: only first term
        # Let me restructure:
        grad = np.zeros(n)
        # First term contribution: (1-h[m-k]) for m in [k, n-1]
        grad[k:] += (1.0 - h[:n - k])
        # Second term contribution: -h[m+k] for m in [0, n-1-k]
        grad[:n - k] -= h[k:2 * k]
        # Wait that's wrong for the index. h[m+k] for m in [0, n-1-k] gives h[k] to h[n-1]
        # So grad[0:n-k] -= h[k:n]
        # Let me redo from scratch
        grad = np.zeros(n)
        # For m in [k, n-1]: contributes (1-h[m-k])
        grad[k:] += 1.0 - h[0:n - k]
        # For m in [0, n-1-k]: contributes -h[m+k]
        grad[0:n - k] -= h[k:n]
        return grad
    else:
        # k < 0, L = -k
        # c_{-L} = sum_{i=0}^{n-1-L} h[i]*(1-h[i+L])
        # d/dh[p] = (1-h[p+L])*1[p<=n-1-L] - h[p-L]*1[p>=L]
        grad = np.zeros(n)
        # First term: (1-h[p+L]) for p in [0, n-1-L]
        grad[0:n - L] += 1.0 - h[L:n]
        # Second term: -h[p-L] for p in [L, n-1]
        grad[L:] -= h[0:n - L]
        return grad


def grad_max_lags(h, lag_indices, n, beta):
    """Gradient of smooth max over a set of lags"""
    corr = compute_corr_vec(h)
    vals = np.array([corr[lag + (n - 1)] for lag in lag_indices])
    vals_safe = np.clip(vals, -1e15, 1e15)
    m = np.max(vals_safe)
    shifted = vals_safe - m
    exps = np.exp(np.clip(beta * shifted, -500, 0))
    total = np.sum(exps)
    if total < 1e-300:
        total = 1e-300
    weights = exps / total

    grad = np.zeros(n)
    for j, lag in enumerate(lag_indices):
        if weights[j] < 1e-15:
            continue
        grad += weights[j] * grad_lag(h, lag, n)
    return grad


def smooth_obj(h, lag_indices, n, beta):
    dx = 2.0 / n
    corr = compute_corr_vec(h)
    vals = np.array([corr[lag + (n - 1)] for lag in lag_indices])
    m = np.max(vals)
    shifted = vals - m
    exps = np.exp(np.clip(beta * shifted, -500, 0))
    return (m + np.log(np.sum(exps)) / beta) * dx


def smooth_obj_full(h, n, beta):
    dx = 2.0 / n
    corr = compute_corr_vec(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 0))
    return (m + np.log(np.sum(exps)) / beta) * dx


def smooth_grad_full(h, n, beta):
    dx = 2.0 / n
    corr = compute_corr_vec(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 0))
    total = np.sum(exps)
    if total < 1e-300:
        total = 1e-300
    weights = exps / total
    # Only compute gradient for lags with significant weight
    grad = np.zeros(len(h))
    n_corr = len(corr)
    for i in range(n_corr):
        if weights[i] < 1e-10:
            continue
        lag = i - (n - 1)
        grad += weights[i] * grad_lag(h, lag, n)
    return grad * dx


def nesterov_optimize(h0, n, lag_indices, beta, lr, momentum, max_iter, time_limit):
    """Nesterov accelerated projected gradient descent"""
    target = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    h_prev = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    t0 = time.time()

    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        if _remaining() < 5.0:
            break

        # Nesterov lookahead
        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)

        # Gradient at y
        grad = grad_max_lags(y, lag_indices, n, beta) * dx

        # Update
        h_new = y - lr * grad
        h_new = project_feasible(h_new, n)

        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

        h_prev = h
        h = h_new

        # Update learning rate (simple decay)
        lr *= 0.9995

    return best_h, best_c5


def binary_local_search(h0, n, max_iters=5000, time_limit=60.0):
    """Local search on binary solutions"""
    target = n / 2.0
    # Round to binary
    h = np.zeros(n)
    sorted_idx = np.argsort(h0)[::-1]
    h[sorted_idx[:n // 2]] = 1.0
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_corr = compute_corr_vec(h).copy()
    t0 = time.time()
    rng = np.random.RandomState(12345)

    no_improve = 0
    max_no_improve = 200

    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        if _remaining() < 5.0:
            break
        if no_improve > max_no_improve:
            # Restart with perturbation
            flip_count = rng.randint(2, 6)
            h = best_h.copy()
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            for _ in range(flip_count):
                if len(ones) > 0 and len(zeros) > 0:
                    oi = rng.choice(ones)
                    zi = rng.choice(zeros)
                    h[oi] = 0.0
                    h[zi] = 1.0
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
            no_improve = 0
            continue

        # Try flipping a random 1 to 0 and a random 0 to 1
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        oi = rng.choice(ones)
        zi = rng.choice(zeros)

        h[oi] = 0.0
        h[zi] = 1.0

        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h.copy()
            no_improve = 0
        else:
            h[oi] = 1.0
            h[zi] = 0.0
            no_improve += 1

    return best_h, best_c5


def binary_local_search_line(h0, n, time_limit=30.0):
    """Local search: flip one bit at a time, scanning all positions"""
    target = n / 2.0
    h = np.zeros(n)
    sorted_idx = np.argsort(h0)[::-1]
    h[sorted_idx[:n // 2]] = 1.0

        if _remaining() < 5.0:
            break
        improved = False
        for i in range(n):
            if h[i] < 0.5:
                continue
            # Try flipping h[i] from 1 to 0
            # Need to flip some 0 to 1 to maintain sum
            # For now, just check: what if we flip i from 1 to 0?
            # We need to maintain sum, so also flip one 0 to 1
            # This is a 2-flip move. Let's just do single flips with sum check later.
            pass
        if not improved:
            break

    # Simpler approach: try all pairs (i,j) where h[i]=1, h[j]=0
    # Swap: h[i]=0, h[j]=1
    h = best_h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(20):
        if time.time() - t0 > time_limit:
            break
        if _remaining() < 5.0:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        any_imp = False
        for i in ones:
            if time.time() - t0 > time_limit:
                break
            for j in zeros:
                h[i] = 0.0

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return h


def make_golden_ratio(n):
    h = np.zeros(n)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    idx = (np.arange(n) * gr) % 1.0
    h[idx < 0.5] = 1.0
    return h
