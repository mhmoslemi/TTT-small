import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Exact correction spread over coordinates closest to 0.5
    diff = target - np.sum(h)
    if abs(diff) > 1e-15 and n > 0:
        order = np.argsort(np.abs(h - 0.5))
        for k in range(min(50, n)):
            h[order[k]] = np.clip(h[order[k]] + diff, 0.0, 1.0)
            diff = target - np.sum(h)
            if abs(diff) < 1e-15:
                break
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 2.0)

def smooth_max_grad(h, beta):
    """Compute smooth max objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    shifted = beta * (corr - max_corr)
    shifted = np.minimum(shifted, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    obj = (max_corr + np.log(sum_exp) / beta) * dx
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    n_corr = len(corr)
    for m in range(n_corr):
        w = weights[m]
        if w < 1e-14:
            continue
        offset = m - (n - 1)
        if offset >= 0:
            n_pairs = n - offset
            if n_pairs > 0:
                grad[:n_pairs] += w * dx * (1.0 - h[offset:offset + n_pairs])
                grad[offset:offset + n_pairs] -= w * dx * h[:n_pairs]
        else:
            abs_off = -offset
            n_pairs = n - abs_off
            if n_pairs > 0:
                grad[:n_pairs] += w * dx * (1.0 - h[abs_off:abs_off + n_pairs])
                grad[abs_off:abs_off + n_pairs] -= w * dx * h[:n_pairs]
    return obj, grad

def slsqp_smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def sum_constraint(x):
    return np.sum(x) - len(x) / 2.0

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def lbfqb_run(h, n, beta, maxiter=400):
    """L-BFGS-B with penalty for sum constraint."""
    penalty = 5000.0

    def obj(x):
        hc = np.clip(x, 0.01, 0.99)
        sm, gd = smooth_max_grad(hc, beta)
        p = penalty * (np.sum(hc) - n / 2.0) ** 2
        gp = 2.0 * penalty * (np.sum(hc) - n / 2.0)
        return sm + p, gd + np.full(n, gp)

    h0 = project_feasible(h, n)
    try:
        res = minimize(obj, h0, jac=True, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-18,
                               'gtol': 1e-12, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def slsqp_run(h, n, beta, maxiter=200):
    try:
        res = minimize(slsqp_smooth_obj, h, args=(beta,), method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': sum_constraint}],
                      options={'maxiter': maxiter, 'ftol': 1e-17, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
