import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)

def sa_binary(h, n, time_limit, rng, T0=3.0, alpha=0.9995, T_min=0.05):
    """Simulated annealing for binary optimization."""
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    ones_count = int(np.sum(h))
    if ones_count > n_half:
        ones_idx = np.where(h == 1)[0]
        to_remove = rng.choice(ones_idx, ones_count - n_half, replace=False)
        h[to_remove] = 0
    elif ones_count < n_half:
        zeros_idx = np.where(h == 0)[0]
        to_add = rng.choice(zeros_idx, n_half - ones_count, replace=False)
        h[to_add] = 1
    
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_energy = float(np.max(corr))
    best_energy = cur_energy
    best_h = h.copy()
    
    T = T0
    start = time.time()
    stagnant = 0
    
    while time.time() - start < time_limit:
        if remaining() < 5:
            break
        
        # Pick a 1 to flip to 0 and a 0 to flip to 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        if i == j:
            continue
        
        # Evaluate swap
        h[i] = 0.0
        h[j] = 1.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_energy = float(np.max(new_corr))
        delta = new_energy - cur_energy

            corr = new_corr

                best_energy = new_energy
                best_h = h.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            h[i] = 1.0
            h[j] = 0.0

        if T < T_min:
            T = T_min
        
        # Periodic restart
        if stagnant > 5000:
            h = best_h.copy()
            cur_energy = best_energy
            corr = np.correlate(h, 1.0 - h, mode='full')
            T = max(T * 2, 1.0)
            stagnant = 0
    
    return best_h, best_energy * dx

def project_feasible(h, n):

def ensure_count(h, n, rng):
    h = h.copy()

    cnt = int(np.sum(h > 0.5))

def soft_energy(corr, beta=5.0):
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -50, 50)
    ex = np.exp(cl)
    return m + np.log(np.sum(ex)) / beta

    # Phase 2: ILS on best
    if remaining() > 30:
        ils_t = min(300.0, max(20.0, remaining() * 0.4))
        h_ils, c5_ils = ils_search(best_h, best_n, ils_t, rng)
        if c5_ils < best_c5:
            best_c5 = c5_ils
            best_h = h_ils.copy()
            best_n = best_n

    # Phase 3: Try other resolutions
    n_candidates = sorted(set([
        best_n - 6, best_n - 4, best_n - 2,
        best_n + 2, best_n + 4, best_n + 6, best_n + 8,
        best_n + 10, best_n + 12, best_n + 14, best_n + 16,
        best_n + 20, best_n + 24, best_n + 30,
        best_n - 8, best_n - 10,
    ]))
    n_candidates = [nn for nn in n_candidates if 120 <= nn <= 600 and nn % 2 == 0 and nn != best_n]

    for n_t in n_candidates:
        if remaining() < 25:
            break
        h_interp = interp_res(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        h_b = ensure_count(h_feas, n_t, rng)

        t_alloc = min(50.0, max(10.0, remaining() * 0.2))
        h_sa2, c5_sa2 = sa_search(h_b, n_t, t_alloc, rng, T0=3.0, alpha=0.9997, T_min=0.04, beta=5.0)

        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
            best_n = n_t

        if remaining() > 15 and c5_sa2 < best_c5 + 0.005:
            h_g2, c5_g2 = greedy_polish(h_sa2, n_t, min(25, remaining() * 0.15), rng)
            if c5_g2 < best_c5:
                best_c5 = c5_g2
                best_h = h_g2.copy()
                best_n = n_t

    # Phase 4: Structural initialization + SA
    if remaining() > 30:
        for alt_seed in [seed + 1, seed + 7, seed + 42]:
            if remaining() < 15:
                break
            h_struct = make_structural(best_n, alt_seed)
            t_struct = min(60.0, max(10.0, remaining() * 0.2))
            h_sr, c5_sr = sa_search(h_struct, best_n, t_struct, rng, T0=5.0, alpha=0.9998, T_min=0.05, beta=4.0)
            if c5_sr < best_c5:
                best_c5 = c5_sr
                best_h = h_sr.copy()
                best_n = best_n

    # Phase 5: More ILS restarts on best
    if remaining() > 40:
        for _ in range(3):
            if remaining() < 15:
                break
            h_pert = best_h.copy()
            k_p = max(3, best_n // 20)
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            for _ in range(k_p):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i], h_pert[j] = h_pert[j], h_pert[i]
            t_il = min(80.0, max(10.0, remaining() * 0.3))
            h_il, c5_il = ils_search(h_pert, best_n, t_il, rng)
            if c5_il < best_c5:
                best_c5 = c5_il
                best_h = h_il.copy()
                best_n = best_n

    # Phase 6: Final SA polish with lower temperature
    if remaining() > 20:
        h_final_bin = ensure_count(best_h, best_n, rng)
        t_fin = min(80.0, max(10.0, remaining() * 0.4))
        h_sf, c5_sf = sa_search(h_final_bin, best_n, t_fin, rng, T0=1.0, alpha=0.9999, T_min=0.01, beta=10.0)
        if c5_sf < best_c5:
            best_c5 = c5_sf
            best_h = h_sf.copy()
            best_n = best_n

    # Phase 7: Final greedy
    if remaining() > 10:
        h_gp, c5_gp = greedy_polish(best_h, best_n, min(40, remaining() * 0.5), rng)
        if c5_gp < best_c5:
            best_c5 = c5_gp
            best_h = h_gp.copy()
            best_n = best_n

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
