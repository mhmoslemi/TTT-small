        # SLSQP with increasing beta
        for beta in [50, 150, 400, 1000]:
            if remaining() < 10:
                break
            n = res_n
            def obj(x):
                return smooth_max_h(x, beta)
            def con_fun(x):
                return np.sum(x) - n / 2.0
            try:
                res = minimize(obj, h_curr, method='SLSQP',
                              bounds=[(0.0, 1.0)] * n,
                              constraints=[{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}],
                              options={'maxiter': 300, 'ftol': 1e-16, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < c5_curr:
                    c5_curr = c5_new
                    h_curr = h_new.copy()
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n
            except:
                pass
    
    # Phase 2: L-BFGS-B refinement at best resolution
    n = best_n
    h_curr = best_h.copy()
    for beta in [200, 500, 1200, 3000]:
        if remaining() < 15:
            break
        def obj_lb(x):
            return smooth_max_h(x, beta)
        def con_fun(x):
            return np.sum(x) - n / 2.0
        try:
            res = minimize(obj_lb, h_curr, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}],
                          options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-14, 'maxfun': 10000})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_curr = h_new.copy()
        except:
            pass
    
    # Phase 3: Binary snap + local search
    h_bin = np.where(best_h > 0.5, 1.0, 0.0)
    h_bin = project_feasible(h_bin, n)
    c5_bin = compute_c5(h_bin)
    h_search = h_bin.copy() if c5_bin < best_c5 else best_h.copy()
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    
    # Single-element perturbation
    for sweep in range(10):
        if remaining() < 5:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if remaining() < 2:
                break
            for trial in [0.0, 1.0]:
                if abs(h_search[i] - trial) < 1e-8:
                    continue
                old = h_search[i]
                h_search[i] = trial
                h_search = project_feasible(h_search, n)
                c5_t = compute_c5(h_search)
                if c5_t < best_c5 - 1e-12:
                    best_c5 = c5_t
                    best_h = h_search.copy()
                    any_imp = True
                    break
                else:
                    h_search[i] = old
                    h_search = project_feasible(h_search, n)
        if not any_imp:
            break
    
    # Phase 4: Pair swaps
    if remaining() > 20:
        h_search = best_h.copy()
        for sweep in range(5):
            if remaining() < 5:
                break
            any_imp = False
            order = rng.permutation(n)
            for i in order:
                if remaining() < 2:
                    break
                for j in range(i + 1, min(i + 15, n)):
                    if abs(h_search[i] - h_search[j]) < 1e-8:
                        continue
                    h_search[i], h_search[j] = h_search[j], h_search[i]
                    h_search = project_feasible(h_search, n)
