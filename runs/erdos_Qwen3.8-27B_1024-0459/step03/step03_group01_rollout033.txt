def slsqp_smooth(h, n, beta, maxiter=400):
    dx_v = 2.0 / n
    tgt = n / 2.0

    def obj(x):
        c = compute_corr(x)
        mm = float(np.max(c))
        sh = c - mm
        cl = np.clip(beta * sh, -50, 50)
        ex = np.exp(cl)
        return (mm + np.log(np.sum(ex)) / beta) * dx_v

    def obj_grad(x):
        c = compute_corr(x)
        mm = float(np.max(c))
        sh = c - mm
        cl = np.clip(beta * sh, -50, 50)
        ex = np.exp(cl)
        w = ex / float(np.sum(ex))
        g = np.zeros(n)
        for k in range(len(c)):
            wi = w[k]
            if wi < 1e-14:
                continue
            s = k - (n - 1)
            if s >= 0:
                np_ = n - s
                g[s:s + np_] += wi * (1.0 - x[:np_])
                g[:np_] -= wi * x[s:s + np_]
            else:
                ao = -s
                np_ = n - ao
                g[:np_] += wi * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wi * x[:np_]
        return g * dx_v

    def sc(x):
        return np.sum(x) - tgt

    def sc_j(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc, 'jac': sc_j}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

                ao = -s
                np_ = n - ao
                g[:np_] += wi * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wi * x[:np_]
        g *= dx_v
        g += pen * (float(np.sum(x)) - tgt)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h


def perturb_binary(h, n, rng, strength=0.15):
    """Perturb a binary solution by flipping some bits."""
    h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    n_flip = max(1, int(n * strength))
    for _ in range(n_flip):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 320
        parent_h = np.zeros(parent_n)
        for i in range(parent_n):
            c = bin(i).count('1')
            parent_h[i] = 1.0 if c % 2 == 1 else 0.0
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h_bin = (best_h > 0.5).astype(float).copy()
    n0 = int(np.sum(h_bin))
    target_n0 = n // 2
    if n0 > target_n0:
        oi = np.where(h_bin == 1.0)[0]
        h_bin[oi[rng.choice(len(oi), n0 - target_n0, replace=False)]] = 0.0
    elif n0 < target_n0:
        zi = np.where(h_bin == 0.0)[0]
        h_bin[zi[rng.choice(len(zi), target_n0 - n0, replace=False)]] = 1.0

    corr_b = compute_corr(h_bin)
    c5_binary = float(np.max(corr_b)) * (2.0 / n)

    if c5_binary < best_c5:
        hc = project_feasible(h_bin, n)
        c5c = compute_c5(hc)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc.copy()

    tl = max(5.0, min(60.0, remaining() * 0.12))
    h_bin, c5_s = sweep_sweep_fast(h_bin, compute_corr(h_bin), 0, 0, n,
                                   max_steps=20000, time_limit=tl, rng=rng)
    if c5_s < best_c5:
        best_c5 = c5_s
        best_h = project_feasible(h_bin, n).copy()

    if remaining() > 8:
        tl2 = max(3.0, min(20.0, remaining() * 0.05))
        h_p, c5_p = pair_swap_search(h_bin, compute_corr(h_bin), n,
                                     max_steps=3000, time_limit=tl2, rng=rng)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = project_feasible(h_p, n).copy()

    h_con = project_feasible(best_h, n)

    for beta in [100, 400, 1200]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h_con, n, beta, maxiter=350)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_con = h_new.copy()

    if remaining() > 6:
        h_as, c5_as = active_set_linear(h_con, n, max_iter=40)
        if c5_as < best_c5:
            best_c5 = c5_as
            best_h = h_as.copy()

    if remaining() > 5:
        for beta in [800, 2500]:
            if remaining() < 6:
                break
            h_new = lgbf_smooth(best_h, n, beta, maxiter=150)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    n_cands = sorted(set([
        best_n - 2, best_n, best_n + 2, best_n + 4,
        best_n + 8, best_n + 16, best_n + 24, best_n + 32
    ]))
    n_cands = [x for x in n_cands if 160 <= x <= 480 and x % 2 == 0]

    for n_t in n_cands:
        if remaining() < 18:
            break
        if n_t == best_n:
            continue
        h_i = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_i, n_t)
        h_b_t = (h_feas > 0.5).astype(float)
        nt0 = int(np.sum(h_b_t))
        tt = n_t // 2
        if nt0 > tt:
            oi = np.where(h_b_t == 1.0)[0]
            h_b_t[oi[rng.choice(len(oi), nt0 - tt, replace=False)]] = 0.0
        elif nt0 < tt:
            zi = np.where(h_b_t == 0.0)[0]
            h_b_t[zi[rng.choice(len(zi), tt - nt0, replace=False)]] = 1.0

        tl_t = max(4.0, min(25.0, remaining() * 0.10))
        h_b_t, c5_bt = sweep_sweep_fast(h_b_t, compute_corr(h_b_t), 0, 0, n_t,
                                        max_steps=10000, time_limit=tl_t, rng=rng)

        h_c_t = project_feasible(h_b_t, n_t)
        c5_ct = compute_c5(h_c_t)

        for beta in [400, 1200]:
            if remaining() < 8:
                break
            h_c_t = slsqp_smooth(h_c_t, n_t, beta, maxiter=250)
            c5_ct = compute_c5(h_c_t)

        if remaining() > 5:
            h_c_t, c5_ct = active_set_linear(h_c_t, n_t, max_iter=30)

        if c5_ct < best_c5:
            best_c5 = c5_ct
            best_h = h_c_t.copy()
            best_n = n_t

    if remaining() > 25:
        n = best_n
        h = best_h.copy()
        h_b_f = (h > 0.5).astype(float)
        nfb = int(np.sum(h_b_f))
        tf = n // 2
        if nfb > tf:
            oi = np.where(h_b_f == 1.0)[0]
            h_b_f[oi[rng.choice(len(oi), nfb - tf, replace=False)]] = 0.0
        elif nfb < tf:
            zi = np.where(h_b_f == 0.0)[0]
            h_b_f[zi[rng.choice(len(zi), tf - nfb, replace=False)]] = 1.0

        rng2 = np.random.RandomState(seed + 100)
        for trial in range(3):
            if remaining() < 10:
                break
            h_pert = perturb_binary(h_b_f, n, rng2, strength=0.08 + 0.04 * trial)
            tl_f = max(3.0, min(10.0, remaining() * 0.15))
            h_pert, c5_pt = sweep_sweep_fast(h_pert, compute_corr(h_pert), 0, 0, n,
                                             max_steps=5000, time_limit=tl_f, rng=rng2)
            h_ct = project_feasible(h_pert, n)
            for beta in [600, 2000]:
                if remaining() < 5:
                    break
                h_ct = slsqp_smooth(h_ct, n, beta, maxiter=200)
            c5_ct = compute_c5(h_ct)
            if c5_ct < best_c5:
                best_c5 = c5_ct
                best_h = h_ct.copy()

    if remaining() > 6:
        for beta in [3000, 8000, 15000]:
            if remaining() < 4:
                break
            h_np = slsqp_smooth(best_h, best_n, beta, maxiter=500)
            c5_np = compute_c5(h_np)
            if c5_np < best_c5:
                best_c5 = c5_np
                best_h = h_np.copy()

    if remaining() > 4:
        h_as_f, c5_as_f = active_set_linear(best_h, best_n, max_iter=50)
        if c5_as_f < best_c5:
            best_c5 = c5_as_f
            best_h = h_as_f.copy()

    best_h = project_feasible(best_h, best_n)
    verify_c5 = compute_c5(best_h)
    if verify_c5 < best_c5:
        best_c5 = verify_c5
    else:
        best_c5 = min(best_c5, verify_c5)

    try:
        ev = evaluate_erdos_solution(best_h)
        if ev < best_c5:
            best_c5 = ev
    except Exception:
        pass

    return (best_h, best_c5, best_n)
