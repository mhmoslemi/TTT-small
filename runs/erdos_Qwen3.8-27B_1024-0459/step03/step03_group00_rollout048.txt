def compute_c5(h):

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    return m + np.log(np.sum(exps)) / beta

def smooth_obj_val(h, beta, n):
    corr = compute_corr(h)
    return smooth_max(corr, beta) * (2.0/n) + 50.0 * (np.sum(h) - n/2.0)**2

def smooth_obj_grad(h, beta, n):
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exponents = np.clip(beta * shifted, -500, 500)
    exps = np.exp(exponents)
    weights = exps / np.sum(exps)
    
    # Compute gradient of weighted sum of correlations
    grad = np.zeros(n)
    n_corr = len(corr)  # 2n-1
    
    for m_idx in range(n_corr):
        w = weights[m_idx]
        if w < 1e-15:

        offset = m_idx - (n - 1)
        if offset >= 0:
            n_terms = n - offset
            # d c[m]/dh[j] for offset >= 0:

            # -h[j+offset] for 0 <= j <= n-1-offset

            grad[0:n_terms] += w * (-h[offset:n])

            grad[o:n] += w * (-h[:n_terms])

def project_feasible(h, n):

    target = n / 2.0
    for _ in range(50):

    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def make_random_bin(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n//2]] = 1.0
    return h

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return h

def make_golden_ratio(n):
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return h

def coordinate_descent(h_best, n, budget_remain, rng, max_sweeps=20):
    c5_best = compute_c5(h_best)
    h = h_best.copy()
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):
        if budget_remain() < 3.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if budget_remain() < 1.0:
                break
            old_val = h[i]
            # Try 0 and 1
            for new_val in [0.0, 1.0]:
                if abs(new_val - old_val) < 1e-8:
                    continue
                h[i] = new_val
                h = project_feasible(h, n)
                c5_test = compute_c5(h)
                if c5_test < c5_best - 1e-12:
                    c5_best = c5_test
                    h_best = h.copy()
                    any_imp = True
                    old_val = new_val
                else:
                    h = project_feasible(h_best, n)
            # Try small perturbations
            for delta in [0.05, -0.05, 0.1, -0.1, 0.2, -0.2, 0.3, -0.3]:
                if budget_remain() < 1.0:
                    break
                trial = h_best.copy()
                trial[i] = np.clip(trial[i] + delta, 0.0, 1.0)
                trial = project_feasible(trial, n)
                c5_test = compute_c5(trial)
                if c5_test < c5_best - 1e-12:
                    c5_best = c5_test
                    h_best = trial.copy()
                    any_imp = True
        if not any_imp:
            break
    
    return h_best, c5_best

Hmm, this analytical gradient for the k-th order statistic is complex. Let me use a simpler approach: just use the softmax (log-sum-exp) gradient for the objective, which is what I had before, and rely on L-BFGS-B's numerical gradient as a fallback.

Actually, let me go back to the log-sum-exp approach which is simpler and well-understood:

f(h) = (max(corr) + (1/beta)*log(sum(exp(beta*(corr - max(corr)))))) * dx + penalty

The gradient is:
g[j] = dx * sum_i w[i] * (dc_i/dh_j) + penalty_grad

where w[i] = exp(beta*(corr[i]-max)) / sum(exp(...)) is the softmax weight.

For the correlation dc_i/dh_j:
As I derived earlier, for correlation index i with offset = i - (n-1):

If offset >= 0:
  dc_i/dh_j = (1-h[j-offset]) for offset <= j <= n-1, minus h[j+offset] for 0 <= j <= n-1-offset
  i.e., grad[j] += (1-h[j-offset]) - h[j+offset] (when both conditions hold)
  More precisely:
  For j in [offset, n-1]: grad[j] += (1-h[j-offset])
  For j in [0, n-1-offset]: grad[j] += -h[j+offset]

If offset < 0 (let o = -offset):
  dc_i/dh_j = (1-h[j+o]) for 0 <= j <= n-1-o, minus h[j-o] for o <= j <= n-1
  For j in [0, n-1-o]: grad[j] += (1-h[j+o])
  For j in [o, n-1]: grad[j] += -h[j-o]

def smooth_obj(h, beta, n):
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    s = np.log(np.sum(np.exp(np.clip(beta * shifted, -500, 500))))
    target = n / 2.0
    penalty = 200.0 * (np.sum(h) - target) ** 2
    return (m + s / beta) * dx + penalty

def smooth_grad(h, beta, n):
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    weights = exps / np.sum(exps)
    
    grad = np.zeros(n)
    
    for i in range(len(corr)):
        w = weights[i]
        if w < 1e-16:
            continue
        off = i - (n - 1)
        if off >= 0:
            nt = n - off
            # (1-h[j-off]) for j in [off, n-1]
            grad[off:] += w * (1.0 - h[:nt])
            # -h[j+off] for j in [0, nt-1]
            grad[:nt] += w * (-h[off:])
        else:
            o = -off
            nt = n - o
            # (1-h[j+o]) for j in [0, nt-1]
            grad[:nt] += w * (1.0 - h[o:])
            # -h[j-o] for j in [o, n-1]
            grad[o:] += w * (-h[:nt])
    
    # Penalty gradient
    target = n / 2.0
    grad += 400.0 * (np.sum(h) - target)
    
    return grad * dx
