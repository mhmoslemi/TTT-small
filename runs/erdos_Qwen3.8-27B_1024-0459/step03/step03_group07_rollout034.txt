        # Spread across multiple coordinates
        n_spread = max(1, int(np.ceil(abs(diff))))
        if diff > 0:
            idxs = np.argsort(1.0 - h)[:n_spread]
            for idx in idxs:
                amt = min(abs(diff) / len(idxs), 1.0 - h[idx])
                h[idx] += amt
                diff -= amt
        else:
            idxs = np.argsort(h)[:n_spread]  # smallest values, wait no, we want to decrease
            idxs = np.argsort(h)[::-1][:n_spread]  # largest values to decrease
            for idx in idxs:
                amt = min(abs(diff) / len(idxs), h[idx])
                h[idx] -= amt
                diff += amt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, n):

                        aff = True
                    if 0 <= j <= n - 1 - k:
                        a_c += (-h[j + k])
                        b_c += h[j + k]
                        aff = True
                    if k == 0:
                        is_q = True
                else:

                    if s <= j <= n - 1:
                        a_c += (1.0 - h[j - s])

                    if s <= j + s <= n - 1:
                        a_c += (-h[j + s])
                        b_c += h[j + s]
                        aff = True

def subgradient_topk(h, n, k=5, iters=500, lr0=0.01, time_limit=None, t0=None):
    """Subgradient descent on top-k average of correlations."""
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    lr = lr0
    patience = 0

        if time_limit and t0 and time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Get top-k indices
        k_eff = min(k, len(corr))
        top_k = np.sort(corr)[-k_eff:]
        threshold = top_k[0] if k_eff < len(corr) else corr.min()
        
        # Gradient: for each lag above threshold (or in top-k), contribute
        # d/dx_j [h[i]*(1-h[i+k])] for relevant i
        active = corr >= threshold - 1e-10
        w = 1.0 / np.sum(active)
        
        grad = np.zeros(n)
        for lidx in np.where(active)[0]:
            lag = lidx - (n - 1)
            if lag >= 0:
                # corr[lidx] = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
                # d/dh[j]: h[j] appears as h[i+lag] (i=j-lag) and (1-h[i]) (i=j)
                if 0 <= j := None:  # placeholder
                    pass
                # Vectorized:
                # Contribution to grad from this lag:
                # For h[j] as h[i+lag]: i = j-lag, need 0 <= j-lag <= n-1-lag
                #   d/dh[j] of h[j]*(1-h[j-lag]) = (1-h[j-lag])
                # For h[j] in (1-h[i]): i=j, need 0 <= j <= n-1-lag
                #   d/dh[j] of h[j+lag]*(1-h[j]) = -h[j+lag]
                if lag < n:
                    # h[j] as first factor: valid for j in [lag, n-1]
                    grad[lag:] += w * (1.0 - h[:n - lag])
                    # h[j] in (1-h[j]): valid for j in [0, n-1-lag]
                    grad[:n - lag] += w * (-h[lag:])
            else:
                s = -lag
                # corr for lag=-s: sum_{i=s}^{n-1} h[i]*(1-h[i-s])
                # d/dh[j]: h[j] as h[i] (i=j, need s<=j<=n-1): (1-h[j-s])
                #          h[j] in (1-h[i-s]) (i-s=j, i=j+s, need s<=j+s<=n-1): -h[j+s]

                    grad[:n - s] += w * (-h[s:])

                patience = 0

def smooth_beta_obj(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 0)
    logsumexp = m + np.log(np.sum(np.exp(shifted))) / beta
    return logsumexp * (2.0 / n)

def slsqp_polish(h, n, time_limit=None, t0=None):
    """SLSQP polish with increasing beta."""
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    
    def eq_fun(x):
        return np.sum(x) - n / 2.0
    
    for beta in [30.0, 100.0, 500.0, 2000.0]:
        if time_limit and t0 and time.time() - t0 > time_limit - 5:
            break
        
        def obj(x):
            return smooth_beta_obj(x, n, beta)
        
        def obj_grad(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(corr)
            shifted = np.clip(beta * (corr - m), -500, 0)
            weights = np.exp(shifted) / np.sum(np.exp(shifted))
            grad_corr = np.zeros(n)
            for lidx in range(len(corr)):
                lag = lidx - (n - 1)
                w = weights[lidx]
                if lag >= 0:
                    if lag < n:
                        grad_corr[lag:] += w * (1.0 - x[:n - lag])
                        grad_corr[:n - lag] += w * (-x[lag:])
                else:
                    s = -lag
                    if s < n:
                        grad_corr[s:] += w * (1.0 - x[:n - s])
                        grad_corr[:n - s] += w * (-x[s:])
            return grad_corr * (2.0 / n)
        
        cons = [{'type': 'eq', 'fun': eq_fun}]
        try:
            res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
