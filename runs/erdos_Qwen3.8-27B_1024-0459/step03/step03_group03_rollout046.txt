    s = float(np.sum(vals))
    if s < 1e-300:
        return mx * (2.0 / n)
    return (mx + np.log(s) / b) * (2.0 / n)


def _slsqp_full(h0, n, beta, maxiter):
    global _ng, _bv, _topk_idx
    _ng = n
    _bv = beta
    _topk_idx = None
    try:
        r = minimize(_sobj_full, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon}],
                     options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return _proj(np.asarray(r.x, dtype=np.float64), n)
    except Exception:
        return _proj(h0.copy(), n)


def _slsqp_topk(h0, n, beta, maxiter, topk_n):
    global _ng, _bv, _topk_idx
    _ng = n
    _bv = beta
    c = np.correlate(h0, 1.0 - h0, mode='full')
    nk = len(c)
    k = min(topk_n, nk)
    _topk_idx = np.argpartition(c, -k)[-k:]
    try:
        r = minimize(_sobj_topk, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon}],
                     options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return _proj(np.asarray(r.x, dtype=np.float64), n)
    except Exception:
        return _proj(h0.copy(), n)


def _ggrad(h, kmax):
    n = len(h)
    s = kmax - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
    else:
        p = -s
        if p > 0:
            g[:n - p] += 1.0 - h[p:n]
            g[p:] -= h[:n - p]
    return g


def _smgrad(h, bw):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(corr))
    exponents = np.clip(bw * (corr - mx), -700, 700)
    w = np.exp(exponents)
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
        sw = 1.0
    w /= sw
    g = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for k in nz:
        g += w[int(k)] * _ggrad(h, int(k))
    return g


def _sgloop(h, n, nit, rng, bws=None):
    if bws is None:
        bws = [30.0, 50.0, 80.0, 120.0, 500.0]
    best = h.copy()
    bc = _c5(best)
    step_list = [0.05, 0.02, 0.01, 0.005, 0.003, 0.001, 0.03, 0.015, 0.008, 0.002]
    bw_idx = 0
    stall = 0
    for it in range(nit):
        if _tl() < 3:
            break
        bw = bws[bw_idx % len(bws)]
        g = _smgrad(best, bw)
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            g = _smgrad(best, bw * 10)
            gn = float(np.linalg.norm(g))
            if gn < 1e-15:
                break
        g /= gn
        moved = False
        for a in step_list:
            h2 = _proj(best - a * g, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-14:
                best = h2
                bc = c2
                moved = True
                stall = 0
                break
        if not moved:
            stall += 1
            if stall >= 2:
                bw_idx += 1
                stall = 0
                corr = np.correlate(best, 1.0 - best, mode='full')
                kmax = int(np.argmax(corr))
                g2 = _ggrad(best, kmax)
                gn2 = float(np.linalg.norm(g2))
                if gn2 > 1e-15:
                    g2 /= gn2
                    for a in [0.04, 0.015, 0.008, 0.004, 0.002]:
                        h2 = _proj(best - a * g2, n)
                        c2 = _c5(h2)
                        if c2 < bc - 1e-14:
                            best = h2
                            bc = c2
                            moved = True
                            stall = 0
                            break
    return best, bc


def _pswap(h, n, nsweep, rng):
    best = h.copy()
    bc = _c5(best)
    for _ in range(nsweep):
        if _tl() < 3:
            break
        ones = np.where(best > 0.98)[0]
        zeros = np.where(best < 0.02)[0]
        if len(ones) >= 2 and len(zeros) >= 2:
            imp = False
            for _ in range(150):
                if _tl() < 2:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h2 = best.copy()
                tmp = h2[i]
                h2[i] = h2[j]
                h2[j] = tmp
                h2 = _proj(h2, n)
                c2 = _c5(h2)
                if c2 < bc - 1e-14:
                    best = h2
                    bc = c2
                    imp = True
                    ones = np.where(best > 0.98)[0]
                    zeros = np.where(best < 0.02)[0]
            if imp:
                continue
            # block moves (swap 2 adjacent ones with 2 adjacent zeros)
            if len(ones) >= 2 and len(zeros) >= 2:
                for _ in range(40):
                    if _tl() < 2:
                        break
                    i1 = ones[rng.randint(len(ones))]
                    i2 = i1 + 1
                    if i2 >= n:
                        i2 = i1 - 1
                    if i2 < 0 or i2 == i1:
                        continue
                    j1 = zeros[rng.randint(len(zeros))]
                    j2 = j1 + 1
                    if j2 >= n:
                        j2 = j1 - 1
                    if j2 < 0 or j2 == j1:
                        continue
                    h2 = best.copy()
                    t1, t2 = h2[i1], h2[i2]
                    h2[i1] = h2[j1]
                    h2[i2] = h2[j2]
                    h2[j1] = t1
                    h2[j2] = t2
                    h2 = _proj(h2, n)
                    c2 = _c5(h2)
                    if c2 < bc - 1e-14:
                        best = h2
                        bc = c2
                        imp = True
                        ones = np.where(best > 0.98)[0]
                        zeros = np.where(best < 0.02)[0]
        else:
            hi = np.where(best > 0.6)[0]
            lo = np.where(best < 0.4)[0]
            if len(hi) < 2 or len(lo) < 2:
                break
            imp = False
            for _ in range(100):
                if _tl() < 2:
                    break
                ii = hi[rng.randint(len(hi))]
                jj = lo[rng.randint(len(lo))]
                h2 = best.copy()
                tmp = h2[ii]
                h2[ii] = h2[jj]
                h2[jj] = tmp
                h2 = _proj(h2, n)
                c2 = _c5(h2)
                if c2 < bc - 1e-14:
                    best = h2
                    bc = c2
                    imp = True
                    hi = np.where(best > 0.6)[0]
                    lo = np.where(best < 0.4)[0]
            if not imp:
                break
    return best, bc


def _squeeze(h, n, nit, rng):
    best = h.copy()
    bc = _c5(best)
    for _ in range(nit):
        if _tl() < 3:
            break
        mid = np.where((best > 0.08) & (best < 0.92))[0]
        if len(mid) < 2:
            break
        imp = False
        for _ in range(60):
            if _tl() < 2:
                break
            idx = mid[rng.randint(len(mid))]
            old = best[idx]
            target = 0.0 if old < 0.5 else 1.0
            delta = target - old
            comp = mid[rng.randint(len(mid))]
            if comp == idx:
                comp = (comp + 1) % n
            h2 = best.copy()
            h2[idx] = target
            h2[comp] = h2[comp] - delta
            h2 = _proj(h2, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-14:
                best = h2
                bc = c2
                imp = True
        if not imp:
            break
    return best, bc


def _cdrefine(h, n, nit, rng):
    best = h.copy()
    bc = _c5(best)
    for _ in range(nit):
        if _tl() < 3:
            break
        nb = np.where((best > 0.02) & (best < 0.98))[0]
        if len(nb) < 2:
            break
        imp = False
        rng.shuffle(nb)
        for idx in nb:
            if _tl() < 2:
                break
            old = best[idx]
            for delta in [0.008, -0.008, 0.015, -0.015, 0.004, -0.004, 0.002, -0.002, 0.02, -0.02]:
                nv = old + delta
                if nv < 0.0 or nv > 1.0:
                    continue
                comp = nb[rng.randint(len(nb))]
                if comp == idx:
                    comp = (comp + 1) % len(nb)
                    comp = nb[comp]
                h2 = best.copy()
                h2[idx] = nv
                h2[comp] += -delta
                h2 = _proj(h2, n)
                c2 = _c5(h2)
                if c2 < bc - 1e-14:
                    best = h2
                    bc = c2
                    imp = True
                    break
            if imp:
                imp = False
    return best, bc


def _full_opt(h0, n, rng):
    h = _proj(h0.copy(), n)
    c5 = _c5(h)

    # Phase A: SLSQP with increasing beta
    for beta in [80, 300, 1000]:
        if _tl() < 20:
            break
        h2 = _slsqp_full(h, n, beta, 300)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2

    # Phase B: Subgradient descent
    h, c5 = _sgloop(h, n, 500, rng)

    # Phase C: Swaps
    h, c5 = _pswap(h, n, 60, rng)

    # Phase D: Top-k SLSQP
    if _tl() > 30:
        for beta in [2000, 5000]:
            if _tl() < 20:
                break
            h2 = _slsqp_topk(h, n, beta, 300, 8)
            c2 = _c5(h2)
            if c2 < c5:
                h, c5 = h2, c2
        h, c5 = _sgloop(h, n, 300, rng)
        h, c5 = _pswap(h, n, 40, rng)

    # Phase E: Squeeze
    if _tl() > 20:
        h, c5 = _squeeze(h, n, 100, rng)
        h, c5 = _sgloop(h, n, 250, rng)

    # Phase F: High-beta SLSQP
    if _tl() > 25:
        for beta in [8000, 15000]:
            if _tl() < 20:
                break
            h2 = _slsqp_full(h, n, beta, 400)
            c2 = _c5(h2)
            if c2 < c5:
                h, c5 = h2, c2
        h, c5 = _sgloop(h, n, 300, rng)
        h, c5 = _pswap(h, n, 30, rng)

    # Phase G: Coordinate descent + final squeeze
    if _tl() > 15:
        h, c5 = _cdrefine(h, n, 150, rng)
        h, c5 = _squeeze(h, n, 80, rng)
        h, c5 = _sgloop(h, n, 200, rng)

    # Phase H: Ultra high beta
    if _tl() > 30:
        h2 = _slsqp_full(h, n, 30000, 500)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2
        h, c5 = _sgloop(h, n, 200, rng)
        h, c5 = _pswap(h, n, 20, rng)

    return h, c5


def _light_opt(h0, n, rng):
    h = _proj(h0.copy(), n)
    c5 = _c5(h)

    for beta in [200, 800]:
        if _tl() < 20:
            break
        h2 = _slsqp_full(h, n, beta, 200)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2

    h, c5 = _sgloop(h, n, 400, rng)
    h, c5 = _pswap(h, n, 50, rng)

    if _tl() > 20:
        h2 = _slsqp_topk(h, n, 3000, 250, 10)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2
        h, c5 = _sgloop(h, n, 300, rng)
        h, c5 = _pswap(h, n, 30, rng)

    if _tl() > 15:
        h, c5 = _squeeze(h, n, 60, rng)
        h, c5 = _sgloop(h, n, 200, rng)

    if _tl() > 20:
        h2 = _slsqp_full(h, n, 10000, 350)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2
        h, c5 = _sgloop(h, n, 150, rng)
        h, c5 = _pswap(h, n, 20, rng)

    return h, c5


def run(seed=42, budget_s=1300, **kwargs):
    global _t0, _tb
    _t0 = time.time()
    _tb = budget_s
    rng = np.random.RandomState(seed)

    try:
        hpar = np.array(initial_h_values, dtype=np.float64)
        npar = len(hpar)
    except Exception:
        npar = 160
        hpar = np.zeros(npar, dtype=np.float64)
        hpar[:npar // 2] = 1.0

    bh = _proj(hpar, npar)
    bc5 = _c5(bh)
    bn = npar
    print("Init n={} C5={:.8f}".format(bn, bc5))

    # Primary optimization at parent resolution
    h, c5 = _full_opt(bh, npar, rng)
    if c5 < bc5:
        bh, bc5, bn = h, c5, npar
        print("Optimized n={} C5={:.8f}".format(npar, c5))
    else:
        bh, bn = bh, npar
        print("No improvement at n={}, C5={:.8f}".format(npar, bc5))

    # Try higher resolutions
    resolutions = []
    for nn in [npar + 20, npar + 40, npar + 80, npar + 20, npar - 4]:
        if nn >= 40 and nn % 2 == 0 and nn > bn:
            resolutions.append(nn)
    # Also try nearby
    for nn in [npar + 4, npar - 4]:
        if nn >= 40 and nn % 2 == 0:
            resolutions.append(nn)
    # Deduplicate and sort
    resolutions = sorted(set([r for r in resolutions if r >= 40]), key=lambda x: abs(x - npar))

    for nn in resolutions:
        if _tl() < 60:
            break
        if nn == bn:
            continue
        hi = _proj(_interp_f(bh, bn, nn), nn)
        h2, c2 = _light_opt(hi, nn, rng)
        if c2 < bc5:
            bh, bc5, bn = h2, c2, nn
            print("Zoom n={} C5={:.8f}".format(nn, c2))

    # Final resolution attempts
    for nn in [200, 220, 240, 260]:
        if _tl() < 50:
            break
        if nn == bn or nn < 40 or nn % 2 != 0:
            continue
        if nn > bn + 100:
            continue
        hi = _proj(_interp_f(bh, bn, nn), nn)
        h2, c2 = _light_opt(hi, nn, rng)
        if c2 < bc5:
            bh, bc5, bn = h2, c2, nn
            print("Try n={} C5={:.8f}".format(nn, c2))

    # Final polish at best resolution
    if _tl() > 30:
        h, c5 = _sgloop(bh, bn, 300, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _pswap(bh, bn, 30, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _squeeze(bh, bn, 50, rng)
        if c5 < bc5:
            bh, bc5 = h, c5

    if _tl() > 20:
        h2 = _slsqp_full(bh, bn, 20000, 400)
        c2 = _c5(h2)
        if c2 < bc5:
            bh, bc5 = h2, c2
        h, c5 = _sgloop(bh, bn, 200, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _pswap(bh, bn, 20, rng)
        if c5 < bc5:
            bh, bc5 = h, c5

    if _tl() > 10:
        h, c5 = _cdrefine(bh, bn, 100, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _squeeze(bh, bn, 40, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _sgloop(bh, bn, 100, rng)
        if c5 < bc5:
            bh, bc5 = h, c5

    # Final verification
    bh = _proj(bh, bn)
    fc5 = _c5(bh)
    try:
        ec = evaluate_erdos_solution(bh)
        if ec is not None and ec < fc5:
            fc5 = ec
    except Exception:
        pass
    print("FINAL n={} C5={:.8f}".format(bn, fc5))
    return (bh, fc5, bn)
