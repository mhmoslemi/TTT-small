def _c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / n)

def _corrv(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj(h, n):
    h = np.array(h, dtype=np.float64)
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(60):
        d = 0.5 * n - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        k = int(np.sum(m))
        if k > 0:
            h[m] += d / k
        else:
            h[:] = 0.5
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _interp(h, ns, nd):
    if ns == nd:
        return h.copy()
    xs = (np.arange(ns) + 0.5) * (2.0 / ns)
    xd = (np.arange(nd) + 0.5) * (2.0 / nd)
    return np.interp(xd, xs, h)

def _ggrad(h, k):
    n = len(h)
    g = np.zeros(n)
    s = k - (n - 1)
    if s >= 0:
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
    else:
        p = -s
        if p > 0:
            g[:n - p] += 1.0 - h[p:n]
            g[p:] -= h[:n - p]
    return g

def _topk_grad(h, top_k, beta):
    n = len(h)
    corr = _corrv(h)
    top_idx = np.argsort(corr)[::-1][:top_k]
    top_vals = corr[top_idx]
    mx = top_vals[0]
    w = np.exp(beta * (top_vals - mx))
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros(len(top_vals))
        w[0] = 1.0
        sw = 1.0
    w /= sw
    g = np.zeros(n)
    for i in range(len(top_idx)):
        g += w[i] * _ggrad(h, int(top_idx[i]))
    return g

def _full_smgrad(h, bw):
    n = len(h)
    corr = _corrv(h)

        w[int(np.argmax(corr))] = 1.0
        sw = 1.0
    w /= sw
    g = np.zeros(n)
    nz = np.where(w > 1e-8)[0]
    for k in nz:
        g += w[int(k)] * _ggrad(h, int(k))
    return g

        c = _corrv(x)
