        h0 = project_feasible(interp_to(best_h, best_n, n), n)
        c5_0 = compute_c5(h0)

        for beta in [100, 500]:
            if remain() < 8.0:
                break
            _G_BETA = float(beta)
            _G_TOPK = min(15, n)

            def obj_fn(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                # Top-k log-sum-exp
                topk_idx = np.argpartition(corr, -_G_TOPK)][-__G_TOPK:] if False else np.argpartition(corr, -_G_TOPK)][-_:G_TOPK:]
                topk_vals = corr[np.argpartition(corr, -_G_TOPK)][-_G_TOPK:]
                m = np.max(topk_vals)
                s = np.sum(np.exp(_G_BETA * (topk_vals - m)))
                return (m + np.log(s) / _G_BETA) * _G_DX

            try:
                res = minimize(obj_fn, h0, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 200, 'ftol': 1e-13, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < c5_0:
                    h0 = h_new
                    c5_0 = c5_new
            except Exception:
                pass

        if c5_0 < best_c5 - 1e-10:
            best_c5 = c5_0
            best_h = h0.copy()
            best_n = n
            print("Phase1: n={}, C5={:.10f}".format(n, best_c5))

    # === Phase 2: Focused top-k optimization ===
    n = best_n
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - _G_TGT}]

    print("\nPhase 2: Focused optimization at n={}".format(n))

    for beta in [200, 400, 800, 1500, 3000, 6000]:
        if remain() < 12.0:
            break
        _G_BETA = float(beta)
        _G_TOPK = min(8, n)

        def obj_topk(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            topk_idx = np.argpartition(corr, -_G_TOPK)][-_:G_TOPK:] if False else np.argpartition(corr, -_G_TOPK)][-_:G_TOPK:]
            topk_vals = corr[np.argpartition(corr, -_G_TOPK)][-_G_TOPK:]
            m = np.max(topk_vals)
            s = np.sum(np.exp(_G_BETA * (topk_vals - m)))
            return (m + np.log(s) / _G_BETA) * _G_DX

        try:
            res = minimize(obj_topk, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 400, 'ftol': 1e-15, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new
                print("  beta={}, C5={:.10f}".format(beta, best_c5))
        except Exception:
            pass

    # === Phase 3: Efficient local search ===
    print("\nPhase 3: Local search at n={}".format(n))
    best_h, best_c5 = do_local_search(best_h, best_n, remain, rng)

    # === Phase 4: Try higher resolution ===
    for n_hi in [2 * best_n, int(best_n * 1.5)]:
        if remain() < 20.0 or n_hi > 768:
            break
        _G_N = n_hi
        _G_TGT = n_hi / 2.0
        _G_DX = 2.0 / n_hi
        bounds_hi = [(0.0, 1.0)] * n_hi
        cons_hi = [{'type': 'eq', 'fun': lambda x, t=_G_TGT: np.sum(x) - t}]

        h_hi = project_feasible(interp_to(best_h, best_n, n_hi), n_hi)
        c5_hi = compute_c5(h_hi)

        _G_BETA = 500.0
        _G_TOPK = min(10, n_hi)

        def obj_hi(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            topk_idx = np.argpartition(corr, -_G_TOPK)][-_G_TOPK:] if False else np.argpartition(corr, -_G_TOPK)][-_G_TOPK:]
            topk_vals = corr[np.argpartition(corr, -_G_TOPK)][-_G_TOPK:]
            m = np.max(topk_vals)
            s = np.sum(np.exp(_G_BETA * (topk_vals - m)))
            return (m + np.log(s) / _G_BETA) * _G_DX

        try:
            res = minimize(obj_hi, h_hi, method='SLSQP', bounds=bounds_hi,
                           constraints=cons_hi, options={'maxiter': 200, 'ftol': 1e-14, 'iprint': 0})
            h_hi_new = project_feasible(res.x, n_hi)
            c5_hi_new = compute_c5(h_hi_new)
            if c5_hi_new < best_c5 - 1e-10:
                best_c5 = c5_hi_new
                best_h = h_hi_new
                best_n = n_hi
                print("Phase4: n={}, C5={:.10f}".format(n_hi, best_c5))
            else:
                # Try local search at higher res
                h_hi_new, c5_hi_new = do_local_search(h_hi_new, n_hi, remain, rng)
                if c5_hi_new < best_c5 - 1e-10:
                    best_c5 = c5_hi_new
                    best_h = h_hi_new
                    best_n = n_hi
                    print("Phase4 LS: n={}, C5={:.10f}".format(n_hi, best_c5))
        except Exception:
            pass

    # === Phase 5: Final refinement ===
    n = best_n
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - _G_TGT}]

    for beta in [5000, 10000, 20000]:
        if remain() < 8.0:
            break
        _G_BETA = float(beta)
        _G_TOPK = min(5, n)

        def obj_final(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            topk_idx = np.argpartition(corr, -_G_TOPK)][-_G_TOPK:] if False else np.argpartition(corr, -_G_TOPK)][-_G_TOPK:]
            topk_vals = corr[np.argpartition(corr, -_G_TOPK)][-_G_TOPK:]
            m = np.max(topk_vals)
            s = np.sum(np.exp(_G_BETA * (topk_vals - m)))
            return (m + np.log(s) / _G_BETA) * _G_DX

        try:
            res = minimize(obj_final, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 300, 'ftol': 1e-16, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass

    # Final local search
    best_h, best_c5 = do_local_search(best_h, best_n, remain, rng)

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)


def do_local_search(h_best, n, remain, rng):
    c5_best = compute_c5(h_best)
    h = h_best.copy()
    dx = 2.0 / n

    # Precompute correlation
    h1 = 1.0 - h
    corr = np.correlate(h, h1, mode='full')
    max_corr = np.max(corr)

    # Single-variable local search with O(n) update per trial
    for sweep in range(60):
        if remain() < 3.0:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if remain() < 2.0:
                break
            old_val = h[i]
            # Try several candidate values
            candidates = []
            for dv in [0.0, 1.0]:
                if abs(dv - old_val) > 1e-8:
                    candidates.append(dv)
            for dv in [-0.3, -0.15, -0.05, 0.05, 0.15, 0.3]:
                nv = old_val + dv
                if 0.0 <= nv <= 1.0 and abs(nv - old_val) > 1e-8:
                    candidates.append(nv)

            best_trial = None
            best_new_c5 = c5_best

            for nv in candidates:
                # Check sum constraint after change
                # We'll project, but for speed, try only if sum stays feasible
                delta = nv - old_val
                # After change, sum changes by delta. We need sum = n/2.
                # So we need to compensate elsewhere. For simplicity, skip if |delta| large.
                if abs(delta) > 0.4:
                    continue

                # Quick feasibility: sum must remain n/2
                # We'll do a small projection: adjust one other variable
                # For efficiency, just check if the max corr would decrease (optimistic)

                # Compute change in correlation
                new_c = corr.copy()
                # h[i] changes by delta
                # For lag k (0-indexed from 0 to 2n-2 in full mode):
                # corr_full[tau] where lag = tau - (n-1)
                # delta effect:
                #   When j = i in sum: h[i] * (1-h[i+lag]) -> add delta*(1-h[i+lag]) if valid
                #   When j+lag = i: h[j] * (1-h[i]) -> add h[j]*(-delta) = -delta*h[j] if valid

                # For lag in range [-(n-1), n-1]:
                # corr_full[n-1+lag] gets:
                #   delta * (1-h[i+lag]) if 0 <= i+lag < n
                #   -delta * h[i-lag] if 0 <= i-lag < n

                changed = False
                for lag in range(-(n - 1), n):
                    tau = n - 1 + lag
                    d_c = 0.0
                    if 0 <= i + lag < n:
                        d_c += delta * (1.0 - h[i + lag])
                    if 0 <= i - lag < n:
                        d_c -= delta * h[i - lag]
                    if d_c != 0.0:
                        new_c[tau] += d_c

                if np.any(new_c < -1e-12):
                    # Shouldn't happen but skip
                    continue

                new_max = np.max(new_c)
                new_c5_val = new_max * dx

                if new_c5_val < best_new_c5 - 1e-12:
                    best_new_c5 = new_c5_val
                    best_trial = (nv, new_c.copy())

            if best_trial is not None:
                nv, new_corr = best_trial
                # Update h and reproject
                delta = nv - old_val
                h[i] = nv
                # Project to maintain sum
                sum_diff = np.sum(h) - n / 2.0
                if abs(sum_diff) > 1e-10:
                    # Compensate: find a variable to adjust
                    mask = (h > 1e-6) & (h < 1.0 - 1e-6) & (np.arange(n) != i)
                    if np.sum(mask) > 0:
                        h[mask] -= sum_diff / np.sum(mask)
                    h = np.clip(h, 0.0, 1.0)
                    # Recompute corr after projection
                    h = project_feasible(h, n)
                    corr = np.correlate(h, 1.0 - h, mode='full')
                else:
                    corr = new_corr

                c5_best = best_new_c5
                h_best = h.copy()
                improved = True

        if not improved and sweep > 5:
            break

    # Pair-swap local search
    if remain() > 5.0:
        for sweep in range(8):
            if remain() < 3.0:
                break
            improved = False
            # Random pairs, focus on nearby pairs
            n_pairs = min(500, n * n // 8)
            for _ in range(n_pairs):
                if remain() < 2.0:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                ai, aj = h[i], h[j]
                if abs(ai - aj) < 1e-8:
                    continue
                # Try swap
                h[i], h[j] = aj, ai
                h_check = project_feasible(h, n)
                c5_check = compute_c5(h_check)
                if c5_check < c5_best - 1e-11:
                    c5_best = c5_check
                    h_best = h_check.copy()
                    improved = True
                else:
                    h = h_best.copy()

            if not improved and sweep > 2:
                break

    return h_best, c5_best
