    m = k - (n - 1)  # actual shift

    else:

        rp = n - mp
        grad[mp:] += (1.0 - h[:rp])

    return grad

        for dn in range(-30, 31, 5):
            if time_left(t0, budget) < 4:
                break
            nt = best_n + dn
            if nt < 80 or nt > 500:
                continue
            ht = interp_h(best_h, best_n, nt)
            ht = project_h(ht, nt)
            ct, _ = compute_c5(ht)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()
                best_n = nt
                print("  Final res n={}: C5={:.8f}".format(nt, ct))
    
    # Final projection and evaluation
    best_h = project_h(best_h, best_n)
    final_c5, _ = compute_c5(best_h)
    
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
