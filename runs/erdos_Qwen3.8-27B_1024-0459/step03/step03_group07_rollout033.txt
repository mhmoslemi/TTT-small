    ones = np.where(h > 0.5)[0]

    best_h = h.copy()

    T = 0.1  # temperature
    T_min = 0.001
    
    for it in range(max_iters):
        if t0 is not None and time.time() - t0 > time_limit:
            break
        # Pick a random 1 and a random 0, swap them
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        c5_new = compute_c5(h)
        
        if c5_new < best_c5:
            best_c5 = c5_new

            if np.random.rand() < np.exp(-(c5_new - best_c5) / T):
                ones[ones == i] = j
                zeros[zeros == j] = i
            else:
                h[i] = 1
                h[j] = 0
        
        # Cool down
        T = max(T_min, T * 0.9999)
    
    return best_h, best_c5
