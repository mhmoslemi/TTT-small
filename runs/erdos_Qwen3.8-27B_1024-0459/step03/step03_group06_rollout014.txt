def interp_to(src_h, src_n, dst_n):

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h


def make_golden_ratio(n):
    h = np.zeros(n)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return h


def make_van_der_corput(n):
    h = np.zeros(n)
    for i in range(n):
        val = i + 1
        rev = 0
        for _ in range(32):
            rev = (rev << 1) | (val & 1)
            val >>= 1
        h[i] = 1.0 if (rev % 2) == 0 else 0.0
    return h


def make_sine(n, freq, phase=0.0):
    t = np.arange(n) / float(n)
    h = 0.5 + 0.35 * np.sin(2.0 * np.pi * freq * t + phase)
    return np.clip(h, 0.0, 1.0)


def make_multi_sine(n, freqs, phases, rng):
    t = np.arange(n) / float(n)
    h = np.full(n, 0.5)
    for f, ph in zip(freqs, phases):
        amp = rng.uniform(0.08, 0.2)
        h += amp * np.sin(2.0 * np.pi * f * t + ph)
    return np.clip(h, 0.0, 1.0)

def projected_gradient_step(h, n, k, beta, lr, n_iter=500):
    """Projected gradient on top-k smooth max."""
    h_work = h.copy()
    best_h = h_work.copy()
    best_c5 = compute_c5(h_work)
    
    for it in range(n_iter):
        if remaining_time() < 2:
            break
        corr = np.correlate(h_work, 1.0 - h_work, mode='full')
        top_idx = np.argsort(corr)[-k:]
        tv = corr[top_idx]
        m = np.max(tv)
        shifted = beta * (tv - m)
        max_sh = np.max(shifted)
        weights = np.exp(shifted - max_sh)
        weights /= np.sum(weights)

        grad = np.zeros(n)
        for idx, j in enumerate(top_idx):
            wj = weights[idx]
            if wj < 1e-12:
                continue
            s = j - (n - 1)
            if s >= 0:
                if s == 0:
                    grad += wj * (1.0 - 2.0 * h_work)
                else:
                    grad[s:] += wj * (1.0 - h_work[:n - s])
                    grad[:n - s] -= wj * h_work[s:]
            else:
                sp = -s
                if sp == 0:
                    grad += wj * (1.0 - 2.0 * h_work)
                else:
                    grad[:n - sp] += wj * (1.0 - h_work[sp:])
                    grad[sp:] -= wj * h_work[:n - sp]

        grad *= 2.0 / n
        grad += 2.0 * 500.0 * (np.sum(h_work) - n / 2.0)

        h_trial = h_work - lr * grad
        h_trial = project_feasible(h_trial, n)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
        h_work = h_trial

        # Adaptive learning rate
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

    return best_h, best_c5

                                       options={'maxiter': 200, 'ftol': 1e-14,
                                                'gtol': 1e-9, 'iprint': 0})
                        h_new = project_feasible(res.x, n)
                        c5_new = compute_c5(h_new)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_new.copy()
                            best_n = n
                            print("  Polished: C5={:.6f}".format(c5_new))
                    except Exception:
                        pass

    # Phase 2: Refined optimization at best resolution
    n = best_n
    _N = n
    _TGT = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h = best_h.copy()
    print("\n=== Final refinement at n={} ===".format(n))

    # Iterative top-k gradient with annealed beta
    if remaining_time() > 30:
        for k_val in [6, 8, 12, 16, 20]:
            if remaining_time() < 10:
                break
            for beta_val in [200, 400, 600]:
                if remaining_time() < 8:
                    break
                _TOP_K = k_val
                _BETA = beta_val
                h_pgd, c5_pgd = projected_gradient_step(h, n, k=k_val, beta=beta_val,
                                                        lr=0.05, n_iter=200)
                if c5_pgd < best_c5:
                    best_c5 = c5_pgd
                    best_h = h_pgd.copy()
                    h = h_pgd.copy()
                    print("  PGD(k={}, beta={}): C5={:.6f}".format(k_val, beta_val, c5_pgd))

    # Final L-BFGS polish
    if remaining_time() > 10:
        for beta_val in [300, 500, 800]:
            if remaining_time() < 6:
                break
            _BETA = beta_val
            _TOP_K = 10
            try:
                res = minimize(topk_smooth_obj, h, jac=topk_smooth_grad,
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 400, 'ftol': 1e-15,
                                        'gtol': 1e-10, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                    print("  Final L-BFGS (beta={}): C5={:.6f}".format(beta_val, c5_new))
            except Exception:
                pass

    # Coordinate descent polish
    if remaining_time() > 15:
        h_cd = best_h.copy()
        c5_cd = best_c5
        for sweep in range(8):
            if remaining_time() < 5:
                break
            order = rng.permutation(n)
            any_imp = False
            for i in order:
                if remaining_time() < 3:
                    break
                old = h_cd[i]
                best_local = c5_cd
                best_val = old
                candidates = [0.0, 1.0, 0.5, max(0.0, old - 0.15), min(1.0, old + 0.15)]
                for nv in candidates:
                    if abs(nv - old) < 1e-8:
                        continue
                    h_cd[i] = nv
                    h_cd = project_feasible(h_cd, n)
                    c5_test = compute_c5(h_cd)
                    if c5_test < best_local:
                        best_local = c5_test
                        best_val = nv
                        h_cd[i] = nv
                        h_cd = project_feasible(h_cd, n)
                    else:
                        h_cd[i] = old
                        h_cd = project_feasible(h_cd, n)
                if best_local < c5_cd:
                    c5_cd = best_local
                    any_imp = True
                else:
                    h_cd[i] = old
                    h_cd = project_feasible(h_cd, n)
            if not any_imp:
                break
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                print("  Coord descent sweep {}: C5={:.6f}".format(sweep, c5_cd))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
