    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g

def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    s = np.sum(ex)
    if s < 1e-300:
        # Use only the max
        k_max = int(np.argmax(corr))
        return corr_grad_per_shift(x, n, k_max) * dx_val
    w = ex / s
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-14:
            g += w[k] * corr_grad_per_shift(x, n, k)
    return g * dx_val

def sum_con(x):
    return np.sum(x) - _TGT

def sum_con_jac(x):
    return np.ones(_N)

def slsqp_smooth(h, n, beta, maxiter=300):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_local_search(h, n, rng, time_limit=30, max_stall=200):
    """Efficient binary local search with greedy + targeted swaps."""
    start = time.time()
    h = h.copy()
    corr = compute_corr(h)
    cur_max = float(corr.max())
    best_h = h.copy()
    best_max = cur_max
    stall = 0
    
    while time.time() - start < time_limit and remaining() > 5:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        found = False
        k_max = int(np.argmax(corr))
        offset = k_max - (n - 1)
        
        # Build candidate lists for targeted search
        if offset >= 0:
            ci = np.array([i for i in ones if i - offset >= 0 and h[i - offset] == 0])
        else:
            ao = -offset
            ci = np.array([i for i in ones if i + ao < n and h[i + ao] == 0])
        
        n_try = 300
        for _ in range(n_try):
            if len(ci) > 0 and rng.random() < 0.5:
                i = ci[rng.randint(len(ci))]
            else:
                i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            
            h[i] = 0.0
            h[j] = 1.0
            new_corr = compute_corr(h)
            new_max = float(new_corr.max())
            
            if new_max < cur_max - 1e-12:
                cur_max = new_max
                corr = new_corr
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
                found = True
                break
            else:
                h[i] = 1.0
                h[j] = 0.0
        
        if found:
            stall = 0
        else:
            stall += 1
            if stall >= max_stall:
                break
    
    return best_h, best_max

def simulated_annealing(h, n, rng, t_start=5.0, t_end=0.1, n_steps=50000, time_limit=60):
    """Simulated annealing for binary pair swaps."""
    start = time.time()
    h = h.copy()
    corr = compute_corr(h)
    cur_max = float(corr.max())
    best_h = h.copy()
    best_max = cur_max
    
    for step in range(n_steps):
        if time.time() - start > time_limit or remaining() < 5:
            break
        
        T = t_start * (t_end / t_start) ** (step / max(n_steps, 1))
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_corr = compute_corr(h)
        new_max = float(new_corr.max())
        delta = new_max - cur_max
        
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            cur_max = new_max
            corr = new_corr
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_max

def multi_start_binary(n, rng, time_per_start=30, n_starts=5):
    """Try multiple initializations and local search."""
    best_h = None
    best_c5 = float('inf')
    
    initializations = []
    
    # Random starts
    for s in range(min(3, n_starts)):
        initializations.append(random_binary(n, rng))
    
    # Thue-Morse
    initializations.append(thue_morse(n, rng))
    
    # Periodic patterns
    for period in [3, 5, 7]:
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if (i % period) < period // 2 else 0.0
        h = make_binary(h, n, rng)
        initializations.append(h)
    
    for h_init in initializations:
        if remaining() < 15:
            break
        
        # Local search
        h_opt, corr_max = binary_local_search(h_init, n, rng, time_limit=time_per_start)
        c5 = corr_max * 2.0 / n
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
    
    return best_h, best_c5

def interpolate(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_new = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    return np.interp(x_new, xp, h_src)

def greedy_refine(h, n, rng, time_limit=30):
    """Intensive greedy refinement after SA."""
    return binary_local_search(h, n, rng, time_limit=time_limit, max_stall=500)

def continuous_refine(h, n, time_limit=60):
    """SLSQP continuous refinement."""
    h_opt = project_feasible(h.copy(), n)
    c5_cur = compute_c5(h_opt)
    
    for beta in [200, 500, 1500, 5000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h_opt, n, beta, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            c5_cur = c5_new
            h_opt = h_new.copy()
    
    return h_opt, c5_cur
