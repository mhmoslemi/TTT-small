            h = np.clip(params, 0.0, 1.0)
            corr = np.correlate(h, 1.0 - h, mode='full')
            val = smooth_max_val(corr, beta)
            pen = 2000.0 * (np.sum(h) - target) ** 2
            return val * dx + pen

        def obj_lb_grad(params):
            h = np.clip(params, 0.0, 1.0)
            g = compute_grad_smooth(h, beta, top_k)
            g += 4000.0 * (np.sum(h) - target) / n
            return g

        try:
            res = minimize(obj_lb, h_work, method='L-BFGS-B', jac=obj_lb_grad,
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 500, 'ftol': 1e-25, 'gtol': 1e-20,
                                    'maxls': 200, 'maxfun': 20000})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n
                h_work = h_new.copy()
                print("  L-BFGS-B beta={}: C5={:.7f}".format(beta, c5_new))
            else:
                h_work = project_feasible(res.x, target)
        except Exception as e:
            print("  L-BFGS-B beta={} error: {}".format(beta, str(e)[:50]))

    # Phase 2: Subgradient descent with extrapolation
    if tleft() > 40:
        n = best_n
        target = n / 2.0
        h = best_h.copy()
        m1 = np.zeros(n)
        m2 = np.zeros(n)
        beta_sub = 500.0

        for it in range(500):
            if tleft() < 5:
                break
            step = 0.05 / (1.0 + 0.02 * it)
            g = compute_grad_smooth(h, beta_sub, top_k=10)
            gn = np.linalg.norm(g)
            if gn < 1e-15:
                break
            g /= gn

            m1 = 0.9 * m1 + 0.1 * g
            m2 = 0.85 * m2 + 0.15 * g
            velocity = m1 + 0.5 * (m2 - m1)

            h_trial = h - step * velocity
            h_trial = project_feasible(h_trial, target)
            c5_t = compute_c5(h_trial)

            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_trial.copy()
                best_n = n
            h = h_trial.copy()

            if it % 50 == 49 and it > 0:
                beta_sub *= 1.5

        print("  After subgrad: C5={:.7f}".format(best_c5))

    # Phase 3: L-BFGS-B on correlated pairs (sum-preserving)
    if tleft() > 40:
        n = best_n
        target = n / 2.0
        dx = 2.0 / n

        h_corr = np.corrcoef(best_h)
        rows, cols = np.where(np.triu(np.ones((n, n), dtype=int), k=1))
        if len(rows) > 3000:
            perm = rng.permutation(len(rows))
            rows = rows[perm[:3000]]
            cols = cols[perm[:3000]]
        else:
            rows = rows[:(cols - rows) >= 1]
            cols = cols[:(cols - rows) >= 1]

        n_pairs = len(rows)
        if n_pairs < 10:
            n_pairs = 0

        if n_pairs > 0:
            beta_cp = 3000
            top_k_cp = 15

            def obj_cp(params):
                h = best_h.copy()
                valid = (params >= -1.0) & (params <= 1.0)
                if not np.all(valid):
                    return 1e6
                for p in range(n_pairs):
                    r, c = rows[p], cols[p]
                    val = params[p]
                    if h[r] + val < -1e-9 or h[r] + val > 1.0 + 1e-9:
                        return 1e6
                    if h[c] - val < -1e-9 or h[c] - val > 1.0 + 1e-9:
                        return 1e6
                    h[r] += val
                    h[c] -= val
                corr = np.correlate(h, 1.0 - h, mode='full')
                return smooth_max_val(corr, beta_cp) * dx

            x0 = np.zeros(n_pairs)
            try:
                res = minimize(obj_cp, x0, method='L-BFGS-B',
                               bounds=[(-0.5, 0.5)] * n_pairs,
                               options={'maxiter': 300, 'ftol': 1e-20,
                                        'maxfun': 10000})
                h_new = best_h.copy()
                for p in range(n_pairs):
                    h_new[rows[p]] += res.x[p]
                    h_new[cols[p]] -= res.x[p]
                h_new = project_feasible(h_new, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  Pair L-BFGS: C5={:.7f}".format(c5_new))
            except Exception:
                pass

    # Phase 4: SLSQP with high-beta smooth max
    if tleft() > 25:
        n = best_n
        target = n / 2.0
        dx = 2.0 / n
        bounds = [(0.0, 1.0)] * n

        for beta_s in [3000, 8000, 15000]:
            if tleft() < 10:
                break
            top_k_s = 15

            def obj_sls(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                return smooth_max_val(corr, beta_s) * dx

            def obj_sls_grad(x):
                return compute_grad_smooth(x, beta_s, top_k_s)

            cons = [{'type': 'eq',
                    'fun': lambda x: np.sum(x) - target,
                    'jac': lambda x: np.ones(n)}]

            try:
                res = minimize(obj_sls, best_h, method='SLSQP', jac=obj_sls_grad,
                               bounds=bounds, constraints=cons,
                               maxiter=600, ftol=1e-22,
                               options={'iprint': 0, 'maxls': 100})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  SLSQP beta={}: C5={:.7f}".format(beta_s, c5_new))
            except Exception:
                pass

    # Phase 5: Intensive local search
    if tleft() > 15:
        n = best_n
        h = best_h.copy()

        delta_sets = [
            [0.001, 0.002, 0.003, 0.005, 0.008, 0.01],
            [0.0005, 0.001, 0.0015, 0.002, 0.003],
            [0.0002, 0.0005, 0.001, 0.0008, 0.0003],
        ]

        for ds_idx, dlist in enumerate(delta_sets):
            n_iter = max(1, int(tleft() * 25))
            n_iter = min(n_iter, 50000)
            improved_any = False
            for i in range(n_iter):
                if tleft() < 3:
                    break
                best_c5, improved = local_search_pair(
                    h, best_c5, n, target, rng, 100, dlist, tleft, 1e-12)
                if improved:
                    improved_any = True
                elif i > 1000 and not improved_any:
                    if ds_idx < len(delta_sets) - 1:
                        continue
                    break
            if improved_any:
                best_h = h.copy()
                print("  Local search round {}: C5={:.7f}".format(ds_idx, best_c5))

    # Phase 6: Very fine final polish
    if tleft() > 10:
        n = best_n
        h = best_h.copy()
        fine_deltas = [0.0001, 0.0002, 0.0005, 0.0003, 0.0008]
        n_iter = max(1, min(int(tleft() * 20), 30000))

        for i in range(n_iter):
            if tleft() < 2:
                break
            best_c5, _ = local_search_pair(
                h, best_c5, n, target, rng, 200, fine_deltas, tleft, 1e-13)

        if compute_c5(h) < best_c5:
            best_c5 = compute_c5(h)
            best_h = h.copy()

    # Final: ensure feasibility
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
