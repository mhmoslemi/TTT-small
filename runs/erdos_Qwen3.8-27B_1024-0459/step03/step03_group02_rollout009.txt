_BETA_GLOBAL = 1000.0

    n = _N_GLOBAL
    beta = _BETA_GLOBAL
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def constraint_sum(x):
    return np.sum(x) - _N_GLOBAL / 2.0

                      options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idxs = np.argsort(np.abs(h - 0.5))
        for idx in idxs[:5]:
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
            if abs(h[idx] - (h[idx] - diff + diff)) < 1.0:
                if 0.0 <= h[idx] <= 1.0:
                    break
            h[idx] = max(0.0, min(1.0, h[idx]))
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)

def smooth_max_obj(x):
    n = _N_GLOBAL
    beta = _BETA_GLOBAL
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def constraint_sum(x):
    return np.sum(x) - _N_GLOBAL / 2.0

def slsqp_optimize(h, n, beta, maxiter=400, ftol=1e-14):
    global _N_GLOBAL, _BETA_GLOBAL
    _N_GLOBAL = n
    _BETA_GLOBAL = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    try:
        res = minimize(smooth_max_obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        h_out = project_feasible(res.x, n)
        if compute_c5(h_out) <= compute_c5(h) + 1e-12:
            return h_out
    except Exception:
        pass
    return h.copy()

def two_coord_optimize(h, i, j):
    c = h[i] + h[j]
    lo = max(0.0, c - 1.0)
    hi = min(1.0, c)
    if hi - lo < 1e-10:
        return h
    best_a = h[i]
    best_val = compute_c5(h)
    grid = np.linspace(lo, hi, 17)
    for a in grid:
        if a < -1e-12 or a > 1.0 + 1e-12:
            continue
        b = c - a
        if b < -1e-12 or b > 1.0 + 1e-12:
            continue
        h[i] = a
        h[j] = b
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_a = a
    step = (hi - lo) / 17.0
    lo2 = max(lo, best_a - 1.5 * step)
    hi2 = min(hi, best_a + 1.5 * step)
    fine = np.linspace(lo2, hi2, 25)
    for a in fine:
        b = c - a
        if b < -1e-12 or b > 1.0 + 1e-12:
            continue
        h[i] = a
        h[j] = b
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_a = a
    h[i] = best_a
    h[j] = c - best_a
    return h

def three_coord_optimize(h, i, j, k):
    s = h[i] + h[j] + h[k]
    lo_i = max(0.0, s - 2.0)
    hi_i = min(1.0, s)
    if hi_i - lo_i < 1e-10:
        return h
    best_t = (h[i], h[j], h[k])
    best_val = compute_c5(h)
    for ai in np.linspace(lo_i, hi_i, 7):
        rem = s - ai
        lo_j = max(0.0, rem - 1.0)
        hi_j = min(1.0, rem)
        if hi_j - lo_j < 1e-10:
            continue
        for aj in np.linspace(lo_j, hi_j, 7):
            ak = rem - aj
            if ak < -1e-10 or ak > 1.0 + 1e-10:
                continue
            h[i] = ai
            h[j] = aj
            h[k] = ak
            val = compute_c5(h)
            if val < best_val:
                best_val = val
                best_t = (ai, aj, ak)
    h[i], h[j], h[k] = best_t
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def active_lag_descent(h, n, max_iter=200):
    """Gradient-like step targeting only the top active lags."""
    for it in range(max_iter):
        corr = compute_corr(h)
        max_corr = np.max(corr)
        top_idx = int(np.argmax(corr))
        offset = top_idx - (n - 1)
        grad = np.zeros(n)
        if offset >= 0:
            n_pairs = n - offset
            if n_pairs <= 0:
                break
            grad[offset:offset + n_pairs] += (1.0 - h[:n_pairs])
            grad[:n_pairs] -= h[offset:offset + n_pairs]
        else:
            ao = -offset
            n_pairs = n - ao
            if n_pairs <= 0:
                break
            grad[:n_pairs] += (1.0 - h[ao:ao + n_pairs])
            grad[ao:ao + n_pairs] -= h[:n_pairs]
        # Project gradient to maintain sum constraint
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for step in [0.01, 0.005, 0.002, 0.001, 0.02]:
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_t = compute_c5(h_trial)
            if c5_t < compute_c5(h) - 1e-15:
                h = h_trial
                improved = True
                break
        if not improved:
            break
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 160
        best_h = np.full(best_n, 0.5)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n

    # Phase 1: SLSQP with escalating precision
    for beta in [80, 200, 600, 1500, 4000]:
        if remaining_time() < 12:
            break
        h_new = slsqp_optimize(best_h.copy(), n, beta, maxiter=500, ftol=1e-15)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Two-coordinate sum-preserving optimization (key new method)
    h = best_h.copy()
    stagnation = 0
    for sweep in range(80):
        if remaining_time() < 5:
            break
        improved = False
        # Paired systematic sweep
        for i in range(n):
            if remaining_time() < 2:
                break
            for j in [i + 1, i + 2, i + 3, i + 5, i + 8, i + 13]:
                if j >= n:
                    break
                hc = h.copy()
                two_coord_optimize(hc, i, j)
                c5_t = compute_c5(hc)
                if c5_t < best_c5 - 1e-14:
                    best_c5 = c5_t
                    h = hc
                    improved = True
                    if remaining_time() < 2:
                        break
        # Random pairs
        for _ in range(n):
            if remaining_time() < 2:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hc = h.copy()
            two_coord_optimize(hc, i, j)
            c5_t = compute_c5(hc)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                h = hc
                improved = True
        if improved:
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation >= 3:
                break

    # Phase 3: SLSQP refine after 2-coord
    if remaining_time() > 20:
        for beta in [1000, 3000, 8000]:
            if remaining_time() < 8:
                break
            h_new = slsqp_optimize(best_h.copy(), n, beta, maxiter=350, ftol=1e-15)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 4: Active-lag targeted descent
    if remaining_time() > 15:
        h_active = active_lag_descent(best_h.copy(), n, max_iter=100)
        c5_a = compute_c5(h_active)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_active.copy()
        # Follow with SLSQP
        if remaining_time() > 10:
            h_new = slsqp_optimize(best_h.copy(), n, 2000, maxiter=200, ftol=1e-15)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 5: Resolution exploration
    for trial_n in [128, 144, 176, 192, 224, 256, 320]:
        if remaining_time() < 20:
            break
        h_interp = interp_to(best_h, best_n, trial_n)
        h_interp = project_feasible(h_interp, trial_n)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5 + 0.004:
            for beta in [200, 800, 2500]:
                if remaining_time() < 8:
                    break
                h_new = slsqp_optimize(h_interp, trial_n, beta, maxiter=300, ftol=1e-14)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = trial_n
            # 2-coord at new resolution
            if remaining_time() > 8 and best_n == trial_n:
                h2 = best_h.copy()
                for s in range(12):
                    if remaining_time() < 3:
                        break
                    for _ in range(trial_n // 2):
                        i = rng.randint(trial_n)
                        j = rng.randint(trial_n)
                        if i == j:
                            continue
                        hc = h2.copy()
                        two_coord_optimize(hc, i, j)
                        c5_t = compute_c5(hc)
                        if c5_t < best_c5 - 1e-14:
                            best_c5 = c5_t
                            h2 = hc
                            best_h = hc
                    best_h = h2.copy()

    # Phase 6: Three-coordinate optimization for stuck solutions
    if remaining_time() > 25:
        h = best_h.copy()
        n = best_n
        for sweep in range(10):
            if remaining_time() < 5:
                break
            improved = False
            for _ in range(n // 2):
                if remaining_time() < 2:
                    break
                idxs = rng.choice(n, size=3, replace=False)
                hc = h.copy()
                three_coord_optimize(hc, int(idxs[0]), int(idxs[1]), int(idxs[2]))
                c5_t = compute_c5(hc)
                if c5_t < best_c5 - 1e-14:
                    best_c5 = c5_t
                    h = hc
                    improved = True
            if not improved:
                break
        best_h = h.copy()

    # Phase 7: Quasi-binary local search
    if remaining_time() > 20:
        n = best_n
        h_bin = best_h.copy()
        # Push values toward 0 or 1
        for _ in range(3):
            for i in range(n):
                if remaining_time() < 3:
                    break
                if h_bin[i] < 0.5:
                    h_try = h_bin.copy()
                    h_try[i] = 0.0
                    rem = h_bin[i]
                    # distribute rem to other non-saturated coords
                    mask = (h_try > 0.01) & (h_try < 0.99)
                    if np.sum(mask) > 0:
                        h_try[mask] += rem / np.sum(mask)
                        h_try = project_feasible(h_try, n)
                    else:
                        h_try = project_feasible(h_try, n)
                    c5_t = compute_c5(h_try)
                    if c5_t < best_c5 - 1e-14:
                        best_c5 = c5_t
                        h_bin = h_try.copy()
                else:
                    h_try = h_bin.copy()
                    h_try[i] = 1.0
                    rem = 1.0 - h_bin[i]
                    mask = (h_try > 0.01) & (h_try < 0.99)
                    if np.sum(mask) > 0:
                        h_try[mask] -= rem / np.sum(mask)
                        h_try = project_feasible(h_try, n)
                    else:
                        h_try = project_feasible(h_try, n)
                    c5_t = compute_c5(h_try)
                    if c5_t < best_c5 - 1e-14:
                        best_c5 = c5_t
                        h_bin = h_try.copy()
        if remaining_time() > 5:
            h_new = slsqp_optimize(h_bin, n, 2000, maxiter=200, ftol=1e-15)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 8: Final ultra-precise SLSQP + 2-coord
    if remaining_time() > 8:
        n = best_n
        for beta in [5000, 12000]:
            if remaining_time() < 4:
                break
            h_new = slsqp_optimize(best_h.copy(), n, beta, maxiter=300, ftol=1e-16)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Final 2-coord polish
    if remaining_time() > 5:
        n = best_n
        h = best_h.copy()
        for _ in range(n):
            if remaining_time() < 2:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hc = h.copy()
            two_coord_optimize(hc, i, j)
            c5_t = compute_c5(hc)
            if c5_t < best_c5 - 1e-15:
                best_c5 = c5_t
                best_h = hc
                h = hc

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
