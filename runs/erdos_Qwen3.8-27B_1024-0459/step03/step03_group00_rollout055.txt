    if budget_fn() > 8:
        best_h, best_c5 = do_subgrad_phase(best_h, best_n, budget_fn, rng, max_iters=50)
    if budget_fn() > 8:
        best_h, best_c5 = do_slsqp_phase(best_h, best_n, budget_fn, [20000, 100000])
    if budget_fn() > 5:
        best_h, best_c5 = do_coord_descent(best_h, best_n, budget_fn, rng, max_sweeps=5)

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
