            if y_best < best_score:
                best_score = y_best
                best_h = h_suggestion.copy()
                
            # Projection to feasible space
            h_projected = project_feasible(h_suggestion)
            if compute_feasibility(h_projected):
                best_h = h_projected
                run coord_descent_stable(best_h, stage)

    return best_h, best_score
