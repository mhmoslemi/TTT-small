def _c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / n)


def _proj(h, n):

                grad[i - p] -= x[i]

def _ggrad(h, kidx):
    n = len(h)
    s = kidx - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        for i in range(n - s):
            g[i] += (1.0 - h[i + s])
            g[i + s] -= h[i]
    else:
        p = -s
        for i in range(p, n):
            g[i] += (1.0 - h[i - p])
            g[i - p] -= h[i]
    return g
