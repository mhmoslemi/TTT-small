import numpy as np
from scipy.optimize import minimize
import time

_N = 0
_TGT = 0.0
_START_TIME = 0.0
_BUDGET_S = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')


def smooth_max_obj(x, beta):
    n = _N
    dx = 2.0 / n
    corr = compute_corr_vec(x)
    m = np.max(corr)
    shifted = corr - m
    sm = (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx
    penalty = 500.0 * (np.sum(x) - _TGT) ** 2
    return sm + penalty


def smooth_max_grad(x, beta):
    n = _N
    dx = 2.0 / n
    corr = compute_corr_vec(x)
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= wsum

    grad = np.zeros(n)
    threshold = 1e-10
    for j in range(2 * n - 1):
        wj = weights[j]
        if wj < threshold:
            continue
        s = j - (n - 1)
        if s >= 0:
            ns = n - s
            if ns > 0:
                grad[:ns] += wj * (-x[s:])
                grad[s:] += wj * (1.0 - x[:ns])
        else:
            sp = -s
            ns = n - sp
            if ns > 0:
                grad[:ns] += wj * (1.0 - x[sp:])
                grad[sp:] += wj * (-x[:ns])

    grad *= dx
    grad += 2.0 * 500.0 * (np.sum(x) - _TGT)
    return grad


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        unsat = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(unsat))
        if cnt > 0:
            h[unsat] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 2.0)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h


def make_sturmian(n, alpha):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (np.floor((i + 1) * alpha) - np.floor(i * alpha)) else 0.0
    return h


def make_sine(n, freq, phase, amp=0.35):
    t = np.arange(n) / float(n)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * t + phase)
    return np.clip(h, 0.0, 1.0)


def binary_sa_informed(h_init, n, iterations, rng, t_start=0.004, t_end=0.00003):
    h = np.where(h_init > 0.5, 1.0, 0.0)
    target_ones = n // 2
    num_ones = int(np.sum(h))
    if num_ones > target_ones:
        ones_idx = np.where(h == 1.0)[0]
        remove = rng.choice(len(ones_idx), num_ones - target_ones, replace=False)
        h[ones_idx[remove]] = 0.0
    elif num_ones < target_ones:
        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
        h[zeros_idx[add]] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])

    for it in range(iterations):
        if remaining_time() < 3:
            break
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break

        T = t_start * (t_end / t_start) ** (float(it) / max(float(iterations), 1.0))

        if it % 500 == 0 or T < 0.001:
            corr = compute_corr_vec(h)
            jmax = np.argmax(corr)
            s = jmax - (n - 1)
            # Find positions contributing to worst lag
            if s >= 0:
                contrib_ones = [i + s for i in range(n - s) if h[i + s] == 1.0 and h[i] == 0.0]
            else:
                sp = -s
                contrib_ones = [i - sp for i in range(sp, n) if h[i - sp] == 1.0 and h[i] == 0.0]

        if T < 0.001 and contrib_ones and len(contrib_ones) > 2 and rng.rand() < 0.6:
            oi = contrib_ones[rng.randint(len(contrib_ones))]
            if h[oi] != 1.0:
                oi = ones_list[rng.randint(len(ones_list))]
            if 0 <= oi < n and h[oi] == 1.0:
                oz = zeros_list[rng.randint(len(zeros_list))]
                h[oi] = 0.0
                h[oz] = 1.0
                new_c5 = compute_c5(h)
                if new_c5 < current_c5 or rng.rand() < np.exp(-(new_c5 - current_c5) / max(T, 1e-15)):
                    current_c5 = new_c5
                    ones_list[ones_list.index(oi)] = oz
                    zeros_list[zeros_list.index(oz)] = oi
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                else:
                    h[oi] = 1.0
                    h[oz] = 0.0
            continue

        i = rng.randint(len(ones_list))
        j = rng.randint(len(zeros_list))
        oi = ones_list[i]
        oz = zeros_list[j]

        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = compute_c5(h)

        if new_c5 < current_c5 or (T > 1e-15 and rng.rand() < np.exp(-(new_c5 - current_c5) / T)):
            current_c5 = new_c5
            ones_list[i] = oz
            zeros_list[j] = oi
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0

    return best_h, best_c5

def continuous_optimize(h0, n, beta_vals, maxiter_each=300):
    global _N, _TGT
    _N = n
    _TGT = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h = h0.copy()
    best_c5 = compute_c5(h)

    for beta in beta_vals:
        if remaining_time() < 8:
            break
        try:
            res = minimize(smooth_max_obj, h, args=(beta,),
                           jac=lambda x, b: smooth_max_grad(x, b),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15,
                                    'gtol': 1e-10, 'iprint': 0, 'maxls': 40, 'maxfun': 3000})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h = h_new
        except Exception:
            pass
    return h, best_c5


def coordinate_descent(h, n, sweeps=8, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    best_c5 = compute_c5(h)
    for sweep in range(sweeps):
        if remaining_time() < 4:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if remaining_time() < 2:
                break
            old_val = h[i]
            current_c5 = compute_c5(h)
            best_val = old_val
            candidates = [0.0, 1.0, 0.5, max(0.0, old_val - 0.25), min(1.0, old_val + 0.25)]
            if old_val > 0.3:
                candidates.append(0.1)
            if old_val < 0.7:
                candidates.append(0.9)
            for nv in candidates:
                if abs(nv - old_val) < 1e-8:
                    continue
                h[i] = nv
                h_feas = project_feasible(h, n)
                c5_test = compute_c5(h_feas)
                if c5_test < current_c5:
                    current_c5 = c5_test
                    best_val = nv
            if best_val != old_val:
                h[i] = best_val
                h = project_feasible(h, n)
                any_imp = True
            else:
                h[i] = old_val
        if not any_imp:
            break
    final_c5 = compute_c5(h)
    return h, min(best_c5, final_c5)
