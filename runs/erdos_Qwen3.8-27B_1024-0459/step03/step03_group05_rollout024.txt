    sum_err = np.sum(h) - target
    return smax + _g['pen'] * sum_err * sum_err

def _grad_lbfgs(x):
    h = np.clip(x, 0.0, 1.0)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    beta = _g['beta']
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w /= np.sum(w)
    grad = np.zeros(n)
    for t in range(2 * n - 1):
        wt = w[t]
        if wt < 1e-14:
            continue
        k = t - (n - 1)
        wtd = wt * dx
        if k >= 0:
            if k == 0:
                grad += wtd * (1.0 - 2.0 * h)
            else:
                grad[k:] += wtd * (1.0 - h[:n - k])
                grad[:n - k] -= wtd * h[k:]
        else:
            ma = -k
            grad[:n - ma] += wtd * (1.0 - h[ma:])
            grad[ma:] -= wtd * h[:n - ma]
    target = _g['target']
    sum_err = np.sum(h) - target
    grad += 2.0 * _g['pen'] * sum_err
    return grad

def _obj_smooth_only(x):
    h = np.clip(x, 0.0, 1.0)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    beta = _g['beta']
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx

def interpolate_h(h_src, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    return np.interp(xn, xp, h_src)

def _subgrad_active(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w /= np.sum(w)
    g = np.zeros(n)
    for t in range(2 * n - 1):
        wt = w[t]
        if wt < 1e-14:
            continue
        k = t - (n - 1)
        wtd = wt * dx
        if k >= 0:
            if k == 0:
                g += wtd * (1.0 - 2.0 * h)
            else:
                g[k:] += wtd * (1.0 - h[:n - k])
                g[:n - k] -= wtd * h[k:]
        else:
            ma = -k
            g[:n - ma] += wtd * (1.0 - h[ma:])
            g[ma:] -= wtd * h[:n - ma]
    return g

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    def time_left():
        return max(0.0, budget_s - (time.time() - t0) - 5.0)

    rng = np.random.RandomState(seed)

    h_par = np.array(initial_h_values, dtype=float)
    n_par = len(h_par)
    c5_par = compute_c5(h_par)
    print("Parent: n={}, C5={:.8f}".format(n_par, c5_par))

    best_h = h_par.copy()
    best_c5 = c5_par
    best_n = n_par

    # Phase 1: Multi-resolution L-BFGS-B screening
    n_candidates = [n_par]
    for delta in [8, 16, 24, 32, 48, 64, 80, 96, 128]:
        for d in [delta, -delta]:
            nn = n_par + d
            if 80 <= nn <= 360 and nn % 2 == 0:
                n_candidates.append(nn)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time_left() < 35:
            break
        target = n_try / 2.0
        h0 = interpolate_h(best_h, best_n, n_try)
        h0 = project_feasible(h0, target)
        c5_0 = compute_c5(h0)

        for beta in [400, 1000, 2500]:
            if time_left() < 12:
                break
            _g['beta'] = beta
            _g['target'] = target
            _g['pen'] = 300.0
            try:
                res = minimize(_obj_lbfgs, h0, jac=_grad_lbfgs, method='L-BFGS-B',
                              bounds=[(0.0, 1.0)] * n_try,
                              options={'maxiter': 400, 'ftol': 1e-18, 'gtol': 1e-15,
                                       'maxls': 60, 'maxfun': 4000})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-9:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n_try
                    h0 = h_new.copy()
                    print("  Screen n={}, beta={}: C5={:.8f}".format(n_try, beta, c5_new))
                else:
                    if c5_new < c5_0:
                        h0 = h_new.copy()
                        c5_0 = c5_new
            except Exception:
                pass

    # Phase 2: Intensive L-BFGS-B at best resolution with increasing beta
    n = best_n
    target = n / 2.0

    for beta in [500, 1000, 2000, 4000, 8000, 15000]:
        if time_left() < 18:
            break
        _g['beta'] = beta
        _g['target'] = target
        _g['pen'] = 800.0
        try:
            res = minimize(_obj_lbfgs, best_h, jac=_grad_lbfgs, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 1000, 'ftol': 1e-20, 'gtol': 1e-17,
                                   'maxls': 100, 'maxfun': 10000})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  Intensive beta={}: C5={:.8f}".format(beta, c5_new))
        except Exception:
            pass

    # Phase 3: Subgradient with momentum and adaptive step
    if time_left() > 40:
        h = best_h.copy()
        momentum = np.zeros(n)
        alpha = 0.015
        decay = 0.91
        beta_sg = 3000

        for it in range(800):
            if time_left() < 5:
                break
            g = _subgrad_active(h, beta_sg)
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-15:
                break
            g /= gnorm
            momentum = decay * momentum - alpha * g
            h_trial = h + momentum
            h_trial = project_feasible(h_trial, target)
            c5_t = compute_c5(h_trial)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_trial.copy()
            h = h_trial.copy()
            if it % 60 == 59:
                alpha *= 0.88
            if it % 100 == 99:
                beta_sg = min(beta_sg * 1.5, 20000)

        print("  After subgradient: C5={:.8f}".format(best_c5))

    # Phase 4: Pair coordinate descent with decreasing step sizes
    if time_left() > 30:
        h = best_h.copy()
        step_sizes = [0.04, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0001]

        for delta in step_sizes:
            if time_left() < 8:
                break
            h = best_h.copy()
            sweeps = 0
            max_sweeps = 12
            while sweeps < max_sweeps and time_left() > 5:
                sweeps += 1
                improved = False
                perms = rng.permutation(n)
                for s in range(n):
                    i = perms[s]
                    j = perms[(s + n // 3) % n]
                    if i == j:
                        continue
                    for direction in [1, -1]:
                        if improved:
                            break
                        di = direction * delta
                        if h[i] + di < -1e-9 or h[i] + di > 1.0 + 1e-9:
                            continue
                        if h[j] - di < -1e-9 or h[j] - di > 1.0 + 1e-9:
                            continue
                        h[i] += di
                        h[j] -= di
                        c5_t = compute_c5(h)
                        if c5_t < best_c5 - 1e-11:
                            best_c5 = c5_t
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i] -= di
                            h[j] += di
                if not improved:
                    break

        print("  After pair descent: C5={:.8f}".format(best_c5))

    # Phase 5: Full single-coordinate sweep (try all j for compensating partner)
    if time_left() > 20:
        h = best_h.copy()
        small_deltas = [0.003, 0.001, 0.0004, 0.0001]
        for delta in small_deltas:
            if time_left() < 6:
                break
            improved_any = True
            while improved_any and time_left() > 4:
                improved_any = False
                h = best_h.copy()
                for i in range(n):
                    if time_left() < 3:
                        break
                    h_i_old = h[i]
                    for di_val in [delta, -delta]:
                        h_try = h_i_old + di_val
                        if h_try < -1e-9 or h_try > 1.0 + 1e-9:
                            continue
                        # Try compensating at various j
                        best_j = -1
                        best_c5_j = best_c5
                        for j in range(n):
                            if j == i:
                                continue
                            hj_try = h[j] - di_val
                            if hj_try < -1e-9 or hj_try > 1.0 + 1e-9:
                                continue
                            h[i] = h_try
                            h[j] = hj_try
                            c5_t = compute_c5(h)
                            if c5_t < best_c5_j:
                                best_c5_j = c5_t
                                best_j = j
                            h[i] = h_i_old
                            h[j] += di_val
                        if best_j >= 0 and best_c5_j < best_c5 - 1e-12:
                            h[i] = h_try
                            h[best_j] = h[best_j] - di_val
                            best_c5 = best_c5_j
                            best_h = h.copy()
                            improved_any = True
                            break

        print("  After full sweep: C5={:.8f}".format(best_c5))

    # Phase 6: Final L-BFGS-B polish
    if time_left() > 10:
        n = best_n
        target = n / 2.0
        for beta in [5000, 10000]:
            if time_left() < 5:
                break
            _g['beta'] = beta
            _g['target'] = target
            _g['pen'] = 2000.0
            try:
                res = minimize(_obj_lbfgs, best_h, jac=_grad_lbfgs, method='L-BFGS-B',
                              bounds=[(0.0, 1.0)] * n,
                              options={'maxiter': 500, 'ftol': 1e-20, 'gtol': 1e-18,
                                       'maxls': 100, 'maxfun': 5000})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  Polish beta={}: C5={:.8f}".format(beta, c5_new))
            except Exception:
                pass

    # Phase 7: Random restarts from perturbed best (if time allows)
    if time_left() > 50:
        for restart in range(5):
            if time_left() < 30:
                break
            n_r = best_n
            target_r = n_r / 2.0
            perturbation = 0.02
            h_pert = best_h + rng.randn(n_r) * perturbation
            h_pert = project_feasible(h_pert, target_r)
            c5_pert = compute_c5(h_pert)

            _g['beta'] = 2000
            _g['target'] = target_r
            _g['pen'] = 500.0
            try:
                res = minimize(_obj_lbfgs, h_pert, jac=_grad_lbfgs, method='L-BFGS-B',
                              bounds=[(0.0, 1.0)] * n_r,
                              options={'maxiter': 600, 'ftol': 1e-19, 'gtol': 1e-16,
                                       'maxls': 80, 'maxfun': 6000})
                h_new = project_feasible(res.x, target_r)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  Restart {}: C5={:.8f}".format(restart, c5_new))
            except Exception:
                pass

    # Final: ensure feasibility and compute
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    # Verify with evaluator if available
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
