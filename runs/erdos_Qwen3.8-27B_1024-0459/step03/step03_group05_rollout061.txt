            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n
                h0 = h_new.copy()
                print("  L-BFGS-B beta={}: C5={:.6f}".format(beta, c5_new))
        except Exception as e:
            print("  Error: {}".format(e))
    
    # Phase 2: Try higher resolutions
    for n_try in [n_par+40, n_par+80, 200, 240, 280]:
        if rem() < 60:
            break
        if n_try == best_n or n_try < 80 or n_try > 400:
            continue
        if n_try % 2 != 0:
            continue
        target_t = n_try / 2.0
        xp = (np.arange(best_n)+0.5)*2.0/best_n
        xn = (np.arange(n_try)+0.5)*2.0/n_try
        h0_t = np.interp(xn, xp, best_h)
        h0_t = project_feasible(h0_t, target_t)
        
        for beta in [200, 500, 1500, 4000]:
            if rem() < 20:
                break
            top_k = min(25, 2*n_try-1)
            
            def obj2(x, n_t=n_try):
                return smooth_max_val(np.correlate(x, 1-x, mode='full'), 2.0/n_t, beta, top_k)
            
            try:
                res = minimize(obj2, h0_t, method='L-BFGS-B',
                              bounds=[(0,1)]*n_try,
                              options={'maxiter': 500, 'ftol': 1e-22, 'gtol': 1e-15,
                                       'maxfun': 6000})
                h_new = project_feasible(res.x, target_t)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n_try
                    h0_t = h_new.copy()
                    print("  L-BFGS-B n={}, beta={}: C5={:.6f}".format(n_try, beta, c5_new))
            except Exception:
                pass
    
    # Phase 3: Subgradient with Nesterov
    if rem() > 40:
        n = best_n
        target = n / 2.0
        h = best_h.copy()
        v = np.zeros(n)
        
        for it in range(800):
            if rem() < 5:
                break
            beta = 300
            top_k = 15
            corr = np.correlate(h, 1-h, mode='full')
            if top_k < len(corr):
                idx_all = np.argsort(corr)[-top_k:]
            else:
                idx_all = np.arange(len(corr))
            cw = corr[idx_all]
            mw = cw.max()
            sw = cw - mw
            w = np.exp(beta * sw)
            w /= np.sum(w)
            
            dx = 2.0/n
            grad = np.zeros(n)
            for t, ki in enumerate(idx_all):
                wt = w[t]
                if wt < 1e-14: continue
                s = ki - (n-1)
                if s >= 0:
                    m_lo, m_hi = 0, n-1-s
                else:
                    m_lo, m_hi = -s, n-1
                if m_lo > m_hi: continue
                t1lo, t1hi = max(m_lo+s,0), min(m_hi+s,n-1)
                if t1lo <= t1hi:
                    grad[t1lo:t1hi+1] += wt
                    p = np.arange(t1lo, t1hi+1)
                    grad[p] -= wt * h[p-s]
                p2 = np.arange(m_lo, m_hi+1)
                grad[p2] -= wt * h[p2+s]
            grad *= dx
            
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15: break
            grad /= gnorm
            
            # Nesterov momentum
            decay = 0.93
            alpha = 0.015 * (1.0 + 0.5*np.cos(it/200.0))
            v = decay * v - alpha * grad
            h_trial = h + v
            h_trial = project_feasible(h_trial, target)
            c5_t = compute_c5(h_trial)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_trial.copy()
            h = h_trial.copy()
        
        print("  After subgradient: C5={:.6f}".format(best_c5))
    
    # Phase 4: Fine coordinate descent
    if rem() > 30:
        n = best_n
        target = n / 2.0
        h = best_h.copy()
        no_improve = 0
        
        for sweep in range(100):
            if rem() < 5:
                break
            improved = False
            for _ in range(4000):
                if rem() < 3:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j: continue
                sign = 1 if h[i] < h[j] else -1

                    di = sign * delta
                    dj = -sign * delta
                    if h[i]+di < -1e-9 or h[i]+di > 1+1e-9: continue
                    if h[j]+dj < -1e-9 or h[j]+dj > 1+1e-9: continue
                    h[i] += di; h[j] += dj
                    c5_t = compute_c5(h)
                    if
