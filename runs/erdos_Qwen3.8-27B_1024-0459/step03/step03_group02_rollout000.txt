```
def project_feasible(y, n):
    h = np.asarray(y, dtype=float).copy()
    if n <= 0: return h
    target = 0.5 * n
    mn = float(np.min(h)); mx = float(np.max(h))
    lo = -mx
    hi = 1.0 - mn
    if hi - lo <= 1e-15:
        np.clip(h,0.0,1.0,out=h)
    else:
        for _ in range(45):
            mid = 0.5*(lo+hi)
            s = np.sum(np.clip(h + mid, 0.0, 1.0))
            if s < target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + 0.5*(lo+hi), 0.0, 1.0)
    # fix residual
    for _ in range(8):
        res = target - float(np.sum(h))
        if abs(res) <= 1e-13:
            break
        if res > 0:
            slack = 1.0 - h
            idx = int(np.argmax(slack))
            add = min(res, slack[idx])
            if add <= 0.0:
                break
            h[idx] += add
        else:
            mass = h
            idx = int(np.argmax(mass))
            sub = min(-res, mass[idx])
            if sub <= 0.0:
                break
            h[idx] -= sub
    np.clip(h, 0.0, 1.0, out=h)
    return h
```

```python
import numpy as np
import time
from scipy import optimize as scipy_opt

_START_TIME = 0.0
_BUDGET_S = 1300.0
_EPS_TIME = 8.0
_RNG = None

_S_N = 0
_S_BETA = 1.0
_S_CACHE_X = None
_S_CACHE_CORR = None
_S_CACHE_VAL = 0.0
_S_CACHE_W = None
_S_CACHE_SUMW = 0.0

_ACTIVE_ALPHAS = (0.10, 0.05, 0.20, 0.025, 0.01, 0.40, 0.005, 0.80)


def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - _EPS_TIME)


def fallback_h(n):
    x = (np.arange(n) + 0.5) / float(n)
    h = 0.5 + 0.42 * np.sin(2.0 * np.pi * x + 0.7) + 0.08 * np.sin(4.0 * np.pi * x - 0.25)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)


def project_feasible(y, n):
    h = np.asarray(y, dtype=float).copy()
    if n <= 0:
        return h
    target = 0.5 * float(n)
    mn = float(np.min(h))
    mx = float(np.max(h))
    lo = -mx
    hi = 1.0 - mn
    if hi - lo <= 1e-15:
        np.clip(h, 0.0, 1.0, out=h)
    else:
        for _ in range(45):
            mid = 0.5 * (lo + hi)
            s = np.sum(np.clip(h + mid, 0.0, 1.0))
            if s < target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(12):
        res = target - float(np.sum(h))
        if abs(res) <= 1e-13:
            break
        if res > 0.0:
            slack = 1.0 - h
            idx = int(np.argmax(slack))
            add = min(res, float(slack[idx]))
            if add <= 0.0:
                break
            h[idx] += add
        else:
            mass = h
            idx = int(np.argmax(mass))
            sub = min(-res, float(mass[idx]))
            if sub <= 0.0:
                break
            h[idx] -= sub
    np.clip(h, 0.0, 1.0, out=h)
    return h


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    corr = compute_corr(h)
    return float(np.max(corr) * (2.0 / float(n)))


def lag_gradient(h, n, offset):
    grad = np.zeros(n, dtype=float)
    if offset >= 0:
        pairs = n - offset
        if pairs <= 0:
            return grad
        grad[offset:offset + pairs] += 1.0 - h[:pairs]
        grad[:pairs] -= h[offset:offset + pairs]
    else:
        d = -offset
        pairs = n - d
        if pairs <= 0:
            return grad
        grad[:pairs] += 1.0 - h[d:]
        grad[d:] -= h[:pairs]
    return grad


def fw_closest(pts, iters=90):
    if len(pts) == 0:
        return None
    if len(pts) == 1:
        return pts[0].copy()
    norms = np.einsum('ij,ij->i', pts, pts)
    v = pts[int(np.argmin(norms))].copy()
    vsq = float(v.dot(v))
    for _ in range(iters):
        dots = pts.dot(v)
        s_idx = int(np.argmin(dots))
        s = pts[s_idx]
        d = s - v
        dn2 = float(d.dot(d))
        if dn2 < 1e-24:
            break
        t = -float(v.dot(d)) / dn2
        if t <= 1e-12:
            break
        if t > 1.0:
            t = 1.0
        v_new = v + t * d
        v_new_sq = float(v_new.dot(v_new))
        if v_new_sq < vsq - 1e-16:
            v = v_new
            vsq = v_new_sq
        else:
            v = v_new
            vsq = v_new_sq
    return v


def active_dir_from_indices(h, n, indices):
    pts = []
    for idx in indices:
        off = int(idx) - (n - 1)
        g = lag_gradient(h, n, off)
        g = g - float(np.mean(g))
        if float(np.linalg.norm(g)) > 1e-12:
            pts.append(g)
    if len(pts) == 0:
        return None
    pts = np.vstack(pts)
    v = fw_closest(pts, 90)
    if v is None:
        return None
    nv = float(np.linalg.norm(v))
    if nv < 1e-8:
        return None
    d = -v
    nd = float(np.linalg.norm(d))
    if nd > 3.0:
        d = d * (3.0 / nd)
    return d


def active_directions(h, n, corr):
    maxc = float(np.max(corr))
    dirs = []

    near = np.where(corr >= maxc - 2.0)[0]
    if len(near) > 16:
        order = np.argsort(corr[near])[-16:]
        near = near[order]
    if len(near) > 0:
        d = active_dir_from_indices(h, n, near)
        if d is not None:
            dirs.append(d)

    sorted_idx = np.argsort(corr)[::-1]
    for cnt in (1, 2, 4, 8):
        d = active_dir_from_indices(h, n, sorted_idx[:cnt])
        if d is not None:
            dirs.append(d)
    return dirs

def _reset_smooth_cache():
    global _S_CACHE_X, _S_CACHE_CORR, _S_CACHE_VAL, _S_CACHE_W, _S_CACHE_SUMW
    _S_CACHE_X = None
    _S_CACHE_CORR = None
    _S_CACHE_VAL = 0.0
    _S_CACHE_W = None
    _S_CACHE_SUMW = 0.0


def _get_smooth_state(x):
    global _S_CACHE_X, _S_CACHE_CORR, _S_CACHE_VAL, _S_CACHE_W, _S_CACHE_SUMW
    n = _S_N
    beta = float(_S_BETA)
    dx = 2.0 / float(n)

    if _S_CACHE_X is None or len(_S_CACHE_X) != len(x) or not np.array_equal(x, _S_CACHE_X):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = float(np.max(corr))
        shift = beta * (corr - m)
        shift = np.minimum(shift, 0.0)
        w = np.exp(shift)
        sw = float(np.sum(w))
        if sw <= 0.0:
            sw = 1.0
            w = np.zeros_like(w)
            w[0] = 1.0
        val = dx * (m + np.log(sw) / beta)

        _S_CACHE_X = np.asarray(x, dtype=float).copy()
        _S_CACHE_CORR = corr
        _S_CACHE_VAL = val
        _S_CACHE_W = w
        _S_CACHE_SUMW = sw

    return _S_CACHE_VAL, _S_CACHE_CORR


def _smooth_obj(x):
    val, _ = _get_smooth_state(x)
    return val


def _smooth_grad(x):
    _, corro = _get_smooth_state(x)
    n = _S_N
    beta = float(_S_BETA)
    dx = 2.0 / float(n)
    w = _S_CACHE_W
    sw = float(_S_CACHE_SUMW)
    if sw <= 0.0:
        return np.zeros(n, dtype=float)

    active = np.where(w > 1e-14)[0]
    g = np.zeros(n, dtype=float)
    for idx in active:
        off = int(idx) - (n - 1)
        g += w[idx] * lag_gradient(_S_CACHE_X, n, off)
    return (dx / sw) * g


def _smooth_constraint(x):
    return float(np.sum(x) - 0.5 * float(_S_N))


def slsqp_maxiter(n, requested):
    if n <= 170:
        cap = 260
    elif n <= 230:
        cap = 170
    elif n <= 330:
        cap = 110
    else:
        cap = 60
    return int(min(max(20, int(requested)), cap))


def smooth_slsqp(h, n, beta, maxiter):
    global _S_N, _S_BETA
    h0 = project_feasible(h, n)
    _S_N = int(n)
    _S_BETA = float(beta)
    _reset_smooth_cache()

    mx = slsqp_maxiter(n, maxiter)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _smooth_constraint}]
    try:
        res = scipy_opt.minimize(
            _smooth_obj,
            h0,
            method='SLSQP',
            jac=_smooth_grad,
            bounds=bounds,
            constraints=cons,
            options={'maxiter': mx, 'ftol': 1e-10, 'iprint': 0}
        )
        h2 = project_feasible(res.x, n)
    except Exception:
        h2 = h0
    c2 = compute_c5(h2)
    return h2, c2


def interp_to_len(src_h, dst_n):
    src_n = len(src_h)
    if src_n == dst_n:
        return project_feasible(src_h, dst_n)
    xp = (np.arange(src_n) + 0.5) * (2.0 / float(src_n))
    xq = (np.arange(dst_n) + 0.5) * (2.0 / float(dst_n))
    y = np.interp(xq, xp, src_h)
    return project_feasible(y, dst_n)

        for beta, req in plans:
            if remaining_time() < need:
                break
            h3, c3 = smooth_slsqp(best_h, n, beta, req)
            if c3 < best_c - 1e-15:
                best_h, best_c = h3, c3

        if remaining_time() > 12.0:
            h4, c4 = active_descent_limited(best_h, n, max_iter=60, time_limit_s=min(10.0, float(budget_s) / 5.0))
            if c4 < best_c:
                best_h, best_c = h4, c4

    return best_h, best_c


def make_candidates(base_n):
    cands = set([base_n])
    for off in (8, 16, 24, 32, 48, 64, 80, 96, 128, 160, 256, 320, 384):
        for m in (base_n + off, base_n - off):
            if 96 <= m <= 512 and (m % 2) == 0:
                cands.add(m)
    for m in (128, 160, 192, 224, 256, 288, 320, 384, 448, 512):
        if 96 <= m <= 512 and (m % 2) == 0:
            cands.add(m)
    lst = [m for m in cands if 96 <= m <= 512 and (m % 2) == 0]
    lst.sort(key=lambda m: (abs(m - base_n), m))
    return lst[:10]


def variant_refine(base_h, base_c, budget_s):
    n = len(base_h)
    v1 = project_feasible(0.5 * (base_h + base_h[::-1]), n)
    v2 = project_feasible(1.0 - base_h, n)
    v3 = project_feasible(1.0 - v1, n)
    for hv in (v1, v2, v3):
        if remaining_time() < 30.0:
            break
        c = compute_c5(hv)
        if c < base_c + 0.005:
            h2, c2 = refine_resolution(hv, n, min(60.0, float(budget_s)))
            if c2 < base_c:
                return h2, c2
    return base_h, base_c


def binary_from_h(h, n):
    k = n // 2
    idx = np.argsort(-h)[:k]
    b = np.zeros(n, dtype=np.int32)
    b[idx] = 1
    return b


def binary_c5(b, n):
    dx = 2.0 / float(n)
    corr = np.correlate(b, 1 - b, mode='full')
    return float(np.max(corr)) * dx


def binary_local_search(b, n, time_limit_s):
    start = time.time()
    b = b.copy().astype(np.int32)
    cur_c = binary_c5(b, n)
    best = b.copy()
    best_c = cur_c
    no_improve = 0

    while time.time() - start < time_limit_s:
        if remaining_time() < 3.0:
            break

        ones = np.nonzero(b == 1)[0]
        zeros = np.nonzero(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = ones[int(_RNG.randint(len(ones)))]
        j = zeros[int(_RNG.randint(len(zeros)))]
        if i == j:
            continue

        old_i, old_j = int(b[i]), int(b[j])
        b[i] = old_j
        b[j] = old_i

        c = binary_c5(b, n)
        if c < best_c - 1e-15:
            best_c = c
            best = b.copy()

        if c < cur_c - 1e-15:
            cur_c = c
            no_improve = 0
        else:
            b[i] = old_i
            b[j] = old_j
            no_improve += 1
            if no_improve > max(250, 8 * n):
                b = best.copy()
                for _ in range(4):
                    ones = np.nonzero(b == 1)[0]
                    zeros = np.nonzero(b == 0)[0]
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    p = ones[int(_RNG.randint(len(ones)))]
                    q = zeros[int(_RNG.randint(len(zeros)))]
                    b[p], b[q] = b[q], b[p]
                cur_c = binary_c5(b, n)
                no_improve = 0

    return best, best_c


def random_perturb_h(h, mode):
    n = len(h)
    r = _RNG
    if mode == 0:
        return project_feasible(h + r.normal(0.0, 0.035, n), n)
    if mode == 1:
        sym = 0.5 * (h + h[::-1])
        return project_feasible(sym + r.normal(0.0, 0.018, n), n)
    if mode == 2:
        comp = 1.0 - h
        return project_feasible(comp + r.normal(0.0, 0.018, n), n)

    b = binary_from_h(h, n)
    for _ in range(5):
        ones = np.nonzero(b == 1)[0]
        zeros = np.nonzero(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        p = ones[int(r.randint(len(ones)))]
        q = zeros[int(r.randint(len(zeros)))]
        b[p], b[q] = b[q], b[p]
    return project_feasible(b.astype(np.float64), n)


def random_restart_segment(base_h, base_c, max_calls):
    best_h = base_h.copy()
    best_c = base_c
    n = len(base_h)

    for r in range(max_calls):
        if remaining_time() < 70.0:
            break
        hv = random_perturb_h(best_h, r % 4)
        c = compute_c5(hv)
        if c < best_c + 0.007:
            h2, c2 = refine_resolution(hv, n, min(55.0, max(20.0, remaining_time() - 20.0)))
            if c2 < best_c:
                best_h, best_c = h2, c2

    return best_h, best_c


def binary_phase(base_h, base_c):
    n0 = len(base_h)
    ns = set()
    for m in (160, 176, 192, 224, 256, n0):
        if 96 <= m <= 320 and (m % 2) == 0:
            ns.add(m)

    for m in sorted(ns, key=lambda x: abs(x - n0)):
        if remaining_time() < 90.0:
            break
        if (m % 2) != 0:
            continue
        hm = interp_to_len(base_h, m)
        b = binary_from_h(hm, m)
        b2, bc2 = binary_local_search(b, m, min(45.0, max(10.0, remaining_time() - 45.0)))
        hb = b2.astype(np.float64)
        c = compute_c5(hb)
        if c < base_c + 0.004:
            h2, c2 = refine_resolution(hb, m, min(80.0, max(20.0, remaining_time() - 25.0)))
            if c2 < base_c:
                return h2, c2
    return base_h, base_c


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S, _RNG
    _START_TIME = time.time()
    _BUDGET_S = float(budget_s)
    _RNG = np.random.RandomState(seed)

    if _RNG is None:
        _RNG = np.random.RandomState(seed)

    try:
        best_h = np.asarray(initial_h_values, dtype=float).copy()
        best_n = int(len(best_h))
    except Exception:
        best_n = 160
        best_h = fallback_h(best_n)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)

    if remaining_time() > 60.0:
        h1, c1 = refine_resolution(best_h, best_n, min(320.0, max(30.0, remaining_time())))
        if c1 < best_c:
            best_h, best_c = h1, c1

    if remaining_time() > 80.0:
        h2, c2 = variant_refine(best_h, best_c, min(100.0, remaining_time()))
        if c2 < best_c:
            best_h, best_c = h2, c2

    if remaining_time() > 150.0:
        h3, c3 = refine_resolution(best_h, best_n, min(180.0, max(40.0, remaining_time())))
        if c3 < best_c:
            best_h, best_c = h3, c3

    cands = make_candidates(best_n)
    if remaining_time() > 60.0:
        for m in cands:
            if remaining_time() < 55.0:
                break
            if m == best_n:
                continue
            per = min(70.0, max(35.0, (remaining_time() - 40.0) / max(1.0, float(len(cands)))))
            hm = interp_to_len(best_h, m)
            h4, c4 = refine_resolution(hm, m, per)
            if c4 < best_c:
                best_h, best_c, best_n = h4, c4, int(m)

    if remaining_time() > 140.0:
        h5, c5 = refine_resolution(best_h, best_n, min(200.0, max(50.0, remaining_time())))
        if c5 < best_c:
            best_h, best_c = h5, c5

    if remaining_time() > 120.0:
        h6, c6 = random_restart_segment(best_h, best_c, 5)
        if c6 < best_c:
            best_h, best_c = h6, c6

    if remaining_time() > 150.0:
        h7, c7 = binary_phase(best_h, best_c)
        if c7 < best_c:
            best_h, best_c, best_n = h7, c7, int(len(h7))

    final_plans = ((2500, 60), (5000, 45), (10000, 30))
    need = 35.0 if best_n <= 220 else (55.0 if best_n <= 320 else 85.0)
    for beta, req in final_plans:
        if remaining_time() < need:
            break
        h8, c8 = smooth_slsqp(best_h, best_n, beta, req)
        if c8 < best_c:
            best_h, best_c = h8, c8

    if remaining_time() > 15.0:
        h9, c9 = active_descent_limited(best_h, best_n, max_iter=80, time_limit_s=min(14.0, remaining_time()))
        if c9 < best_c:
            best_h, best_c = h9, c9

    if remaining_time() > 25.0:
        vs1 = project_feasible(0.5 * (best_h + best_h[::-1]), best_n)
        vs2 = project_feasible(1.0 - best_h, best_n)
        for hv in (vs1, vs2):
            if remaining_time() < 10.0:
                break
            c = compute_c5(hv)
            if c < best_c:
                best_h, best_c = hv, c
                if remaining_time() > 20.0:
                    h10, c10 = active_descent_limited(best_h, best_n, max_iter=30, time_limit_s=min(8.0, remaining_time() - 6.0))
                    if c10 < best_c:
                        best_h, best_c = h10, c10

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        ev = evaluate_erdos_solution(best_h)
        if ev is not None:
            ev = float(ev)
            if np.isfinite(ev) and ev < final_c5:
                final_c5 = ev
    except Exception:
        pass

    return (best_h, final_c5, int(best_n))
```
