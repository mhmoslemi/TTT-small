                if remaining_time() < 8:
                    break
                args = (n_base, k, target, beta, 200.0)
                try:
                    res = minimize(topk_obj, h_polish, jac=topk_grad,
                                   method='L-BFGS-B', bounds=bounds_160,
                                   args=args,
                                   options={'maxiter': 1000, 'ftol': 1e-16,
                                            'gtol': 1e-12, 'iprint': 0, 'maxls': 30})
                    h_new = project_feasible(res.x, n_base)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h_polish = h_new.copy()
                        print("  Polish k={}, beta={}: C5={:.7f}".format(k, beta, c5_new))
                except Exception:
                    pass

    # Phase 5: Multi-resolution - interpolate to higher n and optimize
    print("\n=== Phase 5: Multi-resolution ===")
    for n_hi in [240, 320, 400]:
        if remaining_time() < 40:
            break
        target_hi = n_hi / 2.0
        bounds_hi = [(0.0, 1.0)] * n_hi

        h_interp = interp_to(best_h, best_n, n_hi)
        h_interp = project_feasible(h_interp, n_hi)
        c5_interp = compute_c5(h_interp)

        improved = False
        for k in [4, 6]:
            if remaining_time() < 15:
                break
            for beta in [300, 800]:
                if remaining_time() < 8:
                    break
                args = (n_hi, k, target_hi, beta, 200.0)
                try:
                    res = minimize(topk_obj, h_interp, jac=topk_grad,
                                   method='L-BFGS-B', bounds=bounds_hi,
                                   args=args,
                                   options={'maxiter': 500, 'ftol': 1e-15,
                                            'gtol': 1e-10, 'iprint': 0, 'maxls': 25})
                    h_new = project_feasible(res.x, n_hi)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        best_n = n_hi
                        h_interp = h_new.copy()
                        improved = True
                        print("  n={}, k={}, beta={}: C5={:.7f}".format(n_hi, k, beta, c5_new))
                except Exception:
                    pass

        # Binary SA at higher resolution
        if remaining_time() > 30:
            h_bin_hi = np.where(h_interp > 0.5, 1.0, 0.0)
            h_bin_hi = project_feasible(h_bin_hi, n_hi)
            h_bin_hi2, c5_bin_hi = binary_sa(h_bin_hi, n_hi, 3000, rng, t_start=0.003, t_end=0.0001)
            if c5_bin_hi < best_c5:
                best_c5 = c5_bin_hi
                best_h = h_bin_hi2.copy()
                best_n = n_hi
                print("  n={} binary SA: C5={:.7f}".format(n_hi, c5_bin_hi))

    # Phase 6: Parametric optimization (sinusoidal)
    print("\n=== Phase 6: Parametric ===")
    if remaining_time() > 40:
        n_param = best_n
        n_freqs = 3
        n_params = 3 * n_freqs
        bounds_param = []
        for kf in range(n_freqs):
            bounds_param.append((0.05, 0.45))  # amplitude
            bounds_param.append((0.5, 15.0))   # frequency
            bounds_param.append((0.0, 2.0 * np.pi))  # phase

        # Try a few initial random parameter sets
        best_param_c5 = compute_c5(best_h)
        for trial in range(4):
            if remaining_time() < 15:
                break
            p0 = np.array([rng.uniform(0.1, 0.4) for _ in range(n_freqs)] +
                          [rng.uniform(1, 10) for _ in range(n_freqs)] +
                          [rng.uniform(0, 2 * np.pi) for _ in range(n_freqs)])
            # Reorder: [amp1, freq1, phase1, amp2, freq2, phase2, ...]
            p0 = np.array([0.2, 3.0, 0.5, 0.15, 7.0, 1.5, 0.1, 11.0, 4.0])
            p0[rng.randint(n_params)] += rng.uniform(-0.1, 0.1)

            args_p = (n_param, n_freqs)
            try:
                res = minimize(parametric_obj, p0, method='Nelder-Mead',
                               args=args_p,
                               options={'maxiter': 500, 'xatol': 1e-6, 'fatol': 1e-10,
                                        'adaptive': True})
                h_p = parametric_h(res.x, n_param, n_freqs)
                h_p = project_feasible(h_p, n_param)
                c5_p = compute_c5(h_p)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p.copy()
                    best_n = n_param
                    print("  Parametric: C5={:.7f}".format(c5_p))
            except Exception:
                pass

    # Phase 7: Coordinate descent on best solution
    print("\n=== Phase 7: Coordinate descent ===")
    if remaining_time() > 20:
        h_cd, c5_cd = coordinate_descent(project_feasible(best_h, best_n), best_n, 30, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            print("  Coord descent: C5={:.7f}".format(c5_cd))

    # Phase 8: Final high-beta polish
    print("\n=== Phase 8: Final polish ===")
    if remaining_time() > 15:
        n_final = best_n
        target_final = n_final / 2.0
        bounds_final = [(0.0, 1.0)] * n_final
        h_final = project_feasible(best_h, n_final)

        for beta in [2000, 5000]:
            if remaining_time() < 8:
                break
            for k in [3, 5]:
                if remaining_time() < 5:
                    break
                args = (n_final, k, target_final, beta, 500.0)
                try:
                    res = minimize(topk_obj, h_final, jac=topk_grad,
                                   method='L-BFGS-B', bounds=bounds_final,
                                   args=args,
                                   options={'maxiter': 1500, 'ftol': 1e-17,
                                            'gtol': 1e-12, 'iprint': 0, 'maxls': 30})
                    h_new = project_feasible(res.x, n_final)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h_final = h_new.copy()
                        print("  Final k={}, beta={}: C5={:.7f}".format(k, beta, c5_new))
                except Exception:
                    pass

        # Full smooth max final
        if remaining_time() > 10:
            args = (n_final, target_final, 5000, 500.0)
            try:
                res = minimize(full_smooth_obj, h_final, jac=full_smooth_grad,
                               method='L-BFGS-B', bounds=bounds_final,
                               args=args,
                               options={'maxiter': 800, 'ftol': 1e-17,
                                        'gtol': 1e-12, 'iprint': 0, 'maxls': 30})
                h_new = project_feasible(res.x, n_final)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  Full smooth final: C5={:.7f}".format(c5_new))
            except Exception:
                pass

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, best_c5))
    return (best_h, best_c5, best_n)
