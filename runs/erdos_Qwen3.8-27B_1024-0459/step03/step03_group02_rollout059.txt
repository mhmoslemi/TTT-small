    corr = compute_corr(h)
    dx = 2.0 / n
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.minimum(shifted, 0.0)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    val = (m + np.log(np.sum(exp_shifted)) / beta) * dx
    
    # Gradient: only compute for lags with significant weight
    grad = np.zeros(n)
    # Get top lags by weight
    top_k = np.where(w > 1e-4)[0]
    for k_idx in top_k:
        lag = k_idx - (n - 1)
        if lag == 0:
            # c = sum h[i]*(1-h[i])
            grad += w[k_idx] * (1 - 2*h)
        elif lag > 0:
            # c = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
            n_pairs = n - lag
            # d/dc/dh[i] for i in 0..n-1-lag-1: contributes (1-h[i+lag])
            # d/dc/dh[i] for i in lag..n-1: contributes -h[i-lag]
            term_a = np.zeros(n)
            term_b = np.zeros(n)
            term_a[:n_pairs] = 1.0 - h[lag:lag+n_pairs]
            term_b[lag:] = -h[:n_pairs]
            grad += w[k_idx] * (term_a + term_b)
        else:
            # lag < 0, let lag2 = -lag
            lag2 = -lag
            n_pairs = n - lag2
            # c = sum_{i=lag2}^{n-1} h[i]*(1-h[i-lag2])

            term_b[:n_pairs] = -h[lag2:lag2+n_pairs]
            grad += w[k_idx] * (term_a + term_b)
    
    grad *= dx
    return val, grad

                    best_h = h_t.copy()
                    h = h_t.copy()
                    improved = True
                    break
            if not improved and it % 200 == 100:
                h_new = slsqp_analytic(h, n, 1000, maxiter=200)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()

    # Phase 5: Micro top-lag optimization
    if remaining_time() > 30:
        h_m, c5_m = micro_top_lag(best_h, n, max_iter=200)
        if c5_m < best_c5:
            best_c5 = c5_m
            best_h = h_m.copy()

    # Phase 6: Restart from perturbations
    if remaining_time() > 80:
        for restart_idx in range(4):
            if remaining_time() < 20:
                break
            h_p = perturbed_start(best_h, n, rng, scale=0.05 * (restart_idx + 1))
            h_new = slsqp_analytic(h_p, n, 300, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_new = slsqp_analytic(h_p, n, 2000, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        best_h = slsqp_analytic(best_h, n, 5000, maxiter=400)
        c5_new = compute_c5(best_h)
        if c5_new < best_c5:
            best_c5 = c5_new

    # Phase 7: Resolution exploration
    for trial_n in [128, 192, 256, 320, 480]:
        if remaining_time() < 30:
            break
        h_interp = interp_to(best_h, best_n, trial_n)
        h_interp = project_feasible(h_interp, trial_n)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5 + 0.003:
            h_new = slsqp_analytic(h_interp, trial_n, 300, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = trial_n
            h_new = slsqp_analytic(h_new, trial_n, 2000, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = trial_n

    # Phase 8: Final high-precision SLSQP
    if remaining_time() > 10:
        for beta in [5000, 10000, 20000]:
            if remaining_time() < 5:
                break
            h_new = slsqp_analytic(best_h, best_n, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 9: Final micro polish
    if remaining_time() > 10:
        h_m, c5_m = micro_top_lag(best_h, best_n, max_iter=100)
        if c5_m < best_c5:
            best_c5 = c5_m
            best_h = h_m.copy()

    # Final projection
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
