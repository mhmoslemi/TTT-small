    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-15
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return h

def compute_correlation(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def fast_coord_descent(h, n, sweeps=100, time_limit=None, t0=None):

    C = compute_correlation(h, n)

            if time.time() - t0 > time_limit:
                break
        
        for j in range(n):
            a = min(j, n - 1 - j)
            old = h[j]
            
            # Unaffected max
            if a < n - 1:
                # Unaffected: indices in C where |k| > a
                # C index = k + (n-1), k from -(n-1) to n-1
                # |k| > a means k < -a or k > a
                # C indices: k + (n-1) < -a + (n-1) = n-1-a, or k + (n-1) > a + (n-1) = n+a
                left_max = np.max(C[:n-1-a]) if n-1-a > 0 else 0.0
                right_max = np.max(C[n+a+1:]) if n+a+1 < 2*n-1 else 0.0
                unaffected_max = max(left_max, right_max)
            else:
                unaffected_max = 0.0
            
            # Affected: slopes for k in [-a, a], k≠0
            n_affected = 2 * a
            if n_affected > 0:
                ks = np.arange(-a, a+1)
                ks = ks[ks != 0]
                fidxs = ks + (n-1)
                slopes = 1.0 - h[j + ks] - h[j - ks]
                C_aff = C[fidxs]
            else:
                slopes = np.array([])
                C_aff = np.array([])
            
            # Lag 0: C[0] part
            C0_base = C[n-1] - old * (1.0 - old)
            
            # 1D minimization
            # f(x) = max(unaffected_max, max_k(C_aff[k] + (x-old)*slopes[k]), C0_base + x*(1-x))
            
            # Grid
            xs_grid = np.linspace(0.0, 1.0, 101)
            best_x = old
            best_val = float('inf')
            
            for x in xs_grid:
                val = unaffected_max
                if len(slopes) > 0:
                    aff_vals = C_aff + (x - old) * slopes
                    val = max(val, np.max(aff_vals))
                val0 = C0_base + x * (1.0 - x)
                val = max(val, val0)
                if val < best_val:
                    best_val = val
                    best_x = x

            # Find interval around best_x
            idx = np.argmin(np.abs(xs_grid - best_x))
            lo = max(0.0, xs_grid[max(0, idx-1)])
            hi = min(1.0, xs_grid[min(100, idx+1)])
            
            golden = (np.sqrt(5) - 1) / 2  # ~0.618
            for _ in range(60):

                m2 = lo + golden * (hi - lo)

                    val = unaffected_max
                    if len(slopes) > 0:
                        aff_vals = C_aff + (x - old) * slopes
                        val = max(val, np.max(aff_vals))
                    val0 = C0_base + x * (1.0 - x)
                    return max(val, val0)

                f2 = f_val(m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            
            best_x = (lo + hi) / 2.0

            h[j] = best_x
            delta = best_x - old
            if abs(delta) > 1e-16:
                C[n-1] += best_x * (1 - best_x) - old * (1 - old)
                if a > 0:

                        if k_idx == 0:

            h = h_proj
            C = compute_correlation(h, n)
        c5 = np.max(C) * (2.0 / n)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best = h.copy()
    
    return best, best_c5

def smooth_max_grad(h, n, beta, dx):
    """Compute gradient of smooth max of correlation wrt h."""
    C = compute_correlate(h, n)
    m = np.max(C)
    # Softmax weights
    exp_c = np.exp(beta * (C - m))
    w = exp_c / np.sum(exp_c)
    
    # Gradient of C[k] wrt h[j]
    # C[k] = sum_i h[i]*(1-h[i+k])
    # dC[k]/dh[j] = (1-h[j+k]) - h[j-k] (for valid indices, k≠0)
    # dC[0]/dh[j] = 1 - 2*h[j]
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        if w[k + n-1] < 1e-10:
            continue
        wk = w[k + n-1]
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            # dC[k]/dh[j] = (1-h[j+k]) if j+k < n, plus -h[j-k] if j-k >= 0
            valid_plus = slice(0, n-k) if k > 0 else slice(0, n)
            valid_minus = slice(-k, n) if k > 0 else slice(k, n)
            # Actually: for lag k>0:
            #   term h[j]*(1-h[j+k]) for j=0..n-1-k: grad = (1-h[j+k])
            #   term h[j-k]*(1-h[j]) for j=k..n-1: grad = -h[j-k]
            if k > 0:
                grad[:n-k] += wk * (1.0 - h[k:])
                grad[k:] += wk * (-h[:n-k])
            else:
                ko = -k
                grad[ko:] += wk * (-h[:n-ko])
                grad[:n-ko] += wk * (1.0 - h[ko:])
    
    return grad * dx

    best_h = None
    best_c5 = float('inf')
    best_n = 0
    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn > 10:
            best_h = ph.copy()
            best_c5 = compute_c5(ph)
            best_n = pn
    except Exception:
        pass

    if best_h is None:
        best_n = 160
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)

    target_ns = [200, 256, 320, 400]

    for n in target_ns:
        if time_left() < 90:
            break

        if n == best_n:
            h_init = project_feasible(best_h, n)
        else:
            h_init = interpolate_h(best_h, best_n, n)

        inits = [h_init.copy()]
        for s in [0.02, 0.05, 0.1]:
            inits.append(project_feasible(h_init + s * rng.randn(n), n))

        for h0 in inits:
            if time_left() < 30:
                break

            # Phase 1: Coordinate descent
            tl1 = min(time_left() * 0.4, 250.0)
            h_cd, c5_cd = fast_coord_descent(h0, n, sweeps=300, time_limit=tl1, t0=t0, rng=rng)

            # Phase 2: Smooth gradient
            if time_left() > 30:
                tl2 = min(time_left() * 0.25, 120.0)
                h_sg, c5_sg = smooth_grad_descent(h_cd, n, time_limit=tl2, t0=t0)
                if c5_sg < c5_cd:
                    h_cd, c5_cd = h_sg, c5_sg

            # Phase 3: More coordinate descent
            if time_left() > 20:
                tl3 = min(time_left() * 0.3, 150.0)
                h_cd2, c5_cd2 = fast_coord_descent(h_cd, n, sweeps=200, time_limit=tl3, t0=t0, rng=rng)
                if c5_cd2 < c5_cd:
                    h_cd, c5_cd = h_cd2, c5_cd2

            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n

    # Final polish rounds
    for _ in range(3):
        if time_left() < 15:
            break
        h_pol, c5_pol = fast_coord_descent(best_h, best_n, sweeps=300, time_limit=time_left() * 0.4, t0=t0, rng=rng)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    if time_left() > 20:
        h_sg, c5_sg = smooth_grad_descent(best_h, best_n, time_limit=time_left() * 0.4, t0=t0)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    if time_left() > 15:
        h_pol, c5_pol = fast_coord_descent(best_h, best_n, sweeps=200, time_limit=time_left() * 0.5, t0=t0, rng=rng)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 is not None:
            final_c5 = min(final_c5, ec5)
    except Exception:
        pass

    return (best_h, final_c5, best_n)
