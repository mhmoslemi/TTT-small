def active_weights(q, top, beta):
    order = np.argsort(q)[::-1]
    top = min(top, len(q))
    idxs = order[:top]
    m = q[idxs].max()
    w = np.exp(beta*(q[idxs]-m))
    s = np.sum(w)
    if s <= 0: w = np.ones(len(idxs)); s=len
    w /= s
    return idxs, w

def project_sum_clip(x, target):
    x = np.asarray(x, dtype=float).copy()
    n = x.size
    if n == 0: return x
    lo = -float(np.max(x))
    hi = 1.0 - float(np.min(x))
    for _ in range(70):
       mid = 0.5*(lo+hi)
       s = np.sum(np.clip(x + mid, 0.0, 1.0))
       if s < target:
           lo = mid
       else:
           hi = mid
    h = np.clip(x + 0.5*(lo+hi), 0.0, 1.0)
    # fix residual
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-14:
       mask = (h > 1e-10) & (h < 1.0 - 1e-10)
       cnt = int(np.sum(mask))
       if cnt > 0:
           h[mask] += diff / cnt
       else:
           h[:] += diff / n
       h = np.clip(h, 0.0, 1.0)
       diff = target - float(np.sum(h))
       if abs(diff) > 1e-14:
           # adjust all slightly
           h[:] += diff / n
           h = np.clip(h,0,1)
    return h

def project_sum_clip(x, target):
    x = np.asarray(x, dtype=float).copy()
    n = len(x)
    if n == 0:
        return x
    lo = -float(np.max(x))
    hi = 1.0 - float(np.min(x))
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - float(np.sum(h))
        if abs(diff) > 1e-14:
            h[:] += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(src, n):
    m = len(src)
    if m == n:
        return np.asarray(src, dtype=float).copy()
    if m == 0:
        return np.zeros(n, dtype=float)
    if m == 1:
        return np.full(n, float(src[0]), dtype=float)
    xp = (np.arange(m) + 0.5) * (2.0 / m)
    xn = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(xn, xp, np.asarray(src, dtype=float))


def direct_q(h):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode="full")

def gradient_from_indices(h, idxs, weights):
    n = len(h)
    off = n - 1
    G = np.zeros(n, dtype=float)
    if len(idxs) == 0:
        return G
    if weights is None:
        inv = 1.0 / float(len(idxs))
        wvals = None
    else:
        inv = 1.0
        wvals = np.asarray(weights, dtype=float)
    for ki, idx in enumerate(idxs):
        w = 1.0 if wvals is None else float(wvals[ki])
        if w == 0.0:
            continue
        t = off - int(idx)
        scale = w * inv
        if t > 0:
            m = n - t
            if m > 0:
                G[:m] += scale * (1.0 - h[t:t + m])
                G[t:] -= scale * h[:m]
        elif t < 0:
            r = -t
            m = n - r
            if m > 0:
                G[r:] += scale * (1.0 - h[:m])
                G[:m] -= scale * h[r:]
        else:
            G += scale * (1.0 - 2.0 * h)
    return G


def pair_delta_cont(h, i, j, tas, n):
    A = np.zeros(len(tas), dtype=float)
    ti = i + tas
    mi = (ti >= 0) & (ti < n)
    if np.any(mi):
        A[mi] -= (1.0 - h[ti[mi]])
    tj = j + tas
    mj = (tj >= 0) & (tj < n)
    if np.any(mj):
        A[mj] += (1.0 - h[tj[mj]])
    si = i - tas
    mis = (si >= 0) & (si < n)
    if np.any(mis):
        A[mis] += h[si[mis]]
    sj = j - tas
    mjs = (sj >= 0) & (sj < n)
    if np.any(mjs):
        A[mjs] -= h[sj[mjs]]
    B = np.zeros(len(tas), dtype=float)
    B[n - 1] = -2.0
    t1 = j - i
    if -n + 1 <= t1 <= n - 1:
        B[n - 1 - t1] += 1.0
    t2 = i - j
    if -n + 1 <= t2 <= n - 1:
        B[n - 1 - t2] += 1.0
    return A, B


def try_cont_pair(h, q, i, j, tas, curmax, deriv):

    s = float(np.sum(h) - float(st["target"]))
    val += float(st["pen"]) * s * s
    return float(val)


def smooth_jac(x):
    st = _GSM
    h = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)

def bits_from_h(frac_h, n, rng, noise):
    if len(frac_h) != n:
        frac_h = interpolate_to(frac_h, n)
    vals = np.asarray(frac_h, dtype=float).copy()
    if noise > 0.0:
        vals = vals + rng.normal(0.0, noise, n)
    bits = np.zeros(n, dtype=np.int32)
    idx = np.argsort(vals)[::-1]
    bits[idx[:n // 2]] = 1
    return bits


def random_bits(n, rng):
    bits = np.zeros(n, dtype=np.int32)
    idx = rng.choice(n, size=n // 2, replace=False)
    bits[idx] = 1
    return bits


def structured_bits(n, alpha, phase):
    x = np.arange(n, dtype=float)
    vals = np.sin(2.0 * math.pi * alpha * x + phase)
    vals += 0.5 * np.sin(4.0 * math.pi * alpha * x + 2.0 * phase)
    bits = np.zeros(n, dtype=np.int32)
    idx = np.argsort(vals)[::-1]
    bits[idx[:n // 2]] = 1
    return bits


def apply_binary_swap(bits, one, ones, zeros, i, j, ri, rj):
    bits[i] = 0
    bits[j] = 1
    one[i] = 1
    one[j] = 0
    ones[ri] = ones[-1]
    ones.pop()
    zeros[rj] = zeros[-1]
    zeros.pop()
    zeros.append(i)
    ones.append(j)

        one[i] = 1
        one[j] = 0
        corr = np.correlate(bits, one, mode="full")
        nm = int(np.max(corr))
        deltaC = (nm - curmax) * factor
        accept = False
        if deltaC <= 1e-15:
            accept = True
        else:
            ratio = deltaC / T
            if ratio < 50.0 and rng.random() < math.exp(-ratio):
                accept = True

        if accept:
            curmax = nm
            noimp = 0
            if curmax < bestmax:
                bestmax = curmax
                bestbits = bits.copy()
            apply_binary_swap(bits, one, ones, zeros, i, j, ri, rj)
        else:
            bits[i] = 1
            bits[j] = 0
            one[i] = 0
            one[j] = 1
            noimp += 1

        steps += 1
        if steps % 1000 == 0:
            T = max(floorT, T * 0.99)

        if noimp > 3000:
            T = max(0.015, floorT)
            noimp = 0
            for _ in range(10):
                if time.time() >= deadline:
                    break
                if len(ones) == 0 or len(zeros) == 0:
                    break
                ri = int(rng.randint(len(ones)))
                rj = int(rng.randint(len(zeros)))
                i = ones[ri]
                j = zeros[rj]
                bits[i] = 0
                bits[j] = 1
                one[i] = 1
                one[j] = 0
                corr = np.correlate(bits, one, mode="full")
                nm = int(np.max(corr))
                curmax = nm
                if curmax < bestmax:
                    bestmax = curmax
                    bestbits = bits.copy()
                apply_binary_swap(bits, one, ones, zeros, i, j, ri, rj)

    return bestbits, bestmax


def binary_search_multistart(base_h, n, deadline, seed):
    rng = np.random.RandomState(seed)
    n = int(n)
    if n % 2 != 0:
        n -= 1
    if n < 80:
        n = 160
    seeds = []
    for noise in (1e-6, 0.02, 0.08, 0.2):
        seeds.append(bits_from_h(base_h, n, rng, noise))
    seeds.append(random_bits(n, rng))
    seeds.append(structured_bits(n, 0.6180339887, 0.0))
    seeds.append(structured_bits(n, 0.314, 1.3))

    unique = []
    seen = set()
    for b in seeds:
        key = b.tobytes()
        if key not in seen:
            seen.add(key)
            unique.append(b)
    seeds = unique
    if not seeds:
        return None, 10 ** 9

    bestbits = None
    bestmax = 10 ** 9
    mnum = len(seeds)
    for k, b in enumerate(seeds):
        if time.time() >= deadline:
            break
        remaining = float(deadline) - time.time()
        if remaining < 30.0:
            break
        slice_deadline = time.time() + max(30.0, remaining / float(mnum - k))
        slice_deadline = min(slice_deadline, float(deadline))
        bb, bm = binary_sa(b, slice_deadline, seed + 100 * (k + 1))
        if bm < bestmax:
            bestmax = int(bm)
            bestbits = bb.astype(float)
    return bestbits, bestmax


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + max(10.0, float(budget_s) - 20.0)

    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
    except Exception:
        rng_tmp = np.random.RandomState(int(seed))
        parent = rng_tmp.normal(0.5, 0.1, 160)

    n_par = len(parent)
    if n_par < 80 or n_par % 2 != 0:
        n_par = 2 * max(40, n_par // 2)
        parent = interpolate_to(parent, n_par)
    target_par = n_par / 2.0
    parent = project_sum_clip(parent, target_par)

    best_h = parent.copy()
    best_n = int(n_par)
    best_c = c5_from_q(direct_q(best_h), best_h)

    if float(deadline) - time.time() > 80.0:
        smooth_dead = min(deadline - 20.0, time.time() + min(180.0, 0.18 * float(budget_s)))
        try:
            h_sl, c_sl = refine_smooth(best_h, best_n / 2.0, smooth_dead, int(seed) + 1)
            if c_sl < best_c:
                best_c = c_sl
                best_h = h_sl
                best_n = len(h_sl)
        except Exception:
            pass

    if float(deadline) - time.time() > 80.0:
        cont_dead = min(deadline - 30.0, time.time() + min(360.0, 0.28 * float(budget_s)))
        try:
            h_ol, c_ol = continuous_polish(best_h, best_n / 2.0, cont_dead, int(seed) + 2)
            if c_ol < best_c:
                best_c = c_ol
                best_h = h_ol
                best_n = len(h_ol)
        except Exception:
            pass

    if float(deadline) - time.time() > 100.0:
        bin_dead = min(deadline - 40.0, time.time() + max(150.0, 0.52 * float(budget_s)))
        try:
            bb, bm = binary_search_multistart(best_h, best_n, bin_dead, int(seed) + 3)
            if bb is not None:
                n_bin = len(bb)
                c_bin = float(2.0 * int(bm)) / float(n_bin)
                if c_bin < best_c:
                    best_c = c_bin
                    best_h = project_sum_clip(bb, n_bin / 2.0)
                    best_n = n_bin
        except Exception:
            pass

    if float(deadline) - time.time() > 220.0 and best_c > 0.38085:
        rng_alt = np.random.RandomState(int(seed) + 900)
        n_alt = 192 if best_n <= 160 else 256
        if n_alt % 2 != 0:
            n_alt -= 1
        if n_alt < 80:
            n_alt = 160
        try:
            h_alt = project_sum_clip(interpolate_to(best_h, n_alt), n_alt / 2.0)
            alt_cont_dead = min(deadline - 80.0, time.time() + 0.18 * (deadline - time.time()))
            h_alt, c_alt = continuous_polish(h_alt, n_alt / 2.0, alt_cont_dead, int(seed) + 7)
            if c_alt < best_c:
                best_c = c_alt
                best_h = h_alt
                best_n = n_alt
        except Exception:
            pass

    if float(deadline) - time.time() > 120.0:
        rng_alt = np.random.RandomState(int(seed) + 901)
        try:
            n_try = best_n
            if n_try % 2 != 0:
                n_try -= 1
            if n_try < 80:
                n_try = 160
            b_try = bits_from_h(best_h, n_try, rng_alt, 0.03)
            deadline2 = min(deadline - 25.0, time.time() + max(60.0, 0.15 * (deadline - time.time())))
            bb, bm = binary_sa(b_try, deadline2, int(seed) + 8)
            c_bin = float(2.0 * int(bm)) / float(len(bb))
            if c_bin < best_c:
                best_c = c_bin
                best_h = project_sum_clip(bb.astype(float), len(bb) / 2.0)
                best_n = len(bb)
        except Exception:
            pass

    best_h = project_sum_clip(best_h, best_n / 2.0)
    q = direct_q(best_h)
    final_c = c5_from_q(q, best_h)

    try:
        ev = evaluate_erdos_solution(best_h)
        ev = float(ev)
        if math.isfinite(ev) and ev < final_c:
            final_c = ev
    except Exception:
        pass

    final_c = float(final_c)
    if not math.isfinite(final_c):
        final_c = c5_from_q(direct_q(parent), parent)

    return (best_h, final_c, int(best_n))
