def swap_eval(P,N,h,a,b, dP,dN,b1,b2,n):
    dP.fill(0.0); dN.fill(0.0)
    # a 1->0
    p=a
    if p+1 < n:
        dP[1:n-p] -= (1.0 - h[p+1:n])
        dN[1:n-p] += h[p+1:n]
    if p > 0:
        rev = h[p-1::-1]
        dP[1:p+1] += rev
        dN[1:p+1] -= (1.0 - rev)
    # b 0->1
    p=b
    if p+1 < n:
        dP[1:n-p] += (1.0 - h[p+1:n])
        dN[1:n-p] -= h[p+1:n]
    if p > 0:

    np.add(P,dP,out=b1); np.add(N,dN,out=b2)
    return max(b1.max(), b2.max())

def project_feasible(h,n):
    h=np.array(h,dtype=float).copy()
    np.clip(h,0.0,1.0,out=h)
    target=0.5*n
    for _ in range(30):
       diff=target-h.sum()
       if abs(diff)<1e-14: break
       if diff>0:
           room=1.0-h
           room[room<1e-15]=0.0
           tot=room.sum()
           if tot <= 1e-15:
               h[:] = target/n
               break
           inc=(diff/tot)*room
           h=np.clip(h+inc,0.0,1.0)
       else:
           avail=h
           avail[avail<1e-15]=0.0
           tot=avail.sum()
           if tot <=1e-15:
               h[:]=target/n; break
           dec=((-diff)/tot)*avail

import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

_TGT = 0.0

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME))


def pair_key(x):
    return x[0]


def compute_c5(h):
    arr = np.asarray(h, dtype=float)
    n = len(arr)
    if n <= 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)

    for _ in range(50):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            room[room < 1e-15] = 0.0
            tot = float(np.sum(room))
            if tot <= 1e-15:
                h[:] = target / float(n)
                break
            inc = (diff / tot) * room
            h = np.clip(h + inc, 0.0, 1.0)
        else:
            avail = h
            avail[avail < 1e-15] = 0.0
            tot = float(np.sum(avail))
            if tot <= 1e-15:
                h[:] = target / float(n)
                break
            dec = (-diff / tot) * avail
            h = np.clip(h - dec, 0.0, 1.0)

    for _ in range(8):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-12:
            break
        unsat = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if unsat.size > 0:
            h[unsat] += diff / float(unsat.size)
            h = np.clip(h, 0.0, 1.0)
        else:
            if diff > 0.0:
                idx = int(np.argmax(1.0 - h))
            else:
                idx = int(np.argmax(h))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def interp_to(src_h, dst_n):
    src_h = np.asarray(src_h, dtype=float)
    src_n = len(src_h)
    if src_n == dst_n:
        return src_h.copy()
    if src_n <= 0 or dst_n <= 0:
        return np.zeros(dst_n, dtype=float)
    xp = (np.arange(src_n) + 0.5) * (2.0 / float(src_n))
    xq = (np.arange(dst_n) + 0.5) * (2.0 / float(dst_n))
    return np.interp(xq, xp, src_h)


def binary_objects(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    P = corr[n - 1:].astype(float).copy()
    N = np.empty(n, dtype=float)
    N[0] = P[0]
    if n > 1:
        N[1:] = corr[:n - 1][::-1].astype(float)
    return P, N


def binary_count_max(h):
    P, N = binary_objects(h)
    M = max(float(np.max(P)), float(np.max(N)))
    n = len(h)
    return (2.0 * M) / float(n), M


def to_binary(v, n, rng):
    v = np.asarray(v, dtype=float)
    if len(v) != n:
        v = interp_to(v, n)
    target = int(n // 2)
    idx = np.argsort(-v, kind="mergesort")
    out = np.zeros(n, dtype=float)
    out[idx[:target]] = 1.0
    return out


def make_random_binary(n, rng):

    target = n // 2
    if target > 0:

def make_run_seq(n, rng):
    if n % 2 == 1:
        n = n - 1
    target = n // 2
    max_runs = max(12, int(4.0 * np.sqrt(max(n, 16))))
    max_runs = min(max_runs, max(16, n // 4), 90)
    r = int(rng.randint(10, max_runs + 1))
    r = min(r, max(2, n))
    cuts = np.sort(rng.choice(n, size=r - 1, replace=False))
    lengths = np.diff(np.concatenate(([0], cuts, [n])))

    h = np.zeros(n, dtype=float)
    bit = int(rng.randint(2))
    pos = 0
    for L in lengths:
        L = int(L)
        if L > 0:
            h[pos:pos + L] = float(bit)
            pos += L
        bit = 1 - bit

    cnt = int(np.sum(h))
    if cnt > target:
        ones = np.where(h == 1.0)[0]
        remove = ones[rng.choice(cnt - target, size=cnt - target, replace=False)]
        h[remove] = 0.0

        zeros = np.where(h == 0.0)[0]
        add = zeros[rng.choice(target - cnt, size=target - cnt, replace=False)]
        h[add] = 1.0
    return h

def smooth_box(h, width):

    n = len(h)

    kernel = np.ones(w, dtype=float) / float(w)
    y = np.convolve(h, kernel, mode="same")
    return np.clip(y, 0.0, 1.0)

        s = make_sine(n, f, float(rng.uniform(0.0, 6.283185307179586)), 0.40)
        starts.append(to_binary(s, n, rng))

    starts.append(make_random_binary(n, rng))

    scored = []
    for h in starts:
        c, _ = binary_count_max(h)
        scored.append((c, h))
    scored.sort(key=pair_key)
    if len(scored) > 8:
        scored = scored[:8]
    return scored


def lbfgs_obj(x):
    n = _N
    a = np.asarray(x, dtype=float)
    corr = np.correlate(a, 1.0 - a, mode="full")
    k = int(_TOPK)
    if k < len(corr) and k > 0:
        tv = np.partition(corr, -k)[-k:]
    else:
        tv = corr
    m = float(np.max(tv))
    beta = float(max(1.0, _BETA))
    z = beta * (tv - m)
    logsum = float(np.log(np.sum(np.exp(z))))
    sm = m + logsum / beta
    val = sm * (2.0 / float(n))
    if _PEN > 0.0:
        s = float(np.sum(a)) - float(_TGT)
        val += float(_PEN) * s * s
    return float(val)


def lbfgs_grad(x):
    n = _N
    a = np.asarray(x, dtype=float)
    corr = np.correlate(a, 1.0 - a, mode="full")
    k = int(_TOPK)
    if k < len(corr) and k > 0:
        idx = np.argpartition(corr, -k)[-k:]
    else:
        idx = np.arange(len(corr))
    tv = corr[idx]
    m = float(np.max(tv))
    beta = float(max(1.0, _BETA))
    z = beta * (tv - m)
    w = np.exp(z)
    ws = float(np.sum(w))
    if ws > 0.0:
        w = w / ws
    else:
        w = np.zeros_like(tv)

    grad = np.zeros(n, dtype=float)
    off = n - 1
    for jj ww in zip(idx, w):
        if ww < 1e-15:
            continue
        s = int(jj) - off
        if s == 0:
            grad += ww * (1.0 - 2.0 * a)
        elif s > 0:
            L = n - s
            grad[:L] += ww * (1.0 - a[s:])
            grad[s:] -= ww * a[:L]
        else:
            sp = -s
            L = n - sp
            grad[:L] -= ww * a[sp:]
            grad[sp:] += ww * (1.0 - a[:L])

    grad *= (2.0 / float(n))
    if _PEN > 0.0:
        grad += 2.0 * float(_PEN) * (float(np.sum(a)) - float(_TGT))
    return grad


def refine_continuous(x0, n, time_limit, rng):
    if x0 is None or n <= 0:
        return x0, float("inf")
    if time_limit <= 5.0 or remaining_time() <= 5.0:
        x = project_feasible(x0, n)
        return x, compute_c5(x)

    deadline = min(time.time() + float(time_limit), _START_TIME + _BUDGET_S - 2.0)
    x = project_feasible(x0, n)
    best = x.copy()
    best_c = compute_c5(x)

    global _N, _TGT, _BETA, _TOPK, _PEN
    _N = n
    _TGT = 0.5 * float(n)
    bounds = [(0.0, 1.0)] * n
    scale = max(1, min(2, int(n // 128) + 1))

    configs = [
        (10.0, 8, 1000.0, 160 * scale),
        (30.0, 12, 1000.0, 260 * scale),
        (80.0, 20, 3000.0, 360 * scale),
        (200.0, 32, 5000.0, 460 * scale),
        (50.0, min(90, 2 * n - 1), 5000.0, 560 * scale),
    ]

    for beta, k, pen, mx in configs:
        if time.time() > deadline or remaining_time() < 12:
            break
        _BETA = float(beta)
        _TOPK = int(max(4, min(int(k), 2 * n - 1)))
        _PEN = float(pen)

        x = project_feasible(x, n)
        try:
            mx = int(mx)
            maxfun = max(500, 8 * mx)
            if time.time() < deadline:
                rem_local = max(5.0, deadline - time.time())
                if rem_local < 30.0:
                    mx = min(mx, 100)
                    maxfun = min(maxfun, 700)
                elif rem_local < 90.0:
                    mx = min(mx, 240)
                    maxfun = min(maxfun, 1700)

                res = minimize(
                    lbfgs_obj,
                    x,
                    jac=lbfgs_grad,
                    method="L-BFGS-B",
                    bounds=bounds,
                    options={
                        "maxiter": mx,
                        "maxfun": maxfun,
                        "ftol": 1e-15,
                        "gtol": 1e-9,
                        "iprint": 0,
                        "maxls": 40,
                    },
                )
                if getattr(res, "x", None) is not None:
                    xn = project_feasible(res.x, n)
                    cn = compute_c5(xn)
                    if cn < best_c - 1e-14:
                        best = xn.copy()
                        best_c = cn
                        x = xn.copy()
        except Exception:
            pass

    return best, float(best_c)


def rotation_sweep(h, n):
    h = np.asarray(h, dtype=float)
    if n <= 0:
        return h.copy(), float("inf")
    best_c = compute_c5(h)
    best_r = h.copy()
    for sh in range(1, n):
        r = np.roll(h, sh)
        c = compute_c5(r)
        if c < best_c - 1e-15:
            best_c = c
            best_r = r.copy()
    return best_r, float(best_c)


def consider_solution(h, n, best_h, best_c, best_n):
    if h is None:
        return best_h, best_c, best_n
    h = np.asarray(h, dtype=float)
    if len(h) != n:
        h = interp_to(h, n)
    h = project_feasible(h, n)
    c = compute_c5(h)
    if c < best_c - 1e-12:
        return h.copy(), float(c), int(n)
    return best_h, best_c, best_n


def swap_delta_scratch(h, a, b, dP, dN, n):
    dP.fill(0.0)
    dN.fill(0.0)

    p = int(a)
    if p + 1 < n:
        dP[1:n - p] -= (1.0 - h[p + 1:n])
        dN[1:n - p] += h[p + 1:n]
    if p > 0:
        rev = h[p - 1::-1]
        dP[1:p + 1] += rev
        dN[1:p + 1] -= (1.0 - rev)

    p = int(b)
    if p + 1 < n:
        dP[1:n - p] += (1.0 - h[p + 1:n])
        dN[1:n - p] -= h[p + 1:n]
    if p > 0:
        rev = h[p - 1::-1]
        dP[1:p + 1] -= rev
        dN[1:p + 1] += (1.0 - rev)

    dd = abs(int(a) - int(b))
    dP[dd] += 1.0
    dN[dd] += 1.0


def swap_eval(P, N, h, a, b, dP, dN, b1, b2, n):
    swap_delta_scratch(h, a, b, dP, dN, n)
    np.add(P, dP, out=b1)
    np.add(N, dN, out=b2)
    return float(max(np.max(b1), np.max(b2)))


def generate_move(h, n, P, N, ones, zeros, top_lags, rng, last_a, last_b):
    if len(ones) == 0 or len(zeros) == 0:
        return -1, -1

    r = float(rng.random())
    if r < 0.42 or top_lags is None or len(top_lags) == 0:
        a = int(ones[int(rng.randint(len(ones)))])
        b = int(zeros[int(rng.randint(len(zeros)))])
    elif r < 0.90:
        found = False
        s = int(top_lags[int(rng.randint(len(top_lags)))])
        L = n - s
        if L > 0:
            if P[s] >= N[s]:
                for _ in range(7):
                    i = int(rng.randint(L))
                    if h[i] == 1.0 and h[i + s] == 0.0:
                        a = i
                        b = i + s
                        found = True
                        break
            else:
                for _ in range(7):
                    i = int(rng.randint(L))
                    if h[i] == 0.0 and h[i + s] == 1.0:
                        a = i + s
                        b = i
                        found = True
                        break
        if not found:
            a = int(ones[int(rng.randint(len(ones)))])
            b = int(zeros[int(rng.randint(len(zeros)))])
    else:
        a = int(ones[int(rng.randint(len(ones)))])
        found = False
        for d in (-1, 1, -2, 2, -3, 3, -4, 4):
            j = a + d
            if 0 <= j < n and h[j] == 0.0:
                b = j
                found = True
                break
        if not found:
            b = int(zeros[int(rng.randint(len(zeros)))])

    if a == b:
        b = int(zeros[int(rng.randint(len(zeros)))])

    if (a == last_b and b == last_a) or (a == last_a and b == last_b):
        if rng.random() < 0.82:
            a = int(ones[int(rng.randint(len(ones)))])
            b = int(zeros[int(rng.randint(len(zeros)))])
            if a == b:
                a = int(ones[int(rng.randint(len(ones)))])
                b = int(zeros[int(rng.randint(len(zeros)))])
    return a, b


def binary_local_search(h0, n, time_limit, rng):
    if time_limit <= 5.0 or remaining_time() <= 5.0:
        h = to_binary(h0, n, rng)
        _, M = binary_count_max(h)
        return h, float(M)

    h = to_binary(h0, n, rng)
    P, N = binary_objects(h)
    M = max(float(np.max(P)), float(np.max(N)))
    best_M = M
    best_H = h.copy()

    ones = np.nonzero(h)[0]
    zeros = np.nonzero(h)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return best_H, float(best_M)

    vals = np.maximum(P, N)
    if len(vals) > 0:
        vals[0] = -1.0
    top_lags = np.argsort(-vals)[:min(12, len(vals))]

    dP = np.zeros(n, dtype=float)
    dN = np.zeros(n, dtype=float)
    b1 = np.empty(n, dtype=float)
    b2 = np.empty(n, dtype=float)

    deadline = min(time.time() + float(time_limit), _START_TIME + _BUDGET_S - 3.0)
    T = max(2.0, min(7.0, 0.15 * float(best_M) + 2.0))
    no_improve = 0
    move_count = 0
    last_a = -1000
    last_b = -1000

    while time.time() < deadline:
        batch = 40
        best_nm = M + 1e100
        best_a = -1
        best_b = -1

        for _ in range(batch):
            if time.time() >= deadline:
                break
            a, b = generate_move(h, n, P, N, ones, zeros, top_lags, rng, last_a, last_b)
            if a < 0 or b < 0 or a == b:
                continue
            nm = swap_eval(P, N, h, a, b, dP, dN, b1, b2, n)
            if nm < best_nm - 1e-12:
                best_nm = nm
                best_a = a
                best_b = b
            elif abs(nm - best_nm) <= 1e-9 and best_a < 0 and abs(nm - M) <= 1e-9:
                best_nm = nm
                best_a = a
                best_b = b

        accepted = False
        if best_a >= 0:
            if best_nm < M - 1e-9:
                accepted = True
            elif abs(best_nm - M) <= 1e-9:
                if rng.random() < 0.48:
                    accepted = True
            else:
                delta = float(best_nm - M)
                if T > 1e-4 and delta <= 10.0 and rng.random() < np.exp(-delta / T):
                    accepted = True

        if accepted:
            swap_delta_scratch(h, best_a, best_b, dP, dN, n)
            P += dP
            N += dN
            h[best_a] = 0.0
            h[best_b] = 1.0
            M = best_nm
            last_a, last_b = best_a, best_b
            move_count += 1

            if M < best_M - 1e-9:
                best_M = M
                best_H = h.copy()
                no_improve = 0
            else:
                no_improve += 1

            ones = np.nonzero(h)[0]
            zeros = np.nonzero(h)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break

            if move_count % 100 == 0:
                vals = np.maximum(P, N)
                vals[0] = -1.0
                top_lags = np.argsort(-vals)[:min(12, len(vals))]

            if move_count % 1000 == 0:
                P, N = binary_objects(h)
                M = max(float(np.max(P)), float(np.max(N)))
                if M < best_M - 1e-9:
                    best_M = M
                    best_H = h.copy()
                    no_improve = 0
                if M > best_M + 12.0:
                    h = best_H.copy()
                    P, N = binary_objects(h)
                    M = best_M
                    ones = np.nonzero(h)[0]
                    zeros = np.nonzero(h)[0]
                    no_improve = 0
                    last_a = -1000
                    last_b = -1000

            if no_improve > 4000:
                T *= 0.75
                T = max(T, 0.05)
            if no_improve > 14000:
                h = best_H.copy()
                P, N = binary_objects(h)
                M = best_M
                for _ in range(3 + int(rng.randint(0, 6))):
                    oj = np.nonzero(h)[0]
                    zj = np.nonzero(h == 0.0)[0]
                    if len(oj) == 0 or len(zj) == 0:
                        break
                    aj = int(oj[int(rng.randint(len(oj)))])
                    bj = int(zj[int(rng.randint(len(zj)))])
                    if aj == bj:
                        continue
                    swap_delta_scratch(h, aj, bj, dP, dN, n)
                    P += dP
                    N += dN
                    h[aj] = 0.0
                    h[bj] = 1.0
                P, N = binary_objects(h)
                M = max(float(np.max(P)), float(np.max(N)))
                ones = np.nonzero(h)[0]
                zeros = np.nonzero(h)[0]
                vals = np.maximum(P, N)
                vals[0] = -1.0
                top_lags = np.argsort(-vals)[:min(12, len(vals))]
                no_improve = 5000
                last_a = -1000
                last_b = -1000
        else:
            no_improve += batch
            if no_improve > 4000:
                T *= 0.75
                T = max(T, 0.05)

    return best_H, float(best_M)


def improve_at_n(n, cont_budget, bin_budget, cur_h, cur_n, cur_c, rng):
    if cur_h is None:
        best_h = project_feasible(make_run_seq(n, rng), n)
        best_c = compute_c5(best_h)
        best_n = n
    else:
        best_h = np.array(cur_h, dtype=float).copy()
        best_c = float(cur_c)
        best_n = int(cur_n)

    if n <= 0:
        return best_h, best_c, best_n

    if cont_budget > 10.0 and remaining_time() > 20.0:
        starts = gen_continuous_starts(n, best_h, best_n, rng)
        cont_deadline = min(time.time() + float(cont_budget), _START_TIME + _BUDGET_S - 5.0)
        k = 3 if cont_budget > 180.0 else (2 if cont_budget > 80.0 else 1)
        k = min(k, len(starts))
        for i in range(k):
            left = cont_deadline - time.time()
            if left < 12.0 or remaining_time() < 15.0:
                break
            if k == 1:
                tp = left - 5.0
            elif i == 0:
                tp = 0.52 * left
            elif i == 1:
                tp = 0.36 * left
            else:
                tp = left - 5.0
            tp = max(8.0, min(tp, left - 5.0))
            h, c = refine_continuous(starts[i][1], n, tp, rng)
            best_h, best_c, best_n = consider_solution(h, n, best_h, best_c, best_n)

        if time.time() < cont_deadline and remaining_time() > 10.0:
            base = best_h if len(best_h) == n else interp_to(best_h, n)
            hr, cr = rotation_sweep(base, n)
            best_h, best_c, best_n = consider_solution(hr, n, best_h, best_c, best_n)

    if bin_budget > 10.0 and remaining_time() > 20.0:
        bstarts = gen_binary_starts(n, best_h, best_n, rng)
        bin_deadline = min(time.time() + float(bin_budget), _START_TIME + _BUDGET_S - 5.0)
        top_count = min(3, len(bstarts))

        for i in range(top_count):
            left = bin_deadline - time.time()
            if left < 25.0 or remaining_time() < 20.0:
                break
            if top_count == 1:
                tp = left - 5.0
            elif i == 0:
                tp = 0.55 * left
            elif i == 1:
                tp = 0.38 * left
            else:
                tp = left - 5.0
            tp = max(10.0, min(tp, left - 5.0))

            h_out, M = binary_local_search(bstarts[i][1], n, tp, rng)
            best_h, best_c, best_n = consider_solution(h_out, n, best_h, best_c, best_n)

            if bin_deadline - time.time() > 10.0:
                hr, cr = rotation_sweep(h_out, n)
                best_h, best_c, best_n = consider_solution(hr, n, best_h, best_c, best_n)

            if bin_deadline - time.time() > 35.0:
                h_rot2, M_rot2 = binary_local_search(hr, n, max(10.0, bin_deadline - time.time() - 8.0), rng)
                best_h, best_c, best_n = consider_solution(h_rot2, n, best_h, best_c, best_n)

        left = bin_deadline - time.time()
        if left > 40.0 and remaining_time() > 50.0:
            base = best_h if len(best_h) == n else interp_to(best_h, n)
            hb = to_binary(base, n, rng)
            h_out, M = binary_local_search(hb, n, max(15.0, left - 10.0), rng)
            best_h, best_c, best_n = consider_solution(h_out, n, best_h, best_c, best_n)
            if bin_deadline - time.time() > 10.0:
                hr, cr = rotation_sweep(h_out, n)
                best_h, best_c, best_n = consider_solution(hr, n, best_h, best_c, best_n)

    if remaining_time() > 35.0:
        base = best_h if len(best_h) == n else interp_to(best_h, n)
        extra = max(15.0, min(remaining_time() - 12.0, max(25.0, 0.45 * float(cont_budget))))
        h, c = refine_continuous(base, n, extra, rng)
        best_h, best_c, best_n = consider_solution(h, n, best_h, best_c, best_n)

    return best_h, best_c, best_n


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = None
    parent_n = 0
    try:
        pv = initial_h_values
        if pv is not None:
            p = np.asarray(pv, dtype=float)
            if p.size > 0:
                parent = p
                parent_n = int(len(p))
    except Exception:
        parent = None

    if parent is not None and parent_n > 0:
        pproj = project_feasible(parent, parent_n)
        best_h = pproj.copy()
        best_c = compute_c5(pproj)
        best_n = parent_n
        if remaining_time() > 25.0:
            rr, cr = rotation_sweep(pproj, parent_n)
            best_h, best_c, best_n = consider_solution(rr, parent_n, best_h, best_c, best_n)
    else:
        for n in (160, 200, 240):
            h = make_run_seq(n, rng)
            best_h, best_c, best_n = consider_solution(h, n, best_h, best_c, best_n)

    if best_h is None:
        best_h = project_feasible(make_run_seq(160, rng), 160)
        best_c = compute_c5(best_h)
        best_n = 160

    base_n = int(best_n)
    if base_n % 2 != 0 or base_n < 120 or base_n > 320:
        base_n = 160

    candidates = [base_n, 160, 176, 192, 200, 224, 240, 288]
    pref = []
    for x in candidates:
        if x % 2 == 0 and 120 <= x <= 320:
            pref.append((abs(x - base_n), x))
    pref.sort()
    resolutions = []
    seen = set()
    for _, x in pref:
        if x not in seen:
            resolutions.append(x)
            seen.add(x)
        if len(resolutions) >= 4:
            break
    if not resolutions:
        resolutions = [160]

    for idx, n in enumerate(resolutions):
        rem = remaining_time()
        if rem < 60.0:
            break
        if idx == 0:
            cb = min(160.0, 0.27 * rem)
            bb = min(520.0, 0.52 * rem)
        elif idx == 1:
            cb = min(130.0, 0.25 * rem)
            bb = min(360.0, 0.45 * rem)
        elif idx == 2:
            cb = min(80.0, 0.20 * rem)
            bb = min(210.0, 0.32 * rem)
        else:
            cb = min(45.0, 0.13 * rem)
            bb = min(130.0, 0.22 * rem)
        cb = max(10.0, min(cb, 0.40 * rem))
        bb = max(10.0, min(bb, 0.65 * rem))
        best_h, best_c, best_n = improve_at_n(n, cb, bb, best_h, best_n, best_c, rng)

    if remaining_time() > 45.0 and best_h is not None:
        nfin = int(best_n)
        h, c = refine_continuous(best_h, nfin, min(remaining_time() - 12.0, 120.0), rng)
        best_h, best_c, best_n = consider_solution(h, nfin, best_h, best_c, best_n)

        if nfin % 2 == 0 and remaining_time() > 45.0:
            hb = to_binary(best_h, nfin, rng)
            hbin, M = binary_local_search(hb, nfin, max(15.0, remaining_time() - 18.0), rng)
            best_h, best_c, best_n = consider_solution(hbin, nfin, best_h, best_c, best_n)
            if remaining_time() > 15.0:
                hr, cr = rotation_sweep(hbin, nfin)
                best_h, best_c, best_n = consider_solution(hr, nfin, best_h, best_c, best_n)

    if best_h is None:
        best_h = project_feasible(make_run_seq(160, rng), 160)
        best_n = 160

    best_h = project_feasible(best_h, best_n)
    final_c = compute_c5(best_h)

    try:
        ec = float(evaluate_erdos_solution(best_h))
        if ec < final_c:
            final_c = ec
    except Exception:
        pass

    return (best_h, float(final_c), int(best_n))
