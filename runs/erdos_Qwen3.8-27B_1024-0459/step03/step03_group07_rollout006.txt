            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = corr.max()
    sh = beta * (corr - c_max)
    sh = np.clip(sh, -500.0, 0.0)
    exp_sh = np.exp(sh)
    w_sum = exp_sh.sum()
    if w_sum < 1e-300:
        return None
    weights = exp_sh / w_sum
    
    grad = np.zeros(n)
    for lidx in range(len(corr)):
        if weights[lidx] < 1e-12:
            continue
        lag = lidx - (n - 1)
        w = weights[lidx]
        if lag >= 0:
            grad[:n - lag] += w * (1.0 - h[lag:n])
            grad[lag:] += w * (-h[:n - lag])
        else:
            s = -lag
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] += w * (-h[s:n])
    return grad

def gd_smooth_max(h, n, time_limit, t0):
    h = project_feasible(h, n)
    best = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    for beta in [30.0, 80.0, 200.0, 500.0, 1000.0, 2000.0]:
        if time.time() - t0 > time_limit:
            break
        lr = 0.005
        for it in range(2000):
            if time.time() - t0 > time_limit:
                break
            g = smooth_max_grad(h, beta)
            if g is None:
                continue
            h_new = h - lr * g
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-16:
                best_c5 = c5_new
                best = h_new.copy()
                h = h_new
            elif it > 0:
                lr *= 0.95
            if lr < 1e-10:
                break
    return best

def coord_descent_fast(h, n, sweeps, time_limit, t0):
    """Fast coordinate descent with efficient 1D minimization"""
    h = project_feasible(h, n)
    best = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    for sw in range(sweeps):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        for j in range(n):
            if time.time() - t0 > time_limit:
                break
            old = h[j]
            
            # Compute affected correlations and their linear coefficients
            # corr[lag] = sum_i h[i](1-h[i+lag]) for lag>=0
            # h[j] affects corr[lag] for all lag (since h[j] appears in every formula)
            # But we only need the ones that could be max
            
            # For efficiency, compute the change in each corr when h[j] changes to x
            # corr[lag] as function of x = h[j]:
            #   For lag >= 0 with lag <= j <= n-1: 
            #     term change: x*(1-h[j+lag]) + h[j-lag]*... wait this gets complicated
            # Let me just use the formula:
            #   d corr[lag]/d h[j] is known, so corr_new[lag] ≈ corr[lag] + d*h[j]*(x - old)
            #   But it's actually: for lag>=0:
            #     if j <= n-1-lag: corr changes by (x - old)*(1-h[j+lag])
            #     if j >= lag: corr changes by (x - old)*(-h[j-lag])
            #   Wait no, it's nonlinear in general because h[j] appears in two places
            #   for lag=0: h[j]*(1-h[j]) is quadratic
            
            # Let me just do a direct evaluation for the grid search
            # Affected lags: all lags from -(n-1) to n-1
            # But most have small weight. Let's identify which lags could be near max.
            
            # Actually for a quick CD, let's just evaluate the objective at a few points
            # and pick the best.
            
            # Evaluate at old (current) and a grid of alternatives
            # The key insight: changing h[j] changes at most 2*min(j+1, n-j) correlations
            # Actually it changes correlations at lags -j to n-1-j (from h[j] as first factor)
            # and lags j-n+1 to j (from h[j] as second factor)
            # So the affected lag range is [-(n-1), n-1] in the worst case (all of them)
            
            # For a practical CD, let's evaluate at a small grid
            grid = np.array([old, max(0, old-0.1), min(1, old+0.1), max(0, old-0.05), min(1, old+0.05), 0.0, 1.0, 0.5])
            grid = np.unique(np.clip(grid, 0, 1))
            
            best_x = old
            best_val = float('inf')
            
            for x in grid:
                h[j] = x
                c = corr + (x - old) * 0  # We need to recompute affected parts
                # Actually let's just compute the changed correlations
                # For lag >= 0:
                #   corr[lag] involves h[i+lag]*(1-h[i]) for i in [0, n-1-lag]
                #   h[j] appears as h[i+lag] when i=j-lag (valid if 0<=j-lag<=n-1-lag, i.e., lag<=j<=n-1)
                #   h[j] appears as h[i] when i=j (valid if 0<=j<=n-1-lag)
                #   So contribution to corr[lag] from h[j]:
                #     if lag<=j<=n-1: h[j]*(1-h[j-lag])  [as h[i+lag] with i=j-lag]
                #     if 0<=j<=n-1-lag: h[j]*(1-h[j+lag])  [as h[i] with i=j, but (1-h[i+lag])=(1-h[j+lag])]
                #     Wait, I'm confusing myself. Let me restart.
                #
                # corr[lag] = sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag]) for lag >= 0
                # The j-th pair contributes h[j] * (1-h[j+lag]) if 0<=j<=n-1-lag
                # Also, h[j] can appear as h[i+lag] for i=j-lag, contributing h[j-lag]*(1-h[j]) if lag<=j<=n-1
                #
                # So changing h[old_j]=old to h[j]=x:
                #   If 0<=j<=n-1-lag: corr[lag] changes by (x - old)*(1-h[j+lag])
                #   If lag<=j<=n-1: corr[lag] changes by (x - old)*(-h[j-lag])
                #   Note: for lag=0, both conditions hold (0<=j<=n-1), and we get (x-old)*(1-h[j]) + (x-old)*(-h[j])
                #     But wait, at lag=0, the pair is h[j]*(1-h[j]), so changing h[j] from old to x:
                #     new pair = x*(1-x), old pair = old*(1-old)
                #     change = x(1-x) - old(1-old) = (x-old)(1-x-old)
                #     My formula gives: (x-old)*(1-x) + (x-old)*(-old) = (x-old)(1-x-old). ✓
                #     Wait: formula gives (x-old)*(1-h[j+0]) + (x-old)*(-h[j-0]) = (x-old)*(1-x) + (x-old)*(-old)
                #     But h[j] is being changed, so we should use the NEW value x for the first term?
                #     Actually no: the derivative d corr[0]/d h[j] = (1-h[j]) - h[j] = 1-2h[j]
                #     The actual change is x(1-x) - old(1-old), which is NOT (x-old)(1-2*old) in general.
                #     So linear approximation works for small changes but not exact.
                #
                # For exact computation, I need to handle it properly. Let me just recompute the
                # affected correlations exactly.
                
                # For each lag, compute the exact new value:
                # This is O(n) per grid point, which is expensive.
                
                # Alternative: precompute the "rest" (contribution of all other terms)
                # For lag >= 0:
                #   corr[lag] = sum_{i≠j, i+lag≠j} h[i](1-h[i+lag]) + [terms involving h[j]]
                #   Terms involving h[j] (for lag >= 0):
                #     If 0<=j<=n-1-lag: h[j](1-h[j+lag])
                #     If lag<=j<=n-1 and j+lag != j (i.e., lag≠0): h[j-lag](1-h[j])
                #     If lag=0: h[j](1-h[j]) [only counted once]
                
                # This is getting complicated. Let me just recompute the full correlation.
                # For n~200, np.correlate is fast (~microseconds).
                
                c_full = np.correlate(h, 1.0 - h, mode='full')
                val = np.max(c_full) * dx
                if val < best_val:
                    best_val = val
                    best_x = x
            
            h[j] = best_x
        
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    
    return best

def interpolate_h(h_src, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_new = np.interp(xn, xp, h_src)
    return project_feasible(h_new, n_dst)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Get parent
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn > 10:
            best_h = ph.copy()
            best_c5 = compute_c5(ph)
            best_n = pn
    except Exception:
        pass
    
    if best_h is None:
        best_n = 160
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)
    
    def time_left():
        return max(0.0, budget_s - (time.time() - t0) - 5.0)
    
    # Try multiple resolutions
    # Always include parent resolution
    n_list = [best_n]
    for delta in [-16, -8, 8, 16, 24, 32, -32, -24, -40]:
        nn = best_n + delta
        if 80 <= nn <= 400 and nn not in n_list:
            n_list.append(nn)
    # Also try some absolute values
    for nn in [128, 192, 224, 256, 288, 320]:
        if nn not in n_list and 80 <= nn <= 400:
            n_list.append(nn)
    n_list = sorted(set(n_list))
    
    # Time allocation: more time for parent resolution, less for others
    # Parent gets 50% of time, rest split among others
    n_others = len(n_list) - 1
    time_parent = budget_s * 0.45
    time_other_total = budget_s * 0.40
    # Keep 15% for final polish
    
    # Process parent first
    for n in n_list:
        if time_left() < 30:
            break
        is_parent = (n == best_n)
        
        if is_parent:
            tl = time_left() * 0.45
        else:
            tl = time_left() / max(1, len(n_list) - 1) * 0.5
        
        tl = min(tl, time_left() * 0.8)
        
        # Create starting point
        if n == best_n:
            h_init = project_feasible(best_h, n)
        else:
            h_init = interpolate_h(best_h, best_n, n)
        
        # Multiple initializations
        inits = [h_init.copy()]
        for scale in [0.02, 0.04, 0.08]:
            hp = h_init + scale * rng.randn(n)
            inits.append(project_feasible(hp, n))
        
        # Sine perturbation
        t = np.arange(n) / n
        for freq in [3, 5]:
            phase = rng.rand() * 2 * np.pi
            hf = h_init + 0.03 * np.sin(2 * np.pi * freq * t + phase)
            inits.append(project_feasible(hf, n))
        
        n_inits = len(inits)
        time_per_init = tl / n_inits
        
        for h0 in inits:
            if time_left() < 10:
                break
            tl_i = time_per_init
            
            # Phase 1: Gradient descent with smooth max
            h1 = gd_smooth_max(h0, n, tl_i * 0.5, t0)
            c5_1 = compute_c5(h1)
            
            # Phase 2: Coordinate descent
            h2 = coord_descent_fast(h1, n, 5, tl_i * 0.9, t0)
            c5_2 = compute_c5(h2)
            
            # Phase 3: More gradient descent at higher beta
            h3 = gd_smooth_max(h2, n, tl_i * 0.95, t0)
            c5_3 = compute_c5(h3)
            
            # Take best
            candidates = [(h1, c5_1), (h2, c5_2), (h3, c5_3)]
            cb_h, cb_c5 = min(candidates, key=lambda x: x[1])
            
            if cb_c5 < best_c5:
                best_c5 = cb_c5
                best_h = cb_h.copy()
                best_n = n
    
    # Final polish on best solution

        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
