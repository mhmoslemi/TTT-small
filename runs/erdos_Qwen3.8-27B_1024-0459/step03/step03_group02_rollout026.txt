def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
        else:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for i in order:
                if diff > 0:
                    h[i] = min(1.0, h[i] + diff)
                else:
                    h[i] = max(0.0, h[i] + diff)
                diff = target - np.sum(h)
                if abs(diff) < 1e-15:
                    break
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def smooth_max_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    s = float(np.sum(exp_vals))
    obj_val = (m + np.log(s) / beta) * dx
    weights = exp_vals / s
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-12:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            n_pairs = n - offset
            if n_pairs > 0:
                grad[offset:offset + n_pairs] += w * (1.0 - x[:n_pairs])
                grad[:n_pairs] -= w * x[offset:offset + n_pairs]
        else:
            ao = -offset
            n_pairs = n - ao
            if n_pairs > 0:
                grad[:n_pairs] += w * (1.0 - x[ao:ao + n_pairs])
                grad[ao:ao + n_pairs] -= w * x[:n_pairs]
    grad *= dx
    return obj_val, grad


def slsqp_high_beta(h, n, beta, maxiter=500):
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = float(np.max(corr))
        shifted = np.minimum(beta * (corr - m), 0.0)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = float(np.max(corr))
        shifted = np.minimum(beta * (corr - m), 0.0)
        exp_vals = np.exp(shifted)
        s = float(np.sum(exp_vals))
        weights = exp_vals / s
        grad = np.zeros(n)
        for k in range(len(corr)):
            w = weights[k]
            if w < 1e-12:
                continue
            offset = k - (n - 1)
            if offset >= 0:
                n_pairs = n - offset
                if n_pairs > 0:
                    grad[offset:offset + n_pairs] += w * (1.0 - x[:n_pairs])
                    grad[:n_pairs] -= w * x[offset:offset + n_pairs]
            else:
                ao = -offset
                n_pairs = n - ao
                if n_pairs > 0:
                    grad[:n_pairs] += w * (1.0 - x[ao:ao + n_pairs])
                    grad[ao:ao + n_pairs] -= w * x[:n_pairs]
        return grad * dx

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]

        if res.fun < np.inf:

    except Exception:
        pass
    return h.copy()

def subgrad_direction(h, n, tol_frac=0.2):

    h_nest, c5_nest = nesterov_optimize(h, n, beta=5000, max_iters=600, lr0=0.02)
    if c5_nest < best_c5:
        best_c5 = c5_nest
        best_h = h_nest.copy()
        h = h_nest.copy()

    # Phase 3: Subgradient with careful backtracking
    h = best_h.copy()
    stagnant = 0
    for it in range(3000):
        if remaining_time() < 5:
            break
        grad = subgrad_direction(h, n, tol_frac=0.15)
        if grad is None:
            break
        base_lr = 0.04 / (1.0 + 0.002 * it)
        improved = False
        for mult in [1.0, 0.5, 0.25, 2.0, 0.1, 4.0, 0.05]:
            lr = base_lr * mult
            h_trial = project_feasible(h - lr * grad, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-14:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial.copy()
                improved = True
                stagnant = 0
                break
        if not improved:
            stagnant += 1
        if stagnant > 300 and it % 200 == 100 and remaining_time() > 20:
            h_try = slsqp_high_beta(h, n, 10000, maxiter=300)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                stagnant = 0

    # Phase 4: Active-set QP iteration
    if remaining_time() > 15:
        h_try = active_set_qp(best_h, n, max_iter=15, tol=0.08)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 5: Binary local search from current best
    if remaining_time() > 30:
        h_bin, c5_bin = binary_local_search(best_h, n, max_sweeps=8, rng=rng)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # Also try SLSQP from the binary solution
        if c5_bin < best_c5 + 0.005:
            h_try = slsqp_high_beta(h_bin, n, 5000, maxiter=400)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Phase 6: Multi-resolution search
    for trial_n in [156, 158, 162, 164, 166, 168, 172, 176, 192, 208, 240, 256, 320]:
        if remaining_time() < 18:
            break
        h_interp = interp_to(best_h, best_n, trial_n)
        h_interp = project_feasible(h_interp, trial_n)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5 + 0.003:
            for beta in [2000, 8000]:
                if remaining_time() < 8:
                    break
                h_try = slsqp_high_beta(h_interp, trial_n, beta, maxiter=350)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = trial_n
                    h_interp = h_try.copy()
            # Nesterov at this resolution
            if remaining_time() > 15:
                h_ne, c5_ne = nesterov_optimize(h_interp, trial_n, beta=3000,
                                               max_iters=300, lr0=0.02)
                if c5_ne < best_c5:
                    best_c5 = c5_ne
                    best_h = h_ne.copy()
                    best_n = trial_n

    # Phase 7: Multi-restart with perturbations at best resolution
    n_cur = best_n
    for restart in range(6):
        if remaining_time() < 25:
            break
        h_pert = best_h.copy()
        mask = (h_pert > 0.02) & (h_pert < 0.98)
        nmask = int(np.sum(mask))
        if nmask > 3:
            h_pert[mask] += rng.randn(nmask) * 0.015
            h_pert = project_feasible(h_pert, n_cur)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5 + 0.008:
                h_try = slsqp_high_beta(h_pert, n_cur, 5000, maxiter=350)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()

    # Phase 8: Final ultra-precision SLSQP
    n_final = best_n
    for beta in [20000, 50000, 100000, 200000]:
        if remaining_time() < 6:
            break
        h_try = slsqp_high_beta(best_h, n_final, beta, maxiter=400)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 9: Final subgradient micro-steps
    if remaining_time() > 5:
        h = best_h.copy()
        for it in range(300):
            if remaining_time() < 2:
                break
            grad = subgrad_direction(h, n_final, tol_frac=0.08)
            if grad is None:
                break
            improved = False
            for lr in [0.0005, 0.001, 0.002, 0.005, 0.0002, 0.01]:
                h_trial = project_feasible(h - lr * grad, n_final)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-15:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h = h_trial.copy()
                    improved = True
                    break
            if not improved and it > 50:
                break

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
