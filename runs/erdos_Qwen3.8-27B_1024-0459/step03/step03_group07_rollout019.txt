def smooth_max_obj(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -500.0, 0.0)
    return (m + np.log(np.sum(np.exp(sh))) / beta) * (2.0 / n)

                    v += delta * (1.0 - h[j+k])
                if k <= j:
                    v -= delta * h[j-k]
                max_new = max(max_new, v)
                
                fi2 = n - 1 - k
                if fi2 >= 0:
                    v2 = corr[fi2]
                    if k <= j:
                        v2 += delta * (1.0 - h[j-k])
                    if j + k < n:
                        v2 -= delta * h[j+k]
                    max_new = max(max_new, v2)
            
            if max_new < best_flip_c5 - 1e-10:
                best_flip_c5 = max_new
                best_flip_j = j
        
        if best_flip_j >= 0:
            new_val = 0.0 if h[best_flip_j] > 0.5 else 1.0
            h[best_flip_j] = new_val
            h = project_feasible(h, n)
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_lag_idx = np.argmax(corr)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            break
    
    return best_h

def subgrad_optimize(h, n, iters=2000, time_limit=None, t0=None):
    """Subgradient method on the max of correlations."""
    h = project_feasible(h, n)
    best = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    lr_schedule = 'auto'
    lr = 0.01
    patience = 0
    max_patience = 200
    
    for it in range(iters):
        if time_limit is not None and t0 is not None and time.time() - t0 > time_limit - 2:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        target = np.max(corr)
        
        # Find top lags
        n_lags = 5
        top_ids = np.argsort(corr)[-n_lags:]
        
        # Compute subgradient
        grad = np.zeros(n)
        for lidx in top_ids:
            lag = lidx - (n - 1)
            w = 1.0 / n_lags
            if lag == 0:
                grad += w * (1.0 - 2.0 * h)
            elif lag > 0:
                # corr[lag] = sum_{i=0}^{n-1-lag} h[i](1-h[i+lag])
                # d/dh[i] = (1-h[i+lag]) for i in [0, n-1-lag]
                # d/dh[i] = -h[i-lag] for i in [lag, n-1]
                grad[:n-lag] += w * (1.0 - h[lag:])
                grad[lag:] += w * (-h[:n-lag])
            else:
                s = -lag
                # corr[-s] = sum_{i=s}^{n-1} h[i](1-h[i-s])
                grad[s:] += w * (1.0 - h[:n-s])
                grad[:n-s] += w * (-h[s:])
        
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best = h_new.copy()
            h = h_new
            patience = 0
            lr = min(lr * 1.01, 0.05)
        else:
            h = h_new  # still move
            patience += 1
            lr *= 0.95
            if patience > max_patience:
                # Perturb
                h = project_feasible(h + 0.03 * np.random.randn(n), n)
                lr = 0.01
                patience = 0
    
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return max(0.0, budget_s - (time.time() - t0) - 5.0)
    
    # Get parent
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn > 10:
            best_h = ph.copy()
            best_c5 = compute_c5(ph)
            best_n = pn
    except Exception:
        pass
    
    if best_h is None:
        best_n = 160
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)
    
    start_n = best_n
    start_h = best_h.copy()
    
    # Multi-resolution search
    n_list = [start_n]
    for nn in [128, 144, 160, 176, 192, 208, 224, 240, 256, 288, 320, 384]:
        if nn not in n_list:
            n_list.append(nn)
    n_list = sorted(set(n_list))
    
    # Focus on resolutions near and above the parent
    # Rough time allocation: spend more time on higher resolutions
    for n in n_list:
        if time_left() < 40:
            break
        
        # Skip very small n if we already have a good one
        if n < start_n - 32:
            continue
        
        # Create starting points
        if n == start_n:
            h_init = project_feasible(start_h, n)
        else:
            h_init = interpolate_h(start_h, start_n, n)
        
        # Run fast coordinate descent
        tl_cd = min(time_left() * 0.4, 180.0)
        h_cd = fast_cd(h_init, n, sweeps=25, time_limit=tl_cd, t0=t0)
        c5_cd = compute_c5(h_cd)
        
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n
        
        # Binary local search (quick)
        if time_left() > 30:
            tl_bl = min(time_left() * 0.1, 40.0)
            h_bl = binary_local_search(h_cd, n, time_limit=tl_bl, t0=t0)
            c5_bl = compute_c5(h_bl)
            if c5_bl < best_c5:
                best_c5 = c5_bl
                best_h = h_bl.copy()
                best_n = n
                h_cd = h_bl.copy()
        
        # Subgradient optimization
        if time_left() > 40:
            tl_sg = min(time_left() * 0.15, 80.0)
            h_sg, c5_sg = subgrad_optimize(h_cd, n, iters=3000, time_limit=tl_sg, t0=t0)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()
                best_n = n
            h_cd = h_sg if c5_sg < c5_cd else h_cd
        
        # SLSQP polish
        if time_left() > 30:
            tl_sl = min(time_left() * 0.15, 60.0)
            h_sl, c5_sl = slsqp_optimize(h_cd, n, time_limit=tl_sl, t0=t0)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()
                best_n = n
            h_cd = h_sl if c5_sl < c5_cd else h_cd
        
        # More coordinate descent
        if time_left() > 30:
            tl_cd2 = min(time_left() * 0.15, 120.0)
            h_cd2 = fast_cd(h_cd, n, sweeps=20, time_limit=tl_cd2, t0=t0)
            c5_cd2 = compute_c5(h_cd2)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()
                best_n = n
    
    # Final polish on the best solution
    if time_left() > 40:
        tl1 = time_left() * 0.3
        h_pol = fast_cd(best_h, best_n, sweeps=40, time_limit=tl1, t0=t0)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    
    if time_left() > 30:
        tl2 = time_left() * 0.25
        h_sl2, c5_sl2 = slsqp_optimize(best_h, best_n, time_limit=tl2, t0=t0)
        if c5_sl2 < best_c5:
            best_c5 = c5_sl2
            best_h = h_sl2.copy()
    
    if time_left() > 30:
        tl3 = time_left() * 0.3
        h_pol3 = fast_cd(best_h, best_n, sweeps=30, time_limit=tl3, t0=t0)
        c5_pol3 = compute_c5(h_pol3)
        if c5_pol3 < best_c5:
            best_c5 = c5_pol3
            best_h = h_pol3.copy()
    
    # Final small perturbations and re-polish
    if time_left() > 20:
        for scale in [0.005, 0.01, 0.002]:
            if time_left() < 10:
                break
            for _ in range(3):
                h_pert = project_feasible(best_h + scale * rng.randn(best_n), best_n)
                tl_p = min(time_left() * 0.2, 30.0)
                h_r = fast_cd(h_pert, best_n, sweeps=15, time_limit=tl_p, t0=t0)
                c5_r = compute_c5(h_r)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r.copy()
    
    # Final final polish
    if time_left() > 20:
        h_final = fast_cd(best_h, best_n, sweeps=30, time_limit=time_left() * 0.5, t0=t0)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    # Validate with evaluator
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
