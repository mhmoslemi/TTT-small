def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:

    m = corr[idx].max()
    shifted = corr[idx] - m
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    val = (m + np.log(w_sum) / beta) * dx
    se = np.sum(h) - target
    val += pw * se * se

    grad = np.zeros(n)
    for t in range(len(idx)):
        if w[t] < 1e-14:
            continue
        k = int(idx[t]) - (n - 1)
        if k >= 0:
            al = k
            term1 = np.zeros(n)
            term1[:n - al] = 1.0 - h[al:]
            term2 = np.zeros(n)
            term2[al:] = h[:n - al]
        else:
            al = -k
            term1 = np.zeros(n)
            term1[al:] = 1.0 - h[:n - al]
            term2 = np.zeros(n)
            term2[:n - al] = h[al:]
        grad += (w[t] / w_sum) * (term1 - term2)
    grad *= dx
    grad += 2.0 * pw * se * np.ones(n)
    return val, grad


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, budget_s - (time.time() - t0) - 3.0)

    # Initialize from parent
    h_par = np.array(initial_h_values, dtype=float)
    n_par = len(h_par)
    c5_par = compute_c5(h_par)
    print("Parent: n={} C5={:.6f}".format(n_par, c5_par))

    # Interpolate to n=640
    n = 640
    target = n / 2.0
    xp = (np.arange(n_par) + 0.5) * (2.0 / n_par)
    xn = (np.arange(n) + 0.5) * (2.0 / n)
    h0 = np.interp(xn, xp, h_par)
    h0 = project_feasible(h0, target)
    c5_0 = compute_c5(h0)

    # Also try n=480
    n2 = 480
    target2 = n2 / 2.0
    xn2 = (np.arange(n2) + 0.5) * (2.0 / n2)
    h2 = np.interp(xn2, xp, h_par)
    h2 = project_feasible(h2, target2)
    c5_2 = compute_c5(h2)

    if c5_2 < c5_0:
        n = n2
        target = target2
        h0 = h2
        c5_0 = c5_2

    best_h = h0.copy()
    best_c5 = c5_0
    best_n = n
    print("Start: n={} C5={:.6f}".format(n, best_c5))

    # Phase 1: L-BFGS-B with analytic gradient
    if remaining() > 100:
        bounds = [(0.0, 1.0)] * best_n
        target_cur = best_n / 2.0

        for beta in [150, 300, 600, 1200, 2500, 5000]:
            if remaining() < 20:
                break
            top_k = 30
            pw = 2000.0

            def obj_fn(params):
                h_clip = np.clip(params, 0.0, 1.0)
                v, _ = smooth_obj_and_grad(h_clip, beta, top_k, target_cur, pw)
                return v

            def grad_fn(params):
                h_clip = np.clip(params, 0.0, 1.0)
                _, g = smooth_obj_and_grad(h_clip, beta, top_k, target_cur, pw)
                return g

            try:
                res = minimize(obj_fn, best_h, method='L-BFGS-B', bounds=bounds,
                               jac=grad_fn,
                               options={'maxiter': 800, 'ftol': 1e-20,
                                        'gtol': 1e-14, 'maxls': 80,
                                        'maxfun': 12000})
                h_new = project_feasible(res.x, target_cur)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  L-BFGS b={} C5={:.6f}".format(beta, c5_new))
                elif res.x is not None:
                    h_new = project_feasible(res.x, target_cur)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        print("  L-BFGS b={} (alt) C5={:.6f}".format(beta, c5_new))
            except Exception:
                pass

    # Phase 2: Targeted coordinate descent
    if remaining() > 80:
        n_t = best_n
        target_t = n_t / 2.0
        h = best_h.copy()
        no_improve = 0
        step_list = [0.03, 0.02, 0.01, 0.05, 0.015, 0.005, 0.04, 0.008]

        for sweep in range(500):
            if remaining() < 5:
                break

            # Get top lags for targeting
            corr = np.correlate(h, 1.0 - h, mode='full')
            top_k = 5
            top_idx = np.argsort(corr)[-top_k:]
            current_max_c5 = corr.max() * (2.0 / n_t)

            improved_sweep = False

            # Targeted moves based on overlap structure
            for qi in top_idx:
                if remaining() < 4:
                    break
                k = int(qi) - (n_t - 1)
                al = abs(k)
                if al >= n_t:
                    continue
                # overlap[i] = h[i+al] + h[i-al]
                overlap = np.zeros(n_t)
                overlap[:n_t - al] += h[al:]
                overlap[al:] += h[:n_t - al]

                # Sort by overlap: try moving to high-overlap from low-overlap
                order = np.argsort(overlap)
                # Try top 10 targets and bottom 10 sources
                n_try = 15
                hi_idx = order[(n_t - n_try):]
                lo_idx = order[:n_try]

                for _ in range(n_try * 3):
                    i_cand = hi_idx[rng.randint(n_try)]
                    j_cand = lo_idx[rng.randint(n_try)]
                    if i_cand == j_cand:
                        continue
                    for delta in step_list[:4]:
                        if h[i_cand] + delta > 1.0 or h[j_cand] - delta < 0.0:
                            continue
                        h[i_cand] += delta
                        h[j_cand] -= delta
                        c5_test = compute_c5(h)
                        if c5_test < best_c5:
                            best_c5 = c5_test
                            best_h = h.copy()
                            improved_sweep = True
                        else:
                            h[i_cand] -= delta
                            h[j_cand] += delta
                        break

            # Random pair moves
            n_rand = 1200
            for _ in range(n_rand):
                if remaining() < 3:
                    break
                i = rng.randint(n_t)
                j = rng.randint(n_t)
                if i == j:
                    continue
                si = i if h[i] < h[j] else j
                sj = j if si == i else i
                delta = step_list[rng.randint(len(step_list))]
                if h[si] + delta > 1.0 or h[sj] - delta < 0.0:
                    continue
                h[si] += delta
                h[sj] -= delta
                c5_test = compute_c5(h)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h.copy()
                    improved_sweep = True
                else:
                    h[si] -= delta
                    h[sj] += delta

            if improved_sweep:
                no_improve = 0
            else:
                no_improve += 1

            if no_improve >= 25:
                break

        print("  After targeted CD: C5={:.6f}".format(best_c5))

    # Phase 3: Resolution sweep
    if remaining() > 60:
        xp_best = (np.arange(best_n) + 0.5) * (2.0 / best_n)
        for n_try in [best_n - 32, best_n + 32, 320, 400, 560, 720]:
            if remaining() < 15:
                break
            if n_try < 100 or n_try > 900 or n_try % 2 != 0:
                continue
            t_try = n_try / 2.0
            xq = (np.arange(n_try) + 0.5) * (2.0 / n_try)
            hq = np.interp(xq, xp_best, best_h)
            hq = project_feasible(hq, t_try)
            c5_try = compute_c5(hq)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = hq.copy()
                best_n = n_try
                target = t_try
                print("  Found better at n={}: C5={:.6f}".format(n_try, c5_try))
                # Quick L-BFGS polish at new resolution
                if remaining() > 20:
                    n_pol = best_n
                    t_pol = n_pol / 2.0
                    bnd = [(0.0, 1.0)] * n_pol

                    def obj_pol(params):
                        hc = np.clip(params, 0.0, 1.0)
                        v, _ = smooth_obj_and_grad(hc, 1000, 25, t_pol, 2000.0)
                        return v

                    def grad_pol(params):
                        hc = np.clip(params, 0.0, 1.0)
                        _, g = smooth_obj_and_grad(hc, 1000, 25, t_pol, 2000.0)
                        return g

                    try:
                        res = minimize(obj_pol, best_h, method='L-BFGS-B',
                                       bounds=bnd, jac=grad_pol,
                                       options={'maxiter': 400, 'ftol': 1e-19,
                                                'gtol': 1e-14, 'maxfun': 5000})
                        h_new = project_feasible(res.x, t_pol)
                        c5_new = compute_c5(h_new)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_new.copy()
                            print("    Polish n={} C5={:.6f}".format(n_pol, c5_new))
                    except Exception:
                        pass

    # Phase 4: Fine polishing
    if remaining() > 40:
        n_f = best_n
        t_f = n_f / 2.0
        h = best_h.copy()
        fine_steps = [0.001, 0.002, 0.0005, 0.003, 0.0015]

        for sweep in range(400):
            if remaining() < 4:
                break
            improved = False
            order = rng.permutation(n_f)
            for _ in range(800):
                if remaining() < 2:
                    break
                i = order[rng.randint(n_f)]
                j = order[rng.randint(n_f)]
                if i == j:
                    continue
                delta = fine_steps[rng.randint(len(fine_steps))]
                # Decide direction
                if h[i] < 0.99 and h[j] > 0.01:
                    si, sj = i, j
                elif h[j] < 0.99 and h[i] > 0.01:
                    si, sj = j, i
                else:
                    continue
                if h[si] + delta > 1.0 or h[sj] - delta < 0.0:
                    continue
                h[si] += delta
                h[sj] -= delta
                c5_test = compute_c5(h)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h.copy()
                    improved = True
                else:
                    h[si] -= delta
                    h[sj] += delta
            if not improved:
                # Try systematic scan
                sys_improved = False
                for i in range(0, n_f, 2):
                    if remaining() < 2:
                        break
                    j = (i + n_f // 3) % n_f
                    if j == i:
                        j = (i + 1) % n_f
                    for delta in [0.001, -0.001]:
                        if h[i] + delta > 1.0 or h[i] + delta < 0.0:
                            continue
                        if h[j] - delta > 1.0 or h[j] - delta < 0.0:
                            continue
                        h[i] += delta
                        h[j] -= delta
                        c5_test = compute_c5(h)
                        if c5_test < best_c5:
                            best_c5 = c5_test
                            best_h = h.copy()
                            sys_improved = True
                            improved = True
                        else:
                            h[i] -= delta
                            h[j] += delta
                if not sys_improved:
                    break

        print("  After fine polish: C5={:.6f}".format(best_c5))

    # Phase 5: Final targeted fine-tune on worst lags
    if remaining() > 30:
        n_f2 = best_n
        t_f2 = n_f2 / 2.0
        h = best_h.copy()
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_lags = np.argsort(corr)[-8:]

        for _ in range(2000):
            if remaining() < 3:
                break
            qi = top_lags[rng.randint(len(top_lags))]
            k = int(qi) - (n_f2 - 1)
            al = abs(k)
            if al >= n_f2:
                continue
            overlap = np.zeros(n_f2)
            overlap[:n_f2 - al] += h[al:]
            overlap[al:] += h[:n_f2 - al]
            order = np.argsort(overlap)
            i_c = order[rng.randint(max(1, n_f2 // 4)):]
            j_c = order[:max(1, n_f2 // 4)]
            i = i_c[rng.randint(len(i_c))]
            j = j_c[rng.randint(len(j_c))]
            if i == j:
                continue
            delta = rng.choice([0.002, 0.005, 0.01, 0.003])
            if h[i] + delta > 1.0 or h[j] - delta < 0.0:
                continue
            h[i] += delta
            h[j] -= delta
            c5_test = compute_c5(h)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h.copy()
            else:
                h[i] -= delta
                h[j] += delta

        print("  After lag-target fine: C5={:.6f}".format(best_c5))

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={} C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
