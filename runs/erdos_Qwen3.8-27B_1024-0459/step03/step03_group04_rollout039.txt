        idx = k + n - 1

            if np_ > 0:
                grads[idx, :np_] += (1.0 - h[k:k + np_])

            stuck_count += 1
            # Try very small step
            h_try = project_h(h + 0.01 * d, n)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur:
                h = h_try
                c5_cur = c5_try
                best_alpha = 0.01
            else:
                if stuck_count > 15:
                    break
                rho = max(1.0, rho * 0.8)
                continue
        
        h = project_h(h + best_alpha * d, n)
        c5_cur = compute_c5(h)
        
        if c5_cur < best_c5 - 1e-10:
            best_c5 = c5_cur
            best_h = h.copy()
            stuck_count = 0
            # Slowly increase trust region if making good progress
            rho = min(15.0, rho * 1.02)
        else:
            rho = max(2.0, rho * 0.95)
            stuck_count += 1
    
    return best_h, best_c5

def slsqp_smooth_max(h, n, beta, max_iter=200):
    """SLSQP with smooth max objective."""
    def obj(x):
        return smooth_max_value(x, beta)
    
    def grad(x):
        return smooth_max_obj_grad(x, beta)
    
    def con_eq(x):
        return np.sum(x) - n / 2.0
    
    def con_eq_grad(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_grad}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0})
        h_new = project_h(res.x, n)
        return h_new
    except Exception:
        return h

def local_perturb(h, n, budget_remaining, rng, max_iter=50000):
    """Fine local perturbation search."""
    t0 = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    deltas = [0.001, 0.002, 0.005, 0.01, 0.02, 0.03, 0.05]
    
    for it in range(max_iter):
        if time.time() - t0 > budget_remaining - 2:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for d in deltas:
            improved = False
            if h[i] >= d and h[j] <= 1.0 - d:
                h[i] -= d
                h[j] += d
                c5_try = compute_c5(h)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                    improved = True
                else:
                    h[i] += d
                    h[j] -= d
            if improved:
                break
            if h[j] >= d and h[i] <= 1.0 - d:
                h[j] -= d
                h[i] += d
                c5_try = compute_c5(h)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                    improved = True
                else:
                    h[j] += d
                    h[i] -= d
            if improved:
                break
    
    return best_h, best_c5

def triple_perturb(h, n, budget_remaining, rng, max_iter=20000):
    """Triple position perturbation."""
    t0 = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() - t0 > budget_remaining - 2:
            break
        idx3 = rng.choice(n, size=3, replace=False)
        i, j, k = int(idx3[0]), int(idx3[1]), int(idx3[2])
        d = 0.005 * (1.0 + rng.rand())
        improved = False
        if h[i] >= 2 * d and h[j] <= 1 - d and h[k] <= 1 - d:
            h[i] -= 2 * d
            h[j] += d
            h[k] += d
            c5_try = compute_c5(h)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h.copy()
                improved = True
            else:
                h[i] += 2 * d
                h[j] -= d
                h[k] -= d
        if not improved and h[j] >= d and h[k] >= d and h[i] <= 1 - 2 * d:
            h[j] -= d
            h[k] -= d
            h[i] += 2 * d
            c5_try = compute_c5(h)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h.copy()
                improved = True
            else:
                h[j] += d
                h[k] += d
                h[i] -= 2 * d
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 160
        best_h = make_thue_morse(best_n)
        best_c5 = compute_c5(best_h)
        print("No parent, Thue-Morse: n={}, C5={:.10f}".format(best_n, best_c5))
    
    # Phase 1: Try different resolutions (10% budget)
    phase1_budget = 0.10 * budget
    phase1_end = t0 + phase1_budget
    
    for n_try in [128, 144, 160, 176, 192, 208, 224, 240, 256]:
        if time.time() > phase1_end:
            break
        if n_try < 64 or n_try > 320:
            continue
        h_try = interp_h(best_h, best_n, n_try)
        h_try = project_h(h_try, n_try)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try
            print("  Phase1 n={}: C5={:.10f}".format(n_try, c5_try))
    
    # Also try structured inits at best_n
    for init_fn_args in [('tm', None), ('rand', None), ('sine3', None), ('sine5', None)]:
        if time.time() > phase1_end:
            break
        kind, _ = init_fn_args
        if kind == 'tm':
            h_init = make_thue_morse(best_n)
        elif kind == 'rand':
            h_init = make_random_binary(best_n, rng)
        elif kind == 'sine3':
            h_init = make_sine_init(best_n, 3, np.pi / 4, rng)
        elif kind == 'sine5':
            h_init = make_sine_init(best_n, 5, np.pi / 6, rng)
        else:
            continue
        c5_init = compute_c5(h_init)
        if c5_init < best_c5:
            best_c5 = c5_init
            best_h = h_init.copy()
            print("  Phase1 init {}: C5={:.10f}".format(kind, best_c5))
    
    # Phase 2: LP-based optimization (60% budget)
    phase2_budget = 0.60 * budget
    n = best_n
    h = best_h.copy()
    remaining = budget - (time.time() - t0)
    
    # Split phase 2 into multiple LP runs with different trust region sizes
    lp_splits = [
        (0.30, 10.0),  # 30% of phase2, rho=10
        (0.30, 5.0),   # 30% of phase2, rho=5
        (0.20, 3.0),   # 20% of phase2, rho=3
        (0.20, 1.5),   # 20% of phase2, rho=1.5
    ]
    
    for frac, rho_base in lp_splits:
        if time.time() - t0 > budget - 20:
            break
        sub_budget = min(phase2_budget * frac, budget - (time.time() - t0) - 15)
        if sub_budget < 10:
            break
        h, c5 = lp_optimize(h, n, sub_budget, max_iter=300)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            print("  LP rho={:.1f}: C5={:.10f}".format(rho_base, best_c5))
    
    # Phase 3: Smooth max SLSQP refinement (15% budget)
    phase3_budget = 0.15 * budget
    if time.time() - t0 < budget - 15:
        n = best_n
        h = best_h.copy()
        remaining3 = min(phase3_budget, budget - (time.time() - t0) - 10)
        
        for beta in [50, 100, 200, 500, 1000]:
            if time.time() - t0 > budget - 15:
                break
            sub_b = remaining3 / 5.0
            if sub_b < 5:
                break
            h_new = slsqp_smooth_max(h, n, beta, max_iter=150)
            h_new = project_h(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
                print("  SLSQP beta={}: C5={:.10f}".format(beta, best_c5))
    
    # Phase 4: Local perturbations (10% budget)
    phase4_budget = 0.10 * budget
    if time.time() - t0 < budget - 5:
        n = best_n
        h = best_h.copy()
        remaining4 = min(phase4_budget, budget - (time.time() - t0) - 5)
        
        # Pairwise
        h4a = h.copy()
        h4a, c5_4a = local_perturb(h4a, n, remaining4 * 0.6, rng, max_iter=80000)
        if c5_4a < best_c5:
            best_c5 = c5_4a
            best_h = h4a.copy()
            print("  Local pairwise: C5={:.10f}".format(best_c5))
        
        # Triple
        remaining4b = min(remaining4 * 0.4, budget - (time.time() - t0) - 3)
        if remaining4b > 3:
            h4b = best_h.copy()
            h4b, c5_4b = triple_perturb(h4b, n, remaining4b, rng, max_iter=30000)
            if c5_4b < best_c5:
                best_c5 = c5_4b
                best_h = h4b.copy()
                print("  Triple: C5={:.10f}".format(best_c5))
    
    # Phase 5: Final resolution check and polish
    if time.time() - t0 < budget - 5:
        for dn in range(-16, 17, 4):
            if time.time() - t0 > budget - 3:
                break
            nt = best_n + dn
            if nt < 80 or nt > 300:
                continue
            ht = interp_h(best_h, best_n, nt)
            ht = project_h(ht, nt)
            ct = compute_c5(ht)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()
                best_n = nt
                print("  Res n={}: C5={:.10f}".format(nt, best_c5))
    
    # Final SLSQP polish
    if time.time() - t0 < budget - 3:
        n = best_n
        h = best_h.copy()
        for beta_f in [2000, 5000]:
            if time.time() - t0 > budget - 3:
                break
            h_new = slsqp_smooth_max(h, n, beta_f, max_iter=100)
            h_new = project_h(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    # Final projection and evaluation
    best_h = project_h(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
