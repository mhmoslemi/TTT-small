def _c5(h):
    n = len(h)

def _proj(h, n):

def _smgrad(h, beta, k_top=None):

            wt = np.zeros_like(ct)
            wt[0] = 1.0
            s = 1.0

def _mslsqp(h0, n, beta, maxiter):
    ng = n
    bv = beta

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mx = float(np.max(c))
        sh = c - mx
        sm = float(np.sum(np.exp(bv * sh)))
        kg = int(ng)
        return (mx + np.log(sm) / bv) * (2.0 / kg)

    def con(x):
        return x.sum() - 0.5 * ng

    try:
        r = minimize(obj, h0.copy(), method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': con, 'jac': lambda x: np.ones(n)}],
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h = _proj(np.asarray(r.x, dtype=np.float64), n)
        return h
