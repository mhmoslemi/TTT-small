def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * n
    diff = target - np.sum(h)
    if diff > 0.0:
        order = np.argsort(1.0 - h)[::-1]
        rem = diff
        for idx in order:
            if rem <= 1e-15: break
            cap = 1.0 - h[idx]
            if cap > 0.0:
                add = cap if cap < rem else rem
                h[idx] += add
                rem -= add
    elif diff < 0.0:
        order = np.argsort(h)[::-1]
        rem = -diff
        for idx in order:
            if rem <= 1e-15: break
            sub = h[idx]
            if sub > 0.0:
                if sub > rem: sub = rem
                h[idx] -= sub
                rem -= sub
    # final tiny correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
       # adjust one variable with slack
       if diff > 0:

          h[idx] += add
       else:
          idx = int(np.argmax(h))
          sub = min(-diff, h[idx])
          h[idx] -= sub
    np.clip(h,0,1,out=h)
    return h

def unscaled_c5(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr))


def scaled_c5(h, n):
    return unscaled_c5(h) * (2.0 / float(n))


def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    if h.size != n:
        h = h[:n]
    np.clip(h, 0.0, 1.0, out=h)

    target = 0.5 * n
    diff = target - float(np.sum(h))

    if diff > 1e-14:
        order = np.argsort(1.0 - h)[::-1]
        rem = diff
        for idx in order:
            if rem <= 1e-15:
                break
            cap = 1.0 - h[idx]
            if cap > 0.0:
                add = cap if cap < rem else rem
                h[idx] += add
                rem -= add
    elif diff < -1e-14:
        order = np.argsort(h)[::-1]
        rem = -diff
        for idx in order:
            if rem <= 1e-15:
                break
            val = h[idx]
            if val > 0.0:
                sub = val if val < rem else rem
                h[idx] -= sub
                rem -= sub

    np.clip(h, 0.0, 1.0, out=h)

    for _ in range(12):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            cap = 1.0 - h[idx]
            if cap <= 0.0:
                break
            add = diff if diff < cap else cap
            h[idx] += add
        else:
            idx = int(np.argmax(h))
            val = h[idx]
            if val <= 0.0:
                break
            sub = -diff if -diff < val else val
            h[idx] -= sub

    return h


def interpolate_to(h_src, n_src, n_dst):
    if n_dst == n_src:
        return h_src.copy()
    xp = (np.arange(n_src) + 0.5) * (2.0 / float(n_src))
    xn = (np.arange(n_dst) + 0.5) * (2.0 / float(n_dst))
    h_new = np.interp(xn, xp, h_src)
    return project_feasible(h_new, n_dst)


def construct_fallback(n, rng):
    best_h = None
    best_val = float("inf")

    t = (np.arange(n) + 0.5) / float(n)
    candidates = []

    candidates.append(np.full(n, 0.5, dtype=float))

    h = np.zeros(n, dtype=float)
    h[:n // 2] = 1.0
    candidates.append(h)

    h = np.zeros(n, dtype=float)
    h[::2] = 1.0
    candidates.append(h)

    for _ in range(6):
        h = (rng.rand(n) > 0.5).astype(float)
        candidates.append(h)

    for f in [2, 3, 5, 8, 13, 21]:
        for ph in [0.0, 0.3, 0.7]:
            h = 0.5 + 0.5 * np.cos(2.0 * np.pi * f * t + 2.0 * np.pi * ph)
            candidates.append(h)

    for h in candidates:
        hp = project_feasible(h, n)
        v = unscaled_c5(hp)
        if v < best_val:
            best_val = v
            best_h = hp

    return best_h


def eval_pair(work, om, j, a, A, x):
    z = A - x
    work[j] = x
    work[a] = z
    om[j] = 1.0 - x
    om[a] = 1.0 - z
    corr = np.correlate(work, om, mode="full")
    return float(np.max(corr))


def coordinate_minimize(h, n, j, a, current_unscaled, grid_pts, refine_iters, full_iters):
    old = float(h[j])
    olda = float(h[a])
    A = old + olda

    lo = A - 1.0
    if lo < 0.0:
        lo = 0.0
    hi = A
    if hi > 1.0:
        hi = 1.0

    if hi - lo < 1e-12:
        return old, current_unscaled

    work = h.copy()
    om = 1.0 - work

    m = int(grid_pts)
    if m < 3:
        m = 3

    pts = np.linspace(lo, hi, m)
    pts = np.unique(np.concatenate((pts, np.array([old], dtype=float))))

    vals = np.empty(len(pts), dtype=float)
    best_x = old
    best_val = current_unscaled

    for i in range(len(pts)):
        x = float(pts[i])
        if x < lo:
            x = lo
        elif x > hi:
            x = hi
        val = eval_pair(work, om, j, a, A, x)
        vals[i] = val
        if val < best_val:
            best_val = val
            best_x = x

    if best_val < current_unscaled - 1e-8:
        order = np.argsort(pts)
        pts_s = pts[order]
        vals_s = vals[order]
        ib = int(np.argmin(vals_s))
        if ib > 0 and ib < len(pts_s) - 1:
            lo2 = float(pts_s[ib - 1])
            hi2 = float(pts_s[ib + 1])
        elif ib == 0:
            lo2 = float(pts_s[0])
            hi2 = float(pts_s[1])
        else:
            lo2 = float(pts_s[-2])
            hi2 = float(pts_s[-1])
        iters = int(refine_iters)
    else:
        lo2 = lo
        hi2 = hi
        iters = int(full_iters)

    if hi2 - lo2 > 1e-12 and iters > 0:
        if lo2 < lo:
            lo2 = lo
        if hi2 > hi:
            hi2 = hi

        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c = hi2 - (hi2 - lo2) * gr
        d = lo2 + (hi2 - lo2) * gr

        fc = eval_pair(work, om, j, a, A, c)
        if fc < best_val:
            best_val = fc
            best_x = c

        fd = eval_pair(work, om, j, a, A, d)
        if fd < best_val:
            best_val = fd
            best_x = d

        for _ in range(iters):
            if hi2 - lo2 <= 1e-12:
                break
            if fc < fd:
                hi2 = d
                d = c
                fd = fc
                c = hi2 - (hi2 - lo2) * gr
                fc = eval_pair(work, om, j, a, A, c)
                if fc < best_val:
                    best_val = fc
                    best_x = c
            else:
                lo2 = c
                c = d
                fc = fd
                d = lo2 + (hi2 - lo2) * gr
                fd = eval_pair(work, om, j, a, A, d)
                if fd < best_val:
                    best_val = fd
                    best_x = d

        xmid = 0.5 * (lo2 + hi2)
        fmid = eval_pair(work, om, j, a, A, xmid)
        if fmid < best_val:
            best_val = fmid
            best_x = xmid

    return best_x, best_val


def choose_anchor(h, rng, top_k):
    n = len(h)
    if n <= 1:
        return 0
    k = int(top_k)
    if k <= 1:
        slack = np.minimum(h, 1.0 - h)
        return int(np.argmax(slack))
    if k >= n:
        return int(rng.randint(n))

    slack = np.minimum(h, 1.0 - h)
    choices = np.argpartition(slack, n - k)[-k:]
    return int(choices[rng.randint(len(choices))])


def random_pair_optimize(h, n, current_unscaled, deadline, rng, grid_pts, refine_iters, full_iters):
    if n < 2:
        return h.copy(), current_unscaled

    h = h.copy()
    cur = float(current_unscaled)
    best_uns = cur
    best_h = h.copy()

    max_pairs = 4000

    for _ in range(max_pairs):
        if time.time() >= deadline:
            break
        j = int(rng.randint(n))
        a = int(rng.randint(n - 1))
        if a >= j:
            a += 1

        x_new, val_new = coordinate_minimize(h, n, j, a, cur, grid_pts, refine_iters, full_iters)
        if val_new < cur - 1e-12:
            oldj = float(h[j])
            olda = float(h[a])
            A = oldj + olda
            h[j] = x_new
            h[a] = A - x_new
            cur = val_new
            if cur < best_uns - 1e-12:
                best_uns = cur
                best_h = h.copy()

    return best_h, best_uns


def local_optimize(h, n, deadline, rng, grid_pts, refine_iters, full_iters):
    if n < 2:
        return h.copy(), scaled_c5(h, max(1, n))

    h = project_feasible(h, n)
    cur = unscaled_c5(h)

    if time.time() >= deadline:
        return h.copy(), cur * (2.0 / float(n))

    best_uns = cur
    best_h = h.copy()

    stagn = 0
    sweep = 0
    max_sweeps = 5000
    top_k = min(8, n)

    while sweep < max_sweeps:
        now = time.time()
        if now >= deadline - 12.0:
            break

        h = project_feasible(h, n)
        cur = unscaled_c5(h)

        if cur < best_uns - 1e-12:
            best_uns = cur
            best_h = h.copy()

        start_cur = cur
        anchor = choose_anchor(h, rng, top_k)

        if anchor == 0:
            order = np.arange(1, n)
        elif anchor == n - 1:
            order = np.arange(0, n - 1)
        else:
            order = np.arange(n)
            mask = np.ones(n, dtype=bool)
            mask[anchor] = False
            order = order[mask]

        rng.shuffle(order)

        for idx in order:
            if time.time() >= deadline - 5.0:
                break
            j = int(idx)
            x_new, val_new = coordinate_minimize(h, n, j, anchor, cur, grid_pts, refine_iters, full_iters)
            if val_new < cur - 1e-12:
                oldj = float(h[j])
                olda = float(h[anchor])
                A = oldj + olda
                h[j] = x_new
                h[anchor] = A - x_new
                cur = val_new
                if cur < best_uns - 1e-12:
                    best_uns = cur
                    best_h = h.copy()

        if time.time() >= deadline - 5.0:
            break

        if start_cur - cur < 1e-7 * max(1.0, start_cur):
            stagn += 1
            if stagn <= 3 and deadline - time.time() > 45.0:
                pair_deadline = min(deadline - 5.0, time.time() + min(30.0, 0.25 * (deadline - time.time())))
                g2 = max(3, grid_pts - 3)
                r2 = max(8, refine_iters // 2)
                f2 = max(4, full_iters // 2)
                h2, cur2 = random_pair_optimize(h, n, cur, pair_deadline, rng, g2, r2, f2)
                if cur2 < cur - 1e-12:
                    h = h2
                    cur = cur2
                    if cur < best_uns - 1e-12:
                        best_uns = cur
                        best_h = h.copy()
                    stagn = 0
                else:
                    if stagn >= 2:
                        break
            else:
                if stagn >= 2:
                    break
        else:
            stagn = 0

        sweep += 1

    best_h = project_feasible(best_h, n)
    final_uns = unscaled_c5(best_h)
    if final_uns < best_uns:
        best_uns = final_uns

    return best_h, best_uns * (2.0 / float(n))


def params_for_n(n):
    if n <= 192:
        return 9, 26, 22
    if n <= 320:
        return 8, 24, 20
    if n <= 512:
        return 7, 22, 18
    if n <= 1024:
        return 6, 20, 16
    return 5, 18, 14


def add_candidate(history, h, n, c5s):
    for i in range(len(history)):
        if history[i][1] == n:
            if c5s < history[i][2] - 1e-15:
                history[i] = (h.copy(), n, c5s)
            return
    history.append((h.copy(), n, c5s))
    if len(history) > 8:
        worst = 0
        worst_val = -1.0
        for i in range(len(history)):
            v = history[i][2]
            if v > worst_val:
                worst_val = v
                worst = i
        del history[worst]


def has_candidate_n(cands, n_target):
    for item in cands:
        if item[1] == n_target:
            return True
    return False


def select_final_candidates(history):
    out = []
    if not history:
        return out

    best_i = 0
    for i in range(1, len(history)):
        if history[i][2] < history[best_i][2]:
            best_i = i
    out.append(history[best_i])

    max_n_i = 0
    for i in range(1, len(history)):
        if history[i][1] > history[max_n_i][1]:
            max_n_i = i
    if not has_candidate_n(out, history[max_n_i][1]):
        out.append(history[max_n_i])

    second_i = -1
    second_val = float("inf")
    for i in range(len(history)):
        if i == best_i:
            continue
        if second_val == float("inf") or history[i][2] < second_val:
            second_val = history[i][2]
            second_i = i

    if second_i >= 0:
        cand = history[second_i]
        if not has_candidate_n(out, cand[1]):
            out.append(cand)

    return out


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)

    if budget_s < 20.0:
        try:
            ph = np.array(initial_h_values, dtype=float)
            if ph.size > 10:
                n0 = int(ph.size)
                ph = project_feasible(ph, n0)
                return ph, scaled_c5(ph, n0), n0
        except Exception:
            pass
        n0 = 128
        h0 = construct_fallback(n0, np.random.RandomState(seed))
        return h0, scaled_c5(h0, n0), n0

    rng = np.random.RandomState(seed)

    best_h = None
    best_n = 0
    best_uns = float("inf")

    try:
        ph = np.array(initial_h_values, dtype=float)
        if ph.size > 10:
            best_n = int(ph.size)
            best_h = project_feasible(ph, best_n)
            best_uns = unscaled_c5(best_h)
    except Exception:
        pass

    if best_h is None:
        best_n = 160
        best_h = construct_fallback(best_n, rng)
        best_uns = unscaled_c5(best_h)

    best_scaled = best_uns * (2.0 / float(best_n))

    base_n = best_n
    history = []
    add_candidate(history, best_h, best_n, best_scaled)

    final_reserve = 45.0 if budget_s > 150.0 else max(2.0, 0.1 * budget_s)
    outer_deadline = t0 + max(1.0, budget_s - final_reserve)
    available = max(1.0, budget_s - final_reserve)

    if base_n > 1600:
        resolutions = [base_n]
    else:
        cap = max(1024, int(base_n))
        if base_n <= 256 and budget_s > 500.0:
            high_target = 1024
        else:
            high_target = min(cap, max(800, int(round(5.0 * base_n))))

        raw = [
            base_n,
            2.0 * base_n,
            4.0 * base_n,
            float(high_target),
        ]

        resolutions = []
        for x in raw:
            nn = int(round(float(x)))
            if nn > cap:
                nn = cap
            if nn < 32:
                continue
            if nn not in resolutions:
                resolutions.append(nn)

    if not resolutions:
        resolutions = [base_n]

    if len(resolutions) == 1:
        weights = [1.0]
    elif len(resolutions) == 2:
        weights = [1.0, 1.8]
    elif len(resolutions) == 3:
        weights = [0.9, 1.5, 2.2]
    else:
        weights = [0.8, 1.4, 2.2, 3.0]
        if len(resolutions) > 4:
            extra = len(resolutions) - 4
            weights += [3.2] * extra

    total_w = float(sum(weights))

    for i, n in enumerate(resolutions):
        now = time.time()
        if now >= outer_deadline - 10.0:
            break

        alloc = (available * weights[i]) / total_w if total_w > 0.0 else available
        local_deadline = min(outer_deadline, time.time() + alloc)

        if n == best_n:
            h_init = best_h.copy()
        else:
            h_init = interpolate_to(best_h, best_n, n)

        grid, refi, fulli = params_for_n(n)
        h_new, c5_new = local_optimize(h_init, n, local_deadline, rng, grid, refi, fulli)

        if c5_new < best_scaled - 1e-14:
            best_scaled = c5_new
            best_h = h_new.copy()
            best_n = n

        add_candidate(history, h_new, n, c5_new)

    final_candidates = select_final_candidates(history)
    final_deadline = t0 + max(1.0, budget_s - 3.0)

    for cand in final_candidates:
        now = time.time()
        remaining = final_deadline - now
        if remaining <= 30.0:
            break

        alloc = min(remaining * 0.5, 240.0)
        h0, n0 = cand[0], cand[1]
        grid, refi, fulli = params_for_n(n0)
        local_deadline = min(final_deadline - 10.0, time.time() + alloc)
        h_pol, c5_pol = local_optimize(h0, n0, local_deadline, rng, grid, refi, fulli)

        if c5_pol < best_scaled - 1e-14:
            best_scaled = c5_pol
            best_h = h_pol.copy()
            best_n = n0

        add_candidate(history, h_pol, n0, c5_pol)

    best_h = project_feasible(best_h, best_n)
    final_uns = unscaled_c5(best_h)
    final_c5 = final_uns * (2.0 / float(best_n))

    try:
        ev = evaluate_erdos_solution(best_h)
        if ev is not None:
            ev = float(ev)
            if np.isfinite(ev) and ev < final_c5:
                final_c5 = ev
    except Exception:
        pass

    return best_h, float(final_c5), int(best_n)
