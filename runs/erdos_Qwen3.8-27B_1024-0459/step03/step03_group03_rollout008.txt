                h2[j] += (old - nv)
                h2 = _proj(h2, n)
                c2 = _c5(h2)
                if c2 < bc - 1e-14:
                    best = h2
                    bc = c2
                    imp = True
                    break
    if not imp:
        return best, bc
    return _cd_opt(best, n, nit, rng) if _tl() > 5 else (best, bc)


def _shake(h, n, rng, nshake=None):
    h2 = h.copy()
    if nshake is None:
        nshake = max(4, n // 15)
    idx = rng.choice(n, size=min(nshake, n), replace=False)
    h2[idx] = rng.uniform(0, 1, size=len(idx))
    h2 = _proj(h2, n)
    return h2


def _full_opt(h0, n, rng, time_limit=None):
    h = _proj(h0.copy(), n)
    c5 = _c5(h)
    t_start = time.time()

    for beta in [30, 100, 300, 1000]:
        if _tl() < 12:
            break
        h2 = _slsqp(h, n, beta, 180)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2

    h, c5 = _gradloop(h, n, 400, rng)
    h, c5 = _swap_opt(h, n, 60, rng)
    h, c5 = _gradloop(h, n, 250, rng)
    h, c5 = _cd_opt(h, n, 100, rng)

    if _tl() > 15:
        for beta in [2000, 6000]:
            if _tl() < 12:
                break
            h2 = _slsqp(h, n, beta, 250)
            c2 = _c5(h2)
            if c2 < c5:
                h, c5 = h2, c2
        h, c5 = _gradloop(h, n, 200, rng)
        h, c5 = _swap_opt(h, n, 40, rng, temp=0.001)

    if _tl() > 10:
        h, c5 = _cd_opt(h, n, 80, rng)
        h, c5 = _gradloop(h, n, 150, rng)

    return h, c5


def _fine_tune(h, n, rng):
    best = h.copy()
    bc = _c5(best)
    corr = np.correlate(best, 1.0 - best, mode='full')
    kpeak = int(np.argmax(corr))
    g = _ggrad(best, kpeak)
    gn = float(np.linalg.norm(g))
    if gn > 1e-15:
        g /= gn
        for a in np.arange(0.001, 0.05, 0.001):
            h2 = _proj(best - a * g, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-15:
                best = h2
                bc = c2
    # Try all shifts near the peak
    for dk in [-3, -2, -1, 0, 1, 2, 3]:
        if _tl() < 2:
            break
        k2 = kpeak + dk
        if k2 < 0 or k2 >= len(corr):
            continue
        g2 = _ggrad(best, k2)
        gn2 = float(np.linalg.norm(g2))
        if gn2 > 1e-15:
            g2 /= gn2
            for a in [0.01, 0.005, 0.002, 0.02]:
                h2 = _proj(best - a * g2, n)
                c2 = _c5(h2)
                if c2 < bc - 1e-15:
                    best = h2
                    bc = c2
                    break
    return best, bc


def run(seed=42, budget_s=1300, **kwargs):
    global _t0, _tb
    _t0 = time.time()
    _tb = budget_s
    rng = np.random.RandomState(seed)

    try:
        hpar = np.array(initial_h_values, dtype=np.float64)
        npar = len(hpar)
    except Exception:
        npar = 160
        hpar = np.zeros(npar, dtype=np.float64)
        hpar[:npar // 2] = 1.0

    bh = _proj(hpar, npar)
    bc5 = _c5(bh)
    bn = npar
    print("Init n={} C5={:.8f}".format(bn, bc5))

    h, c5 = _full_opt(bh, npar, rng)
    if c5 < bc5:
        bh, bc5, bn = h, c5, npar
        print("Opt n={} C5={:.8f}".format(npar, c5))

    alts = []
    for delta in [2, -2, 4, -4, 8, -8, 12, -12, 16, -16, 24, 32, 40, -40, 48]:
        nn = npar + delta
        if nn >= 48 and nn % 2 == 0 and nn not in alts:
            alts.append(nn)
    alts.sort(key=lambda x: abs(x - npar))

    for nn in alts:
        if _tl() < 35:
            break
        hi = _interp(bh, bn, nn)
        hi = _proj(hi, nn)
        h2, c2 = _full_opt(hi, nn, rng)
        if c2 < bc5:
            bh, bc5, bn = h2, c2, nn
            print("N={} C5={:.8f}".format(nn, c2))

    for trial in range(4):
        if _tl() < 50:
            break
        nshake = max(5, bn // 12)
        hshake = _shake(bh, bn, rng, nshake)
        h2, c2 = _full_opt(hshake, bn, rng)
        if c2 < bc5:
            bh, bc5 = h2, c2
            print("Shake{} N={} C5={:.8f}".format(trial, bn, c2))

    if _tl() > 50:
        bh, bc5 = _fine_tune(bh, bn, rng)
        h, c5 = _gradloop(bh, bn, 300, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _swap_opt(bh, bn, 50, rng, temp=0.0005)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _cd_opt(bh, bn, 100, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        bh, bc5 = _fine_tune(bh, bn, rng)
        print("Refine N={} C5={:.8f}".format(bn, bc5))

    if _tl() > 20:
        h2 = _slsqp(bh, bn, 8000, 300)
        c2 = _c5(h2)
        if c2 < bc5:
            bh, bc5 = h2, c2
        h, c5 = _gradloop(bh, bn, 200, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        bh, bc5 = _fine_tune(bh, bn, rng)

    if _tl() > 10:
        bh, bc5 = _fine_tune(bh, bn, rng)
        h, c5 = _gradloop(bh, bn, 100, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        bh, bc5 = _fine_tune(bh, bn, rng)

    bh = _proj(bh, bn)
    fc5 = _c5(bh)
    try:
        ec = evaluate_erdos_solution(bh)
        if ec < fc5:
            fc5 = ec
    except Exception:
        pass
    print("FINAL n={} C5={:.8f}".format(bn, fc5))
    return (bh, fc5, bn)
