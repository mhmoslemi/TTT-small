    best_c5 = compute_c5(h)
    momentum_vec = np.zeros(n)
    beta_mom = 0.85
    step_size = 0.02
    n_corr = 2 * n - 1
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining() < 10:
            break
        
        corr = compute_corr(h)
        # Top-K with softmax weighting
        sorted_idx = np.argsort(-corr)
        top = sorted_idx[:top_k]
        weights = np.exp((corr[top] - corr[top[0]]) * 200.0)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for w, k in zip(weights, top):
            grad += w * corr_grad_per_shift(h, n, int(k))
        
        momentum_vec = beta_mom * momentum_vec + grad
        h_trial = h - step_size * momentum_vec
        h = project_feasible(h_trial, n)
        
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            step_size = min(step_size * 1.005, 0.08)
        else:
            if c5_new > best_c5 * 1.005:
                step_size *= 0.95
            else:
                step_size = min(step_size * 1.002, 0.08)
    
    return h, best_c5


def active_set_equalize(h, n, max_iters=200, top_k=3):
    """Equalize top-K correlations via projected gradient."""
    h = h.copy()
    dx_val = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        top = np.argsort(-corr)[:top_k]
        c_max = corr[top[0]]
        c_min = corr[top[-1]]
        if c_max - c_min < 1e-10:
            break
        # Direction: reduce max, increase min
        g_max = corr_grad_per_shift(h, n, int(top[0]))
        g_min = corr_grad_per_shift(h, n, int(top[-1]))
        d = g_min - g_max
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-14:
            break
        d /= dnorm
        improved = False
        for step in [0.005, 0.01, 0.02, 0.05, 0.002, 0.03, 0.08, 0.015]:
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-13:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def binary_local_search(h, n, time_limit=30, rng=None):
    """Binary pair-swap local search."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        ones_idx = np.where(h == 1)[0]
        remove = rng.choice(ones_idx, n_ones - n // 2, replace=False)
        h[remove] = 0.0
    elif n_ones < n // 2:
        zeros_idx = np.where(h == 0)[0]
        add = rng.choice(zeros_idx, n // 2 - n_ones, replace=False)
        h[add] = 1.0
    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx_val
    best_h = h.copy()
    start = time.time()
    stall = 0
    
    for it in range(200000):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        k_max = int(np.argmax(corr))
        offset = k_max - (n - 1)
        # Sample swap candidates biased toward worst-shift contributors
        n_try = 200
        found = False
        for _ in range(n_try):
            if offset >= 0:
                candidates = np.array([i for i in ones if i + offset < n and h[i + offset] == 0])
                if len(candidates) > 0 and rng.rand() < 0.7:
                    i = candidates[rng.randint(len(candidates))]
                else:
                    i = ones[rng.randint(len(ones))]
            elif offset < 0:
                candidates = np.array([i for i in ones if i - offset < n and h[i - offset] == 0])
                if len(candidates) > 0 and rng.rand() < 0.7:
                    i = candidates[rng.randint(len(candidates))]
                else:
                    i = ones[rng.randint(len(ones))]
            else:
                i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            if new_max < cur_max - 1e-12:
                corr = new_corr
                cur_max = new_max
                best_c5 = cur_max * dx_val
                best_h = h.copy()
                found = True
                break
            else:
                h[i] = 1.0
                h[j] = 0.0
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 100:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                corr = compute_corr(h)
                cur_max = float(np.max(corr))
                stall = 0
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 320
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP refinement with increasing beta
    for beta in [100, 300, 1000, 3000, 8000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=350)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Subgradient with momentum
    if remaining() > 20:
        h_sg, c5_sg = subgradient_momentum(h, n, max_iters=5000, time_limit=45, top_k=5)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 3: Pair coordinate descent (main workhorse)
    if remaining() > 15:
        h_pc, c5_pc = pair_coord_descent(h, n, max_passes=500, time_limit=80, rng=rng)
        if c5_pc < best_c5:
            best_c5 = c5_pc
            best_h = h_pc.copy()
            h = h_pc.copy()

    # Phase 4: SLSQP polish after coord descent
    for beta in [2000, 5000, 15000, 30000]:
        if remaining() < 6:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: More coordinate descent
    if remaining() > 15:
        h_pc, c5_pc = pair_coord_descent(h, n, max_passes=400, time_limit=60, rng=rng)
        if c5_pc < best_c5:
            best_c5 = c5_pc
            best_h = h_pc.copy()
            h = h_pc.copy()

    # Phase 6: Active set equalization
    if remaining() > 10:
        h_eq, c5_eq = active_set_equalize(h, n, max_iters=300, top_k=3)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 7: Try other n values
    n_tries = sorted(set([
        n + 2, n + 4, n + 8, n + 16, n + 32,
        max(100, n - 4), max(100, n - 8)
    ]))
    n_tries = [nn for nn in n_tries if 80 <= nn <= 800 and nn % 2 == 0 and nn != n]

    for n_t in n_tries:
        if remaining() < 30:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        
        # SLSQP on new n
        h_opt = h_feas
        for beta in [500, 2000, 8000]:
            if remaining() < 8:
                break
            h_opt = slsqp_smooth(h_opt, n_t, beta, maxiter=250)
        
        # Pair coord descent
        if remaining() > 15:
            h_opt, _ = pair_coord_descent(h_opt, n_t, max_passes=200, time_limit=30, rng=rng)
        
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 8: Final intensive refinement on best solution
    n = best_n
    h = best_h.copy()

    # Final SLSQP
    for beta in [5000, 15000, 40000]:
        if remaining() < 5:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final coordinate descent
    if remaining() > 10:
        h_pc, c5_pc = pair_coord_descent(h, n, max_passes=300,
                                         time_limit=min(40, max(5, remaining() - 10)), rng=rng)
        if c5_pc < best_c5:
            best_c5 = c5_pc
            best_h = h_pc.copy()

    # Final binary local search
    if remaining() > 12:
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        if n_ones != best_n // 2:
            if n_ones > best_n // 2:
                idx = np.where(h_bin == 1)[0]
                h_bin[idx[rng.choice(len(idx), n_ones - best_n // 2, replace=False)]] = 0
            else:
                idx = np.where(h_bin == 0)[0]
                h_bin[idx[rng.choice(len(idx), best_n // 2 - n_ones, replace=False)]] = 1
        h_b, c5_b = binary_local_search(h_bin, best_n,
                                         time_limit=min(25, max(5, remaining() - 8)), rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = best_n

    # Final polish
    if remaining() > 6:
        h_new = slsqp_smooth(best_h, best_n, 20000, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Validate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
