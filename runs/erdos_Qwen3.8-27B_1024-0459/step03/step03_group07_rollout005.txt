    m = 2*n - 1
    base = n - 1
    A = np.zeros(m, dtype=float)
    if j == p:
        return A
    A[base] = 2.0 * (h[p] - h[j])
    hp = h[p]
    # for d loop
    for d in range(1, n):
        ap = 0.0
        am = 0.0
        # derivative for j in plus

def project_feasible(h, n):
    x = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(30):
        s = float(x.sum())
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = x < 1.0 - 1e-14
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            x[mask] += diff / cnt
        else:
            mask = x > 1e-14
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            x[mask] += diff / cnt
        x = np.clip(x, 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
       if diff > 0:
          order = np.argsort(1.0 - x)
          rem = diff
          for idx in order:
             cap = 1.0 - x[idx]
             if cap <= 0.0:
                continue
             add = min(rem, cap)
             x[idx] += add
             rem -= add
             if rem <= 1e-14:
                break
       else:
          order = np.argsort(x)
          rem = -diff
          for idx in order:
             cap = x[idx]
             if cap <= 0.0:
                continue
             sub = min(rem, cap)
             x[idx] -= sub
             rem -= sub
             if rem <= 1e-14:
                break
    return x

                rem -= add
                if rem <= 1e-14:
                    break
        else:
            order = np.argsort(x)[::-1]
            rem = -diff
            for idx in order:
                cap = x[idx]
                if cap <= 0.0:
                    continue
                sub = min(rem, cap)
                x[idx] -= sub
                rem -= sub
                if rem <= 1e-14:
                    break
    return x


def c5_from_array(h):
    n = int(len(h))
    if n <= 0:
        return float("inf")
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / float(n)))


def interpolate_to_n(h_src, n_src, n_dst):
    n_dst = int(n_dst)
    n_src = int(n_src)
    if n_dst <= 0:
        return np.array([], dtype=float)
    if n_src == n_dst:
        return project_feasible(h_src, n_dst)
    xp = (np.arange(n_src, dtype=float) + 0.5) * (2.0 / float(n_src))
    xn = (np.arange(n_dst, dtype=float) + 0.5) * (2.0 / float(n_dst))
    h = np.interp(xn, xp, np.asarray(h_src, dtype=float))
    return project_feasible(h, n_dst)


def lag_derivative_difference(h, j, p, n):
    m = 2 * n - 1
    base = n - 1
    A = np.zeros(m, dtype=float)
    if j == p:
        return A

    A[base] = 2.0 * (float(h[p]) - float(h[j]))

    for d in range(1, n):
        ap = 0.0
        am = 0.0

        if j >= d:
            v = h[j - d]
            ap += 1.0 - v
            am -= v
        if j + d < n:
            v = h[j + d]
            ap -= v
            am += 1.0 - v

        if p >= d:
            v = h[p - d]
            ap -= 1.0 - v
            am += v
        if p + d < n:
            v = h[p + d]
            ap += v
            am -= 1.0 - v

        A[base + d] = ap
        A[base - d] = am

    return A


def feasible_quad_vec(v, max_zero, Al_pos, Cl_pos, Al_neg, Cl_neg,
                      L0, R0,
                      A_base, C_base,
                      A_sp1, C_sp1, A_sp2, C_sp2):
    if max_zero > v + 1e-11:
        return False

    L = float(L0)
    R = float(R0)

    if Al_pos.size > 0:
        val = float(np.min((v - Cl_pos) / Al_pos))
        if val < R:
            R = val

    if Al_neg.size > 0:
        val = float(np.max((v - Cl_neg) / Al_neg))
        if val > L:
            L = val

    if L > R + 1e-11:
        return False

    D = A_sp1 * A_sp1 + 4.0 * (v - C_sp1)
    if D < -1e-9:
        return False
    if D < 0.0:
        D = 0.0
    rs = math.sqrt(D)
    r1 = -0.5 * (A_sp1 + rs)
    r2 = -0.5 * (A_sp1 - rs)
    if r1 > L:
        L = r1
    if r2 < R:
        R = r2
    if L > R + 1e-11:
        return False

    D = A_sp2 * A_sp2 + 4.0 * (v - C_sp2)
    if D < -1e-9:
        return False
    if D < 0.0:
        D = 0.0
    rs = math.sqrt(D)
    r1 = -0.5 * (A_sp2 + rs)
    r2 = -0.5 * (A_sp2 - rs)
    if r1 > L:
        L = r1
    if r2 < R:
        R = r2
    if L > R + 1e-11:
        return False

    qmax = C_base + (A_base * A_base) / 8.0
    if v < qmax - 1e-10:
        D2 = A_base * A_base - 8.0 * (v - C_base)
        if D2 < 0.0:
            D2 = 0.0
        rs2 = math.sqrt(D2)
        b1 = 0.25 * (A_base - rs2)
        b2 = 0.25 * (A_base + rs2)
        if (L > b1 + 1e-10) and (b2 > R + 1e-10):
            return False

    return True


def choose_delta_quad_vec(v, max_zero, Al_pos, Cl_pos, Al_neg, Cl_neg,
                          L0, R0,
                          A_base, C_base,
                          A_sp1, C_sp1, A_sp2, C_sp2,
                          rng, random_select):
    if max_zero > v + 1e-9:
        return 0.0

    L = float(L0)
    R = float(R0)

    if Al_pos.size > 0:
        val = float(np.min((v - Cl_pos) / Al_pos))
        if val < R:
            R = val

    if Al_neg.size > 0:
        val = float(np.max((v - Cl_neg) / Al_neg))
        if val > L:
            L = val

    if L > R + 1e-10:
        return 0.0

    D = A_sp1 * A_sp1 + 4.0 * (v - C_sp1)
    if D < -1e-8:
        return 0.0
    if D < 0.0:
        D = 0.0
    rs = math.sqrt(D)
    r1 = -0.5 * (A_sp1 + rs)
    r2 = -0.5 * (A_sp1 - rs)
    if r1 > L:
        L = r1
    if r2 < R:
        R = r2
    if L > R + 1e-10:
        return 0.0

    D = A_sp2 * A_sp2 + 4.0 * (v - C_sp2)
    if D < -1e-8:
        return 0.0
    if D < 0.0:
        D = 0.0
    rs = math.sqrt(D)
    r1 = -0.5 * (A_sp2 + rs)
    r2 = -0.5 * (A_sp2 - rs)
    if r1 > L:
        L = r1
    if r2 < R:
        R = r2
    if L > R + 1e-10:
        return 0.0

    intervals = []
    qmax = C_base + (A_base * A_base) / 8.0
    if v >= qmax - 1e-9:
        intervals.append((L, R))
    else:
        D2 = A_base * A_base - 8.0 * (v - C_base)
        if D2 < 0.0:
            D2 = 0.0
        rs2 = math.sqrt(D2)
        b1 = 0.25 * (A_base - rs2)
        b2 = 0.25 * (A_base + rs2)

        if L <= b1 + 1e-9:
            rr = min(R, b1)
            if L <= rr + 1e-9:
                intervals.append((L, rr))
        if b2 <= R + 1e-9:
            ll = max(L, b2)
            if ll <= R + 1e-9:
                intervals.append((ll, R))

    valid = []
    total = 0.0
    for a, b in intervals:
        if a < L0:
            a = L0
        if b > R0:
            b = R0
        if b >= a - 1e-12:
            ln = max(0.0, b - a)
            valid.append((a, b, ln))
            total += ln

    if not valid:
        return 0.0

    if len(valid) == 1:
        a, b, ln = valid[0]
        if random_select and rng is not None and ln > 1e-13:
            return float(a + rng.random() * ln)
        return float(0.5 * (a + b))

    if random_select and rng is not None and total > 0.0:
        u = rng.random() * total
        for a, b, ln in valid:
            if u < ln:
                return float(a + u)
            u -= ln

    best_ln = -1.0
    best_x = 0.0
    for a, b, ln in valid:
        if ln > best_ln:
            best_ln = ln
            best_x = 0.5 * (a + b)
    return float(best_x)


def eval_delta_quad_vec(delta, A, C, base, sp1, sp2):
    val = C + A * delta
    d2 = delta * delta
    val[base] -= 2.0 * d2
    val[sp1] += d2
    val[sp2] += d2
    return float(np.max(val))


def coordinate_sweeps_quad(h_in, n, max_sweeps, deadline, pivot, rng,
                           random_order=True, random_select=True):
    n = int(n)
    h = project_feasible(h_in, n)
    c = np.correlate(h, 1.0 - h, mode="full")
    cmax = float(np.max(c))
    c5 = cmax * (2.0 / float(n))

    best_h = h.copy()
    best_c5 = c5
    best_cmax = cmax

    m = 2 * n - 1
    base = n - 1

    adaptive_pivot = (pivot is None)
    if not adaptive_pivot:
        p0 = int(pivot)
        if p0 < 0:
            p0 = 0
        if p0 >= n:
            p0 = n - 1
    else:
        p0 = 0

    last_improve = best_c5
    stagnant = 0

    for sweep in range(int(max_sweeps)):
        if time.time() > deadline:
            break

        if adaptive_pivot:
            p = int(np.argmin(np.abs(h - 0.5)))
        else:
            p = p0

        free = []
        for i in range(n):
            if i != p:
                free.append(i)

        if random_order and rng is not None:
            order = free.copy()
            rng.shuffle(order)
        else:
            order = free

        check_count = 0
        for j in order:
            check_count += 1
            if (check_count & 15) == 0 and time.time() > deadline:
                break

            oj = float(h[j])
            op = float(h[p])

            lo = max(-oj, op - 1.0)
            hi = min(1.0 - oj, op)
            if hi <= lo + 1e-14:
                continue

            A = lag_derivative_difference(h, j, p, n)
            d = abs(j - p)
            sp1 = base + d
            sp2 = base - d

            mask = np.ones(m, dtype=bool)
            mask[base] = False
            mask[sp1] = False
            mask[sp2] = False

            Al_all = A[mask]
            Cl_all = c[mask]

            pos = Al_all > 1e-12
            neg = Al_all < -1e-12
            zero = ~(pos | neg)

            Cl_zero = Cl_all[zero]
            max_zero = float(Cl_zero.max()) if Cl_zero.size > 0 else -1.0

            Al_pos = Al_all[pos]
            Cl_pos = Cl_all[pos]
            Al_neg = Al_all[neg]
            Cl_neg = Cl_all[neg]

            A_base = float(A[base])
            C_base = float(c[base])
            A_sp1 = float(A[sp1])
            C_sp1 = float(c[sp1])
            A_sp2 = float(A[sp2])
            C_sp2 = float(c[sp2])

            low = 0.0
            high = cmax + 1e-11

            for _ in range(34):
                midv = 0.5 * (low + high)
                if feasible_quad_vec(midv, max_zero, Al_pos, Cl_pos, Al_neg, Cl_neg,
                                     lo, hi,
                                     A_base, C_base,
                                     A_sp1, C_sp1, A_sp2, C_sp2):
                    high = midv
                else:
                    low = midv

            chosen = choose_delta_quad_vec(
                high + 1e-10,
                max_zero, Al_pos, Cl_pos, Al_neg, Cl_neg,
                lo, hi,
                A_base, C_base,
                A_sp1, C_sp1, A_sp2, C_sp2,
                rng, random_select
            )

            cand = [0.0, lo, hi]
            if chosen is not None:
                cand.append(chosen)

            best_delta = 0.0
            best_val = cmax

            for delta in cand:
                if delta < lo:
                    delta = lo
                elif delta > hi:
                    delta = hi
                val = eval_delta_quad_vec(delta, A, c, base, sp1, sp2)
                if val < best_val:
                    best_val = val
                    best_delta = float(delta)

            if abs(best_delta) > 1e-14:
                c = c + A * best_delta
                d2 = best_delta * best_delta
                c[base] -= 2.0 * d2
                c[sp1] += d2
                c[sp2] += d2

                h[j] = oj + best_delta
                h[p] = op - best_delta

                new_cmax = float(np.max(c))
                if new_cmax < best_cmax:
                    best_cmax = new_cmax
                    best_c5 = new_cmax * (2.0 / float(n))
                    best_h = h.copy()
                cmax = new_cmax

        if time.time() < deadline - 1.0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode="full")
            cmax_proj = float(np.max(c))

            if cmax_proj < best_cmax + 1e-10:
                cmax = cmax_proj
                if cmax < best_cmax:
                    best_cmax = cmax
                    best_c5 = cmax * (2.0 / float(n))
                    best_h = h.copy()
            else:
                h = best_h.copy()
                c = np.correlate(h, 1.0 - h, mode="full")
                cmax = best_cmax

        if best_c5 < last_improve - 1e-12:
            stagnant = 0
            last_improve = best_c5
        else:
            stagnant += 1

        if stagnant >= 4 and sweep > 4:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    total_budget = max(10.0, float(budget_s))
    reserve = max(8.0, min(25.0, 0.06 * total_budget))
    absolute_deadline = t0 + max(1.0, total_budget - reserve)

    rng = np.random.RandomState(int(seed))

    best_h = None
    best_n = 0
    best_c5 = float("inf")

    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = int(len(ph))
        if pn > 10:
            best_h = project_feasible(ph, pn)
            best_n = pn
            best_c5 = c5_from_array(best_h)
    except Exception:
        best_h = None

    if best_h is None:
        best_n = 160
        tx = (np.arange(best_n, dtype=float) + 0.5) / float(best_n)
        best_h = 0.5 + 0.45 * np.cos(2.0 * np.pi * 2.0 * tx)
        best_h = project_feasible(best_h, best_n)
        best_c5 = c5_from_array(best_h)
        best_n = int(best_n)

    # Primary deep adaptive coordinate phase.
    rem = absolute_deadline - time.time()
    if rem > 30.0:
        alloc = rem * 0.40
        if alloc < 25.0:
            alloc = rem
        deadline = time.time() + alloc
        if deadline > absolute_deadline:
            deadline = absolute_deadline
        try:
            res_h, res_c5 = coordinate_sweeps_quad(
                best_h, best_n, 800, deadline, -1, rng, True, True)
            cand_h = project_feasible(res_h, best_n)
            cand_c5 = c5_from_array(cand_h)
            if cand_c5 < best_c5:
                best_h = cand_h
                best_n = int(len(cand_h))
                best_c5 = cand_c5
        except Exception:
            pass

    # Fixed-pivot phases to try different coordinate axes.
    fixed_pivots = []
    for p in [0, best_n - 1, best_n // 4, (3 * best_n) // 4]:
        if 0 <= p < best_n and p not in fixed_pivots:
            fixed_pivots.append(int(p))

    for p in fixed_pivots:
        rem = absolute_deadline - time.time()
        if rem < 25.0:
            break
        alloc = rem * 0.15
        if alloc < 15.0:
            alloc = min(15.0, rem)
        deadline = time.time() + alloc
        if deadline > absolute_deadline:
            deadline = absolute_deadline
        try:
            res_h, res_c5 = coordinate_sweeps_quad(
                best_h, best_n, 400, deadline, p, rng, True, True)
            cand_h = project_feasible(res_h, best_n)
            cand_c5 = c5_from_array(cand_h)
            if cand_c5 < best_c5:
                best_h = cand_h
                best_n = int(len(cand_h))
                best_c5 = cand_c5
        except Exception:
            pass

    # Symmetrization and complement phases.
    rem = absolute_deadline - time.time()
    if rem > 35.0:
        alloc = rem * 0.10
        if alloc < 12.0:
            alloc = min(12.0, rem)
        deadline = time.time() + alloc
        if deadline > absolute_deadline:
            deadline = absolute_deadline
        try:
            h_sym = project_feasible(0.5 * (best_h + best_h[::-1]), best_n)
            res_h, res_c5 = coordinate_sweeps_quad(
                h_sym, best_n, 250, deadline, -1, rng, True, True)
            cand_h = project_feasible(res_h, best_n)
            cand_c5 = c5_from_array(cand_h)
            if cand_c5 < best_c5:
                best_h = cand_h
                best_n = int(len(cand_h))
                best_c5 = cand_c5
        except Exception:
            pass

    rem = absolute_deadline - time.time()
    if rem > 30.0:
        alloc = rem * 0.10
        if alloc < 10.0:
            alloc = min(10.0, rem)
        deadline = time.time() + alloc
        if deadline > absolute_deadline:
            deadline = absolute_deadline
        try:
            h_comp = project_feasible(1.0 - best_h, best_n)
            res_h, res_c5 = coordinate_sweeps_quad(
                h_comp, best_n, 250, deadline, -1, rng, True, True)
            cand_h = project_feasible(res_h, best_n)
            cand_c5 = c5_from_array(cand_h)
            if cand_c5 < best_c5:
                best_h = cand_h
                best_n = int(len(cand_h))
                best_c5 = cand_c5
        except Exception:
            pass

    # Local perturbations.
    for sigma in (0.02, 0.05, 0.10, 0.20):
        rem = absolute_deadline - time.time()
        if rem < 18.0:
            break
        try:
            h_init = project_feasible(best_h + float(sigma) * rng.randn(best_n), best_n)
            init_c5 = c5_from_array(h_init)
            if init_c5 > best_c5 + 0.08:
                continue
            alloc = rem * 0.08
            if alloc < 8.0:
                alloc = min(8.0, rem)
            deadline = time.time() + alloc
            if deadline > absolute_deadline:
                deadline = absolute_deadline
            res_h, res_c5 = coordinate_sweeps_quad(
                h_init, best_n, 220, deadline, -1, rng, True, True)
            cand_h = project_feasible(res_h, best_n)
            cand_c5 = c5_from_array(cand_h)
            if cand_c5 < best_c5:
                best_h = cand_h
                best_n = int(len(cand_h))
                best_c5 = cand_c5
        except Exception:
            pass

    # One subset-reset perturbation.
    rem = absolute_deadline - time.time()
    if rem > 20.0:
        try:
            h_mod = best_h.copy()
            mm = max(1, best_n // 20)
            idx = rng.choice(best_n, size=mm, replace=False)
            h_mod[idx] = rng.rand(mm)
            h_init = project_feasible(h_mod, best_n)
            init_c5 = c5_from_array(h_init)
            if init_c5 <= best_c5 + 0.10:
                alloc = rem * 0.08
                if alloc < 8.0:
                    alloc = min(8.0, rem)
                deadline = time.time() + alloc
                if deadline > absolute_deadline:
                    deadline = absolute_deadline
                res_h, res_c5 = coordinate_sweeps_quad(
                    h_init, best_n, 200, deadline, -1, rng, True, True)
                cand_h = project_feasible(res_h, best_n)
                cand_c5 = c5_from_array(cand_h)
                if cand_c5 < best_c5:
                    best_h = cand_h
                    best_n = int(len(cand_h))
                    best_c5 = cand_c5
        except Exception:
            pass

    # Multi-resolution refinement.
    res_list = []
    for nn in (192, 224, 256, 288, 320, 384, 448):
        if nn > best_n and nn <= 512 and nn not in res_list:
            res_list.append(int(nn))

    base = max(64, best_n)
    for mult in (1.25, 1.5, 1.75, 2.0):
        nn = int(round(float(base) * float(mult) / 2.0) * 2)
        if nn > best_n and nn <= 512 and nn not in res_list:
            res_list.append(int(nn))
    res_list.sort()

    for nn in res_list:
        rem = absolute_deadline - time.time()
        if rem < 50.0:
            break
        if nn <= best_n:
            continue
        alloc = rem * 0.28
        if alloc < 30.0:
            alloc = min(30.0, rem)
        deadline = time.time() + alloc
        if deadline > absolute_deadline:
            deadline = absolute_deadline
        try:
            h_init = interpolate_to_n(best_h, best_n, nn)
            res_h, res_c5 = coordinate_sweeps_quad(
                h_init, nn, 400, deadline, -1, rng, True, True)
            cand_h = project_feasible(res_h, nn)
            cand_c5 = c5_from_array(cand_h)
            if cand_c5 < best_c5:
                best_h = cand_h
                best_n = int(len(cand_h))
                best_c5 = cand_c5
        except Exception:
            pass

    # Final polish on the best feasible construction.
    rem = absolute_deadline - time.time()
    if rem > 15.0:
        try:
            deadline = absolute_deadline
            res_h, res_c5 = coordinate_sweeps_quad(
                best_h, best_n, 1200, deadline, -1, rng, True, True)
            cand_h = project_feasible(res_h, best_n)
            cand_c5 = c5_from_array(cand_h)
            if cand_c5 < best_c5:
                best_h = cand_h
                best_n = int(len(cand_h))
                best_c5 = cand_c5
        except Exception:
            pass

    best_h = project_feasible(best_h, best_n)
    best_n = int(len(best_h))
    final_c5 = c5_from_array(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        ec5 = float(ec5)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
