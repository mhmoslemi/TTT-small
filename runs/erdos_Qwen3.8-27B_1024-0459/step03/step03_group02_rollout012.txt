        #          for j in [offset, n-1]: -h[j-offset]

        grad[offset:] -= h[:n_pairs]

            break

        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n

    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:

        for idx in order:
            if diff > 0:
                adj = min(diff, 1.0 - h[idx])

            h[idx] += adj
            diff -= adj

    grad = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        if np_ <= 0:
            return grad

        grad[offset:] -= h[:np_]

        ao = -offset
        np_ = n - ao
        if np_ <= 0:
            return grad
        grad[ao:] += (1.0 - h[:np_])
        grad[:np_] -= h[ao:ao + np_]
    return grad
