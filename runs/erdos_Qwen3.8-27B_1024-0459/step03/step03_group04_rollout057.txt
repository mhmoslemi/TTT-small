def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr


def project_h(v, n):
    v = np.array(v, dtype=float).copy()
    v = np.clip(v, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(v)
        if abs(diff) < 1e-15:
            break
        free = (v > 1e-13) & (v < 1.0 - 1e-13)
        cnt = float(np.sum(free))
        if cnt > 1e-10:
            v[free] += diff / cnt
        else:
            v[:] += diff / n
        v = np.clip(v, 0.0, 1.0)
    return v


def smooth_max(vals, beta):
    m = np.max(vals)
    if beta < 1e-4:
        return m
    shifted = np.clip(beta * (vals - m), -500, 0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

    if n_src == n_dst:

    return np.interp(xq, xp, src)
