                        lin[fidx_neg] += (-h[j + s])
                    else:
                        lin[fidx_neg] = -h[j + s]
            
            # Max of unaffected correlations
            affected_idx = set(lin.keys()) | {n - 1}  # lag 0 also affected
            max_unaff = 0.0
            for fi in range(n_full):
                if fi not in affected_idx:
                    if corr[fi] > max_unaff:
                        max_unaff = corr[fi]
            
            # 1D optimization: minimize max(max_unaff, corr[n-1] - old*(1-old) + x*(1-x), max_{fi in lin}(corr[fi] + lin[fi]*(x-old)))
            # subject to 0 <= x <= 1
            
            # Golden section search
            phi = (1.0 + 5**0.5) / 2.0
            a, b = 0.0, 1.0
            c_ = b - (b - a) / phi
            d_ = a + (b - a) / phi
            
            def obj_1d(x):
                val = max_unaff
                # Lag 0
                v0 = corr[n-1] - old * (1.0 - old) + x * (1.0 - x)
                if v0 > val:
                    val = v0
                # Other lags
                for fi, lc in lin.items():
                    v = corr[fi] + lc * (x - old)
                    if v > val:
                        val = v
                return val
            
            fc = obj_1d(c_)
            fd = obj_1d(d_)
            for _ in range(40):
                if fc < fd:
                    b = d_
                    d_ = c_
                    c_ = b - (b - a) / phi
                    fc = obj_1d(c_)
                else:
                    a = c_
                    c_ = d_
                    d_ = a + (b - a) / phi
                    fd = obj_1d(d_)
            
            new_x = (a + b) / 2.0
            h[j] = new_x
            # Update corr
            for fi, lc in lin.items():
                corr[fi] = corr[fi] + lc * (new_x - old)
            corr[n-1] = corr[n-1] - old * (1.0 - old) + new_x * (1.0 - new_x)
            
            if new_x != old:
                improved = True
            
            # Fix sum
            # Actually, we need to maintain sum(h) = n/2
            # The change in sum is new_x - old
            # We need to distribute this among other coordinates
            # For now, skip this and project at end of sweep
            # (The projection might undo some progress, but it's O(n) and fast)
        
        # Project after sweep
        h = project_feasible(h, n)
        corr = compute_corr(h)  # Recompute to avoid accumulated errors
        c5 = np.max(corr) * dx
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        if not improved and sw > 2:
            break
    
    return best_h, best_c5

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - h)[:max(1, int(np.ceil(abs(diff))))]
        else:
            idx = np.argsort(h)[::-1][:max(1, int(np.ceil(abs(diff))))]
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_src, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_new = np.interp(xn, xp, h_src)
    return project_feasible(h_new, n_dst)

def coord_descent(h, n, sweeps=50, time_limit=None, t0=None, anneal_temp=0.0, rng=None):
    h = project_feasible(h, n).copy()
    dx = 2.0 / n
    corr = compute_corr(h)
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    n_full = 2 * n - 1
    phi = (1.0 + 5**0.5) / 2.0

    for sw in range(sweeps):
        if time_limit and t0 and time.time() - t0 > time_limit:
            break
        improved_any = False
        for j in range(n):
            if time_limit and t0 and time.time() - t0 > time_limit:
                break
            old = h[j]
            lin = {}
            for k in range(1, n):
                fidx_pos = (n - 1) + k
                if j <= n - 1 - k:
                    lin[fidx_pos] = lin.get(fidx_pos, 0.0) + (1.0 - h[j + k])
                if k <= j:
                    lin[fidx_pos] = lin.get(fidx_pos, 0.0) - h[j - k]
                fidx_neg = (n - 1) - k
                if k <= j:
                    lin[fidx_neg] = lin.get(fidx_neg, 0.0) + (1.0 - h[j - k])
                if j <= n - 1 - k:
                    lin[fidx_neg] = lin.get(fidx_neg, 0.0) - h[j + k]
            affected = set(lin.keys())
            affected.add(n - 1)
            max_unaff = 0.0
            for fi in range(n_full):
                if fi not in affected:
                    if corr[fi] > max_unaff:
                        max_unaff = corr[fi]
            c0_base = corr[n - 1] - old * (1.0 - old)
            vals_lin = [(fi, lc, corr[fi]) for fi, lc in lin.items()]

            a, b = 0.0, 1.0
            c_pt = b - (b - a) / phi
            d_pt = a + (b - a) / phi
            fc = max(max_unaff, c0_base + c_pt * (1.0 - c_pt))
            for fi, lc, cv in vals_lin:
                v = cv + lc * (c_pt - old)
                if v > fc:
                    fc = v
            fd = max(max_unaff, c0_base + d_pt * (1.0 - d_pt))
            for fi, lc, cv in vals_lin:
                v = cv + lc * (d_pt - old)
                if v > fd:
                    fd = v
            for _ in range(35):
                if fc < fd:
                    b = d_pt
                    d_pt = c_pt
                    c_pt = b - (b - a) / phi
                    fc = max(max_unaff, c0_base + c_pt * (1.0 - c_pt))
                    for fi, lc, cv in vals_lin:
                        v = cv + lc * (c_pt - old)
                        if v > fc:
                            fc = v
                else:
                    a = c_pt
                    c_pt = d_pt
                    d_pt = a + (b - a) / phi
                    fd = max(max_unaff, c0_base + d_pt * (1.0 - d_pt))

            else:
                break
    return best_h, best_c5

def smooth_max_loss(h, n, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -500.0, 0.0)
    return (m + np.log(np.sum(np.exp(sh))) / beta) * (2.0 / n)

def smooth_max_grad(h, n, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -500.0, 0.0)
    exps = np.exp(sh)
    s = np.sum(exps)
    weights = exps / s
    grad = np.zeros(n)
    for fi, w in enumerate(weights):
        if w < 1e-10:
            continue
        lag = fi - (n - 1)
        if lag >= 0:
            k = lag
            for i in range(n - k):
                grad[i] += w * (1.0 - h[i + k])
                grad[i + k] += w * (-h[i])
        else:
            s_abs = -lag
            for i in range(n - s_abs):
                grad[i + s_abs] += w * (1.0 - h[i])
                grad[i] += w * (-h[i + s_abs])
    return grad * (2.0 / n)

def projected_gradient(h, n, iters=200, lr0=0.01, beta=200.0, time_limit=None, t0=None):
    h = project_feasible(h, n).copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    stagn = 0
    for it in range(iters):
        if time_limit and t0 and time.time() - t0 > time_limit:
            break
        g = smooth_max_grad(h, n, beta)
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best = h_new.copy()
            h = h_new
            stagn = 0
        else:
            lr *= 0.8
            stagn += 1
            if stagn > 40:
                break
    return best, best_c5

def slsqp_smooth(h, n, beta, time_limit=None, t0=None):
    h = project_feasible(h, n).copy()
    bounds = [(0.0, 1.0)] * n
    def obj_fn(x):
        return smooth_max_loss(x, n, beta)
    def eq_fn(x):
        return np.sum(x) - n / 2.0
    cons = [{'type': 'eq', 'fun': eq_fn}]
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-12, 'iprint': 0})
        h_out = project_feasible(res.x, n)
        c5_out = compute_c5(h_out)
        return h_out, c5_out
    except Exception:
        return h, compute_c5(h)

def basinhopping_step(h, n, rng, scale, time_limit=None, t0=None):
    hp = h + scale * rng.randn(n)
    hp = project_feasible(hp, n)
    hc, c5c = coord_descent(hp, n, sweeps=8, time_limit=time_limit, t0=t0)
    return hc, c5c
