def subgrad_full_lag(h, k_idx, n):
    """Gradient of corr[k_idx] w.r.t. h"""
    tau = k_idx - (n - 1)
    grad = np.zeros(n)
    if tau == 0:
        grad[:] = 1.0 - 2.0 * h
    elif tau > 0:
        grad[tau:n] += (1.0 - h[:n-tau])
        grad[:n-tau] -= h[tau:n]
    else:
        t = -tau
        grad[:n-t] += (1.0 - h[t:n])
        grad[t:n] -= h[:n-t]
    return grad

def smooth_grad(h, beta, top_k=20):
    """Subgradient of log-sum-exp smoothed max of corr * dx"""
    n = len(h)
    dx = 2.0 / n

    exps = np.exp(beta * (c - mw))
    w = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for t, k_idx in enumerate(idx):
        if w[t] < 1e-12:
