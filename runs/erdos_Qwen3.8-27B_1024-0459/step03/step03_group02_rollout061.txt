        return h.copy()

def nesterov_phase(h, n, beta, n_steps=300, lr0=0.05):
    v = np.zeros(n)
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for t in range(1, n_steps+1):
        if remaining_time() < 5:
            break
        x = y.copy()
        gr = sm_grad(x, n, beta)
        gn = np.linalg.norm(gr)
        if gn < 1e-15:
            break
        lr = lr0 / (1.0 + 0.005 * t)
        v = 0.85 * v - lr * gr
        x_new = x + v
        x_new = project_feasible(x_new, n)
        c5_new = compute_c5(x_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = x_new.copy()
        mu = 2.0 / (t + 2.0)
        y = (1.0 - mu) * y + mu * x_new
    return best_h, best_c5

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def local_search(h, n, rounds=15, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for r in range(rounds):
        if remaining_time() < 3:
            break
        improved = False
        for i in rng.permutation(n):
            if remaining_time() < 2:
                break
            old = h[i]
            for d in [0.02, -0.02, 0.05, -0.05, 0.1, -0.1]:
                h[i] = np.clip(old + d, 0, 1)
                h = project_feasible(h, n)
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True
                    break
            if old != h[i] and old != best_h[i]:
                h = best_h.copy()
            # Pair swap
            j = (i + 1 + rng.randint(0, max(1, n//8))) % n
            if abs(h[i] - h[j]) > 0.15:
                h[i], h[j] = h[j], h[i]
                h = project_feasible(h, n)
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True
                else:
                    h = best_h.copy()
        if not improved:
            break
    return best_h, best_c5
