from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    """Project h to satisfy 0<=h<=1 and sum(h)=target_sum"""
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            adj = diff / np.sum(mask)
            adj = np.clip(adj, -1e-12, 1e-12)
            adj = diff / np.sum(mask)  # don't clip, let the next iteration handle
            h[mask] += adj
        else:
            h[:] += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        # Find closest to 0.5 to adjust
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_and_grad(h, beta, top_k=None):
    """Compute smooth max of correlation and its gradient w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    if top_k is not None and top_k < len(corr):
        idx = np.argpartition(corr, -top_k)[-top_k:]
        corr_work = corr[idx]
    else:
        idx = np.arange(len(corr))
        corr_work = corr
    
    m = np.max(corr_work)
    shifted = corr_work - m
    # Clip for numerical stability
    shifted = np.clip(shifted, -50/beta, 100)
    w = np.exp(beta * shifted)
    sw = np.sum(w)
    if sw < 1e-300:
        w = np.zeros_like(corr_work)
        w[np.argmax(corr_work)] = 1.0
        sw = 1.0
    w /= sw
    
    val = (m + np.log(sw) / beta) * dx
    
    # Gradient: d/dh of sum_k w_k * corr_k * dx
    # corr_k = sum_j h[j] * (1-h[j+k-n+1]) (need to be careful with indexing)
    # Actually let's compute it directly
    grad = np.zeros(n)
    for t, k_idx in enumerate(idx):
        if w[t] < 1e-15:
            continue
        k = k_idx - (n - 1)  # shift
        if k >= 0:
            i_lo = k
            i_hi = n - 1
        else:
            i_lo = 0
            i_hi = n - 1 + k
        if i_lo > i_hi:
            continue
        m_vals = np.arange(i_lo, i_hi + 1)
        j_vals = m_vals - k
        # d(corr_k)/d(h[m]) = 1 - h[j] (for the h[m] term)
        # d(corr_k)/d(h[j]) = -h[m] (for the (1-h[j]) term)
        grad[m_vals] += w[t] * (1.0 - h[j_vals])
        grad[j_vals] += w[t] * (-h[m_vals])
    
    grad *= dx
    return val, grad

        return m
