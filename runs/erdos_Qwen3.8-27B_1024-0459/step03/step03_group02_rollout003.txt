def adjust_sum(h,n):
    target=0.5*n
    h=np.array(h,dtype=float).copy()
    for _ in range(10):
      diff=target-h.sum()
      if abs(diff)<1e-13: break
      if diff>0: mask=h<1-1e-12
      else: mask=h>1e-12
      cnt=np.count_nonzero(mask)
      if cnt>0:
         h[mask] += diff/cnt
         h=np.clip(h,0,1)
      else:
         h += diff/n
         h=np.clip(h,0,1)
    diff=target-h.sum()
    if abs(diff)>1e-12:
       idx=int(np.argmax(1.0-h)) if diff>0 else int(np.argmax(h))
       h[idx]=float(np.clip(h[idx]+diff,0.0,1.0))
       diff=target-h.sum()
    if abs(diff)>1e-12:
       # last resort: distribute
       for j in range(n):
          if diff>0:
             s=min(diff,1.0-h[j]); h[j]+=s; diff-=s
          else:
             s=min(-diff,h[j]); h[j]-=s; diff+=s
          if abs(diff)<1e-14: break
    return h

def phase_time_left(deadline):

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    if n <= 0:
        return float('inf')
    return float(np.max(compute_corr(h)) * (2.0 / n))

def adjust_sum(h, n):
    h = np.array(h, dtype=float).copy()
    if n <= 0:
        return h
    target = 0.5 * n
    for _ in range(12):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            mask = h < (1.0 - 1e-12)
        else:
            mask = h > 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-12:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
        diff = target - float(np.sum(h))

                h[j] -= step

    return h

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

        return h
    h = np.clip(h, 0.0, 1.0)
    return adjust_sum(h, n)

def interp_to(src_h, src_n, dst_n):
    if src_n <= 0 or dst_n <= 0:
        return np.zeros(dst_n)
    if src_n == dst_n:
        out = np.array(src_h, dtype=float).copy()
        return project_feasible(out, dst_n)
    xp = (np.arange(src_n, dtype=float) + 0.5) * (2.0 / src_n)
    xq = (np.arange(dst_n, dtype=float) + 0.5) * (2.0 / dst_n)
    out = np.interp(xq, xp, src_h)
    return project_feasible(out, dst_n)

def max_step(h, d):
    pos = d > 1e-12
    neg = d < -1e-12
    tmax = float('inf')
    if np.any(pos):
        vals = h[pos] / d[pos]
        if vals.size:
            tmax = min(tmax, float(np.min(vals)))
    if np.any(neg):
        vals = (1.0 - h[neg]) / (-d[neg])
        if vals.size:
            tmax = min(tmax, float(np.min(vals)))
    if not np.isfinite(tmax):
        return 0.0
    return tmax

def feasible_direction(h, d):
    d = np.array(d, dtype=float).copy()
    eps = 1e-12
    for _ in range(4):
        bad = ((h <= eps) & (d > 1e-14)) | ((h >= 1.0 - eps) & (d < -1e-14))
        if np.any(bad):
            d[bad] = 0.0
        nonzero = np.abs(d) > 1e-14
        cnt = int(np.count_nonzero(nonzero))
        if cnt > 0:
            d[nonzero] -= float(np.sum(d)) / cnt
            bad = ((h <= eps) & (d > 1e-14)) | ((h >= 1.0 - eps) & (d < -1e-14))
            if np.any(bad):
                d[bad] = 0.0
        if abs(float(np.sum(d))) <= 1e-12 and not np.any(bad):
            break
    return d

def lag_gradients(h, n, p_indices):
    m = len(p_indices)
    G = np.zeros((n, m), dtype=float)
    for col, p in enumerate(p_indices):
        k = int(p - (n - 1))
        if k == 0:
            continue
        if k > 0:
            mlen = n - k
            if mlen > 0:
                G[k:k + mlen, col] += (1.0 - h[:mlen])
                G[:mlen, col] -= h[k:k + mlen]
        else:
            a = -k
            mlen = n - a
            if mlen > 0:
                G[:mlen, col] += (1.0 - h[a:a + mlen])
                G[a:a + mlen, col] -= h[:mlen]
    return G

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    denom = np.arange(1, n + 1, dtype=float)
    rho = int(np.sum(u > (css - 1.0) / denom))
    if rho <= 0:
        return np.ones(n, dtype=float) / n
    theta = float((css[rho - 1] - 1.0) / rho)
    return np.maximum(v - theta, 0.0)

def qp_weights(A):
    m = A.shape[1]
    if m <= 0:
        return None
    if m == 1:
        return np.ones(1, dtype=float)
    Q = A.T @ A
    tr = float(np.trace(Q))
    if tr < 1e-15:
        return np.ones(m, dtype=float) / m
    reg = 1e-9 * tr + 1e-12
    try:
        try:
            maxeig = float(np.linalg.norm(Q, ord=2))
        except Exception:
            maxeig = float(tr)
    except Exception:
        maxeig = float(tr)
    if maxeig < 1e-15:
        return np.ones(m, dtype=float) / m
    I = np.eye(m, dtype=float)
    w = np.ones(m, dtype=float) / m
    try:
        w0 = np.linalg.solve(Q + reg * I, np.ones(m, dtype=float))
        w0 = np.maximum(w0, 0.0)
        s0 = float(np.sum(w0))
        if s0 > 1e-14:
            w0 /= s0
            val0 = float(w0 @ (Q @ w0))
            valu = float(w @ (Q @ w))
            if val0 < valu:
                w = w0
        else:
            raise ValueError
    except Exception:
        w = np.ones(m, dtype=float) / m
    step = 1.0 / (2.0 * maxeig + 1e-12)
    best = w.copy()
    best_val = float(best @ (Q @ best))
    no_improve = 0
    for _ in range(60):
        grad = 2.0 * (Q @ w)
        wn = project_simplex(w - step * grad)
        vn = float(wn @ (Q @ wn))
        if vn < best_val - 1e-13:
            best = wn
            best_val = vn
            w = wn
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 16:
                break
    return best

def active_refine(h, n, phase_budget, deadline=0.0):

            best_h = h.copy()

            no_improve += 1

        order = np.argsort(corr)[::-1]

        if len(active) < 4:
            active = order[:min(4, len(order))]
        active = active[:24]
        G = lag_gradients(h, n, active)
        A = G - np.mean(G, axis=0, keepdims=True)
        w = qp_weights(A)
        if w is None:
            break
        d = feasible_direction(h, A @ w)
        if float(np.linalg.norm(d)) < 1e-12:
            top = order[:min(5, len(order))]
            G2 = lag_gradients(h, n, top)
            A2 = G2 - np.mean(G2, axis=0, keepdims=True)
            d = feasible_direction(h, A2.mean(axis=1))
        nd = float(np.linalg.norm(d))
        if nd < 1e-12:
            break
        d = d / nd
        new_h, new_c5 = line_search_continuous(h, d, n, cur)
        if new_h is not None and new_c5 < cur - 1e-15:
            h = new_h
        else:
            no_improve += 1
            if no_improve > 90:
                break
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def random_binary(n, rng):

def thue_binary(n, rng):
    x = np.zeros(n, dtype=float)
    for i in range(n):
        v = i
        bit = 0
        while v:
            bit ^= (v & 1)
            v >>= 1
        x[i] = float(bit)
    target = n // 2
    cnt = int(round(float(np.sum(x))))
    if cnt > target:
        idx = np.nonzero(x == 1.0)[0]
        need = cnt - target
        if idx.size >= need:
            proc = idx[rng.choice(idx.size, need, replace=False)]
            x[proc] = 0.0
    elif cnt < target:
        idx = np.nonzero(x == 0.0)[0]
        need = target - cnt
        if idx.size >= need:
            proc = idx[rng.choice(idx.size, need, replace=False)]
            x[proc] = 1.0
    return x

def get_edge_list(n):
    if n in _EDGE_CACHE:
        return _EDGE_CACHE[n]
    rows = []
    cols = []
    lags = []
    for k in range(-(n - 1), n):
        if k == 0:
            continue
        p = n - 1 + k
        if k > 0:
            mlen = n - k
            src = np.arange(0, mlen, dtype=np.int32)
            dst = (src + k).astype(np.int32)
        else:
            a = -k
            src = np.arange(a, n, dtype=np.int32)
            dst = (src + k).astype(np.int32)
        if src.size:
            rows.append(src)
            cols.append(dst)
            lags.append(np.full(src.size, p, dtype=np.int32))
    if len(rows) == 0:
        rows = np.zeros(0, dtype=np.int32)
        cols = np.zeros(0, dtype=np.int32)
        lags = np.zeros(0, dtype=np.int32)
    else:
        rows = np.concatenate(rows)
        cols = np.concatenate(cols)
        lags = np.concatenate(lags)
    _EDGE_CACHE[n] = (rows, cols, lags)
    return _EDGE_CACHE[n]

def mincut_reachable(res, s, N):
    vis = np.zeros(N, dtype=bool)
    vis[s] = True
    stack = [s]
    indptr = res.indptr
    indices = res.indices
    data = res.data
    while stack:
        u = stack.pop()
        for idx in range(indptr[u], indptr[u + 1]):
            if data[idx] > 1e-10:
                v = int(indices[idx])
                if not vis[v]:
                    vis[v] = True
                    stack.append(v)
    return vis

def mincut_one(n, weights, lam, eps=1e-7):
    rows_all, cols_all, lags_all = get_edge_list(n)
    N = n + 2
    s = 0
    t = N - 1
    var = np.arange(1, n + 1, dtype=np.int32)
    if rows_all.size:
        data = np.asarray(weights[i] for i in lags_all, dtype=float)
        mask = data > 1e-15
        rr = rows_all[mask]
        cc = cols_all[mask]
        dd = data[mask]
    else:
        rr = np.zeros(0, dtype=np.int32)
        cc = np.zeros(0, dtype=np.int32)
        dd = np.zeros(0, dtype=float)
    source_caps = np.full(n, (lam if lam > 0.0 else 0.0) + eps, dtype=float)
    sink_caps = np.full(n, (-lam if lam < 0.0 else 0.0) + eps, dtype=float)
    r0 = np.concatenate([rr, np.zeros(n, dtype=np.int32), var])
    c0 = np.concatenate([cc, var, np.full(n, t, dtype=np.int32)])
    cap = np.concatenate([dd, source_caps, sink_caps])
    if cap.size == 0:
        x = np.zeros(n, dtype=float)
        return x, 0
    try:
        G = csr_matrix((cap, (r0, c0)), shape=(N, N), dtype=float)
        res = G.copy()
        maximum_flow(G, s, t, residual=res)
        reach = mincut_reachable(res, s, N)
        x = (~reach[1:n + 1]).astype(float)
        return x, int(round(float(np.sum(x))))
    except Exception:
        return None, -1

def get_swap_tables(n):
    if n in _SWAP_CACHE:
        return _SWAP_CACHE[n]
    ps_t = [None] * n
    nb_t = [None] * n
    ps_s = [None] * n
    nb_s = [None] * n
    for a in range(n):
        pt = []
        nbt = []
        ps = []
        nbs = []
        for k in range(-(n - 1), 0):
            i = a - k
            if 0 <= i < n:
                pt.append(n - 1 + k)
                nbt.append(i)
            j = a + k
            if 0 <= j < n:
                ps.append(n - 1 + k)
                nbs.append(j)
        for k in range(1, n):
            i = a - k
            if 0 <= i < n:
                pt.append(n - 1 + k)
                nbt.append(i)
            j = a + k
            if 0 <= j < n:
                ps.append(n - 1 + k)
                nbs.append(j)
        ps_t[a] = np.asarray(pt, dtype=np.int32)
        nb_t[a] = np.asarray(nbt, dtype=np.int32)
        ps_s[a] = np.asarray(ps, dtype=np.int32)
        nb_s[a] = np.asarray(nbs, dtype=np.int32)
    _SWAP_CACHE[n] = (ps_t, nb_t, ps_s, nb_s)
    return _SWAP_CACHE[n]

def eval_flip(x, c, i, n, tables, temp_delta, temp_c):
    ps_t, nb_t, ps_s, nb_s = tables
    temp_delta.fill(0.0)
    da = 1.0 if x[i] < 0.5 else -1.0
    pt = ps_t[i]
    if pt.size:
        temp_delta[pt] += da * (1.0 - x[nb_t[i]])
    ps = ps_s[i]
    if ps.size:
        temp_delta[ps] += -da * x[nb_s[i]]
    temp_c[:] = c
    temp_c += temp_delta
    mc = float(np.max(temp_c))
    sq = float(np.dot(temp_c, temp_c))
    return mc, sq, temp_c

def eval_swap(x, c, a, b, n, tables, temp_delta, temp_c):
    olda = float(x[a])
    oldb = float(x[b])
    if abs(olda - oldb) < 1e-12:
        return float('inf'), float('inf'), temp_c
    da = oldb - olda
    db = olda - oldb
    ps_t, nb_t, ps_s, nb_s = tables
    temp_delta.fill(0.0)
    pt = ps_t[a]
    if pt.size:
        temp_delta[pt] += da * (1.0 - x[nb_t[a]])
    ps = ps_s[a]
    if ps.size:
        temp_delta[ps] += -da * x[nb_s[a]]
    pt = ps_t[b]
    if pt.size:
        temp_delta[pt] += db * (1.0 - x[nb_t[b]])
    ps = ps_s[b]
    if ps.size:
        temp_delta[ps] += -db * x[nb_s[b]]
    dist = b - a
    L = 2 * n - 1
    if dist != 0:
        p1 = n - 1 + dist
        if 0 <= p1 < L:
            temp_delta[p1] -= da * db
        p2 = n - 1 - dist
        if 0 <= p2 < L:
            temp_delta[p2] -= da * db
    temp_c[:] = c
    temp_c += temp_delta
    mc = float(np.max(temp_c))
    sq = float(np.dot(temp_c, temp_c))
    return mc, sq, temp_c

def correct_count_by_flips(x, target, n):
    if n % 2 != 0:
        return x
    x = np.array(x, dtype=float).copy()
    diff = int(target - round(float(np.sum(x))))
    if diff == 0:
        return x
    tables = get_swap_tables(n)
    L = 2 * n - 1
    temp_delta = np.zeros(L, dtype=float)
    temp_c = np.empty(L, dtype=float)
    c = compute_corr(x)
    while diff != 0 and (time.time() < _DEADLINE - 10):
        if diff > 0:
            cand_idx = np.nonzero(x < 0.5)[0]
            newv = 1.0
        else:
            cand_idx = np.nonzero(x > 0.5)[0]
            newv = 0.0
        if cand_idx.size == 0:
            break
        best_i = -1
        best_key = (float('inf'), float('inf'))
        best_c = None
        for i in cand_idx:
            mc, sq, cand = eval_flip(x, c, int(i), n, tables, temp_delta, temp_c)
            key = (mc, sq)
            if key < best_key:
                best_key = key
                best_i = int(i)
                best_c = cand.copy()
        if best_i < 0:
            break
        x[best_i] = newv
        if best_c is not None:
            c = best_c
        diff = int(target - round(float(np.sum(x))))
    return x

def mincut_for_target(n, weights, target, max_flows=18, eps=1e-7, deadline=0.0):
    if n % 2 != 0 or n <= 1:
        return None
    w = np.asarray(weights, dtype=float).copy()
    if w.size != (2 * n - 1):
        w = np.zeros(2 * n - 1, dtype=float)
    w[n - 1] = 0.0
    ssum = float(np.sum(w))
    if ssum < 1e-14:
        w = np.ones(2 * n - 1, dtype=float)
        w[n - 1] = 0.0
        ssum = w.sum()
        if ssum > 0:
            w /= ssum
    x0, cnt0 = mincut_one(n, w, 0.0, eps=eps)
    if x0 is None:
        return None
    best_x = x0.copy()
    best_cnt = cnt0
    best_diff = abs(cnt0 - target)
    if cnt0 == target:
        return x0
    if cnt0 > target:
        lo = 0.0
        hi = float(n) + 1.0
    else:
        lo = -float(n) - 1.0
        hi = 0.0
    for _ in range(max_flows - 1):
        if deadline > 0.0 and time.time() > deadline:
            break
        if (hi - lo) < 1e-10:
            break
        mid = 0.5 * (lo + hi)
        x, cnt = mincut_one(n, w, mid, eps=eps)
        if x is None:
            break
        diff = abs(cnt - target)
        if diff < best_diff:
            best_diff = diff
            best_cnt = cnt
            best_x = x.copy()
        if cnt == target:
            return x
        if cnt > target:
            lo = mid
        else:
            hi = mid
    x = best_x
    x = correct_count_by_flips(x, target, n)
    return x

def make_weights_from_corr(n, c, mode, param, rng):
    L = 2 * n - 1
    w = np.zeros(L, dtype=float)
    if mode == 'exp':
        if c is None or len(c) != L:
            w = np.ones(L, dtype=float)
        else:
            cv = np.asarray(c, dtype=float).copy()
            cm = float(np.max(cv))
            if cm <= 0.0:
                w = np.ones(L, dtype=float)
            else:
                sigma = float(max(1e-6, float(param)))
                z = (cv - cm) / sigma
                np.clip(z, -700.0, 0.0, out=z)
                w = np.exp(z)
    elif mode == 'top':
        if c is None or len(c) != L:
            w = np.ones(L, dtype=float)
        else:
            cv = np.asarray(c, dtype=float)
            m = int(min(L, max(1, param)))
            idx = np.argsort(cv)[-m:]
            w[idx] = 1.0
    elif mode == 'small':
        k = np.arange(-(n - 1), n, dtype=float)
        sc = float(max(1e-6, float(param)))
        w = 1.0 / (1.0 + np.abs(k) / sc)
    elif mode == 'large':
        k = np.arange(-(n - 1), n, dtype=float)
        w = np.abs(k)
    elif mode == 'rand':
        w = rng.random(L)
    else:
        w = np.ones(L, dtype=float)
    w[n - 1] = 0.0
    wx = float(np.max(w))
    if wx > 1e-15:
        cut = 1e-7 * wx
        w[w < cut] = 0.0
    ss = float(np.sum(w))
    if ss <= 1e-14:
        w = np.ones(L, dtype=float)
        w[n - 1] = 0.0
        ss = float(np.sum(w))
    if ss > 0:
        w /= ss
    return w

def binary_local_search(x, n, phase_budget, rng, deadline=0.0):
    if deadline <= 0.0:

    phase_budget = min(float(phase_budget), max(0.0, phase_time_left(deadline)))

    best_c5 = compute_c5(x)
    best_max = float(np.max(c))
    best_sq = float(np.dot(c, c))
    tables = get_swap_tables(n)
    L = 2 * n - 1
    temp_delta = np.zeros(L, dtype=float)
    temp_c = np.empty(L, dtype=float)
    start = time.time()
    accepts = 0
    while (time.time() - start) < phase_budget and phase_time_left(deadline) > 4.0:
        improved = False
        ones = np.nonzero(x > 0.5)[0]
        zeros = np.nonzero(x < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        zeros_seq = rng.permutation(zeros)
        if ones.size * zeros.size <= 25000:
            for a in ones:
                ia = int(a)
                for b in zeros_seq:
                    ib = int(b)
                    if ia == ib:
                        continue
                    mc, sq, cand = eval_swap(x, c, ia, ib, n, tables, temp_delta, temp_c)
                    if mc < best_max - 1e-9 or (abs(mc - best_max) <= 1e-9 and sq < best_sq - 1e-6):
                        ua = float(x[ia])
                        ub = float(x[ib])
                        x[ia] = ub
                        x[ib] = ua
                        c[:] = cand
                        best_max = mc
                        best_sq = sq
                        c5 = best_max * (2.0 / n)
                        if c5 < best_c5 - 1e-13:
                            best_c5 = c5
                            best_x = x.copy()
                        else:
                            best_x = x.copy()
                        accepts += 1
                        if accepts % 25 == 0:
                            c = compute_corr(x)
                            best_max = float(np.max(c))
                            best_sq = float(np.dot(c, c))
                            best_c5 = compute_c5(x)
                            best_x = x.copy()
                        improved = True
                        break
                if improved:
                    break
        else:
            num = 16000
            for _ in range(num):
                if time.time() > deadline - 5:
                    break
                ia = int(ones[rng.randint(ones.size)])
                ib = int(zeros[pick = rng.randint(zeros.size)])
                if ia == ib:
                    continue
                mc, sq, cand = eval_swap(x, c, ia, ib, n, tables, temp_delta, temp_c)
                if mc < best_max - 1e-9 or (abs(mc - best_max) <= 1e-9 and sq < best_sq - 1e-6):
                    ua = float(x[ia])
                    ub = float(x[ib])
                    x[ia] = ub
                    x[ib] = ua
                    c[:] = cand
                    best_max = mc
                    best_sq = sq
                    c5 = best_max * (2.0 / n)
                    if c5 < best_c5 - 1e-13:
                        best_c5 = c5
                        best_x = x.copy()
                    else:
                        best_x = x.copy()
                    accepts += 1
                    if accepts % 25 == 0:
                        c = compute_corr(x)
                        best_max = float(np.max(c))
                        best_sq = float(np.dot(c, c))
                        best_c5 = compute_c5(x)
                        best_x = x.copy()
                    improved = True
                    break
        if not improved:
            break
    best_x = project_feasible(best_x, n)
    best_c5 = compute_c5(best_x)
    return best_x, best_c5

def graph_cut_search(n, h_init, phase_budget, rng, deadline=0.0):
    if deadline <= 0.0:
        deadline = _DEADLINE - 5.0
    phase_budget = min(float(phase_budget), max(0.0, phase_time_left(deadline)))
    if n % 2 != 0 or phase_budget <= 0.0:
        return None, float('inf')
    best_x = None
    best_c5 = float('inf')
    c0 = compute_corr(h_init).copy()
    c0[n - 1] = 0.0
    profiles = [
        ('top', 12),
        ('exp', 0.5),
        ('top', 24),
        ('exp', 1.5),
        ('small', 5),
        ('exp', 3.0),
        ('top', 48),
        ('small', 25),
        ('large', 0),
        ('rand', 0),
    ]
    tstart = time.time()
    for ptype, pval in profiles:
        if time.time() > deadline - 15 or (time.time() - tstart) > phase_budget + 10:
            break
        profile_budget = min(70.0, max(12.0, phase_time_left(deadline)))
        pdeadline = min(deadline, time.time() + profile_budget)
        w = make_weights_from_corr(n, c0, ptype, pval, rng)
        for it in range(3):
            if time.time() > pdeadline - 3:
                break
            x = mincut_for_target(n, w, n // 2, max_flows=18, eps=1e-7, deadline=pdeadline)
            if x is None:
                break
            c = compute_corr(x)
            c5 = compute_c5(x)
            if c5 < best_c5:
                best_c5 = c5
                best_x = x.copy()
            if it < 2:
                cw = c.copy()
                cw[n - 1] = 0.0
                cmax = float(np.max(cw))
                sigma = max(0.2, min(2.0, 0.04 * cmax + 0.2))
                w_new = make_weights_from_corr(n, cw, 'exp', sigma, rng)
                mix = 0.7 if c5 < best_c5 + 1e-12 else 0.35
                w = (1.0 - mix) * w + mix * w_new
                ssum = float(np.sum(w))
                if ssum > 0:
                    w /= ssum
                else:
                    w = make_weights_from_corr(n, cw, 'exp', sigma, rng)
        if best_x is not None and phase_time_left(deadline) > 35:
            local_budget = min(45.0, max(8.0, phase_time_left(deadline)))
            lx, lc = binary_local_search(best_x, n, local_budget, rng, deadline=deadline)
            if lc < best_c5:
                best_c5 = lc
                best_x = lx
    return best_x, best_c5

def build_even_n_list(base_n):
    cand = []
    if base_n > 0 and base_n % 2 == 0:
        cand.append(int(base_n))
    for v in (128, 160, 168, 176, 180, 192, 208, 224, 240, 256, 288, 320):
        if v not in cand:
            cand.append(v)
    pairs = []
    for v in cand:
        if 96 <= v <= 320 and v % 2 == 0:
            pairs.append((abs(v - base_n) if base_n > 0 else abs(v - 160), v))
    pairs.sort()
    out = []
    for _, v in pairs:
        if v not in out:
            out.append(v)
    return out

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _DEADLINE
    _START_TIME = time.time()
    _DEADLINE = time.time() + float(budget_s)
    rng = np.random.RandomState(int(seed))

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        ph = np.asarray(initial_h_values, dtype=float)
        pn = len(ph)
        ph = project_feasible(ph, pn)
        pc5 = compute_c5(ph)
        best_h = ph
        best_c5 = pc5
        best_n = pn
    except Exception:
        best_n = 160
        h0 = thue_binary(best_n, rng)
        best_h = project_feasible(h0, best_n)
        best_c5 = compute_c5(best_h)

    if remaining_time() > 130:
        bh, bc = active_refine(best_h, best_n, min(90.0, remaining_time() - 70.0))
        if bc < best_c5:
            best_c5 = bc
            best_h = bh
            best_n = len(bh)

    n_list = build_even_n_list(best_n)

    idx = 0
    while idx < len(n_list) and remaining_time() > 110:
        n = n_list[idx]
        left_n = len(n_list) - idx
        slice_s = min(260.0, max(50.0, (remaining_time() - 90.0) / float(left_n)))
        if remaining_time() < slice_s + 20:
            slice_s = max(30.0, remaining_time() - 40.0)
        if slice_s <= 0:
            break
        hd = best_h if best_n == n else interp_to(best_h, best_n, n)
        hd = project_feasible(hd, n)
        c5d = compute_c5(hd)
        if c5d < best_c5:
            best_c5 = c5d
            best_h = hd
            best_n = n
        if phase_time_left(_DEADLINE) > 35:
            refine_budget = min(35.0, max(8.0, slice_s * 0.28))
            bh, bc = active_refine(hd, n, refine_budget, deadline=_DEADLINE - 40.0)
            if bc < best_c5:
                best_c5 = bc
                best_h = bh
                best_n = n
        if phase_time_left(_DEADLINE) > 50:
            gc_budget = min(120.0, max(15.0, slice_s * 0.55))
            gx, gc = graph_cut_search(n, best_h, gc_budget, rng, deadline=_DEADLINE - 35.0)
            if gx is not None and gc < best_c5:
                best_c5 = gc
                best_h = gx
                best_n = n
            if gx is not None and phase_time_left(_DEADLINE) > 30:
                bh2, bc2 = active_refine(gx, n, min(35.0, phase_time_left(_DEADLINE) - 25.0), deadline=_DEADLINE - 30.0)
                if bc2 < best_c5:
                    best_c5 = bc2
                    best_h = bh2
                    best_n = n
        idx += 1

    if remaining_time() > 45:
        final_budget = min(360.0, remaining_time() - 25.0)
        bh, bc = active_refine(best_h, best_n, final_budget * 0.55, deadline=_DEADLINE - 20.0)
        if bc < best_c5:
            best_c5 = bc
            best_h = bh
            best_n = len(bh)

    if best_n % 2 == 0 and remaining_time() > 90:
        gx, gc = graph_cut_search(best_n, best_h, remaining_time() - 50.0, rng, deadline=_DEADLINE - 20.0)
        if gx is not None and gc < best_c5:
            best_c5 = gc
            best_h = gx
            best_n = len(gx)
        if gx is not None and remaining_time() > 35:
            bh3, bc3 = active_refine(gx, best_n, min(60.0, remaining_time() - 25.0), deadline=_DEADLINE - 15.0)
            if bc3 < best_c5:
                best_c5 = bc3
                best_h = bh3
                best_n = len(bh3)

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    try:
        ek = evaluate_erdos_solution(best_h)
        if ek < final_c5:
            final_c5 = float(ek)
    except Exception:
        pass
    return (best_h, float(final_c5), int(best_n))
