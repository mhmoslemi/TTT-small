def compute_c5(h):

            if mask.sum() == 0:
                break

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final check
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute over all elements
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        # Re-project
        for _ in range(50):
            d2 = target - np.sum(h)
            if abs(d2) < 1e-13:
                break
            if d2 > 0:
                m = h < 1.0 - 1e-12
                h[m] += d2 / np.sum(m)
            else:
                m = h > 1e-12
                h[m] += d2 / np.sum(m)
            h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, n_sweeps=50, time_limit=None, t0=None):

    best_h = h.copy()
    best_c5 = compute_c5(h)

    L0_idx = n - 1
    n_lags = 2 * n - 1
    
    for sw in range(n_sweeps):
        if time_limit and t0 and (time.time() - t0) > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        for j in range(n):
            if time_limit and t0 and (time.time() - t0) > time_limit:
                break
            
            old = h[j]
            
            # Coefficients for linear terms (L ≠ 0)
            a_coeff = np.zeros(n_lags)
            
            # Region A: L in [j-(n-1), j]
            A_lo = max(0, j - (n-1) - (-(n-1)))  # index into Ls
            A_hi = min(n_lags-1, j - (-(n-1)))
            # Ls[A_lo:A_hi+1] gives the lags in region A
            # Let me just use boolean masks
            A_valid = (Ls >= j - (n-1)) & (Ls <= j)
            B_valid = (Ls >= -j) & (Ls <= n - 1 - j)
            
            if np.any(A_valid):
                idx_A = j - Ls[A_valid].astype(int)
                a_coeff[A_valid] += (1.0 - h[np.clip(idx_A, 0, n-1)])
            if np.any(B_valid):
                idx_B = j + Ls[B_valid].astype(int)
                a_coeff[B_valid] += (-h[np.clip(idx_B, 0, n-1)])
            
            a_coeff[L0_idx] = 0.0
            
            # Ternary search for optimal x
            lo, hi = 0.0, 1.0
            for _ in range(45):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0

                v1 = np.max(corr + d1 * a_coeff)
                q1 = corr[L0_idx] + m1*(1.0 - m1) - old*(1.0 - old)
                if q1 > v1:
                    v1 = q1

                    v2 = q2
                
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            
            h[j] = (lo + hi) / 2.0
        
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
