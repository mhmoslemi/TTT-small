    out = np.zeros(n)
    out[h_idx[:n//2]] = 1.0
    return out

def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off+npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off+npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao+npairs])
        g[ao:ao+npairs] -= x[:npairs]
    return g

def sa_binary(h_bin, n, time_limit, rng, t0=5.0, t1=0.01):
    """Fast SA on binary sequence."""
    h = h_bin.copy()
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_obj = cur_max
    
    # Calibrate iteration speed
    t_st = time.time()
    for _ in range(50):
        _ = compute_corr(h)
    t_corr = (time.time() - t_st) / 50.0
    n_iter = max(1000, int((time_limit * 0.9) / t_corr))
    
    cooling = (t1 / t0) ** (1.0 / max(1, n_iter - 1))
    
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    
    start_time = time.time()
    for it in range(n_iter):
        if time.time() - start_time > time_limit * 0.9:
            break
        if remaining() < 3:
            break
        
        T = t0 * (cooling ** it)
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        
        delta = new_max - cur_max
        if delta <= 0 or rng.rand() < np.exp(-delta / max(T, 1e-15)):
            cur_max = new_max
            corr = new_corr
            # Update ones/zeros efficiently
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if cur_max < best_obj:
                best_obj = cur_max
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_obj

def targeted_continuous(h, n, time_limit, rng):
    """Targeted continuous optimization focusing on top shifts."""
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx_val = 2.0 / n
    
    start_time = time.time()
    
    # Phase 1: Projected gradient on soft-max
    for beta in [100, 500, 2000, 10000]:
        if time.time() - start_time > time_limit * 0.6 or remaining() < 5:
            break
        
        n_steps = 300
        step_size = 0.05 / np.sqrt(n)
        
        for step in range(n_steps):
            if time.time() - start_time > time_limit * 0.6:
                break
            corr = compute_corr(h)
            m = float(np.max(corr))
            shifted = corr - m
            cl = np.clip(beta * shifted, -40, 40)
            ex = np.exp(cl)
            w = ex / np.sum(ex)
            
            g = np.zeros(n)
            for k in range(len(corr)):
                wk = w[k]
                if wk < 1e-10:
                    continue
                g += wk * corr_grad_per_shift(h, n, k)
            
            # Multiple step sizes
            improved = False
            for ss in [step_size, step_size*0.5, step_size*0.1, step_size*2]:
                h_new = h - ss * g
                h_new = project_feasible(h_new, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    h = h_new
                    best_c5 = c5_new
                    best_h = h.copy()
                    improved = True
                    break
            
            if not improved:
                break
    
    # Phase 2: Active set equalization
    for _ in range(200):
        if time.time() - start_time > time_limit or remaining() < 3:
            break
        
        corr = compute_corr(h)
        top3 = np.argsort(-corr)[:3]
        if len(top3) < 2:
            break
        c_max = corr[top3[0]]
        c_2nd = corr[top3[1]]
        
        if c_max - c_2nd < 1e-8:
            # Already equalized at top
            # Try to reduce the abs max
            g = -corr_grad_per_shift(h, n, int(top3[0]))
            for ss in [0.02, 0.01, 0.005, 0.05]:
                h_new = h - ss * g
                h_new = project_feasible(h_new, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    h = h_new
                    best_c5 = c5_new
                    best_h = h.copy()
                    break
        else:
            # Equalize: reduce max, increase second
            g1 = corr_grad_per_shift(h, n, int(top3[0]))
            g2 = corr_grad_per_shift(h, n, int(top3[1]))
            d = g2 - g1
            dn = np.linalg.norm(d)
            if dn < 1e-14:
                break
            d /= dn
            
            for ss in [0.01, 0.02, 0.05, 0.005, 0.08]:
                h_new = h + ss * d
                h_new = project_feasible(h_new, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    h = h_new
                    best_c5 = c5_new
                    best_h = h.copy()
                    break
    
    return best_h, best_c5

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = i & 1
        v = i
        while v > 1:
            v >>= 1
            c ^= (v & 1)
        h[i] = 1.0 if c == 1 else 0.0
    return h
