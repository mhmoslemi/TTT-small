def lag_gradient(h, n, offset):
    """Gradient of corr at lag offset with respect to h."""
    grad = np.zeros(n)
    if offset >= 0:
        n_pairs = n - offset
        if n_pairs <= 0:
            return grad
        # corr = sum_{i=0}^{n_pairs-1} h[i+offset] * (1-h[i])

            candidates = [old_j, 0.0, 1.0, max(0, old_j - 0.1), min(1, old_j + 0.1)]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            step = diff / cnt
            if diff > 0:
                max_step = np.min(1.0 - h[mask])
                step = min(step, max_step - 1e-15)
            else:
                max_step = np.min(h[mask])
                step = max(step, -(max_step - 1e-15))
            if abs(step) < 1e-18:
                break
            h[mask] += step
        else:
            h[:] += diff / n
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-13:
            for j in range(n):
                if 0.0 < h[j] < 1.0:
                    h[j] = np.clip(h[j] + diff2, 0.0, 1.0)
                    break
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def compute_deltas_for_j(h, n, j):
    """Compute d(corr[k])/d(h[j]) for all k."""
    n_lags = 2 * n - 1
    deltas = np.zeros(n_lags)
    old_val = h[j]
    # d=0: index n-1
    deltas[n - 1] = 1.0 - 2.0 * old_val
    # d > 0
    for d in range(1, n):
        idx = n - 1 + d
        dv = 0.0
        if d <= j <= n - 1:
            dv += (1.0 - h[j - d])
        if 0 <= j <= n - 1 - d:
            dv -= h[j + d]
        deltas[idx] = dv
    # d < 0 (a = -d > 0)
    for a in range(1, n):
        idx = n - 1 - a
        dv = 0.0
        if 0 <= j <= n - 1 - a:
            dv += (1.0 - h[j + a])
        if a <= j <= n - 1:
            dv -= h[j - a]
        deltas[idx] = dv
    return deltas

def coord_descent_optimize(h_init, n, time_limit, n_grid=40, n_grid_refine=40):
    """Cyclic coordinate descent with exact line search on the max objective."""
    h = h_init.copy()
    h = project_feasible(h, n)
    corr = compute_all_corr(h)
    dx = 2.0 / n
    c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    best_c5 = c5
    start_time = time.time()
    n_sweeps = 0
    no_improve_count = 0
    
    while time.time() - start_time < time_limit:
        sweep_start = time.time()
        improved = False
        for j in range(n):
            old_val = h[j]
            if old_val < 1e-15 or old_val > 1.0 - 1e-15:
                # Still try to move it
                deltas = compute_deltas_for_j(h, n, j)
                # Check if moving helps
                s_lo = -old_val
                s_hi = 1.0 - old_val
                f_lo = float(np.max(corr + deltas * s_lo))
                f_hi = float(np.max(corr + deltas * s_hi))
                f_cur = float(np.max(corr))
                if f_lo < f_cur - 1e-10 or f_hi < f_cur - 1e-10:
                    best_s = s_lo if f_lo < f_hi else s_hi
                    new_val = float(np.clip(old_val + best_s, 0.0, 1.0))
                    h[j] = new_val
                    actual_d = new_val - old_val
                    if abs(actual_d) > 1e-15:
                        corr += deltas * actual_d
                    improved = True
                continue
            
            deltas = compute_deltas_for_j(h, n, j)
            s_lo = -old_val
            s_hi = 1.0 - old_val
            
            # Grid search
            best_s = 0.0
            best_f = float(np.max(corr))
            s_vals = np.linspace(s_lo, s_hi, n_grid)
            for s_val in s_vals:
                f_val = float(np.max(corr + deltas * s_val))
                if f_val < best_f:
                    best_f = f_val
                    best_s = s_val
            
            # Refine
            if best_s > s_lo + 1e-8 and best_s < s_hi - 1e-8:
                half_range = (s_hi - s_lo) / (2 * n_grid)
                s_vals2 = np.linspace(best_s - half_range, best_s + half_range, n_grid_refine)
                for s_val in s_vals2:
                    if s_val < s_lo or s_val > s_hi:
                        continue
                    f_val = float(np.max(corr + deltas * s_val))
                    if f_val < best_f:
                        best_f = f_val
                        best_s = s_val
            
            if abs(best_s) > 1e-12:
                new_val = float(np.clip(old_val + best_s, 0.0, 1.0))
                h[j] = new_val
                actual_d = new_val - old_val
                if abs(actual_d) > 1e-15:
                    corr += deltas * actual_d
                improved = True
        
        n_sweeps += 1
        # Project periodically
        if n_sweeps % 10 == 0:
            h = project_feasible(h, n)
            corr = compute_all_corr(h)
        
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        if no_improve_count > 50 and time.time() - start_time > time_limit * 0.3:
            # Try a perturbation to escape local optimum
            if time.time() - start_time < time_limit * 0.8:
                rng_p = np.random.RandomState(int(time.time() * 1000) % 100000)
                h_pert = h.copy()
                perturb_idx = rng_p.choice(n, size=n // 8, replace=False)
                h_pert[perturb_idx] += rng_p.uniform(-0.1, 0.1, size=n // 8)
                h_pert = project_feasible(h_pert, n)
                corr_p = compute_all_corr(h_pert)
                c5_p = float(np.max(corr_p) * dx)
                if c5_p < c5 + 0.005:
                    h = h_pert
                    corr = corr_p
                    c5 = c5_p
                no_improve_count = 0
            else:
                break
    
    return best_h, best_c5

def slsqp_refine(h, n, beta, maxiter):
    n_lags = 2 * n - 1
    dx = 2.0 / n
    
    def smooth_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        exp_shifted = np.exp(shifted)
        sum_exp = np.sum(exp_shifted)
        weights = (beta * (corr - m) * exp_shifted / sum_exp - 1.0) * dx / (m + np.log(sum_exp) / beta)
        # This is getting complicated, let's use numerical
        return None
    
    def constraint_sum(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    try:
        res = minimize(smooth_obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_pass(h, n, max_iters, top_k=5):
    """Subgradient descent focusing on top lags."""
    h = h.copy()
    corr = compute_all_corr(h)
    dx = 2.0 / n
    best_c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    
    for it in range(max_iters):
        max_corr = float(np.max(corr))
        # Get top lags
        sorted_idx = np.argsort(corr)[::-1][:top_k]
        # Weight by how close to max
        weights = np.zeros(len(sorted_idx))
        for i, idx in enumerate(sorted_idx):
            weights[i] = max((corr[idx] - max_corr + 1.0), 0.05)
        weights /= np.sum(weights)
        
        # Compute combined gradient
        grad = np.zeros(n)
        for w, idx in zip(weights, sorted_idx):
            offset = idx - (n - 1)
            if offset >= 0:
                np_pairs = n - offset
                if np_pairs > 0:
                    grad[offset:offset+np_pairs] += w * (1.0 - h[:np_pairs])
                    grad[:np_pairs] -= w * h[offset:offset+np_pairs]
            else:
                a = -offset
                np_pairs = n - a
                if np_pairs > 0:
                    grad[:np_pairs] += w * (1.0 - h[a:a+np_pairs])
                    grad[a:a+np_pairs] -= w * h[:np_pairs]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        
        # Backtracking line search
        base_step = 0.05 / (1.0 + it * 0.01)
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = base_step * mult
            h_trial = h - step * grad
            h_trial = np.clip(h_trial, 0.0, 1.0)
            h_trial = project_feasible(h_trial, n)
            corr_trial = compute_all_corr(h_trial)
            c5_trial = float(np.max(corr_trial) * dx)
            if c5_trial < best_c5 - 1e-15:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial.copy()
                corr = corr_trial
                improved = True
                break
    
    return best_h, best_c5

def local_search_pairs(h, n, max_sweeps=50, rng=None):
    """Try paired coordinate changes."""
    if rng is None:
        rng = np.random.RandomState(123)
    h = h.copy()
    corr = compute_all_corr(h)
    dx = 2.0 / n
    best_c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        improved = False
        # Try random pairs
        for trial in range(200):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            old_i, old_j = h[i], h[j]
            # Try: increase i, decrease j (or vice versa) to maintain sum
            for di, dj in [(0.01, -0.01), (-0.01, 0.01), (0.05, -0.05), (-0.05, 0.05)]:
                new_i = np.clip(old_i + di, 0.0, 1.0)
                new_j = np.clip(old_j + dj, 0.0, 1.0)
                h[i], h[j] = new_i, new_j
                h = project_feasible(h, n)
                c5_test = compute_c5(h)
                if c5_test < best_c5 - 1e-14:
                    best_c5 = c5_test
                    best_h = h.copy()
                    corr = compute_all_corr(h)
                    improved = True
                    break
                else:
                    h[i], h[j] = old_i, old_j
                    h = project_feasible(h, n)
            if improved:
                break
        if not improved:
            # Try single coordinate
            for trial in range(100):
                i = rng.randint(n)
                old_i = h[i]
                for d in [0.01, -0.01, 0.03, -0.03]:
                    h[i] = np.clip(old_i + d, 0.0, 1.0)
                    h = project_feasible(h, n)
                    c5_test = compute_c5(h)
                    if c5_test < best_c5 - 1e-14:
                        best_c5 = c5_test
                        best_h = h.copy()
                        corr = compute_all_corr(h)
                        improved = True
                        break
                    else:
                        h[i] = old_i
                        h = project_feasible(h, n)
                if improved:
                    break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return max(0.0, budget_s - (time.time() - start_time) - 5.0)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 160
        h_init = np.zeros(best_n)
        for i in range(best_n):
            v = i
            cnt = 0
            while v:
                cnt ^= (v & 1)
                v >>= 1
            h_init[i] = 1.0 if cnt == 1 else 0.0
        best_h = project_feasible(h_init, best_n)
        best_c5 = compute_c5(best_h)
    
    # Phase 1: Coordinate descent at parent resolution
    if time_left() > 30:
        t1 = min(time_left() - 20, 60.0)
        h_new, c5_new = coord_descent_optimize(best_h, best_n, t1, n_grid=50)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            best_n = best_n
    
    # Phase 2: Try n=320 (interpolated + fresh)
    if time_left() > 120:
        n_320 = 320
        h_320 = interp_to(best_h, best_n, n_320)
        h_320 = project_feasible(h_320, n_320)
        c5_320 = compute_c5(h_320)
        
        if time_left() > 80:
            t2 = min(time_left() - 60, 250.0)
            h_new, c5_new = coord_descent_optimize(h_320, n_320, t2, n_grid=50)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                best_n = n_320
    
    # Phase 3: Subgradient pass on best resolution
    if time_left() > 40:
        t3 = min(time_left() - 20, 80.0)
        max_iters = max(200, int(t3 * 5))
        h_new, c5_new = subgradient_pass(best_h, best_n, max_iters, top_k=5)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 4: Coordinate descent again (slightly different starting point)
    if time_left() > 60:
        # Add small perturbation
        h_pert = best_h.copy()
        n_p = best_n
        perturb_mask = rng.rand(n_p) < 0.3
        h_pert[perturb_mask] += rng.normal(0, 0.05, size=np.sum(perturb_mask))
        h_pert = project_feasible(h_pert, n_p)
        c5_pert = compute_c5(h_pert)
        
        base_h = best_h.copy()
        base_c5 = best_c5
        if c5_pert < best_c5 + 0.002:
            t4 = min(time_left() - 20, 150.0)
            h_new, c5_new = coord_descent_optimize(h_pert, n_p, t4, n_grid=60)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
    
    # Phase 5: SLSQP polish
    if time_left() > 30:
        for beta in [500, 2000]:
            if time_left() < 15:
                break
            h_new = slsqp_refine(best_h, best_n, beta, 200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
    
    # Phase 6: Local pair search
    if time_left() > 20:
        h_new, c5_new = local_search_pairs(best_h, best_n, max_sweeps=20, rng=rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 7: Try n=480 if time permits
    if time_left() > 100:
        n_480 = 480
        h_480 = interp_to(best_h, best_n, n_480)
        h_480 = project_feasible(h_480, n_480)
        c5_480 = compute_c5(h_480)
        if c5_480 < best_c5 + 0.003:
            t7 = min(time_left() - 30, 180.0)
            h_new, c5_new = coord_descent_optimize(h_480, n_480, t7, n_grid=50)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                best_n = n_480
    
    # Phase 8: Final intensive coordinate descent
    if time_left() > 30:
        t8 = min(time_left() - 10, 100.0)
        h_new, c5_new = coord_descent_optimize(best_h, best_n, t8, n_grid=60, n_grid_refine=60)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 9: Final SLSQP polish
    if time_left() > 10:
        h_new = slsqp_refine(best_h, best_n, 3000, 100)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
