def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr


def compute_grad_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    w = exps / np.sum(exps)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-12:
            continue
        tau = k - (n - 1)
        wk = w[k]
        if tau >= 0:
            if tau < n:
                j1 = np.arange(tau, n)
                grad[j1] += wk * (1.0 - h[j1 - tau])
            j2 = np.arange(0, n - tau)
            if len(j2) > 0:
                grad[j2] -= wk * h[j2 + tau]
        else:
            a = -tau
            j1 = np.arange(0, n - a)
            if len(j1) > 0:
                grad[j1] += wk * (1.0 - h[j1 + a])
            j2 = np.arange(a, n)
            grad[j2] -= wk * h[j2 - a]
    return dx * grad


def smooth_max_obj(h, beta):
    n = len(h)

def project_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def projected_gradient_step(h, grad, alpha, n):
    h_new = h - alpha * grad
    h_proj = h - alpha * (grad - np.mean(grad))
    h_proj = np.clip(h_proj, 0.0, 1.0)

        h_proj[mask] += diff / np.sum(mask)
    else:
        h_proj[:] += diff / n
    h_proj = np.clip(h_proj, 0.0, 1.0)

                        best_h = h_tr.copy()
                        h = h_tr.copy()
                        done = True
                if not done and h[j] >= d and h[i] <= 1.0 - d:
                    h_tr = h.copy()
                    h_tr[j] -= d
                    h_tr[i] += d
                    h_tr = project_h(h_tr, n)
                    c5_tr, _ = compute_c5(h_tr)
                    if c5_tr < best_c5:
                        best_c5 = c5_tr
                        best_h = h_tr.copy()
                        h = h_tr.copy()
                        done = True
                if done:
                    break

    # Phase 8: Multi-resolution final check
    if time_left(t0, budget) > 20:
        for dn in [-30, -20, -10, 10, 20, 30, 40, 50, -40, -50]:
            if time_left(t0, budget) < 5:
                break
            nt = best_n + dn
            if nt < 100 or nt > 400:
                continue
            ht = interp_h(best_h, best_n, nt)
            ht = project_h(ht, nt)
            ct, _ = compute_c5(ht)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()
                best_n = nt
                print("  Final Res n={}: C5={:.8f}".format(nt, ct))

    # Phase 9: Final high-precision SLSQP
    if time_left(t0, budget) > 15:
        n = best_n
        _g_n = n
        bounds = [(0.0, 1.0)] * n
        cons_slsqp = [{'type': 'eq', 'fun': con_sum}]
        h = best_h.copy()
        for beta_f in [500, 2000, 5000, 10000]:
            if time_left(t0, budget) < 6:
                break
            _g_beta = beta_f
            _g_active = np.arange(2 * n - 1)
            _g_lam = 10000.0
            try:
                res = minimize(ob_active_pen, h, method='SLSQP',
                               bounds=bounds, constraints=cons_slsqp,
                               options={'maxiter': 300, 'ftol': 1e-20, 'iprint': 0})
                h_new = project_h(res.x, n)
                c5_new, _ = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
            except Exception:
                pass

    # Final cleanup
    best_h = project_h(best_h, best_n)
    final_c5, _ = compute_c5(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
