                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                    best_n = n_target
        
        cur_h = best_h
        cur_n = best_n
        cur_c5 = best_c5
    
    # Phase 3: Gradient descent on best
    if time_left() > 40:
        h_gd, c5_gd = topk_based_gd(best_h, best_n, t0, t0 + min(60, time_left() * 0.3))
        if c5_gd < best_c5:
            best_c5 = c5_gd
            best_h = h_gd.copy()
    
    # Phase 4: SLSQP smooth max polish
    if time_left() > 60:
        h_sp, c5_sp = slsqp_smooth(best_h, best_n, t0, t0 + min(120, time_left() * 0.5))
        if c5_sp < best_c5:
            best_c5 = c5_sp
            best_h = h_sp.copy()
    
    # Phase 5: More CD at best and slightly higher resolution
    if time_left() > 40:
        h_cd2, c5_cd2 = coord_descent(best_h, best_n, 15, t0, t0 + time_left() * 0.4)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
    
    # Try one resolution higher
    if time_left() > 60:
        n_hi = best_n + (32 if best_n % 32 == 0 else 64)
        if n_hi <= 512:
            h_hi = upsample_h(best_h, best_n, n_hi)
            h_cd3, c5_cd3 = coord_descent(h_hi, n_hi, 10, t0, t0 + time_left() * 0.5)
            if c5_cd3 < best_c5:
                best_c5 = c5_cd3
                best_h = h_cd3.copy()
                best_n = n_hi
    
    # Phase 6: Final SLSQP
    if time_left() > 30:
        h_sp2, c5_sp2 = slsqp_smooth(best_h, best_n, t0, t0 + time_left() * 0.5)
        if c5_sp2 < best_c5:
            best_c5 = c5_sp2
            best_h = h_sp2.copy()
    
    # Phase 7: Final CD polish
    if time_left() > 20:
        h_fin, c5_fin = coord_descent(best_h, best_n, 10, t0, t0 + time_left() * 0.7)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    # Validate with evaluator
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 is not None and ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
