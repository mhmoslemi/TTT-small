import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 2.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        unsat = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(unsat) > 0:

        idx = np.argmin(np.abs(h - 0.5))

        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_gradient(h, n, beta, top_k=None):
    """Vectorized gradient of smooth max of correlations."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    if top_k is not None and top_k < len(corr):
        # Only use top-k correlations
        top_idx = np.argsort(corr)[-top_k:]
        used_corr = np.zeros_like(corr)
        used_corr[top_idx] = corr[top_idx]
    else:
        used_corr = corr
    
    m = np.max(used_corr)
    # Avoid overflow
    shifted = beta * (used_corr - m)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    if wsum > 0:
        weights /= wsum
    else:
        weights = np.ones_like(corr) / len(corr)
    
    # Compute gradient: sum_j weights[j] * d(corr_j)/d(h) * dx
    grad = np.zeros(n)
    
    for j in range(2 * n - 1):
        wj = weights[j]
        if wj < 1e-12:
            continue
        s = j - (n - 1)
        if s >= 0:
            # corr = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
            # d/dh[m]: +(1-h[m-s]) for m in [s, n-1], -(h[m+s]) for m in [0, n-1-s]
            grad[s:] += wj * (1.0 - h[:n-s])
            grad[:n-s] -= wj * h[s:]
        else:
            p = -s
            # corr = sum_{i=p}^{n-1} h[i-p]*(1-h[i])
            # d/dh[m]: +(1-h[m+p]) for m in [0, n-1-p], -(h[m-p]) for m in [p, n-1]
            grad[:n-p] += wj * (1.0 - h[p:])
            grad[p:] -= wj * h[:n-p]
    
    grad *= dx
    return grad, corr

def smooth_max_value(h, n, beta, top_k=None):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(corr)[-top_k:]
        used = corr[top_idx]
    else:
        used = corr
    m = np.max(used)
    shifted = beta * (used - m)
    sm = m + np.log(np.sum(np.exp(shifted))) / beta
    return sm * dx

def lbfgs_step(h, n, beta, maxiter=300, top_k=None):
    """One L-BFGS-B optimization step."""
    global _N, _TGT
    
    def obj(x):
        val = smooth_max_value(x, n, beta, top_k)
        pen = 500.0 * (np.sum(x) - n/2.0) ** 2
        return val + pen
    
    def grad(x):
        g, _ = smooth_max_gradient(x, n, beta, top_k)
        g += 2.0 * 500.0 * (np.sum(x) - n/2.0)
        return g
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8, 'iprint': 0, 'maxls': 30})
    return res.x

def topk_gradient_step(h, n, top_k=8, steps=[0.02, 0.04, 0.06, 0.01, 0.08, 0.03, 0.1, 0.15]):
    """Direct gradient step on top-k correlations."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    
    # Compute gradient targeting only top-k
    grad = np.zeros(n)
    for j in top_idx:
        s = j - (n - 1)
        if s >= 0:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] -= h[s:]
        else:
            p = -s
            grad[:n-p] += (1.0 - h[p:])
            grad[p:] -= h[:n-p]
    
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-14:
        return h, False
    
    grad /= (gnorm * dx)  # normalize to unit step in C5 units
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    improved = False
    for step in steps:
        h_trial = h - step * grad
        h_trial = project_feasible(h_trial, n)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            improved = True
    return best_h, improved

def binary_local_search(h, n, iterations, rng, t_start=0.003, t_end=0.00002, time_limit=60):
    """Binary local search with simulated annealing."""
    h = np.where(h > 0.5, 1.0, 0.0)
    num_ones = int(np.sum(h))
    target_ones = n // 2
    
    if num_ones != target_ones:
        if num_ones > target_ones:
            ones_idx = np.where(h == 1.0)[0]
            remove = rng.choice(len(ones_idx), num_ones - target_ones, replace=False)
            h[ones_idx[remove]] = 0.0
        else:
            zeros_idx = np.where(h == 0.0)[0]
            add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
            h[zeros_idx[add]] = 1.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])

    for it in range(iterations):
        if remaining_time() < 3 or (time.time() - start_t) > time_limit:
            break

        j = rng.randint(len(zeros_list))
        oi = ones_list[i]
        oz = zeros_list[j]

        h[oz] = 1.0

    target_ones = n // 2
    if num_ones > target_ones:
        ones_idx = np.where(h == 1.0)[0]
        remove = rng.choice(len(ones_idx), num_ones - target_ones, replace=False)
        h[ones_idx[remove]] = 0.0
    elif num_ones < target_ones:
        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
        h[zeros_idx[add]] = 1.0

    dx = 2.0 / n
    best_c5 = compute_c5(h)
    current_c5 = best_c5

    for sweep in range(max_sweeps):
        if remaining_time() < 2:
            break

        # Periodic kick to escape local minima
        if kick_interval > 0 and sweep > 0 and sweep % kick_interval == 0:
            for _ in range(kick_size):
                i = rng.randint(n)
                h[i] = 1.0 - h[i]
            h = binary_to_feasible(h, n)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_c5_sweep = current_c5
        best_i = -1
        best_j = -1

        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
                if c < best_c5_sweep:
                    best_c5_sweep = c
                    best_i = int(i)
                    best_j = int(j)
                h[i] = 1.0
                h[j] = 0.0

        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            current_c5 = best_c5_sweep
            if current_c5 < best_c5:
                best_c5 = current_c5
        else:
            break

    return h.copy(), best_c5

def smooth_max_grad_vectorized(h, n, beta, top_k=None):
    """Compute gradient of smooth max of correlations (vectorized per shift)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(corr)[-top_k:]
        mask = np.zeros(len(corr), dtype=bool)
        mask[top_idx] = True
        used = np.where(mask, corr, -1e10)
    else:
        used = corr
        mask = np.ones(len(corr), dtype=bool)

    m = np.max(used)
    shifted = beta * (used - m)
    weights = np.exp(shifted)
    weights[~mask] = 0
    wsum = np.sum(weights)
    if wsum < 1e-30:
        return np.zeros(n)
    weights /= wsum

    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = weights[j]
        if wj < 1e-12:
            continue
        s = j - (n - 1)
        if s >= 0:
            grad[s:] += wj * (1.0 - h[:n - s])
            grad[:n - s] -= wj * h[s:]
        else:
            p = -s
            grad[:n - p] += wj * (1.0 - h[p:])
            grad[p:] -= wj * h[:n - p]

    grad *= dx
    return grad

def smooth_max_value(h, n, beta, top_k=None):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(corr)[-top_k:]
        used = corr[top_idx]
    else:
        used = corr
    m = np.max(used)
    shifted = beta * (used - m)
    sm = m + np.log(np.sum(np.exp(shifted))) / beta
    return sm * dx

def lbfgs_continuous(h, n, beta, maxiter=400, top_k=None):
    """L-BFGS-B continuous optimization with smooth max objective."""
    target = n / 2.0

    def obj(x):
        val = smooth_max_value(x, n, beta, top_k)
        pen = 1000.0 * (np.sum(x) - target) ** 2
        return val + pen

    def grad(x):
        g = smooth_max_grad_vectorized(x, n, beta, top_k)
        g += 2.0 * 1000.0 * (np.sum(x) - target)
        return g

    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                            'iprint': 0, 'maxls': 30, 'maxfun': maxiter * 3})
    return res.x

def topk_gradient_descent(h, n, max_iters=300, top_k=10):
    """Direct gradient steps targeting top-k worst correlations."""
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for it in range(max_iters):
        if remaining_time() < 2:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]

        grad = np.zeros(n)
        for j in top_idx:
            s = j - (n - 1)
            if s >= 0:
                grad[s:] += (1.0 - h[:n - s])
                grad[:n - s] -= h[s:]
            else:
                p = -s
                grad[:n - p] += (1.0 - h[p:])
                grad[p:] -= h[:n - p]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= (gnorm * dx)

        improved = False
        for step in [0.01, 0.02, 0.04, 0.06, 0.015, 0.03, 0.08, 0.12]:
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial.copy()
                improved = True
        if not improved:
            break

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Phase 0: Initialize from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.7f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 160
        best_h = make_thue_morse(best_n)
        best_h = binary_to_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)
        print("Thue-Morse: n={}, C5={:.7f}".format(best_n, best_c5))

    # Phase 1: Binary hill climbing at multiple resolutions with many restarts
    resolutions = sorted(set([best_n, 160, 200, 240]))
    resolutions = [nn for nn in resolutions if 80 <= nn <= 300]

    for n in resolutions:
        if remaining_time() < 40:
            break
        print("\n=== Binary hill climb n={} (remaining {:.0f}s) ===".format(n, remaining_time()))

        # Generate diverse starting points
        starts = []
        h_interp = interp_to(best_h, best_n, n)
        starts.append(binary_to_feasible(h_interp, n))
        starts.append(binary_to_feasible(make_thue_morse(n), n))
        starts.append(binary_to_feasible(make_golden_ratio(n), n))

        # Random starts
        for _ in range(3):
            h_rand = np.zeros(n)
            h_rand[rng.choice(n, n // 2, replace=False)] = 1.0
            starts.append(h_rand.copy())

        # Perturbed parent
        for _ in range(2):
            h_pert = best_h.copy() if best_n == n else interp_to(best_h, best_n, n)
            flip_idx = rng.choice(n, size=min(20, n // 10), replace=False)
            h_pert[flip_idx] = 1.0 - h_pert[flip_idx]
            starts.append(binary_to_feasible(h_pert, n))

        for si, h0 in enumerate(starts):
            if remaining_time() < 15:
                break
            print("  Start {}/{}: C5={:.7f}".format(si + 1, len(starts), compute_c5(h0)))

            h_bin, c5_bin = binary_hill_climb(h0, n, max_sweeps=5000, rng=rng,
                                              kick_interval=300, kick_size=4)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                best_n = n
                print("  -> Improved: n={}, C5={:.7f}".format(n, c5_bin))

    # Phase 2: Continuous optimization on best binary solution
    n = best_n
    h = best_h.copy()
    print("\n=== Continuous optimization n={} (C5={:.7f}) ===".format(n, best_c5))

    # L-BFGS with increasing beta
    for beta_val in [50, 100, 200, 500]:
        if remaining_time() < 15:
            break
        h_new = lbfgs_continuous(h, n, beta_val, maxiter=400, top_k=None)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            print("  L-BFGS beta={}: C5={:.7f}".format(beta_val, c5_new))

    # L-BFGS targeting top-k
    for top_k in [4, 8, 16]:
        if remaining_time() < 10:
            break
        h_new = lbfgs_continuous(h, n, 300, maxiter=300, top_k=top_k)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            print("  L-BFGS top-{}: C5={:.7f}".format(top_k, c5_new))

    # Top-k gradient descent
    if remaining_time() > 20:
        h_new, c5_new = topk_gradient_descent(h, n, max_iters=200, top_k=10)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            print("  TopK grad: C5={:.7f}".format(c5_new))

    # Phase 3: Binary re-optimization from continuous solution
    if remaining_time() > 30:
        n = best_n
        h_bin = binary_to_feasible(best_h, n)
        print("\n=== Binary re-opt from continuous (n={}) ===".format(n))
        h_bin, c5_bin = binary_hill_climb(h_bin, n, max_sweeps=3000, rng=rng,
                                          kick_interval=200, kick_size=3)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            print("  Binary re-opt: C5={:.7f}".format(c5_bin))

        # Polish binary with continuous
        if remaining_time() > 15 and c5_bin < best_c5 + 0.001:
            h_cont = h_bin.copy()
            h_new = lbfgs_continuous(h_cont, n, 500, maxiter=300)
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  Polished: C5={:.7f}".format(c5_new))

    # Phase 4: Try higher resolution binary search
    if remaining_time() > 40:
        for n_try in [200, 240, 320]:
            if remaining_time() < 20:
                break
            if n_try == best_n:
                continue
            print("\n=== Binary at n={} ===".format(n_try))
            h_interp = binary_to_feasible(interp_to(best_h, best_n, n_try), n_try)
            h_try, c5_try = binary_hill_climb(h_interp, n_try, max_sweeps=3000, rng=rng,
                                              kick_interval=250, kick_size=5)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
                print("  -> n={}: C5={:.7f}".format(n_try, c5_try))

            # Polish
            if remaining_time() > 12:
                h_new = lbfgs_continuous(h_try, n_try, 300, maxiter=200)
                h_new = project_feasible(h_new, n_try)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n_try
                    print("  Polished n={}: C5={:.7f}".format(n_try, c5_new))

    # Phase 5: Final intensive local search
    if remaining_time() > 15:
        n = best_n
        h = best_h.copy()
        print("\n=== Final polish (n={}) ===".format(n))

        # Several small L-BFGS passes
        for beta_val in [800, 1200, 2000]:
            if remaining_time() < 6:
                break
            h_new = lbfgs_continuous(h, n, beta_val, maxiter=200, top_k=6)
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

        # Final top-k descent
        if remaining_time() > 8:
            h_new, c5_new = topk_gradient_descent(h, n, max_iters=100, top_k=6)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

        # Quick binary check
        if remaining_time() > 10:
            h_bin = binary_to_feasible(best_h, n)
            h_bin, c5_bin = binary_hill_climb(h_bin, n, max_sweeps=1000, rng=rng,
                                              kick_interval=0, kick_size=0)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
