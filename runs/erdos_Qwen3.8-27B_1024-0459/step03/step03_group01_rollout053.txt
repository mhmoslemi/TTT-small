def corr_grad_at_shift(x, n, k):
    g = np.zeros(n)

    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g

            n_candidates.append(nn)

    # Remove current best_n to save time (already optimized)
    n_candidates = [nn for nn in n_candidates if nn != best_n]

    time_per_n = min(80, max(20, remaining() / max(1, len(n_candidates))))

    for n_t in n_candidates:
        if remaining() < 25:
            break
        time_per_n = min(time_per_n, max(15, remaining() / max(1, len(n_candidates) - n_candidates.index(n_t))))

        # Interpolate from best
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        h_bin_t = make_binary_feasible(h_feas, n_t, rng)

        # Also try Thue-Morse start
        h_tm = thue_morse(n_t, rng)

        # Try both starts
        for h_start_bin in [h_bin_t, h_tm]:
            if remaining() < 20:
                break
            # Binary local search
            h_b, c5_b = binary_local_search(h_start_bin.copy(), n_t,
                                            time_limit=min(15, max(5, remaining() - 25)), rng=rng)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_t
            # Continuous refinement from binary
            if remaining() > 25:
                h_cont = h_b.copy()
                for beta in [500, 2000]:
                    if remaining() < 10:
                        break
                    h_cont = slsqp_opt(h_cont, n_t, beta, maxiter=200)
                    c5_cont = compute_c5(h_cont)
                    if c5_cont < best_c5:
                        best_c5 = c5_cont
                        best_h = h_cont.copy()
                        best_n = n_t

        # Simulated annealing on best binary for this n
        if remaining() > 30:
            h_sa_best = make_binary_feasible(interp_to(best_h, best_n, n_t), n_t, rng)
            h_sa, c5_sa = sim_anneal_binary(h_sa_best, n_t,
                                            time_limit=min(25, max(5, remaining() - 30)),
                                            rng=rng, T0=2.5, alpha=0.9996)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_t
                # Continuous polish
                if remaining() > 15:
                    h_cp = slsqp_opt(h_sa, n_t, 2000, maxiter=200)
                    c5_cp = compute_c5(h_cp)
                    if c5_cp < best_c5:
                        best_c5 = c5_cp
                        best_h = h_cp.copy()
                        best_n = n_t

    # Phase 4: Final intensive refinement
    n = best_n
    h = best_h.copy()

    # More aggressive SLSQP
    for beta in [1000, 3000, 8000, 20000]:
        if remaining() < 8:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Shift targeting
    if remaining() > 30:
        h_st, c5_st = shift_targeting(h, n, max_iters=800, time_limit=min(80, remaining() - 20))
        if c5_st < best_c5:
            best_c5 = c5_st
            best_h = h_st.copy()
            h = h_st.copy()

    # Binary fine-tune
    if remaining() > 30:
        h_bin = make_binary_feasible(best_h, best_n, rng)
        h_bl, c5_bl = binary_local_search(h_bin, best_n,
                                          time_limit=min(60, max(10, remaining() - 20)), rng=rng)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()

        # SA polish
        if remaining() > 30:
            h_sa, c5_sa = sim_anneal_binary(h_bl, best_n,
                                            time_limit=min(50, max(10, remaining() - 20)),
                                            rng=rng, T0=1.5, alpha=0.9997)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()

    # Final SLSQP polish
    if remaining() > 15:
        for beta in [5000, 15000, 50000]:
            if remaining() < 5:
                break
            h_new = slsqp_opt(best_h, best_n, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
