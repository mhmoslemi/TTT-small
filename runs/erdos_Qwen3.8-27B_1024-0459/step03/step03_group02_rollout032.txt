        return g
    bounds = [(0, 1)] * len(h)

                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14})
    return project_feasible(res.x, len(h))

    for sweep in range(sweeps):

a_k(i) = (i≥d ? (1-h[i-d]) : 0) - (i≤n-1-d ? h[i+d] : 0) where d = |k-(n-1)|

So corr_k(new) = corr_k(old) + (new-old) * a_k(i)

This means I can update the correlation vector in O(n) instead of O(n²) (for the full correlate). Then finding the max is O(n). Total per coordinate: O(n). Per sweep: O(n²). For n=200, that's 40000 per sweep. Very fast!

            # where d = |k-(n-1)| = |lag|
            
            # For each d from 0 to n-1:
            # a_d = (i>=d ? (1-h[i-d]) : 0) - (i<=n-1-d ? h[i+d] : 0)
            
            # Precompute a_d for d = 0, 1, ..., n-1
            a = np.zeros(n)  # a[d] for lag d
            for d in range(n):
                a[d] = 0.0
                if i >= d:
                    a[d] += (1.0 - h[i-d])
                if i <= n-1-d:
                    a[d] -= h[i+d]
            
            # corr at lag d (both +d and -d have same value)
            # corr[n-1-d] = corr[n-1+d] = C_d
            C = np.zeros(n)  # C[d] for lag d
            for d in range(n):
                C[d] = corr[n-1-d]  # or corr[n-1+d], they're the same
            
            # New corr if h[i] = t: C_d + (t - old) * a[d]
            # Objective: max_d (C_d + (t-old)*a[d]) * dx
            # Minimize over t in [0,1]
            
            # This is max of linear functions of t, which is convex
            # Find minimum by checking breakpoints and bounds
            
            # Simple approach: evaluate at several points
            best_t = old
            best_obj = np.max(C) * dx
            
            # Check t=0 and t=1
            for t in [0.0, 1.0]:

                    best_t = t

            # (t-old)*(a_d - a_e) = C_e - C_d
            # t = old + (C_e - C_d) / (a_d - a_e)
            # Only need to check pairs where a_d ≠ a_e
            
            # For efficiency, just check a few quantized values
            for t in np.linspace(0, 1, 11):
                new_C = C + (t - old) * a
                obj = np.max(new_C) * dx
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_t = t

                h[i] = best_t
                delta = best_t - old
                # Update corr: corr[n-1±d] += delta * a[d]
                for d in range(n):

                        corr[n-1-d] += delta * a[d]
                        if d > 0:

                any_change = True
        
        if not any_change:
            break

        else:

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)

def lbfgs_objective(x, beta, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_s = np.exp(shifted)
    sum_s = np.sum(exp_s)
    sm_val = m + np.log(sum_s) / beta
    sv = np.sum(x) - 0.5 * n
    return sm_val * dx + penalty * sv * sv

def lbfgs_gradient(x, beta, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_s = np.exp(shifted)
    sum_s = np.sum(exp_s)
    weights = exp_s / sum_s
    sv = np.sum(x) - 0.5 * n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-16:
            continue
        d = abs(k - (n - 1))
        n_pairs = n - d
        if n_pairs <= 0:
            continue
        if d == 0:
            grad += wk * (1.0 - 2.0 * x)
        else:
            grad[d:] += wk * (1.0 - x[:n_pairs])
            grad[:n_pairs] -= wk * x[d:]
    grad *= dx
    grad += 2.0 * penalty * sv
    return grad

def run_lbfgs(h, beta, penalty, maxiter=2000):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    res = minimize(lbfgs_objective, h, jac=lambda x: lbfgs_gradient(x, beta, penalty),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-16, 'maxfun': 5000})
    return project_feasible(res.x, n)

def coord_descent_fast(h, n, sweeps=20, penalty=1e6):
    dx = 2.0 / n
    h = h.copy()
    corr_full = np.correlate(h, 1.0 - h, mode='full')
    C = corr_full[:n]  # C[d] = corr[n-1-d] = corr at lag d (symmetric)
    
    for sweep in range(sweeps):
        any_change = False
        for i in range(n):
            old = h[i]
            S_other = np.sum(h) - old
            
            # Compute a[d] = sensitivity of lag-d correlation to h[i]
            a = np.zeros(n)
            for d in range(n):
                val = 0.0
                if i >= d:
                    val += (1.0 - h[i - d])
                if i <= n - 1 - d:
                    val -= h[i + d]
                a[d] = val
            
            # Objective: max_d(C[d] + (t-old)*a[d]) * dx + penalty*(S_other + t - n/2)^2
            # Find best t in [0,1]
            best_t = old
            best_obj = float('inf')
            
            # Evaluate at candidate points
            cands = np.array([0.0, 1.0, old])
            # Add breakpoints from top lags
