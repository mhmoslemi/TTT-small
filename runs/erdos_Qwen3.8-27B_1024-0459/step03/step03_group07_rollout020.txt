def active_set_optimize(h, n, max_iters=100, time_limit=None, t0=None):
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if time_limit and t0 and time.time() - t0 > time_limit:
            break
        
        corr = np.correlate(h, 1-h, mode='full') * dx
        c5 = np.max(corr)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        
        # Find active set (lags within tolerance of max)
        tol = 1e-4
        active = np.where(corr >= c5 - tol)[0]
        
        # Subgradient: average of gradients of active lags
        grad = np.zeros(n)
        for idx in active:
            lag = idx - (n-1)
            if lag >= 0:
                # f_lag = sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag]) * dx
                # df_lag/dh[i] = (1-h[i+lag]) * dx for i < n-lag
                # df_lag/dh[i+lag] = -h[i] * dx for i < n-lag
                for i in range(n - lag):
                    grad[i] += (1-h[i+lag]) * dx
                    grad[i+lag] += -h[i] * dx
            else:
                o = -lag
                for i in range(n - o):
                    grad[i+o] += (1-h[i]) * dx
                    grad[i] += -h[i+o] * dx
        
        grad /= len(active)
        
        # Line search
        for alpha in [0.1, 0.05, 0.02, 0.01, 0.005]:
            h_new = h - alpha * grad
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5:
                h = h_new
                break
        else:
            break
    
    return best

def project_feasible(h, n):

    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

            break
        if diff > 0:

        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:

        else:

            h[i] = 1.0 - a
    # Mix with sine for more structure
    t = np.arange(n) / n
    h = 0.6 * h + 0.4 * (0.5 + 0.5 * np.cos(4 * np.pi * t))
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)

def block_coord_descent(h, n, block_size=4, time_limit=None, t0=None, max_rounds=50):
    """Optimize blocks of coordinates together using a finer grid."""
    h = project_feasible(h, n).copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    
    for rd in range(max_rounds):
        if time_limit and t0 and time.time() - t0 > time_limit:
            break
        improved = False
        for start in range(0, n, block_size):
            if time_limit and t0 and time.time() - t0 > time_limit:
                break
            end = min(start + block_size, n)
            block = h[start:end].copy()
            
            # Try random perturbations of the block
            for trial in range(8):
                if time_limit and t0 and time.time() - t0 > time_limit:
                    break
                h_trial = h.copy()
                if trial == 0:
                    continue
                elif trial == 1:
                    h_trial[start:end] = 1.0 - block
                elif trial == 2:
                    h_trial[start:end] = np.clip(block + 0.2 * (np.random.rand(end-start) - 0.5), 0, 1)
                elif trial == 3:
                    h_trial[start:end] = np.clip(1.0 - block + 0.1 * (np.random.rand(end-start) - 0.5), 0, 1)
                else:
                    h_trial[start:end] = np.random.rand(end - start)
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-12:
                    h = h_trial
                    best_c5 = c5_trial
                    best = h.copy()
                    improved = True
        if not improved:
            break
    
    return best

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return max(0.0, budget_s - (time.time() - t0) - 5.0)
    
    # Load parent
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn > 20:
            best_h = ph.copy()
            best_c5 = compute_c5(ph)
            best_n = pn
    except Exception:
        pass
    
    if best_h is None:
        best_n = 160
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)
    
    # Multi-resolution optimization
    resolutions = []
    base_n = best_n
    for nn in [base_n, base_n + 20, base_n + 40, base_n + 60, base_n + 80]:
        if 100 <= nn <= 400:
            resolutions.append(nn)
    resolutions = sorted(set(resolutions))
    
    for n_target in resolutions:
        if time_left() < 80:
            break
        
        # Initialize at this resolution
        if n_target == best_n and len(best_h) == n_target:
            h_init = project_feasible(best_h, n_target)
        else:
            # Interpolate from current best
            h_init = interpolate_h(best_h, len(best_h), n_target)
        
        # Also create alternative initializations
        inits = [h_init.copy()]
        
        # Two-value structured init
        for ratio in [0.12, 0.18, 0.25]:
            inits.append(two_value_init(n_target, ratio=ratio, rng=rng))
        
        # Perturbed parent
        for scale in [0.03, 0.08]:
            hp = h_init + scale * rng.randn(n_target)
            inits.append(project_feasible(hp, n_target))
        
        # Sine-based
        t = np.arange(n_target) / n_target
        for freq, amp in [(3, 0.3), (5, 0.2), (7, 0.15)]:
            hf = 0.5 + amp * np.sin(2 * np.pi * freq * t + rng.rand() * 2 * np.pi)
            inits.append(project_feasible(hf, n_target))
        
        n_inits = len(inits)
        time_per_init = max(20.0, min(150.0, time_left() * 0.7 / n_inits))
        
        for h0 in inits:
            if time_left() < 30:
                break
            tl = time_left() * 0.85
            h_cur = h0.copy()
            c5_cur = compute_c5(h_cur)
            
            # Phase 1: Coordinate descent
            h_cd = coord_descent(h_cur, n_target, sweeps=30, time_limit=tl * 0.5, t0=t0)
            c5_cd = compute_c5(h_cd)
            if c5_cd < c5_cur:
                h_cur = h_cd
                c5_cur = c5_cd
            
            # Phase 2: Active set subgradient
            if time_left() > 20:
                h_as = active_set_subgrad(h_cur, n_target, max_iters=100, time_limit=tl * 0.7, t0=t0)
                c5_as = compute_c5(h_as)
                if c5_as < c5_cur:
                    h_cur = h_as
                    c5_cur = c5_as
            
            # Phase 3: More coordinate descent (finer)
            if time_left() > 15:
                h_cd2 = coord_descent(h_cur, n_target, sweeps=40, time_limit=tl * 0.85, t0=t0)
                c5_cd2 = compute_c5(h_cd2)
                if c5_cd2 < c5_cur:
                    h_cur = h_cd2
                    c5_cur = c5_cd2
            
            # Phase 4: L-BFGS with smooth max
            if time_left() > 30 and n_target <= 260:
                h_lb = slsqp_smooth(h_cur, n_target, time_limit=tl * 0.95, t0=t0)
                c5_lb = compute_c5(h_lb)
                if c5_lb < c5_cur:
                    h_cur = h_lb
                    c5_cur = c5_lb
            
            # Phase 5: Final coordinate descent polish
            if time_left() > 15:
                h_fin = coord_descent(h_cur, n_target, sweeps=20, time_limit=tl, t0=t0)
                c5_fin = compute_c5(h_fin)
                if c5_fin < c5_cur:
                    h_cur = h_fin
                    c5_cur = c5_fin
            
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n_target
    
    # Final intensive polish on the best solution
    if time_left() > 60:
        h_pol = coord_descent(best_h, best_n, sweeps=50, time_limit=time_left() * 0.4, t0=t0)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    
    if time_left() > 40:
        h_pol2 = active_set_subgrad(best_h, best_n, max_iters=150, time_limit=time_left() * 0.5, t0=t0)
        c5_pol2 = compute_c5(h_pol2)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2.copy()
    
    if time_left() > 30:
        h_pol3 = coord_descent(best_h, best_n, sweeps=30, time_limit=time_left() * 0.5, t0=t0)
        c5_pol3 = compute_c5(h_pol3)
        if c5_pol3 < best_c5:
            best_c5 = c5_pol3
            best_h = h_pol3.copy()
    
    if time_left() > 20 and best_n <= 240:
        h_pol4 = slsqp_smooth(best_h, best_n, time_limit=time_left() * 0.6, t0=t0)
        c5_pol4 = compute_c5(h_pol4)
        if c5_pol4 < best_c5:
            best_c5 = c5_pol4
            best_h = h_pol4.copy()
    
    if time_left() > 15:
        h_pol5 = coord_descent(best_h, best_n, sweeps=30, time_limit=time_left(), t0=t0)
        c5_pol5 = compute_c5(h_pol5)
        if c5_pol5 < best_c5:
            best_c5 = c5_pol5
            best_h = h_pol5.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    # Validate
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
