def compute_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

    m = np.max(corr)

def soft_max_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)

    total = np.sum(exp_vals)
    weights = exp_vals / total

        w = weights[idx]
        if abs(w) < 1e-13:
            continue

            n_pairs = n - offset
            if n_pairs <= 0:

            abs_off = -offset

                continue
            grad[:n_pairs] += w * (1.0 - h[abs_off:abs_off+n_pairs])
            grad[abs_off:abs_off+n_pairs] -= w * h[:n_pairs]
    return grad * dx

    target = n / 2.0

        return soft_max_obj(x, n, beta)

        return soft_max_grad(x, n, beta)

        return np.sum(x) - target

        return [np.sum(x) - target]

    """Compute subgradient from active (near-max) lags."""
