        grad[:np_] += (1.0 - h[a:a+np_])
        grad[a:a+np_] -= h[:np_]
    return grad

def g_lse_obj(x):
    n = _G_N
    beta = _G_BETA
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

def g_lse_grad(x):
    n = _G_N
    beta = _G_BETA
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exps = np.exp(shifted)
    total = np.sum(exps)
    if total < 1e-300:
        return np.zeros(n)
    weights = exps / total
    grad = np.zeros(n)
    for j in range(len(weights)):
        w = weights[j]
        if w < 1e-14:
            continue
        grad += w * lag_grad_vec(x, n, j)
    return grad * (2.0 / n)

def g_sum_fun(x):
    return np.sum(x) - _G_N / 2.0

def g_sum_jac(x):
    return np.ones(_G_N)

def slsqp_refine(h, n, beta, maxiter, ftol):
    global _G_N, _G_BETA
    _G_N = n
    _G_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': g_sum_fun, 'jac': g_sum_jac}]
    try:
        res = minimize(g_lse_obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, jac=g_lse_grad,
                      options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgrad_active(h, n, n_active):
    corr = compute_corr(h)
    m = np.max(corr)
    spread = m - np.min(corr)
    tol = max(spread * 0.1, 1.0)
    active = np.where(corr >= m - tol)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(corr))])
    if len(active) > n_active:
        order = corr[active]
        top = np.argsort(order)[::-1][:n_active]
        active = active[top]
    grad = np.zeros(n)
    for k_idx in active:
        grad += lag_grad_vec(h, n, int(k_idx))
    grad /= len(active)
    return grad, corr, m

def project_feasible(h, n):

    target = n / 2.0
    for _ in range(300):

        if abs(diff) < 1e-17:

        if cnt > 0:
            h[mask] += diff / cnt
        elif n > 0:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-17 and n > 0:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)

def lag_grad_vec(h, n, k_idx):
    grad = np.zeros(n)
    offset = k_idx - (n - 1)
    if offset >= 0:
        np_ = n - offset
        if np_ <= 0:
            return grad
        grad[offset:offset+np_] += (1.0 - h[:np_])
        grad[:np_] -= h[offset:offset+np_]
    else:
        a = -offset
        np_ = n - a
        if np_ <= 0:
            return grad
        grad[:np_] += (1.0 - h[a:a+np_])
        grad[a:a+np_] -= h[:np_]
    return grad

def g_obj(x):
    n = _G_N
    beta = _G_BETA
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

def g_grad(x):
    n = _G_N
    beta = _G_BETA
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exps = np.exp(shifted)
    total = exps.sum()
    if total < 1e-300:
        return np.zeros(n)
    weights = exps / total
    grad = np.zeros(n)
    for j in range(len(weights)):
        w = weights[j]
        if w < 1e-14:
            continue
        grad += w * lag_grad_vec(x, n, j)
    return grad * (2.0 / n)

def g_eq_fun(x):
    return x.sum() - _G_N / 2.0

def g_eq_jac(x):
    return np.ones(_G_N)

def slsqp(h, n, beta, maxiter, ftol):
    global _G_N, _G_BETA
    _G_N = n
    _G_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': g_eq_fun, 'jac': g_eq_jac}]
    try:
        res = minimize(g_obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, jac=g_grad,
                      options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgrad_step(h, n, n_active, tol_frac):
    corr = compute_corr(h)
    m = np.max(corr)
    spread = m - np.min(corr)
    tol = max(spread * tol_frac, 0.5)
    active = np.where(corr >= m - tol)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(corr))])
    if len(active) > n_active:
        idx = np.argsort(corr[active])[::-1][:n_active]
        active = active[idx]
    grad = np.zeros(n)
    for k_idx in active:
        grad += lag_grad_vec(h, n, int(k_idx))
    grad /= len(active)
    return grad, float(m)

def interp_to(src, s_n, d_n):
    if s_n == d_n:
        return src.copy()
    xp = (np.arange(s_n) + 0.5) * 2.0 / s_n
    xn = (np.arange(d_n) + 0.5) * 2.0 / d_n
    return np.interp(xn, xp, src)

def coord_descent(h, n, max_sweeps, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    deltas = [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005,
              -0.05, -0.03, -0.02, -0.01, -0.005, -0.002, -0.001, -0.0005]
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            old = best_h.copy()
            ov = old[i]
            best_v = best_h.copy()
            best_v_c5 = best_c5
            for d in deltas:
                ht = old.copy()
                ht[i] = np.clip(ov + d, 0.0, 1.0)
                ht = project_feasible(ht, n)
                c5t = compute_c5(ht)
                if c5t < best_v_c5 - 1e-16:
                    best_v_c5 = c5t
                    best_v = ht.copy()
            if best_v_c5 < best_c5 - 1e-16:
                best_c5 = best_v_c5
                best_h = best_v
                improved = True
        if not improved:
            break
    return best_h, best_c5

        active = np.array([int(np.argmax(corr))])
    if len(active) > n_active:
        idx = np.argsort(corr[active])[::-1][:n_active]
        active = active[idx]
    grad = np.zeros(n)
    for k_idx in active:
        grad += lag_grad_vec(h, n, int(k_idx))
    grad /= len(active)
    return grad, m

def interp_to(src, s_n, d_n):
    if s_n == d_n:
        return src.copy()
    xp = (np.arange(s_n) + 0.5) * 2.0 / s_n
    xn = (np.arange(d_n) + 0.5) * 2.0 / d_n
    return np.interp(xn, xp, src)

def coord_descent(h, n, max_sweeps, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    deltas = [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005,
              -0.05, -0.03, -0.02, -0.01, -0.005, -0.002, -0.001, -0.0005]
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            old = best_h.copy()
            ov = old[i]
            best_v = best_h
            best_v_c5 = best_c5
            for d in deltas:
                ht = old.copy()
                ht[i] = np.clip(ov + d, 0.0, 1.0)
                ht = project_feasible(ht, n)
                c5t = compute_c5(ht)
                if c5t < best_v_c5 - 1e-17:
                    best_v_c5 = c5t
                    best_v = ht
            if best_v_c5 < best_c5 - 1e-17:
                best_c5 = best_v_c5
                best_h = best_v
                improved = True
        if not improved:
            break
    return best_h, best_c5

def pair_coord_descent(h, n, max_sweeps, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    deltas2 = [0.02, 0.01, 0.005, 0.002, -0.02, -0.01, -0.005, -0.002]
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        pairs_per_sweep = min(n, 40)
        for _ in range(pairs_per_sweep):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            old = best_h.copy()
            for di in deltas2[:4]:
                for dj in deltas2[4:]:
                    ht = old.copy()
                    ht[i] = np.clip(old[i] + di, 0.0, 1.0)
                    ht[j] = np.clip(old[j] + dj, 0.0, 1.0)
                    ht = project_feasible(ht, n)
                    c5t = compute_c5(ht)
                    if c5t < best_c5 - 1e-17:
                        best_c5 = c5t
                        best_h = ht
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
    except Exception:
        parent_n = 160
        parent_h = np.zeros(parent_n)
        for i in range(parent_n):
            parent_h[i] = 1.0 if bin(i).count('1') % 2 else 0.0
        parent_h = project_feasible(parent_h, parent_n)

    best_h = parent_h.copy()
    best_c5 = compute_c5(parent_h)
    best_n = parent_n
    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP sweep from parent (establish good local minimum)
    for beta in [200, 500, 1000, 2000, 5000, 10000, 20000]:
        if remaining_time() < 30:
            break
        h_new = slsqp(h, n, beta, 1000, 1e-19)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-17:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new

    # Phase 2: Extended subgradient descent on active lags
    for it in range(6000):
        if remaining_time() < 10:
            break
        grad, m = subgrad_step(h, n, 3, 0.08)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        improved = False
        base_step = 0.12 / (1.0 + it * 0.0015)
        for mult in [1.0, 0.5, 0.25, 0.1, 2.0, 4.0, 0.05, 0.7, 1.5]:
            step = base_step * mult
            ht = project_feasible(h - step * grad, n)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-17:
                best_c5 = c5t
                best_h = ht.copy()
                h = ht
                improved = True
                break
        if not improved and it % 300 == 150 and remaining_time() > 25:
            h_new = slsqp(h, n, 5000, 600, 1e-19)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-17:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new

    # Phase 3: SLSQP high-precision polish
    if remaining_time() > 20:
        for beta in [5000, 20000, 50000, 100000]:
            if remaining_time() < 12:
                break
            h_new = slsqp(best_h, best_n, beta, 800, 1e-19)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-17:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 4: Fine coordinate descent
    if remaining_time() > 10:
        h_cd, c5_cd = coord_descent(best_h, best_n, 150, rng)
        if c5_cd < best_c5 - 1e-16:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 5: Pair coordinate descent
    if remaining_time() > 10:
        h_p, c5_p = pair_coord_descent(best_h, best_n, 80, rng)
        if c5_p < best_c5 - 1e-16:
            best_c5 = c5_p
            best_h = h_p

    # Phase 6: Post-cd SLSQP
    if remaining_time() > 25:
        for beta in [10000, 50000]:
            if remaining_time() < 12:
                break
            h_new = slsqp(best_h, best_n, beta, 600, 1e-19)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-17:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 7: Multi-resolution exploration
    for trial_n in [144, 152, 156, 164, 168, 176, 192, 200, 224, 256]:
        if remaining_time() < 35:
            break
        h_int = interp_to(best_h, best_n, trial_n)
        h_int = project_feasible(h_int, trial_n)
        c5_int = compute_c5(h_int)
        if c5_int < best_c5 + 0.004:
            h_opt = h_int.copy()
            for beta in [500, 2000, 10000]:
                if remaining_time() < 10:
                    break
                h_opt = slsqp(h_opt, trial_n, beta, 500, 1e-19)
            c5_final = compute_c5(h_opt)
            h_cd_t, c5_cd_t = coord_descent(h_opt, trial_n, 40, rng)
            if c5_cd_t < c5_final:
                h_opt = h_cd_t
                c5_final = c5_cd_t
            if c5_final < best_c5 - 1e-16:
                best_c5 = c5_final
                best_h = h_opt.copy()
                best_n = trial_n

    # Phase 8: Small-n global structures + upsample
    for small_n in [16, 24, 32, 48, 64]:
        if remaining_time() < 50:
            break
        for restart in range(12):
            if remaining_time() < 6:
                break
            r2 = np.random.RandomState(seed * 10000 + small_n * 100 + restart)
            h_sm = r2.uniform(0, 1, small_n)
            h_sm = project_feasible(h_sm, small_n)
            h_sm = slsqp(h_sm, small_n, 300, 400, 1e-16)
            h_sm = slsqp(h_sm, small_n, 2000, 400, 1e-16)
            c5_sm = compute_c5(h_sm)
            h_up = interp_to(h_sm, small_n, best_n)
            h_up = project_feasible(h_up, best_n)
            h_up = slsqp(h_up, best_n, 2000, 500, 1e-19)
            h_up = slsqp(h_up, best_n, 10000, 400, 1e-19)
            c5_up = compute_c5(h_up)
            if c5_up < best_c5 - 1e-16:
                best_c5 = c5_up
                best_h = h_up.copy()

    # Phase 9: Random perturbation + re-optimize
    if remaining_time() > 40:
        for pert_it in range(300):
            if remaining_time() < 5:
                break
            h_pert = best_h.copy()
            n_pert = rng.randint(1, 6)
            idx_pert = rng.choice(best_n, size=n_pert, replace=False)
            for ii in idx_pert:
                h_pert[ii] = np.clip(h_pert[ii] + rng.uniform(-0.15, 0.15), 0.0, 1.0)
            h_pert = project_feasible(h_pert, best_n)
            h_pert = slsqp(h_pert, best_n, 2000, 300, 1e-19)
            c5_p = compute_c5(h_pert)
            if c5_p < best_c5 - 1e-16:
                best_c5 = c5_p
                best_h = h_pert.copy()

    # Phase 10: Final coordinate descent + SLSQP
    if remaining_time() > 8:
        h_cd3, c5_cd3 = coord_descent(best_h, best_n, 100, rng)
        if c5_cd3 < best_c5 - 1e-16:
            best_c5 = c5_cd3
            best_h = h_cd3

    if remaining_time() > 10:
        for beta in [20000, 100000]:
            if remaining_time() < 6:
                break
            h_new = slsqp(best_h, best_n, beta, 500, 1e-19)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-18:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Final ultra-fine coordinate descent
    if remaining_time() > 4:
        h_cd4, c5_cd4 = coord_descent(best_h, best_n, 60, rng)
        if c5_cd4 < best_c5 - 1e-17:
            best_c5 = c5_cd4
            best_h = h_cd4

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        ec = evaluate_erdos_solution(best_h)
        if ec < final_c5:
            final_c5 = ec
    except Exception:
        pass

    return (best_h, final_c5, best_n)
