                c2 = _c5(h2)
                if c2 < bc - 1e-14:
                    best = h2
                    bc = c2
                    imp = True
                    break
    return best, bc


def _optimize_at_res(h0, n, rng):
    """Full optimization pipeline at a given resolution"""
    h = _proj(h0.copy(), n)
    c5 = _c5(h)
    print("  Start n={} C5={:.8f}".format(n, c5))

    # Phase 1: SLSQP with moderate smoothing
    for beta in [50, 200, 500]:
        if _tl() < 20:
            break
        h2 = _slsqp(h, n, beta, 300)
        c2 = _c5(h2)
        if c2 < c5 - 1e-10:
            h, c5 = h2, c2
            print("  SLSQP b={} C5={:.8f}".format(beta, c5))

    # Phase 2: Momentum gradient descent
    if _tl() > 30:
        h, c5 = _momentum_descent(h, n, 800, rng, topm=30, bw=100.0, lr0=0.015, decay=0.9995)
        print("  MomGrad C5={:.8f}".format(c5))

    # Phase 3: Combinatorial swaps
    if _tl() > 20:
        h, c5 = _comb_swap(h, n, 80, rng)
        print("  Swap C5={:.8f}".format(c5))

    # Phase 4: More gradient
    if _tl() > 30:
        h, c5 = _momentum_descent(h, n, 600, rng, topm=50, bw=60.0, lr0=0.01, decay=0.999)
        print("  MomGrad2 C5={:.8f}".format(c5))

    # Phase 5: Coordinate descent
    if _tl() > 20:
        h, c5 = _coord_descent(h, n, 300, rng)
        print("  Coord C5={:.8f}".format(c5))

    # Phase 6: SLSQP with high smoothing
    for beta in [1000, 3000, 8000]:
        if _tl() < 15:
            break
        h2 = _slsqp(h, n, beta, 400)
        c2 = _c5(h2)
        if c2 < c5 - 1e-10:
            h, c5 = h2, c2
            print("  SLSQP hi b={} C5={:.8f}".format(beta, c5))

    # Phase 7: Final gradient polish
    if _tl() > 15:
        h, c5 = _momentum_descent(h, n, 500, rng, topm=20, bw=150.0, lr0=0.008, decay=0.999)
        print("  Final Grad C5={:.8f}".format(c5))

    # Phase 8: Final swaps + coord
    if _tl() > 10:
        h, c5 = _comb_swap(h, n, 30, rng)
        h, c5 = _coord_descent(h, n, 100, rng)
        print("  Final polish C5={:.8f}".format(c5))

    return h, c5


def _alt_init(n, rng, style):
    """Generate alternative initializations"""
    if style == 0:
        # Block: first half 1, second half 0
        h = np.zeros(n)
        h[:n // 2] = 1.0
    elif style == 1:
        # Alternating blocks
        h = np.zeros(n)
        block = n // 8
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
    elif style == 2:
        # Cosine-shaped
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.4 * np.cos(np.pi * x)
    elif style == 3:
        # Two-hump
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.35 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
    else:
        # Random binary
        h = np.zeros(n)
        idx = rng.permutation(n)[:n // 2]
        h[idx] = 1.0
    return _proj(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _t0, _tb
    _t0 = time.time()
    _tb = budget_s
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        hpar = np.array(initial_h_values, dtype=np.float64)
        npar = len(hpar)
    except Exception:
        npar = 160
        hpar = np.zeros(npar)
        hpar[:npar // 2] = 1.0

    bh = _proj(hpar, npar)
    bc5 = _c5(bh)
    bn = npar
    print("Parent n={} C5={:.8f}".format(bn, bc5))

    # Try multiple resolutions
    resolutions = []
    for nn in [npar, npar + 40, npar + 80, npar + 48, npar - 20]:
        if nn >= 32 and nn % 2 == 0 and nn not in resolutions:
            resolutions.append(nn)

    best_overall = bh.copy()
    best_c5 = bc5
    best_n = bn

    for n_try in resolutions:
        if _tl() < 60:
            break
        # Interpolate parent to this resolution
        h_init = _proj(_interp_f(bh, bn, n_try), n_try)
        print("Trying n={} (interpolated)".format(n_try))
        h_opt, c_opt = _optimize_at_res(h_init, n_try, rng)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_overall = h_opt
            best_n = n_try
            print("  ** Improved to C5={:.8f} at n={}".format(c_opt, n_try))

    # Try alternative initializations at best resolution and a couple others
    for n_try in [best_n, best_n + 20 if best_n % 2 == 0 else best_n + 21]:
        if _tl() < 80:
            break
        if n_try < 32 or n_try % 2 != 0:
            continue
        for style in [0, 1, 2, 3, 4]:
            if _tl() < 40:
                break
            h_alt = _alt_init(n_try, rng, style)
            c_alt = _c5(h_alt)
            if c_alt > 0.42:
                continue
            h_opt, c_opt = _optimize_at_res(h_alt, n_try, rng)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_overall = h_opt
                best_n = n_try
                print("  ** Alt style {} n={} C5={:.8f}".format(style, n_try, c_opt))

    # Final intensive polish on the best solution
    if _tl() > 30:
        print("Final intensive polish at n={}".format(best_n))
        rng2 = np.random.RandomState(seed + 100)
        h_pol, c_pol = _optimize_at_res(best_overall, best_n, rng2)
        if c_pol < best_c5:
            best_c5 = c_pol
            best_overall = h_pol
            print("  Polished to C5={:.8f}".format(c_pol))

    # Try tiny resolution adjustments around the best
    for delta in [2, -2, 4, -4, 6, -6]:
        if _tl() < 25:
            break
        nn = best_n + delta
        if nn < 32 or nn % 2 != 0:
            continue
        h_i = _proj(_interp_f(best_overall, best_n, nn), nn)
        h2, c2 = _momentum_descent(h_i, nn, 300, rng, topm=30, bw=80.0, lr0=0.01, decay=0.999)
        h2, c2 = _comb_swap(h2, nn, 20, rng)
        h2, c2 = _momentum_descent(h2, nn, 200, rng, topm=20, bw=150.0, lr0=0.008, decay=0.999)
        if c2 < best_c5:
            best_c5 = c2
            best_overall = h2
            best_n = nn
            print("  n+{}={} C5={:.8f}".format(delta, nn, c2))

    # Final SLSQP push
    if _tl() > 15:
        h2 = _slsqp(best_overall, best_n, 5000, 500)
        c2 = _c5(h2)
        if c2 < best_c5:
            best_c5 = c2
            best_overall = h2
        h2 = _slsqp(best_overall, best_n, 10000, 300)
        c2 = _c5(h2)
        if c2 < best_c5:
            best_c5 = c2
            best_overall = h2

    # Verify and return
    best_overall = _proj(best_overall, best_n)
    fc5 = _c5(best_overall)
    try:
        ec = evaluate_erdos_solution(best_overall)
        if ec < fc5:
            fc5 = ec
    except Exception:
        pass
    print("FINAL n={} C5={:.8f}".format(best_n, fc5))
    return (best_overall, fc5, best_n)
