    """Project h onto {0 <= h <= 1, sum(h) = target_sum}"""

        c_work = corr

    w /= np.sum(w)

    else:
        c = corr
    m = c.max()
    return m + np.log(np.sum(np.exp(beta * (c - m))) / len(c)) / beta

def subgradient_vec(h, beta, top_k=10):

    dx = 2.0 / n
    corr = compute_corr(h)
    
    if top_k < len(corr):
        idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        idx = np.arange(len(corr))
    
    c_work = corr[idx]
    m = c_work.max()
    w = np.exp(beta * (c_work - m))
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for t in range(len(idx)):
        wt = w[t]
        if wt < 1e-12:
            continue
        k = idx[t] - (n - 1)
        if k >= 0:
            i_range = np.arange(k, n)
            grad[i_range] += wt * (1.0 - h[i_range - k])
            j_range = np.arange(n - k)
            grad[j_range] += wt * (-h[j_range + k])
        else:
            kk = -k
            i_range = np.arange(n - kk)
            grad[i_range] += wt * (1.0 - h[i_range + kk])
            j_range = np.arange(kk, n)
            grad[j_range] += wt * (-h[j_range - kk])
    
    return grad * dx
