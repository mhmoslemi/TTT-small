```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def smooth_max(corr, beta):
    """Smooth approximation to max using log-sum-exp."""
    m = np.max(corr)
    if beta * (corr - m).max() < -500:
        return m + 0.0
    exps = np.exp(beta * (corr - m))
    return m + np.log(np.sum(exps)) / beta

def smooth_max_grad(corr, beta):
    """Gradient of smooth max with respect to corr."""
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    weights = exps / np.sum(exps)
    return weights

def compute_corr(h):
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}."""
    h = np.array(h, dtype=np.float64).copy()
    # Bisection on shift
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any numerical issues
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        # Distribute to interior points
        interior = (result > 0.01) & (result < 0.99)
        if np.sum(interior) > 0:
            result[interior] += diff / np.sum(interior)
            result = np.clip(result, 0.0, 1.0)
    return result

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def smooth_max_and_weights(corr, beta):
    """Compute smooth max value and softmax weights."""
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -700, 700)
    exps = np.exp(shifted)
    log_sum = m + np.log(np.sum(exps)) / beta
    weights = exps / np.sum(exps)
    return log_sum, weights

def compute_gradient(h, weights):
    """Compute gradient of smooth max w.r.t. h.
    
    h: array of length n
    weights: array of length 2n-1 (softmax weights for each lag)
    Returns: gradient array of length n
    """
    n = len(h)
    m_idx = n - 1
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        wp = weights[p]
        if abs(wp) < 1e-10:
            continue
        k = p - m_idx  # lag
        if k >= 0:
            # First term: (1 - h[j-k]) for j in [k, n-1]
            if k == 0:
                grad += wp * (1 - 2 * h)
            else:
                # j in [k, n-1]: adds wp*(1-h[j-k])
                grad[k:] += wp * (1 - h[:n-k])
                # j in [0, n-1-k]: subtracts wp*h[j+k]
                grad[:n-k] -= wp * h[k:]
        else:
            k_neg = -k
            if k_neg == 0:
                continue  # already handled
            # First term: (1 - h[j+k_neg]) for j in [0, n-1-k_neg]
            grad[:n-k_neg] += wp * (1 - h[k_neg:])
            # Second term: -h[j-k_neg] for j in [k_neg, n-1]
            grad[k_neg:] -= wp * h[:n-k_neg]
    return grad

def interp_to_fn(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def optimize_at_resolution(h_init, n, target, budget_fn, rng, max_time):
    """Optimize h at given resolution using projected gradient with smooth max."""
    h = h_init.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    
    beta_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000]
    lr = 0.5
    
    for beta in beta_schedule:
        if budget_fn() < 10:
            break
        n_iter = 3000
        lr = 0.3
        
        for it in range(n_iter):
            if budget_fn() < 5:
                break
            
            corr = compute_corr(h)
            F, weights = smooth_max_and_weights(corr, beta)
            
            # Check if we improved
            F_dx = F * (2.0 / n)
            if F_dx < c5_best - 1e-12:
                c5_best = F_dx
                h_best = h.copy()
            
            grad = compute_gradient(h, weights)
            
            # Gradient step
            h_trial = h - lr * grad
            h_trial = project_feasible(h_trial, target)
            
            # Check improvement
            corr_trial = compute_corr(h_trial)
            F_trial, _ = smooth_max_and_weights(corr_trial, beta)
            
            if F_trial < F:
                h = h_trial
            else:
                # Halve step size
                lr *= 0.5
                if lr < 0.001:
                    # Reset step size
                    lr = 0.3
                    break
    
    # Final check with raw objective
    c5_best = compute_c5(h_best)
    c5_cur = compute_c5(h)
    if c5_cur < c5_best:
        c5_best = c5_cur
        h_best = h.copy()
    
    return h_best, c5_best

def coordinate_descent_polish(h, n, budget_fn, rng):
    """Polish with coordinate descent on binary flips and small perturbations."""
    target = n / 2.0
    h = h.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    
    for sweep in range(50):
        if budget_fn() < 3:
            break
```

```python
def project_feasible(h, target):
    h = np.array(h, dtype=np.float64).copy()
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        interior = (result > 0.001) & (result < 0.999)
        cnt = np.sum(interior)
        if cnt > 0:

            result = np.clip(result, 0.0, 1.0)

    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    exps = np.exp(shifted)
    s = np.sum(exps)
    val = m + np.log(s) / beta
    weights = exps / s
    return val, weights


def compute_gradient(h, weights):

        else:

            nnk = n - kn
            grad[:nnk] += wp * (1.0 - h[kn:])

                if abs(nv - old) < 1e-7:
                    continue
                h = h_best.copy()
                h[i] = nv
                h = project_feasible(h, target)
                c5 = compute_c5(h)
                if c5 < c5_best - 1e-11:
                    c5_best = c5
                    h_best = h.copy()
                    any_imp = True
                    old = h_best[i]
            for d in [0.1, -0.1, 0.15, -0.15, 0.2, -0.2, 0.05, -0.05]:
                if budget_fn() < 2.0:
                    break
                trial = h_best.copy()
                trial[i] = np.clip(trial[i] + d, 0.0, 1.0)
                trial = project_feasible(trial, target)
                c5 = compute_c5(trial)
                if c5 < c5_best - 1e-11:
                    c5_best = c5
                    h_best = trial.copy()
                    any_imp = True

        if any_imp:
            no_imp_sweeps = 0
        else:
            no_imp_sweeps += 1
            if no_imp_sweeps >= 4:
                break

    return h_best, c5_best


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = 0
        v = i
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return h


def make_golden_ratio(n):
    h = np.zeros(n)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        fr = (i * gr) % 1.0
        h[i] = 1.0 if fr < 0.5 else 0.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
    except Exception:
        parent_n = 256
        parent_h = make_thue_morse(parent_n)

    parent_h = project_feasible(parent_h, parent_n / 2.0)
    parent_c5 = compute_c5(parent_h)

    best_h = parent_h.copy()
    best_c5 = parent_c5
    best_n = parent_n
    print("Init: n={}, C5={:.8f}".format(best_n, best_c5))

    # Phase 1: Optimize at parent resolution
    if _budget_remain() > 40:
        h1, c5_1 = optimize_multi_beta_restart(parent_h, parent_n, parent_n / 2.0,
                                                _budget_remain, rng)
        if c5_1 < best_c5:
            best_h, best_c5 = h1, c5_1
            print("P1(parent): C5={:.8f}".format(best_c5))

    # Phase 2: Higher resolutions
    res_list = [384, 512, 768]
    for n_try in res_list:
        if _budget_remain() < 80:
            break
        h_init = project_feasible(interp_to(best_h, best_n, n_try), n_try / 2.0)
        h2, c5_2 = optimize_multi_beta_restart(h_init, n_try, n_try / 2.0,
                                                _budget_remain, rng)
        if c5_2 < best_c5:
            best_h, best_c5, best_n = h2, c5_2, n_try
            print("P2(n={}): C5={:.8f}".format(n_try, best_c5))

    # Phase 3: Alternative inits at best resolution
    if _budget_remain() > 60:
        n = best_n
        target = n / 2.0
        inits = []
        inits.append(project_feasible(make_thue_morse(n), target))
        inits.append(project_feasible(make_golden_ratio(n), target))
        h_rand = (rng.rand(n) < 0.5).astype(float)
        inits.append(project_feasible(h_rand, target))

        for h_init in inits:
            if _budget_remain() < 25:
                break
            h3, c5_3 = optimize_smooth_max(h_init, n, target, _budget_remain, rng,
                                           min_time=20.0)
            if c5_3 < best_c5:
                best_h, best_c5 = h3, c5_3
                print("P3(alt): C5={:.8f}".format(best_c5))

    # Phase 4: Coordinate descent polish
    if _budget_remain() > 20:
        best_h, best_c5 = coord_descent_polish(best_h, best_n, _budget_remain, rng)
        print("P4(CD): C5={:.8f}".format(best_c5))

    # Phase 5: L-BFGS-B with low-beta smooth max
    if _budget_remain() > 25:
        n = best_n
        target = n / 2.0
        dx = 2.0 / n
        beta = 200

        def lbfgs_obj(x):
            c = compute_corr(x)
            F, _ = smooth_max_and_weights(c, beta)
            return F * dx

        def lbfgs_grad(x):
            c = compute_corr(x)
            _, w = smooth_max_and_weights(c, beta)
            return compute_gradient(x, w) * dx

        try:
            bounds = [(0.0, 1.0)] * n
            res = minimize(lbfgs_obj, best_h, jac=lbfgs_grad, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 1000, 'ftol': 1e-16,
                                                   'gtol': 1e-12, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
                print("P5(LBFGS): C5={:.8f}".format(best_c5))
        except Exception:
            pass

    # Phase 6: Final coordinate descent
    if _budget_remain() > 15:
        best_h, best_c5 = coord_descent_polish(best_h, best_n, _budget_remain, rng)

    # Final validation
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
```
