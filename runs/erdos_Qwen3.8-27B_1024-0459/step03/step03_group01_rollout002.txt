def random_bit_index(mask, rng):
    if mask == 0: return -1
    c = mask.bit_count()
    k = rng.randint(c)
    x = mask
    while k > 0:
        x ^= x & -x
        k -= 1
    lsb = x & -x
    return lsb.bit_length() - 1

def project_feasible(h, n):
    h = np.asarray(h, dtype=float)
    target = 0.5 * float(n)

    s = float(np.sum(yc))

    y = h

        ps = float(np.sum(proj))

        else:
            lo = mid

def eval_bit_score(A, n, masks):
    maxq = 0
    cnt = 0
    second = 0

        m = masks[d]

            maxq = q1
            cnt = 1

            cnt += 1

            maxq = q2
            cnt = 1
        elif q2 == maxq and maxq > 0:

def get_top_shifts(A, n, masks, top_count):

def random_bit_index(mask, rng):
    if mask == 0:
        return -1
    c = mask.bit_count()
    k = int(rng.randint(c))
    x = mask
    while k > 0:
        x ^= (x & -x)
        k -= 1
    lsb = x & -x
    return lsb.bit_length() - 1


def apply_bit_move(A, n, ones, zeros, i, j):
    A2 = A ^ (1 << i) ^ (1 << j)
    try:
        if (A >> i) & 1:
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
        else:
            zeros.remove(i)
            ones.remove(j)
            zeros.append(j)
            ones.append(i)
    except ValueError:
        mask = np.zeros(n, dtype=bool)
        for idx in range(n):
            if (A2 >> idx) & 1:
                mask[idx] = True
        ones = []
        zeros = []
        for idx in range(n):
            if mask[idx]:
                ones.append(idx)
            else:
                zeros.append(idx)
    return A2, ones, zeros

                    break

def corr_grad_per_shift(x, n, off):
    g = np.zeros(n, dtype=float)
    if off >= 0:
        npairs = n - off
        if npairs > 0:
            g[off:off + npairs] += (1.0 - x[:npairs])
            g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        if npairs > 0:
            g[:npairs] += (1.0 - x[ao:ao + npairs])
            g[ao:ao + npairs] -= x[:npairs]
    return g


def logsum_value_grad(x, n, beta):
    corr = compute_corr(x)
    m = float(np.max(corr))
    shifted = corr - m
    cl = np.clip(beta * shifted, -60.0, 60.0)
    ex = np.exp(cl)
    total = float(np.sum(ex))
    if total <= 0.0:
        return m, np.zeros(n, dtype=float)
    w = ex / total
    val = m + math.log(total) / beta

    g = np.zeros(n, dtype=float)
    idxs = np.nonzero(w > 1e-6)[0]
    for k in idxs:
        wk = float(w[k])
        off = int(k) - (n - 1)
        g += wk * corr_grad_per_shift(x, n, off)
    return val, g


def smooth_obj(x):

def slsqp_smooth(h, n, beta, maxiter=350):

    _NG = n
    _BETA = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

    except Exception:
        return h


def projected_steepest_refine(h, n, beta, time_limit=25.0, maxiter=1200, init_scale=0.25):

    cur = compute_c5(h)

    for _ in range(maxiter):

        d = g - np.mean(g)

        if nrm < 1e-12:
            break

        a = scale / nrm
        improved = False
        for mult in (1.0, 0.5, 0.25, 2.0, 0.1, 4.0, 0.05):

    return h, cur


def linearized_lp_step(h, n, k_active, eps):
    h = np.asarray(h, dtype=float)
    corr = compute_corr(h)
    L = len(corr)
    if L <= 0:
        return None
    if L <= k_active:
        idx = np.arange(L)
    else:
        idx = np.argsort(-corr)[:k_active]

    K = int(len(idx))
    if K <= 0:
        return None

    A_ub = np.zeros((K, n + 1), dtype=float)
    b_ub = np.zeros(K, dtype=float)

    for r, k in enumerate(idx):
        off = int(k) - (n - 1)
        g = corr_grad_per_shift(h, n, off)
        q0 = float(corr[int(k)])
        intercept = q0 - float(np.dot(g, h))
        A_ub[r, 0] = -1.0
        A_ub[r, 1:] = g
        b_ub[r] = intercept

    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, 1:] = 1.0
    b_eq = np.array([0.5 * float(n)], dtype=float)

    c = np.zeros(n + 1, dtype=float)
    c[0] = 1.0

    lo = np.maximum(0.0, h - eps)
    hi = np.minimum(1.0, h + eps)
    bounds = [(None, None)]
    for i in range(n):
        bounds.append((float(lo[i]), float(hi[i])))

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method="highs")
    except Exception:
        return None

    if not getattr(res, "success", False) or res.x is None:
        return None
    x = np.asarray(res.x[1:], dtype=float)
    return project_feasible(x, n)


def lp_refine(h, n, time_limit=120.0, k_active=80, init_eps=0.25):
    h = project_feasible(h, n)
    cur = compute_c5(h)
    start = time.time()

                break

    return h, cur


def active_set_equalize(h, n, max_iters=300):
    h = project_feasible(h, n)
    cur = compute_c5(h)

    for _ in range(max_iters):
        if remaining() < 6:
            break
        corr = compute_corr(h)
        if len(corr) < 2:
            break
        top = np.argsort(-corr)[:2]
        k1 = int(top[0])
        k2 = int(top[1])
        c1 = float(corr[k1])
        c2 = float(corr[k2])
        if c1 - c2 < 1e-10:
            break

        g1 = corr_grad_per_shift(h, n, k1 - (n - 1))
        g2 = corr_grad_per_shift(h, n, k2 - (n - 1))

        if c1 - c2 < 0.10 * max(c1, 1e-9):
            d = g2 - g1
        else:
            d = -g1

        d = d - np.mean(d)
        nrm = float(np.linalg.norm(d))
        if nrm < 1e-14:
            break
        d /= nrm

        improved = False
        for step in (0.05, 0.1, 0.02, 0.2, 0.01, 0.5, 0.03):
            trial = project_feasible(h + step * d, n)
            c = compute_c5(trial)
            if c < cur - 1e-13:
                h = trial
                cur = c
                improved = True
                break

        if not improved:
            break

    return h, cur


def epsilon_polish(h, n):
    h = project_feasible(h, n)
    cur = compute_c5(h)
    for eps in (1e-6, 1e-5, 3e-6, 1e-4, 3e-5):
        trial = h + eps * (0.5 - h)
        trial = np.clip(trial, 0.0, 1.0)
        trial = project_feasible(trial, n)
        c = compute_c5(trial)
        if c < cur - 1e-15:
            h = trial
            cur = c
    return h, cur


def polish_candidate(h, n, total_time):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()

    def elapsed_local():
        return time.time() - start

    if elapsed_local() < 0.35 * total_time and remaining() > 10:
        per = max(4.0, min(22.0, total_time / 10.0))
        for beta in (500, 2000, 8000):
            if elapsed_local() > 0.35 * total_time or remaining() < 8:
                break
            h2, c2 = projected_steepest_refine(best_h, n, beta, time_limit=per, maxiter=900)
            if c2 < best_c - 1e-14:
                best_h = h2
                best_c = c2

    if elapsed_local() < 0.65 * total_time and remaining() > 12 and n <= 700:
        lt = min(160.0, max(12.0, 0.30 * total_time))
        h2, c2 = lp_refine(best_h, n, time_limit=lt, k_active=90, init_eps=0.25)
        if c2 < best_c - 1e-14:
            best_h = h2
            best_c = c2

    if elapsed_local() < 0.85 * total_time and remaining() > 12 and n <= 700:
        for beta in (800, 3000, 12000):
            if elapsed_local() > 0.85 * total_time or remaining() < 8:
                break
            h2 = slsqp_smooth(best_h, n, beta, maxiter=250)
            c2 = compute_c5(h2)
            if c2 < best_c - 1e-14:
                best_h = h2
                best_c = c2

    if elapsed_local() < total_time and remaining() > 8:
        h2, c2 = active_set_equalize(best_h, n, max_iters=250)
        if c2 < best_c - 1e-14:
            best_h = h2
            best_c = c2

    return best_h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET

    _START = time.time()
    _BUDGET = float(budget_s)
    rng = np.random.RandomState(int(seed))

    best_h = None
    best_c5 = 1.0e9
    best_n = 0

    try:
        parent = np.asarray(initial_h_values, dtype=float)
        pn = int(len(parent))
        if pn < 16:
            raise ValueError("parent too small")
        parent = project_feasible(parent, pn)
        pc = compute_c5(parent)
        best_h = parent.copy()
        best_c5 = pc
        best_n = pn
    except Exception:
        pn = 128
        A0, ones0, zeros0 = bit_random_start(pn, rng)
        parent = bit_to_h(A0, pn)
        parent = project_feasible(parent, pn)
        best_h = parent.copy()
        best_c5 = compute_c5(parent)
        best_n = pn

    base_h = best_h.copy()
    base_n = best_n

    for nn in make_n_list(base_n):
        if remaining() < 4:
            break
        if nn == base_n:
            continue
        h = interp_to(base_h, base_n, nn)
        h = project_feasible(h, nn)
        c = compute_c5(h)
        if c < best_c5:
            best_h = h
            best_c5 = c
            best_n = nn

    for nn in (16, 18, 20):
        if remaining() < 4 or best_c5 <= 0.3752:
            break
        masks, _ = get_bit_masks(nn)
        A, q = brute_small_bit(nn, masks, 3.0)
        if A != 0:
            h = bit_to_h(A, nn)
            c = compute_c5(h)
            if c < best_c5:
                best_h = h
                best_c5 = c
                best_n = nn

    if remaining() > 140:
        short_time = min(130.0, max(25.0, remaining() * 0.16))
        h2, c2 = polish_candidate(best_h, best_n, short_time)
        if c2 < best_c5:
            best_h = h2
            best_c5 = c2
            best_n = best_n

    bit_ns = make_bit_n_list(best_n)
    bit_phase_start = time.time()
    bit_available = min(460.0, max(70.0, remaining() * 0.46))

    for idx, nn in enumerate(bit_ns):
        if remaining() < 65:
            break
        elapsed_phase = time.time() - bit_phase_start
        if idx == 0:
            tlimit = min(260.0, bit_available)
        elif idx == 1:
            tlimit = min(140.0, max(35.0, bit_available - elapsed_phase))
        else:
            tlimit = min(65.0, max(20.0, bit_available - elapsed_phase))

        if tlimit < 18:
            break
        if remaining() < tlimit + 45:
            if idx == 0:
                tlimit = max(20.0, remaining() - 45.0)
            else:
                break
        if tlimit < 18:
            break

        if len(best_h) == nn:
            hsrc = best_h
        else:
            hsrc = interp_to(best_h, best_n, nn)
        hsrc = project_feasible(hsrc, nn)

        A, ones, zeros = h_to_bit_state(hsrc, nn)
        masks, _ = get_bit_masks(nn)
        A2, q2 = bit_local_search(A, nn, ones, zeros, masks, rng, tlimit)
        if q2 < 10 ** 9:
            hbin = bit_to_h(A2, nn)
            c = compute_c5(hbin)
            if c < best_c5:
                best_h = hbin
                best_c5 = c
                best_n = nn

    if remaining() > 90:
        cont_time = min(380.0, max(40.0, remaining() * 0.45))
        h2, c2 = polish_candidate(best_h, best_n, cont_time)
        if c2 < best_c5:
            best_h = h2
            best_c5 = c2

    if remaining() > 115:
        if len(best_h) != best_n:
            hsrc = interp_to(best_h, best_n, best_n)
        else:
            hsrc = best_h
        hsrc = project_feasible(hsrc, best_n)
        A, ones, zeros = h_to_bit_state(hsrc, best_n)
        masks, _ = get_bit_masks(best_n)
        A2, q2 = bit_local_search(A, best_n, ones, zeros, masks, rng, min(55.0, remaining() * 0.35))
        if q2 < 10 ** 9:
            hbin = bit_to_h(A2, best_n)
            c = compute_c5(hbin)
            if c < best_c5:
                best_h = hbin
                best_c5 = c

        if remaining() > 55:
            h2, c2 = polish_candidate(best_h, best_n, min(70.0, remaining() * 0.5))
            if c2 < best_c5:
                best_h = h2
                best_c5 = c2

    if remaining() > 25:
        h2, c2 = epsilon_polish(best_h, best_n)
        if c2 < best_c5:
            best_h = h2
            best_c5 = c2

    best_h = project_feasible(best_h, best_n)
    c_local = compute_c5(best_h)
    if c_local < best_c5:
        best_c5 = c_local

    try:
        c_eval = float(evaluate_erdos_solution(best_h))
        if c_eval < best_c5:
            best_c5 = c_eval
    except Exception:
        pass

    return (best_h, float(best_c5), int(best_n))
