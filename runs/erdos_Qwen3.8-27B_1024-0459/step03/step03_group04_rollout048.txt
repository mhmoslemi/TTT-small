    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

        # d/dh[m] for m=k..n-1: (1-h[m-k])
        grad[k:] += w * (1.0 - x[:n-k])
        # d/dh[m] for m=0..n-1-k: -h[m+k]
        grad[:n-k] -= w * x[k:]
    
    return grad * dx

import numpy as np

_g_beta = 1.0
_g_lam = 1000.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def project_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xq = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xq, xp, src)

def time_left(t0, budget):
    return max(0.0, budget - (time.time() - t0))

def _soft_max_vals(vals, beta):
    m = np.max(vals)
    if beta < 1e-6:
        return m
    shifted = vals - m
    exps = np.exp(beta * shifted)
    return m + np.log(np.sum(exps)) / beta

def _soft_max_grad(corr, beta):
    """Returns softmax weights w_j such that d(logsumexp)/d(corr_j) = w_j"""
    mx = np.max(corr)
    if beta < 1e-6:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
        return w
    shifted = corr - mx
    w = np.exp(beta * shifted)
    w /= w.sum()
    return w

def _corr_grad(x, w):
    """Gradient of [sum_k w_k * corr_k] with respect to x.
    w has length 2n-1, indexed as np.correlate output."""
    n = len(x)
    grad = np.zeros(n)
    
    # lag 0 is at index n-1 in the full correlation
    w0 = w[n-1]
    if w0 > 1e-16:
        grad += w0 * (1.0 - 2.0 * x)
    
    # Positive lags k=1..n-1, at indices n..2n-2
    for k in range(1, n):
        wk = w[n-1+k]
        if wk < 1e-16:
            continue
        # corr_k = sum_{i=0}^{n-1-k} x[i]*(1-x[i+k])
        # d/dx[m] = (1-x[m+k]) for m=0..n-1-k
        #          -x[m-k] for m=k..n-1
        grad[:n-k] += wk * (1.0 - x[k:])
        grad[k:] -= wk * x[:n-k]
    
    # Negative lags k=1..n-1, at indices n-2..0
    for k in range(1, n):
        wk = w[n-1-k]
        if wk < 1e-16:
            continue
        # corr_{-k} = sum_{i=0}^{n-1-k} x[i+k]*(1-x[i])
        # d/dx[m] = (1-x[m-k]) for m=k..n-1
        #          -x[m+k] for m=0..n-1-k
        grad[k:] += wk * (1.0 - x[:n-k])
        grad[:n-k] -= wk * x[k:]
    
    return grad

def _obj_pen_pair(x):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    b = _g_beta
    val = _soft_max_vals(corr, b) * dx
    w = _soft_max_grad(corr, b)
    g = _corr_grad(x, w) * dx
    # Penalty for sum constraint
    s = np.sum(x) - n / 2.0
    val += _g_lam * s * s
    g += 2.0 * _g_lam * s
    return val, g

Wait, let me look at the numpy docs more carefully.

np.correlate(a, v, mode='full'):
result[m] = sum_n a[n] * conj(v[n+m-(len(v)-1)]) for valid n

Actually, for 1D:
If len(a) = N, len(v) = M, the full output has length N+M-1.
output[k] = sum_n a[n] * v[n - k + (M-1)] for valid n

This is the cross-correlation. Let me just verify with a small example.

For a = [a0, a1, a2], v = [v0, v1, v2]:
full output has 5 elements.

Using the formula: output[k] = sum_n a[n] * v[n+k-2] (where M-1=2)

k=0: a[0]*v[-2] + a[1]*v[-1] + a[2]*v[0] = a[2]*v[0] (only valid term)
k=1: a[0]*v[-1] + a[1]*v[0] + a[2]*v[1] = a[1]*v[0] + a[2]*v[1]
k=2: a[0]*v[0] + a[1]*v[1] + a[2]*v[2] = a0*v0 + a1*v1 + a2*v2
k=3: a[0]*v[1] + a[1]*v[2] = a0*v1 + a1*v2
k=4: a[0]*v[2] = a0*v2

So with a = h = [a,b,c] and v = 1-h = [1-a, 1-b, 1-c]:

corr[0] = c*(1-a)
corr[1] = b*(1-a) + c*(1-b)
corr[2] = a*(1-a) + b*(1-b) + c*(1-c)
