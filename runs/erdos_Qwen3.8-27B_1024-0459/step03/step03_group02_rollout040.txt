def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):

    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

    target = n / 2.0

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return (m + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-14:
            continue
        offset = k_idx - (n - 1)
        w = weights[k_idx]
        if offset >= 0:
            n_pairs = n - offset
            if n_pairs > 0:
                grad[offset:offset + n_pairs] += w * (1.0 - x[:n_pairs])
                grad[:n_pairs] -= w * x[offset:offset + n_pairs]
        else:
            abs_off = -offset
            n_pairs = n - abs_off
            if n_pairs > 0:
                grad[:n_pairs] += w * (1.0 - x[abs_off:abs_off + n_pairs])
                grad[abs_off:abs_off + n_pairs] -= w * x[:n_pairs]
    return grad * dx

def slsqp_refine(h, n, beta, maxiter=300):
    def obj(x):
        return smooth_obj(x, beta)
    def obj_grad(x):
        return smooth_grad(x, beta)
    def cons_fun(x):
        return np.sum(x) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.success or compute_c5(res.x) < compute_c5(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def generate_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = 0
        v = i
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return project_feasible(h, n)
