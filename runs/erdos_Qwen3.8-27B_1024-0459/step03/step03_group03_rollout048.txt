```python
import numpy as np
from scipy.optimize import minimize
import time

_t0 = 0.0
_tb = 1300.0
_ng = 200
_bv = 100.0


def _tl():
    return max(0.0, _tb - (time.time() - _t0) - 5.0)


def _c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / n)


def _proj(h, n):
    h = np.array(h, dtype=np.float64).copy()
    target = 0.5 * n
    for it in range(80):
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = int(np.sum(free))
        if nf > 0:
            h[free] += diff / nf
        else:
            if diff > 0:
                order = np.argsort(h)
            else:
                order = np.argsort(-h)
            rem = diff
            for i in order:
                if abs(rem) < 1e-14:
                    break
                if rem > 0:
                    add = min(1.0 - h[i], rem)
                    h[i] += add
                    rem -= add
                else:
                    sub = min(h[i], -rem)
                    h[i] -= sub
                    rem += sub
    np.clip(h, 0.0, 1.0, out=h)
    return h


def _interp(h, ns, nd):
    if ns == nd:
        return h.copy()
    xs = (np.arange(ns) + 0.5) * (2.0 / ns)
    xd = (np.arange(nd) + 0.5) * (2.0 / nd)
    return np.interp(xd, xs, h)


def _ggrad(h, k):
    n = len(h)
    s = abs(int(k))
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    g[s:] += 1.0 - h[:n - s]
    g[:n - s] -= h[s:]
    return g


def _smgrad(h, bw):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(corr))
    w = np.exp(bw * (corr - mx))
    sw = float(np.sum(w))
    if sw < 1e-30:
        idx = int(np.argmax(corr))
        w = np.zeros_like(corr)
        w[idx] = 1.0
        sw = 1.0
    w /= sw
    g = np.zeros(n)
    nz = np.where(w > 1e-8)[0]
    for k in nz:
        g += w[int(k)] * _ggrad(h, int(k - (n - 1)))
    return g


def _scon(x):
    return x.sum() - 0.5 * _ng


def _sobj(x):
    n = _ng
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = float(np.max(c))
    b = _bv
    sh = c - mx
    return (mx + np.log(np.exp(b * sh).sum()) / b) * (2.0 / n)


def _slsqp(h0, n, beta, maxiter):
    global _ng, _bv
    _ng = n
    _bv = beta
    try:
        r = minimize(_sobj, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon}],
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return _proj(np.asarray(r.x, dtype=np.float64), n)
    except Exception:
        return _proj(h0.copy(), n)


def _grad_descent(h, n, nit, rng, bw_init=30.0, bw_end=300.0, use_avg=False):
    best = h.copy()
    bc = _c5(best)
    g_avg = np.zeros(n)
    count = 0
    for i in range(nit):
        if _tl() < 3:
            break
        bw = bw_init + (bw_end - bw_init) * (i / max(1, nit))
        g = _smgrad(best, bw)
        count += 1
        if use_avg and count > 10:
            g_avg = 0.85 * g_avg + 0.15 * g
            g = g_avg
        else:
            g_avg = g
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            g = _smgrad(best, bw * 10)
            gn = float(np.linalg.norm(g))
            if gn < 1e-15:
                break
        g /= gn
        improved = False
        for a in [0.08, 0.03, 0.015, 0.006, 0.04, 0.10, 0.012, 0.004]:
            h2 = _proj(best - a * g, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-14:
                best = h2
                bc = c2
                improved = True
                break
        if not improved:
            # Try single max-lag gradient
            corr = np.correlate(best, 1.0 - best, mode='full')
            kmax = int(np.argmax(corr)) - (n - 1)
            g2 = _ggrad(best, kmax)
            gn2 = float(np.linalg.norm(g2))
            if gn2 > 1e-15:
                g2 /= gn2
                for a in [0.06, 0.02, 0.01, 0.004]:
                    h2 = _proj(best - a * g2, n)
                    c2 = _c5(h2)
                    if c2 < bc - 1e-14:
                        best = h2
                        bc = c2
                        improved = True
                        break
    return best, bc


def _sa_local_search(h, n, nsweeps, rng, t0=0.05, t1=0.002):
    best = h.copy()
    bc = _c5(best)
    for sweep in range(nsweeps):
        if _tl() < 3:
            break
        t = t0 * (t1 / t0) ** (sweep / max(1, nsweeps))
        imp_count = 0
        ones = np.where(best > 0.999)[0]
        zeros = np.where(best < 0.001)[0]
        fracs = np.where((best > 0.01) & (best < 0.99))[0]
        total_trials = 200
        for trial in range(total_trials):
            if _tl() < 2:
                break
            r = rng.random()
            if len(ones) > 0 and len(zeros) > 0 and r < 0.5:
                # Swap a 1 and a 0
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h2 = best.copy()
                h2[i], h2[j] = h2[j], h2[i]
                h2 = _proj(h2, n)
                c2 = _c5(h2)
            elif len(fracs) >= 2 and r < 0.7:
                # Adjust two fractional values
                i = fracs[rng.randint(len(fracs))]
                j = fracs[rng.randint(len(fracs))]
                if i == j:
                    continue
                h2 = best.copy()
                delta = (rng.random() - 0.5) * 0.1
                h2[i] += delta
                h2[j] -= delta
                h2 = _proj(h2, n)
                c2 = _c5(h2)
            elif len(ones) > 0 and len(fracs) > 0 and r < 0.85:
                # Swap a 1 with a fractional
                i = ones[rng.randint(len(ones))]
                j = fracs[rng.randint(len(fracs))]
                h2 = best.copy()
                h2[i], h2[j] = h2[j], h2[i]
                h2 = _proj(h2, n)
                c2 = _c5(h2)
            else:
                # Block move: swap a short contiguous block
                blen = rng.randint(2, 6)
                if n < 2 * blen + 2:
                    blen = max(2, n // 4)
                i = rng.randint(0, n - blen)
                j = rng.randint(0, n - blen)
                if abs(i - j) < blen:
                    continue
                h2 = best.copy()
                tmp1 = h2[i:i+blen].copy()
                tmp2 = h2[j:j+blen].copy()
                h2[i:i+blen] = tmp2
                h2[j:j+blen] = tmp1
                h2 = _proj(h2, n)
                c2 = _c5(h2)
            delta_c = c2 - bc
            if delta_c < -1e-14:
                best = h2
                bc = c2
                imp_count += 1
                ones = np.where(best > 0.999)[0]
                zeros = np.where(best < 0.001)[0]
                fracs = np.where((best > 0.01) & (best < 0.99))[0]
            elif delta_c < 0.1 and rng.random() < np.exp(-delta_c / max(t, 1e-8)):
                best = h2
                bc = c2
                ones = np.where(best > 0.999)[0]
                zeros = np.where(best < 0.001)[0]
                fracs = np.where((best > 0.01) & (best < 0.99))[0]
    return best, bc


def _swap_local(h, n, nsweeps, rng):
    best = h.copy()
    bc = _c5(best)
    for _ in range(nsweeps):
        if _tl() < 3:
            break
        ones = np.where(best > 0.999)[0]
        zeros = np.where(best < 0.001)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        imp = False
        for i in range(min(300, len(ones) * len(zeros))):
            if _tl() < 2:
                break
            ii = ones[rng.randint(len(ones))]
            jj = zeros[rng.randint(len(zeros))]
            h2 = best.copy()
            h2[ii], h2[jj] = h2[jj], h2[ii]
            h2 = _proj(h2, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-14:
                best = h2
                bc = c2
                imp = True
                ones = np.where(best > 0.999)[0]
                zeros = np.where(best < 0.001)[0]
        if not imp:
            # Systematic scan
            improved_any = False
            for i in ones:
                if _tl() < 2:
                    break
                for j in zeros:
                    if _tl() < 2:
                        break
                    h2 = best.copy()
                    h2[i], h2[j] = h2[j], h2[i]
                    h2 = _proj(h2, n)
                    c2 = _c5(h2)
                    if c2 < bc - 1e-14:
                        best = h2
                        bc = c2
                        improved_any = True
                        ones = np.where(best > 0.999)[0]
                        zeros = np.where(best < 0.001)[0]
                        break
                if improved_any:
                    break
            if not improved_any:
                break
    return best, bc


def _boundary_adjust(h, n, nit, rng):
    best = h.copy()
    bc = _c5(best)
    for _ in range(nit):
        if _tl() < 3:
            break
        fracs = np.where((best > 0.005) & (best < 0.995))[0]
        if len(fracs) < 2:
            break
        imp = False
        for idx in fracs:
            if _tl() < 2:
                break
            old = best[idx]
            for delta in [0.02, -0.02, 0.01, -0.01, 0.05, -0.05, 0.005, -0.005]:
                nv = old + delta
                if nv < 0.001 or nv > 0.999:
                    continue
                j = fracs[rng.randint(len(fracs))]
                if j == idx:
                    continue
                h2 = best.copy()
                h2[idx] = nv
                h2[j] += (old - nv)
                h2 = _proj(h2, n)
                c2 = _c5(h2)
                if c2 < bc - 1e-14:
                    best = h2
                    bc = c2
                    imp = True
                    break
            if imp:
                fracs = np.where((best > 0.005) & (best < 0.995))[0]
                imp = False
    return best, bc


def _optimize_n(h0, n, rng):
    h = _proj(h0.copy(), n)
    c5 = _c5(h)
    # SLSQP phase
    for beta in [50, 200, 600]:
        if _tl() < 20:
            break
        h2 = _slsqp(h, n, beta, 300)
        c2 = _c5(h2)
        if c2 < c5 - 1e-12:
            h, c5 = h2, c2
    # Gradient descent phase 1
    h, c5 = _grad_descent(h, n, 400, rng, bw_init=20.0, bw_end=200.0)
    # Local search
    h, c5 = _sa_local_search(h, n, 30, rng)
    h, c5 = _swap_local(h, n, 40, rng)
    # Gradient descent phase 2
    h, c5 = _grad_descent(h, n, 300, rng, bw_init=100.0, bw_end=500.0, use_avg=True)
    h, c5 = _swap_local(h, n, 30, rng)
    # SLSQP polish
    if _tl() > 15:
        h2 = _slsqp(h, n, 2000, 400)
        c2 = _c5(h2)
        if c2 < c5 - 1e-12:
            h, c5 = h2, c2
    h, c5 = _grad_descent(h, n, 200, rng, bw_init=200.0, bw_end=800.0, use_avg=True)
    h, c5 = _boundary_adjust(h, n, 100, rng)
    h, c5 = _swap_local(h, n, 20, rng)
    # Final SLSQP
    if _tl() > 12:

        h, c5 = _swap_local(h, n, 15, rng)
```
