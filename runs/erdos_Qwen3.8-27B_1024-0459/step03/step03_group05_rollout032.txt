def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_max_obj(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k < len(corr):
        idx = np.argsort(corr)[-top_k:]
        p = corr[idx]
    else:
        p = corr
    m = np.max(p)
    exps = np.exp(beta * (p - m))
    return (m + np.log(np.sum(exps)) / beta) * dx


def subgradient_top_k(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k < len(corr):
        idx = np.argsort(corr)[-top_k:]
    else:
        idx = np.arange(len(corr))
    p = corr[idx]
    m = np.max(p)
    w = np.exp(beta * (p - m))
    w /= np.sum(w)
    grad = np.zeros(n)
    for t in range(len(idx)):
        if w[t] < 1e-12:
            continue
        s = idx[t] - (n - 1)
        if s >= 0:
            max_i = n - 1 - s
            i_arr = np.arange(0, max_i + 1)
            j_arr = i_arr + s
            grad[i_arr] += w[t] * (1.0 - h[j_arr])
            grad[j_arr] += w[t] * (-h[i_arr])
        else:

            grad[i_arr] += w[t] * (1.0 - h[j_arr])
            grad[j_arr] += w[t] * (-h[i_arr])
    return grad * dx


def coordinate_descent(h, target, best_c5, rng, n_trials, time_fn, budget):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    improved_count = 0
    for _ in range(n_trials):
        if time_fn() > budget - 5:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if h[i] < h[j]:
            di, dj = 1, -1
        else:
            di, dj = -1, 1
        for delta in [0.02, 0.01, 0.005, 0.05, 0.003]:
            if h[i] + di * delta < -1e-9 or h[i] + di * delta > 1.0 + 1e-9:
                continue
            if h[j] + dj * delta < -1e-9 or h[j] + dj * delta > 1.0 + 1e-9:
                continue
            h[i] += di * delta
            h[j] += dj * delta
            c5_test = compute_c5(h)
            if c5_test < best_c5 - 1e-12:
                best_c5 = c5_test
                improved_count += 1
            else:
                h[i] -= di * delta
                h[j] -= dj * delta
                break
    return h, best_c5, improved_count
