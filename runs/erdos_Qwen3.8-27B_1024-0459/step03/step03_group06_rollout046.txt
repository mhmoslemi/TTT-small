import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_N = 0
_BETA = 50.0
_TOPK = 15
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 2.0)


def _topk_indices(corr, topk):
    full = len(corr)
    if topk >= full:
        return np.arange(full), corr
    idx = np.argsort(corr)[-topk:]
    return idx, corr[idx]


def obj_fn(h):
    n = _N
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx, vals = _topk_indices(corr, _TOPK)
    m = np.max(vals)
    sm = (m + np.log(np.sum(np.exp(_BETA * (vals - m))))) / _BETA * dx
    pen = 1e5 * (np.sum(h) - _TGT) ** 2
    return sm + pen


def grad_fn(h):
    n = _N
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx, vals = _topk_indices(corr, _TOPK)
    m = np.max(vals)
    weights = np.exp(_BETA * (vals - m))
    wsum = np.sum(weights)
    weights /= wsum
    grad = np.zeros(n)
    for i, j in enumerate(idx):
        wj = weights[i]
        if wj < 1e-15:
            continue
        s = j - (n - 1)
        if s >= 0:
            grad[s:] += wj * (1.0 - h[:n - s])
            grad[:n - s] -= wj * h[s:]
        else:
            sp = -s
            grad[:n - sp] += wj * (1.0 - h[sp:])
            grad[sp:] -= wj * h[:n - sp]
    grad *= dx
    grad += 2.0 * 1e5 * (np.sum(h) - _TGT)
    return grad


def interpolate(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h


def make_golden(n):
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        fr = (i * gr) % 1.0
        h[i] = 1.0 if abs(fr - 0.5) < 0.25 else 0.0
    return h


def make_sine(n, freq, phase):
    t = np.arange(n) / float(n)
    h = 0.5 + 0.3 * np.sin(2.0 * np.pi * freq * t + phase)
    return np.clip(h, 0.0, 1.0)


def make_multi_sine(n, freqs, phases, rng):
    t = np.arange(n) / float(n)
    h = np.full(n, 0.5)
    for f, ph in zip(freqs, phases):
        amp = rng.uniform(0.08, 0.22)
        h += amp * np.sin(2.0 * np.pi * f * t + ph)
    return np.clip(h, 0.0, 1.0)


def lbfgs_optimize(h0, n, beta, topk, maxiter=2000, ftol=1e-16):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _BETA = beta
    _TOPK = topk
    _TGT = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h0, jac=grad_fn, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol,
                                'gtol': 1e-12, 'iprint': 0, 'maxls': 50})
        return project_feasible(res.x, n), res
    except Exception:
        return h0, None


def binary_sa(h_init, n, n_iters, rng, t0=0.003, t1=0.00002):

    target = n // 2
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) > target:
        rm = rng.choice(len(ones), len(ones) - target, replace=False)
        h[ones[rm]] = 0.0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
    elif len(ones) < target:
        ad = rng.choice(len(zeros), target - len(ones), replace=False)
        h[zeros[ad]] = 1.0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

    best_h = h.copy()

    cur_c5 = best_c5
    olist = ones.tolist()
    zlist = zeros.tolist()

    for it in range(n_iters):

            break
        if not olist or not zlist:
            break
        i = rng.randint(len(olist))
        j = rng.randint(len(zlist))
        oi, oz = olist[i], zlist[j]
        h[oi] = 0.0
        h[oz] = 1.0
        nc = compute_c5(h)
        T = t0 * (t1 / t0) ** (it / max(n_iters, 1))
        if nc < cur_c5 or (T > 1e-16 and rng.rand() < np.exp(-(nc - cur_c5) / T)):
            cur_c5 = nc

            zlist[j] = oi
            if nc < best_c5:

                best_h = h.copy()
        else:

            h[oz] = 0.0
    return best_h, best_c5


def active_set_polish(h, n, n_steps=200, topk=12):
    dx_val = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_steps):
        if remaining() < 3:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(corr)[-topk:]
        grad = np.zeros(n)
        for j in idx:
            s = j - (n - 1)
            if s >= 0:
                grad[s:] += (1.0 - h[:n - s])
                grad[:n - s] -= h[s:]
            else:
                sp = -s
                grad[:n - sp] += (1.0 - h[sp:])
                grad[sp:] -= h[:n - sp]
        gn = np.linalg.norm(grad)
        if gn < 1e-14:

        grad /= gn

            ct = compute_c5(ht)
            if ct < best_c5:

                h = ht
                improved = True
                break

            break
