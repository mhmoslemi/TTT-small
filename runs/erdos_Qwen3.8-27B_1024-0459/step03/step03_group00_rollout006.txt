def project_sum_box(x, target):
    x = np.asarray(x, dtype=float).copy()
    if x.size == 0:
        return x
    x = np.nan_to_num(x, nan=0.0, posinf=1.0, neginf=0.0)

    target = float(np.clip(target, 0.0, float(x.size)))

def apply_flip(c, h, p, s, prec):
    q, idx1, idx2 = prec[p]
    if s == 1:
        c[idx1] += 1 - h[q]
        c[idx2] -= h[q]
    else:
        c[idx1] -= 1 - h[q]
        c[idx2] += h[q]


def eval_swap(c, h, a, b, prec):
    if a == b or h[a] == h[b]:
        return None
    old_a = int(h[a])
    old_b = int(h[b])
    c2 = c.copy()
    if old_a == 0:
        inc = a
        dec = b
    else:
        inc = b
        dec = a
    h[inc] = 1
    apply_flip(c2, h, inc, 1, prec)
    h[dec] = 0
    apply_flip(c2, h, dec, -1, prec)
    M2 = int(np.max(c2))
    hot2 = int(np.count_nonzero(c2 == M2))
    near2 = int(np.count_nonzero(c2 >= M2 - 1))
    h[inc] = old_a
    h[dec] = old_b
    return c2, M2, hot2, near2


def contrib_positions(h, d, N):
    empty = (np.empty(0, dtype=np.int64), np.empty(0, dtype=np.int64))

def soft_obj(x):
    n = int(_G_N)
    h = x
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    z = _G_BETA * (corr - m)
    exps = np.exp(np.clip(z, -700.0, 0.0))
    S = float(np.sum(exps))
    if S <= 0.0 or not np.isfinite(S):
        val = m * _G_DX
    else:
        val = (m + np.log(S) / _G_BETA) * _G_DX
    err = float(np.sum(h) - _G_TGT)
    val += _G_PEN * err * err
    return float(val)


def soft_jac(x):
    n = int(_G_N)
    h = x
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    z = _G_BETA * (corr - m)
    exps = np.exp(np.clip(z, -700.0, 0.0))
    S = float(np.sum(exps))
    grad = np.zeros(n, dtype=float)
    if S > 0.0 and np.isfinite(S):
        w = exps / S
        active = np.flatnonzero(w > 1.0e-9)
        base = n - 1
        for idx in active:
            d = int(idx - base)
            wt = float(w[idx])
            if d > 0:
                ell = n - d
                grad[:ell] -= wt * h[d:]
                grad[d:] += wt * (1.0 - h[:ell])
            elif d < 0:
                e = -d
                ell = n - e
                grad[:ell] += wt * (1.0 - h[e:])
                grad[e:] -= wt * h[:ell]
            else:
                grad += wt * (1.0 - 2.0 * h)
    if _G_PEN > 0.0:
        err = float(np.sum(h) - _G_TGT)
        grad += 2.0 * _G_PEN * err
    return grad

                    method="L-BFGS-B",
                    bounds=bounds,
                    callback=_minimize_timeout_callback,
                    options=options,
                )
                h_new = project_sum_box(res.x, _G_TGT)
                c_new = compute_c5(h_new)
                if c_new < c_cur - 1.0e-12:
                    c_cur = c_new
                    h_cur = h_new
            except TypeError:
                try:
                    res = minimize(
                        soft_obj,
                        h_cur,
                        jac=soft_jac,
                        method="L-BFGS-B",
                        bounds=bounds,
                        options=options,
                    )
                    h_new = project_sum_box(res.x, _G_TGT)
                    c_new = compute_c5(h_new)
                    if c_new < c_cur - 1.0e-12:
                        c_cur = c_new
                        h_cur = h_new
                except Exception:
                    pass
            except Exception:
                pass

        if c_cur < best_c:
            best_h = h_cur.copy()
            best_c = c_cur

    return best_h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET

    _G_START = time.time()
    _G_BUDGET = float(budget_s)

    try:
        seed_int = int(seed) & 0x7FFFFFFF
    except Exception:
        seed_int = 42 & 0x7FFFFFFF
    rng = np.random.RandomState(seed_int)

    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        if parent_h.ndim != 1 or parent_h.size == 0:
            raise ValueError("bad parent")
        best_h = project_sum_box(parent_h, parent_h.size / 2.0)
        best_n = int(parent_h.size)
        best_c = compute_c5(best_h)
    except Exception:
        n0 = 256
        best_h = None
        best_c = 1.0e9
        best_n = n0

        h = make_thue_morse(n0).astype(float)
        h = project_sum_box(h, n0 / 2.0)
        c = compute_c5(h)
        if c < best_c:
            best_h = h
            best_c = c

        h = make_golden_ratio(n0).astype(float)
        h = project_sum_box(h, n0 / 2.0)
        c = compute_c5(h)
        if c < best_c:
            best_h = h
            best_c = c

        h = random_binary(n0, rng).astype(float)
        h = project_sum_box(h, n0 / 2.0)
        c = compute_c5(h)
        if c < best_c:
            best_h = h
            best_c = c

    # Main binary search at a practical resolution.
    n_main = best_n if (best_n % 2 == 0 and 128 <= best_n <= 384) else 256
    rem = get_remaining(10.0)
    if rem > 120.0:
        h_bin = quantize_binary(interp_to(best_h, best_n, n_main), n_main, rng)
        prec = make_prefcomp(n_main)
        allow = min(700.0, 0.5 * rem)
        if allow > 30.0:
            deadline = time.time() + allow
            bh, bM = binary_optimize(h_bin, n_main, prec, deadline, rng)
            c5 = float(bM) * (2.0 / float(n_main))
            if c5 < best_c - 1.0e-12:
                best_h = bh.astype(float)
                best_n = n_main
                best_c = c5

    # Try a few smaller resolutions for better quantization / easier local search.
    rem = get_remaining(10.0)
    alt_ns = []
    for cand in (192, 160, 128):
        if cand != n_main:
            alt_ns.append(cand)
    if alt_ns and rem > 150.0:
        time_per = max(40.0, min(120.0, (rem - 50.0) / float(max(1, len(alt_ns)))))
        for n_alt in alt_ns[:2]:
            if get_remaining(30.0) < 80.0:
                break
            h0 = quantize_binary(interp_to(best_h, best_n, n_alt), n_alt, rng)
            prec = make_prefcomp(n_alt)
            deadline = time.time() + min(time_per, get_remaining(30.0))
            if deadline <= time.time() + 20.0:
                continue
            bh, bM = binary_optimize(h0, n_alt, prec, deadline, rng)
            c5 = float(bM) * (2.0 / float(n_alt))
            if c5 < best_c - 1.0e-12:
                best_h = bh.astype(float)
                best_n = n_alt
                best_c = c5

    # Continuous refinement from the best discrete / interpolated solution.
    rem = get_remaining(15.0)
    if rem > 150.0:
        n_cont = best_n if (best_n % 2 == 0 and 128 <= best_n <= 512) else 256
        starts = []

        h1 = project_sum_box(interp_to(best_h, best_n, n_cont), n_cont / 2.0)
        starts.append(h1)

        try:
            par = np.asarray(initial_h_values, dtype=float)
            if par.size > 0:
                if int(par.size) == n_cont:
                    h2 = par.copy()
                else:
                    h2 = interp_to(par, par.size, n_cont)
                h2 = project_sum_box(h2, n_cont / 2.0)
                starts.append(h2)
        except Exception:
            pass

        if rem > 320.0:
            starts.append(make_fourier_start(n_cont, rng))

        allow = min(450.0, 0.55 * rem)
        if allow > 60.0:
            deadline = time.time() + allow
            ch, cc = continuous_optimize_starts(
                starts, n_cont, [80, 200, 500], 50, deadline, rng
            )
            if ch is not None and cc < best_c - 1.0e-12:
                best_h = ch
                best_n = n_cont
                best_c = cc

    # Optional higher-resolution continuous continuation if enough time remains.
    rem = get_remaining(15.0)
    if rem > 180.0:
        n_cont2 = 384
        starts2 = []
        starts2.append(project_sum_box(interp_to(best_h, best_n, n_cont2), n_cont2 / 2.0))
        try:
            par = np.asarray(initial_h_values, dtype=float)
            if par.size > 0:
                if int(par.size) == n_cont2:
                    h2 = par.copy()
                else:
                    h2 = interp_to(par, par.size, n_cont2)
                h2 = project_sum_box(h2, n_cont2 / 2.0)
                starts2.append(h2)
        except Exception:
            pass
        if rem > 300.0:
            starts2.append(make_fourier_start(n_cont2, rng))

        allow = min(240.0, 0.4 * rem)
        if allow > 60.0:
            deadline = time.time() + allow
            ch, cc = continuous_optimize_starts(
                starts2, n_cont2, [100, 300], 40, deadline, rng
            )
            if ch is not None and cc < best_c - 1.0e-12:
                best_h = ch
                best_n = n_cont2
                best_c = cc

    # Final validation and projection.
    best_h = np.asarray(best_h, dtype=float)
    best_n = int(best_n)
    best_h = project_sum_box(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c = compute_c5(best_h)

    try:
        ev = float(evaluate_erdos_solution(best_h))
        if np.isfinite(ev) and ev < final_c:
            final_c = ev
    except Exception:
        pass

    return best_h, final_c, best_n
