        c5_bin, _ = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            h = h_bin.copy()

        for bsl in range(8):
            if tleft() < 8:
                break
            h_cur = best_h.copy()
            h_cur, c5_bs = binary_swap_search(h_cur, n, max_iters=4000, rng=rng)
            if c5_bs < best_c5:
                best_c5 = c5_bs
                best_h = h_cur.copy()
                h = h_cur.copy()
                print("  BinLS {}: C5={:.10f}".format(bsl, best_c5))

    # Phase 5: Coordinate descent
    if tleft() > 30:
        h_cur = best_h.copy()
        for cd in range(6):
            if tleft() < 6:
                break
            h_cur, c5_cd = coord_descent(h_cur, n, max_sweeps=25, rng=rng)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cur.copy()
                print("  CoordDesc {}: C5={:.10f}".format(cd, best_c5))

    # Phase 6: Basin hopping - perturb and re-optimize
    if tleft() > 40:
        for bh in range(3):
            if tleft() < 15:
                break
            h_pert = best_h + 0.05 * (rng.rand(n) - 0.5)
            h_pert = project_sum(h_pert, n)
            c5_pert, overlaps_pert = compute_c5(h_pert)

            _g_active = find_active_set(overlaps_pert, top_m=15, gap=0.3)
            _g_beta = 500
            _g_lam = 500.0
            try:
                res = minimize(obj_lse, h_pert, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-14})
                h_new = project_sum(res.x, n)
                c5_new, _ = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  BasinHop {}: C5={:.10f}".format(bh, best_c5))
            except Exception:
                pass

    # Phase 7: Final high-precision polish
    if tleft() > 10:
        _g_n = best_n
        _g_dx = 2.0 / best_n
        _g_active = np.arange(2 * best_n - 1)
        _g_beta = 5000
        _g_lam = 1e6
        bounds_f = [(0.0, 1.0)] * best_n
        cons_f = [{'type': 'eq', 'fun': con_sum}]
        try:
            res = minimize(obj_lse_slsqp, best_h, method='SLSQP',
                           bounds=bounds_f, constraints=cons_f,
                           options={'maxiter': 300, 'ftol': 1e-20, 'iprint': 0})
            h_new = project_sum(res.x, best_n)
            c5_new, _ = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    # Final
    best_h = project_sum(best_h, best_n)
    final_c5, _ = compute_c5(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
