    h = np.asarray(h, dtype=float).copy()

    """Compute d(corr)/dh[i] for all shifts."""
    n = len(h)

Let v = S[i] - S[j] (vector of length 2n-1).
We want to minimize max(corr + δ*v) subject to 0 ≤ h[i]+δ ≤ 1 and 0 ≤ h[j]-δ ≤ 1 (for δ > 0, or reversed).

The function max(corr + δ*v) is convex in δ (max of linear functions). So we can minimize it by:
1. Finding all δ where corr[k] + δ*v[k] changes which k is the max

def grad_smooth_max_fast(h, beta, top_k, corr, S):
    n = len(h)
    dx = 2.0 / n
    m = len(corr)
    if top_k is not None and top_k < m:
        idx = np.argsort(corr)[-top_k:]
        cw = corr[idx]
    else:
        idx = np.arange(m)
        cw = corr
    mx = np.max(cw)
    shifted = cw - mx
    w = np.exp(beta * shifted)
    ws = np.sum(w)
    if ws < 1e-300:
        w = np.zeros(len(cw))
        w[np.argmax(cw)] = 1.0
    else:
        w /= ws
    val = (mx + np.log(ws) / beta) * dx
    Sw = S[:, idx]
    g = Sw.T @ w
    g *= dx
    return val, g


def do_gradient_descent(h, n, t_start, budget, max_time, rng):

        return budget_s - (time.time() - t0)

    h_par = np.array(initial_h_values, dtype=float)
    n_par = len(h_par)
    best_h = h_par.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_par
    print("Parent: n={}, C5={:.6f}".format(best_n, best_c5))

    n_list = sorted(set([
        n_par, n_par + 40, n_par + 80, n_par + 120,
        n_par + 160, n_par + 200, n_par + 240, n_par + 320
    ]))
    n_list = [nn for nn in n_list if 120 <= nn <= 520 and nn % 2 == 0]

    for n in n_list:
        if remaining() < 60:
            break
        target = n / 2.0
        h_init = interpolate_h(best_h, best_n, n)
        h_init = project_feasible(h_init, target)
        c5_init = compute_c5(h_init)
        print("n={} init: C5={:.6f}".format(n, c5_init))

        t_phase = time.time()
        h_gd, c5_gd = do_gradient_descent(h_init, n, t0, budget_s, 120, rng)
        if c5_gd < best_c5:
            best_c5 = c5_gd
            best_h = h_gd.copy()
            best_n = n
            print("  GD n={}: C5={:.6f}".format(n, c5_gd))

        if remaining() < 30:
            break

        t_ls = time.time()
        h_ls, c5_ls = do_systematic_local_search(h_gd, n, best_c5, t0, budget_s, 40)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            best_n = n
            print("  LS n={}: C5={:.6f}".format(n, c5_ls))

        if remaining() < 20:
            break

        h_ft, c5_ft = do_fine_tune(h_ls, n, best_c5, rng, t0, budget_s)
        if c5_ft < best_c5:
            best_c5 = c5_ft
            best_h = h_ft.copy()
            best_n = n
            print("  FT n={}: C5={:.6f}".format(n, c5_ft))

    print("After resolution sweep: n={}, C5={:.6f}".format(best_n, best_c5))

    if remaining() > 30:
        h_rs, c5_rs = do_random_local_search(best_h, best_n, best_c5, rng, t0, budget_s, 80)
        if c5_rs < best_c5:
            best_c5 = c5_rs
            best_h = h_rs.copy()
            print("  RLS: C5={:.6f}".format(c5_rs))

    if remaining() > 15:
        h_pl, c5_pl = do_lbfgs_polish(best_h, best_n, best_c5, t0, budget_s)
        if c5_pl < best_c5:
            best_c5 = c5_pl
            best_h = h_pl.copy()
            print("  L-BFGS: C5={:.6f}".format(c5_pl))

    if remaining() > 10:
        h_fin, c5_fin = do_fine_tune(best_h, best_n, best_c5, rng, t0, budget_s)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            print("  Final FT: C5={:.6f}".format(c5_fin))

    if remaining() > 10:
        h_rs2, c5_rs2 = do_random_local_search(best_h, best_n, best_c5, rng, t0, budget_s, 40)
        if c5_rs2 < best_c5:
            best_c5 = c5_rs2
            best_h = h_rs2.copy()
            print("  RLS2: C5={:.6f}".format(c5_rs2))

    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
