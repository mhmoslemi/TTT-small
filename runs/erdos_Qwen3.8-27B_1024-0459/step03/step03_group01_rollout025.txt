    """Compute the change in all C(s) when h[a]: 1→0, h[b]: 0→1."""
    delta = np.zeros(2*n - 1)

    # For shift s > 0: C(s) = sum_i h[i]*(1-h[i+s])
    #   term at i=a: was (1-h[a+s]), now 0. delta: -(1-h[a+s]) if a+s < n
    #   term at i=a-s: was h[a-s]*1, now 0. delta: -h[a-s] if a-s >= 0
    for s in range(1, n):

        if a + s < n:
            delta[n-1+s] -= (1 - h[a+s])
        if a - s >= 0:

        # h[b] 0→1 effect on C(s):

            nc = _corr(h)
            nm = float(np.max(nc))
            if nm < max_c:
                max_c = nm; best_c5 = nm * dx; corr = nc; stall = 0
            else:
                h[i] = 1; h[j] = 0
            continue

        found = False
        # Try ALL bad-good pairs (full scan)
        for i in bad:
            i = int(i)
            for j in good:
                j = int(j)
                h[i] = 0.0; h[j] = 1.0
                nc = _corr(h)
                nm = float(np.max(nc))
                if nm < max_c - 1e-15:
                    max_c = nm
                    best_c5 = nm * dx
                    corr = nc
                    found = True
                    break
                h[i] = 1.0; h[j] = 0.0
            if found:
                break

        if found:
            stall = 0
        else:
            stall += 1
            # 2-swap escape
            if len(ones) >= 2 and len(zeros) >= 2:
                for _ in range(100):
                    i1 = int(ones[rng.randint(len(ones))])
                    i2 = int(ones[rng.randint(len(ones))])
                    j1 = int(zeros[rng.randint(len(zeros))])
                    j2 = int(zeros[rng.randint(len(zeros))])
                    if i1 == i2 or j1 == j2 or j1 in (i1, i2) or j2 in (i1, i2):
                        continue
                    h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
                    nc = _corr(h)
                    nm = float(np.max(nc))
                    if nm < max_c - 1e-15:
                        max_c = nm; best_c5 = nm * dx; corr = nc
                        stall = 0; found = True; break
                    h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0

    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
    except Exception:
        parent_h = None
        parent_n = 320

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _START) - 2.0)

    # Gather initial candidates
    candidates = []  # list of (h_binary, n)

    if parent_h is not None:
        for n_try in [parent_n]:
            h_bin = _make_binary(parent_h, n_try, rng)
            candidates.append((h_bin, n_try))

    # Thue-Morse at various sizes
    for n_tm in [320, 384, 448, 512]:
        tm = _thue_morse(n_tm)
        # Fix to exactly n/2 ones
        n_ones = int(np.sum(tm))
        if n_ones != n_tm // 2:
            tm = _make_binary(tm, n_tm, rng)
        candidates.append((tm, n_tm))

    # Random initializations
    for n_rand in [320, 400, 480, 640]:
        h_rand = _make_binary(np.random.RandomState(seed + n_rand).rand(n_rand), n_rand,
                              np.random.RandomState(seed + n_rand + 1))
        candidates.append((h_rand, n_rand))

    # Sort candidates by parent quality first
    candidates.sort(key=lambda x: x[1] != parent_n)

    # Phase 1: Quick greedy on all candidates (short time each)
    t_alloc1 = 0.25 * budget_s
    t_per = max(5.0, t_alloc1 / len(candidates))
    for idx, (h_init, n) in enumerate(candidates):
        if remaining() < 5:
            break
        h_cur, c5_cur = _greedy_search(h_init.copy(), n, t_per, np.random.RandomState(seed + idx))
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n

    # Phase 2: Thorough search on top candidates
    top_ns = set()
    if parent_n in [c[1] for c in candidates]:
        top_ns.add(parent_n)
    top_ns.add(best_n)

    t_alloc2 = 0.35 * budget_s
    t_per2 = max(10.0, t_alloc2 / max(1, len(top_ns)))

    for n in sorted(top_ns):
        if remaining() < 10:
            break
        # Best init for this n
        h_start = None
        if n == best_n:
            h_start = best_h.copy()
        else:
            for (hc, nn) in candidates:
                if nn == n:
                    h_start = hc.copy()
                    break
            if h_start is None:
                h_start = _make_binary(np.random.RandomState(seed + n).rand(n), n,
                                       np.random.RandomState(seed + n + 1))

        # Multi-shift greedy
        h_cur, c5_cur = _multi_shift_greedy(h_start.copy(), n, t_per2 * 0.4,
                                            np.random.RandomState(seed + n * 7))
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n

        # Exhaustive greedy on best so far
        h_start2 = best_h.copy() if best_n == n else h_start.copy()
        h_cur, c5_cur = _exhaustive_greedy(h_start2.copy(), n, t_per2 * 0.5,
                                           np.random.RandomState(seed + n * 11))
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n

    # Phase 3: Simulated annealing on best solution
    if remaining() > 10:
        n = best_n
        sa_temp = 2.0
        h_cur = best_h.copy()
        t_sa = min(remaining() * 0.4, 200.0)
        h_cur, c5_cur = _greedy_search(h_cur.copy(), n, t_sa,
                                       np.random.RandomState(seed + 999), sa_temp=sa_temp)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n

    # Phase 4: Final intensive greedy on best
    if remaining() > 5:
        n = best_n
        t_final = remaining() * 0.5
        h_cur, c5_cur = _exhaustive_greedy(best_h.copy(), n, t_final,
                                           np.random.RandomState(seed + 7777))
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n

    # Phase 5: Try interpolation to higher res then back
    if remaining() > 15:
        n_hi = best_n * 2
        h_hi = _make_binary(best_h, n_hi, np.random.RandomState(seed + 555))
        h_hi, c5_hi = _greedy_search(h_hi, n_hi, min(remaining() * 0.3, 60.0),
                                     np.random.RandomState(seed + 444))
        # Project back to best_n
        h_back = _make_binary(h_hi, best_n, np.random.RandomState(seed + 333))
        h_back, c5_back = _exhaustive_greedy(h_back, best_n, min(remaining() * 0.2, 30.0),
                                             np.random.RandomState(seed + 222))
        if c5_back < best_c5:
            best_c5 = c5_back
            best_h = h_back.copy()
            best_n = best_n

    # Final validation
    corr_final = _corr(best_h)
    final_c5 = _c5_of(corr_final, best_n)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
