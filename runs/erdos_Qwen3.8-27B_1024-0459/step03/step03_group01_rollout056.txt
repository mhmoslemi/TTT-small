        shifted = corr - m
        cl = np.clip(beta * shifted, -60, 60)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k_idx in range(len(corr)):
            if w[k_idx] > 1e-14:
                g += w[k_idx] * corr_grad_shift(x, k_idx - (n - 1))
        return g * dx_val
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

def random_binary(n, rng):
    """Create a random binary vector with exactly n/2 ones."""
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def thue_morse(n, rng):
    """Thue-Morse based initialization."""
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c == 1 else 0.0
    return binarize(h, n, rng)

def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 320
        parent_h = thue_morse(parent_n, rng)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    # Phase 1: Quick multi-resolution scan (first ~80s)
    resolutions = []
    for delta in [0, -20, -40, 20, 40, 60, -60, 80, -80, 100]:
        nn = parent_n + delta
        if 160 <= nn <= 500 and nn % 2 == 0:
            resolutions.append(nn)
    # Add some absolute values
    for nn in [200, 240, 280, 320, 360, 400, 440]:
        if nn not in resolutions and 160 <= nn <= 500:
            resolutions.append(nn)
    resolutions = sorted(set(resolutions))
    
    quick_results = []
    for n_t in resolutions[:8]:
        if remaining() < 20:
            break
        if n_t == parent_n:
            h_start = best_h.copy()
        else:
            h_start = interp_to(best_h, best_n, n_t)
        h_start = project_feasible(h_start, n_t)
        h_bin = binarize(h_start, n_t, rng)
        
        # Quick local search
        h_ls, c5_ls = local_search_binary(h_bin, n_t, rng, time_limit=min(10, max(3, remaining() - 5)))
        c5_ls = compute_c5(h_ls)
        quick_results.append((n_t, h_ls, c5_ls))
        
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            best_n = n_t
    
    # Phase 2: Intensive SA on top resolutions (~600s)
    top_res = sorted(quick_results, key=lambda x: x[2])[:4]
    
    for n_sa, h_sa, c5_sa in top_res:
        if remaining() < 60:
            break
        # Run SA from multiple starts
        time_per_sa = min(100, max(30, remaining() / 3 - 20))
        
        # SA from the local search result
        h_r, c5_r = simulated_annealing(h_sa, n_sa, rng, time_limit=time_per_sa,
                                         t_init=0.8, t_final=0.03)
        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_sa
        
        # SA from a random start
        if remaining() > time_per_sa + 20:
            h_rand = random_binary(n_sa, rng)
            h_r2, c5_r2 = simulated_annealing(h_rand, n_sa, rng, time_limit=time_per_sa * 0.7,
                                               t_init=1.5, t_final=0.05)
            c5_r2 = compute_c5(h_r2)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2.copy()
                best_n = n_sa
    
    # Phase 3: Iterated local search on best (~200s)
    n_use = best_n
    h_use = best_h.copy()
    h_bin_use = binarize(h_use, n_use, rng)
    
    if remaining() > 50:
        h_ils, c5_ils = iterated_local_search(h_bin_use, n_use, rng,
                                               time_limit=min(150, max(20, remaining() - 30)))
        c5_ils = compute_c5(h_ils)
        if c5_ils < best_c5:
            best_c5 = c5_ils
            best_h = h_ils.copy()
            best_n = n_use
    
    # Also try ILS at a second resolution
    if len(top_res) > 1 and remaining() > 40:
        n2, h2, c52 = top_res[1]
        if n2 != n_use:
            h_bin2 = binarize(h2, n2, rng)
            h_ils2, c5_ils2 = iterated_local_search(h_bin2, n2, rng,
                                                     time_limit=min(80, max(15, remaining() - 20)))
            c5_ils2 = compute_c5(h_ils2)
            if c5_ils2 < best_c5:
                best_c5 = c5_ils2
                best_h = h_ils2.copy()
                best_n = n2
    
    # Phase 4: Final SA with more time on the very best (~150s)
    if remaining() > 50:
        h_best_bin = binarize(best_h, best_n, rng)
        h_fin, c5_fin = simulated_annealing(h_best_bin, best_n, rng,
                                             time_limit=min(120, max(20, remaining() - 30)),
                                             t_init=0.5, t_final=0.02)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = best_n
    
    # Phase 5: Continuous refinement via SLSQP (~100s)
    if remaining() > 40:
        n_pol = best_n
        h_pol = best_h.copy()
        
        for beta in [1500, 5000, 15000]:
            if remaining() < 15:
                break
            h_new = slsqp_continuous(h_pol, n_pol, maxiter=200, beta=beta)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_pol
                h_pol = h_new.copy()
    
    # Phase 6: Final local search on the binarized best
    if remaining() > 20:
        h_final_bin = binarize(best_h, best_n, rng)
        h_fs, c5_fs = local_search_binary(h_final_bin, best_n, rng,
                                           time_limit=min(30, max(5, remaining() - 5)))
        c5_fs = compute_c5(h_fs)
        if c5_fs < best_c5:
            best_c5 = c5_fs
            best_h = h_fs.copy()
            best_n = best_n
    
    # Final continuous polish
    if remaining() > 15:
        h_p2 = slsqp_continuous(best_h, best_n, maxiter=300, beta=10000)
        c5_p2 = compute_c5(h_p2)
        if c5_p2 < best_c5:
            best_c5 = c5_p2
            best_h = h_p2.copy()
    
    # Validate and return
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
