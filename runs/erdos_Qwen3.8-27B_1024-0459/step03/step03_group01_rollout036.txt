import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _START))

def binary_sa(h, n, time_limit, rng, t0=1.0, t1=0.001, cooling='exp'):
    """Simulated annealing on binary sequence with exactly n/2 ones."""
    h = (h > 0.5).astype(float).copy()
    ones_count = int(np.sum(h))
    if ones_count != n // 2:
        # Fix count
        if ones_count > n // 2:
            ones_idx = np.where(h == 1)[0]
            to_remove = rng.choice(ones_idx, ones_count - n // 2, replace=False)
            h[to_remove] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            to_add = rng.choice(zeros_idx, n // 2 - ones_count, replace=False)
            h[to_add] = 1
    
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_corr = corr.copy()
    best_h = h.copy()
    best_max = cur_max
    dx_val = 2.0 / n
    best_c5 = cur_max * dx_val
    
    start = time.time()
    # Precompute about the total number of iterations we can afford
    # Each iteration is roughly O(n) for the delta computation
    # Let's aim for ~1M iterations per 10s for n=320
    n_steps = int(time_limit * 80000)  # rough estimate
    n_steps = min(n_steps, 5000000)
    
    for step in range(n_steps):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 10:
            break
        
        # Temperature
        progress = step / max(1, n_steps)
        T = t0 * (t1 / t0) ** progress
        
        # Choose swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Bias: pick i from positions contributing to max correlation
        k_max = int(np.argmax(corr))
        s_max = k_max - (n - 1)
        
        if s_max > 0:
            # h[i]=1 and h[i+s_max]=0
            candidates = np.array([i for i in ones if i + s_max < n and h[i + s_max] == 0])
        elif s_max < 0:
            s_abs = -s_max
            candidates = np.array([i for i in ones if i - s_abs >= 0 and h[i - s_abs] == 0])
        else:
            candidates = ones
        
        if len(candidates) > 0 and rng.rand() < 0.7:
            i = candidates[rng.randint(len(candidates))]
        else:
            i = ones[rng.randint(len(ones))]
        
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        
        # Compute delta in correlation
        delta = compute_swap_delta(h, i, j, n)
        new_corr = corr + delta
        new_max = float(np.max(new_corr))
        
        # Metropolis acceptance
        delta_max = new_max - cur_max
        if delta_max < 0 or rng.rand() < np.exp(-delta_max / max(T, 1e-10)):
            # Accept
            h[i] = 0
            h[j] = 1
            corr = new_corr
            cur_max = new_max
            
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                best_corr = corr.copy()
                best_c5 = best_max * dx_val
    
    return best_h, best_c5

def compute_swap_delta(h, i, j, n):
    """Compute delta in correlation array when h[i]: 1→0 and h[j]: 0→1."""
    # Using the formula derived:
    # corr at shift s = sum_m h[m] * (1-h[m+s]) for valid m
    # 
    # h[i]: 1→0
    #   As first index (m=i): for shift s where 0≤i+s<n: Δ = -(1-h[i+s])
    #   As second index (m+s=i, so m=i-s): for shift s where 0≤i-s<n: Δ = +h[i-s]
    #
    # h[j]: 0→1
    #   As first index (m=j): for shift s where 0≤j+s<n: Δ = +(1-h[j+s])
    #   As second index (m+s=j, so m=j-s): for shift s where 0≤j-s<n: Δ = -h[j-s]
    
    delta = np.zeros(2*n-1)
    
    # h[i]: 1→0, as first index
    # shift s in [0, n-1-i]: corr[n-1+s] -= (1-h[i+s])
    if n - 1 - i >= 0:
        s_vals = np.arange(0, n - i)
        delta[n-1+s_vals] -= (1.0 - h[i + s_vals])
    
    # h[i]: 1→0, as second index
    # shift s in [-(i), -1] i.e., s = -1, -2, ..., -i: corr[n-1+s] += h[i+s]
    # i+s = i-1, i-2, ..., 0
    if i >= 1:
        s_vals = np.arange(-i, 0)
        delta[n-1+s_vals] += h[i + s_vals]
    
    # h[j]: 0→1, as first index
    # shift s in [0, n-1-j]: corr[n-1+s] += (1-h[j+s])
    if n - 1 - j >= 0:
        s_vals = np.arange(0, n - j)
        delta[n-1+s_vals] += (1.0 - h[j + s_vals])
    
    # h[j]: 0→1, as second index
    # shift s in [-(j), -1]: corr[n-1+s] -= h[j+s]
    if j >= 1:
        s_vals = np.arange(-j, 0)
        delta[n-1+s_vals] -= h[j + s_vals]
    
    return delta
