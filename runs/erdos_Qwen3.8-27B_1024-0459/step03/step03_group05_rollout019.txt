import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(40):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                order = np.argsort(h)[::-1]
            else:
                order = np.argsort(h)
            rem = diff
            for idx in order:
                if rem > 0:
                    room = 1.0 - h[idx]
                    mv = min(room, rem)
                    h[idx] += mv
                    rem -= mv
                else:
                    room = h[idx]
                    mv = min(room, -rem)
                    h[idx] -= mv
                    rem += mv
                if abs(rem) < 1e-15:
                    break
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def topk_value(x, beta, K):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if K < len(corr):
        idx = np.argsort(corr)[-K:]
        vals = corr[idx]
    else:
        vals = corr
    m = vals.max()
    shifted = vals - m
    return (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx


def sum_con_fun(x):
    return np.sum(x) - x[0] - (len(x) / 2.0) + x[0]


def sum_con_fun_wrap(x, target):
    return np.sum(x) - target


def sum_con_jac_wrap(x):
    return np.ones(len(x))
