def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        order = np.argsort(np.abs(h - 0.5))
        for idx in order:
            if h[idx] >= 0.0 and h[idx] <= 1.0:
                new_v = h[idx] + diff
                if 0.0 <= new_v <= 1.0:
                    h[idx] = new_v
                    break
        else:
            h[0] += diff
            h = np.clip(h, 0.0, 1.0)
    # Exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h[np.argmin(np.abs(h - 0.5))] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n

        rev = 0
        for _ in range(20):
            rev = (rev << 1) | (val & 1)
            val >>= 1
        h[i] = 1.0 if (rev & 1) == 0 else 0.0
    return h


def binary_sa(h_init, n, max_iters, rng, t_start, t_end):
    h = np.where(h_init > 0.5, 1.0, 0.0).copy()
    num_ones = int(np.sum(h))
    target_ones = n // 2
    if num_ones > target_ones:
        ones_idx = np.where(h == 1.0)[0]
        remove = rng.choice(len(ones_idx), num_ones - target_ones, replace=False)
        h[ones_idx[remove]] = 0.0
    elif num_ones < target_ones:
        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
        h[zeros_idx[add]] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5

    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])

    cool_rate = (t_end / t_start) ** (1.0 / max(float(max_iters), 1.0))

    for it in range(max_iters):
        if remaining_time() < 5:
            break
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break

        i = rng.randint(len(ones_list))
        j = rng.randint(len(zeros_list))
        oi = ones_list[i]
        oz = zeros_list[j]

        if oi == oz:
            continue

        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = compute_c5(h)

        T = t_start * (cool_rate ** it)

        if new_c5 < current_c5 or (T > 1e-15 and rng.rand() < np.exp(-(new_c5 - current_c5) / T)):
            current_c5 = new_c5
            ones_list[i] = oz
            zeros_list[j] = oi
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0

    return best_h, best_c5


def lgbfsgs_optimize(h0, n, beta, pen_coeff, maxiter, ftol, gtol):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    args = (beta, n, target, pen_coeff)
    try:
        res = minimize(smooth_max_val, h0, jac=smooth_max_grad,
                       method='L-BFGS-B', bounds=bounds,
                       args=args,
                       options={'maxiter': maxiter, 'ftol': ftol,
                                'gtol': gtol, 'iprint': 0,
                                'maxls': 80, 'maxfun': maxiter * 12})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)


def lgbfsgs_topk(h0, n, pen_coeff, top_k, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    args = (n, target, pen_coeff, top_k)
    try:
        res = minimize(topk_obj, h0, jac=topk_grad,
                       method='L-BFGS-B', bounds=bounds,
                       args=args,
                       options={'maxiter': maxiter, 'ftol': 1e-14,
                                'iprint': 0, 'maxls': 80})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)


def optimize_at_resolution(h0, n, rng, time_limit):

        c5_trial = compute_c5(h_trial)
        if c5_trial < best_local_c5:
            best_local_c5 = c5_trial
            best_local = h_trial.copy()

    return best_local, best_local_c5


def coordinate_descent(h, n, rng, sweeps=6):
    target = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for sweep in range(sweeps):
        if remaining_time() < 5:
            break
        order = rng.permutation(n)
        improved = False
        for i in order:
            if remaining_time() < 3:
                break
            old_val = h[i]
            cur_c5 = compute_c5(h)
            cands = [0.0, 1.0, 0.5]
            if 0.0 < old_val < 1.0:
                cands.extend([old_val - 0.15, old_val + 0.15, old_val - 0.05, old_val + 0.05])
            cands = [c for c in cands if 0.0 <= c <= 1.0 and abs(c - old_val) > 1e-8]
            best_val = old_val
            best_h_local = h.copy()
            best_c5_local = cur_c5
            for cv in cands:
                h[i] = cv
                h_feas = project_feasible(h, n)
                c5_test = compute_c5(h_feas)
                if c5_test < best_c5_local:
                    best_c5_local = c5_test
                    best_val = cv
                    best_h_local = h_feas.copy()
            h = best_h_local.copy()
            if best_c5_local < best_c5:
                best_c5 = best_c5_local
                best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.7f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 160
        best_h = project_feasible(make_thue_morse(best_n), best_n)
        best_c5 = compute_c5(best_h)
        print("No parent. Thue-Morse: n={}, C5={:.7f}".format(best_n, best_c5))

    # Phase 1: Optimize at parent resolution
    if remaining_time() > 40:
        print("\n--- Phase 1: Optimize n={} ---".format(best_n))
        h_opt, c5_opt = optimize_at_resolution(best_h, best_n, rng, 60)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            print("  n={}: C5={:.7f}".format(best_n, c5_opt))

        # Try alternative inits at same n
        for init_fn in [make_thue_morse, lambda nn: make_random_binary(nn, rng)]:
            if remaining_time() < 25:
                break
            h_init = project_feasible(init_fn(best_n), best_n)
            h_opt, c5_opt = optimize_at_resolution(h_init, best_n, rng, 30)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = best_n
                print("  Alt init n={}: C5={:.7f}".format(best_n, c5_opt))

    # Phase 2: Higher resolutions
    higher_ns = [240, 320, 420]
    if best_n > 360:
        higher_ns = [best_n + 80, best_n + 160]
    for nn in higher_ns:
        if remaining_time() < 60:
            break
        print("\n--- Phase 2: n={} ---".format(nn))
        h_interp = interp_to(best_h, best_n, nn)
        h_interp = project_feasible(h_interp, nn)
        h_opt, c5_opt = optimize_at_resolution(h_interp, nn, rng, 80)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = nn
            print("  n={}: C5={:.7f}".format(nn, c5_opt))

        # Also try Thue-Morse at this resolution
        if remaining_time() < 35:
            break
        h_tm = project_feasible(make_thue_morse(nn), nn)
        h_opt2, c5_opt2 = optimize_at_resolution(h_tm, nn, rng, 40)
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2.copy()
            best_n = nn
            print("  TM n={}: C5={:.7f}".format(nn, c5_opt2))

    # Phase 3: Binary SA at best resolution (and neighbors)
    if remaining_time() > 80:
        sa_ns = [best_n]
        if best_n + 20 <= 500:
            sa_ns.append(best_n + 20)
        if best_n - 20 >= 80:
            sa_ns.append(best_n - 20)
        sa_ns = list(dict.fromkeys(sa_ns))  # dedupe preserving order

        for nn in sa_ns:
            if remaining_time() < 40:
                break
            print("\n--- Phase 3: Binary SA n={} ---".format(nn))

            # Start from best continuous solution (rounded)
            h_round = np.where(interp_to(best_h, best_n, nn) > 0.5, 1.0, 0.0)
            h_round = project_feasible(h_round, nn)

            # Also try a few random binary inits
            sa_inits = [h_round]
            for _ in range(2):
                sa_inits.append(make_random_binary(nn, rng))

            sa_iters = min(6000, int(remaining_time() * 1.5))
            for sa_h in sa_inits:
                if remaining_time() < 20:
                    break
                h_bin, c5_bin = binary_sa(sa_h, nn, sa_iters, rng, 0.003, 0.00002)
                if c5_bin < best_c5:
                    best_c5 = c5_bin
                    best_h = h_bin.copy()
                    best_n = nn
                    print("  SA n={}: C5={:.7f}".format(nn, c5_bin))

                # Polish binary solution in continuous space
                if remaining_time() > 20 and c5_bin < best_c5 + 0.005:
                    h_pol = lgbfsgs_optimize(h_bin, nn, 300, 200.0, 300, 1e-14, 1e-10)
                    c5_pol = compute_c5(h_pol)
                    if c5_pol < best_c5:
                        best_c5 = c5_pol
                        best_h = h_pol.copy()
                        best_n = nn
                        print("  Polished n={}: C5={:.7f}".format(nn, c5_pol))

    # Phase 4: Coordinate descent refinement
    if remaining_time() > 20:
        print("\n--- Phase 4: Coord descent n={} ---".format(best_n))
        h_cd, c5_cd = coordinate_descent(best_h, best_n, rng, sweeps=5)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            print("  CD: C5={:.7f}".format(c5_cd))

    # Phase 5: Final L-BFGS-B polish at multiple betas
    if remaining_time() > 15:
        print("\n--- Phase 5: Final polish n={} ---".format(best_n))
        for beta in [1500, 3000, 5000]:
            if remaining_time() < 8:
                break
            h_pol = lgbfsgs_optimize(best_h, best_n, beta, 300.0, 300, 1e-15, 1e-11)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                print("  beta={}: C5={:.7f}".format(beta, c5_pol))

    # Phase 6: Try one more resolution jump if time allows
    if remaining_time() > 50:
        extra_n = min(best_n + 100, 560)
        if extra_n > best_n and extra_n % 2 == 0:
            print("\n--- Phase 6: n={} ---".format(extra_n))
            h_ex = project_feasible(interp_to(best_h, best_n, extra_n), extra_n)
            h_ex, c5_ex = optimize_at_resolution(h_ex, extra_n, rng, 50)
            if c5_ex < best_c5:
                best_c5 = c5_ex
                best_h = h_ex.copy()
                best_n = extra_n
                print("  n={}: C5={:.7f}".format(extra_n, c5_ex))

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
