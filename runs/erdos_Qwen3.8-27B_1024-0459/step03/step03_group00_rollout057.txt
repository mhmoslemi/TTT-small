import time

def c5_val(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def c5_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def to_binary(h, n):
    # Threshold to binary maintaining exactly n/2 ones
    h = np.array(h, dtype=float)
    idx = np.argsort(-h)
    result = np.zeros(n)
    result[idx[:n//2]] = 1.0
    return result

def make_tm(n):
    # Thue-Morse
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt else 0.0
    # Ensure exactly n/2 ones
    if not h.sum() == n // 2:
        h = to_binary(h, n)
    return h

def make_golden(n, rng):
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    phase = rng.rand()

    for i in range(n):
        frac = (i * gr + phase) % 1.0

    return to_binary(h, n)

def make_spectral_bin(rng, n):
    """Generate binary sequence from flat spectrum."""
    # Random phasors
    phases = rng.uniform(0, 2*np.pi, n+1)
    amplitudes = rng.uniform(0.5, 1.0, n+1)
    sig = np.zeros(n+1)
    sig[0] = n/4  # mean
    for k in range(1, n):
        sig[k] = amplitudes[k] * np.exp(1j * phases[k])
        sig[n-k] = conj(sig[k])
    h = np.real(np.fft.ifft(sig))[:n]
    return to_binary(h, n)

def hill_climb_binary(h, n, rng, time_limit=30.0):

    h = h.copy()

        n_ones = len(ones)
        n_zeros = len(zeros)
        n_trials = min(n_ones * n_zeros, 5000)

            if time.time() - start > time_limit:
                break
            i_idx = int(rng.randint(n_ones))
            j_idx = int(rng.randint(n_zeros))
            i = int(ones[i_idx])
            j = int(zeros[j_idx])

            h[j] = 1.0
            new_c5 = c5_val(h, n)

            else:

def simulated_annealing(h_init, n, rng, time_limit=60.0):

    h = h_init.copy()
    c5 = c5_val(h, n)
    best_h = h.copy()
    best_c5 = c5
    
    # Scale temperature to C5 differences
    # Typical ΔC5 from one swap ~ 1e-4
    T_start = 5e-4
    T_end = 1e-6
    total = time_limit
    
    while time.time() - start < total:
        elapsed = time.time() - start
        frac = elapsed / total
        T = T_start * (T_end / T_start) ** frac
        
        ones = np.where(h > 0.99)[0]
        zeros = np.where(h < 0.01)[0]
        i = int(ones[int(rng.randint(len(ones)))])
        j = int(zeros[int(rng.randint(len(zeros)))])
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = c5_val(h, n)
        delta = new_c5 - c5
        
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5

def slsqp_smooth(h0, n, beta, maxiter):
    dx = 2.0 / n
    tgt = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        exps = np.exp(np.clip(beta * shifted, -500, 500))
        return (m + np.log(np.sum(exps)) / beta) * dx

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        exps = np.exp(np.clip(beta * shifted, -500, 500))
        sum_exps = np.sum(exps)
        grad_corr = np.zeros_like(corr)
        max_idx = np.argmax(corr)
        grad_corr[max_idx] = 1.0
        if sum_exps > 0:
            grad_corr += beta * exps / sum_exps
        grad_corr *= dx

        grad = np.zeros(n)
        n_corr = len(corr)
        for k_idx in range(n_corr):
            gk = grad_corr[k_idx]
            if abs(gk) < 1e-18:
                continue
            offset = k_idx - (n - 1)
            if offset >= 0:
                n_pairs = n - offset
                grad[offset:offset + n_pairs] += gk * (1.0 - x[offset:offset + n_pairs])
                grad[:n_pairs] -= gk * x[offset:offset + n_pairs]
            else:
                abs_off = -offset
                n_pairs = n - abs_off
                grad[:n_pairs] += gk * (1.0 - x[abs_off:abs_off + n_pairs])
                grad[abs_off:abs_off + n_pairs] -= gk * x[:n_pairs]
        return grad

    def con_fun(x):
        return x.sum() - tgt

    def con_grad(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_grad}]
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)

        h_parent = make_random_bin(rng, n)

    best_h = h_parent.copy()
    best_c5 = c5_val(best_h, n)
    print("Parent: n={}, C5={:.8f}".format(n, best_c5))

    # Phase 1: Diverse binary initializations + hill climb
    print("\n=== Phase 1: Multi-init Hill Climb ===")
    inits = []
    inits.append(('parent_bin', to_binary(h_parent, n)))
    inits.append(('thue_morse', make_thue_morse(n)))
    inits.append(('golden', make_golden(n, rng)))
    for k in range(6):
        inits.append(('rand_{}'.format(k), make_random_bin(rng, n)))
    for k in range(4):
        inits.append(('spectral_{}'.format(k), make_spectral_bin(rng, n)))
    for k in range(3):
        inits.append(('vander_{}'.format(k), make_vandermonde(rng, n)))

    t1_start = time.time()
    t1_total = min(150.0, time_left() * 0.15)
    per_init = max(8.0, t1_total / len(inits))

    for name, h0 in inits:
        if time_left() < per_init + 5:
            break
        h1, c5_1 = hill_climb_binary(h0, n, rng, per_init)
        if c5_1 < best_c5 - 1e-14:
            best_c5 = c5_1
            best_h = h1
            print("  {} C5={:.8f} ***".format(name, c5_1))
        else:
            print("  {} C5={:.8f}".format(name, c5_1))

    # Ensure best is binary
    best_h = to_binary(best_h, n)
    best_c5 = c5_val(best_h, n)
    print("Phase 1 best: C5={:.8f}".format(best_c5))

    # Phase 2: Extended simulated annealing
    print("\n=== Phase 2: Simulated Annealing ===")
    sa_time = min(200.0, time_left() * 0.25)
    if sa_time > 40:
        h_sa, c5_sa = simulated_annealing(best_h, n, rng, sa_time)
        # Hill climb after SA
        h_sa2, c5_sa2 = hill_climb_binary(h_sa, n, rng, min(60.0, time_left() * 0.1))
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2
        print("SA: {:.8f} -> HC: {:.8f}".format(c5_sa, c5_sa2))

    # Phase 3: Iterated local search
    print("\n=== Phase 3: Iterated Local Search ===")
    ils_budget = min(350.0, time_left() * 0.35)
    ils_start = time.time()
    n_ils_iter = 0
    while time.time() - ils_start < ils_budget and time_left() > 60:
        n_flips = int(rng.randint(3, max(4, 3 + n_ils_iter)))
        h_pert = best_h.copy()
        ones = np.where(h_pert > 0.99)[0]
        zeros = np.where(h_pert < 0.01)[0]
        for _ in range(n_flips):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = int(ones[int(rng.randint(len(ones)))])
            j = int(zeros[int(rng.randint(len(zeros)))])
            h_pert[i] = 0.0
            h_pert[j] = 1.0
            ones = np.where(h_pert > 0.99)[0]
            zeros = np.where(h_pert < 0.01)[0]

        hc_time = min(50.0, ils_budget - (time.time() - ils_start), time_left() - 30)
        if hc_time < 5:
            break
        h_is, c5_is = hill_climb_binary(h_pert, n, rng, hc_time)
        if c5_is < best_c5 - 1e-14:
            best_c5 = c5_is
            best_h = h_is
            n_ils_iter = 0
            print("  ILS[{}] improved: C5={:.8f}".format(n_ils_iter, c5_is))
        else:
            n_ils_iter += 1
    print("ILS done: C5={:.8f}".format(best_c5))

    # Phase 4: Best-improvement sweeps
    print("\n=== Phase 4: Best-Improvement Sweeps ===")
    bi_time = min(120.0, time_left() * 0.15)
    if bi_time > 20:
        best_h, best_c5 = best_improvement_sweep(best_h, n, rng, bi_time)
        print("After best-improvement: C5={:.8f}".format(best_c5))

    # Phase 5: Another round of SA + HC from best
    if time_left() > 80:
        print("\n=== Phase 5: Second SA + HC ===")
        sa2_time = min(100.0, time_left() * 0.25)
        h_sa3, c5_sa3 = simulated_annealing(best_h, n, rng, sa2_time)
        h_sa4, c5_sa4 = hill_climb_binary(h_sa3, n, rng, min(50.0, time_left() * 0.1))
        if c5_sa4 < best_c5:
            best_c5 = c5_sa4
            best_h = h_sa4
        print("SA2: {:.8f} -> HC: {:.8f}".format(c5_sa3, c5_sa4))

    # Phase 6: Continuous micro-polish
    if time_left() > 30:
        print("\n=== Phase 6: Continuous Polish ===")
        best_h, best_c5 = continuous_micro_polish(best_h, n, rng, min(60.0, time_left() * 0.3))
        print("After continuous: C5={:.8f}".format(best_c5))

    # Phase 7: SLSQP polish on continuous
    if time_left() > 30:
        print("\n=== Phase 7: SLSQP Polish ===")
        for beta in [200, 500, 1000]:
            if time_left() < 15:
                break
            h_new = slsqp_smooth(best_h, n, beta, 200)
            c5_new = c5_val(h_new, n)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new
                print("  beta={}: C5={:.8f}".format(beta, c5_new))
            else:
                print("  beta={}: no improve ({:.8f})".format(beta, c5_new))

    # Phase 8: Try different resolution
    if time_left() > 60:
        print("\n=== Phase 8: Resolution Search ===")
        for n_alt in [128, 512, 384]:
            if time_left() < 30:
                break
            h_alt = project_feasible(interp_to(best_h, n, n_alt), n_alt)
            # Quick hill climb on binary version
            h_alt_bin = to_binary(h_alt, n_alt)
            h_alt_opt, c5_alt = hill_climb_binary(h_alt_bin, n_alt, rng, min(30.0, time_left() * 0.3))
            # Continuous polish
            h_alt_opt, c5_alt2 = continuous_micro_polish(h_alt_opt, n_alt, rng, min(15.0, time_left() * 0.2))
            if c5_alt2 < best_c5 - 1e-14:
                best_c5 = c5_alt2
                best_h = h_alt_opt
                n = n_alt
                print("  n={}: C5={:.8f} ***".format(n_alt, c5_alt2))
            else:
                print("  n={}: C5={:.8f}".format(n_alt, c5_alt2))

    # Final validation
    best_h = project_feasible(best_h, n)
    final_c5 = c5_val(best_h, n)
    print("\nFINAL: n={}, C5={:.8f}".format(n, final_c5))

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, final_c5, n)
