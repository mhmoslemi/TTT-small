            break
        unsat = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(unsat)
        if cnt > 0:
            h[unsat] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-12:
            idx = np.argmin(np.abs(h - 0.5))
            h[idx] += diff2
            h = np.clip(h, 0.0, 1.0)
    return h


def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = float(bin(i).count('1') % 2)
    return h


def make_shifted_tm(n, shift):
    h = np.zeros(n)
    for i in range(n):
        h[i] = float(bin((i + shift) % n).count('1') % 2)
    return h


def make_sine_pattern(n, rng):
    t = np.linspace(0, 2, n, endpoint=False)
    h = 0.5
    for f in [0.5, 1.5, 3.0, 5.0, 7.0]:
        h += rng.uniform(0.05, 0.2) * np.sin(2 * np.pi * f * t + rng.rand() * 6.28)
    return np.clip(h, 0.0, 1.0)


def make_random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def binary_sa(h_init, n, max_iters, rng, t_start=0.003, t_end=0.00002):
    h = np.where(h_init > 0.5, 1.0, 0.0)
    num_ones = int(np.sum(h))
    target = n // 2
    if num_ones > target:
        oi = np.where(h == 1.0)[0]
        rem = rng.choice(len(oi), num_ones - target, replace=False)
        h[oi[rem]] = 0.0
    elif num_ones < target:
        zi = np.where(h == 0.0)[0]
        add = rng.choice(len(zi), target - num_ones, replace=False)
        h[zi[add]] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])

    for it in range(max_iters):
        if remaining_time() < 3:
            break
        if not ones_list or not zeros_list:
            break
        i = int(rng.randint(len(ones_list)))
        j = int(rng.randint(len(zeros_list)))
        oi = ones_list[i]
        oz = zeros_list[j]

        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = compute_c5(h)

        T = t_start * (t_end / t_start) ** (float(it) / max(float(max_iters), 1.0))
        if new_c5 < cur_c5 or (T > 1e-16 and rng.rand() < np.exp(-(new_c5 - cur_c5) / T)):
            cur_c5 = new_c5
            ones_list[i] = oz
            zeros_list[j] = oi
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0
    return best_h, best_c5


def focus_gradient_step(h, n, top_k=5, lr_scale=0.02):
    """Take a gradient step focused on the top-k worst lags."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    max_c = np.max(corr[top_idx])
    weights = np.zeros(len(corr))
    for j in top_idx:
        weights[j] = np.exp(200.0 * (corr[j] - max_c))
    wsum = np.sum(weights[top_idx])
    if wsum < 1e-30:
        return None
    for j in top_idx:
        weights[j] /= wsum

    grad = np.zeros(n)
    for j in top_idx:
        w = weights[j]
        if w < 1e-10:
            continue
        s = j - (n - 1)
        if s >= 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        else:
            sp = -s
            grad[:n - sp] += w * (1.0 - h[sp:])
            grad[sp:] -= w * h[:n - sp]
    grad *= dx
    return grad


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.7f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 200
        best_h = project_feasible(make_thue_morse(best_n), best_n)
        best_c5 = compute_c5(best_h)
        print("No parent: n={}, C5={:.7f}".format(best_n, best_c5))

    # Phase 1: Multi-resolution continuous optimization
    n_list = sorted(set([best_n, 192, 240, 288, 320, 400]))
    n_list = [n for n in n_list if 120 <= n <= 500]

    for n in n_list:
        if remaining_time() < 35:
            break
        print("\n--- Cont opt n={} ---".format(n))
        bounds = [(0.0, 1.0)] * n

        inits = []
        inits.append(project_feasible(interp_to(best_h, best_n, n), n))
        for sh in [0, 1, 3, 7]:
            inits.append(project_feasible(make_shifted_tm(n, sh), n))
        for _ in range(3):
            inits.append(project_feasible(make_sine_pattern(n, rng), n))
        for _ in range(2):
            inits.append(make_random_binary(n, rng))

        for h0 in inits:
            if remaining_time() < 10:
                break
            c0 = compute_c5(h0)
            if c0 > best_c5 * 1.2:
                continue

            for beta in [40, 120, 400, 900]:
                if remaining_time() < 7:
                    break
                try:
                    res = minimize(topk_obj, h0, jac=topk_grad,
                                   args=(15, beta),
                                   method='L-BFGS-B', bounds=bounds,
                                   options={'maxiter': 400, 'ftol': 1e-16,
                                            'gtol': 1e-12, 'iprint': 0,
                                            'maxls': 40, 'maxfev': 5000})
                    h_new = project_feasible(res.x, n)
                    c_new = compute_c5(h_new)
                    if c_new < c0:
                        h0 = h_new
                        c0 = c_new
                except Exception:
                    pass

            if remaining_time() > 6:
                try:
                    res = minimize(topk_obj, h0, jac=topk_grad,
                                   args=(30, 2000),
                                   method='L-BFGS-B', bounds=bounds,
                                   options={'maxiter': 600, 'ftol': 1e-16,
                                            'gtol': 1e-13, 'iprint': 0,
                                            'maxfev': 8000})
                    h_new = project_feasible(res.x, n)
                    c_new = compute_c5(h_new)
                    if c_new < c0:
                        h0 = h_new
                        c0 = c_new
                except Exception:
                    pass

            if c0 < best_c5:
                best_c5 = c0
                best_h = h0.copy()
                best_n = n
                print("  Improved: n={}, C5={:.7f}".format(n, c0))

    # Phase 2: Binary SA + continuous polish
    if remaining_time() > 80:
        n = best_n
        print("\n--- Binary SA n={} ---".format(n))
        bounds = [(0.0, 1.0)] * n

        starts = [np.where(best_h > 0.5, 1.0, 0.0)]
        for sh in [1, 2, 5]:
            starts.append(project_feasible(make_shifted_tm(n, sh), n))
        for _ in range(3):
            starts.append(make_random_binary(n, rng))

        for sa_h in starts:
            if remaining_time() < 25:
                break
            h_bin, c_bin = binary_sa(sa_h, n, 10000, rng)
            if c_bin < best_c5:
                best_c5 = c_bin
                best_h = h_bin.copy()
                best_n = n
                print("  SA: C5={:.7f}".format(c_bin))

            if remaining_time() > 12:
                for beta in [300, 1000]:
                    if remaining_time() < 7:
                        break
                    try:
                        res = minimize(topk_obj, h_bin, jac=topk_grad,
                                       args=(15, beta),
                                       method='L-BFGS-B', bounds=bounds,
                                       options={'maxiter': 500, 'ftol': 1e-16,
                                                'gtol': 1e-12, 'iprint': 0})
                        h_new = project_feasible(res.x, n)
                        c_new = compute_c5(h_new)
                        if c_new < best_c5:
                            best_c5 = c_new
                            best_h = h_new.copy()
                            best_n = n
                            print("  Polished: C5={:.7f}".format(c_new))
                    except Exception:
                        pass

    # Phase 3: Focus-gradient iterative refinement
    if remaining_time() > 40:
        n = best_n
        h = best_h.copy()
        print("\n--- Focus grad n={} ---".format(n))

        for it in range(2000):
            if remaining_time() < 5:
                break
            grad = focus_gradient_step(h, n, top_k=8)
            if grad is None:
                break
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break

            improved = False
            for lr in [0.03, 0.06, 0.01, 0.1, 0.005, 0.15]:
                h_trial = h - lr * grad
                h_trial = project_feasible(h_trial, n)
                c_trial = compute_c5(h_trial)
                if c_trial < best_c5 - 1e-14:
                    best_c5 = c_trial
                    best_h = h_trial.copy()
                    h = h_trial.copy()
                    improved = True
                    break
            if not improved:
                if remaining_time() < 15:
                    break
                # L-BFGS restart
                bounds = [(0.0, 1.0)] * n
                try:
                    res = minimize(topk_obj, h, jac=topk_grad,
                                   args=(15, 600),
                                   method='L-BFGS-B', bounds=bounds,
                                   options={'maxiter': 300, 'ftol': 1e-16,
                                            'iprint': 0})
                    h_new = project_feasible(res.x, n)
                    c_new = compute_c5(h_new)
                    if c_new < best_c5:
                        best_c5 = c_new
                        best_h = h_new.copy()
                        h = h_new.copy()
                    else:
                        break
                except Exception:
                    break

    # Phase 4: Coordinate descent (thorough)
    if remaining_time() > 25:
        n = best_n
        h = best_h.copy()
        print("\n--- Coord descent n={} ---".format(n))

        for sweep in range(50):
            if remaining_time() < 4:
                break
            any_imp = False
            order = rng.permutation(n)
            for i in order:
                if remaining_time() < 2:
                    break
                old = h[i]
                c_old = compute_c5(h)
                best_val = old
                best_c = c_old
                cands = [0.0, 1.0, 0.5,
                         max(0.0, old - 0.02), min(1.0, old + 0.02),
                         max(0.0, old - 0.08), min(1.0, old + 0.08),
                         max(0.0, old - 0.15), min(1.0, old + 0.15)]
                seen = set()
                for cv in cands:
                    cv_r = round(cv, 6)
                    if cv_r in seen:
                        continue
                    seen.add(cv_r)
                    if abs(cv - old) < 1e-9:
                        continue
                    h[i] = cv
                    h_feas = project_feasible(h, n)
                    c_t = compute_c5(h_feas)
                    if c_t < best_c - 1e-14:
                        best_c = c_t
                        best_val = cv
                if best_val != old:
                    h[i] = best_val
                    h = project_feasible(h, n)
                    any_imp = True
                    if best_c < best_c5:
                        best_c5 = best_c
                        best_h = h.copy()
            if not any_imp:
                break

    # Phase 5: Final ultra-polish
    if remaining_time() > 10:
        n = best_n
        bounds = [(0.0, 1.0)] * n
        print("\n--- Final polish n={} ---".format(n))
        for beta in [1000, 2000, 4000]:
            if remaining_time() < 5:
                break
            try:
                res = minimize(topk_obj, best_h, jac=topk_grad,
                               args=(20, beta),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 800, 'ftol': 1e-17,
                                        'gtol': 1e-14, 'iprint': 0,
                                        'maxfev': 10000})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new)
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_new.copy()
                    print("  Polish beta={}: C5={:.7f}".format(beta, c_new))
            except Exception:
                pass

    # Final binary round-trip check
    if remaining_time() > 8:
        n = best_n
        h_bin = np.where(best_h > 0.5, 1.0, 0.0)
        h_bin = project_feasible(h_bin, n)
        c_bin = compute_c5(h_bin)
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(topk_obj, h_bin, jac=topk_grad,
                           args=(30, 2000),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 500, 'ftol': 1e-16, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
        except Exception:
            pass

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
