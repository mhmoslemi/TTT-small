            cnt = int(np.count_nonzero(mask))

def compute_corr(h):

def corr_value(h, dx):

def smooth_fixed(vals, beta):
    m = float(np.max(vals))
    if beta <= 1e-8:
        return m
    a = beta * (vals - m)
    a = np.clip(a, -700.0, 0.0)
    s = float(np.sum(np.exp(a)))
    if s <= 1e-300:
        return m
    return m + math.log(s) / beta

def project_box_sum(y, target):
    y = np.asarray(y, dtype=np.float64)
    n = len(y)
    if n == 0:
        return y.copy()

    h = np.clip(y, 0.0, 1.0).copy()

def line_search(h, grad, target, dx, beta, c_current, cmax_current, smooth_current):
    n = len(h)
    rms = float(np.sqrt(np.mean(grad * grad)))
    if rms < 1e-13:
        return h, c_current, smooth_current, False

    d = grad / rms
    lrs = (0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125,
           0.00390625, 0.001953125, 0.0009765625)

    for lr in lrs:
        y = h - lr * d
        hp = project_box_sum(y, target)
        if np.linalg.norm(hp - h) < 1e-13:
            continue

        c = compute_corr(hp)
        c5 = float(np.max(c)) * dx
        sm = smooth_max(c, beta)

        if c5 < cmax_current - 1e-12:
            return hp, c, sm, True
        if c5 <= cmax_current + 1e-9 and sm < smooth_current - 1e-10:
            return hp, c, sm, True

    return h, c_current, smooth_current, False

def make_thue_morse(n):
    h = np.zeros(n, dtype=np.float64)
    for i in range(n):
        x = i
        cnt = 0
        while x:
            cnt ^= (x & 1)
            x >>= 1
        if cnt:
            h[i] = 1.0
    return h


def make_golden(n):
    h = np.zeros(n, dtype=np.float64)
    gr = (1.0 + math.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * gr) % 1.0
        if abs(frac - 0.5) < 0.25:
            h[i] = 1.0
    return h

def lp_obj(x):

            except Exception:
                pass

        if remaining(deadline) < 12.0:
            break

    return best_h, best_c


def gradient_pair_moves(h, n, deadline):
    target = 0.5 * float(n)
    dx = 2.0 / float(n)
    best_h = np.array(h, dtype=np.float64).copy()
    best_c = corr_value(best_h, dx)

    if n <= 200:
        steps = 140
        cand_limit = 96
        pos_top = 8
    elif n <= 320:
        steps = 80
        cand_limit = 48
        pos_top = 6
    else:
        steps = 50
        cand_limit = 24
        pos_top = 5

    no_imp = 0

    for _ in range(steps):
        if remaining(deadline) < 3.0:
            break
        if no_imp >= 7:
            break

        h = best_h.copy()
        c0 = compute_corr(h)
        c0max = float(np.max(c0)) * dx

        if c0max < best_c - 1e-13:
            best_c = c0max
            best_h = h.copy()

        grad, _ = active_gradient(h, 150.0, 20, n)
        direction = -grad
        direction = direction - float(np.mean(direction))

        if float(np.sqrt(np.mean(direction * direction))) < 1e-13:
            break

        ordv = np.argsort(direction)
        pos = ordv[-pos_top:][::-1]
        neg = ordv[:pos_top]

        improved = False
        evals = 0
        stop = False

        for i in neg:
            if stop:
                break
            hi = float(h[i])
            if hi <= 1e-12:
                continue
            for j in pos:
                if stop:
                    break
                if i == j:
                    continue
                hj = float(h[j])
                if hj >= 1.0 - 1e-12:
                    continue
                maxd = min(hi, 1.0 - hj)
                for f in (0.5, 0.25, 0.125, 0.0625):
                    if evals >= cand_limit:
                        stop = True
                        break
                    amt = f * maxd
                    if amt < 1e-9:
                        continue

                    tr = h.copy()
                    tr[i] -= amt
                    tr[j] += amt
                    tr = project_box_sum(tr, target)

                    cr = compute_corr(tr)
                    crc = float(np.max(cr)) * dx
                    evals += 1

                    if crc < best_c - 1e-13:
                        best_c = crc
                        best_h = tr.copy()
                        improved = True

        if improved:
            no_imp = 0
        else:
            no_imp += 1

    return best_h, best_c


def random_small_moves(h, n, deadline):
    target = 0.5 * float(n)
    dx = 2.0 / float(n)
    best_h = np.array(h, dtype=np.float64).copy()
    best_c = corr_value(best_h, dx)
    h = best_h.copy()

    rng = np.random.RandomState(20240528)
    deltas = np.array([0.025, 0.0125, 0.00625, 0.003125, 0.0015625], dtype=np.float64)

    if n <= 220:
        max_it = 40000
    elif n <= 340:
        max_it = 20000
    else:
        max_it = 10000

    for _ in range(max_it):
        if remaining(deadline) < 2.0:
            break

        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue

        d = float(rng.choice(deltas))

        if h[i] >= d and h[j] <= 1.0 - d:
            tr = h.copy()
            tr[i] -= d
            tr[j] += d
            tr = project_box_sum(tr, target)
            cr = compute_corr(tr)
            crc = float(np.max(cr)) * dx
            if crc < best_c - 1e-13:
                best_c = crc
                best_h = tr.copy()
                h = tr

        if h[j] >= d and h[i] <= 1.0 - d:
            tr = h.copy()
            tr[j] -= d
            tr[i] += d
            tr = project_box_sum(tr, target)
            cr = compute_corr(tr)
            crc = float(np.max(cr)) * dx
            if crc < best_c - 1e-13:
                best_c = crc
                best_h = tr.copy()
                h = tr

    return best_h, best_c


def mix_and_eval(h1, h2, n):
    h = 0.5 * (np.asarray(h1, dtype=np.float64) + np.asarray(h2, dtype=np.float64))
    h = project_box_sum(h, 0.5 * float(n))
    return h, corr_value(h, 2.0 / float(n))


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget = float(budget_s)

    parent_h = None
    try:
        arr = initial_h_values
        if arr is not None:
            parent_h = np.array(arr, dtype=np.float64).copy()
    except Exception:
        parent_h = None

    if parent_h is None:
        best_n = 160
        best_h, best_c = choose_initial_no_parent(best_n, int(seed))
    else:
        best_n = int(len(parent_h))
        best_h = project_box_sum(parent_h, 0.5 * float(best_n)).copy()
        best_c = corr_value(best_h, 2.0 / float(best_n))

        if best_n % 2 == 0:
            hthr = binarize_nearest(best_h, best_n)
            cthr = corr_value(hthr, 2.0 / float(best_n))
            if cthr < best_c - 1e-13:
                best_c = cthr
                best_h = hthr.copy()

    reserve = min(240.0, max(20.0, 0.16 * budget))
    work = max(0.0, budget - reserve)

    cset = set()
    for dn in range(-30, 51, 3):
        nt = best_n + dn
        if 64 <= nt <= 700:
            cset.add(nt)
    for nt in (128, 160, 192, 224, 256, 320, 384, 448, 512):
        if 64 <= nt <= 700:
            cset.add(nt)

    n_list = sorted(cset)
    scored = []
    for nt in n_list:
        ht = interp_h(best_h, best_n, nt)
        ht = project_box_sum(ht, 0.5 * float(nt))
        ct = corr_value(ht, 2.0 / float(nt))
        scored.append((ct, nt, ht))

    vals = []
    for item in scored:
        vals.append(item[0])
    order = np.argsort(np.array(vals, dtype=np.float64))
    scored_order = []
    for i in order:
        scored_order.append(scored[int(i)])

    phase_ns = []
    phase_ns.append(best_n)
    for item in scored_order[:12]:
        nt = int(item[1])
        if nt not in phase_ns and len(phase_ns) < 4:
            phase_ns.append(nt)

    if len(phase_ns) == 1:
        fracs = [1.0]
    elif len(phase_ns) == 2:
        fracs = [0.60, 0.40]
    elif len(phase_ns) == 3:
        fracs = [0.52, 0.32, 0.16]
    else:
        fracs = [0.45, 0.30, 0.18, 0.07]

    s = 0.0
    for item in fracs:
        s += item
    fracs = [item / s for item in fracs]

    cum = 0.0
    for idx, nt in enumerate(phase_ns):
        seg_end = t0 + work * cum
        cum += fracs[idx]

        if remaining(seg_end) < 20.0:
            break

        if nt == best_n:
            h_init = best_h.copy()
        else:
            h_init = interp_h(best_h, best_n, nt)
            h_init = project_box_sum(h_init, 0.5 * float(nt))

        hh, cc = optimize_active(h_init, nt, seg_end)
        if cc < best_c - 1e-13:
            best_c = cc
            best_h = hh.copy()
            best_n = nt

    base_deadline = t0 + budget - 2.0

    if remaining(base_deadline) > 20.0:
        fset = set()
        for dn in range(-24, 25, 2):
            nt = best_n + dn
            if 48 <= nt <= 800:
                fset.add(nt)
        for nt in (128, 160, 192, 224, 256, 320, 384, 448, 512):
            if 48 <= nt <= 800:
                fset.add(int(nt))

        for nt in sorted(fset):
            ht = interp_h(best_h, best_n, nt)
            ht = project_box_sum(ht, 0.5 * float(nt))
            ct = corr_value(ht, 2.0 / float(nt))
            if ct < best_c - 1e-13:
                best_c = ct
                best_h = ht.copy()
                best_n = nt

        if best_n % 2 == 0:
            ht = binarize_nearest(best_h, best_n)
            ct = corr_value(ht, 2.0 / float(best_n))
            if ct < best_c - 1e-13:
                best_c = ct
                best_h = ht.copy()

        if parent_h is not None and remaining(base_deadline) > 5.0:
            hp = interp_h(parent_h, int(len(parent_h)), best_n)
            hp = project_box_sum(hp, 0.5 * float(best_n))
            hm, cm = mix_and_eval(best_h, hp, best_n)
            if cm < best_c - 1e-13:
                best_c = cm
                best_h = hm.copy()

            if best_n % 2 == 0:
                htp = binarize_nearest(hp, best_n)
                hm, cm = mix_and_eval(best_h, htp, best_n)
                if cm < best_c - 1e-13:
                    best_c = cm
                    best_h = hm.copy()

    if remaining(base_deadline) > 80.0 and best_n <= 256:
        chunk = min(140.0, max(20.0, remaining(base_deadline) / 3.0))
        t1 = time.time() + chunk
        best_h, best_c = polish_slsqp(best_h, best_n, t1)

    if remaining(base_deadline) > 50.0 and best_n <= 256:
        chunk = min(90.0, max(12.0, remaining(base_deadline) / 3.0))
        t2 = time.time() + chunk
        best_h, best_c = polish_lbfgsb(best_h, best_n, t2)

    pair_deadline = t0 + budget - 15.0
    if remaining(pair_deadline) > 12.0:
        best_h, best_c = gradient_pair_moves(best_h, best_n, pair_deadline)

    rand_deadline = t0 + budget - 4.0
    if remaining(rand_deadline) > 6.0:
        best_h, best_c = random_small_moves(best_h, best_n, rand_deadline)

    best_h = project_box_sum(best_h, 0.5 * float(best_n))
    final_c = corr_value(best_h, 2.0 / float(best_n))
    if final_c < best_c:
        best_c = final_c

    try:
        ec = evaluate_erdos_solution(best_h)
        ec = float(ec)
        if ec < best_c:
            best_c = ec
    except Exception:
        pass

    print("FINAL: n={}, C5={:.10f}".format(best_n, best_c))
    return (best_h, float(best_c), int(best_n))
