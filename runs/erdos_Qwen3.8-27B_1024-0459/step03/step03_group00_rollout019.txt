    try:
        res = minimize(obj_smooth, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter,
                       'ftol': ftol, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def do_slsqp_raw(h0, n, maxiter=300):
    dx = 2.0 / n
    def obj_raw(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return float(np.max(corr)) * dx
    def con_f(x):
        return np.sum(x) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_f}]
    try:
        res = minimize(obj_raw, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter,
                       'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def coordinate_descent(h0, n, budget_fn, rng, max_sweeps=20):
    h = h0.copy()
    c5_best = compute_c5(h)
    for sweep in range(max_sweeps):
        if budget_fn() < 3.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if budget_fn() < 1.5:
                break
            old = h[i]
            for v in [0.0, 1.0]:
                if abs(v - old) < 1e-8:
                    continue
                h[i] = v
                h = project_feasible(h, n)
                c = compute_c5(h)
                if c < c5_best - 1e-12:
                    c5_best = c
                    h0 = h.copy()
                    any_imp = True
                else:
                    h = h0.copy()
            for d in [0.03, -0.03, 0.07, -0.07, 0.15, -0.15]:
                if budget_fn() < 1.5:
                    break
                trial = h0.copy()
                trial[i] = np.clip(trial[i] + d, 0.0, 1.0)
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5_best - 1e-12:
                    c5_best = c
                    h0 = trial.copy()
                    any_imp = True
        if not any_imp:
            break
    return h0, c5_best

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xq = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xq, xp, src)

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    total_budget = float(budget_s)

    def remaining():
        return max(0.0, total_budget - (time.time() - t_start) - 5.0)

    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n_init = 256
        best_h = np.zeros(n_init)
        best_h[:n_init // 2] = 1.0
        best_h = make_thue_morse(n_init)

    best_n = len(best_h)
    best_c5 = compute_c5(best_h)
    print("Initial: n={}, C5={:.8f}".format(best_n, best_c5))

    # Phase 1: Quick SLSQP warm-up
    if remaining() > 40:
        print("Phase 1: SLSQP warm-up")
        for beta in [50, 300]:
            if remaining() < 15:
                break
            h_new = do_slsqp(best_h, best_n, beta, maxiter=200, ftol=1e-13)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-10:
                best_h = h_new
                best_c5 = c5_new
                print("  SLSQP beta={}: C5={:.8f}".format(beta, best_c5))

    # Phase 2: LP-based refinement (core algorithm)
    print("Phase 2: LP refinement at n={}".format(best_n))
    n = best_n
    lp_iter = 0
    no_improve = 0

    while remaining() > 120:
        lp_iter += 1
        delta = lp_step(best_h, n)

        if delta is None or np.max(np.abs(delta)) < 1e-10:
            no_improve += 1
            if no_improve > 4:
                if remaining() > 60:
                    h_new = do_slsqp(best_h, n, 500, maxiter=200, ftol=1e-14)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5 - 1e-11:
                        best_h = h_new
                        best_c5 = c5_new
                        no_improve = 0
                        print("  SLSQP escape iter {}: C5={:.8f}".format(lp_iter, best_c5))
                    else:
                        break
                else:
                    break
            continue

        h_new, c5_new, alpha = line_search(best_h, delta, n)

        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new
            no_improve = 0
            if lp_iter % 20 == 0:
                print("  LP iter {}: C5={:.8f} (alpha={:.2f})".format(lp_iter, best_c5, alpha))
        else:
            no_improve += 1
            if no_improve > 8:
                break

        if lp_iter % 30 == 0 and remaining() > 60:
            best_h, best_c5 = coordinate_descent(best_h, n, remaining, rng, max_sweeps=5)
            print("  CD at iter {}: C5={:.8f}".format(lp_iter, best_c5))

    print("  LP phase done: {} iters, C5={:.8f}".format(lp_iter, best_c5))

    # Phase 3: Coordinate descent
    if remaining() > 40:
        print("Phase 3: Coordinate descent")
        best_h, best_c5 = coordinate_descent(best_h, best_n, remaining, rng, max_sweeps=15)
        print("  After CD: C5={:.8f}".format(best_c5))

    # Phase 4: More LP refinement
    if remaining() > 60:
        print("Phase 4: More LP")
        n2 = len(best_h)
        for i in range(100):
            if remaining() < 20:
                break
            delta = lp_step(best_h, n2)
            if delta is None or np.max(np.abs(delta)) < 1e-10:
                if i > 5:
                    break
                continue
            h_new, c5_new, alpha = line_search(best_h, delta, n2)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new
            elif i > 10:
                break
        print("  After LP2: C5={:.8f}".format(best_c5))

    # Phase 5: Try alternative resolutions
    if remaining() > 80:
        print("Phase 5: Try other resolutions")
        for n_try in [192, 288, 320, 160]:
            if remaining() < 30:
                break
            h_try = project_feasible(interp_to(best_h, len(best_h), n_try), n_try)
            c5_try = compute_c5(h_try)
            h_try = do_slsqp(h_try, n_try, 200, maxiter=200, ftol=1e-13)
            c5_try = compute_c5(h_try)
            for _ in range(15):
                if remaining() < 15:
                    break
                d = lp_step(h_try, n_try)
                if d is None or np.max(np.abs(d)) < 1e-10:
                    break
                h_new, c5_new, _ = line_search(h_try, d, n_try)
                if c5_new < c5_try - 1e-12:
                    h_try = h_new
                    c5_try = c5_new
                else:
                    break
            if c5_try < best_c5 - 1e-10:
                best_h = h_try
                best_c5 = c5_try
                print("  n={} gives C5={:.8f}".format(n_try, best_c5))

    # Phase 6: Final polish
    if remaining() > 20:
        print("Phase 6: Final polish")
        n_final = len(best_h)
        best_h = do_slsqp(best_h, n_final, 500, maxiter=300, ftol=1e-14)
        c5_chk = compute_c5(best_h)
        if c5_chk < best_c5:
            best_c5 = c5_chk
        best_h = do_slsqp_raw(best_h, n_final, maxiter=200)
        c5_chk = compute_c5(best_h)
        if c5_chk < best_c5:
            best_c5 = c5_chk

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_n = len(best_h)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
