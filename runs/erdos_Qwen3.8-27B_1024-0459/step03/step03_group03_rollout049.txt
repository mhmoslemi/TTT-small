                break
    return best, bc


def _binary_swap(h, n, nsweeps, rng, tol_bin=0.998):
    best = h.copy()
    bc = _c5(best)
    for sweep in range(nsweeps):
        if _tl() < 5:
            break
        ones = np.where(best > tol_bin)[0]
        zeros = np.where(best < 1.0 - tol_bin)[0]
        if len(ones) == 0 or len(zeros) == 0:
            # Try with looser threshold
            ones = np.where(best > 0.7)[0]
            zeros = np.where(best < 0.3)[0]
            if len(ones) < 2 or len(zeros) < 2:
                break
        improved = False
        attempts = 0
        max_attempts = 200
        while attempts < max_attempts:
            if _tl() < 2:
                break
            attempts += 1
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h2 = best.copy()
            tmp = h2[i]
            h2[i] = h2[j]
            h2[j] = tmp
            h2 = _proj(h2, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-14:
                best = h2
                bc = c2
                improved = True
                ones = np.where(best > tol_bin)[0]
                zeros = np.where(best < 1.0 - tol_bin)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    ones = np.where(best > 0.7)[0]
                    zeros = np.where(best < 0.3)[0]
        if not improved:
            # Try exhaustive for small sets
            if len(ones) <= 30 and len(zeros) <= 30:
                found = False
                for i in ones:
                    if _tl() < 2:
                        break
                    for j in zeros:
                        h2 = best.copy()
                        tmp = h2[i]
                        h2[i] = h2[j]
                        h2[j] = tmp
                        h2 = _proj(h2, n)
                        c2 = _c5(h2)
                        if c2 < bc - 1e-14:
                            best = h2
                            bc = c2
                            found = True
                            ones = np.where(best > tol_bin)[0]
                            zeros = np.where(best < 1.0 - tol_bin)[0]
                            break
                    if found:
                        break
                if not found:
                    break
            else:
                break
    return best, bc


def _coord_descent(h, n, nit, rng):
    best = h.copy()
    bc = _c5(best)
    for it in range(nit):
        if _tl() < 3:
            break
        nb = np.where((best > 0.005) & (best < 0.995))[0]
        if len(nb) < 2:
            break
        improved = False
        for idx in rng.permutation(nb):
            if _tl() < 2:
                break
            old = best[idx]
            others = np.setdiff1d(nb, idx)
            if len(others) == 0:
                continue
            j = others[rng.randint(len(others))]
            for delta in [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.03, -0.03, 0.008, -0.008]:
                nv = old + delta
                if nv < 0.0 or nv > 1.0:
                    continue
                h2 = best.copy()
                h2[idx] = nv
                h2[j] -= delta
                h2 = _proj(h2, n)
                c2 = _c5(h2)
                if c2 < bc - 1e-14:
                    best = h2
                    bc = c2
                    improved = True
                    nb = np.where((best > 0.005) & (best < 0.995))[0]
                    break
        if not improved:
            break
    return best, bc


def _sim_anneal(h, n, nit, rng, t_start=0.03, t_end=0.002):
    best = h.copy()
    bc = _c5(best)
    cur = h.copy()
    cc = bc
    t_span = nit
    imp_streak = 0
    for it in range(nit):
        if _tl() < 3:
            break
        t = t_start * (t_end / t_start) ** (it / max(t_span, 1))
        # Pick a random pair to swap or a random continuous adjustment
        r = rng.random()
        if r < 0.6:
            # Swap
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            h2 = cur.copy()
            tmp = h2[i]
            h2[i] = h2[j]
            h2[j] = tmp
            h2 = _proj(h2, n)
        else:
            # Random continuous perturbation
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.1, 0.1)
            h2 = cur.copy()
            h2[i] += d
            h2[j] -= d
            h2 = _proj(h2, n)
        c2 = _c5(h2)
        delta = c2 - cc
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            cur = h2
            cc = c2
            if cc < bc - 1e-14:
                best = cur.copy()
                bc = cc
                imp_streak = 0
            else:
                imp_streak += 1
    return best, bc


def _full_optimize(h0, n, rng, time_limit=None):
    h = _proj(h0.copy(), n)
    c5 = _c5(h)

    # Phase 1: SLSQP with increasing beta
    for beta in [80, 300, 1000]:
        if _tl() < 20:
            break
        h2 = _slsqp(h, n, beta, 300)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2

    # Phase 2: Exact subgradient descent
    if _tl() > 30:
        h, c5 = _subgrad_descent(h, n, 400, rng, alpha0=0.04)

    # Phase 3: Smoothed gradient descent
    if _tl() > 25:
        h, c5 = _smgrad_descent(h, n, 300, rng, beta=40.0)

    # Phase 4: Binary swap
    if _tl() > 15:
        h, c5 = _binary_swap(h, n, 80, rng)

    # Phase 5: More gradient
    if _tl() > 25:
        h, c5 = _subgrad_descent(h, n, 300, rng, alpha0=0.02)
        h, c5 = _smgrad_descent(h, n, 400, rng, beta=100.0)

    # Phase 6: Coordinate descent
    if _tl() > 15:
        h, c5 = _coord_descent(h, n, 150, rng)

    # Phase 7: High-beta SLSQP polish
    if _tl() > 15:
        h2 = _slsqp(h, n, 3000, 400)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2

    # Phase 8: Final gradient + swap
    if _tl() > 20:
        h, c5 = _smgrad_descent(h, n, 500, rng, beta=200.0)
        h, c5 = _binary_swap(h, n, 60, rng)
        h, c5 = _subgrad_descent(h, n, 300, rng, alpha0=0.01)

    # Phase 9: Ultra-fine polish
    if _tl() > 15:
        h2 = _slsqp(h, n, 8000, 500)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2
        h, c5 = _smgrad_descent(h, n, 400, rng, beta=500.0)
        h, c5 = _coord_descent(h, n, 100, rng)

    # Phase 10: Final swap + gradient
    if _tl() > 10:
        h, c5 = _binary_swap(h, n, 40, rng)
        h, c5 = _smgrad_descent(h, n, 300, rng, beta=1000.0)

    return h, c5


def _multi_start(h0, n, rng, n_starts=3):
    best_h = h0.copy()
    best_c = _c5(h0)
    for s in range(n_starts):
        if _tl() < 40:
            break
        # Different perturbations
        p = rng.randn(n) * 0.02
        h_init = _proj(h0 + p, n)
        h, c = _full_optimize(h_init, n, rng)
        if c < best_c:
            best_h, best_c = h, c
    return best_h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    global _t0, _tb
    _t0 = time.time()
    _tb = budget_s
    rng = np.random.RandomState(seed)

    try:
        hpar = np.array(initial_h_values, dtype=np.float64)
        npar = len(hpar)
    except Exception:
        npar = 160
        hpar = np.zeros(npar, dtype=np.float64)
        hpar[:npar // 2] = 1.0

    # Project parent to satisfy constraints exactly
    bh = _proj(hpar, npar)
    bc5 = _c5(bh)
    bn = npar
    print("Init n={} C5={:.10f}".format(bn, bc5))

    # Primary optimization at parent resolution
    if _tl() > 60:
        h, c5 = _full_optimize(bh, npar, rng)
        if c5 < bc5:
            bh, bc5, bn = h, c5, npar
            print("Opt1 n={} C5={:.10f}".format(bn, bc5))

    # Try multi-start at parent resolution
    if _tl() > 80:
        h, c5 = _multi_start(bh, bn, rng, n_starts=2)
        if c5 < bc5:
            bh, bc5 = h, c5
            print("Multi n={} C5={:.10f}".format(bn, bc5))

    # Try higher resolutions
    for nn in [bn + 4, bn + 8, bn + 16, bn + 24]:
        if _tl() < 45:
            break
        if nn < 32 or nn > 300:
            continue
        # Check that sum constraint works: sum(h) = nn/2, need nn even
        if nn % 2 != 0:
            continue
        hi = _proj(_interp_f(bh, bn, nn), nn)
        h2, c2 = _full_optimize(hi, nn, rng)
        if c2 < bc5:
            bh, bc5, bn = h2, c2, nn
            print("Ups n={} C5={:.10f}".format(bn, c2))
        else:
            # Still do a quick polish at this resolution in case it helps later
            pass

    # Try lower resolution (simpler structure might be better)
    for nn in [bn - 4, bn - 8, 160, 120]:
        if _tl() < 40:
            break
        if nn < 32 or nn % 2 != 0:
            continue
        hi = _proj(_interp_f(bh, bn, nn), nn)
        h2, c2 = _full_optimize(hi, nn, rng)
        if c2 < bc5:
            bh, bc5, bn = h2, c2, nn
            print("Down n={} C5={:.10f}".format(bn, c2))

    # Final intensive polish at best resolution
    if _tl() > 60:
        h, c5 = _full_optimize(bh, bn, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
            print("Polish n={} C5={:.10f}".format(bn, c5))

    # Extra: simulated annealing for escape
    if _tl() > 30:
        h, c5 = _sim_anneal(bh, bn, 300, rng, t_start=0.02, t_end=0.001)
        if c5 < bc5:
            bh, bc5 = h, c5
            print("SA n={} C5={:.10f}".format(bn, c5))

    # Final micro-polish
    if _tl() > 10:
        h, c5 = _smgrad_descent(bh, bn, 500, rng, beta=2000.0)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _binary_swap(bh, bn, 30, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _subgrad_descent(bh, bn, 200, rng, alpha0=0.005)
        if c5 < bc5:
            bh, bc5 = h, c5
        print("Final polish n={} C5={:.10f}".format(bn, bc5))

    # Ensure constraints are met exactly
    bh = _proj(bh, bn)
    fc5 = _c5(bh)

    # Try evaluate_erdos_solution if available
    try:
        ec = evaluate_erdos_solution(bh)
        if ec < fc5:
            fc5 = ec
    except Exception:
        pass

    print("FINAL n={} C5={:.10f}".format(bn, fc5))
    return (bh, fc5, bn)
