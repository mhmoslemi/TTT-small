        npar = len(hpar)
    except Exception:
        npar = 160
        hpar = np.zeros(npar, dtype=np.float64)
        hpar[:npar // 2] = 1.0

    bh = _proj(hpar, npar)
    bc5 = _c5(bh)
    bn = npar
    print("Init n={} C5={:.8f}".format(bn, bc5))

    # Phase 1: Fourier parametric optimization to find good structure
    for nh in [3, 5, 7, 10]:
        if _tl() < 40:
            break
        hf, cf = _fourier_opt(npar, nh, rng, maxiter=1500)
        if hf is not None and cf < bc5:
            bh, bc5 = hf.copy(), cf
            print("Fourier M={} n={} C5={:.8f}".format(nh, npar, cf))

    # Phase 2: Main optimization pipeline at parent resolution
    h, c5 = _full_pipeline(bh, npar, rng)
    if c5 < bc5:
        bh, bc5 = h, c5
        print("Pipeline n={} C5={:.8f}".format(npar, bc5))

    # Phase 3: Multi-resolution search
    resolutions = sorted(set([npar - 4, npar - 2, npar + 2, npar + 4,
                               npar + 6, npar + 8, npar + 10, npar + 12,
                               npar + 16, npar - 6, npar - 8]))
    for nn in resolutions:
        if _tl() < 50:
            break
        if nn < 40 or nn % 2 != 0:
            continue
        hi = _interp(bh, bn, nn)
        hi = _proj(hi, nn)
        # Quick optimize
        h2 = _slsqp(hi, nn, 200, 200)
        c2 = _c5(h2)
        h2 = _grad_descend(h2, nn, 200, rng)
        c2 = _c5(h2)
        if c2 < bc5:
            # Full pipeline on this resolution
            h3, c3 = _full_pipeline(h2, nn, rng)
            if c3 < bc5:
                bh, bc5, bn = h3, c3, nn
                print("Res n={} C5={:.8f}".format(nn, c3))

    # Phase 4: More Fourier at best resolution
    for nh in [8, 12, 15, 20]:
        if _tl() < 30:
            break
        hf, cf = _fourier_opt(bn, nh, rng, maxiter=1000)
        if hf is not None and cf < bc5:
            # Refine from Fourier solution
            h2 = _slsqp(hf, bn, 500, 200)
            c2 = _c5(h2)
            h2, c2 = _grad_descend(h2, bn, 200, rng)
            if c2 < bc5:
                bh, bc5 = h2, c2
                print("Fourier-final M={} n={} C5={:.8f}".format(nh, bn, c2))

    # Phase 5: Final intensive polishing
    if _tl() > 60:
        h, c5 = _perturb_optimize(bh, bn, rng, 15, 0.02)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _grad_descend(bh, bn, 400, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _cont_local(bh, bn, rng, 300)
        if c5 < bc5:
            bh, bc5 = h, c5

    if _tl() > 30:
        for beta in [10000, 30000, 50000]:
            if _tl() < 15:
                break
            h2 = _slsqp(bh, bn, beta, 500)
            c2 = _c5(h2)
            if c2 < bc5:
                bh, bc5 = h2, c2
            h, c5 = _grad_descend(bh, bn, 200, rng)
            if c5 < bc5:
                bh, bc5 = h, c5

    if _tl() > 20:
        h, c5 = _cont_local(bh, bn, rng, 300)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _grad_descend(bh, bn, 200, rng)
        if c5 < bc5:
            bh, bc5 = h, c5

    # Final projection and evaluation
    bh = _proj(bh, bn)
    fc5 = _c5(bh)
    try:
        ec = evaluate_erdos_solution(bh)
        if ec < fc5:
            fc5 = ec
    except Exception:
        pass
    print("FINAL n={} C5={:.8f}".format(bn, fc5))
    return (bh, fc5, bn)
