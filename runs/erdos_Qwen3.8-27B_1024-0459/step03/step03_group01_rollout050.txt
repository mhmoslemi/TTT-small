import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-14:
            g += w[k] * corr_grad_per_shift(x, n, k)
    return g * dx_val


def sum_con(x):
    return np.sum(x) - _TGT


def sum_con_jac(x):
    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=400):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def subgradient_momentum(h, n, time_limit, beta_start=50, beta_end=5000):
    """Custom subgradient method with Nesterov momentum for the smoothed objective."""
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    h = h.copy()
    dx_val = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    v = 0.9
    
    start = time.time()
    n_iters = 0
    total_iters = 5000
    
    for it in range(total_iters):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break
        # anneal beta
        t = it / total_iters
        beta = beta_start * (beta_end / beta_start) ** t
        _BETA = beta
        
        corr = compute_corr(h)
        m = np.max(corr)
        c5_curr = m * dx_val
        
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h.copy()
        
        # Compute subgradient (for the smoothed version)
        shifted = corr - m
        cl = np.clip(beta * shifted, -60, 60)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] > 1e-12:
                g += w[k] * corr_grad_per_shift(h, n, k)
        g *= dx_val
        
        # Nesterov momentum
        momentum = v * momentum - g / (np.linalg.norm(g) + 1e-12)
        h_new = h + momentum
        
        # Project
        h_new = project_feasible(h_new, n)
        
        # Line search
        best_step = 0.0
        for step in [0.01, 0.005, 0.02, 0.002, 0.05]:
            h_try = h + step * momentum
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < c5_curr:
                best_step = step
                break
        
        if best_step > 0:
            h = project_feasible(h + best_step * momentum, n)
        else:
            # No improvement, decay momentum
            momentum *= 0.5
    
    return best_h, best_c5


def simulated_annealing_binary(h, n, time_limit, rng):
    """Simulated annealing on binary representations for the Erdos problem."""
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin))
    target_ones = n // 2
    
    if n_ones > target_ones:
        ones_idx = np.where(h_bin == 1)[0]
        remove = rng.choice(ones_idx, n_ones - target_ones, replace=False)
        h_bin[remove] = 0.0
    elif n_ones < target_ones:
        zeros_idx = np.where(h_bin == 0)[0]
        add = rng.choice(zeros_idx, target_ones - n_ones, replace=False)
        h_bin[add] = 1.0
    
    dx_val = 2.0 / n
    corr = compute_corr(h_bin)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx_val
    best_h = h_bin.copy()
    
    start = time.time()
    T_start = 0.5  # initial "temperature" in correlation units
    T_end = 0.001
    total_iters = 20000
    
    for it in range(total_iters):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break
        
        t = it / total_iters
        T = T_start * (T_end / T_start) ** t
        
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Choose a random pair swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_corr = compute_corr(h_bin)
        new_max = float(np.max(new_corr))
        delta = new_max - cur_max
        
        if delta < 0 or rng.rand() < np.exp(-delta / (T * dx_val + 1e-12)):
            # Accept
            cur_max = new_max
            corr = new_corr
            if cur_max * dx_val < best_c5:
                best_c5 = cur_max * dx_val
                best_h = h_bin.copy()
        else:
            # Reject
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    
    return best_h, best_c5


def directed_pair_search(h, n, time_limit, rng):

            if offset >= 0:

                if len(candidates) > 0 and rng.rand() < 0.8:
                    i = candidates[rng.randint(len(candidates))]
                else:
                    i = ones[rng.randint(len(ones))]
            elif offset < 0:
                ao = -offset
                candidates = np.array([i for i in ones if i + ao < n and h_bin[i + ao] == 0])
                if len(candidates) > 0 and rng.rand() < 0.8:
                    i = candidates[rng.randint(len(candidates))]
                else:
                    i = ones[rng.randint(len(ones))]
            else:

                cur_max = new_max
                if cur_max * dx_val < best_c5:
                    best_c5 = cur_max * dx_val
                    best_h = h_bin.copy()
                found = True
                break
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
        
        if not found:
            stall += 1
            if stall > max_stall:
                # Perturbation: random 2-swap
                i1 = ones[rng.randint(len(ones))]
                i2 = ones[rng.randint(len(ones))]
                if i1 == i2:
                    continue
                j1 = zeros[rng.randint(len(zeros))]
                j2 = zeros[rng.randint(len(zeros))]
                if j1 == j2:
                    continue
                h_bin[i1] = 0.0
                h_bin[i2] = 0.0
                h_bin[j1] = 1.0
                h_bin[j2] = 1.0
                corr = compute_corr(h_bin)
                cur_max = float(np.max(corr))
                ones = np.where(h_bin == 1)[0]
                zeros = np.where(h_bin == 0)[0]
                stall = 0
    
    return best_h, best_c5


def top_k_equalize(h, n, max_iters=500, k=3):
    """Try to equalize top-k correlations using projected gradient."""
    h = h.copy()
    dx_val = 2.0 / n
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        topk = np.argsort(-corr)[:k]
        max_c = corr[topk[0]]
        
        # Check if top-k are approximately equal
        spread = corr[topk[0]] - corr[topk[-1]]
        if spread < 1e-8:
            break
        
        # Gradient direction: reduce max, increase min among top-k
        g_max = corr_grad_per_shift(h, n, int(topk[0]))
        g_min = corr_grad_per_shift(h, n, int(topk[-1]))
        d = g_min - g_max  # reduces max, increases min
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-14:
            break
        d /= dnorm
        
        improved = False
        for step in [0.005, 0.01, 0.02, 0.05, 0.001, 0.03]:
            h_trial = project_feasible(h + step * d, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-12:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            break
    
    return h, best_c5

            best_h = h_fin.copy()

        h_ds_fin, c5_ds_fin = directed_pair_search(h, n,
                                                   time_limit=min(20, max(5, remaining() - 8)), rng=rng)
        if c5_ds_fin < best_c5 - 1e-13:
            best_c5 = c5_ds_fin
            best_h = h_ds_fin.copy()

    # Final SLSQP
    if remaining() > 6:
        h_fin2 = slsqp_smooth(best_h, best_n, 10000, maxiter=300)
        c5_fin2 = compute_c5(h_fin2)
        if c5_fin2 < best_c5 - 1e-13:
            best_c5 = c5_fin2
            best_h = h_fin2.copy()

    # Validate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
