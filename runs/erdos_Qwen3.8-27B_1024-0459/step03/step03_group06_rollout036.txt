def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        unsat = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.sum(unsat)
        if cnt > 0:
            h[unsat] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute among multiple points
        order = np.argsort(np.abs(h - 0.5))
        for idx in order:
            room_up = 1.0 - h[idx]
            room_dn = h[idx]
            if diff > 0 and room_up > 0:
                adj = min(diff, room_up)
                h[idx] += adj
                diff -= adj
            elif diff < 0 and room_dn > 0:
                adj = max(diff, -room_dn)
                h[idx] += adj
                diff -= adj
            if abs(diff) < 1e-14:
                break
    return h

def projected_gradient_step(h, top_k=10, n_steps=200, rng=None):

        tv = corr[top_idx]

        wv /= np.sum(wv)

            wj = wv[idx_i]

                    grad[m_idx] += wj * (-h[m_idx + s])

                    grad[m_idx] += wj * (1.0 - h[m_idx + sp])
                for m_idx in range(sp, n):
                    grad[m_idx] += wj * (-h[m_idx - sp])

        # Add constraint gradient
        grad += 0.5 * (np.sum(h) - n / 2.0)

        gn = np.linalg.norm(grad)
        if gn < 1e-15:

            h_cur = h0.copy()
            c5_cur = c5_0
            
            # Multi-beta L-BFGS-B
            for beta_val in [100, 300, 800]:
                if remaining_time() < 5:
                    break
                _BETA = beta_val
                try:
                    res = minimize(smooth_max_obj, h_cur, jac=smooth_max_grad,
                                   method='L-BFGS-B', bounds=bounds,
                                   options={'maxiter': 300, 'ftol': 1e-15,
                                            'gtol': 1e-10, 'iprint': 0,
                                            'maxls': 40, 'maxfun': 3000})
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur:
                        h_cur = h_new
                        c5_cur = c5_new
                except Exception:
                    pass
            
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n
                print("  L-BFGS: C5={:.7f}".format(best_c5))
    
    # Phase 2: Active-set projected gradient at best resolution
    if remaining_time() > 40:
        n = best_n
        _N = n
        _TGT = n / 2.0
        print("\n=== Active-set PG: n={} ===".format(n))
        
        h_cur = best_h.copy()
        for top_k in [5, 8, 12]:
            if remaining_time() < 15:
                break
            h_cur, c5_cur = projected_gradient_step(h_cur, top_k=top_k, n_steps=30, rng=rng)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                print("  PG(k={}): C5={:.7f}".format(top_k, best_c5))
        
        # L-BFGS-B with active set objective
        bounds = [(0.0, 1.0)] * n
        for top_k in [8, 15]:
            if remaining_time() < 10:
                break
            try:
                res = minimize(lambda x: active_set_obj(x, top_k), best_h,
                               jac=lambda x: active_set_grad(x, top_k),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 200, 'ftol': 1e-15,
                                        'gtol': 1e-10, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  AS-LBFGS(k={}): C5={:.7f}".format(top_k, best_c5))
            except Exception:
                pass
    
    # Phase 3: Try higher resolution with active-set optimization
    for n_high in [320, 480]:
        if remaining_time() < 60:
            break
        if n_high <= best_n:
            continue
        _N = n_high
        _TGT = n_high / 2.0
        print("\n=== High-res n={} ===".format(n_high))
        
        h_interp = project_feasible(interp_to(best_h, best_n, n_high), n_high)
        bounds = [(0.0, 1.0)] * n_high
        
        # L-BFGS-B
        for beta_val in [200, 500]:
            if remaining_time() < 15:
                break
            _BETA = beta_val
            try:
                res = minimize(smooth_max_obj, h_interp, jac=smooth_max_grad,
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 200, 'ftol': 1e-15,
                                        'gtol': 1e-10, 'iprint': 0, 'maxfun': 2000})
                h_new = project_feasible(res.x, n_high)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n_high
                    h_interp = h_new.copy()
                    print("  HiRes LBFGS: C5={:.7f}".format(best_c5))
            except Exception:
                pass
        
        # Active-set PG
        if remaining_time() > 20:
            h_interp, c5_pg = projected_gradient_step(h_interp, top_k=10, n_steps=30, rng=rng)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_interp.copy()
                best_n = n_high
                print("  HiRes PG: C5={:.7f}".format(best_c5))
    
    # Phase 4: Binary local search
    if remaining_time() > 30:
        n = best_n
        print("\n=== Binary LS: n={} ===".format(n))
        
        # Start from thresholded best
        h_bin = np.where(best_h > 0.5, 1.0, 0.0)
        h_bin, c5_bin = binary_local_search(h_bin, n, max_iters=3000, rng=rng)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            print("  BLS: C5={:.7f}".format(best_c5))
        
        # Continue with SA from best binary
        if remaining_time() > 20:
            h_bin2, c5_bin2 = cosine_anneal_sa(best_h, n, n_iters=3000,
                                                 t_start=0.002, t_end=0.00001, rng=rng)
            if c5_bin2 < best_c5:
                best_c5 = c5_bin2
                best_h = h_bin2.copy()
                print("  SA: C5={:.7f}".format(best_c5))
        
        # Polish binary result with continuous
        if remaining_time() > 10:
            _N = n
            _TGT = n / 2.0
            bounds = [(0.0, 1.0)] * n
            for beta_val in [300, 1000]:
                if remaining_time() < 5:
                    break
                _BETA = beta_val
                try:
                    res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                                   method='L-BFGS-B', bounds=bounds,
                                   options={'maxiter': 200, 'ftol': 1e-15,
                                            'gtol': 1e-10, 'iprint': 0})
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        print("  Binary+Polish: C5={:.7f}".format(best_c5))
                except Exception:
                    pass
    
    # Phase 5: Coordinate descent refinement
    if remaining_time() > 15:
        n = best_n
        print("\n=== Coord Descent: n={} ===".format(n))
        h = best_h.copy()
        
        for sweep in range(8):
            if remaining_time() < 5:
                break
            any_imp = False
            order = rng.permutation(n)
            for i in order:
                if remaining_time() < 2:
                    break
                old = h[i]
                best_val = old
                best_c = compute_c5(h)
                
                # Try a few candidate values
                cands = [0.0, 1.0, 0.5]
                if old > 0.1:
                    cands.append(old - 0.15)
                if old < 0.9:
                    cands.append(old + 0.15)
                if old > 0.2:
                    cands.append(0.1)
                if old < 0.8:
                    cands.append(0.9)
                
                for nv in cands:
                    nv = np.clip(nv, 0.0, 1.0)
                    if abs(nv - old) < 1e-8:
                        continue
                    h[i] = nv
                    h_feas = project_feasible(h, n)
                    c_test = compute_c5(h_feas)
                    if c_test < best_c:
                        best_c = c_test
                        best_val = nv
                        h[:] = h_feas
                
                if best_c < best_c5:
                    best_c5 = best_c
                    best_h = h.copy()
                    any_imp = True
            
            if not any_imp:
                break
    
    # Phase 6: Final tight L-BFGS-B
    if remaining_time() > 6:
        n = best_n
        _N = n
        _TGT = n / 2.0
        bounds = [(0.0, 1.0)] * n
        for beta_val in [1000, 2000]:
            if remaining_time() < 4:
                break
            _BETA = beta_val
            try:
                res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 400, 'ftol': 1e-16,
                                        'gtol': 1e-11, 'iprint': 0, 'maxfun': 4000})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  Final polish: C5={:.7f}".format(best_c5))
            except Exception:
                pass
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
