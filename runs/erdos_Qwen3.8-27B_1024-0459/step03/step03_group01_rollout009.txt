                    i = bad_ones[rng.randint(len(bad_ones))]
                else:
                    i = ones[rng.randint(len(ones))]
            else:
                i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

            h[i] = 0.0
            h[j] = 1.0
            new_corr = compute_corr(h)
            new_max = int(np.max(new_corr))
            delta = new_max - cur_max

            if delta < 0 or (delta == 0 and rng.random() < 0.3) or \
               (delta > 0 and T > 0.005 and rng.random() < np.exp(-delta / T)):
                if delta <= 0:
                    corr = new_corr
                    cur_max = new_max
                else:
                    corr = new_corr
                    cur_max = cur_max  # revert
                    h[i] = 1.0
                    h[j] = 0.0
                    # Actually accept if delta > 0 but SA says yes
                    h[i] = 0.0
                    h[j] = 1.0
                    corr = new_corr
                    cur_max = new_max
                if cur_max * dx_val < best_c5:
                    best_c5 = cur_max * dx_val
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0

        elif r < 0.80:
            # Double swap
            if len(ones) < 2 or len(zeros) < 2:
                continue
            i1, i2 = rng.choice(ones, 2, replace=False)
            j1, j2 = rng.choice(zeros, 2, replace=False)
            h[i1] = 0.0; h[i2] = 0.0
            h[j1] = 1.0; h[j2] = 1.0
            new_corr = compute_corr(h)
            new_max = int(np.max(new_corr))
            delta = new_max - cur_max
            if delta < 0 or (delta == 0 and rng.random() < 0.4) or \
               (delta > 0 and T > 0.005 and rng.random() < np.exp(-delta / T)):
                corr = new_corr
                cur_max = new_max
                if cur_max * dx_val < best_c5:
                    best_c5 = cur_max * dx_val
                    best_h = h.copy()
            else:
                h[i1] = 1.0; h[i2] = 1.0
                h[j1] = 0.0; h[j2] = 0.0
        else:
            # Triple swap
            if len(ones) < 3 or len(zeros) < 3:
                continue
            io = rng.choice(ones, 3, replace=False)
            iz = rng.choice(zeros, 3, replace=False)
            h[io] = 0.0
            h[iz] = 1.0
            new_corr = compute_corr(h)
            new_max = int(np.max(new_corr))
            delta = new_max - cur_max
            if delta < 0 or (delta == 0 and rng.random() < 0.5) or \
               (delta > 0 and T > 0.005 and rng.random() < np.exp(-delta / T)):
                corr = new_corr
                cur_max = new_max
                if cur_max * dx_val < best_c5:
                    best_c5 = cur_max * dx_val
                    best_h = h.copy()
            else:
                h[io] = 1.0
                h[iz] = 0.0

    return best_h, best_c5


def greedy_steepest(h_init, n, max_rounds=50, time_limit=30):
    """Try all single swaps each round, take the best improving one."""
    dx_val = 2.0 / n
    h = project_binary(h_init, n, np.random.RandomState(0)).copy()
    corr = compute_corr(h)
    cur_max = int(np.max(corr))
    best_c5 = cur_max * dx_val
    best_h = h.copy()
    start_time = time.time()

    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit or remaining() < 5:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_delta = 0
        best_move = None

        # Subsample for large n to keep runtime manageable
        if len(ones) * len(zeros) > 100000:
            sample_o = np.random.choice(ones, 400, replace=False)
            sample_z = np.random.choice(zeros, 400, replace=False)
        else:
            sample_o = ones
            sample_z = zeros

        for i in sample_o:
            h[i] = 0.0
            for j in sample_z:
                h[j] = 1.0
                nc = compute_corr(h)
                nm = int(np.max(nc))
                delta = nm - cur_max
                if delta < best_delta:
                    best_delta = delta
                    best_move = (i, j)
                h[j] = 0.0
            h[i] = 1.0

        if best_move is None:
            break
        i, j = best_move
        h[i] = 0.0
        h[j] = 1.0
        corr = compute_corr(h)
        cur_max = int(np.max(corr))
        if cur_max * dx_val < best_c5:
            best_c5 = cur_max * dx_val
            best_h = h.copy()

    return best_h, best_c5


def slsqp_polish(h_init, n, beta, maxiter=300):
    """Smooth SLSQP refinement."""
    global _Npol, _BTapol
    _Npol = n
    _BTapol = beta
    dx_val = 2.0 / n

    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -50, 50)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_val

    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -50, 50)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] > 1e-12:
                off = k - (n - 1)
                if off >= 0:
                    npairs = n - off
                    g[off:off + npairs] += (1.0 - x[:npairs])
                    g[:npairs] -= x[off:off + npairs]
                else:
                    ao = -off
                    npairs = n - ao
                    g[:npairs] += (1.0 - x[ao:ao + npairs])
                    g[ao:ao + npairs] -= x[:npairs]
        return g * dx_val

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h_init, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h_init, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 320
        parent_h = make_thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Generate diverse starting points
    starts = {}
    # Parent
    starts[('parent', best_n)] = best_h.copy()

    # Thue-Morse at various n
    for n_tm in [256, 320, 384, 512]:
        if n_tm % 2 == 0:
            tm = make_thue_morse(n_tm)
            c5_tm = compute_c5(tm)
            starts[('tm', n_tm)] = tm
            if c5_tm < best_c5:
                best_c5 = c5_tm
                best_h = tm.copy()
                best_n = n_tm

    # Random balanced
    for trial in range(4):
        n_r = best_n
        rb = make_balanced_random(n_r, rng)
        starts[('rand%d' % trial, n_r)] = rb

    # Phase 1: Main SA search on best n
    n_main = best_n
    h_main = best_h.copy() if best_n == n_main else starts.get(('tm', n_main), best_h.copy())

    # Run SA for the main n
    sa_time = min(700, remaining() * 0.5)
    if remaining() > 30:
        h_sa, c5_sa = sa_multi_point(h_main, n_main, sa_time, rng, t0=3.0, t1=0.01)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n_main

    # Greedy polish on the SA result
    if remaining() > 20:
        h_g, c5_g = greedy_steepest(best_h, best_n, max_rounds=30,
                                     time_limit=min(40, remaining() * 0.15))
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 2: Try other n values
    n_list = sorted(set([best_n - 64, best_n - 32, best_n, best_n + 32, best_n + 64,
                         best_n + 96, best_n + 128, 256, 384, 448, 512]))
    n_list = [nn for nn in n_list if 128 <= nn <= 800 and nn % 2 == 0 and nn != best_n]

    for n_t in n_list:
        if remaining() < 30:
            break

        # Interpolate best solution
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin = project_binary(h_interp, n_t, rng)
        c5_init = compute_c5(h_bin)

        h_new = h_bin.copy()
        # Also try Thue-Morse if n_t is power of 2
        if n_t & (n_t - 1) == 0:
            tm = make_thue_morse(n_t)
            if compute_c5(tm) < c5_init:
                h_new = tm.copy()

        # SA on this n
        sa_t = min(80, remaining() * 0.25)
        if sa_t > 5:
            h_sa2, c5_sa2 = sa_multi_point(h_new, n_t, sa_t, rng, t0=2.0, t1=0.02)
            if c5_sa2 < c5_init:
                h_new = h_sa2
                c5_init = c5_sa2

            # Quick greedy
            if remaining() > 15:
                h_g2, c5_g2 = greedy_steepest(h_new, n_t, max_rounds=15,
                                               time_limit=min(20, remaining() * 0.1))
                if c5_g2 < c5_init:
                    h_new = h_g2
                    c5_init = c5_g2

        if c5_init < best_c5:
            best_c5 = c5_init
            best_h = h_new.copy()
            best_n = n_t

    # Phase 3: Intensive refinement on best
    n = best_n
    h = best_h.copy()

    # More SA rounds
    if remaining() > 100:
        h_sa3, c5_sa3 = sa_multi_point(h, n, min(150, remaining() * 0.3), rng, t0=1.5, t1=0.005)
        if c5_sa3 < best_c5:
            best_c5 = c5_sa3
            best_h = h_sa3.copy()
            h = h_sa3.copy()

    # Greedy steepest descent (try all swaps)
    if remaining() > 20:
        h_g3, c5_g3 = greedy_steepest(best_h, best_n, max_rounds=40,
                                       time_limit=min(60, remaining() * 0.3))
        if c5_g3 < best_c5:
            best_c5 = c5_g3
            best_h = h_g3.copy()

    # Multi-restart SA with perturbations
    if remaining() > 50:
        for restart in range(3):
            if remaining() < 20:
                break
            h_pert = best_h.copy()
            # Perturb: flip 5-15 random pairs
            n_flip = rng.randint(3, 8)
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            fo = rng.choice(ones, n_flip, replace=False)
            fz = rng.choice(zeros, n_flip, replace=False)
            h_pert[fo] = 0.0
            h_pert[fz] = 1.0

            h_r, c5_r = sa_multi_point(h_pert, best_n, min(40, remaining() * 0.3),
                                        rng, t0=2.0, t1=0.01)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 4: Continuous SLSQP polish
    if remaining() > 30:
        h_cont = best_h.copy()
        for beta in [500, 2000, 8000]:
            if remaining() < 15:
                break
            h_pol = slsqp_polish(h_cont, best_n, beta, maxiter=200)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                h_cont = h_pol.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
