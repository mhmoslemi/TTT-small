```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_full(h):
    """Compute exact C5 value"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def soft_max_corr(h, beta, active=None):
    """Soft max of corridor values, optionally restricted to active set"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if active is not None:
        vals = corr[active]
    else:
        vals = corr
    m = np.max(vals)
    if beta < 1e-6:
        return m * dx, m, corr
    exps = np.exp(beta * (vals - m))
    return (m + np.log(np.sum(exps)) / beta) * dx, m, corr

def project_onto_simplex_box(h, n):
    """Project h to [0,1]^n with sum = n/2"""
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Distribute to non-boundary elements
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to_n(h, n_src, n_dst):
    """Interpolate h from n_src points to n_dst points"""
    if n_src == n_dst:
        return h.copy()
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xq = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    return np.interp(xq, xp, h)

def get_active_set(corr, n, gap_frac=0.02, min_count=5, max_count=40):
    """Find active set: lags where corr is close to max"""
    mx = np.max(corr)
    gap = gap_frac * mx
    active = np.where(corr >= mx - gap)[0]
    if len(active) < min_count:
        active = np.argsort(corr)[-min_count:]
    if len(active) > max_count:
        active = np.argsort(corr)[-max_count:]
    return active

def nelder_mead_step(h, n, best_c5, n_iter=200):
    """Nelder-Mead on the free variables"""
    free = (h > 0.01) & (h < 0.99)
    free_idx = np.where(free)[0]
    if len(free_idx) < 2:
        return h, best_c5
    
    # Create a reduced problem
    fixed = h.copy()
    
    def obj_reduced(x_reduced):
        h_trial = fixed.copy()
        h_trial[free_idx] = x_reduced
        h_trial = np.clip(h_trial, 0, 1)
        c5, _ = compute_c5_full(h_trial)
        # Penalty for sum constraint
        pen = 1000.0 * (np.sum(h_trial) - n/2.0)**2
        return c5 + pen
    
    x0 = h[free_idx].copy()
    bounds = [(0, 1)] * len(free_idx)
    
    try:
        res = minimize(obj_reduced, x0, method='Nelder-Mead',
                      bounds=bounds,
                      options={'maxiter': n_iter, 'xatol': 1e-10, 'fatol': 1e-12})
        h_trial = fixed.copy()
        h_trial[free_idx] = res.x
        h_trial = project_onto_simplex_box(h_trial, n)
        c5_trial, _ = compute_c5_full(h_trial)
        if c5_trial < best_c5:
            return h_trial, c5_trial
    except:
        pass
    return h, best_c5
```

```python
            break
        phase_deadline = deadline - 120
        
        h_n = interp_to(best_h, best_n, n)
        h_n = proj(h_n, n)
        c5_n = compute_c5(h_n)[0]
        
        # Fourier optimization at this resolution
        if time.time() > phase_deadline - 30:
            continue
        h_four, c5_four = fourier_optimize(n, 10, min(phase_deadline, time.time() + 60), rng)
        if h_four is not None and c5_four < c5_n:
            h_n = h_four
            c5_n = c5_four
        
        # Active-set optimization
        if time.time() > phase_deadline - 10:
            continue
        h_act, c5_act = active_set_optimize(h_n, n, phase_deadline, rng)
        if c5_act < c5_n:
            h_n = h_act
            c5_n = c5_act
        
        if c5_n < best_c5:
            best_c5 = c5_n
            best_h = h_n
            best_n = n
            print("Res {}: C5={:.8f}".format(n, best_c5))
    
    # === Phase 2: Coordinate descent at best resolution ===
    if time.time() > deadline - 80:
        pass
    else:
        h_cd, c5_cd = coord_descent(best_h, best_n, deadline - 80, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
            print("CoordDec: C5={:.8f}".format(best_c5))
    
    # === Phase 3: Nelder-Mead polish ===
    if time.time() > deadline - 40:
        pass
    else:
        h_nm, c5_nm = nelder_mead_polish(best_h, best_n, deadline - 40)
        if c5_nm < best_c5:
            best_c5 = c5_nm
            best_h = h_nm
            print("NM: C5={:.8f}".format(best_c5))
    
    # === Phase 4: Pairwise fine search ===
    if time.time() > deadline - 20:
        pass
    else:
        h_pw, c5_pw = pairwise_fine(best_h, best_n, deadline - 5, rng)
        if c5_pw < best_c5:
            best_c5 = c5_pw
            best_h = h_pw
            print("Pairwise: C5={:.8f}".format(best_c5))
    
    # === Phase 5: Final multi-resolution check ===
    if time.time() > deadline - 10:
        pass
    else:
        for dn in [-20, -10, 10, 20, 30]:
            if time.time() > deadline - 3:
                break
            nt = best_n + dn
            if nt < 80:
                continue
            h_try = interp_to(best_h, best_n, nt)
            h_try = proj(h_try, nt)
            c5_try = compute_c5(h_try)[0]
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
                best_n = nt
    
    # === Phase 6: Final SLSQP polish ===
    if time.time() > deadline - 5:
        pass
    else:
        n = best_n
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        h = best_h.copy()
        
        def final_obj(x):
            n_l = len(x)
            dx = 2.0 / n_l
            corr = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(corr)
            beta = 5000
            exps = np.exp(beta * (corr - m))
            return (m + np.log(np.sum(exps)) / beta) * dx + 1000*(np.sum(x)-target)**2
        
        try:
            res = minimize(final_obj, h, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
                          options={'maxiter': 200, 'ftol': 1e-17, 'iprint': 0})
            h_try = proj(res.x, n)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
        except:
            pass
    
    # Final projection and evaluation
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)[0]
    
    # Try evaluate_erdos_solution
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except:
        pass
    
    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
```
