    return h


def make_prime_diff_set(n, p=None):
    # Try to use a difference set at a prime near n
    # For prime p, the set of quadratic residues mod p is a difference set
    if p is None:
        p = n
    h = np.zeros(n)
    # Quadratic residues mod p: {i^2 mod p : i=1..(p-1)/2}
    residues = set()
    for i in range(1, (p + 1) // 2):
        residues.add((i * i) % p)
    for i in range(min(n, p)):
        h[i] = 1.0 if i in residues else 0.0
    return h


def multi_flip_sa(h_init, n, iterations, rng, t_start=0.003, t_end=0.00001, flip_count=2):
    h = np.where(h_init > 0.5, 1.0, 0.0)
    num_ones = int(np.sum(h))
    target_ones = n // 2
    if num_ones > target_ones:
        ones_idx = np.where(h == 1.0)[0]
        remove = rng.choice(len(ones_idx), num_ones - target_ones, replace=False)
        h[ones_idx[remove]] = 0.0
    elif num_ones < target_ones:
        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
        h[zeros_idx[add]] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])

    for it in range(iterations):
        if remaining_time() < 5:
            break
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break

        fc = flip_count if flip_count <= min(len(ones_list), len(zeros_list)) else 1
        oi_arr = rng.choice(len(ones_list), fc, replace=False)
        oz_arr = rng.choice(len(zeros_list), fc, replace=False)
        oi_set = set(ones_list[i] for i in oi_arr)
        oz_set = set(zeros_list[i] for i in oz_arr)
        if oi_set & oz_set:
            continue

        h[list(oi_set)] = 0.0
        h[list(oz_set)] = 1.0
        new_c5 = compute_c5(h)

        T = t_start * (t_end / t_start) ** (float(it) / max(float(iterations), 1.0))

        if new_c5 < current_c5 or (T > 1e-15 and rng.rand() < np.exp(-(new_c5 - current_c5) / T)):
            current_c5 = new_c5
            old_oi = [ones_list[i] for i in oi_arr]
            new_oz = [zeros_list[i] for i in oz_arr]
            for idx_old, idx_new in zip(oi_arr, oz_arr):
                ones_list[idx_old] = new_oz[idx_new]
                zeros_list[idx_new] = old_oi[idx_new]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[list(oi_set)] = 1.0
            h[list(oz_set)] = 0.0

    return best_h, best_c5


def top_corr_targeting(h, n, iters=200, top_k=6):
    """Iteratively reduce the top-k correlations by moving h values."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx_val = 2.0 / n

    for it in range(iters):
        if remaining_time() < 4:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[-top_k:]
        current_max = corr[top_idx[0]]

        # Compute gradient only for top-k shifts
        grad = np.zeros(n)
        for j in top_idx:
            s = j - (n - 1)
            if s >= 0:
                grad[s:n] += (1.0 - h[0:n - s])
                grad[0:n - s] -= h[s:n]
            else:
                sp = -s
                grad[0:n - sp] += (1.0 - h[sp:n])
                grad[sp:n] -= h[0:n - sp]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        grad /= gnorm
        improved = False
        for step in [0.02, 0.05, 0.1, 0.015, 0.08, 0.2, 0.03, 0.15]:
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break

        if not improved:
            # Try random perturbation
            for _ in range(20):
                idx = rng_perturb.randint(0, n - 1)
                delta = rng_perturb.uniform(-0.1, 0.1)
                h_trial = h.copy()
                h_trial[idx] += delta
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h = h_trial.copy()
                    improved = True
                    break
            if not improved and it > 50:
                break

    return best_h, best_c5


rng_perturb = np.random.RandomState(12345)


def coordinate_descent_polish(h, n, sweeps=8):
    h = h.copy()
    best_c5 = compute_c5(h)
    dx_val = 2.0 / n
    order = np.arange(n)

    for sweep in range(sweeps):
        if remaining_time() < 3:
            break
        np.random.RandomState(sweep + seed_cd).shuffle(order)
        any_imp = False
        for i in order:
            if remaining_time() < 2:
                break
            old_val = h[i]
            current_c5 = compute_c5(h)
            best_val = old_val
            best_h_local = h.copy()
            best_local_c5 = current_c5

            candidates = [0.0, 1.0, 0.5,
                          max(0.0, old_val - 0.15), min(1.0, old_val + 0.15),
                          max(0.0, old_val - 0.05), min(1.0, old_val + 0.05)]
            if old_val < 0.3:
                candidates.append(0.7)
            if old_val > 0.7:
                candidates.append(0.3)

            for nv in candidates:
                if abs(nv - old_val) < 1e-9:
                    continue
                h[i] = nv
                h_feas = project_feasible(h, n)
                c5_test = compute_c5(h_feas)
                if c5_test < best_local_c5 - 1e-10:
                    best_local_c5 = c5_test
                    best_val = nv
                    best_h_local = h_feas.copy()

            h = best_h_local.copy()
            if best_local_c5 < best_c5:
                best_c5 = best_local_c5
                any_imp = True

        if not any_imp:
            break

    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _N, _BETA, _TGT, _START_TIME, _BUDGET_S, seed_cd
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    seed_cd = seed + 7
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.7f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 160
        best_h = project_feasible(make_thue_morse(best_n), best_n)
        best_c5 = compute_c5(best_h)
        print("No parent. Thue-Morse: n={}, C5={:.7f}".format(best_n, best_c5))

    # Phase 1: Intensive continuous optimization at parent resolution
    n = best_n
    print("\n=== Phase 1: L-BFGS-B at n={} ===".format(n))
    _N = n
    _TGT = n / 2.0
    bounds = [(0.0, 1.0)] * n

    # Multiple beta values, increasing
    for beta_val in [50, 150, 400, 1000, 2000]:
        if remaining_time() < 15:
            break
        _BETA = beta_val
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 600, 'ftol': 1e-15,
                                    'gtol': 1e-10, 'iprint': 0,
                                    'maxls': 100, 'maxfun': 10000})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  beta={}: C5={:.7f}".format(beta_val, best_c5))
        except Exception as e:
            pass

    # Top-k smooth objective
    if remaining_time() > 15:
        try:
            res = minimize(topk_smooth_obj, best_h, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 400, 'ftol': 1e-15, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  TopK: C5={:.7f}".format(best_c5))
        except Exception:
            pass

    # Phase 2: Top-correlation targeting
    if remaining_time() > 30:
        print("\n=== Phase 2: Top-corr targeting ===")
        h_try, c5_try = top_corr_targeting(best_h, n, iters=300, top_k=8)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            print("  Targeting: C5={:.7f}".format(best_c5))

    # Phase 3: Binary multi-flip SA
    if remaining_time() > 60:
        print("\n=== Phase 3: Binary SA ===")
        h_bin_init = np.where(best_h > 0.5, 1.0, 0.0)

        for fc in [1, 2, 3]:
            if remaining_time() < 20:
                break
            sa_h, sa_c5 = multi_flip_sa(h_bin_init, n, 8000, rng,
                                         t_start=0.004, t_end=0.000005,
                                         flip_count=fc)
            if sa_c5 < best_c5:
                best_c5 = sa_c5
                best_h = sa_h.copy()
                print("  SA fc={}: C5={:.7f}".format(fc, best_c5))

            # Polish the binary result with continuous optimization
            if remaining_time() > 15:
                _BETA = 500
                try:
                    res = minimize(smooth_max_obj, sa_h, jac=smooth_max_grad,
                                   method='L-BFGS-B', bounds=bounds,
                                   options={'maxiter': 300, 'ftol': 1e-15, 'iprint': 0})
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        print("  SA+polish fc={}: C5={:.7f}".format(fc, best_c5))
                except Exception:
                    pass

    # Phase 4: Try other resolutions
    other_ns = [200, 240, 320]
    for nn in other_ns:
        if remaining_time() < 60:
            break
        if nn == best_n:
            continue
        print("\n=== Phase 4: n={} ===".format(nn))
        _N = nn
        _TGT = nn / 2.0
        bounds_n = [(0.0, 1.0)] * nn

        h_interp = project_feasible(interp_to(best_h, best_n, nn), nn)
        for beta_val in [200, 600, 1500]:
            if remaining_time() < 15:
                break
            _BETA = beta_val
            try:
                res = minimize(smooth_max_obj, h_interp, jac=smooth_max_grad,
                               method='L-BFGS-B', bounds=bounds_n,
                               options={'maxiter': 400, 'ftol': 1e-15, 'iprint': 0})
                h_new = project_feasible(res.x, nn)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = nn
                    print("  n={}, beta={}: C5={:.7f}".format(nn, beta_val, best_c5))
            except Exception:
                pass

        # Quick SA at new resolution
        if remaining_time() > 30 and best_n == nn:
            h_bin = np.where(best_h > 0.5, 1.0, 0.0)
            sa_h, sa_c5 = multi_flip_sa(h_bin, nn, 5000, rng,
                                         t_start=0.003, t_end=0.00001, flip_count=2)
            if sa_c5 < best_c5:
                best_c5 = sa_c5
                best_h = sa_h.copy()
                print("  SA n={}: C5={:.7f}".format(nn, best_c5))

    # Phase 5: Coordinate descent polish
    if remaining_time() > 15:
        print("\n=== Phase 5: Coord descent ===")
        n = best_n
        _N = n
        _TGT = n / 2.0
        h_pol, c5_pol = coordinate_descent_polish(best_h, n, sweeps=6)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            print("  Coord: C5={:.7f}".format(best_c5))

    # Phase 6: Final intensive L-BFGS-B
    if remaining_time() > 10:
        n = best_n
        _N = n
        _TGT = n / 2.0
        bounds = [(0.0, 1.0)] * n
        for beta_val in [2000, 5000]:
            if remaining_time() < 5:
                break
            _BETA = beta_val
            try:
                res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 800, 'ftol': 1e-15,
                                        'gtol': 1e-11, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  Final beta={}: C5={:.7f}".format(beta_val, best_c5))
            except Exception:
                pass

    # Final top-corr targeting pass
    if remaining_time() > 10:
        n = best_n
        h_tc, c5_tc = top_corr_targeting(best_h, n, iters=100, top_k=4)
        if c5_tc < best_c5:
            best_c5 = c5_tc
            best_h = h_tc.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
