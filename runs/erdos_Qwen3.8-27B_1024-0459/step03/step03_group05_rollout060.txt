    if i_lo > i_hi:
        return grad
    m_vals = np.arange(i_lo, i_hi + 1)
    j_vals = m_vals - k
    grad[m_vals] += (1.0 - h[j_vals])
    grad[j_vals] += (-h[m_vals])
    return grad * dx


def subgradient_topk(h, k_idx, n, dx, weights):
    g = np.zeros(n)
    for wi, ki in zip(weights, k_idx):
        if wi > 1e-12:
            g += wi * grad_single_shift(h, ki, n, dx)
    return g


def deterministic_sweep(h, target, best_c5_ref, max_sweeps):
    n = len(h)
    best = h.copy()
    best_val = compute_c5(best)
    t_s = time.time()
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            if time.time() - t_s > 8:
                return best, best_val
            trial = best.copy()
            for d in [0.005, -0.005, 0.01, -0.01, 0.003, -0.003]:
                if trial[i] + d < -1e-9 or trial[i] + d > 1 + 1e-9:
                    continue
                trial[i] += d
                # Find best compensation
                best_j = -1
                best_j_val = best_val
                candidates = [(j, d) for j in range(n) if j != i and 0 <= trial[j] - d <= 1]
                # Sample a subset for efficiency
                if len(candidates) > 30:
                    rng_l = np.random.RandomState(sweep * n + i)
                    idx = rng_l.choice(len(candidates), 30, replace=False)
                    candidates = [candidates[k] for k in idx]
                for j, dj in candidates:
                    trial[j] -= dj
                    cv = compute_c5(trial)
                    if cv < best_j_val:
                        best_j_val = cv
                        best_j = j
                    trial[j] += dj
                if best_j >= 0:
                    trial[i] += 0  # keep the d
                    trial[best_j] -= d
                    if best_j_val < best_val:
                        best = trial.copy()
                        best_val = best_j_val
                        improved = True
                    break
    return best, best_val

    n_par = len(h_par)

    best_h = h_par.copy()
    best_c5 = c5_par
    best_n = n_par

    # Phase 1: Subgradient with Nesterov at parent resolution
    n = best_n
    target = n / 2.0
    dx = 2.0 / n
