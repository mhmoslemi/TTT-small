import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def get_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(get_corr(h)) * dx)


def smooth_max_val(corr, beta, top_k):
    if top_k is not None and top_k < len(corr):
        idx = np.argpartition(corr, -top_k)[-top_k:]
        c = corr[idx]
    else:
        c = corr
    m = c.max()
    s = np.sum(np.exp(beta * (c - m)))
    return m + np.log(s) / beta


def smooth_max_and_grad(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    if top_k is not None and top_k < len(corr):
        idx = np.argpartition(corr, -top_k)[-top_k:]
        c = corr[idx]
    else:
        idx = np.arange(len(corr))
        c = corr
    m = c.max()
    w = np.exp(beta * (c - m))
    w /= np.sum(w)
    val = (m + np.log(np.sum(np.exp(beta * (c - m)))) / beta) * dx

    grad = np.zeros(n)
    for t in range(len(idx)):
        if w[t] < 1e-13:
            continue
        k = idx[t] - (n - 1)
        if k >= 0:
            i_lo, i_hi = k, n - 1
        else:
            i_lo, i_hi = 0, n - 1 + k
        if i_lo > i_hi:
            continue
        m_vals = np.arange(i_lo, i_hi + 1)
        j_vals = m_vals - k
        grad[m_vals] += w[t] * (1.0 - h[j_vals])
        grad[j_vals] += w[t] * (-h[m_vals])
    return val, grad * dx


def subgrad_topk(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    idx = np.argpartition(corr, -top_k)[-top_k:]
    c = corr[idx]
    m = c.max()
    w = np.exp(beta * (c - m))
    w /= np.sum(w)
    grad = np.zeros(n)
    for t in range(top_k):
        if w[t] < 1e-13:
            continue
        k = idx[t] - (n - 1)
        if k >= 0:
            i_lo, i_hi = k, n - 1
        else:
            i_lo, i_hi = 0, n - 1 + k
        if i_lo > i_hi:
            continue
        m_vals = np.arange(i_lo, i_hi + 1)
        j_vals = m_vals - k
        grad[m_vals] += w[t] * (1.0 - h[j_vals])
        grad[j_vals] += w[t] * (-h[m_vals])
    return grad * dx, c.max() * dx
