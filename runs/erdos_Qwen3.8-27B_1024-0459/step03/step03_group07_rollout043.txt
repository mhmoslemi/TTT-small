def subgradient_polish(h, n, iters=5000, time_limit=None, t0=None):
    """Subgradient method for the max correlation."""
    h = project_feasible(h.copy(), n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best = h.copy()
    best_c5 = np.max(corr) * (2.0/n)
    dx = 2.0/n
    
    # Avg for Nesterov
    x_avg = h.copy()
    
    for it in range(iters):
        if time_limit and t0 and time.time() - t0 > time_limit:
            break
        
        c = np.max(corr)
        if c * dx < best_c5:
            best_c5 = c * dx
            best = h.copy()
        
        # Subgradient: pick a random argmax lag
        max_idx = np.random.choice(np.where(corr == corr.max())[0])
        lag = max_idx - (n-1)
        
        # Compute gradient of corr[lag] w.r.t. h
        grad = np.zeros(n)
        if lag >= 0:
            # corr[n-1+lag] = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
            # d/dh[j]: (1-h[j+lag]) if j <= n-1-lag, minus h[j-lag] if lag <= j
            mask1 = np.arange(n) <= n-1-lag
            mask2 = np.arange(n) >= lag
            grad[mask1 & (np.arange(n)+lag < n)] += (1-h[np.arange(n)[mask1]+lag])
            grad[mask2] += -h[np.arange(n)[mask2]-lag]
        else:
            s = -lag
            # corr[n-1-s] = sum_{i=s}^{n-1} h[i]*(1-h[i-s])
            # d/dh[j]: (1-h[j-s]) if s <= j, minus h[j+s] if j+s < n
            mask1 = np.arange(n) >= s
            mask2 = np.arange(n)+s < n
            grad[mask1] += (1-h[np.arange(n)[mask1]-s])
            grad[mask2] += -h[np.arange(n)[mask2]+s]
        
        # Projected gradient step with decreasing step size
        lr = 1.0 / (it + 1)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        # Nesterov averaging
        alpha = 2.0 / (it + 2)
        x_avg = (1-alpha)*x_avg + alpha*h_new
        
        h = h_new
        x_avg = project_feasible(x_avg, n)

        if c_avg < best_c5:

        h = project_feasible(h - lr * grad, n)
        alpha = 2.0 / (it + 2.0)
        x_avg = (1.0 - alpha) * x_avg + alpha * h
        x_avg = project_feasible(x_avg, n)
    c_avg = compute_c5(x_avg)
    if c_avg < best_c5:
        best = x_avg.copy()
        best_c5 = c_avg
    return best

def smooth_max_obj(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 0.0)
    return (m + np.sum(np.exp(shifted)) / beta) * (2.0 / n)

def slsqp_polish(h, n, time_limit=None, t0=None):
    h = project_feasible(h.copy(), n)
    best = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for beta in [30.0, 100.0, 500.0, 2000.0]:
        if time_limit and t0 and time.time() - t0 > time_limit - 5:
            break
        def obj(x):
            return smooth_max_obj(x, n, beta)
        def obj_grad(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(corr)
            w = np.exp(np.clip(beta * (corr - m), -500.0, 0.0))
            w /= np.sum(w)
            grad = np.zeros(n)
            for ci in range(len(corr)):
                lag = ci - (n - 1)
                ar = np.arange(n)
                if lag >= 0:
                    mask1 = ar < n - lag
                    grad[mask1] += w[ci] * (1.0 - x[ar[mask1] + lag])
                    mask2 = ar >= lag
                    grad[mask2] += w[ci] * (-x[ar[mask2] - lag])
                else:
                    s = -lag
                    mask1 = ar >= s
                    grad[mask1] += w[ci] * (1.0 - x[ar[mask1] - s])
                    mask2 = ar + s < n
                    grad[mask2] += w[ci] * (-x[ar[mask2] + s])
            return grad * (2.0 / n)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 200, 'ftol': 1e-14, 'iprint': 0})
            h = project_feasible(res.x, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
        except Exception:
            pass
    return best

def interpolate_h(h_src, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_new = np.interp(xn, xp, h_src)
    return project_feasible(h_new, n_dst)

def make_structured_h(n, rng, parent_h=None):
    """Create a structured initial guess."""
    t = np.arange(n) / n
    cands = []
    # Sine-based
    for freq in [3, 5, 7, 11]:
        phase = rng.rand() * 2 * np.pi
        amp = 0.3 + 0.5 * rng.rand()
        h = np.clip(0.5 + amp * np.sin(2 * np.pi * freq * t + phase), 0, 1)
        cands.append(project_feasible(h, n))
    # Binary-like: random blocks
    for _ in range(3):
        h = np.zeros(n)
        block_size = max(1, n // (8 + 8 * rng.rand()))
        pos = 0
        val = 0.0
        while pos < n:
            ln = max(1, int(block_size * (0.5 + rng.rand())))
            h[pos:pos+ln] = val
            val = 1.0 - val
            pos += ln
        cands.append(project_feasible(h, n))
    # Parent interpolation with noise
    if parent_h is not None:
        h_interp = interpolate_h(parent_h, len(parent_h), n)
        for noise_scale in [0.05, 0.1, 0.2]:
            hp = h_interp + noise_scale * rng.randn(n)
            cands.append(project_feasible(hp, n))
    return cands

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return max(0.0, budget_s - (time.time() - t0) - 10.0)
    
    # Get parent
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn > 10:
            best_h = ph.copy()
            best_c5 = compute_c5(ph)
            best_n = pn
    except Exception:
        pass
    
    if best_h is None:
        best_n = 160
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)
    
    # Resolutions to try
    n_candidates = [160, 192, 224, 256, 320, 400]
    if best_n not in n_candidates:
        n_candidates.insert(0, best_n)
    n_candidates = sorted(set(n_candidates))
    
    total_time = budget_s - 200  # reserve 200s for polishing
    
    for n in n_candidates:
        if time_left() < 30:
            break
        # Allocate time for this resolution
        n_remaining = len([x for x in n_candidates if x >= n])
        time_per_res = max(30.0, (budget_s - 200 - (time.time() - t0)) / max(1, n_remaining))
        
        # Generate initial guesses
        inits = make_structured_h(n, rng, parent_h=best_h)
        # Add pure random
        inits.append(project_feasible(rng.rand(n), n))
        # Add flat
        inits.append(np.ones(n) * 0.5)
        
        n_inits = len(inits)
        time_per_init = max(10.0, time_per_res / n_inits)
        
        for h0 in inits:
            if time_left() < 15:
                break
            tl = min(time_per_init, time_left() - 5)
            
            # Coordinate descent
            h_cd = fast_coord_descent(h0, n, sweeps=30, time_limit=tl * 0.7, t0=t0)
            c5_cd = compute_c5(h_cd)
            
            # Subgradient polish
            if time_left() > 20:
                h_sg = subgradient_descent(h_cd, n, iters=2000, time_limit=tl * 0.9, t0=t0)
                c5_sg = compute_c5(h_sg)
            else:
                h_sg = h_cd
                c5_sg = c5_cd
            
            # More coordinate descent on best
            if time_left() > 15:
                h_best_local = h_cd if c5_cd < c5_sg else h_sg
                h_cd2 = fast_coord_descent(h_best_local, n, sweeps=20, time_limit=tl, t0=t0)
                c5_cd2 = compute_c5(h_cd2)
            else:
                h_cd2 = h_cd
                c5_cd2 = c5_cd
            
            # Take best
            cands = [(h_cd, c5_cd), (h_sg, c5_sg), (h_cd2, c5_cd2)]
            bh, bc = min(cands, key=lambda x: x[1])
            
            if bc < best_c5:
                best_c5 = bc
                best_h = bh.copy()
                best_n = n
    
    # Final polishing phase
    if time_left() > 60:
        h_pol = fast_coord_descent(best_h, best_n, sweeps=50, time_limit=time_left() * 0.4, t0=t0)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    
    if time_left() > 30:
        h_sp = slsqp_polish(best_h, best_n, time_limit=time_left() * 0.5, t0=t0)
        c5_sp = compute_c5(h_sp)
        if c5_sp < best_c5:
            best_c5 = c5_sp
            best_h = h_sp.copy()
    
    if time_left() > 20:
        h_pol2 = fast_coord_descent(best_h, best_n, sweeps=30, time_limit=time_left() * 0.5, t0=t0)
        c5_pol2 = compute_c5(h_pol2)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2.copy()
    
    # Final projection and validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
