```
    if they're all within epsilon, stop
    compute subgradient of max (or top-k average)
    take a small step in -subgradient direction
    project to feasible set
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(2, len(corr) - 1):
        if w[k] > 1e-14:
            g += w[k] * corr_grad_per_shift(x, n, k)
    return g * dx_val


def sum_con(x):
    return np.sum(x) - _TGT


def sum_con_jac(x):
    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=400):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def direct_max_reduce(h, n, max_iters=3000, top_k=4, step_scale=0.005):
    """Directly reduce the maximum correlation via projected subgradient."""
    h = h.copy()
    dx_val = 2.0 / n
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    for it in range(max_iters):
        if remaining() < 3:
            break
        if it % 200 == 0 and remaining() < 5:
            break
        corr = compute_corr(h)
        m = np.max(corr)
        cur_c5 = m * dx_val
        if cur_c5 < best_c5:
            best_c5 = cur_c5
        # Find top-k active shifts
        top_idx = np.argsort(-corr)[:top_k]
        # Check if all top-k are close
        if corr[top_idx[0]] - corr[top_idx[-1]] < 1e-10:
            break
        # Compute subgradient: weighted average of top-k gradients
        weights = np.zeros(n_corr)
        for idx in top_idx:
            weights[idx] = 1.0 / top_k
        g = np.zeros(n)
        for idx in top_idx:
            g += weights[idx] * corr_grad_per_shift(h, n, int(idx))
        g *= dx_val
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        # Adapt step size
        step = step_scale / (1.0 + 0.01 * it)
        h_trial = h - step * g
        h_trial = project_feasible(h_trial, n)
        new_c5 = compute_c5(h_trial)
        if new_c5 < best_c5 - 1e-13:
            h = h_trial
            best_c5 = new_c5
        else:
            # Halve the step
            h_trial = h - step * 0.5 * g
            h_trial = project_feasible(h_trial, n)
            new_c5 = compute_c5(h_trial)
            if new_c5 < best_c5 - 1e-13:
                h = h_trial
                best_c5 = new_c5
            else:
                if it > 500 and it % 50 == 0:
                    # Slight perturbation
                    rng = np.random.RandomState(it)
                    pert = rng.randn(n) * 0.001
                    h_trial = h + pert
                    h_trial = project_feasible(h_trial, n)
                    if compute_c5(h_trial) < best_c5:
                        h = h_trial
    return h, best_c5


def active_set_iterate(h, n, max_iters=500):
    """Iteratively equalize top correlations using pairwise gradient directions."""
    h = h.copy()
    dx_val = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        m = np.max(corr)
        cur_c5 = m * dx_val
        if cur_c5 < best_c5:
            best_c5 = cur_c5
        top2 = np.argsort(-corr)[:2]
        c1, c2 = corr[top2[0]], corr[top2[1]]
        if c1 - c2 < 1e-12:
            break
        g1 = corr_grad_per_shift(h, n, int(top2[0]))
        g2 = corr_grad_per_shift(h, n, int(top2[1]))
        d = g2 - g1
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-14:
            break
        d /= dnorm
        improved = False
        for step in [0.01, 0.02, 0.05, 0.005, 0.03, 0.08, 0.015, 0.04]:
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-13:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            # Try reducing the max directly
            g1 = -corr_grad_per_shift(h, n, int(top2[0]))
            gnorm = np.linalg.norm(g1)
            if gnorm > 1e-14:
                g1 /= gnorm
                for step in [0.005, 0.01, 0.02, 0.003]:
                    h_trial = h + step * g1
                    h_trial = project_feasible(h_trial, n)
                    c_new = compute_c5(h_trial)
                    if c_new < best_c5 - 1e-13:
                        h = h_trial
                        best_c5 = c_new
                        improved = True
                        break
            if not improved:
                break
    return h, best_c5


def binary_local_search(h, n, max_iters=5000, time_limit=30, rng=None):
    """Binary pair-swap local search with efficient evaluation."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h == 1)[0]
        remove = rng.choice(ones_idx, n_ones - target, replace=False)
        h[remove] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h == 0)[0]
        add = rng.choice(zeros_idx, target - n_ones, replace=False)
        h[add] = 1.0
    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx_val
    start = time.time()
    stall = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break
        k_max = int(np.argmax(corr))
        offset = k_max - (n - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        found = False
        n_try = 200
        for _ in range(n_try):
            if offset >= 0 and abs(offset) < n:
                ci = np.array([i for i in ones if i + offset < n and h[i + offset] == 0])
                if len(ci) > 0 and rng.rand() < 0.6:
                    i = ci[rng.randint(len(ci))]
                else:
                    i = ones[rng.randint(len(ones))]
            elif offset < 0 and abs(offset) < n:
                ci = np.array([i for i in ones if i - offset < n and h[i - offset] == 0])
                if len(ci) > 0 and rng.rand() < 0.6:
                    i = ci[rng.randint(len(ci))]
                else:
                    i = ones[rng.randint(len(ones))]
            else:
                i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            if new_max < cur_max - 1e-12:
                corr = new_corr
                cur_max = new_max
                best_c5 = cur_max * dx_val
                found = True
                break
            else:
                h[i] = 1.0
                h[j] = 0.0
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 60:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                corr = compute_corr(h)
                cur_max = float(np.max(corr))
                stall = 0
    return h, best_c5


def multi_start_optimize(n, parent_h, parent_n, max_time=60, start_seed=0):
    """Try multiple random + structural starts at resolution n."""
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    rng = np.random.RandomState(start_seed)
    best_h = None
    best_c5 = float('inf')

    # Start from parent interpolation
    h0 = interp_to(parent_h, parent_n, n)
    h0 = project_feasible(h0, n)

        h_bin = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h_bin[idx] = 1.0
        starts.append(h_bin)
    # Thue-Morse
    h_tm = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h_tm[i] = 1.0 if c == 1 else 0.0
    starts.append(h_tm)

    t_start = time.time()
    for si, h_s in enumerate(starts):
        if remaining() < 10 or time.time() - t_start > max_time - 5:
            break
        h_cur = h_s.copy()
        # Quick SLSQP
        for beta in [300, 1000, 3000]:
            if remaining() < 8:
                break
            h_cur = slsqp_smooth(h_cur, n, beta, maxiter=200)
        c5_cur = compute_c5(h_cur)
        # Direct max reduce
        h_cur, c5_cur = direct_max_reduce(h_cur, n, max_iters=500, top_k=3)
        # Active set
        h_cur, c5_cur = active_set_iterate(h_cur, n, max_iters=200)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()

    return best_h, best_c5


def perturb_and_reoptimize(h, n, rng, n_perturb=10, time_limit=40):
    """Perturb the solution and re-optimize to find better local optima."""
    dx_val = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    for p in range(n_perturb):
        if time.time() - t_start > time_limit - 5:
            break
        if remaining() < 8:
            break
        # Random perturbation
        mag = 0.01 * (1.0 - p / n_perturb) + 0.002
        h_pert = h + rng.randn(n) * mag
        h_pert = project_feasible(h_pert, n)
        c5_pert = compute_c5(h_pert)
        # Quick optimize
        h_opt = slsqp_smooth(h_pert, n, 1000, maxiter=150)
        h_opt, c5_opt = direct_max_reduce(h_opt, n, max_iters=300, top_k=3)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h = h_opt.copy()
    return best_h, best_c5
```
