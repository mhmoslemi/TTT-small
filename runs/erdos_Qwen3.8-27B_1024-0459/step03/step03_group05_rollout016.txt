                    best_h = h_new.copy()
                    print("  SLSQP beta={}: C5={:.6f}".format(beta, c5_new))
            except Exception:
                pass

    # Phase 5: Final fine coordinate descent
    if remaining() > 10:
        n = best_n
        target = n / 2.0
        h = best_h.copy()
        for _ in range(50000):
            if remaining() < 3:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d_sign = 1.0 if h[i] < h[j] else -1.0
            for delta in [0.002, 0.004, 0.001]:
                di = d_sign * delta
                dj = -d_sign * delta
                if h[i] + di < -1e-9 or h[i] + di > 1.0 + 1e-9:
                    continue
                if h[j] + dj < -1e-9 or h[j] + dj > 1.0 + 1e-9:
                    continue
                h[i] += di
                h[j] += dj
                c5_test = compute_c5(h)
                if c5_test < best_c5 - 1e-14:
                    best_c5 = c5_test
                    best_h = h.copy()
                    break
                h[i] -= di
                h[j] -= dj

    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
