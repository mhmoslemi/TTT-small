def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def smooth_obj_and_grad(h, beta, n):
    """Compute smooth log-sum-exp objective and its gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -600, 600))
    total_exp = np.sum(exps)
    obj = (m + np.log(total_exp) / beta) * dx
    weights = exps / total_exp

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        wk = weights[k]
        if abs(wk) < 1e-18:
            continue
        # First term: (1-h[i+lag]) for valid i where 0<=i<n and 0<=i+lag<n
        if lag >= 0:
            idx = np.arange(0, n - lag)
            grad[idx] += wk * (1.0 - h[idx + lag])
        else:
            idx = np.arange(-lag, n)
            grad[idx] += wk * (1.0 - h[idx + lag])
        # Second term: -h[i-lag] for valid i where 0<=i<n and 0<=i-lag<n
        if lag <= 0:
            lo = max(0, -lag)
            idx = np.arange(lo, n)
            grad[idx] -= wk * h[idx - lag]
        else:
            idx = np.arange(lag, n)
            grad[idx] -= wk * h[idx - lag]

    grad *= dx
    return obj, grad


def penalty_obj(h_flat, n, beta, M):
    h = h_flat[:n]
    val, _ = smooth_obj_and_grad(h, beta, n)
    val += M * (np.sum(h) - n / 2.0) ** 2
    return val


def penalty_grad(h_flat, n, beta, M):
    h = h_flat[:n]
    _, g = smooth_obj_and_grad(h, beta, n)
    g = g.copy()
    g += 2.0 * M * (np.sum(h) - n / 2.0)
    out = np.zeros_like(h_flat)
    out[:n] = g
    return out


def do_lbfgs_optimize(h0, n, beta, M, maxiter, ftol):
    """Run L-BFGS-B on the penalized smooth objective."""
    bounds = [(0.0, 1.0)] * n
    h_start = h0.copy()

    def fun(x):
        return penalty_obj(x, n, beta, M)

    def grad(x):
        return penalty_grad(x, n, beta, M)

    try:
        res = minimize(fun, h_start, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol,
                                'gtol': 1e-12, 'iprint': 0,
                                'maxfun': maxiter * 3})
        return project_feasible(res.x, n)
    except Exception:
        return h0


def do_projected_gradient(h0, n, beta, n_steps, step_sizes):
    """Projected gradient descent with backtracking."""
    h = h0.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for step_idx in range(n_steps):
        obj, grad = smooth_obj_and_grad(h, beta, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Projected gradient step with backtracking
        best_trial = None
        best_trial_c5 = best_c5
        for step in step_sizes:
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_trial_c5 - 1e-12:
                best_trial_c5 = c5_trial
                best_trial = h_trial.copy()

        if best_trial is not None:
            h = best_trial
            best_c5 = best_trial_c5
            best_h = h.copy()

    return best_h, best_c5

        sigma = 0.02 + 0.03 * rng.rand()
        h_pert = h_il + sigma * rng.randn(n_il)
        h_pert = project_feasible(h_pert, n_il)

        # Quick local optimization
        h_opt, c5_opt = do_projected_gradient(h_pert, n_il, beta=500.0,
                                              n_steps=20,
                                              step_sizes=[0.02, 0.05, 0.1, 0.2])
        # L-BFGS-B polish
        if budget_remain() > 40.0:
            h_opt = do_lbfgs_optimize(h_opt, n_il, 1000, 500.0, 100, 1e-14)
            c5_opt = compute_c5(h_opt)

        if c5_opt < c5_il - 1e-11:
            c5_il = c5_opt
            h_il = h_opt.copy()
            if c5_il < best_c5 - 1e-12:
                best_c5 = c5_il
                best_h = h_opt.copy()
                best_n = n_il
            n_restarts += 1
            print("  ILS restart {}: C5={:.8f}".format(n_restarts, best_c5))
        if n_restarts > 50:
            break

    print("  ILS done, best C5={:.8f}".format(best_c5))

    # Phase 4: Top-k lag refinement
    if budget_remain() > 40.0:
        print("\n=== Phase 4: Top-k refinement ===")
        best_h, best_c5 = do_topk_refine(best_h, best_n, budget_remain, top_k=5)
        print("  After top-k: C5={:.8f}".format(best_c5))

    # Phase 5: Coordinate descent
    if budget_remain() > 20.0:
        print("\n=== Phase 5: Coordinate descent ===")
        best_h, best_c5 = do_coordinate_descent(best_h, best_n, budget_remain, rng)
        best_n = len(best_h)
        print("  After coord descent: C5={:.8f}".format(best_c5))

    # Phase 6: Final L-BFGS-B polish at very high beta
    if budget_remain() > 15.0:
        print("\n=== Phase 6: Final polish ===")
        n = best_n
        for beta_val in [5000, 10000, 20000]:
            if budget_remain() < 8.0:
                break
            h_new = do_lbfgs_optimize(best_h, n, beta_val, 5000.0, 200, 1e-16)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new
                print("  beta={}: C5={:.8f}".format(beta_val, best_c5))

    # Phase 7: Final coordinate descent
    if budget_remain() > 10.0:
        best_h, best_c5 = do_coordinate_descent(best_h, best_n, budget_remain, rng)
        best_n = len(best_h)

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
