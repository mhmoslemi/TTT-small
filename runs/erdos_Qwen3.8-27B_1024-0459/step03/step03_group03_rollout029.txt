    for wi, ki in zip(w, idx):
        s = int(ki) - (n - 1)
        g += wi * _grad_lag(h, s)
    return g


def _grad_desc(h, n, nit, topk=5):
    best = h.copy()
    bc = _c5(best)
    for _ in range(nit):
        if _tl() < 5:
            break
        g = _grad_topk(best, topk)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            c = _corr(best)
            kmax = int(np.argmax(c))
            s = kmax - (n - 1)
            g = _grad_lag(best, s)
            gn = float(np.linalg.norm(g))
            if gn < 1e-14:
                break
        g /= gn
        moved = False
        for a in [0.04, 0.02, 0.01, 0.005, 0.002, 0.06, 0.03, 0.015, 0.008]:
            h2 = _proj(best - a * g, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-15:
                best = h2
                bc = c2
                moved = True
                break
        if not moved:
            c = _corr(best)
            kmax = int(np.argmax(c))
            s = kmax - (n - 1)
            g2 = _grad_lag(best, s)
            gn2 = float(np.linalg.norm(g2))
            if gn2 > 1e-14:
                g2 /= gn2
                for a in [0.03, 0.01, 0.005, 0.002, 0.001]:
                    h2 = _proj(best - a * g2, n)
                    c2 = _c5(h2)
                    if c2 < bc - 1e-15:
                        best = h2
                        bc = c2
                        moved = True
                        break
    return best, bc


def _directed_swap(h, n, max_sweeps=400):
    best = h.copy()
    bc = _c5(best)
    for _ in range(max_sweeps):
        if _tl() < 5:
            break
        ones = np.where(best > 0.999)[0]
        zeros = np.where(best < 0.001)[0]
        if len(ones) == 0 or len(zeros) == 0:
            hi = np.where(best > 0.55)[0]
            lo = np.where(best < 0.45)[0]
            if len(hi) == 0 or len(lo) == 0:
                break
            imp = False
            for ii in hi:
                if _tl() < 3:
                    break
                for jj in lo:
                    if ii == jj or abs(best[ii] - best[jj]) < 0.15:
                        continue
                    h2 = best.copy()
                    tmp = h2[ii]
                    h2[ii] = h2[jj]
                    h2[jj] = tmp
                    h2 = _proj(h2, n)
                    c2 = _c5(h2)
                    if c2 < bc - 1e-14:
                        best = h2
                        bc = c2
                        imp = True
                        break
                if imp:
                    break
            if not imp:
                break
            continue
        imp = False
        for ii in ones:
            if _tl() < 3:
                break
            for jj in zeros:
                if ii == jj:
                    continue
                h2 = best.copy()
                h2[ii] = 0.0
                h2[jj] = 1.0
                h2 = _proj(h2, n)
                c2 = _c5(h2)
                if c2 < bc - 1e-14:
                    best = h2
                    bc = c2
                    imp = True
                    ones = np.where(best > 0.999)[0]
                    zeros = np.where(best < 0.001)[0]
                    break
            if imp:
                break
        if not imp:
            break
    return best, bc


def _perturb_reopt(h, n, sigma, rng, nit_gd=150):
    h2 = _proj(h + sigma * rng.randn(n), n)
    c2 = _c5(h2)
    h3 = _slsqp(h2, n, 500, 200)
    c3 = _c5(h3)
    h4, c4 = _grad_desc(h3, n, nit_gd, 4)
    h5, c5v = _directed_swap(h4, n, 100)
    return h5, c5v


def _full_opt(h0, n):
    h = _proj(h0.copy(), n)
    c5 = _c5(h)

    for beta in [30, 100, 300, 800]:
        if _tl() < 20:
            break
        h2 = _slsqp(h, n, beta, 300)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2

    h, c5 = _grad_desc(h, n, 500, 5)
    h, c5 = _directed_swap(h, n, 300)
    h, c5 = _grad_desc(h, n, 300, 3)

    for beta in [1500, 3000]:
        if _tl() < 15:
            break
        h2 = _slsqp(h, n, beta, 400)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2

    h, c5 = _grad_desc(h, n, 300, 3)
    h, c5 = _directed_swap(h, n, 200)
    h, c5 = _grad_desc(h, n, 200, 5)

    if _tl() > 10:
        h2 = _slsqp(h, n, 6000, 500)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2
        h, c5 = _grad_desc(h, n, 200, 3)

    return h, c5


def run(seed=42, budget_s=1300, **kwargs):
    global _t0, _tb
    _t0 = time.time()
    _tb = float(budget_s)
    rng = np.random.RandomState(seed)

    try:
        hpar = np.array(initial_h_values, dtype=np.float64)
        npar = len(hpar)
    except Exception:
        npar = 160
        hpar = np.zeros(npar, dtype=np.float64)
        hpar[:npar // 2] = 1.0

    bh = _proj(hpar, npar)
    bc5 = _c5(bh)
    bn = npar
    print("Init n={} C5={:.8f}".format(bn, bc5))

    h, c5 = _full_opt(bh, npar)
    if c5 < bc5:
        bh, bc5 = h, c5
        print("Opt1 N={} C5={:.8f}".format(npar, c5))

    # Perturbation restarts at same resolution
    for sig in [0.02, 0.01, 0.005]:
        if _tl() < 60:
            break
        hp, cp = _perturb_reopt(bh, bn, sig, rng)
        if cp < bc5:
            bh, bc5 = hp, cp
            print("Perturb s={:.3f} C5={:.8f}".format(sig, cp))

    # Try nearby resolutions
    for nn in [npar + 2, npar + 4, npar - 2, npar - 4, npar + 6]:
        if _tl() < 80:
            break
        if nn < 64 or nn % 2 != 0:
            continue
        hi = _proj(_interp_f(bh, bn, nn), nn)
        h2, c2 = _full_opt(hi, nn)
        if c2 < bc5:
            bh, bc5, bn = h2, c2, nn
            print("N={} C5={:.8f}".format(nn, c2))

    # Final intensive polish
    if _tl() > 40:
        h, c5 = _grad_desc(bh, bn, 500, 3)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _directed_swap(bh, bn, 300)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _grad_desc(bh, bn, 400, 5)
        if c5 < bc5:
            bh, bc5 = h, c5

    if _tl() > 25:
        h2 = _slsqp(bh, bn, 10000, 600)
        c2 = _c5(h2)
        if c2 < bc5:
            bh, bc5 = h2, c2
        h, c5 = _grad_desc(bh, bn, 300, 3)
        if c5 < bc5:
            bh, bc5 = h, c5

    if _tl() > 20:
        hp, cp = _perturb_reopt(bh, bn, 0.003, rng, 100)
        if cp < bc5:
            bh, bc5 = hp, cp
        h, c5 = _grad_desc(bh, bn, 300, 4)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _directed_swap(bh, bn, 150)
        if c5 < bc5:
            bh, bc5 = h, c5

    if _tl() > 15:
        h2 = _slsqp(bh, bn, 15000, 500)
        c2 = _c5(h2)
        if c2 < bc5:
            bh, bc5 = h2, c2
        h, c5 = _grad_desc(bh, bn, 200, 3)
        if c5 < bc5:
            bh, bc5 = h, c5

    bh = _proj(bh, bn)
    fc5 = _c5(bh)
    try:
        ec = evaluate_erdos_solution(bh)
        if ec < fc5:
            fc5 = ec
    except Exception:
        pass
    print("FINAL n={} C5={:.8f}".format(bn, fc5))
    return (bh, fc5, bn)
