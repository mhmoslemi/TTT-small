def compute_c5_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def project_balance(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(free)
        if cnt > 0:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            break
    return h

        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xq = (np.arange(dst_n) + 0.5) * 2.0 / dst_n

            elif h_cur[j] >= d and h_cur[i] <= 1.0 - d:
                h_cur[j] -= d
                h_cur[i] += d
                c5_tr, _ = compute_c5_corr(h_cur)
                if c5_tr < best_c5:
                    best_c5 = c5_tr
                    best_h = h_cur.copy()
                else:
                    h_cur[i] -= d
                    h_cur[j] += d

    # Phase 6: 3-point perturbation
    if time_rem(t0, budget) > 20:
        h_cur = best_h.copy()
        for titer in range(40000):
            if time_rem(t0, budget) < 3:
                break
            idx3 = rng.choice(n, size=3, replace=False)
            i, j, k = int(idx3[0]), int(idx3[1]), int(idx3[2])
            d = 0.001 + 0.01 * rng.rand()
            if h_cur[i] >= 2.0 * d and h_cur[j] <= 1.0 - d and h_cur[k] <= 1.0 - d:
                h_cur[i] -= 2.0 * d
                h_cur[j] += d
                h_cur[k] += d
                c5_tr, _ = compute_c5_corr(h_cur)
                if c5_tr < best_c5:
                    best_c5 = c5_tr
                    best_h = h_cur.copy()
                else:
                    h_cur[j] -= d
                    h_cur[k] -= d
                    h_cur[i] += 2.0 * d

    # Phase 7: Final SLP + SLSQP at potentially different resolutions
    if time_rem(t0, budget) > 30:
        for dn in range(-30, 31, 4):
            if time_rem(t0, budget) < 5:
                break
            nt = best_n + dn
            if nt < 80 or nt > 500:
                continue
            ht = interp_h(best_h, best_n, nt)
            ht = project_balance(ht, nt)
            ct, _ = compute_c5_corr(ht)
            if ct < best_c5 - 1e-10:
                best_c5 = ct
                best_h = ht.copy()
                best_n = nt
                n = nt
                _g_n = n
                print("  Res n={}: {:.8f}".format(nt, best_c5))

    # Phase 8: Final SLSQP at best resolution
    if time_rem(t0, budget) > 10:
        n = best_n
        _g_n = n
        bounds_f = [(0.0, 1.0)] * n
        cons_f = [{'type': 'eq', 'fun': slsqp_con_fun, 'jac': slsqp_con_jac}]
        h_f = best_h.copy()
        _g_beta = 5000
        try:
            res = minimize(smooth_slsqp_obj, h_f, method='SLSQP',
                           bounds=bounds_f, constraints=cons_f,
                           options={'maxiter': 300, 'ftol': 1e-18, 'iprint': 0})
            h_new = project_balance(res.x, n)
            c5_new, _ = compute_c5_corr(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass
        try:
            res = minimize(raw_slsqp_obj, best_h, method='SLSQP',
                           bounds=bounds_f, constraints=cons_f,
                           options={'maxiter': 200, 'ftol': 1e-18, 'iprint': 0})
            h_new = project_balance(res.x, n)
            c5_new, _ = compute_c5_corr(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    # Final validation
    best_h = project_balance(best_h, best_n)
    final_c5, _ = compute_c5_corr(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
