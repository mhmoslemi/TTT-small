def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c == 1 else 0.0
    # Adjust count
    target = n // 2
    cur = int(h.sum())
    if cur > target:
        idx = np.where(h == 1)[0]
        h[idx[:cur - target]] = 0
    elif cur < target:
        idx = np.where(h == 0)[0]
        h[idx[:target - cur]] = 1
    return h

    n = len(a)
    a = a.copy()
    h = periodic_to_full(a)
    corr = compute_corr(h)
    cur_max = int(corr.max())
    best_a = a.copy()
    best_max = cur_max
    start = time.time()
    T = T_init

    while time.time() - start < time_limit:
        if _BUDGET - (time.time() - _START) < 5:
            break
        ones = np.where(a == 1)[0]
        zeros = np.where(a == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        a[i] = 0.0
        a[j] = 1.0
        h = periodic_to_full(a)
        new_corr = compute_corr(h)
        new_max = int(new_corr.max())
        delta = new_max - cur_max
        if delta <= 0 or (T > T_min and rng.random() < np.exp(-delta / max(T, 1e-10))):
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
        else:
            a[i] = 1.0
            a[j] = 0.0
        if T > T_min:
            T *= cool

    best_h = periodic_to_full(best_a)
    return best_h, best_max * (2.0 / (2 * n))


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def continuous_refine(h, n, time_limit, rng):
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        c = compute_corr(x)
        return float(c.max())

    def obj_grad(x):
        c = compute_corr(x)
        km = int(np.argmax(c))
        g = np.zeros(n)
        off = km - (n - 1)
        if off >= 0:
            npairs = n - off
            g[off:off + npairs] += (1.0 - x[:npairs])
            g[:npairs] -= x[off:off + npairs]
        else:
            ao = -off
            npairs = n - ao
            g[:npairs] += (1.0 - x[ao:ao + npairs])
            g[ao:ao + npairs] -= x[:npairs]
        return g

    # Use subgradient method with projection
    x = h.copy()
    lr = 0.02
    best_x = x.copy()
    best_val = obj(x)
    start = time.time()

    for it in range(5000):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break
        g = obj_grad(x)
        x_new = x - lr * g
        x_new = project_feasible(x_new, n)
        val = obj(x_new)
        if val < best_val:
            best_val = val
            best_x = x_new.copy()
        x = x_new
        lr *= 0.999

    return best_x, best_val * dx


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    parent_n = 320
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_h = thue_morse(320)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = 320

    n = best_n
    # Phase 1: Greedy local search from parent (binary)
    h_bin = to_binary(best_h, rng)
    t_start = time.time()
    h_g, c5_g = sa_greedy(h_bin, 200, rng, T_init=2.0, T_min=0.05, cool=0.99998, block_prob=0.2)
    if c5_g < best_c5:
        best_c5 = c5_g
        best_h = h_g.copy()
        best_n = n

    # Phase 2: Iterated SA from parent binary
    if remaining() > 200:
        h_ils, c5_ils = iterated_local_search(h_g, 300, rng, kick_size=6)
        if c5_ils < best_c5:
            best_c5 = c5_ils
            best_h = h_ils.copy()
            best_n = n

    # Phase 3: Try diverse initializations at n
    initializations = [
        thue_morse(n),
        balanced_22(n),
        random_binary(n, np.random.RandomState(seed + 1)),
        random_binary(n, np.random.RandomState(seed + 2)),
        random_binary(n, np.random.RandomState(seed + 3)),
    ]

    for init_h in initializations:
        if remaining() < 40:
            break
        init_c5 = compute_c5(init_h)
        if init_c5 < best_c5:
            best_c5 = init_c5
            best_h = init_h.copy()
        t_alloc = min(60, max(20, remaining() / 4))
        h_sa, c5_sa = sa_greedy(init_h, t_alloc, rng, T_init=3.0, T_min=0.1, cool=0.99997, block_prob=0.3)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n

    # Phase 4: 1-periodic optimization (n/2 bits)
    if n % 2 == 0:
        n_half = n // 2
        if n_half % 2 == 0:  # Need n divisible by 4 for binary periodic
            h_half = to_binary(best_h[:n_half], rng)
            # Ensure n_half/2 ones
            target_half = n_half // 2
            cur_ones = int(h_half.sum())
            if cur_ones > target_half:
                idx = np.where(h_half == 1)[0]
                h_half[idx[:cur_ones - target_half]] = 0
            elif cur_ones < target_half:
                idx = np.where(h_half == 0)[0]
                h_half[idx[:target_half - cur_ones]] = 1

            if remaining() > 100:
                t_alloc = min(200, max(50, remaining() * 0.4))
                h_per, c5_per = sa_periodic(h_half, t_alloc, rng, T_init=2.5, T_min=0.05, cool=0.99998)
                if c5_per < best_c5:
                    best_c5 = c5_per
                    best_h = h_per.copy()
                    best_n = n

            # Also try Thue-Morse periodic
            h_tm_half = thue_morse(n_half)
            if remaining() > 50:
                h_per2, c5_per2 = sa_periodic(h_tm_half, 80, rng, T_init=2.0, T_min=0.1, cool=0.99999)
                if c5_per2 < best_c5:
                    best_c5 = c5_per2
                    best_h = h_per2.copy()
                    best_n = n

    # Phase 5: Multi-resolution exploration
    n_options = sorted(set([n, n - 2, n + 2, n + 4, n + 6, n + 8, n + 12, n + 16]))
    n_options = [nn for nn in n_options if 200 <= nn <= 500 and nn % 2 == 0]

    for n_t in n_options:
        if remaining() < 50:
            break
        if n_t == best_n:
            continue
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = to_binary(h_interp, rng)
        c5_interp = compute_c5(h_bin_t)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_bin_t.copy()
            best_n = n_t

        t_alloc = min(60, max(20, remaining() * 0.15))
        h_sa, c5_sa = sa_greedy(h_bin_t, t_alloc, np.random.RandomState(seed + n_t),
                                T_init=2.5, T_min=0.1, cool=0.99998, block_prob=0.3)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n_t

    # Phase 6: Continuous refinement
    if remaining() > 30:
        n = best_n
        h_cont = best_h.copy().astype(float)
        h_cont = project_feasible(h_cont, n)
        h_ref, c5_ref = continuous_refine(h_cont, n, min(80, remaining() - 10), rng)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            best_n = n

    # Phase 7: Final greedy polish on best
    if remaining() > 30:
        n = best_n
        h_bin_final = to_binary(best_h, rng)
        h_pol, c5_pol = sa_greedy(h_bin_final, min(60, remaining() - 10), rng,
                                   T_init=1.5, T_min=0.02, cool=0.99999, block_prob=0.1)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n

    # Phase 8: Final continuous polish
    if remaining() > 15:
        n = best_n
        h_cont = project_feasible(best_h.copy().astype(float), n)
        h_ref2, c5_ref2 = continuous_refine(h_cont, n, min(30, remaining() - 5), rng)
        if c5_ref2 < best_c5:
            best_c5 = c5_ref2
            best_h = h_ref2.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
