import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def fast_coord_descent(h, n, sweeps=10, time_limit=None, t0=None):
    h = project_feasible(h, n)
    corr = compute_corr(h)
    best = h.copy()
    best_c5 = np.max(corr) * (2.0 / n)
    full_n = 2 * n - 1
    dx = 2.0 / n

    for sw in range(sweeps):
        if time_limit and t0 and time.time() - t0 > time_limit:
            break
        for j in range(n):
            if time_limit and t0 and time.time() - t0 > time_limit:
                break
            old = h[j]

            k_lo = max(-(n - 1), min(-j, j - n + 1))
            k_hi = min(n - 1, max(n - 1 - j, j))
            r_lo = k_lo + (n - 1)
            r_hi = k_hi + (n - 1)

            max_unaffected = 0.0
            if r_lo > 0:
                max_unaffected = max(max_unaffected, np.max(corr[:r_lo]))
            if r_hi < full_n - 1:
                max_unaffected = max(max_unaffected, np.max(corr[r_hi + 1:]))

            affected = []
            for k in range(k_lo, k_hi + 1):
                r = k + (n - 1)
                a = 0.0
                b = 0.0
                is_quad = False
                if 0 <= j + k <= n - 1:
                    a += (1.0 - h[j + k])
                    if k == 0:
                        is_quad = True
                if 0 <= j - k <= n - 1:
                    b += h[j - k]
                    a += -h[j - k]
                    if k == 0:
                        is_quad = True
                if is_quad:
                    rest = corr[r] - old * (1.0 - old)
                    affected.append((rest, 1.0, 0.0, True))
                elif a != 0.0 or b != 0.0:
                    rest = corr[r] - a * old - b
                    affected.append((rest, a, b, False))

            best_x = old
            best_val = np.inf
            grid = np.linspace(0.0, 1.0, 40)
            for x in grid:
                val = max_unaffected
                for (rest, a, b, is_q) in affected:
                    if is_q:
                        v = rest + x - x * x
                    else:
                        v = rest + a * x + b
                    if v > val:
                        val = v
                if val < best_val:
                    best_val = val
                    best_x = x

            span = 1.0 / 40.0
            lo = max(0.0, best_x - span)
            hi = min(1.0, best_x + span)
            for _ in range(45):

                    else:

                    if v > v1:

                    else:

                else:

def smooth_max_fun(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

def topk_fun(x, n, k):
    corr = np.correlate(x, 1.0 - x, mode='full')
    k = min(k, len(corr))
    top = np.sort(corr)[-k:]
    return np.mean(top) * (2.0 / n)

def sum_con_fun(x, n):
    return np.sum(x) - n / 2.0

def slsqp_optimize(h, n, time_limit=None, t0=None):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'args': (n,)}]
    best = h.copy()
    best_c5 = compute_c5(h)

    for beta in [30.0, 100.0, 500.0, 2000.0, 8000.0]:
        if time_limit and t0 and time.time() - t0 > time_limit:
            break
        try:
            res = minimize(smooth_max_fun, h.copy(), args=(n, beta),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-15, 'iprint': 0})
            h = project_feasible(res.x, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
        except Exception:
            pass

    for k in [2, 3, 5, 8]:
        if time_limit and t0 and time.time() - t0 > time_limit:
            break
        try:
            res = minimize(topk_fun, h.copy(), args=(n, k),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 150, 'ftol': 1e-15, 'iprint': 0})
            h = project_feasible(res.x, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
        except Exception:
            pass

    return best

def peak_equalization(h, n, iters=300, time_limit=None, t0=None):
    h = project_feasible(h, n)
    best = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n

    for it in range(iters):
        if time_limit and t0 and time.time() - t0 > time_limit:
            break
        corr = compute_corr(h)
        threshold = np.max(corr) - 2.0
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(corr)[-5:]

        grad = np.zeros(n)
        for r in active:
            lag = r - (n - 1)
            if lag >= 0:
                for i in range(n - lag):
                    grad[i] += (1.0 - h[i + lag])
                    grad[i + lag] += (-h[i])
            else:
                s = -lag
                for i in range(s, n):
                    grad[i] += (1.0 - h[i - s])
                    grad[i - s] += (-h[i])

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        lr = 0.01
        for attempt in range(20):
            h_new = h - lr * grad / grad_norm
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best = h_new.copy()
                h = h_new
                lr = min(lr * 1.05, 0.1)
                break
            else:
                lr *= 0.7
        else:
            # Try perturbation
            h_pert = h + 0.03 * np.random.randn(n)
            h_pert = project_feasible(h_pert, n)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best = h_pert.copy()
                h = h_pert
            else:
                h = project_feasible(h + 0.05 * np.random.randn(n), n)

    return best

def interpolate_h(h_src, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_new = np.interp(xn, xp, h_src)
    return project_feasible(h_new, n_dst)

def make_binary_init(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return project_feasible(h, n)

def make_sine_init(n, rng, base=None):
    t = np.arange(n) / n

    return project_feasible(h, n)
