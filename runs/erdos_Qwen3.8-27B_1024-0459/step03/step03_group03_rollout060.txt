        g /= gn
        moved = False
        for a in [0.05, 0.02, 0.01, 0.04, 0.008, 0.015, 0.03, 0.005, 0.06, 0.012]:
            h2 = _proj(best - a * g, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-15:
                best = h2
                bc = c2
                moved = True
                break
        if not moved:
            corr = _corr(best)
            kmax = int(np.argmax(corr))
            g2 = _g_one(best, kmax)
            gn2 = float(np.linalg.norm(g2))
            if gn2 > 1e-15:
                g2 /= gn2
                for a in [0.04, 0.015, 0.008, 0.02, 0.005, 0.05]:
                    h2 = _proj(best - a * g2, n)
                    c2 = _c5(h2)
                    if c2 < bc - 1e-15:
                        best = h2
                        bc = c2
                        moved = True
                        break
            if not moved:
                break
    return best, bc


def _binary_greedy(h0, n, rng, max_sweeps=50):
    """Greedy binary local search: swap 1<->0 pairs to reduce max correlation."""
    h = (h0 > 0.5).astype(np.float64)
    target = n // 2
    n_ones = int(h.sum())
    if n_ones > target:
        ones = np.where(h == 1)[0]
        order = np.argsort(h0[ones])[:n_ones - target]
        h[ones[order]] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        order = np.argsort(-(h0[zeros]))[:target - n_ones]
        h[zeros[order]] = 1

    cc = _corr(h)
    cmax = float(np.max(cc)) * (2.0 / n)
    best_h = h.copy()
    best_c = cmax

    for sweep in range(max_sweeps):
        if _rem() < 5:
            break
        imp = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Sort by "importance" - positions where h is near boundary
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in ones:
            if _rem() < 3:
                break
            for j in zeros:
                h[i], h[j] = h[j], h[i]
                nc = _corr(h)
                nmx = float(np.max(nc)) * (2.0 / n)
                if nmx < cmax - 1e-15:
                    cmax = nmx
                    cc = nc
                    imp = True
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    break
                else:
                    h[i], h[j] = h[j], h[i]
            if imp:
                imp = False
        if cmax < best_c:
            best_c = cmax
            best_h = h.copy()
        else:
            h = best_h.copy()
            cmax = best_c
            break
    return best_h, best_c


def _binary_sa(h0, n, nsteps, rng, T0=1.0, T1=0.005):
    """Simulated annealing on binary configs."""
    h = (h0 > 0.5).astype(np.float64)
    target = n // 2
    n_ones = int(h.sum())
    if n_ones > target:
        ones = np.where(h == 1)[0]
        order = np.argsort(h0[ones])[:n_ones - target]
        h[ones[order]] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        order = np.argsort(-(h0[zeros]))[:target - n_ones]
        h[zeros[order]] = 1

    cur_cc = _corr(h)
    cur_mx = float(np.max(cur_cc))
    best_h = h.copy()
    best_mx = cur_mx
    dx = 2.0 / n

    for step in range(nsteps):
        if _rem() < 5:
            break
        if step % 10000 == 0 and step > 0:
            T = T0 * (T1 / T0) ** (step / nsteps)
        else:
            T = T0 * (T1 / T0) ** (step / max(1, nsteps))
        if T < T1:
            T = T1

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        h[i], h[j] = h[j], h[i]
        new_cc = _corr(h)
        new_mx = float(np.max(new_cc))

        if new_mx <= cur_mx or rng.rand() < np.exp(-(new_mx - cur_mx) / max(T, 1e-12)):
            cur_mx = new_mx
            cur_cc = new_cc
            if new_mx < best_mx:
                best_mx = new_mx
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]

    return best_h, best_mx * dx


def _slsqp_solve(h0, n, beta, maxiter):
    ng_local = n

    def sobj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mx = float(np.max(c))
        sh = c - mx
        val = (mx + np.log(np.exp(beta * sh).sum()) / beta) * (2.0 / n)
        return val

    def scon(x):
        return x.sum() - 0.5 * n

    try:
        r = minimize(sobj, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': scon}],
                     options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return _proj(np.asarray(r.x, dtype=np.float64), n)
    except Exception:
        return _proj(h0.copy(), n)


def _coord_descent(h, n, nit, rng):
    best = _proj(h.copy(), n)
    bc = _c5(best)
    for it in range(nit):
        if _rem() < 5:
            break
        imp = False
        nb = np.where((best > 0.02) & (best < 0.98))[0]
        if len(nb) < 2:
            break
        for idx in nb:
            if _rem() < 2:
                break
            old = best[idx]
            for delta in [0.01, -0.01, 0.02, -0.02, 0.005, -0.005, 0.015, -0.015]:
                nv = old + delta
                if nv < 0.0 or nv > 1.0:
                    continue
                others = nb[nb != idx]
                if len(others) == 0:
                    break
                j = others[rng.randint(len(others))]
                h2 = best.copy()
                h2[idx] = nv
                h2[j] += -delta
                h2 = _proj(h2, n)
                c2 = _c5(h2)
                if c2 < bc - 1e-15:
                    best = h2
                    bc = c2
                    imp = True
                    break
            if imp:
                imp = False
        nb = np.where((best > 0.02) & (best < 0.98))[0]
    return best, bc


def _interp(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xs = (np.arange(n_from) + 0.5) * (2.0 / n_from)
    xd = (np.arange(n_to) + 0.5) * (2.0 / n_to)
    return np.interp(xd, xs, h)


def _full_optimize(h0, n, rng):
    """Full optimization pipeline at given resolution."""
    h = _proj(h0.copy(), n)
    c5 = _c5(h)

    # Binary SA
    h_bin, c5_bin = _binary_sa(h, n, 300000, rng, T0=0.8, T1=0.01)
    if c5_bin < c5:
        # Use binary as starting point for fractional
        h_init = h_bin.copy()
        for i in range(len(h_init)):
            # Add slight fractional noise to escape binary
            if h_init[i] == 1.0:
                h_init[i] = 0.95
            else:
                h_init[i] = 0.05
        h = _proj(h_init, n)
    else:
        h = _proj(h_bin, n)
        c5 = _c5(h)

    # Greedy binary
    if _rem() > 20:
        h_g, c5_g = _binary_greedy(h, n, rng, max_sweeps=30)
        if c5_g < c5:
            h = _proj(h_g, n)
            c5 = _c5(h)

    # Focus gradient
    if _rem() > 15:
        h2, c2 = _focus_loop(h, n, 400, rng, beta=100.0)
        if c2 < c5:
            h, c5 = h2, c2

    # SLSQP
    if _rem() > 30:
        h2 = _slsqp_solve(h, n, 300, 200)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2

    # More focus gradient
    if _rem() > 15:
        h2, c2 = _focus_loop(h, n, 300, rng, beta=200.0)
        if c2 < c5:
            h, c5 = h2, c2

    # Coordinate descent
    if _rem() > 10:
        h2, c2 = _coord_descent(h, n, 150, rng)
        if c2 < c5:
            h, c5 = h2, c2

    # Final SLSQP with high beta
    if _rem() > 25:
        h2 = _slsqp_solve(h, n, 2000, 300)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2
        h2, c2 = _focus_loop(h, n, 200, rng, beta=500.0)
        if c2 < c5:
            h, c5 = h2, c2

    return h, c5


def run(seed=42, budget_s=1300, **kwargs):
    global _t0, _tb
    _t0 = time.time()
    _tb = budget_s
    rng = np.random.RandomState(seed)

    # Get initial values
    try:
        hpar = np.array(initial_h_values, dtype=np.float64)
        npar = len(hpar)
    except Exception:
        npar = 160
        hpar = np.zeros(npar, dtype=np.float64)
        hpar[:npar // 2] = 1.0

    # Project parent
    bh = _proj(hpar, npar)
    bc5 = _c5(bh)
    bn = npar
    print("Init n={} C5={:.8f}".format(bn, bc5))

    # Phase 1: Full optimization at parent resolution
    h, c5 = _full_optimize(bh, npar, rng)
    if c5 < bc5:
        bh, bc5, bn = h, c5, npar
        print("After main n={} C5={:.8f}".format(npar, c5))

    # Phase 2: Try alternative resolutions
    alts = [npar + 4, npar - 4, npar + 8, npar - 8, npar + 12, npar - 12,
            npar + 20, npar - 20, npar + 40, npar + 60]
    for nn in alts:
        if _rem() < 60:
            break
        if nn < 40 or nn % 2 != 0:
            continue
        hi = _interp(bh, bn, nn)
        hi = _proj(hi, nn)
        h2, c2 = _full_optimize(hi, nn, rng)
        if c2 < bc5:
            bh, bc5, bn = h2, c2, nn
            print("Alt n={} C5={:.8f}".format(nn, c2))

    # Phase 3: Second binary SA with different params + polish
    if _rem() > 40:
        h_bin2, c5_bin2 = _binary_sa(bh, bn, 500000, rng, T0=0.3, T1=0.005)
        if c5_bin2 < bc5:
            # Interpolate: blend binary with current best
            blend = 0.7 * bh + 0.3 * h_bin2
            bh2 = _proj(blend, bn)
            c5b = _c5(bh2)
            if c5b < bc5:
                bh, bc5 = bh2, c5b
            else:
                bh2 = _proj(h_bin2, bn)
                c5b = _c5(bh2)
                if c5b < bc5:
                    bh, bc5 = bh2, c5b
        # Polish
        h2, c2 = _focus_loop(bh, bn, 300, rng, beta=200.0)
        if c2 < bc5:
            bh, bc5 = h2, c2
        h2, c2 = _coord_descent(bh, bn, 200, rng)
        if c2 < bc5:
            bh, bc5 = h2, c2

    # Phase 4: Final SLSQP
    if _rem() > 20:
        h2 = _slsqp_solve(bh, bn, 5000, 400)
        c2 = _c5(h2)
        if c2 < bc5:
            bh, bc5 = h2, c2
        h2, c2 = _focus_loop(bh, bn, 200, rng, beta=1000.0)
        if c2 < bc5:
            bh, bc5 = h2, c2

    # Phase 5: Ultra-fine coordinate descent
    if _rem() > 10:
        h2, c2 = _coord_descent(bh, bn, 300, rng)
        if c2 < bc5:
            bh, bc5 = h2, c2

    # Final projection and evaluation
    bh = _proj(bh, bn)
    fc5 = _c5(bh)
    try:
        ec = evaluate_erdos_solution(bh)
        if ec < fc5:
            fc5 = ec
    except Exception:
        pass
    print("FINAL n={} C5={:.8f}".format(bn, fc5))
    return (bh, fc5, bn)
