import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_N = 0
_BETA = 100.0
_TGT = 0.0
_PEN = 800.0


def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        unsat = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(unsat)
        if cnt > 0:
            h[unsat] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def smooth_max_obj(x):
    n = _N
    dx = 2.0 / n
    beta = _BETA
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    sm = (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx
    penalty = _PEN * (np.sum(x) - _TGT) ** 2
    return sm + penalty


def smooth_max_grad(x):
    n = _N
    dx = 2.0 / n
    beta = _BETA
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    weights /= np.sum(weights)

    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = weights[j]
        if wj < 1e-16:
            continue
        s = j - (n - 1)
        if s >= 0:
            for m_idx in range(s, n):
                grad[m_idx] += wj * (1.0 - x[m_idx - s])
            for m_idx in range(0, n - s):
                grad[m_idx] += wj * (-x[m_idx + s])
        else:
            sp = -s
            for m_idx in range(0, n - sp):
                grad[m_idx] += wj * (1.0 - x[m_idx + sp])
            for m_idx in range(sp, n):
                grad[m_idx] += wj * (-x[m_idx - sp])

    grad *= dx
    grad += 2.0 * _PEN * (np.sum(x) - _TGT)
    return grad


def corr_update_single(h, i, delta):
    """Compute change in correlation when h[i] changes by delta."""
    n = len(h)
    corr_char = np.zeros(2 * n - 1)
    h_new_i = h[i]
    # For shift s >= 0 (index j = n-1+s):
    # h[i] appears as h[j+s] when j = i-s, valid if s <= i <= n-1
    # h[i] appears as h[j] when j = i, valid if i <= n-1-s
    for s in range(0, n):
        j = n - 1 + s
        contrib = 0.0
        if s <= i <= n - 1:
            contrib += delta * (1.0 - h_new_i)  # from h[j+s] term, but h_new_i is old
            # Actually: h[j+s] changes by delta, so C_s changes by delta*(1-h[j]) = delta*(1-h[i-s])
            # But we need the OLD h[i-s], not the new h[i]
            # Since we're only changing h[i], h[i-s] is unchanged unless i-s == i (s=0)
            contrib = 0.0
            # h[i] as h[j+s]: j = i-s, so h[j] = h[i-s]
            contrib += delta * (1.0 - h[i - s])
        if i <= n - 1 - s:
            # h[i] as h[j]: j = i, so h[j+s] = h[i+s]
            contrib += (-delta) * h[i + s]
        corr_char[j] = contrib
    # For shift s < 0 (index j = n-1-s where s = -|s|):
    # C_{-s} = sum_{i=s}^{n-1} h[i-s]*(1-h[i]) for s>0, meaning shift by s
    # Actually np.correlate(h, 1-h) at index j corresponds to:
    # For j >= n-1: shift s = j-(n-1), C = sum h[i+s]*(1-h[i])
    # For j < n-1: shift s = j-(n-1) = -(n-1-j), C = sum h[i]*(1-h[i+s]) where s=n-1-j
    # Let me just use the formula directly
    for s in range(1, n):
        j = n - 1 - s
        contrib = 0.0
        # C at index j: sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
        # h[i] changes by delta when i = i (i.e., i is in [0, n-1-s])
        if i <= n - 1 - s:
            contrib += delta * (1.0 - h[i + s])
        # h[i+s] changes by delta when i+s = i, i.e., i = i-s, valid if i-s >= 0 and i-s <= n-1-s
        if i >= s and i - s <= n - 1 - s:
            contrib += (-delta) * h[i - s]
        corr_char[j] = contrib
    return corr_char


def coord_descent(h, n, max_sweeps=20, rng=None):
    """Coordinate descent with paired updates for the sum constraint."""
    dx = 2.0 / n
    current_c5 = compute_c5(h)

    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for i in order:
            # Find best new value for h[i], adjusting h[(i+n//2)%n] to maintain sum
            j = (i + n // 2) % n
            if j == i:
                j = (i + 1) % n

            old_i = h[i]
            old_j = h[j]

            # 1D search over t: h[i] = old_i + t, h[j] = old_j - t
            # Constraints: 0 <= old_i + t <= 1, 0 <= old_j - t <= 1
            t_lo = max(-old_i, -(1.0 - old_j))
            t_hi = min(1.0 - old_i, old_j)

            if t_hi - t_lo < 1e-12:
                continue

            def obj_1d(t):
                h[i] = old_i + t
                h[j] = old_j - t
                c = compute_c5(h)
                h[i] = old_i
                h[j] = old_j
                return c

            # Sample a few points for initial bracket
            t_mid = (t_lo + t_hi) / 2.0
            best_t = 0.0
            best_val = obj_1d(0.0)

            # Try a few strategic points
            candidates = [best_t, t_mid, t_lo + 0.1*(t_hi-t_lo), t_lo + 0.3*(t_hi-t_lo),
                         t_lo + 0.7*(t_hi-t_lo), t_lo + 0.9*(t_hi-t_lo)]
            for tc in candidates:
                vc = obj_1d(tc)
                if vc < best_val:
                    best_val = vc
                    best_t = tc

            # Golden section search
            lo, hi = t_lo, t_hi
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = obj_1d(c1)
            f2 = obj_1d(c2)
            for _ in range(25):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = obj_1d(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = obj_1d(c2)

            t_opt = (lo + hi) / 2.0
            f_opt = obj_1d(t_opt)

            if f_opt < best_val - 1e-9:
                best_t = t_opt
                best_val = f_opt

            if best_val < current_c5 - 1e-10:
                h[i] = old_i + best_t
                h[j] = old_j - best_t
                h = project_feasible(h, n)
                current_c5 = compute_c5(h)
                improved = True

        if not improved and sweep > 2:
            break

    return h, current_c5


def binary_local_search(h, n, num_iter, rng, double_flip=False):
    """Systematic binary local search."""
    h = np.where(h > 0.5, 1.0, 0.0)
    target_ones = n // 2
    num_ones = int(np.sum(h))
    if num_ones > target_ones:

        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
        h[zeros_idx[add]] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5

    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])

    for it in range(num_iter):
        if remaining_time() < 3:
            break
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break

        if double_flip:
            if len(ones_list) < 2 or len(zeros_list) < 2:
                break
            i1 = rng.randint(len(ones_list))
            j1 = rng.randint(len(ones_list))
            while j1 == i1:
                j1 = rng.randint(len(ones_list))
            i2 = rng.randint(len(zeros_list))
            j2 = rng.randint(len(zeros_list))
            while j2 == i2:
                j2 = rng.randint(len(zeros_list))

            o1, o2 = ones_list[i1], ones_list[j1]
            z1, z2 = zeros_list[i2], zeros_list[j2]

            h[o1] = 0.0
            h[o2] = 0.0
            h[z1] = 1.0
            h[z2] = 1.0
            new_c5 = compute_c5(h)

            if new_c5 < current_c5 - 1e-12:
                current_c5 = new_c5
                ones_list[i1] = z1
                ones_list[j1] = z2
                zeros_list[i2] = o1
                zeros_list[j2] = o2
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[o1] = 1.0
                h[o2] = 1.0
                h[z1] = 0.0
                h[z2] = 0.0
        else:
            i = rng.randint(len(ones_list))
            j = rng.randint(len(zeros_list))
            oi = ones_list[i]
            oz = zeros_list[j]

            h[oi] = 0.0
            h[oz] = 1.0
            new_c5 = compute_c5(h)

            if new_c5 < current_c5 - 1e-12:
                current_c5 = new_c5
                ones_list[i] = oz
                zeros_list[j] = oi
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[oi] = 1.0
                h[oz] = 0.0

    return best_h, best_c5

                if c5_bs2 < best_c5:
                    best_c5 = c5_bs2
                    best_h = h_best_bs2.copy()
                    print("  Binary (double): C5={:.7f}".format(best_c5))

            # Polish binary with continuous optimization
            if remaining_time() > 15:
                _N = n
                _TGT = n / 2.0
                bounds_n = [(0.0, 1.0)] * n
                _BETA = 500
                _PEN = 1000.0
                try:
                    res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                                   method='L-BFGS-B', bounds=bounds_n,
                                   options={'maxiter': 400, 'ftol': 1e-16,
                                            'gtol': 1e-11, 'iprint': 0})
                    h_polished = project_feasible(res.x, n)
                    c5_pol = compute_c5(h_polished)
                    if c5_pol < best_c5:
                        best_c5 = c5_pol
                        best_h = h_polished.copy()
                        print("  Polished: C5={:.7f}".format(best_c5))
                except Exception:
                    pass

            # Random restart for binary
            if remaining_time() > 20:
                h_rand = np.zeros(n)
                idx = rng.permutation(n)
                h_rand[idx[:n // 2]] = 1.0
                h_best_rr, c5_rr = binary_local_search(h_rand, n, 3000, rng, double_flip=False)
                if c5_rr < best_c5:
                    best_c5 = c5_rr
                    best_h = h_best_rr.copy()
                    print("  Restart: C5={:.7f}".format(best_c5))

    # Phase 4: Try n=240 as intermediate
    if remaining_time() > 60:
        print("\n=== Phase 4: Try n=240 ===")
        n3 = 240
        _N = n3
        _TGT = n3 / 2.0
        bounds3 = [(0.0, 1.0)] * n3

        h_interp3 = interp_to(best_h, best_n, n3)
        h0 = project_feasible(h_interp3, n3)

        for beta_val in [100, 300, 800]:
            if remaining_time() < 10:
                break
            _BETA = beta_val
            _PEN = 1000.0
            try:
                res = minimize(smooth_max_obj, h0, jac=smooth_max_grad,
                               method='L-BFGS-B', bounds=bounds3,
                               options={'maxiter': 500, 'ftol': 1e-16,
                                        'gtol': 1e-10, 'iprint': 0})
                h_new = project_feasible(res.x, n3)
                c5_new = compute_c5(h_new)
                if c5_new < compute_c5(h0):
                    h0 = h_new
            except Exception:
                pass

        if remaining_time() > 10:
            h_cd3, c5_cd3 = coord_descent(h0, n3, max_sweeps=10, rng=rng)
            h0 = h_cd3
            c5_240 = compute_c5(h0)
        else:
            c5_240 = compute_c5(h0)

        if c5_240 < best_c5:
            best_c5 = c5_240
            best_h = h0.copy()
            best_n = n3
            print("  n=240: C5={:.7f}".format(best_c5))

    # Phase 5: Final intensive polish
    if remaining_time() > 15:
        print("\n=== Phase 5: Final Polish ===")
        n = best_n
        _N = n
        _TGT = n / 2.0
        bounds_f = [(0.0, 1.0)] * n

        # Very high beta L-BFGS-B
        for beta_val in [1500, 3000]:
            if remaining_time() < 6:
                break
            _BETA = beta_val
            _PEN = 2000.0
            try:
                res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                               method='L-BFGS-B', bounds=bounds_f,
                               options={'maxiter': 800, 'ftol': 1e-16,
                                        'gtol': 1e-11, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  Final beta={}: C5={:.7f}".format(beta_val, best_c5))
            except Exception:
                pass

        # One more coord descent
        if remaining_time() > 8:
            h_cd_f, c5_cd_f = coord_descent(best_h, n, max_sweeps=8, rng=rng)
            if c5_cd_f < best_c5:
                best_c5 = c5_cd_f
                best_h = h_cd_f.copy()
                print("  Final CoordDesc: C5={:.7f}".format(best_c5))

    # Final evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
