def project_feasible(x, target):
    y = np.asarray(x, dtype=float).copy()
    n = y.size
    target = float(target)
    y = np.clip(y, 0.0, 1.0)
    for _ in range(60):
        diff = target - float(np.sum(y))
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            mask = y < 1.0 - 1e-13
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            y[mask] += diff / cnt
        else:
            mask = y > 1e-13
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            y[mask] += diff / cnt
        np.clip(y, 0.0, 1.0, out=y)
    diff = target - float(np.sum(y))
    if abs(diff) > 0.0:
        cand = np.where((y < 1.0 - 1e-12) & (y > 1e-12))[0]
        if cand.size > 0:
            added = False
            if diff > 0.0:
                room = 1.0 - y[cand]
                idxs = cand[room >= diff]
                if idxs.size > 0:
                    y[idxs[0]] += diff
                    added = True
            else:
                room = y[cand]
                idxs = cand[room >= -diff]
                if idxs.size > 0:
                    y[idxs[0]] += diff
                    added = True
            if not added:
                y[cand] += diff / float(cand.size)
                np.clip(y, 0.0, 1.0, out=y)
        else:
            y[:] += diff / float(n)
            np.clip(y, 0.0, 1.0, out=y)
    return y
```

import time
import math
import numpy as np
from scipy.optimize import minimize


def project_feasible(x, target):
    y = np.asarray(x, dtype=float).copy()
    n = int(y.size)
    target = float(target)
    if n <= 0:
        return y

    np.clip(y, 0.0, 1.0, out=y)

    for _ in range(80):
        diff = target - float(np.sum(y))
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            mask = y < (1.0 - 1e-13)
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            y[mask] += diff / float(cnt)
        else:
            mask = y > 1e-13
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            y[mask] += diff / float(cnt)
        np.clip(y, 0.0, 1.0, out=y)

    diff = target - float(np.sum(y))
    if diff != 0.0:
        cand = np.where((y < 1.0 - 1e-12) & (y > 1e-12))[0]
        if cand.size > 0:
            done = False
            if diff > 0.0:
                room = 1.0 - y[cand]
                ok = cand[room >= diff]
                if ok.size > 0:
                    y[ok[0]] += diff
                    done = True
            else:
                room = y[cand]
                ok = cand[room >= (-diff)]
                if ok.size > 0:
                    y[ok[0]] += diff
                    done = True
            if not done:
                y[cand] += diff / float(cand.size)
                np.clip(y, 0.0, 1.0, out=y)
        else:
            y[:] += diff / float(n)
            np.clip(y, 0.0, 1.0, out=y)

    return y


def make_c(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n <= 0:
        return np.array([0.0], dtype=float)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))


def compute_c5(h):
    return float(np.max(make_c(h)))


def compute_smooth(c, cmax, beta):
    c = np.asarray(c, dtype=float)
    cmax = float(cmax)
    z = float(beta) * (c - cmax)
    z = np.minimum(z, 0.0)
    z = np.maximum(z, -745.0)
    return float(np.sum(np.exp(z)))


def top_indices(c, k):
    c = np.asarray(c, dtype=float)
    k = int(min(k, c.size))
    if k <= 0:
        return np.array([], dtype=int)
    if k >= c.size:
        return np.arange(c.size)
    return np.argpartition(c, -k)[-k:]


def apply_single(h, omh, c, idx, delta, n, off, dx):
    delta = float(delta)
    if delta == 0.0:
        return

    old = float(h[idx])
    old_omh = float(omh[idx])
    new = old + delta
    new_omh = old_omh - delta

    if idx < off:
        # As the "head" in terms for negative/positive lags, excluding k=0.
        if idx < off:
            c[idx:off] += dx * delta * omh[idx + 1:n][::-1]

        # As the "tail" in terms where the other factor is before idx.
        if idx > 0:
            c[off - idx:off] -= dx * delta * h[:idx]

        # As the "tail" in terms where the other factor is after idx.
        length_right = n - idx - 1
        if length_right > 0:
            c[off + 1:off + 1 + length_right] -= dx * delta * h[idx + 1:n]

    # Lag k=0 term: h_i(1-h_i) changes by delta * (1 - old - new).
    c[off] += dx * delta * (old_omh + new_omh - 1.0)

    h[idx] = new
    omh[idx] = new_omh


def attempt_pair(h, c, omh, i, delta, j, n, off, dx,
                 old_max, old_smooth=None, smooth_beta=0.0):
    if i == j:
        return False, old_max, old_smooth

    delta = float(delta)
    if delta == 0.0:
        return False, old_max, old_smooth

    hi = float(h[i]) + delta
    hj = float(h[j]) - delta
    if hi < -1e-12 or hi > 1.0 + 1e-12 or hj < -1e-12 or hj > 1.0 + 1e-12:
        return False, old_max, old_smooth

    # Move mass from j to i:
    #   h[i] += delta, h[j] -= delta
    apply_single(h, omh, c, i, delta, n, off, dx)
    apply_single(h, omh, c, j, -delta, n, off, dx)

    new_max = float(np.max(c))
    new_smooth = old_smooth
    accept = new_max < (old_max - 1e-12)

    if (not accept) and old_smooth is not None and smooth_beta > 0.0 and new_max <= old_max + 1e-12:
        new_smooth = compute_smooth(c, new_max, smooth_beta)
        if new_smooth < (old_smooth - 1e-10):
            accept = True

    if accept:
        return True, new_max, new_smooth

    # Revert in reverse order.
    apply_single(h, omh, c, j, delta, n, off, dx)
    apply_single(h, omh, c, i, -delta, n, off, dx)
    return False, old_max, old_smooth


def make_delta(rng, margin):
    margin = float(margin)
    if margin <= 1e-10:
        return 0.0

    clip = 0.9 * margin
    u = float(rng.random_sample())

    if u < 0.08 and margin > 0.9:
        delta = margin
    elif u < 0.22:
        delta = min(0.25, clip) * float(rng.uniform(0.3, 1.0))
    else:
        # log-uniform in [1e-4, 5e-2]
        delta = math.exp(float(rng.uniform(-9.210340372, -2.995732274)))
        if delta > clip:
            delta = clip

    if delta <= 1e-9:
        if clip > 1e-6:
            delta = min(1e-5, clip)
        else:
            delta = 0.0

    return float(delta)


def weighted_gradient(h, weights, n, off, dx):
    g = np.zeros(n, dtype=float)
    nz = np.nonzero(np.abs(weights) > 1e-14)[0]

    for idx in nz:
        w = float(weights[idx])
        if w == 0.0:
            continue
        k = int(idx) - int(off)

        if k == 0:
            g += w * (1.0 - 2.0 * h)
        elif k > 0:
            length = n - k
            if length <= 0:
                continue
            g[k:] += w * (1.0 - h[:length])
            g[:length] -= w * h[k:]
        else:
            a = -k
            length = n - a
            if length <= 0:
                continue
            g[:length] += w * (1.0 - h[a:])
            g[a:] -= w * h[:length]

    g *= float(dx)
    return g


def subgradient_polish(h, target, deadline, seed):
    h = np.asarray(h, dtype=float).copy()
    target = float(target)
    n = int(h.size)
    off = n - 1
    dx = 2.0 / float(n)

    c = make_c(h)
    current_max = float(np.max(c))
    best_h = h.copy()
    best_c = current_max

    if time.time() >= float(deadline):
        return h, c, best_h, best_c

    rng = np.random.RandomState(seed)
    steps = (0.05, 0.02, 0.008, 0.003, 0.001)

    if n <= 256:
        betas = (8, 30, 100, 400)
    else:
        betas = (8, 30, 120)

    for beta in betas:
        stall = 0
        for _ in range(70):
            if time.time() >= float(deadline):
                break

            current_max = float(np.max(c))
            if current_max < best_c - 1e-13:
                best_c = current_max
                best_h = h.copy()

            z = float(beta) * (c - current_max)
            z = np.minimum(z, 0.0)
            z = np.maximum(z, -745.0)
            weights = np.exp(z)
            s = float(np.sum(weights))
            if s <= 0.0:
                break
            weights = weights / s

            g = weighted_gradient(h, weights, n, off, dx)
            d = -g
            d -= float(np.mean(d))
            scale = float(np.max(np.abs(d)))
            if scale < 1e-14:
                break
            d /= scale

            improved = False
            for step in steps:
                if time.time() >= float(deadline):
                    break
                h_trial = project_feasible(h + step * d, target)
                c_trial = make_c(h_trial)
                f_trial = float(np.max(c_trial))
                if f_trial < current_max - 1e-12:
                    h = h_trial
                    c = c_trial
                    current_max = f_trial
                    improved = True
                    break

            if not improved:
                stall += 1
                if stall >= 4:
                    break

    c = make_c(h)
    current_max = float(np.max(c))
    if current_max < best_c - 1e-13:
        best_c = current_max
        best_h = h.copy()

    return h, c, best_h, best_c


def perturb_h(h, target, scale, frac, rng):
    y = np.asarray(h, dtype=float).copy()
    n = int(y.size)
    mask = rng.rand(n) < float(frac)
    y[mask] += float(scale) * rng.randn(int(np.count_nonzero(mask)))
    return project_feasible(y, float(target))


def upsample_h(h, new_n, mode):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    new_n = int(new_n)
    if new_n <= n:
        return project_feasible(h, 0.5 * n)

    x_old = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    x_new = (np.arange(new_n, dtype=float) + 0.5) * (2.0 / float(new_n))

    if mode == "rep":
        idx = np.floor(x_new * (float(n) / 2.0)).astype(int)
        np.clip(idx, 0, n - 1, out=idx)
        y = h[idx]
    else:
        y = np.interp(x_new, x_old, h)

    return project_feasible(y, 0.5 * float(new_n))


def logsum_obj(x, n, beta):
    c = make_c(x)
    m = float(np.max(c))
    z = float(beta) * (c - m)
    z = np.minimum(z, 0.0)
    z = np.maximum(z, -745.0)
    s = float(np.sum(np.exp(z)))
    if s <= 0.0:
        s = 1.0
    return m + float(np.log(s)) / float(beta)


def logsum_jac(x, n, beta):
    c = make_c(x)
    m = float(np.max(c))
    z = float(beta) * (c - m)
    z = np.minimum(z, 0.0)
    z = np.maximum(z, -745.0)
    w = np.exp(z)
    s = float(np.sum(w))
    if s <= 0.0:
        w = np.full(c.shape, 1.0 / c.size)
    else:
        w = w / s

    n = int(n)
    off = n - 1
    dx = 2.0 / float(n)
    return weighted_gradient(np.asarray(x, dtype=float), w, n, off, dx)


def sum_con_fun(x, target):
    return float(np.sum(x) - float(target))


def sum_con_jac(x, target):
    return np.ones(len(x), dtype=float)


def slsqp_polish(h, target, deadline, seed):
    h = np.asarray(h, dtype=float).copy()
    target = float(target)
    n = int(h.size)

    if n > 220 or time.time() >= float(deadline):
        c = make_c(h)
        return h, c, h.copy(), float(np.max(c))

    c = make_c(h)
    best_h = h.copy()
    best_c = float(np.max(c))
    h_cur = h.copy()

    bounds = [(0.0, 1.0)] * n
    cons = ({
        "type": "eq",
        "fun": sum_con_fun,
        "jac": sum_con_jac,
        "args": (target,),
    },)

    if n <= 180:
        betas = (40, 150, 500, 1500)
    else:
        betas = (40, 150, 500)

    for beta in betas:
        if time.time() >= float(deadline):
            break
        try:
            res = minimize(
                logsum_obj,
                h_cur,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                jac=logsum_jac,
                args=(n, float(beta)),
                maxiter=70,
                ftol=1e-12,
                options={"iprint": 0}
            )
            if res is not None and hasattr(res, "x") and res.x is not None:
                h_new = project_feasible(np.asarray(res.x, dtype=float), target)
                c_new = make_c(h_new)
                val = float(np.max(c_new))
                if val < best_c - 1e-13:
                    best_c = val
                    best_h = h_new.copy()
                    h_cur = h_new.copy()
                elif val < best_c + 1e-7:
                    h_cur = h_new.copy()
        except Exception:
            pass

    c_best = make_c(best_h)
    best_c = float(np.max(c_best))
    c_cur = make_c(h_cur)
    return h_cur, c_cur, best_h, best_c


def hill_climb(h, target, deadline, seed):
    h = np.asarray(h, dtype=float).copy()
    target = float(target)
    n = int(h.size)
    off = n - 1
    dx = 2.0 / float(n)

    c = make_c(h)
    omh = 1.0 - h
    current_max = float(np.max(c))

    best_h = h.copy()
    best_c = current_max

    if time.time() >= float(deadline):
        c = make_c(h)
        cf = float(np.max(c))
        if cf < best_c - 1e-14:
            best_c = cf
            best_h = h.copy()
        cb = make_c(best_h)
        return h, c, best_h, float(np.max(cb))

    rng = np.random.RandomState(seed)
    active_k = 24 if n <= 256 else 42
    sample_count = 5 if n <= 256 else 6
    active = top_indices(c, active_k)

    no_improve = 0
    trial = 0
    smooth_enabled = False
    old_smooth = None
    smooth_beta = 90.0

    while time.time() < float(deadline):
        trial += 1

        if trial % 700 == 0:
            active = top_indices(c, active_k)

        if no_improve > 900 and not smooth_enabled:
            smooth_enabled = True

        if smooth_enabled and old_smooth is None:
            old_smooth = compute_smooth(c, current_max, smooth_beta)

        if no_improve > 1800:
            c = make_c(h)
            current_max = float(np.max(c))
            omh = 1.0 - h
            if current_max < best_c - 1e-12:
                best_c = current_max
                best_h = h.copy()
            active = top_indices(c, active_k)
            no_improve = 0
            old_smooth = None

        if active.size == 0:
            active = top_indices(c, active_k)

        idxk = int(active[int(rng.randint(active.size))])
        k = idxk - off

        if k == 0:
            best_i = -1
            best_j = -1
            best_diff = -1.0
            for _ in range(sample_count):
                a = int(rng.randint(n))
                b = int(rng.randint(n))
                if a == b:
                    continue
                if float(h[a]) <= float(h[b]):
                    i = a
                    j = b
                    diff = float(h[b] - h[a])
                else:
                    i = b
                    j = a
                    diff = float(h[a] - h[b])
                if diff > best_diff:
                    best_diff = diff
                    best_i = i
                    best_j = j

            if best_i < 0 or best_diff <= 1e-6:
                no_improve += 1
                continue
            i = best_i
            j = best_j
        else:
            if k > 0:
                length = n - k
                best_r = -1
                best_val = -1.0
                for _ in range(sample_count):
                    r = int(rng.randint(length))
                    val = float(h[r + k] * (1.0 - h[r]))
                    if val > best_val:
                        best_val = val
                        best_r = r
                if best_r < 0:
                    best_r = int(rng.randint(length))
                    best_val = 0.0
                tail = best_r
                head = best_r + k
            else:
                a = -k
                length = n - a
                best_r = -1
                best_val = -1.0
                for _ in range(sample_count):
                    r = int(rng.randint(length))
                    val = float(h[r] * (1.0 - h[r + a]))
                    if val > best_val:
                        best_val = val
                        best_r = r
                if best_r < 0:
                    best_r = int(rng.randint(length))
                    best_val = 0.0
                head = best_r
                tail = best_r + a

            # Mostly move from head to tail, which directly decreases this lag.
            if rng.random_sample() < 0.10:
                i = head
                j = tail
            else:
                i = tail
                j = head

        margin = min(float(h[j]), float(1.0 - h[i]))
        if margin <= 1e-9:
            no_improve += 1
            continue

        delta = make_delta(rng, margin)
        if delta <= 1e-10:
            no_improve += 1
            continue

        old_s = old_smooth if (smooth_enabled and old_smooth is not None) else None
        sb = smooth_beta if old_s is not None else 0.0

        ok, new_max, new_smooth = attempt_pair(
            h, c, omh, i, delta, j, n, off, dx,
            current_max, old_s, sb
        )

        if not ok and margin > 0.5 and delta < 0.5 and rng.random_sample() < 0.20:
            delta2 = min(margin, max(0.10, 2.0 * delta))
            ok2, new_max2, new_smooth2 = attempt_pair(
                h, c, omh, i, delta2, j, n, off, dx,
                current_max, old_s, sb
            )
            if ok2:
                ok = True
                new_max = new_max2
                new_smooth = new_smooth2

        if ok:
            current_max = new_max
            old_smooth = None
            if current_max < best_c - 1e-12:
                best_c = current_max
                best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

    c = make_c(h)
    cf = float(np.max(c))
    if cf < best_c - 1e-14:
        best_c = cf
        best_h = h.copy()

    cb = make_c(best_h)
    best_c = float(np.max(cb))
    return h, c, best_h, best_c


def evaluate_candidate(h, best_h, best_c, best_n):
    try:
        h = np.asarray(h, dtype=float)
    except Exception:
        return best_h, best_c, int(best_n)

    if h.size == 0:
        return best_h, best_c, int(best_n)

    target = 0.5 * float(h.size)
    if (abs(float(np.sum(h)) - target) > 1e-9 or
            np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10)):
        h = project_feasible(h, target)

    c = make_c(h)
    val = float(np.max(c))

    if val < best_c - 1e-13:
        return h.copy(), val, int(h.size)

    return best_h, best_c, int(best_n)


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    budget_s = float(budget_s)
    reserve = 25.0 if budget_s > 100.0 else 8.0
    global_deadline = t_start + max(20.0, budget_s - reserve)

    # Initial construction.
    try:
        h0 = np.asarray(initial_h_values, dtype=float).ravel()
    except Exception:
        h0 = None

    if h0 is None or h0.size < 32:
        rng0 = np.random.RandomState(seed)
        h0 = np.zeros(160, dtype=float)
        h0[rng0.choice(160, 80, replace=False)] = 1.0

    n0 = int(h0.size)
    target0 = 0.5 * float(n0)

    if (abs(float(np.sum(h0)) - target0) > 1e-9 or
            np.any(h0 < -1e-10) or np.any(h0 > 1.0 + 1e-10)):
        h0 = project_feasible(h0, target0)

    c0 = make_c(h0)
    best_c = float(np.max(c0))
    best_h = h0.copy()
    best_n = n0

    max_time = max(30.0, budget_s - 25.0)

    # Early first-order polish on the base resolution.
    if budget_s > 220.0:
        sg_deadline = min(
            global_deadline,
            time.time() + min(85.0, 0.075 * max_time)
        )
        if sg_deadline > time.time() + 25.0:
            h_sg, c_sg, best_sg, _ = subgradient_polish(
                best_h.copy(), 0.5 * float(best_n), sg_deadline, seed + 1011
            )
            best_h, best_c, best_n = evaluate_candidate(best_sg, best_h, best_c, best_n)
            best_h, best_c, best_n = evaluate_candidate(h_sg, best_h, best_c, best_n)

    # Optional local nonlinear polish with SLSQP at moderate resolution.
    if best_n <= 220 and budget_s > 450.0:
        sl_deadline = min(
            global_deadline,
            time.time() + min(55.0, 0.045 * max_time)
        )
        if sl_deadline > time.time() + 18.0:
            h_sl, c_sl, best_sl, _ = slsqp_polish(
                best_h.copy(), 0.5 * float(best_n), sl_deadline, seed + 1021
            )
            best_h, best_c, best_n = evaluate_candidate(best_sl, best_h, best_c, best_n)
            best_h, best_c, best_n = evaluate_candidate(h_sl, best_h, best_c, best_n)

    # Base-resolution local search.
    now = time.time()
    if best_n <= 250:
        base_len = 0.34 * max_time
    elif best_n <= 520:
        base_len = 0.26 * max_time
    else:
        base_len = 0.16 * max_time

    base_deadline = min(global_deadline - 80.0, now + base_len)
    if base_deadline > now + 18.0:
        h_b, c_b, best_b, _ = hill_climb(
            best_h.copy(), 0.5 * float(best_n), base_deadline, seed + 2021
        )
        best_h, best_c, best_n = evaluate_candidate(best_b, best_h, best_c, best_n)
        best_h, best_c, best_n = evaluate_candidate(h_b, best_h, best_c, best_n)

    # Try a finer resolution if it looks promising and time remains.
    now = time.time()
    hi_deadline = min(global_deadline - 45.0, now + 0.33 * max_time)
    if best_n <= 320 and hi_deadline > now + 120.0:
        new_n = 2 * int(best_n)
        if new_n > 640:
            new_n = 640

        if new_n > best_n:
            h_rep = upsample_h(best_h, new_n, "rep")
            c_rep = make_c(h_rep)
            cst_rep = float(np.max(c_rep))

            h_lin = upsample_h(best_h, new_n, "linear")
            c_lin = make_c(h_lin)
            cst_lin = float(np.max(c_lin))

            if min(cst_rep, cst_lin) <= best_c + 0.08:
                if cst_rep <= cst_lin:
                    h_hi = h_rep
                else:
                    h_hi = h_lin

                best_h, best_c, best_n = evaluate_candidate(h_hi, best_h, best_c, best_n)

                if hi_deadline > time.time() + 60.0:
                    h_ht, c_ht, best_ht, _ = hill_climb(
                        h_hi.copy(), 0.5 * float(new_n), hi_deadline, seed + 3031
                    )
                    best_h, best_c, best_n = evaluate_candidate(best_ht, best_h, best_c, best_n)
                    best_h, best_c, best_n = evaluate_candidate(h_ht, best_h, best_c, best_n)

    # Short randomized restarts / re-polish at the current best resolution.
    if best_n <= 640 and time.time() < global_deadline - 40.0:
        rng_restart = np.random.RandomState(seed + 4041)
        for si, scale in enumerate((0.01, 0.025, 0.05)):
            if time.time() >= global_deadline - 45.0:
                break
            rem = global_deadline - 45.0 - time.time()
            rtime = min(75.0, 0.55 * rem, 0.08 * max_time)
            if rtime < 22.0:
                break

            h_p = perturb_h(best_h, 0.5 * float(best_n), scale, 0.25, rng_restart)
            best_h, best_c, best_n = evaluate_candidate(h_p, best_h, best_c, best_n)

            r_deadline = time.time() + rtime
            if r_deadline > time.time() + 10.0:
                h_pr, c_pr, best_pr, _ = hill_climb(
                    h_p.copy(), 0.5 * float(best_n), r_deadline, seed + 5000 + 17 * si
                )
                best_h, best_c, best_n = evaluate_candidate(best_pr, best_h, best_c, best_n)
                best_h, best_c, best_n = evaluate_candidate(h_pr, best_h, best_c, best_n)

    # Final direct polish.
    if best_n <= 640 and time.time() < global_deadline - 10.0:
        f_deadline = min(global_deadline - 5.0, time.time() + min(60.0, 0.05 * max_time))
        if f_deadline > time.time() + 10.0:
            h_f, c_f, best_f, _ = hill_climb(
                best_h.copy(), 0.5 * float(best_n), f_deadline, seed + 6061
            )
            best_h, best_c, best_n = evaluate_candidate(best_f, best_h, best_c, best_n)
            best_h, best_c, best_n = evaluate_candidate(h_f, best_h, best_c, best_n)

    # Final feasibility cleanup, only if needed.
    target_best = 0.5 * float(best_n)
    if (abs(float(np.sum(best_h)) - target_best) > 1e-10 or
            np.any(best_h < -1e-12) or np.any(best_h > 1.0 + 1e-12)):
        best_h = project_feasible(best_h, target_best)

    c_best = make_c(best_h)
    best_c = float(np.max(c_best))

    # Cross-check with the pre-imported evaluator if available.
    try:
        ev = evaluate_erdos_solution(best_h)
        if ev is not None:
            best_c = min(best_c, float(ev))
    except Exception:
        pass

    return best_h, float(best_c), int(best_n)
