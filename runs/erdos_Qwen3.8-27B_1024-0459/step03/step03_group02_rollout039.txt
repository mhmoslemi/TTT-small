                    candidates.append(i)
        elif worst_d < 0:

                if i - worst_d >= 0 and h[i-worst_d] == 0:
                    candidates.append(i)
        
        if not candidates:

            if not ones or not zeros: break
            candidates = [random choice from ones]

        n_try = min(len(candidates), 30)
        selected = random subset of candidates, size n_try

        best_new_max = current_max
        
        for i in selected:
            # For this i, try a few j's
            n_j_try = min(len(zeros), 15)
            js = random subset of zeros, size n_j_try
            
            for j in js:
                # Compute new correlation after swap
                h[i] = 0; h[j] = 1
                new_corr = compute_corr(h)
                new_max = max(new_corr)
                h[i] = 1; h[j] = 0

            i, j, new_corr = best_move
            h[i] = 0; h[j] = 1
            ones.remove(i); ones.add(j)
            zeros.remove(j); zeros.add(i)
            corr = new_corr
            current_max = best_new_max
            stale = 0
            if current_max < best[1]:
                best = (h.copy(), current_max)
        else:
            # SA: accept random move with probability
            if ones and zeros:
                i = random from ones
                j = random from zeros
                h[i] = 0; h[j] = 1
                new_corr = compute_corr(h)
                new_max = max(new_corr)

            T *= cooling

def project_feasible(h, n):

    target = n / 2.0

def lag_gradient(h, n, offset):
    grad = np.zeros(n)
    if offset >= 0:
        n_pairs = n - offset
        if n_pairs <= 0:
            return grad
        grad[offset:offset + n_pairs] += (1.0 - h[:n_pairs])
        grad[:n_pairs] -= h[offset:offset + n_pairs]
    else:
        abs_off = -offset
        n_pairs = n - abs_off
        if n_pairs <= 0:
            return grad
        grad[:n_pairs] += (1.0 - h[abs_off:abs_off + n_pairs])
        grad[abs_off:abs_off + n_pairs] -= h[:n_pairs]
    return grad

def grad_descent_refine(h, n, max_iters=500, lr=0.01):
    """Gradient descent on the smooth max, with projection."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    corr = compute_corr(h)

    for it in range(max_iters):
        # Compute weighted gradient focusing on top lags
        max_corr = np.max(corr)
        # Weights: exponential decay from max
        weights = np.exp(-20.0 * (max_corr - corr))
        total_w = np.sum(weights)
        if total_w < 1e-15:
            break
        weights /= total_w

        grad = np.zeros(n)
        for k_idx in range(len(corr)):
            if weights[k_idx] > 0.01:
                offset = k_idx - (n - 1)
                grad += weights[k_idx] * lag_gradient(h, n, offset)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        grad /= gnorm
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            corr = compute_corr(h)
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break

    return best_h, best_c5

        n = best_n
        for beta in [3000, 8000]:
            if remaining_time() < 5:
                break
            h_new = slsqp_refine(best_h, n, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 7: One more round of binary search from best (in case continuous broke structure)
    if remaining_time() > 20:
        h_bin = np.round(best_h)
        n = best_n
        ones_cnt = int(np.sum(h_bin > 0.5))
        target = n // 2
        if ones_cnt > target:
            idx = np.where(h_bin > 0.5)[0]
            for i in rng.choice(idx, size=ones_cnt - target, replace=False):
                h_bin[i] = 0.0
        elif ones_cnt < target:
            idx = np.where(h_bin < 0.5)[0]
            for i in rng.choice(idx, size=target - ones_cnt, replace=False):
                h_bin[i] = 1.0

        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5 + 0.01:
            h_ls, c5_ls = binary_local_search(h_bin, n, rng,
                                               max_iters=15000, t0=0.02, cooling=0.9998)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
                # Quick continuous refine
                if remaining_time() > 10:
                    h_cr, c5_cr = grad_descent_refine(h_ls, n, max_iters=80, lr=0.005)
                    if c5_cr < best_c5:
                        best_c5 = c5_cr
                        best_h = h_cr.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
