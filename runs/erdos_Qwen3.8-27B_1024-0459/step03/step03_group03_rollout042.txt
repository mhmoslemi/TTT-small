def _subgrad_descent(h, n, nit, topk=3, lr_start=0.01):
    """Subgradient descent on the max objective."""
    best = h.copy()
    bc = _c5(best)
    lr = lr_start
    
    for it in range(nit):
        if _tl() < 5:
            break
        
        corr = _corr(best)
        idx = np.argsort(corr)[::-1][:topk]
        g = np.zeros(n)
        for k in idx:
            g += _grad_k(best, int(k))
        g /= len(idx)
        
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        
        # Line search
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            step = lr * factor
            h2 = _proj(best - step * g, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-15:
                best = h2
                bc = c2
                improved = True
                break
        
        if not improved:
            lr *= 0.7
            if lr < 1e-6:
                break
    
    return best, bc

def _binary_local(h, n, max_iter=500):
    """Local search: try to make h more binary and improve C5."""
    best = _proj(h, n)
    bc = _c5(best)
    rng = np.random.default_rng(42)
    
    for _ in range(max_iter):
        if _tl() < 5:
            break
        improved = False
        
        # Find positions that are "in between" (not close to 0 or 1)
        mid = (best > 0.05) & (best < 0.95)
        mid_idx = np.where(mid)[0]
        
        if len(mid_idx) > 0:
            # Try to push boundary values to 0 or 1
            for idx in mid_idx:
                if _tl() < 3:
                    break
                for target in [0.0, 1.0]:
                    h2 = best.copy()
                    h2[idx] = target
                    h2 = _proj(h2, n)
                    c2 = _c5(h2)
                    if c2 < bc - 1e-15:
                        best = h2
                        bc = c2
                        improved = True
                        break
        
        if not improved and len(mid_idx) == 0:
            # All values are binary-like; try swaps
            ones = np.where(best > 0.5)[0]
            zeros = np.where(best < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            
            ntrials = min(500, len(ones) * len(zeros))
            for _ in range(ntrials):
                if _tl() < 2:
                    break
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
                h2 = best.copy()
                h2[i], h2[j] = 0.0, 1.0
                h2 = _proj(h2, n)
                c2 = _c5(h2)
                if c2 < bc - 1e-15:
                    best = h2
                    bc = c2
                    improved = True
                    ones = np.where(best > 0.5)[0]
                    zeros = np.where(best < 0.5)[0]
        
        if not improved:
            break
    
    return best, bc

def _refine_slqp(h, n, maxiter=200, beta=100):
    """SLSQP with log-sum-exp smoothing of the max."""
    global _ng, _bv
    
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mx = float(np.max(c))
        return (mx + np.log(np.exp(beta * (c - mx)).sum()) / beta) * (2.0 / _ng)
    
    def _obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mx = float(np.max(c))
        w = np.exp(beta * (c - mx))
        sw = w.sum()
        w /= sw
        g = np.zeros(_ng)
        for k_idx in range(len(c)):
            if w[k_idx] > 1e-10:
                g += w[k_idx] * _grad_k(x, k_idx)
        return g * (2.0 / _ng)
    
    def _con(x):
        return x.sum() - 0.5 * _ng
    
    _ng = n
    _bv = beta
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(_obj, h, jac=_obj_grad, method='SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': _con}],
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        result = _proj(np.asarray(res.x, dtype=np.float64), n)
        return result
    except Exception:
        return _proj(h.copy(), n)

def _interp(h, ns, nd):
    if ns == nd:
        return h.copy()
    xs = (np.arange(ns) + 0.5) * (2.0 / ns)
    xd = (np.arange(nd) + 0.5) * (2.0 / nd)
    return np.interp(xd, xs, h)
