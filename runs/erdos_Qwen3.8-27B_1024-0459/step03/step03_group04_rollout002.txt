def project_box_sum(v,n):
   target=n/2.0
   v=np.asarray(v,dtype=float).copy()
   # If v has nan, fallback
   if not np.all(np.isfinite(v)): v=np.clip(v,0,1); ...
   lo = np.min(v) - 1.0
   hi = np.max(v) + 1.0
   # binary search
   for _ in range(80):
      mid=(lo+hi)*0.5
      x=np.clip(v-mid,0.0,1.0)
      s=x.sum()
      if s > target: lo=mid
      else: hi=mid
   x=np.clip(v-mid,0.0,1.0)
   resid=target-x.sum()
   if resid != 0.0:
      if resid > 0:
         cand=np.where(x < 1.0 - 1e-14)[0]
      else:
         cand=np.where(x > 1e-14)[0]
      if len(cand)>0:
         # distribute residual with capacity
         if resid > 0:
            caps=1.0-x[cand]
         else:
            caps=x[cand]
         # Could use one index
         # To maintain box, distribute greedily
         rem=resid
         for idx in cand[np.argsort(-caps) if resid>0 else cand[np.argsort(caps)]]:
             if resid>0: add=min(rem,1-x[idx])
             else: add=max(rem,x[idx])
             x[idx]+=add; rem-=add
             if abs(rem)<1e-15: break
      # if still, small adjust first
      if abs(target-x.sum())>1e-12:
         x += (target-x.sum())/n
         x=np.clip(x,0,1)
   return x

def weighted_gradient(h, offs, weights):
    n = len(h)
    g = np.zeros(n, dtype=float)
    om = 1.0 - h
    for idx, off in enumerate(offs):
        w = weights[idx]
        if w == 0.0: continue
        if off > 0:
            l = n - off
            if l > 0:
                g[:l] += w * om[off:]
                g[off:] -= w * h[:l]
        elif off < 0:
            a = -off
            l = n - a
            if l > 0:
                g[a:] += w * om[:l]
                g[:l] -= w * h[a:]
        else:
            g += w * (1.0 - 2.0 * h)
    return g
```

def select_offsets(vals, top_m):
    m = len(vals)
    if top_m is None or top_m >= m:
        return np.arange(m, dtype=int)
    tm = int(top_m)
    if tm < 8: tm = 8
    if tm > m: tm = m
    order = np.argsort(vals)[::-1]
    top = order[:tm]
    mx = float(vals[order[0]])
    tol = max(1e-9, 1e-7 * max(1.0, abs(mx)))
    near = np.where(vals >= mx - tol)[0]
    combined = np.unique(np.concatenate((top, near)))
    if combined.size > tm * 2:
        idx = np.argsort(vals[combined])[::-1][:tm]
        combined = combined[idx]
    return combined

def project_box_sum(v, n):
    n = int(n)
    x = np.asarray(v, dtype=float).reshape(-1).copy()
    if x.size != n:
        tmp = np.full(n, 0.5, dtype=float)
        m = min(n, x.size)
        tmp[:m] = x[:m]
        x = tmp

    if not np.all(np.isfinite(x)):
        x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)

    target = 0.5 * float(n)

    x = np.clip(x, 0.0, 1.0)
    s = float(np.sum(x))
    if abs(s - target) < 1e-14:
        return x

    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    tau = 0.5 * (lo + hi)

    for _ in range(70):
        tau = 0.5 * (lo + hi)
        y = np.clip(x - tau, 0.0, 1.0)
        s = float(np.sum(y))
        if s > target:
            lo = tau
        else:
            hi = tau

    x = np.clip(x - tau, 0.0, 1.0)

    diff = target - float(np.sum(x))
    if diff > 1e-14:
        cand = np.where(x < 1.0 - 1e-15)[0]
        if cand.size:
            order = np.argsort(1.0 - x[cand])[::-1]
            rem = diff
            for pos in order:
                idx = cand[pos]
                add = min(rem, 1.0 - x[idx])
                if add > 0.0:
                    x[idx] += add
                    rem -= add
                if rem <= 1e-14:
                    break
    elif diff < -1e-14:
        cand = np.where(x > 1e-15)[0]
        if cand.size:
            order = np.argsort(x[cand])[::-1]
            rem = -diff
            for pos in order:
                idx = cand[pos]
                sub = min(rem, x[idx])
                if sub > 0.0:
                    x[idx] -= sub
                    rem -= sub
                if rem <= 1e-14:
                    break

    return float(np.max(corr) * dx)


def values_from_offsets(h, offs):

        else:

def weighted_gradient(h, offs, weights):
    arr = np.asarray(h, dtype=float).reshape(-1)
    n = arr.size
    g = np.zeros(n, dtype=float)
    om = 1.0 - arr
    offs = np.asarray(offs, dtype=int).reshape(-1)
    wts = np.asarray(weights, dtype=float).reshape(-1)
    m = min(offs.size, wts.size)

    for i in range(m):
        off = int(offs[i])
        w = float(wts[i])
        if w == 0.0:
            continue

        if off > 0:
            l = n - off
            if l > 0:
                g[:l] += w * om[off:off + l]
                g[off:] -= w * arr[:l]
        elif off < 0:
            a = -off
            l = n - a
            if l > 0:
                g[a:] += w * om[:l]
                g[:l] -= w * arr[a:a + l]
        else:
            g += w * (1.0 - 2.0 * arr)

    return g

def compute_smooth(x):

            S = 1.0
        weights = exps / S
        F = _g_dx * (mx + float(np.log(S)) / b)

    G = _g_dx * weighted_gradient(arr, offs, weights)

    _g_cache_x = arr.copy()
    _g_cache_F = float(F)
    _g_cache_G = G
    return _g_cache_F, _g_cache_G


def smooth_fun(x):
    F, _ = compute_smooth(x)
    return F


def smooth_jac(x):
    _, G = compute_smooth(x)
    return G


def con_sum(x):
    return float(np.sum(np.asarray(x, dtype=float).reshape(-1))) - 0.5 * float(_g_n)


def con_jac(x):
    n = len(np.asarray(x).reshape(-1))
    return np.ones((1, n), dtype=float)


def select_offsets(vals, top_m):
    m = int(vals.size)
    if top_m is None or top_m >= m:
        return np.arange(m, dtype=int)

    tm = int(top_m)
    if tm < 8:
        tm = 8
    if tm > m:
        tm = m

    order = np.argsort(vals)[::-1]
    top = order[:tm]
    mx = float(vals[order[0]])
    tol = max(1e-9, 1e-7 * max(1.0, abs(mx)))
    near = np.where(vals >= mx - tol)[0]

    combined = np.unique(np.concatenate((top, near)))
    if combined.size > 2 * tm:
        idx = np.argsort(vals[combined])[::-1][:tm]
        combined = combined[idx]

    return combined


def make_binary_random(n, rng):
    n = int(n)
    h = np.zeros(n, dtype=float)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h


def make_thue_morse(n):
    n = int(n)
    idx = np.arange(n, dtype=np.int64)
    p = np.zeros(n, dtype=np.int8)
    b = 0
    while (1 << b) <= max(1, n):
        p ^= (idx >> b) & 1
        b += 1
    return (p & 1).astype(float)


def make_golden(n):
    n = int(n)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    t = np.arange(n, dtype=float)
    frac = (t * gr) % 1.0
    return (np.abs(frac - 0.5) < 0.25).astype(float)


def make_sine(n, f, ph):
    n = int(n)
    t = np.arange(n, dtype=float) / float(n)
    return 0.5 + 0.45 * np.sin(2.0 * np.pi * float(f) * t + float(ph))


def make_ramp(n):
    n = int(n)
    t = np.arange(n, dtype=float) / float(n)
    return np.abs(2.0 * t - 1.0)


def make_lowpass_random(n, rng, win):
    n = int(n)
    if win is None:
        win = 7
    win = int(win)
    if win < 3:
        win = 3
    if win > n:
        win = n

    h = rng.rand(n)
    ker = np.ones(win, dtype=float) / float(win)
    h = np.convolve(h, ker, mode="same")

    mean = float(np.mean(h))
    std = float(np.std(h))
    if std > 1e-12:
        h = 0.5 + 0.85 * (h - mean) / std
    else:
        h = np.full(n, 0.5, dtype=float)

    return h


def make_fourier_random(n, rng):
    n = int(n)
    m = min(max(2, n // 20), 18)
    t = np.arange(n, dtype=float) / float(n)
    s = np.zeros(n, dtype=float)

    for f in range(1, m + 1):
        amp = 1.0 / (float(f) ** 0.7)
        a = rng.normal(0.0, amp)
        b = rng.normal(0.0, amp)
        s += a * np.cos(2.0 * np.pi * float(f) * t)
        s += b * np.sin(2.0 * np.pi * float(f) * t)

    std = float(np.std(s))
    if std > 1e-12:
        s = 0.5 + 0.8 * s / std
    else:
        s = 0.5 + 0.4 * np.sin(2.0 * np.pi * t)

    return s


def smooth_pattern(h, win):
    arr = np.asarray(h, dtype=float).reshape(-1)
    n = arr.size
    w = int(win)
    if w < 3 or n < 4:
        return arr.copy()
    if w > n:
        w = n
    if w % 2 == 0:
        w -= 1
    ker = np.ones(w, dtype=float) / float(w)
    out = np.convolve(arr, ker, mode="same")
    return project_box_sum(out, n)


def process_candidate(cands, h, n):
    n = int(n)
    if n <= 0:
        return None

    arr = np.asarray(h, dtype=float).reshape(-1)
    if arr.size != n:
        tmp = np.full(n, 0.5, dtype=float)
        m = min(n, arr.size)
        tmp[:m] = arr[:m]
        arr = tmp

    ph = project_box_sum(arr, n)
    c = eval_exact(ph)
    if np.isfinite(c):
        cands.append((float(c), n, ph))
    return c


def add_interp_candidate(cands, src, src_n, nt):
    ht = interp_h(src, src_n, nt)
    process_candidate(cands, ht, nt)


def get_c5(c):
    return c[0]


def polish_slsqp(start_h, n, betas, maxiter, t0, budget, active_m=None):
    n = int(n)
    best_h = project_box_sum(start_h, n)
    best_c5 = eval_exact(best_h)

    if time_left(t0, budget) <= 20.0:
        return best_h, best_c5

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_jac}]

    for beta in betas:
        rem = time_left(t0, budget)
        if rem < 25.0:
            break

        x0 = project_box_sum(best_h, n)

        if active_m is None:
            set_problem(n, beta, None)
        else:
            all_offs = all_offsets(n)
            idxs = select_offsets(values_from_offsets(x0, all_offs), int(active_m))
            set_problem(n, beta, all_offs[idxs])

        if n <= 200:
            ips = 6.0
        elif n <= 400:
            ips = 3.0
        else:
            ips = 1.5

        mi_limit = int(max(30, rem * ips * 0.7))
        mi = min(int(maxiter), mi_limit)
        if mi < 30:
            break

        try:
            res = minimize(
                smooth_fun,
                x0,
                method='SLSQP',
                jac=smooth_jac,
                bounds=bounds,
                constraints=cons,
                options={'maxiter': mi, 'ftol': 1e-16, 'iprint': 0}
            )
            if hasattr(res, 'x') and res.x is not None:
                cand = project_box_sum(res.x, n)
                c = eval_exact(cand)
                if c < best_c5 - 1e-15:
                    best_h = cand.copy()
                    best_c5 = c
        except Exception:
            pass

    return best_h, best_c5


def pg_polish(start_h, n, t0, budget, max_iters=1000, active_m=None, beta=300.0):
    n = int(n)
    h = project_box_sum(start_h, n)
    c5 = eval_exact(h)

    if time_left(t0, budget) <= 10.0:
        return h, c5

    all_offs = all_offsets(n)
    if active_m is None or active_m >= 2 * n - 1:
        active = all_offs.copy()
    else:
        active = all_offs[select_offsets(values_from_offsets(h, all_offs), int(active_m))]

    set_problem(n, beta, active)

    step = 0.1
    no_improve = 0

    for it in range(int(max_iters)):
        rem = time_left(t0, budget)
        if rem < 5.0:
            break

        F, grad = compute_smooth(h)
        gnorm = float(np.linalg.norm(grad))

        if gnorm < 1e-14:
            no_improve += 10
        else:
            accepted = False
            for factor in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125, 0.00390625):
                st = step * factor
                ht = project_feasible(h - st * grad, n)
                Ftry, _ = compute_smooth(ht)
                if Ftry < F - 1e-14:
                    h = ht
                    F = Ftry
                    accepted = True
                    if factor == 1.0:
                        step = min(0.5, st * 1.15)
                    elif factor >= 0.5:
                        step = st
                    else:
                        step = st * 1.2
                    no_improve = 0
                    break

            if not accepted:
                step *= 0.5
                no_improve += 1
                if step < 1e-10:
                    no_improve += 60

        if it % 25 == 0:
            ce = eval_exact(h)
            if ce < c5:
                c5 = ce

        if it % 40 == 0:
            if active_m is None or active_m >= 2 * n - 1:
                active = all_offs.copy()
            else:
                idxs = select_offsets(values_from_offsets(h, all_offs), int(active_m))
                active = all_offs[idxs]
            set_problem(n, beta, active)
            F, grad = compute_smooth(h)

        if no_improve > 180:
            break

    ce = eval_exact(h)
    if ce < c5:
        c5 = ce
    return h, c5


def subgrad_polish(start_h, n, t0, budget, max_iters=200):
    n = int(n)
    h = project_box_sum(start_h, n)
    c5 = eval_exact(h)

    if time_left(t0, budget) <= 10.0:
        return h, c5

    all_offs = all_offsets(n)
    no_improve = 0

    for _ in range(int(max_iters)):
        rem = time_left(t0, budget)
        if rem < 5.0:
            break

        vals = values_from_offsets(h, all_offs)
        mx = float(np.max(vals))
        tol = max(1e-9, 1e-8 * max(1.0, abs(mx)))
        near = np.where(vals >= mx - tol)[0]
        if near.size == 0:
            near = np.argsort(vals)[-5:]

        top_m = min(120, max(10, n // 4))
        if near.size > top_m:
            near = near[np.argsort(vals[near])[::-1][:top_m]]

        offs = all_offs[near]
        w = np.full(near.size, 1.0 / float(near.size), dtype=float)
        g = weighted_gradient(h, offs, w)

        accepted = False
        for st in (1e-5, 5e-5, 2e-4, 8e-4, 3e-3, 1e-2, 4e-2):
            ht = project_feasible(h - st * g, n)
            ct = eval_exact(ht)
            if ct < c5 - 1e-14:
                h = ht
                c5 = ct
                accepted = True
                no_improve = 0
                break

        if not accepted:
            no_improve += 1
            if no_improve > 35:
                break

    return h, c5


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    rng = np.random.RandomState(int(seed))

    best_h = None
    best_n = 160
    best_c5 = float("inf")

    try:
        raw = np.asarray(initial_h_values, dtype=float).reshape(-1)
        if raw.size > 0:
            bn = int(raw.size)
            bh = project_box_sum(raw, bn)
            bc = eval_exact(bh)
            if np.isfinite(bc):
                best_h = bh
                best_n = bn
                best_c5 = bc
    except Exception:
        best_h = None

    if best_h is None:
        fallback = np.zeros(160, dtype=float)
        fallback[80:160] = 1.0
        best_h = project_box_sum(fallback, 160)
        best_n = 160
        best_c5 = eval_exact(best_h)

    cands = []
    process_candidate(cands, best_h, best_n)

    base = int(best_n)
    generated = set([base])

    lower = max(48, base - 24)
    upper = min(640, base + 24)
    for nt in range(lower, upper + 1):
        if nt in generated:
            continue
        if time_left(t0, budget) < 4.0:
            break
        add_interp_candidate(cands, best_h, best_n, nt)
        generated.add(nt)

    fixed_ns = [64, 80, 96, 112, 128, 144, 160, 176, 192, 208, 224, 256, 288, 320, 352, 384, 416, 448, 480]
    for nt in fixed_ns:
        if nt in generated:
            continue
        if not (48 <= nt <= 640):
            continue
        if time_left(t0, budget) < 4.0:
            break
        add_interp_candidate(cands, best_h, best_n, nt)
        generated.add(nt)

    struct_ns = []
    for nt in [base, base + 32, base + 96, 192, 256, 320, 384]:
        if 80 <= nt <= 500 and nt not in struct_ns:
            struct_ns.append(nt)

    for nt in struct_ns:
        if time_left(t0, budget) < 30.0:
            break
        process_candidate(cands, make_thue_morse(nt), nt)
        process_candidate(cands, make_golden(nt), nt)
        process_candidate(cands, make_binary_random(nt, rng), nt)
        process_candidate(cands, make_lowpass_random(nt, rng, 5 + nt // 60), nt)
        process_candidate(cands, make_fourier_random(nt, rng), nt)
        process_candidate(cands, make_ramp(nt), nt)
        for f in (2, 3, 5, 7):
            process_candidate(cands, make_sine(nt, f, 0.0), nt)
            process_candidate(cands, make_sine(nt, f, 1.0), nt)

    if cands:
        cands.sort(key=get_c5)
        cc, cn, ch = cands[0]
        if cc < best_c5:
            best_c5 = float(cc)
            best_n = int(cn)
            best_h = np.array(ch, dtype=float).copy()

    # Light polishing of a few promising alternative resolutions.
    max_quick = 0
    rem = time_left(t0, budget)
    if rem > 900.0:
        max_quick = 2
    elif rem > 600.0:
        max_quick = 1

    quick_done = set()
    for item in cands:
        if len(quick_done) >= max_quick:
            break
        cc, cn, ch = item
        if cn == best_n:
            continue
        if cc > best_c5 + 0.02:
            continue
        if time_left(t0, budget) < 90.0:
            break

        quick_done.add(int(cn))
        if cn <= 600:
            active_m = None
            maxiter = 140 if cn <= 250 else 100
            betas = (60, 200)
        else:
            active_m = min(60, max(30, cn // 8))
            maxiter = 80
            betas = (60, 200)

        h2, c2 = polish_slsqp(ch, cn, betas, maxiter, t0, budget, active_m=active_m)
        if c2 < best_c5:
            best_c5 = float(c2)
            best_n = int(cn)
            best_h = np.array(h2, dtype=float).copy()

    # Deep polish the current best resolution.
    if time_left(t0, budget) > 160.0:
        if best_n <= 200:
            betas = (40, 150, 400, 1000, 2000)
            maxiter = 600
            active_m = None
        elif best_n <= 350:
            betas = (40, 150, 400, 1200)
            maxiter = 420
            active_m = None
        elif best_n <= 650:
            betas = (50, 200, 700, 1500)
            maxiter = 260
            active_m = None
        else:
            betas = (50, 200, 700)
            maxiter = 160
            active_m = min(80, max(40, best_n // 8))

        h2, c2 = polish_slsqp(best_h, best_n, betas, maxiter, t0, budget, active_m=active_m)
        if c2 < best_c5:
            best_c5 = float(c2)
            best_n = int(len(h2))
            best_h = np.array(h2, dtype=float).copy()

    # If time remains, try one more nearby candidate.
    rem = time_left(t0, budget)
    if rem > 260.0 and cands:
        done = set([int(best_n)])
        tried = 0
        for item in cands:
            cc, cn, ch = item
            if int(cn) in done:
                continue
            if cc > best_c5 + 0.02:
                continue
            done.add(int(cn))
            tried += 1

            if cn <= 300:
                betas = (50, 200, 700)
                maxiter = 260
                active_m = None
            elif cn <= 600:
                betas = (50, 200, 700)
                maxiter = 170
                active_m = None
            else:
                betas = (50, 200, 700)
                maxiter = 110
                active_m = min(80, max(40, cn // 8))

            h2, c2 = polish_slsqp(ch, cn, betas, maxiter, t0, budget, active_m=active_m)
            if c2 < best_c5:
                best_c5 = float(c2)
                best_n = int(cn)
                best_h = np.array(h2, dtype=float).copy()
                done.add(int(cn))

            if tried >= 1:
                break

    # Light smoothing sweep.
    if time_left(t0, budget) > 20.0:
        for win in (3, 5, 7):
            hs = smooth_pattern(best_h, win)
            hc = eval_exact(hs)
            if hc < best_c5:
                best_h = np.array(hs, dtype=float).copy()
                best_c5 = float(hc)

    # Projected-gradient smoothing.
    if time_left(t0, budget) > 100.0:
        if best_n <= 700:
            active_m = None
        else:
            active_m = min(80, max(40, best_n // 8))
        h2, c2 = pg_polish(best_h, best_n, t0, budget, max_iters=900, active_m=active_m, beta=300.0)
        if c2 < best_c5:
            best_c5 = float(c2)
            best_n = int(len(h2))
            best_h = np.array(h2, dtype=float).copy()

    # Exact active-set subgradient polish.
    if time_left(t0, budget) > 40.0:
        maxit = 180 if best_n <= 200 else (90 if best_n <= 400 else 50)
        h2, c2 = subgrad_polish(best_h, best_n, t0, budget, max_iters=maxit)
        if c2 < best_c5:
            best_c5 = float(c2)
            best_n = int(len(h2))
            best_h = np.array(h2, dtype=float).copy()

    # Final resolution micro-sweep around the best optimized function.
    if time_left(t0, budget) > 20.0:
        improved = False
        best_nt = best_n
        best_ht = best_h
        best_ct = best_c5
        for delta in range(-10, 11):
            nt = best_n + delta
            if nt < 48 or nt > 700:
                continue
            ht = interp_h(best_h, best_n, nt)
            ht = project_box_sum(ht, nt)
            ct = eval_exact(ht)
            if ct < best_ct:
                best_ct = ct
                best_nt = nt
                best_ht = ht.copy()
                improved = True

        if improved:
            best_n = int(best_nt)
            best_h = np.array(best_ht, dtype=float).copy()
            best_c5 = float(best_ct)

            if time_left(t0, budget) > 90.0:
                if best_n <= 300:
                    betas = (80, 300)
                    maxiter = 120
                    active_m = None
                elif best_n <= 600:
                    betas = (80, 300)
                    maxiter = 90
                    active_m = None
                else:
                    betas = (80, 300)
                    maxiter = 60
                    active_m = min(70, max(35, best_n // 8))

                h2, c2 = polish_slsqp(best_h, best_n, betas, maxiter, t0, budget, active_m=active_m)
                if c2 < best_c5:
                    best_c5 = float(c2)
                    best_n = int(len(h2))
                    best_h = np.array(h2, dtype=float).copy()

    best_h = project_box_sum(best_h, best_n)
    final_c5 = eval_exact(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if np.isfinite(ec5) and float(ec5) < best_c5:
            best_c5 = float(ec5)
    except Exception:
        pass

    return (best_h, float(best_c5), int(best_n))
