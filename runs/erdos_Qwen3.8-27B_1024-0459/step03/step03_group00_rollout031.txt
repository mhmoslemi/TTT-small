def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr)) * dx


def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    return m + np.log(np.sum(exps)) / beta

        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        cnt = int(np.sum(mask))
        if cnt > 0:

        else:
