def _slsqp_opt(h, n, beta, maxiter):
    """SLSQP with softmax approximation of max."""
    global _ng, _bv
    _ng = n
    _bv = beta
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mx = float(np.max(c))
        b = _bv
        sh = c - mx
        # Clamp to avoid overflow
        sh = np.clip(sh, -50.0/b, 0)
        return (mx + np.log(np.exp(b * sh).sum()) / b) * (2.0 / n)
    
    def cons(x):
        return x.sum() - 0.5 * n
    
    try:
        r = minimize(obj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': cons}],
                     options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return _proj(np.asarray(r.x, dtype=np.float64), n)
    except Exception:
        return _proj(h.copy(), n)

    """Interpolate from n_from to n_to grid points."""

    x_new = (np.arange(n_to) + 0.5) * (2.0 / n_to)
    return np.interp(x_new, xs, h)

def _tl():
    return max(0.0, _tb - (time.time() - _t0) - 5.0)

def _c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / n)

def _proj(h, n):
    h = np.array(h, dtype=np.float64)
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(200):
        d = 0.5 * n - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        k = int(np.sum(m))
        if k > 0:
            h[m] += d / k
        else:
            h[:] = 0.5
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _fix_binary(h, n, rng):
    """Ensure h is binary with exactly n/2 ones."""
    h = (h > 0.5).astype(np.float64)
    s = int(h.sum())
    target = n // 2
    if s > target:
        ones = np.where(h == 1)[0]
        for _ in range(s - target):
            i = ones[rng.randint(len(ones))]
            h[i] = 0

        zeros = np.where(h == 0)[0]

            j = zeros[rng.randint(len(zeros))]

def _grad_at_lag(h, m):
    """Gradient of correlation at shift m w.r.t. h.
    m is the shift: corr(m) = sum appropriate h[i]*(1-h[i+m or i-m])
    """
    n = len(h)
    g = np.zeros(n)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif m > 0:
        # corr = sum_{i=m}^{n-1} h[i] * (1-h[i-m])
        g[m:] += (1.0 - h[:n-m])
        g[:n-m] -= h[m:]
    else:
        p = -m
        # corr = sum_{i=0}^{n-1-p} h[i] * (1-h[i+p])
        g[:n-p] += (1.0 - h[p:])
        g[p:] -= h[:n-p]
    return g

def _multi_grad(h, n, beta):
    """Smoothed max gradient using softmax over correlations."""
    c = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(c))
    w = np.exp(-beta * (mx - c))
    sw = float(w.sum())
    w /= sw
    
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] > 1e-12:
            m = k - (n - 1)
            g += w[k] * _grad_at_lag(h, m)
    return g

def _proj_grad(h, n, max_iters, rng):

    best = h.copy()
    bc5 = _c5(best)

        if _tl() < 3:
            break

        beta = betas[it % len(betas)]
        g = _multi_grad(best, n, beta)
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            break
        g /= gn
        
        moved = False
        for a in steps:
            h2 = _proj(best - a * g, n)
            c2 = _c5(h2)
            if c2 < bc5 - 1e-15:
                best = h2
                bc5 = c2
                moved = True
                break

            kmax = int(np.argmax(c))
            m = kmax - (n - 1)
            g2 = _grad_at_lag(best, m)
            gn2 = float(np.linalg.norm(g2))

def _slsqp(h0, n, beta, maxiter):
    """SLSQP optimization with softmax-max objective."""
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mx = float(np.max(c))
        sh = c - mx
        sh = np.clip(sh, -100.0, 0.0)
        logsum = mx + np.log(np.exp(beta * sh).sum()) / beta
        return logsum * (2.0 / n)
    
    def cons(x):
        return x.sum() - 0.5 * n
    
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mx = float(np.max(c))
        sh = c - mx
        sh = np.clip(sh, -100.0, 0.0)
        w = np.exp(beta * sh)
        sw = float(w.sum())
        w /= sw
        g = np.zeros(n)
        for k in range(len(c)):
            if w[k] > 1e-10:
                m = k - (n - 1)
                g += w[k] * _grad_at_lag(x, m)
        return g * (2.0 / n)
    
    try:
        r = minimize(obj, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': cons, 'jac': lambda x: np.ones(n)}],
                     options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return _proj(np.asarray(r.x, dtype=np.float64), n)
    except Exception:
        try:
            r = minimize(obj, h0, method='SLSQP',
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq', 'fun': cons}],
                         options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
            return _proj(np.asarray(r.x, dtype=np.float64), n)
        except Exception:
            return _proj(h0.copy(), n)

import numpy as np
from scipy.optimize import minimize
import time

_t0 = 0.0
_tb = 1300.0


def _tl():
    return max(0.0, _tb - (time.time() - _t0) - 4.0)


def _c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / n)


def _proj(h, n):
    h = np.array(h, dtype=np.float64)
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(200):
        d = 0.5 * n - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        k = int(np.sum(m))
        if k > 0:
            h[m] += d / k
        else:
            h[:] = 0.5
        np.clip(h, 0.0, 1.0, out=h)
    d = 0.5 * n - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] + d)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + d)
    return h


def _fix_binary(h, n, rng):
    h = (h > 0.5).astype(np.float64)
    s = int(h.sum())
    target = n // 2
    if s > target:
        ones = np.where(h == 1)[0]
        for _ in range(s - target):
            i = ones[rng.randint(len(ones))]
            h[i] = 0
            ones = np.where(h == 1)[0]
    elif s < target:
        zeros = np.where(h == 0)[0]
        for _ in range(target - s):
            j = zeros[rng.randint(len(zeros))]
            h[j] = 1
            zeros = np.where(h == 0)[0]
    return h


def _grad_at_lag(h, m):
    n = len(h)
    g = np.zeros(n)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif m > 0:
        g[m:] += (1.0 - h[:n - m])
        g[:n - m] -= h[m:]
    else:
        p = -m
        g[:n - p] += (1.0 - h[p:])
        g[p:] -= h[:n - p]
    return g


def _multi_grad(h, n, beta):
    c = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(c))
    sh = c - mx
    np.clip(sh, -80.0 / beta, 0.0, out=sh)
    w = np.exp(beta * sh)
    sw = float(w.sum())
    if sw < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        sw = 1.0
    w /= sw
    g = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for k in nz:
        m = int(k) - (n - 1)
        g += w[int(k)] * _grad_at_lag(h, m)
    return g


def _binary_sa(h_init, n, t_start, cool, max_iters, rng):
    h = _fix_binary(h_init, n, rng)
    c5 = _c5(h)
    best = h.copy()
    best_c5 = c5
    T = t_start
    for it in range(max_iters):
        if _tl() < 5:
            break
        T = t_start * (cool ** it)
        if T < 1e-9:
            T = 1e-9
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5_new = _c5(h)
        delta = c5_new - c5
        if delta <= 0 or np.log(rng.random() + 1e-300) > -delta / T:
            c5 = c5_new
            if c5 < best_c5:
                best = h.copy()
                best_c5 = c5
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best, best_c5


def _binary_ls(h_init, n, max_iters, rng):
    h = _fix_binary(h_init, n, rng)
    c5 = _c5(h)
    best = h.copy()
    best_c5 = c5
    iterations = 0
    while _tl() > 10 and iterations < max_iters:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        for _ in range(min(3000, max_iters - iterations)):
            if _tl() < 5:
                break
            iterations += 1
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            c5_new = _c5(h)
            if c5_new < c5 - 1e-15:
                c5 = c5_new
                if c5 < best_c5:
                    best = h.copy()
                    best_c5 = c5
                improved = True
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        if not improved:
            n_pert = 1 + rng.randint(4)
            for _ in range(n_pert):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:

def _proj_grad(h, n, max_iters, rng):
    best = h.copy()
    bc5 = _c5(best)
    betas = [8, 30, 100, 400]
    steps = [0.08, 0.04, 0.02, 0.01, 0.005, 0.05, 0.015, 0.008, 0.003, 0.012]
    for it in range(max_iters):
        if _tl() < 3:
            break
        beta = betas[it % len(betas)]
        g = _multi_grad(best, n, beta)
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            break
        g /= gn
        moved = False
        for a in steps:
            h2 = _proj(best - a * g, n)
            c2 = _c5(h2)
            if c2 < bc5 - 1e-15:
                best = h2
                bc5 = c2
                moved = True
                break
        if not moved:
            c = np.correlate(best, 1.0 - best, mode='full')
            kmax = int(np.argmax(c))
            m = kmax - (n - 1)
            g2 = _grad_at_lag(best, m)
            gn2 = float(np.linalg.norm(g2))
            if gn2 > 1e-15:
                g2 /= gn2
                for a in steps[:5]:

            c2 = _c5(h2)
            if c2 < bc5 - 1e-15:
                best = h2
                bc5 = c2
    return best, bc5


def _slsqp_obj(x, n, beta):
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = float(np.max(c))
    sh = c - mx
    np.clip(sh, -80.0 / max(beta, 1.0), 0.0, out=sh)
    return (mx + np.log(np.exp(beta * sh).sum()) / beta) * (2.0 / n)


def _slsqp_cons(x, n):
    return x.sum() - 0.5 * n


def _slsqp(h0, n, beta, maxiter):
    try:
        r = minimize(_slsqp_obj, h0, args=(n, beta), method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _slsqp_cons, 'args': (n,)}],
                     options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return _proj(np.asarray(r.x, dtype=np.float64), n)
    except Exception:
        return _proj(h0.copy(), n)


def _interp_up(h, nf, nt):
    if nf == nt:
        return h.copy()
    xs = (np.arange(nf) + 0.5) * (2.0 / nf)
    xn = (np.arange(nt) + 0.5) * (2.0 / nt)
    return np.interp(xn, xs, h)
