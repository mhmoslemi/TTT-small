def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1][:max(1, int(np.ceil(abs(diff))))]
            h[idx] += diff / len(idx)
        else:
            idx = np.argsort(h)[::-1][:max(1, int(np.ceil(abs(diff))))]
            h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value_and_weights(corr, beta):
    m = np.max(corr)
    sh = beta * (corr - m)
    sh = np.clip(sh, -500.0, 0.0)
    exp_sh = np.exp(sh)
    w = exp_sh / np.sum(exp_sh)
    val = m + np.log(np.sum(exp_sh)) / beta
    return val, w

def slsqp_smooth_max(x0, n, beta, maxiter=200):
    dx = 2.0 / n
    best_x = x0.copy()
    c0 = corr_full(x0)
    v0, _ = smooth_max_value_and_weights(c0, beta)
    best_val = v0 * dx

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        v, _ = smooth_max_value_and_weights(c, beta)
        return v * dx

    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        _, w = smooth_max_value_and_weights(c, beta)
        g = np.zeros(n)
        for k_idx in range(len(c)):
            wk = w[k_idx]
            if wk < 1e-10:
                continue
            k = k_idx - (n - 1)
            wk_dx = wk * dx
            if k == 0:
                g += wk_dx * (1.0 - 2.0 * x)
            elif k > 0:
                part1 = np.zeros(n)
                part1[k:] = 1.0 - x[:n - k]
                part2 = np.zeros(n)
                part2[:n - k] = x[k:]
                g += wk_dx * (part1 - part2)
            else:
                s = -k
                part1 = np.zeros(n)
                part1[:n - s] = 1.0 - x[s:]
                part2 = np.zeros(n)
                part2[s:] = x[:n - s]
                g += wk_dx * (part1 - part2)
        return g

    def cons_fun(x):
        return np.sum(x) - n / 2.0

    def cons_grad(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_grad}]

    try:
        res = minimize(obj, x0, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
        if res.x is not None:
            x_new = project_feasible(res.x, n)
            v_new = compute_c5(x_new)
            c_new = corr_full(x_new)
            vn, _ = smooth_max_value_and_weights(c_new, beta)
            if vn * dx < best_val:
                best_val = vn * dx
                best_x = x_new
    except Exception:
        pass
    return best_x, best_val

def coord_descent_fast(h, n, corr, sweeps=20, time_limit=None, t0=None, cap_time=None):
    dx = 2.0 / n
    best = h.copy()
    best_corr = corr.copy()
    best_c5 = np.max(corr) * dx
    j_arr = np.arange(n)

    for sw in range(sweeps):
        if cap_time and t0 and time.time() - t0 > cap_time - 2:
            break
        order = np.random.permutation(n)
        for j in order:
            if cap_time and t0 and time.time() - t0 > cap_time - 1:
                break
            old = h[j]
            M = max(j, n - 1 - j)
            k_lo = max(-(n - 1), -M)
            k_hi = min(n - 1, M)
            num_aff = k_hi - k_lo + 1

            rest = np.zeros(num_aff)
            a_coef = np.zeros(num_aff)
            is_zero = np.zeros(num_aff, dtype=bool)

            for idx in range(num_aff):
                k = k_lo + idx
                fidx = k + (n - 1)
                if k == 0:
                    is_zero[idx] = True
                    rest[idx] = corr[fidx] - old + old * old
                    a_coef[idx] = 1.0
                else:
                    coeff = 0.0
                    if k >= 0:
                        if k <= j <= n - 1:
                            coeff += (1.0 - h[j - k])
                        if 0 <= j <= n - 1 - k:
                            coeff += (-h[j + k])
                    else:
                        s = -k
                        if 0 <= j <= n - 1 - s:
                            coeff += (1.0 - h[j + s])
                        if s <= j <= n - 1:
                            coeff += (-h[j - s])
                    rest[idx] = corr[fidx] - coeff * old
                    a_coef[idx] = coeff

            if k_lo > 0:
                unaff_left = corr[0:k_lo + (n - 1)]
            else:
                unaff_left = np.array([0.0])
            if k_hi < n - 1:
                unaff_right = corr[k_hi + 1 + (n - 1):]
            else:
                unaff_right = np.array([0.0])
            max_unaffected = 0.0
            if len(unaff_left) > 0:
                max_unaffected = max(max_unaffected, np.max(unaff_left))
            if len(unaff_right) > 0:
                max_unaffected = max(max_unaffected, np.max(unaff_right))

            xs_coarse = np.linspace(0.0, 1.0, 40)
            aff_vals = np.zeros((num_aff, 40))
            for idx in range(num_aff):
                if is_zero[idx]:
                    aff_vals[idx, :] = rest[idx] + xs_coarse - xs_coarse * xs_coarse
                else:
                    aff_vals[idx, :] = rest[idx] + a_coef[idx] * xs_coarse
            obj_vals = np.maximum(aff_vals.max(axis=0), max_unaffected)
            best_x_coarse = xs_coarse[np.argmin(obj_vals)]

            lo = max(0.0, best_x_coarse - 0.03)
            hi = min(1.0, best_x_coarse + 0.03)
            for _ in range(45):
                m1 = lo + (hi - lo) * 0.381966
                m2 = hi - (hi - lo) * 0.381966
                v1 = max_unaffected
                v2 = max_unaffected
                for idx in range(num_aff):
                    if is_zero[idx]:
                        cv1 = rest[idx] + m1 - m1 * m1
                        cv2 = rest[idx] + m2 - m2 * m2
                    else:
                        cv1 = rest[idx] + a_coef[idx] * m1
                        cv2 = rest[idx] + a_coef[idx] * m2
                    if cv1 > v1:
                        v1 = cv1
                    if cv2 > v2:
                        v2 = cv2
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            best_x = (lo + hi) / 2.0

            h[j] = best_x
            for idx in range(num_aff):
                fidx = (k_lo + idx) + (n - 1)
                if is_zero[idx]:
                    corr[fidx] = rest[idx] + best_x - best_x * best_x
                else:
                    corr[fidx] = rest[idx] + a_coef[idx] * best_x

            c5_now = max(max_unaffected, np.max(corr[k_lo + (n - 1):k_hi + 1 + (n - 1)])) * dx
            if c5_now < best_c5:
                best_c5 = c5_now
                best = h.copy()
                best_corr = corr.copy()

        h = project_feasible(h, n)
        corr = corr_full(h)
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
            best_corr = corr.copy()

    return best, best_c5

def interpolate_h(h_src, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_new = np.interp(xn, xp, h_src)
    return project_feasible(h_new, n_dst)

def topk_smooth_obj(x, n, k, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    top_k = np.sort(corr)[-k:]
    m = np.max(top_k)
    sh = beta * (top_k - m)
    sh = np.clip(sh, -500.0, 0.0)
    return dx * (m + np.log(np.sum(np.exp(sh))) / beta)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return max(0.0, budget_s - (time.time() - t0) - 5.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn > 10:
            best_h = ph.copy()
            best_c5 = compute_c5(ph)
            best_n = pn
    except Exception:
        pass

    if best_h is None:
        best_n = 200
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)

    start_h = best_h.copy()
    start_n = best_n

    n_targets = sorted(set([start_n, 176, 192, 208, 224, 256]))
    n_targets = [nn for nn in n_targets if 120 <= nn <= 320]

    all_candidates = []

    for n_trial in n_targets:
        if time_left() < 80:
            break

        if n_trial == start_n:
            h_init = project_feasible(start_h, n_trial)
        else:
            h_init = interpolate_h(start_h, start_n, n_trial)

        inits = [h_init.copy()]
        for sc in [0.01, 0.03, 0.08]:
            hp = h_init + sc * rng.randn(n_trial)
            inits.append(project_feasible(hp, n_trial))
        t_arr = (np.arange(n_trial) + 0.5) / n_trial
        for freq in [2, 3, 4]:
            ph = rng.rand() * 2 * np.pi
            hf = 0.7 * h_init + 0.3 * np.clip(0.5 + 0.45 * np.cos(2 * np.pi * freq * t_arr + ph), 0, 1)
            inits.append(project_feasible(hf, n_trial))

        for h0 in inits:
            if time_left() < 40:
                break

            h_curr = h0.copy()

            for beta in [100.0, 500.0, 2000.0, 10000.0]:
                if time_left() < 20:
                    break
                h_curr, _ = slsqp_smooth_max(h_curr, n_trial, beta, maxiter=150)

            corr_c = corr_full(h_curr)
            h_curr, c5_cd = coord_descent_fast(h_curr, n_trial, corr_c, sweeps=8,
                                               time_limit=None, t0=t0,
                                               cap_time=time_left() + t0 - 5)

            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_curr.copy()
                best_n = n_trial

            all_candidates.append((h_curr.copy(), n_trial, c5_cd))

    # Sort candidates and polish the best ones
    all_candidates.sort(key=lambda x: x[2])
    polish_list = all_candidates[:min(5, len(all_candidates))]

    for h_cand, n_cand, c5_cand in polish_list:
        if time_left() < 60:
            break
        h_curr = h_cand.copy()
        corr_c = corr_full(h_curr)

        for beta in [5000.0, 20000.0, 50000.0]:
            if time_left() < 20:
                break
            h_curr, _ = slsqp_smooth_max(h_curr, n_cand, beta, maxiter=100)
            h_curr = project_feasible(h_curr, n_cand)

        corr_c = corr_full(h_curr)
        h_curr, c5_cd = coord_descent_fast(h_curr, n_cand, corr_c, sweeps=15,
                                           time_limit=None, t0=t0,
                                           cap_time=time_left() + t0 - 3)
        c5_cur = compute_c5(h_curr)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_curr.copy()
            best_n = n_cand

    # Final intensive polishing
    if time_left() > 30:
        h_final = best_h.copy()
        n_final = best_n
        corr_f = corr_full(h_final)

        for _ in range(3):
            if time_left() < 15:
                break
            h_final, c5_f = coord_descent_fast(h_final, n_final, corr_f, sweeps=25,
                                               time_limit=None, t0=t0,
                                               cap_time=time_left() + t0 - 3)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_final.copy()
            corr_f = corr_full(h_final)

        if time_left() > 20:
            h_final = best_h.copy()
            for beta in [20000.0, 80000.0]:
                if time_left() < 10:
                    break
                h_final, _ = slsqp_smooth_max(h_final, best_n, beta, maxiter=100)
                h_final = project_feasible(h_final, best_n)
                c5_sp = compute_c5(h_final)
                if c5_sp < best_c5:
                    best_c5 = c5_sp
                    best_h = h_final.copy()
                    best_n = len(h_final)
            corr_f = corr_full(best_h)
            h_final, c5_f = coord_descent_fast(best_h, best_n, corr_f, sweeps=20,
                                               time_limit=None, t0=t0,
                                               cap_time=time_left() + t0 - 2)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_final.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
