def compute_c5(h):
    """Compute C5 = max(corr) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr_array(h):
    """Compute full correlation array"""
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    """Project h onto feasible set: 0<=h<=1, sum(h)=n/2"""
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Adjust values that are strictly inside (0,1)
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            # All at boundaries; use a more careful approach
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Find index with most room
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(src_h, src_n, dst_n):

    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def obj_smooth(x):

    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _G_BETA

    exps = np.exp(np.clip(beta * shifted, -500, 500))
    s = np.sum(exps)
    if s < 1e-300:
        return m * dx
    return (m + np.log(s) / beta) * dx

def make_thue_morse(n):
    """Thue-Morse binary sequence"""
    # Pad to next power of 2 if needed
    p2 = 1
    while p2 < n:
        p2 *= 2
    h_full = np.zeros(p2)
    for i in range(p2):
        cnt = 0
        v = i
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h_full[i] = 1.0 if cnt == 1 else 0.0
    h = h_full[:n].copy()
    # Ensure exactly n/2 ones
    num_ones = int(np.sum(h))
    if num_ones > n//2:
        idx = np.where(h == 1.0)[0]
        for i in idx[(num_ones - n//2):]:
            h[i] = 0.0

        for i in idx[:n//2-num_ones]:
            h[i] = 1.0

def make_golden_ratio(n):

def projected_subgradient_nesterov(h, n, budget_limit, rng, lr=0.05, momentum=0.9):
    """Projected subgradient with Nesterov momentum"""
    global _G_N, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    
    dx = 2.0 / n

    best_h = h.copy()
    best_cost = compute_c5(h)
    
    t_start = time.time()
    
    for it in range(100000):
        if time.time() - t_start > budget_limit:
            break
        if remaining_time() < 5.0:
            break
        
        # Nesterov lookahead
        h_y = h - momentum * v
        
        # Compute subgradient of max_k corr[k] * dx
        corr = np.correlate(h_y, 1.0 - h_y, mode='full')
        max_idx = np.argmax(corr)
        
        # Subgradient at the active constraint
        grad = np.zeros(n)
        offset = int(max_idx) - (n - 1)
        if offset >= 0:
            n_pairs = n - offset
            if n_pairs > 0:
                grad[offset:offset+n_pairs] += (1.0 - h_y[:n_pairs])
                grad[:n_pairs] -= h_y[offset:offset+n_pairs]
        else:
            abs_off = -offset
            n_pairs = n - abs_off
            if n_pairs > 0:
                grad[:n_pairs] += (1.0 - h_y[abs_off:abs_off+n_pairs])
                grad[abs_off:abs_off+n_pairs] -= h_y[:n_pairs]
        
        # Scale gradient (it's already per-dx since we multiply by dx in cost)
        v = momentum * v + grad
        h = h - lr * v
        h = project_feasible(h, n)
        
        cost = compute_c5(h)
        if cost < best_cost:
            best_cost = cost
            best_h = h.copy()
        
        # Adaptive learning rate
        if it % 100 == 0:
            lr = max(0.001, lr * 0.999)
    
    return best_h, best_cost
