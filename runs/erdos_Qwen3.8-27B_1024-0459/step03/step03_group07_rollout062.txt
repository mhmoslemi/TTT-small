            temp = max(0.003, temp * 0.97)
        else:
            if rng.rand() < 0.1:
                h = h_opt.copy()
            temp = max(0.003, temp * 0.96)
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    try:
        ph = np.array(initial_h_values, dtype=float)
        best_n = len(ph)
        best_h = ph.copy()
        best_c5 = compute_c5(ph)
    except Exception:
        best_n = 160
        best_h = project_feasible(np.ones(best_n) * 0.5, best_n)
        best_c5 = compute_c5(best_h)
    def time_left():
        return max(0.0, budget_s - (time.time() - t0) - 15.0)
    # Phase 1: Quick CD at parent resolution
    if time_left() > 40:
        h1 = project_feasible(best_h, best_n)
        h1, c1 = coord_descent(h1, best_n, max_sweeps=80, time_limit=time_left() * 0.25, t0=t0)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()
    # Phase 2: n=320
    n2 = 320
    if time_left() > 200:
        h2 = interpolate_h(best_h, best_n, n2)
        tl2 = time_left() * 0.30
        h2, c2 = coord_descent(h2, n2, max_sweeps=300, time_limit=tl2, t0=t0)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()
            best_n = n2
        # Basin hopping at n=320
        if time_left() > 100:
            tl_bh = time_left() * 0.45
            h2b, c2b = basin_hopping(h2, n2, rng, max_iters=500, time_limit=tl_bh, t0=t0)
            if c2b < best_c5:
                best_c5 = c2b
                best_h = h2b.copy()
                best_n = n2
            # More CD on best
            if time_left() > 30:
                h2c, c2c = coord_descent(best_h, best_n, max_sweeps=100, time_limit=time_left() * 0.3, t0=t0)
                if c2c < best_c5:
                    best_c5 = c2c
                    best_h = h2c.copy()
    # Phase 3: n=480
    n3 = 480
    if time_left() > 150:
        h3 = interpolate_h(best_h, best_n, n3)
        tl3 = time_left() * 0.30
        h3, c3 = coord_descent(h3, n3, max_sweeps=200, time_limit=tl3, t0=t0)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3.copy()
            best_n = n3
        if time_left() > 80:
            tl_bh3 = time_left() * 0.40
            h3b, c3b = basin_hopping(h3, n3, rng, max_iters=300, time_limit=tl_bh3, t0=t0)
            if c3b < best_c5:
                best_c5 = c3b
                best_h = h3b.copy()
                best_n = n3
    # Phase 4: n=640
    n4 = 640
    if time_left() > 120:
        h4 = interpolate_h(best_h, best_n, n4)
        tl4 = time_left() * 0.35
        h4, c4 = coord_descent(h4, n4, max_sweeps=150, time_limit=tl4, t0=t0)
        if c4 < best_c5:
            best_c5 = c4
            best_h = h4.copy()
            best_n = n4
        if time_left() > 50:
            h4b, c4b = basin_hopping(h4, n4, rng, max_iters=150, time_limit=time_left() * 0.35, t0=t0)
            if c4b < best_c5:
                best_c5 = c4b
                best_h = h4b.copy()
                best_n = n4
    # Phase 5: Final polish
    if time_left() > 30:
        hp, cp = coord_descent(best_h, best_n, max_sweeps=200, time_limit=time_left() * 0.6, t0=t0)
        if cp < best_c5:
            best_c5 = cp
            best_h = hp.copy()
    if time_left() > 20:
        hp2, cp2 = coord_descent(best_h, best_n, max_sweeps=100, time_limit=time_left() * 0.8, t0=t0)
        if cp2 < best_c5:
            best_c5 = cp2
            best_h = hp2.copy()
    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    return (best_h, final_c5, best_n)
