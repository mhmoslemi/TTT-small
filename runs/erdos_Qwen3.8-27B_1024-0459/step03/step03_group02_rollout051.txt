def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):

            break

            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:

                    h[j] = np.clip(h[j] + diff2, 0.0, 1.0)
                    break

def smooth_max_obj(x):

    dx = 2.0 / n

    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    shifted = np.maximum(shifted, -60.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def smooth_max_grad(x):
    n = _N_GLOBAL
    beta = _BETA_GLOBAL
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    shifted = np.maximum(shifted, -60.0)
    weights = np.exp(shifted)
    total_w = np.sum(weights)
    if total_w < 1e-30:
        return np.zeros(n)
    weights /= total_w
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-14:
            continue
        offset = k_idx - (n - 1)
        w = weights[k_idx]
        if offset >= 0:
            n_pairs = n - offset
            if n_pairs <= 0:
                continue
            grad[offset:offset + n_pairs] += w * (1.0 - x[:n_pairs])
            grad[:n_pairs] -= w * x[offset:offset + n_pairs]
        else:
            abs_off = -offset
            n_pairs = n - abs_off
            if n_pairs <= 0:
                continue
            grad[:n_pairs] += w * (1.0 - x[abs_off:abs_off + n_pairs])
            grad[abs_off:abs_off + n_pairs] -= w * x[:n_pairs]
    return grad * dx

    return np.sum(x) - _N_GLOBAL / 2.0

def slsqp_refine(h, n, beta, maxiter=300, ftol=1e-18):

    _N_GLOBAL = n
    _BETA_GLOBAL = beta

            for k_idx in top_lags:
                offset = k_idx - (n - 1)
                grad += 0.5 * lag_gradient(h, n, offset)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            grad /= gnorm
            improved = False
            for step in [0.001, 0.002, 0.0005, 0.003, 0.0008]:
                h_trial = h - step * grad
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-17:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h = h_trial.copy()
                    improved = True
                    break
            if not improved:
                break

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
