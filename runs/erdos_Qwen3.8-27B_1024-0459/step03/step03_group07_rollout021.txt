import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Fix residual
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0:
                amt = min(1.0 - h[i], diff)
                h[i] += amt
                diff -= amt
            else:
                amt = min(h[i], -diff)
                h[i] -= amt
                diff += amt
            if abs(diff) < 1e-15:
                break
    return h

def coord_descent(h, n, max_sweeps=50, time_limit=None, t0=None):
    """Fast vectorized coordinate descent."""
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    grid_coarse = np.linspace(0.0, 1.0, 101)
    
    for sweep in range(max_sweeps):
        if time_limit and t0 and time.time() - t0 > time_limit + 2:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        any_change = False
        
        for j in range(n):
            if time_limit and t0 and time.time() - t0 > time_limit + 2:
                break
            old = h[j]
            
            # For each lag d, compute slope and check if affected
            # Affected lags: d where 0<=j+d<n OR 0<=j-d<n
            d_lo = max(-(n-1), j-(n-1))  # from second factor condition
            d_hi = min(n-1, n-1-j)        # from first factor condition
            
            # Actually, let me just compute for all d
            # d ranges from -(n-1) to n-1
            ds = np.arange(-(n-1), n)
            
            # Slopes: for each d
            # First factor: 0<=j+d<n → d in [-j, n-1-j]
            # Second factor: 0<=j-d<n → d in [j-n+1, j]
            
            first_valid = (j + ds >= 0) & (j + ds < n)  # h[j] is first factor, pair is h[j+d]
            second_valid = (j - ds >= 0) & (j - ds < n)  # h[j] is in (1-h[.]), pair is h[j-d]
            
            slopes = np.zeros(len(ds))
            for idx in range(len(ds)):
                d = ds[idx]
                s = 0.0
                if first_valid[idx]:
                    s += (1.0 - h[j + d])
                if second_valid[idx]:
                    s += (-h[j - d])
                slopes[idx] = s
            
            affected = first_valid | second_valid
            # Note: for d=0, first_valid and second_valid are both True, slope = (1-h[j]) + (-h[j]) = 1-2h[j]
            # But d=0 is the quadratic case
            
            max_unaff = 0.0
            if np.any(~affected):
                max_unaff = np.max(corr[~((ds >= -(n-1)) & (ds < n)) | ~affected]) if np.any(~affected) else 0.0
                # Simpler:
                unaff_mask = ~affected
                if np.any(unaff_mask):
                    max_unaff = np.max(corr[unaff_mask])
            
            # Quadratic term for d=0
            quad_idx = n - 1  # d=0 → fi = 0 + (n-1) = n-1
            quad_rest = corr[quad_idx] - old * (1.0 - old)
            
            # Affine terms (excluding d=0)
            aff_mask = affected & (ds != 0)
            if np.any(aff_mask):
                aff_slopes = slopes[aff_mask]
                aff_offsets = corr[aff_mask] - aff_slopes * old
            else:
                aff_slopes = np.array([0.0])
                aff_offsets = np.array([np.inf])  # No affine constraints
            
            # Evaluate f(x) on coarse grid
            # f(x) = max(max_unaff, max_i(aff_offsets[i] + aff_slopes[i]*x), quad_rest + x - x²)
            args = [np.full(101, max_unaff)]
            if np.any(aff_mask):
                affine_grid = aff_offsets[None, :] + grid_coarse[:, None] * aff_slopes[None, :]
                args.append(np.max(affine_grid, axis=1))
            q_grid = quad_rest + grid_coarse - grid_coarse**2
            args.append(q_grid)
            
            f_vals = np.maximum.reduce(args)
            best_i = np.argmin(f_vals)
            best_x = grid_coarse[best_i]
            
            # Refine
            if 0 < best_i < 100:
                lo = grid_coarse[best_i - 1]
                hi = grid_coarse[best_i + 1]
            else:
                lo = max(0.0, best_x - 0.01)
                hi = min(1.0, best_x + 0.01)
            
            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                v1 = max_unaff
                if np.any(aff_mask):
                    v1 = max(v1, np.max(aff_offsets + aff_slopes * m1))
                v1 = max(v1, quad_rest + m1 - m1*m1)
                v2 = max_unaff
                if np.any(aff_mask):
                    v2 = max(v2, np.max(aff_offsets + aff_slopes * m2))
                v2 = max(v2, quad_rest + m2 - m2*m2)
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            
            best_x = (lo + hi) / 2.0
            
            if abs(best_x - old) > 1e-10:
                h[j] = best_x
                any_change = True
        
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        
        if not any_change:
            break
    
    return best_h

def slsqp_opt(h, n, time_limit=None, t0=None):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
    
    for beta in [50, 200, 1000, 5000]:
        if time_limit and t0 and time.time() - t0 > time_limit - 5:
            break

            def obj(x, b=beta):
                corr = np.correlate(x, 1.0 - x, mode='full')
                m_val = np.max(corr)
                ea = np.clip(b * (corr - m_val), -500, 0)
                return (m_val + np.log(np.sum(np.exp(ea))) / b) * (2.0 / n)
            
            res = minimize(obj, h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 300, 'ftol': 1e-13, 'iprint': 0})
            if res.x is not None:
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
        except Exception:
            pass
    
    return best_h
