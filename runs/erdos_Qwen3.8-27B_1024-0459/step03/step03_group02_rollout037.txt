```python
def flip_effect(h, j, l, k, n):
    """Compute change in c_k when h[j]: 0→1, h[l]: 1→0."""
    # Valid i range: 0 to n-1-k
    max_i = n - 1 - k
    if max_i < 0:
        return 0
    
    affected = set()
    if 0 <= j <= max_i:
        affected.add(j)
    if 0 <= j-k <= max_i:
        affected.add(j-k)
    if 0 <= l <= max_i:
        affected.add(l)
    if 0 <= l-k <= max_i:
        affected.add(l-k)
    
    delta = 0
    for i in affected:
        # Old values
        old_hi = h[i]
        old_hik = h[i+k]
        old_term = old_hi * (1 - old_hik)
        
        # New values
        new_hi = 1.0 if i == j else (0.0 if i == l else h[i])
        new_hik = 1.0 if (i+k) == j else (0.0 if (i+k) == l else h[i+k])
        new_term = new_hi * (1 - new_hik)
        
        delta += new_term - old_term
    
    return delta
```
