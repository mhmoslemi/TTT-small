        mask1 = (js >= 0) & (js < n)
        topgrad[k_full, mask1] -= h[js[mask1]]
        js2 = j - s
        mask2 = (js2 >= 0) & (js2 < n)
        topgrad[k_full, mask2] -= h[js2[mask2]]
    return topgrad

def grad_shift(h, n, s):
    """Gradient of corr_s w.r.t. h (without the constant 1)."""
    j = np.arange(n)
    grad = np.zeros(n)
    js = j + s
    mask1 = (js >= 0) & (js < n)
    grad[mask1] -= h[js[mask1]]
    js2 = j - s
    mask2 = (js2 >= 0) & (js2 < n)
    grad[mask2] -= h[js2[mask2]]
    return grad

def soft_max_obj(x, n, active, beta, lam):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    vals = corr[active]
    m = np.max(vals)
    if beta < 1e-4:
        sme = m
    else:
        shifted = vals - m
        exps = np.exp(beta * shifted)
        sme = m + np.log(np.sum(exps)) / beta
    pen = lam * (np.sum(x) - n / 2.0) ** 2
    return sme * dx + pen

def soft_max_obj_full(x, n, beta, lam):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    if beta < 1e-4:
        sme = m
    else:
        shifted = corr - m
        exps = np.exp(beta * shifted)
        sme = m + np.log(np.sum(exps)) / beta
    pen = lam * (np.sum(x) - n / 2.0) ** 2
    return sme * dx + pen

def sum_constraint(x, n):
    return np.sum(x) - n / 2.0

def subgradient_step(h, n, top_k, temperature, step_scale):
    """Take a projected subgradient step targeting the worst shifts.
    Returns new h and the C5 value."""
    _, corr = compute_c5(h)
    # Weight shifts by how close they are to the max
    mx = np.max(corr)
    if mx < 1e-15:
        return h, 0.0
    # Use exponential weighting
    weights = np.exp(temperature * (corr - mx) / mx)
    # Only use top_k shifts
    top_idx = np.argsort(corr)[-top_k:]
    other_weights = np.zeros_like(weights)
    other_weights[top_idx] = weights[top_idx]
    w_sum = np.sum(other_weights)
    if w_sum < 1e-15:
        return h, np.max(corr) * 2.0 / n
    other_weights /= w_sum
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for k_full in top_idx:
        w = other_weights[k_full]
        if w < 1e-8:
            continue
        s = k_full - (n - 1)
        j = np.arange(n)
        js = j + s
        m1 = (js >= 0) & (js < n)
        grad[m1] -= w * h[js[m1]]
        js2 = j - s
        m2 = (js2 >= 0) & (js2 < n)
        grad[m2] -= w * h[js2[m2]]
    
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h, np.max(corr) * 2.0 / n
    grad /= gn
    
    # Backtracking line search
    best_h_local = h
    best_c5_local, _ = compute_c5(h)
    alpha0 = step_scale
    for i in range(20):
        alpha = alpha0 * (0.7 ** i)
        if alpha < 1e-8:
            break
        h_try = h - alpha * grad
        h_try = project_h(h_try, n)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5_local:
            best_c5_local = c5_try
            best_h_local = h_try
        else:
            if i >= 3:
                break
    return best_h_local, best_c5_local

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def elapsed():
        return time.time() - t0
    
    def remaining():
        return max(0.0, budget_s - elapsed())
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Initialize from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5, _ = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 200
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5, _ = compute_c5(best_h)
    
    # Phase 1: Multi-resolution scan
    res_list = [180, 200, 220, 240, 280, 320, 360, 400, 480, 520]
    for n_try in res_list:
        if remaining() < 60:
            break
        h_try = interp_h(best_h, best_n, n_try)
        h_try = project_h(h_try, n_try)
        c5_try, _ = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try
    
    # Phase 2: Main optimization at best resolution
    n = best_n
    h = best_h.copy()
    
    # Also try a few other resolutions in parallel (keep best per resolution)
    alt_ns = []
    for dn in [-40, -20, 20, 40, 80]:
        nt = n + dn
        if 100 <= nt <= 600:
            alt_ns.append(nt)
    
    # Track solutions at multiple resolutions
    sols = {n: h.copy()}
    for nt in alt_ns:
        ht = interp_h(h, n, nt)
        ht = project_h(ht, nt)
        sols[nt] = ht
    
    # Optimization loop
    for outer in range(200):
        if remaining() < 30:
            break
        
        # Work on the best resolution
        h = sols[n].copy()
        c5_cur, corr = compute_c5(h)
        
        improved = False
        
        # Subgradient method with adaptive parameters
        for temp in [5.0, 15.0, 40.0, 100.0]:
            if remaining() < 10:
                break
            top_k = min(30, 2 * n - 1)
            for step_scale in [0.3, 0.15, 0.08, 0.04]:
                if remaining() < 5:
                    break
                # Take multiple subgradient steps
                for sg_iter in range(50):
                    if remaining() < 3:
                        break
                    h, c5_sg = subgradient_step(h, n, top_k, temp, step_scale)
                    if c5_sg < best_c5:
                        best_c5 = c5_sg
                        best_h = h.copy()
                        sols[n] = h.copy()
                        improved = True
        
        # L-BFGS-B with soft-max on active set
        if remaining() < 15:
            break
        _, corr = compute_c5(h)
        mx = np.max(corr)
        # Active set: shifts within 97% of max
        thresh = mx * 0.97
        active = np.where(corr >= thresh)[0]
        if len(active) > 40:
            active = np.argsort(corr)[-40:]
        
        for beta in [50, 200, 500, 1000]:
            if remaining() < 8:
                break
            bounds = [(0.0, 1.0)] * n
            def obj_func(x):
                return soft_max_obj(x, n, active, beta, 2000.0)
            try:
                res = minimize(obj_func, h, method='L-BFGS-B', bounds=bounds,
                             options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-13, 'maxls': 50})
                h_new = project_h(res.x, n)
                c5_new, _ = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    sols[n] = h_new.copy()
                    h = h_new.copy()
                    improved = True
            except Exception:
                pass
        
        # SLSQP with equality constraint
        if remaining() < 20:
            break
        for beta in [200, 500, 1500]:
            if remaining() < 8:
                break
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                     'jac': lambda x: np.ones(n)}]
            def obj_slsqp(x):
                return soft_max_obj(x, n, active, beta, 0.0)
            try:
                res = minimize(obj_slsqp, h, method='SLSQP', bounds=bounds,
                             constraints=cons,
                             options={'maxiter': 150, 'ftol': 1e-15, 'iprint': 0})
                h_new = project_h(res.x, n)
                c5_new, _ = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    sols[n] = h_new.copy()
                    h = h_new.copy()
                    improved = True
            except Exception:
                pass
        
        if not improved and outer > 5:
            # Try switching to a different resolution
            if remaining() < 30:
                break
            best_alt_c5 = best_c5
            for nt in alt_ns:
                if nt not in sols:
                    continue
                ht = interp_h(best_h, best_n, nt)
                ht = project_h(ht, nt)
                ct, _ = compute_c5(ht)
                if ct < best_alt_c5:
                    best_alt_c5 = ct
                    sols[nt] = ht.copy()
                    h = ht.copy()
                    # Switch to this resolution
                    if nt != n:
                        n = nt
                        break
    
    # Phase 3: Local search with simulated annealing
    if remaining() > 20:
        n = best_n
        h = best_h.copy()
        T = 0.001  # initial temperature
        cooling = 0.995
        
        for it in range(80000):
            if remaining() < 3:
                break
            # Pick 1-3 coordinates to modify
            n_mod = 1 + (1 if rng.rand() < 0.3 else 0) + (1 if rng.rand() < 0.1 else 0)
            idx = rng.choice(n, size=min(n_mod, n), replace=False)
            d = 0.005 + 0.03 * rng.rand()
            
            h_try = h.copy()
            valid = True
            for k in range(min(n_mod, n)):
                i = int(idx[k])
                if rng.rand() < 0.5:
                    # Try to decrease
                    if h_try[i] >= d / min(n_mod, n):
                        h_try[i] -= d / min(n_mod, n)
                    else:
                        valid = False
                        break
                else:
                    # Try to increase
                    if h_try[i] <= 1.0 - d / min(n_mod, n):
                        h_try[i] += d / min(n_mod, n)
                    else:
                        valid = False
                        break
            
            if not valid:
                continue
            
            h_try = project_h(h_try, n)
            c5_try, _ = compute_c5(h_try)
            
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                sols[n] = h_try.copy()
                h = h_try.copy()
            else:
                # Simulated annealing: accept with probability exp(-delta/T)
                delta = c5_try - best_c5
                if delta < 0.01 and rng.rand() < np.exp(-delta / T):
                    h = h_try.copy()
            T *= cooling
    
    # Phase 4: Final high-precision optimization
    if remaining() > 15:
        n = best_n
        h = best_h.copy()
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        
        # Full soft-max with very high beta (near-exact max)
        for beta_f in [2000, 5000, 10000]:
            if remaining() < 8:
                break
            def obj_full(x):
                return soft_max_obj_full(x, n, beta_f, 0.0)
            try:
                res = minimize(obj_full, h, method='SLSQP', bounds=bounds,
                             constraints=cons,
                             options={'maxiter': 100, 'ftol': 1e-16, 'iprint': 0})
                h_new = project_h(res.x, n)
                c5_new, _ = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
            except Exception:
                pass
    
    # Phase 5: Final multi-resolution check and swap
    if remaining() > 10:
        for dn in range(-50, 51, 10):
            if remaining() < 4:
                break
            if dn == 0:
                continue
            nt = best_n + dn
            if nt < 80 or nt > 600:
                continue
            ht = interp_h(best_h, best_n, nt)
            ht = project_h(ht, nt)
            ct, _ = compute_c5(ht)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()
                best_n = nt
                # Quick optimize at new resolution
                if remaining() > 20:
                    h2 = ht.copy()
                    _, corr2 = compute_c5(h2)
                    mx2 = np.max(corr2)
                    act2 = np.where(corr2 >= mx2 * 0.95)[0]
                    if len(act2) > 30:
                        act2 = np.argsort(corr2)[-30:]
                    bounds2 = [(0.0, 1.0)] * nt
                    for beta2 in [200, 500]:
                        if remaining() < 10:
                            break
                        def obj2(x):
                            return soft_max_obj(x, nt, act2, beta2, 2000.0)
                        try:
                            res2 = minimize(obj2, h2, method='L-BFGS-B', bounds=bounds2,
                                         options={'maxiter': 100, 'ftol': 1e-15, 'gtol': 1e-12})
                            h2_new = project_h(res2.x, nt)
                            c5_2n, _ = compute_c5(h2_new)
                            if c5_2n < best_c5:
                                best_c5 = c5_2n
                                best_h = h2_new.copy()
                                h2 = h2_new.copy()
                        except Exception:
                            pass
    
    # Final projection and evaluation
    best_h = project_h(best_h, best_n)
    final_c5, _ = compute_c5(best_h)
    
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
