    """Compute a subgradient of the max correlation and return updated h."""

            if imp:
                best_h = h_cd.copy()
            else:
                if sweep > 30:
                    break
        print("  After cyclic CD: C5={:.8f}".format(best_c5))
    
    # === PHASE 3: Try other resolutions ===
    n_tries = sorted(set([best_n, best_n-16, best_n+16, best_n+32, best_n-32,
                          best_n+64, best_n-48, best_n+48, best_n+96]))
    n_tries = [x for x in n_tries if 80 <= x <= 320 and x % 2 == 0]
    
    for n_try in n_tries:
        if tleft() < 50:
            break
        if n_try == best_n:
            continue
        
        target_t = n_try / 2.0
        # Interpolate from current best
        xp = (np.arange(best_n) + 0.5) * 2.0 / best_n
        xn = (np.arange(n_try) + 0.5) * 2.0 / n_try
        h_interp = np.interp(xn, xp, best_h)
        h_interp = project_feasible(h_interp, target_t)
        bounds_t = [(0.0, 1.0)] * n_try
        
        for beta in [50, 200, 600, 1500]:
            if tleft() < 15:
                break
            res = minimize(smooth_obj, h_interp, method='L-BFGS-B', bounds=bounds_t,
                          args=(target_t, beta, 10000.0, 20),
                          options={'maxiter': 400, 'ftol': 1e-22, 'gtol': 1e-18,
                                   'maxls': 40, 'maxfun': 4000})
            h_new = project_feasible(res.x, target_t)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_try
                h_interp = h_new.copy()
                print("  n={}, beta={}: C5={:.8f}".format(n_try, beta, best_c5))
            else:
                h_interp = project_feasible(res.x, target_t)
        
        # Quick CD at this resolution
        if tleft() > 20 and best_n == n_try:
            h_cd2 = best_h.copy()
            for sweep in range(50):
                if tleft() < 5:
                    break
                h_cd2, best_c5, imp = coord_descent_step(h_cd2, target_t, n_try, j_offsets, deltas_small, best_c5, rng)
                if imp:
                    best_h = h_cd2.copy()
                else:
                    break
    
    # === PHASE 4: Subgradient descent with momentum ===
    if tleft() > 40:
        n = best_n
        target = n / 2.0
        h_sg = best_h.copy()
        vel = np.zeros(n)
        alpha = 0.03
        mu = 0.85
        
        for it in range(600):
            if tleft() < 4:
                break
            
            grad, corr, idx_top, top_vals = subgradient_one_step(h_sg, target, n, 100, 8)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm
            
            vel = mu * vel - alpha * grad
            h_trial = h_sg + vel
            h_trial = project_feasible(h_trial, target)
            c5_trial = compute_c5(h_trial)
            
            if c5_trial < best_c5 - 1e-13:
                best_c5 = c5_trial
                best_h = h_trial.copy()
            
            h_sg = h_trial.copy()
            
            if it % 30 == 29:
                alpha *= 0.93
                mu *= 0.99
        
        print("  After subgradient: C5={:.8f}".format(best_c5))
    
    # === PHASE 5: Random perturbations + local search ===
    if tleft() > 50:
        n = best_n
        target = n / 2.0
        
        for trial in range(300):
            if tleft() < 8:
                break
            
            # Perturbation size decreases over trials
            mag = 0.03 * (1.0 / (1.0 + 0.02 * trial))
            h_pert = best_h + rng.randn(n) * mag
            h_pert = project_feasible(h_pert, target)
            
            # Quick local search
            c5_p = compute_c5(h_pert)
            if c5_p > best_c5 + 0.005:
                continue  # Perturbation was too far
            
            h_local = h_pert.copy()
            for cd_sweep in range(30):
                if tleft() < 3:
                    break
                h_local, c5_p, imp = coord_descent_step(h_local, target, n, j_offsets, deltas_small, best_c5 if c5_p >= best_c5 else c5_p, rng)
                if imp:
                    if c5_p < best_c5:
                        best_c5 = c5_p
                        best_h = h_local.copy()
                else:
                    break
        
        print("  After random restarts: C5={:.8f}".format(best_c5))
    
    # === PHASE 6: Final ultra-fine coordinate descent ===
    if tleft() > 20:
        n = best_n
        target = n / 2.0
        h_fin = best_h.copy()
        very_small_deltas = [0.0005, 0.0002, 0.0001, 0.0003, 0.0008]
        offsets_fine = [1, n-1, 2, n-2, 3, n-3, n//4, n//4+1, 5, n-5]
        
        for sweep in range(150):
            if tleft() < 3:
                break
            h_fin, best_c5, imp = coord_descent_step(h_fin, target, n, offsets_fine, very_small_deltas, best_c5, rng)
            if imp:
                best_h = h_fin.copy()
            else:
                if sweep > 20:
                    break
    
    # Final projection
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass
    
    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
