import cvxpy as cp

def solve_cvxpy(h_init, n, active_lags, time_limit=None, t0=None):
    dx = 2.0 / n
    h = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n / 2.0
    ]
    
    # For k=0: sum h[i]² >= n/2 - t/dx
    constraints.append(cp.sum_squares(h) >= n/2.0 - t/dx)
    
    # For each active lag k != 0
    for k in active_lags:
        if k == 0:
            continue
        if k > 0:
            # sum h[i]*(1-h[i+k]) for i=0..n-1-k
            # = sum h[i] - sum h[i]*h[i+k] for i=0..n-1-k
            # z_i = h[i]*h[i+k] for i=0..n-1-k
            nz = n - k
            z = cp.Variable(nz)
            for i in range(nz):
                constraints.append(z[i] <= h[i])
                constraints.append(z[i] <= h[i+k])
                constraints.append(z[i] >= h[i] + h[i+k] - 1)
                constraints.append(z[i] >= 0)
            constraints.append(cp.sum(h[:nz]) - cp.sum(z) <= t/dx)
        else:
            kk = -k
            # sum h[i]*(1-h[i-kk]) for i=kk..n-1
            # = sum h[i] - sum h[i]*h[i-kk] for i=kk..n-1
            nz = n - kk
            z = cp.Variable(nz)
            for i in range(nz):
                constraints.append(z[i] <= h[i+kk])
                constraints.append(z[i] <= h[i])
                constraints.append(z[i] >= h[i+kk] + h[i] - 1)
                constraints.append(z[i] >= 0)
            constraints.append(cp.sum(h[kk:]) - cp.sum(z) <= t/dx)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        if h.value is not None:
            h_sol = np.clip(h.value, 0, 1)
            # Project
            h_sol = project_feasible(h_sol, n)
            return h_sol, t.value
    except Exception:
        pass
    return None, None

If lag > 0:

The z-linearization for 0≤a,b≤1: z = a*b iff:

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            j = int(np.argmin(h))
            h[j] = min(1.0, h[j] + diff)
        else:
            j = int(np.argmax(h))
            h[j] = max(0.0, h[j] + diff)
    return h

        hf = fast_cd(best_h, best_n, sweeps=30, time_limit=time_left(), t0=t0)
        c5f = compute_c5(hf)
        if c5f < best_c5:
            best_c5 = c5f
            best_h = hf

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
