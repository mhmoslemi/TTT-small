    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        # Distribute among interior points
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            # All at bounds, need to adjust
            if diff > 0:

                idx = np.argmin(h)
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + diff)

    n = len(h)
    g = np.zeros(n)

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            break
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 0.01) & (h < 0.99)

            h[mask] += diff / np.sum(mask)

    n = len(corr)
    if top_k and top_k < n:
        idx = np.argsort(corr)[-top_k:]
        m = corr[idx].max()
        shifted = corr[idx] - m

    else:

        m = corr[idx].max()
        shifted = corr[idx] - m

    else:

        shifted = corr - m

        return np.arange(n), w
