        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-14:
            for j in range(n):
                if h[j] > 1e-8 and h[j] < 1.0 - 1e-8:
                    h[j] = np.clip(h[j] + diff2, 0.0, 1.0)
                    break
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def _sm_obj(x):
    n = _N_G
    beta = _BETA_G
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def _sm_eq(x):
    return np.sum(x) - _N_G / 2.0

def _mm_obj(x):
    return x[_N_G]

def _mm_ineq(x):
    n = _N_G
    dx = _DX_G
    h = x[:n]
    t = x[n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - dx * corr[_LAGS_G]

def _mm_eq(x):
    return np.sum(x[:_N_G]) - _N_G / 2.0

def slsqp_smooth(h, n, beta, maxiter=300):
    global _N_G, _BETA_G
    _N_G = n
    _BETA_G = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sm_eq}]
    try:
        res = minimize(_sm_obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def slsqp_minimax(h_init, n, max_rounds=12, max_slsqp_iter=200):
    global _N_G, _DX_G, _LAGS_G
    _N_G = n
    _DX_G = 2.0 / n
    h = h_init.copy()
    dx = 2.0 / n

    for rnd in range(max_rounds):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        max_corr = np.max(corr)
        t_init = max_corr * dx + 1e-8

        threshold = max_corr - 0.5
        active = np.where(corr >= threshold)[0]
        top3 = np.argsort(corr)[::-1][:3]
        active = np.unique(np.concatenate([active, top3]))
        active = active[np.argsort(active)]

        _LAGS_G = active
        x0 = np.concatenate([h, [t_init]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': _mm_ineq},
            {'type': 'eq', 'fun': _mm_eq}
        ]
        try:
            res = minimize(_mm_obj, x0, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_slsqp_iter, 'ftol': 1e-14, 'iprint': 0})
            if res.x is not None:
                h_new = project_feasible(res.x[:n], n)
                c5_new = compute_c5(h_new)
                c5_old = compute_c5(h)
                if c5_new < c5_old - 1e-16:
                    h = h_new
                else:
                    if rnd > 2:
                        break
        except Exception:
            break
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def binarize_and_refine(h, n, threshold=0.15):
    """Round near-binary values, then re-optimize."""
    h_bin = h.copy()
    h_bin[h_bin < threshold] = 0.0
    h_bin[h_bin > 1.0 - threshold] = 1.0
    h_bin = project_feasible(h_bin, n)
    return h_bin

def pair_transfer_search(h, n, max_sweeps=6, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        any_imp = False
        order = rng.permutation(n)
        for ii in range(len(order)):
            if remaining_time() < 1:
                break
            i = int(order[ii])
            hi = best_h[i]
            if hi < 0.08 or hi > 0.92:
                continue
            for j in range(i + 1, min(i + 25, n)):
                hj = best_h[j]
                if abs(hi - hj) < 0.04:
                    continue
                for frac in [0.25, 0.5, 0.75]:
                    amt = (hi - hj) * frac
                    if abs(amt) < 0.01:
                        continue
                    h_trial = best_h.copy()
                    h_trial[i] -= amt
                    h_trial[j] += amt
                    h_trial = project_feasible(h_trial, n)
                    c5_t = compute_c5(h_trial)
                    if c5_t < best_c5 - 1e-15:
                        best_c5 = c5_t
                        best_h = h_trial.copy()
                        any_imp = True
                        break
            if not any_imp and ii > n // 3:
                break
        if not any_imp:
            break
    return best_h, best_c5

def perturb_restart(h, n, rng, sigma=0.05):
    """Create a perturbed version for restart."""
    h_p = h.copy()
    mask = (h_p > 0.05) & (h_p < 0.95)
    if np.sum(mask) > 0:
        h_p[mask] += rng.normal(0, sigma, np.sum(mask))
    idx = rng.choice(n, size=min(5, n), replace=False)
    for i in idx:
        h_p[i] = 1.0 - h_p[i]
    h_p = project_feasible(h_p, n)
    return h_p

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S, _N_G, _BETA_G, _LAGS_G, _DX_G
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 160
        best_h = np.full(best_n, 0.5)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: Escalating-beta smooth SLSQP
    for beta in [30, 80, 200, 500, 1000, 2000, 5000, 10000]:
        if remaining_time() < 12:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-16:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Minimax SLSQP
    if remaining_time() > 25:
        h_mm = slsqp_minimax(best_h, best_n, max_rounds=15, max_slsqp_iter=250)
        c5_mm = compute_c5(h_mm)
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm.copy()
        if remaining_time() > 10:
            h_pol = slsqp_smooth(best_h, best_n, 8000, maxiter=300)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 3: Binarize + re-optimize
    if remaining_time() > 30:
        h_bin = binarize_and_refine(best_h, best_n, threshold=0.12)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5 + 0.001:
            h_b2 = slsqp_smooth(h_bin, best_n, 500, maxiter=300)
            c5_b2 = compute_c5(h_b2)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()

    # Phase 4: Multi-resolution exploration
    trial_ns = [152, 156, 164, 168, 172, 176, 180, 184, 192, 200, 208, 224, 240, 256, 288, 320, 384, 448, 512, 640]
    for trial_n in trial_ns:
        if remaining_time() < 18:
            break
        h_interp = interp_to(best_h, best_n, trial_n)
        h_interp = project_feasible(h_interp, trial_n)
        c5_interp = compute_c5(h_interp)

        if c5_interp < best_c5 + 0.004:
            maxit = 300 if trial_n <= 256 else 200
            for beta in [200, 1000, 3000]:
                if remaining_time() < 6:
                    break
                h_new = slsqp_smooth(h_interp, trial_n, beta, maxiter=maxit)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = trial_n
                    h_interp = h_new.copy()

            if remaining_time() > 12 and trial_n <= 320:
                h_mm = slsqp_minimax(h_interp, trial_n, max_rounds=8, max_slsqp_iter=150)
                c5_new = compute_c5(h_mm)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_mm.copy()
                    best_n = trial_n

    # Phase 5: Pair transfer combinatorial search
    if remaining_time() > 20:
        h_comb, c5_comb = pair_transfer_search(best_h, best_n, max_sweeps=8, rng=rng)
        if c5_comb < best_c5:
            best_c5 = c5_comb
            best_h = h_comb.copy()
        if remaining_time() > 8:
            h_pol = slsqp_smooth(best_h, best_n, 5000, maxiter=200)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 6: Perturbation restarts
    for restart in range(4):
        if remaining_time() < 30:
            break
        h_p = perturb_restart(best_h, best_n, rng, sigma=0.04)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5 + 0.01:
            h_p2 = slsqp_smooth(h_p, best_n, 500, maxiter=300)
            c5_p2 = compute_c5(h_p2)
            if c5_p2 < best_c5:
                best_c5 = c5_p2
                best_h = h_p2.copy()

    # Phase 7: Final ultra-precision polish
    if remaining_time() > 10:
        for beta in [10000, 20000, 50000]:
            if remaining_time() < 4:
                break
            h_new = slsqp_smooth(best_h, best_n, beta, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 8: Final minimax
    if remaining_time() > 12:
        h_mm = slsqp_minimax(best_h, best_n, max_rounds=10, max_slsqp_iter=200)
        c5_mm = compute_c5(h_mm)
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm.copy()

    # Phase 9: Final combinatorial sweep
    if remaining_time() > 8:
        h_comb, c5_comb = pair_transfer_search(best_h, best_n, max_sweeps=4, rng=rng)
        if c5_comb < best_c5:
            best_c5 = c5_comb
            best_h = h_comb.copy()

    # Phase 10: Ultimate polish
    if remaining_time() > 5:
        h_new = slsqp_smooth(best_h, best_n, 30000, maxiter=150)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
