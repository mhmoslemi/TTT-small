def binary_sa(h_init, n, iters, rng, t0=0.003, t1=0.00003):
    h = np.where(h_init > 0.5, 1.0, 0.0)
    tgt_ones = n // 2
    n_ones = int(np.sum(h))
    if n_ones > tgt_ones:
        ones = np.where(h == 1.0)[0]
        rem = rng.choice(len(ones), n_ones - tgt_ones, replace=False)
        h[ones[rem]] = 0.0
    elif n_ones < tgt_ones:
        zeros = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros), tgt_ones - n_ones, replace=False)
        h[zeros[add]] = 1.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    
    for it in range(iters):
        if remaining_time() < 2:
            break
        i = rng.randint(len(ones_list))
        j = rng.randint(len(zeros_list))
        oi = ones_list[i]
        oz = zeros_list[j]
        
        h[oi] = 0.0
        h[oz] = 1.0
        nc5 = compute_c5(h)
        
        T = t0 * (t1 / t0) ** (it / max(iters, 1))
        if nc5 < cur_c5 or (T > 1e-15 and rng.rand() < np.exp(-(nc5 - cur_c5) / T)):
            cur_c5 = nc5
            ones_list[i] = oz
            zeros_list[j] = oi
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0
    
    return best_h, best_c5

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    return h

def make_sine(n, freq, phase, amp=0.3):
    t = np.arange(n) / float(n)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * t + phase)
    return np.clip(h, 0.0, 1.0)

def make_multi_sine(n, rng):
    t = np.arange(n) / float(n)
    h = 0.5
    for f in [2, 3, 5, 7]:
        h += rng.uniform(0.05, 0.15) * np.sin(2.0 * np.pi * f * t + rng.rand() * 6.28)
    return np.clip(h, 0.0, 1.0)
