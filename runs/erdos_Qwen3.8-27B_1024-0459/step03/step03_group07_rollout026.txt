import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - h)  # most room first
        else:
            idx = np.argsort(h)[::-1]  # largest first
        for i in range(min(len(idx), abs(n))):
            adj = min(diff, 1.0 - h[idx[i]]) if diff > 0 else max(diff, -h[idx[i]])
            h[idx[i]] += adj
            diff -= adj
            if abs(diff) < 1e-15:
                break
    return h

    quad_rest: the rest for the quadratic term (x - x^2), or None

def check_feasible(T, a, b, quad_rest, max_unaff):
    if T < max_unaff:
        return False, 0.0, 1.0
    x_lo = 0.0
    x_hi = 1.0
    # Affine constraints: a_i * x + b_i <= T
    # For a_i > 0: x <= (T - b_i) / a_i
    pos_mask = a > 1e-15
    neg_mask = a < -1e-15
    if np.any(pos_mask):
        x_hi = min(x_hi, np.min((T - b[pos_mask]) / a[pos_mask]))
    if np.any(neg_mask):
        x_lo = max(x_lo, np.min((T - b[neg_mask]) / a[neg_mask]))
    # Wait, for a_i < 0: a_i * x + b_i <= T => x >= (T - b_i) / a_i (dividing by negative flips)
    # Actually: a_i * x <= T - b_i. If a_i < 0: x >= (T - b_i) / a_i (since dividing by negative)
    # But (T - b_i) / a_i when a_i < 0: this is the LOWER bound
    # So x_lo = max(x_lo, max of (T - b_i)/a_i for a_i < 0)
    # Wait no: a_i * x + b_i <= T => a_i * x <= T - b_i
    # If a_i > 0: x <= (T - b_i) / a_i (upper bound)
    # If a_i < 0: x >= (T - b_i) / a_i (lower bound, and dividing by neg flips)
    # (T - b_i) / a_i when a_i < 0: e.g., a_i = -1, T - b_i = 2: x >= 2 / (-1) = -2. So x >= -2, which is always true.
    # Let me redo: a_i = -c (c > 0). -c*x + b_i <= T => -c*x <= T - b_i => c*x >= b_i - T => x >= (b_i - T) / c = (T - b_i) / a_i
    # So x_lo = max(x_lo, max_i (T - b_i) / a_i) for a_i < 0
    
    # Quadratic: quad_rest + x - x^2 <= T => x - x^2 <= T - quad_rest
    # x^2 - x + (T - quad_rest) >= 0
    # Let c = T - quad_rest. x^2 - x + c >= 0.
    # If c <= 0: always true (since x^2 - x + c has discriminant 1-4c >= 1 > 0, and the parabola is >= 0 outside the roots, and both roots are in [0,1]... wait)
    # Actually x^2 - x + c >= 0. Discriminant = 1 - 4c.
    # If 1 - 4c < 0 (c > 0.25): always positive, constraint always satisfied.
    # If 1 - 4c >= 0: roots are (1 ± sqrt(1-4c))/2. Parabola >= 0 outside [r1, r2].
    # So constraint: x notin (r1, r2), i.e., x <= r1 or x >= r2.
    
    if quad_rest is not None:
        c = T - quad_rest
        if c <= 0.25:
            disc = 1 - 4 * c
            r1 = (1 - np.sqrt(disc)) / 2
            r2 = (1 + np.sqrt(disc)) / 2
            # x must be in [0, r1] ∪ [r2, 1]
            # Check if [x_lo, x_hi] intersects this set
            # Left part: [x_lo, min(x_hi, r1)]
            # Right part: [max(x_lo, r2), x_hi]
            left_ok = min(x_hi, r1) >= x_lo
            right_ok = max(x_lo, r2) <= x_hi
            if left_ok and not right_ok:
                x_hi = min(x_hi, r1)
            elif right_ok and not left_ok:
                x_lo = max(x_lo, r2)
            elif not left_ok and not right_ok:
                return False, 0.0, 1.0
            # If both ok, keep [x_lo, x_hi] (but it might span the gap (r1, r2))
            # In that case, we should pick one side. For minimax, either works.
            # Let's keep the whole interval and handle it at the end.
    
    if x_lo > x_hi + 1e-14:
        return False, 0.0, 1.0
    x_lo = max(0.0, x_lo)
    x_hi = min(1.0, x_hi)
    return True, x_lo, x_hi

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)[::-1]
        for i in range(len(idx)):
            if abs(diff) < 1e-15:
                break
            if diff > 0:
                adj = min(diff, 1.0 - h[idx[i]])
            else:
                adj = max(diff, -h[idx[i]])
            h[idx[i]] += adj
            diff -= adj
    return h

            if not any(np.allclose(h_i, h_u, atol=1e-6) for h_u in unique_inits):
                unique_inits.append(h_i)
        inits = unique_inits
        
        # Allocate time per init
        n_inits = len(inits)
        time_per_init = max(20.0, min(200.0, time_left() / n_inits - 15.0))
        
        for h0 in inits:
            if time_left() < 30:
                break
            # Run coordinate descent
            tl = min(time_per_init, time_left() - 15)
            n_sweeps = max(5, int(tl / 0.05))  # estimate ~50ms per sweep
            h_cd, c5_cd = coord_descent_fast(h0, n, n_sweeps, t0, budget_s, reserve_s=15.0)
            
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n
            
            # If improved significantly, do more sweeps
            if c5_cd < best_c5 + 0.001 and time_left() > 60:
                tl2 = min(time_per_init * 0.5, time_left() - 15)
                if tl2 > 20:
                    h_cd2, c5_cd2 = coord_descent_fast(h_cd, n, max(10, int(tl2 / 0.05)), t0, budget_s, reserve_s=15.0)
                    if c5_cd2 < best_c5:
                        best_c5 = c5_cd2
                        best_h = h_cd2.copy()
                        best_n = n
    
    # Final intensive polish on best solution
    if time_left() > 40:
        h_pol, c5_pol = coord_descent_fast(best_h, best_n, max(20, int(time_left() * 0.5 / 0.05)), t0, budget_s, reserve_s=8.0)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    
    # One more round if time
    if time_left() > 20:
        h_pol2, c5_pol2 = coord_descent_fast(best_h, best_n, max(10, int(time_left() * 0.7 / 0.05)), t0, budget_s, reserve_s=5.0)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    # Validate with evaluator
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
