def project_feasible(h, n):

def interp_to(src_h, src_n, dst_n):

    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n

def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g


def smooth_obj_topk(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA

    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def smooth_grad_topk(x, k_limit=20):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    s = np.sum(ex)
    w = ex / s
    idx = np.argsort(-w)[:k_limit]
    g = np.zeros(n)
    for k in idx:
        if w[k] > 1e-12:
            g += w[k] * corr_grad_per_shift(x, n, int(k))
    return g * dx_val


def slsqp_smooth(h, n, beta, maxiter=300, topk=20):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n

    def _sum_con(x):
        return np.sum(x) - _TGT

    def _sum_jac(x):
        return np.ones(_N)

    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac}]
    try:
        def _obj(x):
            return smooth_obj_topk(x)

        def _grad(x):
            return smooth_grad_topk(x, k_limit=topk)

        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

    higher_ns = [nn for nn in higher_ns if 100 <= nn <= 1200 and nn % 2 == 0]
    higher_ns = sorted(set(higher_ns))

    for n_t in higher_ns:
        if remaining() < 40:
            break
        # Interpolate best solution
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)

        # SLSQP at this resolution
        h_opt = h_feas
        for beta in [500, 2000, 6000]:
            if remaining() < 15:
                break
            h_opt = slsqp_smooth(h_opt, n_t, beta, maxiter=200, topk=12)

        c5_opt = compute_c5(h_opt)

        # Binary local search (random sampling for larger n)
        if remaining() > 25:
            h_bin_t = (h_opt > 0.5).astype(float)
            n_ones_t = int(np.sum(h_bin_t))
            target_t = n_t // 2
            if n_ones_t > target_t:
                idx = np.where(h_bin_t == 1)[0]
                h_bin_t[rng.choice(idx, n_ones_t - target_t, replace=False)] = 0.0
            elif n_ones_t < target_t:
                idx = np.where(h_bin_t == 0)[0]
                h_bin_t[rng.choice(idx, target_t - n_ones_t, replace=False)] = 1.0
            h_bt, c5_bt = full_binary_local_search(h_bin_t, n_t,
                                                   time_limit=min(40, remaining() - 10), rng=rng)
            if c5_bt < c5_opt:
                c5_opt = c5_bt
                h_opt = h_bt

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 4: Try a few random structural inits at parent and similar n
    if remaining() > 80:
        for trial_seed in [seed + 1, seed + 7, seed + 13]:
            if remaining() < 30:
                break
            trng = np.random.RandomState(trial_seed)
            h_rand = np.zeros(n)
            indices = trng.choice(n, n // 2, replace=False)
            h_rand[indices] = 1.0

            # Quick local search
            h_r, c5_r = full_binary_local_search(h_rand, n, time_limit=min(30, remaining() - 15), rng=trng)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n

    # Phase 5: Final intensive refinement
    n = best_n
    h = best_h.copy()

    # SLSQP with high beta
    if remaining() > 20:
        for beta in [3000, 8000, 20000]:
            if remaining() < 8:
                break
            h_new = slsqp_smooth(h, n, beta, maxiter=300, topk=10)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Final binary polish
    if remaining() > 15:
        h_final_bin = (best_h > 0.5).astype(float)
        n_ones_f = int(np.sum(h_final_bin))
        target_f = best_n // 2
        if n_ones_f > target_f:
            idx = np.where(h_final_bin == 1)[0]
            h_final_bin[rng.choice(idx, n_ones_f - target_f, replace=False)] = 0.0
        elif n_ones_f < target_f:
            idx = np.where(h_final_bin == 0)[0]
            h_final_bin[rng.choice(idx, target_f - n_ones_f, replace=False)] = 1.0
        h_fb, c5_fb = full_binary_local_search(h_final_bin, best_n,
                                               time_limit=min(40, remaining() - 5), rng=rng)
        if c5_fb < best_c5:
            best_c5 = c5_fb
            best_h = h_fb.copy()

    # Final SLSQP on best
    if remaining() > 10:
        h_final = project_feasible(best_h, best_n)
        for beta in [5000, 15000]:
            if remaining() < 5:
                break
            h_p = slsqp_smooth(h_final, best_n, beta, maxiter=200, topk=8)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    # Try to verify with evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
        elif eval_c5 > best_c5 + 1e-8:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
