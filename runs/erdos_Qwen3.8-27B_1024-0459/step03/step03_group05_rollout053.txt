import numpy as np

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    for _ in range(40):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(free))
        if cnt == 0:
            break
        h[free] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_max_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    e = np.exp(beta * shifted)
    se = np.sum(e)
    w = e / se
    val = (m + np.log(se) / beta) * dx

    grad = np.zeros(n)
    for t in range(2 * n - 1):
        wt = w[t]
        if wt < 1e-12:
            continue
        s = t - (n - 1)
        if s >= 0:
            grad[s:] += wt * (1.0 - h[:n - s])
            grad[:n - s] -= wt * h[s:]
        else:
            ss = -s
            grad[:n - ss] += wt * (1.0 - h[ss:])
            grad[ss:] -= wt * h[:n - ss]
    grad *= dx
    return val, grad

    y = h_init.copy()
    best_x = h_init.copy()
    best_c5 = compute_c5(h_init)

        if time.time() - t0 > budget_s - 5.0:
            break
        val_y, grad_y = smooth_max_obj_grad(y, beta)
        lr = base_lr / (1.0 + 0.015 * i)
        x_new = y - lr * grad_y
        x_new = project_feasible(x_new, target)
        momentum = i / (float(i) + 2.0)
        y_new = x_new + momentum * (x_new - x)
        y_new = project_feasible(y_new, target)
        c5_new = compute_c5(x_new)

        x = x_new
        y = y_new
