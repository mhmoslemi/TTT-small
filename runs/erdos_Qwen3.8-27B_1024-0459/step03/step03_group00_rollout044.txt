        if budget_fn() < 8.0:
            break
        _G_BETA = float(beta)
        try:
            res = minimize(obj_smooth, h_cur, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per_beta, 'ftol': 1e-15, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-12:
                c5_cur = c5_new
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, c5_cur


def binary_local_search(h, n, budget_fn, rng, max_sweeps=50):
    """After continuous optimization, try binary refinement."""
    h_bin = (h >= 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h_bin))
    if n_ones > n // 2:
        # Flip some 1s to 0s (the "least important" ones)
        one_indices = np.where(h_bin == 1.0)[0]
        # Try to find which ones to flip based on local impact
        flips_needed = n_ones - n // 2
        # Use the continuous values as a guide
        order = np.argsort(h[one_indices])[::-1]  # flip the "least committed" ones first
        for i in range(min(flips_needed, len(order))):
            h_bin[one_indices[order[i]]] = 0.0
    elif n_ones < n // 2:
        zero_indices = np.where(h_bin == 0.0)[0]
        flips_needed = n // 2 - n_ones
        order = np.argsort(h[zero_indices])  # flip the "most committed" zeros
        for i in range(min(flips_needed, len(order))):
            h_bin[zero_indices[order[i]]] = 1.0
    
    c5_bin = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = c5_bin
    
    # Local search: try flipping pairs
    for sweep in range(max_sweeps):
        if budget_fn() < 3.0:
            break
        improved = False
        for _ in range(5):  # Try 5 random flips per sweep
            if budget_fn() < 2.0:
                break
            i, j = rng.choice(n, size=2, replace=False)
            if h_bin[i] == h_bin[j]:
                continue  # Would break the sum constraint
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            c5_test = compute_c5(h_bin)
            if c5_test < best_c5 - 1e-10:
                best_c5 = c5_test
                best_h = h_bin.copy()
                improved = True
            else:
                h_bin[i], h_bin[j] = h_bin[j], h_bin[i]  # Undo
        if not improved and sweep > 5:
            break
    
    return best_h, best_c5


def do_projected_gradient(h, n, budget_fn, rng, n_iter=200):
    """Projected subgradient descent with adaptive step size."""
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    dx = 2.0 / n
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    step = 0.1
    
    for it in range(n_iter):
        if budget_fn() < 3.0:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        n_corr = len(corr)
        # Subgradient of max: pick the argmax lag
        k_max = np.argmax(corr)
        offset = k_max - (n - 1)
        
        grad = np.zeros(n)
        if offset >= 0:
            n_pairs = n - offset
            grad[offset:offset + n_pairs] += (1.0 - h_cur[:n_pairs])
            grad[:n_pairs] -= h_cur[offset:offset + n_pairs]
        else:
            abs_off = -offset
            n_pairs = n - abs_off
            grad[:n_pairs] += (1.0 - h_cur[abs_off:abs_off + n_pairs])
            grad[abs_off:abs_off + n_pairs] -= h_cur[:n_pairs]
        
        grad *= dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Try multiple step sizes
        best_step = None
        for s in [step, step * 0.5, step * 0.25, step * 2.0, step * 0.75]:
            h_trial = h_cur - s * grad / gnorm * gnorm  # = h_cur - s * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_cur - 1e-12:
                best_step = s
                h_cur = h_trial
                c5_cur = c5_trial
                break
        
        if best_step is not None:
            step = min(step * 1.2, 0.5)
        else:
            step = max(step * 0.7, 0.005)
    
    return h_cur, c5_cur
