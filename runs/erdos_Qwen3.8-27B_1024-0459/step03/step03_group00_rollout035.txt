```python
def compute_grad(h, w):
    """Compute gradient of smooth objective.
    h: array of length n
    w: array of length 2n-1 (weights for each correlation lag)
    Returns: gradient array of length n
    """
    n = len(h)
    grad = np.zeros(n)
    for k in range(2*n-1):
        if abs(w[k]) < 1e-15: continue
        shift = k - (n-1)  # can be negative
        if shift >= 0:
            # corr[k] = sum_{i=0}^{n-1-shift} h[i]*(1-h[i+shift])
            # d/dh[i] for i=0..n-1-shift: (1-h[i+shift])
            # d/dh[i] for i=shift..n-1: -h[i-shift]
            for i in range(n - shift):
                grad[i] += w[k] * (1 - h[i + shift])
            for i in range(shift, n):
                grad[i] += w[k] * (-h[i - shift])
        else:
            abs_shift = -shift
            # corr[k] = sum_{i=0}^{n-1-abs_shift} h[i+abs_shift]*(1-h[i])
            # d/dh[i] for i=0..n-1-abs_shift: -h[i+abs_shift]  (from (1-h[i]) term)
            # d/dh[i] for i=abs_shift..n-1: (1-h[i-abs_shift])  (from h[i+abs_shift] term)
            for i in range(n - abs_shift):
                grad[i] += w[k] * (-h[i + abs_shift])
            for i in range(abs_shift, n):
                grad[i] += w[k] * (1 - h[i - abs_shift])
    return grad
```

- Sum constraint: add penalty term μ*(sum(h) - n/2)² to the objective
- The smooth objective: f(h) = (1/β)*log(∑_k exp(β*corr[k]*dx)) + μ*(sum(h)-n/2)²
- Gradient: computed efficiently

For the gradient of the log-sum-exp:
Let S = ∑_k exp(β*c_k) where c_k = corr[k]*dx
f_sle = (1/β)*log(S)
∂f_sle/∂h[i] = (1/S)*∑_k exp(β*c_k)*β*dx*∂corr[k]/∂h[i]
= dx * ∑_k p_k * ∂corr[k]/∂h[i]
where p_k = exp(β*c_k)/S

And ∂corr[k]/∂h[i] as derived earlier.

For the penalty term:
∂(μ*(sum(h)-n/2)²)/∂h[i] = 2μ*(sum(h)-n/2)

def sle_obj_grad(h, beta, mu, target_sum):
    """Log-sum-exp smooth approximation + sum penalty.
    Returns (objective, gradient).
    """
    n = len(h)

    exps = np.exp(np.clip(shifted, -500, 0))
    log_S = np.log(np.sum(exps))
    c_max_val = c_max + log_S / beta  # this is the SLE value
    
    # Weights
    p = exps / np.sum(exps)  # probability weights
    
    # Gradient of SLE part: dx * sum_k p_k * d(corr[k])/dh
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k in range(n_corr):
        if p[k] < 1e-12:
            continue
        shift = k - (n - 1)

            # d corr[k] / d h[i] for i=0..n_pairs-1: (1 - h[i+shift])
            # d corr[k] / d h[i] for i=shift..n-1: -h[i-shift]

            # d/dh[i] for i=abs_shift..n-1: (1-h[i-abs_shift])  (from h[i+abs_shift] term)
            grad[:n_pairs] += p[k] * (-h[abs_shift:abs_shift+n_pairs])
            grad[abs_shift:] += p[k] * (1.0 - h[:n_pairs])
    
    grad *= dx
    
    # Gradient of penalty
    sum_diff = np.sum(h) - target_sum
    grad += 2.0 * mu * sum_diff
    
    obj_val = c_max_val + mu * sum_diff * sum_diff
    return obj_val, grad

import numpy as np
from scipy.optimize import minimize
import time

_G_T0 = 0.0
_G_BUDGET = 1300.0


def _tl():
    return max(0.0, _G_BUDGET - (time.time() - _G_T0) - 5.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(60):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmax((h > 0.05) & (h < 0.95)))
        if 0 <= idx < len(h):
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def slo_obj_grad(h, beta, mu, target_sum):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = np.max(c)
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exps = np.exp(shifted)
    S = np.sum(exps)
    if S < 1e-300:
        sle_val = c_max
        grad = np.zeros(n)
    else:
        sle_val = c_max + np.log(S) / beta
        p = exps / S
        grad = np.zeros(n)
        n_corr = 2 * n - 1
        for k in range(n_corr):
            if p[k] < 1e-10:
                continue
            s = (n - 1) - k
            if s >= 0:
                n_pairs = n - s
                if n_pairs <= 0:
                    continue
                grad[:n_pairs] += p[k] * (1.0 - h[s:s + n_pairs])
                grad[s:] += p[k] * (-h[:n_pairs])
            else:
                ab = -s
                n_pairs = n - ab
                if n_pairs <= 0:
                    continue
                grad[ab:] += p[k] * (1.0 - h[:n_pairs])
                grad[:n_pairs] += p[k] * (-h[ab:ab + n_pairs])
        grad *= dx
    sum_diff = np.sum(h) - target_sum
    grad += 2.0 * mu * sum_diff
    obj_val = sle_val + mu * sum_diff * sum_diff
    return obj_val, grad


def raw_obj_grad(h, beta, mu, target_sum):
    """Use top-k weighted gradient for near-exact max."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    top_k = min(8, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:top_k]
    c_max = c[top_idx[0]]
    shifted = beta * (c[top_idx] - c_max)
    shifted = np.clip(shifted, -200.0, 0.0)
    exps = np.exp(shifted)
    S = np.sum(exps)
    if S < 1e-300:
        obj_val = c_max
        grad = np.zeros(n)
    else:
        sle_val = c_max + np.log(S) / beta
        p = exps / S
        grad = np.zeros(n)
        for idx_k in range(top_k):
            k = top_idx[idx_k]
            w = p[idx_k]
            s = (n - 1) - k
            if s >= 0:
                n_pairs = n - s
                if n_pairs <= 0:
                    continue
                grad[:n_pairs] += w * (1.0 - h[s:s + n_pairs])
                grad[s:] += w * (-h[:n_pairs])
            else:
                ab = -s
                n_pairs = n - ab
                if n_pairs <= 0:
                    continue
                grad[ab:] += w * (1.0 - h[:n_pairs])
                grad[:n_pairs] += w * (-h[ab:ab + n_pairs])
        grad *= dx
    sum_diff = np.sum(h) - target_sum
    grad += 2.0 * mu * sum_diff
    obj_val = sle_val + mu * sum_diff * sum_diff
    return obj_val, grad


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = 0
        v = i
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return h


def make_golden_ratio(n):
    h = np.zeros(n)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return h


def lbgfb_optimize(h_init, n, target_sum, beta, mu, maxiter, ftol, gtol):
    bounds = [(0.0, 1.0)] * n
    res = minimize(slo_obj_grad, h_init, method='L-BFGS-B',
                   args=(beta, mu, target_sum),
                   bounds=bounds, jac=True,
                   options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol, 'iprint': 0})
    h_new = project_feasible(res.x, target_sum)
    return h_new


def topk_optimize(h_init, n, target_sum, beta, mu, maxiter):
    bounds = [(0.0, 1.0)] * n
    res = minimize(raw_obj_grad, h_init, method='L-BFGS-B',
                   args=(beta, mu, target_sum),
                   bounds=bounds, jac=True,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12, 'iprint': 0})
    h_new = project_feasible(res.x, target_sum)
    return h_new


def subgradient_phase(h, n, target_sum, max_iters):
    c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    for it in range(max_iters):
        if _tl() < 8.0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        top_k = min(5, 2 * n - 1)
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for k_idx in top_idx:
            kk = int(k_idx)
            s = (n - 1) - kk
            if s >= 0:
                n_pairs = n - s
                if n_pairs <= 0:
                    continue
                grad[:n_pairs] += (1.0 - h[s:s + n_pairs])
                grad[s:] += (-h[:n_pairs])
            else:
                ab = -s
                n_pairs = n - ab
                if n_pairs <= 0:
                    continue
                grad[ab:] += (1.0 - h[:n_pairs])
                grad[:n_pairs] += (-h[ab:ab + n_pairs])
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        improved = False
        for step in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.03, 0.08, 0.15, 0.4]:
            h_try = h - step * grad
            h_try = project_feasible(h_try, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < c5 - 1e-11:
                c5 = c5_try
                h = h_try.copy()
                best_h = h.copy()
                improved = True
                break
        if not improved:
            if _tl() > 30 and it % 10 == 5:
                try:

                    c5_new = compute_c5(h_new)
                    if c5_new < c5 - 1e-11:
                        c5 = c5_new
                        h = h_new
                        best_h = h.copy()
                        improved = True
                except Exception:
                    pass
            if not improved and it > 30:
                break
    return best_h, c5
