    for k in range(len(corr)):
        if w[k] > 1e-14:
            g += w[k] * corr_grad_per_shift(x, n, k)
    return g * dx_val


def sum_con(x):
    return np.sum(x) - _TGT


def sum_con_jac(x):
    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=400):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def make_thue_morse(n, seed_val):
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c == 1 else 0.0
    # Fix count
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        rem = rng.choice(ones, n_ones - target, replace=False)
        h[rem] = 0.0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        add = rng.choice(zeros, target - n_ones, replace=False)
        h[add] = 1.0
    return h


def make_periodic(n, seed_val):
    """Create periodic-like pattern with perturbation."""
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    # Try pattern 1010 with some block structure
    period = rng.choice([4, 6, 8, 10, 12, 7, 9, 11])
    pattern = rng.randint(0, 2, period)
    # Ensure half ones in pattern
    half = period // 2
    if int(np.sum(pattern)) > half:
        ones = np.where(pattern == 1)[0]
        pattern[rng.choice(ones, int(np.sum(pattern)) - half, replace=False)] = 0
    elif int(np.sum(pattern)) < half:
        zeros = np.where(pattern == 0)[0]
        pattern[rng.choice(zeros, half - int(np.sum(pattern)), replace=False)] = 1
    for i in range(n):
        h[i] = pattern[i % period]
    return h


def simulated_annealing_binary(h, n, time_limit, rng, t0=1.0, cool=0.99997):
    """Simulated annealing for binary sequence with pair swaps."""
    h = h.copy()
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n
    best_c5 = cur_max * dx
    T = t0
    start = time.time()
    it_count = 0

    while time.time() - start < time_limit and remaining() > 5:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        delta = new_max - cur_max
        if delta < 0 or (T > 0.01 and rng.rand() < np.exp(-delta / T)):
            cur_max = new_max
            corr = new_corr
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                best_c5 = new_max * dx
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cool
        it_count += 1
    return best_h, best_c5


def local_search_binary(h, n, max_rounds=200, time_limit=60, rng=None):
    """Targeted local search: minimize the top correlations."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n
    best_c5 = cur_max * dx
    start = time.time()

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break
        corr = compute_corr(h)
        cur_max = float(np.max(corr))
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
            best_c5 = best_max * dx

        k_worst = int(np.argmax(corr))
        offset = k_worst - (n - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        found = False
        # Biased sampling: prefer ones that contribute to worst lag
        n_try = 800
        for _ in range(n_try):
            if offset >= 0 and len(ones) > 0:
                ci = [i for i in ones if i + offset < n and h[int(i + offset)] == 0]
                if ci and rng.rand() < 0.8:
                    i_val = ci[rng.randint(len(ci))]
                else:
                    i_val = ones[rng.randint(len(ones))]
            elif offset < 0 and len(ones) > 0:
                ci = [i for i in ones if i - offset < n and h[int(i - offset)] == 0]
                if ci and rng.rand() < 0.8:
                    i_val = ci[rng.randint(len(ci))]
                else:
                    i_val = ones[rng.randint(len(ones))]
            else:
                i_val = ones[rng.randint(len(ones))]
            j_val = zeros[rng.randint(len(zeros))]
            h[int(i_val)] = 0.0
            h[int(j_val)] = 1.0
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            if new_max < cur_max - 1e-12:
                cur_max = new_max
                corr = new_corr
                found = True
            else:
                h[int(i_val)] = 1.0
                h[int(j_val)] = 0.0
        if not found:
            break
    return best_h, best_c5


def double_swap_search(h, n, time_limit, rng):
    """Try 2-for-2 swaps (swap two ones with two zeros)."""
    h = h.copy()
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n
    best_c5 = cur_max * dx
    start = time.time()
    stall = 0

    while time.time() - start < time_limit and remaining() > 5:
        if stall > 2000:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i1 = ones[rng.randint(len(ones))]
        j1 = ones[rng.randint(len(ones))]
        while j1 == i1:
            j1 = ones[rng.randint(len(ones))]
        k1 = zeros[rng.randint(len(zeros))]
        l1 = zeros[rng.randint(len(zeros))]
        while l1 == k1:
            l1 = zeros[rng.randint(len(zeros))]
        # Also try single swap
        if rng.rand() < 0.4:
            i_val = ones[rng.randint(len(ones))]
            j_val = zeros[rng.randint(len(zeros))]
            h[int(i_val)] = 0.0
            h[int(j_val)] = 1.0
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            if new_max < cur_max - 1e-12:
                cur_max = new_max
                stall = 0
                if new_max < best_max:
                    best_max = new_max
                    best_h = h.copy()
                    best_c5 = best_max * dx
            else:
                h[int(i_val)] = 1.0
                h[int(j_val)] = 0.0
                stall += 1
        else:
            h[int(i1)] = 0.0
            h[int(j1)] = 0.0
            h[int(k1)] = 1.0
            h[int(l1)] = 1.0
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            if new_max < cur_max - 1e-12:
                cur_max = new_max
                stall = 0
                if new_max < best_max:
                    best_max = new_max
                    best_h = h.copy()
                    best_c5 = best_max * dx
            else:
                h[int(i1)] = 1.0
                h[int(j1)] = 1.0
                h[int(k1)] = 0.0
                h[int(l1)] = 0.0
                stall += 1
    return best_h, best_c5


def active_set_equalize(h, n, max_iters=200):
    h = h.copy()
    dx_val = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        top2 = np.argsort(-corr)[:2]
        c1, c2 = corr[int(top2[0])], corr[int(top2[1])]
        if c1 - c2 < 1e-10:
            break
        g1 = corr_grad_per_shift(h, n, int(top2[0]))
        g2 = corr_grad_per_shift(h, n, int(top2[1]))
        d = g2 - g1
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-14:
            d = -g1
            dnorm = np.linalg.norm(d)
            if dnorm < 1e-14:
                break
        d /= dnorm
        improved = False
        for step in [0.01, 0.02, 0.05, 0.005, 0.03, 0.08]:
            h_trial = project_feasible(h + step * d, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-12:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 320
        parent_h = make_thue_morse(parent_n, seed)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Binary optimization at parent resolution
    n = best_n
    h_bin = (best_h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones != n // 2:
        if n_ones > n // 2:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[rng.choice(len(idx), n_ones - n // 2, replace=False)]] = 0
        else:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[rng.choice(len(idx), n // 2 - n_ones, replace=False)]] = 1

    # SA at parent n
    sa_time = min(120, max(30, remaining() * 0.15))
    h_sa, c5_sa = simulated_annealing_binary(h_bin, n, sa_time, rng, t0=0.8, cool=0.99998)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()
        best_n = n
        h_bin = h_sa.copy()

    # Local search
    ls_time = min(120, max(20, remaining() * 0.12))
    h_ls, c5_ls = local_search_binary(h_sa, n, max_rounds=500, time_limit=ls_time, rng=rng)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls.copy()
        best_n = n
        h_bin = h_ls.copy()

    # Double swap
    ds_time = min(80, max(15, remaining() * 0.08))
    h_ds, c5_ds = double_swap_search(h_ls, n, ds_time, rng)
    if c5_ds < best_c5:
        best_c5 = c5_ds
        best_h = h_ds.copy()
        best_n = n

    # Phase 2: Try multiple resolutions
    res_list = [320, 318, 322, 384, 400, 480, 512, 640]
    # Add parent n if not in list
    if best_n not in res_list:
        res_list.insert(0, best_n)
    # Sort by expected difficulty (prefer n where target is more achievable)
    res_list = [r for r in res_list if 160 <= r <= 800 and r % 2 == 0]
    res_list = sorted(set(res_list))

    for n_t in res_list:
        if remaining() < 60:
            break
        if n_t == best_n and best_c5 < 0.38:
            continue

        # Initialize at this resolution
        # Interpolate from best
        h_interp = interp_to(best_h, best_n, n_t)
        h_int_bin = (h_interp > 0.5).astype(float)
        n_ones_t = int(np.sum(h_int_bin))
        if n_ones_t != n_t // 2:
            if n_ones_t > n_t // 2:
                idx = np.where(h_int_bin == 1)[0]
                h_int_bin[idx[rng.choice(len(idx), n_ones_t - n_t // 2, replace=False)]] = 0
            else:
                idx = np.where(h_int_bin == 0)[0]
                h_int_bin[idx[rng.choice(len(idx), n_t // 2 - n_ones_t, replace=False)]] = 1

        # Also try Thue-Morse and random initializations
        h_tm = make_thue_morse(n_t, seed + n_t)
        h_rand = np.zeros(n_t)
        h_rand[rng.choice(n_t, n_t // 2, replace=False)] = 1.0

        # Try each initialization with SA
        inits = [h_int_bin, h_tm, h_rand]
        sa_time_t = min(80, max(20, remaining() * 0.08))
        for init_h in inits:
            if remaining() < 30:
                break
            time_per_sa = sa_time_t / len(inits)
            h_sa_t, c5_sa_t = simulated_annealing_binary(init_h, n_t, time_per_sa, rng,
                                                         t0=1.0, cool=0.99997)
            if c5_sa_t < best_c5:
                best_c5 = c5_sa_t
                best_h = h_sa_t.copy()
                best_n = n_t

        # Local search on best at this n
        if remaining() > 30:
            h_ls_t = (best_h if best_n == n_t else h_int_bin).copy()
            ls_time_t = min(60, max(10, remaining() * 0.06))
            h_ls_t, c5_ls_t = local_search_binary(h_ls_t, n_t, max_rounds=300,
                                                   time_limit=ls_time_t, rng=rng)
            if c5_ls_t < best_c5:
                best_c5 = c5_ls_t
                best_h = h_ls_t.copy()
                best_n = n_t

    # Phase 3: Continuous refinement on best binary solution
    n = best_n
    h = (best_h > 0.5).astype(float)
    # Ensure exact count
    n_ones = int(np.sum(h))
    if n_ones != n // 2:
        if n_ones > n // 2:
            idx = np.where(h == 1)[0]
            h[idx[rng.choice(len(idx), n_ones - n // 2, replace=False)]] = 0
        else:
            idx = np.where(h == 0)[0]
            h[idx[rng.choice(len(idx), n // 2 - n_ones, replace=False)]] = 1
    h = project_feasible(h, n)

    # SLSQP with increasing beta
    for beta in [300, 1000, 3000, 8000, 15000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Active set equalize
    if remaining() > 10:
        h_eq, c5_eq = active_set_equalize(best_h, n, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 4: Intensive binary search on best solution
    if remaining() > 20:
        h_bin_final = (best_h > 0.5).astype(float)
        n_ones_f = int(np.sum(h_bin_final))
        if n_ones_f != n // 2:
            if n_ones_f > n // 2:
                idx = np.where(h_bin_final == 1)[0]
                h_bin_final[idx[rng.choice(len(idx), n_ones_f - n // 2, replace=False)]] = 0
            else:
                idx = np.where(h_bin_final == 0)[0]
                h_bin_final[idx[rng.choice(len(idx), n // 2 - n_ones_f, replace=False)]] = 1

        # Aggressive local search
        ls_time_f = min(100, max(20, remaining() * 0.15))
        h_ls_f, c5_ls_f = local_search_binary(h_bin_final, n, max_rounds=1000,
                                               time_limit=ls_time_f, rng=rng)

        # Double swap
        if remaining() > 15:
            ds_time_f = min(60, max(10, remaining() * 0.1))
            h_ds_f, c5_ds_f = double_swap_search(h_ls_f, n, ds_time_f, rng)
            if c5_ds_f < c5_ls_f:
                h_ls_f, c5_ls_f = h_ds_f, c5_ds_f

        if c5_ls_f < best_c5:
            best_c5 = c5_ls_f
            best_h = h_ls_f.copy()

        # Final SLSQP on best binary
        if remaining() > 15:
            h_cont = project_feasible(h_ls_f.copy(), n)
            for beta in [5000, 12000, 25000]:
                if remaining() < 8:
                    break
                h_new = slsqp_smooth(h_cont, n, beta, maxiter=600)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_cont = h_new.copy()

    # Phase 5: Final polish
    n = best_n
    if remaining() > 10:
        h_eq2, c5_eq2 = active_set_equalize(best_h, n, max_iters=200)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
