import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -80, 80)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -80, 80)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-14:
            off = k - (n - 1)
            if off >= 0:
                npairs = n - off
                g[off:off + npairs] += w[k] * (1.0 - x[:npairs])
                g[:npairs] -= w[k] * x[off:off + npairs]
            else:
                ao = -off
                npairs = n - ao
                g[:npairs] += w[k] * (1.0 - x[ao:ao + npairs])
                g[ao:ao + npairs] -= w[k] * x[:npairs]
    return g * dx_val


def sum_con(x):
    return np.sum(x) - _TGT


def sum_con_jac(x):
    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=500):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})

    except Exception:

def active_set_equalize(h, n, max_iters=300, top_k=3):
    """Equalize top-k correlations via projected gradient with adaptive step."""
    h = h.copy()
    dx_val = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:top_k]
        top_vals = corr[top_idx]
        spread = top_vals[0] - top_vals[-1]
        if spread < 1e-9:
            break
        # Weighted gradient: reduce top correlations, emphasize the highest
        weights = np.zeros(len(corr))
        max_w = top_vals[0] - top_vals[-1]
        if max_w > 1e-15:
            for idx in top_idx:
                weights[idx] = (top_vals[0] - top_vals[idx] + 1e-6) / max_w
        else:
            weights[top_idx] = 1.0
        weights /= np.sum(weights)
        g = np.zeros(n)
        for ki in top_idx:
            k = int(ki)
            w_val = weights[top_idx.tolist().index(k)]
            off = k - (n - 1)
            if off >= 0:
                npairs = n - off
                g[off:off + npairs] += w_val * (1.0 - h[:npairs])
                g[:npairs] -= w_val * h[off:off + npairs]
            else:
                ao = -off
                npairs = n - ao
                g[:npairs] += w_val * (1.0 - h[ao:ao + npairs])
                g[ao:ao + npairs] -= w_val * h[:npairs]
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g /= gnorm
        improved = False
        for step in [0.005, 0.01, 0.02, 0.003, 0.05, 0.002, 0.08, 0.015]:
            h_trial = h - step * g
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-13:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            # Try smaller steps
            for step in [0.001, 0.0005, 0.002, 0.0008]:
                h_trial = h - step * g
                h_trial = project_feasible(h_trial, n)
                c_new = compute_c5(h_trial)
                if c_new < best_c5 - 1e-13:
                    h = h_trial
                    best_c5 = c_new
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def binary_local_search(h, n, time_limit=30, rng=None):

        h[remove] = 0.0
    elif n_ones < n // 2:
        zeros_idx = np.where(h == 0)[0]
        add = rng.choice(zeros_idx, n // 2 - n_ones, replace=False)
        h[add] = 1.0

    dx_val = 2.0 / n
    corr = compute_corr(h)

    best_c5 = cur_max * dx_val
    start = time.time()
    stall = 0

    while time.time() - start < time_limit and remaining() > 5:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_improvement = 1e-10
        best_i = -1
        best_j = -1
        best_new_corr = None

        # For each one-position, compute the effect vector
        # Then for each zero-position, compute combined effect
        n_corr = 2 * n - 1
        # Precompute shift indices
        shifts = np.arange(-(n - 1), n)

                    if i < n - kp:

                        dk += h[i - kp]

                        if j + k < n and j + k >= k:

                            dk -= h[j - kp]

                        best_j = j
                        best_new_corr = new_corr

            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = best_new_corr

def multi_start_binary(n, seed_val, rng):
    """Generate diverse binary starting points."""
    h = np.zeros(n)
    mode = seed_val % 5
    if mode == 0:
        # Thue-Morse
        for i in range(n):
            c = 0
            v = i
            while v:
                c ^= (v & 1)
                v >>= 1
            h[i] = 1.0 if c == 1 else 0.0
    elif mode == 1:
        # Alternating blocks
        block = max(1, n // 40)
        for i in range(n):
            h[i] = 1.0 if (i // block) % 2 == 0 else 0.0
    elif mode == 2:
        # Random
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    elif mode == 3:
        # Fibonacci-like
        for i in range(n):
            h[i] = 1.0 if (i * 7 + 3) % n < n // 2 else 0.0
    else:
        # Sinusoidal
        h[:] = 0.5 + 0.5 * np.sin(2 * np.pi * 3 * np.arange(n) / n)
        h = (h > 0.5).astype(float)
        # Fix count
        cnt = int(np.sum(h))
        if cnt > n // 2:
            idx = np.where(h == 1)[0]
            h[idx[rng.choice(len(idx), cnt - n // 2, replace=False)]] = 0
        elif cnt < n // 2:
            idx = np.where(h == 0)[0]
            h[idx[rng.choice(len(idx), n // 2 - cnt, replace=False)]] = 1
    # Fix count for all modes
    cnt = int(np.sum(h))
    if cnt > n // 2:
        idx = np.where(h == 1)[0]
        h[idx[rng.choice(len(idx), cnt - n // 2, replace=False)]] = 0
    elif cnt < n // 2:
        idx = np.where(h == 0)[0]
        h[idx[rng.choice(len(idx), n // 2 - cnt, replace=False)]] = 1
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 320
        parent_h = multi_start_binary(parent_n, 0, rng)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Quick SLSQP refinement of parent
    n = best_n
    h = best_h.copy()
    for beta in [200, 1000, 5000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n
            h = h_new.copy()

    # Active set equalize
    if remaining() > 10:
        h_eq, c5_eq = active_set_equalize(h, n, max_iters=200, top_k=3)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 2: Multi-resolution search
    n_tries = sorted(set([
        best_n, best_n - 20, best_n + 20,
        best_n + 40, best_n + 60, best_n + 80,
        best_n + 120, best_n + 160,
        int(best_n * 1.5), int(best_n * 2),
        200, 240, 280, 400, 480, 560, 640, 720, 800
    ]))
    n_tries = [nn for nn in n_tries if 80 <= nn <= 1200 and nn % 2 == 0]
    n_tries = list(dict.fromkeys(n_tries))  # dedupe preserving order

    for n_t in n_tries:
        if remaining() < 20:
            break
        if n_t == best_n:
            continue

        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)

        # SLSQP refinement
        h_opt = h_feas.copy()
        for beta in [300, 1500, 8000]:
            if remaining() < 10:
                break
            h_opt = slsqp_smooth(h_opt, n_t, beta, maxiter=300)
        c5_opt = compute_c5(h_opt)

        # Active set
        h_opt, c5_opt = active_set_equalize(h_opt, n_t, max_iters=150, top_k=3)

        if c5_opt < best_c5 - 1e-6:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 3: Binary local search on best n
    n_bin = best_n
    h_bin = (best_h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones != n_bin // 2:
        if n_ones > n_bin // 2:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[rng.choice(len(idx), n_ones - n_bin // 2, replace=False)]] = 0
        else:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[rng.choice(len(idx), n_bin // 2 - n_ones, replace=False)]] = 1

    t_limit = min(60, max(10, remaining() - 30))
    h_b, c5_b = fast_binary_search(h_bin, n_bin, time_limit=t_limit, rng=rng)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
        best_n = n_bin

    # Phase 4: Try different binary starts at best_n
    for trial in range(5):
        if remaining() < 15:
            break
        h_start = multi_start_binary(best_n, seed + 100 + trial * 7, rng)
        h_b2, c5_b2 = fast_binary_search(h_start, best_n,
                                         time_limit=min(15, max(5, remaining() - 10)), rng=rng)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

    # Phase 5: Final intensive SLSQP polish on best solution
    n = best_n
    h = best_h.copy()
    for beta in [3000, 10000, 30000, 80000]:
        if remaining() < 6:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Active set with more iterations
    if remaining() > 10:
        h_eq, c5_eq = active_set_equalize(h, n, max_iters=500, top_k=4)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Another SLSQP pass after equalization
    for beta in [15000, 50000, 150000]:
        if remaining() < 5:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final active set
    if remaining() > 8:
        h_eq, c5_eq = active_set_equalize(h, n, max_iters=300, top_k=5)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Binary fine-tune at the end
    if remaining() > 15:
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        if n_ones != best_n // 2:
            if n_ones > best_n // 2:
                idx = np.where(h_bin == 1)[0]
                h_bin[idx[rng.choice(len(idx), n_ones - best_n // 2, replace=False)]] = 0
            else:
                idx = np.where(h_bin == 0)[0]
                h_bin[idx[rng.choice(len(idx), best_n // 2 - n_ones, replace=False)]] = 1
        h_b3, c5_b3 = fast_binary_search(h_bin, best_n,
                                         time_limit=min(20, max(5, remaining() - 8)), rng=rng)
        if c5_b3 < best_c5:
            best_c5 = c5_b3
            best_h = h_b3.copy()
            best_n = best_n
            # Re-optimize continuous
            if remaining() > 8:
                h_new = slsqp_smooth(best_h, best_n, 20000, maxiter=300)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
