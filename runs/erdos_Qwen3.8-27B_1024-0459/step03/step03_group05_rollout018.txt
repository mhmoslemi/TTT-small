```python
def project_box_sum(y, target):
    """Project y onto {0≤x≤1, sum x = target}"""
    n = len(y)
    # Bisection on lambda
    lo, hi = -10, 10
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(y - mid, 0, 1)
        if np.sum(x) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo+hi)/2, 0, 1)
    # Adjust sum exactly
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        mask = (x > 1e-10) & (x < 1-1e-10)
        if np.sum(mask) > 0:
            x[mask] += diff / np.sum(mask)
    return x
```

When h[i] changes by δ:

where s(k) = k-(n-1).

This is O(n) to compute for all k, same as the full gradient. So I can compute the change in C5 in O(n) per single-variable perturbation. That makes coordinate descent much faster.

```python
                break
            h[i] -= delta
            h[j] += delta
    return h


def interpolate_h(h_src, n_src, n_dst):
    """Interpolate h from n_src points to n_dst points."""
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_new = np.interp(xn, xp, h_src)
    return h_new


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    def remaining():
        return max(0.0, budget_s - (time.time() - t0) - 5.0)

    rng = np.random.RandomState(seed)

    h_par = np.array(initial_h_values, dtype=float)
    n_par = len(h_par)
    c5_par = compute_c5_val(h_par)
    print("Parent: n={}, C5={:.8f}".format(n_par, c5_par))

    best_h = h_par.copy()
    best_c5 = c5_par
    best_n = n_par

    # Phase 1: Subgradient descent at parent resolution
    n = n_par
    target = n / 2.0
    h = best_h.copy()
    beta_sw = 30.0
    top_k = 8

    for it in range(3000):
        if remaining() < 15:
            break
        h, c5_new = subgradient_iteration(h, target, beta_sw, top_k)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            best_n = n
            if it % 100 == 0:
                print("  Subgrad it {}: C5={:.8f}".format(it, best_c5))
        if it > 0 and it % 200 == 0:
            beta_sw *= 1.3
            top_k = min(12, top_k + 1)

    print("After Phase 1 (subgradient n={}): C5={:.8f}".format(n, best_c5))

    # Phase 2: Coordinate descent sweeps
    if remaining() > 30:
        n = best_n
        target = n / 2.0
        h = best_h.copy()
        for sweep in range(15):
            if remaining() < 10:
                break
            h = coord_sweep(h, target, [0.02, 0.01, 0.005, 0.002])
            c5_new = compute_c5_val(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                best_n = n
                print("  Coord sweep {}: C5={:.8f}".format(sweep, best_c5))
            if compute_c5_val(h) >= best_c5 + 1e-10:
                h = best_h.copy()

    # Phase 3: Try complement 1-h
    if remaining() > 30:
        n = best_n
        target = n / 2.0
        h_comp = 1.0 - best_h.copy()
        h_comp = project_box_sum(h_comp, target)
        c5_comp = compute_c5_val(h_comp)
        print("Complement C5={:.8f}".format(c5_comp))
        if c5_comp < best_c5:
            best_c5 = c5_comp
            best_h = h_comp.copy()
            best_n = n

    # Phase 4: Multi-resolution search
    if remaining() > 60:
        h_base = best_h.copy()
        n_base = best_n
        for n_try in [n_base - 2, n_base + 2, n_base + 4, n_base + 6, n_base - 4, n_base + 8]:
            if remaining() < 25:
                break
            if n_try < 40 or n_try > 300:
                continue
            target_try = n_try / 2.0
            h_try = interpolate_h(h_base, n_base, n_try)
            h_try = project_box_sum(h_try, target_try)
            c5_try = compute_c5_val(h_try)

            # Subgradient at this resolution
            for it in range(1500):
                if remaining() < 10:
                    break
                h_try, c5_try = subgradient_iteration(h_try, target_try, 40.0, 8)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    best_n = n_try
                    h_base = best_h.copy()
                    n_base = best_n
                    print("  Res n={} it {}: C5={:.8f}".format(n_try, it, best_c5))

            # Quick coord sweep
            if remaining() > 15:
                h_try = best_h.copy() if best_n == n_try else interpolate_h(best_h, best_n, n_try)
                h_try = project_box_sum(h_try, target_try)
                for sw in range(5):
                    if remaining() < 8:
                        break
                    h_try = coord_sweep(h_try, target_try, [0.01, 0.005, 0.002])
                    c5_try = compute_c5_val(h_try)
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        best_n = n_try
                        print("  Res n={} sweep {}: C5={:.8f}".format(n_try, sw, best_c5))

    # Phase 5: Pair coordinate descent (random)
    if remaining() > 20:
        n = best_n
        target = n / 2.0
        h = best_h.copy()
        deltas = [0.01, 0.005, 0.003, 0.002, 0.001]
        for batch in range(500):
            if remaining() < 5:
                break
            h = pair_sweep(h, target, deltas, rng, 500)
            c5_new = compute_c5_val(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                best_n = n
                if batch % 50 == 0:
                    print("  Pair batch {}: C5={:.8f}".format(batch, best_c5))
            else:
                h = best_h.copy()

    # Phase 6: Fine-tuning with very small steps
    if remaining() > 10:
        n = best_n
        target = n / 2.0
        h = best_h.copy()
        order = rng.permutation(n)
        for sweep in range(30):
            if remaining() < 4:
                break
            improved = False
            for i in order:
                for d in [0.002, 0.001, 0.003]:
                    for sign in [1, -1]:
                        h[i] += sign * d
                        h = project_box_sum(h, target)
                        c5_test = compute_c5_val(h)
                        if c5_test < best_c5 - 1e-15:
                            best_c5 = c5_test
                            best_h = h.copy()
                            improved = True
                            break
                        else:
                            h[i] -= sign * d
                            h = project_box_sum(h, target)
                    if improved:
                        break
                if not improved:
                    h = best_h.copy()
            if not improved:
                break

    # Phase 7: Final subgradient polish
    if remaining() > 8:
        n = best_n
        target = n / 2.0
        h = best_h.copy()
        for it in range(500):
            if remaining() < 3:
                break
            h, c5_new = subgradient_iteration(h, target, 50.0, 12)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                best_n = n

    # Final projection and validation
    best_h = project_box_sum(best_h, best_n / 2.0)
    final_c5 = compute_c5_val(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.8f}".format(best_n, best_c5))
    return (best_h, best_c5, best_n)
```
