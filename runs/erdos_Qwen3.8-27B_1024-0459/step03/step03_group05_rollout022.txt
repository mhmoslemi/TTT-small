import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            break
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, beta, top_k=50):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    if top_k < len(corr):
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
    else:
        c = corr
    m = c.max()
    shifted = c - m
    return (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx


def smooth_grad(h, beta, top_k=50):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    if top_k < len(corr):
        idx_all = np.argsort(corr)[-top_k:]
    else:
        idx_all = np.arange(len(corr))
    c = corr[idx_all]
    m = c.max()
    w = np.exp(beta * (c - m))
    w /= np.sum(w)
    grad = np.zeros(n)
    for t in range(len(idx_all)):
        if w[t] < 1e-14:
            continue
        k = idx_all[t]
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        if j_lo > j_hi:
            continue
        js = np.arange(j_lo, j_hi + 1)
        ih = k - js
        grad[js] += w[t] * (1.0 - 2.0 * h[ih])
    return grad * dx
