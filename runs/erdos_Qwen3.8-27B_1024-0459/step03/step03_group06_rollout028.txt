def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

            h[unsat] += diff / cnt
        else:

def topk_gradient(h, k):
    """Compute gradient of sum of top-k correlations (unnormalized)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-k:]

                grad[m] -= h[m + s]

    return sm + penalty

def smooth_max_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    weights /= np.sum(weights)
    
    grad = np.zeros(n)

        wj = weights[j]
        if wj < 1e-14:
            continue
        s = j - (n - 1)

                grad[m_idx] += wj * (1.0 - x[m_idx - s])

import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_BETA_GLOBAL = 100.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_all_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        unsat = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(unsat)
        if cnt > 0:
            h[unsat] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        candidates = np.where((h > 0.005) & (h < 0.995))[0]
        if len(candidates) > 0:
            idx = candidates[np.argmin(np.abs(h[candidates] - 0.5))]
        else:
            idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def topk_gradient(h, k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-k:]
    grad = np.zeros(n)
    for j in top_idx:
        s = j - (n - 1)
        if s >= 0:
            for m in range(s, n):
                grad[m] += (1.0 - h[m - s])
            for m in range(0, n - s):
                grad[m] -= h[m + s]
        else:
            sp = -s
            for m in range(0, n - sp):
                grad[m] += (1.0 - h[m + sp])
            for m in range(sp, n):
                grad[m] -= h[m - sp]
    return grad * dx


def smooth_max_obj(x):
    n = len(x)
    dx = 2.0 / n
    beta = _BETA_GLOBAL
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    sm = (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx
    penalty = 1000.0 * (np.sum(x) - n / 2.0) ** 2
    return sm + penalty


def smooth_max_grad(x):
    n = len(x)
    dx = 2.0 / n
    beta = _BETA_GLOBAL
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    total_w = np.sum(weights)
    if total_w < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights /= total_w
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = weights[j]
        if wj < 1e-14:
            continue
        s = j - (n - 1)
        if s >= 0:
            for m_idx in range(s, n):
                grad[m_idx] += wj * (1.0 - x[m_idx - s])
            for m_idx in range(0, n - s):
                grad[m_idx] += wj * (-x[m_idx + s])
        else:
            sp = -s
            for m_idx in range(0, n - sp):
                grad[m_idx] += wj * (1.0 - x[m_idx + sp])
            for m_idx in range(sp, n):
                grad[m_idx] += wj * (-x[m_idx - sp])
    grad *= dx
    grad += 2.0 * 1000.0 * (np.sum(x) - n / 2.0)
    return grad

def focused_gradient_descent(h, n, k, max_iters, rng):
    dx_val = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    stagnation = 0

    for it in range(max_iters):
        if remaining_time() < 5:
            break
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        c5_cur = np.max(corr) * dx_val
        grad = topk_gradient(cur_h, k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad_dir = -grad / gnorm
        improved = False
        for alpha in [0.04, 0.08, 0.02, 0.15, 0.01, 0.1, 0.2, 0.005]:
            h_trial = project_feasible(cur_h + alpha * grad_dir, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_cur - 1e-12:
                cur_h = h_trial
                c5_cur = c5_trial
                improved = True
                break
            elif c5_trial < c5_cur:
                cur_h = h_trial
                c5_cur = c5_trial
                improved = True

        if not improved:
            for alpha in [0.003, 0.001, 0.0005]:
                h_trial = project_feasible(cur_h + alpha * grad_dir, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < c5_cur:
                    cur_h = h_trial
                    c5_cur = c5_trial
                    improved = True
                    break

        if improved and c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = cur_h.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 30:
                break
    return best_h, best_c5


def binary_sa(h_init, n, max_iterations, rng, t_start=0.003, t_end=0.00002):
    h = np.where(h_init > 0.5, 1.0, 0.0)
    target_ones = n // 2
    num_ones = int(np.sum(h == 1.0))
    if num_ones > target_ones:
        ones_idx = np.where(h == 1.0)[0]
        remove = rng.choice(len(ones_idx), num_ones - target_ones, replace=False)
        h[ones_idx[remove]] = 0.0
    elif num_ones < target_ones:
        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
        h[zeros_idx[add]] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    ones_list = list(set(np.where(h == 1.0)[0].tolist()))
    zeros_list = list(set(np.where(h == 0.0)[0].tolist()))

    for it in range(max_iterations):
        if remaining_time() < 3:
            break
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        i = rng.randint(len(ones_list))
        j = rng.randint(len(zeros_list))
        oi = ones_list[i]
        oz = zeros_list[j]
        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = compute_c5(h)
        T = t_start * (t_end / t_start) ** (float(it) / max(float(max_iterations), 1.0))
        if new_c5 < current_c5 or (T > 1e-15 and rng.rand() < np.exp(-(new_c5 - current_c5) / T)):
            current_c5 = new_c5
            ones_list[i] = oz
            zeros_list[j] = oi
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0
    return best_h, best_c5


def coordinate_descent(h, n, sweeps, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()

    for sw in range(sweeps):
        if remaining_time() < 4:
            break
        improved_sweep = False
        order = rng.permutation(n)
        for i in order:
            if remaining_time() < 2:
                break
            old_val = cur_h[i]
            cur_c5 = compute_c5(cur_h)
            best_local_c5 = cur_c5
            best_local_h = cur_h.copy()
            candidates = [0.0, 1.0, 0.5, max(0.0, old_val - 0.15), min(1.0, old_val + 0.15)]
            if old_val > 0.4:
                candidates.append(0.1)
            if old_val < 0.6:
                candidates.append(0.9)

            for nv in candidates:
                if abs(nv - old_val) < 1e-9:
                    continue
                cur_h[i] = nv
                h_feas = project_feasible(cur_h, n)
                c5_test = compute_c5(h_feas)
                if c5_test < best_local_c5:
                    best_local_c5 = c5_test
                    best_local_h = h_feas.copy()
            cur_h = best_local_h.copy()
            if best_local_c5 < best_c5:
                best_c5 = best_local_c5
                best_h = cur_h.copy()
                improved_sweep = True
        if not improved_sweep:
            break
    return best_h, best_c5
