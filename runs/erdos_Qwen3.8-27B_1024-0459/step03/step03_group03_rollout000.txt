from scipy.optimize import minimize
import time

_T0 = 0.0
_TB = 1300.0
_NG = 160
_BV = 100.0


def _tl(reserve=0.0):
    return max(0.0, _TB - (time.time() - _T0) - reserve)


def _time_left(start_t, limit, reserve=0.0):
    a = _TB - (time.time() - _T0) - reserve
    b = float(limit) - (time.time() - start_t)
    return max(0.0, min(a, b))


def _target_count(target_c, n):
    v = float(target_c) * (float(n) / 2.0)
    return int(np.floor(v - 1e-9))


def _project(h, n):
    n = int(n)
    y = np.asarray(h, dtype=np.float64).ravel()
    if y.size != n:
        y = y[:n] if y.size > n else np.concatenate([y, np.zeros(n - y.size)])
    np.clip(y, 0.0, 1.0, out=y)
    S = 0.5 * n

    lo, hi = -1.0, 1.0
    for _ in range(90):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y + mid, 0.0, 1.0)))
        if s < S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(40):
        d = S - float(np.sum(x))
        if abs(d) < 1e-15:
            break
        free = (x > 1e-15) & (x < 1.0 - 1e-15)
        k = int(np.sum(free))
        if k > 0:
            x[free] += d / float(k)
        else:
            if d > 0:
                idx = int(np.argmin(x))
                x[idx] = min(1.0, x[idx] + d)
            else:
                idx = int(np.argmax(x))
                x[idx] = max(0.0, x[idx] + d)
        np.clip(x, 0.0, 1.0, out=x)

    # final tiny correction on free coordinates
    d = S - float(np.sum(x))
    if abs(d) > 1e-16:
        free = (x > 1e-15) & (x < 1.0 - 1e-15)
        k = int(np.sum(free))
        if k > 0:
            x[free] += d / float(k)
            np.clip(x, 0.0, 1.0, out=x)
        else:
            x[*np.argmin(x)] = min(1.0, x[np.argmin(x)] + d) if d > 0 else max(0.0, x[np.argmax(x)] + d)

    return x


def _corr(h):
    h = np.asarray(h, dtype=np.float64).ravel()
    return np.correlate(h, 1.0 - h, mode="full")


def _c5(h):
    h = np.asarray(h, dtype=np.float64).ravel()
    n = h.size
    if n == 0:
        return 1e100
    c = _corr(h)
    return float(np.max(c)) * (2.0 / float(n))


def _ggrad(h, t):
    h = np.asarray(h, dtype=np.float64).ravel()
    n = h.size
    s = int(t) - (n - 1)
    g = np.zeros(n, dtype=np.float64)
    if s >= 0:
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
    else:
        p = -s
        if p > 0:
            g[:n - p] += 1.0 - h[p:n]
            g[p:] -= h[:n - p]
    return g


def _tangent_dir(g):
    g = np.asarray(g, dtype=np.float64).ravel()
    g = g - np.mean(g)
    gn = float(np.linalg.norm(g))
    if gn < 1e-14:
        return None
    return -g / gn


def _scon(x):
    return float(np.sum(x)) - 0.5 * float(_NG)


def _scon_jac(x):
    return np.ones_like(x, dtype=np.float64)


def _sm_obj(x):
    x = np.asarray(x, dtype=np.float64)
    n = int(_NG)
    b = float(_BV)
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(c))
    v = np.exp(b * (c - m))
    s = float(np.sum(v))
    if s <= 0.0:
        return m * (2.0 / n)
    return (m + np.log(s) / b) * (2.0 / n)


def _sm_jac(x):
    x = np.asarray(x, dtype=np.float64)
    n = int(_NG)
    b = float(_BV)
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(c))
    v = np.exp(b * (c - m))
    s = float(np.sum(v))
    if s <= 0.0:
        return np.zeros(n, dtype=np.float64)
    w = v / s
    nz = np.flatnonzero(w > 1e-12)
    g = np.zeros(n, dtype=np.float64)
    for t in nz:
        g += float(w[int(t)]) * _ggrad(x, int(t))
    return (2.0 / n) * g


def _slsqp(h0, n, beta, maxiter):
    global _NG, _BV
    _NG = int(n)
    _BV = float(beta)
    x0 = np.asarray(h0, dtype=np.float64).ravel()
    x0 = _project(x0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _scon, "jac": _scon_jac}]
    try:
        r = minimize(
            _sm_obj,
            x0,
            method="SLSQP",
            jac=_sm_jac,
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-15, "maxls": 30, "iprint": 0},
        )
        x = np.asarray(r.x, dtype=np.float64)
    except Exception:
        x = x0
    return _project(x, n)


def _active_loop(h, n, rng, start_t, limit, reserve, max_iter=4000):
    n = int(n)
    h = np.asarray(h, dtype=np.float64).ravel().copy()
    h = _project(h, n)
    best_c = _c5(h)
    klist = [1, 2, 3, 5, 8, 13, 21, 34]
    no_imp = 0

    if n <= 200:
        steps = [0.05, 0.02, 0.10, 0.008, 0.03, 0.06, 0.015, 0.004]
    else:
        steps = [0.03, 0.012, 0.06, 0.005, 0.02, 0.04, 0.010, 0.003]

    for it in range(int(max_iter)):
        if _time_left(start_t, limit, reserve) <= 0.0:
            break
        c = _corr(h)
        L = c.size
        K = klist[it % len(klist)]
        if K >= L:
            idx = np.arange(L)
        else:
            idx = np.argpartition(c, -K)[-K:]

        g = np.zeros(n, dtype=np.float64)
        for t in idx:
            g += _ggrad(h, int(t))
        g /= float(idx.size)

        d = _tangent_dir(g)
        improved = False
        if d is not None:
            for a in steps:
                h2 = _project(h + a * d, n)
                c2 = _c5(h2)
                if c2 < best_c - 1e-13:
                    h = h2
                    best_c = c2
                    improved = True
                    break

        if not improved and rng.rand() < 0.25:
            if n > 1:
                h2 = _project(h + 0.008 * rng.standard_normal(n), n)
                c2 = _c5(h2)
                if c2 < best_c - 1e-13:
                    h = h2
                    best_c = c2
                    improved = True

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 90:
                break

    return h, best_c


def _pair_loop(h, n, rng, start_t, limit, reserve, max_iter=2500):
    n = int(n)
    h = np.asarray(h, dtype=np.float64).ravel().copy()
    h = _project(h, n)
    best_c = _c5(h)
    fracs = [0.5, 0.25, 0.75, 0.1, 0.9, 0.35, 0.65, 0.15, 0.85, 0.05, 0.95]
    no_imp = 0

    for it in range(int(max_iter)):
        if _time_left(start_t, limit, reserve) <= 0.0:
            break
        c = _corr(h)
        tmax = int(np.argmax(c))
        g = _ggrad(h, tmax)

        pairs = []
        if float(np.max(g) - np.min(g)) > 1e-12:
            pairs.append((int(np.argmax(g)), int(np.argmin(g))))

        npos = max(2, n // 10)
        hi = np.argpartition(g, -npos)[-npos:]
        lo = np.argpartition(g, npos)[:npos]
        for _ in range(3):
            pairs.append((int(hi[rng.randint(npos)]), int(lo[rng.randint(npos)])))
        for _ in range(2):
            pairs.append((int(rng.randint(n)), int(rng.randint(n))))

        improved = False
        for i, j in pairs:
            if i == j:
                continue
            for a, b in ((i, j), (j, i)):
                tmaxv = min(float(h[a]), 1.0 - float(h[b]))
                if tmaxv <= 1e-12:
                    continue
                for fr in fracs:
                    if _time_left(start_t, limit, reserve) <= 0.0:
                        break
                    t = fr * tmaxv
                    h2 = h.copy()
                    h2[a] -= t
                    h2[b] += t
                    if h2[a] < -1e-12 or h2[b] > 1.0 + 1e-12:
                        continue
                    np.clip(h2, 0.0, 1.0, out=h2)
                    c2 = _c5(h2)
                    if c2 < best_c - 1e-13:
                        h = h2
                        best_c = c2
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 70:
                break

    return h, best_c


def _bin_count(bits):
    bits = np.asarray(bits, dtype=np.int32).ravel()
    c = np.correlate(bits, 1 - bits, mode="full")
    return int(np.max(c))


def _to_bits(x, n):
    n = int(n)
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size != n:
        x = x[:n] if x.size > n else np.concatenate([x, np.zeros(n - x.size)])
    k = n // 2
    bits = np.zeros(n, dtype=np.int32)
    if k <= 0:
        return bits
    if k >= n:
        bits[:] = 1
        return bits
    idx = np.argsort(x)[::-1][:k]
    bits[idx] = 1
    s = int(bits.sum())
    if s > k:
        pos = np.flatnonzero(bits)
        if pos.size >= s - k:
            bits[pos[:s - k]] = 0
    elif s < k:
        pos = np.flatnonzero(1 - bits)
        if pos.size >= k - s:
            bits[pos[:k - s]] = 1
    return bits


def _balance_bits(bits, rng):
    bits = np.asarray(bits, dtype=np.int32).ravel().copy()
    n = bits.size
    k = n // 2
    s = int(bits.sum())
    if s > k:
        pos = np.flatnonzero(bits)
        if pos.size > s - k:
            bits[pos[rng.permutation(pos.size)[:s - k]]] = 0
    elif s < k:
        pos = np.flatnonzero(1 - bits)
        if pos.size > k - s:
            bits[pos[rng.permutation(pos.size)[:k - s]]] = 1
    return bits


def _random_bits(n, rng):
    n = int(n)
    bits = np.zeros(n, dtype=np.int32)
    if n > 1:
        bits[rng.permutation(n)[:n // 2]] = 1
    return bits


def _pick2(rng, m):
    m = int(m)
    if m < 2:
        return 0, 1
    i = int(rng.randint(m))
    j = int(rng.randint(m - 1))
    if j >= i:
        j += 1
    return i, j


def _binary_starts(h0, n, rng, max_starts=8):
    n = int(n)
    starts = []

    if h0 is not None and np.asarray(h0).size == n:
        starts.append(_to_bits(h0, n))

    pos = np.arange(n, dtype=np.float64)
    for f in (2, 3, 5, 8, 12, 16):
        if len(starts) >= max_starts:
            break
        vals = np.sin(2.0 * np.pi * f * pos / n)
        starts.append(_to_bits(vals, n))

    for _ in range(2):
        if len(starts) >= max_starts:
            break
        vals = np.zeros(n, dtype=np.float64)
        freqs = rng.choice([2, 3, 4, 5, 6, 8, 12, 16], size=min(8, np.arange(2, 17).size), replace=False)
        for f in freqs:
            vals += rng.uniform(0.5, 1.0) * np.sin(2.0 * np.pi * f * pos / n + rng.uniform(0.0, 2.0 * np.pi))
        starts.append(_to_bits(vals, n))

    for bl in (4, 6, 8, 12, 16, 24, 32):
        if len(starts) >= max_starts:
            break
        if bl >= n:
            continue
        bits = ((np.arange(n) % bl) < (bl // 2)).astype(np.int32)
        starts.append(_balance_bits(bits, rng))

    for _ in range(3):
        if len(starts) >= max_starts:
            break
        starts.append(_random_bits(n, rng))

    uniq = []
    seen = set()
    for b in starts:
        b = np.asarray(b, dtype=np.int32).ravel()
        if b.size != n:
            continue
        b = np.clip(b, 0, 1).astype(np.int32)
        key = bytes(b.astype(np.uint8))
        if key in seen:
            continue
        seen.add(key)
        uniq.append(b)
    return uniq[:max_starts]


def _binary_full_improve(bits, n, rng, start_t, limit, reserve, target_count=None):
    n = int(n)
    bits = np.asarray(bits, dtype=np.int32).ravel().copy()
    bits = _balance_bits(bits, rng)
    current = _bin_count(bits)
    best_count = current
    best_bits = bits.copy()

    def reached():
        return target_count is not None and int(best_count) <= int(target_count)

    passes = 0
    while _time_left(start_t, limit, reserve) > 0.5 and passes < 60 and not reached():
        ones = np.flatnonzero(bits)
        zeros = np.flatnonzero(1 - bits)
        if ones.size == 0 or zeros.size == 0:
            break

        pass_best = current
        bi, bj = -1, -1

        for ii in range(ones.size):
            if _time_left(start_t, limit, reserve) <= 0.5:
                break
            i = int(ones[ii])
            bits[i] = 0
            for jj in range(zeros.size):
                if _time_left(start_t, limit, reserve) <= 0.5:
                    break
                j = int(zeros[jj])
                bits[j] = 1
                cnt = _bin_count(bits)
                bits[j] = 0
                if cnt < pass_best:
                    pass_best = cnt
                    bi, bj = i, j
            bits[i] = 1

        if bi >= 0:
            bits[bi] = 0
            bits[bj] = 1
            current = pass_best
            passes += 1
            if current < best_count:
                best_count = current
                best_bits = bits.copy()
        else:
            break

    if _time_left(start_t, limit, reserve) > 20.0 and not reached():
        no_imp = 0
        while _time_left(start_t, limit, reserve) > 10.0 and no_imp < 70 and not reached():
            ones = np.flatnonzero(bits)
            zeros = np.flatnonzero(1 - bits)
            if ones.size < 2 or zeros.size < 2:
                break

            n_cand = 180 if n <= 200 else 70
            best_delta = 10 ** 9
            best_pos = None
            best_cnt = current

            for _ in range(n_cand):
                if _time_left(start_t, limit, reserve) <= 1.0:
                    break
                i1, i2 = _pick2(rng, ones.size)
                j1, j2 = _pick2(rng, zeros.size)
                a1, a2 = int(ones[i1]), int(ones[i2])
                b1, b2 = int(zeros[j1]), int(zeros[j2])

                bits[a1] = 0
                bits[a2] = 0
                bits[b1] = 1
                bits[b2] = 1
                cnt = _bin_count(bits)
                bits[a1] = 1
                bits[a2] = 1
                bits[b1] = 0
                bits[b2] = 0

                dlt = cnt - current
                if dlt < best_delta:
                    best_delta = dlt
                    best_pos = (a1, a2, b1, b2)
                    best_cnt = cnt
                    if dlt < 0:
                        break

            accept = False
            if best_pos is not None:
                if best_delta < 0:
                    accept = True
                elif best_delta == 0 and rng.rand() < 0.03:
                    accept = True
                elif best_delta == 1 and rng.rand() < 0.002:
                    accept = True

            if accept:
                a1, a2, b1, b2 = best_pos
                bits[a1] = 0
                bits[a2] = 0
                bits[b1] = 1
                bits[b2] = 1
                current = best_cnt
                if current < best_count:
                    best_count = current
                    best_bits = bits.copy()
                no_imp = 0 if best_delta < 0 else max(0, no_imp - 1)
            else:
                no_imp += 1

            if no_imp > 25 and _time_left(start_t, limit, reserve) > 5.0:
                ones = np.flatnonzero(bits)
                zeros = np.flatnonzero(1 - bits)
                if ones.size > 0 and zeros.size > 0:
                    i = int(ones[rng.randint(ones.size)])
                    j = int(zeros[rng.randint(zeros.size)])
                    bits[i] = 0
                    bits[j] = 1
                    cnt = _bin_count(bits)
                    bits[i] = 1
                    bits[j] = 0
                    if cnt <= current + 2 and rng.rand() < 0.35:
                        bits[i] = 0
                        bits[j] = 1
                        current = cnt
                        if current < best_count:
                            best_count = current
                            best_bits = bits.copy()
                no_imp = 0

    return best_bits, best_count


def _binary_optimize(h0, n, rng, start_t, limit, reserve, target_count=None):
    n = int(n)
    best_bits = None
    best_count = 10 ** 9

    def reached():
        return target_count is not None and int(best_count) <= int(target_count)

    starts = _binary_starts(h0, n, rng, max_starts=6)
    if not starts:
        starts = [_random_bits(n, rng)]

    # Allocate more time to the parent-derived start.
    for idx, b in enumerate(starts):
        if _time_left(start_t, limit, reserve) < 12.0 or reached():
            break
        left = _time_left(start_t, limit, reserve)
        remaining = max(1, len(starts) - idx)
        if idx == 0:
            per = min(75.0, max(12.0, left * 0.45))
        else:
            per = min(35.0, max(8.0, left / (remaining + 1.0)))
        if per <= 1.0:
            continue

        st_local = time.time()
        b2, c2 = _binary_full_improve(b, n, rng, st_local, per, reserve, target_count)
        if c2 < best_count:
            best_count = int(c2)
            best_bits = np.asarray(b2, dtype=np.int32).copy()

    # Iterated kicks from the best binary pattern.
    if best_bits is not None and _time_left(start_t, limit, reserve) > 25.0 and not reached():
        for _ in range(3):
            if _time_left(start_t, limit, reserve) < 12.0 or reached():
                break
            kicked = best_bits.copy()
            for _ in range(5):
                ones = np.flatnonzero(kicked)
                zeros = np.flatnonzero(1 - kicked)
                if ones.size == 0 or zeros.size == 0:
                    break
                i = int(ones[rng.randint(ones.size)])
                j = int(zeros[rng.randint(zeros.size)])
                kicked[i] = 0
                kicked[j] = 1
            per = min(30.0, _time_left(start_t, limit, reserve) * 0.35)
            if per <= 2.0:
                break
            b2, c2 = _binary_full_improve(kicked, n, rng, time.time(), per, reserve, target_count)
            if c2 < best_count:
                best_count = int(c2)
                best_bits = np.asarray(b2, dtype=np.int32).copy()

    return best_bits, best_count


def _interp(h, n1, n2):
    n1 = int(n1)
    n2 = int(n2)
    h = np.asarray(h, dtype=np.float64).ravel()
    if h.size != n1:
        h = h[:n1] if h.size > n1 else np.concatenate([h, np.zeros(n1 - h.size)])
    if n1 == n2:
        return h.copy()
    xs = (np.arange(n1) + 0.5) * (2.0 / n1)
    xd = (np.arange(n2) + 0.5) * (2.0 / n2)
    return np.interp(xd, xs, h)


def _optimize_fractional(h0, n, rng, start_t, limit, reserve, do_binary=True):
    n = int(n)
    h = _project(h0, n)
    best_c = _c5(h)

    left = _time_left(start_t, limit, reserve)
    if left > 20.0:
        t1 = min(170.0, 0.35 * left)
        h2, c2 = _active_loop(h, n, rng, start_t, t1, reserve)
        if c2 < best_c:
            h, best_c = h2, c2

    left = _time_left(start_t, limit, reserve)
    if left > 90.0:
        if n <= 192:
            h2 = _slsqp(h, n, 220, 90)
        elif n <= 256:
            h2 = _slsqp(h, n, 120, 55)
        else:
            h2 = _slsqp(h, n, 80, 35)
        c2 = _c5(h2)
        if c2 < best_c:
            h, best_c = h2, c2

    left = _time_left(start_t, limit, reserve)
    if left > 60.0:
        t3 = min(110.0, 0.25 * left)
        h2, c2 = _pair_loop(h, n, rng, start_t, t3, reserve)
        if c2 < best_c:
            h, best_c = h2, c2

    if do_binary and n <= 320:
        left = _time_left(start_t, limit, reserve)
        if left > 130.0:
            per = min(150.0, 0.30 * left)
            tgt = _target_count(0.380849999, n)
            b, cnt = _binary_optimize(h, n, rng, start_t, per, reserve, tgt)
            if b is not None:
                cb = float(cnt) * (2.0 / n)
                if cb < best_c:
                    h = np.asarray(b, dtype=np.float64).copy()
                    best_c = cb
                left = _time_left(start_t, limit, reserve)
                if cb < best_c + 0.005 and left > 60.0:
                    h2 = np.asarray(b, dtype=np.float64).copy()
                    h2, c2 = _active_loop(h2, n, rng, start_t, min(80.0, left), reserve)
                    if c2 < best_c:
                        h, best_c = h2, c2

    left = _time_left(start_t, limit, reserve)
    if left > 100.0:
        if n <= 192:
            h2 = _slsqp(h, n, 750, 60)
        elif n <= 256:
            h2 = _slsqp(h, n, 350, 35)
        else:
            h2 = _slsqp(h, n, 220, 20)
        c2 = _c5(h2)
        if c2 < best_c:
            h, best_c = h2, c2

    left = _time_left(start_t, limit, reserve)
    if left > 45.0:
        h2, c2 = _active_loop(h, n, rng, start_t, left, reserve)
        if c2 < best_c:
            h, best_c = h2, c2

    h = _project(h, n)
    best_c = _c5(h)
    return h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _TB
    _T0 = time.time()
    _TB = float(max(1.0, budget_s))
    rng = np.random.RandomState(int(seed))

    try:
        hpar = np.array(initial_h_values, dtype=np.float64).ravel()
    except Exception:
        hpar = None

    if hpar is None or hpar.size < 16:
        npar = 160
        hpar = np.zeros(npar, dtype=np.float64)
        hpar[rng.permutation(npar)[:npar // 2]] = 1.0
    else:
        npar = int(hpar.size)
        if npar % 2 != 0:
            if npar > 2:
                npar -= 1
                hpar = _interp(hpar, npar + 1, npar)
            else:
                npar = 160
                hpar = np.zeros(npar, dtype=np.float64)
                hpar[rng.permutation(npar)[:npar // 2]] = 1.0
        if npar < 48:
            hpar = _interp(hpar, npar, 160)
            npar = 160
        elif npar > 448:
            hpar = _interp(hpar, npar, 256)
            npar = 256

    best_n = int(npar)
    best_h = _project(hpar, best_n)
    best_c = _c5(best_h)

    # Main fine-tuning of the inherited construction.
    if _tl(130.0) > 20:
        lim = min(420.0, _tl(130.0))
        h2, c2 = _optimize_fractional(best_h, best_n, rng, time.time(), lim, 130.0, False)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Binary-window search at a few nearby even resolutions.
    if _tl(190.0) > 180:
        alts = [
            best_n,
            best_n + 16,
            best_n - 16,
            best_n + 32,
            best_n - 32,
            best_n + 48,
            best_n + 64,
            best_n - 48,
        ]
        chosen = []
        seen = set()
        for nn in alts:
            nn = int(nn)
            if nn % 2 != 0:
                continue
            if nn < 96 or nn > 384:
                continue
            if nn in seen:
                continue
            seen.add(nn)
            chosen.append(nn)
            if len(chosen) >= 5:
                break

        bin_budget = min(520.0, _tl(190.0))
        for idx, nn in enumerate(chosen):
            if _tl(190.0) < 30.0 or bin_budget < 25:
                break
            if idx == 0:
                per = min(200.0, bin_budget * 0.45)
            else:
                per = min(80.0, bin_budget * 0.18)
            per = max(12.0, per)
            if per < 12.0:
                continue

            if nn == best_n:
                seed_h = best_h
            else:
                seed_h = _interp(best_h, best_n, nn)

            tgt = _target_count(min(0.380849999, best_c), nn)
            b, cnt = _binary_optimize(seed_h, nn, rng, time.time(), per, 190.0, tgt)
            if b is not None:
                cb = float(cnt) * (2.0 / float(nn))
                if cb < best_c - 1e-12:
                    best_h = np.asarray(b, dtype=np.float64).copy()
                    best_n = nn
                    best_c = cb

    # Final continuous polish of the best known discrete pattern.
    if _tl(70.0) > 40:
        lim = min(700.0, _tl(70.0))
        h2, c2 = _optimize_fractional(best_h, best_n, rng, time.time(), lim, 70.0, True)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Short high-frequency tail polish.
    if _tl(25.0) > 25:
        h2, c2 = _active_loop(best_h, best_n, rng, time.time(), min(70.0, _tl(25.0)), 25.0)
        if c2 < best_c:
            best_h, best_c = h2, c2

    if _tl(25.0) > 25:
        h2, c2 = _pair_loop(best_h, best_n, rng, time.time(), min(50.0, _tl(25.0)), 25.0)
        if c2 < best_c:
            best_h, best_c = h2, c2

    best_h = _project(best_h, best_n)
    final_c = _c5(best_h)

    try:
        ec = evaluate_erdos_solution(best_h)
        ec = float(ec)
        if np.isfinite(ec) and ec < final_c:
            final_c = ec
    except Exception:
        pass

    return (best_h, float(final_c), int(best_n))
