def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mid = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mid))
        if cnt > 0:
            h[mid] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, rng):
    n = len(h)

    threshold = 0.5
    b = (h >= threshold).astype(float)
    n_ones = int(np.sum(b))
    if n_ones != half:
        if n_ones > half:
            ones = np.where(b == 1)[0]
            rem = rng.choice(ones, n_ones - half, replace=False)
            b[rem] = 0
        else:
            zeros = np.where(b == 0)[0]
            add = rng.choice(zeros, half - n_ones, replace=False)
            b[add] = 1
