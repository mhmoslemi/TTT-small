def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def project_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xq = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xq, xp, src)

def make_m_sequence(n, rng):
    # Generate an m-sequence-like pattern
    h = np.zeros(n)
    # Use a simple LFSR
    reg = rng.randint(1, 2**16, dtype=int)
    mask = 0x10000
    for i in range(n):
        h[i] = (reg >> 15) & 1
        new_bit = (reg ^ (reg >> 1) ^ (reg >> 12) ^ (reg >> 15)) & 1
        reg = ((reg << 1) | new_bit) & (mask - 1)
    return h

def smooth_max(vals, beta):
    m = np.max(vals)
    if beta < 1e-6:
        return m.copy() if isinstance(m, np.ndarray) else m
    exps = np.exp(beta * (vals - m))
    return m + np.log(np.sum(exps)) / beta

def corr_grad(h, m, n):
    """Gradient of correlation at lag m with respect to h."""
    grad = np.zeros(n)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m > 0:
        # corr = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
        lo = 0
        hi = n - m  # exclusive
        # For j in [0, hi): contributes 1 - h[j+m]
        grad[lo:hi] += (1.0 - h[m:m + (hi - lo)])
        # For j in [m, n): contributes -h[j-m]
        grad[m:] -= h[:n - m]
    else:

        lo = 0
        hi = n - m
        grad[lo:hi] += (1.0 - h[m:m + (hi - lo)])
        grad[m:] -= h[:n - m]

    m = np.max(vals)
    if beta < 1e-6:

        weights = np.ones(len(top_indices)) / len(top_indices)
    else:

    grad = np.zeros(n)
    for idx, w in zip(top_indices, weights):
        lag = idx - (n - 1)
        grad += w * corr_grad(h, lag, n)

**Structured local search idea**:
For the correlation at the worst lag m:
corr_m = sum_{i=0}^{n-1-m} h[i] * (1-h[i+m])

To reduce this, we want to:
- Decrease h[i] for positions where h[i] is large and h[i+m] is small (reduces the product)
- Or increase h[i+m] for positions where h[i] is large and h[i+m] is small
- But we need to maintain sum(h) = n/2, so we need to compensate elsewhere

A good move: find i where h[i] is large and h[i+m] is small, then:
- Decrease h[i] by δ
- Find j (not related to i or i+m) where h[j+m'] is small for other critical lags, and increase h[j] by δ

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr


def project_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_h(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n

    m = np.max(vals)

    e = np.exp(beta * (vals - m))

    m = np.max(vals)

    e = np.exp(beta * (vals - m))
    return e / np.sum(e)
