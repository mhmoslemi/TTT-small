```python
            for L in range(1, n - j):
                a_list.append(-h[j + L])
                b_list.append(corr[L + n - 1] - old * (-h[j + L]))
                aff_idx.append(L + n - 1)
            
            # L < 0, s=-L, h[j] in first factor h[i]: i=j, need 1<=s<=j
            for s in range(1, j + 1):
                a_list.append(1.0 - h[j - s])
                b_list.append(corr[n - 1 - s] - old * (1.0 - h[j - s]))
                aff_idx.append(n - 1 - s)
            
            # L < 0, s=-L, h[j] in second factor (1-h[i-s]): i=j+s, need 1<=s, j+s<=n-1 → s<=n-1-j
            for s in range(1, n - j):
                a_list.append(-h[j + s])
                b_list.append(corr[n - 1 - s] - old * (-h[j + s]))
                aff_idx.append(n - 1 - s)
            
            # M_unaff
            if len(aff_idx) < 2 * n - 1:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[aff_idx] = False
                M_unaff = np.max(corr[mask])
            else:
                M_unaff = 0.0

            a_arr = np.array(a_list) if a_list else np.zeros(1)
            b_arr = np.array(b_list) if b_list else np.zeros(1)
            
            # Grid
            xs = np.linspace(0.0, 1.0, 51)
            if len(a_list) > 0:
                lin_vals = np.max(a_arr[:, None] * xs[None, :] + b_arr[:, None], axis=0)
            else:
                lin_vals = np.full_like(xs, -np.inf)
            quad_vals = b0 + xs - xs * xs
            obj = np.maximum.reduce([np.full_like(xs, M_unaff), quad_vals, lin_vals])
            bi = np.argmin(obj)
            lo = xs[max(0, bi - 1)]
            hi = xs[min(50, bi + 1)]
            
            # Golden section
            for _ in range(80):
                m1 = lo + 0.381966 * (hi - lo)
                m2 = hi - 0.381966 * (hi - lo)
                if abs(hi - lo) < 1e-14:
                    break
                v1 = M_unaff
                q1 = b0 + m1 - m1 * m1
                if q1 > v1: v1 = q1
                if len(a_list) > 0:
                    l1 = np.max(a_arr * m1 + b_arr)
                    if l1 > v1: v1 = l1
                v2 = M_unaff
                q2 = b0 + m2 - m2 * m2
                if q2 > v2: v2 = q2
                if len(a_list) > 0:
                    l2 = np.max(a_arr * m2 + b_arr)
                    if l2 > v2: v2 = l2
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            
            new_x = (lo + hi) / 2.0
            h[j] = new_x
            corr = np.correlate(h, 1.0 - h, mode='full')
        
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
```

```python
def subgradient_step(h, n, top_k=20, lr=0.01):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-top_k:]
    w = 1.0 / top_k
    grad = np.zeros(n)
    for m in idx:
        L = m - (n - 1)
        s = abs(L)
        if s == 0:
            # Special: L=0, gradient of h[j]*(1-h[j]) = 1-2h[j]
            grad += w * (1 - 2*h)
        else:
            grad[s:] += w * (1 - h[:n-s])
            grad[:n-s] += w * (-h[s:])
    return h - lr * grad, np.max(corr), idx
```

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-13
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_h(h, top_k=20):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-top_k:]
    w = 1.0 / top_k
    grad = np.zeros(n)
    for m in idx:
        s = abs(m - (n - 1))
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            grad[s:] += w * (1.0 - h[:n-s])
            grad[:n-s] += w * (-h[s:])
    return grad, corr, idx

def subgradient_optimize(h, n, time_limit, t0, iters=20000, k_schedule=None):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    # Learning rate schedule: start large, decay
    # Use Polyak-like or simple geometric
    for it in range(iters):
        if time.time() - t0 > time_limit:
            break
        
        # Adaptive k
        if k_schedule:
            frac = it / max(1, iters)
            k = max(3, int(k_schedule[0] + (k_schedule[1] - k_schedule[0]) * frac))
        else:
            k = max(3, int(5 + 20 * (1.0 - it / iters)))
        
        lr = 0.05 / (1.0 + 0.01 * it)
        
        grad, corr, idx = subgradient_h(h, k)
        h_new = project_feasible(h - lr * grad, n)
        c5_new = np.max(corr) * dx  # use old corr for this estimate
        # Actually recompute
        c5_check = compute_c5(h_new)
        
        if c5_check < best_c5:
            best_c5 = c5_check
            best_h = h_new.copy()
            h = h_new
        else:
            # Try smaller step
            h_new2 = project_feasible(h - 0.5 * lr * grad, n)
            c5_2 = compute_c5(h_new2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h_new2.copy()
                h = h_new2
            else:

                if it % 500 == 499:
                    h = project_feasible(h + 0.02 * np.random.RandomState(it).randn(n), n)
    
    return best_h, best_c5
