```python
        corr = compute_corr(h)
        cmax = np.max(corr)
        # Active set: shifts within 1.0 of max
        active = np.where(corr > cmax - 1.0)[0]
        
        # Build LP: minimize t s.t. c_k + g_k·(h-h0) ≤ t
        # variables: h (n vars) + t (1 var)
        # constraints: 0 ≤ h ≤ 1, sum(h) = n/2
        x = cp.Variable(n)
        t = cp.Variable()
        
        constraints = [x >= 0, x <= 1, cp.sum(x) == n/2.0]
        
        for k in active:
            off = k - (n - 1)
            c_k = corr[k]
            # Gradient of corr[k] w.r.t. h
            g_k = np.zeros(n)
            if off >= 0:
                npairs = n - off
                g_k[off:off+npairs] = (1.0 - h[:npairs])
                g_k[:npairs] -= h[off:off+npairs]
            else:
                ao = -off
                npairs = n - ao
                g_k[:npairs] = (1.0 - h[ao:ao+npairs])
                g_k[ao:ao+npairs] -= h[:npairs]
            
            # Linearized: c_k + g_k · (x - h) ≤ t
            # => g_k · x ≤ t - c_k + g_k · h
            constraints.append(g_k @ x <= t - c_k + g_k @ h)
        
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS)  # or another solver
        
        if prob.status == 'optimal':
            h_new = x.value
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                # Damped step
                for alpha in [1.0, 0.5, 0.25, 0.1]:
                    h_trial = alpha * h_new + (1 - alpha) * h
                    h_trial = project_feasible(h_trial, n)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial <= best_c5:
                        h = h_trial
                        best_c5 = c5_trial
                        break
                corr_new = compute_corr(h)
                cmax_new = np.max(corr_new)
                if cmax_new >= cmax - 1e-10:
                    break
        else:
            break
    
    return h, best_c5
```

This is a fundamentally different approach from the previous code! It uses LP subproblems within an SQP framework, which should be more accurate than the softmax approximation.

Let me also add:
- Multiple active set sizes (threshold 0.5, 1.0, 2.0)
- Different initial points
- Binary local search as a final step

Now, performance considerations:
- Each SQP iteration solves an LP with ~n + 1 variables and ~2n + |active| + 1 constraints
- For n=320 and |active|≈50, that's ~321 vars and ~700 constraints
- ECOS should handle this in < 1 second
- So we can do ~100+ SQP iterations in a few minutes

Let me also think about the binary case. For binary h, the problem is purely combinatorial. The pair-swap local search is the standard approach. But maybe I can improve it with:
- Tabu search (forbid recent swaps for T iterations)
- Simulated annealing (accept worse moves with some probability)
- Multi-start with diverse initializations

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):

def slsqp_optimize(h, n, beta=1000, maxiter=300):
    """Softmax-based SLSQP optimization."""
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = beta * (corr - m)
        cl = np.clip(shifted, -50, 50)
        return (m + np.log(np.sum(np.exp(cl))) / beta) * dx
    
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = beta * (corr - m)
        cl = np.clip(shifted, -50, 50)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] > 1e-12:
                g += w[k] * get_gradient_shift(x, n, k)
        return g * dx
    
    def constraint(x):
        return np.sum(x) - target
    
    cons = [{'type': 'eq', 'fun': constraint, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h
