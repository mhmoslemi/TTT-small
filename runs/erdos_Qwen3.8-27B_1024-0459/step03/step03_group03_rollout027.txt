import numpy as np
from scipy.optimize import minimize
import time

_t0 = 0.0
_tb = 1300.0
_ng = 0
_bv = 100.0


def _tl():
    return max(0.0, _tb - (time.time() - _t0) - 3.0)


def _c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / n)


def _proj(h, n):
    h = np.array(h, dtype=np.float64)
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(50):
        d = 0.5 * n - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        k = int(np.sum(m))
        if k > 0:
            h[m] += d / k
        else:
            h[:] = 0.5
        np.clip(h, 0.0, 1.0, out=h)
    return h


def _interp_f(h, ns, nd):
    if ns == nd:
        return h.copy()
    xs = (np.arange(ns) + 0.5) * (2.0 / ns)
    xd = (np.arange(nd) + 0.5) * (2.0 / nd)
    return np.interp(xd, xs, h)


def _scon(x):
    return x.sum() - 0.5 * _ng


def _sobj(x):
    n = _ng
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = float(np.max(c))
    b = _bv
    sh = c - mx
    return (mx + np.log(np.exp(np.clip(b * sh, -700, 0)).sum()) / b) * (2.0 / n)


def _slsqp(h0, n, beta, maxiter):
    global _ng, _bv
    _ng = n
    _bv = beta
    try:
        r = minimize(_sobj, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon}],
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return _proj(np.asarray(r.x, dtype=np.float64), n)
    except Exception:
        return _proj(h0.copy(), n)


def _ggrad(h, kmax):
    n = len(h)
    s = kmax - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
    else:
        p = -s
        if p > 0:
            g[:n - p] += 1.0 - h[p:n]
            g[p:] -= h[:n - p]
    return g


def _smgrad(h, bw):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(corr))
    w = np.exp(bw * (corr - mx))
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
        sw = 1.0
    w /= sw
    g = np.zeros(n)
    nz = np.where(w > 1e-8)[0]
    for k in nz:
        g += w[int(k)] * _ggrad(h, int(k))
    return g


def _sgdescent(h, n, nit, rng, bw=50.0):
    best = h.copy()
    bc = _c5(best)
    for it in range(nit):
        if _tl() < 3:
            break
        corr = np.correlate(best, 1.0 - best, mode='full')
        mx_idx = int(np.argmax(corr))
        g_exact = _ggrad(best, mx_idx)
        g_sm = _smgrad(best, bw)
        g = 0.4 * g_exact + 0.6 * g_sm
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            g = g_exact
            gn = float(np.linalg.norm(g))
            if gn < 1e-15:
                break
        g /= gn
        moved = False
        for a in [0.08, 0.03, 0.01, 0.02, 0.005, 0.05, 0.015, 0.008, 0.003]:
            h2 = _proj(best - a * g, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-15:
                best = h2
                bc = c2
                moved = True
                break
        if not moved:
            gn2 = float(np.linalg.norm(g_exact))
            if gn2 > 1e-15:
                g2 = g_exact / gn2
                for a in [0.04, 0.015, 0.007, 0.02, 0.004]:
                    h2 = _proj(best - a * g2, n)
                    c2 = _c5(h2)

                        moved = True

        if not moved:

                h2 = _proj(best - a * g, n)

                    best = h2
                    bc = c2
                    break
    return best, bc
