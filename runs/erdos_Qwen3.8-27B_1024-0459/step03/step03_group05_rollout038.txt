```python
    for t in range(len(idx)):
        w = weights[t]
        if w < 1e-15:
            continue
        k = idx[t] - (n - 1)
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            grad[k:] += w * (1.0 - h[:-k]) * dx
            grad[:n-k] += w * (-h[k:]) * dx
        else:
            kp = -k
            grad[:n-kp] += w * (1.0 - h[kp:]) * dx
            grad[kp:] += w * (-h[:-kp]) * dx
    
    return val, grad

def active_set_projected_gradient(h, target_sum, beta, top_k=15, max_iter=500, 
                                   alpha_init=0.1, verbose=False):
    """Projected gradient with backtracking on the smooth max objective."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    
    alpha = alpha_init
    momentum = np.zeros(n)
    mu = 0.9  # momentum coefficient
    
    for it in range(max_iter):

        # Add penalty gradient for sum constraint
        sum_err = np.sum(h) - target_sum
        grad += 10.0 * sum_err * np.ones(n)

        momentum = mu * momentum - alpha * grad
        h_trial = project_feasible(h + momentum, target_sum)

        if c5_trial < best_val:
            best_val = c5_trial
            best_h = h_trial.copy()
        
        # Backtracking if no improvement
        if c5_trial > val:
            alpha *= 0.8
            continue
        
        h = h_trial.copy()
        
        # Decay alpha slowly
        if it % 50 == 49:
            alpha *= 0.95
        
        if verbose and it % 100 == 0:
            print("    iter {}: C5={:.6f}, alpha={:.4f}".format(it, best_val, alpha))
    
    return best_h, best_val
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def active_set_gradient(h, beta, top_k=15):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    
    if top_k < len(corr):
        idx = np.argpartition(-corr, top_k)[:top_k]
    else:
        idx = np.arange(len(corr))
    
    c = corr[idx]
    m = c.max()
    shifted = c - m
    exps = np.exp(beta * shifted)
    w_sum = exps.sum()
    weights = exps / w_sum
    val = m + np.log(w_sum) / beta
    
    grad = np.zeros(n)
    for t in range(len(idx)):
        w = weights[t]
        if w < 1e-15:
            continue
        k = idx[t] - (n - 1)
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            grad[k:] += w * (1.0 - h[:-k]) * dx
            grad[:n-k] += w * (-h[k:]) * dx
        else:
            kp = -k
            grad[:n-kp] += w * (1.0 - h[kp:]) * dx
            grad[kp:] += w * (-h[:-kp]) * dx
    
    return val, grad

def lbfgs_refine(h0, n, target, betas, max_iter=300, verbose=False):
    """Run L-BFGS-B with smooth max objective at increasing beta."""
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    for beta in betas:
        def obj(params):
            val, _ = active_set_gradient(params, beta, top_k=20)
            sum_err = np.sum(params) - target
            pen = 100.0 * sum_err * sum_err
            return val + pen
        
        def obj_grad(params):
            val, grad = active_set_gradient(params, beta, top_k=20)
            sum_err = np.sum(params) - target
            grad = grad + 200.0 * sum_err * np.ones(n)
            return grad
        
        try:
            res = minimize(obj, best_h, jac=obj_grad, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxls': 50})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                if verbose:
                    print("    L-BFGS beta={}: C5={:.6f} (iter={})".format(beta, c5_new, res.nit))
            else:
                # Still use the result as starting point for next beta
                best_h = h_new.copy()
        except Exception as e:
            if verbose:
                print("    L-BFGS beta={} failed: {}".format(beta, str(e)[:50]))
    
    return best_h, best_c5

def projected_gradient_descent(h0, n, target, beta, top_k, max_iter, alpha_init, 
                                momentum_coeff=0.85, verbose=False):
    """Projected gradient with momentum and backtracking."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    alpha = alpha_init
    
    for it in range(max_iter):
        val, grad = active_set_gradient(h, beta, top_k)
        # Penalty for constraint
        sum_err = np.sum(h) - target
        grad = grad + 5.0 * sum_err * np.ones(n)
        
        momentum = momentum_coeff * momentum - alpha * grad
        h_trial = project_feasible(h + momentum, target)
        c5_trial = compute_c5(h_trial)
        
        if c5_trial < best_c5 - 1e-15:
            best_c5 = c5_trial
            best_h = h_trial.copy()
        
        if c5_trial > val + 0.01:
            # Backtrack
            alpha *= 0.7
            if alpha < 1e-6:
                break
            continue
        
        h = h_trial.copy()
        
        if it % 100 == 99:
            alpha *= 0.93
    
    return best_h, best_c5

def coordinate_descent(h0, n, target, max_sweeps=30, samples_per_sweep=2000, 
                       rng=None, verbose=False):
    """Random pairwise coordinate descent."""
    if rng is None:
        rng = np.random.RandomState(0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    deltas = [0.01, 0.005, 0.02, 0.003, 0.05, 0.015]
    no_improve = 0
    
    for sweep in range(max_sweeps):
        improved = False
        for _ in range(samples_per_sweep):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            # Move mass from higher to lower (or both directions)
            for d in deltas:
                for sign in [1, -1]:
                    di = sign * d
                    dj = -sign * d
                    hi_new = h[i] + di
                    hj_new = h[j] + dj
                    if hi_new < -1e-9 or hi_new > 1.0 + 1e-9:
                        continue
                    if hj_new < -1e-9 or hj_new > 1.0 + 1e-9:
                        continue
                    h[i] += di
                    h[j] += dj
                    c5_test = compute_c5(h)
                    if c5_test < best_c5:
                        best_c5 = c5_test
                        best_h = h.copy()
                        improved = True
                        break
                    h[i] -= di
                    h[j] -= dj
                if improved:
                    break
            if improved:
                # Continue from the improved point
                continue
        
        if improved:
            no_improve = 0
            h = best_h.copy()
        else:
            no_improve += 1
            h = best_h.copy()
            if no_improve >= 5:
                break
    
    return best_h, best_c5

def interpolate_to_resolution(h_from, n_from, n_to):
    """Interpolate h from n_from to n_to resolution."""
    target = n_to / 2.0
    x_from = (np.arange(n_from) + 0.5) * 2.0 / n_from
    x_to = (np.arange(n_to) + 0.5) * 2.0 / n_to
    h_interp = np.interp(x_to, x_from, h_from)
    return project_feasible(h_interp, target)

def structured_perturbation(h, n, rng, amplitude=0.05):
    """Add a structured (Fourier) perturbation to h."""
    h_pert = h.copy()
    # Choose 1-3 random frequencies
    num_freqs = rng.randint(1, 4)

        t = np.arange(n)
        h_pert += amp * np.cos(2 * np.pi * freq * t / n + phase)
    return project_feasible(h_pert, n / 2.0)
```
