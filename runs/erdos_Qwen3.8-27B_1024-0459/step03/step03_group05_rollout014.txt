def compute_c5(h):

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def smooth_max(h, beta, top_k=20):
    cv = corr_values(h)
    if top_k < len(cv):
        idx = np.argpartition(cv, -top_k)[-top_k:]
        c = cv[idx]
    else:
        c = cv

    s = beta * (c - m)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / beta
