import time

def compute_c5_vals(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5, corr, dx

def project_h_to_box_sum(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Distribute among non-boundary elements
        mask_lo = h <= 1e-14
        mask_hi = h >= 1.0 - 1e-14
        mask_mid = ~(mask_lo | mask_hi)
        cnt = np.sum(mask_mid)
        if cnt > 0:
            h[mask_mid] += diff / cnt
        else:
            # All at bounds - adjust proportionally
            if diff > 0:
                # Need to increase sum: increase zeros
                h[mask_lo] += diff / max(1, np.sum(mask_lo))
            else:
                h[mask_hi] += diff / max(1, np.sum(mask_hi))
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_reproject(src, src_n, dst_n):
    """Interpolate h from src_n to dst_n points and project."""
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xq = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    h_q = np.interp(xq, xp, src)
    return project_h_to_box_sum(h_q, dst_n)

        elif diff > 0:

            c2 = int(np.sum(mask_lo))
            if c2 > 0:
                h[mask_lo] += diff / c2
        else:
            mask_hi = h >= 1.0 - 1e-14
            c2 = int(np.sum(mask_hi))
            if c2 > 0:
                h[mask_hi] += diff / c2
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_reproject(src, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xq = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    h_q = np.interp(xq, xp, src)
    return project_h_to_box_sum(h_q, dst_n)

def compute_d_for_pair(h, n, i, j):
    """Compute d[m] = ∂corr_full[m]/∂h[j] - ∂corr_full[m]/∂h[i] for all m."""
    k_all = np.arange(-(n - 1), n)
    d = np.zeros(2 * n - 1)
    # For h[j]: first factor k ∈ [j-n+1, j], contribution (1-h[j-k])
    lo1, hi1 = j - n + 1, j
    mask1 = (k_all >= lo1) & (k_all <= hi1)
    idx1 = k_all[mask1]
    d[mask1] += (1.0 - h[j - idx1])
    # For h[j]: second factor k ∈ [-j, n-1-j], contribution -h[j+k]
    lo2, hi2 = -j, n - 1 - j
    mask2 = (k_all >= lo2) & (k_all <= hi2)
    idx2 = k_all[mask2]
    d[mask2] -= h[j + idx2]
    # For h[i] (subtracted)
    lo1i, hi1i = i - n + 1, i
    mask1i = (k_all >= lo1i) & (k_all <= hi1i)
    idx1i = k_all[mask1i]
    d[mask1i] -= (1.0 - h[i - idx1i])
    lo2i, hi2i = -i, n - 1 - i
    mask2i = (k_all >= lo2i) & (k_all <= hi2i)
    idx2i = k_all[mask2i]
    d[mask2i] += h[i + idx2i]
    return d

def cd_pair_line_search(corr, d, h_i, h_j):
    """Find optimal delta for pair transfer. Returns (best_delta, improved)."""
    delta_lo = max(-h_j, h_i - 1.0)
    delta_hi = min(h_i, 1.0 - h_j)
    if delta_hi - delta_lo < 1e-14:
        return 0.0, False
    lo, hi = delta_lo, delta_hi
    best_delta = 0.0
    best_val = np.max(corr)
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        vals = corr + d * mid
        f_mid = np.max(vals)
        if f_mid < best_val - 1e-15:
            best_val = f_mid
            best_delta = mid
        active = np.where(vals >= f_mid - 1e-12)[0]
        d_act = d[active]
        if np.all(d_act > 1e-10):
            hi = mid
        elif np.all(d_act < -1e-10):
            lo = mid
        else:
            best_delta = mid
            if f_mid < best_val:
                best_val = f_mid
            break
    if best_val < np.max(corr) - 1e-14:
        return best_delta, True
    return 0.0, False

def pairwise_cd(h, n, dx, corr, max_sweeps=30, t0=None, budget=None, rng=None):
    """Pairwise coordinate descent. Returns h, c5."""
    best_c5 = np.max(corr) * dx
    for sweep in range(max_sweeps):
        if t0 is not None and budget is not None:
            if time.time() - t0 > budget - 5:
                break
        improved_total = 0
        pairs = []
        for i in range(n):
            for j in range(i + 1, n):
                pairs.append((i, j))
        if rng is not None:
            rng.shuffle(pairs)
        for i, j in pairs:
            hi, hj = h[i], h[j]
            d = compute_d_for_pair(h, n, i, j)
            delta, improved = cd_pair_line_search(corr, d, hi, hj)
            if improved and abs(delta) > 1e-14:
                h[i] -= delta
                h[j] += delta
                corr = corr + d * delta
                new_c5 = np.max(corr) * dx
                if new_c5 < best_c5:
                    best_c5 = new_c5
                improved_total += 1
        if improved_total == 0:
            break
    return h, best_c5, corr

def obj_smooth(h, n, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if beta < 1e-6:
        return m * dx
    exps = np.exp(beta * (corr - m))
    return (m + np.log(np.sum(exps)) / beta) * dx

def sum_constraint(h, n):
    return np.sum(h) - n / 2.0

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1') & 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return project_h_to_box_sum(h, n)

def make_sine_init(n, freq, phase):
    t = np.arange(n) / float(n)
    h = 0.5 + 0.45 * np.sin(2.0 * np.pi * freq * t + phase)
    return project_h_to_box_sum(h, n)

def make_binary_optimum(n, rng):
    """Simple greedy: place ones to minimize max correlation."""
    h = np.zeros(n)
    ones_to_place = n // 2
    for _ in range(ones_to_place):
        best_pos = 0
        best_c5 = float('inf')
        for pos in range(n):
            if h[pos] < 0.1:
                h[pos] = 1.0
                c5, _, _ = compute_c5_vals(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_pos = pos
                h[pos] = 0.0
        h[best_pos] = 1.0
    return project_h_to_box_sum(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget = budget_s
    rng = np.random.RandomState(seed)

    def time_left():
        return max(0.0, budget - (time.time() - t0))

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Initialize from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        best_h = parent_h.copy()
        best_n = parent_n
        best_c5, _, _ = compute_c5_vals(best_h)
        print("Parent: n={}, C5={:.8f}".format(best_n, best_c5))
    except Exception:
        best_n = 160
        best_h = make_thue_morse(best_n)
        best_c5, _, _ = compute_c5_vals(best_h)
        print("No parent: n={}, C5={:.8f}".format(best_n, best_c5))

    # Phase 1: Try multiple resolutions via interpolation
    for n_try in [120, 140, 160, 170, 180, 190, 200, 220, 240, 280, 320]:
        if time_left() < 30:
            break
        h_try = interp_reproject(best_h, best_n, n_try)
        c5_try, _, _ = compute_c5_vals(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try
            print("  Res n={}: C5={:.8f}".format(n_try, c5_try))

    # Phase 2: Try structured initializations at current best_n
    n = best_n
    candidates = [best_h.copy()]
    candidates.append(make_thue_morse(n))
    for freq in [2, 3, 4, 5, 7]:
        for ph in [0.0, 0.5, 1.0, np.pi / 2]:
            candidates.append(make_sine_init(n, freq, ph))
    # Random binary
    hb = np.zeros(n)
    idx = rng.permutation(n)
    hb[idx[:n//2]] = 1.0
    candidates.append(hb)

    for hc in candidates:
        c5_c, _, _ = compute_c5_vals(hc)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = hc.copy()
            print("  Better init: C5={:.8f}".format(best_c5))

    # Phase 3: Main optimization loop - alternate CD and L-BFGS-B
    n = best_n
    h = best_h.copy()
    c5_cur, corr_cur, dx = compute_c5_vals(h)
    bounds = [(0.0, 1.0)] * n

    for outer in range(40):
        if time_left() < 60:
            break

        # CD phase (main workhorse)
        h, c5_cd, corr_cur = pairwise_cd(h, n, dx, corr_cur,
                                          max_sweeps=15, t0=t0, budget=budget,
                                          rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h.copy()

        # L-BFGS-B polish
        if time_left() > 30:
            for beta in [50, 150, 400]:
                if time_left() < 10:
                    break
                try:
                    res = minimize(
                        lambda x: obj_smooth(x, n, dx, beta) + 100.0 * (np.sum(x) - n / 2.0) ** 2,
                        h, method='L-BFGS-B', bounds=bounds,
                        options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-12}
                    )
                    h_new = project_h_to_box_sum(res.x, n)
                    c5_new, _, _ = compute_c5_vals(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h = h_new.copy()
                        c5_cur, corr_cur, dx = compute_c5_vals(h)
                except Exception:
                    pass

        # SLSQP with constraint
        if time_left() > 20:
            for beta in [200, 600]:
                if time_left() < 8:
                    break
                try:
                    res = minimize(
                        lambda x: obj_smooth(x, n, dx, beta),
                        h, method='SLSQP', bounds=bounds,
                        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}],
                        options={'maxiter': 200, 'ftol': 1e-16, 'iprint': 0}
                    )
                    h_new = project_h_to_box_sum(res.x, n)
                    c5_new, _, _ = compute_c5_vals(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h = h_new.copy()
                        c5_cur, corr_cur, dx = compute_c5_vals(h)
                except Exception:
                    pass

        # Check convergence
        c5_now, corr_now, _ = compute_c5_vals(h)
        if abs(c5_now - best_c5) < 1e-10 and c5_now >= best_c5 - 1e-10:
            if outer > 10:
                break

        if outer % 5 == 0:
            print("  Outer {}: C5={:.8f}".format(outer, best_c5))

    # Phase 4: Perturbation-based escape
    if time_left() > 30:
        n = best_n
        h = best_h.copy()
        deltas = [0.003, 0.005, 0.01, 0.015, 0.02, 0.04]
        for piter in range(15000):
            if time_left() < 5:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in deltas:
                found = False
                if h[i] >= d and h[j] <= 1.0 - d:
                    h_tr = h.copy()
                    h_tr[i] -= d
                    h_tr[j] += d
                    c5_tr, _, _ = compute_c5_vals(h_tr)
                    if c5_tr < best_c5:
                        best_c5 = c5_tr
                        best_h = h_tr.copy()
                        h = h_tr.copy()
                        found = True
                if not found and h[j] >= d and h[i] <= 1.0 - d:
                    h_tr = h.copy()
                    h_tr[j] -= d
                    h_tr[i] += d
                    c5_tr, _, _ = compute_c5_vals(h_tr)
                    if c5_tr < best_c5:
                        best_c5 = c5_tr
                        best_h = h_tr.copy()
                        h = h_tr.copy()
                        found = True
                if found:
                    break

    # Phase 5: CD after perturbation
    if time_left() > 20:
        n = best_n
        h = best_h.copy()
        c5_cur, corr_cur, dx = compute_c5_vals(h)
        h, c5_cd, corr_cur = pairwise_cd(h, n, dx, corr_cur,
                                          max_sweeps=20, t0=t0, budget=budget,
                                          rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h.copy()

    # Phase 6: Final resolution sweep
    if time_left() > 20:
        for dn in range(-30, 31, 5):
            if time_left() < 5:
                break
            if dn == 0:
                continue
            nt = best_n + dn
            if nt < 80 or nt > 400:
                continue
            ht = interp_reproject(best_h, best_n, nt)
            ct, _, _ = compute_c5_vals(ht)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()
                best_n = nt
                print("  Final res n={}: C5={:.8f}".format(nt, ct))

    # Phase 7: Final high-precision CD polish
    if time_left() > 15:
        n = best_n
        h = best_h.copy()
        c5_cur, corr_cur, dx = compute_c5_vals(h)
        h, c5_final, corr_cur = pairwise_cd(h, n, dx, corr_cur,
                                             max_sweeps=30, t0=t0, budget=budget,
                                             rng=None)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h.copy()

    # Phase 8: Final L-BFGS-B polish
    if time_left() > 10:
        n = best_n
        h = best_h.copy()
        bounds = [(0.0, 1.0)] * n
        for beta in [1000, 3000]:
            if time_left() < 5:
                break
            try:
                res = minimize(
                    lambda x: obj_smooth(x, n, 2.0/n, beta) + 500.0*(np.sum(x)-n/2.0)**2,
                    h, method='L-BFGS-B', bounds=bounds,
                    options={'maxiter': 500, 'ftol': 1e-17, 'gtol': 1e-13}
                )
                h_new = project_h_to_box_sum(res.x, n)
                c5_new, _, _ = compute_c5_vals(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    # Final projection and evaluation
    best_h = project_h_to_box_sum(best_h, best_n)
    final_c5, _, _ = compute_c5_vals(best_h)
    if final_c5 > best_c5:
        best_c5 = min(best_c5, final_c5)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < best_c5:
            best_c5 = ec5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, best_c5))
    return (best_h, best_c5, best_n)
