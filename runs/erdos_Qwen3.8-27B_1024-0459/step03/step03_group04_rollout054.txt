    cons = [{'type': 'eq', 'fun': con_sum}]
    try:
        res = minimize(obj_slsqp_act, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

def do_slsqp_full(h, n, bounds, beta, maxiter):
    _G['n'] = n
    _G['active'] = list(range(2 * n - 1))
    _G['beta'] = beta
    cons = [{'type': 'eq', 'fun': con_sum}]
    try:
        res = minimize(obj_slsqp_full, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

def gradient_directed_search(h, n, best_c5, rng, max_iters, t0, budget):
    """Use gradient of active correlations to guide 2-swap moves."""
    h = h.copy()
    deltas_small = [0.001, 0.002, 0.003, 0.005]
    deltas_med = [0.008, 0.01, 0.015, 0.02, 0.03]
    for it in range(max_iters):
        if tl(t0, budget) < 3:
            break
        if it % 500 == 0:
            _, corr = compute_c5(h)
            top5 = np.argsort(corr)[-5:]
            n_pts = len(h)
            grad = np.zeros(n_pts)
            for k_idx in top5:
                offset = k_idx - (n_pts - 1)
                if offset >= 0:
                    np_ = n_pts - offset
                    grad[offset:offset + np_] += (1.0 - h[:np_])
                    grad[:np_] -= h[offset:offset + np_]
                else:
                    ao = -offset
                    np_ = n_pts - ao
                    grad[:np_] += (1.0 - h[ao:ao + np_])
                    grad[ao:ao + np_] -= h[:np_]
            # Gradient direction: decrease where grad is positive, increase where negative
            # Pick top candidates for decrease and increase
            dec_cands = np.argsort(grad)[-20:]      # most positive gradient -> decrease
            inc_cands = np.argsort(grad)[:20]       # most negative gradient -> increase
            deltas = deltas_small if it % 1000 == 0 else deltas_med
            improved = False
            for _ in range(10):
                if tl(t0, budget) < 1:
                    break
                i = int(dec_cands[rng.randint(len(dec_cands))])
                j = int(inc_cands[rng.randint(len(inc_cands))])
                if i == j:
                    continue
                for d in deltas:
                    if h[i] >= d and h[j] <= 1.0 - d:
                        ht = h.copy()
                        ht[i] -= d
                        ht[j] += d
                        c5t, _ = compute_c5(ht)
                        if c5t < best_c5 - 1e-13:
                            best_c5 = c5t
                            h = ht
                            improved = True
                            break
                    if h[j] >= d and h[i] <= 1.0 - d:
                        ht = h.copy()
                        ht[j] -= d
                        ht[i] += d
                        c5t, _ = compute_c5(ht)
                        if c5t < best_c5 - 1e-13:
                            best_c5 = c5t
                            h = ht
                            improved = True
                            break
                    if improved:
                        break
            if not improved and it % 100 == 0:
                # Random escape: perturb 5 random positions
                idx5 = rng.choice(n_pts, size=5, replace=False)
                h[idx5] += rng.randn(5) * 0.02
                h = project_feasible(h, n_pts)
    return h, best_c5

def random_2swap(h, n, best_c5, rng, max_iters, t0, budget):
    h = h.copy()
    deltas = [0.002, 0.005, 0.01, 0.02, 0.04]
    for _ in range(max_iters):
        if tl(t0, budget) < 2:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for d in deltas:
            if h[i] >= d and h[j] <= 1.0 - d:
                ht = h.copy()
                ht[i] -= d
                ht[j] += d
                c5t, _ = compute_c5(ht)
                if c5t < best_c5 - 1e-13:
                    best_c5 = c5t
                    h = ht
                    break
            if h[j] >= d and h[i] <= 1.0 - d:
                ht = h.copy()
                ht[j] -= d
                ht[i] += d
                c5t, _ = compute_c5(ht)
                if c5t < best_c5 - 1e-13:
                    best_c5 = c5t
                    h = ht
                    break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5, _ = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 160
        best_h = np.full(best_n, 0.5)
        best_c5, _ = compute_c5(best_h)
        print("No parent: n={}, C5={:.8f}".format(best_n, best_c5))

    # === Phase 1: Resolution sweep (broad) ===
    sweep_ns = list(range(130, 290, 2))
    for nc in sweep_ns:
        if tl(t0, budget) < 5:
            break
        hc = interp_n(best_h, best_n, nc)
        hc = project_feasible(hc, nc)
        cc, _ = compute_c5(hc)
        if cc < best_c5 - 1e-10:
            best_c5 = cc
            best_h = hc.copy()
            best_n = nc
            print("  Sweep n={}: C5={:.8f}".format(nc, cc))

    n = best_n
    h = best_h.copy()
    bounds = [(0.0, 1.0)] * n

    # === Phase 2: Progressive active-set optimization ===
    stall_count = 0
    prev_c5 = best_c5

    for rnd in range(40):
        if tl(t0, budget) < 80:
            break

        c5_cur, corr_cur = compute_c5(h)

        # Try multiple active set sizes
        for top_m in [12, 18, 25]:
            if tl(t0, budget) < 15:
                break
            active = find_top_shifts(corr_cur, top_m, 0.88)

            # L-BFGS-B with moderate beta
            h_new = do_lbfgsb(h, n, bounds, active, 150, 50.0, 600)
            c5_new, _ = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
                c5_cur = c5_new

        # SLSQP with tighter beta
        if tl(t0, budget) > 25:
            c5_cur2, corr_cur2 = compute_c5(h)
            active = find_top_shifts(corr_cur2, 18, 0.92)
            h_new = do_slsqp_act(h, n, bounds, active, 400, 250)
            c5_new, _ = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
                c5_cur = c5_new

        # Track stalls
        if best_c5 < prev_c5 - 1e-10:
            stall_count = 0
        else:
            stall_count += 1
        prev_c5 = best_c5

        if stall_count >= 6:
            # Restart with perturbation
            h = best_h.copy()
            h += rng.randn(n) * 0.03
            h = project_feasible(h, n)
            stall_count = 0
            print("  Restart with perturbation at round {}".format(rnd))

        if rnd % 5 == 0:
            cp, _ = compute_c5(h)
            print("  Round {}: C5={:.8f}".format(rnd, cp))

    # === Phase 3: Full-correlation SLSQP polish (high beta) ===
    if tl(t0, budget) > 30:
        for beta_f in [800, 2000, 5000]:
            if tl(t0, budget) < 12:
                break
            h_new = do_slsqp_full(best_h, best_n, bounds, beta_f, 150)
            c5_new, _ = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  FullSLSQP beta={}: C5={:.8f}".format(beta_f, best_c5))

    # === Phase 4: Gradient-directed local search ===
    if tl(t0, budget) > 30:
        max_iters = int(min(30000, tl(t0, budget) * 30))
        h, best_c5 = gradient_directed_search(best_h, best_n, best_c5, rng, max_iters, t0, budget)
        print("  After grad search: C5={:.8f}".format(best_c5))

    # === Phase 5: Random 2-swap refinement ===
    if tl(t0, budget) > 20:
        max_iters = int(min(30000, tl(t0, budget) * 40))
        best_h, best_c5 = random_2swap(best_h, best_n, best_c5, rng, max_iters, t0, budget)
        print("  After rand search: C5={:.8f}".format(best_c5))

    # === Phase 6: Multi-resolution final check (fine) ===
    if tl(t0, budget) > 20:
        for dn in range(-12, 13, 1):
            if tl(t0, budget) < 3:
                break
            if dn == 0:
                continue
            nt = best_n + dn
            if nt < 100 or nt > 320:
                continue
            ht = interp_n(best_h, best_n, nt)
            ht = project_feasible(ht, nt)
            ct, _ = compute_c5(ht)
            if ct < best_c5 - 1e-10:
                bounds_t = [(0.0, 1.0)] * nt
                _G['n'] = nt
                c5_t, corr_t = compute_c5(ht)
                act_t = find_top_shifts(corr_t, 20, 0.88)
                ht2 = do_lbfgsb(ht, nt, bounds_t, act_t, 200, 100.0, 400)
                ct2, _ = compute_c5(ht2)
                if ct2 < best_c5 - 1e-10:
                    best_c5 = ct2
                    best_h = ht2.copy()
                    best_n = nt
                    print("  FineRes n={}: C5={:.8f}".format(nt, ct2))

    # === Phase 7: Final SLSQP squeeze ===
    if tl(t0, budget) > 8:
        n_f = best_n
        bounds_f = [(0.0, 1.0)] * n_f
        h_f = best_h.copy()
        for bf in [3000, 8000, 15000]:
            if tl(t0, budget) < 4:
                break
            h_new = do_slsqp_full(h_f, n_f, bounds_f, bf, 100)
            c5_new, _ = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_f = h_new.copy()

    # Final
    best_h = project_feasible(best_h, best_n)
    final_c5, _ = compute_c5(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
