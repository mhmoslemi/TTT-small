        if remaining_time() < 2:
            break
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        i = rng.randint(len(ones_list))
        j = rng.randint(len(zeros_list))
        oi = ones_list[i]
        oz = zeros_list[j]

        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = compute_c5(h)
        T = t_start * (t_end / t_start) ** (float(it) / max(float(iters), 1.0))

        if new_c5 < cur_c5 or (T > 1e-15 and rng.rand() < np.exp(-(new_c5 - cur_c5) / T)):
            cur_c5 = new_c5
            ones_list[i] = oz
            zeros_list[j] = oi
            no_improve = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
        else:
            h[oi] = 1.0
            h[oz] = 0.0
            no_improve += 1
            if no_improve > 500:
                no_improve = 0
                v = np.zeros(n)
                for jj in range(2 * n - 1):
                    s = jj - (n - 1)
                    if s >= 0:
                        a = np.arange(s, n)
                        v[a] += (1.0 - h[a - s])
                        b = np.arange(0, n - s)
                        v[b] -= h[b + s]
                    else:
                        sp = -s
                        a = np.arange(0, n - sp)
                        v[a] += (1.0 - h[a + sp])
                        b = np.arange(sp, n)
                        v[b] -= h[b - sp]
                flip_idx = int(np.argmin(v))
                if h[flip_idx] == 1.0:
                    z = int(rng.randint(len(zeros_list)))
                    h[flip_idx] = 0.0
                    h[zeros_list[z]] = 1.0
                    ones_list.remove(flip_idx)
                    ones_list.append(zeros_list[z])
                    zeros_list[z] = flip_idx
                    cur_c5 = compute_c5(h)
                elif h[flip_idx] == 0.0:
                    o = int(rng.randint(len(ones_list)))
                    h[flip_idx] = 1.0
                    h[ones_list[o]] = 0.0
                    zeros_list.append(ones_list[o])
                    zeros_list.remove(flip_idx)
                    ones_list[o] = flip_idx
                    cur_c5 = compute_c5(h)
    return best_h, best_c5


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    return h


def make_m_sequence(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h


def make_structured(n, freq, phase, amp):
    t = np.arange(n) / float(n)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * t + phase)
    return np.clip(h, 0.0, 1.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def coordinate_descent(h, n, sweeps=4, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    best_c5 = compute_c5(h)
    for sw in range(sweeps):
        if remaining_time() < 2:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if remaining_time() < 1:
                break
            old = h[i]
            best_local = h.copy()
            best_local_c5 = compute_c5(h)
            for nv in [0.0, 1.0, 0.5, max(0.0, old - 0.3), min(1.0, old + 0.3),
                       max(0.0, old - 0.15), min(1.0, old + 0.15)]:
                if abs(nv - old) < 1e-9:
                    continue
                h[i] = nv
                h_feas = project_feasible(h, n)
                c5_test = compute_c5(h_feas)
                if c5_test < best_local_c5:
                    best_local_c5 = c5_test
                    best_local = h_feas.copy()
            h = best_local.copy()
            if best_local_c5 < best_c5:
                best_c5 = best_local_c5
                any_imp = True
        if not any_imp:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.7f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 160
        best_h = make_thue_morse(best_n)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)
        print("Thue-Morse: n={}, C5={:.7f}".format(best_n, best_c5))

    n_list = [best_n]
    for nn in [160, 192, 200, 224, 240, 256, 288, 320]:
        if nn not in n_list and 120 <= nn <= 384:
            n_list.append(nn)
    n_list = sorted(set(n_list))

    # Phase 1: Multi-resolution heavy-ball optimization
    for n in n_list:
        if remaining_time() < 30:
            break
        print("\n--- n={} ---".format(n))

        inits = []
        h_interp = interp_to(best_h, best_n, n)
        inits.append(project_feasible(h_interp, n))
        inits.append(project_feasible(make_thue_morse(n), n))
        inits.append(project_feasible(make_m_sequence(n, rng), n))
        inits.append(project_feasible(make_structured(n, 3, 0.5, 0.3), n))
        inits.append(project_feasible(make_structured(n, 7, 1.2, 0.25), n))
        inits.append(project_feasible(make_structured(n, 11, 3.7, 0.2), n))
        if n % 16 == 0:
            tm16 = make_thue_morse(128 if n >= 128 else 64)
            if len(tm16) < n:
                # Tile or pad
                h_tiled = np.tile(tm16, n // len(tm16) + 1)[:n]
                inits.append(project_feasible(h_tiled, n))
            else:
                inits.append(project_feasible(tm16[:n], n))

        for h0 in inits:
            if remaining_time() < 15:
                break
            c5_init = compute_c5(h0)
            if c5_init > best_c5 * 1.15:
                continue

            # Heavy ball with different params
            for (k, lr0, mu, steps) in [(10, 0.08, 0.85, 300), (5, 0.05, 0.9, 400), (15, 0.06, 0.8, 200)]:
                if remaining_time() < 8:
                    break
                h_new, c5_new = heavy_ball(h0, n, steps, k, lr0, mu)
                if c5_new < c5_init:
                    h0 = h_new
                    c5_init = c5_new

            if remaining_time() < 6:
                continue

            # L-BFGS-B polish
            for beta in [200, 500]:
                if remaining_time() < 5:
                    break
                h_new, c5_new = lfgb_optimize(h0, n, beta, 150)
                if c5_new < c5_init:
                    h0 = h_new
                    c5_init = c5_new

            if c5_init < best_c5:
                best_c5 = c5_init
                best_h = h0.copy()
                best_n = n
                print("  HB improved: n={}, C5={:.7f}".format(n, best_c5))

    # Phase 2: Intensive optimization at best resolution
    n = best_n
    h = best_h.copy()
    print("\n--- Intensive (n={}) ---".format(n))

    # Multiple heavy-ball passes with decreasing k
    for (k, lr0, mu, steps) in [(8, 0.06, 0.85, 500), (5, 0.04, 0.9, 500),
                                  (3, 0.03, 0.92, 500), (10, 0.05, 0.85, 300),
                                  (5, 0.04, 0.9, 400), (3, 0.02, 0.93, 600)]:
        if remaining_time() < 10:
            break
        h_new, c5_new = heavy_ball(h, n, steps, k, lr0, mu)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            print("  HB k={}: C5={:.7f}".format(k, best_c5))

    # L-BFGS-B with increasing beta
    for beta in [300, 600, 1000, 1500]:
        if remaining_time() < 6:
            break
        h_new, c5_new = lfgb_optimize(h, n, beta, 200)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            print("  L-FG-B beta={}: C5={:.7f}".format(beta, best_c5))

    # Phase 3: Coordinate descent
    if remaining_time() > 10:
        h_new, c5_new = coordinate_descent(h, n, sweeps=6, rng=rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            print("  CoordDescent: C5={:.7f}".format(best_c5))

    # Phase 4: Binary local search
    if remaining_time() > 30:
        print("\n--- Binary LS (n={}) ---".format(n))
        h_bin = np.where(best_h > 0.5, 1.0, 0.0)
        h_bin = project_feasible(h_bin, n)

        # Ensure exactly n/2 ones for binary
        num_ones = int(np.sum(h_bin))
        target_ones = n // 2
        if num_ones != target_ones:
            h_b = np.zeros(n)
            idx = rng.permutation(n)
            h_b[idx[:target_ones]] = 1.0
            # Use parent structure as guide
            h_guide = np.where(best_h > 0.5, 1.0, 0.0)
            h_b = project_feasible(h_guide, n)

        for trial in range(3):
            if remaining_time() < 10:
                break
            if trial == 0:
                start_h = h_guide
            else:
                start_h = make_m_sequence(n, rng)
            h_bin_new, c5_bin = binary_local_search(start_h, n, 3000, rng)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin_new.copy()
                print("  BinLS trial {}: C5={:.7f}".format(trial, best_c5))
                # Polish binary result with continuous
                if remaining_time() > 8:
                    h_pol = project_feasible(h_bin_new, n)
                    h_pol, c5_pol = lfgb_optimize(h_pol, n, 500, 100)
                    if c5_pol < best_c5:
                        best_c5 = c5_pol
                        best_h = h_pol.copy()
                        print("  Polish: C5={:.7f}".format(best_c5))

    # Phase 5: Interleaved multi-resolution refinement
    if remaining_time() > 40:
        print("\n--- Multi-res refinement ---")
        for n_alt in [n - 16, n + 16, n + 32]:
            if n_alt < 100 or n_alt > 400 or remaining_time() < 15:
                continue
            h_alt = interp_to(best_h, best_n, n_alt)
            h_alt = project_feasible(h_alt, n_alt)
            h_alt, c5_alt = heavy_ball(h_alt, n_alt, 300, 8, 0.06, 0.85)
            if remaining_time() > 6:
                h_alt, c5_alt = lfgb_optimize(h_alt, n_alt, 500, 150)
            if c5_alt < best_c5:
                best_c5 = c5_alt
                best_h = h_alt.copy()
                best_n = n_alt
                print("  Alt n={}: C5={:.7f}".format(n_alt, best_c5))

    # Phase 6: Final intensive push
    if remaining_time() > 15:
        n_final = best_n
        h_final = best_h.copy()
        # Very focused optimization on top 3 lags
        for k in [3, 2, 4]:
            if remaining_time() < 5:
                break
            h_final, c5_final = heavy_ball(h_final, n_final, 400, k, 0.02, 0.93)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()
                print("  Final k={}: C5={:.7f}".format(k, best_c5))

        if remaining_time() > 5:
            h_final, c5_final = lfgb_optimize(h_final, n_final, 800, 100)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, best_c5))
    return (best_h, best_c5, best_n)
