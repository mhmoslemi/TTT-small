def smooth_grad(h, beta, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    max_c = np.max(corr)
    exps = np.exp(beta * (corr - max_c))
    s = np.sum(exps)
    w = exps / s  # weights, shape (2n-1,)
    obj = (max_c + np.log(s) / beta) * dx
    
    grad = np.zeros(n)
    
    # k = 0 (m = n-1): corr = sum h[i]*(1-h[i]), grad = 1-2h[i]
    grad += w[n-1] * (1.0 - 2.0 * h)
    
    # k > 0 (m = n-1+k):
    for k in range(1, n):
        L = n - k
        wk = w[n-1+k]
        if wk == 0: continue
        # (1-h[i+k]) for i in [0, L-1]
        grad[:L] += wk * (1.0 - h[k:])
        # -h[i-k] for i in [k, n-1]
        grad[k:] -= wk * h[:L]
    
    # d > 0 (m = n-1-d):

        L = n - d
        wd = w[n-1-d]
        if wd == 0: continue
        # (1-h[i-d]) for i in [d, n-1]
        grad[d:] += wd * (1.0 - h[:L])
        # -h[i+d] for i in [0, L-1]
        grad[:L] -= wd * h[d:]
    
    return obj, dx * grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, beta, n, penalty):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted = np.clip(shifted, -500, 500)
    exps = np.exp(shifted)
    s = np.sum(exps)
    w = exps / s
    obj = (max_c + np.log(s) / beta) * dx
    grad = np.zeros(n)
    grad += w[n - 1] * (1.0 - 2.0 * h)
    for k in range(1, n):
        L = n - k
        wk = w[n - 1 + k]
        if wk < 1e-15:
            continue
        grad[:L] += wk * (1.0 - h[k:])
        grad[k:] -= wk * h[:L]
    for d in range(1, n):
        L = n - d
        wd = w[n - 1 - d]
        if wd < 1e-15:
            continue
        grad[d:] += wd * (1.0 - h[:L])
        grad[:L] -= wd * h[d:]
    obj *= 1.0
    grad *= dx
    s_diff = np.sum(h) - n / 2.0
    obj += penalty * s_diff * s_diff
    grad += 2.0 * penalty * s_diff
    return obj, grad

def penalty_objective(h, beta, n, penalty):
    return smooth_obj_and_grad(h, beta, n, penalty)[0]

def penalty_jacobian(h, beta, n, penalty):
    return smooth_obj_and_grad(h, beta, n, penalty)[1]

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = 0
        v = i
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return h

def make_golden_ratio(n):
    h = np.zeros(n)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return h

def make_random_bin(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def make_antisymmetric(rng, n):
    half = n // 2
    h = np.zeros(n)
    for i in range(half):
        val = rng.rand()
        h[i] = val
        h[i + half] = 1.0 - val
    if n % 2 == 1:
        h[n - 1] = 0.5
    return h

def make_sinusoidal(n, freq, phase, amp):
    t = np.arange(n) / float(n)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * t + phase)
    return np.clip(h, 0.0, 1.0)

def do_lbfgs(h0, n, beta, penalty, maxiter, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n
    h_cur = h0.copy()
    
    def fun(x):
        return smooth_obj_and_grad(x, beta, n, penalty)[0]
    
    def jac(x):
        return smooth_obj_and_grad(x, beta, n, penalty)[1]
    
    try:
        res = minimize(fun, h_cur, method='L-BFGS-B', jac=jac,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14,
                                'gtol': 1e-8, 'maxfun': maxiter * 3,
                                'iprint': 0})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h_cur

def do_slqp(h0, n, beta, maxiter, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        return smooth_obj_and_grad(x, beta, n, 0.0)[0]
    
    def jac(x):
        return smooth_obj_and_grad(x, beta, n, 0.0)[1]
    
    def con_sum(x):
        return np.sum(x) - n / 2.0
    
    cons = [{'type': 'eq', 'fun': con_sum}]
    h_cur = h0.copy()
    try:
        res = minimize(fun, h_cur, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13,
                                'iprint': 0})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h_cur

def do_coord_descent(h_best, n, budget_check, rng, max_sweeps=20):
    c5_best = compute_c5(h_best)
    h = h_best.copy()
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):
        if budget_check() < 3.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if budget_check() < 2.0:
                break
            old_val = h[i]
            for new_val in [0.0, 1.0, 0.1, 0.4, 0.6, 0.9]:
                if abs(new_val - old_val) < 1e-7:
                    continue
                h[i] = new_val
                h = project_feasible(h, n)
                c5_test = compute_c5(h)
                if c5_test < c5_best - 1e-10:
                    c5_best = c5_test
                    h_best = h.copy()
                    any_imp = True
                else:
                    h[i] = h_best[i]
                    h = h_best.copy()
            for delta in [0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.2, -0.2]:
                if budget_check() < 2.0:
                    break
                trial = h_best.copy()
                trial[i] = np.clip(trial[i] + delta, 0.0, 1.0)
                trial = project_feasible(trial, n)
                c5_test = compute_c5(trial)
                if c5_test < c5_best - 1e-10:
                    c5_best = c5_test
                    h_best = trial.copy()
                    any_imp = True
        if not any_imp and sweep > 2:
            break
    return h_best, c5_best

def do_subgradient(h_best, n, budget_check, rng, max_iters=200):
    c5_best = compute_c5(h_best)
    h = h_best.copy()
    dx = 2.0 / n
    
    for it in range(max_iters):
        if budget_check() < 5.0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = 5
        top_indices = np.argsort(corr)[-top_k:]
        top_indices = top_indices[np.argsort(corr[top_indices])[::-1]]
        
        grad = np.zeros(n)
        for k_idx in top_indices:
            offset = int(k_idx) - (n - 1)
            if offset >= 0:
                n_pairs = n - offset
                if n_pairs <= 0:
                    continue
                grad[offset:offset + n_pairs] += (1.0 - h[:n_pairs])
                grad[:n_pairs] -= h[offset:offset + n_pairs]
            else:
                abs_off = -offset
                n_pairs = n - abs_off
                if n_pairs <= 0:
                    continue
                grad[:n_pairs] += (1.0 - h[abs_off:abs_off + n_pairs])
                grad[abs_off:abs_off + n_pairs] -= h[:n_pairs]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        improved = False
        for step in [0.05, 0.1, 0.2, 0.3, 0.15, 0.08, 0.4]:
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-10:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
        if not improved:
            # Add small random perturbation to escape local optimum
            if it % 10 == 9 and budget_check() > 20:
                h = h_best.copy()
                h += rng.randn(n) * 0.01
                h = project_feasible(h, n)
                if compute_c5(h) > c5_best + 0.01:
                    h = h_best.copy()
    return h_best, c5_best
