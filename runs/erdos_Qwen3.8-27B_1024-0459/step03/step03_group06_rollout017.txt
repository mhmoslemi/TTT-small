```python
            grad[0:n-s] += w * (1.0 - h[s:n])
            grad[s:n] -= w * h[0:n-s]
        else:
            sp = -s
            # C(-sp) = sum_{i=sp}^{n-1} h[i]*(1-h[i-sp])
            # d/dh[i] for i in [sp, n-1]: (1-h[i-sp])
            # d/dh[i] for i in [0, n-1-sp]: -h[i+sp]
            grad[sp:n] += w * (1.0 - h[0:n-sp])
            grad[0:n-sp] -= w * h[sp:n]
    return grad * dx
```

```python
def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:

            h[idx] = min(1.0, h[idx] + diff)

            idx = np.argmax(h * (1.0 - h))

def smooth_max_grad(x):
    n = _N
    dx = 2.0 / n
    beta = _BETA
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)

    h = np.zeros(n)

def make_sturmian(n, alpha):
    h = np.zeros(n)
    s = 0.0
    for i in range(n):
        if s < alpha:
            h[i] = 1.0
            s += alpha
        else:
            h[i] = 0.0
            s -= 1.0 - alpha
    s2 = 0.5
    for i in range(n):
        if s2 < alpha:
            s2 += alpha
        else:
            s2 -= 1.0 - alpha
    return h

        if remaining_time() < 2:
            break
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        
        i = rng.randint(len(ones_list))
        j = rng.randint(len(zeros_list))
        oi = ones_list[i]
        oz = zeros_list[j]
        
        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = compute_c5(h)
        
        T = t_start * (t_end / t_start) ** (float(it) / max(float(max_iters), 1.0))
        
        if new_c5 < current_c5 or (T > 1e-16 and rng.rand() < np.exp(-(new_c5 - current_c5) / T)):
            current_c5 = new_c5
            ones_list[i] = oz
            zeros_list[j] = oi
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0
        
        # Occasional double swap
        if it % 500 == 250 and len(ones_list) > 1 and len(zeros_list) > 1:
            i2 = rng.randint(len(ones_list))
            j2 = rng.randint(len(zeros_list))
            if i2 == i or j2 == j:
                continue
            oi2 = ones_list[i2]
            oz2 = zeros_list[j2]
            h[oi2] = 0.0
            h[oz2] = 1.0
            h[ones_list[i]] = 0.0
            h[zeros_list[j]] = 1.0
            
            new_c5_2 = compute_c5(h)
            if new_c5_2 < current_c5 + 0.001:
                ones_list[i] = zeros_list[j]
                zeros_list[j] = oi
                ones_list[i2] = zeros_list[j2]
                zeros_list[j2] = oi2
                current_c5 = new_c5_2
                if new_c5_2 < best_c5:
                    best_c5 = new_c5_2
                    best_h = h.copy()
            else:
                h[oi2] = 1.0
                h[oz2] = 0.0
                h[ones_list[i]] = 1.0
                h[zeros_list[j]] = 0.0
    
    return best_h, best_c5


def coordinate_descent_continuous(h, n, sweeps=10, rng=None):
    if rng is None:
        rng = np.random.RandomState(1)
    h = h.copy()
    best_c5 = compute_c5(h)
    dx_val = 2.0 / n
    
    for sweep in range(sweeps):
        if remaining_time() < 3:
            break
        order = rng.permutation(n)
        any_imp = False
        for i in order:
            if remaining_time() < 1.5:
                break
            old_val = h[i]
            c0 = compute_c5(h)
            
            # Try a few candidate values
            cands = [0.0, 1.0, 0.5]
            if old_val > 0.2:
                cands.append(max(0.0, old_val - 0.25))
                cands.append(0.1)
            if old_val < 0.8:
                cands.append(min(1.0, old_val + 0.25))
                cands.append(0.9)
            # Add golden ratio offset
            cands.append((old_val + 0.618) % 1.0)
            
            best_val = old_val
            best_h_trial = h.copy()
            best_local = c0
            
            for nv in cands:
                if abs(nv - old_val) < 1e-8:
                    continue
                h[i] = nv
                h_feas = project_feasible(h, n)
                c_test = compute_c5(h_feas)
                if c_test < best_local:
                    best_local = c_test
                    best_val = nv
                    best_h_trial = h_feas.copy()
            
            h = best_h_trial.copy()
            if best_local < best_c5:
                best_c5 = best_local
                any_imp = True
        
        if not any_imp:
            break
    
    return h, best_c5


def subgradient_step(h, n, top_k=4, lr=0.05):
    dx_val = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    
    grad = np.zeros(n)
    for j in top_idx:
        s = j - (n - 1)
        if s >= 0:
            if n - s > 0:
                grad[0:n-s] += (1.0 - h[s:n])
                grad[s:n] -= h[0:n-s]
        else:
            sp = -s
            if n - sp > 0:
                grad[sp:n] += (1.0 - h[0:n-sp])
                grad[0:n-sp] -= h[sp:n]
    
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-14:
        return h
    grad /= gnorm
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new


def multi_start_continuous(h0, n, rng, n_starts=4):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    bounds = [(0.0, 1.0)] * n
    
    inits = [h0.copy()]
    for _ in range(n_starts - 1):
        noisy = h0 + rng.normal(0, 0.1, n)
        inits.append(project_feasible(noisy, n))
    
    for beta_val in [50, 100, 300, 800]:
        if remaining_time() < 5:
            break
        _BETA = beta_val
        _N = n
        _TGT = n / 2.0
        for h_init in inits[:2]:
            if remaining_time() < 4:
                break
            try:
                res = minimize(smooth_max_obj, h_init, jac=smooth_max_grad,
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 300, 'ftol': 1e-14,
                                        'gtol': 1e-10, 'iprint': 0,
                                        'maxls': 40, 'maxfun': 3000})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass
    
    # Top-k penalty refinement
    if remaining_time() > 5:
        try:
            _N = n
            _TGT = n / 2.0
            res = minimize(topk_penalty_obj, best_h, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 200, 'ftol': 1e-14,
                                    'iprint': 0, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S, _N, _BETA, _TGT, _TOP_K
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.7f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 160
        best_h = project_feasible(make_thue_morse(best_n), best_n)
        best_c5 = compute_c5(best_h)
        parent_h = best_h.copy()
        parent_n = best_n
        print("No parent. Thue-Morse: n={}, C5={:.7f}".format(best_n, best_c5))
    
    # Phase 1: Multi-resolution continuous optimization
    n_candidates = [best_n, 200, 240, 280, 320]
    n_candidates = sorted(set([n for n in n_candidates if 100 <= n <= 400]))
    
    for n in n_candidates:
        if remaining_time() < 30:
            break
        
        _N = n
        _TGT = n / 2.0
        print("\n=== Continuous opt n={} (remaining {:.0f}s) ===".format(n, remaining_time()))
        
        # Interpolate parent
        h_interp = project_feasible(interp_to(best_h, best_n, n), n)
        c5_interp = compute_c5(h_interp)
        
        # Structured inits
        h_thue = project_feasible(make_thue_morse(n), n)
        h_sturm1 = project_feasible(make_sturmian_v2(n, 0.5 - 0.5/np.sqrt(5.0)), n)
        h_sturm2 = project_neasible = project_feasible(make_sturmian_v2(n, 0.5 + 0.1/np.sqrt(5.0)), n)
        h_rand = make_random_binary(n, rng)
        
        inits = [h_interp, h_thue, h_rand]
        if c5_interp > best_c5:
            inits = [h_thue, h_rand, h_interp]
        
        for h0 in inits:
            if remaining_time() < 12:
                break
            h_opt, c5_opt = multi_start_continuous(h0, n, rng, n_starts=2)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
                print("  Continuous n={}: C5={:.7f}".format(n, c5_opt))
        
        # Binary search from best continuous at this n
        if remaining_time() > 20:
            h_bin = np.where(best_h > 0.5, 1.0, 0.0)
            h_bin = project_feasible(h_bin, n)
            h_sa, c5_sa = binary_swap_search(h_bin, n, max_iters=15000, rng=rng)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n
                print("  Binary SA n={}: C5={:.7f}".format(n, c5_sa))
            
            # Also try binary from thue-morse
            h_tm_bin = make_thue_morse(n)
            # Fix count
            ones = np.sum(h_tm_bin)
            if ones != n // 2:
                idx = np.where(h_tm_bin == 1.0)[0] if ones > n//2 else np.where(h_tm_bin == 0.0)[0]
                h_tm_bin[idx[:abs(int(ones - n//2))]] = 0.0 if ones > n//2 else 1.0
            h_sa2, c5_sa2 = binary_swap_search(h_tm_bin, n, max_iters=10000, rng=rng)
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2.copy()
                best_n = n
                print("  Binary TM n={}: C5={:.7f}".format(n, c5_sa2))
    
    # Phase 2: Aggressive binary local search at best resolution
    if remaining_time() > 30:
        n = best_n
        print("\n=== Aggressive binary search n={} (remaining {:.0f}s) ===".format(n, remaining_time()))
        
        h_bin = np.where(best_h > 0.5, 1.0, 0.0)
        h_bin = project_feasible(h_bin, n)
        
        # Multiple restarts of SA
        for restart in range(5):
            if remaining_time() < 8:
                break
            if restart == 0:
                h_start = h_bin.copy()
            else:
                h_start = make_random_binary(n, rng)
            
            h_sa, c5_sa = binary_swap_search(h_start, n, max_iters=20000, rng=rng,
                                             temper=0.002 * (0.7 ** restart))
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n
                print("  SA restart {}: C5={:.7f}".format(restart, c5_sa))
        
        # Subgradient + binary polish
        if remaining_time() > 10:
            h_sg = best_h.copy()
            for step in range(200):
                if remaining_time() < 3:
                    break
                h_sg = subgradient_step(h_sg, n, top_k=5, lr=0.03)
                c5_sg = compute_c5(h_sg)
                if c5_sg < best_c5:
                    best_c5 = c5_sg
                    best_h = h_sg.copy()
                    break
                # Also try rounding
                h_round = np.where(h_sg > 0.5, 1.0, 0.0)
                h_round = project_feasible(h_round, n)
                c5_round = compute_c5(h_round)
                if c5_round < best_c5:
                    best_c5 = c5_round
                    best_h = h_round.copy()
    
    # Phase 3: Continuous refinement from best binary
    if remaining_time() > 15:
        n = best_n
        _N = n
        _TGT = n / 2.0
        print("\n=== Continuous refinement n={} (remaining {:.0f}s) ===".format(n, remaining_time()))
        
        h_cont = project_feasible(best_h * 0.7 + 0.5 * 0.3, n)
        h_cont = np.clip(h_cont, 0, 1)
        h_cont = project_feasible(h_cont, n)
        
        for beta_val in [200, 500, 1000, 2000]:
            if remaining_time() < 5:
                break
            _BETA = beta_val
            bounds = [(0.0, 1.0)] * n
            try:
                res = minimize(smooth_max_obj, h_cont, jac=smooth_max_grad,
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 400, 'ftol': 1e-15,
                                        'gtol': 1e-11, 'iprint': 0,
                                        'maxls': 50, 'maxfun': 4000})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_cont = h_new.copy()
                    print("  Beta {}: C5={:.7f}".format(beta_val, c5_new))
            except Exception:
                pass
        
        # Coordinate descent
        if remaining_time() > 8:
            h_cd, c5_cd = coordinate_descent_continuous(best_h, n, sweeps=8, rng=rng)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                print("  CoordDesc: C5={:.7f}".format(c5_cd))
    
    # Phase 4: Final binary polish (short SA)
    if remaining_time() > 10:
        n = best_n
        h_fin = np.where(best_h > 0.5, 1.0, 0.0)
        h_fin = project_feasible(h_fin, n)
        h_fin2, c5_fin = binary_swap_search(h_fin, n, max_iters=15000, rng=rng)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin2.copy()
            print("Final binary: C5={:.7f}".format(c5_fin))
        
        # One more continuous pass
        if remaining_time() > 5:
            _N = n
            _TGT = n / 2.0
            _BETA = 1500
            bounds = [(0.0, 1.0)] * n
            try:
                res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 300, 'ftol': 1e-15,
                                        'gtol': 1e-12, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
```
