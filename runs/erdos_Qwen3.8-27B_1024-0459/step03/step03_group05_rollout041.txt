def project_feasible(h, target_sum):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break

        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = np.sum(mask)
