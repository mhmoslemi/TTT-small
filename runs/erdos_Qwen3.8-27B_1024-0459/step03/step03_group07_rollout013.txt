        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)
    
    # Resolutions to try
    resolutions = [192, 224, 256, 288]
    # Always include parent resolution
    if best_n not in resolutions:
        resolutions.append(best_n)
    resolutions = sorted(set(resolutions))
    
    # Allocate time: give more to higher resolutions
    # Total available
    total_time = budget_s - 60  # reserve 60s for final polish
    
    # Split time among resolutions (weighted toward higher n)
    weights = {}
    for r in resolutions:
        weights[r] = r ** 1.2 if r != best_n else r ** 1.2 * 0.5
    
    wsum = sum(weights.values())
    time_alloc = {}
    for r in resolutions:
        time_alloc[r] = total_time * weights[r] / wsum
    
    for n in resolutions:
        if time_left() < 40:
            break
        tl = min(time_alloc[n], time_left() - 20)
        if tl < 30:
            tl = time_left() - 20
        
        # Initialize at this resolution
        if n == best_n:
            h_init = project_feasible(best_h, n)
        else:
            h_init = interpolate_h(best_h, best_n, n)
        
        # Generate initializations
        inits = [h_init.copy()]
        
        # Perturbed
        for scale in [0.03, 0.08, 0.15]:
            hp = h_init + scale * rng.randn(n)
            inits.append(project_feasible(hp, n))
        
        # Sine-modulated around parent
        t = np.arange(n) / n
        for freq in [2, 3, 4]:
            phase = rng.rand() * 2 * np.pi
            mod = 0.05 * np.cos(2 * np.pi * freq * t * 2 + phase)
            hs = project_feasible(h_init + mod, n)
            inits.append(hs)
        
        # Pure sine
        for freq in [3, 5, 7]:
            phase = rng.rand() * 2 * np.pi
            hs = 0.5 + 0.35 * np.cos(2 * np.pi * freq * t + phase)
            hs = project_feasible(np.clip(hs, 0, 1), n)
            inits.append(hs)
        
        n_inits = len(inits)
        time_per_init = tl / n_inits
        
        for h0 in inits:
            if time_left() < 20:
                break
            tl_i = min(time_per_init, time_left() - 5)
            if tl_i < 10:
                break
            
            # Phase 1: Coordinate descent
            h1 = coord_descent(h0, n, sweeps=15, time_limit=tl_i * 0.5, t0=t0)
            c5_1 = compute_c5(h1)
            
            if time_left() < 15:
                if c5_1 < best_c5:
                    best_c5 = c5_1
                    best_n = n
                    best_h = h1.copy()
                continue
            
            # Phase 2: L-BFGS smooth max
            h2 = lbfgs_smooth(h1, n, time_limit=tl_i * 0.3, t0=t0)
            c5_2 = compute_c5(h2)
            
            if time_left() < 15:
                cands = [(h1, c5_1), (h2, c5_2)]
                ch, cc = min(cands, key=lambda x: x[1])
                if cc < best_c5:
                    best_c5 = cc
                    best_n = n
                    best_h = ch.copy()
                continue
            
            # Phase 3: SLSQP polish
            h3 = slsqp_polish(h2, n, time_limit=tl_i * 0.15, t0=t0)
            c5_3 = compute_c5(h3)
            
            # Phase 4: More CD
            h4 = coord_descent(h3, n, sweeps=20, time_limit=tl_i * 0.08, t0=t0)
            c5_4 = compute_c5(h4)
            
            # Take best
            candidates = [(h1, c5_1), (h2, c5_2), (h3, c5_3), (h4, c5_4)]
            ch, cc = min(candidates, key=lambda x: x[1])
            if cc < best_c5:
                best_c5 = cc
                best_n = n
                best_h = ch.copy()
    
    # Final intensive polish on best solution
    if time_left() > 30:
        h_p1 = coord_descent(best_h, best_n, sweeps=40, time_limit=time_left() * 0.35, t0=t0)
        c5_p1 = compute_c5(h_p1)
        if c5_p1 < best_c5:
            best_c5 = c5_p1
            best_h = h_p1.copy()
    
    if time_left() > 25:
        h_p2 = lbfgs_smooth(best_h, best_n, time_limit=time_left() * 0.25, t0=t0)
        c5_p2 = compute_c5(h_p2)
        if c5_p2 < best_c5:
            best_c5 = c5_p2
            best_h = h_p2.copy()
    
    if time_left() > 20:
        h_p3 = coord_descent(best_h, best_n, sweeps=30, time_limit=time_left() * 0.2, t0=t0)
        c5_p3 = compute_c5(h_p3)
        if c5_p3 < best_c5:
            best_c5 = c5_p3
            best_h = h_p3.copy()
    
    if time_left() > 15:
        h_p4 = slsqp_polish(best_h, best_n, time_limit=time_left() * 0.15, t0=t0)
        c5_p4 = compute_c5(h_p4)
        if c5_p4 < best_c5:
            best_c5 = c5_p4
            best_h = h_p4.copy()
    
    # Final coordinate descent
    if time_left() > 10:
        h_final = coord_descent(best_h, best_n, sweeps=20, time_limit=max(5.0, time_left() - 3), t0=t0)
        c5_f = compute_c5(h_final)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_final.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    # Validate
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
