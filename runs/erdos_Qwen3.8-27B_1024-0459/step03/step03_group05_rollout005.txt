    y=h - float(np.mean(h))
    dx=2.0/n

def choose_active_lags(h, max_lags=20, gap=None, include_zero=True):
   corr=corr_fft(h); M=len(corr); n=len(h)
   neg_start=M-(n-1)
   valid=np.concatenate((np.arange(n), np.arange(neg_start, M)))
   c=corr[valid]
   order=np.argsort(c)[::-1]
   mx=float(c[order[0]])
   if gap is None: gap = max(1e-5, 0.03*mx)
   min_count = max(1, min(max_lags, max(2, max_lags//4)))

      if float(corr[0]) >= mx - max(gap, 1e-4) or len(lags) < 2:

def project_feasible(h, target):
    h = np.asarray(h, dtype=float).reshape(-1)
    if h.size == 0:
       return h
    target=float(target)
    h = np.where(np.isfinite(h), h, 0.5)
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    # If target impossible? clamp
    if target <= 0:
       return np.zeros_like(h)
    if target >= h.size:
       return np.ones_like(h)
    for _ in range(45):
       mid = 0.5*(lo+hi)
       s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
       if s > target:
          lo = mid
       else:
          hi = mid
    lam = 0.5*(lo+hi)
    p = np.clip(h - lam, 0.0, 1.0)
    diff = target - float(np.sum(p))
    if abs(diff) > 1e-13:
       if diff > 0:
          cand = np.where(p < 1.0 - abs(diff))[0]
          if cand.size:
             idx = int(cand[np.argmax(1.0 - p[cand])])
             p[idx] = min(1.0, p[idx] + diff)
          else:
             p += diff / float(p.size)
       else:
          need = -diff
          cand = np.where(p > need)[0]
          if cand.size:
             idx = int(cand[np.argmax(p[cand])])
             p[idx] = max(0.0, p[idx] + diff)
          else:

def grad_lags_weighted(h,lags,weights):
  n=len(h); g=np.zeros(n,float)

def next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def project_feasible(h, target):
    h = np.asarray(h, dtype=float).reshape(-1)
    if h.size == 0:
        return h
    target = float(target)
    h = np.where(np.isfinite(h), h, 0.5)
    n = h.size
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    p = np.clip(h - lam, 0.0, 1.0)

    diff = target - float(np.sum(p))
    if abs(diff) > 1e-13:

            cand = np.where(p < 1.0 - abs(diff))[0]

def corr_fft(h):
    n = len(h)
    M = next_pow2(max(1, 2 * n - 1))
    F = np.fft.rfft(h, M)
    V = np.fft.rfft(1.0 - h, M)
    return np.fft.irfft(np.conj(F) * V, M)


def c5_fft(h):
    n = len(h)
    if n == 0:
        return 0.0
    c = corr_fft(h)
    M = len(c)
    a = float(np.max(c[:n]))
    if n > 1:
        b = float(np.max(c[M - (n - 1):M]))
    else:
        b = 0.0
    return float(max(a, b) * (2.0 / float(n)))


def c5_exact(h):
    n = len(h)
    if n == 0:
        return 0.0
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / float(n)))


def choose_active_lags(h, max_lags=20, gap=None, include_zero=True):
    n = len(h)
    if n <= 1:
        return [0]
    c = corr_fft(h)
    M = len(c)
    valid = np.concatenate((np.arange(n), np.arange(M - n + 1, M)))
    vals = c[valid].astype(float, copy=False)
    vals = np.where(np.isfinite(vals), vals, -1e100)
    order = np.argsort(vals, kind='mergesort')[::-1]
    max_lags = int(min(max_lags, len(order)))
    mx = float(vals[order[0]])
    if gap is None:
        gap = max(1e-5, 0.02 * max(1.0, mx))
    else:
        gap = float(gap)
    min_count = max(1, min(max_lags, max(2, max_lags // 4)))

    sel = []
    for idx in order:
        if len(sel) >= max_lags:
            break
        if len(sel) >= min_count and float(vals[idx]) < mx - gap:
            break
        sel.append(int(valid[idx]))

    lags = []
    for pos in sel:
        k = pos if pos < n else pos - M
        if k not in lags:
            lags.append(k)
    if include_zero and 0 not in lags:
        v0 = float(c[0])
        if (np.isfinite(v0) and v0 >= mx - max(gap, 1e-4)) or len(lags) < 2:
            lags.insert(0, 0)

def smooth_max_vals(vals, beta):
    vals = np.asarray(vals, dtype=float)
    if vals.size == 0:
        return 0.0
    mx = float(np.max(vals))
    if vals.size == 1:
        return mx
    beta = float(beta)
    if beta <= 1e-12:
        return mx
    e = np.exp(beta * (vals - mx))
    return float(mx + float(np.log(np.sum(e))) / beta)


def smooth_weights(vals, beta):
    vals = np.asarray(vals, dtype=float)
    if vals.size == 0:
        return np.array([], dtype=float)
    mx = float(np.max(vals))
    if vals.size == 1:
        return np.array([1.0], dtype=float)
    beta = float(beta)
    if beta <= 1e-12:
        w = np.zeros(vals.size, dtype=float)
        w[np.argmax(vals) == np.argmax(vals)] = 1.0
        return w
    e = np.exp(beta * (vals - mx))
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros(vals.size, dtype=float)
        w[np.argmax(vals)] = 1.0
        return w
    return e / s


def vals_lags(h, lags):
    n = len(h)
    out = []
    for k in lags:
        if k == 0:
            out.append(float(np.sum(h * (1.0 - h))))
        elif k > 0:
            out.append(float(np.sum(h[:-k] * (1.0 - h[k:]))))
        else:
            q = -k
            out.append(float(np.sum(h[q:] * (1.0 - h[:-q]))))
    return np.asarray(out, dtype=float)


def grad_lags_weighted(h, lags, weights):
    n = len(h)
    g = np.zeros(n, dtype=float)
    for k, w in zip(lags, weights):
        if w == 0.0 or not np.isfinite(w):
            continue
        if k == 0:
            g += w * (1.0 - 2.0 * h)
        elif k > 0:
            g[:-k] += w * (1.0 - h[k:])
            g[k:] -= w * h[:-k]
        else:
            q = -k
            g[q:] += w * (1.0 - h[:-q])
            g[:-q] -= w * h[q:]
    return g


class ActiveSmoothObj:
    def __init__(self, n, target, dx, beta, lags, penalty):
        self.n = int(n)
        self.target = float(target)
        self.dx = float(dx)
        self.beta = float(beta)
        self.lags = [int(x) for x in lags]
        self.penalty = float(penalty)

    def __call__(self, x):
        h = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
        s = float(np.sum(h) - self.target)
        vals = vals_lags(h, self.lags)
        val = smooth_max_vals(vals, self.beta) * self.dx
        if self.penalty > 0.0:
            val += self.penalty * s * s
        if not np.isfinite(val):
            return 1e100
        return float(val)

    def gradient(self, x):
        h = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
        s = float(np.sum(h) - self.target)
        vals = vals_lags(h, self.lags)
        w = smooth_weights(vals, self.beta)
        g = grad_lags_weighted(h, self.lags, w) * self.dx
        if self.penalty > 0.0:
            g += (2.0 * self.penalty * s) * np.ones(self.n, dtype=float)
        g = np.where(np.isfinite(g), g, 0.0)
        return g


def lbfgs_active(h, beta, lags, maxiter=80):
    n = len(h)
    if n == 0:
        return h, c5_fft(h)
    target = float(n) / 2.0
    dx = 2.0 / float(n)
    c0 = c5_fft(h)
    obj = ActiveSmoothObj(n, target, dx, beta, lags, 150.0 if n > 512 else 100.0)
    x0 = np.asarray(h, dtype=float).copy()
    try:
        bounds = [(0.0, 1.0)] * n
        options = {
            'maxiter': int(maxiter),
            'ftol': 1e-15,
            'gtol': 1e-9,
            'maxls': 30,
            'maxcor': 20

                       bounds=bounds, options=options)

        c2 = c5_fft(h2)
        if not np.isfinite(c2):

        return h2, float(c2)
    except Exception:

def gradient_polish(h, deadline, top_m=16, beta0=80.0, alpha0=0.012):
    n = len(h)
    target = float(n) / 2.0
    dx = 2.0 / float(n)
    h = project_feasible(h, target)
    c = c5_fft(h)

    beta = float(beta0)

    while time.time() < deadline and alpha > 1e-7:
        lags = choose_active_lags(h, max_lags=top_m)
        vals = vals_lags(h, lags)
        w = smooth_weights(vals, beta)
        g = grad_lags_weighted(h, lags, w) * dx
        g -= np.mean(g)
        norm = float(np.linalg.norm(g))
        if not np.isfinite(norm) or norm < 1e-14:
            break
        g /= norm

        best_h = None
        best_c = c
        for mult in (1.0, 0.6, 0.35, 0.18, 0.09):
            ht = project_feasible(h - alpha * mult * g, target)
            ct = c5_fft(ht)
            if np.isfinite(ct) and ct < best_c:
                best_c = ct
                best_h = ht
        if best_h is not None:
            h = best_h
            c = best_c
            no_improve = 0
            alpha = min(0.08, alpha * 1.03)
        else:
            no_improve += 1
            alpha *= 0.72
            if no_improve > 30:
                break
        beta *= 1.01
    return h, float(c)


def pair_polish(h, deadline, top_m=12, beta=150.0):

    if n <= 1 or time.time() >= deadline:

def resample_h(h, n_new):
    n_old = len(h)
    n_new = int(n_new)
    if n_new <= 0:
        return project_feasible(h.copy(), max(1, n_old) / 2.0)
    if n_new % 2 != 0:
        n_new += 1
    if n_new == n_old:
        return project_feasible(np.asarray(h, dtype=float).copy(), n_new / 2.0)
    xp = (np.arange(n_old) + 0.5) * (2.0 / float(n_old))
    xn = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    hn = np.interp(xn, xp, h)
    return project_feasible(hn, n_new / 2.0)

def fourier_h_from_x(x, params, M):
    n = len(x)
    target = float(n) / 2.0
    p = np.asarray(params, dtype=float).reshape(-1).copy()
    need = 2 * int(M) + 2
    if len(p) < need:
        tmp = np.zeros(need, dtype=float)
        tmp[:len(p)] = p
        tmp[-2] = 1.0
        tmp[-1] = 0.0
        p = tmp
    else:
        p = p[:need]

    s = np.zeros(n, dtype=float)
    for m in range(1, M + 1):
        a = float(np.clip(p[2 * m - 2], -2.0, 2.0))
        ph = float(p[2 * m - 1])
        s += a * np.sin(np.pi * m * x + ph)
    sc = float(np.clip(p[2 * M], 0.15, 5.0))
    bias = float(np.clip(p[2 * M + 1], -0.2, 0.2))
    h = 0.5 + bias + 0.5 * np.tanh(s / sc)
    h += (target / float(n) - float(np.mean(h)))
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, target)


def fourier_init_from_h(h, n, M):
    n = int(n)
    if n <= 0:
        n = 128
    if n % 2 != 0:
        n += 1
    y = np.asarray(h, dtype=float).reshape(-1)
    if y.size != n:
        y = resample_h(y, n)
    y = y - float(np.mean(y))
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    dx = 2.0 / float(n)
    params = np.zeros(2 * M + 2, dtype=float)
    for m in range(1, M + 1):
        ang = np.pi * m * x
        A = dx * float(np.sum(y * np.sin(ang)))
        B = dx * float(np.sum(y * np.cos(ang)))
        amp = float(np.hypot(A, B))
        amp = float(np.clip(amp, -2.0, 2.0))
        ph = float(np.arctan2(B, A)) if amp > 1e-8 else 0.0
        params[2 * m - 2] = amp
        params[2 * m - 1] = ph
    params[2 * M] = 1.0
    params[2 * M + 1] = 0.0
    return params

def quantize_exact(h, n=None, p=36):
    h = np.asarray(h, dtype=float).reshape(-1)
    if n is None:
        n = len(h)
    n = int(n)
    if n <= 0:
        return np.array([], dtype=float)
    if n % 2 != 0:
        n += 1
        h = resample_h(h, n)
        n = len(h)

    scale = 1 << int(p)
    target = (n // 2) * scale
    hi = np.where(np.isfinite(h), h, 0.5)
    hi = np.clip(hi, 0.0, 1.0)

    ints = np.rint(hi * float(scale)).astype(np.int64)
    ints = np.clip(ints, 0, scale)

    total = int(ints.sum(dtype=np.int64))
    diff = int(target) - total
    if diff > 0:
        d = np.int64(diff)
        for i in range(n):
            room = int(np.int64(scale) - ints[i])
            if d <= room:
                ints[i] = ints[i] + d
                break
            else:
                ints[i] = np.int64(scale)
                d -= room
    elif diff < 0:
        need = np.int64(-diff)
        for i in range(n):
            take = int(min(need, ints[i]))
            if take > 0:
                ints[i] = ints[i] - take
                need -= take
                if need == 0:
                    break

    return ints.astype(np.float64) / float(scale)

    pool = {}
    update_pool(pool, best_opt, best_opt_c)

    phase1_end = min(deadline - 220.0, t0 + min(320.0, 0.20 * budget_s + 60.0))
    if time.time() < phase1_end:
        h1, c1 = local_refine(best_opt, phase1_end, top_m=24, use_pairs=True)
        if c1 < best_opt_c:
            best_opt = h1.copy()
            best_opt_c = c1
        update_pool(pool, best_opt, best_opt_c)
        best_final_h, best_final_c, best_final_n = consider_final(best_opt, best_final_h, best_final_c, best_final_n)

    exp_deadline = min(deadline - 250.0, t0 + min(950.0, 0.60 * budget_s))
    n_opts = [256, 384, 512]
    if budget_s > 1100:
        n_opts.append(768)

    for nn in n_opts:
        if time.time() > exp_deadline - 70.0:
            break
        if nn == len(best_opt):
            continue
        cand = resample_h(best_opt, nn)
        cc = float(c5_fft(cand))
        update_pool(pool, cand, cc)
        best_final_h, best_final_c, best_final_n = consider_final(cand, best_final_h, best_final_c, best_final_n)

        td = min(exp_deadline - 50.0, time.time() + 70.0)
        if time.time() < td and cc <= best_opt_c + 0.003:
            h2, c2 = local_refine(cand, td, top_m=14, use_pairs=False)
            if c2 < best_opt_c:
                best_opt = h2.copy()
                best_opt_c = c2
            update_pool(pool, h2, c2)
            best_final_h, best_final_c, best_final_n = consider_final(h2, best_final_h, best_final_c, best_final_n)
        elif cc <= best_opt_c + 0.01 and time.time() < exp_deadline - 40.0:
            td2 = min(exp_deadline - 35.0, time.time() + 25.0)
            if time.time() < td2:
                h2, c2 = local_refine(cand, td2, top_m=10, use_pairs=False)
                if c2 < best_opt_c:
                    best_opt = h2.copy()
                    best_opt_c = c2
                update_pool(pool, h2, c2)
                best_final_h, best_final_c, best_final_n = consider_final(h2, best_final_h, best_final_c, best_final_n)

    fourier_deadline = min(deadline - 200.0, t0 + min(1100.0, 0.75 * budget_s))
    if time.time() < fourier_deadline - 80.0:
        n_f = 512
        if 512 <= len(best_opt) <= 768:
            n_f = len(best_opt)
        for M in (3, 5):
            if time.time() > fourier_deadline - 70.0:
                break
            fobj = FourierObj(n_f, M)
            bounds = fourier_bounds(M)
            x0 = None
            if differential_evolution is not None:
                maxiter_de = 18 if M <= 3 else 12
                popsize_de = 8 if time.time() < fourier_deadline - 150.0 else 6
                try:
                    de = differential_evolution(
                        fobj, bounds,
                        seed=int(seed) + M,
                        maxiter=maxiter_de,
                        popsize=popsize_de,
                        tol=1e-4,
                        polish=False,
                        workers=1
                    )
                    x0 = np.asarray(de.x, dtype=float)
                except Exception:
                    x0 = None
            if x0 is None:
                x0 = np.array([float(rng.uniform(lo, hi)) for lo, hi in bounds], dtype=float)

            inits = [x0]
            inits.append(fourier_init_from_h(resample_h(best_opt, n_f), n_f, M))
            inits.append(np.array([float(rng.uniform(lo, hi)) for lo, hi in bounds], dtype=float))

            for xi in inits:
                if time.time() > fourier_deadline - 45.0:
                    break
                xbest = np.asarray(xi, dtype=float).copy()
                try:
                    resN = minimize(fobj, xbest, method='Nelder-Mead',
                                    options={'maxiter': 120, 'xatol': 1e-5, 'fatol': 1e-8})
                    xbest = np.asarray(resN.x, dtype=float)
                except Exception:
                    pass
                h_f = fourier_h_from_x(fobj.x, xbest, M)
                c_f = float(c5_fft(h_f))
                update_pool(pool, h_f, c_f)
                best_final_h, best_final_c, best_final_n = consider_final(h_f, best_final_h, best_final_c, best_final_n)
                if c_f < best_opt_c:
                    best_opt = h_f.copy()
                    best_opt_c = c_f

                if c_f <= best_opt_c + 0.001 and time.time() < fourier_deadline - 25.0:
                    t_try = min(fourier_deadline - 15.0, time.time() + 40.0)
                    if time.time() < t_try:
                        h_l, c_l = local_refine(h_f, t_try, top_m=12, use_pairs=False)
                        if c_l < best_opt_c:
                            best_opt = h_l.copy()
                            best_opt_c = c_l
                        update_pool(pool, h_l, c_l)
                        best_final_h, best_final_c, best_final_n = consider_final(h_l, best_final_h, best_final_c, best_final_n)

    if time.time() < deadline - 12.0:
        items = sorted(pool.items(), key=pool_sort_key)[:4]
        for idx, item in enumerate(items):
            nn, (hh, cc) = item
            if time.time() > deadline - 12.0:
                break
            if idx == 0:
                sub = deadline - 8.0
            elif idx == 1:
                sub = min(deadline - 45.0, time.time() + 180.0)
            else:
                sub = min(deadline - 70.0, time.time() + 60.0)
            if time.time() >= sub:
                continue
            h2, c2 = local_refine(hh, sub, top_m=14, use_pairs=(time.time() < deadline - 80.0))
            if c2 < best_opt_c:
                best_opt = h2.copy()
                best_opt_c = c2
            update_pool(pool, h2, c2)
            best_final_h, best_final_c, best_final_n = consider_final(h2, best_final_h, best_final_c, best_final_n)
            if c2 >= cc + 1e-12 and idx == 0:
                pass

    best_final_h, best_final_c, best_final_n = consider_final(best_opt, best_final_h, best_final_c, best_final_n)
    if best_final_h is None:
        best_final_h, best_final_c, best_final_n = final_candidate(h_par)

    try:
        exact_c = float(c5_exact(best_final_h))
        if np.isfinite(exact_c) and exact_c < float(best_final_c):
            best_final_c = exact_c
    except Exception:
        pass

    ev = globals().get('evaluate_erdos_solution', None)
    if ev is not None:
        try:
            ev_c = ev(best_final_h)
            ev_c = float(ev_c)
            if np.isfinite(ev_c) and ev_c < float(best_final_c):
                best_final_c = ev_c
        except Exception:
            pass

    return best_final_h, float(best_final_c), int(best_final_n)
