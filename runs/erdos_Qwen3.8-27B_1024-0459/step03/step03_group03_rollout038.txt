def _swap_delta(h, n, i, j):
    """Compute delta C for swapping h[i]:1->0 and h[j]:0->1."""
    # Range for position i: k in [n-1-i, 2n-2-i]
    # Delta at k: 2*h[i+k-(n-1)] - 1
    # Let t = k - (n-1), then i+k-(n-1) = i+t, valid for 0<=i+t<=n-1, t in [-i, n-1-i]
    # k = n-1+t, so k in [n-1-i, n-1+n-1-i] = [n-1-i, 2n-2-i]
    
    delta = np.zeros(2*n-1)
    
    # Contribution from position i (1->0): delta[k] += 2*h[i+k-(n-1)] - 1
    # k from n-1-i to 2n-2-i
    k_start_i = n - 1 - i
    k_end_i = 2 * n - 2 - i
    t_start = -i
    t_end = n - 1 - i
    # i+k-(n-1) = i + (k-(n-1)) = i + t where t = k-(n-1)
    # For k in [k_start_i, k_end_i], t in [t_start, t_end]
    # i+t ranges from 0 to n-1
    delta[k_start_i:k_end_i+1] += 2.0 * h[i+k_start_i-(n-1) : i+k_end_i-(n-1)+1] - 1.0
    
    # Contribution from position j (0->1): delta[k] += 1 - 2*h[j+k-(n-1)]
    k_start_j = n - 1 - j
    k_end_j = 2 * n - 2 - j
    delta[k_start_j:k_end_j+1] += 1.0 - 2.0 * h[j+k_start_j-(n-1) : j+k_end_j-(n-1)+1]
    
    return delta
