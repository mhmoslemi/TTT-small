```python
def fast_smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    # For each shift s = j - (n-1), j = 0, ..., 2n-2
    for j in range(2*n - 1):
        wj = weights[j]
        if wj < 1e-12:
            continue
        s = j - (n - 1)
        if s >= 0:
            # h[m] appears as h[i+s] for m in [s, n-1]: contrib (1-h[m-s])
            # h[m] appears as h[i] for m in [0, n-1-s]: contrib -h[m+s]
            grad[s:] += wj * (1.0 - h[:n-s])
            grad[:n-s] -= wj * h[s:]
        else:
            sp = -s
            # h[m] appears as h[i-sp] for m in [0, n-1-sp]: contrib (1-h[m+sp])
            # h[m] appears as h[i] for m in [sp, n-1]: contrib -h[m-sp]
            grad[:n-sp] += wj * (1.0 - h[sp:])
            grad[sp:] -= wj * h[:n-sp]
    grad *= dx
    return grad
```

```python
        print("Parent: n={}, C5={:.7f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 200
        best_h = project_feasible(make_thue_morse(best_n), best_n)
        best_c5 = compute_c5(best_h)
        print("Thue-Morse: n={}, C5={:.7f}".format(best_n, best_c5))

    # Phase 1: L-BFGS polish at parent resolution
    print("\n=== Phase 1: L-BFGS polish n={} ===".format(best_n))
    n = best_n
    for beta in [50, 100, 200, 400, 800, 1500]:
        if _remaining() < 12:
            break
        h_new = run_lbfgs(best_h, n, beta, max_iter=300, max_fun=2500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            print("  beta={}: C5={:.7f}".format(beta, c5_new))

    # Phase 2: Binary local search at multiple starting points
    if _remaining() > 40:
        n = best_n
        print("\n=== Phase 2: Binary local search n={} ===".format(n))

        starts = []
        # From current best (thresholded)
        starts.append(np.where(best_h > 0.5, 1.0, 0.0))
        # Thue-Morse
        starts.append(make_thue_morse(n))
        # Rudin-Shapiro
        starts.append(make_rudin_shapiro(n))
        # Sturmian
        golden = (1.0 + np.sqrt(5.0)) / 2.0
        starts.append(make_sturmian(n, 1.0 / golden, 0.5))
        # Random binaries
        for _ in range(3):
            starts.append(random_binary(n, rng))

        for si, h_s in enumerate(starts):
            if _remaining() < 12:
                break
            h_s = project_feasible(h_s, n)
            h_ls, c5_ls = binary_local_search(h_s, n, max_iters=40000, rng=rng, patience=4000)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = project_feasible(h_ls, n)
                best_n = n
                print("  Start {}: C5={:.7f}".format(si, c5_ls))

        # Polish best with continuous L-BFGS
        if _remaining() > 15:
            for beta in [200, 500, 1000]:
                if _remaining() < 8:
                    break
                h_new = run_lbfgs(best_h, n, beta, max_iter=250, max_fun=2000)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  Polish beta={}: C5={:.7f}".format(beta, c5_new))

    # Phase 3: Higher resolution optimization
    for n_try in [200, 240, 280]:
        if _remaining() < 50:
            break
        if n_try <= best_n:
            continue
        print("\n=== Phase 3: n={} ===".format(n_try))

        h_interp = interp_h(best_h, best_n, n_try)
        h_interp = project_feasible(h_interp, n_try)

        inits = [h_interp]
        inits.append(project_feasible(make_thue_morse(n_try), n_try))
        inits.append(project_feasible(make_rudin_shapiro(n_try), n_try))
        inits.append(random_binary(n_try, rng))
        inits.append(project_feasible(make_sine(n_try, 3.0, rng.rand() * 6.28, 0.35), n_try))
        inits.append(project_feasible(make_sine(n_try, 5.0, rng.rand() * 6.28, 0.35), n_try))
        inits.append(project_feasible(make_multi_freq(n_try, [2, 5, 9],
                    [0.15, 0.12, 0.08], [0.3, 1.8, 4.2], rng), n_try))

        for h0 in inits:
            if _remaining() < 15:
                break
            h0 = project_feasible(h0, n_try)
            c5_cur = compute_c5(h0)
            if c5_cur > best_c5 * 1.05 + 0.01:
                continue

            for beta in [80, 200, 500]:
                if _remaining() < 10:
                    break
                h_new = run_lbfgs(h0, n_try, beta, max_iter=200, max_fun=2000)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur:
                    h0 = h_new
                    c5_cur = c5_new

            # Binary local search at this resolution
            if _remaining() > 20 and c5_cur < best_c5 + 0.01:
                h_bin = np.where(h0 > 0.5, 1.0, 0.0)
                h_bin = project_feasible(h_bin, n_try)
                h_ls, c5_ls = binary_local_search(h_bin, n_try,
                                                   max_iters=30000, rng=rng, patience=3000)
                if c5_ls < c5_cur:
                    c5_cur = c5_ls
                    h0 = project_feasible(h_ls, n_try)

            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h0.copy()
                best_n = n_try
                print("  n={}: C5={:.7f}".format(n_try, c5_cur))

    # Phase 4: Active-set gradient refinement
    if _remaining() > 20:
        n = best_n
        print("\n=== Phase 4: Active-set refinement n={} ===".format(n))
        h = best_h.copy()

        for outer in range(50):
            if _remaining() < 4:
                break

            grad = active_set_gradient(h, n, top_k=6, beta_act=50.0)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-13:
                break

            unit_grad = grad / gnorm
            cur_c5 = compute_c5(h)
            improved = False
            for step in [0.04, 0.08, 0.15, 0.02, 0.1, 0.2, 0.06, 0.3]:
                h_trial = h - step * unit_grad
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5:
                    h = h_trial.copy()
                    cur_c5 = c5_trial
                    improved = True
                    if c5_trial < best_c5:
                        best_c5 = c5_trial
                        best_h = h.copy()
                        print("  Iter {}: C5={:.7f}".format(outer, c5_trial))
                    break

            if not improved:
                # Try L-BFGS nudge
                if _remaining() > 8:
                    h_new = run_lbfgs(h, n, 400, max_iter=100, max_fun=800)
                    c5_new = compute_c5(h_new)
                    if c5_new < cur_c5:
                        h = h_new.copy()
                        cur_c5 = c5_new
                    else:
                        break

    # Phase 5: Final multi-beta polish
    if _remaining() > 10:
        n = best_n
        print("\n=== Phase 5: Final polish ===".format(n))
        for beta in [600, 1200, 2500]:
            if _remaining() < 5:
                break
            h_new = run_lbfgs(best_h, n, beta, max_iter=400, max_fun=3000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  beta={}: C5={:.7f}".format(beta, c5_new))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
```
