def binary_sa(h_init, n, iters, rng, t_start=0.004, t_end=0.00002):
    h = np.where(h_init > 0.5, 1.0, 0.0).copy()
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    target = n // 2
    if len(ones) > target:
        rem = rng.choice(len(ones), len(ones) - target, replace=False)
        for idx in rem:
            h[ones[idx]] = 0.0
            zeros.append(ones[idx])
            ones[idx] = -1
        ones = [x for x in ones if x >= 0]
    elif len(ones) < target:
        add = rng.choice(len(zeros), target - len(ones), replace=False)
        for idx in add:
            h[zeros[idx]] = 1.0
            ones.append(zeros[idx])
            zeros[idx] = -1
        zeros = [x for x in zeros if x >= 0]

    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5

    for it in range(iters):
        if remaining_time() < 3:
            break
        if not ones or not zeros:
            break
        oi = ones[rng.randint(len(ones))]
        oz = zeros[rng.randint(len(zeros))]
        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = compute_c5(h)
        T = t_start * (t_end / t_start) ** (float(it) / max(float(iters), 1))
        if new_c5 < cur_c5 or (T > 1e-15 and rng.rand() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-15))):
            cur_c5 = new_c5
            ones[ones.index(oi)] = oz
            zeros[zeros.index(oz)] = oi
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0
    return best_h, best_c5

    if len(ones) > target:
        rem = rng.choice(len(ones), len(ones) - target, replace=False)
        for idx in rem:
            h[ones[idx]] = 0.0
            zeros.append(ones[idx])
            ones[idx] = -1
        ones = [x for x in ones if x >= 0]
    elif len(ones) < target:
        add = rng.choice(len(zeros), target - len(ones), replace=False)
        for idx in add:
            h[zeros[idx]] = 1.0
            ones.append(zeros[idx])
            zeros[idx] = -1
        zeros = [x for x in zeros if x >= 0]

    best_c5 = compute_c5(h)
    best_h = h.copy()

    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        order_o = rng.permutation(len(ones))
        for ii in order_o:
            oi = ones[ii]
            order_z = rng.permutation(len(zeros))
            for jj in order_z:
                oz = zeros[jj]
                h[oi] = 0.0
                h[oz] = 1.0
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h.copy()
                    ones[ii] = oz
                    zeros[jj] = oi
                    improved = True
                else:
                    h[oi] = 1.0
                    h[oz] = 0.0
        if not improved:
            break
    return best_h, best_c5


def coordinate_descent(h0, n, n_sweeps=10, rng=None):
    h = h0.copy()
    for sweep in range(n_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        order = rng.permutation(n) if rng is not None else np.arange(n)
        for i in order:
            if remaining_time() < 3:
                break
            old = h[i]
            cur_c5 = compute_c5(h)
            best_val = old
            best_c = cur_c5
            cands = [0.0, 1.0, 0.5,
                     max(0.0, old + 0.15), min(1.0, old - 0.15),
                     max(0.0, old + 0.4), min(1.0, old - 0.4)]
            for cv in cands:
                if abs(cv - old) < 1e-9:
                    continue
                h[i] = cv
                hf = project_feasible(h, n)
                c = compute_c5(hf)
                if c < best_c:
                    best_c = c
                    best_val = cv
                    h = hf.copy()
            if best_c < cur_c5 - 1e-12:
                improved = True
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S, _G_N, _G_BETA, _G_TARGET, _G_TOP_IDX
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
    except Exception:
        parent_h = None
        parent_n = 160
        parent_c5 = float('inf')

    if parent_h is not None:
        best_h = parent_h.copy()
        best_n = parent_n
        best_c5 = parent_c5
    else:
        best_n = 160
        best_h = project_feasible(make_thue_morse(best_n), best_n)
        best_c5 = compute_c5(best_h)

    print("Initial: n={}, C5={:.7f}".format(best_n, best_c5))

    # Phase 1: L-BFGS at parent resolution with increasing beta
    n = best_n
    target = n / 2.0
    h = project_feasible(best_h, n)
    for beta in [50, 100, 200, 400, 800, 1500]:
        if remaining_time() < 12:
            break
        h = lbfgs_smooth_max(h, n, beta, target, maxiter=2000)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            print("  n={}, beta={}: C5={:.7f}".format(n, beta, best_c5))

    # Phase 2: Top-k iterative optimization at parent resolution
    if remaining_time() > 30:
        h = topk_iterate(best_h, best_n, best_n / 2.0, n_rounds=8, k=5, maxiter=600)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            print("  TopK iter: C5={:.7f}".format(best_c5))

    # Phase 3: Multi-resolution optimization
    res_list = [200, 240, 320]
    for n_r in res_list:
        if remaining_time() < 60:
            break
        target_r = n_r / 2.0
        inits = [project_feasible(interp_to(best_h, best_n, n_r), n_r)]
        for _ in range(2):
            inits.append(project_feasible(make_random_binary(n_r, rng), n_r))
        inits.append(project_feasible(make_thue_morse(n_r), n_r))
        inits.append(project_feasible(make_multi_sine(n_r, rng), n_r))

        for h0 in inits:
            if remaining_time() < 30:
                break
            c5_0 = compute_c5(h0)
            if c5_0 > best_c5 * 1.2:
                continue
            for beta in [100, 300, 600]:
                if remaining_time() < 15:
                    break
                h_r = lbfgs_smooth_max(h0, n_r, beta, target_r, maxiter=1500)
                c5_r = compute_c5(h_r)
                if c5_r < c5_0:
                    h0 = h_r
                    c5_0 = c5_r
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h0.copy()
                    best_n = n_r
                    print("  n_r={}, beta={}: C5={:.7f}".format(n_r, beta, best_c5))
            if remaining_time() > 20:
                h_tk = topk_iterate(h0, n_r, target_r, n_rounds=4, k=5, maxiter=500)
                c5_tk = compute_c5(h_tk)
                if c5_tk < best_c5:
                    best_c5 = c5_tk
                    best_h = h_tk.copy()
                    best_n = n_r
                    print("  n_r={} topK: C5={:.7f}".format(n_r, best_c5))

    # Phase 4: Binary local search + SA
    if remaining_time() > 50:
        n_sa = best_n
        h_bin = np.where(best_h > 0.5, 1.0, 0.0)
        h_bin = project_feasible(h_bin, n_sa)
        h_ls, c5_ls = binary_local_search(h_bin, n_sa, max_sweeps=15, rng=rng)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            print("  BinLS: C5={:.7f}".format(best_c5))

        for trial in range(3):
            if remaining_time() < 30:
                break
            if trial == 0:
                h_start = h_ls
            else:
                h_start = project_feasible(make_random_binary(n_sa, rng), n_sa)
            h_s, c5_s = binary_sa(h_start, n_sa, 4000, rng)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
                print("  BinSA: C5={:.7f}".format(best_c5))
            h_ls2, c5_ls2 = binary_local_search(h_s, n_sa, max_sweeps=10, rng=rng)
            if c5_ls2 < best_c5:
                best_c5 = c5_ls2
                best_h = h_ls2.copy()
                print("  BinSA+LS: C5={:.7f}".format(best_c5))

        # Polish binary with continuous
        if remaining_time() > 25:
            for beta in [200, 500, 1000]:
                if remaining_time() < 10:
                    break
                h_p = lbfgs_smooth_max(best_h, n_sa, beta, n_sa / 2.0, maxiter=1500)
                c5_p = compute_c5(h_p)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p.copy()
                    print("  Polish beta={}: C5={:.7f}".format(beta, best_c5))

    # Phase 5: Coordinate descent on best solution
    if remaining_time() > 25:
        h_cd = coordinate_descent(best_h, best_n, n_sweeps=8, rng=rng)
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            print("  CoordDesc: C5={:.7f}".format(best_c5))

    # Phase 6: Final high-beta polish with top-k alternating
    if remaining_time() > 20:
        n_f = best_n
        target_f = n_f / 2.0
        h_f = best_h.copy()
        for beta in [2000, 5000, 10000]:
            if remaining_time() < 8:
                break
            h_f = lbfgs_smooth_max(h_f, n_f, beta, target_f, maxiter=3000)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()
                print("  Final beta={}: C5={:.7f}".format(beta, best_c5))
        if remaining_time() > 15:
            h_tk = topk_iterate(h_f, n_f, target_f, n_rounds=6, k=4, maxiter=1000)
            c5_tk = compute_c5(h_tk)
            if c5_tk < best_c5:
                best_c5 = c5_tk
                best_h = h_tk.copy()
                print("  Final TopK: C5={:.7f}".format(best_c5))

    # Phase 7: Final coordinate + polish
    if remaining_time() > 10:
        h_cd2 = coordinate_descent(best_h, best_n, n_sweeps=5, rng=rng)
        c5_cd2 = compute_c5(h_cd2)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
        if remaining_time() > 5:
            h_p2 = lbfgs_smooth_max(best_h, best_n, 5000, best_n / 2.0, maxiter=2000)
            c5_p2 = compute_c5(h_p2)
            if c5_p2 < best_c5:
                best_c5 = c5_p2
                best_h = h_p2.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
