import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def binary_init_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') & 1
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        idx = np.where(h == 1)[0]
        for k in range(n_ones - n // 2):
            h[idx[k]] = 0.0
    elif n_ones < n // 2:
        idx = np.where(h == 0)[0]
        for k in range(n // 2 - n_ones):
            h[idx[k]] = 1.0
    return h


def fix_binary_count(h, n, rng):
    h = h.copy()
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        idx = np.where(h == 1)[0]
        if len(idx) > 0:
            to_remove = rng.choice(idx, min(n_ones - n // 2, len(idx)), replace=False)
            h[to_remove] = 0.0
    elif n_ones < n // 2:
        idx = np.where(h == 0)[0]
        if len(idx) > 0:
            to_add = rng.choice(idx, min(n // 2 - n_ones, len(idx)), replace=False)
            h[to_add] = 1.0
    return h


def corr_delta_one_swap(h, i, j, n):
    """Compute delta corr when h[i]: 1->0 and h[j]: 0->1. Returns (delta_corr, new_max_estimate)."""
    h[i] = 0.0
    h[j] = 1.0
    new_corr = np.correlate(h, 1.0 - h, mode='full')
    h[i] = 1.0
    h[j] = 0.0
    return new_corr


def corr_delta_two_swap(h, i1, i2, j1, j2, n):
    """Compute corr after 2-swap: h[i1]=h[i2]=0, h[j1]=h[j2]=1."""
    h[i1] = 0.0
    h[i2] = 0.0
    h[j1] = 1.0
    h[j2] = 1.0
    new_corr = np.correlate(h, 1.0 - h, mode='full')
    h[i1] = 1.0
    h[i2] = 1.0
    h[j1] = 0.0
    h[j2] = 0.0
    return new_corr


def targeted_local_search(h, n, time_limit, rng):
    """Aggressive local search with 1-swap and 2-swap targeting worst shifts."""
    h = (h > 0.5).astype(float).copy()
    h = fix_binary_count(h, n, rng)
    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx_val
    start = time.time()
    stall = 0
    total_perturb = 0

    while time.time() - start < time_limit and remaining() > 5:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Get top 3 worst shifts
        top3 = np.argsort(-corr)[:3]
        k_max = int(top3[0])
        offset = k_max - (n - 1)

        # Identify contributors to worst shift
        if offset >= 0 and offset < n:
            contrib = np.array([i for i in ones if i + offset < n and h[i + offset] == 0])
        elif offset < 0 and -offset < n:
            ao = -offset
            contrib = np.array([i for i in ones if i + ao < n and h[i + ao] == 0])
        else:
            contrib = ones

        found = False
        n_try_1 = 300
        n_try_2 = 150

        # Phase A: 1-swaps biased toward worst-shift contributors
        for _ in range(n_try_1):
            if time.time() - start > time_limit:
                break
            if len(contrib) > 0 and rng.rand() < 0.75:
                i = contrib[rng.randint(len(contrib))]
            else:
                i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            new_corr = corr_delta_one_swap(h, i, j, n)
            new_max = float(np.max(new_corr))
            if new_max < cur_max - 1e-12:
                corr = new_corr
                cur_max = new_max
                best_c5 = cur_max * dx_val
                found = True
                break

        # Phase B: 2-swaps for more powerful moves
        if not found:
            for _ in range(n_try_2):
                if time.time() - start > time_limit:
                    break
                # Bias toward contributors
                if len(contrib) >= 2 and rng.rand() < 0.5:
                    i1, i2 = rng.choice(contrib, 2, replace=False)
                else:
                    i1, i2 = rng.choice(ones, 2, replace=False)
                j1, j2 = rng.choice(zeros, 2, replace=False)
                new_corr = corr_delta_two_swap(h, i1, i2, j1, j2, n)
                new_max = float(np.max(new_corr))

                    corr = new_corr
                    cur_max = new_max
                    best_c5 = cur_max * dx_val
                    # Also try which 2-swap configuration is better
                    # Try the other pairing

                    nm2 = float(np.max(new_corr2))
                    if nm2 < new_max:
                        corr = new_corr2
                        cur_max = nm2
                        best_c5 = cur_max * dx_val
                    found = True
                    break

        if found:
            stall = 0
        else:
            stall += 1
            if stall > 40:
                # Perturbation: random 3-swap for diversity
                if len(ones) >= 3 and len(zeros) >= 3:
                    i1, i2, i3 = rng.choice(ones, 3, replace=False)
                    j1, j2, j3 = rng.choice(zeros, 3, replace=False)
                    h[i1] = 0.0

                cur_max = float(np.max(corr))
                stall = 0
                total_perturb += 1
                if total_perturb > 500:
                    break

    return h, best_c5


def smooth_obj(h, n, beta):
    dx_val = 2.0 / n
    corr = compute_corr(h)
    m = float(np.max(corr))
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def smooth_grad_full(h, n, beta):
    dx_val = 2.0 / n
    corr = compute_corr(h)
    m = float(np.max(corr))
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk > 1e-13:
            off = k - (n - 1)
            if off >= 0:
                npairs = n - off
                g[off:off + npairs] += wk * (1.0 - h[:npairs])
                g[:npairs] -= wk * h[off:off + npairs]
            else:
                ao = -off
                npairs = n - ao
                g[:npairs] += wk * (1.0 - h[ao:ao + npairs])
                g[ao:ao + npairs] -= wk * h[:npairs]
    return g * dx_val


def slsqp_refine(h, n, beta, maxiter=250):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0

    def obj_fn(x):
        return smooth_obj(x, n, beta)

    def grad_fn(x):
        return smooth_grad_full(x, n, beta)

    def con_fun(x):
        return np.sum(x) - target

    def con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def multi_start_search(n, n_starts, time_limit, rng):
    """Run local search from multiple starts, return best."""
    best_h = None
    best_c5 = float('inf')
    per_start = max(5, time_limit // n_starts)

    for s in range(n_starts):
        if remaining() < 5:
            break
        # Diversified initialization
        h_init = binary_init_thue_morse(n)
        # Perturb with random swaps
        n_perturb = n // 8
        for _ in range(n_perturb):
            ones = np.where(h_init == 1)[0]
            zeros = np.where(h_init == 0)[0]
            if len(ones) > 0 and len(zeros) > 0:
                a = ones[rng.randint(len(ones))]
                b = zeros[rng.randint(len(zeros))]
                h_init[a], h_init[b] = h_init[b], h_init[a]

        tl = min(per_start, max(3, remaining() * 0.8 / (n_starts - s)))
        h_out, c5_out = targeted_local_search(h_init, n, tl, rng)
        if c5_out < best_c5:
            best_c5 = c5_out
            best_h = h_out.copy()

    return best_h, best_c5
