        h_blend2 = 0.5 * h_init + 0.5 * blk
        inits.append(project_feasible(h_blend2, n))
        
        n_inits = len(inits)
        time_per_init = max(30.0, (time_left() * 0.7) / n_inits)
        
        for h0 in inits:
            if time_left() < 30:
                break
            tl = time_per_init
            
            # Phase 1: Fast coordinate descent (bulk of time)
            tl_cd = tl * 0.55
            h_cd, c5_cd = coord_descent_fast(h0, n, sweeps=200, time_limit=tl_cd, t0=t0)
            
            # Phase 2: SLSQP polish
            if time_left() > 15:
                tl_slq = min(30.0, (time_left() - 10) * 0.15)
                h_sp, c5_sp = slsqp_polish(h_cd, n, time_limit=tl_slq, t0=t0)
            else:
                h_sp, c5_sp = h_cd, c5_cd
            
            # Phase 3: More coordinate descent on the polished result
            if time_left() > 20:
                tl_cd2 = min(60.0, (time_left() - 10) * 0.2)
                h_cd2, c5_cd2 = coord_descent_fast(h_sp, n, sweeps=300, time_limit=tl_cd2, t0=t0)
            else:
                h_cd2, c5_cd2 = h_sp, c5_sp
            
            # Phase 4: Pair swaps
            if time_left() > 15:
                h_ps, c5_ps = pair_swap_move(h_cd2, n, n_pairs=200, rng=rng)
                if c5_ps < c5_cd2:
                    h_cd2, c5_cd2 = h_ps, c5_ps
            
            # Take best
            candidates = [(h_cd, c5_cd), (h_sp, c5_sp), (h_cd2, c5_cd2)]
            bh = min(candidates, key=lambda x: x[1])[0]
            bc5 = min(candidates, key=lambda x: x[1])[1]
            
            if bc5 < best_c5:
                best_c5 = bc5
                best_h = bh.copy()
                best_n = n
    
    # Final intensive polish on the best solution
    if time_left() > 40:
        tl_final = time_left() * 0.5
        h_f, c5_f = coord_descent_fast(best_h, best_n, sweeps=500, time_limit=tl_final, t0=t0)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    
    if time_left() > 25:
        tl_slq2 = min(40.0, time_left() * 0.4)
        h_s2, c5_s2 = slsqp_polish(best_h, best_n, time_limit=tl_slq2, t0=t0)
        if c5_s2 < best_c5:
            best_c5 = c5_s2
            best_h = h_s2.copy()
    
    if time_left() > 20:
        h_f2, c5_f2 = coord_descent_fast(best_h, best_n, sweeps=500, time_limit=time_left() * 0.6, t0=t0)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()
    
    if time_left() > 15:
        h_ps2, c5_ps2 = pair_swap_move(best_h, best_n, n_pairs=500, rng=rng)
        if c5_ps2 < best_c5:
            best_c5 = c5_ps2
            best_h = h_ps2.copy()
    
    if time_left() > 10:
        h_f3, c5_f3 = coord_descent_fast(best_h, best_n, sweeps=300, time_limit=time_left() * 0.8, t0=t0)
        if c5_f3 < best_c5:
            best_c5 = c5_f3
            best_h = h_f3.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    # Validate
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
