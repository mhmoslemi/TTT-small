def try_resolution(h_src, n_src, nn):

def scan_range_update(best_h, best_c, best_n, start, end, step, reserve=30.0):
    if end < start:
        return best_h, best_c, best_n
    step = max(1, int(step))

        if cand is not None and c < best_c - 1.0e-15:

            best_c = c

    return best_h, best_c, best_n


def make_sturmian(n, alpha, phase):

def make_multisine(n, rng):
    n = int(n)
    t = np.arange(n, dtype=float) / float(n)
    h = 0.5 * np.ones(n, dtype=float)
    freqs = [1, 2, 3, 5, 8, 13]
    for f in freqs:
        if f >= max(2, n // 2):
            continue
        amp = float(rng.uniform(0.04, 0.22))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        h += amp * np.sin(2.0 * np.pi * f * t + ph)
    h = 0.5 + 0.5 * np.tanh(1.8 * (h - 0.5))
    return project_box_sum(h, n)


def make_random_balanced(n, rng):

    h = np.zeros(n, dtype=float)
    if n % 2 == 0:

        h[idx] = 1.0
        zero = np.where(h == 0.0)[0]
        if len(zero) > 0:
            h[zero[0]] = 0.5
    return project_box_sum(h, n)


def make_thue_morse(n):

def make_candidates(n, ref_h, ref_n, rng, count):

    if ref_h is not None:

        cands.append(project_box_sum(nearest_to(ref_h, n), n))

        base = interp_to(ref_h, n)
        pert = base + 0.05 * rng.normal(0.0, 1.0, n)
        cands.append(project_box_sum(pert, n))

        if n % 2 == 0:
            cands.append(make_binary_from(ref_h, n))

    alphas = [0.6180339887, 0.3819660113, 0.569840291, 0.430159709, 0.333333333, 0.754877666]
    use_alpha = 2 if count <= 2 else 5
    for alpha in alphas[:use_alpha]:
        ph = float(rng.uniform(0.0, 2.0))
        cands.append(make_sturmian(n, alpha, ph))

    cands.append(make_multisine(n, rng))
    if count > 3:
        cands.append(make_multisine(n, rng))
    if n % 2 == 0:
        cands.append(make_random_balanced(n, rng))
        if count > 3:
            cands.append(make_thue_morse(n))

    scored = []
    for i, h in enumerate(cands):
        if h is None or len(h) != n:
            continue
        h = project_box_sum(h, n)
        c = compute_c5(h)

        scored.append((c, i))

    chosen = []
    for _, i in scored[:int(count)]:
        chosen.append(cands[i])

    if not chosen:
        h = make_multisine(n, rng)
        chosen.append(h)
    return chosen

        amp = float(params[k])
        ph = float(params[m + k])
        h += amp * np.sin(2.0 * np.pi * freqs[k] * t + ph)
    h = 0.5 + 0.5 * np.tanh(1.6 * (h - 0.5))
    return project_box_sum(h, n)


def fourier_obj(params):
    h = h_from_fourier(np.asarray(params, dtype=float), _FOUR_N)
    return compute_c5(h)


def run_fourier_search(n, seed, max_time_s):
    global _FOUR_N, _FOUR_BOUNDS, _PHASE_END
    if max_time_s <= 12.0:
        return None
    n = int(n)
    _FOUR_N = n
    m = 6
    _FOUR_BOUNDS = ((0.0, 0.35),) * m + ((0.0, 2.0 * np.pi),) * m
    _PHASE_END = time.time() + max_time_s
    try:
        res = differential_evolution(
            fourier_obj,
            _FOUR_BOUNDS,
            seed=int(seed),
            maxiter=12,
            popsize=10,
            tol=1.0e-5,
            polish=False,
            init="latinhypercube",
        )
        h = h_from_fourier(np.asarray(res.x, dtype=float), n)
        return project_box_sum(h, n)
    except Exception:
        return None


def build_binary_lists(b):
    return np.flatnonzero(b > 0.5), np.flatnonzero(b < 0.5)


def binary_candidates(b, n, top_k=8):
    corr = compute_c5_array(b)
    active = np.argsort(corr)[-int(top_k):]
    swaps = []
    seen = set()

    for j in active:
        j = int(j)
        s = j - (n - 1)
        if s > 0:
            arr = np.where((b[:n - s] == 0.0) & (b[s:] == 1.0))[0]
            if len(arr) > 40:
                arr = arr[::max(1, len(arr) // 40)]
            for a in arr:
                p = int(a + s)
                q = int(a)
                if p != q and 0 <= p < n and 0 <= q < n and b[p] == 1.0 and b[q] == 0.0:
                    key = (p, q)
                    if key not in seen:
                        seen.add(key)
                        swaps.append(key)
        elif s < 0:
            sp = -s
            arr = np.where((b[sp:] == 1.0) & (b[:n - sp] == 0.0))[0]
            if len(arr) > 40:
                arr = arr[::max(1, len(arr) // 40)]
            for a in arr:
                p = int(a)
                q = int(a + sp)
                if p != q and 0 <= p < n and 0 <= q < n and b[p] == 1.0 and b[q] == 0.0:
                    key = (p, q)
                    if key not in seen:
                        seen.add(key)
                        swaps.append(key)

    if len(swaps) > 140:
        swaps = swaps[:140]
    return swaps


def binary_local(b, n, rng, reserve=30.0):
    set_state(n)
    cur = project_box_sum(b, n)
    best = cur.copy()
    cur_c = compute_c5(cur)
    best_c = cur_c
    ones, zeros = build_binary_lists(cur)
    dx = 2.0 / float(n)
    pert_tol = 3.0 * dx
    stagn = 0

    while work_left(reserve) > float(reserve):
        if len(ones) == 0 or len(zeros) == 0:
            break

        swaps = binary_candidates(cur, n, top_k=8)
        if len(swaps) < 50:
            for _ in range(30):
                p = int(ones[rng.randint(len(ones))])
                q = int(zeros[rng.randint(len(zeros))])
                swaps.append((p, q))

        improved = False
        for p, q in swaps:
            if work_left(reserve) <= float(reserve):
                break
            if len(ones) == 0 or len(zeros) == 0:
                break
            if cur[p] != 1.0 or cur[q] != 0.0:
                continue
            cur[p] = 0.0
            cur[q] = 1.0
            c = compute_c5(cur)
            if c < cur_c - 1.0e-14:
                cur_c = c
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best = cur.copy()
                ones, zeros = build_binary_lists(cur)
                break
            else:
                cur[p] = 1.0
                cur[q] = 0.0

        if improved:
            stagn = 0
            continue

        accepted = False
        for _ in range(300):
            if work_left(reserve) <= float(reserve):
                break
            if len(ones) == 0 or len(zeros) == 0:
                break
            p = int(ones[rng.randint(len(ones))])
            q = int(zeros[rng.randint(len(zeros))])
            cur[p] = 0.0
            cur[q] = 1.0
            c = compute_c5(cur)
            if c < cur_c - 1.0e-14:
                cur_c = c
                accepted = True
                if cur_c < best_c:
                    best_c = cur_c
                    best = cur.copy()
                ones, zeros = build_binary_lists(cur)
                break
            else:
                cur[p] = 1.0
                cur[q] = 0.0

        if accepted:
            stagn = 0
            continue

        perturbed = False
        pert_count = 0
        for _ in range(60):
            if work_left(reserve) <= float(reserve):
                break
            if len(ones) == 0 or len(zeros) == 0:
                break
            p = int(ones[rng.randint(len(ones))])
            q = int(zeros[rng.randint(len(zeros))])
            cur[p] = 0.0
            cur[q] = 1.0
            c = compute_c5(cur)
            if c < cur_c + pert_tol:
                cur_c = c
                perturbed = True
                pert_count += 1
                if cur_c < best_c:
                    best_c = cur_c
                    best = cur.copy()
                ones, zeros = build_binary_lists(cur)
                if pert_count >= 4:
                    break
            else:
                cur[p] = 1.0
                cur[q] = 0.0

        if not perturbed:
            stagn += 1
            if stagn > 3:
                break
        else:
            stagn = 0

    return best, best_c, cur, cur_c


def unique_sorted_by_first(tups):
    if not tups:
        return []
    tups = list(tups)
    tups.sort()
    seen = set()
    out = []
    for _, x in tups:
        if x not in seen:
            seen.add(x)
            out.append(x)
    return out


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S, _PHASE_END

    _START_TIME = time.time()
    _BUDGET_S = float(budget_s)
    rng = np.random.RandomState(int(seed))

    best_h = None
    best_c = float("inf")
    best_n = 160

    try:
        parent = np.array(initial_h_values, dtype=float, copy=True)
        pn = len(parent)
        if pn > 0:
            parent = project_box_sum(parent, pn)
            pc = compute_c5(parent)
            best_h = parent
            best_c = pc
            best_n = pn
    except Exception:
        pass

    if best_h is None:
        if remaining_time() < 30.0:
            best_n = 160
            best_h = make_multisine(best_n, rng)
            best_h = project_box_sum(best_h, best_n)
        else:
            best_n = 160
            cands = make_candidates(best_n, None, None, rng, 3)
            best_h = cands[0]
        best_c = compute_c5(best_h)

    if best_h is None:
        best_n = 160
        best_h = make_multisine(best_n, rng)
        best_c = compute_c5(best_h)

    if remaining_time() > 150:
        low = max(80, best_n - 90)
        high = min(640, best_n + 180)
        best_h, best_c, best_n = scan_range_update(best_h, best_c, best_n, low, high, 2, reserve=100.0)

    opt_base = int(best_n) if 80 <= best_n <= 320 else 240

    # Fourier global-ish search at a representative resolution.
    if remaining_time() > 300:
        _PHASE_END = time.time() + min(80.0, remaining_time() - 120.0)
        h4 = run_fourier_search(opt_base, rng.randint(1, 1000000), 50.0)
        if h4 is not None:
            c4 = compute_c5(h4)
            if c4 < best_c - 1.0e-12:
                best_h = h4
                best_c = c4
                best_n = opt_base
            if c4 < best_c + 0.02 and opt_base >= 80 and opt_base <= 320:
                cand, cc = improve_continuous(h4, opt_base, deep=False)
                if cc < best_c - 1.0e-12:
                    best_h = cand
                    best_c = cc
                    best_n = int(opt_base)

    # Deep refinement around the current best resolution.
    if remaining_time() > 160:
        _PHASE_END = time.time() + min(340.0, remaining_time() - 150.0)
        n1 = int(opt_base)
        cands = make_candidates(n1, best_h, best_n, rng, 3)
        for h0 in cands:
            if work_left(40.0) <= 40.0:
                break
            cand, cc = improve_continuous(h0, n1, deep=True)
            if cc < best_c - 1.0e-12:
                best_h = cand
                best_c = cc
                best_n = int(n1)
        if remaining_time() > 220:
            low = max(80, n1 - 50)
            high = min(520, n1 + 90)
            best_h, best_c, best_n = scan_range_update(best_h, best_c, best_n, low, high, 2, reserve=160.0)

    # Explore neighboring resolutions.
    extras = [
        opt_base - 48,
        opt_base - 32,
        opt_base - 16,
        opt_base + 16,
        opt_base + 32,
        opt_base + 48,
        opt_base + 64,
        144,
        160,
        184,
        208,
        232,
        256,
        288,
    ]
    tups = []
    for nn in extras:
        nn = int(nn)
        if 80 <= nn <= 320:
            tups.append((abs(nn - opt_base), nn))
    n_order = unique_sorted_by_first(tups)

    if remaining_time() > 240:
        _PHASE_END = time.time() + min(420.0, remaining_time() - 220.0)
        for nn in n_order:
            nn = int(nn)
            if work_left(45.0) <= 45.0:
                break
            if nn == opt_base:
                continue
            cands = make_candidates(nn, best_h, best_n, rng, 2)
            if not cands:
                continue
            h0 = cands[0]
            cand, cc = improve_continuous(h0, nn, deep=False)
            if cc < best_c - 1.0e-12:
                best_h = cand
                best_c = cc
                best_n = nn
            if remaining_time() < 280:
                break
            if remaining_time() > 260:
                low = max(80, nn - 35)
                high = min(480, nn + 55)
                best_h, best_c, best_n = scan_range_update(best_h, best_c, best_n, low, high, 2, reserve=220.0)

    # Binary local search from continuous best and random restarts.
    if remaining_time() > 180:
        _PHASE_END = time.time() + min(200.0, remaining_time() - 90.0)

        bin_ns = []
        if 80 <= best_n <= 320 and best_n % 2 == 0:
            bin_ns.append(int(best_n))
        if 80 <= opt_base <= 320 and opt_base % 2 == 0:
            bin_ns.append(int(opt_base))
        if (opt_base + 16) % 2 == 0 and 80 <= opt_base + 16 <= 320:
            bin_ns.append(int(opt_base + 16))
        if (opt_base - 16) % 2 == 0 and 80 <= opt_base - 16 <= 320:
            bin_ns.append(int(opt_base - 16))
        bin_ns = unique_sorted_by_first([(0, x) for x in bin_ns])

        for nn in bin_ns:
            if work_left(35.0) <= 35.0:
                break
            nn = int(nn)
            if nn % 2 != 0:
                continue

            b = make_binary_from(best_h, nn)
            res = binary_local(b, nn, rng, reserve=25.0)
            if len(res) == 4:
                b2, b2_c, cur_b, cur_c = res
            else:
                b2, b2_c, cur_b, cur_c = res[0], res[1], res[0], res[1]

            if b2_c < best_c - 1.0e-12:
                best_h = b2
                best_c = b2_c
                best_n = nn

            for h0, c0 in ((b2, b2_c), (cur_b, cur_c)):
                if work_left(25.0) <= 25.0:
                    break
                if c0 < best_c + 0.025:
                    cand, cc = improve_continuous(h0, nn, deep=False)
                    if cc < best_c - 1.0e-12:
                        best_h = cand
                        best_c = cc
                        best_n = nn

            if work_left(35.0) > 35.0:
                br = make_random_balanced(nn, rng)
                res = binary_local(br, nn, rng, reserve=20.0)
                if len(res) == 4:
                    b2, b2_c, cur_b, cur_c = res
                else:
                    b2, b2_c, cur_b, cur_c = res[0], res[1], res[0], res[1]
                if b2_c < best_c - 1.0e-12:
                    best_h = b2
                    best_c = b2_c
                    best_n = nn
                if work_left(20.0) > 20.0 and b2_c < best_c + 0.025:
                    cand, cc = improve_continuous(b2, nn, deep=False)
                    if cc < best_c - 1.0e-12:
                        best_h = cand
                        best_c = cc
                        best_n = nn

    # Final multi-resolution scanning.
    if remaining_time() > 80:
        centers = [int(best_n), int(opt_base), 160]
        for ctr in centers:
            if remaining_time() < 40:
                break
            low = max(80, ctr - 50)
            high = min(640, ctr + 80)
            best_h, best_c, best_n = scan_range_update(best_h, best_c, best_n, low, high, 1, reserve=40.0)

        if remaining_time() > 70:
            best_h, best_c, best_n = scan_range_update(best_h, best_c, best_n, 80, 320, 5, reserve=50.0)
        if remaining_time() > 55:
            best_h, best_c, best_n = scan_range_update(best_h, best_c, best_n, 322, 640, 10, reserve=40.0)
        if remaining_time() > 35:
            low = max(80, best_n - 20)
            high = min(640, best_n + 25)
            best_h, best_c, best_n = scan_range_update(best_h, best_c, best_n, low, high, 1, reserve=25.0)

    # Final polish at the selected resolution.
    if remaining_time() > 45 and best_h is not None:
        if best_n <= 320:
            _PHASE_END = time.time() + min(70.0, remaining_time() - 12.0)
            cand, cc = improve_continuous(best_h, best_n, deep=False)
            if cc < best_c - 1.0e-12:
                best_h = cand
                best_c = cc
                best_n = int(best_n)
        else:
            if best_n <= 512:
                _PHASE_END = time.time() + min(45.0, remaining_time() - 12.0)
                cand, cc = coordinate_polish(best_h, best_n, max_sweeps=1)
                if cc < best_c - 1.0e-12:
                    best_h = cand
                    best_c = cc
                    best_n = int(best_n)

    if remaining_time() > 25 and best_h is not None and best_n <= 512:
        _PHASE_END = time.time() + max(2.0, remaining_time() - 8.0)
        cand, cc = subgradient_polish(best_h, best_n, max_iters=30)
        if cc < best_c - 1.0e-12:
            best_h = cand
            best_c = cc
            best_n = int(best_n)

    # Final feasibility normalization.
    best_h = project_box_sum(best_h, best_n)
    best_n = int(best_n)

    for _ in range(8):
        diff = float(best_n) / 2.0 - float(np.sum(best_h))
        if abs(diff) < 1.0e-14:
            break
        free = (best_h > 1.0e-14) & (best_h < 1.0 - 1.0e-14)
        if np.sum(free) == 0:
            break
        best_h[free] += diff / float(np.sum(free))
        np.clip(best_h, 0.0, 1.0, out=best_h)

    final_c = compute_c5(best_h)
    try:
        ev = evaluate_erdos_solution(best_h)
        if isinstance(ev, (float, int, np.floating, np.integer)):
            ev = float(ev)
            if ev < final_c:
                final_c = ev
    except Exception:
        pass

    return (best_h, float(final_c), best_n)
