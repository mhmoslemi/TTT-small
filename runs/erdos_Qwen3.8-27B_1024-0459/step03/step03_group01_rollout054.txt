    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 320
        parent_h = np.ones(parent_n)
        parent_h[parent_n // 2:] = 0.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    # Phase 1: Binary SA on parent
    h_bin = to_binary(best_h, best_n, rng)
    h_sa, c5_sa = simulated_annealing_binary(h_bin, best_n, rng, max_iter=50000, T_start=2.0, T_end=0.05)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()
    
    # Greedy local search
    h_g, c5_g = greedy_local_search(h_sa, best_n, rng, max_iter=15000)
    if c5_g < best_c5:
        best_c5 = c5_g
        best_h = h_g.copy()
    
    # Multistart SA
    h_ms, c5_ms = multistart_sa(h_g, best_n, rng, num_restarts=3, iters_per_restart=20000)
    if c5_ms < best_c5:
        best_c5 = c5_ms
        best_h = h_ms.copy()
        best_n = best_n
    
    # Phase 2: Try other resolutions
    resolutions = sorted(set([
        best_n - 40, best_n - 20, best_n, best_n + 20, best_n + 40,
        best_n + 60, best_n + 80, best_n + 100, best_n + 120,
        int(best_n * 1.25), int(best_n * 1.5), int(best_n * 1.75), int(best_n * 2.0)
    ]))
    resolutions = [r for r in resolutions if 120 <= r <= 800 and r % 2 == 0]
    
    for n_t in resolutions:
        if remaining() < 30:
            break
        if n_t == best_n:
            continue
        
        # Interpolate best solution to new resolution
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        h_bin_t = to_binary(h_feas, n_t, rng)
        
        # SA at this resolution
        h_sa_t, c5_sa_t = simulated_annealing_binary(h_bin_t, n_t, rng, max_iter=30000, T_start=1.5, T_end=0.05)
        
        # Greedy
        h_g_t, c5_g_t = greedy_local_search(h_sa_t, n_t, rng, max_iter=8000)
        
        c5_cur = c5_g_t
        h_cur = h_g_t
        if c5_sa_t < c5_cur:
            c5_cur = c5_sa_t
