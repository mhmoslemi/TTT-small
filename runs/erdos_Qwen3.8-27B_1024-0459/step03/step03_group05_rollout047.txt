def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            break
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _smooth_max_grad(h, beta, n, dx, top_k=None):
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k is not None and top_k < len(corr):
        idx = np.argsort(corr)[-top_k:]
        cw = corr[idx]
        mw = cw.max()
        sw = cw - mw
        w = np.exp(beta * sw)
        w /= np.sum(w)
        work_idx = idx

        mw = corr.max()
