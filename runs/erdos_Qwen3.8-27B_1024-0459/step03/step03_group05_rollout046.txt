∂C_k/∂h_i = (1 - h_{i+k}) - h_{i-k} (for valid indices)

This is O(n * K) to compute all sensitivities, then O(n^2 * K) to evaluate all moves. For n=160, K=10: 160*10 + 25600*10 = 257,600 operations. Very fast!

1. Compute correlations, identify top K lags
2. Compute ∂C_k/∂h_i for all i and top K lags
3. For each pair (i,j), compute the predicted change in max C_k
4. Execute the best move
5. Repeat

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_corr(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k is not None and top_k < len(corr):
        idx = np.argsort(corr)[-top_k:]
        m = corr[idx].max()
        shifted = corr[idx] - m
        sse = np.log(np.sum(np.exp(beta * shifted))) / beta
        return (m + sse) * dx
    else:
        m = np.max(corr)
        shifted = corr - m
        sse = np.log(np.sum(np.exp(beta * shifted))) / beta
        return (m + sse) * dx


def compute_sensitivities(h, top_lags, n):
    K = len(top_lags)
    sens = np.zeros((K, n))
    for idx, m in enumerate(top_lags):
        k = m - (n - 1)

            if k > 0 and n - k > 0:
                sens[idx, k:n] -= h[0:n-k]
            elif k == 0:
                sens[idx, :] -= h[:]
        else:
            kp = -k
            if n - kp > 0:
                sens[idx, kp:n] += (1.0 - h[0:n-kp])

def gradient_smooth_max(h, beta, top_k=20):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k < len(corr):
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
    else:
        idx = np.arange(len(corr))
        c = corr
    m = c.max()
    shifted = c - m
    w = np.exp(beta * shifted)
    sw = w.sum()
    if sw < 1e-30:
        w = np.ones(len(c)) / len(c)
    else:
        w /= sw
    top_lags = idx
    sens = compute_sensitivities(h, top_lags, n)
    grad = w @ sens * dx
    return grad, m * dx

                        best_c5 = c5_t
                        best_h = h.copy()
                        h = best_h.copy()
                        rand_improved = True
                        improved = True
                    else:
                        h[i] -= d
                        h[j] += d
                if not rand_improved and not improved:
                    if sweep > 60:
                        break

            if sweep % 50 == 0 and sweep > 0:
                print("  Coord sv={}: C5={:.8f}".format(sweep, best_c5))
        print("  PhaseC done: C5={:.8f}".format(best_c5))

    # Phase D: Single-coordinate fine tuning
    if remaining() > 40:
        n = best_n
        target = n / 2.0
        h = best_h.copy()

        for sweep in range(200):
            if remaining() < 4:
                break
            improved_this_sweep = False
            order = rng.permutation(n)
            for i in order:
                if remaining() < 3:
                    break
                for d in [0.002, 0.001, 0.005, 0.0005]:
                    if h[i] + d > 1.0:
                        continue
                    # Find compensating j
                    j = rng.randint(n)
                    if j == i:
                        j = (j + 1) % n
                    if h[j] - d < 0:
                        continue
                    h[i] += d
                    h[j] -= d
                    c5_t = compute_c5(h)
                    if c5_t < best_c5 - 1e-14:
                        best_c5 = c5_t
                        best_h = h.copy()
                        improved_this_sweep = True
                        break
                    else:
                        h[i] -= d
                        h[j] += d
            if not improved_this_sweep:
                if sweep > 30:
                    break
                h = best_h.copy()

    # Phase E: Try interpolation to slightly different resolutions for final polish
    if remaining() > 40:
        for n_try in [best_n + 8, best_n - 8, best_n + 16]:
            if remaining() < 20:
                break
            if n_try % 2 != 0 or n_try < 40 or n_try > 800:
                continue
            target = n_try / 2.0
            xp = (np.arange(best_n) + 0.5) * 2.0 / best_n
            xn = (np.arange(n_try) + 0.5) * 2.0 / n_try
            h_try = np.interp(xn, xp, best_h)
            h_try = project_feasible(h_try, target)

            for beta in [1000, 3000]:
                if remaining() < 10:
                    break

                def obj_polish(params, _n=n_try, _target=target, _beta=beta):
                    h = np.clip(params, 0.0, 1.0)
                    se = np.sum(h) - _target
                    pen = 3000.0 * se * se
                    return smooth_max_corr(h, _beta, top_k=20) + pen

                try:
                    res = minimize(obj_polish, h_try, method='L-BFGS-B',
                                   bounds=[(0.0, 1.0)] * n_try,
                                   options={'maxiter': 800, 'ftol': 1e-22, 'gtol': 1e-18,
                                            'maxls': 80, 'maxfun': 8000})
                    h_new = project_feasible(res.x, target)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        best_n = n_try
                        print("  Polish n={}, b={}: C5={:.8f}".format(n_try, beta, c5_new))
                except Exception:
                    pass

    # Final: verify and return
    best_h = project_feasible(best_h, best_n / 2.0)
    # Ensure sum is exactly n/2
    diff = best_n / 2.0 - np.sum(best_h)
    if abs(diff) > 1e-12:
        # Distribute adjustment to interior points
        interior = (best_h > 0.01) & (best_h < 0.99)
        if np.any(interior):
            best_h[interior] += diff / np.sum(interior)
        best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, best_n / 2.0)

    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
