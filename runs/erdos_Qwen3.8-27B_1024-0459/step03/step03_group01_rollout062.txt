```python
def nesterov_smooth(h, n, beta, lr=0.01, steps=5000):
    global _N, _BETA
    _N = n
    _BETA = beta
    h = h.copy()
    x_prev = h.copy()
    t_k = 1.0
    # Estimate Lipschitz constant (rough)
    g0 = smooth_grad(h)
    L = max(np.linalg.norm(g0) / max(lr, 1e-6), 1.0)
    
    for k in range(steps):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        mu = (t_k - 1.0) / t_new
        y = h + mu * (h - x_prev)
        y = np.clip(y, 0.0, 1.0)
        g = smooth_grad(y)
        x_new = y - (1.0 / L) * g
        x_new = project_feasible(x_new, n)
        x_prev = h
        h = x_new
        t_k = t_new
```

```python
import time
```

```python
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)

    return (m + np.log(np.sum(ex)) / beta) * dx_val

    g = np.zeros(n)

            g[:npairs] -= x[offset:offset + npairs]

        npairs = n - ao
        if npairs > 0:
            g[:npairs] += omh[ao:ao + npairs]
            g[ao:ao + npairs] -= x[:npairs]
    return g

            i1, i2 = rng.choice(ones, 2, replace=False)
            j1, j2 = rng.choice(zeros, 2, replace=False)
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
            nc = np.correlate(h, 1.0 - h, mode='full')
            nm = float(np.max(nc))
            d = nm - cur_max
            if d < -1e-15 or (T > 1e-10 and rng.rand() < np.exp(-max(d, 0.0) / T)):
                corr = nc
                cur_max = nm
                if cur_max * dx_val < best_c5 - 1e-15:
                    best_c5 = cur_max * dx_val
                    best_h = h.copy()
            else:
                h[i1] = 1.0
                h[i2] = 1.0
                h[j1] = 0.0
                h[j2] = 0.0
            if it % 5000 == 0:
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]

    # Final hill-climbing on best
    h = best_h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(corr))
    hc_start = time.time()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(50000):
        if time.time() - hc_start > 8 or remaining() < 5:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        nc = np.correlate(h, 1.0 - h, mode='full')
        nm = float(np.max(nc))
        if nm < cur_max - 1e-15:
            corr = nc
            cur_max = nm
            best_c5 = min(best_c5, cur_max * dx_val)
            best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0

    best_c5 = min(best_c5, compute_c5(best_h))
    return best_h, best_c5


def perturb_restart(h, n, rng, n_restarts, beta):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for r in range(n_restarts):
        if remaining() < 8:
            break
        h_p = h.copy()
        noise = rng.randn(n) * 0.03
        h_p = np.clip(h_p + noise, 0.0, 1.0)
        h_p = project_feasible(h_p, n)
        h_p = slsqp_run(h_p, n, beta, 200)
        c5 = compute_c5(h_p)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_p.copy()
    return best_h, best_c5


def thue_morse_init(n, rng):
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        ones = np.where(h == 1)[0]
        h[rng.choice(ones, n_ones - n // 2, replace=False)] = 0.0
    elif n_ones < n // 2:
        zeros = np.where(h == 0)[0]
        h[rng.choice(zeros, n // 2 - n_ones, replace=False)] = 1.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Phase 0: Load parent
    try:
        ph = np.array(initial_h_values, dtype=float).copy()
        pn = len(ph)
    except Exception:
        pn = 320
        ph = np.zeros(pn)
        ph[:pn // 2] = 1.0

    bh = ph.copy()
    bc5 = compute_c5(bh)
    bn = pn

    # Phase 1: SLSQP refinement at parent n with escalating beta
    h = bh.copy()
    for beta in [800, 3000, 10000]:
        if remaining() < 10:
            break
        h = slsqp_run(h, bn, beta, 350)
        c5 = compute_c5(h)
        if c5 < bc5:
            bc5 = c5
            bh = h.copy()

    # Perturbation restarts at parent n
    if remaining() > 20:
        bh, bc5 = perturb_restart(bh, bn, rng, 4, 8000)

    # Phase 2: Multi-n exploration
    n_candidates = [bn, bn - 2, bn + 2, bn + 4, bn + 6, bn + 10,
                    bn + 20, bn + 40, bn + 60, bn + 100]
    n_candidates = sorted(set([nn for nn in n_candidates if 120 <= nn <= 500 and nn % 2 == 0]))

    for nt in n_candidates:
        if remaining() < 30:
            break

        if nt == bn:
            hi = bh.copy()
        else:
            hi = interp_to(bh, bn, nt)
            hi = project_feasible(hi, nt)

        # SLSQP at this n
        hc = hi.copy()
        for beta in [2000, 8000]:
            if remaining() < 10:
                break
            hc = slsqp_run(hc, nt, beta, 250)
        c5c = compute_c5(hc)
        if c5c < bc5:
            bc5 = c5c
            bh = hc.copy()
            bn = nt

        # SA at this n
        if remaining() > 30:
            tl = min(35, max(8, int(remaining() * 0.18)))
            hb, cb = sa_binary(hc, nt, 250000, 0.4, 0.002, rng, tl)
            if cb < bc5:
                bc5 = cb
                bh = hb.copy()
                bn = nt

            # Refine SA result with SLSQP
            if remaining() > 12:
                hr = project_feasible(hb, nt)
                hr = slsqp_run(hr, nt, 8000, 250)
                c5r = compute_c5(hr)
                if c5r < bc5:
                    bc5 = c5r
                    bh = hr.copy()
                    bn = nt

        # Try Thue-Morse init at this n
        if remaining() > 25 and nt != bn:
            htm = thue_morse_init(nt, rng)
            htm_c = slsqp_run(htm, nt, 5000, 200)
            c5t = compute_c5(htm_c)
            if c5t < bc5:
                bc5 = c5t
                bh = htm_c.copy()
                bn = nt

    # Phase 3: Intensive refinement on best solution
    n = bn
    h = bh.copy()
    for beta in [5000, 15000, 30000]:
        if remaining() < 8:
            break
        h = slsqp_run(h, n, beta, 400)
        c5 = compute_c5(h)
        if c5 < bc5:
            bc5 = c5
            bh = h.copy()

    # Perturbation restarts on best
    if remaining() > 15:
        bh, bc5 = perturb_restart(bh, n, rng, 5, 15000)

    # One more SA round on best
    if remaining() > 15:
        hb, cb = sa_binary(bh, n, 150000, 0.2, 0.001, rng,
                           min(20, max(5, int(remaining() * 0.3))))
        if cb < bc5:
            bc5 = cb
            bh = hb.copy()
            bn = n
            # Quick SLSQP polish
            if remaining() > 8:
                hr = project_feasible(bh, n)
                hr = slsqp_run(hr, n, 15000, 300)
                c5r = compute_c5(hr)
                if c5r < bc5:
                    bc5 = c5r
                    bh = hr.copy()

    # Final polish
    if remaining() > 6:
        h = bh.copy()
        h = slsqp_run(h, n, 25000, 300)
        c5 = compute_c5(h)
        if c5 < bc5:
            bc5 = c5
            bh = h.copy()

    # Ensure feasibility
    bh = project_feasible(bh, bn)
    fc5 = compute_c5(bh)
    if fc5 > bc5 + 1e-10:
        bc5 = fc5

    try:
        ec5 = evaluate_erdos_solution(bh)
        if ec5 < bc5:
            bc5 = ec5
    except Exception:
        pass

    return (bh, bc5, bn)
```
