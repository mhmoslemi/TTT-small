def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def project_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)

            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)

        target_sum = n / 2.0
        sum_offset = current_sum - old_val - target_sum  # sum + t - old - target

        # Vectorized: for each t in grid, compute all corr values
        # Linear part: C_arr + A_arr * t_grid (shape: grid_size x n_full)
        lin_part = C_arr[None, :] + A_arr[None, :] * t_grid[:, None]
        # Quadratic part (d=0): t - t^2 + r0
        quad_part = t_grid - t_grid ** 2 + r0
        # The d=0 index in C_arr/A_arr:
        d0_idx = n - 1
        # Replace the d=0 entry in lin_part with quad_part
        lin_part[:, d0_idx] = quad_part
        # Penalty
        pen_part = lam * (sum_offset + t_grid) ** 2

        # Total objective at each grid point
        obj_vals = np.max(lin_part, axis=1) + pen_part

        best_idx = np.argmin(obj_vals)
        best_t = t_grid[best_idx]
        best_obj = obj_vals[best_idx]

        h[j] = best_t
        current_sum += (best_t - old_val)

    return h


def coord_descent_full(h, n, n_sweeps=50, grid_size=101, lam=5000.0,
                       t0=None, budget=1300.0):
    """Multiple coordinate descent sweeps."""
    best_c5, _ = compute_c5(h)
    h = project_sum(h, n)

    for sweep in range(n_sweeps):
        if t0 is not None and time_left(t0, budget) < 10:
            break
        h = coord_descent_one_sweep(h, n, grid_size, lam)
        h = project_sum(h, n)
        c5_cur, _ = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
    return h, best_c5


def subgradient_step(h, n, top_m=15, beta_smooth=30.0):
    """Compute smoothed subgradient of C5 w.r.t. h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    exps = np.exp(beta_smooth * (corr - mx))
    weights = exps / np.sum(exps)

    grad = np.zeros(n)
    n_full = 2 * n - 1
    for di, d in enumerate(range(-(n - 1), n)):
        k = d + (n - 1)
        if weights[k] < 1e-10:
            continue
        # grad[d] = -h[j+d]*1[valid] + (1-h[j-d])*1[valid]
        if d >= 0:
            npv = n - d
            if npv > 0:
                grad[:npv] += -weights[k] * h[d:d + npv]
                grad[d:d + npv] += weights[k] * (1.0 - h[:npv])
        else:
            a = -d
            npv = n - a
            if npv > 0:
                grad[a:a + npv] += -weights[k] * h[:npv]
                grad[:npv] += weights[k] * (1.0 - h[a:a + npv])

    return grad


def nesterov_optimize(h, n, max_iter=2000, t0=None, budget=1300.0):
    """Nesterov-accelerated projected subgradient."""
    best_c5, _ = compute_c5(h)
    best_h = h.copy()
    y = h.copy()
    v = np.zeros(n)
    beta = 0.9
    momentum = 0.0

    for it in range(max_iter):
        if t0 is not None and time_left(t0, budget) < 10:
            break

        # Nesterov step
        t_k = 2.0 / (it + 2.0)
        y_new = h + (t_k - 1.0) * (h - y) * momentum
        y = y_new

        # Compute subgradient
        g = subgradient_step(y, n, top_m=15, beta_smooth=30.0)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        g /= g_norm

        # Adaptive step size (Barzilai-Borwein style)
        alpha = np.min([1.0 / (it + 1), 0.05])

        # Update
        h_new = h - alpha * g
        h_new = project_sum(h_new, n)

        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

        h = h_new

    return best_h, best_c5
