def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(np.maximum(h, 1.0 - h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_subgrad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    w = np.exp(beta * shifted)
    wsum = np.sum(w)
    if wsum < 1e-30:
        return np.zeros(n)
    w = w / wsum
    grad = np.zeros(n)
    omh = 1.0 - h
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-10:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            length = n - k
            j = np.arange(length)
            grad[j] += wk * omh[j + k]
            grad[j + k] -= wk * h[j]
        else:
            s = -k
            length = n - s
            j = np.arange(length)
            grad[j] += wk * omh[j + s]
            grad[j + s] -= wk * h[j]
    return grad * dx


def nesterov_subgrad_descent(h, target, beta_start, beta_end, alpha0, n_iters, max_time, t0):
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    mu = 0.9
    best = h.copy()
    best_val = compute_c5(best)
    it_per_sec = max(1, n_iters // max(1, int(max_time)))
    for it in range(n_iters):
        if time.time() - t0 > max_time - 5:
            break
        frac = it / max(1, n_iters - 1)
        beta = beta_start + (beta_end - beta_start) * frac
        alpha = alpha0 * (1.0 - 0.7 * frac)
        # Nesterov lookahead
        h_look = h + mu * v
        h_look = project_feasible(h_look, target)
        g = compute_subgrad(h_look, beta)
        v = mu * v - alpha * g
        h_new = h + v
        h_new = project_feasible(h_new, target)
        # Backtracking line search
        c5 = compute_c5(h_new)
        if c5 < best_val:
            best_val = c5
            best = h_new.copy()
        h = h_new
    return h, best, best_val


def systematic_local_search(h, target, max_time, t0, deltas):
    n = len(h)
    h = h.copy()
    best = h.copy()
    best_val = compute_c5(best)
    for sweep in range(2000):
        if time.time() - t0 > max_time - 5:
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                if time.time() - t0 > max_time - 5:
                    break
                hi = h[i]
                hj = h[j]
                for d in deltas:
                    if hi + d > 1.0 + 1e-9 or hj - d < -1e-9:
                        continue
                    h[i] = hi + d
                    h[j] = hj - d
                    cv = compute_c5(h)

                        best = h.copy()
                        improved = True

                    h[i] = hi + d

                    # Try reverse

                    h[i] = hi - d
                    h[j] = hj + d
                    cv = compute_c5(h)

                        improved = True

def simulated_annealing(h, target, t_start, t_end, n_iters, max_time, t0, rng):
    n = len(h)

    best = h.copy()
    best_val = compute_c5(best)

        if time.time() - t0 > max_time - 5:
            break
        frac = it / max(1, n_iters - 1)
        temp = t_start * (t_end / t_start) ** frac
        # Random move: pick 2 indices, transfer
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:

        d = rng.uniform(-0.05, 0.05)
        hi = h[i]
        hj = h[j]
        if hi + d > 1.0 or hj - d < 0.0:

        h[i] += d
        h[j] -= d
        c5_new = compute_c5(h)
        delta = c5_new - best_val
        if delta < 0 or rng.rand() < np.exp(-delta / temp):
            h = project_feasible(h, target)
            if c5_new < best_val:
                best_val = c5_new
                best = h.copy()
        else:
            h[i] = hi
            h[j] = hj
    return h, best, best_val


def binary_relaxation(h, target):
    n = len(h)
    # Push values toward 0 or 1, maintaining sum constraint
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    # Adjust to meet sum constraint
    diff = target - np.sum(h_bin)
    if abs(diff) > 1e-10:
        # Find best edges to adjust
        if diff > 0:
            order = np.argsort(h_bin)[::-1]  # closest to 1 first (will decrease)
            for idx in order:
                rem = diff
                if h_bin[idx] - rem >= 0:
                    h_bin[idx] -= rem
                    break
                else:
                    rem -= h_bin[idx]
                    h_bin[idx] = 0.0
                    diff -= h_bin[idx]
        else:
            order = np.argsort(h_bin)  # closest to 0 first
            for idx in order:
                rem = -diff
                if 1.0 - h_bin[idx] - rem >= 0:
                    h_bin[idx] += rem
                    break
                else:
                    rem -= (1.0 - h_bin[idx])
                    h_bin[idx] = 1.0
                    diff += h_bin[idx]
    return project_feasible(h_bin, target)


def smooth_max_obj(params, target, beta, pw):
    h = np.clip(params, 0.0, 1.0)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    val = (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx
    sum_err = np.sum(h) - target
    val += pw * sum_err * sum_err
    return val

    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
