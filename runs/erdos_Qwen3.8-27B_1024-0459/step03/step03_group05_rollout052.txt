                if remaining() < 2:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d_sign = 1.0 if h[i] < h[j] else -1.0
                for delta in [0.004, 0.002, 0.001, 0.006, 0.008]:
                    di = d_sign * delta
                    dj = -d_sign * delta
                    if h[i] + di < -1e-9 or h[i] + di > 1.0 + 1e-9:
                        continue
                    if h[j] + dj < -1e-9 or h[j] + dj > 1.0 + 1e-9:
                        continue
                    h[i] += di
                    h[j] += dj
                    c5_test = compute_c5(h)
                    if c5_test < best_c5:
                        best_c5 = c5_test
                        best_h = h.copy()
                        improved = True
                        break
                    h[i] -= di
                    h[j] -= dj
            if not improved:
                # Try smaller steps
                n_trials2 = min(2000, int(remaining() * 150))
                for _ in range(n_trials2):
                    if remaining() < 2:
                        break
                    i = rng.randint(n)
                    j = rng.randint(n)
                    if i == j:
                        continue
                    d_sign = 1.0 if h[i] < h[j] else -1.0
                    for delta in [0.0005, 0.001, 0.0015]:
                        di = d_sign * delta
                        dj = -d_sign * delta
                        if h[i] + di < -1e-9 or h[i] + di > 1.0 + 1e-9:
                            continue
                        if h[j] + dj < -1e-9 or h[j] + dj > 1.0 + 1e-9:
                            continue
                        h[i] += di
                        h[j] += dj
                        c5_test = compute_c5(h)
                        if c5_test < best_c5:
                            best_c5 = c5_test
                            best_h = h.copy()
                            improved = True
                            break
                        h[i] -= di
                        h[j] -= dj
                if not improved:
                    break
            h = best_h.copy()

    # Phase 4: SLSQP with very high beta smooth max
    if remaining() > 15:
        n = best_n
        target = n / 2.0
        dx = 2.0 / n

        for beta, topk in [(1000, 3), (3000, 3), (6000, 2), (10000, 2)]:
            if remaining() < 8:
                break

            def slsqp_obj(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                if topk < len(corr):
                    idx = np.argpartition(corr, -topk)[-topk:]
                else:
                    idx = np.arange(len(corr))
                vals = corr[idx]
                m = vals.max()
                shifted = vals - m
                return (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx

            try:
                res = minimize(slsqp_obj, best_h.copy(), method='SLSQP',
                               bounds=[(0.0, 1.0)] * n,
                               constraints=[{'type': 'eq',
                                             'fun': lambda x: np.sum(x) - target}],
                               maxiter=300, ftol=1e-18)
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  SLSQP beta={}, topk={}: C5={:.6f}".format(beta, topk, c5_new))
            except Exception:
                pass

    # Phase 5: Exhaustive single-coordinate scan
    if remaining() > 8:
        n = best_n
        target = n / 2.0
        h = best_h.copy()
        improvement_found = True
        while improvement_found and remaining() > 3:
            improvement_found = False
            for i in range(n):
                if remaining() < 2:
                    break
                for delta in [0.0005, -0.0005, 0.001, -0.001, 0.002, -0.002]:
                    di = delta
                    if h[i] + di < 0 or h[i] + di > 1:
                        continue
                    # Spread compensation among 3 random indices
                    comp = -di
                    candidates = rng.choice(n, size=3, replace=False)
                    candidates = candidates[candidates != i]
                    if len(candidates) == 0:
                        continue
                    per_j = comp / len(candidates)
                    valid = True
                    for j in candidates:
                        if h[j] + per_j < -1e-9 or h[j] + per_j > 1.0 + 1e-9:
                            valid = False
                            break
                    if not valid:
                        continue
                    for j in candidates:
                        h[j] += per_j
                    h[i] += di
                    c5_test = compute_c5(h)
                    if c5_test < best_c5:
                        best_c5 = c5_test
                        best_h = h.copy()
                        improvement_found = True
                        break
                    for j in candidates:
                        h[j] -= per_j
                    h[i] -= di
                if improvement_found:
                    h = best_h.copy()

    # Phase 6: Final fine-tune with Nesterov at very small step
    if remaining() > 10:
        n = best_n
        target = n / 2.0
        h = best_h.copy()
        y = h.copy()
        t_mom = 1.0
        beta = 5000
        topk = 3
        step = 0.003

        for it in range(200):
            if remaining() < 2:
                break
            g = smooth_max_grad(y, beta, topk)
            h_new = y - step * g
            h_new = project_feasible(h_new, target)
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
            y = h_new + ((t_mom - 1.0) / t_new) * (h_new - h)
            y = np.clip(y, 0.0, 1.0)
            t_mom = t_new
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new.copy()
            if it % 30 == 0 and it > 0:
                step *= 0.8

    # Final
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
