import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 4.0)


def compute_corr(h):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode="full")


def max_count(h):
    return float(np.max(compute_corr(h)))


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return 1.0
    return float(np.max(compute_corr(h)) * (2.0 / n))


def project_box_sum(x, target):
    x = np.asarray(x, dtype=float).copy()
    n = x.size
    if n == 0:
        return x
    target = float(target)
    if not np.all(np.isfinite(x)):
        return np.full(n, target / n, dtype=float)

    x = np.clip(x, 0.0, 1.0)
    s = float(np.sum(x))
    if abs(s - target) <= 1e-13:
        return x

    lo = -float(np.max(x))
    hi = 1.0 - float(np.min(x))
    if lo > hi:
        lo, hi = hi, lo

    for _ in range(45):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid

    x = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)

    diff = float(target - np.sum(x))
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if np.any(free):
            x[free] += diff / float(np.sum(free))
            x = np.clip(x, 0.0, 1.0)

    return x


def interp_to(src_h, src_n, dst_n):
    if int(src_n) == int(dst_n):
        return np.asarray(src_h, dtype=float).copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    xq = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(xq, xp, np.asarray(src_h, dtype=float))


def repair_binary(h, n, rng):
    h = np.asarray(h, dtype=float).copy()
    h = (h > 0.5).astype(float)
    m = n // 2
    cur = int(np.sum(h))

    if cur > m:
        ones = np.nonzero(h == 1.0)[0]
        if len(ones) >= cur - m:
            rem = rng.choice(ones, cur - m, replace=False)
            h[rem] = 0.0
    elif cur < m:
        zeros = np.nonzero(h == 0.0)[0]
        if len(zeros) >= m - cur:
            add = rng.choice(zeros, m - cur, replace=False)
            h[add] = 1.0
    return h


def make_random_binary(n, rng):
    h = np.zeros(n, dtype=float)
    m = n // 2
    if m > 0:
        idx = rng.choice(n, m, replace=False)
        h[idx] = 1.0
    return h


def make_thue(n, rng=None):
    h = np.zeros(n, dtype=float)
    for i in range(n):
        if (bin(i).count("1") & 1) == 0:
            h[i] = 1.0
    if rng is not None:
        h = repair_binary(h, n, rng)
    return h


def corr_grad(h, n, k):
    g = np.zeros(n, dtype=float)
    off = int(k) - (n - 1)
    if off >= 0:
        m = n - off
        if m > 0:
            g[off:off + m] += (1.0 - h[:m])
            g[:m] -= h[off:off + m]
    else:
        ao = -off
        m = n - ao
        if m > 0:
            g[:m] += (1.0 - h[ao:ao + m])
            g[ao:ao + m] -= h[:m]
    return g


def project_simplex(v):
    v = np.asarray(v, dtype=float)
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=float)
    cond = u - (cssv - 1.0) / idx > 0.0
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, 1.0 / n, dtype=float)
    r = int(rho[-1])
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lambda_simplex(A, r):
    lam = np.full(r, 1.0 / r, dtype=float)
    try:
        L = 2.0 * float(np.linalg.norm(A, 2))
        if not np.isfinite(L) or L <= 0.0:
            L = 2.0
        step = 1.0 / L
        for _ in range(220):
            grad = 2.0 * (A.dot(lam))
            new_lam = project_simplex(lam - step * grad)
            if np.linalg.norm(new_lam - lam) < 1e-11:
                lam = new_lam
                break
            lam = new_lam
        return lam
    except Exception:
        return None


def active_direction(h, F, idx):
    r = len(idx)
    n = len(h)
    if r == 0:
        return None

    P = []
    for k in idx:
        g = corr_grad(h, n, int(k))
        g -= g.mean()
        P.append(g)

    if r == 1:
        p = P[0]
    else:
        Pm = np.vstack(P)
        try:
            A = Pm.dot(Pm.T)
            lam = solve_lambda_simplex(A, r)
            if lam is None:
                lam = np.full(r, 1.0 / r, dtype=float)
            p = Pm.T.dot(lam)
        except Exception:
            p = np.mean(Pm, axis=0)

    p = np.asarray(p, dtype=float)
    p -= p.mean()
    norm = float(np.linalg.norm(p))
    if norm < 1e-12:
        return None
    return -p / norm


def safe_step(h, d, rho):
    t = float(rho)
    if t <= 0.0:
        return 0.0

    pos = d > 1e-14
    if np.any(pos):
        lim = float(np.min((1.0 - h[pos]) / d[pos]))
        if lim < t:
            t = lim

    neg = d < -1e-14
    if np.any(neg):
        lim = float(np.min((-h[neg]) / (-d[neg])))
        if lim < t:
            t = lim

    if t < 0.0:
        t = 0.0
    return t


def line_search_active(h, d, rho, n, cur_M):
    target = 0.5 * n
    best_h = h
    best_M = float(cur_M)
    improved = False

    st = safe_step(h, d, rho)
    if st > 1e-13:
        for frac in (1.0, 0.6, 0.35, 0.2, 0.1, 0.05):
            t = st * frac
            if t < 1e-12:
                continue
            x = h + t * d
            if (x.min() < -1e-12 or x.max() > 1.0 + 1e-12 or

            M = float(np.max(compute_corr(x)))
            if M < best_M - 1e-10:
                best_M = M

            improved = True

def smooth_obj(x):
    n = int(_N)
    corr = compute_corr(x)
    m = float(np.max(corr))
    b = float(_BETA)
    cl = np.clip(b * (corr - m), -50.0, 50.0)
    s = float(np.sum(np.exp(cl)))
    return float((m + np.log(s) / b) * (2.0 / n))


def smooth_grad(x):
    n = int(_N)
    b = float(_BETA)
    corr = compute_corr(x)
    m = float(np.max(corr))
    cl = np.clip(b * (corr - m), -50.0, 50.0)
    ex = np.exp(cl)
    se = float(np.sum(ex))
    if se <= 0.0:
        return np.zeros(n, dtype=float)

    w = ex / se
    idx = np.nonzero(w > 1e-8)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(w))], dtype=int)

    g = np.zeros(n, dtype=float)
    for k in idx:
        g += w[k] * corr_grad(x, n, int(k))

    return g * (2.0 / n)


def sum_con(x):
    return float(np.sum(x) - _TGT)


def sum_con_jac(x):
    return np.ones(int(_N), dtype=float)


def slsqp_smooth(h, n, beta, maxiter=200):
    global _N, _BETA, _TGT
    n = int(n)
    _N = n
    _BETA = float(beta)
    _TGT = 0.5 * n

    x0 = project_box_sum(np.asarray(h, dtype=float), 0.5 * n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]

    try:
        res = minimize(
            smooth_obj,
            x0,
            jac=smooth_grad,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': 1e-14, 'iprint': 0}
        )
        if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):
            return project_box_sum(res.x, 0.5 * n)
    except Exception:
        pass

    return x0


def corr_fft_binary(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    N = 2 * n - 1
    a = h
    b = (1.0 - h)[::-1]
    f = np.fft.rfft(a, N)
    g = np.fft.rfft(b, N)
    c = np.fft.irfft(f * g, N)
    return np.rint(c)


def binary_local(h, n, time_limit=30.0, rng=None):

    cur = int(np.max(corr_fft_binary(h)))
    best_h = h.copy()
    best_cur = cur

    t0 = time.time()
    stall = 0

    while time.time() - t0 < time_limit and remaining() > 5.0:
        ones = np.nonzero(h == 1.0)[0]
        zeros = np.nonzero(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        improved = False
        tries = 32
        for _ in range(tries):
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])

            h[i] = 0.0
            h[j] = 1.0
            cnew = int(np.max(corr_fft_binary(h)))

            if cnew < cur - 1e-9:
                cur = cnew
                best_h = h.copy()
                best_cur = cur
                improved = True
                break

            h[i] = 1.0
            h[j] = 0.0

        if improved:
            stall = 0
        else:
            stall += 1
            if stall % 18 == 0:
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                h[i] = 0.0
                h[j] = 1.0
                cur = int(np.max(corr_fft_binary(h)))
                if cur < best_cur:
                    best_h = h.copy()
                    best_cur = cur
                stall = 0
            if stall > 450:
                break

    best_h = repair_binary(best_h, n, rng)
    return best_h, float(max_count(best_h) * (2.0 / n))


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT

    _START = time.time()
    _BUDGET = float(budget_s)
    rng = np.random.RandomState(int(seed))

    best_h = None
    best_n = 0
    best_c5 = float("inf")

    try:
        ph = np.asarray(initial_h_values, dtype=float)
        if ph is not None and len(ph) > 0:
            best_n = int(len(ph))
            best_h = project_box_sum(ph, 0.5 * best_n)
            best_c5 = compute_c5(best_h)
    except Exception:
        best_h = None

    if best_h is None:
        best_n = 320
        best_h = make_random_binary(best_n, rng)
        best_c5 = compute_c5(best_h)

    if remaining() > 25.0:
        h2, c2 = active_refine(
            best_h, best_n,
            max_iter=320,
            time_limit=min(110.0, max(25.0, remaining() * 0.16)),
            rng=rng
        )
        c2 = compute_c5(h2)
        if np.isfinite(c2) and c2 < best_c5 - 1e-13:
            best_c5 = float(c2)
            best_h = h2
            best_n = int(len(h2))

    if remaining() > 45.0 and best_n <= 800:
        maxiter_parent = 180 if best_n <= 420 else 110
        for beta in (80.0, 300.0, 1000.0):
            if remaining() < 20.0:
                break
            h2 = slsqp_smooth(best_h, best_n, beta, maxiter=maxiter_parent)
            c2 = compute_c5(h2)
            if np.isfinite(c2) and c2 < best_c5 - 1e-13:
                best_c5 = float(c2)
                best_h = h2
                best_n = int(len(h2))

    if remaining() > 30.0:
        h2, c2 = active_refine(
            best_h, best_n,
            max_iter=260,
            time_limit=min(90.0, max(25.0, remaining() * 0.13)),
            rng=rng
        )
        c2 = compute_c5(h2)
        if np.isfinite(c2) and c2 < best_c5 - 1e-13:
            best_c5 = float(c2)
            best_h = h2
            best_n = int(len(h2))

    cand_n = []
    for factor in (1.25, 1.5, 2.0, 2.5):
        nn = int(round(best_n * factor / 2.0) * 2)
        if 220 <= nn <= 900 and nn != best_n:
            cand_n.append(nn)
    cand_n = sorted(set(cand_n))

    for nn in cand_n:
        if remaining() < 55.0:
            break

        local_h = interp_to(best_h, best_n, nn)
        local_h = project_box_sum(local_h, 0.5 * nn)
        local_c5 = compute_c5(local_h)

        if remaining() > 25.0:
            ah, ac = active_refine(
                local_h, nn,
                max_iter=220,
                time_limit=min(80.0, max(20.0, remaining() * 0.18)),
                rng=rng
            )
            ac = compute_c5(ah)
            if np.isfinite(ac) and ac < local_c5:
                local_h = ah
                local_c5 = float(ac)

        if nn <= 650 and remaining() > 55.0:
            maxiter_hi = 120 if nn <= 520 else 80
            for beta in (300.0, 1000.0):
                if remaining() < 20.0:
                    break
                sh = slsqp_smooth(local_h, nn, beta, maxiter=maxiter_hi)
                sc = compute_c5(sh)
                if np.isfinite(sc) and sc < local_c5:
                    local_h = sh
                    local_c5 = float(sc)

            if remaining() > 18.0:
                ah, ac = active_refine(
                    local_h, nn,
                    max_iter=180,
                    time_limit=min(50.0, max(12.0, remaining() * 0.16)),
                    rng=rng
                )
                ac = compute_c5(ah)
                if np.isfinite(ac) and ac < local_c5:
                    local_h = ah
                    local_c5 = float(ac)

        if np.isfinite(local_c5) and local_c5 < best_c5 - 1e-13:
            best_c5 = float(local_c5)
            best_h = local_h
            best_n = int(nn)

    if remaining() > 55.0:
        thue_ns = [best_n, 640, 800]
        seen = set()
        for nn in thue_ns:
            if remaining() < 35.0 or nn < 220 or nn > 900:
                continue
            if nn in seen:
                continue
            seen.add(nn)
            th = make_thue(nn, rng)
            ah, ac = active_refine(
                th, nn,
                max_iter=180,
                time_limit=min(55.0, max(20.0, remaining() * 0.18)),
                rng=rng
            )
            ac = compute_c5(ah)
            if np.isfinite(ac) and ac < best_c5 - 1e-13:
                best_c5 = float(ac)
                best_h = ah
                best_n = int(nn)

    if remaining() > 70.0:
        bin_ns = []
        for nn in (400, 480, 512, 640):
            if 220 <= nn <= 900 and nn % 2 == 0 and nn not in bin_ns:
                bin_ns.append(nn)

        for nn in bin_ns:
            if remaining() < 45.0:
                break
            binit = (best_h > 0.5).astype(float)
            if len(binit) != nn:
                binit = (interp_to(best_h, best_n, nn) > 0.5).astype(float)
            bh, bc = binary_local(
                binit, nn,
                time_limit=min(35.0, max(15.0, remaining() * 0.22)),
                rng=rng
            )
            bc = compute_c5(bh)
            if np.isfinite(bc) and bc < best_c5 - 1e-13:
                best_c5 = float(bc)
                best_h = bh
                best_n = int(nn)

    best_h = project_box_sum(best_h, 0.5 * best_n)
    final_c5 = compute_c5(best_h)

    if np.isfinite(final_c5):
        best_c5 = float(final_c5)

    try:
        eval_c5 = float(evaluate_erdos_solution(best_h))
        if np.isfinite(eval_c5) and eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, float(best_c5), int(best_n))
