def binary_local_search(h, n, max_time=60.0, rng=None):
    """Greedy binary local search with pair swaps."""
    if rng is None:
        rng = np.random.RandomState(42)
    
    h = np.where(h > 0.5, 1.0, 0.0).copy()
    # Ensure exactly n/2 ones
    ones = np.sum(h)
    target = n // 2
    if ones > target:
        idx = np.where(h == 1)[0]
        remove = rng.choice(len(idx), ones - target, replace=False)
        h[idx[remove]] = 0
    elif ones < target:
        idx = np.where(h == 0)[0]
        add = rng.choice(len(idx), target - ones, replace=False)
        h[idx[add]] = 1
    
    best_c5 = compute_c5(h)
    start_time = time.time()
    
    while time.time() - start_time < max_time:
        improved = False
        ones_list = np.where(h == 1)[0]
        zeros_list = np.where(h == 0)[0]
        
        # Try all pairs (or a large random subset if too many)
        n_pairs = len(ones_list) * len(zeros_list)
        if n_pairs > 50000:
            # Random subset
            n_try = 50000
            for _ in range(n_try):
                i = ones_list[rng.randint(len(ones_list))]
                j = zeros_list[rng.randint(len(zeros_list))]
                h[i] = 0
                h[j] = 1
                c5 = compute_c5(h)
                h[i] = 1
                h[j] = 0
                if c5 < best_c5:
                    h[i] = 0
                    h[j] = 1
                    best_c5 = c5
                    improved = True
                    break
        else:
            for i in ones_list:
                for j in zeros_list:
                    h[i] = 0

                    h[i] = 1
                    h[j] = 0
                    if c5 < best_c5:

                        h[j] = 1
                        best_c5 = c5
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:

def binary_sa(h, n, max_time=60.0, rng=None, t_start=0.003, t_end=0.0001):
    """Binary simulated annealing with pair swaps."""
    h = np.where(h > 0.5, 1.0, 0.0).copy()
    ones = np.sum(h)
    target = n // 2
    if ones > target:
        idx = np.where(h == 1)[0]
        remove = rng.choice(len(idx), ones - target, replace=False)
        h[idx[remove]] = 0
    elif ones < target:
        idx = np.where(h == 0)[0]
        add = rng.choice(len(idx), target - ones, replace=False)
        h[idx[add]] = 1
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    start_time = time.time()
    it = 0
    total_iters = 200000  # estimate
    
    while time.time() - start_time < max_time:
        it += 1
        ones_list = np.where(h == 1)[0]
        zeros_list = np.where(h == 0)[0]
        
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h)

            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h, best_c5

    h = np.zeros(n)
    pattern = np.zeros(period)
    pattern[ones_per_period] = 1.0
    for i in range(n):
        h[i] = pattern[(i + phase) % period]
    return h

def binary_greedy_search(h_init, n, max_time=60.0, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = np.where(h_init > 0.5, 1.0, 0.0).copy()
    target = n // 2
    ones = np.sum(h)
    if ones > target:
        idx = np.where(h == 1)[0]
        remove = rng.choice(len(idx), int(ones - target), replace=False)
        h[idx[remove]] = 0
    elif ones < target:
        idx = np.where(h == 0)[0]
        add = rng.choice(len(idx), int(target - ones), replace=False)
        h[idx[add]] = 1

    best_c5 = compute_c5(h)
    start = time.time()

    while time.time() - start < max_time:
        improved = False
        ones_list = np.where(h == 1)[0].tolist()
        zeros_list = np.where(h == 0)[0].tolist()

        best_swap_c5 = best_c5
        best_i, best_j = -1, -1

        for i in ones_list:
            for j in zeros_list:
                h[i] = 0.0
                h[j] = 1.0
                c5 = compute_c5(h)
                if c5 < best_swap_c5:
                    best_swap_c5 = c5
                    best_i, best_j = i, j
                h[i] = 1.0
                h[j] = 0.0

        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            best_c5 = best_swap_c5
            improved = True

        if not improved:
            break

    return h, best_c5

def binary_sa(h_init, n, max_time=30.0, rng=None, t0=0.003, t1=0.00005):
    h = np.where(h_init > 0.5, 1.0, 0.0).copy()
    target = n // 2
    ones = np.sum(h)
    if ones > target:
        idx = np.where(h == 1)[0]
        remove = rng.choice(len(idx), int(ones - target), replace=False)
        h[idx[remove]] = 0
    elif ones < target:
        idx = np.where(h == 0)[0]
        add = rng.choice(len(idx), int(target - ones), replace=False)
        h[idx[add]] = 1

    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    start = time.time()
    it = 0
    total_est = int(max_time * 5000)

    while time.time() - start < max_time:
        it += 1
        ones_list = np.where(h == 1)[0]
        zeros_list = np.where(h == 0)[0]
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        T = t0 * (t1 / t0) ** (it / max(total_est, 1))
        if new_c5 < cur_c5 or (T > 1e-15 and rng.rand() < np.exp(-(new_c5 - cur_c5) / T)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0

    return best_h, best_c5

def subgradient_step(h, n, step_size=0.02):
    """One subgradient step on the active (max) shift."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = np.argmax(corr) - (n - 1)

    grad = np.zeros(n)
    if s >= 0:
        for m in range(n):
            if m <= n - 1 - s:
                grad[m] += 1.0 - h[m + s]
            if m >= s:
                grad[m] += -h[m - s]
    else:
        sp = -s
        for m in range(n):
            if m + sp <= n - 1:
                grad[m] += 1.0 - h[m + sp]
            if m >= sp:
                grad[m] += -h[m - sp]

    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

def smooth_max_c5(h, K=5, beta=100.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(corr)[-K:]
    top_vals = corr[top_idx]
    m = np.max(top_vals)
    sm = m + np.log(np.sum(np.exp(beta * (top_vals - m)))) / beta
    return sm, top_idx

def smooth_max_grad(h, K=5, beta=100.0):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    top_idx = np.argsort(corr)[-K:]
    top_vals = corr[top_idx]
    m = np.max(top_vals)
    weights = np.exp(beta * (top_vals - m))
    weights /= np.sum(weights)

    grad = np.zeros(n)
    for wi, j in zip(weights, top_idx):
        s = int(j) - (n - 1)
        if s >= 0:
            for m2 in range(n):
                if m2 <= n - 1 - s:
                    grad[m2] += wi * (1.0 - h[m2 + s])
                if m2 >= s:
                    grad[m2] += wi * (-h[m2 - s])
        else:
            sp = -s
            for m2 in range(n):
                if m2 + sp <= n - 1:
                    grad[m2] += wi * (1.0 - h[m2 + sp])
                if m2 >= sp:
                    grad[m2] += wi * (-h[m2 - sp])

    return grad * dx, top_idx

def coordinate_descent(h, n, max_time=30.0, rng=None):
    h = h.copy()
    start = time.time()
    best_c5 = compute_c5(h)

    while time.time() - start < max_time:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() - start > max_time:
                break
            old = h[i]
            c5_old = compute_c5(h)
            best_val = old
            best_c = c5_old
            candidates = [0.0, 1.0, 0.5, max(0.0, old - 0.15), min(1.0, old + 0.15),
                          max(0.0, old - 0.05), min(1.0, old + 0.05)]
            for nv in candidates:
                if abs(nv - old) < 1e-10:
                    continue
                h[i] = nv
                h_feas = project_feasible(h, n)
                c = compute_c5(h_feas)
                if c < best_c:
                    best_c = c
                    best_val = nv
                    h = h_feas.copy()
            h[i] = best_val
            h = project_feasible(h, n)
            if compute_c5(h) < best_c5 - 1e-10:
                best_c5 = compute_c5(h)
                improved = True

        if not improved:
            break

    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _start_time, _budget_s
    _start_time = time.time()
    _budget_s = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.7f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 160
        best_h = make_thue_morse(best_n)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)
        print("No parent. Thue-Morse n={}, C5={:.7f}".format(best_n, best_c5))

    # Phase 1: Binary greedy search at parent resolution
    n1 = best_n
    h_bin = np.where(best_h > 0.5, 1.0, 0.0)
    h_bin = project_feasible(h_bin, n1)
    t_alloc = min(200.0, remaining_time() * 0.25)
    h_bin, c5_bin = binary_greedy_search(h_bin, n1, max_time=t_alloc, rng=rng)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
        best_n = n1
        print("Binary greedy n={}: C5={:.7f}".format(n1, c5_bin))

    # Phase 2: Binary SA restarts
    if remaining_time() > 60:
        t_alloc = min(80.0, remaining_time() * 0.12)
        n_restarts = max(2, int(t_alloc / 25))
        for r in range(n_restarts):
            if remaining_time() < 20:
                break
            pert = best_h + rng.randn(n1) * 0.3
            pert = np.clip(pert, 0, 1)
            pert = project_feasible(pert, n1)
            h_sa, c5_sa = binary_sa(pert, n1, max_time=t_alloc / n_restarts - 2, rng=rng,
                                     t0=0.004 * (0.7 ** r), t1=0.00003)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n1
                print("SA restart {}: C5={:.7f}".format(r, c5_sa))

    # Phase 3: Try other resolutions with binary search
    other_ns = [120, 200, 240]
    for nn in other_ns:
        if remaining_time() < 40:
            break
        t_alloc = min(60.0, remaining_time() * 0.1)
        h_interp = interp_to(best_h, best_n, nn)
        h_bin = np.where(h_interp > 0.5, 1.0, 0.0)
        h_bin = project_feasible(h_bin, nn)
        h_bin, c5_b = binary_greedy_search(h_bin, nn, max_time=t_alloc, rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_bin.copy()
            best_n = nn
            print("Binary greedy n={}: C5={:.7f}".format(nn, c5_b))

        if remaining_time() > 20:
            h_tm = make_thue_morse(nn)
            h_tm = project_feasible(h_tm, nn)
            h_tm, c5_tm = binary_greedy_search(h_tm, nn, max_time=min(20.0, remaining_time() * 0.08), rng=rng)
            if c5_tm < best_c5:
                best_c5 = c5_tm
                best_h = h_tm.copy()
                best_n = nn
                print("TM binary n={}: C5={:.7f}".format(nn, c5_tm))

    # Phase 4: Continuous optimization
    n_best = best_n
    h_work = best_h.copy()
    bounds = [(0.0, 1.0)] * n_best

    for K in [3, 5, 8]:
        if remaining_time() < 30:
            break
        for beta in [50, 150, 400, 800]:
            if remaining_time() < 10:
                break
            for trial in range(2):
                if remaining_time() < 8:
                    break
                h0 = h_work + rng.randn(n_best) * 0.02 * (trial + 1)
                h0 = project_feasible(h0, n_best)

                obj_fn = lambda x: smooth_max_c5(x, K=K, beta=beta)[0] + 1000.0 * (np.sum(x) - n_best / 2.0) ** 2
                grad_fn = lambda x: smooth_max_grad(x, K=K, beta=beta)[0] + 2000.0 * (np.sum(x) - n_best / 2.0) * np.ones(n_best)

                try:
                    res = minimize(obj_fn, h0, jac=grad_fn, method='L-BFGS-B',
                                   bounds=bounds, options={'maxiter': 500, 'ftol': 1e-16,
                                                           'gtol': 1e-12, 'iprint': 0, 'maxls': 80})
                    h_new = project_feasible(res.x, n_best)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h_work = h_new.copy()
                        print("L-BFGS K={}, beta={}, trial={}: C5={:.7f}".format(K, beta, trial, c5_new))
                except Exception:
                    pass

    # Phase 5: Subgradient method on active shift
    if remaining_time() > 30:
        h_sub = best_h.copy()
        t_alloc = min(80.0, remaining_time() * 0.15)
        start_sub = time.time()
        step = 0.03
        for it in range(5000):
            if time.time() - start_sub > t_alloc:
                break
            h_sub = subgradient_step(h_sub, n_best, step_size=step)
            c5_sub = compute_c5(h_sub)
            if c5_sub < best_c5:
                best_c5 = c5_sub
                best_h = h_sub.copy()
            if it % 100 == 0:
                step *= 0.95

    # Phase 6: Final L-BFGS polish with high beta
    if remaining_time() > 15:
        for beta in [1000, 2000, 5000]:
            if remaining_time() < 6:
                break
            for K in [2, 3, 5]:
                if remaining_time() < 5:
                    break
                h0 = best_h.copy()
                obj_fn = lambda x: smooth_max_c5(x, K=K, beta=beta)[0] + 5000.0 * (np.sum(x) - n_best / 2.0) ** 2
                grad_fn = lambda x: smooth_max_grad(x, K=K, beta=beta)[0] + 10000.0 * (np.sum(x) - n_best / 2.0) * np.ones(n_best)
                try:
                    res = minimize(obj_fn, h0, jac=grad_fn, method='L-BFGS-B',
                                   bounds=bounds, options={'maxiter': 1000, 'ftol': 1e-18,
                                                           'gtol': 1e-14, 'iprint': 0, 'maxls': 100})
                    h_new = project_feasible(res.x, n_best)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        print("Polish beta={}, K={}: C5={:.7f}".format(beta, K, c5_new))
                except Exception:
                    pass

    # Phase 7: Coordinate descent
    if remaining_time() > 15:
        h_cd, c5_cd = coordinate_descent(best_h, n_best, max_time=min(30.0, remaining_time() * 0.2), rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            print("Coord descent: C5={:.7f}".format(c5_cd))

    # Phase 8: Final final L-BFGS
    if remaining_time() > 5:
        h0 = best_h.copy()
        for K in [2, 3]:
            beta = 3000
            obj_fn = lambda x: smooth_max_c5(x, K=K, beta=beta)[0] + 10000.0 * (np.sum(x) - n_best / 2.0) ** 2
            grad_fn = lambda x: smooth_max_grad(x, K=K, beta=beta)[0] + 20000.0 * (np.sum(x) - n_best / 2.0) * np.ones(n_best)
            try:
                res = minimize(obj_fn, h0, jac=grad_fn, method='L-BFGS-B',
                               bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-18,
                                                       'gtol': 1e-15, 'iprint': 0, 'maxls': 120})
                h_new = project_feasible(res.x, n_best)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h0 = h_new.copy()
            except Exception:
                pass

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
