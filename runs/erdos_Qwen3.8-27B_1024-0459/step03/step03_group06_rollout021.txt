def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        unsat = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(unsat)
        if cnt > 0:
            h[unsat] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:

    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining_time():

def smooth_max_obj_vec(h, n, beta, top_k=None):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k is not None:
        idx = np.argsort(corr)[-top_k:]
        corr_sub = corr[idx]
    else:
        corr_sub = corr
    m = np.max(corr_sub)

    sm = m + np.log(np.sum(np.exp(beta * shifted))) / beta
    penalty = 500.0 * (np.sum(h) - n / 2.0) ** 2

        T = t_start * (t_end / t_start) ** (float(it) / max(float(max_iters), 1.0))
        
        if new_c5 < current_c5 or (T > 1e-15 and rng.rand() < np.exp(-(new_c5 - current_c5) / T)):
            current_c5 = new_c5
            ones_list[i] = oz
            zeros_list[j] = oi
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0
    
    return best_h, best_c5


def coordinate_descent(h, n, rng, sweeps=8):
    dx_val = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for sweep in range(sweeps):
        if remaining_time() < 3:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if remaining_time() < 2:
                break
            old_val = h[i]
            best_local_c5 = best_c5
            best_local_h = h.copy()
            
            candidates = [0.0, 1.0, 0.5, max(0.0, old_val - 0.25), min(1.0, old_val + 0.25)]
            if old_val > 0.3:
                candidates.append(0.1)
            if old_val < 0.7:
                candidates.append(0.9)
            
            for nv in candidates:
                if abs(nv - old_val) < 1e-8:
                    continue
                h[i] = nv
                h_feas = project_feasible(h, n)
                c5_test = compute_c5(h_feas)
                if c5_test < best_local_c5:
                    best_local_c5 = c5_test
                    best_local_h = h_feas.copy()
            h = best_local_h.copy()
            
            if best_local_c5 < best_c5:
                best_c5 = best_local_c5
                any_imp = True
        if not any_imp:
            break
    
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.7f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 160
        best_h = make_thue_morse(best_n)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)
        print("No parent. Thue-Morse: n={}, C5={:.7f}".format(best_n, best_c5))
    
    # Multi-resolution optimization
    n_list = [160, 200, 240, 320]
    n_list = sorted(set(n_list))
    
    for n in n_list:
        if remaining_time() < 30:
            break
        print("\n--- n={} (remaining {:.0f}s) ---".format(n, remaining_time()))
        
        bounds = [(0.0, 1.0)] * n
        h_interp = interp_to(best_h, best_n, n)
        h_interp = project_feasible(h_interp, n)
        
        inits = [h_interp]
        # Thue-Morse
        inits.append(project_feasible(make_thue_morse(n), n))
        # Golden ratio
        inits.append(project_feasible(make_golden_ratio(n), n))
        # Multi-sine
        freqs = rng.randint(2, 12, 4)
        phases = rng.uniform(0, 6.28, 4)
        inits.append(project_feasible(make_multi_sine(n, freqs, phases), n))
        # Sine
        inits.append(project_feasible(make_sine(n, rng.randint(2,8), rng.rand()*6.28), n))
        # Random binary
        for _ in range(2):
            h_bin = np.zeros(n)
            idx = rng.permutation(n)
            h_bin[idx[:n//2]] = 1.0
            inits.append(h_bin)
        
        local_best_h = h_interp.copy()
        local_best_c5 = compute_c5(h_interp)
        
        for h0 in inits:
            if remaining_time() < 12:
                break
            c5_cur = compute_c5(h0)
            if c5_cur > best_c5 * 1.15:
                continue
            
            # Try multiple beta values for smooth max
            for beta_val in [100, 300, 800]:
                if remaining_time() < 8:
                    break
                try:
                    def obj_fn(x, beta=beta_val, nn=n):
                        return smooth_max_obj_vec(x, nn, beta)
                    def grad_fn(x, beta=beta_val, nn=n):
                        return smooth_max_grad_vec(x, nn, beta)
                    
                    res = minimize(obj_fn, h0, jac=grad_fn,
                                   method='L-BFGS-B', bounds=bounds,
                                   options={'maxiter': 600, 'ftol': 1e-15,
                                            'gtol': 1e-10, 'iprint': 0,
                                            'maxls': 40, 'maxfun': 8000})
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur:
                        h0 = h_new
                        c5_cur = c5_new
                except Exception:
                    pass
            
            # Top-k targeted
            if remaining_time() > 6:
                try:
                    for tk in [8, 15]:
                        if remaining_time() < 5:
                            break
                        def obj_fn2(x, nn=n, al=tk):
                            dx = 2.0/nn
                            corr = np.correlate(x, 1.0-x, mode='full')
                            top = np.sort(corr)[::-1][:al]
                            return np.sum(top)*dx + 500.0*(np.sum(x)-nn/2.0)**2
                        res2 = minimize(obj_fn2, h0, method='L-BFGS-B',
                                       bounds=bounds,
                                       options={'maxiter': 300, 'ftol': 1e-14,
                                                'iprint': 0, 'maxls': 30})
                        h_new2 = project_feasible(res2.x, n)
                        c5_new2 = compute_c5(h_new2)
                        if c5_new2 < c5_cur:
                            h0 = h_new2
                            c5_cur = c5_new2
                except Exception:
                    pass
            
            if c5_cur < local_best_c5:
                local_best_c5 = c5_cur
                local_best_h = h0.copy()
            
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h0.copy()
                best_n = n
                print("  Improved: n={}, C5={:.7f}".format(n, c5_cur))
        
        if local_best_c5 < best_c5:
            best_c5 = local_best_c5
            best_h = local_best_h.copy()
            best_n = n
    
    # Binary SA at best resolution and nearby
    sa_n_list = [best_n]
    if remaining_time() > 60:
        for alt_n in [best_n - 20, best_n + 20, 160, 200]:
            if 80 <= alt_n <= 400 and alt_n != best_n:
                sa_n_list.append(alt_n)
    
    for sa_n in sa_n_list[:3]:
        if remaining_time() < 20:
            break
        print("\n--- Binary SA n={} ---".format(sa_n))
        
        h_start = interp_to(best_h, best_n, sa_n)
        h_start = project_feasible(h_start, sa_n)
        
        sa_inits = [h_start]
        sa_inits.append(np.where(h_start > 0.5, 1.0, 0.0))
        for _ in range(3):
            h_bin = np.zeros(sa_n)
            idx = rng.permutation(sa_n)
            h_bin[idx[:sa_n//2]] = 1.0
            sa_inits.append(h_bin)
        
        for sa_h in sa_inits:
            if remaining_time() < 12:
                break
            max_iters = min(5000, int(remaining_time() * 50))
            h_bin, c5_bin = shift_aware_sa(sa_h, sa_n, rng, max_iters)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                best_n = sa_n
                print("  SA: n={}, C5={:.7f}".format(sa_n, c5_bin))
            
            # Polish with continuous
            if remaining_time() > 8 and c5_bin < best_c5 * 1.01:
                bounds_sa = [(0.0, 1.0)] * sa_n
                for beta_val in [200, 600]:
                    if remaining_time() < 5:
                        break
                    try:
                        def obj_fn3(x, beta=beta_val, nn=sa_n):
                            return smooth_max_obj_vec(x, nn, beta)
                        def grad_fn3(x, beta=beta_val, nn=sa_n):
                            return smooth_max_grad_vec(x, nn, beta)
                        res3 = minimize(obj_fn3, h_bin, jac=grad_fn3,
                                       method='L-BFGS-B', bounds=bounds_sa,
                                       options={'maxiter': 400, 'ftol': 1e-15,
                                                'gtol': 1e-10, 'iprint': 0})
                        h_new3 = project_feasible(res3.x, sa_n)
                        c5_new3 = compute_c5(h_new3)
                        if c5_new3 < best_c5:
                            best_c5 = c5_new3
                            best_h = h_new3.copy()
                            best_n = sa_n
                            print("  Polished: n={}, C5={:.7f}".format(sa_n, c5_new3))
                    except Exception:
                        pass
    
    # Coordinate descent polish
    if remaining_time() > 10:
        print("\n--- Coordinate descent n={} ---".format(best_n))
        h_cd, c5_cd = coordinate_descent(best_h, best_n, rng, sweeps=10)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            print("  CD: C5={:.7f}".format(best_c5))
    
    # Final L-BFGS-B polish with very high beta
    if remaining_time() > 6:
        n_fin = best_n
        bounds_fin = [(0.0, 1.0)] * n_fin
        for beta_val in [1000, 2000]:
            if remaining_time() < 4:
                break
            try:
                def obj_fn4(x, beta=beta_val, nn=n_fin):
                    return smooth_max_obj_vec(x, nn, beta)
                def grad_fn4(x, beta=beta_val, nn=n_fin):
                    return smooth_max_grad_vec(x, nn, beta)
                res4 = minimize(obj_fn4, best_h, jac=grad_fn4,
                               method='L-BFGS-B', bounds=bounds_fin,
                               options={'maxiter': 800, 'ftol': 1e-16,
                                        'gtol': 1e-11, 'iprint': 0})
                h_new4 = project_feasible(res4.x, n_fin)
                c5_new4 = compute_c5(h_new4)
                if c5_new4 < best_c5:
                    best_c5 = c5_new4
                    best_h = h_new4.copy()
                    print("  Final polish: C5={:.7f}".format(best_c5))
            except Exception:
                pass
    
    # Second coordinate descent pass
    if remaining_time() > 5:
        h_cd2, c5_cd2 = coordinate_descent(best_h, best_n, rng, sweeps=6)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
