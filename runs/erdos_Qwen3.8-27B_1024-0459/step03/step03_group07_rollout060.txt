import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-13
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_h(h_src, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_new = np.interp(xn, xp, h_src)
    return project_feasible(h_new, n_dst)

def coord_descent_fast(h, n, n_sweeps, t0, time_limit, rng):
    h = project_feasible(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best = h.copy()
    best_c5 = float(np.max(corr)) * (2.0 / n)
    two_n_1 = 2 * n - 1

    for sweep in range(n_sweeps):
        if time.time() - t0 >= time_limit:
            break
        order = rng.permutation(n)
        for j in order:
            if time.time() - t0 >= time_limit:
                break
            old = h[j]
            l_min = max(-j, j - (n - 1))
            l_max = min(n - 1 - j, j)
            f_min = l_min + (n - 1)
            f_max = l_max + (n - 1)
            f_min = max(0, f_min)
            f_max = min(two_n_1 - 1, f_max)
            if f_min > f_max:
                continue
            n_affected = f_max - f_min + 1

            A = np.zeros(n_affected)
            lag0_pos = -1
            for idx in range(n_affected):
                fidx = f_min + idx
                l = fidx - (n - 1)
                if l == 0:
                    lag0_pos = idx
                else:
                    a = 0.0
                    if 0 <= j + l <= n - 1:
                        a += (1.0 - h[j + l])
                    if 0 <= j - l <= n - 1:
                        a -= h[j - l]
                    A[idx] = a

            base = corr[f_min:f_max + 1].copy()

            if n_affected < two_n_1:
                corr_m = corr.copy()
                corr_m[f_min:f_max + 1] = -1e18
                max_ua = float(np.max(corr_m))
            else:
                max_ua = -1e18

            xs = np.linspace(0.0, 1.0, 40)
            best_x = old
            best_val = 1e18
            for x in xs:
                delta = x - old
                vals = base + A * delta
                if lag0_pos >= 0:
                    vals[lag0_pos] = base[lag0_pos] + x * (1.0 - x) - old * (1.0 - old)
                val = max(max_ua, float(np.max(vals)))
                if val < best_val:
                    best_val = val
                    best_x = x

            lo = max(0.0, best_x - 0.015)
            hi = min(1.0, best_x + 0.015)
            for _ in range(22):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                d1 = m1 - old
                v1 = max(max_ua, float(np.max(base + A * d1)))
                if lag0_pos >= 0:
                    v1 = max(max_ua, float(np.max(np.where(
                        np.arange(n_affected) == lag0_pos,
                        base[lag0_pos] + m1 * (1.0 - m1) - old * (1.0 - old),
                        base + A * d1))))
                d2 = m2 - old
                v2 = max(max_ua, float(np.max(base + A * d2)))
                if lag0_pos >= 0:
                    v2 = max(max_ua, float(np.max(np.where(
                        np.arange(n_affected) == lag0_pos,
                        base[lag0_pos] + m2 * (1.0 - m2) - old * (1.0 - old),
                        base + A * d2))))
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1

            h[j] = (lo + hi) / 2.0

        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        c5 = float(np.max(corr)) * (2.0 / n)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best = h.copy()

    return best, best_c5

def subgradient_step(h, n, top_k, lr, t0, time_limit, max_iters):
    h = project_feasible(h, n).copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)

    for it in range(max_iters):
        if time.time() - t0 >= time_limit:
            break
        thresh = np.sort(corr)[-top_k][-1]
        active = np.where(corr >= thresh - 1e-12)[0]
        if len(active) == 0:
            active = np.argsort(corr)[-top_k:]
        w = 1.0 / len(active)

        grad = np.zeros(n)
        for a in active:
            lag = a - (n - 1)
            if lag >= 0:
                for i in range(n - lag):
                    grad[i] += w * (1.0 - h[i + lag])
                    grad[i + lag] += w * (-h[i])
            else:
                s = -lag
                for i in range(n - s):
                    grad[i + s] += w * (1.0 - h[i])
                    grad[i] += w * (-h[i + s])

        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
            h = h_new
            corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        else:
            lr *= 0.9
            if lr < 1e-6:
                break

    return best, best_c5

def block_cd(h, n, block_size, n_sweeps, t0, time_limit, rng):
    h = project_feasible(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best = h.copy()

            l_max_all = max(min(n - 1 - j, j) for j in idx)

            f_max = min(two_n_1 - 1, l_max_all + (n - 1))

                        val = max(max_ua, float(np.max(vals)))

def simulated_anneal(h, n, t0, time_limit, rng, n_steps=2000):
    h = project_feasible(h, n).copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_c5 = best_c5
    T = 0.01
    T_min = 0.0005

    elapsed0 = time.time() - t0
    step_time = max(0.001, (time_limit - elapsed0) / n_steps)

    for step in range(n_steps):
        if time.time() - t0 >= time_limit:
            break
        # Pick a random subset of indices to perturb
        n_perturb = rng.randint(1, max(2, n // 10) + 1)
        idx = rng.choice(n, size=n_perturb, replace=False)
        delta = rng.randn(n_perturb) * 0.05
        h_new = h.copy()
        h_new[idx] = np.clip(h_new[idx] + delta, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)

        if c5_new < current_c5 or rng.rand() < np.exp(-(c5_new - current_c5) / T):
            h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best = h.copy()

        T = max(T_min, T * 0.995)

    return best, best_c5
