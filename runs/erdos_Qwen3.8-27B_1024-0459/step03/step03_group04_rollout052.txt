            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xq = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xq, xp, src)


def find_active(corr, m=20, gap=1.0):
    mx = np.max(corr)
    thresh = mx - gap
    act = np.where(corr >= thresh)[0]
    if len(act) > m:
        order = np.argsort(corr[act])[::-1]
        act = act[order[:m]]
    if len(act) < 3:
        act = np.argsort(corr)[::-1][:m]
    return act


def grad_corr_j(h, j, n):
    kk = j - (n - 1)
    grad = np.zeros(n)
    if kk >= 0:
        nk = n - kk
        grad[kk:] += (1.0 - h[:nk])
        grad[:nk] -= h[kk:]
    else:
        m = -kk
        nk = n - m
        grad[:nk] += (1.0 - h[m:])
        grad[m:] -= h[:nk]
    return grad


def make_obj_smooth(beta, active, n):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        vals = corr[active]
        mm = np.max(vals)
        if beta < 1:
            return mm * dx
        exps = np.exp(beta * (vals - mm))
        return (mm + np.log(np.sum(exps)) / beta) * dx
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        vals = corr[active]
        mm = np.max(vals)
        if beta < 1:
            idx = np.argmax(vals)
            return mm * dx, grad_corr_j(x, active[idx], n) * dx
        weights = beta * np.exp(beta * (vals - mm))
        s = np.sum(weights)
        if s < 1e-300:
            idx = np.argmax(vals)
            return mm * dx, grad_corr_j(x, active[idx], n) * dx
        weights /= s
        sme = mm + np.log(s) / beta
        g = np.zeros(n)
        for idx, jj in enumerate(active):
            g += weights[idx] * grad_corr_j(x, jj, n)
        return sme * dx, g * dx
    return obj, obj_grad


def make_constraint(n):
    def con(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    return con, con_jac


def make_binary_random(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return h


def make_stream(n):
    h = np.zeros(n)
    for i in range(n):
        if i == 0:
            h[i] = 0.0
        else:
            prev = (i * (i - 1) // 2) % 2
            h[i] = 1.0 if prev else 0.0
    target = n // 2
    cur = np.sum(h)
    if cur < target:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:int(target - cur)]:
            h[idx] = 1.0
    elif cur > target:
        ones = np.where(h == 1)[0]
        for idx in ones[:int(cur - target)]:
            h[idx] = 0.0
    return h


def time_left(t0, budget):
    return max(0.0, budget - (time.time() - t0))


def slsqp_round(h, n, active, beta, maxiter):
    dx = 2.0 / n
    obj_fn, obj_grad_fn = make_obj_smooth(beta, active, n)
    con_fn, con_jac_fn = make_constraint(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fn, 'jac': con_jac_fn}]
    res = minimize(obj_grad_fn, h, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
    if res.success or res.fun < obj_fn(h):
        return project_feasible(res.x, n)
    return h


def coordinate_descent(h, n, best_c5, max_rounds=50):
    dx = 2.0 / n
    h = h.copy()
    improved = True
    rounds = 0
    while improved and rounds < max_rounds:
        improved = False
        rounds += 1
        c5_cur, corr = compute_c5(h)
        for i in range(n):
            if c5_cur >= best_c5 + 1e-12:
                break
            for direction in [-1, 1]:
                h_tr = h.copy()
                h_tr[i] += direction * 0.5
                h_tr2 = project_feasible(h_tr, n)
                c5_tr, _ = compute_c5(h_tr2)
                if c5_tr < c5_cur - 1e-14:
                    h = h_tr2
                    c5_cur = c5_tr
                    improved = True
                    break
            # Small fractional moves
            for direction in [-1, 1]:
                for eps in [0.01, 0.05, 0.1]:
                    h_tr = h.copy()
                    h_tr[i] += direction * eps
                    if h_tr[i] < 0 or h_tr[i] > 1:
                        continue
                    h_tr2 = project_feasible(h_tr, n)
                    c5_tr, _ = compute_c5(h_tr2)
                    if c5_tr < c5_cur - 1e-14:
                        h = h_tr2
                        c5_cur = c5_tr
                        improved = True
                        break
        if c5_cur < best_c5:
            best_c5 = c5_cur
    return h, best_c5


def pair_swap_local(h, n, best_c5, max_rounds=20):
    dx = 2.0 / n
    h = h.copy()
    improved = True
    rounds = 0
    while improved and rounds < max_rounds:
        improved = False
        rounds += 1
        c5_cur, _ = compute_c5(h)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for i in ones:
            for j in zeros:
                h_tr = h.copy()
                h_tr[i] = 0.0
                h_tr[j] = 1.0
                c5_tr, _ = compute_c5(h_tr)
                if c5_tr < c5_cur - 1e-14:
                    h = h_tr
                    c5_cur = c5_tr
                    improved = True
                    if c5_cur < best_c5:
                        best_c5 = c5_cur
                    break
            if improved:
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
    return h, best_c5


def simulated_annealing(h, n, best_c5, t0, budget, seed, max_iter=50000):
    rng = np.random.RandomState(seed)
    h = h.copy()
    c5_cur, _ = compute_c5(h)
    T = 0.002
    T_min = 0.00001
    cooling = 0.9995
    n_steps = 0
    while n_steps < max_iter:
        if time_left(t0, budget) < 5:
            break
        n_steps += 1
        move_type = rng.randint(4)
        h_tr = h.copy()
        if move_type == 0:
            i = rng.randint(n)
            h_tr[i] = 1.0 - h[i]
            h_tr = project_feasible(h_tr, n)
        elif move_type == 1:
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                tmp = h[i]
                h_tr[i] = h[j]
                h_tr[j] = tmp
        elif move_type == 2:
            i = rng.randint(n)
            d = rng.uniform(0.01, 0.2)
            h_tr[i] += d if h_tr[i] < 1 else -d
            h_tr[i] = np.clip(h_tr[i], 0, 1)
            h_tr = project_feasible(h_tr, n)
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                h_tr[i] -= 0.1
                h_tr[j] += 0.1
                h_tr = project_feasible(h_tr, n)
        if np.allclose(h_tr, h):
            continue
        c5_tr, _ = compute_c5(h_tr)
        delta = c5_tr - c5_cur
        if delta < -1e-14:
            h = h_tr
            c5_cur = c5_tr
            if c5_cur < best_c5:
                best_c5 = c5_cur
        else:
            if T > T_min and rng.rand() < np.exp(-delta / T):
                h = h_tr
                c5_cur = c5_tr
        T = max(T * cooling, T_min)
    return h, best_c5


def high_res_attempt(src_h, src_n, dst_n, t0, budget):
    if time_left(t0, budget) < 60:
        return None, float('inf'), dst_n
    h = interpolate_to(src_h, src_n, dst_n)
    h = project_feasible(h, dst_n)
    c5, corr = compute_c5(h)
    active = find_active(corr, m=20, gap=1.0)
    bounds = [(0.0, 1.0)] * dst_n
    for beta in [500, 2000, 8000]:
        if time_left(t0, budget) < 15:
            break
        h = slsqp_round(h, dst_n, active, beta, 200)
        c5, corr = compute_c5(h)
        active = find_active(corr, m=20, gap=1.0)
    return h, c5, dst_n


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5, _ = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 160
        best_h = make_thue_morse(best_n)
        best_h = project_feasible(best_h, best_n)
        best_c5, _ = compute_c5(best_h)
        print("No parent: n={}, C5={:.8f}".format(best_n, best_c5))

    # Phase 1: Multi-resolution exploration
    res_list = [best_n]
    for rn in [160, 200, 240, 280, 320, 400, 480, 560, 640]:
        if rn != best_n:
            res_list.append(rn)
    res_list = sorted(set(res_list))

    for rn in res_list:
        if time_left(t0, budget) < 80:
            break
        if rn == best_n:
            continue
        ht, ct, _ = high_res_attempt(best_h, best_n, rn, t0, budget)
        if ht is not None and ct < best_c5:
            best_c5 = ct
            best_h = ht
            best_n = rn
            print("  Res n={}: C5={:.8f}".format(rn, best_c5))

    # Also try structured inits at best_n
    n = best_n
    for pat_fn_name in ['thue_morse', 'stream', 'random']:
        if time_left(t0, budget) < 30:
            break
        if pat_fn_name == 'thue_morse':
            hp = make_thue_morse(n)
        elif pat_fn_name == 'stream':
            hp = make_stream(n)
        else:
            hp = make_binary_random(n, rng)
        hp = project_feasible(hp, n)
        cp, _ = compute_c5(hp)
        if cp < best_c5:
            best_c5 = cp
            best_h = hp
            print("  Structured init: C5={:.8f}".format(cp))

    # Phase 2: Main SLSQP optimization loop
    n = best_n
    h = best_h.copy()
    bounds = [(0.0, 1.0)] * n
    cons_fn, cons_jac = make_constraint(n)
    cons = [{'type': 'eq', 'fun': cons_fn, 'jac': cons_jac}]

    outer = 0
    while time_left(t0, budget) > 120:
        outer += 1
        c5_cur, corr = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()

        # Adaptive active set
        gap_decrease = max(0.3, 1.5 - outer * 0.15)
        m_active = min(30, 10 + outer * 2)
        active = find_active(corr, m=m_active, gap=gap_decrease)
        active = np.array(sorted(active))

        improved = False

        # SLSQP rounds with increasing beta
        for beta in [300, 1000, 3000, 10000, 30000]:
            if time_left(t0, budget) < 20:
                break
            obj_fn, obj_grad_fn = make_obj_smooth(beta, active, n)
            try:
                res = minimize(obj_grad_fn, h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-18, 'iprint': 0})
                if hasattr(res, 'x') and res.x is not None:
                    h_new = project_feasible(res.x, n)
                    c5_new, _ = compute_c5(h_new)
                    if c5_new < best_c5 - 1e-14:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h = h_new.copy()
                        improved = True
            except Exception:
                pass

        # Coordinate descent
        if time_left(t0, budget) > 40:
            h, best_c5 = coordinate_descent(h, n, best_c5, max_rounds=10)
            if np.any(h != best_h):
                improved = True

        if improved:
            print("  Outer {}: C5={:.8f} (active={}, beta=30000)".format(outer, best_c5, len(active)))
        else:
            if time_left(t0, budget) > 60:
                # Try full-set SLSQP
                _active_all = np.arange(2 * n - 1)
                for beta_f in [2000, 8000]:
                    if time_left(t0, budget) < 15:
                        break
                    obj_fn_f, obj_grad_fn_f = make_obj_smooth(beta_f, _active_all, n)
                    try:
                        res = minimize(obj_grad_fn_f, h, method='SLSQP',
                                       bounds=bounds, constraints=cons,
                                       options={'maxiter': 200, 'ftol': 1e-18, 'iprint': 0})
                        h_new = project_feasible(res.x, n)
                        c5_new, _ = compute_c5(h_new)
                        if c5_new < best_c5 - 1e-14:
                            best_c5 = c5_new
                            best_h = h_new.copy()
                            h = h_new.copy()
                            improved = True
                    except Exception:
                        pass

            if not improved:
                break

        # Update h from best
        h = best_h.copy()

    # Phase 3: Simulated annealing
    if time_left(t0, budget) > 80:
        n = best_n
        h = best_h.copy()
        h, best_c5 = simulated_annealing(h, n, best_c5, t0, budget, seed, max_iter=30000)
        if h is not None:
            c5_sa, _ = compute_c5(h)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h.copy()
        print("  After SA: C5={:.8f}".format(best_c5))

    # Phase 4: Second round of SLSQP at potentially new best
    if time_left(t0, budget) > 40:
        n = best_n
        h = best_h.copy()
        c5_cur, corr = compute_c5(h)
        active = find_active(corr, m=15, gap=0.3)
        for beta in [5000, 20000, 50000]:
            if time_left(t0, budget) < 10:
                break
            obj_fn, obj_grad_fn = make_obj_smooth(beta, active, n)
            try:
                res = minimize(obj_grad_fn, h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-19, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new, _ = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
            except Exception:
                pass

    # Phase 5: Fine-grained local search (pair moves)
    if time_left(t0, budget) > 30:
        n = best_n
        h = best_h.copy()
        c5_cur, _ = compute_c5(h)
        for _round in range(5):
            if time_left(t0, budget) < 5:
                break
            improved_local = False
            ones_idx = np.where(h > 0.9)[0]
            zeros_idx = np.where(h < 0.1)[0]
            frac_idx = np.where((h > 0.1) & (h < 0.9))[0]
            for i in frac_idx:
                for target_val in [0.0, 1.0, 0.5]:
                    h_tr = h.copy()
                    h_tr[i] = target_val
                    h_tr = project_feasible(h_tr, n)
                    c5_tr, _ = compute_c5(h_tr)
                    if c5_tr < c5_cur - 1e-13:
                        h = h_tr
                        c5_cur = c5_tr
                        improved_local = True
                        if c5_cur < best_c5:
                            best_c5 = c5_cur
                            best_h = h.copy()
                if improved_local:
                    break
            if not improved_local:
                break
        print("  After fine local: C5={:.8f}".format(best_c5))

    # Phase 6: Try one more resolution jump if time allows
    if time_left(t0, budget) > 50:
        for dn in [40, 80, -40]:
            if time_left(t0, budget) < 30:
                break
            nt = best_n + dn
            if nt < 100 or nt > 700:
                continue
            ht, ct, _ = high_res_attempt(best_h, best_n, nt, t0, budget)
            if ht is not None and ct < best_c5:
                best_c5 = ct
                best_h = ht
                best_n = nt
                print("  Final res n={}: C5={:.8f}".format(nt, best_c5))

    # Phase 7: Final L-BFGS-B polish with penalty
    if time_left(t0, budget) > 20:
        n = best_n
        h = best_h.copy()
        c5_cur, corr = compute_c5(h)
        active = find_active(corr, m=15, gap=0.5)

        def lbfgs_obj(x):
            corr_l = np.correlate(x, 1.0 - x, mode='full')
            vals = corr_l[active]
            mm = np.max(vals)
            beta_l = 5000.0
            exps = np.exp(beta_l * (vals - mm))
            sme = mm + np.log(np.sum(exps)) / beta_l
            pen = 1000.0 * (np.sum(x) - n / 2.0) ** 2
            return sme * (2.0 / n) + pen

        try:
            res = minimize(lbfgs_obj, h, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 5000, 'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c5_new, _ = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  L-BFGS-B: C5={:.8f}".format(best_c5))
        except Exception:
            pass

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5, _ = compute_c5(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
