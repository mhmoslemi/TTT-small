```python
def fast_smooth_max_grad(x, beta, top_k=20):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    # Get top-k indices
    idx = np.argpartition(corr, -top_k)[-top_k:]
    corr_top = corr[idx]
    m = np.max(corr_top)
    W = np.exp(beta * (corr_top - m))
    W /= np.sum(W)
    
    grad = np.zeros(n)
    for ki, j in enumerate(idx):
        wj = W[ki]
        if wj < 1e-12:
            continue
        s = j - (n - 1)
        if s >= 0:
            # corr[j] = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
            # d/dh[m] = (1-h[m-s]) for m in [s, n-1]  (from h[i+s] term)
            #           -h[m+s] for m in [0, n-1-s]  (from 1-h[i] term)
            grad[s:] += wj * (1.0 - x[:n-s])  # (1-h[m-s]) for m=s,...,n-1
            grad[:n-s] += wj * (-x[s:])  # -h[m+s] for m=0,...,n-1-s
        else:
            sp = -s
            # corr[j] = sum_{i=sp}^{n-1} h[i-sp]*(1-h[i])
            # d/dh[m] = (1-h[m+sp]) for m in [0, n-1-sp]  (from h[i-sp] term)
            #           -h[m-sp] for m in [sp, n-1]  (from 1-h[i] term)
            grad[:n-sp] += wj * (1.0 - x[sp:])  # (1-h[m+sp]) for m=0,...,n-1-sp
            grad[sp:] += wj * (-x[:n-sp])  # -h[m-sp] for m=sp,...,n-1
    
    grad *= dx
    return grad
```

```python
def _coord_descent(h, n, n_sweeps, rng, deadline):
    best_c5 = _compute_c5(h)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.time() > deadline:
                break
            old_v = h[i]
            best_v = old_v
            best_local = best_c5
            h[i] = old_v
            cands = [0.0, 1.0, 0.5, max(0.0, old_v - 0.25), min(1.0, old_v + 0.25)]
            if old_v > 0.35:
                cands.append(0.15)
            if old_v < 0.65:
                cands.append(0.85)
            cands.append(max(0.0, min(1.0, old_v - 0.08)))
            cands.append(max(0.0, min(1.0, old_v + 0.08)))
            for v in cands:
                if abs(v - old_v) < 1e-9:
                    continue
                h[i] = v
                h_p = _project(h, n)
                c = _compute_c5(h_p)
                if c < best_local:
                    best_local = c
                    best_v = v
                    h = h_p.copy()
            h[i] = best_v
            h = _project(h, n)
            if best_local < best_c5:
                best_c5 = best_local
                improved = True
        if not improved:
            break
    return h, best_c5

def _interp(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)

def _thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    return h

def _golden(n):
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        fr = (i * gr) % 1.0
        h[i] = 1.0 if 0.25 < fr < 0.75 else 0.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s
    rng = np.random.RandomState(seed)

    def tl():
        return max(0.0, budget_s - (time.time() - t_start) - 3.0)

    # Initialize from parent
    best_h = np.array(initial_h_values, dtype=float).copy()
    best_n = len(best_h)
    best_c5 = _compute_c5(best_h)
    print("Parent: n={}, C5={:.7f}".format(best_n, best_c5))

    # Phase 1: Binary SA on parent resolution
    if tl() > 60:
        n = best_n
        h_bin = np.where(best_h > 0.5, 1.0, 0.0)
        h_bin = _project(h_bin, n)
        c5_bin = _compute_c5(h_bin)
        h_bin, c5_bin = _sa_binary(h_bin, n, 80000, rng, 0.003, 0.00001, t_start + tl())
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            best_n = n
            print("SA1: C5={:.7f}".format(best_c5))

    # Phase 2: Continuous optimization with multiple betas
    if tl() > 40:
        n = best_n
        tgt = n / 2.0
        bounds = [(0.0, 1.0)] * n
        h_work = best_h.copy()
        for beta_val in [100, 300, 1000, 3000]:
            if tl() < 10:
                break
            pen = 5000.0
            try:
                res = minimize(_make_beta_obj(beta_val, tgt, pen), h_work,
                               jac=_make_beta_grad(beta_val, tgt, pen, 40),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 5000, 'ftol': 1e-15,
                                        'gtol': 1e-10, 'iprint': 0, 'maxls': 60})
                h_new = _project(res.x, n)
                c5_new = _compute_c5(h_new)
                if c5_new < _compute_c5(h_work):
                    h_work = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_work.copy()
                    best_n = n
                    print("LBFGS beta={}: C5={:.7f}".format(beta_val, best_c5))
            except Exception:
                pass
        # Coordinate descent polish
        if tl() > 15:
            h_work, c5_cd = _coord_descent(h_work, n, 15, rng, t_start + tl())
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_work.copy()
                print("CD: C5={:.7f}".format(best_c5))

    # Phase 3: Try different resolutions
    for n_trial in [200, 240, 320]:
        if tl() < 50:
            break
        print("--- n={} ---".format(n_trial))
        tgt = n_trial / 2.0
        bounds = [(0.0, 1.0)] * n_trial

        # Interpolate best to new resolution
        h_try = _interp(best_h, best_n, n_trial)
        h_try = _project(h_try, n_trial)
        c5_try = _compute_c5(h_try)

        # Also try structured inits
        h_tm = _project(_thue_morse(n_trial), n_trial)
        h_tm_c5 = _compute_c5(h_tm)
        h_gr = _project(_golden(n_trial), n_trial)
        h_gr_c5 = _compute_c5(h_gr)
        h_rand = np.zeros(n_trial)
        h_rand[rng.permutation(n_trial)[:n_trial // 2]] = 1.0

        inits = [h_try, h_tm, h_gr, h_rand]

        for h0 in inits:
            if tl() < 10:
                break
            c5_0 = _compute_c5(h0)
            if c5_0 > best_c5 * 1.05:
                continue
            # Binary SA
            if tl() > 20:
                h_sa, c5_sa = _sa_binary(h0, n_trial, 30000, rng, 0.002, 0.000005, t_start + tl())
                if c5_sa < c5_0:
                    h0 = h_sa
                    c5_0 = c5_sa
            # L-BFGS
            for bv in [200, 1500]:
                if tl() < 8:
                    break
                try:
                    res = minimize(_make_beta_obj(bv, tgt, 5000.0), h0,
                                   jac=_make_beta_grad(bv, tgt, 5000.0, 40),
                                   method='L-BFGS-B', bounds=bounds,
                                   options={'maxiter': 3000, 'ftol': 1e-15, 'iprint': 0})
                    h_new = _project(res.x, n_trial)
                    c5_new = _compute_c5(h_new)
                    if c5_new < c5_0:
                        h0 = h_new
                        c5_0 = c5_new
                except Exception:
                    pass
            if c5_0 < best_c5:
                best_c5 = c5_0
                best_h = h0.copy()
                best_n = n_trial
                print("  n={}: C5={:.7f}".format(n_trial, best_c5))

    # Phase 4: Final intensive polish on best
    n = best_n
    tgt = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h_work = best_h.copy()

    if tl() > 30:
        for beta_val in [5000, 10000, 20000]:
            if tl() < 8:
                break
            try:
                res = minimize(_make_beta_obj(beta_val, tgt, 10000.0), h_work,
                               jac=_make_beta_grad(beta_val, tgt, 10000.0, 50),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 8000, 'ftol': 1e-16, 'gtol': 1e-11,
                                        'iprint': 0, 'maxls': 80})
                h_new = _project(res.x, n)
                c5_new = _compute_c5(h_new)
                if c5_new < _compute_c5(h_work):
                    h_work = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_work.copy()
                    print("Polish beta={}: C5={:.7f}".format(beta_val, best_c5))
            except Exception:
                pass

    if tl() > 10:
        h_work, c5_cd = _coord_descent(h_work, n, 25, rng, t_start + tl())
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_work.copy()
            print("Final CD: C5={:.7f}".format(best_c5))

    # Phase 5: One more SA round on the polished continuous solution
    if tl() > 20:
        h_bin = np.where(best_h > 0.5, 1.0, 0.0)
        h_bin = _project(h_bin, best_n)
        h_sa, c5_sa = _sa_binary(h_bin, best_n, 50000, rng, 0.001, 0.000001, t_start + tl())
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            print("SA2: C5={:.7f}".format(best_c5))

    # Final validation
    best_h = _project(best_h, best_n)
    final_c5 = _compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    print("FINAL: n={}, C5={:.7f}".format(best_n, best_c5))
    return (best_h, best_c5, best_n)
```
