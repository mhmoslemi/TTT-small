    return np.concatenate([C[-(n - 1):], C[:n]])


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=np.float64), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mid = np.where((h > 0.05) & (h < 0.95))[0]
        if len(mid) > 0:
            h[mid[0]] += diff
        else:
            h[int(np.argmax(h > 0.5))] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500.0, 0.0)
    exps = np.exp(shifted)
    return (m + np.log(np.sum(exps)) / beta)


def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 0.0)
    exps = np.exp(shifted)
    total = np.sum(exps)
    if total < 1e-300:
        return np.ones(len(corr)) / len(corr)
    return exps / total


def obj_smooth_max(h, beta):
    corr = fast_full_corr(h, 1.0 - h)
    return smooth_max_val(corr, beta) * (2.0 / len(h))


def grad_smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = fast_full_corr(h, 1.0 - h)
    w = smooth_max_weights(corr, beta)
    grad = np.zeros(n)
    w0 = w[n - 1]
    if w0 > 1e-12:
        grad += w0 * (1.0 - 2.0 * h)
    for j in range(1, n):
        wj = w[n - 1 + j] + w[n - 1 - j]
        if wj < 1e-12:
            continue
        grad[j:] += wj * (1.0 - h[:n - j])
        grad[:n - j] -= wj * h[j:]
    return dx * grad


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = 0
        v = i
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return h


def make_golden_ratio(n):
    h = np.zeros(n)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return h


def make_random_bin(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    half = n // 2
    h[idx[:half]] = 1.0
    return h


def make_m_sequence(n):
    """M-sequence based initialization (good autocorrelation properties)"""
    h = np.zeros(n)
    period = 1
    while period + 1 < n:
        period *= 2
    tm = make_thue_morse(period + 1)
    h[:min(n, len(tm))] = tm[:min(n, len(tm))]
    return h


def optimize_lbfgsb(h0, n, beta, max_iter, time_limit):
    """Optimize using L-BFGS-B with penalty for sum constraint"""
    target = n / 2.0
    pen_w = 100.0

    def obj(z):
        h = z
        sm = obj_smooth_max(h, beta)
        pen = pen_w * (np.sum(h) - target) ** 2
        return sm + pen

    def grad(z):
        g = grad_smooth_max(z, beta)
        g += 2.0 * pen_w * (np.sum(z) - target) * np.ones(n)
        return g

    bounds = [(0.0, 1.0)] * n
    t0 = time.time()
    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-15,
                                'gtol': 1e-10, 'maxfun': 5000,
                                'iprint': 0})
        h_new = project_feasible(res.x, n)
    except Exception:
        h_new = h0
    if time.time() - t0 > time_limit:
        return h0
    return h_new


def optimize_slsqp(h0, n, beta, max_iter):
    """Optimize using SLSQP with constraint"""
    target = n / 2.0
    dx = 2.0 / n

    def obj(z):
        return obj_smooth_max(z, beta)

    def grad(z):
        return grad_smooth_max(z, beta)

    def con_fun(z):
        return np.sum(z) - target

    def con_grad(z):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_grad}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        h_new = project_feasible(res.x, n)
    except Exception:
        h_new = h0
    return h_new


def binary_swap_improve(h, n, max_sweeps=20, time_limit=60.0):
    """Try pairwise swaps and flips to improve binary-like solutions"""
    c5_best = compute_c5(h)
    h_best = h.copy()
    t0 = time.time()

    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        any_imp = False
        for i in range(n):
            for j in range(i + 1, n):
                if time.time() - t0 > time_limit:
                    break
                if h_best[i] == h_best[j]:
                    continue
                h_try = h_best.copy()
                h_try[i], h_try[j] = h_try[j], h_try[i]
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < c5_best - 1e-12:
                    c5_best = c5_try
                    h_best = h_try
                    any_imp = True
        if not any_imp:
            break
    return h_best, c5_best


def coordinate_refine(h, n, time_limit=30.0):
    """Fine coordinate-wise perturbation"""
    c5_best = compute_c5(h)
    h_best = h.copy()
    t0 = time.time()
    deltas = [0.02, -0.02, 0.05, -0.05, 0.1, -0.1]

    for sweep in range(50):
        if time.time() - t0 > time_limit:
            break
        any_imp = False
        for i in range(n):
            if time.time() - t0 > time_limit:
                break
            for d in deltas:
                trial = h_best.copy()
                trial[i] = np.clip(trial[i] + d, 0.0, 1.0)
                trial = project_feasible(trial, n)
                c5_try = compute_c5(trial)
                if c5_try < c5_best - 1e-12:
                    c5_best = c5_try
                    h_best = trial
                    any_imp = True
                    break
        if not any_imp and sweep > 5:
            break
    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _start_time, _budget
    _start_time = time.time()
    _budget = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = make_thue_morse(parent_n)
        parent_h = project_feasible(parent_h, parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("No parent. TM: n={}, C5={:.8f}".format(parent_n, best_c5))

    # Phase 1: Quick multi-resolution SLSQP
    print("\n=== Phase 1: Multi-resolution SLSQP ===")
    n_candidates = sorted(set([best_n, 128, 192, 256, 320, 384, 512]))
    n_candidates = [nn for nn in n_candidates if 64 <= nn <= 512]

    phase1_results = []
    for n in n_candidates:
        if time_left() < 20.0:
            break
        print("  n={}...".format(n), end=" ", flush=True)

        # Multiple initializations
        h_interp = project_feasible(interp_to(best_h, best_n, n), n)
        inits = [h_interp]
        for _ in range(2):
            inits.append(project_feasible(make_random_bin(rng, n), n))
        inits.append(project_feasible(make_thue_morse(n), n))
        inits.append(project_feasible(make_golden_ratio(n), n))

        for h0 in inits:
            if time_left() < 10.0:
                break
            h_cur = h0.copy()
            c5_cur = compute_c5(h_cur)
            for beta in [20, 80, 300]:
                if time_left() < 5.0:
                    break
                h_new = optimize_slsqp(h_cur, n, beta, max_iter=200)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur:
                    h_cur = h_new
                    c5_cur = c5_new
            if c5_cur < best_c5 - 1e-10:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n
            phase1_results.append((c5_cur, n, h_cur.copy()))
        print("best so far: {:.8f}".format(best_c5))

    # Sort and pick top
    phase1_results.sort(key=lambda x: x[0])

    # Phase 2: L-BFGS-B at better resolutions (faster for large n)
    print("\n=== Phase 2: L-BFGS-B optimization ===")
    n_phase2 = sorted(set([best_n, 256, 384, 512, 640, 768, 1024]))
    n_phase2 = [nn for nn in n_phase2 if 128 <= nn <= 1024]

    # Get best starting point per resolution
    best_start = {}
    for c5_r, n_r, h_r in phase1_results:
        if n_r not in best_start or c5_r < best_start[n_r][0]:
            best_start[n_r] = (c5_r, h_r.copy())

    for n in n_phase2:
        if time_left() < 30.0:
            break
        print("  n={}...".format(n), end=" ", flush=True)

        # Get best starting point for this n
        if n in best_start:
            h_cur = best_start[n][1].copy()
        else:
            h_cur = project_feasible(interp_to(best_h, best_n, n), n)
        c5_cur = compute_c5(h_cur)

        # L-BFGS-B with increasing beta
        for beta in [30, 100, 300, 1000, 3000, 8000]:
            if time_left() < 15.0:
                break
            max_iter = 400 if beta < 500 else 200
            t_limit = min(40.0, time_left() - 5.0)
            if t_limit < 5.0:
                break
            h_new = optimize_lbfgsb(h_cur, n, beta, max_iter, t_limit)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-12:
                c5_cur = c5_new
                h_cur = h_new

        # SLSQP final polish at this resolution
        if time_left() > 15.0:
            for beta in [3000, 8000]:
                if time_left() < 8.0:
                    break
                h_new = optimize_slsqp(h_cur, n, beta, max_iter=300)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-12:
                    c5_cur = c5_new
                    h_cur = h_new

        if c5_cur < best_c5 - 1e-10:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n
            print("improved! C5={:.8f}".format(best_c5))
        else:
            print("C5={:.8f}".format(c5_cur))

        if time_left() > 30.0:
            best_start[n] = (c5_cur, h_cur.copy())

    # Phase 3: Very high beta refinement at best resolution
    print("\n=== Phase 3: High-beta refinement ===")
    n_best = best_n
    for beta in [5000, 10000, 20000, 50000]:
        if time_left() < 15.0:
            break
        print("  beta={}...".format(beta), end=" ", flush=True)
        h_new = optimize_lbfgsb(best_h, n_best, beta, max_iter=500,
                                time_limit=min(50.0, time_left() - 5.0))
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new
        print("C5={:.8f}".format(best_c5))

    # SLSQP with very high beta
    if time_left() > 15.0:
        for beta in [20000, 50000]:
            if time_left() < 10.0:
                break
            h_new = optimize_slsqp(best_h, n_best, beta, max_iter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new
        print("  After SLSQP: C5={:.8f}".format(best_c5))

    # Phase 4: Try alternative resolutions for improvement
    print("\n=== Phase 4: Alternative resolutions ===")
    alt_ns = [nn for nn in [320, 400, 480, 560, 640, 800, 960]
              if nn != best_n and 160 <= nn <= 960]
    for n in alt_ns[:4]:
        if time_left() < 30.0:
            break
        h_cur = project_feasible(interp_to(best_h, best_n, n), n)
        c5_cur = compute_c5(h_cur)
        # Quick L-BFGS-B
        for beta in [500, 2000, 10000]:
            if time_left() < 10.0:
                break
            t_limit = min(30.0, time_left() - 5.0)
            if t_limit < 5.0:
                break
            h_new = optimize_lbfgsb(h_cur, n, beta, max_iter=300, time_limit=t_limit)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-12:
                c5_cur = c5_new
                h_cur = h_new
        if c5_cur < best_c5 - 1e-10:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n
            print("  n={}: improved to {:.8f}".format(n, best_c5))

    # Phase 5: Binary swap improvement (works well for near-binary solutions)
    print("\n=== Phase 5: Binary swap / coordinate refinement ===")
    # Threshold to binary for swap search
    h_bin = (best_h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, best_n)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5 + 0.005:
        # Binary is close, try swaps
        t_limit = min(120.0, time_left() - 10.0)
        if t_limit > 10.0:
            h_sw, c5_sw = binary_swap_improve(h_bin, best_n,
                                              max_sweeps=30, time_limit=t_limit)
            if c5_sw < best_c5 - 1e-10:
                best_c5 = c5_sw
                best_h = h_sw.copy()
            print("  Binary swap: C5={:.8f}".format(c5_sw))

    # Coordinate refinement on the continuous solution
    if time_left() > 20.0:
        t_limit = min(60.0, time_left() - 10.0)
        h_coord, c5_coord = coordinate_refine(best_h, best_n, time_limit=t_limit)
        if c5_coord < best_c5 - 1e-10:
            best_c5 = c5_coord
            best_h = h_coord.copy()
        print("  Coord refine: C5={:.8f}".format(c5_coord))

    # Phase 6: Final very high-beta polish
    print("\n=== Phase 6: Final polish ===")
    if time_left() > 20.0:
        for beta in [50000, 100000, 200000]:
            if time_left() < 10.0:
                break
            n_cur = len(best_h)
            t_limit = min(40.0, time_left() - 5.0)
            if t_limit < 5.0:
                break
            h_new = optimize_lbfgsb(best_h, n_cur, beta, max_iter=500,
                                    time_limit=t_limit)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new
        # One more SLSQP
        if time_left() > 10.0:
            h_new = optimize_slsqp(best_h, len(best_h), 100000, max_iter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new
        print("  Final: C5={:.8f}".format(best_c5))

    # Phase 7: Try one more resolution jump
    if time_left() > 40.0 and best_n < 640:
        print("\n=== Phase 7: Higher resolution attempt ===")
        n_jump = min(best_n * 2, 1024)
        h_jump = project_feasible(interp_to(best_h, best_n, n_jump), n_jump)
        c5_jump = compute_c5(h_jump)
        for beta in [500, 5000, 50000]:
            if time_left() < 15.0:
                break
            t_limit = min(50.0, time_left() - 5.0)
            if t_limit < 5.0:
                break
            h_new = optimize_lbfgsb(h_jump, n_jump, beta, max_iter=400,
                                    time_limit=t_limit)
            c5_new = compute_c5(h_new)
            if c5_new < c5_jump - 1e-12:
                c5_jump = c5_new
                h_jump = h_new
        if c5_jump < best_c5 - 1e-10:
            best_c5 = c5_jump
            best_h = h_jump.copy()
            best_n = n_jump
            print("  Jump to n={}: C5={:.8f}".format(n_jump, best_c5))

    # Final: validate and ensure exact feasibility
    best_h = project_feasible(best_h, len(best_h))
    # Verify sum is exact
    n_final = len(best_h)
    target = n_final / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-14:
        # Distribute among interior points
        mask = (best_h > 0.01) & (best_h < 0.99)
        if np.sum(mask) > 0:
            best_h[mask] += diff / np.sum(mask)
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(len(best_h), final_c5))
    return (best_h, final_c5, len(best_h))
