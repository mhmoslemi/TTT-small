```python
def smooth_max_grad_vec(x, n, beta):
    dx = 2.0 / n
    one_minus = 1.0 - x
    corr = np.correlate(x, one_minus, mode='full')
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    weights /= np.sum(weights)
    
    # For each shift s, compute the gradient contribution
    # grad[m] += w_s * (1-x[m+s]) for valid m (where m is first index)
    # grad[m] += w_s * (-x[m-s]) for valid m (where m is second index)
    
    grad = np.zeros(n)
    
    for s in range(-n+1, n):
        w = weights[s + n - 1]
        if w < 1e-14:
            continue
        if s >= 0:
            # C_s = sum_{i=0}^{n-1-s} x[i]*(1-x[i+s])
            # d/dx[m]: first part gives (1-x[m+s]) for m in [0, n-1-s]
            #           second part gives -x[m-s] for m in [s, n-1]
            if s == 0:
                grad += w * (1.0 - 2.0 * x)
            else:
                idx_first = np.arange(0, n - s)
                grad[idx_first] += w * (1.0 - x[idx_first + s])
                idx_second = np.arange(s, n)
                grad[idx_second] -= w * x[idx_second - s]
        else:
            sp = -s
            # C_{-sp} = sum_{i=sp}^{n-1} x[i]*(1-x[i-sp])
            # d/dx[m]: first part gives (1-x[m-sp]) for m in [sp, n-1]
            #           second part gives -x[m+sp] for m in [0, n-1-sp]
            idx_first = np.arange(sp, n)
            grad[idx_first] += w * (1.0 - x[idx_first - sp])
            idx_second = np.arange(0, n - sp)
            grad[idx_second] -= w * x[idx_second + sp]
    
    grad *= dx
    return grad
```

```python
def smooth_max_grad(x, n, beta, tgt):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w /= np.sum(w)

    grad = np.zeros(n)
    for s in range(-n + 1, n):
        ws = w[s + n - 1]
        if ws < 1e-14:
            continue
        if s == 0:
            grad += ws * (1.0 - 2.0 * x)
        elif s > 0:
            idx1 = np.arange(0, n - s)
            grad[idx1] += ws * (1.0 - x[idx1 + s])
            idx2 = np.arange(s, n)
            grad[idx2] -= ws * x[idx2 - s]
        else:
            sp = -s
            idx1 = np.arange(sp, n)
            grad[idx1] += ws * (1.0 - x[idx1 - sp])
            idx2 = np.arange(0, n - sp)
            grad[idx2] -= ws * x[idx2 + sp]

    grad *= dx
    grad += 2.0 * 1e6 * (np.sum(x) - tgt)
    return grad


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    return h


def make_golden(n):
    h = np.zeros(n)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if abs(frac - 0.5) < 0.25 else 0.0
    return h


def make_vdc(n):
    h = np.zeros(n)
    for i in range(n):
        v = i + 1
        r = 0
        for _ in range(24):
            r = (r << 1) | (v & 1)
            v >>= 1
        h[i] = 1.0 if (r & 3) == 0 or (r & 3) == 1 else 0.0
    return h


def make_sine(n, freq, phase):
    t = np.arange(n) / float(n)
    h = 0.5 + 0.3 * np.sin(2.0 * np.pi * freq * t + phase)
    return np.clip(h, 0.0, 1.0)


def make_multi_sine(n, rng):
    t = np.arange(n) / float(n)
    h = 0.5 + 0.15 * np.sin(2.0 * np.pi * 3.0 * t + rng.rand() * 6.28)
    h += 0.12 * np.sin(2.0 * np.pi * 7.0 * t + rng.rand() * 6.28)
    h += 0.08 * np.sin(2.0 * np.pi * 11.0 * t + rng.rand() * 6.28)
    return np.clip(h, 0.0, 1.0)


def binary_local_search(h, n, max_passes=5, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h = np.where(h > 0.5, 1.0, 0.0).copy()
    tgt = n // 2
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) > tgt:
        to_remove = rng.choice(len(ones), len(ones) - tgt, replace=False)
        h[ones[to_remove]] = 0.0
    elif len(ones) < tgt:
        to_add = rng.choice(len(zeros), tgt - len(ones), replace=False)
        h[zeros[to_add]] = 1.0
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for p in range(max_passes):
        if remaining_time() < 3:
            break
        improved = False
        for oi in range(len(ones)):
            if remaining_time() < 3:
                break
            pos_1 = ones[oi]
            for oz in range(len(zeros)):
                pos_0 = zeros[oz]
                if pos_1 == pos_0:
                    continue
                h[pos_1] = 0.0
                h[pos_0] = 1.0
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    best_h = h.copy()
                    ones[oi] = pos_0
                    zeros[oz] = pos_1
                    improved = True
                else:
                    h[pos_1] = 1.0
                    h[pos_0] = 0.0
        if not improved:
            break
    return best_h, best_c5


def hybrid_local_search(h, n, rng, max_iter=200):
    """Allow continuous values, try small perturbations on each coordinate."""
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        if remaining_time() < 3:
            break
        idx = rng.randint(n)
        old = h[idx]
        best_local = best_c5
        best_val = old
        candidates = [0.0, 1.0, 0.5, max(0.0, old - 0.25), min(1.0, old + 0.25)]
        if old > 0.2:
            candidates.append(0.1)
        if old < 0.8:
            candidates.append(0.9)
        h_save = h.copy()
        for cv in candidates:
            if abs(cv - old) < 1e-9:
                continue
            h[idx] = cv
            h_feas = project_feasible(h, n)
            c5_try = compute_c5(h_feas)
            if c5_try < best_local:
                best_local = c5_try
                best_val = cv
                h = h_feas.copy()
        if best_val == old:
            h = h_save.copy()
        else:
            best_c5 = best_local
            h = project_feasible(h, n)
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.7f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 160
        best_h = make_thue_morse(best_n)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)
        print("No parent. Thue-Morse: n={}, C5={:.7f}".format(best_n, best_c5))

    n_list = sorted(set([best_n, 120, 140, 160, 170, 180, 200, 240, 280]))
    n_list = [nn for nn in n_list if 80 <= nn <= 320]

    for n in n_list:
        if remaining_time() < 20:
            break
        tgt = n / 2.0
        bounds = [(0.0, 1.0)] * n
        print("\n=== n={} ===".format(n))

        # Build initializations
        inits = []
        h_interp = project_feasible(interp_to(best_h, best_n, n), n)
        inits.append(h_interp)
        inits.append(project_feasible(make_thue_morse(n), n))
        inits.append(project_feasible(make_golden(n), n))
        inits.append(project_feasible(make_vdc(n), n))
        inits.append(project_feasible(make_sine(n, 3, rng.rand() * 6.28), n))
        inits.append(project_feasible(make_multi_sine(n, rng), n))
        # Random binary
        for _ in range(2):
            hb = np.zeros(n)
            idx = rng.permutation(n)
            hb[idx[:n // 2]] = 1.0
            inits.append(hb)

        for h0 in inits:
            if remaining_time() < 8:
                break
            c5_cur = compute_c5(h0)
            if c5_cur > best_c5 * 1.05 and c5_cur > 0.42:
                continue

            # Multi-beta L-BFGS-B
            for beta_val in [30, 100, 300, 800, 2000]:
                if remaining_time() < 5:
                    break
                try:
                    res = minimize(
                        lambda x: smooth_max_obj(x, n, beta_val, tgt),
                        h0,
                        jac=lambda x: smooth_max_grad(x, n, beta_val, tgt),
                        method='L-BFGS-B',
                        bounds=bounds,
                        options={'maxiter': 600, 'ftol': 1e-15,
                                 'gtol': 1e-10, 'iprint': 0,
                                 'maxls': 80, 'maxfun': 8000}
                    )
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur:
                        h0 = h_new
                        c5_cur = c5_new
                except Exception:
                    pass

            # Binary local search
            if remaining_time() > 10:
                h_bin = np.where(h0 > 0.5, 1.0, 0.0)
                h_bin = project_feasible(h_bin, n)
                h_b, c5_b = binary_local_search(h_bin, n, max_passes=6, rng=rng)
                if c5_b < c5_cur:
                    h0 = h_b
                    c5_cur = c5_b

            # Polish with very high beta
            if remaining_time() > 5:
                for beta_val in [3000, 8000]:
                    if remaining_time() < 4:
                        break
                    try:
                        res = minimize(
                            lambda x: smooth_max_obj(x, n, beta_val, tgt),
                            h0,
                            jac=lambda x: smooth_max_grad(x, n, beta_val, tgt),
                            method='L-BFGS-B',
                            bounds=bounds,
                            options={'maxiter': 1000, 'ftol': 1e-16,
                                     'gtol': 1e-11, 'iprint': 0,
                                     'maxls': 100, 'maxfun': 10000}
                        )
                        h_new = project_feasible(res.x, n)
                        c5_new = compute_c5(h_new)
                        if c5_new < c5_cur:
                            h0 = h_new
                            c5_cur = c5_new
                    except Exception:
                        pass

            # Hybrid local search (continuous coordinate descent)
            if remaining_time() > 15:
                h0, c5_cur = hybrid_local_search(h0, n, rng, max_iter=300)

            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h0.copy()
                best_n = n
                print("  Improved: n={}, C5={:.7f}".format(n, c5_cur))

    # Intensive final polish on best solution
    n_final = best_n
    tgt = n_final / 2.0
    bounds = [(0.0, 1.0)] * n_final
    h = best_h.copy()

    print("\n=== Final polish (n={}) ===".format(n_final))
    for beta_val in [2000, 5000, 10000, 20000]:
        if remaining_time() < 5:
            break
        try:
            res = minimize(
                lambda x: smooth_max_obj(x, n_final, beta_val, tgt),
                h,
                jac=lambda x: smooth_max_grad(x, n_final, beta_val, tgt),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 2000, 'ftol': 1e-17,
                         'gtol': 1e-12, 'iprint': 0,
                         'maxls': 100, 'maxfun': 20000}
            )
            h_new = project_feasible(res.x, n_final)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
                print("  Beta={}: C5={:.7f}".format(beta_val, best_c5))
        except Exception:
            pass

    # Final hybrid local search
    if remaining_time() > 10:
        h, c5_hs = hybrid_local_search(h, n_final, rng, max_iter=500)
        if c5_hs < best_c5:
            best_c5 = c5_hs
            best_h = h.copy()
            print("  Hybrid: C5={:.7f}".format(best_c5))

    # Final binary search + continuous polish on best
    if remaining_time() > 15:
        h_bin = np.where(best_h > 0.5, 1.0, 0.0)
        h_bin, c5_bin = binary_local_search(h_bin, n_final, max_passes=8, rng=rng)
        if remaining_time() > 8:
            for beta_val in [5000, 15000]:
                if remaining_time() < 5:
                    break
                try:
                    res = minimize(
                        lambda x: smooth_max_obj(x, n_final, beta_val, tgt),
                        h_bin,
                        jac=lambda x: smooth_max_grad(x, n_final, beta_val, tgt),
                        method='L-BFGS-B',
                        bounds=bounds,
                        options={'maxiter': 2000, 'ftol': 1e-17,
                                 'gtol': 1e-12, 'iprint': 0}
                    )
                    h_new = project_feasible(res.x, n_final)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        print("  Bin+Polish: C5={:.7f}".format(best_c5))
                except Exception:
                    pass
        elif c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
```
