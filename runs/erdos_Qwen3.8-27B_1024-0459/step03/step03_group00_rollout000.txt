    same_streak = 0

    while _time_left() > max(25.0, 0.08 * time_limit):
        batch += 1
        if batch % 128 == 0:
            temperature *= 0.94
            same_streak = max(0, same_streak - 1)

        bsize = 240
        best_delta = 10 ** 9
        best_Bc = None
        best_p = -1
        best_q = -1

        for _ in range(bsize):
            if len(ones) == 0 or len(zeros) == 0:
                break
            iq = int(rng.randint(len(ones)))
            ip = int(rng.randint(len(zeros)))
            q = int(ones[iq])
            p = int(zeros[ip])
            if p == q:
                continue
            val, Bc = _swap_candidate_value(B, bits, p, q, n)
            delta = val - cur
            if delta < best_delta:
                best_delta = delta
                best_Bc = Bc
                best_p = p
                best_q = q

        if best_delta < 0:
            bits[best_p] = 1
            bits[best_q] = 0
            ones[iq_b] = best_q
            zeros[ip_b] = best_p
            B = best_Bc
            cur = int(np.max(B))
            same_streak = 0
        else:
            accept = False
            if best_delta == 0:
                accept = True
            else:
                if temperature > 1e-4:
                    prob = np.exp(-float(best_delta) / max(1e-6, temperature))
                    accept = (rng.rand() < prob)
                    same_streak += 1
                else:
                    same_streak += 1

            if accept and best_Bc is not None:
                bits[best_p] = 1
                bits[best_q] = 0
                # Update list entries robustly.
                if best_p in zeros:
                    zi = zeros.index(best_p)
                    zeros[zi] = best_q
                if best_q in ones:
                    oi = ones.index(best_q)
                    ones[oi] = best_p
                B = best_Bc
                cur = int(np.max(B))

            if same_streak > 31:
                # Random shake: perform a feasible random restart around current.
                mix = 0.25
                vals = bits.astype(float)
                vals = vals * (1.0 - mix) + (0.5 + rng.normal(0.0, 0.18, size=n)) * mix
                newbits = _binary_from_values(vals, n)
                newB = _bits_to_counts(newbits, n)
                newcur = int(np.max(newB))
                if newcur <= cur + 2:
                    bits = newbits
                    B = newB
                    cur = newcur
                    ones = list(np.flatnonzero(bits == 1))
                    zeros = list(np.flatnonzero(bits == 0))
                    same_streak = 0

        if (time.time() - t0) > time_limit:
            break

    c5 = (2.0 / n) * cur
    return bits, B, cur, c5


def _continuous_polish(h0, n, rng, time_limit):
    h = _project_feasible(h0, n)
    cur = _c5_count(h)
    idx0 = None
    best = h.copy()
    best_c = cur

    settings = []
    for beta in [8.0, 18.0, 40.0, 90.0, 180.0, 350.0, 700.0, 1400.0, 2800.0]:
        for maxiter in [70, 120, 180]:
            settings.append((beta, maxiter))

    done = set()
    for beta, maxiter in settings:
        if _time_left() < 12.0:
            break
        key = (beta, maxiter)
        if key in done:
            continue
        done.add(key)

        _G_N = n
        _G_DX = 2.0 / float(n)
        _G_BETA = float(beta)
        bounds = [(0.0, 1.0)] * n
        cons = {'type': 'eq', 'fun': _con_sum, 'jac': lambda x: np.ones(n)}

        try:
            res = minimize(
                _smooth_obj,
                h,
                jac=_smooth_jac,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0}
            )
            h2 = _project_feasible(res.x, n)
            c2 = _c5_count(h2)
            if c2 < cur - 1e-12:
                h = h2
                cur = c2
            else:
                h2b = _project_feasible(h + 0.05 * rng.normal(0.0, 1.0, size=n), n)
                cb = _c5_count(h2b)
                if cb < cur:
                    h = h2b
                    cur = cb

            if cur < best_c - 1e-12:
                best_c = cur
                best = h.copy()
        except Exception:
            pass

    # coordinate-like pairwise continuous polish
    h = best.copy()
    cur = best_c
    for sweep in range(12 if n <= 384 else 7):
        if _time_left() < 15.0 or (time.time() - (_G_BUDGET - time_limit)) > time_limit:
            break
        order = rng.permutation(n)
        improved = False
        for i in order:
            if _time_left() < 10.0:
                break
            old = h[i]
            for j in range(0, n, 23):
                if j == i:
                    continue
                if _time_left() < 10.0:
                    break
                # move mass between i and j while keeping sum
                vi = h[i]
                vj = h[j]
                for eps in [0.05, -0.05, 0.12, -0.12, 0.22, -0.22]:
                    ni = vi + eps
                    nj = vj - eps
                    if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
                        continue
                    trial = h.copy()
                    trial[i] = np.clip(ni, 0.0, 1.0)
                    trial[j] = np.clip(nj, 0.0, 1.0)
                    trial = _project_feasible(trial, n)
                    cval = _c5_count(trial)
                    if cval < cur - 1e-12:
                        h = trial
                        cur = cval
                        improved = True
                        break
        if not improved:
            break

    if cur < best_c - 1e-12:
        best = h.copy()
        best_c = cur

    return best, best_c


def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(int(seed))

    best_h = None
    best_c = float('inf')
    best_n = 0

    try:
        parent = np.asarray(initial_h_values, dtype=float)
        pn = int(len(parent))
        pc = _c5_count(parent)
        best_h = parent.copy()
        best_c = pc
        best_n = pn
    except Exception:
        pn = 256
        parent = np.zeros(pn)
        parent[rng.choice(pn, size=pn // 2, replace=False)] = 1.0
        best_h = parent.copy()
        best_c = _c5_count(parent)
        best_n = pn

    # Continuous polishing from many resolutions.
    cont_ns = sorted(set([
        best_n, 256, 288, 320, 352, 384, 416, 448, 480, 512, 576, 640
    ]))
    cont_ns = [nn for nn in cont_ns if 128 <= nn <= 704]

    polish_budget_total = max(120.0, 0.42 * _G_BUDGET)
    per_n = max(35.0, polish_budget_total / max(1, len(cont_ns)))

    for nn in cont_ns:
        if _time_left() < 45.0:
            break
        h0 = _project_feasible(_interp_to(best_h, best_n, nn), nn)
        # also try direct parent interpolation if parent differs
        if pn != nn:
            h0b = _project_feasible(_interp_to(parent, pn, nn), nn)
            if _c5_count(h0b) < _c5_count(h0):
                h0 = h0b

        cand, cc = _continuous_polish(h0, nn, rng, per_n)
        if cc < best_c - 1e-12:
            best_c, best_n, best_h = cc, nn, cand.copy()

    # Binary search from several inits/resolutions
    # The binary search uses exact integer counts; it can cross continuous local bars.
    bin_ns = sorted(set([best_n, 256, 320, 384, 448, 512, 640, 768, 896]))
    bin_ns = [nn for nn in bin_ns if 192 <= nn <= 896]

    inits_by_n = {}
    for nn in bin_ns:
        inits = []
        inits.append(_binary_from_values(best_h, nn))
        inits.append(_binary_from_values(parent, nn))
        inits.append(_thue_morse(nn))
        inits.append(_van_der_corput(nn))
        inits.append(_mech_bits(nn, 0.38196601125, rng.rand() * 2.0))
        inits.append(_mech_bits(nn, 0.42, rng.rand() * 2.0))
        inits.append(_sin_bits(nn, 3, rng.rand() * 2.0))
        inits.append(_sin_bits(nn, 7, rng.rand() * 2.0))
        inits.append(_random_bits(rng, nn))
        # keep unique
        uniq = []
        seen = set()
        for b in inits:
            key = (int(b.sum()), tuple(b.tolist()) if nn <= 384 else None)
            if key is not None and key in seen:
                continue
            uniq.append(b)
            seen.add(key)
        inits_by_n[nn] = uniq

    # Estimate how long one binary search will take by using time budgets.
    total_bin_budget = max(120.0, _time_left() - 60.0)
    weights = {}
    for nn in bin_ns:
        # Larger n is slower roughly n log n; bias smaller n for more coverage.
        weights[nn] = 1.0 / max(1.0, float(nn) / 256.0)
    tw = float(sum(weights.values()))
    for nn in bin_ns:
        if _time_left() < 40.0:
            break
        tlimit = max(25.0, (total_bin_budget * weights[nn] / max(1e-9, tw)))
        # Never spend beyond remaining safety
        tlimit = min(tlimit, max(25.0, _time_left() - 35.0))
        for init_bits in inits_by_n[nn]:
            if _time_left() < 30.0:
                break
            tleft = _time_left() - 35.0
            if tleft <= 10.0:
                break
            # If initial is already bad, give less time.
            B0 = _bits_to_counts(init_bits, nn)
            c0 = int(np.max(B0)) * (2.0 / nn)
            if c0 < best_c:
                ttl = tleft
            else:
                ttl = min(tleft, max(18.0, 0.45 * tlimit))

            bits, B, mc, cc = _binary_search(init_bits, nn, rng, ttl)
            if cc < best_c - 1e-12:
                best_c, best_n = cc, nn
                best_h = bits.astype(float)

    # Final continuous polish from the best binary / continuous hybrid.
    if _time_left() > 90.0 and best_h is not None:
        # Convert best binary to continuous if it improved, but also keep continuous n.
        fn = max(256, min(640, int(round(best_n))))
        if int(len(best_h)) != fn:
            fh = _project_feasible(_interp_to(best_h, int(len(best_h)), fn), fn)
        else:
            fh = _project_feasible(best_h, fn)
        # Make a light continuous version: clip but keep exactly n/2 sum.
        fh2 = 0.85 * fh + 0.15 * 0.5
        fh2 = _project_feasible(fh2, fn)
        if _c5_count(fh2) > _c5_count(fh):
            fh2 = fh
        cand, cc = _continuous_polish(fh2, fn, rng, max(45.0, min(260.0, _time_left() - 40.0)))
        if cc < best_c - 1e-12:
            best_c, best_n, best_h = cc, fn, cand.copy()

    # Validate and return
    best_h = _project_feasible(best_h, int(len(best_h)))
    final_c = _c5_count(best_h)

    try:
        ev = float(evaluate_erdos_solution(best_h))
        if ev < final_c:
            final_c = ev
    except Exception:
        pass

    return (best_h, float(final_c), int(len(best_h)))
