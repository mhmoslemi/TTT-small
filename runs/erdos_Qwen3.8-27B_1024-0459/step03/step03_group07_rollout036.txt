def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-16:

        np.clip(h, 0.0, 1.0, out=h)

            best_x = xs[gi]

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def coord_desc(h, n, sweeps=20, t0=None, time_limit=None):

def interp_h(h_src, n_src, n_dst):
    px = (np.arange(n_src) + 0.5) * 2.0 / n_src
    pxn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    return project_feasible(np.interp(pxn, px, h_src), n_dst)

def smooth_slsqp(h, n, beta, t0=None, time_limit=None):
    from scipy.optimize import minimize
    h0 = project_feasible(h, n).copy()
    best = h0.copy()
    best_c5 = compute_c5(h0)
    dx = 2.0 / n

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        s = np.clip(beta * (c - m), -400.0, 0.0)
        return (m + np.log(np.sum(np.exp(s))) / beta) * dx

    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        s = np.clip(beta * (c - m), -400.0, 0.0)
        w = np.exp(s) / np.sum(np.exp(s))
        g = np.zeros(n)
        for k in range(n):
            idx = n - 1 + k
            wk = w[idx]
            if k == 0:
                g += wk * (1.0 - 2.0 * x)
            else:
                for j in range(n):
                    d = 0.0
                    if j <= n - 1 - k:
                        d += 1.0 - x[j + k]
                    if j >= k:
                        d -= x[j - k]
                    g[j] += (wk + w[n - 1 - k]) * d
        return g * dx

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]

    for b in [beta]:
        if t0 is not None and time.time() - t0 > time_limit:
            break
        try:
            res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-16, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best = h_new.copy()
                h0 = h_new.copy()
        except Exception:
            pass
    return best, best_c5
