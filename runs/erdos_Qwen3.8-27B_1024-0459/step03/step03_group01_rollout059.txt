        g[off:off + npairs] += (1.0 - x[:npairs])

        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]

    return g * dx_val

        h[i] = float(c)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[:n_ones - target]] = 0.0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[:target - n_ones]] = 1.0
    return h


def sturmian(n, alpha, rho=0.0):
    h = np.zeros(n)
    seq = [(i + 1) * alpha + rho for i in range(n)]
    seq = seq + [i * alpha + rho for i in range(n)]
    first_half = seq[:n]
    second_half = seq[n:2 * n]
    for i in range(n):
        frac1 = first_half[i] - np.floor(first_half[i])
        frac2 = second_half[i] - np.floor(second_half[i])
        h[i] = 1.0 if frac1 > frac2 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[:n_ones - target]] = 0.0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[:target - n_ones]] = 1.0
    return h


def exhaustive_swap_local(h, n, max_rounds=30, time_limit=40):
    h = h.astype(float).copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    cur_max = float(np.max(compute_corr(h)))
    best_c5 = cur_max * (2.0 / n)
    start = time.time()

    for rnd in range(max_rounds):
        if time.time() - start > time_limit or remaining() < 5:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_improve = 1e-14
        best_a, best_b = -1, -1
        new_max_best = cur_max

        for a in ones:
            ai = int(a)
            for b in zeros:
                bi = int(b)
                h[ai] = 0.0
                h[bi] = 1.0
                new_max = float(np.max(compute_corr(h)))
                h[ai] = 1.0
                h[bi] = 0.0

                if new_max < new_max_best:
                    new_max_best = new_max
                    best_a, best_b = ai, bi

        if best_a >= 0:
            h[best_a] = 0.0
            h[best_b] = 1.0
            cur_max = new_max_best
            best_c5 = cur_max * (2.0 / n)
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
        else:
            break

    return h, best_c5


def sample_swap_local(h, n, max_iters=5000, time_limit=25, rng=None):
    if rng is None:
        rng = np.random.RandomState(99)
    h = h.astype(float).copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    cur_max = float(np.max(compute_corr(h)))
    best_c5 = cur_max * (2.0 / n)
    start = time.time()
    stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit or remaining() < 5:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break

        found = False
        n_try = 100
        for _ in range(n_try):
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            h[a] = 0.0
            h[b] = 1.0
            new_max = float(np.max(compute_corr(h)))
            if new_max < cur_max - 1e-14:
                cur_max = new_max
                best_c5 = cur_max * (2.0 / n)
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
                found = True
                break
            else:
                h[a] = 1.0
                h[b] = 0.0

        if found:
            stall = 0
        else:
            stall += 1
            if stall > 50:
                a = ones[rng.randint(len(ones))]
                b = zeros[rng.randint(len(zeros))]
                h[a] = 0.0
                h[b] = 1.0
                cur_max = float(np.max(compute_corr(h)))
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
                stall = 0

    return h, best_c5


def simulated_annealing(h, n, time_limit=20, T_start=4.0, T_end=0.005, seed_val=42):
    h = h.astype(float).copy()
    rng = np.random.RandomState(seed_val)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    cur_val = float(np.max(compute_corr(h)))
    best_h = h.copy()
    best_val = cur_val
    start = time.time()
    n_iter = 8000

    for it in range(n_iter):
        if time.time() - start > time_limit or remaining() < 5:
            break
        T = T_start * (T_end / T_start) ** (float(it) / float(n_iter))

        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h[a] = 0.0
        h[b] = 1.0
        new_val = float(np.max(compute_corr(h)))
        delta = new_val - cur_val

        if delta < 0 or rng.rand() < np.exp(max(-delta / max(T, 1e-8), -50)):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
        else:
            h[a] = 1.0
            h[b] = 0.0

    return best_h, best_val * (2.0 / n)


def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    s = np.sum(ex)
    if s < 1e-30:
        return np.zeros(n)
    w = ex / s
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-14:
            g += w[k] * corr_grad_per_shift(x, n, k)
    return g * dx_val


def sum_con(x):
    return np.sum(x) - _TGT


def sum_con_jac(x):
    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=400):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def active_set_refine(h, n, max_iters=200):
    h = h.astype(float).copy()
    dx_val = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        top2 = np.argsort(-corr)[:2]
        c1, c2 = corr[top2[0]], corr[top2[1]]
        if c1 - c2 < 1e-10:
            break
        g1 = corr_grad_per_shift(h, n, int(top2[0]))
        g2 = corr_grad_per_shift(h, n, int(top2[1]))
        d = g2 - g1
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-14:
            d = -g1
            dnorm = np.linalg.norm(d)
            if dnorm < 1e-14:
                break
        d /= dnorm
        improved = False
        for step in [0.01, 0.02, 0.05, 0.005, 0.03, 0.08, 0.015, 0.04]:
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-12:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def ensure_binary(h, n, rng=None):
    h = (h > 0.5).astype(float)
    target = n // 2
    n_ones = int(np.sum(h))
    if rng is None:
        rng = np.random.RandomState(0)
    if n_ones > target:
        idx = np.where(h == 1)[0]
        sel = rng.choice(idx, n_ones - target, replace=False)
        h[sel] = 0.0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        sel = rng.choice(idx, target - n_ones, replace=False)
        h[sel] = 1.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 320
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Binary local search on parent
    n = best_n
    h_bin = ensure_binary(best_h, n, rng)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Exhaustive swap on parent
    time_left = remaining()
    if time_left > 25:
        h_opt, c5_opt = exhaustive_swap_local(h_bin, n, max_rounds=min(25, int(time_left // 3)), time_limit=min(50, int(time_left / 3)))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h_bin = h_opt.copy()

    # SA on parent
    if remaining() > 15:
        h_sa, c5_sa = simulated_annealing(h_bin, n, time_limit=min(20, remaining() - 5), T_start=5.0, T_end=0.003, seed_val=777)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            h_bin = h_sa.copy()
        h_es, c5_es = exhaustive_swap_local(h_bin, n, max_rounds=min(20, int(remaining() // 4)), time_limit=min(40, int(remaining() / 4)))
        if c5_es < best_c5:
            best_c5 = c5_es
            best_h = h_es.copy()

    # Phase 2: Try other resolutions
    res_list = [160, 200, 240, 280, 320, 360, 400, 440, 480, 560, 640, 720, 800]
    res_list = [r for r in res_list if r != best_n and r % 2 == 0]

    for n_t in res_list:
        if remaining() < 25:
            break

        # Thue-Morse init
        h_tm = thue_morse(n_t)
        if n_t <= 400:
            h_opt, c5_opt = exhaustive_swap_local(h_tm, n_t, max_rounds=min(20, int(remaining() // 5)), time_limit=min(30, int(remaining() / 5)))
        else:
            h_opt, c5_opt = sample_swap_local(h_tm, n_t, max_iters=3000, time_limit=min(20, int(remaining() / 5)), rng=rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

        # Sturmian init
        if remaining() < 20:
            break
        alpha = (n_t // 2 + 1) / float(n_t)
        h_st = sturmian(n_t, alpha, 0.3)
        if n_t <= 400:
            h_opt2, c5_opt2 = exhaustive_swap_local(h_st, n_t, max_rounds=min(15, int(remaining() // 5)), time_limit=min(25, int(remaining() / 5)))
        else:
            h_opt2, c5_opt2 = sample_swap_local(h_st, n_t, max_iters=2000, time_limit=min(15, int(remaining() / 5)), rng=rng)
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2.copy()
            best_n = n_t

        # Random init
        if remaining() < 15:
            break
        idx_r = rng.choice(n_t, n_t // 2, replace=False)
        h_r = np.zeros(n_t)
        h_r[idx_r] = 1.0
        if n_t <= 360:
            h_opt3, c5_opt3 = exhaustive_swap_local(h_r, n_t, max_rounds=min(15, int(remaining() // 6)), time_limit=min(20, int(remaining() / 6)))
        else:
            h_opt3, c5_opt3 = sample_swap_local(h_r, n_t, max_iters=2000, time_limit=min(15, int(remaining() / 6)), rng=rng)
        if c5_opt3 < best_c5:
            best_c5 = c5_opt3
            best_h = h_opt3.copy()
            best_n = n_t

    # Phase 3: Intensive refinement on best
    n = best_n
    h_bin = ensure_binary(best_h, n, rng)

    # Multiple SA + exhaustive cycles
    for cycle in range(6):
        if remaining() < 12:
            break
        T_s = 3.0 + 0.5 * cycle
        h_sa, c5_sa = simulated_annealing(h_bin, n, time_limit=min(15, remaining() - 5), T_start=T_s, T_end=0.002, seed_val=100 + cycle)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            h_bin = h_sa.copy()
        if remaining() < 8:
            break
        if n <= 400:
            h_es, c5_es = exhaustive_swap_local(h_bin, n, max_rounds=min(25, int(remaining() // 5)), time_limit=min(35, int(remaining() / 5)))
        else:
            h_es, c5_es = sample_swap_local(h_bin, n, max_iters=4000, time_limit=min(25, int(remaining() / 5)), rng=rng)
        if c5_es < best_c5:
            best_c5 = c5_es
            best_h = h_es.copy()
            h_bin = h_es.copy()

    # Phase 4: SLSQP continuous refinement
    if remaining() > 12:
        h_cont = best_h.copy()
        for beta in [300, 1000, 3000, 8000]:
            if remaining() < 7:
                break
            h_new = slsqp_smooth(h_cont, n, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_cont = h_new.copy()

    # Active set refinement
    if remaining() > 8:
        h_eq, c5_eq = active_set_refine(best_h, n, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 5: Final binary polish
    if remaining() > 10:
        h_bin2 = ensure_binary(best_h, n, rng)
        if n <= 400:
            h_f, c5_f = exhaustive_swap_local(h_bin2, n, max_rounds=min(30, int(remaining() // 4)), time_limit=min(40, int(remaining() / 4)))
        else:
            h_f, c5_f = sample_swap_local(h_bin2, n, max_iters=5000, time_limit=min(30, int(remaining() / 4)), rng=rng)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final SLSQP
    if remaining() > 8:
        for beta in [5000, 12000]:
            if remaining() < 5:
                break
            h_new = slsqp_smooth(best_h, n, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
