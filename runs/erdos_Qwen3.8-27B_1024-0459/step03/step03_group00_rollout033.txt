    h = h.copy()

            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        
        # Top K lags
        top_k = 5
        top_idx = np.argsort(corr)[-top_k:]

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

        cnt = int(np.sum(mask))
        if cnt > 0:

        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def lp_descent_step(B, n, n_lags=5, step_size=0.1):
    """Use linear programming to find a descent direction from a binary solution."""
    dx = 2.0 / n
    corr = get_corr(B)
    max_c = float(np.max(corr))
    top_k = np.argsort(corr)[-n_lags:]

    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        offset = k - (n - 1)
        if offset >= 0:
            for i in range(n - offset):
                G[k, i] += (1.0 - B[i + offset])
                G[k, i + offset] -= B[i]
        else:
            abs_off = -offset
            for i in range(n - abs_off):
                G[k, i] += (1.0 - B[i + abs_off])
                G[k, i + abs_off] -= B[i]

    c = np.zeros(n + 1)
    c[n] = 1.0

    A_ub = np.zeros((n_lags, n + 1))
    b_ub = np.zeros(n_lags)
    for ii, kk in enumerate(top_k):
        A_ub[ii, :n] = G[kk, :]
        A_ub[ii, n] = 1.0
        b_ub[ii] = -1e-8

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds_lp = [(None, None)] * n + [(0, None)]
    for i in range(n):
        if B[i] < 1e-10:
            bounds_lp[i] = (0.0, 0.0)
        elif B[i] > 1.0 - 1e-10:
            bounds_lp[i] = (0.0, 0.0)
        else:
            bounds_lp[i] = (-B[i], 1.0 - B[i])

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs')
        if res.success and res.fun > 0:
            eps = res.x[:n]
            h_new = np.clip(B + step_size * eps, 0.0, 1.0)
            h_new = project_feasible(h_new, n)
            return h_new
    except Exception:
        pass
    return None

            best_h = h_cur.copy()

    # Phase 4: More SLSQP on the refined solution
    print("\n=== Phase 4: Final SLSQP ===")
    if time_fn() > 10.0:
        for beta in [5000, 15000, 50000]:
            if time_fn() < 5.0:
                break
            best_h, best_c5 = slsqp_optimize(best_h, best_n, beta, 200, time_fn)
            print("  beta={}: C5={:.10f}".format(beta, best_c5))

    # Phase 5: Coordinate descent polish
    print("\n=== Phase 5: Coordinate Descent ===")
    if time_fn() > 8.0:
        best_h, best_c5 = coordinate_descent(best_h, best_n, time_fn, rng, n_sweeps=5)
        print("  After CD: C5={:.10f}".format(best_c5))

    # Phase 6: Try LP descent again from current (possibly non-binary) solution
    if time_fn() > 5.0:
        h_rounded = (best_h >= 0.5).astype(float)
        target = best_n // 2
        cur = int(np.sum(h_rounded))
        if cur > target:
            idx = np.where(h_rounded == 1)[0]
            for ii in idx[:cur - target]:
                h_rounded[ii] = 0.0
        elif cur < target:
            idx = np.where(h_rounded == 0)[0]
            for ii in idx[:target - cur]:
                h_rounded[ii] = 1.0
        for _ in range(5):
            if time_fn() < 3.0:
                break
            h_lp = lp_descent_step(h_rounded, best_n, n_lags=7, step_size=0.08)
            if h_lp is not None:
                c5_lp = compute_c5(h_lp)
                if c5_lp < best_c5 - 1e-12:
                    best_c5 = c5_lp
                    best_h = h_lp.copy()
                    h_rounded = (best_h >= 0.5).astype(float)
                    cur = int(np.sum(h_rounded))
                    if cur > target:
                        idx = np.where(h_rounded == 1)[0]
                        for ii in idx[:cur - target]:
                            h_rounded[ii] = 0.0
                    elif cur < target:
                        idx = np.where(h_rounded == 0)[0]
                        for ii in idx[:target - cur]:
                            h_rounded[ii] = 1.0

    # Phase 7: One more SLSQP pass if time allows
    if time_fn() > 8.0:
        best_h, best_c5 = slsqp_optimize(best_h, best_n, 20000, 150, time_fn)
        print("  Final SLSQP: C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
