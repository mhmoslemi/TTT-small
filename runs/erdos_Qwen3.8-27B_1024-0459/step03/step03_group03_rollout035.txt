```python
import numpy as np
from scipy.optimize import minimize
import time

_t0 = 0.0
_tb = 1300.0
_ng = 0
_bv = 100.0

def _tl():
    return max(0.0, _tb - (time.time() - _t0) - 5.0)

def _c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / n)

def _proj(h, n):
    h = np.array(h, dtype=np.float64)
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(30):
        d = 0.5 * n - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        k = int(np.sum(m))
        if k > 0:
            h[m] += d / k
        else:
            h[:] = 0.5
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _interp(h, ns, nd):
    if ns == nd:
        return h.copy()
    xs = (np.arange(ns) + 0.5) * (2.0 / ns)
    xd = (np.arange(nd) + 0.5) * (2.0 / nd)
    return np.interp(xd, xs, h)

def _scon(x):
    return x.sum() - 0.5 * _ng

def _sobj(x):
    n = _ng
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = float(np.max(c))
    b = _bv
    sh = c - mx
    # Numerical stability
    sh = np.clip(sh, -300.0, 0.0)
    return (mx + np.log(np.exp(b * sh).sum()) / b) * (2.0 / n)

def _slsqp(h0, n, beta, maxiter):
    global _ng, _bv
    _ng = n
    _bv = beta
    try:
        r = minimize(_sobj, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon}],
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return _proj(np.asarray(r.x, dtype=np.float64), n)
    except Exception:
        return _proj(h0.copy(), n)

def _ggrad(h, k):
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    else:
        p = -s
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g

def _smgrad(h, beta, topm=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    nc = len(corr)
    if topm is not None and topm < nc:
        idx = np.argsort(corr)[-topm:]
        vals = corr[idx]
        vmax = float(vals.max())
        w = np.exp(beta * (vals - vmax))
        sw = float(w.sum())
        if sw < 1e-30:
            return np.zeros(n)
        w /= sw
        g = np.zeros(n)
        for i in range(len(idx)):
            if w[i] > 1e-10:
                g += w[i] * _ggrad(h, int(idx[i]))
        return g
    else:
        mx = float(np.max(corr))
        w = np.exp(beta * (corr - mx))
        sw = float(w.sum())
        if sw < 1e-30:
            return np.zeros(n)
        w /= sw
        g = np.zeros(n)
        nz = np.where(w > 1e-10)[0]
        for k in nz:
            g += w[int(k)] * _ggrad(h, int(k))
        return g

def _grad_descent(h, n, nit, rng, topm=50):
    best = h.copy()
    bc = _c5(best)
    alpha = 0.05
    for it in range(nit):
        if _tl() < 5:
            break
        g = _smgrad(best, 100.0, topm)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            g = _smgrad(best, 500.0, topm=80)
            gn = float(np.linalg.norm(g))
            if gn < 1e-14:
                break
        g /= gn
        # Backtracking line search
        moved = False
        a = alpha
        for _bt in range(15):
            h2 = _proj(best - a * g, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-15:
                best = h2
                bc = c2
                alpha = min(a * 1.5, 0.1)
                moved = True
                break
            a *= 0.6
        if not moved:
            alpha *= 0.7
            if alpha < 1e-5:
                break
    return best, bc

def _local_search(h, n, nit, rng):
    best = h.copy()
    bc = _c5(best)
    for sweep in range(nit):
        if _tl() < 5:
            break
        imp = False
        # Random pairwise swaps
        for _ in range(200):
            if _tl() < 3:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            h2 = best.copy()
            tmp = h2[i]
            h2[i] = h2[j]
            h2[j] = tmp
            h2 = _proj(h2, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-14:
                best = h2
                bc = c2
                imp = True
        # Block swaps (length 2-3)
        for _ in range(80):
            if _tl() < 3:
                break
            bl = rng.randint(1, 4)  # block length 1-3
            i = rng.randint(0, n - bl)
            j = rng.randint(0, n - bl)
            if abs(i - j) < bl:
                continue
            h2 = best.copy()
            h2[i:i+bl], h2[j:j+bl] = h2[j:j+bl].copy(), h2[i:i+bl].copy()
            h2 = _proj(h2, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-14:
                best = h2
                bc = c2
                imp = True
        # Nudge: slightly adjust a point and redistribute
        for _ in range(100):
            if _tl() < 3:

            delta = 0.01 * (1 if rng.rand() < 0.5 else -1)
            j = rng.randint(n)

            h2 = best.copy()

            h2[j] -= delta
            h2 = _proj(h2, n)
            c2 = _c5(h2)
```

For the objective gradient, I need to be careful. The smooth max is:
f(x) = (mx + log(sum(exp(beta*(c-mx)))) / beta) * (2/n)

where mx = max(c). The issue is that mx itself depends on x, making the gradient complex.

A cleaner formulation: let's use the "soft max" without explicitly tracking the max:
f(x) = log(sum(exp(beta * c_k))) / beta * (2/n)

This is the log-sum-exp divided by beta, which approximates max(c_k) * (2/n) for large beta.

The gradient is:
df/dx = (2/n) * (1/beta) * beta * sum_k (exp(beta*c_k) / sum_j exp(beta*c_j)) * grad_k
      = (2/n) * sum_k w_k * grad_k

where w_k = exp(beta*c_k) / sum_j exp(beta*c_j)

This is much cleaner! And the gradient is well-defined everywhere.

But there's a numerical stability issue: for large beta, exp(beta*c_k) can overflow. The solution is to subtract the max:
w_k = exp(beta*(c_k - max_c)) / sum_j exp(beta*(c_j - max_c))

And the objective value is:
f = (max_c + log(sum(exp(beta*(c - max_c))))) / beta * (2/n)
