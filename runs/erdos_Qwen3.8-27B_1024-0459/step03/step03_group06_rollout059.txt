def smooth_max_obj(x):

def fast_gradient(h, w):
    """Compute gradient of weighted sum of correlations."""
    n = len(h)
    grad = np.zeros(n)
    shifts = np.arange(-(n-1), n)
    for i, s in enumerate(shifts):
        wi = w[i]

def smooth_max_grad(x):

    dx = 2.0 / n

def binary_2swap(h, n, iterations, rng):

        h[ones_idx[remove]] = 0.0

        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)

def fast_gradient(h, w):
    n = len(h)
    grad = np.zeros(n)
    # w has length 2n-1, w[i] corresponds to shift s = i - (n-1)
    for i in range(2 * n - 1):
        wi = w[i]
        if abs(wi) < 1e-15:
            continue
        s = i - (n - 1)
        if s >= 0:
            # (1-h[m+s]) for m in [0, n-1-s]
            grad[:n-s] += wi * (1.0 - h[s:])
            # -h[m-s] for m in [s, n-1]
            grad[s:] -= wi * h[:n-s]
        else:
            sp = -s
            # (1-h[m-sp]) for m in [sp, n-1]
            grad[sp:] += wi * (1.0 - h[:n-sp])
            # -h[m+sp] for m in [0, n-1-sp]
            grad[:n-sp] -= wi * h[sp:]
    return grad

import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0


def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 2.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def fast_gradient(h, w):
    n = len(h)
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if abs(wi) < 1e-15:
            continue
        s = i - (n - 1)
        if s >= 0:
            grad[:n - s] += wi * (1.0 - h[s:])
            grad[s:] -= wi * h[:n - s]
        else:
            sp = -s
            grad[sp:] += wi * (1.0 - h[:n - sp])
            grad[:n - sp] -= wi * h[sp:]
    return grad


def smooth_max_obj(x):
    n = _N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    sm = (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx
    penalty = 5000.0 * (np.sum(x) - _TGT) ** 2
    return sm + penalty


def smooth_max_grad(x):
    n = _N
    dx = 2.0 / n
    beta = _BETA
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    total = np.sum(weights)
    weights /= total
    grad = fast_gradient(x, weights) * dx
    penalty_coeff = 5000.0
    grad += 2.0 * penalty_coeff * (np.sum(x) - _TGT)
    return grad


def topk_obj(x, top_k=12):
    n = _N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    tv = corr[top_idx]
    m = np.max(tv)
    beta = 120.0
    sm = (m + np.log(np.sum(np.exp(beta * (tv - m)))) / beta) * dx
    penalty = 5000.0 * (np.sum(x) - _TGT) ** 2
    return sm + penalty


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        unsat = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(unsat)
        if cnt > 0:
            h[unsat] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def make_golden_ratio(n):
    h = np.zeros(n)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return project_feasible(h, n)


def make_binary_random(n, rng):

def interp_to(src_h, src_n, dst_n):

    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-15,
                                    'gtol': 1e-11, 'iprint': 0,
                                    'maxls': 40, 'maxfun': 4000})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    if remaining_time() > 8:
        try:
            res = minimize(topk_obj, best_h, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 500, 'ftol': 1e-15,
                                    'iprint': 0, 'maxls': 40,
                                    'maxfun': 6000})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    return best_h, best_c5


def binary_2swap(h, n, iterations, rng):
    h = np.where(h > 0.5, 1.0, 0.0).copy()
    target_ones = n // 2
    num_ones = int(np.sum(h))
    if num_ones > target_ones:
        ones_idx = np.where(h == 1.0)[0]
        remove = rng.choice(len(ones_idx), num_ones - target_ones, replace=False)
        h[ones_idx[remove]] = 0.0
    elif num_ones < target_ones:
        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
        h[zeros_idx[add]] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5

    for it in range(iterations):
        if remaining_time() < 3:
            break
        ones_list = np.where(h == 1.0)[0]
        zeros_list = np.where(h == 0.0)[0]
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break

        oi = ones_list[rng.randint(len(ones_list))]
        oz = zeros_list[rng.randint(len(zeros_list))]

        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = compute_c5(h)

        T = 0.003 * (0.00003 / 0.003) ** (float(it) / max(float(iterations), 1.0))

        if new_c5 < current_c5 or (T > 1e-12 and rng.rand() < np.exp(-(new_c5 - current_c5) / T)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0

    return best_h, best_c5


def binary_hill_climb(h, n, rounds, rng):
    h = np.where(h > 0.5, 1.0, 0.0).copy()
    target_ones = n // 2
    num_ones = int(np.sum(h))
    if num_ones != target_ones:
        if num_ones > target_ones:
            ones_idx = np.where(h == 1.0)[0]
            remove = rng.choice(len(ones_idx), num_ones - target_ones, replace=False)
            h[ones_idx[remove]] = 0.0
        else:
            zeros_idx = np.where(h == 0.0)[0]
            add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
            h[zeros_idx[add]] = 1.0

    best_c5 = compute_c5(h)

    for rnd in range(rounds):
        if remaining_time() < 3:
            break
        improved = False
        ones_list = np.where(h == 1.0)[0]
        zeros_list = np.where(h == 0.0)[0]

        ntrials = min(300, max(50, len(ones_list) * len(zeros_list) // 40))
        for _ in range(ntrials):
            if remaining_time() < 1:
                break
            oi = ones_list[rng.randint(len(ones_list))]
            oz = zeros_list[rng.randint(len(zeros_list))]
            h[oi] = 0.0
            h[oz] = 1.0
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                improved = True
            else:
                h[oi] = 1.0
                h[oz] = 0.0

        if not improved:
            break

    return h, best_c5


def gradient_descent_minmax(h, n, max_iter=300):
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for it in range(max_iter):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        m = np.max(corr) * dx
        top_k = 5
        top_idx = np.argsort(corr)[-top_k:]
        w = np.zeros(2 * n - 1)
        w[top_idx] = 1.0 / top_k

        grad = fast_gradient(h, w)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm

        improved = False
        for step in [1.0, 0.5, 2.0, 0.25, 0.1, 4.0, 0.75, 0.05]:
            h_trial = h - step * grad * dx
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-12:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break

        if not improved:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S, _N, _BETA, _TGT
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.6f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 160
        best_h = make_thue_morse(best_n)
        best_c5 = compute_c5(best_h)
        print("Default Thue-Morse: n={}, C5={:.6f}".format(best_n, best_c5))

    # Phase 1: Continuous optimization at parent resolution
    print("\n=== Phase 1: Continuous optimization (n={}) ===".format(best_n))
    h_opt, c5_opt = optimize_continuous(best_h, best_n, [30, 80, 200, 500, 1000], maxiter=500)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()
        print("  Continuous: C5={:.7f}".format(best_c5))

    # Subgradient minmax at parent resolution
    if remaining_time() > 30:
        h_gs, c5_gs = gradient_descent_minmax(best_h, best_n, max_iter=300)
        if c5_gs < best_c5:
            best_c5 = c5_gs
            best_h = h_gs.copy()
            print("  GradDesc: C5={:.7f}".format(best_c5))
        # Polish after grad desc
        h_pol, c5_pol = optimize_continuous(h_gs, best_n, [500, 1000, 2000], maxiter=300)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            print("  Polished: C5={:.7f}".format(best_c5))

    # Phase 2: Multi-resolution
    print("\n=== Phase 2: Multi-resolution ===")
    for n_try in [best_n + 16, best_n + 40, best_n + 80, 200, 240, 280, 320, 360]:
        if remaining_time() < 35:
            break
        if n_try < 120 or n_try > 400 or n_try == best_n:
            continue
        print("\n  --- n={} ---".format(n_try))

        h_int = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        h_opt2, c5_opt2 = optimize_continuous(h_int, n_try, [50, 200, 600], maxiter=350)
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2.copy()
            best_n = n_try
            print("  Continuous n={}: C5={:.7f}".format(n_try, best_c5))

        # Binary at this resolution
        if remaining_time() > 20:
            h_bin = np.where(best_h > 0.5, 1.0, 0.0)
            h_bin = project_feasible(h_bin, n_try)
            h_bc, c5_bc = binary_hill_climb(h_bin, n_try, 30, rng)
            if c5_bc < best_c5:
                best_c5 = c5_bc
                best_h = h_bc.copy()
                best_n = n_try
                print("  BinaryHC n={}: C5={:.7f}".format(n_try, best_c5))

            h_pol2, c5_pol2 = optimize_continuous(h_bc, n_try, [300, 800], maxiter=300)
            if c5_pol2 < best_c5:
                best_c5 = c5_pol2
                best_h = h_pol2.copy()
                best_n = n_try
                print("  Bi2Cont n={}: C5={:.7f}".format(n_try, best_c5))

        # Downsample back to original and re-optimize
        if remaining_time() > 25 and n_try > best_n - 64:
            h_down = project_feasible(interp_to(best_h, best_n, min(best_n, n_try)), min(best_n, n_try))
            h_down_opt, c5_down = optimize_continuous(h_down, min(best_n, n_try), [500, 1500], maxiter=250)
            if c5_down < best_c5:
                best_c5 = c5_down
                best_h = h_down_opt.copy()
                best_n = min(best_n, n_try)
                print("  Downsample: C5={:.7f}".format(best_c5))

    # Phase 3: Aggressive binary search at best resolution
    print("\n=== Phase 3: Binary search (n={}) ===".format(best_n))
    n = best_n
    h_bin = np.where(best_h > 0.5, 1.0, 0.0)
    h_bin = project_feasible(h_bin, n)

    # Deterministic hill climb first
    h_hc, c5_hc = binary_hill_climb(h_bin, n, 50, rng)
    if c5_hc < best_c5:
        best_c5 = c5_hc
        best_h = h_hc.copy()
        print("  HillClimb: C5={:.7f}".format(best_c5))

    # SA iterations
    for sa_round in range(6):
        if remaining_time() < 25:
            break
        if sa_round > 0:
            h_pert = h_hc.copy()
            n_swaps = max(3, n // 30)
            for _ in range(n_swaps):
                oi = np.where(h_pert == 1.0)[0][rng.randint(n // 2)]
                oz = np.where(h_pert == 0.0)[0][rng.randint(n // 2)]
                h_pert[oi], h_pert[oz] = h_pert[oz], h_pert[oi]
            h_sa_start = project_feasible(h_pert, n)
        else:
            h_sa_start = h_hc.copy()

        h_sa, c5_sa = binary_2swap(h_sa_start, n, 8000, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            h_hc = h_sa.copy()
            print("  SA round {}: C5={:.7f}".format(sa_round, best_c5))

        h_hc2, c5_hc2 = binary_hill_climb(h_sa, n, 40, rng)
        if c5_hc2 < best_c5:
            best_c5 = c5_hc2
            best_h = h_hc2.copy()
            h_hc = h_hc2.copy()
            print("  HC after SA: C5={:.7f}".format(best_c5))

        # Polish with continuous
        if remaining_time() > 15:
            h_pol3, c5_pol3 = optimize_continuous(h_hc, n, [300, 800, 2000], maxiter=400)
            if c5_pol3 < best_c5:
                best_c5 = c5_pol3
                best_h = h_pol3.copy()
                print("  Cont polish: C5={:.7f}".format(best_c5))

    # Phase 4: Extended subgradient + L-BFGS alternating
    print("\n=== Phase 4: Alternating optimization (n={}) ===".format(best_n))
    n = best_n
    h = best_h.copy()

    for alt_round in range(8):
        if remaining_time() < 20:
            break

        # Subgradient step
        h_gs2, c5_gs2 = gradient_descent_minmax(h, n, max_iter=200)
        if c5_gs2 < best_c5:
            best_c5 = c5_gs2
            best_h = h_gs2.copy()
            h = h_gs2.copy()

        # L-BFGS polish
        h_lb, c5_lb = optimize_continuous(h, n, [500, 1500, 3000], maxiter=400)
        if c5_lb < best_c5:
            best_c5 = c5_lb
            best_h = h_lb.copy()
            h = h_lb.copy()
            print("  Alt round {}: C5={:.7f}".format(alt_round, best_c5))
        else:
            break

    # Phase 5: Try SLSQP for equality-constrained optimization
    if remaining_time() > 20:
        print("\n=== Phase 5: SLSQP ===")
        n = best_n
        _N = n
        _TGT = n / 2.0

        def slsqp_fun(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(corr)
            beta = 300.0
            sh = corr - m
            return (m + np.log(np.sum(np.exp(beta * sh))) / beta) * (2.0 / n)

        def slsqp_grad(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(corr)
            beta = 300.0
            w = np.exp(beta * (corr - m))
            w /= np.sum(w)
            return fast_gradient(x, w) * (2.0 / n)

        def slsqp_con_eq(x):
            return np.sum(x) - n / 2.0

        def slsqp_con_eq_grad(x):
            return np.ones(n)

        cons = [{'type': 'eq', 'fun': slsqp_con_eq, 'jac': slsqp_con_eq_grad}]
        bounds = [(0.0, 1.0)] * n

        try:
            res = minimize(slsqp_fun, best_h, jac=slsqp_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-12,
                                    'disp': 0})
            h_sl = project_feasible(res.x, n)
            c5_sl = compute_c5(h_sl)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()
                print("  SLSQP: C5={:.7f}".format(best_c5))
        except Exception:
            pass

    # Phase 6: Final high-precision polish
    print("\n=== Phase 6: Final polish ===")
    n = best_n
    _N = n
    _TGT = n / 2.0
    bounds = [(0.0, 1.0)] * n

    for beta_val in [2000, 5000, 10000]:
        if remaining_time() < 4:
            break
        _BETA = beta_val
        try:
            res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 600, 'ftol': 1e-16,
                                    'gtol': 1e-13, 'iprint': 0,
                                    'maxls': 50, 'maxfun': 8000})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  Beta {}: C5={:.7f}".format(beta_val, best_c5))
        except Exception:
            pass

    # Final binary check
    if remaining_time() > 8:
        h_bin_f = np.where(best_h > 0.5, 1.0, 0.0)
        h_bin_f = project_feasible(h_bin_f, n)
        c5_bin_f = compute_c5(h_bin_f)
        if c5_bin_f < best_c5:
            best_c5 = c5_bin_f
            best_h = h_bin_f.copy()
            print("  Final binary: C5={:.7f}".format(best_c5))

    # Try one more resolution jump if time allows
    if remaining_time() > 30 and n < 350:
        n2 = min(n + 64, 400)
        if n2 != n:
            h_make = best_n  # temp
            h_int2 = project_feasible(interp_to(best_h, n, n2), n2)
            h_opt3, c5_opt3 = optimize_continuous(h_int2, n2, [300, 1000], maxiter=300)
            if c5_opt3 < best_c5:
                best_c5 = c5_opt3
                best_h = h_opt3.copy()
                best_n = n2
                print("  Higher res n={}: C5={:.7f}".format(n2, best_c5))

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
