import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xq = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xq, xp, src)

def soft_max_val(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    maxs = np.max(shifted)
    exps = np.exp(beta * (shifted - maxs))
    s = np.sum(exps)
    return (m + maxs + np.log(s) / beta) * dx

def soft_max_obj_grad(h, n, beta, lam=0.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    maxs = np.max(shifted)
    exps = np.exp(beta * (shifted - maxs))
    w = exps / np.sum(exps)
    val = (m + maxs + np.log(np.sum(exps)) / beta) * dx

    # Gradient
    grad = np.zeros(n)
    for t in range(2 * n - 1):
        if w[t] < 1e-14:
            continue
        k = t - (n - 1)
        wk = w[t] * dx
        if k >= 0:
            nk = n - k
            if nk > 0:
                grad[k:n] += wk * (1.0 - h[0:nk])
                grad[0:nk] -= wk * h[k:n]
        else:
            ka = -k
            nk = n - ka
            if nk > 0:
                grad[0:nk] += wk * (1.0 - h[ka:n])
                grad[ka:n] -= wk * h[0:nk]
    
    if lam > 0:
        s = np.sum(h)
        target = n / 2.0
        val += lam * (s - target) ** 2
        grad += 2.0 * lam * (s - target)
    
    return val, grad

def rtt_step(h, n, i, j, d, direction):
    """Try moving d from one index to another. Returns new h or None."""
    if direction == 0:
        if h[i] < d or h[j] > 1.0 - d:
            return None
    else:
        if h[j] < d or h[i] > 1.0 - d:
            return None
    h_new = h.copy()
    if direction == 0:
        h_new[i] -= d
        h_new[j] += d
    else:
        h_new[j] -= d
        h_new[i] += d
    return h_new

    """Fast evaluation focusing on active constraints."""
    dx = 2.0 / n
    c5v, corr = compute_c5(h)
    return c5v, corr

            res = minimize(soft_max_obj_grad, h, method='L-BFGS-B',
                           args=(n, beta, lam),
                           jac=True,
                           bounds=bounds,
                           options={'maxiter': 300, 'ftol': 1e-15,
                                     'gtol': 1e-10, 'maxls': 50})
            h_new = project_h(res.x, n)
            c5_new, _ = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
                print("  L-BFGS b={}: C5={:.10f}".format(beta, best_c5))
        except Exception:
            pass
    
    # Phase 3: Active-set focused optimization
    n = best_n
    h = best_h.copy()
    
    for outer in range(20):
        if time.time() - t0 > 0.70 * budget:
            break
        
        c5_cur, corr = compute_c5(h)
        mx = np.max(corr)
        thresh = mx - 0.05 * mx
        active = np.where(corr >= thresh)[0]
        if len(active) < 3:
            active = np.argsort(corr)[-10:]
        if len(active) > 40:
            active = np.argsort(corr[active])
            active = active[active]  # reorder
            active = np.where(corr >= mx - 0.1 * mx)[0]
            if len(active) > 40:
                active = np.argsort(corr)[-25:]
        
        for beta in [500, 1500, 3000]:
            if time.time() - t0 > 0.69 * budget:
                break
            # Customize: only optimize on active set via higher penalty on others
            # Actually, use full soft-max but with very high beta to focus on max
            try:
                res = minimize(soft_max_obj_grad, h, method='L-BFGS-B',
                               args=(n, beta, 5000.0),
                               jac=True, bounds=bounds,
                               options={'maxiter': 200, 'ftol': 1e-16,
                                         'gtol': 1e-11, 'maxls': 40})
                h_new = project_h(res.x, n)
                c5_new, _ = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
            except Exception:
                pass
    
    print("After active-set: C5={:.10f}".format(best_c5))
    
    # Phase 4: Record-to-record traveling (RTT)
    n = best_n
    h = best_h.copy()
    rtt_time_limit = min(0.15 * budget, max(10, budget - (time.time() - t0) - 60))
    rtt_end = time.time() + rtt_time_limit
    
    # Adaptive tolerance
    tol = max(0.001, best_c5 * 0.005)
    record = best_c5 + tol
    consecutive_rejects = 0
    max_rejects = 5000
    
    d_choices = [0.001, 0.002, 0.003, 0.005, 0.008, 0.012, 0.02, 0.03, 0.05, 0.08]
    
    step_count = 0
    while time.time() < rtt_end:
        step_count += 1
        
        # Adaptive: shrink tolerance over time
        progress = 1.0 - (rtt_end - time.time()) / rtt_time_limit
        tol = max(0.0001, best_c5 * 0.005 * (1.0 - progress * 0.8))
        record = best_c5 + tol
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = d_choices[rng.randint(len(d_choices))]
        direction = rng.randint(2)
        
        h_new = rtt_step(h, n, i, j, d, direction)
        if h_new is None:
            continue
        
        c5_new, _ = compute_c5(h_new)
        
        if c5_new < record:
            h = h_new
            consecutive_rejects = 0
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        else:
            consecutive_rejects += 1
            # If too many rejects, try smaller steps or different pairs
            if consecutive_rejects > max_rejects:
                consecutive_rejects = 0
                tol = max(0.00005, best_c5 * 0.001)
                record = best_c5 + tol
    
    print("After RTT ({} steps): C5={:.10f}".format(step_count, best_c5))
    
    # Phase 5: SLSQP final polish at high beta
    n = best_n
    h = best_h.copy()
    
    if time.time() - t0 < budget - 30:
        for beta_f in [2000, 5000, 10000]:
            if time.time() - t0 > budget - 15:
                break
            try:
                cons = [{'type': 'eq', 'fun': lambda x, nn=n: np.sum(x) - nn / 2.0}]
                res = minimize(soft_max_obj_grad, h, method='SLSQP',
                               args=(n, beta_f, 0.0),
                               jac=True, bounds=bounds, constraints=cons,
                               options={'maxiter': 200, 'ftol': 1e-16, 'iprint': 0})
                h_new = project_h(res.x, n)
                c5_new, _ = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                    print("  SLSQP b={}: C5={:.10f}".format(beta_f, best_c5))
            except Exception:
                pass
    
    # Phase 6: Fine pairwise moves (very small steps)
    n = best_n
    h = best_h.copy()
    fine_end = min(time.time() + (budget - (time.time() - t0)) * 0.3, t0 + budget - 10)
    
    tiny_ds = [0.0005, 0.001, 0.002, 0.003, 0.005, 0.008]
    fine_count = 0
    while time.time() < fine_end:
        fine_count += 1
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = tiny_ds[rng.randint(len(tiny_ds))]
        direction = rng.randint(2)
        
        h_new = rtt_step(h, n, i, j, d, direction)
        if h_new is None:
            continue
        
        c5_new, _ = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    print("After fine RTT ({} steps): C5={:.10f}".format(fine_count, best_c5))
    
    # Phase 7: Try nearby resolutions one more time
    if time.time() - t0 < budget - 15:
        for dn in [-10, -5, 5, 10, 15, 20]:
            if time.time() - t0 > budget - 5:
                break
            nt = best_n + dn
            if nt < 60 or nt > 500 or nt % 2 != 0:
                continue
            ht = interp_h(best_h, best_n, nt)
            ht = project_h(ht, nt)
            ct, _ = compute_c5(ht)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()
                best_n = nt
                print("  Final res n={}: C5={:.10f}".format(nt, ct))
    
    # Final projection and evaluation
    best_h = project_h(best_h, best_n)
    final_c5, _ = compute_c5(best_h)
    
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
