def project_feasible(h, target):

    np.clip(h, 0.0, 1.0, out=h)

        s = float(np.sum(h))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            pos = room > 1e-12
            if not np.any(pos):
                break
            tot = float(np.sum(room[pos]))
            if tot <= 1e-15:
                break
            scale = diff / tot
            if scale > 1.0:
                scale = 1.0
            h[pos] += room[pos] * scale

            avail = h

def project_feasible(h, target):
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(30):
        s = float(np.sum(h))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            pos = room > 1e-12
            if not np.any(pos):
                break
            tot = float(np.sum(room[pos]))
            if tot <= 1e-15:
                break
            scale = diff / tot
            if scale > 1.0:
                scale = 1.0
            h[pos] += room[pos] * scale
        else:
            avail = h
            pos = avail > 1e-12
            if not np.any(pos):
                break
            tot = float(np.sum(avail[pos]))
            if tot <= 1e-15:
                break
            scale = -diff / tot
            if scale > 1.0:
                scale = 1.0
            h[pos] -= avail[pos] * scale
        np.clip(h, 0.0, 1.0, out=h)
    h[h < 1e-15] = 0.0
    h[h > 1.0 - 1e-15] = 1.0
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def compute_f_direct(h, n, h1m=None):
    if h1m is None:
        h1m = 1.0 - h
    dx = 2.0 / n
    f = np.empty(2 * n - 1, dtype=float)
    off = n - 1
    for k in range(0, n):
        l = n - k
        if l > 0:
            f[off + k] = dx * float(np.dot(h[:l], h1m[k:]))
    for k in range(1, n):
        a = k
        l = n - k
        if l > 0:
            f[off - k] = dx * float(np.dot(h[a:], h1m[:l]))
    return f


def update_single(h, h1m, hrev, f, r, d, n, dx, tmp1, tmp2):
    dx_d = dx * d
    if dx_d == 0.0:
        return
    s1 = n - 1 - r
    np.multiply(h1m, dx_d, out=tmp1)
    f[s1:s1 + n] += tmp1
    s2 = r
    np.multiply(hrev, -dx_d, out=tmp2)
    f[s2:s2 + n] += tmp2
    f[n - 1] -= dx * d * d
    h[r] += d
    h1m[r] -= d


def apply_move(h, h1m, hrev, f, target, source, d, n, dx, tmp1, tmp2):
    update_single(h, h1m, hrev, f, target, d, n, dx, tmp1, tmp2)
    update_single(h, h1m, hrev, f, source, -d, n, dx, tmp1, tmp2)


def make_weights(f, beta, top_count, gap):
    m = float(np.max(f))
    sel = np.where(f >= m - gap)[0]
    if sel.size == 0:
        sel = np.array([int(np.argmax(f))])
    elif top_count < f.size and sel.size > top_count:
        sel = np.argpartition(f, -top_count)[-top_count:]
    if beta <= 0.0:
        w = np.zeros(f.size, dtype=float)
        w[sel] = 1.0 / float(sel.size)
        return w
    exps = np.exp(beta * (f[sel] - m))
    s = float(np.sum(exps))
    w = np.zeros(f.size, dtype=float)
    if s > 0.0:
        w[sel] = exps / s
    return w


def compute_gradient(weights, h, n):
    E = np.zeros(n, dtype=float)
    h1m = 1.0 - h
    off = n - 1
    for idx, w in enumerate(weights):
        if w <= 1e-15:
            continue
        k = idx - off
        if k >= 0:
            l = n - k
            if l <= 0:
                continue
            E[:l] += w * h1m[k:]
            E[k:] -= w * h[:l]
        else:
            a = -k
            l = n - a
            E[a:] += w * h1m[:l]
            E[:l] -= w * h[a:]
    return E


def resync_state(h, h1m, f, n):
    np.clip(h, 0.0, 1.0, out=h)

    diff = target - float(np.sum(h))

    h1m[:] = 1.0 - h
    ff = compute_f_direct(h, n, h1m)
    np.copyto(f, ff)
    return float(np.max(f))

def resample_nearest(h, n_new):
    n_old = len(h)
    dx_old = 2.0 / n_old
    x_new = (np.arange(n_new, dtype=float) + 0.5) * (2.0 / n_new)
    idx = np.floor(x_new / dx_old).astype(int)
    np.clip(idx, 0, n_old - 1, out=idx)
    return h[idx].copy()


def resample_linear(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old, dtype=float) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new, dtype=float) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)


def make_variants(h):
    out = []
    out.append(h)
    comp = 1.0 - h
    out.append(comp)
    rev = h[::-1]
    out.append(rev)
    out.append(1.0 - rev)
    out.append(0.5 * (h + rev))
    out.append(0.5 * (h + 1.0 - rev))
    out.append(0.5 * (comp + rev))
    n = len(h)
    for shift in (1, 2, 3):
        if 0 < shift < n:
            out.append(np.roll(h, shift))
            out.append(np.roll(comp, shift))
    if n >= 8:
        shift = n // 4
        out.append(np.roll(h, shift))
        out.append(np.roll(comp, shift))
    return out


def fix_solution_even(h, n):
    if n % 2 == 0:
        return h, n
    nn = n - 1 if n > 100 else n + 1
    h2 = resample_linear(h, nn)

def polish_objective(x):
    n = POL_N
    if n <= 0:
        return 0.0
    dx = POL_DX
    target = POL_TARGET
    beta = POL_BETA
    pen = POL_PEN
    f = compute_f_direct(x, n)
    m = float(np.max(f))
    shifted = beta * (f - m)
    s = float(np.sum(np.exp(shifted)))
    if s <= 0.0:
        val = m
    else:
        val = m + math.log(s) / beta
    se = float(np.sum(x)) - target
    return float(val + pen * se * se)


def polish_gradient(x):
    n = POL_N
    dx = POL_DX
    target = POL_TARGET
    beta = POL_BETA
    pen = POL_PEN
    f = compute_f_direct(x, n)
    m = float(np.max(f))
    shifted = beta * (f - m)
    e = np.exp(shifted)
    s = float(np.sum(e))
    if s > 0.0:
        w = e / s
    else:
        w = np.zeros_like(f)
        w[int(np.argmax(f))] = 1.0
    g = compute_gradient(w, x, n)
    g *= dx
    se = float(np.sum(x)) - target
    g += 2.0 * pen * se
    return g


def polish_lbfgs(h, t_end, betas):
    global POL_N, POL_DX, POL_TARGET, POL_BETA, POL_PEN
    n = len(h)
    target = 0.5 * n
    dx = 2.0 / n
    best_h = project_feasible(h, target).astype(float).copy()
    best_c = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        if time.time() > t_end - 4.0:
            break
        POL_N = n
        POL_DX = dx
        POL_TARGET = target
        POL_BETA = float(beta)
        POL_PEN = 1.0e5
        start = best_h.copy()
        try:
            res = minimize(
                polish_objective,
                start,
                jac=polish_gradient,
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": 120, "ftol": 1e-18, "gtol": 1e-12, "maxfun": 1200}
            )
            xres = getattr(res, "x", None) if res is not None else None
            if xres is not None:
                cand = project_feasible(xres, target)
                c = compute_c5(cand)
                if c < best_c - 1e-14:
                    best_h = cand
                    best_c = c
        except Exception:
            pass
    best_h = project_feasible(best_h, target)
    best_c = compute_c5(best_h)
    return best_h, float(best_c)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget = max(1.0, float(budget_s))
    deadline = t0 + budget

    try:
        parent_raw = initial_h_values
    except NameError:
        parent_raw = np.full(160, 0.5)
        parent_raw[:80] = 1.0

    parent = np.asarray(parent_raw, dtype=float).copy()
    parent = project_feasible(parent, 0.5 * len(parent))
    parent, n_par = fix_solution_even(parent, len(parent))

    if budget < 30.0:
        c = compute_c5(parent)
        try:
            ec = float(evaluate_erdos_solution(parent))
            if ec < c:
                c = ec
        except Exception:
            pass
        return parent, float(c), int(n_par)

    rng = np.random.RandomState(seed)

    best_h = parent.copy()
    best_n = n_par
    best_c = compute_c5(best_h)

    for v in make_variants(parent):
        if time.time() > deadline - 120.0:
            break
        vh = project_feasible(v, 0.5 * n_par)
        c = compute_c5(vh)
        if c < best_c - 1e-14:
            best_c = c
            best_h = vh
            best_n = n_par

    n_set = set()
    for base in (128, 144, 160, 176, 192, 208, 224, 256):
        if 96 <= base <= 320 and base % 2 == 0:
            n_set.add(base)
    for d in (-64, -32, -16, 0, 16, 32, 48, 64, 96, 128):
        nn = n_par + d
        if 96 <= nn <= 320 and nn % 2 == 0:
            n_set.add(nn)
    n_list = sorted(n_set)

    all_cands = []
    for n in n_list:
        if time.time() > deadline - 150.0:
            break
        local = []
        if n == n_par:
            bases = [parent.copy()]
        else:
            bases = [resample_nearest(parent, n), resample_linear(parent, n)]
        for b in bases:
            variants = make_variants(b)
            for v in variants:
                vh = project_feasible(v, 0.5 * n)
                c = compute_c5(vh)
                local.append((c, vh, n))
        local.sort(key=tuple_key0)
        for i in range(min(2, len(local))):
            all_cands.append(local[i])

    all_cands.append((best_c, best_h, best_n))
    all_cands.sort(key=tuple_key0)

    top = []
    seen = {}
    for c, hh, nn in all_cands:
        cnt = seen.get(nn, 0)
        if cnt < 2:
            top.append((c, hh, nn))
            seen[nn] = cnt + 1
        if len(top) >= 5:
            break

    total_alloc = max(10.0, deadline - time.time() - 35.0)
    fracs = [0.40, 0.26, 0.16, 0.10, 0.08]

    for i, item in enumerate(top):
        c0, hh, nn = item
        if time.time() > deadline - 35.0:
            break
        remaining = deadline - time.time() - 35.0
        if remaining < 20.0:
            break
        if i < len(fracs):
            t_alloc = total_alloc * fracs[i]
        else:
            t_alloc = total_alloc * 0.05
        t_alloc = min(t_alloc, remaining)
        if t_alloc < 20.0:
            continue
        end = time.time() + t_alloc
        top_count = max(40, min(120, nn // 2))
        bh, bc, bnn = local_search(hh, end, rng, beta=40.0, top_count=top_count,
                                   gap=0.004, base_temp=1.0e-5)
        bh = project_feasible(bh, 0.5 * bnn)
        bc = compute_c5(bh)
        try:
            ec = float(evaluate_erdos_solution(bh))
            if ec < bc:
                bc = ec
        except Exception:
            pass
        if bc < best_c - 1e-14:
            best_c = bc
            best_h = bh
            best_n = bnn

    if time.time() < deadline - 35.0:
        remaining = deadline - time.time() - 25.0
        if remaining > 30.0:
            t_pol = min(120.0, remaining)
            best_h, best_c = polish_lbfgs(best_h, time.time() + t_pol, (50.0, 100.0, 200.0))

    best_h, best_n = fix_solution_even(best_h, best_n)
    best_h = project_feasible(best_h, 0.5 * best_n)
    final_c = compute_c5(best_h)
    try:
        ec = float(evaluate_erdos_solution(best_h))
        if ec < final_c:
            final_c = ec
    except Exception:
        pass

    return best_h, float(final_c), int(best_n)
