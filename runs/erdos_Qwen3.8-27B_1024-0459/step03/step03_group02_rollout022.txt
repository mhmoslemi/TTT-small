```python
def _corr_grad_single(x, n, j):
    """Gradient of correlation[j] with respect to x."""
    grad = np.zeros(n)
    offset = j - (n - 1)
    if offset >= 0:
        # corr = sum_{i=offset}^{n-1} x[i]*(1-x[i-offset])
        # d/dx[k] = (1-x[k-offset]) for offset <= k <= n-1
        #          - x[k+offset] for 0 <= k <= n-1-offset
        lo1, hi1 = offset, n  # range for first term: k in [offset, n-1]
        lo2, hi2 = 0, n - offset  # range for second term: k in [0, n-1-offset]
        grad[lo1:hi1] += (1.0 - x[lo1 - offset : hi1 - offset])  # 1 - x[k-offset] for k in [offset, n-1]
        grad[lo2:hi2] -= x[lo2 + offset : hi2 + offset]  # x[k+offset] for k in [0, n-1-offset]
    else:
        o = -offset
        # corr = sum_{i=0}^{n-1-o} x[i]*(1-x[i+o])
        # d/dx[k] = (1-x[k+o]) for 0 <= k <= n-1-o
        #          - x[k-o] for o <= k <= n-1
        lo1, hi1 = 0, n - o  # first term range
        lo2, hi2 = o, n  # second term range
        grad[lo1:hi1] += (1.0 - x[lo1 + o : hi1 + o])  # 1 - x[k+o] for k in [0, n-1-o]
        grad[lo2:hi2] -= x[lo2 - o : hi2 - o]  # x[k-o] for k in [o, n-1]
    return grad
```
