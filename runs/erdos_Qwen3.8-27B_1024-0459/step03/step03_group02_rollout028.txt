        # Negative lags (which in the full corr are at index n-1-d)
        # corr[n-1-d] = sum_{i=d}^{n-1} h[i]*(1-h[i-d])
        # When h[j] goes 0->1:
        #   As h[i]: i=j (if j>=d): delta += (1-h[j-d])
        #   As h[i-d]: i=j+d (if j+d<n, i.e., j<n-d): delta += -(1-h[j+d]) = -(1-h[j+d])
        for d in range(1, n):
            if j >= d:
                delta[n-1-d] += (1.0 - h[j-d])
            if j + d < n:
                delta[n-1-d] -= (1.0 - h[j+d])
    else:  # flipping 1 -> 0
        for d in range(1, n):
            if j >= d:
                delta[n-1+d] -= (1.0 - h[j-d])
            if j + d < n:
                delta[n-1+d] += h[j+d]
        for d in range(1, n):
            if j >= d:
                delta[n-1-d] -= (1.0 - h[j-d])
            if j + d < n:
                delta[n-1-d] += (1.0 - h[j+d])
    
    return delta
```

Hmm, this O(n) update per flip is good but doing it for all n positions gives O(n²) per evaluation. Then finding the best flip is another O(n²). Total per iteration: O(n²). For n=160, that's 25600 ops per iteration. Very fast.

But wait, I need to be more careful. When I flip h[j], the correlation changes for ALL lags d. But I only need to compute change in the MAX correlation. So:
- Current max = M at lag k*.
- After flip j: new corr at lag k* = corr[k*] + delta[k*].
- New max could be at lag k* or at some other lag.

To properly evaluate, I need to check all lags. But I can be smarter:
- The new max is max(old_max - |delta[k*]|, max over other lags of corr[k] + delta[k]).
- The "other lags" max: for each lag k ≠ k*, the value is corr[k] + delta[k]. Since |delta[k]| ≤ 1 (it's a single bit change), the new max is at most old_max + 1.
- So I can focus on lags that are within 1 of the current max.

This is a key optimization! Let me implement this.

Actually, for the pair-swap (flip one 0 to 1 AND one 1 to 0), the change in each correlation is bounded by 2. So I should look at lags within 2 of the current max.

```python
import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 2.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def to_binary(h, n):
    target = int(n / 2)
    h_float = np.array(h, dtype=float)
    # Sort by value, assign top n/2 to 1
    order = np.argsort(-h_float)
    h_bin = np.zeros(n, dtype=float)
    h_bin[order[:target]] = 1.0
    return h_bin

def thue_morse(n):
    """Generate Thue-Morse sequence of length n."""
    h = np.zeros(n, dtype=float)
    for i in range(n):
        # Count bits
        cnt = 0
        v = i
        while v:
            cnt += (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt & 1 else 0.0
    return h

def make_binary_feasible(h_init, n):
    """Make h binary with exactly n/2 ones, starting from h_init."""
    target = int(n / 2)
    h = np.array(h_init, dtype=float)
    # Get values sorted
    order = np.argsort(-h)
    h_bin = np.zeros(n, dtype=float)
    h_bin[order[:target]] = 1.0
    return h_bin

def fast_delta_for_swap(h, i, j, n):
    """Compute correlation delta when h[i] goes 0->1 and h[j] goes 1->0.
    Returns delta array of length 2n-1."""
    # For h[i]: 0 -> 1
    # Positive lag d: corr[n-1+d] gets (1-h[i-d]) if i>=d, and -h[i+d] if i+d<n
    # Negative lag d: corr[n-1-d] gets (1-h[i-d]) if i>=d, and -(1-h[i+d]) if i+d<n
    # For h[j]: 1 -> 0
    # Positive lag d: corr[n-1+d] gets -(1-h[j-d]) if j>=d, and +h[j+d] if j+d<n
    # Negative lag d: corr[n-1-d] gets -(1-h[j-d]) if j>=d, and +(1-h[j+d]) if j+d<n
    
    delta = np.zeros(2*n - 1)
    
    # h[i] 0->1
    # Positive lags d=1..n-1
    d_pos = np.arange(1, n)  # d values
    # Condition i >= d: d <= i
    mask1 = d_pos <= i
    mask2 = (i + d_pos) < n
    # For positive lags:
    # delta[n-1+d] += (1-h[i-d]) for d <= i
    # delta[n-1+d] -= h[i+d] for i+d < n
    if np.any(mask1):
        idxs = n - 1 + d_pos[mask1]
        delta[idxs] += (1.0 - h[i - d_pos[mask1]])
    if np.any(mask2):
        idxs = n - 1 + d_pos[mask2]
        delta[idxs] -= h[i + d_pos[mask2]]
    # For negative lags:
    # delta[n-1-d] += (1-h[i-d]) for d <= i
    # delta[n-1-d] -= (1-h[i+d]) for i+d < n
    if np.any(mask1):
        idxs = n - 1 - d_pos[mask1]
        delta[idxs] += (1.0 - h[i - d_pos[mask1]])
    if np.any(mask2):
        idxs = n - 1 - d_pos[mask2]
        delta[idxs] -= (1.0 - h[i + d_pos[mask2]])
    
    # h[j] 1->0
    mask3 = d_pos <= j
    mask4 = (j + d_pos) < n
    if np.any(mask3):
        idxs = n - 1 + d_pos[mask3]
        delta[idxs] -= (1.0 - h[j - d_pos[mask3]])
    if np.any(mask4):
        idxs = n - 1 + d_pos[mask4]
        delta[idxs] += h[j + d_pos[mask4]]
    if np.any(mask3):
        idxs = n - 1 - d_pos[mask3]
        delta[idxs] -= (1.0 - h[j - d_pos[mask3]])
    if np.any(mask4):
        idxs = n - 1 - d_pos[mask4]
        delta[idxs] += (1.0 - h[j + d_pos[mask4]])
    
    return delta

def binary_local_search(h_bin, n, max_sweeps=50, rng=None):
    """Exhaustive pair-swap local search on binary sequence."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = h_bin.copy()
    corr = compute_corr(h)
    max_c = np.max(corr)
    target = int(n / 2)
    
    one_idx = np.where(h == 1)[0].tolist()
    zero_idx = np.where(h == 0)[0].tolist()
    
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        shuffled_zeros = zero_idx[:]
        rng.shuffle(shuffled_zeros)
        for i in shuffled_zeros:
            if remaining_time() < 3:
                break
            # For each 1-position j, check if swapping i and j helps
            # We only need to check if the new max < current max
            for j in one_idx:
                # Quick check: if i and j are far apart, the delta is spread out
                # Compute delta
                delta = fast_delta_for_swap(h, i, j, n)
                new_max_lag = np.argmax(corr + delta)
                new_max = corr[new_max_lag] + delta[new_max_lag]
                if new_max < max_c - 1e-12:
                    # Apply
                    h[i] = 1.0
                    h[j] = 0.0
                    zero_idx.remove(i)
                    one_idx.append(i)
                    one_idx.remove(j)
                    zero_idx.append(j)
                    corr = corr + delta
                    max_c = new_max
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    
    return h, max_c * (2.0 / n)

def simulated_annealing_binary(h_bin, n, t0=1.0, t_end=0.01, n_iters=50000, rng=None):
    """Simulated annealing on binary sequences via pair swaps."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = h_bin.copy()
    corr = compute_corr(h)
    max_c = np.max(corr)
    target = int(n / 2)
    
    one_idx = np.where(h == 1)[0].tolist()
    zero_idx = np.where(h == 0)[0].tolist()
    
    best_h = h.copy()
    best_max = max_c
    
    for it in range(n_iters):
        if remaining_time() < 5:
            break
        T = t0 * (t_end / t0) ** (it / max(n_iters, 1))
        
        # Pick random pair
        i = zero_idx[rng.randint(len(zero_idx))]
        j = one_idx[rng.randint(len(one_idx))]
        
        delta = fast_delta_for_swap(h, i, j, n)
        new_max = np.max(corr + delta)
        delta_e = new_max - max_c
        
        if delta_e < 0 or rng.random() < np.exp(-delta_e / max(T, 1e-10)):
            # Accept
            h[i] = 1.0
            h[j] = 0.0
            zero_idx.remove(i)
            one_idx.append(i)
            one_idx.remove(j)
            zero_idx.append(j)
            corr = corr + delta
            max_c = new_max
            
            if max_c < best_max:
                best_max = max_c
                best_h = h.copy()
    
    return best_h, best_max * (2.0 / n)

def continuous_refine(h, n, maxiter=500):
    """SLSQP refinement with smooth max objective."""
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        beta = 200.0
        shifted = np.minimum(beta * (corr - m), 0.0)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        beta = 200.0
        shifted = np.minimum(beta * (corr - m), 0.0)
        weights = np.exp(shifted) / np.sum(np.exp(shifted))
        # Gradient of corr[k] w.r.t. x
        # corr[n-1+d] = sum_i x[i+d](1-x[i])
        # d/dx[j] of corr[n-1+d]:
        #   if j = i+d (i=j-d): (1-x[j-d])
        #   if j = i: -x[j+d]
        grad = np.zeros(n)
        for d in range(n):
            # corr[n-1+d]
            idx = n - 1 + d
            w = weights[idx]
            if d == 0:
                # corr[n-1] = sum x[i](1-x[i]), grad = 1-2x
                grad += w * (1.0 - 2.0 * x)
            else:
                # positive direction
                n_pairs = n - d
                grad[d:d+n_pairs] += w * (1.0 - x[:n_pairs])
                grad[:n_pairs] -= w * x[d:d+n_pairs]
                # negative direction (corr[n-1-d])
                w_neg = weights[n-1-d]
                grad[:n_pairs] += w_neg * (1.0 - x[d:d+n_pairs])
                grad[d:d+n_pairs] -= w_neg * x[:n_pairs]
        return grad * dx
    
    def constraint_fun(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
