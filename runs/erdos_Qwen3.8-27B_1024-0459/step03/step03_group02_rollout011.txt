        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def lag_gradient(h, n, m):
    grad = np.zeros(n)
    if m >= 0:
        n_pairs = n - m
        if n_pairs <= 0:
            return grad
        grad[:n_pairs] += (1.0 - h[m:n])
        grad[m:] += (-h[:n_pairs])
    else:
        abs_m = -m
        n_pairs = n - abs_m
        if n_pairs <= 0:
            return grad
        grad[abs_m:] += (1.0 - h[:n_pairs])
        grad[:n_pairs] += (-h[abs_m:n])
    return grad

def _obj_func(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = _BETA_OPT * (corr - m)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    return (m + np.log(sum_exp) / _BETA_OPT) * (2.0 / _N_OPT)

def _grad_func(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = _BETA_OPT * (corr - m)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    grad = np.zeros(_N_OPT)
    significant = np.where(weights > 1e-12)[0]
    for k_idx in significant:
        m_lag = k_idx - (_N_OPT - 1)
        grad += weights[k_idx] * lag_gradient(x, _N_OPT, m_lag)
    grad *= (2.0 / _N_OPT)
    return grad

def _sum_constraint(x):
    return np.sum(x) - _N_OPT / 2.0

def _sum_jac(x):
    return np.ones(_N_OPT)

def lbfgsb_optimize(h_init, n, beta, maxiter):
    global _N_OPT, _BETA_OPT
    _N_OPT = n
    _BETA_OPT = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jac}]
    try:
        res = minimize(_obj_func, h_init, jac=_grad_func,
                      method='L-BFGS-B', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12,
                               'maxls': 100, 'maxfun': maxiter * 5, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h_init.copy()

def subgrad_phase(h, n, max_iters, target_count=5):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    stall = 0
    
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        max_corr = np.max(corr)
        threshold = max_corr - max(1.0, 0.03 * max_corr)
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])
        elif len(active) > target_count:
            order = np.argsort(corr[active])[::-1][:target_count]
            active = active[order]
        
        grad = np.zeros(n)
        for k_idx in active:
            m = k_idx - (n - 1)
            grad += lag_gradient(h, n, m)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        improved = False
        base_step = 0.06 * 0.85 ** stall / (1.0 + 0.003 * it)
        for mult in [1.0, 0.5, 2.0, 0.25, 0.1, 4.0, 0.05, 8.0, 0.01]:
            step = base_step * mult
            h_t = h - step * grad
            h_t = project_feasible(h_t, n)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5 - 1e-15:
                best_c5, best_h = c5_t, h_t.copy()
                h = h_t.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 30:
                break
    return best_h, best_c5

def coord_descent(h, n, max_sweeps=100):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        imp = False
        for i in range(n):
            if remaining_time() < 1:
                break
            for dj in [1, 2, 3, n//2, n//3]:
                j = (i + dj) % n
                if i == j:
                    continue
                for delta in [0.02, 0.05, 0.01, 0.08, -0.02, -0.05, -0.01]:
                    if h[i] - delta < -1e-10 or h[j] + delta > 1 + 1e-10:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c5_t = compute_c5(h)
                    if c5_t < best_c5 - 1e-14:
                        best_c5, best_h = c5_t, h.copy()
                        imp = True
                        break
                    else:
                        h[i] += delta
                        h[j] -= delta
                else:
                    continue
                break
            else:
                continue
            break
        if not imp:
            break
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src_h)
