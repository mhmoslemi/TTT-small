    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)


def obj_lbf(h, n, beta, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    smooth_max = (m + np.log(np.sum(exps)) / beta) * dx
    sum_pen = lam * (np.sum(h) - n / 2.0) ** 2
    return smooth_max + sum_pen


def grad_lbf(h, n, beta, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    weights = exps / np.sum(exps)
    grad_corr = np.zeros(2 * n - 1)
    for k_idx in range(n_corr):
        if weights[k_idx] < 1e-12:
            continue
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            n_pairs = n - offset
            if n_pairs <= 0:
                continue
            grad_corr[:n_pairs] += (1.0 - h[offset:offset + n_pairs])
            grad_corr[offset:offset + n_pairs] -= h[:n_pairs]
        else:
            abs_off = -offset
            n_pairs = n - abs_off
            if n_pairs <= 0:
                continue
            grad_corr[abs_off:abs_off + n_pairs] += (1.0 - h[:n_pairs])
            grad_corr[:n_pairs] -= h[abs_off:abs_off + n_pairs]
    grad = dx * grad_corr
    sum_diff = np.sum(h) - n / 2.0
    grad += 2.0 * lam * sum_diff
    return grad


def obj_slsqp(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    return (m + np.log(np.sum(exps)) / beta) * dx


def con_sum(h, n):
    return np.sum(h) - n / 2.0


def nesterov_subgrad(h_init, n, budget_remain, top_k=5, max_iters=300):
    h_best = h_init.copy()
    c5_best = compute_c5(h_best)
    h = h_init.copy()
    h_prev = h_init.copy()
    base_step = 0.03

    for it in range(max_iters):
        if budget_remain() < 3.0:
            break
        t = (it + 1) / max_iters
        theta = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = theta * h - (theta - 1.0) * h_prev
        y = project_feasible(y, n)

        dx = 2.0 / n
        corr = np.correlate(y, 1.0 - y, mode='full')
        n_corr = len(corr)
        top_idx = np.argsort(corr)[::-1][:top_k]

        grad = np.zeros(n)
        for k_idx in top_idx:
            offset = int(k_idx) - (n - 1)
            if offset >= 0:
                npairs = n - offset
                if npairs <= 0:
                    continue
                grad[:npairs] += (1.0 - y[offset:offset + npairs])
                grad[offset:offset + npairs] -= y[:npairs]
            else:
                aoff = -offset
                npairs = n - aoff
                if npairs <= 0:
                    continue
                grad[aoff:aoff + npairs] += (1.0 - y[:npairs])
                grad[:npairs] -= y[aoff:aoff + npairs]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm

        step = base_step / (1.0 + 0.005 * it)
        h_new = y - step * grad
        h_new = project_feasible(h_new, n)

        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-12:
            c5_best = c5_new
            h_best = h_new.copy()

        h_prev = h.copy()
        h = h_new

    return h_best, c5_best


def pairwise_swap(h_init, n, budget_remain, max_sweeps=50):
    h_best = h_init.copy()
    c5_best = compute_c5(h_best)
    h = h_init.copy()

    ones_idx = np.where(h > 0.5)[0]
    zeros_idx = np.where(h < 0.5)[0]

    for sweep in range(max_sweeps):
        if budget_remain() < 3.0:
            break
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full')
        n_corr = len(corr)
        top_lags = np.argsort(corr)[::-1][:8]

        order = np.random.permutation(len(ones_idx))
        for oi in order:
            if budget_remain() < 2.0:
                break
            i = int(ones_idx[oi]) if oi < len(ones_idx) else -1
            if i < 0:
                break

            # Try swapping h[i] with each zero position j (small candidates)
            candidates = zeros_idx[np.random.choice(len(zeros_idx), min(50, len(zeros_idx)), replace=False)]
            best_delta = 0.0
            best_j = -1
            for j in candidates:
                j = int(j)
                if j == i:
                    continue
                # Try: decrease h[i] by delta, increase h[j] by delta
                for delta in [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]:
                    new_h = h.copy()
                    new_h[i] = max(0.0, h[i] - delta)
                    new_h[j] = min(1.0, h[j] + delta)
                    # Adjust sum
                    diff = (n / 2.0) - np.sum(new_h)
                    if abs(diff) > 1e-10:
                        new_h = project_feasible(new_h, n)
                    c5_t = compute_c5(new_h)
                    if c5_t < c5_best - 1e-11:
                        best_delta = delta
                        best_j = j
                        c5_best = c5_t
                        h_best = new_h.copy()
                        improved = True
                        break
                if best_delta > 0:
                    break
            if best_delta > 0:
                h = h_best.copy()
                ones_idx = np.where(h > 0.5)[0]
                zeros_idx = np.where(h < 0.5)[0]

        if not improved and sweep > 5:
            break

    return h_best, c5_best


def coord_descent_fine(h_init, n, budget_remain, max_sweeps=20):
    h_best = h_init.copy()
    c5_best = compute_c5(h_best)
    h = h_init.copy()
    rng = np.random.RandomState(12345)

    for sweep in range(max_sweeps):
        if budget_remain() < 3.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if budget_remain() < 2.0:
                break
            old = h[i]
            for new_val in [0.0, 1.0, 0.5]:
                if abs(new_val - old) < 1e-6:
                    continue
                h[i] = new_val
                h = project_feasible(h, n)
                c5_t = compute_c5(h)
                if c5_t < c5_best - 1e-12:
                    c5_best = c5_t
                    h_best = h.copy()
                    improved = True
                    old = new_val
                else:
                    h = h_best.copy()
            for delta in [0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.15, -0.15]:
                if budget_remain() < 1.5:
                    break
                trial = h_best.copy()
                trial[i] = np.clip(trial[i] + delta, 0.0, 1.0)
                trial = project_feasible(trial, n)
                c5_t = compute_c5(trial)
                if c5_t < c5_best - 1e-12:
                    c5_best = c5_t
                    h_best = trial.copy()
                    improved = True
        if not improved and sweep > 3:
            break

    return h_best, c5_best


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = 0
        v = i
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return h


def make_golden_ratio(n):
    h = np.zeros(n)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return h


def lbfqs_optimize(h_init, n, budget_remain, beta_seq, lam_seq, max_iter_each):
    h_cur = h_init.copy()
    c5_cur = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for beta, lam in zip(beta_seq, lam_seq):
        if budget_remain() < 10.0:
            break
        try:
            res = minimize(
                lambda x: obj_lbf(x, n, beta, lam),
                h_cur,
                jac=lambda x: grad_lbf(x, n, beta, lam),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': max_iter_each, 'ftol': 1e-18, 'iprint': -1}
            )
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-13:
                c5_cur = c5_new
                h_cur = h_new
        except Exception:
            pass

    # Constrained SLSQP polish
    if budget_remain() > 15.0:
        for beta in [200, 500, 1000]:
            if budget_remain() < 8.0:
                break
            try:
                res = minimize(
                    lambda x: obj_slsqp(x, n, beta),
                    h_cur,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': lambda x: con_sum(x, n)}],
                    options={'maxiter': 300, 'ftol': 1e-14, 'iprint': 0}
                )
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-13:
                    c5_cur = c5_new
                    h_cur = h_new
            except Exception:
                pass

    return h_cur, c5_cur


def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = make_thue_morse(parent_n)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("No parent. Thue-Morse: n={}, C5={:.8f}".format(parent_n, parent_c5))

    # Phase 1: Optimize at original resolution with L-BFGS-B + SLSQP
    print("\n=== Phase 1: L-BFGS-B at n={} ===".format(best_n))
    n = best_n
    h_opt, c5_opt = lbfqs_optimize(
        best_h, n, remaining_time,
        beta_seq=[30, 60, 120, 250],
        lam_seq=[10.0, 50.0, 200.0, 500.0],
        max_iter_each=150
    )
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()
        best_n = n
        print("  Phase1 L-BFGS: C5={:.8f}".format(best_c5))

    # Phase 2: Try higher resolution n=512
    print("\n=== Phase 2: Higher resolution n=512 ===")
    n_hi = 512
    h_hi = project_feasible(interp_to(best_h, best_n, n_hi), n_hi)
    c5_hi = compute_c5(h_hi)
    print("  Interpolated C5={:.8f}".format(c5_hi))

    h_hi, c5_hi = lbfqs_optimize(
        h_hi, n_hi, remaining_time,
        beta_seq=[30, 60, 120, 250, 500],
        lam_seq=[10.0, 50.0, 200.0, 500.0, 1000.0],
        max_iter_each=100
    )
    if c5_hi < best_c5:
        best_c5 = c5_hi
        best_h = h_hi.copy()
        best_n = n_hi
        print("  Phase2 n=512: C5={:.8f}".format(best_c5))
    else:
        print("  Phase2 n=512 no improvement ({}): keeping n={}".format(c5_hi, best_n))

    # Phase 3: Nesterov subgradient at best resolution
    if remaining_time() > 30.0:
        print("\n=== Phase 3: Nesterov Subgradient ===")
        n = best_n
        best_h, best_c5 = nesterov_subgrad(best_h, n, remaining_time, top_k=5, max_iters=200)
        print("  After Nesterov: n={}, C5={:.8f}".format(n, best_c5))

    # Phase 4: Pairwise swap local search
    if remaining_time() > 30.0:
        print("\n=== Phase 4: Pairwise Swap ===")
        n = best_n
        best_h, best_c5 = pairwise_swap(best_h, n, remaining_time, max_sweeps=40)
        print("  After pairwise swap: C5={:.8f}".format(best_c5))

    # Phase 5: Fine coordinate descent
    if remaining_time() > 20.0:
        print("\n=== Phase 5: Coordinate Descent ===")
        n = best_n
        best_h, best_c5 = coord_descent_fine(best_h, n, remaining_time, max_sweeps=15)
        print("  After coord descent: C5={:.8f}".format(best_c5))

    # Phase 6: Second round at potentially different resolution
    if remaining_time() > 60.0:
        print("\n=== Phase 6: Second optimization round ===")
        # Try n=384 as well
        n_alt = 384
        h_alt = project_feasible(interp_to(best_h, best_n, n_alt), n_alt)
        c5_alt = compute_c5(h_alt)
        h_alt, c5_alt = lbfqs_optimize(
            h_alt, n_alt, remaining_time,
            beta_seq=[50, 100, 200, 400],
            lam_seq=[50.0, 100.0, 300.0, 500.0],
            max_iter_each=120
        )
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt.copy()
            best_n = n_alt
            print("  Phase6 n=384 improved: C5={:.8f}".format(best_c5))

        # Also re-optimize at best_n with more iterations
        n = best_n
        h2, c5_2 = lbfqs_optimize(
            best_h, n, remaining_time,
            beta_seq=[100, 200, 500, 1000],
            lam_seq=[200.0, 500.0, 1000.0, 2000.0],
            max_iter_each=200
        )
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n
            print("  Phase6 re-opt n={}: C5={:.8f}".format(n, best_c5))

    # Phase 7: Final Nesterov + swap
    if remaining_time() > 20.0:
        print("\n=== Phase 7: Final refinement ===")
        n = best_n
        best_h, best_c5 = nesterov_subgrad(best_h, n, remaining_time, top_k=3, max_iters=100)
        if remaining_time() > 10.0:
            best_h, best_c5 = pairwise_swap(best_h, n, remaining_time, max_sweeps=20)
        print("  After Phase7: C5={:.8f}".format(best_c5))

    # Phase 8: Final SLSQP polish
    if remaining_time() > 10.0:
        print("\n=== Phase 8: Final SLSQP polish ===")
        n = best_n
        bounds = [(0.0, 1.0)] * n
        for beta in [1000, 3000]:
            if remaining_time() < 5.0:
                break
            try:
                res = minimize(
                    lambda x: obj_slsqp(x, n, beta),
                    best_h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': lambda x: con_sum(x, n)}],
                    options={'maxiter': 400, 'ftol': 1e-15, 'iprint': 0}
                )
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h_new
            except Exception:
                pass
        print("  Final: C5={:.8f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
