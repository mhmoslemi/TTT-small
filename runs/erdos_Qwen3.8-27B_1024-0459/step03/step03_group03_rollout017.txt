def _ggrad_single(h, d):
    """Gradient of c[n-1+d] w.r.t. h, where d is the lag offset."""
    n = len(h)
    g = np.zeros(n)
    if d == 0:
        # c = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
        # dc/dh[j] = 1 - 2*h[j]
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        # c = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
        # dc/dh[j] = (1-h[j+d])*[j+d<n] - h[j-d]*[j-d>=0]
        for j in range(n):
            if j + d < n:
                g[j] += 1.0 - h[j + d]
            if j - d >= 0:
                g[j] -= h[j - d]
    else:
        d2 = -d
        # c = sum_{i=d2}^{n-1} h[i]*(1-h[i-d2])  ... wait
        # Actually c[n-1-d2] = sum_i h[i]*(1-h[i-d2]) where 0<=i-d2<n
        # = sum_{i=d2}^{n-1} h[i]*(1-h[i-d2])
        # dc/dh[j] = (1-h[j-d2])*[j-d2>=0] - h[j+d2]*[j+d2<n]
        for j in range(n):
            if j - d2 >= 0:
                g[j] += 1.0 - h[j - d2]
            if j + d2 < n:
                g[j] -= h[j + d2]
    return g

For d < 0 (let d2 = -d):
