            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        except Exception:
            pass
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return max(0.0, budget_s - (time.time() - t0) - 5.0)
    
    # Get parent solution
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn > 10:
            best_h = ph.copy()
            best_c5 = compute_c5(ph)
            best_n = pn
    except Exception:
        pass
    
    if best_h is None:
        best_n = 256
        best_h = project_feasible(np.ones(best_n) * 0.5, best_n)
        best_c5 = compute_c5(best_h)
    
    # Multi-resolution search
    resolutions = [256, 320, 384, 512]
    # Always include parent resolution
    if best_n not in resolutions:
        resolutions.append(best_n)
    resolutions = sorted(set(resolutions))
    
    # Time allocation per resolution
    n_res = len(resolutions)
    time_per_res = (budget_s * 0.7) / n_res  # 70% for main search
    time_remaining_search = budget_s * 0.25  # 25% for diversification
    time_polish = budget_s * 0.05  # 5% for final polish
    
    for n in resolutions:
        if time_left() < 30:
            break
        
        res_time = min(time_per_res, time_left() * 0.4)
        if res_time < 20:
            res_time = time_left() * 0.4
        
        # Create starting points
        if n == best_n:
            h_start = best_h.copy()
        else:
            h_start = interpolate_h(best_h, best_n, n)
        
        # Generate diverse initializations
        inits = []
        inits.append(h_start.copy())
        
        # Binary-like initialization
        t = np.arange(n) / n
        for freq in [3, 5, 7, 11]:
            phase = rng.rand() * 2 * np.pi
            hf = np.clip(0.5 + 0.45 * np.cos(2 * np.pi * freq * t + phase), 0, 1)
            inits.append(project_feasible(hf, n))
        
        # Random perturbations of parent
        for scale in [0.05, 0.1]:
            hp = h_start + scale * rng.randn(n)
            inits.append(project_feasible(hp, n))
        
        # Golden-ratio inspired pattern
        golden = (1.0 + np.sqrt(5.0)) / 2.0
        positions = np.floor(np.arange(n) * golden).astype(int) % n
        h_golden = np.zeros(n)
        h_golden[:n // 2] = 1.0
        h_golden = project_feasible(h_golden, n)
        inits.append(h_golden)
        
        # Sort-based: arrange by cosine
        h_cos = np.clip(0.5 + 0.4 * np.sin(2 * np.pi * np.arange(n) * 4.0 / n), 0, 1)
        inits.append(project_feasible(h_cos, n))
        
        time_per_init = max(10.0, res_time / len(inits))
        
        for h0 in inits:
            if time_left() < 15:
                break
            tl = min(time_per_init, time_left() * 0.8)
            
            # Phase 1: Efficient coordinate descent (main workhorse)
            h_cd, c5_cd = efficient_cd(h0, n, sweeps=200, time_limit=tl * 0.7, t0=t0)
            
            # Phase 2: Subgradient for diversification
            if time_left() > 10:
                h_sg, c5_sg = subgradient_step(h_cd, n, lr=2.0/n, iters=3000, time_limit=tl * 0.15, t0=t0)
            else:
                h_sg, c5_sg = h_cd, c5_cd
            
            # Phase 3: CD again from subgradient result
            if time_left() > 10:
                h_cd2, c5_cd2 = efficient_cd(h_sg, n, sweeps=150, time_limit=tl * 0.15, t0=t0)
            else:
                h_cd2, c5_cd2 = h_sg, c5_sg
            
            # Take best from this init
            candidates = [(h_cd, c5_cd), (h_sg, c5_sg), (h_cd2, c5_cd2)]
            cands_best = min(candidates, key=lambda x: x[1])
            
            if cands_best[1] < best_c5:
                best_c5 = cands_best[1]
                best_h = cands_best[0].copy()
                best_n = n
    
    # SLSQP polish on best solution
    if time_left() > 20:
        h_pol, c5_pol = slsqp_polish(best_h, best_n, time_limit=min(60, time_left() * 0.4), t0=t0)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol
    
    # Final CD polish
    if time_left() > 20:
        h_final, c5_final = efficient_cd(best_h, best_n, sweeps=300, time_limit=time_left() * 0.5, t0=t0)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    
    # One more SLSQP
    if time_left() > 10:
        h_pol2, c5_pol2 = slsqp_polish(best_h, best_n, time_limit=min(30, time_left() * 0.6), t0=t0)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2
    
    # Final CD
    if time_left() > 10:
        h_f2, c5_f2 = efficient_cd(best_h, best_n, sweeps=200, time_limit=time_left() * 0.8, t0=t0)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    # Validate
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
