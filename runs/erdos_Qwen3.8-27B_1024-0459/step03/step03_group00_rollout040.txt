def project_waterfill(v, target):

    n = len(v)
    v = v.astype(float)
    
    def sum_clip(lam):
        return np.sum(np.clip(v - lam, 0.0, 1.0))
    
    # Bisection for lambda
    lo, hi = v.min() - 1.0, v.max() + 1.0
    # sum_clip is decreasing in lam

    if sum_clip(lo) < target:
