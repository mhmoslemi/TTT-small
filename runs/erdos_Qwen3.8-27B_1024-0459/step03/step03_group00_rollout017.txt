    """Project to binary with exactly n/2 ones."""
    h = np.array(h, dtype=float).copy()
    # Sort by value, set top n/2 to 1, rest to 0
    idx = np.argsort(-h)
    h[0] = 0  # dummy
    h_new = np.zeros(n)
    h_new[idx[:n//2]] = 1.0
    return h_new

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = 0
        v = i
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return h

def make_golden_ratio(n):
    h = np.zeros(n)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return h

def make_random_bin(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def binary_local_search(h, n, rng, max_sweeps=200, time_limit=100):
    """Local search for binary sequences. Try swaps."""
    start = time.time()
    c5 = compute_c5(h)
    h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def bit_count(i):
    cnt = 0
    v = i
    while v:
        cnt += (v & 1)
        v >>= 1
    return cnt

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bit_count(i) % 2 == 1 else 0.0
    return h

def make_golden_ratio(n):
    h = np.zeros(n)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        h[i] = 1.0 if ((i * gr) % 1.0) < 0.5 else 0.0
    return h

def make_random_bin(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[: n // 2]] = 1.0
    return h

def exhaustive_swap_ls(h, n, rng, max_sweeps=200, time_budget=100.0):
    """Exhaustive local search: try all swaps, take best improvement each sweep."""
    start = time.time()
    h = h.copy()
    c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_budget:
            break
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        best_c5 = c5
        best_i, best_j = -1, -1
        
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c5_trial = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                if c5_trial < best_c5 - 1e-14:
                    best_c5 = c5_trial
                    best_i, best_j = i, j
        
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            c5 = best_c5
        else:
            break
    
    return h, c5

            h_par_bin[ones[-diff:]] = 0.0
    inits.append(h_par_bin)
    names.append("parent_bin")
    
    inits.append(make_thue_morse(n_work))
    names.append("thue_morse")
    
    inits.append(make_golden_ratio(n_work))
    names.append("golden")
    
    for ri in range(4):
        inits.append(make_random_bin(rng, n_work))
        names.append("rand_{}".format(ri))
    
    inits.append(make_pascal(n_work, rng))
    names.append("pascal")
    
    # Evaluate all inits
    init_results = []
    for name, h_init in zip(names, inits):
        c5_val = compute_c5(h_init)
        init_results.append((c5_val, name, h_init.copy()))
        print("  {}: C5={:.10f}".format(name, c5_val))
    
    init_results.sort(key=lambda x: x[0])
    
    # Update best if any init is better
    for c5_val, name, h_val in init_results:
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_val.copy()
            best_n = n_work
            print("  New best from {}: C5={:.10f}".format(name, best_c5))
    
    # Phase 2: Exhaustive local search on top 3 inits
    print("\n=== Phase 2: Exhaustive Local Search ===")
    for idx in range(min(3, len(init_results))):
        if remaining() < 60.0:
            break
        c5_init, name, h_init = init_results[idx]
        print("  LS on {} (initial C5={:.10f})...".format(name, c5_init))
        
        t_phase = max(20.0, min(120.0, remaining() / max(1, 3 - idx)))
        h_ls, c5_ls = exhaustive_ls(h_init, n_work, max_sweeps=500, time_budget=t_phase)
        
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            best_n = n_work
            print("  Improved! {}: C5={:.10f}".format(name, best_c5))
        else:
            print("  No improvement (best LS C5={:.10f})".format(c5_ls))
        
        init_results[idx] = (c5_ls, name, h_ls.copy())
    
    # Phase 3: Simulated annealing on best solution
    print("\n=== Phase 3: Simulated Annealing ===")
    if remaining() > 40.0:
        sa_budget = min(150.0, remaining() * 0.3)
        h_sa, c5_sa = simulated_annealing(best_h, best_n, rng, time_budget=sa_budget, t0=0.03, tmin=0.002)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = len(h_sa)
            print("  SA improved: C5={:.10f}".format(best_c5))
        else:
            print("  SA no improvement (SA C5={:.10f})".format(c5_sa))
    
    # Phase 4: More SA with different params + local search
    print("\n=== Phase 4: SA + LS combos ===")
    if remaining() > 30.0:
        for trial in range(3):
            if remaining() < 20.0:
                break
            h_try = best_h.copy()
            # Perturb: random swaps
            for _ in range(rng.randint(3, 8)):
                ones = np.where(h_try == 1.0)[0]
                zeros = np.where(h_try == 0.0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_try[i] = 0.0
                h_try[j] = 1.0
            
            sa_b = max(15.0, remaining() / 4.0)
            h_sa, c5_sa = simulated_annealing(h_try, best_n, rng, time_budget=sa_b * 0.6, t0=0.05, tmin=0.001)
            h_ls, c5_ls = exhaustive_ls(h_sa, best_n, max_sweeps=200, time_budget=sa_b * 0.4)
            
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
                best_n = len(h_ls)
                print("  Trial {}: C5={:.10f}".format(trial, best_c5))
            else:
                print("  Trial {}: no improvement (C5={:.10f})".format(trial, c5_ls))
    
    # Phase 5: Alpha exchange refinement
    print("\n=== Phase 5: Alpha Exchange ===")
    if remaining() > 15.0:
        h_ref, c5_ref = alpha_exchange_refine(best_h, best_n, time_budget=min(30.0, remaining() * 0.3))
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            best_n = len(h_ref)
            print("  Alpha refined: C5={:.10f}".format(best_c5))
        else:
            print("  No alpha improvement")
    
    # Phase 6: Try n=512
    print("\n=== Phase 6: n=512 ===")
    if remaining() > 80.0:
        n512 = 512
        # Interpolate and threshold
        xp = (np.arange(best_n) + 0.5) * 2.0 / best_n
        x_new = (np.arange(n512) + 0.5) * 2.0 / n512
        h_interp = np.interp(x_new, xp, best_h)
        h512 = np.zeros(n512)
        h512[h_interp < 0.5] = 0.0
        h512[h_interp >= 0.5] = 1.0
        if np.sum(h512) != n512 // 2:
            ones = np.where(h512 > 0.5)[0]
            zeros = np.where(h512 < 0.5)[0]
            diff = n512 // 2 - int(np.sum(h512 > 0.5))
            if diff > 0:
                h512[zeros[:diff]] = 1.0
            elif diff < 0:
                h512[ones[-diff:]] = 0.0
        
        c5_512 = compute_c5(h512)
        print("  Interpolated to n=512: C5={:.10f}".format(c5_512))
        
        # Also try Thue-Morse and golden at 512
        h_tm512 = make_thue_morse(n512)
        c5_tm512 = compute_c5(h_tm512)
        print("  ThueMorse n=512: C5={:.10f}".format(c5_tm512))
        
        h_gr512 = make_golden_ratio(n512)
        c5_gr512 = compute_c5(h_gr512)
        print("  Golden n=512: C5={:.10f}".format(c5_gr512))
        
        # Pick best init for 512
        cands_512 = [(c5_512, h512), (c5_tm512, h_tm512), (c5_gr512, h_gr512)]
        cands_512.sort(key=lambda x: x[0])
        h512_best = cands_512[0][1].copy()
        
        # Random swap LS at 512
        ls_budget = min(120.0, remaining() * 0.5)
        h512_ls, c5_512_ls = random_swap_ls(h512_best, n512, rng, time_budget=ls_budget)
        
        if c5_512_ls < best_c5:
            best_c5 = c5_512_ls
            best_h = h512_ls.copy()
            best_n = n512
            print("  n=512 improved! C5={:.10f}".format(best_c5))
        else:
            print("  n=512 no improvement (best C5={:.10f})".format(c5_512_ls))
        
        # Also try exhaustive LS at 512 if possible
        if remaining() > 40.0:
            h512_es, c5_512_es = exhaustive_ls(h512_ls, n512, max_sweeps=50, time_budget=min(60.0, remaining() * 0.4))
            if c5_512_es < best_c5:
                best_c5 = c5_512_es
                best_h = h512_es.copy()
                best_n = n512
                print("  n=512 exhaustive improved! C5={:.10f}".format(best_c5))
    
    # Phase 7: Continuous SLSQP polish
    print("\n=== Phase 7: Continuous Polish ===")
    if remaining() > 30.0:
        polish_budget = min(80.0, remaining() * 0.4)
        h_pol, c5_pol = continuous_optimize(best_h, best_n, remaining, time_budget=polish_budget)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = len(h_pol)
            print("  Continuous polish: C5={:.10f}".format(best_c5))
        else:
            print("  No continuous improvement")
    
    # Phase 8: Final local search on best
    print("\n=== Phase 8: Final LS ===")
    if remaining() > 15.0:
        h_final, c5_final = exhaustive_ls(best_h, best_n, max_sweeps=200, time_budget=min(40.0, remaining() * 0.5))
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
            print("  Final LS: C5={:.10f}".format(best_c5))
    
    # Phase 9: Coordinate descent (small perturbations)
    if remaining() > 10.0:
        h_cd = best_h.copy()
        c5_cd = best_c5
        n_cd = best_n
        for _ in range(200):
            if remaining() < 5.0:
                break
            i = rng.randint(n_cd)
            old = h_cd[i]
            for delta in [0.01, 0.02, 0.05, 0.1, -0.01, -0.02, -0.05, -0.1]:
                trial = h_cd.copy()
                trial[i] = np.clip(trial[i] + delta, 0.0, 1.0)
                trial = project_feasible(trial, n_cd)
                c5_t = compute_c5(trial)
                if c5_t < c5_cd - 1e-14:
                    c5_cd = c5_t
                    h_cd = trial.copy()
                    if c5_cd < best_c5:
                        best_c5 = c5_cd
                        best_h = h_cd.copy()
                    break
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
