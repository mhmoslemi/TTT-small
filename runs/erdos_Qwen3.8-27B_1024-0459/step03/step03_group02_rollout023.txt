def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Adjust to maintain sum
        # Use Wasserstein-like projection
        for _ in range(100):
            diff = target - np.sum(h)
            if abs(diff) < 1e-14:
                break
            mask = (h > 1e-10) & (h < 1.0 - 1e-10)
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
            else:
                h[:] += diff / n
            h = np.clip(h, 0.0, 1.0)
    # Exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12 and n > 0:
        # Find the element closest to 0.5
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
        # Re-fix if needed
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-12:
            for j in range(n):
                if 0 < h[j] < 1:
                    h[j] += diff2
                    h[j] = np.clip(h[j], 0.0, 1.0)
                    break
    return h
