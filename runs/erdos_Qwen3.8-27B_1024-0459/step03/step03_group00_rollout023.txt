    return dx * m ** (1.0 - p) * grad

# For n=256, n² = 65K which is fine. For n=512, n² = 262K, still OK.
# But doing this in a Python loop over 2n-1 lags is slow. Let me vectorize.

def obj_pnorm_vec(h, p, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_pos = np.maximum(corr, 0.0)
    sp = np.sum(corr_pos ** p)
    if sp < 1e-30:
        return 0.0
    return (sp ** (1.0 / p)) * dx

def con_sum(h):
    return np.sum(h) - len(h) / 2.0

def make_fourier_init(n, rng, n_harmonics=8):
    """Create a periodic-like initialization using Fourier series"""
    t = np.arange(n) / float(n)
    h = np.ones(n) * 0.5
    for k in range(1, n_harmonics + 1):
        amp = rng.uniform(0.05, 0.3) / k
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2 * np.pi * k * t * 2 + phase)  # *2 because domain is [0,2]
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)

def make_wavelet_init(n, rng, n_basis=12):
    """Initialize using a few dyadic wavelets"""
    h = np.ones(n) * 0.5
    for _ in range(n_basis):
        # Random wavelet-like pattern
        freq = rng.randint(2, max(3, n // 4))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4)
        t = np.arange(n) / float(n)
        h += amp * np.sin(2 * np.pi * freq * t + phase)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)

def subgradient_refine(h, n, budget_remain, rng, max_iters=500):
    """Subgradient method on the raw max objective with adaptive step sizes"""
    dx = 2.0 / n
    c5_best = compute_c5(h)
    h_best = h.copy()
    step_scale = 0.1
    
    for it in range(max_iters):
        if budget_remain() < 3.0:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        
        # Find active lags (within 5% of max)
        threshold = max_val * 0.95
        active = np.where(corr >= threshold)[0]
        
        # Compute subgradient: average gradient of active lags
        grad = np.zeros(n)
        for k_idx in active:
            s = k_idx - (n - 1)
            if s >= 0:
                n_pairs = n - s
                grad[:n_pairs] += (1.0 - h[s:n])
                grad[s:] -= h[:n_pairs]
            else:
                sp = -s
                n_pairs = n - sp
                grad[sp:] += (1.0 - h[:n_pairs])
                grad[:n_pairs] -= h[sp:n]
        
        # Average over active set
        if len(active) > 0:
            grad /= len(active)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        grad /= gnorm
        
        # Adaptive step size
        improved = False
        for step in [step_scale * 0.25, step_scale * 0.5, step_scale, step_scale * 1.5, step_scale * 2]:
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-10:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                step_scale = min(step_scale * 1.1, 0.5)
                improved = True
                break
            elif c5_trial > c5_best + 0.01:
                step_scale = max(step_scale * 0.7, 0.01)
        
        if not improved:
            step_scale *= 0.9
            if step_scale < 0.005:
                break
    
    return h_best, c5_best

def coordinate_search(h, n, budget_remain, rng, max_sweeps=20):
    """Local coordinate search with random perturbations"""
    dx = 2.0 / n
    c5_best = compute_c5(h)
    h_best = h.copy()
    
    for sweep in range(max_sweeps):
        if budget_remain() < 3.0:
            break
        any_imp = False
        order = rng.permutation(n)
        
        for i in order:
            if budget_remain() < 1.5:
                break
            old = h[i]
            
            # Try a few perturbation sizes
            for delta in [0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.0, 1.0]:
                if abs(delta) < 1e-9 and abs(1.0 - old) < 1e-9:
                    continue
                if abs(delta) >= 1.0:  # flip to 0 or 1
                    new_val = 1.0 if old < 0.5 else 0.0
                else:
                    new_val = np.clip(old + delta, 0.0, 1.0)
                if abs(new_val - old) < 1e-8:
                    continue
                
                trial = h_best.copy()
                trial[i] = new_val
                trial = project_feasible(trial, n)
                c5_test = compute_c5(trial)
                if c5_test < c5_best - 1e-11:
                    c5_best = c5_test
                    h_best = trial.copy()
                    any_imp = True
                    h = trial.copy()
                    break  # restart with new h
        
        if not any_imp and sweep > 2:
            break
    
    return h_best, c5_best
