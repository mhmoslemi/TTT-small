                        i = ii[pos]
                        j = i + k
                        # Try several small adjustments
                        for delta in [0.002, 0.001, 0.004, 0.0005, 0.003]:
                            # Move mass from larger to smaller
                            if h[i] < h[j]:
                                di, dj = delta, -delta
                            else:
                                di, dj = -delta, delta
                            if h[i] + di < 0 or h[i] + di > 1:
                                continue
                            if h[j] + dj < 0 or h[j] + dj > 1:
                                continue
                            h[i] += di
                            h[j] += dj
                            c5t = compute_c5(h)
                            if c5t < best_c5 - 1e-11:
                                best_c5 = c5t
                                best_h = h.copy()
                                improved = True
                                break
                            h[i] -= di
                            h[j] -= dj
                        if improved:
                            break
                else:
                    k2 = -k
                    jj = np.arange(k2, n)
                    diff_h = h[jj - k2] - h[jj]
                    for pos in np.argsort(np.abs(diff_h))[::-1][:30]:
                        j = jj[pos]
                        i = j - k2
                        for delta in [0.002, 0.001, 0.004, 0.0005, 0.003]:
                            if h[j] < h[i]:
                                dj, di = delta, -delta
                            else:
                                dj, di = -delta, delta
                            if h[j] + dj < 0 or h[j] + dj > 1:
                                continue
                            if h[i] + di < 0 or h[i] + di > 1:
                                continue
                            h[j] += dj
                            h[i] += di
                            c5t = compute_c5(h)
                            if c5t < best_c5 - 1e-11:
                                best_c5 = c5t
                                best_h = h.copy()
                                improved = True
                                break
                            h[j] -= dj
                            h[i] -= di
                        if improved:
                            break

            if improved:
                no_improve = 0
            else:
                no_improve += 1
            if no_improve >= 30:
                break
            if not improved and sweep % 50 == 49:
                h = best_h.copy()

        print("Phase 3 done: C5=%.7f" % best_c5)

    # ===== Phase 4: L-BFGS-B restart with perturbation =====
    if tl() > 30:
        n = best_n
        target = n / 2.0
        for perturb_round in range(5):
            if tl() < 15:
                break
            h0 = best_h + rng.randn(n) * 0.005
            h0 = project_feasible(h0, target)
            for beta in [5000, 10000, 20000]:
                if tl() < 8:
                    break
                pw = 2000.0

                def obj_pert(x):
                    hc = np.clip(x, 0.0, 1.0)
                    corr = np.correlate(hc, 1.0 - hc, mode='full')
                    ii = np.argsort(corr)[-50:]
                    cc = corr[ii]
                    m = cc.max()
                    sh = cc - m
                    sm = (m + np.log(np.sum(np.exp(beta * sh))) / beta)
                    se = hc.sum() - target
                    return sm * (2.0 / n) + pw * se * se

                try:
                    res = minimize(obj_pert, h0, method='L-BFGS-B',
                                   bounds=[(0.0, 1.0)] * n,
                                   options={'maxiter': 400, 'ftol': 1e-20,
                                            'gtol': 1e-18, 'maxls': 50,
                                            'maxfun': 4000})
                    hn = project_feasible(res.x, target)
                    c5n = compute_c5(hn)
                    if c5n < best_c5 - 1e-12:
                        best_c5 = c5n
                        best_h = hn.copy()
                        best_n = n
                        print("  Perturb round %d beta=%d: C5=%.7f" % (perturb_round, beta, c5n))
                        h0 = hn.copy()
                except Exception:
                    pass

        print("Phase 4 done: C5=%.7f" % best_c5)

    # ===== Phase 5: Ultra-fine random perturbations =====
    if tl() > 5:
        n = best_n
        target = n / 2.0
        h = best_h.copy()
        accept_count = 0

        for it in range(10000):
            if tl() < 1:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            scale = 0.001 if accept_count > 100 else 0.003
            d = rng.uniform(0.0002, scale)
            if h[i] < h[j]:
                dd = d
            else:
                dd = -d
            if h[i] + dd < 1e-9 or h[i] + dd > 1.0 - 1e-9:
                continue
            if h[j] - dd < 1e-9 or h[j] - dd > 1.0 - 1e-9:
                continue
            h[i] += dd
            h[j] -= dd
            c5t = compute_c5(h)
            if c5t < best_c5 - 1e-12:
                best_c5 = c5t
                best_h = h.copy()
                accept_count += 1
            else:
                h[i] -= dd
                h[j] += dd

        print("Phase 5 done: C5=%.7f (accepted %d)" % (best_c5, accept_count))

    # ===== Final projection and verification =====
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("FINAL: n=%d, C5=%.7f" % (best_n, final_c5))
    return (best_h, final_c5, best_n)
