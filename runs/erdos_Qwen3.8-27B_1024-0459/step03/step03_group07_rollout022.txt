import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def topk_gradient(corr, h, k, n):
    """Compute gradient of sum of top-k correlation values w.r.t. h."""
    topk_idx = np.argsort(corr)[-k:]
    grad = np.zeros(n)
    for idx in topk_idx:
        lag = idx - (n - 1)
        if lag >= 0:
            # c(lag) = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
            for j in range(n):
                g = 0.0
                # h[j] as h[i+lag]: i = j-lag, need 0<=j-lag<=n-1-lag → lag<=j<=n-1
                if lag <= j <= n - 1:
                    g += (1.0 - h[j - lag])
                # h[j] as h[i]: i = j, need 0<=j<=n-1-lag
                if 0 <= j <= n - 1 - lag:
                    g += (-h[j + lag])
                grad[j] += g
        else:
            s = -lag
            # c(-s) = sum_{i=s}^{n-1} h[i]*(1-h[i-s])
            for j in range(n):
                g = 0.0
                # h[j] as h[i]: i=j, need s<=j<=n-1
                if s <= j <= n - 1:
                    g += (1.0 - h[j - s])
                # h[j] as h[i-s]: i=j+s, need 0<=j<=n-1-s
                if 0 <= j <= n - 1 - s:
                    g += (-h[j + s])
                grad[j] += g
    return grad / k
```

def topk_gradient_vec(corr, h, k, n):
    topk_idx = np.argsort(corr)[-k:]
    grad = np.zeros(n)
    for idx in topk_idx:
        lag = idx - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += (1.0 - h[:n-lag])
                grad[:n-lag] += (-h[lag:])
        else:
            s = -lag
            if s < n:
                grad[s:] += (1.0 - h[:n-s])
                grad[:n-s] += (-h[s:])
    return grad / k
```

This is O(k*n) per call, fully vectorized. For k=10, n=240: about 2400 ops. Very fast.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)

            h[mask] += diff / cnt

            mask = h > 0.0

def grad_single_lag(h, lag, n):
    """Vectorized gradient of correlation at a single lag."""
    grad = np.zeros(n)

        if lag < n:
            grad[lag:] += (1.0 - h[:n-lag])
            grad[:n-lag] += (-h[lag:])
    else:

        if s < n:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] += (-h[s:])
    return grad

def grad_topk(h, k, n):
    """Gradient of mean of top-k correlation values."""
    corr = compute_corr(h)
    topk_idx = np.argsort(corr)[-k:]
    grad = np.zeros(n)

        lag = idx - (n-1)
        grad += grad_single_lag(h, lag, n)
