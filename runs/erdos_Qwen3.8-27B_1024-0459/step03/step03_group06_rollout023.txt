    n = len(h)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def fast_grad_smooth_max(h, beta):
    """Fast vectorized gradient of smooth max of correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    wsum = weights.sum()
    if wsum < 1e-300:
        return np.zeros(n), m * dx
    weights /= wsum
    
    g = np.zeros(n)
    # Vectorized over lags
    for idx in range(2 * n - 1):
        wj = weights[idx]
        if wj < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[k:] += wj * (1.0 - h[:n-k])
            g[:n-k] -= wj * h[k:]
        else:
            kp = -k
            g[:n-kp] += wj * (1.0 - h[kp:])
            g[kp:] -= wj * h[:n-kp]
    g *= dx
    return g, m * dx

def project_feasible(h, n):

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        order = np.argsort(np.abs(h - 0.5))
        remaining = diff
        for i in order:
            if abs(remaining) < 1e-14:
                break
            if remaining > 0:
                add = min(remaining, 1.0 - h[i])
                h[i] += add
                remaining -= add
            else:
                sub = min(-remaining, h[i])
                h[i] -= sub
                remaining += sub
    return h


def interp_to(src, dst_n):
    n_src = len(src)
    if n_src == dst_n:
        return src.copy()

    x_new = (np.arange(dst_n) + 0.5) / dst_n
    return np.interp(x_new, xp, src)


def _obj_val(x, n, beta, tgt):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    np.clip(shifted, -500, 500, out=shifted)
    sm = m + np.log(np.sum(np.exp(shifted))) / beta
    val = sm * dx
    val += 2000.0 * (np.sum(x) - tgt) ** 2
    return val


def _obj_grad(x, n, beta, tgt):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    np.clip(shifted, -500, 500, out=shifted)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.zeros(2 * n - 1)
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= wsum

    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wj = weights[idx]
        if wj < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[k:] += wj * (1.0 - x[:n - k])
            g[:n - k] -= wj * x[k:]
        else:
            kp = -k
            g[:n - kp] += wj * (1.0 - x[kp:])
            g[kp:] -= wj * x[:n - kp]
    g *= dx
    g += 4000.0 * (np.sum(x) - tgt)
    return g


def _fun_wrapper(x, n, beta, tgt):
    return _obj_val(x, n, beta, tgt)


def _jac_wrapper(x, n, beta, tgt):
    return _obj_grad(x, n, beta, tgt)


def binary_sa(h_init, n, n_iters, rng):
    h = np.where(h_init > 0.5, 1.0, 0.0).copy()
    target_ones = n // 2
    num_ones = int(h.sum())
    if num_ones > target_ones:
        ones = np.where(h == 1)[0]
        idx_rm = rng.choice(len(ones), num_ones - target_ones, replace=False)
        h[ones[idx_rm]] = 0.0
    elif num_ones < target_ones:
        zeros = np.where(h == 0)[0]
        idx_add = rng.choice(len(zeros), target_ones - num_ones, replace=False)
        h[zeros[idx_add]] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5

    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])

    T_end = 5e-6

        i = rng.randint(len(ones_list))
        j = rng.randint(len(zeros_list))
        oi = ones_list[i]
        oz = zeros_list[j]

        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = compute_c5(h)

        T = T_start * (T_end / T_start) ** (float(it) / max(float(n_iters), 1.0))

        if new_c5 < cur_c5 or (T > 1e-16 and rng.rand() < np.exp(-(new_c5 - cur_c5) / T)):
            cur_c5 = new_c5
            ones_list[i] = oz
            zeros_list[j] = oi
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0

    return best_h, best_c5
