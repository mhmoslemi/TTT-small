import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_binary(h, n, rng):
    """Ensure h is binary with exactly n//2 ones."""
    h = (h > 0.5).astype(float).copy()
    target = n // 2
    n_ones = int(h.sum())
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[rng.choice(idx, n_ones - target, replace=False)] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[rng.choice(idx, target - n_ones, replace=False)] = 1
    return h

def guided_local_search(h, n, time_limit, rng):
    """Guided local search: target the worst shifts."""
    h = h.astype(float).copy()
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n
    
    start = time.time()
    stall = 0
    
    while time.time() - start < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Find worst shift
        k_max = int(np.argmax(corr))
        s = k_max - (n - 1)
        
        # Find contributing pairs
        if s > 0:
            # Pairs (i, i+s) with h[i]=1, h[i+s]=0
            contrib_1 = [i for i in ones if i + s < n and h[i+s] == 0]
            contrib_0 = [i for i in zeros if i >= s and h[i-s] == 1]
        elif s < 0:
            t = -s
            contrib_1 = [i for i in ones if i - t >= 0 and h[i-t] == 0]
            contrib_0 = [i for i in zeros if i + t < n and h[i+t] == 1]
        else:
            contrib_1 = list(ones)
            contrib_0 = list(zeros)
        
        # Try swapping a contributing 1 with a non-contributing 0 (or vice versa)
        found = False
        n_try = min(200, len(ones) * len(zeros))
        
        for _ in range(n_try):
            if contrib_1 and rng.rand() < 0.6:
                a = contrib_1[rng.randint(len(contrib_1))]
            else:
                a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            if a == b:
                continue
            
            h[a] = 0.0
            h[b] = 1.0
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            
            if new_max < cur_max - 1e-12:
                cur_max = new_max
                corr = new_corr
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
                found = True
                break
            else:
                h[a] = 1.0
                h[b] = 0.0
        
        if not found:
            stall += 1
            if stall > 50:
                # Perturbation: accept a slightly worse solution
                a = ones[rng.randint(len(ones))]
                b = zeros[rng.randint(len(zeros))]
                h[a] = 0.0
                h[b] = 1.0
                corr = compute_corr(h)
                cur_max = float(np.max(corr))
                stall = 0
            if stall > 200:
                break
    
    return best_h, best_max * dx
