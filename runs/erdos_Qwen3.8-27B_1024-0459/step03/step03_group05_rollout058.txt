def exact_project(h, target_sum):
    h = np.asarray(h, dtype=float)
    if abs(np.sum(h) - target_sum) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_val(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k is not None and top_k < len(corr):
        c = np.sort(corr)[-top_k:]
    else:
        c = corr
    m = c.max()
    return (m + np.log(np.sum(np.exp(beta * (c - m)))) / beta) * dx

def smooth_max_grad(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k is not None and top_k < len(corr):

        if a_s <= a_e:

        if lag >= 0:
            b_s, b_e = lag, n - 1
        else:

        if b_s <= b_e:
