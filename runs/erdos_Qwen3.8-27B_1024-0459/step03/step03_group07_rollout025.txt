def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final fixup
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx[:max(1, int(np.ceil(abs(diff))))]] += diff / max(1, int(np.ceil(abs(diff))))
            h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_h(h_src, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_new = np.interp(xn, xp, h_src)
    return project_feasible(h_new, n_dst)

def fast_coord_descent(h_init, n, sweeps=20, time_limit=None, t0=None):
    h = project_feasible(h_init, n).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_c5 = np.max(corr) * dx
    best = h.copy()
    
    n_grid = 150
    n_refine = 45
    
    for sw in range(sweeps):
        if time_limit and t0 and time.time() - t0 > time_limit:
            break
        
        for j in range(n):
            old_h = h[j]
            
            # Determine affected lag range
            lo_k = max(-(n - 1), min(-j, j - (n - 1)))
            hi_k = min(n - 1, max(n - 1 - j, j))
            
            lags = np.arange(lo_k, hi_k + 1)
            fidx = lags + (n - 1)
            curr = corr[fidx]
            L = len(lags)
            
            # Validity masks
            v1 = (j + lags >= 0) & (j + lags < n)
            v2 = (j - lags >= 0) & (j - lags < n)
            is_zero = (lags == 0)
            nonzero = ~is_zero
            
            # For non-zero lags: c[lag] = rest + a*x
            # old_contrib = old_h*(1-h[j+lag])*v1 + h[j-lag]*(1-old_h)*v2
            # a = (1-h[j+lag])*v1 - h[j-lag]*v2
            a_arr = np.zeros(L)
            old_c = np.zeros(L)
            if np.any(v1 & nonzero):
                mask = v1 & nonzero
                a_arr[mask] += (1.0 - h[j + lags[mask]])
                old_c[mask] += old_h * (1.0 - h[j + lags[mask]])
            if np.any(v2 & nonzero):
                mask = v2 & nonzero
                a_arr[mask] -= h[j - lags[mask]]
                old_c[mask] += h[j - lags[mask]] * (1.0 - old_h)
            
            rest_nz = curr[nonzero] - old_c[nonzero]
            a_nz = a_arr[nonzero]
            
            # For lag=0: c[0] = rest + x - x^2
            rest_z = np.array([])
            if np.any(is_zero):
                rest_z = curr[is_zero] - old_h * (1.0 - old_h)
            
            # Max of unaffected lags
            aff_set = set(fidx.tolist())
            # Efficient: max of all minus affected
            M_unaff = np.max(np.where(np.isin(np.arange(2*n-1), fidx), -np.inf, corr))
            if M_unaff == -np.inf:
                M_unaff = 0.0
            
            # 1D minimization: f(x) = max(M, max(rest_nz + a_nz*x), rest_z + x - x^2)
            xs = np.linspace(0.0, 1.0, n_grid)
            
            # Vectorized evaluation
            if len(a_nz) > 0:
                lin_vals = rest_nz[:, None] + a_nz[:, None] * xs[None, :]
                lin_max = np.max(lin_vals, axis=0)
            else:
                lin_max = np.zeros(n_grid)
            
            if len(rest_z) > 0:
                quad_vals = rest_z[0] + xs - xs * xs
            else:
                quad_vals = np.zeros(n_grid)
            
            obj = np.maximum(np.maximum(M_unaff, lin_max), quad_vals)
            best_idx = np.argmin(obj)
            best_x = xs[best_idx]
            
            # Refine
            lo = max(0.0, best_x - 2.0 / n_grid)
            hi = min(1.0, best_x + 2.0 / n_grid)
            for _ in range(n_refine):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                v1 = M_unaff
                if len(a_nz) > 0:
                    v1 = max(v1, np.max(rest_nz + a_nz * m1))
                if len(rest_z) > 0:
                    v1 = max(v1, rest_z[0] + m1 - m1 * m1)
                v2 = M_unaff
                if len(a_nz) > 0:
                    v2 = max(v2, np.max(rest_nz + a_nz * m2))
                if len(rest_z) > 0:
                    v2 = max(v2, rest_z[0] + m2 - m2 * m2)
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            best_x = (lo + hi) / 2.0
            
            # Update h and corr
            h[j] = best_x
            for idx_i, fi in enumerate(fidx):
                if is_zero[idx_i]:
                    corr[fi] = rest_z[0] + best_x - best_x * best_x
                else:
                    corr[fi] = rest_nz[idx_i - (1 if idx_i > np.where(is_zero, [0], [0])[0] and np.sum(is_zero[:idx_i+1]) > 0 else 0)] + a_nz[idx_i - (1 if np.sum(is_zero[:idx_i+1]) > 0 else 0)] * best_x
        
        # Recompute corr at end of sweep (safety)
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    
    return best

def fast_coord_descent_v2(h_init, n, sweeps=20, time_limit=None, t0=None):
    """Cleaner vectorized CD avoiding the indexing bug."""
    h = project_feasible(h_init, n).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_c5 = np.max(corr) * dx
    best = h.copy()
    
    n_grid = 150
    n_refine = 45
    xs_grid = np.linspace(0.0, 1.0, n_grid)
    
    for sw in range(sweeps):
        if time_limit and t0 and time.time() - t0 > time_limit:
            break
        
        for j in range(n):
            old_h = h[j]
            
            # Affected lags
            lo_k = max(-(n-1), min(-j, j-(n-1)))
            hi_k = min(n-1, max(n-1-j, j))
            lags = np.arange(lo_k, hi_k+1)
            fidx = lags + (n-1)
            curr_corr = corr[fidx].copy()
            L = len(lags)
            
            # Masks
            v1 = (j + lags >= 0) & (j + lags < n)
            v2 = (j - lags >= 0) & (j - lags < n)
            iz = (lags == 0)
            
            if np.sum(iz) == 1:
                # Build per-lag rest and a (only for non-zero lags)
                nz_idx = np.where(~iz)[0]
                z_idx = np.where(iz)[0][0]
                
                nz_lags = lags[nz_idx]
                nz_fidx = fidx[nz_idx]
                nz_v1 = v1[nz_idx]
                nz_v2 = v2[nz_idx]
                
                a_nz = np.zeros(len(nz_idx))
                oc_nz = np.zeros(len(nz_idx))
                if np.any(nz_v1):
                    a_nz[nz_v1] += (1.0 - h[j + nz_lags[nz_v1]])
                    oc_nz[nz_v1] += old_h * (1.0 - h[j + nz_lags[nz_v1]])
                if np.any(nz_v2):
                    a_nz[nz_v2] -= h[j - nz_lags[nz_v2]]
                    oc_nz[nz_v2] += h[j - nz_lags[nz_v2]] * (1.0 - old_h)
                
                rest_nz = curr_corr[nz_idx] - oc_nz
                rest_z = curr_corr[z_idx] - old_h * (1.0 - old_h)
            else:
                # No zero lag affected (shouldn't happen for valid j, but safety)
                a_nz = np.zeros(L)
                oc_nz = np.zeros(L)
                if np.any(v1):
                    a_nz[v1] += (1.0 - h[j + lags[v1]])
                    oc_nz[v1] += old_h * (1.0 - h[j + lags[v1]])
                if np.any(v2):
                    a_nz[v2] -= h[j - lags[v2]]
                    oc_nz[v2] += h[j - lags[v2]] * (1.0 - old_h)
                rest_nz = curr_corr - oc_nz
                rest_z = None
            
            # M_unaff
            all_idx = np.arange(2*n-1)
            aff_mask = np.zeros(2*n-1, dtype=bool)
            aff_mask[fidx] = True
            M_unaff = np.max(np.where(aff_mask, -1e9, corr))
            if M_unaff < 0:
                M_unaff = 0.0
            
            # Evaluate on grid
            if len(a_nz) > 0:
                lin_m = np.max(rest_nz[:, None] + a_nz[:, None] * xs_grid[None, :], axis=0)
            else:
                lin_m = np.full(n_grid, -1e9)
            if rest_z is not None:
                quad_m = rest_z + xs_grid - xs_grid*xs_grid
            else:
                quad_m = np.full(n_grid, -1e9)
            obj = np.maximum(np.maximum(M_unaff, lin_m), quad_m)
            
            bi = np.argmin(obj)
            bx = xs_grid[bi]
            
            # Refine
            lo_b = max(0.0, bx - 3.0/n_grid)
            hi_b = min(1.0, bx + 3.0/n_grid)
            for _ in range(n_refine):
                m1 = lo_b + (hi_b - lo_b)/3.0
                m2 = hi_b - (hi_b - lo_b)/3.0
                v1 = M_unaff
                if len(a_nz) > 0:
                    v1 = max(v1, float(np.max(rest_nz + a_nz * m1)))
                if rest_z is not None:
                    v1 = max(v1, rest_z + m1 - m1*m1)
                v2 = M_unaff
                if len(a_nz) > 0:
                    v2 = max(v2, float(np.max(rest_nz + a_nz * m2)))
                if rest_z is not None:
                    v2 = max(v2, rest_z + m2 - m2*m2)
                if v1 < v2:
                    hi_b = m2
                else:
                    lo_b = m1
            bx = (lo_b + hi_b) / 2.0
            
            # Update
            h[j] = bx
            for ii in range(L):
                if iz[ii]:
                    corr[fidx[ii]] = rest_z + bx - bx*bx
                else:
                    corr[fidx[ii]] = rest_nz[ii] + a_nz[ii] * bx
        
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    
    return best

def topk_subgradient(h_init, n, time_limit=None, t0=None):
    h = project_feasible(h_init, n).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best = h.copy()
    
    n_lags = 2 * n - 1
    
    for K in [5, 10, 20, 40]:
        lr = 0.005
        stuck = 0
        max_iter = 800
        for it in range(max_iter):
            if time_limit and t0 and time.time() - t0 > time_limit:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best = h.copy()
                stuck = 0
            else:
                stuck += 1
            
            # Top-K gradient
            top_k = np.argsort(corr)[-K:]
            grad = np.zeros(n)
            for fi in top_k:
                lag = fi - (n - 1)
                if lag == 0:
                    grad += (1.0 - 2.0 * h) / K
                elif lag > 0:
                    jj1 = np.arange(0, n - lag)
                    grad[jj1] += (1.0 - h[jj1 + lag]) / K
                    jj2 = np.arange(lag, n)
                    grad[jj2] -= h[jj2 - lag] / K
                else:
                    s = -lag
                    jj1 = np.arange(s, n)
                    grad[jj1] += (1.0 - h[jj1 - s]) / K
                    jj2 = np.arange(0, n - s)
                    grad[jj2] -= h[jj2 + s] / K
            
            h_new = h - lr * grad
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best = h_new.copy()
                h = h_new
            else:
                lr *= 0.95
                if stuck > 50:
                    h = project_feasible(h + 0.03 * np.random.randn(n), n)
                    lr = 0.005
                    stuck = 0
    
    return best

def slsqp_polish(h_init, n, time_limit=None, t0=None):
    h = project_feasible(h_init, n).copy()
    best_c5 = compute_c5(h)
    best = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    def eqfun(x):
        return np.sum(x) - n / 2.0
    
    cons = [{'type': 'eq', 'fun': eqfun, 'jac': lambda x: np.ones(n)}]
    
    for beta in [100.0, 500.0, 2000.0, 10000.0]:
        if time_limit and t0 and time.time() - t0 > time_limit - 3:
            break
        
        def obj_smooth(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(corr)
            sh = np.clip(beta * (corr - m), -500.0, 0.0)
            return (m + np.log(np.sum(np.exp(sh))) / beta) * (2.0 / n)
        
        try:
            res = minimize(obj_smooth, h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 200, 'ftol': 1e-12, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    # Top-K smooth objective
    for k in [5, 10]:
        if time_limit and t0 and time.time() - t0 > time_limit - 2:
            break
        def obj_topk(x, kk=k):
            corr = np.correlate(x, 1.0 - x, mode='full')
            idx = np.argsort(corr)[-kk:]
            return np.mean(corr[idx]) * (2.0 / n)
        try:
            res = minimize(obj_topk, h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 150, 'ftol': 1e-12, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return max(0.0, budget_s - (time.time() - t0) - 5.0)
    
    # Get parent
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn > 10:
            best_h = ph.copy()
            best_c5 = compute_c5(ph)
            best_n = pn
    except Exception:
        pass
    
    if best_h is None:
        best_n = 160
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)
    
    # Try multiple resolutions
    resolutions = [best_n, 180, 200, 220, 240, 260]
    resolutions = sorted(set([r for r in resolutions if 100 <= r <= 300]))
    
    # Phase 1: Coordinate descent at each resolution
    for n in resolutions:
        if time_left() < 80:
            break
        
        # Interpolate to this resolution
        if n == best_n:
            h_init = project_feasible(best_h, n)
        else:
            h_init = interpolate_h(best_h, best_n, n)
        
        # Run CD
        h_cd = fast_coord_descent_v2(h_init, n, sweeps=12, time_limit=time_left() * 0.4, t0=t0)
        c5_cd = compute_c5(h_cd)
        
        # More CD sweeps
        h_cd2 = fast_coord_descent_v2(h_cd, n, sweeps=15, time_limit=time_left() * 0.5, t0=t0)
        c5_cd2 = compute_c5(h_cd2)
        
        # Subgradient
        if time_left() > 60:
            h_sg = topk_subgradient(h_cd2, n, time_limit=time_left() * 0.7, t0=t0)
            c5_sg = compute_c5(h_sg)
            # CD after subgradient
            h_cd3 = fast_coord_descent_v2(h_sg, n, sweeps=10, time_limit=time_left() * 0.8, t0=t0)
            c5_cd3 = compute_c5(h_cd3)
        else:
            h_cd3 = h_cd2
            c5_cd3 = c5_cd2
        
        cands = [(h_cd, c5_cd), (h_cd2, c5_cd2), (h_cd3, c5_cd3)]
        cb = min(cands, key=lambda x: x[1])
        if cb[1] < best_c5:
            best_c5 = cb[1]
            best_h = cb[0].copy()
            best_n = n
    
    # Phase 2: SLSQP polish on best
    if time_left() > 40:
        h_sp = slsqp_polish(best_h, best_n, time_limit=time_left() * 0.5, t0=t0)
        c5_sp = compute_c5(h_sp)
        if c5_sp < best_c5:
            best_c5 = c5_sp
            best_h = h_sp.copy()
    
    # Phase 3: Final CD polish
    if time_left() > 30:
        h_fin = fast_coord_descent_v2(best_h, best_n, sweeps=25, time_limit=time_left() * 0.6, t0=t0)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
    
    # Phase 4: Another round with perturbation
    if time_left() > 50:
        # Try a few perturbations of the best
        for trial in range(3):
            if time_left() < 20:
                break
            h_pert = project_feasible(best_h + 0.02 * rng.randn(best_n), best_n)
            h_tr = fast_coord_descent_v2(h_pert, best_n, sweeps=10, time_limit=min(40.0, time_left()*0.5), t0=t0)
            c5_tr = compute_c5(h_tr)
            if c5_tr < best_c5:
                best_c5 = c5_tr
                best_h = h_tr.copy()
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    # Try evaluator
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
