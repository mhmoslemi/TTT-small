def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr


def project_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_h(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xq = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xq, xp, src)

def lse_func(vals, beta):
    m = np.max(vals)
    if beta < 1e-6:
        return m
    exps = np.exp(beta * (vals - m))
    return m + np.log(np.sum(exps)) / beta

    sweep = 0
    while improved and sweep < n_sweeps:

                grad[offset:offset + np_] += (1.0 - h[:np_])

                           options={'maxiter': 2000, 'ftol': 1e-19, 'gtol': 1e-15, 'maxls': 100})
            h_new = project_h(res.x, n)
            c5_new, _ = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
                print("  LBFGS beta={}: C5={:.8f}".format(beta, best_c5))
        except Exception:
            pass

        if time_left(t0, budget) > 12:
            act_s = active
            beta_s = beta
            n_s = n

            def obj_slsqp_inner(x):
                return obj_lse_slsqp(x, n_s, beta_s, act_s)

            cons = [{'type': 'eq', 'fun': lambda x: sum_constraint(x, n)}]
            try:
                res = minimize(obj_slsqp_inner, h, method='SLSQP', bounds=bounds,
                               constraints=cons,
                               options={'maxiter': 500, 'ftol': 1e-18, 'iprint': 0})
                h_new = project_h(res.x, n)
                c5_new, _ = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                    print("  SLSQP beta={}: C5={:.8f}".format(beta, best_c5))
            except Exception:
                pass

    # Phase 3: Coordinate descent on LSE (exact 1D per coordinate)
    if time_left(t0, budget) > 20:
        for beta_cd in [50, 100, 200, 400, 800]:
            if time_left(t0, budget) < 5:
                break
            _, corr = compute_c5(h)
            active = find_active_set(corr, gap=0.3, top_max=50)
            h = coord_descent_lse(h, n, beta_cd, active, n_sweeps=4, seed=seed)
            c5_new, _ = compute_c5(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                print("  CD beta={}: C5={:.8f}".format(beta_cd, best_c5))

    # Phase 4: Simulated annealing
    if time_left(t0, budget) > 30:
        h_sa, c5_sa = simulated_annealing(best_h.copy(), n, t0, budget,
                                          max_iter=40000, seed=seed + 1)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            h = h_sa.copy()
            n = n
            print("  SA: C5={:.8f}".format(best_c5))

    # Phase 5: Nelder-Mead on free coordinates
    if time_left(t0, budget) > 15:
        h_nm = nelder_mead_free(best_h.copy(), n, max_iter=400)
        c5_nm, _ = compute_c5(h_nm)
        if c5_nm < best_c5:
            best_c5 = c5_nm
            best_h = h_nm.copy()
            h = h_nm.copy()
            print("  NM: C5={:.8f}".format(best_c5))

    # Phase 6: Subgradient with top-4 lags
    if time_left(t0, budget) > 20:
        h_sg, c5_sg = subgradient_step(best_h.copy(), n, t0, budget,
                                       n_iter=1000, seed=seed + 2)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()
            print("  Subgrad: C5={:.8f}".format(best_c5))

    # Phase 7: High-beta full LSE polish
    if time_left(t0, budget) > 10:
        for beta_f in [1000, 2000, 5000]:
            if time_left(t0, budget) < 5:
                break
            n_f = n
            beta_bf = beta_f

            def obj_full_inner(x):
                return obj_full_lse_pen(x, n_f, beta_bf)

            try:
                res = minimize(obj_full_inner, h, method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 3000, 'ftol': 1e-19, 'gtol': 1e-16})
                h_new = project_h(res.x, n)
                c5_new, _ = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                    print("  FullLBFGS beta={}: C5={:.8f}".format(beta_f, best_c5))
            except Exception:
                pass

    # Phase 8: Random pair perturbation (fine)
    if time_left(t0, budget) > 10:
        n = best_n
        h = best_h.copy()
        for it in range(80000):
            if time_left(t0, budget) < 2:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(0.001, 0.015)
            h_tr = h.copy()
            h_tr[i] -= d
            h_tr[j] += d
            h_tr = project_h(h_tr, n)
            c5_tr, _ = compute_c5(h_tr)
            if c5_tr < best_c5:
                best_c5 = c5_tr
                best_h = h_tr.copy()
                h = h_tr.copy()

    # Phase 9: Re-check at nearby resolutions
    if time_left(t0, budget) > 10:
        for dn in range(-30, 31, 5):
            if time_left(t0, budget) < 3:
                break
            if dn == 0:
                continue
            nt = best_n + dn
            if nt < 80 or nt > 600:
                continue
            ht = interp_h(best_h, best_n, nt)
            ht = project_h(ht, nt)
            ct, _ = compute_c5(ht)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()
                best_n = nt
                print("  Res n={}: C5={:.8f}".format(nt, ct))

    # Phase 10: Final coordinate descent + SLSQP at best resolution
    if time_left(t0, budget) > 10:
        n = best_n
        h = best_h.copy()
        bounds_final = [(0.0, 1.0)] * n
        _, corr = compute_c5(h)
        active = find_active_set(corr, gap=0.2, top_max=30)
        h = coord_descent_lse(h, n, 500, active, n_sweeps=3, seed=seed + 10)
        c5_cd, _ = compute_c5(h)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h.copy()

        if time_left(t0, budget) > 8:
            act_fin = active
            beta_fin = 2000
            n_fin = n

            def obj_fin(x):
                return obj_lse_slsqp(x, n_fin, beta_fin, act_fin)

            cons_fin = [{'type': 'eq', 'fun': lambda x: sum_constraint(x, n_fin)}]
            try:
                res = minimize(obj_fin, h, method='SLSQP', bounds=bounds_final,
                               constraints=cons_fin,
                               options={'maxiter': 500, 'ftol': 1e-18, 'iprint': 0})
                h_new = project_h(res.x, n)
                c5_new, _ = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    # Final projection and evaluation
    best_h = project_h(best_h, best_n)
    final_c5, _ = compute_c5(best_h)
    final_c5 = min(final_c5, best_c5)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
