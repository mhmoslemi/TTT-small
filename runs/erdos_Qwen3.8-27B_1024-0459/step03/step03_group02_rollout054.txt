def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        elif n > 0:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        # Distribute among all elements proportionally
        adj = np.full(n, 1.0)
        adj[h >= 1.0] = 0.0
        adj[h <= 0.0] = 0.0
        total_w = np.sum(adj)
        if total_w > 0:
            h += diff * adj / total_w
            h = np.clip(h, 0.0, 1.0)
        # Final exact fix
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-12:
            idx = int(np.argmax(np.abs(h - 0.5)))
            h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
        diff3 = target - np.sum(h)
        if abs(diff3) > 1e-12:
            for j in range(n):
                    if 0 < h[j] < 1:
                        h[j] += diff3
                        break
            h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def lag_gradient(h, n, offset):
    """Gradient of corr[offset + (n-1)] with respect to h."""
    grad = np.zeros(n)
    if offset >= 0:
        n_pairs = n - offset
        if n_pairs <= 0:
            return grad
        # corr = sum_{i=0}^{n_pairs-1} h[i+offset] * (1-h[i])
        # d/d h[i] = -h[i+offset] for i in [0, n_pairs)
        # d/d h[i+offset] = (1-h[i]) for i in [0, n_pairs)
        grad[offset:offset + n_pairs] += (1.0 - h[:n_pairs])
        grad[:n_pairs] -= h[offset:offset + n_pairs]
    else:
        abs_off = -offset
        n_pairs = n - abs_off
        if n_pairs <= 0:
            return grad
        # corr = sum_{i=abs_off}^{n-1} h[i] * (1-h[i-abs_off])
        # d/d h[i] = (1-h[i-abs_off]) for i in [abs_off, n)
        # d/d h[i-abs_off] = -h[i] for i in [abs_off, n)
        grad[abs_off:] += (1.0 - h[:n_pairs])
        grad[:n_pairs] -= h[abs_off:]
    return grad

def active_set_descent(h, n, max_iter=500, max_active=5, tol=1e-10):
    """Subgradient descent on the active set (top lags)."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        corr = compute_corr(h)
        max_corr = np.max(corr)
        # Find active lags (within 2% of max)
        threshold = max_corr - 0.02 * max_corr
        active = np.where(corr >= threshold)[0]
        if len(active) > max_active:
            active = np.argsort(corr)[::-1][:max_active]
        
        # Compute combined gradient
        grad = np.zeros(n)
        for k_idx in active:
            offset = k_idx - (n - 1)
            grad += lag_gradient(h, n, offset)
        grad /= len(active)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        
        # Exact line search: minimize max over active lags
        best_step = 0.0
        best_val = best_c5
        # Try a range of steps
        for step in [0.005, 0.01, 0.02, 0.05, 0.01, 0.002, 0.03, 0.08, 0.015, 0.04]:
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_val:
                best_val = c5_trial
                best_step = step
        
        if best_step == 0.0:
            # Try smaller steps
            for step in [0.001, 0.0005, 0.002, 0.003]:
                h_trial = h - step * grad
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_val:
                    best_val = c5_trial
                    best_step = step
        
        if best_step > 0:
            h = h - best_step * grad
            h = project_feasible(h, n)
            c5_val = compute_c5(h)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
            if best_c5 - c5_val < tol:
                # Check if we've made very little progress
                if it > 100 and best_c5 - c5_val < 1e-8:
                    pass  # continue anyway
        else:
            if it > 50:
                break
    
    return best_h, best_c5

def slsqp_refine(h, n, beta, maxiter=500):
    """SLSQP with log-sum-exp smoothing."""
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        # Log-sum-exp
        shifted = np.minimum(beta * (corr - m), 0.0)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx
    
    def constraint(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint}]
    
    try:
        res = minimize(obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        result = project_feasible(res.x, n)
        return result
    except Exception:
        return h.copy()

def coordinate_descent(h, n, max_sweeps=20, coord_tol=1e-8):
    """Coordinate descent: try adjusting each h[i] slightly."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            old_val = h[i]
            # Try small perturbations
            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                h_trial = h.copy()
                h_trial[i] = old_val + delta
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-12:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h = h_trial.copy()
                    improved = True
                    break
            if not improved and sweep > 5:
                # Try even smaller
                for delta in [0.001, -0.001, 0.002, -0.002]:
                    h_trial = h.copy()
                    h_trial[i] = old_val + delta
                    h_trial = project_feasible(h_trial, n)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < best_c5 - 1e-12:
                        best_c5 = c5_trial
                        best_h = h_trial.copy()
                        h = h_trial.copy()
                        improved = True
                        break
        if not improved:
            break
    
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def two_lag_equalize(h, n, lag1, lag2, max_iter=100):
    """Try to equalize two specific lags by adjusting h."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        corr = compute_corr(h)
        c1 = corr[lag1]
        c2 = corr[lag2]
        if abs(c1 - c2) < 1e-10:
            break
        
        # Gradient of (c1 - c2)
        g1 = lag_gradient(h, n, lag1 - (n-1))
        g2 = lag_gradient(h, n, lag2 - (n-1))
        grad = (c1 - c2) * (g1 - g2) / (c1 + c2 + 1)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        for step in [0.001, 0.005, 0.01, 0.002]:
            h_trial = h - step * grad / gnorm
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial.copy()
                break
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 160
        h = np.zeros(best_n)
        for i in range(best_n):
            cnt = 0
            v = i
            while v:
                cnt ^= (v & 1)
                v >>= 1
            h[i] = 1.0 if cnt == 1 else 0.0
        best_h = project_feasible(h, best_n)
        best_c5 = compute_c5(best_h)
        best_n = len(best_h)
    
    # Phase 1: Active-set subgradient with exact line search
    if remaining_time() > 100:
        h = best_h.copy()
        n = best_n
        h, c5_val = active_set_descent(h, n, max_iter=2000, max_active=5)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
    
    # Phase 2: SLSQP refinement with various betas
    for beta in [500, 1000, 2000, 5000]:
        if remaining_time() < 15:
            break
        n = best_n
        h_new = slsqp_refine(best_h, n, beta, 300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 3: Coordinate descent
    if remaining_time() > 60:
        h, c5_val = coordinate_descent(best_h, best_n, max_sweeps=30)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
    
    # Phase 4: Multi-resolution exploration
    for trial_n in [150, 170, 180, 200, 220, 240, 260, 280, 320]:
        if remaining_time() < 30:
            break
        # Make even
        if trial_n % 2 != 0:
            trial_n += 1
        h_interp = interp_to(best_h, best_n, trial_n)
        h_interp = project_feasible(h_interp, trial_n)
        c5_interp = compute_c5(h_interp)
        
        if c5_interp < best_c5 + 0.002:
            # Refine at this resolution
            h_ref, c5_ref = active_set_descent(h_interp, trial_n, max_iter=500, max_active=4)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
                best_n = trial_n
            else:
                h_new = slsqp_refine(h_interp, trial_n, 1000, 200)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = trial_n
    
    # Phase 5: Two-lag equalization on the top few lags
    if remaining_time() > 30:
        n = best_n
        h = best_h.copy()
        corr = compute_corr(h)
        sorted_idx = np.argsort(corr)[::-1]
        top_lags = sorted_idx[:min(4, len(sorted_idx))]
        
        for i in range(len(top_lags)):
            for j in range(i+1, len(top_lags)):
                if remaining_time() < 5:
                    break
                h, c5_val = two_lag_equalize(h, n, top_lags[i], top_lags[j], max_iter=50)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h.copy()
    
    # Phase 6: Final SLSQP polish
    if remaining_time() > 10:
        n = best_n
        for beta in [3000, 5000, 10000]:
            if remaining_time() < 5:
                break
            h_new = slsqp_refine(best_h, n, beta, 200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
