                        best_c5 = c5_new
                        best_h = h0.copy()
                        best_n = n
                        print("  LBFGS b={}: C5={:.6f}".format(beta, c5_new))
            
            # Top-k smooth (focus on worst shifts)
            if _remaining() > 10:
                h_new, c5_new = topk_l_bfgs_run(h0, n, 16, 200, tgt, maxiter=200, bounds=bounds)
                if c5_new < c0:
                    h0 = h_new
                    c0 = c5_new
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h0.copy()
                        best_n = n
                        print("  TopK: C5={:.6f}".format(c5_new))
            
            # Coordinate descent polish
            if _remaining() > 10:
                h_cd, c5_cd = coord_descent(h0, n, rng, max_sweeps=8, time_limit=10)
                if c5_cd < c0:
                    h0 = h_cd
                    c0 = c5_cd
                    if c5_cd < best_c5:
                        best_c5 = c5_cd
                        best_h = h0.copy()
                        best_n = n
                        print("  CoordDesc: C5={:.6f}".format(c5_cd))
        
        if _remaining() < 30:
            break
    
    # Binary SA on best resolution
    if _remaining() > 50:
        n = best_n
        print("\n=== Binary SA (n={}) ===".format(n))
        tgt = n / 2.0
        bounds = [(0.0, 1.0)] * n
        
        sa_inits = [project_feasible(best_h, n)]
        for _ in range(3):
            sa_inits.append(make_random_binary(n, rng))
        sa_inits.append(make_thue_morse(n))
        
        max_sa_iters = int(min(50000, _remaining() * 30))
        for si, h_sa in enumerate(sa_inits):
            if _remaining() < 15:
                break
            h_bin, c5_bin = binary_sa(h_sa, n, max_sa_iters, rng, t0=0.004, t1=0.00003)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                best_n = n
                print("  SA[{}]: C5={:.6f}".format(si, c5_bin))
            
            # Polish with continuous optimization
            if _remaining() > 10:
                for beta in [500, 1000]:
                    if _remaining() < 6:
                        break
                    h_new, c5_new = l_bfgs_run(h_bin, n, beta, tgt, maxiter=200, bounds=bounds)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        print("  SA+LBFGS b={}: C5={:.6f}".format(beta, c5_new))
    
    # Perturbation + reoptimize
    if _remaining() > 20:
        n = best_n
        tgt = n / 2.0
        bounds = [(0.0, 1.0)] * n
        print("\n=== Perturb+Reopt (n={}) ===".format(n))
        h_p, c5_p = perturb_reopt(best_h, n, rng, n_trials=4, bounds=bounds)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
            print("  Perturb: C5={:.6f}".format(c5_p))
    
    # Final coordinate descent sweep
    if _remaining() > 10:
        n = best_n
        print("\n=== Final Coord Descent ===")
        h_fd, c5_fd = coord_descent(best_h, n, rng, max_sweeps=10, time_limit=10)
        if c5_fd < best_c5:
            best_c5 = c5_fd
            best_h = h_fd.copy()
    
    # Final high-beta polish
    if _remaining() > 6:
        n = best_n
        tgt = n / 2.0
        bounds = [(0.0, 1.0)] * n
        for beta in [1500, 3000]:
            if _remaining() < 4:
                break
            h_new, c5_new = l_bfgs_run(best_h, n, beta, tgt, maxiter=400, bounds=bounds)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  Final polish b={}: C5={:.6f}".format(beta, c5_new))
    
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
