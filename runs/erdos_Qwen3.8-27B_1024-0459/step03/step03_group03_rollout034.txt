import numpy as np
from scipy.optimize import minimize
import time

_t0 = 0.0
_tb = 1300.0

def _tl():
    return max(0.0, _tb - (time.time() - _t0) - 2.0)

def _c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj(h, n):
    h = np.array(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        d = 0.5 * n - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        k = int(np.sum(m))
        if k > 0:
            h[m] += d / k
        else:
            h[:] = 0.5
        h = np.clip(h, 0.0, 1.0)
    return h

def _grad_lag(h, k, n):
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[s:] += (1.0 - h[:n-s])
            g[:n-s] -= h[s:]
    else:
        p = -s
        if p > 0:
            g[:n-p] += (1.0 - h[p:])
            g[p:] -= h[:n-p]
    return g

def _subgrad_topk(h, n, T, max_iters, alpha_init, rng):
    """Fermi-Walk style subgradient with top-T lags"""
    h = h.copy()
    bc = _c5(h)
    alpha = alpha_init
    for it in range(max_iters):
        if _tl() < 2:
            break
        corr = _corr(h)
        idx = np.argsort(corr)[-T:]
        g = np.zeros(n)
        for k in idx:
            g += _grad_lag(h, int(k), n)
        g /= T
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        # Try multiple step sizes
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*2.0]:
            h2 = _proj(h - a * g, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-12:
                h = h2
                bc = c2
                alpha = min(a * 1.05, 0.1)
                improved = True
                break
        if not improved:
            alpha *= 0.7
            if alpha < 1e-6:
                break

    n = len(h)

    return np.correlate(h, 1.0 - h, mode='full')

def _proj(h, n):
    h = np.array(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)

        d = 0.5 * n - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        k = int(np.sum(m))

            h[m] += d / k

def _grad_lag(h, k, n):
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[s:] += (1.0 - h[:n-s])
            g[:n-s] -= h[s:]
    else:
        p = -s
        if p > 0:
            g[:n-p] += (1.0 - h[p:])
            g[p:] -= h[:n-p]
    return g

        if c5 < best_c5:
            best_h, best_c5 = h, c5
            print("Phase3-greedy C5={:.8f}".format(c5))

    if _tl() > 15:
        h, c5 = _subgrad_topk(best_h, best_n, 3, 200, 0.02, rng)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
            print("Phase3-finalSG C5={:.8f}".format(c5))

    if _tl() > 10:
        h, c5 = _coord_descent(best_h, best_n, 800, rng)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
            print("Phase3-coord C5={:.8f}".format(c5))

    if _tl() > 8:
        h, c5 = _subgrad_topk(best_h, best_n, 2, 150, 0.01, rng)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Final projection and evaluation
    best_h = _proj(best_h, best_n)
    final_c5 = _c5(best_h)

    try:
        ec = evaluate_erdos_solution(best_h)
        if ec < final_c5:
            final_c5 = ec
    except Exception:
        pass

    print("FINAL n={} C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
