    for idx in range(len(corr)):
        w = weights[idx] * dx
        if w < 1e-10:
            continue
        lag = int(lags[idx])
        if lag >= 0:
            # x[j]*(1-x[j-lag]) for j from lag to n-1
            # d/dx[i]: (1-x[i-lag]) if i>=lag, minus x[i+lag] if i+lag<n (i.e., i < n-lag)
            valid_forward = np.arange(n) >= lag  # i-lag valid means i >= lag
            valid_backward = np.arange(n) < (n - lag)  # i+lag valid means i < n-lag
            
            # (1-x[i-lag]) for i >= lag
            f = (1.0 - x[lag:])  # shape (n-lag,)
            grad[lag:] += w * f
            # -x[i+lag] for i < n-lag
            b = -x[:n-lag]  # shape (n-lag,)
            grad[:n-lag] += w * b
        else:
            abs_lag = -lag
            # x[j]*(1-x[j+abs_lag]) for j from 0 to n-1-abs_lag
            # d/dx[i]: (1-x[i+abs_lag]) if i+abs_lag<n, minus x[i-abs_lag] if i>=abs_lag
            valid_forward = np.arange(n) < (n - abs_lag)  # i+abs_lag valid
            valid_backward = np.arange(n) >= abs_lag  # i-abs_lag valid
            
            f = (1.0 - x[abs_lag:])  # (1-x[i+abs_lag]) for i < n-abs_lag
            grad[:n-abs_lag] += w * f
            b = -x[:n-abs_lag]  # -x[i-abs_lag] for i >= abs_lag
            grad[abs_lag:] += w * b
    
    return grad

        if remaining_time() < 25.0:
            break
        print("\n--- Phase 1: n={} ---".format(n))

        # Multiple initializations
        inits = []
        h_interp = project_feasible(interp_to(best_h, best_n, n), n)
        inits.append(h_interp)

        h_noise = make_noise_init(n, rng, base=h_interp)
        inits.append(h_noise)

        h_banded = make_banded_init(n, rng, base=None)
        inits.append(h_banded)

        # Random binary
        h_bin = np.zeros(n)
        perm = rng.permutation(n)
        h_bin[perm[:n // 2]] = 1.0
        inits.append(h_bin)

        for h0 in inits:
            if remaining_time() < 15.0:
                break
            h_cur = h0.copy()
            c5_cur = compute_c5(h_cur)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n

            # Quick SLSQP
            dx = 2.0 / n
            target = n / 2.0
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': con_sum_fn, 'args': (n,)}]
            try:
                res = minimize(obj_softmax, h_cur, method='SLSQP',
                               args=(100.0,),
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 200, 'ftol': 1e-13, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur:
                    h_cur = h_new
                    c5_cur = c5_new
            except Exception:
                pass

            n_results.append((c5_cur, n, h_cur.copy()))
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n

        print("  Phase 1 n={} best: C5={:.8f}".format(n, min([r[0] for r in n_results if r[1] == n], default=1.0)))

    # Sort results
    n_results.sort(key=lambda x: x[0])
    top_resolutions = []
    for c5_r, n_r, h_r in n_results:
        if n_r not in top_resolutions:
            top_resolutions.append(n_r)
        if len(top_resolutions) >= 3:
            break
    if best_n not in top_resolutions:
        top_resolutions.append(best_n)

    print("\nPhase 2 resolutions: {}".format(top_resolutions))

    # Phase 2: Intensive SLSQP at top resolutions
    for n in top_resolutions:
        if remaining_time() < 60.0:
            break
        print("\n--- Phase 2: n={} ---".format(n))

        # Get best starting point for this n
        h_start = project_feasible(interp_to(best_h, best_n, n), n)
        c5_start = compute_c5(h_start)
        for c5_r, n_r, h_r in n_results:
            if n_r == n and c5_r < c5_start:
                h_start = h_r.copy()
                c5_start = c5_r
                break

        h_opt, c5_opt = do_slsqp_phase(h_start, n, 30.0, rng, tag="n={}".format(n))
        if c5_opt < best_c5 - 1e-10:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n
            print("  Phase 2 improved: n={}, C5={:.8f}".format(n, best_c5))

    # Phase 3: Subgradient at best resolution
    if remaining_time() > 40.0:
        print("\n--- Phase 3: Subgradient ---")
        max_iters = int(min(500, remaining_time() // 1.5))
        best_h, best_c5 = do_subgradient_phase(best_h, best_n, max_iters, rng)
        best_n = len(best_h)
        print("  After subgradient: n={}, C5={:.8f}".format(best_n, best_c5))

    # Phase 4: More SLSQP refinement (second pass with higher beta)
    if remaining_time() > 60.0:
        print("\n--- Phase 4: SLSQP 2nd pass ---")
        n = best_n
        dx = 2.0 / n
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_sum_fn, 'args': (n,)}]

        for beta in [3000, 6000, 10000, 15000]:
            if remaining_time() < 20.0:
                break
            try:
                res = minimize(obj_softmax, best_h, method='SLSQP',
                               args=(float(beta),),
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-16, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h_new
                    print("  beta={} improved: C5={:.8f}".format(beta, best_c5))
            except Exception:
                pass

        # Lp with very high p
        for p in [50, 100, 200]:
            if remaining_time() < 15.0:
                break
            try:
                res = minimize(obj_lp, best_h, method='SLSQP',
                               args=(float(p),),
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-16, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h_new
                    print("  Lp p={} improved: C5={:.8f}".format(p, best_c5))
            except Exception:
                pass

    # Phase 5: Coordinate descent
    if remaining_time() > 30.0:
        print("\n--- Phase 5: Coordinate Descent ---")
        max_sweeps = int(min(20, remaining_time() // 8))
        best_h, best_c5 = do_coordinate_descent(best_h, best_n, max_sweeps, rng)
        best_n = len(best_h)
        print("  After coord descent: n={}, C5={:.8f}".format(best_n, best_c5))

    # Phase 6: Simulated annealing on active coords
    if remaining_time() > 40.0:
        print("\n--- Phase 6: Simulated Annealing ---")
        max_iter_sa = int(min(800, remaining_time() // 0.8))
        best_h, best_c5 = do_simulated_annealing(best_h, best_n, max_iter_sa, rng)
        best_n = len(best_h)
        print("  After SA: n={}, C5={:.8f}".format(best_n, best_c5))

    # Phase 7: Final subgradient + SLSQP
    if remaining_time() > 30.0:
        print("\n--- Phase 7: Final Polish ---")
        n = best_n
        max_iters = int(min(200, remaining_time() // 2))
        best_h, best_c5 = do_subgradient_phase(best_h, n, max_iters, rng)

        if remaining_time() > 15.0:
            dx = 2.0 / n
            target = n / 2.0
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': con_sum_fn, 'args': (n,)}]
            for beta in [8000, 12000]:
                if remaining_time() < 10.0:
                    break
                try:
                    res = minimize(obj_softmax, best_h, method='SLSQP',
                                   args=(float(beta),),
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 300, 'ftol': 1e-16, 'iprint': 0})
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5 - 1e-12:
                        best_c5 = c5_new
                        best_h = h_new
                except Exception:
                    pass

    # Phase 8: Final coordinate descent
    if remaining_time() > 15.0:
        max_sweeps = int(min(10, remaining_time() // 5))
        best_h, best_c5 = do_coordinate_descent(best_h, best_n, max_sweeps, rng)
        best_n = len(best_h)

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
