def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr, dx

def project_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        cnt = np.sum(free)
        if cnt > 0:
            h[free] += diff / cnt
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            i = np.argmin(h)
        else:
            i = np.argmax(h)
        h[i] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()

    xq = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xq, xp, src)
