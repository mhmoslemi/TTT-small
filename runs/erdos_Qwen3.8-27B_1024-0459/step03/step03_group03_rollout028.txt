import numpy as np
from scipy.optimize import minimize
import time

_t0 = 0.0
_tb = 1300.0
_ng = 0
_bv = 100.0


def _tl():
    return max(0.0, _tb - (time.time() - _t0) - 3.0)


def _c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / n)


def _proj(h, n):
    h = np.array(h, dtype=np.float64)
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(30):
        d = 0.5 * n - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        k = int(np.sum(m))
        if k > 0:
            h[m] += d / k
        else:
            h[:] = 0.5
        np.clip(h, 0.0, 1.0, out=h)
    return h


def _binarize(h):
    n = len(h)
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n // 2]] = 1.0
    return b


def _interp_f(h, ns, nd):
    if ns == nd:
        return h.copy()
    xs = (np.arange(ns) + 0.5) * (2.0 / ns)
    xd = (np.arange(nd) + 0.5) * (2.0 / nd)
    return np.interp(xd, xs, h)


def _scon(x):
    return x.sum() - 0.5 * _ng


def _sobj(x):
    n = _ng
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = float(np.max(c))
    b = _bv
    sh = c - mx
    return (mx + np.log(np.exp(b * sh).sum()) / b) * (2.0 / n)


def _slsqp(h0, n, beta, maxiter):
    global _ng, _bv
    _ng = n
    _bv = beta
    try:
        r = minimize(_sobj, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon}],
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return _proj(np.asarray(r.x, dtype=np.float64), n)
    except Exception:
        return _proj(h0.copy(), n)


def _ggrad(h, kmax):
    n = len(h)
    s = kmax - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
    else:
        p = -s
        if p > 0:
            g[:n - p] += 1.0 - h[p:n]
            g[p:] -= h[:n - p]
    return g


def _smgrad(h, bw):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(corr))
    w = np.exp(bw * (corr - mx))
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
        sw = 1.0
    w /= sw
    g = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for k in nz:
        g += w[int(k)] * _ggrad(h, int(k))
    return g


def _sgloop(h, n, nit, rng):

    bc = _c5(best)

            break

        if gn < 1e-15:
            g = _smgrad(best, 500.0)
            gn = float(np.linalg.norm(g))
            if gn < 1e-15:
                break
        g /= gn
        moved = False
        for a in [0.06, 0.025, 0.01, 0.005, 0.04, 0.08, 0.015, 0.003, 0.012]:
            h2 = _proj(best - a * g, n)

def _sa_binary(h, n, niter, rng, T0, T1):
    """Simulated annealing on binary strings with pair and block moves."""
    best = h.copy()
    bc = _c5(best)
    cur = best.copy()
    cc = bc
    for it in range(niter):
        if _tl() < 5:
            break
        T = T0 * (T1 / T0) ** (it / max(1, niter))
        if T < 1e-12:
            T = 1e-12
        ones = np.where(cur > 0.999)[0]
        zeros = np.where(cur < 0.001)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        h2 = cur.copy()
        rmove = rng.randint(10)
        if rmove < 6:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h2[i], h2[j] = 0.0, 1.0
        elif rmove < 9:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            kk = ones[rng.randint(len(ones))]
            ll = zeros[rng.randint(len(zeros))]
            if kk == i or ll == j:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h2[i], h2[j] = 0.0, 1.0
            else:
                h2[i] = 0.0
                h2[j] = 1.0
                h2[kk] = 0.0
                h2[ll] = 1.0
        else:
            blen = min(rng.randint(2, 6), len(ones) // 2)
            i = ones[rng.randint(len(ones))]
            block_i = [i + d for d in range(blen) if 0 <= i + d < n]
            block_i = [x for x in block_i if cur[x] > 0.999]
            if len(block_i) < 2:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h2[i], h2[j] = 0.0, 1.0
            else:
                j = zeros[rng.randint(len(zeros))]
                block_j = [j + d for d in range(blen) if 0 <= j + d < n]
                block_j = [x for x in block_j if cur[x] < 0.001]
                if len(block_j) < 2:
                    h2[i], h2[j] = 0.0, 1.0
                else:
                    for bi in block_i[:len(block_j)]:
                        h2[bi] = 0.0
                    for bj in block_j[:len(block_i)]:
                        h2[bj] = 1.0
        c2 = _c5(h2)
        if c2 < cc:
            cur = h2
            cc = c2
            if cc < bc:
                best = cur.copy()
                bc = cc
        else:
            if rng.rand() < np.exp(-(c2 - cc) / T):
                cur = h2
                cc = c2
    return best, bc


def _ils(h, n, rng, nrestarts):
    """Iterated local search: perturb best, re-optimize."""
    best = h.copy()
    bc = _c5(best)
    for _ in range(nrestarts):
        if _tl() < 30:
            break
        hp = best.copy()
        nflip = rng.randint(3, 12)
        ones = np.where(hp > 0.999)[0]
        zeros = np.where(hp < 0.001)[0]
        if len(ones) < nflip * 2 or len(zeros) < nflip * 2:
            continue
        perm_o = rng.permutation(len(ones))[:nflip]
        perm_z = rng.permutation(len(zeros))[:nflip]
        for idx in perm_o:
            hp[ones[idx]] = 0.0
        for idx in perm_z:
            hp[zeros[idx]] = 1.0
        hp = _proj(hp, n)
        hc, cc = _greedy_bin(hp, n, 30, rng)
        hc2, cc2 = _sgloop(hc, n, 80, rng)
        if cc2 < bc:
            best = hc2
            bc = cc2
    return best, bc

def _thue_morse(n):
    """Generate Thue-Morse based binary pattern with n/2 ones."""
    t = np.zeros(n)
    for i in range(n):
        tc = bin(i).count('1')
        t[i] = 1.0 if tc % 2 == 0 else 0.0
    if t.sum() > n / 2:
        idx1 = np.where(t > 0.5)[0]
        rng_local = np.random.RandomState(1)
        pick = rng_local.choice(idx1, size=int(t.sum() - n // 2), replace=False)
        t[pick] = 0.0
    elif t.sum() < n / 2:
        idx0 = np.where(t < 0.5)[0]
        rng_local = np.random.RandomState(1)
        pick = rng_local.choice(idx0, size=int(n // 2 - t.sum()), replace=False)
        t[pick] = 1.0
    return t
