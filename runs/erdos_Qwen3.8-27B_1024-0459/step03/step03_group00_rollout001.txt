def project_feasible(h, n):
    h = np.asarray(h, dtype=float)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0).astype(float, copy=True)
    target = 0.5 * float(n)
    for _ in range(100):
        s = float(np.sum(h))
        diff = target - s
        if abs(diff) <= 1e-14:
            break
        if diff > 0.0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            h = np.full(n, target / float(n), dtype=float)
            break
        h[mask] += diff / float(cnt)
        np.clip(h, 0.0, 1.0, out=h)
    # exact correction
    for _ in range(30):
        s = float(np.sum(h))
        diff = target - s
        if diff == 0.0:
            break
        if diff > 0.0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        idxs = np.where(mask)[0]
        if idxs.size == 0:
            if diff > 0.0:
                idx = int(np.argmin(h))  # most space?
            else:
                idx = int(np.argmax(h))
        else:
            if diff > 0.0:

def pair_try(c, h, i, j, d, n, cur_max, tol):
    cm = c.copy()
    di = float(d)
    dj = -di
    eps = 1e-12 * max(1.0, float(cur_max))
    for s in range(-n+1, n):
       L = n - (s if s>=0 else -s)
       start = 0 if s >=0 else -s
       end = start + L
       ch = 0.0
       if start <= i < end:
           q = i + s
           ch += di * (1.0 - h[q])
           if q == i:
               ch -= di * di
           elif q == j:
               ch -= di * dj
       if start <= j < end:
           q = j + s
           ch += dj * (1.0 - h[q])
           if q == i:
               ch -= dj * di
           elif q == j:
               ch -= dj * dj
       p = i - s
       if start <= p < end:
           ch -= h[p] * di
       p = j - s
       if start <= p < end:
           ch -= h[p] * dj
       if ch != 0.0:
           idx = s + n - 1
           nv = cm[idx] + ch
           if nv > cur_max + eps:
               return None
           cm[idx] = nv
    new_max = float(np.max(cm))
    if new_max < cur_max - tol:
        return (cm, new_max)
    return None

import numpy as np
from scipy.optimize import minimize

def remaining_time():
    return max(0.0, float(_G_BUDGET) - (time.time() - _G_START) - 5.0)


def set_globals(n, beta):
    global _G_N, _G_TGT, _G_DX, _G_BETA, _G_ONES, _G_HASH, _G_HAVE_GRAD
    _G_N = int(n)
    _G_TGT = 0.5 * float(_G_N)
    _G_DX = 2.0 / float(max(1, _G_N))
    _G_BETA = float(beta)
    _G_ONES = np.ones(max(1, _G_N), dtype=float)
    if _G_HASH is None or len(_G_HASH) != _G_N:
        _G_HASH = np.linspace(0.0, 1.0, max(1, _G_N), dtype=float)
    _G_HAVE_GRAD = False


def checksum(x):
    global _G_HASH
    x = np.asarray(x, dtype=float)
    n = len(x)
    if _G_HASH is None or len(_G_HASH) != n:
        _G_HASH = np.linspace(0.0, 1.0, max(1, n), dtype=float)
    return (
        float(x[0]),
        float(x[-1]),
        float(np.sum(x)),
        float(np.sum(x * x)),
        float(np.max(x)),
        float(np.min(x)),
        float(np.dot(x, _G_HASH)),
    )


def correlate_direct(h):
    return np.correlate(h, 1.0 - h, mode="full")


def correlate_h(h):
    n = len(h)
    if n <= 700:
        return correlate_direct(h)
    b = 1.0 - h[::-1]
    conv = fftconvolve(h, b, mode="full")
    return conv[::-1]

    h = np.asarray(h, dtype=float)
    n = len(h)
    if n <= 0:
        return float("inf")
    c = correlate_direct(h)
    return float(np.max(c)) * (2.0 / float(n))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0).astype(float, copy=True)
    target = 0.5 * float(n)

    for _ in range(80):
        s = float(np.sum(h))
        diff = target - s
        if abs(diff) <= 1e-14:
            break
        if diff > 0.0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            h = np.full(n, 0.5, dtype=float)
            break
        h[mask] += diff / float(cnt)
        np.clip(h, 0.0, 1.0, out=h)

    for _ in range(30):
        s = float(np.sum(h))
        diff = target - s
        if diff == 0.0:
            break
        if diff > 0.0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        idxs = np.where(mask)[0]
        if idxs.size == 0:
            if diff > 0.0:
                idx = int(np.argmin(h))
            else:
                idx = int(np.argmax(h))
        else:
            if diff > 0.0:
                idx = int(idxs[np.argmax(1.0 - h[idxs])])
            else:
                idx = int(idxs[np.argmax(h[idxs])])
        h[idx] += diff
        if h[idx] < 0.0:
            h[idx] = 0.0
        elif h[idx] > 1.0:
            h[idx] = 1.0
    return h


def make_block(n):
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    if n % 2 == 1:
        h[n // 2] = 0.5
    return project_feasible(h, n)


def make_thue_morse(n):
    h = np.zeros(n, dtype=float)
    for i in range(n):
        cnt = 0
        v = i
        while v:
            cnt ^= (v & 1)
            v >>= 1
        if cnt:
            h[i] = 1.0
    return project_feasible(h, n)


def make_golden_ratio(n):
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    i = np.arange(n, dtype=float)
    frac = np.mod(i * gr, 1.0)
    h = (frac < 0.5).astype(float)
    return project_feasible(h, n)


def make_random_bin(rng, n):
    h = np.zeros(n, dtype=float)
    idx = rng.permutation(n)
    h[idx[: n // 2]] = 1.0
    if n % 2 == 1:
        h[idx[n // 2]] = 0.5
    return project_feasible(h, n)


def make_sinusoidal(rng, n, freq, phase):
    t = np.arange(n, dtype=float) / float(max(1, n))
    h = 0.5 + 0.42 * np.sin(2.0 * np.pi * freq * t + phase)
    return project_feasible(h, n)


def interp_to(src_h, src_n, dst_n):
    src_h = np.asarray(src_h, dtype=float)
    src_n = int(src_n)
    dst_n = int(dst_n)
    if src_n == dst_n:
        return project_feasible(src_h, dst_n)
    xp = (np.arange(src_n) + 0.5) * (2.0 / float(src_n))
    xn = (np.arange(dst_n) + 0.5) * (2.0 / float(dst_n))
    return project_feasible(np.interp(xn, xp, src_h), dst_n)


def softmax_weights(c, beta):
    c = np.asarray(c, dtype=float)
    m = float(np.max(c))
    if not np.isfinite(beta) or beta <= 0.0:
        return np.full(c.shape, 1.0 / float(len(c)), dtype=float), m
    z = beta * (c - m)
    np.clip(z, -745.0, 709.0, out=z)
    ex = np.exp(z)
    s = float(np.sum(ex))
    if s <= 0.0 or not np.isfinite(s):
        w = np.full(c.shape, 1.0 / float(len(c)), dtype=float)
        return w, m
    w = ex / s
    val = m + np.log(s) / beta
    return w, float(val)


def weighted_grad_from_w(h, w):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return np.empty(0, dtype=float)
    center = n - 1
    m = 2 * n - 1
    w = np.asarray(w, dtype=float)

    cum = np.cumsum(w, dtype=float)
    low = center - np.arange(n, dtype=int)
    high = 2 * n - 2 - np.arange(n, dtype=int)
    pre = np.empty(m, dtype=float)
    pre[0] = 0.0
    pre[1:] = cum[:-1]
    a = cum[high] - pre[low]

    k = np.zeros(m, dtype=float)
    k[center] = -2.0 * w[center]
    if n > 1:
        plus = w[center + 1:]
        minus = w[center - 1::-1]
        vals = -(plus + minus)
        k[center + 1:] = vals
        k[center - 1::-1] = vals

    if n <= 700:
        conv = np.convolve(h, k, mode="full")
    else:
        conv = fftconvolve(h, k, mode="full")
    b = conv[center:center + n]
    return a + b


def smooth_obj(x):
    global _G_LAST_VAL, _G_LAST_GRAD, _G_HAVE_GRAD, _G_LAST_CS, _G_LAST_W, _G_LAST_C
    h = np.asarray(x, dtype=float)
    cs = checksum(h)
    if _G_HAVE_GRAD and _G_LAST_CS is not None and cs == _G_LAST_CS:
        return _G_LAST_VAL
    c = correlate_h(h)
    w, val = softmax_weights(c, _G_BETA)
    _G_LAST_VAL = float(val) * _G_DX
    _G_LAST_W = w
    _G_LAST_C = c
    _G_LAST_CS = cs
    _G_HAVE_GRAD = False
    return _G_LAST_VAL


def smooth_jac(x):
    global (
        _G_LAST_VAL, _G_LAST_GRAD, _G_HAVE_GRAD, _G_LAST_CS,
        _G_LAST_W, _G_LAST_C,
    )
    h = np.asarray(x, dtype=float)
    cs = checksum(h)
    if _G_HAVE_GRAD and _G_LAST_CS is not None and cs == _G_LAST_CS:
        return _G_LAST_GRAD
    c = correlate_h(h)
    w, val = softmax_weights(c, _G_BETA)
    g = weighted_grad_from_w(h, w) * _G_DX
    _G_LAST_VAL = float(val) * _G_DX
    _G_LAST_W = w
    _G_LAST_C = c
    _G_LAST_CS = cs
    _G_LAST_GRAD = g
    _G_HAVE_GRAD = True
    return g


def con_sum(x):
    return float(np.sum(x) - _G_TGT)


def con_sum_jac(x):
    if _G_ONES is None:
        return np.ones(max(1, _G_N), dtype=float)
    return _G_ONES.copy()


def slsqp_polish(h_in, n, beta_list, maxiter_list, ftol):
    n = int(n)
    h0 = project_feasible(h_in, n)
    current_c5 = compute_c5(h0)
    if not np.isfinite(current_c5):
        return h0, float("inf")

    best_h = h0.copy()
    best_c5 = current_c5
    h = h0.copy()
    current_c5 = best_c5

    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": con_sum, "jac": con_sum_jac}]

    for bi, beta in enumerate(beta_list):
        rem = remaining_time()
        if rem < 25.0:
            break
        base = maxiter_list[bi] if bi < len(maxiter_list) else 60
        scale = 1.0
        if n > 512:
            scale = 0.55
        elif n > 320:
            scale = 0.75
        maxit = int(float(base) * scale)

        if rem < 260.0:
            maxit = min(maxit, 90)
        if rem < 130.0:
            maxit = min(maxit, 50)
        if rem < 70.0:
            maxit = min(maxit, 25)
        maxit = max(10, maxit)

        set_globals(n, float(beta))
        try:
            res = minimize(
                smooth_obj,
                h,
                method="SLSQP",
                jac=smooth_jac,
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": maxit, "ftol": ftol, "iprint": 0},
            )
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if not np.isfinite(c5_new):
                continue
            if c5_new < current_c5 - 1e-11:
                current_c5 = float(c5_new)
                h = h_new.copy()
                if current_c5 < best_c5 - 1e-12:
                    best_c5 = current_c5
                    best_h = h.copy()
            elif abs(c5_new - current_c5) <= 1e-9:
                h = h_new.copy()
                current_c5 = float(c5_new)
                if current_c5 < best_c5 - 1e-12:
                    best_c5 = current_c5
                    best_h = h.copy()
        except Exception:
            pass

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, float(best_c5)


def subgradient_polish(h_in, n, beta_base=50.0, max_iters=80):
    n = int(n)
    h = project_feasible(h_in, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    if not np.isfinite(current_c5):
        return best_h, float("inf")

    alphas = (
        0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
        0.0005, 0.0002, 0.1, 0.0001,
    )

    for it in range(max_iters):
        if remaining_time() < 15.0:
            break
        c = correlate_direct(h)
        beta = min(15000.0, float(beta_base) * (1.0 + 0.10 * float(it)))
        w, _ = softmax_weights(c, beta)
        g = weighted_grad_from_w(h, w)
        g = g - np.mean(g)
        gn = float(np.linalg.norm(g))
        if gn < 1e-12:
            break
        direction = -g / gn

        improved = False
        for a in alphas:
            if remaining_time() < 10.0:
                break
            trial = project_feasible(h + float(a) * direction, n)
            cval = compute_c5(trial)
            if np.isfinite(cval) and cval < current_c5 - 1e-10:
                h = trial
                current_c5 = float(cval)
                improved = True
                if current_c5 < best_c5 - 1e-12:
                    best_c5 = current_c5
                    best_h = h.copy()
                break
        if not improved:
            break

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, float(best_c5)


def make_deltas(dmax):
    dmax = float(dmax)
    if dmax <= 1e-14:
        return []
    out = []
    fr = (
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.001953125,
        0.0009765625, 0.00048828125,
    )
    for f in fr:
        d = dmax * f
        if d > 1e-14:
            out.append(min(d, dmax * (1.0 - 1e-15)))
    absol = (
        0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
        0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5,
    )
    for a_val in absol:
        if a_val <= 1e-14:
            continue
        if a_val < dmax:
            out.append(min(a_val, dmax * (1.0 - 1e-15)))

    clean = []
    for d in out:
        dup = False
        for e in clean:
            if abs(d - e) <= 1e-15 * max(1.0, abs(e)):
                dup = True
                break
        if not dup:
            clean.append(d)
    clean.sort(reverse=True)
    return clean


def pair_try(c, h, i, j, d, n, cur_max, tol):
    cm = c.copy()
    di = float(d)
    dj = -di
    eps = 1e-12 * max(1.0, float(cur_max))
    center = n - 1

    for s in range(-n + 1, n):
        if s >= 0:
            length = n - s
            start = 0
        else:
            length = n + s
            start = -s
        end = start + length
        ch = 0.0

        if start <= i < end:
            q = i + s
            ch += di * (1.0 - h[q])
            if q == i:
                ch -= di * di
            elif q == j:
                ch -= di * dj

        if start <= j < end:
            q = j + s
            ch += dj * (1.0 - h[q])
            if q == i:
                ch -= dj * di
            elif q == j:
                ch -= dj * dj

        p = i - s
        if start <= p < end:
            ch -= h[p] * di

        p = j - s
        if start <= p < end:
            ch -= h[p] * dj

        if ch != 0.0:
            idx = s + center
            nv = cm[idx] + ch
            if nv > cur_max + eps:
                return None
            cm[idx] = nv

    new_max = float(np.max(cm))
    if new_max < cur_max - tol:
        return cm, new_max
    return None


def try_pair_move(c, h, i, j, dmax, n, cur_max):
    deltas = make_deltas(dmax)
    if not deltas:
        return None
    tol = 1e-12 * max(1.0, float(cur_max))
    best = None
    best_max = float(cur_max)
    for d in deltas:
        res = pair_try(c, h, i, j, float(d), n, cur_max, tol)
        if res is not None:
            cm, new_max = res
            if new_max < best_max - tol:
                best_max = new_max
                best = cm, new_max, float(d)
    return best


def coordinate_refine(h_in, n, rng):
    n = int(n)
    dx = 2.0 / float(n)

    best_h = project_feasible(h_in, n)
    best_c5 = compute_c5(best_h)
    if not np.isfinite(best_c5):
        return best_h, float("inf")

    h = best_h.copy()
    min_rem = 30.0
    sweeps = 0
    beta_cycle = (50.0, 200.0, 1000.0, 5000.0, 2000.0, 500.0)

    while remaining_time() > min_rem + 10.0:
        sweeps += 1
        prev_best = best_c5

        h = project_feasible(h, n)
        c = correlate_direct(h)
        cur_max = float(np.max(c))
        cur_c5 = cur_max * dx
        if np.isfinite(cur_c5) and cur_c5 < best_c5 - 1e-12:
            best_c5 = float(cur_c5)
            best_h = h.copy()

        beta = float(beta_cycle[sweeps % len(beta_cycle)])
        w, _ = softmax_weights(c, beta)
        g = weighted_grad_from_w(h, w)
        g = g - np.mean(g)

        order = np.argsort(g)
        low = order[:12]
        high = order[-12:]

        local_improved = False

        for jj in high:
            if remaining_time() <= min_rem:
                break
            for ii in low:
                if ii == jj:
                    continue
                if g[ii] >= g[jj]:
                    continue
                dmax = min(1.0 - float(h[ii]), float(h[jj]))
                if dmax <= 1e-13:
                    continue
                res = try_pair_move(c, h, int(ii), int(jj), dmax, n, cur_max)
                if res is not None:
                    cm, new_max, d = res
                    h[ii] += d
                    h[jj] -= d
                    if h[ii] < 0.0:
                        h[ii] = 0.0
                    elif h[ii] > 1.0:
                        h[ii] = 1.0
                    if h[jj] < 0.0:
                        h[jj] = 0.0
                    elif h[jj] > 1.0:
                        h[jj] = 1.0
                    c = cm
                    cur_max = new_max
                    local_improved = True

        if local_improved and remaining_time() > min_rem + 10.0:
            c = correlate_direct(h)
            cur_max = float(np.max(c))
            w, _ = softmax_weights(c, 1000.0)
            g = weighted_grad_from_w(h, w)
            g = g - np.mean(g)
            for _ in range(90):
                if remaining_time() <= min_rem:
                    break
                ii = int(rng.randint(n))
                jj = int(rng.randint(n))
                if ii == jj:
                    continue
                if g[ii] >= g[jj]:
                    ii, jj = jj, ii
                dmax = min(1.0 - float(h[ii]), float(h[jj]))
                if dmax <= 1e-13:
                    continue
                res = try_pair_move(c, h, ii, jj, dmax, n, cur_max)
                if res is not None:
                    cm, new_max, d = res
                    h[ii] += d
                    h[jj] -= d
                    if h[ii] < 0.0:
                        h[ii] = 0.0
                    elif h[ii] > 1.0:
                        h[ii] = 1.0
                    if h[jj] < 0.0:
                        h[jj] = 0.0
                    elif h[jj] > 1.0:
                        h[jj] = 1.0
                    c = cm
                    cur_max = new_max
                    local_improved = True

        h = project_feasible(h, n)
        final_c5 = compute_c5(h)
        if np.isfinite(final_c5) and final_c5 < best_c5 - 1e-12:
            best_c5 = float(final_c5)
            best_h = h.copy()

        if final_c5 >= prev_best - 1e-11:
            break
        if sweeps > 160:
            break

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, float(best_c5)


def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(int(seed))

    best_h = None
    best_n = 0
    best_c5 = float("inf")
    results = []

    try:
        ph = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        ph = None

    if ph is not None and len(ph) > 0:
        n0 = len(ph)
        h0 = project_feasible(ph, n0)
        c0 = compute_c5(h0)
        best_h = h0.copy()
        best_n = int(n0)
        best_c5 = float(c0)
        if np.isfinite(c0):
            results.append((float(c0), int(n0), h0.copy()))
    else:
        n0 = 256
        h0 = make_block(n0)
        c0 = compute_c5(h0)
        best_h = h0.copy()
        best_n = int(n0)
        best_c5 = float(c0)
        if np.isfinite(c0):
            results.append((float(c0), int(n0), h0.copy()))

    raw_ns = [
        best_n, 128, 160, 192, 224, 256,
        288, 320, 384, 512,
    ]
    seen_ns = []
    for nn in raw_ns:
        nn = int(nn)
        if 96 <= nn <= 768 and nn not in seen_ns:
            seen_ns.append(nn)
    dist_ns = [(abs(nn - best_n), nn) for nn in seen_ns]
    dist_ns.sort()
    candidates = [nn for _, nn in dist_ns]
    quick_ns = [nn for nn in candidates if nn != best_n]

    limit = 0
    if remaining_time() > 800.0:
        limit = 7
    elif remaining_time() > 450.0:
        limit = 5
    elif remaining_time() > 220.0:
        limit = 3

    quick_betas = [10.0, 60.0, 300.0]
    quick_iters = [45, 70, 80]

    for nn in quick_ns[:limit]:
        if remaining_time() < 120.0:
            break
        h_try = interp_to(best_h, best_n, nn)
        h_try, c_try = slsqp_polish(h_try, nn, quick_betas, quick_iters, 1e-12)
        if np.isfinite(c_try):
            if c_try < best_c5 - 1e-12:
                best_c5 = float(c_try)
                best_h = h_try.copy()
                best_n = int(nn)
            results.append((float(c_try), int(nn), h_try.copy()))

    if remaining_time() > 260.0:
        sn = int(best_n)
        starts = []
        starts.append(make_thue_morse(sn))
        starts.append(make_golden_ratio(sn))
        starts.append(make_random_bin(rng, sn))
        starts.append(make_sinusoidal(rng, sn, 3.0, float(rng.rand() * 6.283185307179586)))

        for _m in range(2):
            h_mut = best_h.astype(float, copy=True)
            if len(h_mut) > 0:
                mask = rng.rand(sn) < 0.15
                if np.any(mask):
                    cnt = int(np.count_nonzero(mask))
                    h_mut[mask] += (rng.rand(cnt) - 0.5) * 0.30
            starts.append(project_feasible(h_mut, sn))

        medium_betas = [30.0, 150.0, 600.0]
        medium_iters = [55, 75, 75]
        used = 0
        max_starts = 3 if remaining_time() < 550.0 else 5
        for h_start in starts:
            if remaining_time() < 140.0 or used >= max_starts:
                break
            used += 1
            h_try, c_try = slsqp_polish(h_start, sn, medium_betas, medium_iters, 1e-12)
            if np.isfinite(c_try):
                if c_try < best_c5 - 1e-12:
                    best_c5 = float(c_try)
                    best_h = h_try.copy()
                    best_n = int(sn)
                results.append((float(c_try), int(sn), h_try.copy()))

    if results:
        neg = [(-float(results[i][0]), i) for i in range(len(results))]
        neg.sort()
        top_idx = []
        top_seen = []
        for _, idx in neg:
            nn = int(results[idx][1])
            if nn not in top_seen:
                top_seen.append(nn)
                top_idx.append(idx)
            if len(top_idx) >= 2:
                break

        full_betas = [10.0, 40.0, 150.0, 500.0, 2000.0, 8000.0, 25000.0, 80000.0]
        full_iters = [70, 95, 110, 110, 90, 70, 55, 45]

        for idx in top_idx:
            if remaining_time() < 180.0:
                break
            nn = int(results[idx][1])
            h_start = results[idx][2].astype(float, copy=True)
            h_try, c_try = slsqp_polish(h_start, nn, full_betas, full_iters, 1e-13)
            if np.isfinite(c_try):
                if c_try < best_c5 - 1e-12:
                    best_c5 = float(c_try)
                    best_h = h_try.copy()
                    best_n = int(nn)
                results.append((float(c_try), int(nn), h_try.copy()))

    if remaining_time() > 90.0:
        best_h, best_c5 = subgradient_polish(best_h, best_n, 100.0, 70)
        if np.isfinite(best_c5):
            results.append((float(best_c5), int(best_n), best_h.copy()))

    if remaining_time() > 70.0:
        best_h, best_c5 = coordinate_refine(best_h, best_n, rng)
        if np.isfinite(best_c5):
            results.append((float(best_c5), int(best_n), best_h.copy()))

    if remaining_time() > 70.0:
        best_h, best_c5 = slsqp_polish(
            best_h,
            best_n,
            [1000.0, 8000.0, 50000.0],
            [55, 45, 35],
            1e-13,
        )
        if np.isfinite(best_c5):
            results.append((float(best_c5), int(best_n), best_h.copy()))

    if remaining_time() > 55.0:
        best_h, best_c5 = coordinate_refine(best_h, best_n, rng)
        if np.isfinite(best_c5):
            results.append((float(best_c5), int(best_n), best_h.copy()))

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    if not np.isfinite(final_c5):
        for c_tr, n_tr, h_tr in results:
            if np.isfinite(c_tr):
                h_chk = project_feasible(h_tr, n_tr)
                c_chk = compute_c5(h_chk)
                if np.isfinite(c_chk) and (not np.isfinite(final_c5) or c_chk < final_c5):
                    final_c5 = float(c_chk)
                    best_h = h_chk
                    best_n = int(n_tr)
        if not np.isfinite(final_c5):
            best_h = make_block(256)
            best_n = 256
            final_c5 = compute_c5(best_h)

    try:
        ev = evaluate_erdos_solution(best_h)
        ev = float(ev)
        if np.isfinite(ev) and ev < final_c5 + 1e-12:
            final_c5 = min(float(final_c5), ev)
    except Exception:
        pass

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_n = int(len(best_h))

    return best_h, float(final_c5), best_n
