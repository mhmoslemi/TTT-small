    shifted = np.minimum(beta * (corr - m), 30.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def _smooth_max_grad(h):
    n = _N_G
    dx = 2.0 / n
    beta = _BETA_G
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 30.0)
    exp_s = np.exp(shifted)
    s = exp_s.sum()
    weights = exp_s / s  # softmax weights
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        offset = k_idx - (n - 1)
        a = abs(offset)
        n1 = n - a
        if offset >= 0:
            # corr = sum_{i=0}^{n1-1} h[i]*(1-h[i+a])
            # d/dh[j from h term]: j in [0, n1-1]: (1-h[j+a])
            # d/dh[j from (1-h) term]: j in [a, n-1]: -h[j-a]
            if a == 0:
                grad[:] += w * dx * (1.0 - 2.0 * h)
            else:
                grad[:n1] += w * dx * (1.0 - h[a:a+n1])
                grad[a:] += w * dx * (-h[:n1])
        else:
            # corr = sum_{i=0}^{n1-1} h[i+a]*(1-h[i])
            # d/dh[j from h term]: j in [a, n-1]: (1-h[j-a])
            # d/dh[j from (1-h) term]: j in [0, n1-1]: -h[j+a]
            grad[a:] += w * dx * (1.0 - h[:n1])
            grad[:n1] += w * dx * (-h[a:a+n1])
    return grad

def constraint_fun(x):
    return np.sum(x) - _N_G / 2.0

def constraint_grad(x):
    return np.ones(_N_G)

def slsqp_step(h, n, beta, maxiter=300, ftol=1e-18):
    global _N_G, _BETA_G
    _N_G = n
    _BETA_G = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_grad}]
    try:
        res = minimize(
            _smooth_max_val, h, method='SLSQP',
            jac=_smooth_max_grad,
            bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    fp = src
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, fp)

def subgrad_direction(h, n, top_k=3, tol=0.5):
    """Compute a subgradient direction reducing the top correlations."""
    corr = compute_corr(h)
    max_c = corr.max()
    # Active set: lags within tol of max
    active = np.where(corr >= max_c - tol)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(corr))])
    grad = np.zeros(n)
    for k_idx in active:
        offset = k_idx - (n - 1)
        a = abs(offset)
        n1 = n - a
        if offset >= 0:
            if a == 0:
                grad[:] += (1.0 - 2.0 * h)
            else:
                grad[:n1] += (1.0 - h[a:a+n1])
                grad[a:] += (-h[:n1])
        else:
            grad[a:] += (1.0 - h[:n1])
            grad[:n1] += (-h[a:a+n1])
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return None
    return grad / gnorm

def psaa_step(h, n, rng, step_size=0.02):
    """Simplified SPSA step."""
    corr = compute_corr(h)
    max_c = corr.max()
    dx = 2.0 / n
    # Pick a random active lag
    active = np.where(corr >= max_c - 0.3)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(corr))])
    k_idx = int(active[rng.randint(len(active))])
    offset = k_idx - (n - 1)
    a = abs(offset)
    n1 = n - a
    g = np.zeros(n)
    if offset >= 0:
        if a == 0:
            g[:] = (1.0 - 2.0 * h)
        else:
            g[:n1] = (1.0 - h[a:a+n1])
            g[a:] = (-h[:n1])
    else:
        g[a:] = (1.0 - h[:n1])
        g[:n1] = (-h[a:a+n1])
    gnorm = np.linalg.norm(g)
    if gnorm < 1e-16:
        return h.copy()
    g /= gnorm
    # Random perturbation for SPSA-like effect
    delta = rng.randn(n) * 0.01
    h_new = h - step_size * (g + delta)
    return project_feasible(h_new, n)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(mask.sum())
        if cnt > 0:

            h[:] += diff / n

def interp(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xq = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xq, xp, src)

def sg_direction(h, n, tol=0.3):
    corr = compute_corr(h)
    mx = corr.max()
    act = np.where(corr >= mx - tol)[0]
    if len(act) == 0:
        act = np.array([int(np.argmax(corr))])
    g = np.zeros(n)
    for k in act:
        d = k - (n - 1)
        a = abs(d)
        n1 = n - a
        if d >= 0:
            if a == 0:
                g[:] += (1.0 - 2.0 * h)
            else:
                g[:n1] += (1.0 - h[a:a + n1])
                g[a:a + n1] += (-h[:n1])
        else:
            g[a:a + n1] += (1.0 - h[:n1])
            g[:n1] += (-h[a:a + n1])
    gn = np.linalg.norm(g)
    if gn < 1e-16:
        return None
    return g / gn
