def project_box_sum(z, target):
    z = np.asarray(z, dtype=float).copy()
    n = len(z)
    target = float(np.clip(target, 0.0, float(n)))
    s = np.sum(np.clip(z,0.0,1.0))
    if abs(s-target)<1e-13:
        y=np.clip(z,0,1)
    else:
        lo = float(np.min(z)) - 1.0
        hi = float(np.max(z)) + 1.0
        for _ in range(80):
            mid = 0.5*(lo+hi)
            y = np.clip(z-mid, 0.0, 1.0)
            sy = float(np.sum(y))
            if sy > target:
                lo = mid
            else:
                hi = mid
        y = np.clip(z-0.5*(lo+hi), 0.0, 1.0)
    diff = target - float(np.sum(y))
    if abs(diff) > 1e-13:
        # distribute
        for _ in range(20):
            if diff > 0:
                mask = y < 1.0 - 1e-12
            else:
                mask = y > 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0: break
            y[mask] += diff / cnt
            y = np.clip(y,0,1)
            diff = target - float(np.sum(y))
            if abs(diff) < 1e-13: break
    if abs(diff) > 1e-12:
        # last resort
        if diff > 0:
            idx = int(np.argmax(1.0-y))
            room = 1.0-y[idx]
            y[idx] = min(1.0, y[idx]+diff)
        else:
            idx = int(np.argmax(y))
            room = y[idx]
            y[idx] = max(0.0, y[idx]+diff)
    return y
```

def project_box_sum(z, target):
    z = np.asarray(z, dtype=float).copy()
    n = z.shape[0]
    if n == 0:
        return z
    target = float(target)
    if target < 0.0:
        target = 0.0
    elif target > float(n):
        target = float(n)

    y = np.clip(z, 0.0, 1.0)
    if abs(float(np.sum(y)) - target) > 1e-12:
        lo = float(np.min(z)) - 1.0
        hi = float(np.max(z)) + 1.0
        for _ in range(80):
            mid = 0.5 * (lo + hi)
            y = np.clip(z - mid, 0.0, 1.0)
            sy = float(np.sum(y))
            if sy > target:
                lo = mid
            else:
                hi = mid
        y = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)

    diff = target - float(np.sum(y))
    for _ in range(20):
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            mask = y < 1.0 - 1e-12
        else:
            mask = y > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        y[mask] += diff / cnt
        y = np.clip(y, 0.0, 1.0)
        diff = target - float(np.sum(y))

    if abs(diff) > 1e-12:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - y))
            y[idx] = min(1.0, y[idx] + diff)
        else:
            idx = int(np.argmax(y))
            y[idx] = max(0.0, y[idx] + diff)
    return y

def logsum_values(corr, beta):
    m = float(np.max(corr))
    size = corr.shape[0]
    if beta <= 1e-8:
        w = np.full(size, 1.0 / float(size), dtype=float)
        return m, w

    sh = beta * (corr - m)
    sh = np.clip(sh, -700.0, 0.0)
    e = np.exp(sh)
    s = float(np.sum(e))
    if s <= 0.0:
        e[:] = 0.0
        idx = np.where(corr >= m - 1e-12)[0]
        e[idx] = 1.0
        s = float(np.sum(e))
    w = e / s
    val = m + float(np.log(s)) / beta
    return val, w


def corr_grad_weighted(h, w):
    h = np.asarray(h, dtype=float)
    n = h.shape[0]
    g = np.zeros(n, dtype=float)
    c = n - 1

    # Positive lags: c_l = sum h[i+l] * (1 - h[i])
    for l in range(0, n):
        wt = float(w[c + l])
        if wt == 0.0:
            continue
        m = n - l
        if l == 0:
            g[:] += wt * (1.0 - h[:m])
        else:
            g[l:n] += wt * (1.0 - h[:m])
        g[:m] += -wt * h[l:n]

    # Negative lags: c_-s = sum h[i] * (1 - h[i+s])
    for s in range(1, n):
        wt = float(w[c - s])
        if wt == 0.0:
            continue
        m = n - s
        g[:m] += wt * (1.0 - h[s:n])
        g[s:n] += -wt * h[:m]

    return g


def smooth_obj_full(h, n, beta):
    corr = compute_corr(h)

    return float(val) * (2.0 / float(n))

    corr = compute_corr(h)
    _, w = logsum_values(corr, beta)
    g = corr_grad_weighted(h, w)
    return (2.0 / float(n)) * g

def pair_eval(h, j, k, s, x):

        yy = 0.0

    h[k] = yy
    return compute_c5(h)


def pair_line_search(h, j, k, s, lo, hi, grid, refine1, refine2):

    best_val = pair_eval(h, j, k, s, best_x)
    if hi <= lo:
        return best_x, best_val

    refine2 = int(refine2)

def pair_coord_descent(h, n, max_sweeps, time_end, rng, grid=32, refine1=20, refine2=12):
    h = project_box_sum(np.asarray(h, dtype=float), n / 2.0)
    best = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c

    max_sweeps = int(max_sweeps)
    for _ in range(max_sweeps):
        if time.time() > time_end:
            break
        h = project_box_sum(h, n / 2.0)
        cur_c = compute_c5(h)
        if cur_c < best_c:
            best_c = cur_c
            best = h.copy()

        if time.time() > time_end - 5.0:
            break

        perm = rng.permutation(n)
        for j in perm:
            if time.time() > time_end:
                break
            k = choose_partner(h, j, rng)
            if k == j:
                k = (j + 1) % n
            oldj = float(h[j])
            oldk = float(h[k])
            s = oldj + oldk
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo <= 1e-14:
                continue

            _, cnew = pair_line_search(h, j, k, s, lo, hi, grid, refine1, refine2)
            if cnew < cur_c:
                cur_c = cnew
            if cur_c < best_c - 1e-15:
                best_c = cur_c
                best = h.copy()

    h = project_box_sum(best, n / 2.0)
    c = compute_c5(h)
    return h, c

        try:
            res = minimize(
                smooth_obj_sym,
                best_y,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                jac=smooth_jac_sym,
                args=(int(n), float(beta)),
                options={"maxiter": int(maxit), "ftol": 1e-12, "catol": 1e-10, "disp": 0},
            )
        except Exception:
            continue
        if res is None:
            continue
        x = np.asarray(res.x, dtype=float)
        if x.shape[0] != m or not np.all(np.isfinite(x)):
            continue
        x = project_sym_y(x, n)
        h = make_full_even(x, n)
        c = compute_c5(h)
        if c < best_c - 1e-14:
            best_c = c
            best_y = x

    for k, maxit in topk_plan:
        if time.time() > time_end - 8.0:
            break
        try:
            res = minimize(
                topk_obj_sym,
                best_y,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                jac=topk_jac_sym,
                args=(int(n), int(k)),
                options={"maxiter": int(maxit), "ftol": 1e-12, "catol": 1e-10, "disp": 0},
            )
        except Exception:
            continue
        if res is None:
            continue
        x = np.asarray(res.x, dtype=float)
        if x.shape[0] != m or not np.all(np.isfinite(x)):
            continue
        x = project_sym_y(x, n)
        h = make_full_even(x, n)
        c = compute_c5(h)
        if c < best_c - 1e-14:
            best_c = c
            best_y = x

    best_h = make_full_even(best_y, n)
    return best_h, best_c


def interpolate_h(h_src, n_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    if n_src == n_dst:
        return project_box_sum(h_src, n_dst / 2.0)
    xp = (np.arange(n_src) + 0.5) * (2.0 / float(n_src))
    xn = (np.arange(n_dst) + 0.5) * (2.0 / float(n_dst))
    h_new = np.interp(xn, xp, h_src)
    return project_box_sum(h_new, n_dst / 2.0)


def default_initial(n):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = 0.5 + 0.5 * np.cos(np.pi * x)
    return project_box_sum(h, n / 2.0)


def make_cosine_start(n):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = 0.5 + 0.5 * np.cos(np.pi * x)
    return project_box_sum(h, n / 2.0)


def optimize_start(h0, n, time_end, rng, do_full=True, max_final_sweeps=3):
    h = project_box_sum(np.asarray(h0, dtype=float), n / 2.0)
    c = compute_c5(h)

    if time.time() < time_end - 12.0:
        hp, cp = pair_coord_descent(h, n, 1, time_end - 18.0, rng, 24, 14, 8)
        if cp < c:
            h, c = hp, cp

    if do_full and n <= 220 and time.time() < time_end - 30.0:
        hf, cf = slsqp_full_optimize(
            h,
            n,
            full_smooth_plan(n),
            full_topk_plan(n),
            time_end - 12.0,
        )
        if cf < c:
            h, c = hf, cf

    if time.time() < time_end - 25.0:
        y = sym_y_from_h(h, n)
        hs, cs = slsqp_sym_optimize(
            y,
            n,
            sym_smooth_plan(n),
            sym_topk_plan(n),
            time_end - 12.0,
        )
        if cs < c:
            h, c = hs, cs

    if time.time() < time_end - 10.0:
        sweeps = int(max(1, max_final_sweeps if n <= 180 else max_final_sweeps - 1))
        hp, cp = pair_coord_descent(h, n, sweeps, time_end - 3.0, rng, 32, 20, 12)
        if cp < c:
            h, c = hp, cp

    return h, c


def finalize_solution(h0, n, time_end, rng):
    h = project_box_sum(np.asarray(h0, dtype=float), n / 2.0)
    c = compute_c5(h)

    if time.time() < time_end - 8.0:
        hp, cp = pair_coord_descent(h, n, 1, time_end - 12.0, rng, 24, 14, 8)
        if cp < c:
            h, c = hp, cp

    if n <= 200 and time.time() < time_end - 25.0:
        hf, cf = slsqp_full_optimize(
            h,
            n,
            final_full_smooth_plan(n),
            final_full_topk_plan(n),
            time_end - 10.0,
        )
        if cf < c:
            h, c = hf, cf

    if time.time() < time_end - 20.0:
        y = sym_y_from_h(h, n)
        hs, cs = slsqp_sym_optimize(
            y,
            n,
            final_sym_smooth_plan(n),
            final_sym_topk_plan(n),
            time_end - 10.0,
        )
        if cs < c:
            h, c = hs, cs

    if time.time() < time_end - 5.0:
        sweeps = 5 if n <= 180 else 3
        hp, cp = pair_coord_descent(h, n, sweeps, time_end - 2.0, rng, 36, 22, 14)
        if cp < c:
            h, c = hp, cp

    return h, c


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_n = 160
    best_c = float("inf")

    try:
        ph = np.array(initial_h_values, dtype=float)
        if ph.ndim != 1:
            ph = ph.ravel()
        pn = ph.shape[0]
        if pn >= 8 and np.all(np.isfinite(ph)):
            ph = project_box_sum(ph, pn / 2.0)
            best_h = ph
            best_n = pn
            best_c = compute_c5(ph)
    except Exception:
        pass

    if best_h is None:
        best_n = 160
        best_h = default_initial(best_n)
        best_c = compute_c5(best_h)

    if best_n % 2 != 0:
        newn = best_n + 1
        if newn < 16:
            newn = 16
        best_h = interpolate_h(best_h, best_n, newn)
        best_n = newn
        best_c = compute_c5(best_h)

    if best_n < 48:
        best_h = interpolate_h(best_h, best_n, 160)
        best_n = 160
        best_c = compute_c5(best_h)
    elif best_n > 384:
        best_h = interpolate_h(best_h, best_n, 320)
        best_n = 320
        best_c = compute_c5(best_h)

    base_n = best_n
    if base_n % 2 != 0:
        base_n += 1
    if base_n < 48:
        base_n = 160
    if base_n > 384:
        base_n = 320

    res_list = []
    seen = set()
    for v in [
        base_n,
        base_n - 16,
        base_n + 16,
        144,
        160,
        176,
        192,
        208,
        224,
        240,
        256,
        288,
        320,
    ]:
        v = int(v)
        if v % 2 == 0 and 48 <= v <= 384 and v not in seen:
            seen.add(v)
            res_list.append(v)

    hard_end = t0 + budget - 20.0
    search_end = t0 + budget - 80.0
    if search_end < t0 + 30.0:
        search_end = t0 + max(10.0, budget * 0.5)
    if search_end > hard_end:
        search_end = hard_end

    for idx, n in enumerate(res_list):
        remaining = search_end - time.time()
        if remaining < 70.0:
            break

        if idx == 0:
            alloc = min(650.0, max(300.0, remaining * 0.45))
        else:
            alloc = min(260.0, max(80.0, remaining / float(max(1, len(res_list) - idx))))

        if time.time() + alloc > search_end:
            alloc = search_end - time.time()
        if alloc < 50.0:
            break

        res_end = time.time() + alloc

        if n == best_n:
            h0 = best_h.copy()
        else:
            h0 = interpolate_h(best_h, best_n, n)

        h_r, c_r = optimize_start(h0, n, res_end, rng, do_full=True, max_final_sweeps=3)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r
            best_n = n

        if time.time() < res_end - 60.0:
            hp = project_box_sum(h0 + 0.03 * rng.randn(n), n / 2.0)
            h_alt, c_alt = optimize_start(hp, n, res_end - 10.0, rng, do_full=False, max_final_sweeps=1)
            if c_alt < best_c:
                best_c = c_alt
                best_h = h_alt
                best_n = n

        if time.time() < res_end - 45.0:
            hp = make_cosine_start(n)
            h_alt, c_alt = optimize_start(hp, n, res_end - 10.0, rng, do_full=False, max_final_sweeps=1)
            if c_alt < best_c:
                best_c = c_alt
                best_h = h_alt
                best_n = n

    final_end = hard_end - 5.0
    if time.time() < final_end - 20.0:
        h_f, c_f = finalize_solution(best_h, best_n, final_end, rng)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f
            best_n = best_n

    safe_h = best_h.copy()
    safe_c = best_c
    best_h = project_box_sum(best_h, best_n / 2.0)
    final_c = compute_c5(best_h)
    if final_c > safe_c + 1e-12:
        best_h = safe_h
        final_c = safe_c

    try:
        ec = evaluate_erdos_solution(best_h)
        ec = float(ec)
        if np.isfinite(ec) and ec < final_c:
            final_c = ec
    except Exception:
        pass

    return (best_h, float(final_c), int(best_n))
