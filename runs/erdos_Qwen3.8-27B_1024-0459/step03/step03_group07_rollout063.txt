                    va = np.max(intercepts + slopes * m1)
                    if va > f1:
                        f1 = va
                f2 = unaff_max
                v0 = base_0 + m2 * (1.0 - m2)
                if v0 > f2:
                    f2 = v0
                if len(slopes) > 0:
                    va = np.max(intercepts + slopes * m2)
                    if va > f2:
                        f2 = va
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            
            new_x = (lo + hi) / 2.0
            h[j] = new_x
            corr[n - 1] = base_0 + new_x * (1.0 - new_x)
            if len(aff_arr) > 0:
                corr[aff_arr] = intercepts + slopes * new_x
        
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = np.max(corr) * (2.0 / n)
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stagn = 0
        else:
            stagn += 1
        
        if stagn >= restart_every:
            if rng is not None:
                scale = noise_scale * (1.0 + 0.5 * min(stagn / restart_every, 3.0))
                h = project_feasible(h + scale * rng.randn(n), n)
                corr = np.correlate(h, 1.0 - h, mode='full')
            stagn = 0
    
    return best_h, best_c5

def targeted_cd(h, n, time_limit=None, t0=None, top_k=10, sweeps=50):
    """CD that focuses on reducing the top-k correlations."""
    h = project_feasible(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    ncorr = 2 * n - 1
    best_c5 = np.max(corr) * (2.0 / n)
    best_h = h.copy()
    
    for sw in range(sweeps):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        
        # Identify top-k correlations
        top_idx = np.argsort(corr)[-top_k:]
        top_set = set(top_idx.tolist())
        
        for j in range(n):
            old = h[j]
            base_0 = corr[n - 1] - old * (1.0 - old)
            
            aff_fidx_list = []
            a_list = []
            b_list = []
            
            for L in range(1, n):
                fidx = L + n - 1
                av = 0.0
                bv = 0.0
                if L <= j:
                    av = 1.0 - h[j - L]
                if j <= n - 1 - L:
                    bv = h[j + L]
                if av != 0.0 or bv != 0.0:
                    aff_fidx_list.append(fidx)
                    a_list.append(av)
                    b_list.append(bv)
            
            for s in range(1, n):
                fidx = n - 1 - s
                av = 0.0
                bv = 0.0
                if j <= n - 1 - s:
                    av = 1.0 - h[j + s]
                if j >= s:
                    bv = h[j - s]
                if av != 0.0 or bv != 0.0:
                    aff_fidx_list.append(fidx)
                    a_list.append(av)
                    b_list.append(bv)
            
            if len(aff_fidx_list) > 0:
                aff_arr = np.array(aff_fidx_list, dtype=np.int64)
                a_arr = np.array(a_list, dtype=np.float64)
                b_arr = np.array(b_list, dtype=np.float64)
                slopes = a_arr - b_arr
                intercepts = corr[aff_arr].copy() - slopes * old
                
                # Weighted objective: emphasize top-k correlations
                weights = np.ones(len(aff_arr))
                for ii, fi in enumerate(aff_arr):
                    if fi in top_set:
                        weights[ii] = 2.0
            else:
                aff_arr = np.array([], dtype=np.int64)
                slopes = np.array([])
                intercepts = np.array([])
                weights = np.array([])
            
            mask = np.ones(ncorr, dtype=bool)
            mask[n - 1] = False
            if len(aff_arr) > 0:
                mask[aff_arr] = False
            unaff_max = np.max(corr[mask]) if mask.any() else -1e30
            
            lo = 0.0
            hi = 1.0
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = unaff_max
                v0 = base_0 + m1 * (1.0 - m1)
                if v0 > f1:
                    f1 = v0
                if len(slopes) > 0:
                    va = np.max(intercepts + slopes * m1)
                    if va > f1:
                        f1 = va
                f2 = unaff_max
                v0 = base_0 + m2 * (1.0 - m2)
                if v0 > f2:
                    f2 = v0
                if len(slopes) > 0:
                    va = np.max(intercepts + slopes * m2)
                    if va > f2:
                        f2 = va
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            
            new_x = (lo + hi) / 2.0
            h[j] = new_x
            corr[n - 1] = base_0 + new_x * (1.0 - new_x)
            if len(aff_arr) > 0:
                corr[aff_arr] = intercepts + slopes * new_x
        
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = np.max(corr) * (2.0 / n)
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
    
    return best_h, best_c5

def subgradient_step(h, n, top_k=5, lr=0.01, time_limit=None, t0=None, iters=2000, rng=None):
    """Subgradient method targeting top-k correlations."""
    h = project_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    lr_decay = 0.9995
    
    for it in range(iters):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        
        grad = np.zeros(n)
        for lidx in top_idx:
            lag = lidx - (n - 1)
            w = 1.0 / top_k
            if lag >= 0:
                # c[lag] = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
                # d/dh[j]: (1-h[j-lag]) if j>=lag, and -h[j+lag] if j+lag<n
                valid1 = np.minimum(lag, n)
                grad[:n - lag] += w * 0  # will compute properly
                # For h[j] as h[i+lag]: i=j-lag, need j>=lag and j-lag<=n-1-lag i.e. j<=n-1
                # Contribution: (1-h[j-lag]) for j in [lag, n-1]
                if lag < n:
                    grad[lag:] += w * (1.0 - h[:n - lag])
                # For h[j] in (1-h[i]): i=j, need j<=n-1-lag
                # Contribution: -h[j+lag] for j in [0, n-1-lag]
                if lag < n:
                    grad[:n - lag] += w * (-h[lag:])
            else:
                s = -lag
                # c[-s] = sum_{i=s}^{n-1} h[i-s]*(1-h[i])
                # For h[j] as h[i-s]: i=j+s, need j+s<=n-1 i.e. j<=n-1-s
                # Contribution: (1-h[j+s]) for j in [0, n-1-s]
                if s < n:
                    grad[:n - s] += w * (1.0 - h[s:])
                # For h[j] in (1-h[i]): i=j, need j>=s
                # Contribution: -h[j-s] for j in [s, n-1]
                if s < n:
                    grad[s:] += w * (-h[:n - s])
        
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            lr *= lr_decay
            if lr < 1e-8 and rng is not None:
                h = project_feasible(h + 0.02 * rng.randn(n), n)
                lr = 0.01
        
        if it % 200 == 0:
            lr = max(lr, 0.005)
    
    return best_h, best_c5

def interpolate_h(h_src, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_new = np.interp(xn, xp, h_src)
    return project_feasible(h_new, n_dst)

def binarize_soft(h, n, tau=0.1):
    """Push h toward 0 or 1, with soft transition near 0.5."""
    h = np.array(h, dtype=float).copy()
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    # Blend: keep some continuity
    h_new = h * (1.0 - tau) + h_bin * tau
    return project_feasible(h_new, n)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return max(0.0, budget_s - (time.time() - t0) - 5.0)
    
    # Get parent
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn > 10:
            best_h = ph.copy()
            best_c5 = compute_c5(ph)
            best_n = pn
    except Exception:
        pass
    
    if best_h is None:
        best_n = 160
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)
    
    # Phase 1: Intensive optimization at parent resolution
    if time_left() > 200:
        h_opt, c5_opt = fast_cd(best_h, best_n, sweeps=10000,
                                time_limit=time_left() * 0.5, t0=t0,
                                rng=rng, restart_every=50, noise_scale=0.02)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    if time_left() > 100:
        h_opt, c5_opt = targeted_cd(best_h, best_n, time_limit=time_left() * 0.3, t0=t0, top_k=8, sweeps=200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 2: Try other resolutions
    resolutions_to_try = []
    for nn in [128, 144, 200, 240, 256, 320]:
        if nn != best_n and 64 <= nn <= 384:
            resolutions_to_try.append(nn)
    
    res_time_each = min(80.0, max(30.0, time_left() / max(1, len(resolutions_to_try)) * 0.7))
    
    for n_try in resolutions_to_try:
        if time_left() < res_time_each + 20:
            break
        
        # Interpolate current best to this resolution
        h_init = interpolate_h(best_h, best_n, n_try)
        
        # Try a few inits
        inits = [h_init]
        h_noisy = project_feasible(h_init + 0.03 * rng.randn(n_try), n_try)
        inits.append(h_noisy)
        h_bin = binarize_soft(h_init, n_try, tau=0.3)
        inits.append(h_bin)
        
        time_per_init = max(10.0, (res_time_each - 10) / 3.0)
        
        for h0 in inits:
            if time_left() < 20:
                break
            h_opt, c5_opt = fast_cd(h0, n_try, sweeps=5000,
                                    time_limit=time_per_init, t0=t0,
                                    rng=rng, restart_every=40, noise_scale=0.02)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_try
    
    # Phase 3: Subgradient refinement on best
    if time_left() > 60:
        h_opt, c5_opt = subgradient_step(best_h, best_n, top_k=5, lr=0.005,
                                         time_limit=time_left() * 0.3, t0=t0,
                                         iters=5000, rng=rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 4: Final intensive polish
    if time_left() > 30:
        h_opt, c5_opt = fast_cd(best_h, best_n, sweeps=20000,
                                time_limit=time_left() * 0.6, t0=t0,
                                rng=rng, restart_every=80, noise_scale=0.01)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    if time_left() > 20:
        h_opt, c5_opt = targeted_cd(best_h, best_n, time_limit=time_left() * 0.4, t0=t0, top_k=5, sweeps=300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
