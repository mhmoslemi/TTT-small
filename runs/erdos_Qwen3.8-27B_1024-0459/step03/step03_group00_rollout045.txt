        # Try multiple step sizes
        improved = False
        for step in [step_t, step_t * 0.5, step_t * 2, step_t * 0.25]:
            h_trial = h_cur - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-10:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h_cur = h_trial.copy()
                improved = True
                step_t = min(step_t * 1.2, 0.2)
                break
        if not improved:
            step_t = max(step_t * 0.7, 0.005)
        
        # Occasionally do SLSQP polish
        if it % 30 == 29 and budget_fn is not None and budget_fn() > 20:
            h_sl = slsqp_optimize(h_cur, n, 200, 100)
            c5_sl = compute_c5(h_sl)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()
                h_cur = h_sl.copy()
    
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("No parent. Block: n={}, C5={:.8f}".format(parent_n, best_c5))

    def br():
        return remaining_time()

    # Phase 1: Try parent at higher resolutions with quick polish
    print("\n=== Phase 1: Multi-resolution quick optimization ===")
    n_candidates = sorted(set([best_n, 256, 288, 320, 384, 448, 512]))
    n_candidates = [nn for nn in n_candidates if 128 <= nn <= 512]
    
    candidates = []
    for n in n_candidates:
        if br() < 40.0:
            break
        _G_N = n
        _G_TGT = n / 2.0
        _G_DX = 2.0 / n
        
        h_interp = project_feasible(interp_to(best_h, best_n, n), n)
        c5_interp = compute_c5(h_interp)
        
        # Quick SLSQP polish
        for beta in [50, 200]:
            if br() < 10.0:
                break
            h_new = slsqp_optimize(h_interp, n, beta, 200)
            c5_new = compute_c5(h_new)
            if c5_new < c5_interp:
                h_interp = h_new
                c5_interp = c5_new
        
        candidates.append((c5_interp, n, h_interp.copy()))
        print("  n={}: C5={:.8f}".format(n, c5_interp))
    
    if candidates:
        candidates.sort(key=lambda x: x[0])
        c0, n0, h0 = candidates[0]
        if c0 < best_c5:
            best_c5 = c0
            best_h = h0
            best_n = n0

    # Phase 2: Targeted SLSQP at best resolution with increasing beta
    print("\n=== Phase 2: Targeted multi-beta SLSQP ===")
    n_work = best_n
    _G_N = n_work
    _G_TGT = n_work / 2.0
    _G_DX = 2.0 / n_work
    
    h_work = best_h.copy()
    c5_work = best_c5
    
    # Also prepare alternatives
    inits = [h_work]
    
    # Thue-Morse at working resolution
    if n_work % 2 == 0:
        h_tm = make_thue_morse(n_work)
        h_tm = project_feasible(h_tm, n_work)
        inits.append(h_tm)
    
    # Golden ratio
    h_gr = make_golden_ratio(n_work)
    h_gr = project_feasible(h_gr, n_work)
    inits.append(h_gr)
    
    # For each init, do escalating beta optimization
    for h_init in inits:
        if br() < 30.0:
            break
        h_cur = h_init.copy()
        c5_cur = compute_c5(h_cur)
        
        for beta in [30, 80, 150, 300, 600, 1200, 2500, 5000]:
            if br() < 10.0:
                break
            h_new = slsqp_optimize(h_cur, n_work, beta, 400)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-12:
                c5_cur = c5_new
                h_cur = h_new
        
        # Raw objective polish
        if br() > 15.0:
            h_new = slsqp_raw(h_cur, n_work, 300)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur:
                c5_cur = c5_new
                h_cur = h_new
        
        if c5_cur < best_c5 - 1e-10:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n_work
            print("  Improved via {}: C5={:.8f}".format(inits.index(h_init), best_c5))
    
    # Phase 3: Subgradient min-max
    print("\n=== Phase 3: Subgradient min-max ===")
    if br() > 60.0:
        h_sg, c5_sg = subgradient_minmax(best_h, best_n, max_iters=300, budget_fn=br, rng=rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg
            best_n = len(h_sg)
            print("  After subgradient: C5={:.8f}".format(best_c5))
    
    # Phase 4: Binary local search (simulated annealing)
    print("\n=== Phase 4: Binary local search ===")
    if br() > 40.0:
        # Round to binary
        h_bin_init = best_h.copy()
        h_bin_init[h_bin_init > 0.5] = 1.0
        h_bin_init[h_bin_init <= 0.5] = 0.0
        # Fix sum
        num_ones = int(np.sum(h_bin_init))
        target_ones = best_n // 2
        if num_ones > target_ones:
            ones_idx = np.where(h_bin_init == 1.0)[0]
            remove = rng.choice(ones_idx, size=num_ones - target_ones, replace=False)
            h_bin_init[remove] = 0.0
        elif num_ones < target_ones:
            zeros_idx = np.where(h_bin_init == 0.0)[0]
            add = rng.choice(zeros_idx, size=target_ones - num_ones, replace=False)
            h_bin_init[add] = 1.0
        
        c5_bin_init = compute_c5(h_bin_init)
        print("  Binary init: C5={:.8f}".format(c5_bin_init))
        
        h_bs, c5_bs = binary_local_search(h_bin_init, best_n, max_sweeps=300, budget_fn=br, rng=rng)
        if c5_bs < best_c5:
            best_c5 = c5_bs
            best_h = h_bs.copy()
            best_n = len(h_bs)
            print("  After binary search: C5={:.8f}".format(best_c5))
        else:
            print("  Binary search didn't improve (got {:.8f})".format(c5_bs))
    
    # Phase 5: Continuous polish after binary search
    print("\n=== Phase 5: Continuous polish ===")
    if br() > 20.0:
        h_polish = best_h.copy()
        for beta in [1000, 3000, 8000]:
            if br() < 8.0:
                break
            h_new = slsqp_optimize(h_polish, best_n, beta, 500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new
                h_polish = h_new
        
        # Raw polish
        if br() > 10.0:
            h_new = slsqp_raw(h_polish, best_n, 400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
    
    # Phase 6: Try other resolutions from the best solution
    print("\n=== Phase 6: Resolution exploration ===")
    for n_try in [best_n + 32, best_n + 64, best_n - 32]:
        if br() < 30.0:
            break
        if n_try < 128 or n_try > 640 or n_try % 2 != 0:
            continue
        _G_N = n_try
        _G_TGT = n_try / 2.0
        _G_DX = 2.0 / n_try
        
        h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        c5_interp = compute_c5(h_interp)
        
        for beta in [200, 800, 2000]:
            if br() < 10.0:
                break
            h_new = slsqp_optimize(h_interp, n_try, beta, 300)
            c5_new = compute_c5(h_new)
            if c5_new < c5_interp:
                h_interp = h_new
                c5_interp = c5_new
        
        if c5_interp < best_c5 - 1e-10:
            best_c5 = c5_interp
            best_h = h_interp
            best_n = n_try
            print("  Resolved at n={}: C5={:.8f}".format(n_try, best_c5))
    
    # Phase 7: Final targeted worst-lag optimization
    print("\n=== Phase 7: Final worst-lag target ===")
    if br() > 20.0:
        corr = compute_corr(best_h)
        top_lags = np.argsort(corr)[-5:]
        print("  Worst lags: {}, values: {}".format(
            top_lags.tolist(), corr[top_lags].tolist()))
        
        # Focus on minimizing the worst lag specifically
        # Use SLSQP with very high beta to approximate the max
        for beta in [5000, 15000]:
            if br() < 5.0:
                break
            _G_BETA = float(beta)
            h_new = slsqp_optimize(best_h, best_n, beta, 600)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new
    
    # Phase 8: Tiny perturbations around best solution
    if br() > 15.0:
        print("\n=== Phase 8: Perturbation search ===")
        for _ in range(5):
            if br() < 5.0:
                break
            h_pert = best_h + rng.randn(best_n) * 0.02
            h_pert = project_feasible(h_pert, best_n)
            h_pert = slsqp_optimize(h_pert, best_n, 500, 200)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5 - 1e-12:
                best_c5 = c5_pert
                best_h = h_pert
        
        # Also try slight binary nudges
        for _ in range(10):
            if br() < 3.0:
                break
            h_try = best_h.copy()
            # Nudge a few values toward 0 or 1
            idxs = rng.choice(best_n, size=min(5, best_n), replace=False)
            for idx in idxs:
                if h_try[idx] > 0.5:
                    h_try[idx] = np.clip(h_try[idx] + 0.1, 0, 1)
                else:
                    h_try[idx] = np.clip(h_try[idx] - 0.1, 0, 1)
            h_try = project_feasible(h_try, best_n)
            h_try = slsqp_optimize(h_try, best_n, 500, 150)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
