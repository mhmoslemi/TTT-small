def fix_sum(x,n):
    x=np.asarray(x,dtype=float).copy()
    x=np.clip(x,0.0,1.0)
    target=0.5*float(n)
    for _ in range(30):
       s=np.sum(x)
       diff=target-s
       if abs(diff) < 1e-14:
           break
       if diff > 0:
          order=np.argsort(x)
          rem=diff
          for idx in order:
             space=1.0-x[idx]
             if space <= 1e-15: continue
             take=min(space, rem)
             x[idx] += take
             rem -= take
             if rem <= 1e-15: break
       else:
          order=np.argsort(-x)
          rem=-diff
          for idx in order:
             space=x[idx]
             if space <= 1e-15: continue
             take=min(space, rem)
             x[idx] -= take
             rem -= take
             if rem <= 1e-15: break
       x=np.clip(x,0.0,1.0)
    # If still residual due all saturated improbable, spread over all
    diff=target-np.sum(x)
    if abs(diff) > 1e-13:
       if diff > 0:
          x += diff/float(n)
       else:
          x += diff/float(n)
       x=np.clip(x,0.0,1.0)
    return x

def eval_c5(h):
    n = len(h)
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return (float(np.max(corr)), corr)


def fix_sum(x, n):
    x = np.asarray(x, dtype=float).copy()
    x = np.clip(x, 0.0, 1.0)
    target = 0.5 * float(n)
    for _ in range(40):
        s = float(np.sum(x))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            order = np.argsort(x)
            rem = diff
            for idx in order:
                space = 1.0 - x[idx]
                if space <= 1e-15:
                    continue
                take = min(space, rem)
                x[idx] += take
                rem -= take
                if rem <= 1e-15:
                    break
        else:
            order = np.argsort(-x)
            rem = -diff
            for idx in order:
                space = x[idx]
                if space <= 1e-15:
                    continue
                take = min(space, rem)
                x[idx] -= take
                rem -= take
                if rem <= 1e-15:
                    break
        x = np.clip(x, 0.0, 1.0)
    diff = target - float(np.sum(x))

        x += diff / float(n)

    return x


def project_box_sum(y, n):
    y = np.asarray(y, dtype=float).copy()
    if y.size != n:
        z = np.full(int(n), 0.5, dtype=float)
        m = min(y.size, int(n))
        z[:m] = y[:m]
        y = z
    y = np.clip(y, 0.0, 1.0)

        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_value(v, beta):
    m = float(np.max(v))
    beta = float(beta)
    if beta < 1e-8:
        return m
    e = np.exp(beta * (v - m))
    s = float(np.sum(e))
    if s <= 0.0:
        return m
    return m + math.log(s) / beta


def smooth_weights(v, beta):
    m = float(np.max(v))
    beta = float(beta)
    w = np.zeros(int(len(v)), dtype=float)
    if beta < 1e-8:
        idx = np.where(v >= m - 1e-12)[0]
        if len(idx) == 0:
            idx = np.array([int(np.argmax(v))], dtype=int)
        w[idx] = 1.0 / float(len(idx))
    else:
        e = np.exp(beta * (v - m))
        s = float(np.sum(e))
        if s > 0.0:
            w = e / s
    return w


def grad_from_weights(h, w):
    n = len(h)
    y = np.asarray(w, dtype=float)
    if y.size != 2 * n - 1:
        z = np.zeros(2 * n - 1, dtype=float)
        m = min(y.size, 2 * n - 1)
        z[:m] = y[:m]
        y = z
    g1 = np.convolve(1.0 - h, y, mode='full')
    g2 = np.convolve(h, y[::-1], mode='full')
    return g1[n - 1:2 * n - 1] - g2[n - 1:2 * n - 1]


def obj_pen_smooth(x):
    n = _G_N
    corr = np.correlate(x, 1.0 - x, mode='full') * _G_DX
    if _G_ACT is None or len(_G_ACT) == 2 * n - 1:
        va = corr
        val = smooth_value(va, _G_BETA)
    else:
        va = corr[_G_ACT]
        val = smooth_value(va, _G_BETA)
    diff = float(np.sum(x) - 0.5 * n)
    val += 0.5 * _G_LAM * diff * diff
    return float(val)


def jac_pen_smooth(x):
    n = _G_N
    corr = np.correlate(x, 1.0 - x, mode='full') * _G_DX
    if _G_ACT is None or len(_G_ACT) == 2 * n - 1:
        wf = smooth_weights(corr, _G_BETA) * _G_DX
    else:
        wf = np.zeros(2 * n - 1, dtype=float)
        wf[_G_ACT] = smooth_weights(corr[_G_ACT], _G_BETA)
        wf *= _G_DX
    g = grad_from_weights(x, wf)
    diff = float(np.sum(x) - 0.5 * n)
    if _G_LAM != 0.0:
        g = g + (_G_LAM * diff)
    return g


def obj_eq_smooth(x):
    n = _G_N
    corr = np.correlate(x, 1.0 - x, mode='full') * _G_DX
    if _G_ACT is None or len(_G_ACT) == 2 * n - 1:
        va = corr
    else:
        va = corr[_G_ACT]
    return float(smooth_value(va, _G_BETA))


def jac_eq_smooth(x):
    n = _G_N
    corr = np.correlate(x, 1.0 - x, mode='full') * _G_DX
    if _G_ACT is None or len(_G_ACT) == 2 * n - 1:
        wf = smooth_weights(corr, _G_BETA) * _G_DX
    else:
        wf = np.zeros(2 * n - 1, dtype=float)
        wf[_G_ACT] = smooth_weights(corr[_G_ACT], _G_BETA)
        wf *= _G_DX
    return grad_from_weights(x, wf)


def eq_sum(x):
    return float(np.sum(x) - 0.5 * _G_N)


def eq_sum_jac(x):
    return np.ones(_G_N, dtype=float)


def grad_single_d(h, d):
    n = len(h)
    g = np.zeros(n, dtype=float)
    d = int(d)
    if d == 0:
        return 1.0 - 2.0 * h
    if d > 0:
        g[d:] += 1.0 - h[:n - d]
        g[:n - d] -= h[d:]
    else:
        e = -d
        g[:n - e] += 1.0 - h[e:]
        g[e:] -= h[:n - e]
    return g


def select_active(corr, M, gap):
    n = len(corr)
    M = int(M)
    if M < 1:
        M = 1
    if M > n:
        M = n
    mx = float(np.max(corr))
    near = np.where(corr >= mx - float(gap))[0].astype(int)
    top = np.argsort(-corr)[:M].astype(int)
    idx = np.unique(np.concatenate((top, near)))
    order = np.argsort(-corr[idx])
    idx = idx[order[:M]]
    return idx.astype(int)


def solve_lbfgs_penalty(h0, n, beta, lam, act, bounds, maxiter):

    _G_LAM = float(lam)
    _G_ACT = act
    try:

    except Exception:

    x = res.x

def solve_slsqp_smooth(h0, n, beta, act, bounds, maxiter):
    global _G_N, _G_DX, _G_BETA, _G_ACT
    _G_N = int(n)
    _G_DX = 2.0 / float(n)
    _G_BETA = float(beta)
    _G_ACT = act
    try:
        res = minimize(

            h0,
            method='SLSQP',
            jac=jac_eq_smooth,
            bounds=bounds,

    except Exception:
        return None
    x = res.x
    if x is not None and np.all(np.isfinite(x)):

def mm_obj(y):
    return float(y[-1])


def mm_obj_jac(y):
    g = np.zeros(len(y), dtype=float)
    g[-1] = 1.0
    return g


def mm_incons(y):
    h = y[:-1]
    t = float(y[-1])
    corr = np.correlate(h, 1.0 - h, mode='full') * _G_DX
    return corr[_G_ACT] - t


def mm_con_jac(y):
    h = y[:-1]
    n = _G_N
    act = _G_ACT
    m = len(act)
    J = np.empty((m, n + 1), dtype=float)
    for r in range(m):
        p = int(act[r])
        d = p - (n - 1)
        J[r, :n] = grad_single_d(h, d) * _G_DX
        J[r, n] = -1.0
    return J


def mm_eqfun(y):
    return float(np.sum(y[:-1]) - 0.5 * _G_N)


def mm_eqjac(y):
    g = np.zeros(len(y), dtype=float)
    g[:-1] = 1.0
    return g


def mm_eqfun_arr(y):
    return np.array([float(np.sum(y[:-1]) - 0.5 * _G_N)], dtype=float)


def mm_eqjac_arr(y):
    g = np.zeros(len(y), dtype=float)
    g[:-1] = 1.0
    return g.reshape(1, -1)


def solve_minmax_slsqp(h0, n, act, bounds, maxiter):

    _G_N = int(n)
    _G_DX = 2.0 / float(n)
    _G_ACT = act
    corr = np.correlate(h0, 1.0 - h0, mode='full') * _G_DX
    t0 = float(np.max(corr)) + 1e-6
    if t0 < 0.0:
        t0 = 0.0
    if t0 > 1.0:
        t0 = 1.0
    y0 = np.empty(n + 1, dtype=float)
    y0[:n] = h0
    y0[n] = t0
    pb = list(bounds) + [(0.0, 1.0)]
    try:
        res = minimize(
            mm_obj,
            y0,
            method='SLSQP',
            jac=mm_obj_jac,
            bounds=pb,
            constraints=[
                {'type': 'ineq', 'fun': mm_incons, 'jac': mm_con_jac},
                {'type': 'eq', 'fun': mm_eqfun, 'jac': mm_eqjac}
            ],
            options={'maxiter': int(maxiter), 'ftol': 1e-13, 'iprint': 0}
        )
    except Exception:
        return None
    x = res.x
    if x is not None and np.all(np.isfinite(x)):
        return project_box_sum(x[:-1], n)
    return None


def solve_minmax_trust(h0, n, act, bounds, maxiter):
    if NonlinearConstraint is None or trust_constr is None:
        return None
    global _G_N, _G_DX, _G_ACT
    _G_N = int(n)
    _G_DX = 2.0 / float(n)
    _G_ACT = act
    corr = np.correlate(h0, 1.0 - h0, mode='full') * _G_DX
    t0 = float(np.max(corr)) + 1e-5

def linearize_step(h0, n, act, box):
    if linprog is None:
        return None
    n = int(n)
    h0 = np.asarray(h0, dtype=float)
    corr = np.correlate(h0, 1.0 - h0, mode='full') * (2.0 / float(n))
    v = corr[act]
    m = len(act)
    if m <= 0:
        return None
    A = np.empty((m, n + 1), dtype=float)
    b = np.empty(m, dtype=float)
    for i in range(m):
        p = int(act[i])
        d = p - (n - 1)
        g = grad_single_d(h0, d) * (2.0 / float(n))
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = float(np.dot(g, h0)) - float(v[i])

    if box is None:
        hbounds = [(0.0, 1.0)] * n
    else:
        bx = float(box)
        hbounds = []
        for val in h0:
            lo = float(val) - bx
            hi = float(val) + bx
            if lo < 0.0:
                lo = 0.0
            if hi > 1.0:
                hi = 1.0
            hbounds.append((lo, hi))
    c = np.zeros(n + 1, dtype=float)
    c[n] = 1.0
    Aeq = np.ones((1, n), dtype=float)
    beq = np.array([0.5 * float(n)], dtype=float)
    bounds = hbounds + [(0.0, 1.0)]
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq,
                      bounds=bounds, method='highs')
    except Exception:
        try:
            res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=beq,
                          bounds=bounds)
        except Exception:
            return None
    if res.x is not None and np.all(np.isfinite(res.x)):
        x = np.clip(res.x[:n], 0.0, 1.0)
        return project_box_sum(x, n)
    return None


def lp_cycles(h, n, best_c5, t0, budget, rounds, max_no_improve):
    if linprog is None:
        return h, best_c5
    best_h = np.asarray(h, dtype=float).copy()
    best = float(best_c5)
    no = 0
    for _ in range(int(rounds)):
        if time_left(t0, budget) < 25.0:
            break
        if n <= 200:
            sizes = (15, 30, 50, 80)
        else:
            sizes = (15, 30, 50)
        improved = False
        c, corr = eval_c5(best_h)
        for M in sizes:
            if time_left(t0, budget) < 12.0:
                break
            if M <= 30:
                gap = 1e-5
            elif M <= 60:
                gap = 5e-5
            else:
                gap = 2e-4
            act = select_active(corr, M, gap)
            for box in (0.6, 0.3, 0.12):
                if time_left(t0, budget) < 5.0:
                    break
                x = linearize_step(best_h, n, act, box)
                if x is None:
                    continue
                for a in (1.0, 0.5, 0.25, 0.125):
                    cand = best_h + a * (x - best_h)
                    cand = project_box_sum(cand, n)
                    c2, _ = eval_c5(cand)
                    if c2 < best - 1e-13:
                        best = c2
                        best_h = cand
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if improved:
            no = 0
        else:
            no += 1
            if no >= int(max_no_improve):
                break
    return best_h, best


def subgradient_refine(h, n, best_c5, t0, budget, seed):
    best_h = np.asarray(h, dtype=float).copy()
    best = float(best_c5)
    no = 0
    rng = np.random.RandomState(seed + 17)
    for it in range(500):
        if time_left(t0, budget) < 3.0:
            break
        c, corr = eval_c5(best_h)
        M = 8 + int(rng.randint(0, 12))
        act = select_active(corr, M, 1e-5)
        dx = 2.0 / float(n)
        wf = np.zeros(2 * n - 1, dtype=float)
        wf[act] = dx / float(len(act))
        g = grad_from_weights(best_h, wf)
        g -= np.mean(g)
        gn = float(np.linalg.norm(g))
        if gn < 1e-12:
            break
        d = -g / gn
        improved = False
        for step in (0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005):
            cand = project_box_sum(best_h + step * d, n)
            c2, _ = eval_c5(cand)
            if c2 < best - 1e-13:
                best = c2
                best_h = cand
                improved = True
                break
        if improved:
            no = 0
        else:
            no += 1
            if no > 35:
                break
    return best_h, best


def local_pair_moves(h, n, best_c5, t0, budget, seed):
    best_h = np.asarray(h, dtype=float).copy()
    best = float(best_c5)
    rng = np.random.RandomState(seed + 101)
    high_idx = np.nonzero(best_h > 0.35)[0]
    low_idx = np.nonzero(best_h < 0.65)[0]
    deltas = (0.00025, 0.0005, 0.001, 0.002, 0.004, 0.008, 0.015)
    no_improve = 0
    for it in range(12000):
        if time_left(t0, budget) < 2.0:
            break
        if len(high_idx) > 0 and len(low_idx) > 0:
            i = int(high_idx[int(rng.randint(len(high_idx)))])
            j = int(low_idx[int(rng.randint(len(low_idx)))])
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        if i == j:
            continue
        improved = False
        for d in deltas:
            dd = float(d)
            if best_h[i] >= dd and best_h[j] <= 1.0 - dd:
                oi = best_h[i]
                oj = best_h[j]
                best_h[i] = oi - dd
                best_h[j] = oj + dd
                c, _ = eval_c5(best_h)
                if c < best - 1e-13:
                    best = c
                    best_h = best_h.copy()
                    improved = True
                    break
                best_h[i] = oi
                best_h[j] = oj
            if best_h[j] >= dd and best_h[i] <= 1.0 - dd:
                oj = best_h[j]
                oi = best_h[i]
                best_h[j] = oj - dd
                best_h[i] = oi + dd
                c, _ = eval_c5(best_h)
                if c < best - 1e-13:
                    best = c
                    best_h = best_h.copy()
                    improved = True
                    break
                best_h[j] = oj
                best_h[i] = oi
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3500:
                break
    return best_h, best


def polish_all(h0, n, t0, budget, seed):
    h0 = project_box_sum(np.asarray(h0, dtype=float), n)
    best_h = h0.copy()
    best_c5, _ = eval_c5(h0)

    set_global_n(n)
    full_act = np.arange(2 * n - 1)
    bounds = [(0.0, 1.0)] * int(n)

    if time_left(t0, budget) > 30.0:
        for beta in (30.0, 80.0, 200.0):
            if time_left(t0, budget) < 12.0:
                break
            x = solve_lbfgs_penalty(best_h, n, beta, 100.0, full_act, bounds, 120)
            best_h, best_c5, _ = consider_candidate(x, n, best_h, best_c5)

    best_h, best_c5 = lp_cycles(best_h, n, best_c5, t0, budget, 30, 15)

    if time_left(t0, budget) > 40.0:
        for beta in (100.0, 300.0):
            if time_left(t0, budget) < 15.0:
                break
            x = solve_slsqp_smooth(best_h, n, beta, full_act, bounds, 120)
            best_h, best_c5, _ = consider_candidate(x, n, best_h, best_c5)
        if time_left(t0, budget) > 25.0:
            c, corr = eval_c5(best_h)
            for M in (25, 45):
                if time_left(t0, budget) < 15.0:
                    break
                act = select_active(corr, M, 1e-5)
                x = solve_slsqp_smooth(best_h, n, 400.0, act, bounds, 120)
                best_h, best_c5, _ = consider_candidate(x, n, best_h, best_c5)

    best_h, best_c5 = lp_cycles(best_h, n, best_c5, t0, budget, 25, 12)

    if time_left(t0, budget) > 40.0:
        mm_no = 0
        trust_used = False
        for M in (12, 18, 25, 35, 50, 70, 95):
            if time_left(t0, budget) < 30.0:
                break
            if mm_no >= 4:
                break
            c, corr = eval_c5(best_h)
            if M <= 30:
                gap = 1e-5
            elif M <= 60:
                gap = 2e-4
            else:
                gap = 4e-4
            act = select_active(corr, M, gap)
            x = solve_minmax_slsqp(best_h, n, act, bounds, 140)
            improved = False
            best_h, best_c5, improved = consider_candidate(x, n, best_h, best_c5)
            if not improved and not trust_used and time_left(t0, budget) > 70.0:
                x = solve_minmax_trust(best_h, n, act, bounds, 60)
                best_h, best_c5, improved = consider_candidate(x, n, best_h, best_c5)
                trust_used = True
            if improved:
                mm_no = 0
                best_h, best_c5 = lp_cycles(best_h, n, best_c5, t0, budget, 8, 5)
            else:
                mm_no += 1

    best_h, best_c5 = lp_cycles(best_h, n, best_c5, t0, budget, 18, 10)

    if time_left(t0, budget) > 20.0:
        best_h, best_c5 = subgradient_refine(best_h, n, best_c5, t0, budget, seed)

    if time_left(t0, budget) > 20.0:
        best_h, best_c5 = local_pair_moves(best_h, n, best_c5, t0, budget, seed)

    best_h = project_box_sum(best_h, n)
    best_c5, _ = eval_c5(best_h)
    return best_h, float(best_c5)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    rng = np.random.RandomState(seed)

    parent = None
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
    except Exception:
        parent = None

    if parent is None or len(parent) < 16:
        n0 = 160
        h0 = make_sine_init(n0, rng)
    else:
        n0 = int(len(parent))
        h0 = project_box_sum(parent, n0)

    best_h = h0.copy()
    best_c5, _ = eval_c5(best_h)

    mres = candidate_n_list(len(best_h))
    for nt in mres:
        if time_left(t0, budget) < 8.0:
            break
        if nt == len(best_h):
            continue
        ht = interp_h(best_h, len(best_h), nt)
        ht = project_box_sum(ht, nt)
        ct, _ = eval_c5(ht)
        if ct < best_c5 - 1e-12:
            best_c5 = ct
            best_h = ht

    rem = time_left(t0, budget)
    if rem > 40.0:
        alloc = rem - 30.0 if rem > 80.0 else 0.5 * rem
        local_budget = make_local_budget(t0, budget, alloc)
        h2, c2 = polish_all(best_h, len(best_h), t0, local_budget, seed)
        best_h, best_c5, _ = consider_candidate(h2, len(h2), best_h, best_c5)

    rem = time_left(t0, budget)
    if rem > 80.0:
        current_n = len(best_h)
        alt = []
        for dd in (-24, -16, -10, -6, 6, 10, 16, 24, 36, 56):
            _add_unique(alt, current_n + dd)
        for n in (128, 144, 160, 176, 180, 192, 200, 224, 240, 256, 280, 300, 320):
            _add_unique(alt, n)
        keyed = []
        for n in alt:
            keyed.append((abs(n - current_n), n))
        keyed.sort()
        alt = [n for _, n in keyed]
        for nt in alt:
            rem = time_left(t0, budget)
            if rem < 70.0:
                break
            if nt == len(best_h):
                continue
            ht = interp_h(best_h, len(best_h), nt)
            ht = project_box_sum(ht, nt)
            ct, _ = eval_c5(ht)
            best_h, best_c5, _ = consider_candidate(ht, nt, best_h, best_c5)
            if ct <= best_c5 + 0.003 and rem > 120.0:
                alloc = min(220.0, rem - 35.0)
                local_budget = make_local_budget(t0, budget, alloc)
                h2, c2 = polish_all(ht, nt, t0, local_budget, seed + 997 + (nt % 100))
                best_h, best_c5, _ = consider_candidate(h2, nt, best_h, best_c5)

    rem = time_left(t0, budget)
    if rem > 10.0:
        mres2 = candidate_n_list(len(best_h))
        for nt in mres2[:18]:
            if time_left(t0, budget) < 4.0:
                break
            if nt == len(best_h):
                continue
            ht = interp_h(best_h, len(best_h), nt)
            ht = project_box_sum(ht, nt)
            best_h, best_c5, _ = consider_candidate(ht, nt, best_h, best_c5)

    if time_left(t0, budget) > 30.0:
        best_h, best_c5 = local_pair_moves(best_h, len(best_h), best_c5, t0, budget, seed + 2000)

    best_h = project_box_sum(best_h, len(best_h))
    final_c5, _ = eval_c5(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if isinstance(ec5, tuple) or isinstance(ec5, list):
            ec5 = ec5[0]
        ec5 = float(ec5)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    return (best_h, float(final_c5), int(len(best_h)))
