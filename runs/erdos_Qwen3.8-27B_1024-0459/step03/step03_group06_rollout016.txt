    top_idx = np.argsort(corr)[-top_k:]

def binary_multi_swap(h, n, num_swaps, rng):
    """Perform multi-swap: flip num_swaps 1s to 0s and num_swaps 0s to 1s."""
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]

    add = rng.choice(len(zeros), num_swaps, replace=False)
    h[ones[remove]] = 0.0
    h[zeros[add]] = 1.0
    return h
