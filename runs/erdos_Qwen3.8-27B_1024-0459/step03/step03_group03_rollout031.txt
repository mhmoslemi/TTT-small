        r = minimize(
            lambda x: _pobj(x, n, beta, topk),
            h0, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: _pcon(x, n)}],
            options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
        )
        return _proj(np.asarray(r.x, dtype=np.float64), n)
    except Exception:
        return _proj(h0.copy(), n)


def _grad_loop(h, n, nit, rng, topk=5):
    best = h.copy()
    bc = _c5(best)
    betas = [20, 50, 100, 300, 1000, 3000, 8000]
    for it in range(nit):
        if _tl() < 4:
            break
        beta = betas[min(it // 40, len(betas) - 1)]
        g = _smooth_grad(best, beta, topk)
        gn = float(np.linalg.norm(g))
        if gn < 1e-16:
            g = _smooth_grad(best, 50, 3)
            gn = float(np.linalg.norm(g))
            if gn < 1e-16:
                break
        g = -g / gn
        improved = False
        alphas = [0.04, 0.015, 0.008, 0.02, 0.005, 0.03, 0.01, 0.003, 0.05, 0.002]
        for a in alphas:
            h2 = _proj(best + a * g, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-15:
                best = h2
                bc = c2
                improved = True
                break
        if not improved:
            c = _corr(best)
            kmax = int(np.argmax(c))
            g2 = _grad_single(best, kmax)
            gn2 = float(np.linalg.norm(g2))
            if gn2 > 1e-16:
                g2 = -g2 / gn2
                for a in [0.03, 0.01, 0.005, 0.015, 0.02]:
                    h2 = _proj(best + a * g2, n)
                    c2 = _c5(h2)
                    if c2 < bc - 1e-15:
                        best = h2
                        bc = c2
                        improved = True
                        break
        if not improved and it > 10 and it % 30 == 0:
            pert = rng.randn(n) * 0.01
            h2 = _proj(best + pert, n)
            c2 = _c5(h2)
            if c2 < bc + 0.001:
                best = h2
                bc = c2
    return best, bc


def _topkd_loop(h, n, nit, rng, topk=3):
    best = h.copy()
    bc = _c5(best)
    for it in range(nit):
        if _tl() < 4:
            break
        g = _topk_grad(best, topk)
        gn = float(np.linalg.norm(g))
        if gn < 1e-16:
            break
        g = -g / gn
        improved = False
        for a in [0.02, 0.01, 0.005, 0.03, 0.015, 0.008, 0.003]:
            h2 = _proj(best + a * g, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-15:
                best = h2
                bc = c2
                improved = True
                break
        if not improved:
            g = _smooth_grad(best, 100, 5)
            gn = float(np.linalg.norm(g))
            if gn < 1e-16:
                break
            g = -g / gn
            for a in [0.03, 0.01, 0.005]:
                h2 = _proj(best + a * g, n)
                c2 = _c5(h2)
                if c2 < bc - 1e-15:
                    best = h2
                    bc = c2
                    improved = True
                    break
    return best, bc


def _swap_local(h, n, nsweeps, rng):
    best = h.copy()
    bc = _c5(best)
    for _ in range(nsweeps):
        if _tl() < 4:
            break
        ones = np.where(best > 0.95)[0]
        zeros = np.where(best < 0.05)[0]
        if len(ones) == 0 or len(zeros) == 0:
            hi = np.where(best > 0.6)[0]
            lo = np.where(best < 0.4)[0]
            if len(hi) == 0 or len(lo) == 0:
                break
        else:
            hi = ones
            lo = zeros
        found = False
        for _ in range(200):
            if _tl() < 3:
                break
            i = hi[rng.randint(len(hi))]
            j = lo[rng.randint(len(lo))]
            h2 = best.copy()
            tmp = h2[i]
            h2[i] = h2[j]
            h2[j] = tmp
            h2 = _proj(h2, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-15:
                best = h2
                bc = c2
                found = True
                if best[i] > 0.95:
                    hi = np.where(best > 0.95)[0]
                    lo = np.where(best < 0.05)[0]
                else:
                    hi = np.where(best > 0.6)[0]
                    lo = np.where(best < 0.4)[0]
        if not found:
            ones = np.where(best > 0.95)[0]
            zeros = np.where(best < 0.05)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            found2 = False
            for i in ones:
                if _tl() < 3:
                    break
                for j in zeros:
                    h2 = best.copy()
                    tmp = h2[i]
                    h2[i] = h2[j]
                    h2[j] = tmp
                    h2 = _proj(h2, n)
                    c2 = _c5(h2)
                    if c2 < bc - 1e-15:
                        best = h2
                        bc = c2
                        found2 = True
                        break
                if found2:
                    break
            if not found2:
                break
    return best, bc


def _coord_descent(h, n, nit, rng):
    best = h.copy()
    bc = _c5(best)
    for _ in range(nit):
        if _tl() < 4:
            break
        nb = np.where((best > 0.02) & (best < 0.98))[0]
        if len(nb) < 2:
            break
        improved = False
        for idx in nb:
            if _tl() < 3:
                break
            old = best[idx]
            for delta in [0.008, -0.008, 0.015, -0.015, 0.004, -0.004, 0.02, -0.02, 0.002, -0.002]:
                nv = old + delta
                if nv < 0.0 or nv > 1.0:
                    continue
                others = [j for j in nb if j != idx]
                if len(others) == 0:
                    break
                j = others[rng.randint(len(others))]
                h2 = best.copy()
                h2[idx] = nv
                h2[j] += -delta
                h2 = _proj(h2, n)
                c2 = _c5(h2)
                if c2 < bc - 1e-15:
                    best = h2
                    bc = c2
                    improved = True
                    break
        if not improved:
            break
    return best, bc


def _optimize_full(h0, n, rng, time_budget):
    h = _proj(h0.copy(), n)
    c5 = _c5(h)

    start = time.time()
    elapsed_limit = time_budget

    for beta, tk, mi in [(80, 10, 200), (300, 6, 250), (1000, 4, 300)]:
        if _tl() < 15 or (time.time() - start) > elapsed_limit * 0.35:
            break
        h2 = _slsqp(h, n, beta, tk, mi)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2

    h, c5 = _grad_loop(h, n, 500, rng, topk=5)
    h, c5 = _topkd_loop(h, n, 200, rng, topk=3)
    h, c5 = _swap_local(h, n, 80, rng)
    h, c5 = _grad_loop(h, n, 400, rng, topk=4)
    h, c5 = _topkd_loop(h, n, 150, rng, topk=3)
    h, c5 = _swap_local(h, n, 60, rng)
    h, c5 = _coord_descent(h, n, 200, rng)

    if _tl() > 20 and (time.time() - start) < elapsed_limit * 0.7:
        for beta, tk, mi in [(2000, 3, 350), (5000, 3, 400)]:
            if _tl() < 15:
                break
            h2 = _slsqp(h, n, beta, tk, mi)
            c2 = _c5(h2)
            if c2 < c5:
                h, c5 = h2, c2
        h, c5 = _grad_loop(h, n, 300, rng, topk=3)
        h, c5 = _swap_local(h, n, 40, rng)
        h, c5 = _topkd_loop(h, n, 200, rng, topk=3)
        h, c5 = _coord_descent(h, n, 150, rng)

    if _tl() > 10 and (time.time() - start) < elapsed_limit * 0.9:
        h2 = _slsqp(h, n, 10000, 2, 500)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2
        h, c5 = _grad_loop(h, n, 200, rng, topk=3)
        h, c5 = _swap_local(h, n, 30, rng)

    return h, c5


def _multi_restart(h0, n, rng, nstarts, tper):
    best_h = h0.copy()
    best_c = _c5(h0)
    for s in range(nstarts):
        if _tl() < tper + 5:
            break
        if s == 0:
            h_try = h0.copy()
        elif s == 1:
            h_try = _proj(h0 + rng.randn(n) * 0.02, n)
        elif s == 2:
            h_try = _proj(h0 + rng.randn(n) * 0.05, n)
        else:
            h_try = _proj(h0 + rng.randn(n) * 0.1, n)
        h2, c2 = _optimize_full(h_try, n, rng, tper)
        if c2 < best_c:
            best_h, best_c = h2, c2
    return best_h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    global _t0, _tb
    _t0 = time.time()
    _tb = budget_s
    rng = np.random.RandomState(seed)

    try:
        hpar = np.array(initial_h_values, dtype=np.float64)
        npar = len(hpar)
    except Exception:
        npar = 160
        hpar = np.zeros(npar, dtype=np.float64)
        hpar[:npar // 2] = 1.0

    bh = _proj(hpar, npar)
    bc5 = _c5(bh)
    bn = npar
    print("Init n={} C5={:.8f}".format(bn, bc5))

    # Phase 1: Optimize at parent resolution with multiple restarts
    t_per = min(150, _tl() * 0.3)
    h, c5 = _multi_restart(bh, npar, rng, 3, t_per)
    if c5 < bc5:
        bh, bc5, bn = h, c5, npar
        print("Phase1 n={} C5={:.8f}".format(npar, c5))

    # Phase 2: Resolution sweep - try perturbations in resolution
    res_list = [npar + 2, npar - 2, npar + 4, npar - 4, npar + 6, npar - 6,
                npar + 8, npar + 10, npar + 14, npar + 20]
    for nn in res_list:
        if _tl() < 80:
            break
        if nn < 40:
            continue
        hi = _proj(_interp(bh, bn, nn), nn)
        t_budget = min(120, _tl() * 0.25)
        if t_budget < 40:
            break
        h2, c2 = _optimize_full(hi, nn, rng, t_budget)
        if c2 < bc5:
            bh, bc5, bn = h2, c2, nn
            print("ResN={} C5={:.8f}".format(nn, c2))

    # Phase 3: Try significantly different resolutions
    for nn in [npar + 30, npar + 40, npar + 60, npar - 10]:
        if _tl() < 120:
            break
        if nn < 40:
            continue
        hi = _proj(_interp(bh, bn, nn), nn)
        t_budget = min(100, _tl() * 0.3)
        if t_budget < 40:
            break
        h2, c2 = _optimize_full(hi, nn, rng, t_budget)
        if c2 < bc5:
            bh, bc5, bn = h2, c2, nn
            print("HiRes N={} C5={:.8f}".format(nn, c2))

    # Phase 4: Final polish on best solution
    if _tl() > 40:
        h, c5 = _grad_loop(bh, bn, 400, rng, topk=3)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _topkd_loop(bh, bn, 300, rng, topk=3)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _swap_local(bh, bn, 50, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _coord_descent(bh, bn, 200, rng)
        if c5 < bc5:
            bh, bc5 = h, c5

    if _tl() > 20:
        h2 = _slsqp(bh, bn, 5000, 2, 500)
        c2 = _c5(h2)
        if c2 < bc5:
            bh, bc5 = h2, c2
        h, c5 = _grad_loop(bh, bn, 200, rng, topk=3)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _swap_local(bh, bn, 30, rng)
        if c5 < bc5:
            bh, bc5 = h, c5

    # Final tiny perturbations
    if _tl() > 15:
        for _ in range(50):
            if _tl() < 5:
                break
            h2 = _proj(bh + rng.randn(bn) * 0.001, bn)
            c2 = _c5(h2)
            if c2 < bc5 - 1e-15:
                bh = h2
                bc5 = c2

    bh = _proj(bh, bn)
    fc5 = _c5(bh)
    try:
        ec = evaluate_erdos_solution(bh)
        if ec < fc5:
            fc5 = ec
    except Exception:
        pass
    print("FINAL n={} C5={:.8f}".format(bn, fc5))
    return (bh, fc5, bn)
