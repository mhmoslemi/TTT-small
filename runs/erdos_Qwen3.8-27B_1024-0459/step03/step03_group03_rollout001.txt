        hi = np.where(h > 0.65)[0]
        loii = np.where(h < 0.35)[0]
        if hi.size == 0 or loii.size == 0:
            break
        anypp = False
        for _ in range(100):
            if stop and time.time() > stop:
                break
            i = int(hi[rng.randint(hi.size)])
            jj = int(loii[rng.randint(loii.size)])
            h2 = h.copy()
            tmp = float(h2[i])
            h2[i] = float(h2[jj])
            h2[jj] = tmp
            c2 = _full_c5(h2)
            if c2 < _full_c5(h) - 1e-12:
                h = h2
                anypp = True
        if not anypp:
            break
    bh = h.copy()
    bc = _full_c5(bh)
    return (bh, bc)


def _refine(h0, n, stop, budget_s):

def _upsample(h, ns, nd):
    if ns == nd or ns <= 0 or nd <= 0:
        return np.asarray(h, dtype=np.float64).copy()
    xs = (np.arange(ns, dtype=np.float64) + 0.5) * (2.0 / ns)
    xd = (np.arange(nd, dtype=np.float64) + 0.5) * (2.0 / nd)
    return np.interp(xd, xs, np.asarray(h, dtype=np.float64))


def _beatty(n, alpha, beta):
    h = np.zeros(n, dtype=np.float64)
    yy = (np.arange(n, dtype=np.float64) * alpha - np.floor(np.arange(n, dtype=np.float64) * alpha) + beta)
    yy = yy - np.floor(yy)
    tt = np.sort(yy)
    cut = tt[n // 2 - 1]
    kkk = np.where(yy <= cut)[0]
    h[kkk] = 1.0
    return h


def _init(n, par):
    cands = []
    for alpha in ((1.0 + np.sqrt(5.0)) / 2.0, (1.0 + np.sqrt(2.0)) / 2.0, np.pi / 5.0, np.exp(1.0) / 2.3, 0.381966, 0.2360679):
        for bb in (0.0, 0.33):
            c = _beatty(n, float(alpha), float(bb))
            c = _proj_box(c, n)
            cands.append(c)
    if par is not None:
        cp = _proj_box(_upsample(par, len(par), n), n)
        cands.append(cp)
    best = None
    bc = 1e300
    for c in cands:
        cv = _full_c5(c)
        if cv < bc:
            bc = cv
            best = c
    return best
