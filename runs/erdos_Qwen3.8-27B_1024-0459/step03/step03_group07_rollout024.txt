    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(1.0 - h)[:max(1, int(np.ceil(abs(diff))))]
            h[idx] += diff / len(idx)
        else:
            idx = np.argsort(h)[:max(1, int(np.ceil(abs(diff))))]
            h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, n, beta):
    h = x
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return (max_c + log_sum / beta) * (2.0 / n)

def smooth_max_obj_and_grad(x, n, beta):
    h = x
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    obj = (max_c + np.log(sum_exp) / beta) * (2.0 / n)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        w = weights[k]
        if abs(w) < 1e-12:
            continue
        if s >= 0:
            if s < n:
                grad[:n - s] += w * (1.0 - h[s:n])
                grad[s:] += -w * h[:n - s]
        else:
            t = -s
            if t < n:
                grad[t:] += w * (1.0 - h[:n - t])
                grad[:n - t] += -w * h[t:]
    grad *= (2.0 / n)
    return obj, grad

def topk_obj_and_grad(x, n, k):
    h = x
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-k:]
    top_val = np.mean(corr[top_idx]) * (2.0 / n)
    weights = np.zeros(2 * n - 1)
    weights[top_idx] = 1.0 / k
    grad = np.zeros(n)
    for ki in top_idx:
        s = ki - (n - 1)
        w = weights[ki]
        if s >= 0:
            if s < n:
                grad[:n - s] += w * (1.0 - h[s:n])
                grad[s:] += -w * h[:n - s]
        else:
            t = -s
            if t < n:
                grad[t:] += w * (1.0 - h[:n - t])
                grad[:n - t] += -w * h[t:]
    grad *= (2.0 / n)
    return top_val, grad

def penalty_obj_and_grad(x, n, beta, lam):
    h = x
    om, og = smooth_max_obj_and_grad(h, n, beta)
    e = np.sum(h) - n / 2.0
    obj = om + lam * e * e
    grad = og + 2.0 * lam * e
    return obj, grad

def interpolate_h(h_src, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_new = np.interp(xn, xp, h_src)
    return project_feasible(h_new, n_dst)

def lfbgbs_optimize(h0, n, beta, lam, maxiter, t0, time_limit):
    bounds = [(0.0, 1.0)] * n
    best = h0.copy()
    best_c5 = compute_c5(h0)
    def fobj(x):
        return penalty_obj_and_grad(x, n, beta, lam)[0]
    def fgrad(x):
        return penalty_obj_and_grad(x, n, beta, lam)[1]
    try:
        res = minimize(fobj, h0, jac=fgrad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new
    except Exception:
        pass
    return best, best_c5

def slsqp_optimize(h0, n, beta, maxiter, t0, time_limit):
    bounds = [(0.0, 1.0)] * n
    best = h0.copy()
    best_c5 = compute_c5(h0)
    def fobj(x):
        return smooth_max_obj(x, n, beta)
    def fjac(x):
        return smooth_max_obj_and_grad(x, n, beta)[1]
    def feq(x):
        return np.sum(x) - n / 2.0
    cons = [{'type': 'eq', 'fun': feq}]
    try:
        res = minimize(fobj, h0, jac=fjac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new
    except Exception:
        pass
    return best, best_c5

def coord_descent_fast(h, n, sweeps, t0, time_limit):
    h = project_feasible(h, n)
    best = h.copy()
    best_c5 = compute_c5(h)
    for sw in range(sweps):
        if time.time() - t0 > time_limit:
            break
        for j in range(n):
            if time.time() - t0 > time_limit:
                break
            old = h[j]
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_unaffected = 0.0
            a_vals = {}
            b_vals = {}
            rest_vals = {}
            for lag in range(-(n - 1), n):
                fidx = lag + (n - 1)
                affects = False
                coeff = 0.0
                const_term = 0.0
                if lag >= 0:
                    if j <= n - 1 - lag:
                        coeff += (1.0 - h[j + lag])
                        affects = True
                    if lag <= j <= n - 1:
                        coeff += (-h[j - lag])
                        const_term += h[j - lag]
                        affects = True
                else:
                    s = -lag
                    if 0 <= j <= n - 1 - s:
                        coeff += (1.0 - h[j + s])
                        affects = True
                    if s <= j <= n - 1:
                        coeff += (-h[j - s])
                        const_term += h[j - s]
                        affects = True
                if affects:
                    if lag == 0:
                        a_vals[fidx] = 1.0
                        b_vals[fidx] = 0.0
                        rest_vals[fidx] = corr[fidx] - old * (1.0 - old)
                    else:
                        a_vals[fidx] = coeff
                        b_vals[fidx] = const_term
                        rest_vals[fidx] = corr[fidx] - coeff * old - const_term
                else:
                    if corr[fidx] > max_unaffected:
                        max_unaffected = corr[fidx]
            xs = np.linspace(0.0, 1.0, 80)
            best_x = old
            best_val = float('inf')
            for x in xs:
                val = max_unaffected
                for fidx in a_vals:
                    if fidx == n - 1:
                        v = rest_vals[fidx] + x - x * x
                    else:
                        v = rest_vals[fidx] + a_vals[fidx] * x + b_vals[fidx]
                    if v > val:
                        val = v
                if val < best_val:
                    best_val = val
                    best_x = x
            lo = max(0.0, best_x - 0.02)
            hi = min(1.0, best_x + 0.02)
            for _ in range(30):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                v1 = max_unaffected
                for fidx in a_vals:
                    if fidx == n - 1:
                        v = rest_vals[fidx] + m1 - m1 * m1
                    else:
                        v = rest_vals[fidx] + a_vals[fidx] * m1 + b_vals[fidx]
                    if v > v1:
                        v1 = v
                v2 = max_unaffected
                for fidx in a_vals:
                    if fidx == n - 1:
                        v = rest_vals[fidx] + m2 - m2 * m2
                    else:
                        v = rest_vals[fidx] + a_vals[fidx] * m2 + b_vals[fidx]
                    if v > v2:
                        v2 = v
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            h[j] = (lo + hi) / 2.0
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = 160

    def time_left():
        return max(0.0, budget_s - (time.time() - t0) - 10.0)

    best_h = None
    best_c5 = float('inf')
    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn > 10:
            if pn == n:
                best_h = ph.copy()
            else:
                best_h = interpolate_h(ph, pn, n)
            best_c5 = compute_c5(best_h)
    except Exception:
        pass

    if best_h is None:
        best_h = np.ones(n) * 0.5
        best_h = project_feasible(best_h, n)
        best_c5 = compute_c5(best_h)

    # Generate diverse initial points
    inits = [best_h.copy()]
    for scale in [0.03, 0.06, 0.10]:
        hp = best_h + scale * rng.randn(n)
        inits.append(project_feasible(hp, n))
    t_arr = np.arange(n) / n
    for freq in [2, 3, 5, 7]:
        phase = rng.rand() * 2 * np.pi
        pattern = 0.5 + 0.45 * np.cos(2.0 * np.pi * freq * t_arr + phase)
        hf = 0.6 * best_h + 0.4 * np.clip(pattern, 0, 1)
        inits.append(project_feasible(hf, n))
    # Triangle wave
    tri = np.abs(2.0 * (t_arr - 0.5))
    ht = 0.5 + 0.4 * np.cos(np.pi * tri)
    inits.append(project_feasible(0.5 * best_h + 0.5 * np.clip(ht, 0, 1), n))
    # Asymmetric step
    step = np.where(t_arr < 0.4, 0.9, np.where(t_arr < 0.8, 0.3, 0.6))
    inits.append(project_feasible(0.5 * best_h + 0.5 * step, n))

    # Main optimization loop
    for init_i, h0 in enumerate(inits):
        if time_left() < 60:
            break
        h_cur = h0.copy()
        c5_cur = compute_c5(h_cur)

        # Phase 1: L-BFGS-B with increasing beta and penalty
        for beta, lam, miter in [(30, 50.0, 200), (100, 200.0, 200), (300, 500.0, 150), (800, 1000.0, 150)]:
            if time_left() < 30:
                break
            h_new, c5_new = lfbgbs_optimize(h_cur, n, beta, lam, miter, t0, budget_s - 10)
            if c5_new < c5_cur:
                h_cur = h_new
                c5_cur = c5_new
            elif time_left() < 30 + 30 * (3 - init_i):
                break

        # Phase 2: SLSQP with smooth max
        if time_left() > 40:
            h_new, c5_new = slsqp_optimize(h_cur, n, 1000, 200, t0, budget_s - 15)
            if c5_new < c5_cur:
                h_cur = h_new
                c5_cur = c5_new

        # Phase 3: Top-k SLSQP
        if time_left() > 30:
            for k in [5, 10]:
                if time_left() < 20:
                    break
                h_new, c5_new = topk_slsqp(h_cur, n, k, 150, t0, budget_s - 10)
                if c5_new < c5_cur:
                    h_cur = h_new
                    c5_cur = c5_new

        # Track best
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()

    # Final coordinate descent polish on best solution
    if time_left() > 40:
        h_cd, c5_cd = coord_descent_fast(best_h, n, sweeps=12, t0=t0, time_limit=time_left() + t0 - 5)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Another SLSQP round
    if time_left() > 30:
        h_sp, c5_sp = slsqp_optimize(best_h, n, 2000, 300, t0, budget_s - 8)
        if c5_sp < best_c5:
            best_c5 = c5_sp
            best_h = h_sp.copy()

    # More coordinate descent
    if time_left() > 25:
        h_cd2, c5_cd2 = coord_descent_fast(best_h, n, sweeps=15, t0=t0, time_limit=time_left() + t0 - 5)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    # Final SLSQP
    if time_left() > 15:
        h_sp2, c5_sp2 = slsqp_optimize(best_h, n, 5000, 200, t0, budget_s - 5)
        if c5_sp2 < best_c5:
            best_c5 = c5_sp2
            best_h = h_sp2.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)

    # Validate with evaluator
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    return (best_h, final_c5, n)
