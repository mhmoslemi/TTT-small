```python
    # But this is only valid when the indices are in range!
    # 1-h[p+s] is valid only when 0 ≤ p+s < n (otherwise no such term in the sum)
    # h[p-s] is valid only when 0 ≤ p-s < n (otherwise no such term)

            s = s_idx - (n-1)
            val = 0.0

                val += 1.0 - h[p+s]
            if 0 <= p-s < n:
                val -= h[p-s]
            F[p, s_idx] = val
    return F
```

```python
def compute_F(h, n):
    F = np.zeros((n, 2*n-1))
    indices = np.arange(n)
    for s_idx in range(2*n-1):
        s = s_idx - (n-1)
        # (1-h[p+s]) term: valid when 0 <= p+s < n
        ps = indices + s
        mask1 = (ps >= 0) & (ps < n)
        F[mask1, s_idx] += (1 - h[ps[mask1]])
        # -h[p-s] term: valid when 0 <= p-s < n
        ps2 = indices - s
        mask2 = (ps2 >= 0) & (ps2 < n)
        F[mask2, s_idx] -= h[ps2[mask2]]
    return F
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_F_matrix(h, n):
    F = np.zeros((n, 2 * n - 1))
    indices = np.arange(n)
    for s_idx in range(2 * n - 1):
        s = s_idx - (n - 1)
        ps = indices + s
        mask1 = (ps >= 0) & (ps < n)
        F[mask1, s_idx] += (1.0 - h[ps[mask1]])
        ps2 = indices - s
        mask2 = (ps2 >= 0) & (ps2 < n)
        F[mask2, s_idx] -= h[ps2[mask2]]
    return F


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        unsat = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(unsat)
        if cnt > 0:
            h[unsat] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def active_set_grad(x, top_k=12):
    n = _N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    
    grad = np.zeros(n)
    for j in top_idx:
        s = j - (n - 1)
        if s >= 0:
            m_range = np.arange(s, n)
            grad[m_range] += (1.0 - x[m_range - s])
            m_range2 = np.arange(0, n - s)
            grad[m_range2] -= x[m_range2 + s]
        else:
            sp = -s
            m_range = np.arange(0, n - sp)
            grad[m_range] += (1.0 - x[m_range + sp])
            m_range2 = np.arange(sp, n)
            grad[m_range2] -= x[m_range2 - sp]
    
    grad *= dx
    pen_coeff = 500.0
    grad += 2.0 * pen_coeff * (np.sum(x) - _TGT)
    return grad


def binary_local_search(h, n, max_iters=50000, rng=None, verbose=False):
    """Fast binary local search using F-matrix trick."""
    dx = 2.0 / n
    h = np.where(h > 0.5, 1.0, 0.0).copy()
    num_ones = int(np.sum(h))
    target_ones = n // 2
    if num_ones > target_ones:
        ones_idx = np.where(h == 1.0)[0]
        remove = rng.choice(len(ones_idx), num_ones - target_ones, replace=False)
        h[ones_idx[remove]] = 0.0
    elif num_ones < target_ones:
        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
        h[zeros_idx[add]] = 1.0
    
    corr = compute_corr(h)
    best_c5 = np.max(corr) * dx
    current_c5 = best_c5
    F = compute_F_matrix(h, n)
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    no_improve_count = 0
    tabu_iters = 20
    tabu_set = set()
    
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        
        # Find candidate swaps
        best_delta = 0.0
        best_i = -1
        best_j = -1
        best_new_max = current_c5
        
        # Strategy 1: Try swaps targeting the max shift
        max_shifts = np.argsort(corr)[-8:]
        cand_pairs = []
        for s_idx in max_shifts:
            # Find best i (h=0) maximizing F[i, s_idx]
            if len(zeros) == 0 or len(ones) == 0:
                break
            f_zeros = F[zeros, s_idx]
            f_ones = F[ones, s_idx]
            # We want F[i,s] - F[j,s] to be most negative (reduces corr at s)
            # i.e., minimize F[i,s] - F[j,s] = maximize F[j,s] - F[i,s]
            best_i_local = zeros[np.argmin(f_zeros)]
            best_j_local = ones[np.argmax(f_ones)]
            key = (best_i_local, best_j_local)
            if key not in tabu_set:
                cand_pairs.append(key)
            # Also try reversed: maximize F[i,s] - F[j,s] for other shifts
            best_i_local2 = zeros[np.argmax(f_zeros)]
            best_j_local2 = ones[np.argmin(f_ones)]
            key2 = (best_i_local2, best_j_local2)
            if key2 not in tabu_set:
                cand_pairs.append(key2)
        
        # Strategy 2: Random samples
        for _ in range(20):
            if len(zeros) == 0 or len(ones) == 0:
                break
            ri = zeros[rng.randint(len(zeros))]
            rj = ones[rng.randint(len(ones))]
            key = (ri, rj)
            if key not in tabu_set:
                cand_pairs.append(key)
        
        # Evaluate candidates
        for (ci, cj) in cand_pairs:
            new_corr = corr + F[ci] - F[cj]
            new_max = np.max(new_corr) * dx
            if new_max < best_new_max:
                best_new_max = new_max
                best_i = ci
                best_j = cj
                best_delta = best_new_max - current_c5
        
        if best_i >= 0 and best_delta < -1e-12:
            # Apply swap
            h[best_i] = 1.0
            h[best_j] = 0.0
            corr = corr + F[best_i] - F[best_j]
            F = compute_F_matrix(h, n)
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            current_c5 = np.max(corr) * dx
            no_improve_count = 0
            tabu_set.add((best_i, best_j))
            tabu_set.add((best_j, best_i))
            
            if current_c5 < best_c5:
                best_c5 = current_c5
            
            if verbose and it % 1000 == 0:
                print("    iter {}: C5={:.7f}".format(it, current_c5))
        else:
            no_improve_count += 1
            if best_i >= 0:
                tabu_set.add((best_i, best_j))
            if len(tabu_set) > 200:
                tabu_set.clear()
            if no_improve_count > 500:
                break
    
    return h, best_c5

def continuous_optimize(h0, n, max_iter=300, beta=100):

        c5_new = compute_c5(h_new)

                print("  Restart {}: C5={:.7f}".format(restart, c5_b))
    
    # Phase 2: SA on binary
    if remaining_time() > 80:
        n = best_n
        print("\n--- Phase 2: Binary SA (n={}) ---".format(n))
        h_bin = np.where(best_h > 0.5, 1.0, 0.0).copy()
        h_b, c5_b = binary_sa(h_bin, n, iterations=200000, rng=rng, t_start=0.004, t_end=0.00003)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n
            print("  SA: C5={:.7f}".format(c5_b))
        
        # Second SA with different params
        if remaining_time() > 40:
            h_b2, c5_b2 = binary_sa(best_h, n, iterations=100000, rng=rng, t_start=0.002, t_end=0.00001)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()
                print("  SA2: C5={:.7f}".format(c5_b2))
    
    # Phase 3: Continuous optimization at multiple resolutions
    n_list = sorted(set([best_n]))
    # Add higher resolutions if time permits
    if remaining_time() > 120:
        n_list.append(best_n + 40)
    if remaining_time() > 180:
        n_list.append(best_n + 80)
    if remaining_time() > 250:
        n_list.append(320)
    
    n_list = sorted(set([nn for nn in n_list if 80 <= nn <= 400]))
    
    for n in n_list:
        if remaining_time() < 40:
            break
        print("\n--- Phase 3: Continuous opt (n={}) ---".format(n))
        
        h_interp = project_feasible(interp_to(best_h, best_n, n), n)
        c5_cur = compute_c5(h_interp)
        
        inits = [h_interp]
        # Add binary initialization
        h_bin_n = np.where(h_interp > 0.5, 1.0, 0.0).copy()
        h_bin_n = project_feasible(h_bin_n, n)
        inits.append(h_bin_n)
        # Thue-Morse at this n
        inits.append(project_feasible(make_thue_morse(n), n))
        inits.append(project_feasible(make_golden_ratio(n), n))
        
        for h0 in inits:
            if remaining_time() < 20:
                break
            c5_0 = compute_c5(h0)
            
            # Binary local search at this resolution
            if remaining_time() > 60 and c5_0 < best_c5 * 1.02:
                h_b, c5_b = binary_local_search(h0, n, max_iters=30000, rng=rng)
                if c5_b < best_c5:
                    best_c5 = c5_b
                    best_h = h_b.copy()
                    best_n = n
                    h0 = h_b.copy()
                    c5_0 = c5_b
                    print("  Bin LS n={}: C5={:.7f}".format(n, c5_b))
            
            # Continuous optimization with increasing beta
            for beta_val in [50, 150, 400]:
                if remaining_time() < 10:
                    break
                h_new, c5_new = continuous_optimize(h0, n, max_iter=200, beta=beta_val)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n
                    h0 = h_new.copy()
                    print("  Cont n={} beta={}: C5={:.7f}".format(n, beta_val, c5_new))
                elif c5_new < c5_0:
                    h0 = h_new.copy()
                    c5_0 = c5_new
    
    # Phase 4: Final polish at best resolution
    if remaining_time() > 20:
        n = best_n
        print("\n--- Phase 4: Final polish (n={}) ---".format(n))
        
        # Binary local search again
        h_b, c5_b = binary_local_search(best_h, n, max_iters=30000, rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            print("  Polish BS: C5={:.7f}".format(c5_b))
        
        # High-beta continuous
        for beta_val in [500, 1000, 2000]:
            if remaining_time() < 8:
                break
            h_new, c5_new = continuous_optimize(best_h, n, max_iter=300, beta=beta_val)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  Polish CB={} : C5={:.7f}".format(beta_val, c5_new))
        
        # Coordinate descent
        if remaining_time() > 15:
            h_cd = project_feasible(best_h, n)
            dx_val = 2.0 / n
            for sweep in range(5):
                if remaining_time() < 5:
                    break
                any_imp = False
                order = rng.permutation(n)
                for i in order:
                    if remaining_time() < 2:
                        break
                    old_val = h_cd[i]
                    best_local = compute_c5(h_cd)
                    for nv in [0.0, 1.0, 0.5, max(0.0, old_val - 0.15), min(1.0, old_val + 0.15)]:
                        if abs(nv - old_val) < 1e-8:
                            continue
                        h_cd[i] = nv
                        h_feas = project_feasible(h_cd, n)
                        c5_test = compute_c5(h_feas)
                        if c5_test < best_local:
                            best_local = c5_test
                            h_cd[i] = nv
                            h_cd = h_feas.copy()
                            any_imp = True
                    h_cd[i] = h_cd[i]  # keep best
                    if best_local < best_c5:
                        best_c5 = best_local
                        best_h = h_cd.copy()
                if not any_imp:
                    break
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
```
