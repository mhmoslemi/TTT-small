```python
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    # For numerical stability
    shifted -= np.max(shifted)  # already 0 since m is max
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
```

```python
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    exp_vals = np.exp(shifted - np.max(shifted))
    w = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if w[k] < 1e-15:

        d = k - (n - 1)
        t = abs(d)
        if d >= 0:
            if t == 0:
                grad[:] += w[k] * (1.0 - 2.0*h)
            else:
                grad[t:] += w[k] * (1.0 - h[:n-t])
                grad[:n-t] -= w[k] * h[t:]
```

```python
        zeros.add(a)
```

```python
def compute_swap_costs(h, n):
    """Compute cost and gain matrices for binary swap local search."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    n_ones = len(ones)
    n_zeros = len(zeros)
    
    cost_matrix = np.zeros((n_ones, n_corr))
    for i, a in enumerate(ones):
        for k in range(n_corr):
            d = k - (n - 1)
            val = 0.0
            # h[a] going 1→0: 
            # Term where a is the "i" position: was 1*(1-h[a-d]), becomes 0. Loss: -(1-h[a-d]) if valid
            ad = a - d
            if 0 <= ad < n:
                val -= (1.0 - h[ad])
            # Term where a is the "i-d" position: was h[a+d]*(1-1)=0, becomes h[a+d]*(1-0). Gain: +h[a+d] if valid
            apd = a + d
            if 0 <= apd < n:
                val += h[apd]
            cost_matrix[i, k] = val
    
    gain_matrix = np.zeros((n_zeros, n_corr))
    for i, b in enumerate(zeros):
        for k in range(n_corr):
            d = k - (n - 1)
            val = 0.0
            # h[b] going 0→1:
            # Term where b is the "i" position: was 0, becomes 1*(1-h[b-d]). Gain: +(1-h[b-d]) if valid
            bd = b - d

                val -= h[bpd]
            gain_matrix[i, k] = val
    
    return corr, cost_matrix, gain_matrix, ones, zeros
```

But this is O(n³) in Python (n/2 entries × n_corr values × constant work). For n=160: 80 × 319 × 2 = 51K iterations. Each is just a few operations. In Python, maybe 50-100ms. That's fine for one pass.

But I can vectorize it! For a given a:
- The "loss" term: -(1-h[a-d]) for d where 0≤a-d<n, i.e., a-n+1 ≤ d ≤ a
  In terms of k: d=k-(n-1), so a-n+1 ≤ k-(n-1) ≤ a, i.e., 2a-n+2 ≤ k ≤ 2a+1... wait let me redo.
  
  0 ≤ a-d < n means 0 ≤ a-(k-(n-1)) < n, i.e., k-(n-1) ≤ a and a-(k-(n-1)) < n
  i.e., k ≤ a+n-1 and k ≥ a-n+1
  So k ranges from max(0, a-n+1) to min(2n-2, a+n-1)
  
  And the value is -(1-h[a-d]) = -(1-h[2(n-1)+a-k])... this is getting complicated.

Let me just use a different vectorization. For a given a:
- d ranges from -(n-1) to (n-1)
- a-d ranges from a-(n-1) to a+(n-1)
- Valid when 0 ≤ a-d < n, i.e., -(n-1) - a + 1... 

You know what, for n=160, the Python loop version takes ~50ms per pass. That's fine. Let me just use the Python loop version for clarity and correctness, and rely on the numpy vectorized max computation for the pair search.

Actually, I just realized there might be an even simpler approach. Let me precompute the full correlation, then for each possible swap, compute the max of the modified correlation using vectorized numpy. The key insight is:

new_corr = corr - cost_a + gain_b (element-wise)
new_max = np.max(new_corr)

For n=160 with 80×80=6400 pairs, each requiring a 319-element max: that's 6400 × 319 = 2M operations. In a single numpy broadcast:
```python
new_vals = np.max(corr[None, None, :] - cost[None, :, :] + gain[:, None, :], axis=2)
```
This creates a tensor of shape (80, 80, 319) = 2M elements = 16MB. Fine.

Then argmin gives the best swap in O(80×80) = 6400 ops.

Total per pass: O(n²) for cost/gain computation + O(n³) for broadcast (but vectorized) = maybe 20-50ms.

OK, I think my approach is solid. Let me now write the complete program.

```python
def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 2.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        unsat = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(unsat)
        if cnt > 0:
            h[unsat] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def continuous_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    bc = beta * corr
    bc_max = np.max(bc)
    exp_vals = np.exp(bc - bc_max)
    sum_exp = np.sum(exp_vals)
    obj = (bc_max + np.log(sum_exp)) / beta * dx
    penalty = 100.0 * (np.sum(h) - n / 2.0) ** 2
    obj += penalty
    w = exp_vals / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        t = abs(d)
        if t == 0:
            grad[:] += wk * (1.0 - 2.0 * h)
        else:
            grad[t:] += wk * (1.0 - h[:n - t])
            grad[:n - t] -= wk * h[t:]
    grad *= beta * dx
    grad += 2.0 * 100.0 * (np.sum(h) - n / 2.0)
    return obj, grad

def continuous_obj(h, _beta_cache):
    return continuous_obj_grad(h, _beta_cache[0])[0]

def continuous_grad(h, _beta_cache):
    return continuous_obj_grad(h, _beta_cache[0])[1]

def binary_local_search(h, n, max_passes=2000, time_limit=None):
    h = np.array(h, dtype=float).copy()
    best_c5 = compute_c5(h)
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    dx = 2.0 / n

    for pass_num in range(max_passes):
        if remaining_time() < 5:
            break
        if time_limit is not None and (time.time() - _START_TIME) > time_limit:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break

        n_ones = len(ones)
        n_zeros = len(zeros)

        cost_matrix = np.zeros((n_ones, n_corr))
        for i, a in enumerate(ones):
            h[a] = 0.0
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            cost_matrix[i] = new_corr - corr
            h[a] = 1.0

        gain_matrix = np.zeros((n_zeros, n_corr))
        for i, b in enumerate(zeros):
            h[b] = 1.0
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            gain_matrix[i] = new_corr - corr
            h[b] = 0.0

        new_vals = np.max(corr[None, None, :] - cost_matrix[None, :, :] + gain_matrix[:, None, :], axis=2)
        min_val = np.min(new_vals)

        if min_val * dx >= best_c5 - 1e-14:
            break

        j_best, i_best = np.unravel_index(np.argmin(new_vals), new_vals.shape)
        a_best = ones[i_best]
        b_best = zeros[j_best]

        h[a_best] = 0.0
        h[b_best] = 1.0
        ones.remove(a_best)
        zeros.remove(b_best)
        ones.append(b_best)
        zeros.append(a_best)

        corr = np.correlate(h, 1.0 - h, mode='full')
        best_c5 = min_val * dx

    return h, best_c5

    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    return h

def make_sturmian(n, alpha):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if ((i + 1) * alpha % 1.0) < alpha else 0.0
    # Ensure exactly n/2 ones

        h[ones_idx[remove]] = 0.0
    elif num_ones < n // 2:
        zeros_idx = np.where(h == 0.0)[0]
        add = np.random.choice(len(zeros_idx), n // 2 - num_ones, replace=False)
        h[zeros_idx[add]] = 1.0
    return h
```
