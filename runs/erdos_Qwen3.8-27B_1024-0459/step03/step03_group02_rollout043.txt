def slsqp_smooth_max_obj(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def constraint_sum(x, n):
    return np.sum(x) - n / 2.0

def slsqp_refine(h, n, beta, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [
        {'type': 'eq', 'fun': lambda x: constraint_sum(x, n)}
    ]
    try:
        res = minimize(slsqp_smooth_max_obj, h, args=(n, beta),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def interp_to(src_h, src_n, dst_n):

    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-12:
            for j in range(n):
                if 0 < h[j] < 1:
                    h[j] = np.clip(h[j] + diff2, 0.0, 1.0)
                    break
    return h

def thue_morse_seq(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    rng = np.random.RandomState(42)
    if n_ones > target:
        one_pos = np.where(h == 1)[0]
        rng.shuffle(one_pos)
        for p in one_pos[:n_ones - target]:
            h[p] = 0.0
    elif n_ones < target:
        zero_pos = np.where(h == 0)[0]
        rng.shuffle(zero_pos)
        for p in zero_pos[:target - n_ones]:
            h[p] = 1.0
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def slsqp_active_set(h, n, n_active, beta, maxiter):
    """SLSQP targeting top-n_active lags only."""
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[::-1][:n_active]

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        top_c = c[top_idx]
        mx = np.max(top_c)
        shifted = np.minimum(beta * (top_c - mx), 0.0)
        return (mx + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

    def cons_fun(x):
        return np.sum(x) - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_full_max(h, n, beta, maxiter):
    """SLSQP with full smooth-max objective."""
    def obj(x):

        shifted = np.minimum(beta * (c - mx), 0.0)

    def cons_fun(x):

                if remaining_time() < 3:
