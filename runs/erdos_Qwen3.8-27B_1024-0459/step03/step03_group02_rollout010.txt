def projected_gradient_step(h, n, max_lags=5, step_candidates=None):
    """Project gradient step targeting top correlation lags."""
    corr = compute_corr(h)
    max_corr = np.max(corr)
    # Get top lags
    sorted_idx = np.argsort(corr)[::-1]
    active = sorted_idx[:min(max_lags, len(sorted_idx))]
    
    # Compute gradient as weighted sum of per-lag gradients
    grad = np.zeros(n)
    total_weight = 0
    for k_idx in active:
        w = max(corr[k_idx] - max_corr + 1.0, 0.1)
        offset = k_idx - (n - 1)
        # Gradient of sum_i h[i+offset]*(1-h[i]) w.r.t. h
        # For positive offset: d/dh[j] sum_i h[i+offset]*(1-h[i])
        # For j in [offset, offset+n-1-offset): contribution from h[j] is (1-h[j-offset])
        # minus contribution when j is the i: -h[j+offset]
        n_pairs = n - abs(offset)
        if offset >= 0:
            grad[offset:offset+n_pairs] += w * (1.0 - h[:n_pairs])
            grad[:n_pairs] -= w * h[offset:offset+n_pairs]
        else:
            a = -offset
            grad[:n_pairs] += w * (1.0 - h[a:a+n_pairs])
            grad[a:a+n_pairs] -= w * h[:n_pairs]
        total_weight += w
    
    if total_weight < 1e-15:
        return None, None
    grad /= total_weight
    
    if step_candidates is None:
        step_candidates = [0.01, 0.005, 0.002, 0.001, 0.02, 0.05]
    
    return grad, step_candidates

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:

    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 60.0)
    dx = 2.0 / n
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def projected_gradient(h, n, top_k=5):
    corr = compute_corr(h)
    max_corr = np.max(corr)
    sorted_idx = np.argsort(corr)[::-1]
    active = sorted_idx[:top_k]
    grad = np.zeros(n)
    for k_idx in active:
        w = max(corr[k_idx] - max_corr + 1.0, 0.05)
        offset = k_idx - (n - 1)
        n_pairs = n - abs(offset)
        if n_pairs <= 0:
            continue
        if offset >= 0:
            grad[offset:offset + n_pairs] += w * (1.0 - h[:n_pairs])
            grad[:n_pairs] -= w * h[offset:offset + n_pairs]
        else:
            a = -offset
            grad[:n_pairs] += w * (1.0 - h[a:a + n_pairs])
            grad[a:a + n_pairs] -= w * h[:n_pairs]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return None
    return grad / gnorm
