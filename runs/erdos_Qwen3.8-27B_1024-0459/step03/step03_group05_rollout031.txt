    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_all_shifts(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def overlap_grad(h, k_shift, n):
    """
    Gradient of the overlap at a specific shift.
    k_shift: index in 'full' correlation (0 to 2n-2)
    Actual index shift: k = k_shift - (n-1)
    """
    dx = 2.0 / n
    k = k_shift - (n - 1)
    grad = np.zeros(n)
    
    if k >= 0:
        # O(k) = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])*dx
        # dO/dh[j] for j in [0, n-1-k]: +dx*(1-h[j+k])
        # dO/dh[j] for j in [k, n-1]:   -dx*h[j-k]
        idx1 = np.arange(n - k)
        grad[idx1] += dx * (1.0 - h[idx1 + k])
        idx2 = np.arange(n - k, n)
        grad[idx2] += -dx * h[idx2 - k]
    elif k < 0:
        kk = -k
        # O(-kk) = sum_{i=kk}^{n-1} h[i]*(1-h[i-kk])*dx
        # dO/dh[j] for j in [kk, n-1]: +dx*(1-h[j-kk])
        # dO/dh[j] for j in [0, n-1-kk]: -dx*h[j+kk]
        idx1 = np.arange(kk, n)
        grad[idx1] += dx * (1.0 - h[idx1 - kk])
        idx2 = np.arange(n - kk)
        grad[idx2] += -dx * h[idx2 + kk]
    else:
        # k = 0: O(0) = sum h[i]*(1-h[i])*dx
        # dO/dh[j] = dx*(1-2*h[j])
        grad[:] = dx * (1.0 - 2.0 * h)
    
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(40):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def overlap_grad(h, k_shift, n):
    dx = 2.0 / n
    k = k_shift - (n - 1)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    elif k > 0:
        if k < n:
            grad[k:n] += dx * (1.0 - h[0:n - k])
            grad[0:n - k] -= dx * h[k:n]
    else:
        p = -k
        if p < n:
            grad[0:n - p] += dx * (1.0 - h[p:n])
            grad[p:n] -= dx * h[0:n - p]
    return grad


def smooth_max_obj(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    if top_k < len(corr):
        idx = np.argsort(corr)[-top_k:]
        m = corr[idx].max()

        return (m + np.log(np.sum(np.exp(beta * shifted))) / beta)
    else:
        m = np.max(corr)
        shifted = corr - m
        return (m + np.log(np.sum(np.exp(beta * shifted))) / beta)
