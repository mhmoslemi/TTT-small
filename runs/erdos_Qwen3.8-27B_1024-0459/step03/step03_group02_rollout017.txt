import numpy as np

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-15 and n > 0:
        # Spread among closest to 0.5
        order = np.argsort(np.abs(h - 0.5))
        for idx in order:
            delta = np.clip(diff, -(h[idx]), 1.0 - h[idx])
            h[idx] += delta
            diff -= delta
            if abs(diff) < 1e-15:
                break
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def smooth_max_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    # Numerically stable log-sum-exp
    shifted = np.minimum(beta * (corr - m), 50.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    val = (m + log_sum / beta) * dx
    
    # Gradient
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if weights[k_idx] < 1e-15:
            continue
        offset = k_idx - (n - 1)
        absoff = abs(offset)
        n_pairs = n - absoff
        if n_pairs <= 0:
            continue
        if offset >= 0:
            grad[offset:offset + n_pairs] += weights[k_idx] * (1.0 - x[:n_pairs])
            grad[:n_pairs] -= weights[k_idx] * x[offset:offset + n_pairs]
        else:
            grad[:n_pairs] += weights[k_idx] * (1.0 - x[absoff:absoff + n_pairs])
            grad[absoff:absoff + n_pairs] -= weights[k_idx] * x[:n_pairs]
    grad *= dx
    return val, grad

def smooth_max_obj(x, n, beta):
    val, _ = smooth_max_obj_and_grad(x, n, beta)
    return val

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        order = np.argsort(np.abs(h - 0.5))
        for idx in order:
            delta = np.clip(diff, -h[idx], 1.0 - h[idx])
            h[idx] += delta
            diff -= delta
            if abs(diff) < 1e-14:
                break

def lag_gradient(h, n, offset):
    grad = np.zeros(n)
    absoff = abs(offset)
    n_pairs = n - absoff
    if n_pairs <= 0:
        return grad
    if offset >= 0:
        grad[offset:offset + n_pairs] += (1.0 - h[:n_pairs])
        grad[:n_pairs] -= h[offset:offset + n_pairs]
    else:
        grad[:n_pairs] += (1.0 - h[absoff:absoff + n_pairs])
        grad[absoff:absoff + n_pairs] -= h[:n_pairs]
    return grad

def smooth_max_val(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 60.0)
    logsum = np.log(np.sum(np.exp(shifted)))
    return (m + logsum / beta) * dx

def smooth_max_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 60.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    weights = exp_shifted / wsum
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if weights[k_idx] < 1e-14:
            continue
        offset = k_idx - (n - 1)
        absoff = abs(offset)
        n_pairs = n - absoff
        if n_pairs <= 0:
            continue
        w = weights[k_idx]
        if offset >= 0:
            grad[offset:offset + n_pairs] += w * (1.0 - x[:n_pairs])
            grad[:n_pairs] -= w * x[offset:offset + n_pairs]
        else:
            grad[:n_pairs] += w * (1.0 - x[absoff:absoff + n_pairs])
            grad[absoff:absoff + n_pairs] -= w * x[:n_pairs]
    grad *= dx
    return grad
