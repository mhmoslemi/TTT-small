```python
def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):

            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
```

```python
def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n)

def project_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-13
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        # Distribute among interior points
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            idx = np.argsort(np.abs(h - 0.5))[:max(1, int(abs(diff)))]
            h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

                a = 0.0

                if j <= n - 1 - L:
                    a += (1.0 - h[j + L])
                # h[j] as h[i+L] (i=j-L): valid if 0 ≤ j-L ≤ n-1-L, i.e., L ≤ j ≤ n-1
                if L <= j:
                    a += (-h[j - L])
                
                if a != 0.0:
                    a_arr[fidx] = a
                    base_arr[fidx] = corr[fidx] - a * old
                    is_affected[fidx] = True
            
            # Negative lags: L = -1, ..., -(n-1): fidx = n-1+L = n-1-|L|
            for L in range(1, n):
                fidx = n - 1 - L  # lag = -L
                a = 0.0
                # corr[-L] = Σ_{i=L}^{n-1} h[i]*(1-h[i-L])
                # h[j] as h[i] (i=j): valid if L ≤ j ≤ n-1
                if j >= L:
                    a += (1.0 - h[j - L])
                # h[j] as h[i-L] (i=j+L): valid if L ≤ j+L ≤ n-1, i.e., 0 ≤ j ≤ n-1-L
                if j + L <= n - 1:
                    a += (-h[j + L])
                
                if a != 0.0:
                    a_arr[fidx] = a
                    base_arr[fidx] = corr[fidx] - a * old
                    is_affected[fidx] = True
            
            # Max of unaffected lags
            if is_affected.all():
                max_unaffected = 0.0  # Actually should be -inf conceptually, but corr ≥ 0
            else:
                max_unaffected = corr[~is_affected].max() if (~is_affected).any() else 0.0
            
            # 1D minimization: minimize f(x) = max(max_unaffected, max_{affected, non-quad}(a*x+base), quad_val)
            # where quad_val = x - x² + base_arr[n-1]
            # subject to 0 ≤ x ≤ 1
            
            # Use grid + ternary refinement
            best_x = old
            best_val = float('inf')
            
            # Coarse grid
            grid = np.linspace(0.0, 1.0, 40)
            for x in grid:
                val = max_unaffected
                # Quad term
                vq = x - x * x + base_arr[n-1]
                if vq > val:
                    val = vq
                # Linear terms
                for idx in np.where(is_affected)[0]:
                    if idx == n - 1:
                        continue  # already handled as quad
                    v = a_arr[idx] * x + base_arr[idx]
                    if v > val:
                        val = v
                if val < best_val:
                    best_val = val
                    best_x = x
            
            # Ternary refinement
            lo = max(0.0, best_x - 0.03)
            hi = min(1.0, best_x + 0.03)
            for _ in range(60):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                v1 = max_unaffected
                vq1 = m1 - m1*m1 + base_arr[n-1]
                if vq1 > v1: v1 = vq1
                for idx in np.where(is_affected)[0]:
                    if idx == n - 1: continue
                    v = a_arr[idx] * m1 + base_arr[idx]
                    if v > v1: v1 = v
                v2 = max_unaffected
                vq2 = m2 - m2*m2 + base_arr[n-1]
                if vq2 > v2: v2 = vq2
                for idx in np.where(is_affected)[0]:
                    if idx == n - 1: continue
                    v = a_arr[idx] * m2 + base_arr[idx]
                    if v > v2: v2 = v
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            
            new_x = (lo + hi) / 2.0
            h[j] = new_x
            
            # Update correlation incrementally
            delta = new_x - old
            for idx in np.where(is_affected)[0]:
                if idx == n - 1:
                    # Quad update: corr[n-1] changes from old*(1-old) to new_x*(1-new_x)
                    corr[n-1] += new_x * (1.0 - new_x) - old * (1.0 - old)
                else:
                    corr[idx] += a_arr[idx] * delta
        
        # End of sweep
        c5 = float(corr.max()) * 2.0 / n
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full').copy()
    
    return best_h
```

```python
affected_idx = np.where(is_affected)[0]
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-13
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            idx = np.argsort(np.abs(h - 0.5))[:max(1, int(np.ceil(abs(diff))))]
            h[idx] += diff / max(1, len(idx))
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_h(h_src, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_new = np.interp(xn, xp, h_src)
    return project_feasible(h_new, n_dst)

def coord_descent(h0, n, max_sweeps, budget_sec, t0):
    h = project_feasible(h0.copy(), n)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_h = h.copy()
    best_c5 = float(corr.max()) * 2.0 / n
    deadline = t0 + budget_sec

    for sweep in range(max_sweeps):
        if time.time() > deadline - 3:
            break

        for j in range(n):
            if time.time() > deadline - 3:
                break

            old = h[j]

            a_arr = np.zeros(2 * n - 1)
            base_arr = np.zeros(2 * n - 1)
            is_affected = np.zeros(2 * n - 1, dtype=bool)

            # Lag 0
            a_arr[n - 1] = 1.0
            base_arr[n - 1] = corr[n - 1] - old * (1.0 - old)
            is_affected[n - 1] = True

            # Positive lags
            for L in range(1, n):
                fidx = n - 1 + L
                a = 0.0
                if j <= n - 1 - L:
                    a += (1.0 - h[j + L])
                if L <= j:
                    a += (-h[j - L])
                if a != 0.0:
                    a_arr[fidx] = a
                    base_arr[fidx] = corr[fidx] - a * old
                    is_affected[fidx] = True

            # Negative lags
            for L in range(1, n):
                fidx = n - 1 - L
                a = 0.0
                if j >= L:
                    a += (1.0 - h[j - L])
                if j + L <= n - 1:
                    a += (-h[j + L])
                if a != 0.0:
                    a_arr[fidx] = a
                    base_arr[fidx] = corr[fidx] - a * old
                    is_affected[fidx] = True

            aff_idx = np.where(is_affected)[0]
            lin_idx = aff_idx[aff_idx != n - 1]
            a_vals = a_arr[lin_idx]
            base_vals = base_arr[lin_idx]
            base0 = base_arr[n - 1]

            if (~is_affected).any():
                max_unaff = float(corr[~is_affected].max())
            else:
                max_unaff = 0.0

            def eval_obj_1d(x):
                quad_v = x - x * x + base0
                lin_max = float((a_vals * x + base_vals).max()) if len(a_vals) > 0 else 0.0
                return max(max_unaff, quad_v, lin_max)

            # Coarse grid
            grid = np.linspace(0.0, 1.0, 50)
            best_x = old
            best_val = float('inf')
            for x in grid:
                val = eval_obj_1d(x)
                if val < best_val:
                    best_val = val
                    best_x = x

            # Ternary refinement
            lo = max(0.0, best_x - 0.04)
            hi = min(1.0, best_x + 0.04)
            for _ in range(70):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                v1 = eval_obj_1d(m1)
                v2 = eval_obj_1d(m2)
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            new_x = (lo + hi) / 2.0

            # Update h and corr incrementally
            h[j] = new_x
            delta = new_x - old
            for idx in aff_idx:
                if idx == n - 1:
                    corr[n - 1] += new_x * (1.0 - new_x) - old * (1.0 - old)
                else:
                    corr[idx] += a_arr[idx] * delta

        c5 = float(corr.max()) * 2.0 / n
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full').copy()

    return best_h

def slsqp_smooth_max(x_u, n, beta, constraint_sum):
    h = project_feasible(x_u, n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -100.0, 0.0)
    return (m + np.log(np.exp(shifted).sum()) / beta) * (2.0 / n)

def slsqp_topk(x_u, n, k):
    h = project_feasible(x_u, n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top = np.sort(corr)[::-1][:k]
    return float(top.mean()) * (2.0 / n)

def slsqp_polish(h0, n, budget_sec, t0):
    h = project_feasible(h0.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    deadline = t0 + budget_sec
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]

    betas = [100.0, 500.0, 2000.0, 10000.0]
    for beta in betas:
        if time.time() > deadline - 10:
            break
        try:
            res = minimize(
                lambda x: slsqp_smooth_max(x, n, beta, 1.0),
                h, method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-15, 'iprint': 0}
            )
            h = project_feasible(res.x, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        except Exception:
            pass

    for k in [3, 5, 10, 20]:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(
                lambda x: slsqp_topk(x, n, k),
                h, method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-15, 'iprint': 0}
            )
            h = project_feasible(res.x, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        except Exception:
            pass

    return best_h

def target_peak_reduction(h0, n, budget_sec, t0, rng):
    """Targeted local search: focus on reducing the top lags."""
    h = project_feasible(h0.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    deadline = t0 + budget_sec

    while time.time() < deadline - 5:
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = np.argsort(corr)[::-1][:10]
        max_lag = top_k[0] - (n - 1)

        # For the top lag, try to reduce correlation by adjusting pairs
        L_abs = abs(max_lag)
        if L_abs == 0:
            # Lag 0: reduce by making h more binary (push values toward 0 or 1)
            for j in range(n):
                if h[j] < 0.5:
                    h[j] = max(0.0, h[j] - 0.02)
                else:
                    h[j] = min(1.0, h[j] + 0.02)
            h = project_feasible(h, n)
        else:
            # For lag L: reduce corr by making h[i] and h[i+L] more similar
            step = 0.03
            for _ in range(200):
                if time.time() > deadline - 5:
                    break
                # Pick random pair at distance L
                if max_lag > 0:
                    i = rng.randint(0, n - L_abs)
                    j2 = i + L_abs
                else:
                    j2 = rng.randint(L_abs, n)
                    i = j2 - L_abs

                # If h[i] and h[j2] are "misaligned", push them closer
                diff_h = h[i] - h[j2]
                if abs(diff_h) > 0.1:
                    adjust = step * np.sign(diff_h)
                    h[i] -= adjust
                    h[j2] += adjust
                    h = np.clip(h, 0.0, 1.0)
                    h = project_feasible(h, n)

        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
        elif c5 > best_c5 + 0.01:
            h = best_h.copy()

    return best_h
```
