    """Project onto {h: 0<=h<=1, sum(h)=target_sum}."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to increase sum: increase elements that are < 1
            mask = h < 1.0 - 1e-12
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
        else:
            # Need to decrease sum: decrease elements that are > 0
            mask = h > 1e-12
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    h = np.clip(h, 0.0, 1.0)
    # Now enforce sum(h) = s
    # If sum(h) < s: need to add (s - sum(h)) spread among h[i] < 1
    # If sum(h) > s: need to remove (sum(h) - s) spread among h[i] > 0

            mask = h < 1.0 - 1e-12
            if not mask.any():

            h[mask] += diff / mask.sum()

            mask = h > 1e-12
            if not mask.any():
                break
            h[mask] += diff / mask.sum()

    except Exception:
        parent_n = 320
        parent_h = make_thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Projected gradient at parent resolution
    n = best_n
    h = best_h.copy()
    beta_sched = [50, 100, 300, 1000, 3000, 8000]
    h_new, obj_new = projected_gradient(h, n, beta_sched, max_iter=4000,
                                         step_init=0.02, momentum=0.85, time_limit=80)
    if obj_new < best_c5:
        best_c5 = obj_new
        best_h = h_new.copy()
        best_n = n

    # SLSQP refinement at parent n
    for beta in [500, 2000, 5000]:
        if remaining() < 10:
            break
        h_s = slsqp_smooth(best_h, best_n, beta, maxiter=250)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
            best_n = best_n

    # Active set
    if remaining() > 15:
        h_a, c5_a = active_set_refine(best_h, best_n, max_iters=400, time_limit=25)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a.copy()

    # Phase 2: Explore other resolutions
    n_candidates = sorted(set([
        best_n, best_n + 16, best_n + 32, best_n + 64,
        384, 400, 448, 480, 512, 560, 640
    ]))
    n_candidates = [nn for nn in n_candidates if 200 <= nn <= 700]
    # Remove best_n to try others first
    n_candidates = [nn for nn in n_candidates if nn != best_n] + [best_n]

    for n_try in n_candidates:
        if remaining() < 40:
            break
        # Interpolate from best
        h_interp = interp_to(best_h, best_n, n_try)
        h_feas = project_box_sum(h_interp, n_try / 2.0)
        c5_init = compute_c5(h_feas)

        # Quick PG
        beta_sched2 = [80, 300, 1000, 3000]
        h_opt, c5_opt = projected_gradient(h_feas, n_try, beta_sched2, max_iter=2500,
                                            step_init=0.015, momentum=0.85, time_limit=min(50, remaining() - 20))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_try

        # SLSQP on this resolution
        if remaining() > 20 and c5_opt < best_c5 + 0.005:
            h_s2 = slsqp_smooth(h_opt, n_try, 2000, maxiter=200)
            c5_s2 = compute_c5(h_s2)
            if c5_s2 < best_c5:
                best_c5 = c5_s2
                best_h = h_s2.copy()
                best_n = n_try

        # Active set
        if remaining() > 15:
            h_a2, c5_a2 = active_set_refine(best_h, best_n, max_iters=300, time_limit=15)
            if c5_a2 < best_c5:
                best_c5 = c5_a2
                best_h = h_a2.copy()

    # Phase 3: Intensive refinement at best resolution
    n = best_n
    h = best_h.copy()

    # More PG with finer beta schedule
    beta_sched3 = [200, 500, 1500, 4000, 10000, 20000]
    h_new3, obj_new3 = projected_gradient(h, n, beta_sched3, max_iter=5000,
                                           step_init=0.01, momentum=0.9,
                                           time_limit=min(120, remaining() - 30))
    if obj_new3 < best_c5:
        best_c5 = obj_new3
        best_h = h_new3.copy()

    # SLSQP with multiple betas
    for beta in [2000, 5000, 12000, 25000]:
        if remaining() < 8:
            break
        h_s3 = slsqp_smooth(best_h, n, beta, maxiter=350)
        c5_s3 = compute_c5(h_s3)
        if c5_s3 < best_c5:
            best_c5 = c5_s3
            best_h = h_s3.copy()

    # Active set final
    if remaining() > 10:
        h_a3, c5_a3 = active_set_refine(best_h, n, max_iters=600, time_limit=min(40, remaining() - 10))
        if c5_a3 < best_c5:
            best_c5 = c5_a3
            best_h = h_a3.copy()

    # Phase 4: Try a few more nearby resolutions with the best solution
    for n_near in [best_n - 4, best_n + 4, best_n + 8, best_n - 8]:
        if remaining() < 20:
            break
        if n_near < 100 or n_near > 800:
            continue
        h_near = interp_to(best_h, best_n, n_near)
        h_near = project_box_sum(h_near, n_near / 2.0)
        h_no, c5_no = projected_gradient(h_near, n_near, [500, 2000, 8000], max_iter=2000,
                                          step_init=0.012, momentum=0.88, time_limit=min(25, remaining() - 10))
        if c5_no < best_c5:
            best_c5 = c5_no
            best_h = h_no.copy()
            best_n = n_near

    # Phase 5: Final SLSQP polish
    for beta in [8000, 15000, 30000, 50000]:
        if remaining() < 5:
            break
        h_final = slsqp_smooth(best_h, best_n, beta, maxiter=400)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final validation
    best_h = project_box_sum(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
