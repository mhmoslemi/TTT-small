        h[i] = 1.0 if parity else 0.0
    # Fix sum to be n/2
    s = int(np.sum(h))
    if s < n // 2:
        zeros = np.where(h == 0)[0]
        h[zeros[:n//2 - s]] = 1.0
    elif s > n // 2:
        ones = np.where(h == 1)[0]
        h[ones[:s - n//2]] = 0.0
    return h

def corr_gradient(h, active_indices, n):
    """Compute gradient of sum of corr[k] for k in active_indices"""
    grad = np.zeros(n)
    for k in active_indices:
        if k <= n - 1:
            # corr[k] = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            np_ = n - k
            for i in range(np_):
                # d corr[k] / d h[i] includes contributions from both h[i] and h[i+k] terms
                pass
            # Vectorized:
            # d/dh[i] for i in [0, np_-1]: (1-h[i+k]) from h[i]*(1-h[i+k])
            # d/dh[j] for j in [k, n-1]: -h[j-k] from h[j-k]*(1-h[j])
            grad[:np_] += (1.0 - h[k:k+np_])
            grad[k:k+np_] -= h[:np_]
        else:
            # k > n-1, so k = (n-1) + m where m >= 1
            # This is negative shift m
            m = k - (n - 1)
            # corr[k] = sum_{i=m}^{n-1} h[i]*(1-h[i-m])
            np_ = n - m
            grad[m:m+np_] += (1.0 - h[:np_])
            grad[:np_] -= h[m:m+np_]
    return grad

def lse_obj(h, n, beta, active=None):
    """Log-sum-exp smoothed max of correlation"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    if active is not None and len(active) > 0:
        vals = corr[active]
    else:
        vals = corr
    m = np.max(vals)
    if beta < 1e-6:
        return m * (2.0/n)
    exps = np.exp(beta * (vals - m))
    return (m + np.log(np.sum(exps)) / beta) * (2.0/n)

def make_trunc_sine(n, freq, amp, phase):
    t = np.arange(n) / float(n)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * t + phase)
    return h
