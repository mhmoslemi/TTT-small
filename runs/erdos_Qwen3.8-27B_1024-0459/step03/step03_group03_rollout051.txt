    xs = (np.arange(ns) + 0.5) * (2.0 / ns)
    xd = (np.arange(nd) + 0.5) * (2.0 / nd)

    mx = float(np.max(c))

def _c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / n)


def _proj(h, n):

        if abs(d) < 1e-13:

def _interp(h, nd):
    ns = len(h)
    if ns == nd:
        return h.copy()
    xs = (np.arange(ns) + 0.5) * (2.0 / ns)
    xd = (np.arange(nd) + 0.5) * (2.0 / nd)
    return np.interp(xd, xs, h)


def _scon(x):

    sh = c - mx
    return (mx + np.log(np.sum(np.exp(b * sh))) / b) * (2.0 / n)

    n = len(h)
    if lag >= 0:
        for i in range(n - lag):
            g[i] += (1.0 - h[i + lag])
            g[i + lag] -= h[i]
    else:
        p = -lag
        for i in range(p, n):
            g[i] += (1.0 - h[i - p])
            g[i - p] -= h[i]


def _smgrad(h, bw):

        lag = int(k_idx) - (n - 1)
        tmp = np.zeros(n)
