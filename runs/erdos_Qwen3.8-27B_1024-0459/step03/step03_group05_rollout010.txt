                    continue
                noisy = contribs * (1.0 + 0.3 * rng.standard_normal(len(contribs)))
                idx = int(np.argmax(noisy))
                i, j = idx, idx + m
                d = 0.025
                if h_work[j] - d >= 0 and h_work[i] + d <= 1:
                    saved_i, saved_j = j, i
                    saved_vi, saved_vj = h_work[j], h_work[i]
                    h_work[j] -= d
                    h_work[i] += d
                else:
                    d = 0.012
                    if h_work[j] - d >= 0 and h_work[i] + d <= 1:
                        saved_i, saved_j = j, i
                        saved_vi, saved_vj = h_work[j], h_work[i]
                        h_work[j] -= d
                        h_work[i] += d
                    else:
                        continue

        elif r < 0.82:
            # Random pair transfer
            i = int(rng.randint(N))
            j = int(rng.randint(N))
            while j == i:
                j = int(rng.randint(N))
            d = rng.choice([0.025, -0.025, 0.012, -0.012, 0.04, -0.04, 0.015, -0.015])
            if 0 <= h_work[i] + d and 0 <= h_work[j] - d:
                saved_i, saved_j = i, j
                saved_vi, saved_vj = h_work[i], h_work[j]
                h_work[i] += d
                h_work[j] -= d
            else:
                continue

        elif r < 0.93:
            # Triple transfer
            i = int(rng.randint(N))
            j = int(rng.randint(N))
            ell = int(rng.randint(N))
            if i == j or i == ell or j == ell:
                continue
            d1 = rng.choice([0.015, -0.015, 0.01, -0.01])
            d2 = rng.choice([0.015, -0.015, 0.01, -0.01])
            d3 = -(d1 + d2)
            if (0 <= h_work[i] + d1 and 0 <= h_work[j] + d2 and 0 <= h_work[ell] + d3):
                saved_i, saved_j = i, j
                saved_vi, saved_vj = h_work[i], h_work[j]
                h_work[i] += d1
                h_work[j] += d2
                h_work[ell] += d3
            else:
                continue

        else:
            # Global kick
            n_pts = rng.randint(3, 8)
            for _ in range(n_pts):
                i = int(rng.randint(N))
                h_work[i] = max(0.0, min(1.0, h_work[i] + rng.randn() * 0.02))
            h_work = project_feasible(h_work, target)
            c5_work, _ = eval_c5(h_work)
            if c5_work < best_c5:
                best_c5 = c5_work
                best_h = h_work.copy()
                best_n = N
            continue

        c5_new, _ = eval_c5(h_work)

        # Accept/reject
        if c5_new <= c5_work:
            c5_work = c5_new
        elif rng.random() < np.exp(-(c5_new - c5_work) / temp):
            c5_work = c5_new
        else:
            # Revert
            if saved_i >= 0:
                h_work[saved_i] = saved_vi
                h_work[saved_j] = saved_vj
                if saved_j != saved_i:
                    pass

        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()
            best_n = N

        if it_count % 300 == 0:
            temp *= 0.94
            if temp < 0.0015:
                temp = 0.0015

        if it_count % 1000 == 0:
            print("  it {}: t={:.4f} cur={:.6f} best={:.6f}".format(
                it_count, temp, c5_work, best_c5))

    print("Phase 1: iters={}, C5={:.6f}".format(it_count, best_c5))

    # Phase 2: Fine local search at top shifts
    if tl() > 15:
        h = best_h.copy()
        step = 0.004
        stuck = 0

        while tl() > 4:
            c5_curr, corr_curr = eval_c5(h)
            improved_sweep = False

            top_idx = int(np.argmax(corr_curr))
            k = top_idx - (N - 1)

            if k != 0:
                if k > 0:
                    contribs = h[k:N] * (1.0 - h[0:N - k])
                    if len(contribs) > 1:
                        order = np.argsort(-contribs)
                        for top_i in order[:5]:
                            i = int(top_i)
                            j = i + k
                            for d in [step, step * 0.5, step * 1.5]:
                                if h[j] - d >= 0 and h[i] + d <= 1:
                                    h[j] -= d
                                    h[i] += d
                                    c5_t, _ = eval_c5(h)
                                    if c5_t < best_c5:
                                        best_c5 = c5_t
                                        best_h = h.copy()
                                        improved_sweep = True
                                    else:
                                        h[j] += d
                                        h[i] -= d
                                if improved_sweep:
                                    break
                            if improved_sweep:
                                break
                else:
                    m = -k
                    contribs = h[0:N - m] * (1.0 - h[m:N])
                    if len(contribs) > 1:
                        order = np.argsort(-contribs)
                        for top_i in order[:5]:
                            i = int(top_i)
                            j = i + m
                            for d in [step, step * 0.5, step * 1.5]:
                                if h[j] - d >= 0 and h[i] + d <= 1:
                                    h[j] -= d
                                    h[i] += d
                                    c5_t, _ = eval_c5(h)
                                    if c5_t < best_c5:
                                        best_c5 = c5_t
                                        best_h = h.copy()
                                        improved_sweep = True
                                    else:
                                        h[j] += d
                                        h[i] -= d
                                if improved_sweep:
                                    break
                            if improved_sweep:
                                break

            # Also try second-worst shift
            if not improved_sweep and tl() > 6:
                sorted_idx = np.argsort(-corr_curr)
                for si in sorted_idx[1:3]:
                    if improved_sweep:
                        break
                    k2 = int(si) - (N - 1)
                    if k2 == 0:
                        continue
                    if k2 > 0:
                        contribs2 = h[k2:N] * (1.0 - h[0:N - k2])
                        if len(contribs2) > 1:
                            top_i2 = int(np.argmax(contribs2))
                            i2, j2 = top_i2, top_i2 + k2
                            for d in [step, step * 0.5]:
                                if h[j2] - d >= 0 and h[i2] + d <= 1:
                                    h[j2] -= d
                                    h[i2] += d
                                    c5_t, _ = eval_c5(h)
                                    if c5_t < best_c5:
                                        best_c5 = c5_t
                                        best_h = h.copy()
                                        improved_sweep = True
                                    else:
                                        h[j2] += d
                                        h[i2] -= d
                                if improved_sweep:
                                    break
                    else:
                        m2 = -k2
                        contribs2 = h[0:N - m2] * (1.0 - h[m2:N])
                        if len(contribs2) > 1:
                            top_i2 = int(np.argmax(contribs2))
                            i2, j2 = top_i2, top_i2 + m2
                            for d in [step, step * 0.5]:
                                if h[j2] - d >= 0 and h[i2] + d <= 1:
                                    h[j2] -= d
                                    h[i2] += d
                                    c5_t, _ = eval_c5(h)
                                    if c5_t < best_c5:
                                        best_c5 = c5_t
                                        best_h = h.copy()
                                        improved_sweep = True
                                    else:
                                        h[j2] += d
                                        h[i2] -= d
                                if improved_sweep:
                                    break

            if not improved_sweep:
                # Random fine perturbation
                for _ in range(5):
                    i = int(rng.randint(N))
                    j = int(rng.randint(N))
                    while j == i:
                        j = int(rng.randint(N))
                    d = rng.choice([step, -step, step * 0.5, -step * 0.5])
                    if 0 <= h[i] + d and 0 <= h[j] - d:
                        h[i] += d
                        h[j] -= d
                        c5_t, _ = eval_c5(h)
                        if c5_t < best_c5:
                            best_c5 = c5_t
                            best_h = h.copy()
                            improved_sweep = True
                            break
                        else:
                            h[i] -= d
                            h[j] += d

            if not improved_sweep:
                stuck += 1
                if stuck > 30:
                    step *= 0.85
                    stuck = 0
                    if step < 0.0008:
                        break
                h = best_h.copy()
            else:
                h = best_h.copy()

        print("Phase 2: C5={:.6f}".format(best_c5))

    # Phase 3: Even finer polish
    if tl() > 8:
        h = best_h.copy()
        micro_step = 0.001
        for it in range(3000):
            if tl() < 2:
                break
            c5_curr, corr_curr = eval_c5(h)
            top_idx = int(np.argmax(corr_curr))
            k = top_idx - (N - 1)
            if k == 0:
                break
            if k > 0:
                contribs = h[k:N] * (1.0 - h[0:N - k])
                if len(contribs) < 2:
                    break
                idx = int(np.argmax(contribs))
                i, j = idx, idx + k
            else:
                m = -k
                contribs = h[0:N - m] * (1.0 - h[m:N])
                if len(contribs) < 2:
                    break
                idx = int(np.argmax(contribs))
                i, j = idx, idx + m

            improved = False
            for d in [micro_step, micro_step * 2, micro_step * 0.5]:
                if h[j] - d >= 0 and h[i] + d <= 1:
                    h[j] -= d
                    h[i] += d
                    c5_t, _ = eval_c5(h)
                    if c5_t < best_c5:
                        best_c5 = c5_t
                        best_h = h.copy()
                        improved = True
                    else:
                        h[j] += d
                        h[i] -= d
                if improved:
                    break
            if not improved:
                # Try random micro-moves
                i2 = int(rng.randint(N))
                j2 = int(rng.randint(N))
                while j2 == i2:
                    j2 = int(rng.randint(N))
                d = micro_step * rng.choice([1, -1, 2, -2])
                if 0 <= h[i2] + d and 0 <= h[j2] - d:
                    h[i2] += d
                    h[j2] -= d
                    c5_t, _ = eval_c5(h)
                    if c5_t < best_c5:
                        best_c5 = c5_t
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i2] -= d
                        h[j2] += d
                if not improved:
                    h = best_h.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5, _ = eval_c5(best_h)

    try:
        e = evaluate_erdos_solution(best_h)
        if e < final_c5:
            final_c5 = e
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
