    return float(np.max(corr) * dx)


def get_remaining(t0, budget_s):
    return max(0.0, budget_s - (time.time() - t0) - 5.0)


def smooth_obj_p(x):
    beta = _opt_params['beta']
    n_pts = _opt_params['n_pts']
    target = _opt_params['target']
    pen_w = _opt_params['pen_w']
    dx = 2.0 / n_pts
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    shifted = corr - m
    sm = (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx
    se = np.sum(x) - target
    return sm + pen_w * se * se


def smooth_obj_c(x):
    beta = _opt_params['beta']
    n_pts = _opt_params['n_pts']
    dx = 2.0 / n_pts
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    shifted = corr - m
    return (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx


def sum_eq_fun(x):
    return np.sum(x) - len(x) / 2.0


def sum_eq_jac(x):
    return np.ones(len(x))


def interpolate_h(h_src, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_new = np.interp(xn, xp, h_src)
    return project_feasible(h_new, n_dst / 2.0)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = np.array(initial_h_values, dtype=float).copy()
    best_n = len(best_h)
    best_c5 = compute_c5(best_h)
    print("Parent: n={}, C5={:.8f}".format(best_n, best_c5))

    # Phase 1: L-BFGS-B at parent resolution with increasing beta
    n = best_n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h_work = best_h.copy()

    for beta in [200, 500, 1000, 3000, 8000, 20000]:
        if get_remaining(t0, budget_s) < 5:
            break
        _opt_params['beta'] = beta
        _opt_params['n_pts'] = n
        _opt_params['target'] = target
        _opt_params['pen_w'] = 1000.0
        try:
            res = minimize(smooth_obj_p, h_work, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 500, 'ftol': 1e-20, 'gtol': 1e-18, 'maxls': 50})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  P1 beta={}: C5={:.8f}".format(beta, c5_new))
            h_work = h_new.copy()
        except Exception:
            pass

    # Phase 2: Multi-resolution L-BFGS-B
    if get_remaining(t0, budget_s) > 200:
        base_h = best_h.copy()
        base_n = best_n

        res_candidates = []
        for d in [20, 40, 60, 80, 100, 120, 140, -20, -40]:
            nn = base_n + d
            if 100 <= nn <= 500 and nn % 2 == 0:
                res_candidates.append(nn)
        res_candidates = sorted(set(res_candidates))

        for n_try in res_candidates:
            if get_remaining(t0, budget_s) < 25:
                break
            target_t = n_try / 2.0
            h0 = interpolate_h(base_h, base_n, n_try)
            bounds_t = [(0.0, 1.0)] * n_try

            for beta in [500, 2000, 8000, 20000]:
                if get_remaining(t0, budget_s) < 5:
                    break
                _opt_params['beta'] = beta
                _opt_params['n_pts'] = n_try
                _opt_params['target'] = target_t
                _opt_params['pen_w'] = 1000.0
                try:
                    res = minimize(smooth_obj_p, h0, method='L-BFGS-B',
                                   bounds=bounds_t,
                                   options={'maxiter': 500, 'ftol': 1e-20, 'gtol': 1e-18, 'maxls': 50})
                    h_new = project_feasible(res.x, target_t)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        best_n = n_try
                        base_h = h_new.copy()
                        base_n = n_try
                        print("  P2 n={}, beta={}: C5={:.8f}".format(n_try, beta, c5_new))
                    h0 = h_new.copy()
                except Exception:
                    pass

    # Phase 3: SLSQP with exact constraint at best resolution
    if get_remaining(t0, budget_s) > 80:
        n = best_n
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        h_work = best_h.copy()
        cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]

        for beta in [1000, 3000, 10000, 30000]:
            if get_remaining(t0, budget_s) < 5:
                break
            _opt_params['beta'] = beta
            _opt_params['n_pts'] = n
            try:
                res = minimize(smooth_obj_c, h_work, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               maxiter=800, ftol=1e-20)
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  P3 SLSQP beta={}: C5={:.8f}".format(beta, c5_new))
                h_work = h_new.copy()
            except Exception:
                pass

    # Phase 4: Aggressive coordinate descent with pair moves
    if get_remaining(t0, budget_s) > 150:
        n = best_n
        target = n / 2.0
        h = best_h.copy()
        deltas = [0.01, 0.005, 0.02, 0.003, 0.001, 0.0005, 0.05, 0.015, 0.008]

        for sweep in range(200):
            if get_remaining(t0, budget_s) < 3:
                break
            improved = False
            num_trials = 3000
            for _ in range(num_trials):
                if get_remaining(t0, budget_s) < 1:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                direction = 1.0 if h[i] < h[j] else -1.0
                for delta in deltas:
                    di = direction * delta
                    dj = -direction * delta
                    if h[i] + di < -1e-9 or h[i] + di > 1.0 + 1e-9:
                        continue
                    if h[j] + dj < -1e-9 or h[j] + dj > 1.0 + 1e-9:
                        continue
                    h[i] += di
                    h[j] += dj
                    c5_test = compute_c5(h)
                    if c5_test < best_c5:
                        best_c5 = c5_test
                        best_h = h.copy()
                        improved = True
                        break
                    h[i] -= di
                    h[j] -= dj
            if not improved:
                h = best_h.copy()

    # Phase 5: Nelder-Mead local polish
    if get_remaining(t0, budget_s) > 60:
        n = best_n
        target = n / 2.0
        _opt_params['beta'] = 5000
        _opt_params['n_pts'] = n
        _opt_params['target'] = target
        _opt_params['pen_w'] = 2000.0

        # Try from slightly perturbed starting points
        for attempt in range(3):
            if get_remaining(t0, budget_s) < 10:
                break
            if attempt == 0:
                h_start = best_h.copy()
            else:
                h_start = best_h + rng.normal(0, 0.005 * (attempt + 1), n)
                h_start = project_feasible(h_start, target)

            try:
                res = minimize(smooth_obj_p, h_start, method='Nelder-Mead',
                               options={'maxiter': 5000, 'xatol': 1e-10, 'fatol': 1e-14, 'adaptive': True})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  P5 NM attempt={}: C5={:.8f}".format(attempt, c5_new))
            except Exception:
                pass

    # Phase 6: Very fine coordinate descent
    if get_remaining(t0, budget_s) > 60:
        n = best_n
        target = n / 2.0
        h = best_h.copy()
        fine_deltas = [0.002, -0.002, 0.001, -0.001, 0.0005, -0.0005, 0.003, -0.003]

        for sweep in range(150):
            if get_remaining(t0, budget_s) < 3:
                break
            improved = False
            order = rng.permutation(n)
            for i in order:
                if get_remaining(t0, budget_s) < 1:
                    break
                for delta in fine_deltas:
                    if h[i] + delta < 0 or h[i] + delta > 1:
                        continue
                    # Compensate at a random j
                    h[i] += delta
                    rem = target - np.sum(h)
                    # Find j with room in the right direction
                    if rem > 0:
                        candidates = [j for j in order[:50] if j != i and 1.0 - h[j] > abs(rem) + 1e-10]
                    else:
                        candidates = [j for j in order[:50] if j != i and h[j] > abs(rem) + 1e-10]
                    if not candidates:
                        mask = np.ones(n, dtype=bool)
                        mask[i] = False
                        if rem > 0:
                            candidates = np.where(mask & (1.0 - h > abs(rem) + 1e-10))[0].tolist()
                        else:
                            candidates = np.where(mask & (h > abs(rem) + 1e-10))[0].tolist()
                    if not candidates:
                        h[i] -= delta
                        continue
                    j = candidates[0]
                    h[j] += rem
                    c5_test = compute_c5(h)
                    if c5_test < best_c5:
                        best_c5 = c5_test
                        best_h = h.copy()
                        improved = True
                        break
                    h[j] -= rem
                    h[i] -= delta
            if not improved:
                h = best_h.copy()

    # Phase 7: Final SLSQP polish with very high beta
    if get_remaining(t0, budget_s) > 20:
        n = best_n
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]

        for beta in [10000, 30000, 100000]:
            if get_remaining(t0, budget_s) < 3:
                break
            _opt_params['beta'] = beta
            _opt_params['n_pts'] = n
            try:
                res = minimize(smooth_obj_c, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               maxiter=2000, ftol=1e-22)
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  P7 SLSQP beta={}: C5={:.8f}".format(beta, c5_new))
            except Exception:
                pass

    # Phase 8: Random restart with quick optimization
    if get_remaining(t0, budget_s) > 60:
        base_h = best_h.copy()
        base_n = best_n

        for restart in range(3):
            if get_remaining(t0, budget_s) < 20:
                break
            # Create a perturbed version
            h_rand = base_h + rng.normal(0, 0.05, base_n)
            h_rand = project_feasible(h_rand, base_n / 2.0)

            # Quick L-BFGS-B
            bounds_r = [(0.0, 1.0)] * base_n
            _opt_params['beta'] = 2000
            _opt_params['n_pts'] = base_n
            _opt_params['target'] = base_n / 2.0
            _opt_params['pen_w'] = 1000.0
            try:
                res = minimize(smooth_obj_p, h_rand, method='L-BFGS-B',
                               bounds=bounds_r,
                               options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-16})
                h_new = project_feasible(res.x, base_n / 2.0)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  P8 restart {}: C5={:.8f}".format(restart, c5_new))
            except Exception:
                pass

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
