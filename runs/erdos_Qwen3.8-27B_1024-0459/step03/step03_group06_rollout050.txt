def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        unsat = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(unsat)
        if cnt > 0:
            h[unsat] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def smooth_max_obj(x, n, beta, target_sum):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    sm = (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx
    penalty = 1000.0 * (np.sum(x) - target_sum) ** 2
    return sm + penalty


def smooth_max_grad(x, n, beta, target_sum):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    wsum = np.sum(weights)
    weights /= wsum

    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = weights[j]
        if wj < 1e-14:
            continue
        s = j - (n - 1)
        if s >= 0:
            lo = max(s, 0)
            hi = min(n, n - s + 0)
            for m2 in range(s, n):
                grad[m2] += wj * (1.0 - x[m2 - s])
            for m2 in range(0, n - s):
                grad[m2] += wj * (-x[m2 + s])
        else:
            sp = -s
            for m2 in range(0, n - sp):
                grad[m2] += wj * (1.0 - x[m2 + sp])
            for m2 in range(sp, n):
                grad[m2] += wj * (-x[m2 - sp])
    grad *= dx
    pen_coeff = 1000.0
    grad += 2.0 * pen_coeff * (np.sum(x) - target_sum)
    return grad


def topk_smooth_obj(x, n, top_k, beta, target_sum):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    tv = corr[top_idx]
    m = np.max(tv)
    sm = (m + np.log(np.sum(np.exp(beta * (tv - m)))) / beta) * dx
    penalty = 1000.0 * (np.sum(x) - target_sum) ** 2
    return sm + penalty


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    return h


def make_periodic_anti(n, period):
    h = np.zeros(n)
    for i in range(n):
        p = i % period
        h[i] = 1.0 if p < period // 2 else 0.0
    return h


def make_multi_freq(n, rng, n_freqs=4):
    t = np.arange(n) / float(n)
    h = 0.5
    for _ in range(n_freqs):
        freq = rng.randint(2, 15)
        amp = rng.uniform(0.08, 0.2)
        phase = rng.rand() * 2.0 * np.pi
        h += amp * np.sin(2.0 * np.pi * freq * t + phase)
    return np.clip(h, 0.0, 1.0)


def greedy_binary(h, n, max_sweeps=50, rng=None):
    """Greedy binary local search: try flipping each position, keep if improves C5."""
    dx = 2.0 / n
    corr = compute_corr(h)
    best_c5 = np.max(corr) * dx
    target_ones = n // 2
    num_ones = int(np.sum(h > 0.5))

    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        order = np.arange(n)
        if rng is not None:
            order = rng.permutation(n)
        for m in order:
            if remaining_time() < 3:
                break
            # Try opposite value
            new_val = 1.0 - h[m]
            h[m] = new_val
            new_corr = compute_corr(h)
            new_c5 = np.max(new_corr) * dx
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                corr = new_corr
                improved = True
            else:
                h[m] = 1.0 - new_val
        if not improved:
            break
    return h, best_c5


def swap_sa(h, n, iterations, rng, t_start=0.006, t_end=0.00003):
    """Simulated annealing with swap moves (maintain n/2 ones)."""
    dx = 2.0 / n
    corr = compute_corr(h)
    best_c5 = np.max(corr) * dx
    current_c5 = best_c5
    best_h = h.copy()

    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    if not ones or not zeros:
        return best_h, best_c5

    for it in range(iterations):
        if remaining_time() < 4:
            break
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))
        oa, zb = ones[i], zeros[j]

        h[oa] = 0.0
        h[zb] = 1.0
        new_corr = compute_corr(h)
        new_c5 = np.max(new_corr) * dx

        T = t_start * (t_end / t_start) ** (float(it) / max(float(iterations), 1.0))
        accept = new_c5 < current_c5 or (T > 1e-18 and rng.rand() < np.exp(-(new_c5 - current_c5) / T))

        if accept:
            current_c5 = new_c5
            corr = new_corr
            ones[i] = zb
            zeros[j] = oa
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oa] = 1.0
            h[zb] = 0.0

    return best_h, best_c5


def continuous_optimize(h0, n, max_time, rng, target_sum):
    """Run L-BFGS-B with adaptive beta schedule."""
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    bounds = [(0.0, 1.0)] * n
    t0 = time.time()

    for beta in [20, 50, 150, 400, 800, 2000]:
        if time.time() - t0 > max_time * 0.8 or remaining_time() < 5:
            break
        try:
            res = minimize(
                lambda x: smooth_max_obj(x, n, beta, target_sum),
                best_h,
                jac=lambda x: smooth_max_grad(x, n, beta, target_sum),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-10,
                         'iprint': 0, 'maxls': 60, 'maxfun': 8000}
            )
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    # Top-k focused optimization
    if time.time() - t0 < max_time * 0.5 and remaining_time() > 5:
        for top_k in [4, 8]:
            try:
                res = minimize(
                    lambda x: topk_smooth_obj(x, n, top_k, 200.0, target_sum),
                    best_h,
                    method='L-BFGS-B',
                    bounds=bounds,
                    options={'maxiter': 400, 'ftol': 1e-14, 'iprint': 0,
                             'maxls': 50, 'maxfun': 6000}
                )
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
            except Exception:
                pass

    return best_h, best_c5


def coordinate_descent(h, n, target_sum, max_rounds=10, rng=None):
    """Coordinate descent: try setting each coordinate to a few candidate values."""
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    h = project_feasible(h, n)

    for rnd in range(max_rounds):
        if remaining_time() < 4:
            break
        improved = False
        order = np.arange(n)
        if rng is not None:
            order = rng.permutation(n)
        for i in order:
            if remaining_time() < 2:
                break
            old_val = h[i]
            best_local = best_c5
            best_h_local = h.copy()

            candidates = [0.0, 1.0]
            if 0.1 < old_val < 0.9:
                candidates.extend([0.5, max(0.0, old_val - 0.3), min(1.0, old_val + 0.3)])

            for cv in candidates:
                if abs(cv - old_val) < 1e-9:
                    continue
                h[i] = cv
                h_feas = project_feasible(h, n)
                c5_t = compute_c5(h_feas)
                if c5_t < best_local - 1e-14:
                    best_local = c5_t
                    best_h_local = h_feas.copy()

            h = best_h_local.copy()
            if best_local < best_c5 - 1e-14:
                best_c5 = best_local
                h = best_h_local.copy()
                improved = True

        if not improved:
            break

    return project_feasible(h, n), best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Phase 0: Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.7f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 160
        best_h = project_feasible(make_thue_morse(best_n), best_n)
        best_c5 = compute_c5(best_h)
        print("No parent. TM: n={}, C5={:.7f}".format(best_n, best_c5))

    # Phase 1: Fast binary greedy on parent resolution
    n1 = best_n
    print("\n--- Phase 1: Greedy binary (n={}) ---".format(n1))
    h_bin = np.where(best_h > 0.5, 1.0, 0.0)
    # Fix sum
    target_ones = n1 // 2
    num_ones = int(np.sum(h_bin > 0.5))
    if num_ones != target_ones:
        if num_ones > target_ones:
            ones_idx = np.where(h_bin > 0.5)[0]
            for i in range(num_ones - target_ones):
                h_bin[ones_idx[i]] = 0.0
        else:
            zeros_idx = np.where(h_bin <= 0.5)[0]
            for i in range(target_ones - num_ones):
                h_bin[zeros_idx[i]] = 1.0
    best_h_b, best_c5_b = greedy_binary(h_bin, n1, max_sweeps=30, rng=rng)
    if best_c5_b < best_c5:
        best_c5 = best_c5_b
        best_h = best_h_b.copy()
        print("  Greedy: C5={:.7f}".format(best_c5))

    # Phase 2: SA on binary
    if remaining_time() > 60:
        print("\n--- Phase 2: SA binary (n={}) ---".format(n1))
        for trial in range(3):
            if remaining_time() < 20:
                break
            h_sa = best_h_b.copy() if trial == 0 else np.where(best_h > 0.5, 1.0, 0.0)
            if trial > 0:
                # Perturb
                nswaps = rng.randint(3, 15)
                ones = list(np.where(h_sa > 0.5)[0])
                zeros = list(np.where(h_sa <= 0.5)[0])
                for _ in range(nswaps):
                    a = rng.randint(len(ones))
                    b = rng.randint(len(zeros))
                    h_sa[ones[a]], h_sa[zeros[b]] = h_sa[zeros[b]], h_sa[ones[a]]
                    ones[a], zeros[b] = zeros[b], ones[a]
            iters = int(min(20000, remaining_time() * 8))
            h_sa, c5_sa = swap_sa(h_sa, n1, iters, rng, t_start=0.005, t_end=0.00002)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                print("  SA trial {}: C5={:.7f}".format(trial, c5_sa))

    # Phase 3: Continuous optimization at parent resolution
    if remaining_time() > 100:
        print("\n--- Phase 3: Continuous opt (n={}) ---".format(n1))
        target_sum = n1 / 2.0
        inits = [best_h.copy()]
        # Add a few structured inits
        inits.append(project_feasible(make_thue_morse(n1), n1))
        inits.append(project_feasible(make_periodic_anti(n1, 7), n1))
        inits.append(project_feasible(make_periodic_anti(n1, 9), n1))
        inits.append(project_feasible(make_multi_freq(n1, rng, 5), n1))

        for h0 in inits:
            if remaining_time() < 30:
                break
            h_c, c5_c = continuous_optimize(h0, n1, max_time=80.0, rng=rng,
                                            target_sum=target_sum)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c
                best_n = n1
                print("  Cont: C5={:.7f}".format(c5_c))

    # Phase 4: Multi-resolution - try higher n
    if remaining_time() > 200:
        for n2 in [200, 256, 320]:
            if remaining_time() < 80:
                break
            print("\n--- Phase 4: n={} ---".format(n2))

            h_interp = interp_to(best_h, best_n, n2)
            h_interp = project_feasible(h_interp, n2)
            target_sum = n2 / 2.0

            inits2 = [h_interp]
            # Binary projection
            h_bin2 = np.where(h_interp > 0.5, 1.0, 0.0)
            num_o2 = int(np.sum(h_bin2 > 0.5))
            if num_o2 != n2 // 2:
                if num_o2 > n2 // 2:
                    oi = np.where(h_bin2 > 0.5)[0]
                    for k in range(num_o2 - n2 // 2):
                        h_bin2[oi[k]] = 0.0
                else:
                    zi = np.where(h_bin2 <= 0.5)[0]
                    for k in range(n2 // 2 - num_o2):
                        h_bin2[zi[k]] = 1.0
            inits2.append(h_bin2)
            inits2.append(project_feasible(make_thue_morse(n2), n2))
            inits2.append(project_feasible(make_multi_freq(n2, rng, 5), n2))

            for h0 in inits2:
                if remaining_time() < 25:
                    break
                # Quick continuous opt
                h_c, c5_c = continuous_optimize(h0, n2, max_time=40.0, rng=rng,
                                                target_sum=target_sum)
                if c5_c < best_c5:
                    best_c5 = c5_c
                    best_h = h_c
                    best_n = n2
                    print("  n={} cont: C5={:.7f}".format(n2, c5_c))

                # If continuous improved, do binary greedy on it
                if remaining_time() > 15 and c5_c < best_c5 + 0.001:
                    h_b2 = np.where(h_c > 0.5, 1.0, 0.0)
                    num_o = int(np.sum(h_b2 > 0.5))
                    if num_o != n2 // 2:
                        if num_o > n2 // 2:
                            oi = np.where(h_b2 > 0.5)[0]
                            for k in range(num_o - n2 // 2):
                                h_b2[oi[k]] = 0.0
                        else:
                            zi = np.where(h_b2 <= 0.5)[0]
                            for k in range(n2 // 2 - num_o):
                                h_b2[zi[k]] = 1.0
                    h_b2, c5_b2 = greedy_binary(h_b2, n2, max_sweeps=15, rng=rng)
                    if c5_b2 < best_c5:
                        best_c5 = c5_b2
                        best_h = h_b2
                        best_n = n2
                        print("  n={} bin: C5={:.7f}".format(n2, c5_b2))

    # Phase 5: Coordinate descent polish at best resolution
    if remaining_time() > 30:
        n_pol = best_n
        print("\n--- Phase 5: Coord descent (n={}) ---".format(n_pol))
        h_cd, c5_cd = coordinate_descent(best_h, n_pol, n_pol / 2.0,
                                         max_rounds=6, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
            print("  CD: C5={:.7f}".format(c5_cd))

    # Phase 6: Final binary greedy + SA at best resolution
    if remaining_time() > 20:
        n_fin = best_n
        h_fb = np.where(best_h > 0.5, 1.0, 0.0)
        num_o = int(np.sum(h_fb > 0.5))
        if num_o != n_fin // 2:
            if num_o > n_fin // 2:
                oi = np.where(h_fb > 0.5)[0]
                for k in range(num_o - n_fin // 2):
                    h_fb[oi[k]] = 0.0
            else:
                zi = np.where(h_fb <= 0.5)[0]
                for k in range(n_fin // 2 - num_o):
                    h_fb[zi[k]] = 1.0
        h_fb, c5_fb = greedy_binary(h_fb, n_fin, max_sweeps=10, rng=rng)
        if c5_fb < best_c5:
            best_c5 = c5_fb
            best_h = h_fb
            print("  Final bin: C5={:.7f}".format(best_c5))

        if remaining_time() > 15:
            iters = int(min(8000, remaining_time() * 6))
            h_sa, c5_sa = swap_sa(best_h, n_fin, iters, rng, t_start=0.003, t_end=0.00001)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa
                print("  Final SA: C5={:.7f}".format(best_c5))

    # Phase 7: Ultra-fine continuous polish
    if remaining_time() > 15:
        n_pol = best_n
        target_sum = n_pol / 2.0
        bounds = [(0.0, 1.0)] * n_pol
        for beta in [2000, 5000]:
            if remaining_time() < 8:
                break
            try:
                res = minimize(
                    lambda x: smooth_max_obj(x, n_pol, beta, target_sum),
                    best_h,
                    jac=lambda x: smooth_max_grad(x, n_pol, beta, target_sum),
                    method='L-BFGS-B',
                    bounds=bounds,
                    options={'maxiter': 600, 'ftol': 1e-15, 'gtol': 1e-11,
                             'iprint': 0, 'maxls': 80, 'maxfun': 10000}
                )
                h_new = project_feasible(res.x, n_pol)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
                    print("  Ultra polish beta={}: C5={:.7f}".format(beta, c5_new))
            except Exception:
                pass

    # Final: project and evaluate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
