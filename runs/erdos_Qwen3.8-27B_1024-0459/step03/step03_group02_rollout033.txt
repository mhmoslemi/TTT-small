def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

            h[mask] += diff / cnt
        else:

    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)

    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def smooth_max_grad(x):

    beta = _BETA_GLOBAL

def lag_gradient(h, n, offset):
    """Gradient of the k-th lag correlation value w.r.t. h."""
    grad = np.zeros(n)
    if offset >= 0:
        n_pairs = n - offset
        if n_pairs <= 0:
            return grad
        # c = sum_{i=0}^{n_pairs-1} h[i] * (1-h[i+offset])
        # d c / dh[j] = (1-h[j+offset]) for j < n_pairs
        #              + (-h[j-offset]) for j >= offset
        grad[:n_pairs] += (1.0 - h[offset:offset + n_pairs])
        grad[offset:] -= h[:n_pairs]
    else:
        abs_off = -offset
        n_pairs = n - abs_off
        if n_pairs <= 0:
            return grad
        # c = sum_{i=abs_off}^{n-1} h[i] * (1-h[i-abs_off])
        #   = sum_{i=0}^{n_pairs-1} h[i+abs_off] * (1-h[i])
        # d c / dh[j] = (1-h[j-abs_off]) for j >= abs_off (as the first factor)
        #              + (-h[j+abs_off]) for j < n_pairs (as the second factor)
        grad[abs_off:] += (1.0 - h[:n_pairs])
        grad[:n_pairs] -= h[abs_off:abs_off + n_pairs]
    return grad

    dx = 2.0 / n
    max_val = np.max(corr) * dx

    m = len(active)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-18:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = np.argsort(h)[:min(10, n)]
                for i in idx:
                    adj = min(0.01, diff)
                    h[i] = min(1.0, h[i] + adj)
                    diff -= adj
                    if abs(diff) < 1e-15:
                        break
            else:
                idx = np.argsort(h)[-min(10, n):]
                for i in idx:
                    adj = max(-0.01, diff)
                    h[i] = max(0.0, h[i] + adj)
                    diff -= adj
                    if abs(diff) < 1e-15:
                        break
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15 and n > 0:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 1:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)

        if abs(diff2) > 1e-15:

def smooth_max_obj(x):
    n = _N_GLOBAL
    beta = _BETA_GLOBAL
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def smooth_max_grad(x):
    n = _N_GLOBAL
    beta = _BETA_GLOBAL
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    if total < 1e-300:
        return np.zeros(n)
    weights = exp_vals / total
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-10:
            continue
        offset = k_idx - (n - 1)
        grad += weights[k_idx] * lag_grad(x, n, offset)
    return grad * dx

def constraint_sum(x):
    return np.sum(x) - _N_GLOBAL / 2.0

def constraint_sum_jac(x):
    return np.ones(_N_GLOBAL) / _N_GLOBAL

def lag_grad(h, n, offset):
    grad = np.zeros(n)
    if offset >= 0:
        npairs = n - offset
        if npairs <= 0:
            return grad
        grad[:npairs] += (1.0 - h[offset:offset + npairs])
        grad[offset:] -= h[:npairs]
    else:
        ao = -offset
        npairs = n - ao
        if npairs <= 0:
            return grad
        grad[ao:] += (1.0 - h[:npairs])
        grad[:npairs] -= h[ao:ao + npairs]
    return grad

def exact_subgrad(h, n, frac=0.01):
    corr = compute_corr(h)
    mx = np.max(corr)
    threshold = mx - frac * mx
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    weights = np.zeros(len(corr))
    for k in active:
        weights[k] = max(corr[k] - threshold, 0.1)
    tw = np.sum(weights[active])
    if tw < 1e-15:
        weights[active] = 1.0 / len(active)
        tw = 1.0
    grad = np.zeros(n)
    for k in active:
        if weights[k] / tw < 1e-12:
            continue
        offset = k - (n - 1)
        grad += (weights[k] / tw) * lag_grad(h, n, offset)
    return grad

    global _N_GLOBAL, _BETA_GLOBAL
    _N_GLOBAL = n
    _BETA_GLOBAL = beta
    bounds = [(0.0, 1.0)] * n

                      bounds=bounds, constraints=cons,
                      jac=smooth_max_grad,
                      options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        if res.success or res.fun < smooth_max_obj(h):
            return project_feasible(res.x, n)
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    mx = np.max(corr)
    active = np.where(corr >= mx - tol_frac * mx)[0]
    m = len(active)
    if m == 0:
        active = np.array([np.argmax(corr)])
        m = 1
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    for j, k_idx in enumerate(active):
        offset = k_idx - (n - 1)
        g = lag_grad(h, n, offset)
        A_ub[j, :n] = g
        A_ub[j, n] = -1.0
        lin_const = corr[k_idx] - g @ h
        b_ub[j] = -lin_const
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0

                     options={'presolve': True, 'time_limit': 10})

        pass

        if c5_cd < best_c5 - 1e-15:
            if n_cd != best_n:
                h_up = interp_to(h_cd, n_cd, best_n)
                h_up = project_feasible(h_up, best_n)
                c5_up = compute_c5(h_up)
                if c5_up < best_c5:
                    best_c5 = c5_up
                    best_h = h_up.copy()
            else:
                best_c5 = c5_cd
                best_h = h_cd.copy()
            _current_best[0] = best_c5
            _current_best[1] = best_h.copy()

    # Phase 7: Final micro-polish with tiny steps
    if remaining_time() > 10:
        h = best_h.copy()
        n = best_n
        for it in range(500):
            if remaining_time() < 3:
                break
            corr = compute_corr(h)
            top_k = int(np.argmax(corr))
            offset = top_k - (n - 1)
            grad = lag_grad(h, n, offset)
            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                break
            grad /= gn
            improved = False
            for step in [0.0005, 0.001, 0.002, 0.0003, 0.003]:
                h_trial = h - step * grad
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-17:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h = h_trial.copy()
                    improved = True
                    _current_best[0] = best_c5
                    _current_best[1] = best_h.copy()
                    break
            if not improved and it % 50 == 25 and remaining_time() > 8:
                h_new = slsqp_refine(h, n, 5000, 200, ftol=1e-18)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-17:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                    _current_best[0] = best_c5
                    _current_best[1] = best_h.copy()

    # Phase 8: Try a few random perturbations and re-optimize
    if remaining_time() > 40:
        for attempt in range(5):
            if remaining_time() < 15:
                break
            perturb = 0.02 * rng.randn(len(best_h))
            h_pert = best_h + perturb
            h_pert = project_feasible(h_pert, best_n)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5 + 0.005:
                h_new = slsqp_refine(h_pert, best_n, 3000, 300)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-16:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    _current_best[0] = best_c5
                    _current_best[1] = best_h.copy()

    # Final validation and cleanup
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
