**Approach: Reduction to a sequence of convex problems**

The max over k of corr[k] can be upper-bounded by t. So we're solving:
minimize t
subject to corr[k] * dx <= t for all k
sum(h) = n/2
0 <= h <= 1

This is a bilinear program (non-convex). But we can linearize:
corr[k] = sum h[i]*(1-h[i+k]) = sum h[i] - sum h[i]*h[i+k]

The term sum h[i]*h[i+k] is the issue. We could upper bound it using:
h[i]*h[i+k] <= (h[i]^2 + h[i+k]^2)/2

So corr[k] >= sum h[i] - sum (h[i]^2 + h[i+k]^2)/2

This gives a lower bound on corr[k], not useful for upper-bounding t.

Alternatively, for the minimization: we want to minimize max corr[k]. A convex relaxation would give a lower bound, not an upper bound. So this approach doesn't directly help.

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()

        else:
            h[mask] += diff / cnt

    m = np.max(corr)
    shifted = corr - m
    return m + np.log(np.sum(np.exp(beta * shifted))) / beta

def subgradient_c5(h, beta, top_k):

    dx = 2.0 / n

    if top_k > 0 and top_k < len(corr):

        idx = np.arange(len(corr))

        return np.zeros(n)

            rem = budget_s - (time.time() - t0) - 2.0
            if rem < 5:
                break
            g = subgradient_c5(h, 200, 25)
            gn = np.linalg.norm(g)
            if gn < 1e-15:
                break
            g /= gn

            vel = 0.92 * vel - lr * g
            h_trial = project_feasible(h + vel, target)
            c5_trial, _ = compute_c5(h_trial)

            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial
            else:
                vel *= 0.4
                lr = max(0.0005, lr * 0.88)
                h = project_feasible(h + vel, target)

            if it % 100 == 0 and it > 0:
                lr = max(0.0005, lr * 0.75)

        print("  After P3: C5={:.8f}".format(best_c5))

    # Phase 4: Gradient-guided pairwise coordinate descent
    rem = budget_s - (time.time() - t0) - 2.0
    if rem > 20:
        n = best_n
        target = n / 2.0
        h = best_h.copy()
        no_improve = 0

        for sweep in range(500):
            rem = budget_s - (time.time() - t0) - 2.0
            if rem < 4:
                break
            improved = False
            g = subgradient_c5(h, 100, 20)

            for _ in range(6000):
                rem = budget_s - (time.time() - t0) - 2.0
                if rem < 3:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                sign = 1 if g[i] < g[j] else -1
                for delta in [0.01, 0.005, 0.02, 0.003, 0.04, 0.002, 0.015]:
                    di = sign * delta
                    dj = -sign * delta
                    if h[i] + di < -1e-9 or h[i] + di > 1.0 + 1e-9:
                        continue
                    if h[j] + dj < -1e-9 or h[j] + dj > 1.0 + 1e-9:
                        continue
                    h[i] += di
                    h[j] += dj
                    c5_test, _ = compute_c5(h)
                    if c5_test < best_c5 - 1e-12:
                        best_c5 = c5_test
                        best_h = h.copy()
                        improved = True
                        break
                    h[i] -= di
                    h[j] -= dj

            if not improved:
                no_improve += 1
                if no_improve >= 12:
                    break
            else:
                no_improve = 0

        print("  After P4: C5={:.8f}".format(best_c5))

    # Phase 5: Perturbation + re-optimization to escape local minima
    rem = budget_s - (time.time() - t0) - 2.0
    if rem > 25:
        n = best_n
        target = n / 2.0
        dx = 2.0 / n
        bounds = make_bounds(n)

        for pr in range(4):
            rem = budget_s - (time.time() - t0) - 2.0
            if rem < 18:
                break
            noise_scale = [0.015, 0.04, 0.08, 0.15][pr]
            h_pert = best_h + rng.randn(n) * noise_scale
            h_pert = project_feasible(h_pert, target)

            for beta in [200, 600, 1500]:
                rem = budget_s - (time.time() - t0) - 2.0
                if rem < 10:
                    break
                try:
                    f = partial(obj_smoothed, n=n, beta=beta, top_k=30, target=target, dx=dx)
                    g = partial(grad_smoothed, n=n, beta=beta, top_k=30, target=target, dx=dx)
                    res = minimize(f, h_pert, jac=g, method='L-BFGS-B', bounds=bounds,
                                   options={'maxiter': 600, 'ftol': 1e-19, 'gtol': 1e-14,
                                            'maxls': 40, 'maxfun': 8000})
                    h_new = project_feasible(res.x, target)
                    c5_new, _ = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        print("  P5 r={}, b={}: C5={:.8f}".format(pr, beta, c5_new))
                except Exception:
                    pass

    # Phase 6: Ultra-fine pairwise moves
    rem = budget_s - (time.time() - t0) - 2.0
    if rem > 10:
        n = best_n
        target = n / 2.0
        h = best_h.copy()
        no_improve = 0

        for sweep in range(30):
            rem = budget_s - (time.time() - t0) - 2.0
            if rem < 3:
                break
            improved = False
            order = rng.permutation(n)
            for i in order:
                rem = budget_s - (time.time() - t0) - 2.0
                if rem < 2:
                    break
                for delta in [0.001, -0.001, 0.002, -0.002, 0.003]:
                    if h[i] + delta < -1e-9 or h[i] + delta > 1.0 + 1e-9:
                        continue
                    h[i] += delta
                    j = (i + 1) % n
                    if h[j] - delta < -1e-9 or h[j] - delta > 1.0 + 1e-9:
                        h[i] -= delta
                        continue
                    h[j] -= delta
                    c5_test, _ = compute_c5(h)
                    if c5_test < best_c5 - 1e-13:
                        best_c5 = c5_test
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
                    if improved:
                        break
            if not improved:
                no_improve += 1
                if no_improve >= 3:
                    break
            else:
                no_improve = 0

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5, _ = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
