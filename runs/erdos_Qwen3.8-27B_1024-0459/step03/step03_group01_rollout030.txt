import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _START) - 5.0)


def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c == 1 else 0.0
    cnt = int(np.sum(h))
    target = n // 2
    if cnt > target:
        ones = np.where(h == 1)[0]
        rng_fix = np.random.RandomState(12345)
        rng_fix.shuffle(ones)
        h[ones[:cnt - target]] = 0.0
    elif cnt < target:
        zeros = np.where(h == 0)[0]
        rng_fix = np.random.RandomState(67890)
        rng_fix.shuffle(zeros)
        h[zeros[:target - cnt]] = 1.0
    return h


def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def inverse_thue_morse(n):
    """Thue-Morse complemented (swap 0 and 1) for diversity"""
    h = thue_morse(n)
    return 1.0 - h


def targeted_binary_search(h, n, time_limit=40, rng=None, verbose=False):

    if rng is None:

    elif n_ones < target:
        zeros_idx = np.where(h == 0)[0]
        add = rng.choice(zeros_idx, target - n_ones, replace=False)
        h[add] = 1.0

    corr = compute_corr(h)
    cur_max = float(np.max(corr))

    start_t = time.time()

            n_try_j = min(40, len(zeros))

def projected_gradient_refine(h, n, iters=500, time_limit=15):
    """Projected gradient for continuous refinement."""
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start_t = time.time()
    alpha = 0.05
    beta_sm = 500.0

    for it in range(iters):
        if time.time() - start_t > time_limit or remaining_time() < 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        shifted = corr - m
        ex = np.exp(np.clip(beta_sm * shifted, -50, 50))
        w = ex / np.sum(ex)

        # Gradient of soft-max
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] > 1e-13:
                off = k - (n - 1)
                if off >= 0:
                    npairs = n - off
                    g[off:off + npairs] += (1.0 - h[:npairs]) * w[k]
                    g[:npairs] -= h[off:off + npairs] * w[k]
                else:
                    ao = -off
                    npairs = n - ao
                    g[:npairs] += (1.0 - h[ao:ao + npairs]) * w[k]
                    g[ao:ao + npairs] -= h[:npairs] * w[k]

        # Step
        h_new = h - alpha * g * dx

        # Project
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target - np.sum(h_new)
        mask = (h_new > 1e-10) & (h_new < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h_new[mask] += diff / np.sum(mask)
        h_new = np.clip(h_new, 0.0, 1.0)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            alpha = min(alpha * 1.1, 0.3)
        else:
            alpha *= 0.7

    return best_h, best_c5


def slsqp_refine(h, n, beta=1000, maxiter=300):
    """SLSQP continuous refinement."""
    if remaining_time() < 8:
        return h, compute_c5(h)
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        ex = np.exp(np.clip(beta * shifted, -60, 60))
        return (m + np.log(np.sum(ex)) / beta) * dx

    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        ex = np.exp(np.clip(beta * shifted, -60, 60))
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] > 1e-12:
                off = k - (n - 1)
                if off >= 0:
                    npairs = n - off
                    g[off:off + npairs] += (1.0 - x[:npairs])
                    g[:npairs] -= x[off:off + npairs]
                else:
                    ao = -off
                    npairs = n - ao
                    g[:npairs] += (1.0 - x[ao:ao + npairs])
                    g[ao:ao + npairs] -= x[:npairs]
                g *= w[k]
        return g * dx

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-15:
            return h_new, c5_new
    except Exception:
        pass
    return h, compute_c5(h)
