def compute_c5_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr


def _grad_corr_csr(h, active):
    """Compute d(corr_active)/d(h) as a matrix (len(active) x n)."""
    n = len(h)
    m = len(active)
    G = np.zeros((m, n))
    for idx, k in enumerate(active):
        o = k - (n - 1)
        if o == 0:
            G[idx, :] = 1.0 - 2.0 * h
        elif o > 0:
            G[idx, 0:n-o] += 1.0 - h[o:n]
            G[idx, o:n] -= h[0:n-o]
        else:
            a = -o
            G[idx, a:n] += 1.0 - h[0:n-a]
            G[idx, 0:n-a] -= h[a:n]
    return G


def _lse_obj(h, active, beta, lam):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr[active].astype(float)
    vmax = vals.max()
    if beta < 1e-8:
        f = vmax * (2.0 / n)
    else:
        f = (vmax + np.log(np.sum(np.exp(beta * (vals - vmax)))) / beta) * (2.0 / n)
    s = np.sum(h) - n / 2.0
    f += lam * s * s
    return f


def _lse_grad(h, active, beta, lam):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr[active].astype(float)
