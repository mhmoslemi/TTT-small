    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

def get_coeffs(h, j, n, corr):
    """Get the linear coefficients and rest values for coordinate j."""
    # Returns: coeff array (2n-1,), rest array (2n-1,), quad_rest scalar
    coeff = np.zeros(2*n-1)
    for k in range(1, n):
        # Positive lag k
        if j + k < n:
            coeff[k + n-1] += (1.0 - h[j+k])
        if j >= k:
            coeff[k + n-1] -= h[j-k]
        # Negative lag -k
        if j >= k:
            coeff[n-1-k] += (1.0 - h[j-k])
        if j + k < n:
            coeff[n-1-k] -= h[j+k]
    # k=0 is quadratic
    quad_rest = corr[n-1] - h[j] * (1.0 - h[j])
    rest = corr - coeff * h[j]
    rest[n-1] = quad_rest
    return coeff, rest, quad_rest

            idx_non_quad = non_zero_idx[non_zero_idx != n-1]
            if len(idx_non_quad) > 0:
                a = coeff[idx_non_quad]  # shape (m,)
                r = rest[idx_non_quad]  # shape (m,)
                # linear_vals[i] = max_k(r[k] + a[k] * (grid[i] - old))

            else:
                lin = np.zeros(len(grid_coarse))
            quad = quad_rest + grid_coarse * (1.0 - grid_coarse)
            f_coarse = np.maximum(np.maximum(lin, quad), max_unaff)
            
            best_i = int(np.argmin(f_coarse))
            best_x = grid_coarse[best_i]

                step = grid_coarse[1] - grid_coarse[0]
                lo = max(0.0, best_x - step)

Actually, for a swap (i: +δ, j: -δ), the change in correlation is:
Δc_k = δ * [coeff_i(k)] + (-δ) * [coeff_j(k)] = δ * (coeff_i(k) - coeff_j(k))

where coeff_i(k) is the coefficient of h[i] in c_k (same as computed in get_coeffs).

So the new correlation is c_k + δ * (coeff_i(k) - coeff_j(k)).

The new max is max_k (c_k + δ * (dcoeff_k)) where dcoeff_k = coeff_i(k) - coeff_j(k).

            a = coeff[non_quad_idx]
            r = rest[non_quad_idx]
            if len(a) > 0:
                lin = np.max(r + np.outer(grid - old, a), axis=1)
            else:
                lin = np.zeros(len(grid))
            quad = quad_rest + grid * (1.0 - grid)
            f_coarse = np.maximum(np.maximum(lin, quad), max_unaff)

            best_i = int(np.argmin(f_coarse))
            best_x = grid[best_i]

            if 0 < best_i < len(grid) - 1:
                step = grid[1] - grid[0]
                lo = max(0.0, best_x - step * 1.5)
                hi = min(1.0, best_x + step * 1.5)
                fine = np.linspace(lo, hi, min(1000, grid_size * 5))
                if len(a) > 0:
                    lin_f = np.max(r + np.outer(fine - old, a), axis=1)
                else:
                    lin_f = np.zeros(len(fine))
                quad_f = quad_rest + fine * (1.0 - fine)
                f_fine = np.maximum(np.maximum(lin_f, quad_f), max_unaff)
                best_x = fine[int(np.argmin(f_fine))]

            h[j] = best_x

        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 3:
                break
        if abs(c5 - old_best_c5) < 1e-12 and sweep > 10:
            break

    return best_h, best_c5

def swap_search(h, n, time_limit=None, t0=None, max_rounds=30, pairs_per_round=80, seed=777):
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    grid_d = np.linspace(0.0, 1.0, 80)

    for rnd in range(max_rounds):
        if time_limit and t0 and time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        any_imp = False

        for _ in range(pairs_per_round):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            old_i = h[i]
            old_j = h[j]
            d_lo = max(-old_i, old_j - 1.0)
            d_hi = min(1.0 - old_i, old_j)
            if d_hi - d_lo < 1e-10:
                continue

            coeff_i = compute_coeff_vector(h, i, n)
            coeff_j = compute_coeff_vector(h, j, n)
            dcoeff = coeff_i - coeff_j

            deltas = np.linspace(d_lo, d_hi, 60)
            best_d = 0.0
            best_val = float(np.max(corr))

            for d in deltas:
                if abs(d) < 1e-12:
                    continue
                new_corr = corr + dcoeff * d
                new_corr[n - 1] += 2.0 * d * (old_j - old_i) - 2.0 * d * d
                idx_cross = []
                if 0 < j - i < n:
                    idx_cross.append(j - i + n - 1)
                if 0 < i - j < n:
                    idx_cross.append(i - j + n - 1)
                if 0 < j - i < n:
                    k = j - i
                    fidx = k + n - 1
                    orig_term = old_i * (1.0 - old_j)
                    new_term = (old_i + d) * (1.0 - old_j + d)
                    lin_part = dcoeff[fidx] * d
                    new_corr[fidx] += (new_term - orig_term) - lin_part
                if 0 < i - j < n:
                    k = i - j
                    fidx = k + n - 1
                    orig_term = old_j * (1.0 - old_i)
                    new_term = (old_j - d) * (1.0 - old_i - d)
                    lin_part = dcoeff[fidx] * d
                    new_corr[fidx] += (new_term - orig_term) - lin_part
                val = float(np.max(new_corr))
                if val < best_val - 1e-12:
                    best_val = val
                    best_d = d

            if abs(best_d) > 1e-10:
                h[i] = float(np.clip(old_i + best_d, 0.0, 1.0))
                h[j] = float(np.clip(old_j - best_d, 0.0, 1.0))
                any_imp = True

        if any_imp:
            h = project_feasible(h, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        elif rnd > 2:
            break

    return best_h, best_c5
