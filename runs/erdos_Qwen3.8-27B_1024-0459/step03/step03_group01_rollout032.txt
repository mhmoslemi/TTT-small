def compute_c5(h):

    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):

        else:

def binary_sa(h, n, time_limit, rng):
    """Simulated annealing for binary swap search."""
    h = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h))
    target_ones = n // 2
    if n_ones > target_ones:
        ones_idx = np.where(h == 1)[0]
        rem = rng.choice(ones_idx, n_ones - target_ones, replace=False)
        h[rem] = 0.0
    elif n_ones < target_ones:
        zeros_idx = np.where(h == 0)[0]
        add = rng.choice(zeros_idx, target_ones - n_ones, replace=False)
        h[add] = 1.0
    
    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()
    
    T_init = 5.0  # initial temperature (in correlation units)
    T_final = 0.1
    start = time.time()
    total_steps = 0
    improve_interval = 50000
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while time.time() - start < time_limit:
        if remaining() < 5:
            break
        elapsed = time.time() - start
        progress = elapsed / time_limit
        T = T_init * (T_final / T_init) ** progress

        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        
        if new_max <= cur_max:
            # Accept
            cur_max = new_max
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_h = h.copy()
        elif rng.rand() < np.exp(-(new_max - cur_max) / T):
            # Accept worse
            cur_max = new_max
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            # Reject
            h[i] = 1.0
            h[j] = 0.0
            new_corr = corr  # restore
            corr = compute_corr(h)  # recompute to be safe
            cur_max = float(np.max(corr))
        
        total_steps += 1
    
    return best_h, best_max * dx

    cl = np.clip(b * (corr - m), -50.0, 50.0)

def _corr_grad_shift(x, n, k):
    g = np.zeros(n)
    d = n - 1 - k
    if d >= 0:
        npairs = n - d
        g[:npairs] += (1.0 - x[d:d + npairs])
        g[d:d + npairs] -= x[:npairs]
    else:
        ao = -d
        npairs = n - ao
        g[ao:ao + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[ao:ao + npairs]
    return g


def _smooth_grad(x):

    corr = compute_corr(x)
    m = np.max(corr)
    b = _BETA
    cl = np.clip(b * (corr - m), -50.0, 50.0)

    m = np.max(corr)
    b = _BETA
    cl = np.clip(b * (corr - m), -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(cl))) / b) * (2.0 / n)


def _sgrad(x):
    n = _N
    corr = compute_corr(x)
    m = np.max(corr)
    b = _BETA
    cl = np.clip(b * (corr - m), -50.0, 50.0)
    ex = np.exp(cl)
    s = ex.sum()
    w = ex / s
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = n - 1 - k
        if d >= 0:

            g[d:d + np_] -= wk * x[:np_]

            g[ao:ao + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[ao:ao + np_]

import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 2.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    xn = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(xn, xp, src_h)


def to_binary(h, n, rng):
    h = np.asarray(h, dtype=float)
    b = (h >= 0.5).astype(float)
    no = int(b.sum())
    t = n // 2
    if no > t:
        oi = np.where(b == 1)[0]
        order = np.argsort(h[oi])
        b[oi[order[:no - t]]] = 0.0
    elif no < t:
        zi = np.where(b == 0)[0]
        order = np.argsort(-h[zi])
        b[zi[order[:t - no]]] = 1.0
    return b


def make_tm(n, rng):
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        if c == 1:
            h[i] = 1.0
    no = int(h.sum())
    t = n // 2
    if no > t:
        oi = np.where(h == 1)[0]
        h[rng.choice(oi, no - t, replace=False)] = 0.0
    elif no < t:
        zi = np.where(h == 0)[0]
        h[rng.choice(zi, t - no, replace=False)] = 1.0
    return h


def binary_greedy(h_start, n, time_limit, rng):
    h = to_binary(h_start, n, rng)
    dx = 2.0 / n
    corr = compute_corr(h)
    cur_mx = float(np.max(corr))
    best_mx = cur_mx
    best_h = h.copy()
    sw = time.time()
    stall = 0
    while time.time() - sw < time_limit:
        if remaining() < 3:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        km = int(np.argmax(corr))
        off = km - (n - 1)
        if off >= 0:
            contrib = ones[(ones + off < n) & (h[ones + off] == 0)]
        elif off < 0:
            ao = -off
            contrib = ones[(ones >= ao) & (h[ones - ao] == 0)]
        else:
            contrib = ones
        found = False
        ntry = 400
        for _ in range(ntry):
            if len(contrib) > 0 and rng.rand() < 0.8:
                i = int(contrib[rng.randint(len(contrib))])
            else:
                i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            nc = compute_corr(h)
            nm = float(np.max(nc))
            h[i] = 1.0
            h[j] = 0.0

                    best_h = h.copy()
                found = True
                break

            stall = 0
        else:
            stall += 1
            if stall > 50:

    return best_h, best_mx * dx

def _sobj(x):
    n = _N
    corr = compute_corr(x)
    m = np.max(corr)
    b = _BETA
    cl = np.clip(b * (corr - m), -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(cl))) / b) * (2.0 / n)


def _sgrad(x):
    n = _N
    corr = compute_corr(x)
    m = np.max(corr)
    b = _BETA
    cl = np.clip(b * (corr - m), -50.0, 50.0)
    ex = np.exp(cl)
    s = ex.sum()
    w = ex / s
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = n - 1 - k
        if d >= 0:
            np_ = n - d
            g[:np_] += wk * (1.0 - x[d:d + np_])
            g[d:d + np_] -= wk * x[:np_]
        else:
            ao = -d
            np_ = n - ao
            g[ao:ao + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[ao:ao + np_]
    return g * (2.0 / n)


def _scon(x):
    return np.sum(x) - _TGT


def _sjac(x):
    return np.ones(_N)


def slsqp_refine(h, n, beta, maxiter):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n * 0.5
    _BETA = beta
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _scon, 'jac': _sjac}]
    try:
        res = minimize(_sobj, h, jac=_sgrad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def bin_refine_and_sa(h_start, n, time_budget, rng):
    """Combined greedy + SA pipeline for one resolution."""
    h = to_binary(h_start, n, rng)
    t_greedy = min(20.0, time_budget * 0.3)
    t_sa = time_budget - t_greedy - 2.0
    if t_greedy > 5:

        h2, c2 = binary_sa(h, n, t_sa, rng, T0=3.0, T1=0.005, n_restarts=2)
        if c2 < c:
            h, c = h2, c2
    return h, c
