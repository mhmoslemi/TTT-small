```python
def compute_grad_smooth(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = weights[j]
        if wj < 1e-15:
            continue
        s = j - (n - 1)
        if s >= 0:
            # (1-h[m-s]) for m in [s, n-1]
            grad[s:] += wj * (1.0 - x[:n-s])
            # -h[m+s] for m in [0, n-1-s]
            grad[:n-s] -= wj * x[s:]
        else:
            sp = -s
            grad[:n-sp] += wj * (1.0 - x[sp:])
            grad[sp:] -= wj * x[:n-sp]
    
    grad *= dx
    return grad
```

```python
        ones = np.where(h == 1.0)[0]
        rem = rng.choice(len(ones), num_ones - target_ones, replace=False)
        h[ones[rem]] = 0.0
    elif num_ones < target_ones:
        zeros = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros), target_ones - num_ones, replace=False)
        h[zeros[add]] = 1.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    
    for it in range(n_iters):
        if remaining() < 3:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))
        oi, oz = ones[i], zeros[j]
        
        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = compute_c5(h)
        
        T = t0 * (t1 / t0) ** (it / max(n_iters, 1))
        if new_c5 < cur_c5 or (T > 1e-16 and rng.rand() < np.exp(-(new_c5 - cur_c5) / T)):
            cur_c5 = new_c5
            ones[i] = oz
            zeros[j] = oi
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0
    
    return best_h, best_c5

def coord_descent(h, n, rng, n_sweeps=8):

    best_h = h.copy()
    
    for sweep in range(n_sweeps):
        if remaining() < 4:
            break

        order = rng.permutation(n)
        for i in order:
            if remaining() < 2:
                break
            old = h[i]
            best_local = best_c5
            best_local_h = h.copy()
            best_local_val = old
            
            cands = [0.0, 1.0]
            if 0 < old < 1:
                cands.extend([0.1, 0.3, 0.5, 0.7, 0.9, max(0, old-0.2), min(1, old+0.2)])
            
            for nv in cands:
                if abs(nv - old) < 1e-10:
                    continue
                h[i] = nv
                hf = project_feasible(h, n)
                c5 = compute_c5(hf)
                if c5 < best_local:
                    best_local = c5
                    best_local_val = nv
                    best_local_h = hf.copy()
            
            h = best_local_h.copy()
            if best_local < best_c5:
                best_c5 = best_local
                best_h = h.copy()
                improved = True
        
        if not improved:
            break
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_G = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    for _ in range(50):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += d / cnt
        else:
            h += d / n
        h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += d
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_fn(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    b = _G['beta']
    m = corr.max()
    e = np.exp(b * (corr - m))
    sm = (m + np.log(e.sum()) / b) * dx
    pen = 500.0 * (x.sum() - _G['target']) ** 2
    return sm + pen

def jac_fn(x):
    n = len(x)
    dx = 2.0 / n
    b = _G['beta']
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    e = np.exp(b * (corr - m))
    w = e / e.sum()
    
    g = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-14:
            continue
        s = j - (n - 1)
        if s >= 0:
            g[s:] += wj * (1.0 - x[:n - s])
            g[:n - s] -= wj * x[s:]
        else:
            sp = -s
            g[:n - sp] += wj * (1.0 - x[sp:])
            g[sp:] -= wj * x[:n - sp]
    g *= dx

def sa(h0, n, iters, rng, t0=0.004, t1=0.00003):

    tgt = n // 2

    elif len(ones) < tgt:
        ad = rng.choice(len(zeros), tgt - len(ones), replace=False)
        h[zeros[ad]] = 1.0
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    bh = h.copy()

        if not ones or not zeros:

        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))

def cdescent(h, n, rng, sweeps=8):

    bh = h.copy()

            break

                break

            blh = h.copy()

            if old > 0.15:
                cands.append(max(0.0, old - 0.25))
            if old < 0.85:
                cands.append(min(1.0, old + 0.25))
            for cv in cands:
                if abs(cv - old) < 1e-9:
                    continue
                h[i] = cv
                hf = proj(h, n)
                c = compute_c5(hf)
                if c < bl:
                    bl = c
                    blh = hf.copy()
            h = blh.copy()
            if bl < bc:
                bc = bl
                bh = h.copy()
                imp = True
        if not imp:
            break
    return bh, bc
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_G = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def proj(h, n):
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    tgt = n * 0.5
    for _ in range(60):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += d / cnt
        else:
            h += d / n
        h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += d
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_fn(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    b = _G['beta']
    m = corr.max()
    e = np.exp(b * (corr - m))
    sm = (m + np.log(e.sum()) / b) * dx
    pen = 500.0 * (x.sum() - _G['target']) ** 2
    return sm + pen

def jac_fn(x):
    n = len(x)
    dx = 2.0 / n
    b = _G['beta']
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    e = np.exp(b * (corr - m))
    w = e / e.sum()
    g = np.zeros(n)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-14:
            continue
        s = j - (n - 1)
        if s >= 0:
            g[s:] += wj * (1.0 - x[:n - s])
            g[:n - s] -= wj * x[s:]
        else:
            sp = -s
            g[:n - sp] += wj * (1.0 - x[sp:])
            g[sp:] -= wj * x[:n - sp]
    g *= dx
    g += 1000.0 * (x.sum() - _G['target'])
    return g

def interp_to(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xq = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xq, xp, src)

def rem():
    return max(0.0, _G['budget'] - (time.time() - _G['t0']) - 1.5)

def tm(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 else 0.0
    return h

def gr_seq(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if abs((i * phi) % 1.0 - 0.5) < 0.25 else 0.0
    return h

def vdc(n):
    h = np.zeros(n)
    for i in range(n):
        v = i + 1
        r = 0
        for _ in range(20):
            r = (r << 1) | (v & 1)
            v >>= 1
        h[i] = 1.0 if r % 2 == 0 else 0.0
    return h

def sine_fn(n, f, p):
    t = np.arange(n) / float(n)
    return np.clip(0.5 + 0.32 * np.sin(2.0 * np.pi * f * t + p), 0.0, 1.0)

def multi_sine(n, rng):
    t = np.arange(n) / float(n)
    h = 0.5
    for _ in range(3):
        f = rng.randint(2, 12)
        p = rng.rand() * 6.28
        a = rng.uniform(0.08, 0.2)
        h += a * np.sin(2.0 * np.pi * f * t + p)
    return np.clip(h, 0.0, 1.0)

def sa(h0, n, iters, rng, t0=0.004, t1=0.00002):
    h = np.where(h0 > 0.5, 1.0, 0.0)
    tgt = n // 2
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) > tgt:
        rm = rng.choice(len(ones), len(ones) - tgt, replace=False)
        h[ones[rm]] = 0.0
    elif len(ones) < tgt:
        ad = rng.choice(len(zeros), tgt - len(ones), replace=False)
        h[zeros[ad]] = 1.0
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    bh = h.copy()
    bc = compute_c5(h)
    cc = bc
    for it in range(iters):
        if rem() < 3:
            break
        if not ones or not zeros:
            break
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))
        oi, oz = ones[i], zeros[j]
        h[oi] = 0.0
        h[oz] = 1.0
        nc = compute_c5(h)
        T = t0 * (t1 / t0) ** (it / max(iters, 1))
        if nc < cc or (T > 1e-16 and rng.random() < np.exp(-(nc - cc) / T)):
            cc = nc
            ones[i] = oz
            zeros[j] = oi
            if nc < bc:
                bc = nc
                bh = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0
    return bh, bc

def cdescent(h, n, rng, sweeps=10):

            if old > 0.2:
                cands.append(max(0.0, old - 0.2))
            if old < 0.8:
                cands.append(min(1.0, old + 0.2))
            if 0.0 < old < 1.0:
                cands.append(max(0.0, old - 0.1))
                cands.append(min(1.0, old + 0.1))
            for cv in cands:
                if abs(cv - old) < 1e-9:
                    continue
                h[i] = cv
                hf = proj(h, n)
                c = compute_c5(hf)
                if c < bl:
                    bl = c
                    blh = hf.copy()
            h = blh.copy()
            if bl < bc:
                bc = bl
                bh = h.copy()
                imp = True
        if not imp:
            break
    return bh, bc

    bnds = [(0.0, 1.0)] * n

                   options={'maxiter': maxiter, 'ftol': 1e-15,
                            'gtol': 1e-10, 'iprint': 0,
                            'maxls': 30, 'maxfun': maxiter * 12})
    return proj(res.x, n)
```
