from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-7) & (h < 1.0 - 1e-7)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    return np.correlate(h, 1.0 - h, mode='full')

    h = np.clip(params, 0.0, 1.0)
    se = np.sum(h) - target
    pen = pw * se * se

    m = np.max(corr)

    val = (m + np.log(np.sum(np.exp(beta * sh))) / beta) * dx
