        c5_work = min(c5_bin, best_c5)

        h_cd, c5_cd = binary_cd_smart(h_work, n, remaining, max_sweeps=20, rng=rng)
        if c5_cd < c5_work - 1e-13:
            c5_work = c5_cd
            h_work = h_cd

        if remaining() > 5.0:
            _G_N = n
            _G_TGT = n / 2.0
            _G_DX = 2.0 / n
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': con_sum}]
            _G_BETA = 5000.0
            try:
                res = minimize(obj_smooth, h_work, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-15, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < c5_work - 1e-14:
                    c5_work = c5_new
                    h_work = h_new
            except Exception:
                pass

        if c5_work < best_c5 - 1e-12:
            best_c5 = c5_work
            best_h = h_work.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
