def fast_correlate_binary(h):
    """Compute np.correlate(h, 1-h, mode='full') using FFT."""
    n = len(h)
    f1 = np.fft.fft(h, 2*n)
    f2 = np.fft.fft(1-h, 2*n)
    corr = np.fft.ifft(f1 * np.conj(f2)).real[:2*n-1]
    return corr

                h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_corr(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def smooth_objective(params):
    h = np.clip(params, 0.0, 1.0)
    se = np.sum(h) - _g_target
    corr = np.correlate(h, 1.0 - h, mode='full')
    return lse_corr(corr, _g_beta) * _g_dx + _g_pw * se * se

    global _g_target, _g_beta, _g_pw, _g_dx
    
    t0 = time.time()

        return max(0.0, budget_s - (time.time() - t0) - 5.0)
