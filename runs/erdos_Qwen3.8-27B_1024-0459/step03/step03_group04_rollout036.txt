                    h[j] -= d
                    h[i] += d
                    h = project_h(h, n)
                    c5_try, _ = compute_c5(h)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h.copy()
                        break
                    else:
                        h[j] += d
                        h[i] -= d
                        h = project_h(h, n)

    # Phase 7: Final resolution check
    if time.time() - t0 < budget * 0.98:
        for dn in range(-5, 6):
            if dn == 0:
                continue
            nt = best_n + dn
            if nt < 80:
                continue
            ht = interp_h(best_h, best_n, nt)
            ht = project_h(ht, nt)
            ct, _ = compute_c5(ht)
            if ct < best_c5 - 1e-14:
                best_c5 = ct
                best_h = ht.copy()
                best_n = nt

    # Final projection and evaluation
    best_h = project_h(best_h, best_n)
    final_c5, _ = compute_c5(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
