def project_box_sum(z, target):
    z = np.asarray(z, dtype=float)
    if z.size == 0:
        return z
    if not np.all(np.isfinite(z)):
        z = np.where(np.isfinite(z), z, 0.5)
    target = float(target)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(z - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(60):
        h = np.clip(h, 0.0, 1.0)
        diff = target - float(np.sum(h))
        if abs(diff) <= 1e-14 * max(1.0, target):

def compute_c5_corr(h):
    n = int(len(h))
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = float(np.max(corr)) * dx
    return c5, corr


def grad_lag(h, d):
    n = len(h)
    g = np.zeros(n, dtype=float)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
        return g
    if d > 0:
        L = n - d
        if L <= 0:
            return g
        g[:L] += 1.0
        g[:L] -= h[d:]
        g[d:] -= h[:L]
        return g
    else:
        j = -d
        L = n - j
        if L <= 0:
            return g
        g[j:] += 1.0
        g[j:] -= h[:L]
        g[:L] -= h[j:]
        return g


def select_active(corr, max_active, rel):
    max_active = max(1, int(max_active))
    m = float(np.max(corr))
    tol = max(1e-12, float(rel) * m)
    idx = np.where(corr >= m - tol)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(corr))], dtype=int)
    if idx.size > max_active:
        vals = corr[idx]
        order = np.argsort(vals)[::-1][:max_active]
        idx = idx[order]
    elif idx.size < 3:
        idx = np.argsort(corr)[-max_active:]
    return idx


def grad_from_corr(corr, h, idx, weights):
    n = len(h)
    g = np.zeros(n, dtype=float)
    if idx is None or len(idx) == 0:
        return g
    if weights is None:
        w = 1.0 / float(len(idx))
        for kk in idx:
            g += w * grad_lag(h, int(kk) - (n - 1))
        return g
    weights = np.asarray(weights, dtype=float)
    ws = float(np.sum(weights))
    if ws <= 0.0:
        return g
    for ii, kk in enumerate(idx):
        ww = float(weights[ii]) / ws
        if ww > 1e-18:
            g += ww * grad_lag(h, int(kk) - (n - 1))
    return g


def lp_one_step(h, n, rho, mu, max_active):
    c, corr = compute_c5_corr(h)
    active = select_active(corr, max_active, 0.05)
    if len(active) == 0:
        return None, c, False

    use_l1 = (n <= 288)
    row_len = (2 * n + 1) if use_l1 else (n + 1)
    rows = []
    rhs = []

    for idx in active:
        d = int(idx) - (n - 1)
        g = grad_lag(h, d)
        rhs_val = float(corr[idx] - np.dot(g, h))
        row = np.zeros(row_len, dtype=float)
        row[:n] = g
        row[n] = -1.0
        rows.append(row)
        rhs.append(rhs_val)

    A_ub = np.vstack(rows).astype(float)
    b_ub = np.array(rhs, dtype=float)

    if use_l1:
        A_l1 = np.zeros((2 * n, row_len), dtype=float)
        b_l1 = np.zeros(2 * n, dtype=float)
        for i in range(n):
            A_l1[2 * i, i] = 1.0
            A_l1[2 * i, n + 1 + i] = -1.0
            b_l1[2 * i] = h[i]
            A_l1[2 * i + 1, i] = -1.0
            A_l1[2 * i + 1, n + 1 + i] = -1.0
            b_l1[2 * i + 1] = -h[i]
        A_ub = np.vstack((A_ub, A_l1))
        b_ub = np.concatenate((b_ub, b_l1))

        A_eq = np.zeros((1, row_len), dtype=float)
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0], dtype=float)

        c_obj = np.zeros(row_len, dtype=float)
        c_obj[n] = 1.0
        c_obj[n + 1:] = float(mu)

        bounds = []
        for val in h:
            lo = max(0.0, float(val) - float(rho))
            hi = min(1.0, float(val) + float(rho))
            bounds.append((lo, hi))
        bounds.append((0.0, None))
        for _ in range(n):
            bounds.append((0.0, None))
    else:
        A_eq = np.zeros((1, row_len), dtype=float)
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0], dtype=float)

        c_obj = np.zeros(row_len, dtype=float)
        c_obj[n] = 1.0

        bounds = []
        for val in h:
            lo = max(0.0, float(val) - float(rho))
            hi = min(1.0, float(val) + float(rho))
            bounds.append((lo, hi))
        bounds.append((0.0, None))

    try:
        try:
            res = linprog(
                c_obj,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
            )
        except TypeError:
            res = linprog(
                c_obj,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
            )
    except Exception:
        return None, c, False

    if res is None or getattr(res, "status", 1) != 0 or getattr(res, "x", None) is None:
        return None, c, False

    y = np.asarray(res.x[:n], dtype=float)
    if not np.all(np.isfinite(y)):
        return None, c, False
    y = project_box_sum(y, n / 2.0)
    c_new, _ = compute_c5_corr(y)
    if c_new < c - 1e-14:
        return y, c_new, True
    return y, c_new, False


def lp_search(h0, n, stop, rng):
    h = project_box_sum(h0, n / 2.0)
    c, _ = compute_c5_corr(h)
    if not np.isfinite(c):
        return h, float("inf")

    best_h = h.copy()
    best_c = c
    rho = 0.08
    mus = (0.001, 0.005, 0.02, 0.0005, 0.05)
    mu_i = 0
    stuck = 0
    last_imp = time.time()
    max_active = 20 if n > 180 else 25

    while time.time() < stop - 1.5:
        y, c_new, improved = lp_one_step(h, n, rho, mus[mu_i], max_active)
        if improved and y is not None:
            h = y
            c = c_new
            if c < best_c:
                best_c = c
                best_h = h.copy()
                last_imp = time.time()
            rho = min(0.35, rho * 1.15 + 0.005)
            stuck = 0
        else:
            rho *= 0.65
            stuck += 1
            if rho < 1e-5 or stuck > 25:
                h = project_box_sum(h + 0.03 * (rng.rand(n) - 0.5), n / 2.0)
                c, _ = compute_c5_corr(h)
                if c > best_c + 0.01:
                    h = best_h.copy()
                    c = best_c
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                    last_imp = time.time()
                rho = 0.12
                mu_i = (mu_i + 1) % len(mus)
                stuck = 0
                if time.time() - last_imp > 120.0:
                    break

    if c < best_c:
        best_h = h.copy()
        best_c = c
    return project_box_sum(best_h, n / 2.0), best_c


def binary_hill_climb(h0, n, stop, rng):
    target = n / 2.0
    h = balance_binary(h0, target)
    h = project_box_sum(h, target)
    oh = 1.0 - h
    c, _ = compute_c5_corr(h)
    if not np.isfinite(c):
        return h, float("inf")

    best_h = h.copy()
    best_c = c
    dx = 2.0 / float(n)
    R = max(4, int(n // 20))
    stuck = 0

    while time.time() < stop - 0.75:
        ones = np.where(h > 0.5)[0]
        if ones.size == 0:
            break

        improved = False
        order = rng.permutation(ones) if rng.rand() < 0.35 else ones
        for i in order:
            if time.time() > stop - 1.0:
                break
            lo = max(0, int(i) - R)
            hi = min(n, int(i) + R + 1)
            for j in range(lo, hi):
                if j == int(i):
                    continue
                if h[j] > 0.5:
                    continue
                h[i] = 0.0
                oh[i] = 1.0
                h[j] = 1.0
                oh[j] = 0.0
                c2 = float(np.max(np.correlate(h, oh, mode="full")) * dx)
                if c2 < c - 1e-14:
                    c = c2
                    if c < best_c:
                        best_c = c
                        best_h = h.copy()
                    improved = True
                    break
                else:
                    h[i] = 1.0
                    oh[i] = 0.0
                    h[j] = 0.0
                    oh[j] = 1.0
            if improved:
                break

        if improved:
            stuck = 0
            continue

        zeros = np.where(h < 0.5)[0]
        if zeros.size == 0:
            break

        base_h = h.copy()
        base_oh = oh.copy()
        base_ones = np.where(base_h > 0.5)[0]
        base_zeros = np.where(base_h < 0.5)[0]
        best_i = -1
        best_j = -1
        best_c2 = c

        attempts = 25
        for _ in range(attempts):
            if base_ones.size == 0 or base_zeros.size == 0:
                break
            i = int(base_ones[rng.randint(base_ones.size)])
            j = int(base_zeros[rng.randint(base_zeros.size)])
            if i == j:
                continue
            base_h[i] = 0.0
            base_oh[i] = 1.0
            base_h[j] = 1.0
            base_oh[j] = 0.0
            c2 = float(np.max(np.correlate(base_h, base_oh, mode="full")) * dx)
            if c2 < best_c2 - 1e-14:
                best_c2 = c2
                best_i = i
                best_j = j
            base_h[i] = 1.0
            base_oh[i] = 0.0
            base_h[j] = 0.0
            base_oh[j] = 1.0

        if best_i >= 0:
            h[best_i] = 0.0
            oh[best_i] = 1.0
            h[best_j] = 1.0
            oh[best_j] = 0.0
            c = best_c2
            if c < best_c:
                best_c = c
                best_h = h.copy()
            stuck = 0
            continue

        stuck += 1
        if stuck > 45 and rng.rand() < 0.15:
            old_h = h.copy()
            old_oh = oh.copy()
            old_c = c
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if ones.size and zeros.size:
                for _ in range(3):
                    i = int(ones[rng.randint(ones.size)])
                    j = int(zeros[rng.randint(zeros.size)])
                    if i == j:
                        continue
                    h[i] = 1.0 - h[i]
                    h[j] = 1.0 - h[j]
                    oh[i] = 1.0 - oh[i]
                    oh[j] = 1.0 - oh[j]
                c2 = float(np.max(np.correlate(h, oh, mode="full")) * dx)
                if c2 < best_c + 0.01:
                    c = c2
                    if c < best_c:
                        best_c = c
                        best_h = h.copy()
                else:
                    h = old_h
                    oh = old_oh
                    c = old_c
            stuck = 0

        if stuck > 110:
            break

    return project_box_sum(best_h, n / 2.0), best_c


def subgrad_search(h0, n, stop, rng):
    h = project_box_sum(h0, n / 2.0)
    c, corr = compute_c5_corr(h)
    if not np.isfinite(c):
        return h, float("inf")
    best_h = h.copy()
    best_c = c
    no_acc = 0
    last_imp = time.time()

    while time.time() < stop - 0.5:
        if c < best_c:
            best_c = c
            best_h = h.copy()
            last_imp = time.time()

        improved = False
        for q in (1, 2, 4, 8, 12, 18):
            if q >= corr.size:
                break
            if time.time() > stop - 0.5:
                break
            idx = np.argsort(corr)[-q:]
            g = grad_from_corr(corr, h, idx, None)
            g -= float(np.mean(g))
            gn = float(np.linalg.norm(g))
            if gn < 1e-13:
                continue
            ddir = -g / gn
            for step in (0.5, 0.2, 0.07, 0.025, 0.008, 0.0025, 0.0008):
                hn = project_box_sum(h + float(step) * ddir, n / 2.0)
                cn, _ = compute_c5_corr(hn)
                if cn < c - 1e-13:
                    h = hn
                    c = cn
                    improved = True
                    break
            if improved:
                break

        if not improved:
            for q, beta in ((10, 100.0), (20, 400.0), (30, 1000.0)):
                if q >= corr.size:
                    break
                if time.time() > stop - 0.5:
                    break
                idx = np.argsort(corr)[-q:]
                vals = corr[idx].astype(float)
                m = float(np.max(vals))
                w = np.exp(float(beta) * (vals - m))
                ws = float(np.sum(w))
                if ws <= 0.0:
                    continue
                g = grad_from_corr(corr, h, idx, w / ws)
                g -= float(np.mean(g))
                gn = float(np.linalg.norm(g))
                if gn < 1e-13:
                    continue
                ddir = -g / gn
                for step in (0.3, 0.1, 0.035, 0.012, 0.004, 0.0012):
                    hn = project_box_sum(h + float(step) * ddir, n / 2.0)
                    cn, _ = compute_c5_corr(hn)
                    if cn < c - 1e-13:
                        h = hn
                        c = cn
                        improved = True
                        break
                if improved:
                    break

        if improved:
            no_acc = 0
            if c < best_c:
                best_c = c
                best_h = h.copy()
                last_imp = time.time()
        else:
            no_acc += 1
            if no_acc > 25:
                h = project_box_sum(h + 0.02 * (rng.rand(n) - 0.5), n / 2.0)
                c, corr = compute_c5_corr(h)
                if c > best_c + 0.005:
                    h = best_h.copy()
                    c = best_c
                    corr, _ = compute_c5_corr(h)
                else:
                    if c < best_c:
                        best_c = c
                        best_h = h.copy()
                        last_imp = time.time()
                no_acc = 0
                if time.time() - last_imp > 130.0:
                    break
            if no_acc > 60:
                break

    return project_box_sum(best_h, n / 2.0), best_c


def pair_search(h0, n, stop, rng):
    h = project_box_sum(h0, n / 2.0)
    c, _ = compute_c5_corr(h)
    if not np.isfinite(c):
        return h, float("inf")
    best_h = h.copy()
    best_c = c
    deltas = (0.0007, 0.002, 0.005, 0.015, 0.04)
    no_acc = 0

    while time.time() < stop - 0.5:
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        improved = False
        for sign in (-1, 1):
            if improved:
                break
            for d in deltas:
                if sign == 1:
                    if h[i] >= float(d) and h[j] <= 1.0 - float(d):
                        h[i] -= d
                        h[j] += d
                        cn, _ = compute_c5_corr(h)
                        if cn < c - 1e-13:
                            c = cn
                            improved = True
                            break
                        h[i] += d
                        h[j] -= d
                else:
                    if h[i] <= 1.0 - float(d) and h[j] >= float(d):
                        h[i] += d
                        h[j] -= d
                        cn, _ = compute_c5_corr(h)
                        if cn < c - 1e-13:
                            c = cn
                            improved = True
                            break
                        h[i] -= d
                        h[j] += d

        if improved:
            no_acc = 0
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            no_acc += 1
            if no_acc > 45:
                h = project_box_sum(h + 0.01 * (rng.rand(n) - 0.5), n / 2.0)
                c, _ = compute_c5_corr(h)
                if c > best_c + 0.005:
                    h = best_h.copy()
                    c = best_c
                else:
                    if c < best_c:
                        best_c = c
                        best_h = h.copy()
                no_acc = 0
            if no_acc > 90:
                break

    return project_box_sum(best_h, n / 2.0), best_c

def _opt_objective(x):
    x = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(x)):
        return 1e100
    h = project_box_sum(x, _opt_target)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = _opt_active
    if idx is None or len(idx) == 0:
        return float(np.max(corr)) * _opt_dx
    vals = corr[idx].astype(float)
    m = float(np.max(vals))
    b = float(_opt_beta)
    if b < 1e-5:
        return m * _opt_dx
    exps = np.exp(b * (vals - m))
    s = float(np.sum(exps))
    if s <= 0.0:
        return m * _opt_dx
    return (m + float(np.log(s)) / b) * _opt_dx


def _opt_grad(x):
    x = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(x)):
        return np.zeros_like(x)
    h = project_box_sum(x, _opt_target)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = _opt_active
    if idx is None or len(idx) == 0:
        return np.zeros_like(x)
    n = _opt_n
    vals = corr[idx].astype(float)
    m = float(np.max(vals))
    b = float(_opt_beta)
    if b < 1e-5:
        w = np.zeros(len(idx), dtype=float)
        w[int(np.argmax(vals))] = 1.0
    else:
        w = np.exp(b * (vals - m))
        s = float(np.sum(w))
        if s <= 0.0:
            w = np.zeros(len(idx), dtype=float)
            w[0] = 1.0
        else:
            w = w / s
    gh = np.zeros(n, dtype=float)
    for ii, kk in enumerate(idx):
        if w[ii] > 1e-18:
            gh += w[ii] * grad_lag(h, int(kk) - (n - 1))
    gh *= _opt_dx

    mask = (h > 1e-12) & (h < 1.0 - 1e-12)
    mpos = int(np.count_nonzero(mask))
    gx = np.zeros_like(x)
    if mpos > 0:
        gv = gh[mask]
        gx[mask] = gv - float(np.mean(gv))
    return gx


def lbfgs_lse_search(h0, n, stop, rng):
    global _opt_n, _opt_target, _opt_dx, _opt_active, _opt_beta
    h = project_box_sum(h0, n / 2.0)
    c, corr = compute_c5_corr(h)
    if not np.isfinite(c):
        return h, float("inf")
    best_h = h.copy()
    best_c = c

    cases = ((80.0, 12), (250.0, 25), (800.0, 38))
    for beta, am in cases:
        if time.time() > stop - 6.0:
            break
        if c < best_c:
            best_c = c
            best_h = h.copy()
        _, corr = compute_c5_corr(h)
        am = max(1, min(int(am), 2 * n - 1))
        idx = select_active(corr, am, 0.05)
        _opt_n = int(n)
        _opt_target = float(n / 2.0)
        _opt_dx = 2.0 / float(n)
        _opt_active = idx
        _opt_beta = float(beta)
        x = h.copy()
        try:
            res = minimize(
                _opt_objective,
                x,
                jac=_opt_grad,
                method="L-BFGS-B",
                options={"maxiter": 70, "ftol": 1e-15, "gtol": 1e-12, "maxls": 40},
            )
            h2 = project_box_sum(np.asarray(res.x, dtype=float), n / 2.0)
            c2, _ = compute_c5_corr(h2)
            if c2 < c - 1e-14:
                h = h2
                c = c2
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
        except Exception:
            pass

    if c < best_c:
        best_c = c
        best_h = h.copy()
    return project_box_sum(best_h, n / 2.0), best_c


def balance_binary(src, target):
    src = np.asarray(src, dtype=float)
    target = float(target)
    n = src.size
    if n == 0:
        return src.copy()
    if not np.all(np.isfinite(src)):
        src = np.where(np.isfinite(src), src, 0.5)
    h = (src >= 0.5).astype(float)
    cnt = float(np.sum(h))
    need = target - cnt
    if need > 1e-12:
        k = int(round(need))
        zeros = np.where(h < 0.5)[0]
        if zeros.size >= k and k > 0:
            vals = src[zeros]
            order = np.argsort(vals)[::-1][:k]
            h[zeros[order]] = 1.0
    elif need < -1e-12:
        k = int(round(-need))
        ones = np.where(h >= 0.5)[0]
        if ones.size >= k and k > 0:
            vals = src[ones]
            order = np.argsort(vals)[:k]
            h[ones[order]] = 0.0

    diff = target - float(np.sum(h))
    if abs(diff) > 1e-12:
        if diff > 0.0:
            idx = np.where(h < 1.0)[0]
            if idx.size:
                h[idx[0]] += min(diff, 1.0 - h[idx[0]])
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size:
                h[idx[0]] += max(diff, -h[idx[0]])
    return h


def make_binary_random(n, rng):
    h = np.zeros(int(n), dtype=float)
    if n <= 0:
        return h
    k = int(n // 2)
    if k > 0:
        idx = rng.choice(n, size=k, replace=False)
        h[idx] = 1.0
    return h


def make_thue_morse(n):
    h = np.zeros(int(n), dtype=float)
    for i in range(int(n)):
        if (bin(i).count("1") & 1):
            h[i] = 1.0
    return balance_binary(h, n / 2.0)


def make_golden(n):
    n = int(n)
    if n == 0:
        return np.zeros(0, dtype=float)
    vals = (np.arange(n, dtype=float) * 0.6180339887498949) % 1.0
    return balance_binary(vals, n / 2.0)


def make_periodic_pattern(n, rng, period):
    n = int(n)
    p = max(2, int(period))
    half = max(0, p // 2)
    pat = np.zeros(p, dtype=float)
    if half > 0 and p >= half:
        pos = rng.choice(p, size=half, replace=False)
        pat[pos] = 1.0
    reps = (n + p - 1) // p
    score = np.tile(pat, reps)[:n]
    return balance_binary(score, n / 2.0)


def make_random_smooth(n, rng):
    n = int(n)
    if n == 0:
        return np.zeros(0, dtype=float)
    t = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
    val = 0.5 + np.zeros(n, dtype=float)
    nterms = 2 + int(rng.randint(0, 4))
    freq_hi = max(1, n // 6)
    for _ in range(nterms):
        f = 1 + int(rng.randint(0, freq_hi))
        a = 0.05 + 0.35 * float(rng.rand())
        ph = 2.0 * np.pi * float(rng.rand())
        val += a * np.cos(2.0 * np.pi * float(f) * t + ph)
    return project_box_sum(val, n / 2.0)


def make_block_pattern(n, rng):
    n = int(n)
    score = np.zeros(n, dtype=float)
    pos = 0
    while pos < n:
        L = 1 + int(rng.randint(0, 7))
        end = min(n, pos + L)
        val = float(rng.rand())
        score[pos:end] = val
        pos = end
    return balance_binary(score, n / 2.0)


def resample_h(src, src_n, dst_n):
    src = np.asarray(src, dtype=float)
    src_n = int(src_n)
    dst_n = int(dst_n)
    target = dst_n / 2.0
    if dst_n == src_n:
        return project_box_sum(src.copy(), target)
    if dst_n < src_n:
        out = np.empty(dst_n, dtype=float)
        for i in range(dst_n):
            a = (i * src_n) // dst_n
            b = ((i + 1) * src_n) // dst_n
            if b <= a:
                b = a + 1
            out[i] = float(np.mean(src[a:b]))
    else:
        xp = (np.arange(src_n, dtype=float) + 0.5) * (2.0 / float(src_n))
        xq = (np.arange(dst_n, dtype=float) + 0.5) * (2.0 / float(dst_n))
        out = np.interp(xq, xp, src)
    return project_box_sum(out, target)


def safe_eval_candidate(n, h):
    try:
        h = project_box_sum(np.asarray(h, dtype=float), n / 2.0)
        if h.size != int(n):
            return None
        if not np.all(np.isfinite(h)):
            return None
        c, _ = compute_c5_corr(h)
        if not np.isfinite(c):
            return None
        return (float(c), h)
    except Exception:
        return None


def generate_initial_pool(parent, n_list, seed, stop):
    rng = np.random.RandomState(seed)
    pool = []
    for n in n_list:
        if time.time() > stop - 0.5:
            break
        local = []
        n = int(n)

        if parent is not None:
            res = safe_eval_candidate(n, resample_h(parent, len(parent), n))
            if res is not None:
                local.append(res)
            res = safe_eval_candidate(n, balance_binary(parent, n / 2.0))
            if res is not None:
                local.append(res)

        for _ in range(3):
            res = safe_eval_candidate(n, make_binary_random(n, rng))
            if res is not None:
                local.append(res)

        res = safe_eval_candidate(n, make_thue_morse(n))
        if res is not None:
            local.append(res)

        res = safe_eval_candidate(n, make_golden(n))
        if res is not None:
            local.append(res)

        for p in (8, 10, 12, 16, 20, 24, 32):
            res = safe_eval_candidate(n, make_periodic_pattern(n, rng, p))
            if res is not None:
                local.append(res)

        for _ in range(4):
            res = safe_eval_candidate(n, make_random_smooth(n, rng))
            if res is not None:
                local.append(res)

        res = safe_eval_candidate(n, make_block_pattern(n, rng))
        if res is not None:
            local.append(res)

        if local:
            vals = []
            for item in local:
                vals.append(item[0])
            order = np.argsort(vals)
            take = min(2, len(order))
            for ii in range(take):
                idx = int(order[ii])
                pool.append((local[idx][1].copy(), float(local[idx][0]), n))
    return pool


def select_top_keys(best_per_n, k):
    keys = list(best_per_n.keys())
    if not keys:
        return []
    vals = []
    for kk in keys:
        vals.append(float(best_per_n[kk][0]))
    order = np.argsort(vals)
    out = []
    m = min(k, len(order))
    for i in range(m):
        out.append(keys[int(order[i])])
    return out


def refine_candidate(h0, n, c0, stop, rng):
    n = int(n)
    h0 = project_box_sum(h0, n / 2.0)
    c, _ = compute_c5_corr(h0)
    if not np.isfinite(c):
        return h0, float("inf")
    best_h = h0.copy()
    best_c = c
    total = max(0.0, float(stop) - time.time())
    if total <= 1.0:
        return best_h, best_c

    stop1 = min(float(stop), time.time() + max(5.0, 0.16 * total))
    bh, bc = binary_hill_climb(best_h, n, stop1, rng)
    if bc < best_c:
        best_h = bh.copy()
        best_c = bc

    stop2 = min(float(stop), time.time() + max(15.0, 0.45 * total))
    if stop2 > time.time():
        lh, lc = lp_search(best_h, n, stop2, rng)
        if lc < best_c:
            best_h = lh.copy()
            best_c = lc

    stop3 = min(float(stop), time.time() + max(10.0, 0.24 * total))
    if stop3 > time.time():
        gh, gc = subgrad_search(best_h, n, stop3, rng)
        if gc < best_c:
            best_h = gh.copy()
            best_c = gc

    stop4 = min(float(stop), time.time() + max(5.0, 0.09 * total))
    if stop4 > time.time():
        ph, pc = pair_search(best_h, n, stop4, rng)
        if pc < best_c:
            best_h = ph.copy()
            best_c = pc

    stop5 = min(float(stop), time.time() + max(5.0, 0.08 * total))
    if stop5 > time.time():
        fh, fc = lbfgs_lse_search(best_h, n, stop5, rng)
        if fc < best_c:
            best_h = fh.copy()
            best_c = fc

    best_h = project_box_sum(best_h, n / 2.0)
    best_c, _ = compute_c5_corr(best_h)
    return best_h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 1300.0
    hard_end = t0 + budget - min(5.0, 0.05 * budget)
    reserve = min(120.0, max(30.0, 0.08 * budget))
    rng = np.random.RandomState(seed)

    parent = None
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        if parent.ndim != 1 or parent.size == 0:
            parent = None
        else:
            parent = project_box_sum(parent, parent.size / 2.0)
    except Exception:
        parent = None

    best_h = None
    best_c = float("inf")
    best_n = 0

    if parent is not None:
        best_n = int(parent.size)
        best_h = parent.copy()
        best_c, _ = compute_c5_corr(best_h)
    else:
        best_n = 160
        best_h = project_box_sum(make_binary_random(best_n, rng), best_n / 2.0)
        best_c, _ = compute_c5_corr(best_h)

    base_ns = [80, 96, 128, 160, 192, 224, 256]
    if 48 <= best_n <= 384:
        base_ns.append(best_n)
        base_ns.append(best_n - 16)
        base_ns.append(best_n + 16)
        base_ns.append(best_n + 32)
    n_set = set()
    for nn in base_ns:
        nn = int(nn)
        if nn % 2 != 0:
            nn += 1
        if 48 <= nn <= 384:
            n_set.add(nn)
    n_list = sorted(n_set)

    pool_stop = min(hard_end - 60.0, t0 + min(120.0, 0.09 * budget))
    try:
        pool = generate_initial_pool(parent, n_list, seed, pool_stop)
    except Exception:
        pool = []

    best_per_n = {}
    if best_h is not None:
        best_per_n[int(best_n)] = (float(best_c), np.asarray(best_h, dtype=float).copy())

    for hh, cc, nn in pool:
        try:
            nn = int(nn)
            cc = float(cc)
            if not np.isfinite(cc):
                continue
            prev = best_per_n.get(nn)
            if prev is None or cc < float(prev[0]):
                best_per_n[nn] = (cc, np.asarray(hh, dtype=float).copy())
            if cc < best_c:
                best_c = cc
                best_n = nn
                best_h = np.asarray(hh, dtype=float).copy()
        except Exception:
            pass

    if time.time() < hard_end - reserve - 90.0:
        de_stop = min(hard_end - reserve - 30.0, time.time() + min(75.0, 0.06 * budget))
        try:
            de_h, de_c = run_de(96, de_stop, rng)
            if de_h is not None and np.isfinite(de_c):
                de_n = int(len(de_h))
                de_h = project_box_sum(de_h, de_n / 2.0)
                de_c2, _ = compute_c5_corr(de_h)
                de_c = min(float(de_c), float(de_c2))
                prev = best_per_n.get(de_n)
                if prev is None or de_c < float(prev[0]):
                    best_per_n[de_n] = (de_c, np.asarray(de_h, dtype=float).copy())
                if de_c < best_c:
                    best_c = de_c
                    best_n = de_n
                    best_h = np.asarray(de_h, dtype=float).copy()
        except Exception:
            pass

    top_ns = select_top_keys(best_per_n, 5)
    if len(top_ns) == 1:
        weights = (1.0,)
    else:
        weights = (0.38, 0.25, 0.18, 0.12, 0.07)

    for ci, nn in enumerate(top_ns):
        if time.time() >= hard_end - reserve - 20.0:
            break
        item = best_per_n.get(nn)
        if item is None:
            continue
        base_h, base_c = item
        rem = max(0.0, hard_end - reserve - time.time())
        if rem <= 10.0:
            break
        w = float(weights[min(ci, len(weights) - 1)])
        rem_ws = 0.0
        for ri in range(ci, len(top_ns)):
            rem_ws += float(weights[min(ri, len(weights) - 1)])
        if rem_ws <= 0.0:
            w = 1.0 / max(1, len(top_ns) - ci)
        else:
            w = w / rem_ws
        alloc = rem * w
        alloc = min(alloc, 360.0)
        stop_cand = time.time() + max(25.0, alloc)
        stop_cand = min(stop_cand, hard_end - max(10.0, reserve - 30.0))
        if stop_cand <= time.time() + 12.0:
            break
        try:
            h2, c2 = refine_candidate(base_h, nn, base_c, stop_cand, rng)
            c2 = float(c2)
            if np.isfinite(c2):
                prev = best_per_n.get(nn)
                if prev is None or c2 < float(prev[0]):
                    best_per_n[nn] = (c2, np.asarray(h2, dtype=float).copy())
                if c2 < best_c:
                    best_c = c2
                    best_n = int(nn)
                    best_h = np.asarray(h2, dtype=float).copy()
        except Exception:
            pass

    for dn in (-64, -48, -32, -16, -8, 8, 16, 32, 48, 64):
        if time.time() > hard_end - 6.0:
            break
        nt = int(best_n) + int(dn)
        if nt % 2 != 0:
            nt += 1
        if nt < 48 or nt > 384:
            continue
        try:
            ht, ct = safe_eval_candidate(nt, resample_h(best_h, best_n, nt))
            if ht is not None and ct < best_c:
                best_h = ht
                best_c = ct
                best_n = nt
                lp_stop = min(hard_end - 10.0, time.time() + min(12.0, max(4.0, 0.05 * (hard_end - time.time()))))
                if lp_stop > time.time():
                    h2, c2 = lp_search(best_h, best_n, lp_stop, rng)
                    c2 = float(c2)
                    if np.isfinite(c2) and c2 < best_c:
                        best_h = h2
                        best_c = c2
        except Exception:
            pass

    if time.time() < hard_end - 18.0:
        sstop = min(hard_end - 5.0, time.time() + min(25.0, max(6.0, 0.10 * (hard_end - time.time()))))
        try:
            h2, c2 = lp_search(best_h, best_n, sstop, rng)
            c2 = float(c2)
            if np.isfinite(c2) and c2 < best_c:
                best_h = h2
                best_c = c2
        except Exception:
            pass

    if time.time() < hard_end - 12.0:
        sstop = min(hard_end - 5.0, time.time() + min(18.0, max(5.0, 0.08 * (hard_end - time.time()))))
        try:
            h2, c2 = subgrad_search(best_h, best_n, sstop, rng)
            c2 = float(c2)
            if np.isfinite(c2) and c2 < best_c:
                best_h = h2
                best_c = c2
        except Exception:
            pass

    if time.time() < hard_end - 6.0:
        sstop = min(hard_end - 3.0, time.time() + min(8.0, max(3.0, 0.04 * (hard_end - time.time()))))
        try:
            h2, c2 = pair_search(best_h, best_n, sstop, rng)
            c2 = float(c2)
            if np.isfinite(c2) and c2 < best_c:
                best_h = h2
                best_c = c2
        except Exception:
            pass

    best_h = project_box_sum(best_h, best_n / 2.0)
    final_c, _ = compute_c5_corr(best_h)
    try:
        ec = evaluate_erdos_solution(best_h)
        ec = float(ec)
        if np.isfinite(ec) and ec < final_c:
            final_c = ec
    except Exception:
        pass

    return (np.asarray(best_h, dtype=float), float(final_c), int(best_n))
