            h2[i], h2[j] = h2[j], h2[i]
            h2 = _proj(h2, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-14:
                best = h2
                bc = c2
                imp = True
                ones = np.where(best > 0.999)[0]
                zeros = np.where(best < 0.001)[0]
                mid_hi = np.where((best > 0.6) & (best <= 0.999))[0]
                mid_lo = np.where((best >= 0.001) & (best < 0.4))[0]
                all_hi = np.concatenate([ones, mid_hi]) if len(mid_hi) > 0 else ones
                all_lo = np.concatenate([zeros, mid_lo]) if len(mid_lo) > 0 else zeros
        if not imp:
            break
    return best, bc


def _coord_descent(h, n, rng, nit):
    best = h.copy()
    bc = _c5(best)
    for _it in range(nit):
        if _tl() < 5:
            break
        imp = False
        free = np.where((best > 0.02) & (best < 0.98))[0]
        if len(free) < 2:
            break
        for idx in free:
            if _tl() < 3:
                break
            old = best[idx]
            deltas = [0.02, -0.02, 0.04, -0.04, 0.01, -0.01, 0.005, -0.005]
            rng.shuffle(deltas)
            for d in deltas:
                nv = old + d
                if nv < 0.0 or nv > 1.0:
                    continue
                partner = free[rng.randint(len(free))]
                if partner == idx:
                    partner = (partner + 1) % len(free)
                h2 = best.copy()
                h2[idx] = nv
                h2[partner] -= d
                h2 = _proj(h2, n)
                c2 = _c5(h2)
                if c2 < bc - 1e-14:
                    best = h2
                    bc = c2
                    imp = True
                    break
    return best, bc


def _full_opt(h0, n, rng):
    h = _proj(h0.copy(), n)
    c5 = _c5(h)

    # Phase A: Smooth gradient descent with annealing
    h, c5 = _sgdesc(h, n, rng, 500, 30.0, 500.0, 0.5)
    # Phase B: SLSQP
    for beta in [100, 400, 1500]:
        if _tl() < 20:
            break
        h2 = _slsqp(h, n, beta, 200)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2
    # Phase C: Local search
    h, c5 = _local_swap(h, n, rng, 50, 100)
    # Phase D: More gradient
    h, c5 = _sgdesc(h, n, rng, 400, 100.0, 3000.0, 0.1)
    # Phase E: Coordinate descent
    h, c5 = _coord_descent(h, n, rng, 150)
    # Phase F: Final SLSQP
    if _tl() > 30:
        h2 = _slsqp(h, n, 5000, 400)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2
    # Phase G: Final local search
    h, c5 = _local_swap(h, n, rng, 30, 80)
    h, c5 = _sgdesc(h, n, rng, 200, 200.0, 10000.0, 0.05)

    return h, c5


def run(seed=42, budget_s=1300, **kwargs):
    global _t0, _tb
    _t0 = time.time()
    _tb = budget_s
    rng = np.random.RandomState(seed)

    try:
        hpar = np.array(initial_h_values, dtype=np.float64)
        npar = len(hpar)
    except Exception:
        npar = 160
        hpar = np.zeros(npar, dtype=np.float64)
        hpar[:npar // 2] = 1.0

    bh = _proj(hpar, npar)
    bc5 = _c5(bh)
    bn = npar
    print("Init n={} C5={:.8f}".format(bn, bc5))

    # Multi-resolution exploration
    res_list = [npar]
    for candidate in [npar - 4, npar + 4, npar - 8, npar + 8, npar + 12, npar - 12,
                      npar + 16, int(npar * 1.5), int(npar * 2)]:
        if candidate >= 32 and candidate % 2 == 0 and candidate not in res_list:
            res_list.append(candidate)

    # Sort by proximity to npar for efficiency
    res_list.sort(key=lambda x: abs(x - npar))

    for nn in res_list:
        if _tl() < 80:
            break
        hi = _proj(_interp(bh, bn, nn), nn)
        # Quick optimization
        h2, c2 = _sgdesc(hi, nn, rng, 300, 30.0, 300.0, 0.3)
        if c2 < bc5:
            bh, bc5, bn = h2, c2, nn
            print("MultiRes N={} C5={:.8f}".format(nn, c2))

    # Main optimization at best resolution
    h, c5 = _full_opt(bh, bn, rng)
    if c5 < bc5:
        bh, bc5 = h, c5
        print("MainOpt N={} C5={:.8f}".format(bn, c5))

    # Try slightly different resolutions around best
    for nn in [bn - 2, bn + 2, bn + 4, bn - 4, bn + 6]:
        if _tl() < 60:
            break
        if nn < 32 or nn % 2 != 0:
            continue
        hi = _proj(_interp(bh, bn, nn), nn)
        h2, c2 = _full_opt(hi, nn, rng)
        if c2 < bc5:
            bh, bc5, bn = h2, c2, nn
            print("ResTune N={} C5={:.8f}".format(nn, c2))

    # Final polish
    if _tl() > 30:
        h, c5 = _sgdesc(bh, bn, rng, 300, 500.0, 20000.0, 0.02)
        if c5 < bc5:
            bh, bc5 = h, c5
    if _tl() > 20:
        h2 = _slsqp(bh, bn, 8000, 500)
        c2 = _c5(h2)
        if c2 < bc5:
            bh, bc5 = h2, c2
    if _tl() > 15:
        h, c5 = _local_swap(bh, bn, rng, 40, 120)
        if c5 < bc5:
            bh, bc5 = h, c5
    if _tl() > 10:
        h, c5 = _coord_descent(bh, bn, rng, 100)
        if c5 < bc5:
            bh, bc5 = h, c5
    if _tl() > 10:
        h, c5 = _sgdesc(bh, bn, rng, 200, 5000.0, 50000.0, 0.0)
        if c5 < bc5:
            bh, bc5 = h, c5

    bh = _proj(bh, bn)
    fc5 = _c5(bh)
    try:
        ec = evaluate_erdos_solution(bh)
        if ec < fc5:
            fc5 = ec
    except Exception:
        pass
    print("FINAL n={} C5={:.8f}".format(bn, fc5))
    return (bh, fc5, bn)
