        ntr = min(len(ones) * len(zeros), 800)
        for _ in range(ntr):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h[i], h[j] = 0.0, 1.0
            c5n = _c5(h)
            if c5n < c5m - 1e-12:
                c5m = c5n
                imp = True
            else:
                h[i], h[j] = 1.0, 0.0
            if _ % 60 == 0:
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
        if c5m < bc:
            bc, bh = c5m, h.copy()
        if sa:
            T *= 0.96
            if T < 1e-6:
                sa = False
        if not imp and not sa:
            break
        if not imp and sa and T < 1e-5:
            break
    return bh, bc


def _sys_pairs(h, n, max_rounds=20):
    h = h.copy()
    c5m = _c5(h)
    bh, bc = h.copy(), c5m
    for rd in range(max_rounds):
        if _remain() < 5:
            break
        imp = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in ones:
            if _remain() < 3:
                break
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c5n = _c5(h)
                if c5n < c5m - 1e-12:
                    c5m = c5n
                    imp = True
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                else:
                    h[i], h[j] = 1.0, 0.0
                if _remain() < 3:
                    break
        if c5m < bc:
            bc, bh = c5m, h.copy()
        if not imp:
            break
    return bh, bc


def _sa_pairs(h, n, rng, n_iters=50000, T0=0.005, T1=1e-5):
    h = h.copy()
    c5m = _c5(h)
    bh, bc = h.copy(), c5m
    T = T0
    cool = (T1 / T0) ** (1.0 / max(n_iters, 1))
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(n_iters):
        if _remain() < 3:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i], h[j] = 0.0, 1.0
        c5n = _c5(h)
        delta = c5n - c5m
        if delta < -1e-12 or rng.random() < np.exp(-delta * n / (2.0 * T + 1e-12)):
            c5m = c5n
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if c5n < bc:
                bc, bh = c5n, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T = max(T * cool, T1)
    return bh, bc


def _triple_swap(h, n, rng, n_iters=20000):
    h = h.copy()
    c5m = _c5(h)
    bh, bc = h.copy(), c5m
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(n_iters):
        if _remain() < 5:
            break
        if len(ones) < 1 or len(zeros) < 2:
            break
        i = ones[rng.randint(len(ones))]
        zs = np.delete(zeros, np.where(zeros == i)[0])
        if len(zs) < 2:
            break
        j, k = zs[rng.choice(len(zs), size=2, replace=False)]
        old = h[i], h[j], h[k]
        h[i] = 0.0
        h[j] = 1.0
        h[k] = 0.0
        c5n = _c5(h)
        if c5n < c5m - 1e-12:
            c5m = c5n
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if c5n < bc:
                bc, bh = c5n, h.copy()
        else:
            h[i], h[j], h[k] = old
    return bh, bc


def _slsqp_refine(h, n, maxiter=300, beta=200):
    global _G_N, _G_TGT, _G_DX, _G_BETA
    _G_N, _G_TGT, _G_DX, _G_BETA = n, n / 2.0, 2.0 / n, float(beta)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_sum}]

    def _obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        ex = np.exp(np.clip(_G_BETA * sh, -400, 400))
        return (m + np.log(np.sum(ex)) / _G_BETA) * _G_DX

    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return _project(res.x, n)
    except Exception:
        return h.copy()


def _con_sum(x):
    return np.sum(x) - _G_TGT


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_TGT, _G_DX, _G_START, _G_BUDGET, _G_RNG
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h, best_c5, best_n = None, float('inf'), 0

    # Get parent
    try:
        ph = np.array(initial_h_values, dtype=float).copy()
        pn = len(ph)
        pc5 = _c5(ph)
        best_h, best_c5, best_n = ph.copy(), pc5, pn
        print("Parent: n={}, C5={:.8f}".format(pn, pc5))
    except Exception:
        pn = 256
        ph = _rand_bin(pn, rng)
        best_h, best_c5, best_n = ph.copy(), _c5(ph), pn
        print("No parent. Random: n={}, C5={:.8f}".format(pn, best_c5))

    # Phase 1: Binary local search at n=256 with multiple inits
    n1 = 256
    _setup(n1, budget_s, seed)
    inits = []

    # Thue-Morse
    inits.append((_thue_morse(n1, rng), "TM"))
    # Golden ratio
    inits.append((_golden(n1, rng), "GR"))
    # Parent rounded
    pr = np.where(ph[:min(pn, n1)] > 0.5, 1.0, 0.0) if pn >= n1 else _interp(ph, pn, n1)
    pr = np.where(pr > 0.5, 1.0, 0.0)
    no = int(np.sum(pr))
    if no > n1 // 2:
        idx = np.where(pr == 1)[0]
        pr[rng.choice(idx, no - n1 // 2, replace=False)] = 0
    elif no < n1 // 2:
        idx = np.where(pr == 0)[0]
        pr[rng.choice(idx, n1 // 2 - no, replace=False)] = 1
    inits.append((pr, "ParentBin"))

    # Random
    for _ in range(3):
        inits.append((_rand_bin(n1, rng), "Rand"))

    # Interpolated parent (non-binary, for later SLSQP)
    h_interp = _project(_interp(ph, pn, n1), n1)
    inits.append((h_interp, "Interp"))

    phase1_best = float('inf')
    phase1_h = None
    for h0, name in inits:
        if _remain() < 30:
            break
        c50 = _c5(h0)
        print("  n={}: {} -> C5={:.8f}".format(n1, name, c50))

        # Binary local search
        if np.allclose(h0, np.round(h0), atol=0.1):
            hb, cb = _binary_ls(h0, n1, rng, max_sweeps=300)
        else:
            # Round to binary first
            hr = np.where(h0 > 0.5, 1.0, 0.0)
            no = int(np.sum(hr))
            if no > n1 // 2:
                idx = np.where(hr == 1)[0]
                hr[rng.choice(idx, no - n1 // 2, replace=False)] = 0
            elif no < n1 // 2:
                idx = np.where(hr == 0)[0]
                hr[rng.choice(idx, n1 // 2 - no, replace=False)] = 1
            hb, cb = _binary_ls(hr, n1, rng, max_sweeps=300)

        if cb < phase1_best:
            phase1_best = cb
            phase1_h = hb.copy()
            print("    {} improved: C5={:.8f}".format(name, cb))

    if phase1_h is not None and phase1_best < best_c5:
        best_c5, best_h, best_n = phase1_best, phase1_h.copy(), n1

    # Phase 2: Systematic pair search on best binary
    if _remain() > 20 and phase1_h is not None:
        print("Phase 2: Systematic pairs at n={}".format(n1))
        h2, c5_2 = _sys_pairs(phase1_h, n1, max_rounds=15)
        if c5_2 < best_c5:
            best_c5, best_h, best_n = c5_2, h2.copy(), n1
            print("  Improved: C5={:.8f}".format(c5_2))

    # Phase 3: Simulated annealing with pair swaps
    if _remain() > 30:
        print("Phase 3: SA pairs at n={}".format(n1))
        h3, c5_3 = _sa_pairs(best_h, best_n, rng, n_iters=80000, T0=0.008)
        if c5_3 < best_c5:
            best_c5, best_h, best_n = c5_3, h3.copy(), best_n
            print("  Improved: C5={:.8f}".format(c5_3))

    # Phase 4: Triple swaps
    if _remain() > 30:
        print("Phase 4: Triple swaps at n={}".format(best_n))
        h4, c5_4 = _triple_swap(best_h, best_n, rng, n_iters=30000)
        if c5_4 < best_c5:
            best_c5, best_h, best_n = c5_4, h4.copy(), best_n
            print("  Improved: C5={:.8f}".format(c5_4))

    # Phase 5: More SA + systematic
    if _remain() > 40:
        print("Phase 5: Second SA at n={}".format(best_n))
        h5, c5_5 = _sa_pairs(best_h, best_n, rng, n_iters=60000, T0=0.004)
        if c5_5 < best_c5:
            best_c5, best_h, best_n = c5_5, h5.copy(), best_n
        # Quick systematic
        h5b, c5_5b = _sys_pairs(best_h, best_n, max_rounds=5)
        if c5_5b < best_c5:
            best_c5, best_h, best_n = c5_5b, h5b.copy(), best_n
        print("  C5={:.8f}".format(best_c5))

    # Phase 6: Try n=384
    if _remain() > 60:
        n6 = 384
        _setup(n6, budget_s, seed + 1)
        h6i = _project(_interp(best_h, best_n, n6), n6)
        h6r = np.where(h6i > 0.5, 1.0, 0.0)
        no = int(np.sum(h6r))
        if no > n6 // 2:
            idx = np.where(h6r == 1)[0]
            h6r[rng.choice(idx, no - n6 // 2, replace=False)] = 0
        elif no < n6 // 2:
            idx = np.where(h6r == 0)[0]
            h6r[rng.choice(idx, n6 // 2 - no, replace=False)] = 1

        # Also try Thue-Morse and golden at n6
        h6tm = _thue_morse(n6, rng)
        h6gr = _golden(n6, rng, 0.7)

        for h0, nm in [(h6r, "Interp6"), (h6tm, "TM6"), (h6gr, "GR6"), (_rand_bin(n6, rng), "Rand6")]:
            if _remain() < 20:
                break
            c5x = _c5(h0)
            hx, c5x = _binary_ls(h0, n6, rng, max_sweeps=200)
            if c5x < best_c5:
                best_c5, best_h, best_n = c5x, hx.copy(), n6
                print("  n={} {} improved: C5={:.8f}".format(n6, nm, c5x))

    # Phase 7: Try n=512
    if _remain() > 60:
        n7 = 512
        _setup(n7, budget_s, seed + 2)
        h7i = _project(_interp(best_h, best_n, n7), n7)
        h7r = np.where(h7i > 0.5, 1.0, 0.0)
        no = int(np.sum(h7r))
        if no > n7 // 2:
            idx = np.where(h7r == 1)[0]
            h7r[rng.choice(idx, no - n7 // 2, replace=False)] = 0
        elif no < n7 // 2:
            idx = np.where(h7r == 0)[0]
            h7r[rng.choice(idx, n7 // 2 - no, replace=False)] = 1
        h7tm = _thue_morse(n7, rng)

        for h0, nm in [(h7r, "Interp7"), (h7tm, "TM7"), (_rand_bin(n7, rng), "Rand7")]:
            if _remain() < 20:
                break
            c5x = _c5(h0)
            hx, c5x = _binary_ls(h0, n7, rng, max_sweeps=150)
            if c5x < best_c5:
                best_c5, best_h, best_n = c5x, hx.copy(), n7
                print("  n={} {} improved: C5={:.8f}".format(n7, nm, c5x))

    # Phase 8: Non-binary SLSQP refinement on best solution
    if _remain() > 20:
        print("Phase 8: SLSQP refine at n={}".format(best_n))
        _setup(best_n, budget_s, seed + 3)
        h8 = _project(best_h, best_n)
        for beta in [100, 300, 1000]:
            if _remain() < 8:
                break
            h8 = _slsqp_refine(h8, best_n, maxiter=200, beta=beta)
            c5x = _c5(h8)
            if c5x < best_c5:
                best_c5 = c5x
                best_h = h8.copy()
        print("  SLSQP C5={:.8f}".format(_c5(best_h)))

    # Phase 9: Final binary local search pass
    if _remain() > 15:
        print("Phase 9: Final binary LS")
        h9r = np.where(best_h > 0.5, 1.0, 0.0)
        no = int(np.sum(h9r))
        n9 = len(best_h)
        if no > n9 // 2:
            idx = np.where(h9r == 1)[0]
            h9r[rng.choice(idx, no - n9 // 2, replace=False)] = 0
        elif no < n9 // 2:
            idx = np.where(h9r == 0)[0]
            h9r[rng.choice(idx, n9 // 2 - no, replace=False)] = 1
        h9, c5_9 = _binary_ls(h9r, n9, rng, max_sweeps=200)
        if c5_9 < best_c5:
            best_c5, best_h, best_n = c5_9, h9.copy(), n9

        # Also keep non-binary if better
        hb_nb = _project(best_h, best_n)
        c5_nb = _c5(hb_nb)
        if c5_nb < best_c5:
            best_c5, best_h = c5_nb, hb_nb

    # Phase 10: One more SA round
    if _remain() > 20:
        h10, c5_10 = _sa_pairs(best_h, best_n, rng, n_iters=40000, T0=0.006)
        if c5_10 < best_c5:
            best_c5, best_h, best_n = c5_10, h10.copy(), best_n
        h10b, c5_10b = _sys_pairs(best_h, best_n, max_rounds=5)
        if c5_10b < best_c5:
            best_c5, best_h, best_n = c5_10b, h10b.copy(), best_n

    # Final projection and evaluation
    best_h = _project(best_h, best_n)
    final_c5 = _c5(best_h)

    try:
        ec = evaluate_erdos_solution(best_h)
        if float(ec) < final_c5:
            final_c5 = float(ec)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
