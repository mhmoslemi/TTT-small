At a kink, it's the convex hull of the b_k for all k that achieve the max.

2. Find k* at t_max. If b_{k*} ≤ 0, minimum is at t_max.
3. Otherwise, there's a kink in (t_min, t_max). Find it by "walking" along the envelope.

- Start at t_min, with active set S = {k: a_k + t_min*b_k = φ(t_min)}
- The next kink is at the smallest t > t_min where some k ∉ S becomes active

    for it in range(max_passes):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break

        for i in range(n):
            # Find j to pair with (to maintain sum constraint)
            # Simple strategy: pair with position that's "opposite" (i + n//2) % n
            j = (i + n // 2) % n
            if j == i:
                j = (i + 1) % n
            
            # Current values
            hi = h[i]
            hj = h[j]
            
            # t range: h[i] + t ∈ [0,1], h[j] - t ∈ [0,1]
            t_min = max(0.0, hj - 1.0)
            t_max = min(1.0 - hi, hj)
            # Wait: h[i]_new = hi + t, h[j]_new = hj - t
            # h[i]_new ∈ [0,1]: t ∈ [-hi, 1-hi]
            # h[j]_new ∈ [0,1]: t ∈ [hj-1, hj]
            t_min = max(-hi, hj - 1.0)
            t_max = min(1.0 - hi, hj)
            
            if t_max - t_min < 1e-10:
                continue
            
            # Golden section search for optimal t
            gr = (np.sqrt(5) - 1) / 2
            lo, hi_t = t_min, t_max
            c = hi_t - gr * (hi_t - lo)
            d = lo + gr * (hi_t - lo)
            
            def phi(t):
                h[i] = hi + t
                h[j] = hj - t
                c2 = compute_corr(h)
                val = np.max(c2)
                return val
            
            fc = phi(c)
            fd = phi(d)
            
            for _ in range(40):
                if fc < fd:
                    hi_t = d
                    d = c
                    fd = fc
                    c = hi_t - gr * (hi_t - lo)
                    fc = phi(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi_t - lo)
                    fd = phi(d)
            
            t_opt = (lo + hi_t)
