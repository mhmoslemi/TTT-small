    h, c5 = _swap_search(h, n, 300, rng)
    for beta in [5000, 15000]:
        if _tl() < 15:
            break
        h2 = _slsqp(h, n, beta, 350)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2
    h, c5 = _subgrad_loop(h, n, 600, 1000, 0.01, rng)
    h, c5 = _coord_descent(h, n, 300, rng)
    h01 = _round01(h, n)
    c01 = _c5(h01)
    if c01 < c5:
        h, c5 = h01, c01
    h, c5 = _swap_search(h, n, 400, rng)
    if _tl() > 15:
        h2 = _slsqp(h, n, 30000, 400)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2
        h, c5 = _subgrad_loop(h, n, 400, 5000, 0.008, rng)
        h, c5 = _coord_descent(h, n, 200, rng)
    return h, c5


def run(seed=42, budget_s=1300, **kwargs):
    global _t0, _tb
    _t0 = time.time()
    _tb = budget_s
    rng = np.random.RandomState(seed)

    try:
        hpar = np.array(initial_h_values, dtype=np.float64)
        npar = len(hpar)
    except Exception:
        npar = 160
        hpar = np.zeros(npar, dtype=np.float64)
        hpar[:npar // 2] = 1.0

    # Start at higher resolution: upsample parent to n=200
    n0 = 200
    h_start = _proj(_interp(hpar, npar, n0), n0)
    c_start = _c5(h_start)
    bh = h_start.copy()
    bc5 = c_start
    bn = n0
    print("Start n={} C5={:.8f}".format(bn, bc5))

    # Optimise at n=200
    h, c5 = _optimize_at_res(bh, n0, rng)
    if c5 < bc5:
        bh, bc5, bn = h, c5, n0
        print("N=200 C5={:.8f}".format(c5))

    # Try n=240
    if _tl() > 250:
        n1 = 240
        h1 = _proj(_interp(bh, bn, n1), n1)
        h1, c1 = _optimize_at_res(h1, n1, rng)
        if c1 < bc5:
            bh, bc5, bn = h1, c1, n1
            print("N=240 C5={:.8f}".format(c1))

    # Try n=280
    if _tl() > 250:
        n2 = 280
        h2 = _proj(_interp(bh, bn, n2), n2)
        h2, c2 = _optimize_at_res(h2, n2, rng)
        if c2 < bc5:
            bh, bc5, bn = h2, c2, n2
            print("N=280 C5={:.8f}".format(c2))

        h3 = _proj(_interp(bh, bn, n3), n3)

        if c3 < bc5:
            bh, bc5, bn = h3, c3, n3
            print("N=320 C5={:.8f}".format(c3))

    # Final polish at best resolution
    if _tl() > 30:
        h, c5 = _swap_search(bh, bn, 500, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        if _tl() > 20:
            h2 = _slsqp(bh, bn, 20000, 300)
            c2 = _c5(h2)
            if c2 < bc5:

        if _tl() > 15:
            h, c5 = _subgrad_loop(bh, bn, 500, 8000, 0.005, rng)

            h, c5 = _coord_descent(bh, bn, 200, rng)
            if c5 < bc5:
                bh, bc5 = h, c5

    # Ensure constraint satisfied
    bh = _proj(bh, bn)
    fc5 = _c5(bh)
    try:
        ec = evaluate_erdos_solution(bh)
        if ec < fc5:
            fc5 = ec
    except Exception:
        pass
    print("FINAL n={} C5={:.8f}".format(bn, fc5))
    return (bh, fc5, bn)
