def _c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / n)


def _proj(h, n):
    h = np.array(h, dtype=np.float64)
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(35):
        d = 0.5 * n - h.sum()
        if abs(d) < 1e-15:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        k = int(np.sum(m))
        if k > 0:
            h[m] += d / k
        else:
            h[:] = 0.5
        np.clip(h, 0.0, 1.0, out=h)
    return h


def _interp_f(h, ns, nd):
    if ns == nd:
        return h.copy()
    xs = (np.arange(ns) + 0.5) * (2.0 / ns)
    xd = (np.arange(nd) + 0.5) * (2.0 / nd)
    return np.interp(xd, xs, h)


def _scon(x):
    return x.sum() - 0.5 * _ng


def _sobj(x):
    n = _ng
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = float(np.max(c))
    b = _bv
    sh = c - mx
    return (mx + np.log(np.exp(b * sh).sum()) / b) * (2.0 / n)


def _slsqp(h0, n, beta, maxiter):
    global _ng, _bv
    _ng = n
    _bv = beta
    try:
        r = minimize(_sobj, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon}],
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return _proj(np.asarray(r.x, dtype=np.float64), n)
    except Exception:
        return _proj(h0.copy(), n)


def _ggrad(h, kmax):
    n = len(h)
    g = np.zeros(n)
    s = kmax - (n - 1)
    if s >= 0:
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
    else:
        p = -s
        if p > 0:
            g[:n - p] += 1.0 - h[p:n]
            g[p:] -= h[:n - p]
    return g


def _smgrad(h, bw):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(corr))
    w = np.exp(bw * (corr - mx))
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
        sw = 1.0
    w /= sw
    g = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for k in nz:
        g += w[int(k)] * _ggrad(h, int(k))
    return g


def _topk_grad(h, topk):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_lags = np.argsort(corr)[-topk:]
    g = np.zeros(n)
    for k in top_lags:
        g += _ggrad(h, int(k))
    return g


def _pgd(h, n, nit, rng, bw):
    best = h.copy()
    bc = _c5(best)
    stall = 0
    for it in range(nit):
        if _tl() < 3:
            break
        g = _smgrad(best, bw)
        gn = float(np.linalg.norm(g))
        if gn < 1e-16:
            g2 = _topk_grad(best, 3)
            gn = float(np.linalg.norm(g2))
            if gn > 1e-16:
                g = g2
            else:
                break
        g /= float(np.linalg.norm(g))
        moved = False
        for a in [0.04, 0.015, 0.006, 0.002, 0.025, 0.01, 0.004, 0.001]:
            h2 = _proj(best - a * g, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-15:
                best = h2
                bc = c2
                moved = True
                stall = 0
                break
        if not moved:
            stall += 1
            if stall > 30:
                corr = np.correlate(best, 1.0 - best, mode='full')
                kmax = int(np.argmax(corr))
                g2 = _ggrad(best, kmax)
                gn2 = float(np.linalg.norm(g2))
                if gn2 > 1e-16:
                    g2 /= gn2
                    for a in [0.03, 0.01, 0.005, 0.002]:
                        h2 = _proj(best - a * g2, n)
                        c2 = _c5(h2)
                        if c2 < bc - 1e-15:
                            best = h2
                            bc = c2
                            moved = True
                            stall = 0
                            break
            if stall > 80 and rng.random() < 0.05:
                vals = [0.005, 0.01, 0.02, -0.005, -0.01, -0.02]
                v = vals[rng.randint(len(vals))]
                h2 = _proj(best + v, n)
                c2 = _c5(h2)
                if c2 < bc:
                    best = h2
                    bc = c2
                    stall = 0
        if not moved and stall > 200:
            break
    return best, bc


def _binary_refine(h, n, nsweep, rng):
    best = h.copy()
    bc = _c5(best)
    for sweep in range(nsweep):
        if _tl() < 3:
            break
        ones = np.where(best > 0.45)[0]
        zeros = np.where(best < 0.55)[0]
        if len(ones) == 0 or len(zeros) == 0:
            hi = np.where(best > 0.7)[0]
            lo = np.where(best < 0.3)[0]
            if len(hi) == 0 or len(lo) == 0:
                break
            imp = False
            for _ in range(120):
                if _tl() < 2:
                    break
                ii = hi[rng.randint(len(hi))]
                jj = lo[rng.randint(len(lo))]
                h2 = best.copy()
                tmp = h2[ii]
                h2[ii] = h2[jj]
                h2[jj] = tmp
                h2 = _proj(h2, n)
                c2 = _c5(h2)
                if c2 < bc - 1e-15:
                    best = h2
                    bc = c2
                    imp = True
            if not imp:
                break
            continue
        imp = False
        for _ in range(250):
            if _tl() < 2:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h2 = best.copy()
            tmp = h2[i]
                h2[i] = h2[j]
            h2[j] = tmp
            h2 = _proj(h2, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-15:
                best = h2
                bc = c2
                imp = True
                ones = np.where(best > 0.45)[0]
                zeros = np.where(best < 0.55)[0]
        if not imp:
            best_imp = False
            for i in ones[:min(len(ones), 100)]:
                if _tl() < 2:
                    break
                for j in zeros[:min(len(zeros), 100)]:
                    h2 = best.copy()
                    tmp = h2[i]
                    h2[i] = h2[j]
                    h2[j] = tmp
                    h2 = _proj(h2, n)
                    c2 = _c5(h2)
                    if c2 < bc - 1e-15:
                        best = h2
                        bc = c2
                        best_imp = True
                        ones = np.where(best > 0.45)[0]
                        zeros = np.where(best < 0.55)[0]
                        break
                if best_imp:
                    imp = True
                    break
        if not imp:
            break
    return best, bc


def _frac_refine(h, n, nit, rng):
    """Refine fractional values at transition points"""
    best = h.copy()
    bc = _c5(best)
    for _ in range(nit):
        if _tl() < 3:
            break
        nb = np.where((best > 0.05) & (best < 0.95))[0]
        if len(nb) < 2:
            break
        imp = False
        for idx in nb[:80]:
            if _tl() < 2:
                break
            old = best[idx]
            deltas = [0.02, -0.02, 0.01, -0.01, 0.03, -0.03, 0.005, -0.005]
            rng.shuffle(deltas)
            for delta in deltas:
                nv = old + delta
                if nv < 0.0 or nv > 1.0:
                    continue
                others = nb[(nb != idx)]
                if len(others) == 0:
                    break
                j = others[rng.randint(len(others))]
                h2 = best.copy()
                h2[idx] = nv
                h2[j] += (old - nv)
                h2 = _proj(h2, n)
                c2 = _c5(h2)
                if c2 < bc - 1e-15:
                    best = h2
                    bc = c2
                    imp = True
                    break
            if imp:
                imp = False
        if not imp:
            break
    return best, bc


def _full_opt(h0, n, rng, time_limit):
    h = _proj(h0.copy(), n)
    c5 = _c5(h)
    t_start = time.time()

    for beta in [50, 200, 500]:
        if _tl() < 15:
            break
        h2 = _slsqp(h, n, beta, 300)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2

    h, c5 = _pgd(h, n, 500, rng, 50)
    h, c5 = _binary_refine(h, n, 80, rng)
    h, c5 = _frac_refine(h, n, 150, rng)

    if _tl() > 20:
        h2 = _slsqp(h, n, 1000, 350)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2

    h, c5 = _pgd(h, n, 400, rng, 100)
    h, c5 = _binary_refine(h, n, 60, rng)

    if _tl() > 15:
        h2 = _slsqp(h, n, 3000, 400)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2

    h, c5 = _pgd(h, n, 300, rng, 200)
    h, c5 = _binary_refine(h, n, 50, rng)
    h, c5 = _frac_refine(h, n, 100, rng)

    if _tl() > 10:
        top_lags = np.argsort(np.correlate(h, 1.0 - h, mode='full'))[-3:]
        g = np.zeros(n)
        for k in top_lags:
            g += _ggrad(h, int(k))
        gn = float(np.linalg.norm(g))
        if gn > 1e-16:
            g /= gn
            for a in [0.02, 0.008, 0.003]:
                h2 = _proj(h - a * g, n)
                c2 = _c5(h2)
                if c2 < c5:
                    h, c5 = h2, c2
                    break

    if _tl() > 10:
        h2 = _slsqp(h, n, 8000, 500)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2
        h, c5 = _pgd(h, n, 200, rng, 300)
        h, c5 = _binary_refine(h, n, 40, rng)

    return h, c5


def run(seed=42, budget_s=1300, **kwargs):
    global _t0, _tb
    _t0 = time.time()
    _tb = budget_s
    rng = np.random.RandomState(seed)

    try:
        hpar = np.array(initial_h_values, dtype=np.float64)
        npar = len(hpar)
    except Exception:
        npar = 160
        hpar = np.zeros(npar, dtype=np.float64)
        hpar[:npar // 2] = 1.0

    bh = _proj(hpar, npar)
    bc5 = _c5(bh)
    bn = npar
    print("Init n={} C5={:.8f}".format(bn, bc5))

    # Phase 1: Polish at parent resolution
    h, c5 = _full_opt(bh, npar, rng, 200)
    if c5 < bc5:
        bh, bc5, bn = h, c5, npar
        print("Polished npar N={} C5={:.8f}".format(npar, c5))

    # Phase 2: Try higher resolutions (most promising for improvement)
    cand_res = set()
    for d in [64, 48, 32, 16, -16, -32, 8, -8, 48, -48]:
        nn = npar + d
        if nn >= 32 and nn % 2 == 0:
            cand_res.add(nn)
    cand_res.add(256)
    cand_res.add(320)
    cand_res.discard(npar)

    # Sort by priority: prefer higher resolution for finer control
    cand_list = sorted(cand_res, key=lambda x: -x)[:6]

    best_res = bn
    for nn in cand_list:
        if _tl() < 80:
            break
        h_up = _proj(_interp_f(bh, bn, nn), nn)
        h2, c2 = _full_opt(h_up, nn, rng, 300)
        if c2 < bc5:
            bh, bc5, bn = h2, c2, nn
            print("N={} C5={:.8f}".format(nn, c2))

    # Phase 3: Try a couple more resolutions near the best
    near = set()
    for d in [-16, -8, 8, 16, -32, 32]:
        nn = bn + d
        if nn >= 32 and nn % 2 == 0 and nn not in cand_list:
            near.add(nn)
    for nn in sorted(near, key=lambda x: abs(x - bn))[:3]:
        if _tl() < 70:
            break
        h_up = _proj(_interp_f(bh, bn, nn), nn)
        h2, c2 = _full_opt(h_up, nn, rng, 200)
        if c2 < bc5:
            bh, bc5, bn = h2, c2, nn
            print("Near N={} C5={:.8f}".format(nn, c2))

    # Phase 4: Final intensive polish at best resolution
    if _tl() > 25:
        h, c5 = _pgd(bh, bn, 400, rng, 80)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _binary_refine(bh, bn, 60, rng)
        if c5 < bc5:
            bh, bc5 = h, c5

    if _tl() > 20:
        h2 = _slsqp(bh, bn, 5000, 400)
        c2 = _c5(h2)
        if c2 < bc5:
            bh, bc5 = h2, c2
        h, c5 = _pgd(bh, bn, 300, rng, 150)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _binary_refine(bh, bn, 40, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _frac_refine(bh, bn, 150, rng)
        if c5 < bc5:
            bh, bc5 = h, c5

    if _tl() > 12:
        h2 = _slsqp(bh, bn, 10000, 400)
        c2 = _c5(h2)
        if c2 < bc5:
            bh, bc5 = h2, c2
        h, c5 = _pgd(bh, bn, 200, rng, 250)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _binary_refine(bh, bn, 30, rng)
        if c5 < bc5:
            bh, bc5 = h, c5

    if _tl() > 8:
        corr = np.correlate(bh, 1.0 - bh, mode='full')
        top3 = np.argsort(corr)[-3:]
        g = np.zeros(bn)
        for k in top3:
            g += _ggrad(bh, int(k))
        gn = float(np.linalg.norm(g))
        if gn > 1e-16:
            g /= gn
            for a in [0.015, 0.006, 0.002, 0.01, 0.004]:
                h2 = _proj(bh - a * g, bn)
                c2 = _c5(h2)
                if c2 < bc5:
                    bh, bc5 = h2, c2
                    break

    bh = _proj(bh, bn)
    fc5 = _c5(bh)
    try:
        ec = evaluate_erdos_solution(bh)
        if isinstance(ec, (float, int)) and ec < fc5:
            fc5 = float(ec)
    except Exception:
        pass
    print("FINAL n={} C5={:.8f}".format(bn, fc5))
    return (bh, fc5, bn)
