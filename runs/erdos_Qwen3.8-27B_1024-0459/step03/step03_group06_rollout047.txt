                break
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def vectorized_smooth_max_obj(x, n, beta, tgt):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    # Numerically stable log-sum-exp
    sm = m + np.log(np.sum(np.exp(beta * shifted))) / beta
    penalty = 100.0 * (np.sum(x) - tgt) ** 2
    return sm * dx + penalty

def vectorized_smooth_max_grad(x, n, beta, tgt):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exp_vals = np.exp(beta * shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    # Vectorized gradient computation
    for j in range(2 * n - 1):
        wj = weights[j]
        if wj < 1e-15:
            continue
        s = j - (n - 1)
        if s >= 0:
            grad[s:] += wj * (1.0 - x[:n-s])
            grad[:n-s] -= wj * x[s:]
        else:
            sp = -s
            grad[:n-sp] += wj * (1.0 - x[sp:])
            grad[sp:] -= wj * x[:n-sp]
    
    grad *= dx
    # Penalty gradient
    pen_coeff = 100.0
    grad += 2.0 * pen_coeff * (np.sum(x) - tgt)
    return grad

def topk_obj(x, n, top_k, beta, tgt):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    tv = corr[top_idx]
    m = np.max(tv)
    sm = m + np.log(np.sum(np.exp(beta * (tv - m)))) / beta
    penalty = 100.0 * (np.sum(x) - tgt) ** 2
    return sm * dx + penalty

def topk_grad(x, n, top_k, beta, tgt):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    tv = corr[top_idx]
    m = np.max(tv)
    exp_vals = np.exp(beta * (tv - m))
    weights_top = exp_vals / np.sum(exp_vals)
    
    weights_full = np.zeros(2*n-1)
    for i, j in enumerate(top_idx):
        weights_full[j] = weights_top[i]
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wj = weights_full[j]
        if wj < 1e-15:
            continue
        s = j - (n - 1)
        if s >= 0:
            grad[s:] += wj * (1.0 - x[:n-s])
            grad[:n-s] -= wj * x[s:]
        else:
            sp = -s
            grad[:n-sp] += wj * (1.0 - x[sp:])
            grad[sp:] -= wj * x[:n-sp]
    
    grad *= dx
    pen_coeff = 100.0
    grad += 2.0 * pen_coeff * (np.sum(x) - tgt)
    return grad

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h

def make_golden_ratio(n):
    h = np.zeros(n)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return h

def make_van_der_corput(n):
    h = np.zeros(n)
    for i in range(n):
        val = i + 1
        rev = 0
        for _ in range(20):
            rev = (rev << 1) | (val & 1)
            val >>= 1
        h[i] = 1.0 if (rev % 2) == 0 else 0.0
    return h

def make_low_discrepancy(n, alpha):
    """Liouville-like sequence"""
    h = np.zeros(n)
    for i in range(n):
        val = 0
        x = i + 1
        while x > 0:
            val += (x % 2) * (2 ** bin(x).count('1') - 1)  # Hmm this is getting complex
            x //= 2
        # Simpler: use bit-reversal parity
        cnt = bin(i).count('1')
        h[i] = 1.0 if (cnt * alpha) % 2 < 1 else 0.0
    return h

def binary_sa_fast(h_init, n, max_iters, rng, t_start=0.003, t_end=0.00005):
    """Faster binary SA with O(n) evaluation per step using diff correlation"""
    h = np.where(h_init > 0.5, 1.0, 0.0)
    target_ones = n // 2
    num_ones = int(np.sum(h))
    if num_ones > target_ones:
        ones_idx = np.where(h == 1.0)[0]
        remove = rng.choice(len(ones_idx), num_ones - target_ones, replace=False)
        h[ones_idx[remove]] = 0.0
    elif num_ones < target_ones:
        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
        h[zeros_idx[add]] = 1.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    
    ones_list = np.where(h == 1.0)[0].tolist()
    zeros_list = np.where(h == 0.0)[0].tolist()
    
    # Pre-compute correlation for fast updates
    corr = compute_corr(h)
    
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        
        i = rng.randint(len(ones_list))
        j = rng.randint(len(zeros_list))
        oi = ones_list[i]
        oz = zeros_list[j]
        
        # Apply move
        h[oi] = 0.0
        h[oz] = 1.0
        
        # Fast update of correlation
        # h changed: oi went 1->0, oz went 0->1
        # 1-h changed: oi went 0->1, oz went 1->0
        # C_k = sum_i h[i+k]*(1-h[i])
        # Change in C_k:
        # Contribution from oi in (1-h[.]): was 0, now 1 => adds h[oi+k] for each k
        # Contribution from oi in h[.+k]: was h[oi+k] terms, now 0 => removes terms where i+k=oi => i=oi-k
        # Contribution from oz in h[.+k]: was 0, now 1 => adds (1-h[oz-k]) terms... 
        # This is getting complex. Let me just recompute.
        
        new_c5 = compute_c5(h)
        T = t_start * (t_end / t_start) ** (float(it) / max(float(max_iters), 1.0))
        
        if new_c5 < current_c5 or (T > 1e-15 and rng.rand() < np.exp(-(new_c5 - current_c5) / T)):
            current_c5 = new_c5
            ones_list[i] = oz
            zeros_list[j] = oi
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0
    
    return best_h, best_c5

def binary_sa_inplace(h_init, n, max_iters, rng, t_start=0.003, t_end=0.00005):
    """Binary SA that directly optimizes the max correlation value"""
    h = np.where(h_init > 0.5, 1.0, 0.0)
    target_ones = n // 2
    num_ones = int(np.sum(h))
    if num_ones > target_ones:
        ones_idx = np.where(h == 1.0)[0]
        remove = rng.choice(len(ones_idx), num_ones - target_ones, replace=False)
        h[ones_idx[remove]] = 0.0
    elif num_ones < target_ones:
        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
        h[zeros_idx[add]] = 1.0
    
    dx = 2.0 / n
    best_h = h.copy()
    corr = compute_corr(h)
    best_maxcorr = np.max(corr)
    current_maxcorr = best_maxcorr
    
    ones_set = set(np.where(h == 1.0)[0].tolist())
    zeros_set = set(np.where(h == 0.0)[0].tolist())
    
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        
        oi = rng.choice(list(ones_set))
        oz = rng.choice(list(zeros_set))
        
        h[oi] = 0.0
        h[oz] = 1.0
        new_corr = compute_corr(h)
        new_maxcorr = np.max(new_corr)
        T = t_start * (t_end / t_start) ** (float(it) / max(float(max_iters), 1.0))
        
        accept = new_maxcorr < current_maxcorr or (T > 1e-15 and rng.rand() < np.exp(-(new_maxcorr - current_maxcorr) / (T * n)))
        
        if accept:
            current_maxcorr = new_maxcorr
            corr = new_corr
            ones_set.remove(oi)
            zeros_set.remove(oz)
            ones_set.add(oz)
            zeros_set.add(oi)
            if new_maxcorr < best_maxcorr:
                best_maxcorr = new_maxcorr
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0
    
    return best_h, best_maxcorr * dx

def local_search_binary(h, n, rng, max_rounds=100):
    """Try all single-element swaps and keep improvements"""
    dx = 2.0 / n
    h = h.copy()
    corr = compute_corr(h)
    current_max = np.max(corr)
    best_max = current_max
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for rnd in range(max_rounds):
        if remaining_time() < 5:
            break
        improved = False
        for oi in ones:
            for oz in zeros:
                h[oi] = 0.0
                h[oz] = 1.0
                new_corr = compute_corr(h)
                new_max = np.max(new_corr)
                if new_max < current_max:
                    current_max = new_max
                    if new_max < best_max:
                        best_max = new_max
                    improved = True
                    break
                else:
                    h[oi] = 1.0
                    h[oz] = 0.0
            if not improved:
                # Try a random swap
                oi = rng.choice(ones)
                oz = rng.choice(zeros)
                h[oi] = 0.0
                h[oz] = 1.0
                new_corr = compute_corr(h)
                new_max = np.max(new_corr)
                T = 0.001
                if new_max < current_max or rng.rand() < np.exp(-(new_max - current_max) / T):
                    current_max = new_max
                    if new_max < best_max:
                        best_max = new_max
                    improved = True
                else:
                    h[oi] = 1.0
                    h[oz] = 0.0
                break
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        if not improved:
            break
    
    return h, best_max * dx
