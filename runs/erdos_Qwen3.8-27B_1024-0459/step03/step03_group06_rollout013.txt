def unit_linear(p,h,n):
  L=zeros(2n-1); off=n-1
  # positive s 0..min(p,n-1): L[off+s] += 1 - h[p-s]
  # positive s 0..min(n-1-p,n-1): L[off+s] -= h[p+s]

def balance_binary(h, n, rng):
    h = np.asarray(h, dtype=float).copy()

    parity = np.zeros(n, dtype=int)
    for i in range(1, n):
        parity[i] = int(parity[i >> 1]) ^ (i & 1)
    return parity.astype(float)


def make_golden_ratio(n):
    gr = (1.0 + math.sqrt(5.0)) / 2.0
    x = (np.arange(n, dtype=float) * gr) % 1.0
    return (x < 0.5).astype(float)


def make_van_der_corput(n):
    h = np.zeros(n, dtype=float)
    for k in range(1, n + 1):
        x = k
        r = 0
        for _ in range(16):
            r = (r << 1) | (x & 1)
            x >>= 1
        if (r & 1) == 0:
            h[k - 1] = 1.0
    return h


def make_block(n, block):
    if block <= 1:
        block = 2
    p = np.arange(n) % block
    half = max(1, block // 2)
    return (p < half).astype(float)


def make_starts(n, best_h, best_n, rng):
    starts = []

    if best_h is not None and best_n > 0:
        hi = interp_to(best_h, n)
        hi = project_feasible(hi, n)
        starts.append(hi)
        hb = np.where(hi > 0.5, 1.0, 0.0)
        hb = balance_binary(hb, n, rng)
        starts.append(hb)

    starts.append(balance_binary(make_thue_morse(n), n, rng))
    starts.append(balance_binary(make_golden_ratio(n), n, rng))
    starts.append(balance_binary(make_van_der_corput(n), n, rng))
    starts.append(balance_binary(make_block(n, 6), n, rng))
    starts.append(balance_binary(make_block(n, 10), n, rng))
    starts.append(make_random_binary(n, rng))
    starts.append(make_random_binary(n, rng))

    return starts


def add_unit_linear(A, p, h, sign, n):
    off = n - 1
    if sign == 0.0:
        return

    if p > 0:
        lv = h[p - 1:p - 1:-1]
        A[off + 1:off + p + 1] += sign * (1.0 - lv)
        A[off - 1:off - p - 1:-1] += sign * (-lv)

    q = n - 1 - p
    if q > 0:
        rv = h[p + 1:p + q + 1]
        A[off + 1:off + q + 1] += sign * (-rv)
        A[off - 1:off - q - 1:-1] += sign * (1.0 - rv)

    A[off] += sign * (1.0 - 2.0 * h[p])


def active_grad(c, h, n, top_k):
    L = 2 * n - 1
    off = n - 1
    top_k = int(max(1, min(int(top_k), L)))

    maxc = float(np.max(c))
    tol = 4.0 * (2.0 / float(n))
    idx = np.where(c >= maxc - tol)[0]

    if idx.size == 0:
        idx = np.argpartition(c, -top_k)[-top_k:]
    elif idx.size > top_k:
        idx = np.argpartition(c, -top_k)[-top_k:]

    if idx.size < 3:
        k = max(3, top_k)
        idx = np.argpartition(c, -k)[-k:]

    g = np.zeros(n, dtype=float)
    for jj in idx:
        jj = int(jj)
        s = jj - off
        if s > 0:
            r = s
            g[r:n] += 1.0 - h[0:n - r]
            g[0:n - r] -= h[r:n]
        elif s < 0:
            r = -s
            g[0:n - r] += 1.0 - h[r:n]
            g[r:n] -= h[0:n - r]
        else:
            g += 1.0 - 2.0 * h

    g -= float(np.mean(g))
    return g


def apply_pair(h, c, src, sink, t, A, n):
    off = n - 1
    d = abs(int(src) - int(sink))
    t = float(t)

    h[src] -= t
    h[sink] += t
    if h[src] < 0.0:
        h[src] = 0.0
    if h[sink] > 1.0:
        h[sink] = 1.0

    if t != 0.0:
        c += t * A
        t2 = t * t
        c[off] += -2.0 * t2
        if d > 0:
            c[off + d] += t2
            c[off - d] += t2

    return c


def _pair_eval_max(c, A, taus, n, off, d):
    M = c[np.newaxis, :] + taus[:, np.newaxis] * A[np.newaxis, :]
    t2 = taus * taus
    M[:, off] += -2.0 * t2
    if d > 0:
        M[:, off + d] += t2
        M[:, off - d] += t2
    return np.max(M, axis=1)


def pair_search(h, c, n, src, sink, grid=0, refine=-1, positive_only=False):
    off = n - 1
    L = 2 * n - 1
    dx = 2.0 / float(n)

    src = int(src)
    sink = int(sink)
    T = min(float(h[src]), 1.0 - float(h[sink]))
    if T < 1e-14:
        return 0.0, float(np.max(c) * dx), None

    A = np.zeros(L, dtype=float)
    add_unit_linear(A, sink, h, 1.0, n)
    add_unit_linear(A, src, h, -1.0, n)

    if grid <= 0:
        grid = 40 if n <= 200 else 30
        if not isinstance(grid, int):
            grid = int(grid)
    grid = max(10, int(grid))

    pts = np.linspace(0.0, T, grid)
    if T > 1e-7:
        rel = np.array(
            [1e-5, 3e-5, 1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2,
             0.1, 0.25, 0.35, 0.5, 0.65, 0.75, 0.9], dtype=float)
        extra = rel * T
        extra = extra[(extra > 1e-12) & (extra < T)]
        if extra.size:
            pts = np.unique(np.concatenate((pts, extra)))

    d = abs(src - sink)
    vals = _pair_eval_max(c, A, pts, n, off, d)
    ii = int(np.argmin(vals))
    t = float(pts[ii])
    val = float(vals[ii] * dx)

    cur_val = float(np.max(c) * dx)

    if positive_only and t <= 1e-13:
        vp = vals.copy()
        vp[0] = np.inf
        ii = int(np.argmin(vp))
        t = float(pts[ii])
        val = float(vals[ii] * dx)

    if refine != 0 and t > 1e-14 and len(pts) > 2:
        if refine > 0:
            refn = int(refine)
        else:
            refn = 52 if n <= 200 else 36
        lo = float(pts[ii - 1]) if ii > 0 else 0.0
        hi = float(pts[ii + 1]) if ii + 1 < len(pts) else T
        if hi <= lo + 1e-14:
            if ii == 0:
                hi = float(pts[1])
            elif ii + 1 >= len(pts):
                lo = float(pts[-2])
            else:
                hi = lo
        if hi > lo + 1e-14:
            rt = np.linspace(lo, hi, refn)
            rv = _pair_eval_max(c, A, rt, n, off, d)
            ir = int(np.argmin(rv))
            rtv = float(rv[ir] * dx)
            if rtv < val:
                t = float(rt[ir])
                val = rtv

    if positive_only:
        if t > 1e-13:
            return t, val, A
        return 0.0, cur_val, None

    if t > 1e-13 and val < cur_val - 1e-15:
        return t, val, A
    return 0.0, cur_val, None


def pair_local_search(h_in, c5_in, n, rng, deadline):
    h = project_feasible(h_in, n).copy()
    c = compute_corr(h)
    dx = 2.0 / float(n)
    cur = float(np.max(c) * dx)
    best_h = h.copy()
    best_c = cur

    off = n - 1
    top_k = 12 if n <= 256 else 18
    M = 9 if n <= 220 else 6
    rand_pairs = 24 if n <= 220 else 12
    grid_local = 38 if n <= 200 else 30

    temp = 0.004
    temp_min = 1e-6
    no_improve = 0
    accept_count = 0

    while time.time() < deadline:
        if accept_count > 0 and (accept_count % 32) == 0:
            h = np.clip(h, 0.0, 1.0)
            c = compute_corr(h)
            cur = float(np.max(c) * dx)

        g = active_grad(c, h, n, top_k)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        order_src = np.argsort(g)[::-1]
        order_sink = np.argsort(g)

        srcs = []
        for ii in order_src:
            ii = int(ii)
            if len(srcs) >= M:
                break
            if h[ii] > 1e-9:
                srcs.append(ii)
        while len(srcs) < 2:
            ii = int(rng.randint(n))
            if h[ii] > 1e-9 and ii not in srcs:
                srcs.append(ii)
            else:
                break

        sinks = []
        for ii in order_sink:
            ii = int(ii)
            if len(sinks) >= M:
                break
            if h[ii] < 1.0 - 1e-9:
                sinks.append(ii)
        while len(sinks) < 2:
            ii = int(rng.randint(n))
            if h[ii] < 1.0 - 1e-9 and ii not in sinks:
                sinks.append(ii)
            else:
                break

        pairs = []
        for si in srcs:
            for ki in sinks:
                if si == ki:
                    continue
                if h[si] > 1e-9 and h[ki] < 1.0 - 1e-9:
                    pairs.append((si, ki))

        for _ in range(rand_pairs):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            if h[i] > 1e-9 and h[j] < 1.0 - 1e-9:
                pairs.append((i, j))
            if h[j] > 1e-9 and h[i] < 1.0 - 1e-9:
                pairs.append((j, i))

        max_pairs = 170 if n <= 200 else 80
        if len(pairs) > max_pairs:
            pairs = pairs[:max_pairs]

        best_pair = None
        best_val = cur
        tested = 0

        for src, sink in pairs:
            if time.time() >= deadline:
                break
            if g[src] - g[sink] < -1e-7:
                continue
            t, val, A = pair_search(h, c, n, src, sink, grid=grid_local, refine=-1)
            tested += 1
            if t > 0.0 and val < best_val - 1e-14:
                best_val = val
                best_pair = (src, sink, t, A)
            if tested % 64 == 0 and time.time() >= deadline:
                break

        if best_pair is not None:
            src, sink, t, A = best_pair
            c = apply_pair(h, c, src, sink, t, A, n)
            cur = best_val
            accept_count += 1
            no_improve = 0
            if cur < best_c - 1e-15:
                best_c = cur
                best_h = h.copy()
            temp = max(temp_min, temp * 0.988)
        else:
            no_improve += 1

            if temp > 1.1 * temp_min:
                found = False
                for _ in range(22):
                    if time.time() >= deadline:
                        break
                    i = int(rng.randint(n))
                    j = int(rng.randint(n))
                    if i == j:
                        continue
                    if h[i] > 1e-9 and h[j] < 1.0 - 1e-9:
                        src, sink = i, j
                    elif h[j] > 1e-9 and h[i] < 1.0 - 1e-9:
                        src, sink = j, i
                    else:
                        continue
                    t, val, A = pair_search(
                        h, c, n, src, sink,
                        grid=22 if n <= 200 else 16,
                        refine=0,
                        positive_only=True)
                    if t > 1e-12 and val < cur + temp:
                        c = apply_pair(h, c, src, sink, t, A, n)
                        cur = val
                        accept_count += 1
                        no_improve = 0
                        if cur < best_c - 1e-15:
                            best_c = cur
                            best_h = h.copy()
                        found = True
                        break
                temp = max(temp_min, temp * (0.975 if found else 0.992))

            if cur > best_c + 0.0018:
                h = best_h.copy()
                c = compute_corr(h)
                cur = float(np.max(c) * dx)
                no_improve = 0

            if no_improve > 90:
                break

    h = project_feasible(h, n)
    c = compute_corr(h)
    cur = float(np.max(c) * dx)
    if cur < best_c:
        best_c = cur
        best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, best_c


def all_pairs_sweep(h_in, n, rng, deadline):
    h = project_feasible(h_in, n).copy()
    c = compute_corr(h)
    dx = 2.0 / float(n)
    cur = float(np.max(c) * dx)
    best_h = h.copy()
    best_c = cur

    grid_ap = 20 if n <= 160 else 15
    accept = 0

    while time.time() < deadline:
        improved = False

        for i in range(n):
            if time.time() >= deadline:
                break
            for j in range(i + 1, n):
                if time.time() >= deadline:
                    break

                if h[i] > 1e-9 and h[j] < 1.0 - 1e-9:
                    t, val, A = pair_search(h, c, n, i, j, grid=grid_ap, refine=0)
                    if t > 1e-12 and val < cur - 1e-13:
                        c = apply_pair(h, c, i, j, t, A, n)
                        cur = val
                        improved = True
                        accept += 1
                        if accept % 80 == 0:
                            h = np.clip(h, 0.0, 1.0)
                            c = compute_corr(h)
                            cur = float(np.max(c) * dx)
                        if cur < best_c:
                            best_c = cur
                            best_h = h.copy()

                if h[j] > 1e-9 and h[i] < 1.0 - 1e-9:
                    t, val, A = pair_search(h, c, n, j, i, grid=grid_ap, refine=0)
                    if t > 1e-12 and val < cur - 1e-13:
                        c = apply_pair(h, c, j, i, t, A, n)
                        cur = val
                        improved = True
                        accept += 1
                        if accept % 80 == 0:
                            h = np.clip(h, 0.0, 1.0)
                            c = compute_corr(h)
                            cur = float(np.max(c) * dx)
                        if cur < best_c:
                            best_c = cur
                            best_h = h.copy()

        if not improved:
            break

    h = project_feasible(h, n)
    c = compute_corr(h)
    cur = float(np.max(c) * dx)
    if cur < best_c:
        best_c = cur
        best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, best_c


def active_gradient_steps(h_in, n, deadline, max_iter=400):
    h = project_feasible(h_in, n).copy()
    c = compute_corr(h)
    dx = 2.0 / float(n)
    cur = float(np.max(c) * dx)
    best_h = h.copy()
    best_c = cur

    top_k = 16 if n <= 256 else 22
    steps = [0.003, 0.006, 0.012, 0.025, 0.05, 0.10, 0.18]
    no_improve = 0

    for _ in range(int(max_iter)):
        if time.time() >= deadline:
            break

        g = active_grad(c, h, n, top_k)
        if not np.all(np.isfinite(g)):
            break
        if float(np.linalg.norm(g)) < 1e-13:
            break

        best_val = cur
        best_cand = None
        for a in steps:
            cand = h - a * g
            cand = project_feasible(cand, n)
            cv = compute_c5(cand)
            if cv < best_val - 1e-14:
                best_val = cv
                best_cand = cand

        if best_cand is not None:
            h = best_cand
            c = compute_corr(h)
            cur = best_val
            no_improve = 0
            if cur < best_c:
                best_c = cur
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 10:
                break

    h = project_feasible(h, n)
    c = compute_corr(h)
    cur = float(np.max(c) * dx)
    if cur < best_c:
        best_c = cur
        best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, best_c


_SM_N = 0
_SM_BETA = 100.0
_SM_TGT = 0.0
_SM_PEN = 500.0


def _smooth_obj(x):
    n = _SM_N
    if n <= 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = compute_corr(x)
    m = float(np.max(corr))
    sh = corr - m
    e = np.exp(_SM_BETA * sh)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        sm = m
    else:
        sm = m + math.log(s) / _SM_BETA
    val = sm * dx
    diff = float(np.sum(x)) - _SM_TGT
    val += _SM_PEN * diff * diff
    return float(val)


def _smooth_grad(x):
    n = _SM_N
    if n <= 0:
        return np.zeros(0, dtype=float)
    dx = 2.0 / float(n)
    corr = compute_corr(x)
    m = float(np.max(corr))
    e = np.exp(_SM_BETA * (corr - m))
    s = float(np.sum(e))
    off = n - 1

    if s <= 0.0 or not np.isfinite(s):
        w = np.full(2 * n - 1, 1.0 / float(2 * n - 1))
    else:
        w = e / s

    g = np.zeros(n, dtype=float)
    for jj in range(2 * n - 1):
        wt = float(w[jj])
        if wt < 1e-9:
            continue
        s2 = jj - off
        if s2 > 0:
            r = s2
            g[r:n] += wt * (1.0 - x[0:n - r])
            g[0:n - r] -= wt * x[r:n]
        elif s2 < 0:
            r = -s2
            g[0:n - r] += wt * (1.0 - x[r:n])
            g[r:n] -= wt * x[0:n - r]
        else:
            g += wt * (1.0 - 2.0 * x)

    g *= dx
    diff = float(np.sum(x)) - _SM_TGT
    g += 2.0 * _SM_PEN * diff
    return g


def l_bfgs_polish(h_in, n, betas, max_iter, deadline):
    if minimize is None:
        return h_in, compute_c5(h_in)

    h = project_feasible(h_in, n).copy()
    best_c = compute_c5(h)
    bounds = [(0.0, 1.0)] * int(n)
    global _SM_N, _SM_BETA, _SM_TGT, _SM_PEN

    for beta in betas:
        if time.time() >= deadline:
            break
        _SM_N = int(n)
        _SM_BETA = float(beta)
        _SM_TGT = 0.5 * float(n)
        _SM_PEN = 400.0

        h0 = project_feasible(h, n)
        mi = int(max_iter) if n <= 250 else int(max(80, 0.6 * max_iter))
        try:
            res = minimize(
                _smooth_obj, h0, jac=_smooth_grad,
                method="L-BFGS-B", bounds=bounds,
                options={
                    "maxiter": mi,
                    "ftol": 1e-15,
                    "gtol": 1e-10,
                    "iprint": 0,
                    "maxls": 25
                })
            if res.x is not None and np.all(np.isfinite(res.x)):
                hnew = project_feasible(res.x, n)
                cnew = compute_c5(hnew)
                if np.isfinite(cnew) and cnew < best_c:
                    h = hnew
                    best_c = cnew
        except Exception:
            pass

    h = project_feasible(h, n)
    best_c = compute_c5(h)
    return h, best_c


def optimize_n(h_in, n, rng, deadline):
    h = project_feasible(h_in, n).copy()
    c5 = compute_c5(h)
    total = max(0.0, deadline - time.time())
    if total <= 0.0:
        return h, c5

    if total > 35.0:
        d0 = min(deadline, time.time() + min(70.0, 0.20 * total))
        h, c5 = active_gradient_steps(h, n, d0, max_iter=500)

    if deadline - time.time() > 25.0:
        h, c5 = pair_local_search(h, c5, n, rng, deadline - 5.0)

    if n <= 192 and deadline - time.time() > 80.0:
        d1 = min(deadline - 5.0, time.time() + max(40.0, 0.25 * total))
        h2, c2 = all_pairs_sweep(h, n, rng, d1)
        if c2 < c5:
            h, c5 = h2, c2

    if deadline - time.time() > 35.0:
        h, c5 = pair_local_search(h, c5, n, rng, deadline - 2.0)

    return h, c5


def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    hard_deadline = start + max(60.0, float(budget_s)) - 8.0
    rng = np.random.RandomState(int(seed))

    best_h = None
    best_c5 = float("inf")
    best_n = 160

    parent = globals().get("initial_h_values")
    if parent is not None:
        ph = np.asarray(parent, dtype=float)
        pn = int(len(ph))
        if pn > 2:
            if (np.all(ph >= -1e-8) and np.all(ph <= 1.0 + 1e-8) and
                    abs(float(np.sum(ph)) - 0.5 * float(pn)) < 1e-7):
                h0 = np.clip(ph, 0.0, 1.0)
            else:
                h0 = project_feasible(ph, pn)
            c0 = compute_c5(h0)
            if np.isfinite(c0):
                best_h = h0
                best_c5 = c0
                best_n = pn

    if best_h is None:
        base_n = 160
        starts = make_starts(base_n, None, 0, rng)
        for h in starts:
            h = project_feasible(h, base_n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
                best_n = base_n
        if best_h is None:
            best_h = make_random_binary(160, rng)
            best_h = project_feasible(best_h, 160)
            best_c5 = compute_c5(best_h)
            best_n = 160

    reserve = 110.0
    avail = max(60.0, hard_deadline - time.time() - reserve)
    parent_phase = 0.45 * avail
    other_phase = avail - parent_phase

    other_n = []
    other_w = []
    for n in [128, 160, 192, 256, 320]:
        if n != best_n and 80 <= n <= 320:
            if n <= 128:
                w = 1.0
            elif n <= 160:
                w = 0.92
            elif n <= 192:
                w = 0.85
            elif n <= 256:
                w = 0.72
            else:
                w = 0.60
            other_n.append(int(n))
            other_w.append(w)

    phase_times = {}
    if best_n > 0:
        phase_times[int(best_n)] = parent_phase
    tw = sum(other_w)
    if tw > 0:
        for n, w in zip(other_n, other_w):
            phase_times[int(n)] = other_phase * (w / tw)

    order = [int(best_n)]
    for n in [128, 160, 192, 256, 320]:
        if n not in order:
            order.append(n)

    for n in order:
        if time.time() >= hard_deadline - 35.0:
            break
        phase_t = float(phase_times.get(n, 0.0))
        if phase_t < 28.0:
            continue

        phase_deadline = min(hard_deadline - 20.0, time.time() + phase_t)
        starts = make_starts(n, best_h, best_n, rng)

        scored = []
        for idx, h in enumerate(starts):
            if time.time() >= phase_deadline - 6.0:
                break
            h = project_feasible(h, n)
            c5 = compute_c5(h)
            if not np.isfinite(c5):
                continue
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n
            scored.append((c5, idx, h))

        scored.sort()

        selected = []
        for c5, idx, h in scored:
            if len(selected) >= 3:
                break
            if len(selected) == 0 or c5 < best_c5 + 0.02:
                selected.append(h)
        if not selected and scored:
            selected.append(scored[0][2])

        for idx, h in enumerate(selected):
            if time.time() >= phase_deadline - 5.0:
                break
            rem_phase = phase_deadline - time.time()
            if idx == 0 and len(selected) > 1:
                sub_deadline = time.time() + max(45.0, 0.62 * rem_phase)
            else:
                sub_deadline = phase_deadline

            h = project_feasible(h, n)
            c5 = compute_c5(h)
            if c5 > best_c5 + 0.035 and sub_deadline - time.time() > 70.0:
                sub_deadline = time.time() + 70.0

            h2, c2 = optimize_n(h, n, rng, sub_deadline)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()
                best_n = n

    if (time.time() < hard_deadline - 75.0 and best_h is not None and minimize is not None):
        db = min(hard_deadline - 20.0, time.time() + 65.0)
        h2, c2 = l_bfgs_polish(best_h, best_n, [160.0, 420.0], 220, db)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()
            best_n = best_n

    if time.time() < hard_deadline - 25.0 and best_h is not None:
        db = min(hard_deadline - 8.0, time.time() + 30.0)
        h2, c2 = pair_local_search(best_h, best_c5, best_n, rng, db)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()
            best_n = best_n

    best_h = project_feasible(best_h, int(best_n))
    final_c5 = compute_c5(best_h)

    evalfn = globals().get("evaluate_erdos_solution")
    if evalfn is not None:
        try:
            ev = evalfn(best_h)
            if np.isfinite(ev) and ev < final_c5:
                final_c5 = float(ev)
        except Exception:
            pass

    return (best_h, float(final_c5), int(best_n))
