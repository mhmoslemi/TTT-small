def compute_delta_corr(h, i, j, n):
    """Compute change in all correlations when h[i]:1->0, h[j]:0->1."""
    # h is binary
    delta = np.zeros(2*n - 1)
    
    # For each lag m from -(n-1) to (n-1):
    # delta[m + (n-1)] = change in R(m)
    
    for m in range(-(n-1), n):
        if m == 0:
            continue
        dm = 0.0
        if m > 0:
            # R(m) = sum_{a=0}^{n-1-m} h[a+m]*(1-h[a])
            # h[i] 1->0:
            #   as (1-h[a]) at a=i: was 0, becomes h[i+m] (if i+m < n-m+0... i.e., i+m < n)
            #   Actually a ranges 0 to n-1-m, so a=i valid if i <= n-1-m
            if i <= n-1-m:
                dm += h[i+m]  # (1-h[i]) goes 0->1, so term h[i+m]*(1-h[i]) goes 0 -> h[i+m]
            #   as h[a+m] at a=i-m: was h[i]*(1-h[i-m]) = (1-h[i-m]), becomes 0
            if i-m >= 0 and i-m <= n-1-m:
                dm -= (1 - h[i-m])
            # h[j] 0->1:
            #   as (1-h[a]) at a=j: was h[j+m]*1 = h[j+m], becomes 0
            if j <= n-1-m:
                dm -= h[j+m]
            #   as h[a+m] at a=j-m: was 0, becomes (1-h[j-m])
            if j-m >= 0 and j-m <= n-1-m:
                dm += (1 - h[j-m])
        else:
            p = -m
            # R(-p) = sum_{a=p}^{n-1} h[a-p]*(1-h[a])
            # h[i] 1->0:
            #   as h[a-p] at a=i+p: was (1-h[i+p]), becomes 0
            if i+p >= p and i+p < n:  # a = i+p, need a >= p and a < n
                dm -= (1 - h[i+p])
            #   as (1-h[a]) at a=i: was 0, becomes h[i-p]
            if i >= p and i < n:
                dm += h[i-p]
            # h[j] 0->1:
            #   as h[a-p] at a=j+p: was 0, becomes (1-h[j+p])
            if j+p >= p and j+p < n:
                dm += (1 - h[j+p])
            #   as (1-h[a]) at a=j: was h[j-p], becomes 0
            if j >= p and j < n:
                dm -= h[j-p]
        
        delta[m + (n-1)] = dm
    
    return delta

def slsqp_refine(h, n, time_limit):
    """Continuous SLSQP refinement."""
    n_local = n
    target = n / 2.0
    
    def obj(x):
        dx_v = 2.0 / n_local
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        beta = 3000.0
        sh = np.clip(beta * (corr - m), -60, 60)
        return (m + np.log(np.sum(np.exp(sh))) / beta) * dx_v
    
    def obj_grad(x):
        dx_v = 2.0 / n_local
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        beta = 3000.0
        sh = np.clip(beta * (corr - m), -60, 60)
        ex = np.exp(sh)
        w = ex / np.sum(ex)
        
        g = np.zeros(n_local)
        for k in range(2 * n_local - 1):
            if w[k] < 1e-14:
                continue
            off = k - (n_local - 1)
            if off >= 0:
                np_ = n_local - off
                g[off:off + np_] += (1.0 - x[:np_])
                g[:np_] -= x[off:off + np_]
            else:
                ao = -off
                np_ = n_local - ao
                g[:np_] += (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= x[:np_]
        return g * dx_v
    
    def con_fun(x):
        return np.sum(x) - target
    
    def con_jac(x):
        return np.ones(n_local)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-18, 'iprint': 0})
