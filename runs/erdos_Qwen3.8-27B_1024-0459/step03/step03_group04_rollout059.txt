def project_h(h, n):

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break

For d ≥ 0: c[j] = sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
dc/dh[m] = (1-h[m]) * [m ∈ [0, n-1-d]] + (-h[m-d]) * [m-d ∈ [0, n-1-d], i.e., m ∈ [d, n-1]]

For d < 0, a = -d: c[j] = sum_{i=a}^{n-1} h[i-a]*(1-h[i])
dc/dh[m] = (1-h[m]) * [m ∈ [a, n-1]] + (-h[m+a]) * [m+a ∈ [a, n-1], i.e., m ∈ [0, n-1-a]]

Great. Now I can implement the min-t formulation.

Variables: [h[0], ..., h[n-1], t]
Objective: minimize t
Constraints:
- sum(h) = n/2 (equality)
- For each active j: c[j](h) - t ≤ 0 (inequality, quadratic)
- Bounds: 0 ≤ h[i] ≤ 1, t ≥ 0 (or some lower bound)

This is a QCQP that SLSQP should handle.

_g_active = None

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def project_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nfree = int(np.sum(free))
        if nfree > 0:
            h[free] += diff / nfree
            h = np.clip(h, 0.0, 1.0)
        else:
            break
    diff = target - np.sum(h)
    if abs(diff) > 0.0:
        for i in range(n):
            nv = h[i] + diff
            if 0.0 <= nv <= 1.0:
                h[i] = nv
                break
    return h

    if src_n == dst_n:
        return src.copy()

    xq = (np.arange(dst_n) + 0.5) * 2.0 / dst_n

            m = n - a
            J[i, a:a+m] = (1.0 - h[a:a+m])
            J[i, :m] -= h[a:a+m]

    return J
