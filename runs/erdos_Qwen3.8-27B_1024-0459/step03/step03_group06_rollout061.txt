```python
def swap_delta_vec(h, i, j, n):
    """Vectorized swap delta."""
    delta = np.zeros(2 * n - 1)
    
    # i removal from h[n+k]: k ranges from max(0, i-(n-1)) to min(2n-2, i)
    k_lo = max(0, i - (n - 1))
    k_hi = min(2 * n - 2, i)
    if k_lo <= k_hi:
        n_indices = i - np.arange(k_lo, k_hi + 1)  # n = i - k
        delta[k_lo:k_hi+1] -= (1.0 - h[n_indices])
    
    # i removal from (1-h[n]): k ranges from max(0, -i) to min(2n-2, n-1-i)
    k_lo = 0  # k >= 0 always, and k >= -i is always true since k >= 0
    k_hi = min(2 * n - 2, n - 1 - i)
    if k_lo <= k_hi:
        delta[k_lo:k_hi+1] += h[i:i+k_hi+1]
    
    # j addition to h[n+k]: k ranges from max(0, j-(n-1)) to min(2n-2, j)
    k_lo = max(0, j - (n - 1))
    k_hi = min(2 * n - 2, j)
    if k_lo <= k_hi:
        n_indices = j - np.arange(k_lo, k_hi + 1)
        delta[k_lo:k_hi+1] += (1.0 - h[n_indices])
    
    # j addition to (1-h[n]): k ranges from 0 to min(2n-2, n-1-j)
    k_hi = min(2 * n - 2, n - 1 - j)
    if 0 <= k_hi:
        delta[0:k_hi+1] -= h[j:j+k_hi+1]
    
    return delta
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        unsat = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(unsat)
        if cnt > 0:
            h[unsat] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def fast_smooth_grad(h, beta, target_sum, n):
    """Vectorized gradient of smooth-max objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for k in range(n):
        w = weights[n - 1 + k]
        if w < 1e-15:
            continue
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
    for s in range(1, n):
        w = weights[n - 1 - s]
        if w < 1e-15:
            continue
        grad[:n - s] += w * (1.0 - h[s:])
        grad[s:] -= w * h[:n - s]
    grad *= dx
    grad += 2.0 * 500.0 * (np.sum(h) - target_sum)
    return grad

def fast_smooth_obj(h, beta, target_sum, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sm = (m + np.log(np.sum(np.exp(beta * (corr - m))))) / beta * dx
    penalty = 500.0 * (np.sum(h) - target_sum) ** 2
    return sm + penalty

def swap_delta_vec(h, i, j, n):
    """Compute change in full correlation when h[i]:1->0, h[j]:0->1. Vectorized."""
    delta = np.zeros(2 * n - 1)
    # i was 1, now 0:
    # h[n+k] position: n+k=i -> n=i-k, valid 0<=i-k<n, 0<=k<=2n-2
    klo = max(0, i - (n - 1))
    khi = min(2 * n - 2, i)
    if klo <= khi:
        n_idx = i - np.arange(klo, khi + 1)
        delta[klo:khi + 1] -= (1.0 - h[n_idx])
    # (1-h[n]) position: n=i, h[n+k]=h[i+k], valid 0<=i+k<n
    khi2 = min(2 * n - 2, n - 1 - i)
    if khi2 >= 0:
        delta[0:khi2 + 1] += h[i:i + khi2 + 1]
    # j was 0, now 1:
    # h[n+k] position: n+k=j -> n=j-k
    klo = max(0, j - (n - 1))
    khi = min(2 * n - 2, j)
    if klo <= khi:
        n_idx = j - np.arange(klo, khi + 1)
        delta[klo:khi + 1] += (1.0 - h[n_idx])
    # (1-h[n]) position: n=j, h[n+k]=h[j+k], valid 0<=j+k<n
    khi2 = min(2 * n - 2, n - 1 - j)
    if khi2 >= 0:
        delta[0:khi2 + 1] -= h[j:j + khi2 + 1]
    return delta

def binary_local_search(h_bin, n, max_time=100.0, rng=None, verbose=True):
    """Efficient binary local search using fast swap evaluation."""
    if rng is None:
        rng = np.random.RandomState(0)
    h = h_bin.copy()
    t0 = time.time()
    
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    
    corr = compute_full_corr(h)
    cur_max = np.max(corr)
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n
    
    no_improve = 0
    max_no_improve = 5000
    
    while time.time() - t0 < max_time:
        # Random restart if stuck
        if no_improve > max_no_improve:
            # Try 200 random perturbations
            improved_any = False
            for _ in range(200):
                if len(ones_list) < 2 or len(zeros_list) < 2:
                    break
                i = rng.randint(len(ones_list))
                j = rng.randint(len(zeros_list))
                oi, oz = ones_list[i], zeros_list[j]
                delta = swap_delta_vec(h, oi, oz, n)
                new_corr = corr + delta
                new_max = np.max(new_corr)
                if new_max <= cur_max:
                    h[oi] = 0.0
                    h[oz] = 1.0
                    corr = new_corr
                    cur_max = new_max
                    ones_list[i] = oz
                    zeros_list[j] = oi
                    improved_any = True
                    if cur_max < best_max:
                        best_max = cur_max
                        best_h = h.copy()
            if not improved_any:
                break
            no_improve = 0
        
        # Try random swaps
        i = rng.randint(len(ones_list))
        j = rng.randint(len(zeros_list))
        oi, oz = ones_list[i], zeros_list[j]
        
        delta = swap_delta_vec(h, oi, oz, n)
        new_corr = corr + delta
        new_max = np.max(new_corr)
        
        if new_max <= cur_max:
            h[oi] = 0.0
            h[oz] = 1.0
            corr = new_corr
            cur_max = new_max
            ones_list[i] = oz
            zeros_list[j] = oi
            no_improve = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            no_improve += 1
    
    return best_h, best_max * dx

def binary_local_search_targeted(h_bin, n, max_time=100.0, rng=None, top_k=5):
    """Targeted binary local search focusing on active lags."""
    if rng is None:
        rng = np.random.RandomState(0)
    h = h_bin.copy()
    t0 = time.time()
    dx = 2.0 / n
    
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    
    corr = compute_full_corr(h)
    cur_max = np.max(corr)
    best_h = h.copy()
    best_max = cur_max
    
    no_improve = 0
    
    while time.time() - t0 < max_time:
        if no_improve > 10000:
            break
        
        # Get active lags
        active = np.argsort(corr)[-top_k:]
        active_vals = corr[active]
        
        best_swap = None
        best_new_max = cur_max
        best_delta = None
        
        # Try targeted swaps: positions in active lags
        candidates = []
        for _ in range(50):
            i = rng.randint(len(ones_list))
            j = rng.randint(len(zeros_list))
            candidates.append((i, j))
        
        for idx_i, idx_j in candidates:
            oi, oz = ones_list[idx_i], zeros_list[idx_j]
            delta = swap_delta_vec(h, oi, oz, n)
            # Quick check on active lags
            new_active = active_vals + delta[active]
            quick_max = np.max(new_active)
            if quick_max < best_new_max:
                # Full check
                new_max = np.max(corr + delta)
                if new_max <= cur_max:
                    best_new_max = new_max
                    best_swap = (idx_i, idx_j)
                    best_delta = delta
        
        if best_swap is not None:
            idx_i, idx_j = best_swap
            oi, oz = ones_list[idx_i], zeros_list[idx_j]
            h[oi] = 0.0
            h[oz] = 1.0
            corr = corr + best_delta
            cur_max = np.max(corr)
            ones_list[idx_i] = oz
            zeros_list[idx_j] = oi
            no_improve = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            no_improve += 1
            # Random perturbation to escape
            if no_improve % 2000 == 0:
                for _ in range(10):
                    i = rng.randint(len(ones_list))
                    j = rng.randint(len(zeros_list))
                    oi, oz = ones_list[i], zeros_list[j]
                    delta = swap_delta_vec(h, oi, oz, n)
                    h[oi] = 0.0
                    h[oz] = 1.0
                    corr = corr + delta
                    ones_list[i] = oz
                    zeros_list[j] = oi
                cur_max = np.max(corr)
                no_improve = 0
    
    return best_h, best_max * dx

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h

def make_binary_random(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def make_msequence(n, rng):
    """Generate a pseudo-m-sequence based binary pattern."""
    # Use a simple LFSR-like construction
    period = 127  # 2^7 - 1
    h = np.zeros(n)
    # Use tukey's 4-tap Fibo LFSR
    state = np.array([1, 0, 0, 0, 0, 0, 0])
    for i in range(n):
        h[i] = state[0]
        feedback = state[0] ^ state[3]  # taps at 0 and 3
        state = np.roll(state, 1)
        state[0] = feedback
    # Adjust to have exactly n/2 ones
    num_ones = int(np.sum(h))
    if num_ones > n // 2:
        ones = np.where(h == 1)[0]
        remove_idx = rng.choice(ones, num_ones - n // 2, replace=False)
        h[remove_idx] = 0
    elif num_ones < n // 2:
        zeros = np.where(h == 0)[0]
        add_idx = rng.choice(zeros, n // 2 - num_ones, replace=False)
        h[add_idx] = 1
    return h

def make_strided(n, rng):
    """Create pattern with structured gaps."""
    h = np.zeros(n)
    # Place ones at positions that are multiples of various strides
    base_stride = rng.randint(3, 8)
    positions = set()
    for start in range(base_stride):
        for pos in range(start, n, base_stride):
            if len(positions) < n // 2:
                positions.add(pos)
            else:
                break
    for p in positions:
        h[p] = 1.0
    # Fill remaining
    if np.sum(h) < n // 2:
        remaining = np.where(h == 0)[0]
        extra = int(n // 2 - np.sum(h))
        add_idx = rng.choice(remaining, extra, replace=False)
        h[add_idx] = 1
    return h

def continuous_optimize(h0, n, max_time=100.0, rng=None, verbose=False):
    """L-BFGS-B continuous optimization."""
    t0 = time.time()
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_h = project_feasible(h0, n)
    best_c5 = compute_c5(best_h)
    
    for beta in [50, 150, 400, 800]:
        if time.time() - t0 > max_time - 5:
            break
        def obj(x):
            return fast_smooth_obj(x, beta, target_sum, n)
        def grad(x):
            return fast_smooth_grad(x, beta, target_sum, n)
        
        try:
            res = minimize(obj, best_h, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': 300, 'ftol': 1e-15,
                                   'gtol': 1e-10, 'iprint': 0, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                if verbose:
                    print("    beta={}, C5={:.6f}".format(beta, c5_new))
        except Exception:
            pass
    
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
```
