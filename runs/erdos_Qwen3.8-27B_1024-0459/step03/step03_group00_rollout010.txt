def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def to_binary(h, n):

    h = np.array(h, dtype=float)
    # Sort by value
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:n//2]] = 1.0
    return b
