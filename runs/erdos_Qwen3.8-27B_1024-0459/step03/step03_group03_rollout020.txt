from scipy.optimize import minimize
import time

_t0 = 0.0
_tb = 1300.0
_ng = 0
_bv = 100.0

def _tl():
    return max(0.0, _tb - (time.time() - _t0) - 5.0)

def _c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / n)

def _proj(h, n):
    h = np.array(h, dtype=np.float64).copy()
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * n
    for _ in range(60):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0 - 1e-10
            k = int(mask.sum())
            if k == 0:
                break
            h[mask] += d / k
            np.clip(h, 0.0, 1.0, out=h)
        else:
            mask = h > 1e-10
            k = int(mask.sum())
            if k == 0:
                break
            h[mask] += d / k
            np.clip(h, 0.0, 1.0, out=h)
    # Final exact correction
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # Spread over all non-boundary elements
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        k = int(mask.sum())
        if k > 0:
            h[mask] += diff / k
            np.clip(h, 0.0, 1.0, out=h)
        else:
            idx = int(np.argmin(np.abs(h - 0.5)))
            h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def _interp(h, ns, nd):

        return h.copy()
    xs = (np.arange(ns) + 0.5) * (2.0 / ns)
    xd = (np.arange(nd) + 0.5) * (2.0 / nd)
    return np.interp(xd, xs, h)

def _scon(x):
    return x.sum() - 0.5 * _ng

def _sobj(x):

    c = np.correlate(x, 1.0 - x, mode='full')
    mx = float(np.max(c))
    b = _bv
    sh = c - mx
    return (mx + np.log(np.exp(b * sh).sum()) / b) * (2.0 / n)

def _sgrad(x):
    n = _ng
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = float(np.max(c))
    b = _bv
    sh = c - mx
    w = np.exp(b * sh)
    sw = w.sum()

        w[int(np.argmax(c))] = 1.0

def _proj_bssum(z, n):
    """Project z onto {h: 0<=h_i<=1, sum(h)=n/2}."""
    target = 0.5 * n
    lo, hi = z.min() - 1.0, z.max()
    for _ in range(60):
        tau = 0.5 * (lo + hi)
        h = np.clip(z - tau, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = tau
        elif s < target:
            hi = tau
        else:
            break
    h = np.clip(z - tau, 0.0, 1.0)
    # Fix any remaining numerical error
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Spread over non-bound elements
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        k = mask.sum()
        if k > 0:
            h[mask] += diff / k
        else:
            idx = np.argmin(np.abs(h - 0.5))
            h[idx] += diff
            h[idx] = np.clip(h[idx], 0.0, 1.0)
    return h

def _feasibility_obj(h, T):
    """Sum of violations of corr_k <= T."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    violations = np.maximum(c * (2.0/n) - T, 0)
    return violations.sum()

def _feasibility_grad(h, T):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
    active = c > T + 1e-10
    if not active.any():
        return np.zeros(n)
    # Gradient only from active constraints
    g = np.zeros(n)
    for k in np.where(active)[0]:
        ki = int(k)
        s = ki - (n-1)
        if s >= 0:
            g[:n-s] += (1.0 - h[s:])
            g[s:] += (-h[:n-s])
        else:
            p = -s
            g[p:] += (1.0 - h[:n-p])
            g[:n-p] += (-h[p:])
    g *= (2.0/n)
    return g

    z = h.copy()
    target = 0.5 * n
    lo = z.min() - 2.0
    hi = z.max() + 2.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        h_tmp = np.clip(z - tau, 0.0, 1.0)
        s = h_tmp.sum()
        if s > target:
            lo = tau
        elif s < target:
            hi = tau
        else:
            break
    h = np.clip(z - tau, 0.0, 1.0)
    # Fix residual
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        k = int(mask.sum())
        if k > 0:
            h[mask] += diff / k
            np.clip(h, 0.0, 1.0, out=h)
        else:
            for idx in np.argsort(np.abs(h - 0.5)):
                h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
                if abs(target - h.sum()) < 1e-13:
                    break
                diff = target - h.sum()
    return h

def _interp(h, ns, nd):
    if ns == nd:
        return h.copy()
    xs = (np.arange(ns) + 0.5) * (2.0 / ns)
    xd = (np.arange(nd) + 0.5) * (2.0 / nd)
    return np.interp(xd, xs, h)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_at_lag(h, k, n):
    """Gradient of corr[k] w.r.t. h."""
    g = np.zeros(n)
    s = k - (n - 1)
    if s >= 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] += (-h[:n-s])
    else:
        p = -s
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] += (-h[p:])
    return g

def _sobj(x):
    n = _ng
    c = _corr(x)
    mx = float(np.max(c))
    b = _bv
    sh = c - mx
    return (mx + np.log(np.exp(b * sh).sum()) / b) * (2.0 / n)

def _sgrad(x):
    n = _ng
    c = _corr(x)
    mx = float(np.max(c))
    b = _bv
    sh = c - mx
    w = np.exp(b * sh)
    sw = w.sum()
    if sw < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        sw = 1.0
    w /= sw
    g = np.zeros(n)
    nz = np.where(w > 5e-5)[0]
    for k in nz:
        ki = int(k)
        wk = float(w[ki])
        g += wk * _grad_at_lag(x, ki, n)
    g *= (2.0 / n)
    return g

def _scon(x):
    return x.sum() - 0.5 * _ng

def _scon_grad(x):
    return np.ones(len(x))

def _lbfgs(h0, n, beta, maxiter=300, gtol=1e-14):
    global _ng, _bv
    _ng = n
    _bv = beta
    try:
        r = minimize(_sobj, h0, jac=_sgrad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon, 'jac': _scon_grad}],
                     options={'maxiter': maxiter, 'gtol': gtol, 'ftol': 1e-16,
                              'iprint': 0, 'maxls': 40, 'maxfun': maxiter * 40})
        return _proj(np.asarray(r.x, dtype=np.float64), n)
    except Exception:
        return _proj(h0.copy(), n)

def _pgstep(h, g, n, steps):
    """Try a sequence of step sizes in direction -g (or +g)."""
    best_c = _c5(h)
    for a in steps:
        h2 = _proj(h - a * g, n)
        c2 = _c5(h2)
        if c2 < best_c - 1e-14:
            return h2, c2
    # Also try +g direction
    for a in steps:
        h2 = _proj(h + a * g, n)
        c2 = _c5(h2)
        if c2 < best_c - 1e-14:
            return h2, c2
    return h, best_c

def _pgrad(h0, n, nit, beta_seq):
    """Projected gradient descent with cycling beta values."""
    best = h0.copy()
    bc = _c5(best)
    global _ng
    _ng = n
    for it in range(nit):
        if _tl() < 2:
            break
        beta = beta_seq[it % len(beta_seq)]
        _bv = beta
        g = _sgrad(best)
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        g = g / gn
        best, bc = _pgstep(best, g, n, [0.08, 0.04, 0.02, 0.01, 0.005, 0.015, 0.03, 0.06])
        if not isinstance(best, np.ndarray) or _c5(best) == bc:
            # Try exact max-lag gradient
            c = _corr(best)
            kmax = int(np.argmax(c))
            g2 = _grad_at_lag(best, kmax, n)
            gn2 = np.linalg.norm(g2)
            if gn2 > 1e-13:
                g2 /= gn2
                best, bc = _pgstep(best, g2, n, [0.05, 0.02, 0.01, 0.005])
    return best, bc

def _swapsearch(h, n, ntrials, rng):
    """Random swap local search."""
    best = h.copy()
    bc = _c5(best)
    for _ in range(ntrials):
        if _tl() < 2:
            break
        imp = False
        for _ in range(200):
            if _tl() < 1:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            h2 = best.copy()
            tmp = h2[i]
            h2[i] = h2[j]
            h2[j] = tmp
            h2 = _proj(h2, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-14:
                best = h2
                bc = c2
                imp = True
        if not imp:
            break
    return best, bc

def _slidsearch(h, n, ntrials, rng):
    """Slide individual values toward 0 or 1."""
    best = h.copy()
    bc = _c5(best)
    for _ in range(ntrials):
        if _tl() < 2:
            break
        imp = False
        fracs = np.where((best > 0.02) & (best < 0.98))[0]
        if len(fracs) < 2:
            break
        for _ in range(300):
            if _tl() < 1:
                break
            i = fracs[rng.randint(len(fracs))]
            j = fracs[rng.randint(len(fracs))]
            if i == j:
                continue
            for delta in [0.05, -0.05, 0.1, -0.1, 0.02, -0.02]:
                ni = best[i] + delta
                nj = best[j] - delta
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h2 = best.copy()
                h2[i] = ni
                h2[j] = nj
                h2 = _proj(h2, n)
                c2 = _c5(h2)
                if c2 < bc - 1e-14:
                    best = h2
                    bc = c2
                    imp = True
                    fracs = np.where((best > 0.02) & (best < 0.98))[0]
                    break
        if not imp:
            break
    return best, bc

def _optimize(h0, n, rng, time_limit=0):
    """Full optimization pipeline."""
    h = _proj(h0.copy(), n)
    c5 = _c5(h)
    t_start = time.time()
    
    # Phase 1: L-BFGS-B at multiple smoothing levels
    for beta in [20, 50, 100, 300, 800, 2000]:
        if _tl() < 10 or (time_limit > 0 and time.time() - t_start > time_limit):
            break
        h2 = _lbfgs(h, n, beta, 250)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2
    
    # Phase 2: Projected gradient
    if _tl() > 10:
        h, c5 = _pgrad(h, n, 200, [100, 300, 800, 2000, 5000])
    
    # Phase 3: Local search
    if _tl() > 10:
        h, c5 = _swapsearch(h, n, 50, rng)
    
    # Phase 4: Slide search
    if _tl() > 10:
        h, c5 = _slidsearch(h, n, 50, rng)
    
    # Phase 5: L-BFGS polish
    if _tl() > 10:
        h2 = _lbfgs(h, n, 5000, 300)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2
        h2 = _lbfgs(h, n, 15000, 200)
        c2 = _c5(h2)
        if c2 < c5:
            h, c5 = h2, c2
    
    # Phase 6: Final local search
    if _tl() > 10:
        h, c5 = _pgrad(h, n, 150, [2000, 5000, 10000])
        h, c5 = _swapsearch(h, n, 40, rng)
        h, c5 = _slidsearch(h, n, 40, rng)
    
    return h, c5
