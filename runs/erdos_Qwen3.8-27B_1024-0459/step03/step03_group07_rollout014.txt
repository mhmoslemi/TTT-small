```python
def weighted_subgrad(h, n, weights):
    """
    h: shape (n,)
    weights: shape (2n-1,), softmax of correlations
    Returns: gradient of Σ w_τ * c[τ] with respect to h, shape (n,)
    """
    grad = np.zeros(n)
    for fidx, w in enumerate(weights):
        if w < 1e-12:
            continue
        lag = fidx - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n-lag] += w * (1.0 - h[lag:])
            grad[lag:] += w * (-h[:n-lag])
        else:
            s = -lag
            grad[s:] += w * (1.0 - h[:n-s])
            grad[:n-s] += w * (-h[s:])
    return grad
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = np.argsort(1.0 - h)[:max(1, int(np.ceil(abs(diff))))]
        else:
            idx = np.argsort(h)[-max(1, int(np.ceil(abs(diff)))):]
        h[idx] = np.clip(h[idx] + diff / len(idx), 0, 1)
    return h

def fast_coord_descent(h, n, sweeps, time_limit, t0, grid_size=200):

    xs_col = xs[:, np.newaxis]  # (grid_size, 1)
    all_idx = np.arange(2 * n - 1)
    mask_lag0 = np.ones(2 * n - 1, dtype=bool)
    mask_lag0[n - 1] = False
    
    for sweep in range(sweeps):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        for j in range(n):
            old = h[j]
            # Compute coefficients
            a = np.zeros(2 * n - 1)
            # Positive lags τ = 1..n-1, fidx = n-1+τ = n..2n-2
            if n > 1:
                taus = np.arange(1, n)
                v1 = taus < (n - j)  # j + τ < n
                v2 = taus <= j        # j - τ >= 0
                a_pos = np.zeros(n)  # a[0..n-1] for lags 0..n-1
                a_pos[1:][v1] += (1.0 - h[j + taus[v1]])
                a_pos[1:][v2] += (-h[j - taus[v2]])
                a_pos[0] = 1.0
                a[n-1:] = a_pos
            # Lag 0
            a[n-1] = 1.0
            # Negative lags: fidx = n-1-s, s=1..n-1, so fidx = 0..n-2
            if n > 1:
                s_vals = np.arange(1, n)
                v3 = s_vals <= j
                v4 = s_vals <= (n - 1 - j)
                a_neg = np.zeros(n - 1)
                a_neg[v3] += (1.0 - h[j - s_vals[v3]])
                a_neg[v4] += (-h[j + s_vals[v4]])
                a[:n-1] = a_neg
            
            # Rest
            rest = corr.copy() - a * old
            rest[n-1] = corr[n-1] - old + old * old
            
            # Evaluate on grid

                v2 = rest + a * m2

                else:

def softmax_subgrad(h, n, beta):
    """Compute softmax-weighted subgradient of max c[τ]."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    c_max = np.max(corr)
    w = np.exp(beta * (corr - c_max))
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.full(2*n-1, 1.0/(2*n-1))
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for fidx in range(2 * n - 1):
        wi = w[fidx]
        if wi < 1e-12:
            continue
        lag = fidx - (n - 1)
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            m = n - lag
            grad[:m] += wi * (1.0 - h[lag:lag+m])
            grad[lag:] += wi * (-h[:m])
        else:
            s = -lag
            m = n - s
            grad[s:] += wi * (1.0 - h[:m])
            grad[:m] += wi * (-h[s:])
    return grad, c_max
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = float(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-15
            cnt = float(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(1.0 - h)[:max(1, int(np.ceil(abs(diff) * 2)))]
        else:
            idx = np.argsort(h)[-max(1, int(np.ceil(abs(diff) * 2))):]
        h[idx] = np.clip(h[idx] + diff / len(idx), 0.0, 1.0)
    return h

def softmax_subgrad(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    w = np.exp(beta * (corr - c_max))
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for fidx in range(2 * n - 1):
        wi = w[fidx]
        if wi < 1e-12:
            continue
        lag = fidx - (n - 1)
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            m = n - lag
            grad[:m] += wi * (1.0 - h[lag:lag + m])
            grad[lag:] += wi * (-h[:m])
        else:
            s = -lag
            m = n - s
            grad[s:] += wi * (1.0 - h[:m])
            grad[:m] += wi * (-h[s:])
    return grad, float(c_max)
```
