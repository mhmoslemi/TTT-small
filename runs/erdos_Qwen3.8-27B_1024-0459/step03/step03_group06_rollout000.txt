def _sm_obj(h, beta, topk):
    n = len(h)
    corr = _corr_full(h)
    if topk is None or topk >= len(corr):
        vals = corr
        mx = np.max(vals)
        v = vals - mx
        w = np.exp(beta * v)
    else:
        idx = np.argpartition(corr, -topk)[-topk:]
        vals = corr[idx]
        mx = np.max(vals)
        v = vals - mx
        w = np.exp(beta * v)
    z = w.sum()
    return (mx + np.log(z) / beta) * (2.0 / n)

import numpy as np
import time
import math
from scipy.optimize import minimize


def _rem_at(t0, bt):
    return max(0.0, float(bt) - (time.time() - t0) - 6.0)


def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5_of(h):
    return float(np.max(_corr_full(h))) * (2.0 / len(h))


def _project(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(60):
        r = tgt - float(np.sum(h))
        if abs(r) < 1e-15:
            break
        caps = (1.0 - h) if r > 0.0 else h
        tot = float(caps.sum())
        if tot < 1e-15:
            break
        h = h + (r / tot) * caps
        np.clip(h, 0.0, 1.0, out=h)
    r = tgt - float(np.sum(h))
    if abs(r) > 1e-15:
        j = int(np.argmin(np.abs(h - 0.5)))
        h[j] += r
        if h[j] < 0.0 or h[j] > 1.0:
            h2 = h.copy()
            h2[:] = 0.5
            h = h2
            r = tgt - float(np.sum(h))
            if abs(r) > 1e-15:
                k = int(np.argmin(np.abs(h - 0.5)))
                h[k] += r
    return h


def _resample(src, n):
    m = len(src)
    if m == n:
        return src.copy()
    xs = (np.arange(m) + 0.5) * (2.0 / m)
    xn = (np.arange(n) + 0.5) * (2.0 / n)
    out = np.interp(xn, xs, src)
    return _project(out, n)


def _buildE(h):
    n = len(h)
    ph = 1.0 - h
    E = np.zeros((2 * n - 1, n), dtype=float)
    for jm in range(2 * n - 1):
        s = jm - (n - 1)
        row = E[jm]
        if s >= 0:
            row[:n - s] += ph[s:]
            row[s:] -= h[:n - s]
        else:
            t = -s
            row[t:] += ph[:n - t]
            row[:n - t] -= h[t:]
    return E


def _sm_obj(x, beta, topk):
    corr = _corr_full(x)
    k = len(corr)
    if (topk is None) or (topk >= k):
        idx = range(k)
    else:
        idx = np.argpartition(corr, -int(topk))[-int(topk):]
    vals = corr[idx]
    mx = float(np.max(vals))
    wsum = 0.0
    for v in vals:
        wsum += math.exp(float(beta) * (float(v) - mx))
    return (mx + math.log(wsum) / float(beta)) * (2.0 / len(x))


def _sm_grad(x, beta, topk):
    corr = _corr_full(x)
    k = len(corr)
    if (topk is None) or (topk >= k):
        idx = np.arange(k)
    else:
        idx = np.argpartition(corr, -int(topk))[-int(topk):]
    vals = corr[idx]
    mx = float(np.max(vals))
    wsum = 0.0
    for v in vals:
        wsum += math.exp(float(beta) * (float(v) - mx))
    wsum = 1.0 / wsum
    E = _buildE(x)
    g = np.zeros(len(x), dtype=float)
    for j in idx:
        c = math.exp(float(beta) * (float(corr[j]) - mx)) * wsum
        if c > 1e-16:
            g += c * E[j]
    return g * (2.0 / len(x))


def _iters(n):
    n = int(n)
    return int(min(1100, max(70, int(300000.0 / max(1, n)))))


def _lbfgs_solve(h, n, beta, topk, maxiter):
    bounds = [(0.0, 1.0)] * int(n)
    res = minimize(_sm_obj, h, jac=_sm_grad, args=(float(beta), topk),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': int(maxiter), 'ftol': 1e-16,
                            'gtol': 1e-12, 'maxls': 40, 'maxfun': int(4 * max(1, maxiter)),
                            'iprint': 0})
    out = _project(np.asarray(res.x, dtype=float), int(n))
    return out


def _grad_check(h):
    beta, topk = 250.0, 14
    g = _sm_grad(h, beta, topk)
    n = len(h)
    ps = [n // 4, n // 2, 3 * n // 4]
    eps = 1e-7
    for p in ps:
        if p < 1 or p >= n - 1:
            continue
        if h[p] < 1e-4 or h[p] > 1.0 - 1e-4:
            continue
        h2 = h.copy()
        h2[p] = h[p] + eps
        h3 = h.copy()
        h3[p] = h[p] - eps
        fp = _sm_obj(h2, beta, topk)
        fm = _sm_obj(h3, beta, topk)
        num = (fp - fm) / (2.0 * eps)
        if abs(num - g[p]) > 0.25 * (abs(num) + abs(g[p])) + 1e-5:
            return False
    return True


def _tem_tm(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (bin(i).count('1') & 1) else 0.0
    return _project(h, n)


def _tem_golden(n):
    h = np.zeros(n)
    gr = 0.5 * (1.0 + math.sqrt(5.0))
    for i in range(n):
        f = fmod(i * gr, 1.0) if False else (i * gr) % 1.0
        h[i] = 1.0 if abs(f - 0.5) < 0.25 else 0.0
    return _project(h, n)


def _tem_vdc(n):
    h = np.zeros(n)
    for i in range(n):
        v = i + 1
        r = 0
        for _ in range(24):
            r = (r << 1) | (v & 1)
            v >>= 1
            if v == 0:
                break
        h[i] = 1.0 if (r & 1) == 0 else 0.0
    return _project(h, n)


def _tem_runs(n, rng):
    h = np.zeros(n)
    i = 0
    bit = 1
    while i < n:
        L = int(rng.randint(2, 5))
        L = min(L, n - i)
        h[i:i + L] = bit
        bit = 1 - bit
        i += L
    ones = int(float(h.sum()))
    tgt = n // 2
    if ones > tgt:
        idx = rng.choice(np.where(h == 1.0)[0], ones - tgt, replace=False)
        h[idx] = 0.0
    elif ones < tgt:
        idx = rng.choice(np.where(h == 0.0)[0], tgt - ones, replace=False)
        h[idx] = 1.0
    return _project(h, n)


def _tem_iso(n, rng):
    m = n // 2
    if m < 2:
        m = 2
    g = (rng.rand(m) < 0.5).astype(float)
    ones = int(float(g.sum()))
    tgt = m // 2
    if ones > tgt:
        idx = rng.choice(np.where(g == 1.0)[0], ones - tgt, replace=False)
        g[idx] = 0.0
    elif ones < tgt:
        idx = rng.choice(np.where(g == 0.0)[0], tgt - ones, replace=False)
        g[idx] = 1.0
    h = np.concatenate([g, g])
    h = h[:n]
    return _project(h, n)


def _tem_comp(n, rng):
    m = n // 4
    if m < 1:
        return _tem_runs(n, rng)
    g = (rng.rand(m) < 0.5).astype(float)
    h = np.concatenate([g, 1.0 - g, g, 1.0 - g])
    return _project(h[:n], n)


def _tem_anti(pool, n, rng):
    bn, bc, bh = _pool_best(pool)
    m = n // 2
    g = _resample(bh, max(2, m))
    g = g[:m]
    h = np.concatenate([g, 1.0 - g])
    return _project(h[:n], n)


def _tem_sine(n, rng):
    f = int(rng.randint(2, max(3, n // 6)))
    phi = rng.uniform(0.0, 2.0 * math.pi)
    A = rng.uniform(0.30, 0.48)
    xs = (np.arange(n) + 0.5) * 2.0 / n
    h = 0.5 + A * np.sin(2.0 * math.pi * f * xs + phi)
    return _project(h, n)


def _tem_create(n, rng, pool):
    out = []
    out.append(_tem_tm(n))
    out.append(_tem_golden(n))
    out.append(_tem_vdc(n))
    out.append(_tem_runs(n, rng))
    if n >= 16:
        out.append(_tem_iso(n, rng))
        out.append(_tem_comp(n, rng))
    if len(pool) > 0:
        out.append(_tem_anti(pool, n, rng))
    out.append(_tem_sine(n, rng))
    out.append(_tem_sine(n, rng))
    return out


def _pool_best(pool):
    bn, bc, bh = None, float('inf'), None
    for (nn, item) in pool.items():
        c5 = float(item[0])
        if c5 < bc:
            bn, bc, bh = nn, c5, item[1]
    return bn, bc, bh


def _pool_all(pool):
    out = []
    for (nn, item) in pool.items():
        out.append((float(item[0]), nn, item[1]))
    out.sort()
    return out


def _feed(pool, h, tag):
    n = len(h)
    h = _project(h, n)
    c5 = _c5_of(h)
    old = pool.get(n)
    if old is None or c5 < float(old[0]) - 1e-12:
        pool[n] = (float(c5), h.copy())
        print("  pool n=%d c5=%.7f [%s]" % (n, c5, tag))
    return float(c5)


def _roll_best(h):
    n = len(h)
    besth = h.copy()
    bestc = _c5_of(h)
    step = 1 if n <= 720 else 2
    for q in range(step, n, step):
        hr = np.roll(h, q)
        c5 = _c5_of(hr)
        if c5 < bestc - 1e-12:
            bestc = c5
            besth = hr.copy()
    return besth, bestc


def _lbf_sweep(pool, n, inits, settings, t_lim, tag):
    t0 = time.time()
    cand = []
    seq = 0
    for h0 in inits:
        c5 = _c5_of(h0)
        cand.append((c5, seq, h0))
        seq += 1
    cand.sort()
    for (beta, topk, iters) in settings:
        for i in range(min(6, len(cand))):
            if time.time() - t0 > t_lim:
                break
            c5b, _, hbase = cand[i]
            try:
                hnew = _lbfgs_solve(hbase, n, beta, topk, iters)
            except Exception:
                continue
            c5n = _c5_of(hnew)
            if c5n < c5b - 1e-10:
                cand.append((c5n, seq, hnew))
                cand.sort()
                seq += 1
                _feed(pool, hnew, tag + " beta%d" % int(beta))
            if time.time() - t0 > t_lim:
                break


def _polish2ex(h, n, t_lim, rng):
    h = _project(np.asarray(h, dtype=float).copy(), n)
    V = _corr_full(h)
    U = float(np.max(V))
    dx = 2.0 / n
    t0 = time.time()
    lift = 3.4e-7 / dx
    K = 10000 if n <= 240 else 4500
    idle = 0
    while (time.time() - t0) < t_lim and idle < 3:
        E = _buildE(h)
        did = 0
        T = U - lift
        for _ in range(K):
            a = int(rng.randint(0, n))
            b = int(rng.randint(0, n))
            if a == b:
                continue
            ha = h[a]
            hb = h[b]
            emax = min(ha, 1.0 - hb)
            if emax <= 1e-11:
                continue
            dvec = E[b] - E[a]
            mpos = dvec > 1e-10
            ub = np.inf
            if np.any(mpos):
                ub = float(np.min((T - V[mpos]) / dvec[mpos]))
            if ub <= 0.0:
                continue
            mneg = dvec < -1e-10
            L = 0.0
            if np.any(mneg):
                L = float(np.max((V[mneg] - T) / (-dvec[mneg])))
            q = (n - 1) + (b - a)
            if 0 <= q < (2 * n - 1):
                dq = float(dvec[q])
                disc = dq * dq + 4.0 * (T - V[q])
                if disc < 0.0:
                    continue
                rs = math.sqrt(disc)
                r1 = (-dq - rs) * 0.5
                r2 = (-dq + rs) * 0.5
                if r1 > L:
                    L = r1
                if r2 < ub:
                    ub = r2
            if L < 0.0:
                L = 0.0
            if ub <= L:
                continue
            emax = min(ha, 1.0 - h[b])
            if emax <= L:
                continue
            hi = ub if ub < emax else emax
            e = 0.5 * (L + hi)
            if e <= 1e-13:
                continue
            h[a] -= e
            h[b] += e
            V = _corr_full(h)
            U2 = float(np.max(V))
            if U2 < U - 1e-12:
                U = U2
                T = U - lift
                did += 1
            else:
                h[a] += e
                h[b] -= e
        if did == 0:
            idle += 1
    return h, U * (2.0 / n)


def _polish_chain(h, n, t_lim, rng):
    t0 = time.time()
    h = _project(np.array(h, dtype=float), n)
    third = t_lim / 3.0
    h, _ = _polish2ex(h, n, min(third, t_lim - 1.0), rng)
    if t_lim - (time.time() - t0) > 4.0:
        try:
            h = _lbfgs_solve(h, n, 1000.0, 6, 140)
        except Exception:
            pass
    left = t_lim - (time.time() - t0) - 1.0
    if left > 2.0:
        h, _ = _polish2ex(h, n, left, rng)
    return h

    if _rem_at(t0, BT) > 8.0 and bn <= 720:
