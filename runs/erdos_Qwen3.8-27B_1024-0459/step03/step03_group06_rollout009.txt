Lagrange multipliers: w_i ∝ exp(C_{s_i} / β) for large β, which is exactly the softmax!

def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):

def smooth_max_obj(x):
    n = _N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _BETA
    shifted = beta * (corr - m)

    val = (m + np.log(np.sum(np.exp(shifted))) / beta) * dx
    penalty = 10000.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty


def smooth_max_grad(x):
    n = _N
    dx = 2.0 / n
    beta = _BETA
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    sw = np.sum(weights)
    if sw < 1e-300:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= sw

    grad = np.zeros(n)

    for s in range(0, n):
        w = weights[n - 1 + s] * dx
        if w < 1e-18:
            continue
        # C_s = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
        # d/dh[j]: (1-h[j+s]) for j in [0, n-1-s], -h[j-s] for j in [s, n-1]
        if n - s > 0:
            grad[0:n - s] += w * (1.0 - x[s:n])
            grad[s:n] += w * (-x[0:n - s])

    for p in range(1, n):
        w = weights[n - 1 - p] * dx
        if w < 1e-18:
            continue
        # C_{-p} = sum_{m=p}^{n-1} h[m]*(1-h[m-p])
        # d/dh[j]: (1-h[j-p]) for j in [p, n-1], -h[j+p] for j in [0, n-1-p]
        if n - p > 0:
            grad[p:n] += w * (1.0 - x[0:n - p])
            grad[0:n - p] += w * (-x[p:n])

    pen_coeff = 10000.0
    grad += 2.0 * pen_coeff * (np.sum(x) - _TGT)
    return grad


def topk_smooth_obj(x, K=5):

    corr = np.correlate(x, 1.0 - x, mode='full')
    top_idx = np.argsort(corr)[-K:]

    beta = 500.0
    shifted = beta * (tv - m)
    shifted = np.clip(shifted, -300, 300)
    val = (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

    return val + penalty


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        for _ in range(n):
            idx = np.argmin(np.abs(h - 0.5))
            adj = diff
            lo, hi = 0.0, 1.0
            if h[idx] + adj < lo or h[idx] + adj > hi:
                continue
            h[idx] += adj
            break
    h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

    return h


def active_set_grad(h, top_k=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]

    grad = np.zeros(n)
    for j in top_idx:
        s = j - (n - 1)
        if s >= 0:
            if n - s > 0:
                grad[0:n - s] += (1.0 - h[s:n])
                grad[s:n] += (-h[0:n - s])
        else:
            p = -s
            if n - p > 0:
                grad[p:n] += (1.0 - h[0:n - p])
                grad[0:n - p] += (-h[p:n])

    grad *= dx
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return None
    return grad / gnorm


def binary_sa(h_init, n, max_iters, rng, t_start=0.003, t_end=0.00003):
    h = np.where(h_init > 0.5, 1.0, 0.0)
    target_ones = n // 2
    num_ones = int(np.sum(h))
    if num_ones > target_ones:
        ones_idx = np.where(h == 1.0)[0]
        remove_idx = rng.choice(len(ones_idx), num_ones - target_ones, replace=False)
        h[ones_idx[remove_idx]] = 0.0
    elif num_ones < target_ones:
        zeros_idx = np.where(h == 0.0)[0]
        add_idx = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
        h[zeros_idx[add_idx]] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5

    ones_list = np.where(h == 1.0)[0].tolist()
    zeros_list = np.where(h == 0.0)[0].tolist()

    if len(ones_list) == 0 or len(zeros_list) == 0:
        return best_h, best_c5

    ratio = t_end / t_start
    for it in range(max_iters):
        if remaining_time() < 2:
            break
        i = rng.randint(len(ones_list))
        j = rng.randint(len(zeros_list))
        oi = ones_list[i]
        oz = zeros_list[j]

        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = compute_c5(h)

        T = t_start * (ratio ** (float(it) / max(float(max_iters), 1.0)))
        accept = False
        if new_c5 < current_c5:
            accept = True
        elif T > 1e-18 and rng.rand() < np.exp(-(new_c5 - current_c5) / T):
            accept = True

        if accept:
            current_c5 = new_c5
            ones_list[i] = oz
            zeros_list[j] = oi
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0

    return best_h, best_c5
