import time
import scipy.optimize as opt
from scipy import signal


def compute_correlation(h):
    """Compute full correlation of h with 1-h using FFT for speed."""
    n = len(h)
    N = 2 * (2 * n)  # sufficiently large for FFT
    fh = np.fft.rfft(h, N)
    fg = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(fh * np.conj(fg), N)
    # Extract the linear correlation from the circular one
    # np.correlate(a,b,'full')[k] = sum_i a[i]*b[i+k-(n-1)]
    # The FFT gives circular correlation, which matches linear for zero-padded signals
    # corr_full[k] for k=0..2n-2
    result = np.zeros(2 * n - 1)
    result[:n] = corr[n - 1:]
    result[n:] = corr[:n - 1][::-1]
    return result


def compute_c5(h):
    """Compute C5 value."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlation(h)
    return np.max(corr) * dx


def compute_c5_direct(h):

def project_feasible(h, n):

            break

            if cnt == 0:

                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
            except Exception:
                pass
    return best_h, best_c5


def coordinate_descent(h, n, sweeps=50, time_limit=None, t0=None):
    """Coordinate descent minimizing max correlation."""
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    for sw in range(sweeps):
        if time_limit and t0 and time.time() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time_limit and t0 and (time.time() - t0) % 5 > 4.8:
                if time.time() - t0 > time_limit:
                    break
            old = h[j]
            # Compute current correlation
            corr = compute_correlation(h)
            current_max = np.max(corr) * dx
            
            # Try binary search for optimal h[j]
            lo, hi = 0.0, 1.0
            for _ in range(20):
                mid = (lo + hi) / 2.0
                h[j] = mid
                corr_new = compute_correlation(h)
                new_max = np.max(corr_new) * dx
                if new_max < compute_c5_direct(best_h) or new_max < current_max:
                    lo = mid
                else:
                    hi = mid
            h[j] = (lo + hi) / 2.0
            h = project_feasible(h, n)
            new_c5 = compute_c5(h)
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c5


def fast_coordinate_descent(h, n, sweeps=30, time_limit=None, t0=None):
    """Faster coordinate descent using precomputed correlation updates."""
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    for sw in range(sweeps):
        if time_limit and t0 and time.time() - t0 > time_limit:
            break
        corr = compute_correlation(h)
        max_unaffected = np.max(corr)
        for j in range(n):
            if time_limit and t0 and time.time() - t0 > time_limit:
                break
            old = h[j]
            # For each lag k, determine how c_k changes with h[j]
            # c_k = sum_i h[i](1-h[i+d_k]) where d_k = k-(n-1)
            # h[j] affects c_k if j is in the valid range for lag d_k
            # New c_k = c_k - old*(1-h[j+d_k]) - h[j-d_k]*(1-old) + new*(1-h[j+d_k]) + h[j-d_k]*(1-new)
            #          = c_k + (new-old)*[(1-h[j+d_k]) - h[j-d_k]] (when both terms present)
            # We need to track which lags are affected and compute the new max
            
            # Efficient approach: for each lag, compute the linear coefficient
            # c_k as function of x = h[j]: c_k(x) = A_k * x + B_k (for non-diagonal)
            # or c_k(x) = x - x^2 + B_k (for k = n-1, i.e., lag 0)
            
            a_coeffs = np.zeros(2 * n - 1)
            b_coeffs = np.zeros(2 * n - 1)
            quadratic_mask = np.zeros(2 * n - 1, dtype=bool)
            
            for k in range(2 * n - 1):
                d = k - (n - 1)
                if d == 0:
                    # c_k includes h[j]*(1-h[j]) = x - x^2
                    if 0 <= j < n:
                        quadratic_mask[k] = True
                        # c_0 = sum_i h[i](1-h[i]) = (sum_{i≠j} h[i](1-h[i])) + x - x^2
                        # The "rest" is c_0 - old*(1-old)
                        b_coeffs[k] = corr[k] - old * (1.0 - old)
                        a_coeffs[k] = 1.0  # coefficient of x (the -x^2 is handled separately)
                else:
                    # c_k(x) = c_k - old*(1-h[j+d]) - h[j-d]*(1-old) + x*(1-h[j+d]) + h[j-d]*(1-x)
                    # = c_k + x*[(1-h[j+d]) - h[j-d]] + old*[h[j-d] - (1-h[j+d])]
                    # = [c_k + old*(h[j-d] - (1-h[j+d]))] + x*[(1-h[j+d]) - h[j-d]]
                    affects = False
                    coeff = 0.0
                    if d > 0:
                        if j + d < n:
                            coeff += (1.0 - h[j + d])
                            affects = True
                        if j - d >= 0:
                            coeff += (-h[j - d])
                            affects = True
                    else:
                        s = -d
                        if j - s >= 0:
                            coeff += (1.0 - h[j - s])
                            affects = True
                        if j + s < n:
                            coeff += (-h[j + s])
                            affects = True
                    if affects:
                        a_coeffs[k] = coeff
                        b_coeffs[k] = corr[k] - coeff * old
                    else:
                        a_coeffs[k] = 0.0
                        b_coeffs[k] = corr[k]
            
            # Now find x that minimizes max over all k of:
            # - For quadratic: b_coeffs[k] + x - x^2
            # - For linear: b_coeffs[k] + a_coeffs[k] * x
            # - For unaffected: b_coeffs[k] (constant)
            
            # Use golden section search on [0, 1]
            phi = (np.sqrt(5.0) - 1.0) / 2.0
            a, b = 0.0, 1.0
            c = b - phi * (b - a)
            d_pt = a + phi * (b - a)
            
            def eval_at(x):
                vals_lin = b_coeffs + a_coeffs * x
                val = np.max(vals_lin)
                if np.any(quadratic_mask):
                    qidx = np.where(quadratic_mask)[0]
                    qvals = b_coeffs[qidx] + x - x * x
                    val = max(val, np.max(qvals))
                return val
            
            fc = eval_at(c)
            fd = eval_at(d_pt)
            for _ in range(30):
                if fc < fd:
                    b = d_pt
                    d_pt = c
                    fd = fc
                    c = b - phi * (b - a)
                    fc = eval_at(c)
                else:
                    a = c
                    c = d_pt
                    fc = fd
                    d_pt = a + phi * (b - a)
                    fd = eval_at(d_pt)
            
            best_x = (a + b) / 2.0
            h[j] = best_x
            h = project_feasible(h, n)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5


def interpolate_h(h_src, n_src, n_dst):
    """Interpolate h from one resolution to another."""
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_new = np.interp(xn, xp, h_src)
    return project_feasible(h_new, n_dst)


def generate_fourier_init(n, rng, n_modes=5):
    """Generate a Fourier-based initialization."""
    t = np.arange(n) / n
    h = np.zeros(n)
    for m in range(1, n_modes + 1):
        amp = rng.uniform(0.05, 0.25) / np.sqrt(m)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * m * t + phase)
    h = 0.5 + h
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)


def generate_block_init(n, rng, n_blocks=8):
    """Generate a block pattern initialization."""
    block_size = n // n_blocks
    h = np.zeros(n)
    for i in range(n_blocks):
        val = rng.uniform(0.2, 0.8)
        start = i * block_size
        end = start + block_size if i < n_blocks - 1 else n
        h[start:end] = val
    return project_feasible(h, n)


def subgradient_method(h, n, iters=200, time_limit=None, t0=None):
    """Subgradient method with projection for min-max problem."""
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    lr = 0.3
    
    for it in range(iters):
        if time_limit and t0 and time.time() - t0 > time_limit:
            break
        corr = compute_correlation(h)
        max_val = np.max(corr)
        
        # Get subgradient: pick argmax lag(s)
        top_k = min(5, 2 * n - 1)
        top_idx = np.argsort(corr)[-top_k:]
        
        # Compute subgradient at these lags
        grad = np.zeros(n) / top_k
        for k in top_idx:
            d = k - (n - 1)
            if d >= 0:
                grad[:n - d] += (1.0 - h[d:])
                grad[d:] += (-h[:n - d])
            else:
                s = -d
                grad[s:] += (1.0 - h[:n - s])
                grad[:n - s] += (-h[s:])
        
        h_new = h - lr * grad * dx
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.02, 0.5)
        else:
            lr *= 0.9
            if lr < 0.01:
                lr = 0.01
    
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return max(0.0, budget_s - (time.time() - t0) - 5.0)
    
    # Get parent solution
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn > 10:
            best_h = ph.copy()
            best_c5 = compute_c5_direct(ph)
            best_n = pn
    except NameError:
        pass
    
    if best_h is None:
        best_n = 160
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)
    
    # Phase 1: Multi-resolution exploration with log-sum-exp
    # Try several resolutions
    resolutions = [160, 192, 224, 256, 288]
    
    # Filter to reasonable range
    resolutions = [r for r in resolutions if 80 <= r <= 350]
    
    # Include parent resolution
    if best_n not in resolutions:
        resolutions.append(best_n)
    resolutions = sorted(set(resolutions))
    
    solutions = []  # list of (h, c5, n)
    
    for n in resolutions:
        if time_left() < 50:
            break
        
        # Create starting points
        inits = []
        if n == best_n:
            inits.append(best_h.copy())
        else:
            inits.append(interpolate_h(best_h, best_n, n))
        
        # Fourier init
        inits.append(generate_fourier_init(n, rng, n_modes=4))
        # Another Fourier with different params
        inits.append(generate_fourier_init(n, rng, n_modes=7))
        # Block init
        inits.append(generate_block_init(n, rng, n_blocks=6))
        
        # Allocate time for this resolution
        n_inits = len(inits)
        time_per_n = max(30.0, time_left() / max(1, len(resolutions) - resolutions.index(n)))
        time_per_init = max(10.0, time_per_n / n_inits)
        
        for h0 in inits:
            if time_left() < 15:
                break
            
            tl = min(time_per_init * 1.5, time_left() * 0.4)
            
            # Step 1: LSE + SLSQP with moderate beta
            h_opt, c5_opt = optimize_lse_slsqp(
                h0, n,
                betas=[50.0, 200.0, 500.0],
                maxiter_per_beta=50,
                time_limit=tl * 0.7,
                t0=t0
            )
            
            # Step 2: Subgradient polish
            if time_left() > 20:
                h_sub, c5_sub = subgradient_method(
                    h_opt, n, iters=100,
                    time_limit=min(30, time_left() * 0.3),
                    t0=t0
                )
                if c5_sub < c5_opt:
                    h_opt, c5_opt = h_sub, c5_sub
            
            # Step 3: High-beta LSE
            if time_left() > 20:
                h_hi, c5_hi = optimize_lse_slsqp(
                    h_opt, n,
                    betas=[2000.0, 5000.0],
                    maxiter_per_beta=80,
                    time_limit=min(60, time_left() * 0.3),
                    t0=t0
                )
                if c5_hi < c5_opt:
                    h_opt, c5_opt = h_hi, c5_hi
            
            solutions.append((h_opt, c5_opt, n))
    
    # Update best from Phase 1
    for h_s, c5_s, n_s in solutions:
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
            best_n = n_s
    
    # Phase 2: Focused optimization on best 2-3 solutions
    if time_left() > 100:
        # Sort solutions by quality
        solutions.sort(key=lambda x: x[1])
        top_solutions = []
        seen_ns = set()
        for h_s, c5_s, n_s in solutions:
            if n_s not in seen_ns and c5_s < best_c5 + 0.02:
                top_solutions.append((h_s, c5_s, n_s))
                seen_ns.add(n_s)
            if len(top_solutions) >= 3:
                break
        
        for h_s, c5_s, n_s in top_solutions:
            if time_left() < 60:
                break
            
            tl = min(200.0, time_left() * 0.3)
            
            # Higher beta LSE
            h_opt, c5_opt = optimize_lse_slsqp(
                h_s, n_s,
                betas=[1000.0, 3000.0, 10000.0, 30000.0],
                maxiter_per_beta=100,
                time_limit=tl * 0.6,
                t0=t0
            )
            
            # Subgradient
            if time_left() > 30:
                h_sub, c5_sub = subgradient_method(
                    h_opt, n_s, iters=200,
                    time_limit=min(40, time_left() * 0.2),
                    t0=t0
                )
                if c5_sub < c5_opt:
                    h_opt, c5_opt = h_sub, c5_sub
            
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_s
    
    # Phase 3: Coordinate descent polish
    if time_left() > 30:
        h_cd, c5_cd = fast_coordinate_descent(
            best_h, best_n,
            sweeps=30,
            time_limit=min(120, time_left() * 0.5),
            t0=t0
        )
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Phase 4: Try resolution change (interpolate to nearby resolution and re-optimize)
    if time_left() > 30:
        for dn in [-16, 16, -32, 32]:
            if time_left() < 20:
                break
            n_try = best_n + dn
            if n_try < 80 or n_try > 400:
                continue
            h_try = interpolate_h(best_h, best_n, n_try)
            h_opt, c5_opt = optimize_lse_slsqp(
                h_try, n_try,
                betas=[5000.0],
                maxiter_per_beta=80,
                time_limit=min(30, time_left() * 0.3),
                t0=t0
            )
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_try
    
    # Phase 5: Final coordinate descent
    if time_left() > 15:
        h_cd, c5_cd = fast_coordinate_descent(
            best_h, best_n,
            sweeps=50,
            time_limit=min(60, time_left() * 0.7),
            t0=t0
        )
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Final: ensure feasibility and compute exact C5
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5_direct(best_h)
    
    # Try evaluator
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
