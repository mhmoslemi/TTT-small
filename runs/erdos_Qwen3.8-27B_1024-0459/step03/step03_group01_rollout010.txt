import time

_BUDGET = 1300.0

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):

def binary_sa(h, n, time_limit, rng, t0=1.0, t1=0.01, max_iters=2000000):
    """Simulated annealing on binary sequences with pair swaps."""
    h = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        ones_idx = np.where(h == 1)[0]
        remove = rng.choice(ones_idx, n_ones - n//2, replace=False)
        h[remove] = 0.0
    elif n_ones < n // 2:
        zeros_idx = np.where(h == 0)[0]
        add = rng.choice(zeros_idx, n//2 - n_ones, replace=False)
        h[add] = 1.0
    
    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx
    best_h = h.copy()
    start = time.time()
    n_corr = len(corr)
    
    for it in range(max_iters):
        if time.time() - start > time_limit or remaining() < 5:
            break
        # Temperature schedule
        frac = it / max_iters
        T = t0 * (t1 / t0) ** frac
        
        # Pick a swap: one 1 -> 0 and one 0 -> 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        
        h[i] = 0.0
        h[j] = 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        delta = new_max - cur_max
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_max = new_max
            corr = new_corr
            if cur_max < best_c5 / dx:
                best_c5 = cur_max * dx
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5

def make_quadratic_residue(n, p):
    """Create sequence based on quadratic residues mod p, padded/truncated to n."""
    h = np.zeros(n)
    residues = set()
    for x in range(1, p):
        residues.add((x * x) % p)
    for i in range(n):
        if i % p in residues:
            h[i] = 1.0
    return h
