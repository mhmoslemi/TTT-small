    z = np.exp(beta * shifted)
    z_sum = np.sum(z)
    return (m + np.log(z_sum) / beta) * dx, z / z_sum, idx, m


def objective_lbfgs(params, n, beta, top_k, penalty_w):
    h = np.clip(params, 0.0, 1.0)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    val, _, _, _ = smooth_max_topk(corr, beta, top_k)
    sum_err = np.sum(h) - target
    return val + penalty_w * sum_err * sum_err


def grad_lbfgs(params, n, beta, top_k, penalty_w):
    h = np.clip(params, 0.0, 1.0)
    target = n / 2.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = get_topk_shifts(corr, top_k)
    vals = corr[idx]
    m = vals.max()
    shifted = vals - m
    z = np.exp(beta * shifted)
    z /= np.sum(z)
    grad = np.zeros(n)
    for t in range(len(idx)):
        if z[t] < 1e-14:
            continue
        k_idx = idx[t]
        k = k_idx - (n - 1)
        if k >= 0:
            i_lo = k
            i_hi = n - 1
            m_vals = np.arange(i_lo, i_hi + 1)
            j_vals = m_vals - k
            grad[m_vals] += z[t] * (1.0 - h[j_vals])
            grad[j_vals] += z[t] * (-h[m_vals])
        else:
            i_lo = 0
            i_hi = n - 1 + k
            m_vals = np.arange(i_lo, i_hi + 1)
            j_vals = m_vals - k
            grad[m_vals] += z[t] * (1.0 - h[j_vals])
            grad[j_vals] += z[t] * (-h[m_vals])
    grad *= dx
    sum_err = np.sum(h) - target
    grad += 2.0 * penalty_w * sum_err
    return grad


def exact_coord_descent(h, n, best_c5, max_sweeps=200, rng=None, time_limit=100.0, t_start=None):
    if t_start is None:
        t_start = time.time()
    if rng is None:
        rng = np.random.RandomState(0)
    target = n / 2.0
    improved_total = False
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        improved_this_sweep = False
        order = rng.permutation(n)
        for i in order:
            if time.time() - t_start > time_limit:
                break
            old_val = h[i]
            candidates = [0.0, 1.0]
            if old_val > 0.0:
                candidates.append(old_val - 0.01)
            if old_val < 1.0:
                candidates.append(old_val + 0.01)
            if old_val > 0.005:
                candidates.append(0.005)
            if old_val < 0.995:
                candidates.append(0.995)
            best_trial_val = None
            best_trial_c5 = best_c5
            for cv in candidates:
                if cv < -1e-9 or cv > 1.0 + 1e-9:
                    continue
                h[i] = cv
                # find j to compensate
                diff = old_val - cv
                if abs(diff) < 1e-12:
                    c5_t = compute_c5_exact(h)
                    if c5_t < best_trial_c5:
                        best_trial_c5 = c5_t
                        best_trial_val = cv
                else:
                    # find best j to adjust
                    # try a few candidates for j
                    j_tries = list(range(max(0, i-3), min(n, i+4)))
                    if i not in j_tries:
                        j_tries.append(i)
                    for j in j_tries:
                        if j == i:
                            continue
                        h[j] = h[j] - diff
                        if h[j] < -1e-9 or h[j] > 1.0 + 1e-9:
                            h[j] = old_val - h[i] + h[i]  # revert
                            h[j] = 0.0
                            h[i] = old_val
                            # try another approach: distribute
                            break
                        c5_t = compute_c5_exact(h)
                        if c5_t < best_trial_c5:
                            best_trial_c5 = c5_t
                            best_trial_val = (cv, j, h[j])
                        h[j] = h[j] + diff  # revert j
                    if best_trial_val is None or isinstance(best_trial_val, tuple) is False:
                        # try distributing diff among neighbors
                        if diff > 0:
                            give = [j for j in j_tries if j != i and h[j] >= diff]
                        else:
                            give = [j for j in j_tries if j != i and h[j] + diff >= 0]
                        if give:
                            j = give[0]
                            h[j] -= diff
                            c5_t = compute_c5_exact(h)
                            if c5_t < best_trial_c5:
                                best_trial_c5 = c5_t
                                best_trial_val = (cv, j, h[j])
                            h[j] += diff
            if best_trial_val is not None:
                if isinstance(best_trial_val, tuple):
                    cv, j, jval = best_trial_val
                    h[i] = cv
                    h[j] = jval
                else:
                    h[i] = best_trial_val
                best_c5 = best_trial_c5
                improved_this_sweep = True
                improved_total = True
            else:
                h[i] = old_val
        if not improved_this_sweep:
            break
    return h, best_c5, improved_total


def two_point_descent(h, n, best_c5, rng, max_sweeps=50, time_limit=60.0, t_start=None):
    if t_start is None:
        t_start = time.time()
    target = n / 2.0
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for _ in range(2000):
            if time.time() - t_start > time_limit:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            if h[i] < h[j]:
                d = 1
            else:
                d = -1
            for delta in [0.02, 0.01, 0.005, 0.05, 0.002]:
                di = d * delta
                dj = -d * delta
                vi = h[i] + di
                vj = h[j] + dj
                if vi < -1e-9 or vi > 1.0 + 1e-9 or vj < -1e-9 or vj > 1.0 + 1e-9:
                    continue
                h[i] = vi
                h[j] = vj
                c5_t = compute_c5_exact(h)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    improved = True
                    break
                h[i] -= di
                h[j] -= dj
            if improved:
                break
        if not improved:
            break
    return h, best_c5


def binary_threshold(h, target_sum, thresh_lo=0.1, thresh_hi=0.9):
    h2 = h.copy()
    h2[h2 < thresh_lo] = 0.0
    h2[h2 > thresh_hi] = 1.0
    # Fix sum
    current_sum = np.sum(h2)
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase some values
        free = np.where((h2 > 0.01) & (h2 < 0.99))[0]
        if len(free) > 0:
            h2[free] = np.clip(h2[free] + diff / len(free), 0.0, 1.0)
        else:
            h2 = project_feasible(h2, target_sum)
    elif diff < 0:
        free = np.where((h2 > 0.01) & (h2 < 0.99))[0]
        if len(free) > 0:
            h2[free] = np.clip(h2[free] + diff / len(free), 0.0, 1.0)
        else:
            h2 = project_feasible(h2, target_sum)
    h2 = project_feasible(h2, target_sum)
    return h2


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    def remaining():
        return max(0.0, budget_s - (time.time() - t0) - 2.0)

    rng = np.random.RandomState(seed)

    # Get parent
    h_par = np.array(initial_h_values, dtype=float).copy()
    n_par = len(h_par)
    c5_par = compute_c5_exact(h_par)
    print("Parent: n={}, C5={:.8f}".format(n_par, c5_par))

    best_h = h_par.copy()
    best_c5 = c5_par
    best_n = n_par

    # Phase 1: L-BFGS-B with active-set smooth max at parent resolution
    if remaining() > 60:
        n = n_par
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n

        for beta in [80, 150, 300, 600, 1200, 2500, 5000]:
            if remaining() < 15:
                break
            for top_k in [5, 8, 12]:
                if remaining() < 10:
                    break
                res = minimize(
                    objective_lbfgs, best_h,
                    method='L-BFGS-B',
                    bounds=bounds,
                    args=(n, beta, top_k, 1e5),
                    jac=lambda p: grad_lbfgs(p, n, beta, top_k, 1e5),
                    options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-14,
                             'maxls': 80, 'maxfun': 20000, 'maxcor': 30}
                )
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5_exact(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n
                    print("  L-BFGS n={}, beta={}, topk={}: C5={:.8f}".format(n, beta, top_k, c5_new))
                else:
                    # Still use the projected result as starting point
                    h_alt = project_feasible(res.x, target)
                    c5_alt = compute_c5_exact(h_alt)
                    if c5_alt < c5_par + 0.01:
                        best_h = h_alt.copy()

    # Phase 2: SLSQP with smooth max for polishing
    if remaining() > 30:
        n = best_n
        target = n / 2.0
        dx = 2.0 / n
        bounds = [(0.0, 1.0)] * n

        for beta in [500, 1000, 2000, 4000]:
            if remaining() < 8:
                break
            for top_k in [5, 8]:
                if remaining() < 5:
                    break
                def obj_slsqp(x):
                    corr = np.correlate(x, 1.0 - x, mode='full')
                    idx = np.argpartition(corr, -top_k)[-top_k:]
                    vals = corr[idx]
                    m = vals.max()
                    shifted = vals - m
                    z = np.exp(beta * shifted)
                    return (m + np.log(np.sum(z)) / beta) * dx

                def cons_fun(x):
                    return np.sum(x) - target

                def cons_jac(x):
                    return np.ones(n)

                try:
                    res = minimize(obj_slsqp, best_h, method='SLSQP',
                                   bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                                   maxiter=500, ftol=1e-16)
                    h_new = project_feasible(res.x, target)
                    c5_new = compute_c5_exact(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        print("  SLSQP beta={}, topk={}: C5={:.8f}".format(beta, top_k, c5_new))
                except Exception:
                    pass

    # Phase 3: Exact coordinate descent (key improvement)
    if remaining() > 60:
        n = best_n
        h = best_h.copy()
        time_limit_cd = min(remaining() - 5.0, 200.0)
        h, c5_cd, imp = exact_coord_descent(h, n, best_c5, max_sweeps=500,
                                            rng=np.random.RandomState(seed + 1),
                                            time_limit=time_limit_cd, t_start=time.time())
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h.copy()
            print("  After coord descent: C5={:.8f}".format(best_c5))

    # Phase 4: Two-point descent for global-ish moves
    if remaining() > 40:
        n = best_n
        h = best_h.copy()
        time_limit_tp = min(remaining() - 5.0, 120.0)
        h, c5_tp = two_point_descent(h, n, best_c5, np.random.RandomState(seed + 2),
                                     max_sweeps=100, time_limit=time_limit_tp,
                                     t_start=time.time())
        if c5_tp < best_c5:
            best_c5 = c5_tp
            best_h = h.copy()
            print("  After 2-point: C5={:.8f}".format(best_c5))

    # Phase 5: Binary thresholding + re-optimize
    if remaining() > 40:
        n = best_n
        target = n / 2.0
        h_bin = binary_threshold(best_h, target, 0.15, 0.85)
        c5_bin = compute_c5_exact(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            print("  Binary threshold: C5={:.8f}".format(best_c5))

        # Re-optimize from binary
        if remaining() > 20:
            bounds = [(0.0, 1.0)] * n
            beta = 2000
            top_k = 5
            res = minimize(
                objective_lbfgs, best_h,
                method='L-BFGS-B', bounds=bounds,
                args=(n, beta, top_k, 1e5),
                jac=lambda p: grad_lbfgs(p, n, beta, top_k, 1e5),
                options={'maxiter': 2000, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': 10000}
            )
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5_exact(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  L-BFGS after binary: C5={:.8f}".format(best_c5))

    # Phase 6: Try higher resolutions
    if remaining() > 80:
        for n_try in [n_par + 40, n_par + 80, n_par + 120, n_par + 160]:
            if remaining() < 40:
                break
            if n_try % 2 != 0:
                n_try += 1
            if n_try < 100 or n_try > 600:
                continue
            target_t = n_try / 2.0
            xp = (np.arange(best_n) + 0.5) * 2.0 / best_n
            xn = (np.arange(n_try) + 0.5) * 2.0 / n_try
            h0 = np.interp(xn, xp, best_h)
            h0 = project_feasible(h0, target_t)
            c5_0 = compute_c5_exact(h0)

            bounds_t = [(0.0, 1.0)] * n_try
            for beta in [300, 800, 2000]:
                if remaining() < 15:
                    break
                top_k = 6
                res = minimize(
                    objective_lbfgs, h0,
                    method='L-BFGS-B', bounds=bounds_t,
                    args=(n_try, beta, top_k, 1e5),
                    jac=lambda p: grad_lbfgs(p, n_try, beta, top_k, 1e5),
                    options={'maxiter': 2000, 'ftol': 1e-18, 'gtol': 1e-14,
                             'maxls': 50, 'maxfun': 10000}
                )
                h_new = project_feasible(res.x, target_t)
                c5_new = compute_c5_exact(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n_try
                    h0 = h_new.copy()
                    print("  Higher res n={}, beta={}: C5={:.8f}".format(n_try, beta, c5_new))
                else:
                    h_alt = project_feasible(res.x, target_t)
                    if compute_c5_exact(h_alt) < c5_0 + 0.005:
                        h0 = h_alt.copy()

            # Quick coord descent at this resolution
            if remaining() > 20 and best_n == n_try:
                time_limit_cd = min(remaining() - 10.0, 60.0)
                h_cd, c5_cd, _ = exact_coord_descent(best_h, n_try, best_c5,
                                                     max_sweeps=200,
                                                     rng=np.random.RandomState(seed + n_try),
                                                     time_limit=time_limit_cd,
                                                     t_start=time.time())
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                    print("  CD at n={}: C5={:.8f}".format(n_try, best_c5))

    # Phase 7: Final exact coordinate descent polish
    if remaining() > 30:
        n = best_n
        h = best_h.copy()
        time_limit_final = min(remaining() - 3.0, 100.0)
        h, c5_fin, _ = exact_coord_descent(h, n, best_c5, max_sweeps=300,
                                           rng=np.random.RandomState(seed + 99),
                                           time_limit=time_limit_final, t_start=time.time())
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h.copy()
            print("  Final CD: C5={:.8f}".format(best_c5))

    # Phase 8: Very fine two-point moves
    if remaining() > 15:
        n = best_n
        h = best_h.copy()
        time_limit_final2 = min(remaining() - 2.0, 40.0)
        h, c5_f2 = two_point_descent(h, n, best_c5, np.random.RandomState(seed + 200),
                                     max_sweeps=50, time_limit=time_limit_final2,
                                     t_start=time.time())
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h.copy()
            print("  Final 2pt: C5={:.8f}".format(best_c5))

    # Final projection and verification
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5_exact(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
