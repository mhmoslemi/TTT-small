import numpy as np
from scipy.optimize import minimize
import time
import math

_START = 0.0
_BUDGET = 1300.0
_GL_N = 0
_GL_BETA = 1.0


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 2.0)


def project_feasible(x, n):
    x = np.array(x, dtype=float).copy()
    np.clip(x, 0.0, 1.0, out=x)
    target = 0.5 * float(n)
    s = float(np.sum(x))
    if abs(s - target) <= 1e-12:
        return x

    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        smid = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if smid > target:
            hi = mid
        else:
            lo = mid

    lam = 0.5 * (lo + hi)
    y = np.clip(x + lam, 0.0, 1.0)
    diff = target - float(np.sum(y))
    free = (y > 1e-12) & (y < 1.0 - 1e-12)
    if np.any(free):
        y[free] += diff / float(np.sum(free))
    np.clip(y, 0.0, 1.0, out=y)
    return y


def is_feasible(h, n, tol=1e-8):
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    return abs(float(np.sum(h)) - 0.5 * float(n)) <= tol


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / float(n)
    return float(np.max(compute_corr(h)) * dx)


def interp_linear(src, sn, dn):
    if sn == dn:
        return src.copy()
    old = (np.arange(sn) + 0.5) * (2.0 / float(sn))
    new = (np.arange(dn) + 0.5) * (2.0 / float(dn))
    return np.interp(new, old, src)


def interp_nearest(src, sn, dn):
    if sn == dn:
        return src.copy()
    scale = float(sn) / float(dn)
    idx = np.floor((np.arange(dn) + 0.5) * scale).astype(int)
    idx %= sn
    return src[idx].copy()


def make_structural_h(n, seed_val):
    h = np.zeros(n, dtype=float)
    for i in range(n):
        v = i
        p = 0
        while v:
            p ^= 1
            v &= v - 1
        if p == 1:
            h[i] = 1.0
    return project_feasible(h, n)


def make_n_list(parent_n, budget_s):
    base = int(max(16, int(parent_n)))
    max_n = 1200 if budget_s > 800.0 else 900
    if base > max_n:
        max_n = max(max_n, base + 128)

    cand = []
    factors = [0.5, 0.6, 0.75, 1.0, 1.125, 1.25, 1.5, 1.75, 2.0, 2.5, 3.0]
    for f in factors:
        cand.append(int(round(base * float(f))))
    deltas = [0, 16, 32, 48, 64, 96, 128, 192, 256, 384, 512]
    for d in deltas:
        cand.append(base + d)
        cand.append(base - d)
    fixed = [256, 320, 384, 448, 512, 576, 640, 704, 768, 832, 896, 960, 1024, 1120, 1152]
    for nn in fixed:
        cand.append(nn)

    seen = set()
    pairs = []
    for nn in cand:
        nn = int(nn)
        if nn < 0:
            continue
        if nn % 2 == 1:
            nn += 1
        if 128 <= nn <= max_n and nn not in seen:
            seen.add(nn)
            pairs.append((abs(nn - base), nn))
    pairs.sort()
    out = []
    for p in pairs:
        out.append(p[1])
    return out


def smooth_obj(x):
    n = _GL_N
    corr = compute_corr(x)
    m = float(np.max(corr))
    arg = _GL_BETA * (corr - m)
    np.clip(arg, -50.0, 0.0, out=arg)
    ex = np.exp(arg)
    s = float(np.sum(ex))
    if s <= 0.0:
        s = 1.0
    return (m + math.log(s) / _GL_BETA) * (2.0 / float(n))


def smooth_grad(x):
    n = _GL_N
    corr = compute_corr(x)
    m = float(np.max(corr))
    arg = _GL_BETA * (corr - m)
    np.clip(arg, -50.0, 0.0, out=arg)
    ex = np.exp(arg)
    s = float(np.sum(ex))
    if s <= 0.0:
        ex = np.zeros_like(corr)
        ex[np.argmax(corr)] = 1.0
        s = 1.0
    w = ex / s

    g = np.zeros(n, dtype=float)
    idx0 = n - 1
    idxs = np.flatnonzero(w > 1e-9)
    for idx in idxs:
        wt = float(w[idx])
        if wt <= 1e-12:
            continue
        d = int(idx) - idx0
        if d == 0:
            g += wt * (1.0 - 2.0 * x)
        elif d > 0:
            N = n - d
            g[d:d + N] += wt * (1.0 - x[:N])
            g[:N] -= wt * x[d:d + N]
        else:
            l = -d
            N = n - l
            g[:N] += wt * (1.0 - x[l:l + N])
            g[l:l + N] -= wt * x[:N]
    g *= (2.0 / float(n))
    return g


def sum_con(x):
    return float(np.sum(x)) - 0.5 * float(_GL_N)


def sum_con_jac(x):
    return np.ones(_GL_N, dtype=float)


def slsqp_smooth(h, n, beta, maxiter):
    global _GL_N, _GL_BETA
    n = int(n)
    maxiter = int(maxiter)
    if maxiter <= 0:
        return h.copy()
    _GL_N = n
    _GL_BETA = float(beta)
    h0 = h.copy()
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{"type": "eq", "fun": sum_con, "jac": sum_con_jac}]
        res = minimize(
            smooth_obj,
            h0,
            jac=smooth_grad,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-12, "iprint": 0}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h0


def gradient_from_indices(h, idxs, weights):
    n = len(h)
    g = np.zeros(n, dtype=float)
    if len(idxs) == 0:
        return g
    idx0 = n - 1
    for t in range(len(idxs)):
        idx = int(idxs[t])
        wt = float(weights[t])
        if wt <= 1e-12:
            continue
        d = idx - idx0
        if d == 0:
            g += wt * (1.0 - 2.0 * h)
        elif d > 0:
            N = n - d
            g[d:d + N] += wt * (1.0 - h[:N])
            g[:N] -= wt * h[d:d + N]
        else:
            l = -d
            N = n - l
            g[:N] += wt * (1.0 - h[l:l + N])
            g[l:l + N] -= wt * h[:N]
    return g


def top_gradient(h, corr, count=12):
    n = len(h)
    m = len(corr)
    count = max(1, min(int(count), m))
    idx = np.argsort(corr)[-count:]
    idx = idx[::-1]
    w = np.ones(count, dtype=float) / float(count)
    return gradient_from_indices(h, idx, w)


def pair_eval_delta(base, A, q1, q2, delta, n):
    val = base + A * delta
    idx0 = n - 1
    d2 = delta * delta
    val[idx0] -= 2.0 * d2
    val[q1] += d2
    if q2 != q1:
        val[q2] += d2
    return float(np.max(val))


def pair_build_A(h, a, b, n):
    A = np.zeros(2 * n - 1, dtype=float)
    hrev = h[::-1]
    one_minus_hrev = 1.0 - hrev

    A[a:n + a] += one_minus_hrev
    A[n - 1 - a:2 * n - 1 - a] -= h

    A[b:n + b] -= one_minus_hrev
    A[n - 1 - b:2 * n - 1 - b] += h
    return A


def pair_grid_with_coeffs(base, A, q1, q2, lo, hi, n_grid, refine_iters, n):
    cur_lo = float(lo)
    cur_hi = float(hi)
    n_grid = max(2, int(n_grid))
    refine_iters = max(1, int(refine_iters))
    idx0 = n - 1
    span = cur_hi - cur_lo
    if span <= 1e-14:
        d = 0.5 * (cur_lo + cur_hi)
        return d, pair_eval_delta(base, A, q1, q2, d, n)

    best_delta = 0.5 * (cur_lo + cur_hi)
    best_val = float("inf")

    for it in range(refine_iters):
        grid = np.linspace(cur_lo, cur_hi, n_grid)
        vals = np.outer(grid, A) + base
        d2 = grid * grid
        vals[:, idx0] -= 2.0 * d2
        vals[:, q1] += d2
        if q2 != q1:
            vals[:, q2] += d2
        rowmax = vals.max(axis=1)
        i = int(np.argmin(rowmax))
        bd = float(grid[i])
        bv = float(rowmax[i])
        if bv < best_val:
            best_val = bv
            best_delta = bd

        if it == refine_iters - 1:
            break

        step = (cur_hi - cur_lo) / float(n_grid - 1)
        new_lo = max(float(lo), bd - step)
        new_hi = min(float(hi), bd + step)
        if new_hi - new_lo < 1e-14:
            if bd <= 0.5 * (float(lo) + float(hi)):
                new_hi = min(float(hi), bd + step)
            else:
                new_lo = max(float(lo), bd - step)
        if new_hi <= new_lo:
            break
        cur_lo, cur_hi = new_lo, new_hi

    best_delta = min(max(0.0, best_delta), float(hi))
    best_val = pair_eval_delta(base, A, q1, q2, best_delta, n)
    return best_delta, best_val


def pair_assess(h, corr, a, b, hi, n_grid, refine_iters):
    n = len(h)
    if a == b:
        return None
    dmax = float(min(h[b], 1.0 - h[a]))
    if dmax <= 1e-14:
        return None
    hi = min(float(hi), dmax)
    if hi <= 1e-14:
        return None
    A = pair_build_A(h, a, b, n)
    idx0 = n - 1
    q1 = idx0 + (b - a)
    q2 = idx0 + (a - b)
    bd, bv = pair_grid_with_coeffs(corr, A, q1, q2, 0.0, hi, n_grid, refine_iters, n)
    return bd, bv, A, q1, q2


def random_perturb(h, corr, n, rng, count=20, max_delta=0.1):
    h = h.copy()
    corr = corr.copy()
    idx0 = n - 1
    for _ in range(max(0, int(count))):
        a = int(rng.randint(n))
        b = int(rng.randint(n))
        if a == b:
            b = (b + 1) % n
        dmax1 = float(min(h[b], 1.0 - h[a]))
        dmax2 = float(min(h[a], 1.0 - h[b]))
        if dmax1 <= 1e-12 and dmax2 <= 1e-12:
            continue
        if dmax1 >= dmax2:
            inc, dec, md = a, b, dmax1
        else:
            inc, dec, md = b, a, dmax2
        upper = min(float(max_delta), 0.75 * md)
        if upper <= 1e-10:
            continue
        delta = float(rng.uniform(1e-5, upper))
        A = pair_build_A(h, inc, dec, n)
        q1 = idx0 + (dec - inc)
        q2 = idx0 + (inc - dec)
        h[inc] += delta
        h[dec] -= delta
        corr += A * delta
        d2 = delta * delta
        corr[idx0] -= 2.0 * d2
        corr[q1] += d2
        corr[q2] += d2

    if not is_feasible(h, n, 1e-7):
        h = project_feasible(h, n)
        corr = compute_corr(h)
    return h, corr

                    corr_cur[q1] += d2
                    corr_cur[q2] += d2
                    cur_max = float(np.max(corr_cur))
                    stall = 0

    h_out = project_feasible(best_h, n)
    c_out = compute_c5(h_out)
    if is_feasible(best_h, n, 1e-7):
        c_raw = compute_c5(best_h)
        if c_raw < c_out:
            h_out = best_h
            c_out = c_raw
    return h_out, c_out


def rotation_search(h, n, time_limit_s=20.0):
    h0 = np.array(h, dtype=float).copy()
    n = int(n)
    if n == 0:
        return h0, float("inf")
    best_h = h0.copy()
    best_c = compute_c5(h0)
    start = time.time()

    if n <= 512:
        soft = 40
    elif n <= 900:
        soft = 25
    else:
        soft = 15

    shifts = []
    for s in range(1, soft + 1):
        shifts.append(s)
        shifts.append(n - s)

    for sh in shifts:
        if time.time() - start > time_limit_s or remaining() < 2.0:
            break
        hr = np.roll(h0, sh)
        c = compute_c5(hr)
        if c < best_c:
            best_c = c
            best_h = hr.copy()

    rng = np.random.RandomState(987654321)
    for _ in range(12):
        if time.time() - start > time_limit_s or remaining() < 2.0:
            break
        sh = int(rng.randint(n))
        hr = np.roll(h0, sh)
        c = compute_c5(hr)
        if c < best_c:
            best_c = c
            best_h = hr.copy()

    return best_h, best_c


def replace_best_if_better(cand, best_h, best_c, best_n):
    if cand is None:
        return best_h, best_c, best_n
    hh = np.array(cand, dtype=float).copy()
    n = int(len(hh))
    if n <= 0:
        return best_h, best_c, best_n
    if not is_feasible(hh, n, 1e-7):
        hh = project_feasible(hh, n)
    cc = compute_c5(hh)
    if cc < best_c:
        return hh, cc, n
    return best_h, best_c, best_n


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = int(len(parent_h))
    except Exception:
        parent_n = 320
        parent_h = make_structural_h(parent_n, int(seed))

    if parent_n <= 0:
        parent_n = 320
        parent_h = make_structural_h(parent_n, int(seed))

    if not is_feasible(parent_h, parent_n, 1e-6):
        parent_h = project_feasible(parent_h, parent_n)

    best_h = parent_h.copy()
    best_n = int(parent_n)
    best_c = compute_c5(best_h)

    if remaining() <= 5.0:
        return best_h, best_c, best_n

    if best_n <= 640 and remaining() > 70.0:
        ht = slsqp_smooth(best_h, best_n, 500.0, 60)
        best_h, best_c, best_n = replace_best_if_better(ht, best_h, best_c, best_n)

    if best_n <= 640 and remaining() > 55.0:
        ht = slsqp_smooth(best_h, best_n, 2500.0, 60)
        best_h, best_c, best_n = replace_best_if_better(ht, best_h, best_c, best_n)

    if remaining() > 30.0:
        ht, ct = rotation_search(best_h, best_n, 20.0)
        best_h, best_c, best_n = replace_best_if_better(ht, best_h, best_c, best_n)

    t_parent = min(320.0, max(20.0, remaining() * 0.45))
    if t_parent > 35.0:
        ht, ct = pair_monte_carlo(best_h, best_n, t_parent, rng)
        best_h, best_c, best_n = replace_best_if_better(ht, best_h, best_c, best_n)

    trials = make_n_list(best_n, budget_s)
    trial_list = []
    for nn in trials:
        if nn != best_n:
            trial_list.append(nn)
        if len(trial_list) >= 7:
            break

    reserve = max(60.0, 0.18 * float(budget_s))
    for idx, nn in enumerate(trial_list):
        if remaining() <= reserve + 35.0:
            break

        h_lin = project_feasible(interp_linear(best_h, best_n, nn), nn)
        c_lin = compute_c5(h_lin)
        h_ner = project_feasible(interp_nearest(best_h, best_n, nn), nn)
        c_ner = compute_c5(h_ner)
        h0 = h_ner if c_ner < c_lin else h_lin
        best_h, best_c, best_n = replace_best_if_better(h0, best_h, best_c, best_n)

        h1 = h0
        if nn <= 384 and remaining() > reserve + 55.0:
            h1 = slsqp_smooth(h1, nn, 1000.0, 40)
            best_h, best_c, best_n = replace_best_if_better(h1, best_h, best_c, best_n)

        rem_trials = max(0.0, remaining() - reserve - 20.0)
        t_alloc = max(20.0, min(140.0, rem_trials / max(1, len(trial_list) - idx) * 0.9))
        if remaining() > reserve + 40.0:
            ht, ct = pair_monte_carlo(h1, nn, t_alloc, rng)
            best_h, best_c, best_n = replace_best_if_better(ht, best_h, best_c, best_n)
            if remaining() > reserve + 15.0:
                ht, ct = rotation_search(best_h, best_n, min(15.0, remaining() - reserve - 8.0))
                best_h, best_c, best_n = replace_best_if_better(ht, best_h, best_c, best_n)

    if best_n <= 512 and remaining() > 50.0:
        ht = slsqp_smooth(best_h, best_n, 5000.0, 60)
        best_h, best_c, best_n = replace_best_if_better(ht, best_h, best_c, best_n)

    if remaining() > 25.0:
        ht, ct = rotation_search(best_h, best_n, min(20.0, remaining() - 8.0))
        best_h, best_c, best_n = replace_best_if_better(ht, best_h, best_c, best_n)

    t_final = max(15.0, remaining() - 30.0)
    if t_final > 25.0:
        ht, ct = pair_monte_carlo(best_h, best_n, t_final, rng)
        best_h, best_c, best_n = replace_best_if_better(ht, best_h, best_c, best_n)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if isinstance(eval_c5, (tuple, list)):
            eval_c5 = eval_c5[0]
        eval_c5 = float(eval_c5)
        if eval_c5 < best_c:
            best_c = eval_c5
    except Exception:
        pass

    return best_h, best_c, int(best_n)
