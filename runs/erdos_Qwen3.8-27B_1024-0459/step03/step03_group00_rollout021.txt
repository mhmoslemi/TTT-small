def obj_lob(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    b = _G_BETA
    s = corr - m
    e = np.exp(np.clip(b * s, -400, 400))
    val = (m + np.log(np.sum(e)) / b) * _G_DX
    if _G_PEN > 0:
        val += _G_PEN * (np.sum(x) - _G_TGT) ** 2
    if _G_FOCUS >= 0 and _G_FOCUS < len(corr):
        val += 0.01 * corr[_G_FOCUS] * _G_DX
    return float(val)

def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c else 0.0
    return h

def make_rand_bin(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def make_sine(rng, n, f1, f2):
    t = np.arange(n) / n
    h = 0.5 + 0.4 * np.sin(2.0 * np.pi * f1 * t) + 0.25 * np.sin(2.0 * np.pi * f2 * t + rng.rand() * 6.28)
    return np.clip(h, 0.0, 1.0)

def slsqp_run(h0, n, beta, maxiter, use_eq=True):
    global _G_N, _G_TGT, _G_DX, _G_BETA, _G_PEN, _G_FOCUS
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    _G_BETA = float(beta)
    _G_PEN = 0.0
    _G_FOCUS = -1
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    try:
        res = minimize(obj_lob, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def lbfgsb_run(h0, n, beta, maxiter, pen=100.0):
    global _G_N, _G_TGT, _G_DX, _G_BETA, _G_PEN, _G_FOCUS
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    _G_BETA = float(beta)
    _G_PEN = float(pen)
    _G_FOCUS = -1
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_lob, h0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def coord_descent(h, n, max_sweeps, rng, tlimit):
    c5 = compute_c5(h)
    h_best = h.copy()
    for sweep in range(max_sweeps):
        if tlimit() < 2.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if tlimit() < 1.0:
                break
            v0 = h_best[i]
            for v in [0.0, 1.0]:
                if abs(v - v0) < 1e-9:
                    continue
                trial = h_best.copy()
                trial[i] = v
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 5e-9:
                    c5 = c
                    h_best = trial.copy()
                    improved = True
            for d in [0.15, -0.15, 0.08, -0.08, 0.25, -0.25, 0.04, -0.04]:
                if tlimit() < 1.0:
                    break
                trial = h_best.copy()
                trial[i] = np.clip(trial[i] + d, 0.0, 1.0)
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 5e-9:
                    c5 = c
                    h_best = trial.copy()
                    improved = True
        if not improved and sweep >= 2:
            break
    return h_best, c5

def focused_iteration(h, n, rng, tlimit, max_rounds=50):
    """Iteratively identify the active lag and focus on reducing it."""
    c5 = compute_c5(h)
    h_best = h.copy()
    for rnd in range(max_rounds):
        if tlimit() < 3.0:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode='full')
        k_active = int(np.argmax(corr))
        cmax = corr[k_active]
        # SLSQP with focus on this lag
        global _G_FOCUS
        _G_N = n
        _G_TGT = n / 2.0
        _G_DX = 2.0 / n
        _G_BETA = 500.0
        _G_PEN = 0.0
        _G_FOCUS = k_active
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
        try:
            res = minimize(obj_lob, h_best, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 200, 'ftol': 1e-14, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new)
            if c_new < c5 - 1e-10:
                c5 = c_new
                h_best = h_new
        except Exception:
            pass
        _G_FOCUS = -1
    return h_best, c5

def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX, _G_PEN, _G_FOCUS
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    _G_PEN = 0.0
    _G_FOCUS = -1
    rng = np.random.RandomState(seed)

    def tlimit():
        return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = make_tm(parent_n)
        parent_h = project_feasible(parent_h, parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("No parent. TM: n={}, C5={:.8f}".format(parent_n, best_c5))

    # Phase 1: Intensive optimization at parent resolution
    if tlimit() > 100:
        print("\n=== Phase 1: n={} intensive ===".format(best_n))
        n = best_n
        h_cur = best_h.copy()
        c5_cur = best_c5

        beta_schedule = [30, 60, 100, 200, 400, 800, 1500, 3000, 5000, 8000]
        for bv in beta_schedule:
            if tlimit() < 15:
                break
            h_new = slsqp_run(h_cur, n, bv, maxiter=400)
            c_new = compute_c5(h_new)
            if c_new < c5_cur - 1e-12:
                c5_cur = c_new
                h_cur = h_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n
        print("  After multi-beta: C5={:.8f}".format(best_c5))

        # Focused iteration
        if tlimit() > 30:
            h_cur, c5_cur = focused_iteration(h_cur, n, rng, tlimit, max_rounds=30)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n
            print("  After focused: C5={:.8f}".format(best_c5))

    # Phase 2: Try other resolutions
    if tlimit() > 80:
        print("\n=== Phase 2: other resolutions ===")
        n_tries = sorted(set([best_n, 200, 256, 320, 384, 448, 512]))
        n_tries = [x for x in n_tries if 128 <= x <= 512 and x % 2 == 0]

        for n in n_tries:
            if tlimit() < 40:
                break
            if n == best_n:
                continue

            h0 = project_feasible(interp_h(best_h, best_n, n), n)
            c5_0 = compute_c5(h0)

            # Also try random binary init
            h_rand = project_feasible(make_rand_bin(rng, n), n)
            c5_r = compute_c5(h_rand)

            # Also try Thue-Morse
            h_tm = project_feasible(make_tm(n), n)
            c5_t = compute_c5(h_tm)

            inits = [(h0, c5_0), (h_rand, c5_r), (h_tm, c5_t)]
            inits.sort(key=lambda x: x[1])

            h_cur = inits[0][0].copy()
            c5_cur = inits[0][1]

            for bv in [50, 150, 400, 1000, 3000]:
                if tlimit() < 10:
                    break
                if n <= 320:
                    h_new = slsqp_run(h_cur, n, bv, maxiter=350)
                else:
                    h_new = lbfgsb_run(h_cur, n, bv, maxiter=300, pen=50.0)
                    h_new = slsqp_run(h_new, n, bv, maxiter=200)
                c_new = compute_c5(h_new)
                if c_new < c5_cur - 1e-12:
                    c5_cur = c_new
                    h_cur = h_new

            if c5_cur < best_c5 - 1e-9:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n
                print("  n={} improved: C5={:.8f}".format(n, best_c5))
            else:
                print("  n={} best: C5={:.8f} (no improvement over {:.8f})".format(n, c5_cur, best_c5))

    # Phase 3: Aggressive coordinate descent at best resolution
    if tlimit() > 20:
        print("\n=== Phase 3: coordinate descent n={} ===".format(best_n))
        best_n = len(best_h)
        best_h, best_c5 = coord_descent(best_h, best_n, max_sweeps=40, rng=rng, tlimit=tlimit)
        print("  After coord descent: C5={:.8f}".format(best_c5))

    # Phase 4: Random perturbation + re-optimization to escape local optima
    if tlimit() > 60:
        print("\n=== Phase 4: perturbation escapes ===")
        n = best_n
        for attempt in range(8):
            if tlimit() < 15:
                break
            # Random perturbation
            h_pert = best_h.copy()
            n_pert = rng.randint(3, 15)
            idxs = rng.choice(n, size=n_pert, replace=False)
            for idx in idxs:
                h_pert[idx] = np.clip(h_pert[idx] + rng.randn() * 0.3, 0.0, 1.0)
            h_pert = project_feasible(h_pert, n)
            c5_p = compute_c5(h_pert)

            if c5_p < best_c5 * 1.05:
                h_cur = h_pert.copy()
                c5_cur = c5_p
                for bv in [200, 800, 2000]:
                    if tlimit() < 8:
                        break
                    h_new = slsqp_run(h_cur, n, bv, maxiter=250)
                    c_new = compute_c5(h_new)
                    if c_new < c5_cur - 1e-12:
                        c5_cur = c_new
                        h_cur = h_new
                if c5_cur < best_c5 - 1e-9:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n
                    print("  Perturb {} improved: C5={:.8f}".format(attempt, best_c5))

    # Phase 5: Final focused refinement
    if tlimit() > 20:
        print("\n=== Phase 5: final focused ===")
        best_n = len(best_h)
        best_h, best_c5 = focused_iteration(best_h, best_n, rng, tlimit, max_rounds=40)
        print("  After final focused: C5={:.8f}".format(best_c5))

    # Phase 6: Final coordinate descent pass
    if tlimit() > 10:
        best_n = len(best_h)
        best_h, best_c5 = coord_descent(best_h, best_n, max_sweeps=30, rng=rng, tlimit=tlimit)
        print("  After final coord: C5={:.8f}".format(best_c5))

    # Phase 7: Ultra-fine SLSQP polish
    if tlimit() > 8:
        best_n = len(best_h)
        for bv in [5000, 10000, 20000]:
            if tlimit() < 5:
                break
            h_new = slsqp_run(best_h, best_n, bv, maxiter=500)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-13:
                best_c5 = c_new
                best_h = h_new

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    _G_PEN = 0.0
    _G_FOCUS = -1
    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
