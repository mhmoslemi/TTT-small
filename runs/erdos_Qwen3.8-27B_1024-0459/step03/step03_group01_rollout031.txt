    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    s = np.sum(ex)

def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g

def smooth_grad(x):

    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)

    if s < 1e-300:

    else:
        w = ex / s
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-14:
            g += w[k] * corr_grad_per_shift(x, n, k)

    _TGT = n / 2.0

        if remaining() < 5:

        obj = smooth_obj(h)
        grad = smooth_grad(h)
        gnorm = np.linalg.norm(grad)

        alpha = 0.1
        improved = False

    n = len(h)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

        if w[k] > 1e-13:
            off = k - (n - 1)
            if off >= 0:
                g[:n - off] += 1.0 - x[off:]
                g[off:] -= x[:n - off]
            else:
                a = -off
                g[a:] += 1.0 - x[:n - a]
                g[:n - a] -= x[a:]
    return g * (2.0 / n)


def _scf(x):
    return x.sum() - _TG


def _scj(x):
    return np.ones(_NG)


def _slsqp(h, n, beta, mi=300):
    global _NG, _BG2, _TG
    _NG = n
    _BG2 = beta
    _TG = n / 2.0
    try:
        r = minimize(_sobj, h, jac=_sgrad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scf, 'jac': _scj}],
                     options={'maxiter': mi, 'ftol': 1e-15, 'iprint': 0})
        return _proj(r.x, n)
    except Exception:
        return h


def _tobin(h, n, sd):
    r = np.random.RandomState(sd)
    b = (h > 0.5).astype(float)
    cnt = int(b.sum())
    if cnt > n // 2:
        idx = np.where(b == 1)[0]
        b[idx[r.choice(len(idx), cnt - n // 2, replace=False)]] = 0.0
    elif cnt < n // 2:
        idx = np.where(b == 0)[0]
        b[idx[r.choice(len(idx), n // 2 - cnt, replace=False)]] = 1.0
    return b


def _sa(h, n, tl, sd=42):
