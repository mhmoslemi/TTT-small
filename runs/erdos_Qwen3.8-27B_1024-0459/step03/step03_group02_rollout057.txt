        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-14 and n > 0:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = min(1.0, max(0.0, h[idx] + diff))
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def smooth_max_obj(x):
    n = _N_GLOBAL
    beta = _BETA_GLOBAL
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    if m < -1e10:
        return 0.0
    shifted = np.minimum(beta * (corr - m), 50.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def constraint_sum(x):
    return np.sum(x) - _N_GLOBAL / 2.0

def slsqp_refine(h, n, beta, maxiter):
    global _N_GLOBAL, _BETA_GLOBAL
    _N_GLOBAL = n
    _BETA_GLOBAL = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    try:
        res = minimize(smooth_max_obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def golden_minimax(cons_arr, slope_arr, lo, hi):
    """Minimize max_d(cons_arr[d] + slope_arr[d]*x) over x in [lo, hi]."""
    if hi - lo < 1e-15:
        return lo
    if abs(slope_arr).max() < 1e-15:
        return (lo + hi) / 2.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = float(np.max(cons_arr + c * slope_arr))
    fd = float(np.max(cons_arr + d * slope_arr))
    for _ in range(50):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = float(np.max(cons_arr + c * slope_arr))
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = float(np.max(cons_arr + d * slope_arr))
    best_x = (a + b) * 0.5
    best_val = float(np.max(cons_arr + best_x * slope_arr))
    v0 = float(np.max(cons_arr + lo * slope_arr))
    v1 = float(np.max(cons_arr + hi * slope_arr))
    if v0 < best_val:
        return lo
    elif v1 < best_val:
        return hi
    return best_x

def coord_1d(h, n, i):
    """Optimize coordinate i. Returns optimal value."""
    h_save = h[i]
    h[i] = 0.0
    corr_0 = compute_corr(h)
    h[i] = 1.0
    corr_1 = compute_corr(h)
    h[i] = h_save
    slope = corr_1 - corr_0
    return golden_minimax(corr_0, slope, 0.0, 1.0)
