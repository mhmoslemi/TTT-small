def compute_smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # corr has 2n-1 entries, index k corresponds to shift (k - n + 1)
    m = np.max(corr)
    if beta < 1e-6:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
    else:
        exps = np.exp(beta * (corr - m))
        weights = exps / np.sum(exps)
    
    # For each shift m_k (where m_k = k - n + 1), the gradient is:
    # dF_k/dh[j] = (1-h[j+m_k]) - h[j-m_k] (with boundary handling)
    # This is: shifted(1-h, -m_k) - shifted(h, m_k)
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if abs(weights[k]) < 1e-10:
            continue
        shift = k - n + 1  # actual shift value
        # Contribution from (1-h[j+shift]) for valid j:
        # j+shift must be in [0, n-1], so j in [-shift, n-1-shift]
        # And j must be in [0, n-1]
        if shift >= 0:
            # (1-h[j+shift]): j in [0, n-1-shift]
            n1 = n - shift
            grad[:n1] += weights[k] * (1.0 - h[shift:])
            # -h[j-shift]: j in [shift, n-1]
            grad[shift:] += weights[k] * (-h[:n1])
        else:
            s = -shift
            # (1-h[j-shift]) = (1-h[j+s]): j in [0, n-1-s]
            n1 = n - s
            grad[:n1] += weights[k] * (1.0 - h[s:])
            # -h[j+shift] = -h[j-s]: j in [s, n-1]
            grad[s:] += weights[k] * (-h[:n1])
    
    return grad * dx, corr * dx, np.max(corr) * dx

            adjust = diff / cnt
            # Check if adjustment would violate bounds
            if adjust > 0:

            if i == j:
                continue
            found = False
            for d in deltas:
                if h[i] >= d and h[j] <= 1.0 - d:
                    h_tr = h.copy()
                    h_tr[i] -= d
                    h_tr[j] += d
                    c5_tr, _ = compute_c5(h_tr)
                    if c5_tr < best_c5:
                        best_c5 = c5_tr
                        best_h = h_tr.copy()
                        h = h_tr.copy()
                        found = True
                        break
                if h[j] >= d and h[i] <= 1.0 - d:
                    h_tr = h.copy()
                    h_tr[j] -= d
                    h_tr[i] += d
                    c5_tr, _ = compute_c5(h_tr)
                    if c5_tr < best_c5:
                        best_c5 = c5_tr
                        best_h = h_tr.copy()
                        h = h_tr.copy()
                        found = True
                        break

    # Phase 6: Triple moves
    if tl() > 10:
        n = best_n
        h = best_h.copy()
        for titer in range(30000):
            if tl() < 2:
                break
            idx3 = rng.choice(n, size=3, replace=False)
            i, j, k = int(idx3[0]), int(idx3[1]), int(idx3[2])
            d = 0.005 * (1.0 + rng.rand() * 3.0)
            if h[i] >= 2.0 * d and h[j] <= 1.0 - d and h[k] <= 1.0 - d:
                h_tr = h.copy()
                h_tr[i] -= 2.0 * d
                h_tr[j] += d
                h_tr[k] += d
                c5_tr, _ = compute_c5(h_tr)
                if c5_tr < best_c5:
                    best_c5 = c5_tr
                    best_h = h_tr.copy()
                    h = h_tr.copy()
                    continue
            if h[j] >= d and h[k] >= d and h[i] <= 1.0 - 2.0 * d:
                h_tr = h.copy()
                h_tr[j] -= d
                h_tr[k] -= d
                h_tr[i] += 2.0 * d
                c5_tr, _ = compute_c5(h_tr)
                if c5_tr < best_c5:
                    best_c5 = c5_tr
                    best_h = h_tr.copy()
                    h = h_tr.copy()
                    continue

    # Phase 7: Final polish with very high beta SLSQP
    if tl() > 10:
        n = best_n
        h = best_h.copy()
        _g_n = n
        _g_target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        for beta_f in [2000, 5000, 10000]:
            if tl() < 4:
                break
            _g_beta = beta_f
            _g_active = np.arange(2 * n - 1)
            try:
                def _final_obj(x):
                    n_f = _g_n
                    dx_f = 2.0 / n_f
                    corr_f = np.correlate(x, 1.0 - x, mode='full')
                    m_f = np.max(corr_f)
                    b_f = _g_beta
                    exps = np.exp(b_f * (corr_f - m_f))
                    return (m_f + np.log(np.sum(exps)) / b_f) * dx_f

                def _final_con(x):
                    return np.sum(x) - _g_target

                res = minimize(_final_obj, h, method='SLSQP',
                             bounds=bounds,
                             constraints=[{'type': 'eq', 'fun': _final_con}],
                             options={'maxiter': 400, 'ftol': 1e-17, 'iprint': 0})
                h_new = project_h(res.x, n)
                c5_new, _ = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                    print("  Final SLSQP beta={}: C5={:.10f}".format(beta_f, c5_new))
            except Exception:
                pass

    # Final
    best_h = project_h(best_h, best_n)
    final_c5, _ = compute_c5(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
