```python
def active_set_lp(h, n, n_active=20, margin=0.5):
    """Solve the LP to find the best linearized step."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    
    # Identify active lags
    active_threshold = max_corr - margin
    active_indices = np.where(corr >= active_threshold)[0]
    
    if len(active_indices) == 0:
        return None
    
    # Build constraint matrix for the LP
    # Variables: δ[0], ..., δ[n-1], η
    # Minimize: -η (to maximize η) → coeffs: [0,...,0, -1]
    # Constraints:
    #   For each active lag k: sum_i grad_k[i] * δ[i] + η ≤ 0  (i.e., grad·δ ≤ -η)
    #   sum δ = 0
    #   -h[i] ≤ δ[i] ≤ 1-h[i]  → bounds
    
    n_lags = len(active_indices)
    c = np.zeros(n + 1)
    c[-1] = -1.0  # minimize -η
    
    # Inequality constraints: A_ub @ x ≤ b_ub
    # For each active lag: sum_i grad_k[i]*δ[i] - (-η) ≤ 0 → sum_i grad_k[i]*δ[i] + η ≤ 0
    # Wait: we want grad_k · δ ≤ -η, i.e., grad_k · δ + η ≤ 0
    # So A_ub row: [grad_k[0], ..., grad_k[n-1], 1], b_ub = 0
    
    A_ub_rows = []
    b_ub_vals = []
    for idx in active_indices:
        offset = idx - (n - 1)
        grad = lag_gradient(h, n, offset)
        row = np.append(grad, 1.0)
        A_ub_rows.append(row)
        b_ub_vals.append(0.0)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    
    # Equality constraint: sum δ = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    
    # Bounds: -h[i] ≤ δ[i] ≤ 1-h[i], η ≤ some cap (say 0.1)
    bounds = [(-h[i], 1-h[i]) for i in range(n)]
    bounds.append((0.0, 0.1))  # η ≥ 0, η ≤ 0.1 (cap the step)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0], bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    
    return res.x[:n]  # the δ step direction
```

Looking at this problem, I need to beat the current record of 0.380903 and aim for ≤ 0.38085. The previous code achieved 0.380903 using subgradient descent + SLSQP.

My key innovation: replace the subgradient method with a **linear programming (LP) active-set method**. At each iteration, I:
1. Identify all "active" lags (near the maximum)
2. Solve an exact LP to find the optimal descent direction that simultaneously reduces ALL active lags while preserving the sum constraint

This is fundamentally different from the previous weighted-gradient approach because the LP finds the *exact* best compromise direction over the linearized feasible set, not an approximate one.

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_N_GLOBAL = 160
_BETA_GLOBAL = 500.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15 and n > 0:
        best_j = -1
        best_cost = 999.0
        for j in range(n):
            v = h[j] + diff
            if 0.0 <= v <= 1.0:
                cost = min(v, 1.0 - v)
                if cost < best_cost:
                    best_cost = cost
                    best_j = j
            else:
                cost = min(abs(v), abs(v - 1.0)) + 1.0
                if cost < best_cost:
                    best_cost = cost
                    best_j = j
        if best_j >= 0:
            h[best_j] += diff
        else:
            h[:] += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, n, offset):
    g = np.zeros(n)
    if offset >= 0:
        m = n - offset
        if m <= 0:
            return g
        g[:m] -= h[offset:offset + m]
        g[offset:offset + m] += 1.0 - h[:m]
    else:
        a = -offset
        m = n - a
        if m <= 0:
            return g
        g[:m] += 1.0 - h[a:a + m]
        g[a:a + m] -= h[:m]
    return g

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def lp_step(h, n, margin, max_active):
    corr = compute_corr(h)
    mx = float(np.max(corr))
    active = np.where(corr >= mx - margin)[0]
    if len(active) == 0:
        return None
    if len(active) > max_active:
        ord_idx = np.argsort(corr[active])[::-1]
        active = active[ord_idx[:max_active]]
    nv = n + 1
    c = np.zeros(nv)
    c[-1] = 1.0
    na = len(active)
    A_ub = np.zeros((na, nv))
    for j in range(na):
        ki = int(active[j])
        off = ki - (n - 1)
        g = lag_gradient(h, n, off)
        A_ub[j, :n] = g
        A_ub[j, n] = -1.0
    b_ub = np.zeros(na)
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    bounds = [(-float(h[i]), float(1.0 - h[i])) for i in range(n)]
    bounds.append((None, 0.1))
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if res.status != 0:
        return None
    if res.x[-1] >= -1e-12:
        return None
    return res.x[:n].copy()

def smax_val(x):
    n = _N_GLOBAL
    beta = _BETA_GLOBAL
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    sh = np.minimum(beta * (corr - m), 0.0)
    return (m + np.log(np.sum(np.exp(sh))) / beta) * dx

def smax_grad(x):
    n = _N_GLOBAL
    beta = _BETA_GLOBAL
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    sh = np.minimum(beta * (corr - m), 0.0)
    w = np.exp(sh)
    ws = float(np.sum(w))
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k] / ws
        if wk < 1e-13:
            continue
        off = k - (n - 1)
        grad += wk * lag_gradient(x, n, off)
    return dx * grad

def sum_con(x):
    return np.sum(x) - _N_GLOBAL / 2.0

def slsqp_refine(h, n, beta, niter):
    global _N_GLOBAL, _BETA_GLOBAL
    _N_GLOBAL = n
    _BETA_GLOBAL = beta
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(smax_val, h, jac=smax_grad, method='SLSQP',
                     bounds=bnds,
                     constraints=[{'type': 'eq', 'fun': sum_con}],
                     options={'maxiter': niter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(r.x, n)
    except Exception:
        return h.copy()

def interp_h(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)

def do_lp_phase(h, n, max_iters):
    best_h_local = h.copy()
    best_c5_local = compute_c5(h)
    stall = 0
    for it in range(max_iters):
        if remaining_time() < 10:
            break
        delta = lp_step(h, n, 1.0, 60)
        if delta is None:
            delta = lp_step(h, n, 2.0, 100)
        if delta is None:
            delta = lp_step(h, n, 5.0, 150)
        if delta is None:
            corr = compute_corr(h)
            top = int(np.argmax(corr))
            off = top - (n - 1)
            g = lag_gradient(h, n, off)
            g = g - np.mean(g)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            delta = -g / gn
        improved = False
        for a in [1.0, 0.5, 0.25, 0.75, 0.4, 0.15, 0.1, 0.05, 0.02, 0.01, 0.005]:
            ht = h + a * delta
            ht = project_feasible(ht, n)
            c5t = compute_c5(ht)
            if c5t < best_c5_local - 1e-15:
                best_c5_local = c5t
                best_h_local = ht.copy()
                h = ht.copy()
                improved = True
                break
        if improved:
            stall = 0
        else:
            stall += 1
        if stall > 0 and stall % 150 == 0 and remaining_time() > 30:
            hn = slsqp_refine(h, n, 500, 100)
            c5n = compute_c5(hn)
            if c5n < best_c5_local:
                best_c5_local = c5n
                best_h_local = hn.copy()
                h = hn.copy()
                stall = 0
        if stall >= 600:
            break
    return best_h_local, best_c5_local

def do_slsqp_phase(h, n):
    best_h_local = h.copy()
    best_c5_local = compute_c5(h)
    for beta in [200, 500, 1000, 2000, 5000, 8000]:
        if remaining_time() < 12:
            break
        hn = slsqp_refine(h, n, beta, 400)
        c5n = compute_c5(hn)
        if c5n < best_c5_local:
            best_c5_local = c5n
            best_h_local = hn.copy()
            h = hn.copy()
    return best_h_local, best_c5_local

def do_local_search(h, n, rng, max_sweeps):
    best_h_local = h.copy()
    best_c5_local = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 8:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if remaining_time() < 3:
                break
            ov = h[i]
            for nv in [0.0, 1.0, 0.5, 0.25, 0.75]:
                if abs(nv - ov) < 1e-9:
                    continue
                h[i] = nv
                h = project_feasible(h, n)
                c5t = compute_c5(h)
                if c5t < best_c5_local - 1e-14:
                    best_c5_local = c5t
                    best_h_local = h.copy()
                    any_imp = True
                    break
                else:
                    h = best_h_local.copy()
            for j in range(i + 1, min(i + 10, n)):
                if remaining_time() < 1:
                    break
                if abs(h[i] - h[j]) < 1e-9:
                    continue
                h[i], h[j] = h[j], h[i]
                h = project_feasible(h, n)
                c5t = compute_c5(h)
                if c5t < best_c5_local - 1e-14:
                    best_c5_local = c5t
                    best_h_local = h.copy()
                    any_imp = True
                else:
                    h = best_h_local.copy()
        if not any_imp:
            break
    return best_h_local, best_c5_local
```
