    h = np.where(h_init > 0.5, 1.0, 0.0)
    target_ones = n // 2
    num_ones = int(np.sum(h))
    if num_ones > target_ones:
        ones_idx = np.where(h == 1.0)[0]
        remove = rng.choice(len(ones_idx), num_ones - target_ones, replace=False)
        h[ones_idx[remove]] = 0.0
    elif num_ones < target_ones:
        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
        h[zeros_idx[add]] = 1.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    
    no_improve_count = 0
    for it in range(max_iter):
        if remaining_time() < 5:
            break
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        
        i = rng.randint(len(ones_list))
        j = rng.randint(len(zeros_list))
        oi = ones_list[i]
        oz = zeros_list[j]
        
        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = compute_c5(h)
        
        T = t_start * (t_end / t_start) ** (float(it) / max(float(max_iter), 1.0))
        
        if new_c5 < current_c5:
            current_c5 = new_c5
            ones_list[i] = oz
            zeros_list[j] = oi
            no_improve_count = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        elif T > 1e-15 and rng.rand() < np.exp(-(new_c5 - current_c5) / T):
            current_c5 = new_c5
            ones_list[i] = oz
            zeros_list[j] = oi
            no_improve_count += 1
        else:
            h[oi] = 1.0
            h[oz] = 0.0
            no_improve_count += 1
        
        # Restart if stuck too long
        if no_improve_count > 5000:
            h = best_h.copy()
            current_c5 = best_c5
            ones_list = list(np.where(h == 1.0)[0])
            zeros_list = list(np.where(h == 0.0)[0])
            no_improve_count = 0
    
    return best_h, best_c5

def multi_swap_local_search(h_init, n, max_iter, rng, k=2):
    """Try swapping k ones with k zeros simultaneously"""
    h = np.where(h_init > 0.5, 1.0, 0.0)
    target_ones = n // 2
    num_ones = int(np.sum(h))
    if num_ones != target_ones:
        if num_ones > target_ones:
            ones_idx = np.where(h == 1.0)[0]
            remove = rng.choice(len(ones_idx), num_ones - target_ones, replace=False)
            h[ones_idx[remove]] = 0.0
        else:
            zeros_idx = np.where(h == 0.0)[0]
            add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
            h[zeros_idx[add]] = 1.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    
    for it in range(max_iter):
        if remaining_time() < 3:
            break
        
        # Try random k-swap
        if len(ones_list) < k or len(zeros_list) < k:
            break
            
        sel_ones = rng.choice(len(ones_list), k, replace=False)
        sel_zeros = rng.choice(len(zeros_list), k, replace=False)
        
        old_ones = [ones_list[i] for i in sel_ones]
        old_zeros = [zeros_list[i] for i in sel_zeros]
        
        for pos in old_ones:
            h[pos] = 0.0
        for pos in old_zeros:
            h[pos] = 1.0
        
        new_c5 = compute_c5(h)
        
        if new_c5 < current_c5:
            current_c5 = new_c5
            for i, pos in enumerate(old_ones):
                ones_list[sel_ones[i]] = old_zeros[i]
            for i, pos in enumerate(old_zeros):
                zeros_list[sel_zeros[i]] = old_ones[i]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            for pos in old_ones:
                h[pos] = 1.0
            for pos in old_zeros:
                h[pos] = 0.0
    
    return best_h, best_c5

def active_shift_optimize(h, n, top_k=3, n_iter=200, rng=None):
    """Optimize to specifically reduce the top-k correlation shifts"""
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    dx = 2.0 / n
    
    for it in range(n_iter):
        if remaining_time() < 3:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        curr_max = corr[top_idx[0]]
        
        # Compute gradient only from top-k shifts
        grad = np.zeros(n)
        for j in top_idx:
            s = j - (n - 1)
            if s >= 0:
                for m_idx in range(s, n):
                    grad[m_idx] += (1.0 - h[m_idx - s])
                for m_idx in range(0, n - s):
                    grad[m_idx] += (-h[m_idx + s])
            else:
                sp = -s
                for m_idx in range(0, n - sp):
                    grad[m_idx] += (1.0 - h[m_idx + sp])
                for m_idx in range(sp, n):
                    grad[m_idx] += (-h[m_idx - sp])
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        grad /= gnorm
        
        # Line search
        best_trial = None
        best_trial_c5 = compute_c5(h)
        for step in [0.01, 0.02, 0.05, 0.1, 0.03, 0.08]:
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_trial_c5:
                best_trial_c5 = c5_trial
                best_trial = h_trial.copy()
        
        if best_trial is None:
            # Try smaller steps
            for step in [0.005, 0.002, 0.015]:
                h_trial = h - step * grad
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_trial_c5:
                    best_trial_c5 = c5_trial
                    best_trial = h_trial.copy()
        
        if best_trial is None:
            break
        
        h = best_trial
    
    return h, compute_c5(h)

def coordinate_descent(h, n, n_sweeps=10, rng=None):
    """Coordinate descent: optimize one h[i] at a time"""
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for sweep in range(n_sweeps):
        if remaining_time() < 3:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            old_val = h[i]
            # Try a few candidate values
            candidates = [0.0, 1.0, 0.5,
                         max(0.0, old_val - 0.1), min(1.0, old_val + 0.1),
                         max(0.0, old_val - 0.05), min(1.0, old_val + 0.05)]
            best_local_c5 = best_c5
            best_local_h = h.copy()
            
            for nv in candidates:
                if abs(nv - old_val) < 1e-8:
                    continue
                h[i] = nv
                h_feas = project_feasible(h, n)
                c5_test = compute_c5(h_feas)
                if c5_test < best_local_c5:
                    best_local_c5 = c5_test
                    best_local_h = h_feas.copy()
            
            h = best_local_h.copy()
            if best_local_c5 < best_c5:
                best_c5 = best_local_c5
                any_imp = True
        
        if not any_imp:
            break
    
    return h, best_c5
