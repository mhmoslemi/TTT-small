        diff2 = target_sum - np.sum(h)
        if abs(diff2) < 1e-15:
            break
        mask2 = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask2) > 0:
            h[mask2] += diff2 / np.sum(mask2)
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_max_value_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    h_clipped = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h_clipped, 1.0 - h_clipped, mode='full') * dx

    if top_k is not None and top_k < len(corr):
        idx = np.argsort(corr)[-top_k:]
        m = corr[idx].max()
        shifted = corr[idx] - m
        w = np.exp(beta * shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.zeros(len(idx))
            w[-1] = 1.0
        else:
            w /= w_sum
        smax = m + np.log(w_sum) / beta
    else:
        m = np.max(corr)
        shifted = corr - m
        w = np.exp(beta * shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.zeros(len(corr))
            w[np.argmax(corr)] = 1.0
        else:
            w /= w_sum
        smax = m + np.log(w_sum) / beta

    grad = np.zeros(n)
    if top_k is not None and top_k < len(corr):
        work_idx = idx
        work_w = w
    else:
        work_idx = np.arange(len(corr))
        work_w = w

    for t in range(len(work_idx)):
        if work_w[t] < 1e-14:
            continue
        k = work_idx[t] - (n - 1)
        if k >= 0:
            i_lo = k
            i_hi = n - 1
        else:
            i_lo = 0
            i_hi = n - 1 + k
        if i_lo > i_hi:
            continue
        m_vals = np.arange(i_lo, i_hi + 1)
        j_vals = m_vals - k
        grad[m_vals] += work_w[t] * (1.0 - h_clipped[j_vals])
        grad[j_vals] += work_w[t] * (-h_clipped[m_vals])

    return smax, grad * dx


def run_opt_lbfgs(h0, n, target, beta, max_time):
    t_start = time.time()
    dx = 2.0 / n
    pw = 1000.0 + beta * 0.5

    def obj(params):
        if time.time() - t_start > max_time:
            return np.inf
        h = np.clip(params, 0.0, 1.0)
        sum_err = np.sum(h) - target
        val, _ = smooth_max_value_grad(h, beta, top_k=25)
        return val + pw * sum_err * sum_err

    def obj_grad(params):
        h = np.clip(params, 0.0, 1.0)
        sum_err = np.sum(h) - target
        val, g = smooth_max_value_grad(h, beta, top_k=25)
        g = g + 2.0 * pw * sum_err
        return val + pw * sum_err * sum_err, g

    try:
        res = minimize(obj_grad, h0, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       jac=True,
                       options={'maxiter': 600, 'ftol': 1e-22, 'gtol': 1e-18,
                                'maxls': 60, 'maxfun': 8000, 'iprint': 0})
        h_new = project_feasible(res.x, target)
    except Exception:
        h_new = project_feasible(h0, target)
    return h_new


def run_opt_slsqp(h0, n, target, beta, max_time):
    t_start = time.time()
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]

    def obj(x):
        if time.time() - t_start > max_time:
            return 1e6
        h = np.clip(x, 0.0, 1.0)
        val, _ = smooth_max_value_grad(h, beta, top_k=20)
        return val

    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, maxiter=500, ftol=1e-20,
                       options={'iprint': 0, 'maxiter': 500})
        h_new = project_feasible(res.x, target)
    except Exception:
        h_new = project_feasible(h0, target)
    return h_new


def coord_descent(h0, n, target, best_c5, max_time, rng):
    t_start = time.time()
    h = h0.copy()
    current_c5 = compute_c5(h)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    no_improve = 0

    while time.time() - t_start < max_time and no_improve < 12:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() - t_start > max_time - 1:
                break
            for d in deltas:
                for j in rng.choice(n, size=min(8, n), replace=False):
                    if j == i:
                        continue
                    for sign in [1, -1]:
                        di = sign * d
                        dj = -sign * d
                        if h[i] + di < -1e-10 or h[i] + di > 1.0 + 1e-10:
                            continue
                        if h[j] + dj < -1e-10 or h[j] + dj > 1.0 + 1e-10:
                            continue
                        h[i] += di
                        h[j] += dj
                        c5_test = compute_c5(h)
                        if c5_test < best_c5 - 1e-12:
                            best_c5 = c5_test
                            current_c5 = c5_test
                            improved = True
                            break
                        h[i] -= di
                        h[j] -= dj
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            h = project_feasible(h, target)
    return h, best_c5


def momentum_subgradient(h0, n, target, best_c5, max_time):
    t_start = time.time()
    h = h0.copy()
    moment = np.zeros(n)
    alpha = 0.02
    decay = 0.85

    while time.time() - t_start < max_time:
        g, _ = smooth_max_value_grad(h, beta=200, top_k=15)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g /= gnorm
        moment = decay * moment - alpha * g
        h_trial = h + moment
        h_trial = project_feasible(h_trial, target)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
        h = h_trial
        if (int((time.time() - t_start) * 2)) % 50 == 0:
            alpha *= 0.95
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    def rem():
        return max(0.0, budget_s - (time.time() - t0) - 5.0)

    rng = np.random.RandomState(seed)

    if 'initial_h_values' in dir() or True:
        try:
            h_par = np.array(initial_h_values, dtype=float)
        except NameError:
            n_init = 160
            h_par = np.ones(n_init) * 0.5
            h_par = project_feasible(h_par, n_init / 2.0)

    n_par = len(h_par)
    c5_par = compute_c5(h_par)
    print("Start: n={}, C5={:.7f}".format(n_par, c5_par))

    best_h = h_par.copy()
    best_c5 = c5_par
    best_n = n_par

    # Phase 1: Optimize at parent resolution
    if rem() > 40:
        target = n_par / 2.0
        h = best_h.copy()
        for beta in [200, 500, 1500, 4000, 10000, 25000]:
            if rem() < 5:
                break
            tm = min(rem(), 35.0)
            h = run_opt_lbfgs(h, n_par, target, beta, tm)
            c5_test = compute_c5(h)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h.copy()
                print("  Phase1 n={}, beta={}: C5={:.7f}".format(n_par, beta, c5_test))
        if rem() > 15:
            h, best_c5 = coord_descent(best_h, n_par, n_par / 2.0, best_c5, min(rem(), 30), rng)
            if best_c5 < c5_par:
                best_h = h.copy()
                best_n = n_par
                print("  Phase1 coord: C5={:.7f}".format(best_c5))

    # Phase 2: Multi-resolution optimization
    n_candidates = []
    base = best_n
    for delta in [0, 20, 40, 60, 100, 140, 200, 280, 360]:
        nn = base + delta
        if nn % 2 != 0:
            nn += 1
        if 100 <= nn <= 600:
            n_candidates.append(nn)
    n_candidates = sorted(set(n_candidates))

    for target_n in n_candidates:
        if rem() < 25:
            break
        target_sum = target_n / 2.0

        xp = (np.arange(best_n) + 0.5) * 2.0 / best_n
        xn = (np.arange(target_n) + 0.5) * 2.0 / target_n
        h = np.interp(xn, xp, best_h)
        h = project_feasible(h, target_sum)

        for beta in [300, 800, 2000, 6000, 15000, 40000]:
            if rem() < 5:
                break
            tm = min(rem() * 0.6 / max(1, 4), 40.0)
            h = run_opt_lbfgs(h, target_n, target_sum, beta, tm)
            c5_test = compute_c5(h)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h.copy()
                best_n = target_n
                print("  Phase2 n={}, beta={}: C5={:.7f}".format(target_n, beta, c5_test))

        if rem() > 20:
            h[0] = h  # ensure it's our best
            h, c5_cd = coord_descent(h, target_n, target_sum, best_c5, min(rem(), 25), rng)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h.copy()
                best_n = target_n
                print("  Phase2 coord n={}: C5={:.7f}".format(target_n, best_c5))

    # Phase 3: SLSQP polish at best resolution
    if rem() > 20:
        target = best_n / 2.0
        h = best_h.copy()
        for beta in [500, 2000, 8000, 20000]:
            if rem() < 5:
                break
            tm = min(rem(), 25.0)
            h = run_opt_slsqp(h, best_n, target, beta, tm)
            c5_test = compute_c5(h)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h.copy()
                print("  Phase3 SLSQP beta={}: C5={:.7f}".format(beta, c5_test))

    # Phase 4: Momentum subgradient
    if rem() > 15:
        target = best_n / 2.0
        h, best_c5 = momentum_subgradient(best_h, best_n, target, best_c5, min(rem(), 30))
        best_h = h.copy()
        print("  Phase4 momentum: C5={:.7f}".format(best_c5))

    # Phase 5: Very fine coordinate descent
    if rem() > 10:
        target = best_n / 2.0
        h = best_h.copy()
        h, best_c5 = coord_descent(h, best_n, target, best_c5, min(rem(), 40), rng)
        if best_c5 < compute_c5(best_h):
            best_h = h.copy()
        print("  Phase5 fine: C5={:.7f}".format(best_c5))

    # Phase 6: Final L-BFGS polish with very high beta
    if rem() > 10:
        target = best_n / 2.0
        h = best_h.copy()
        for beta in [50000, 100000, 200000]:
            if rem() < 4:
                break
            tm = min(rem(), 20.0)
            h = run_opt_lbfgs(h, best_n, target, beta, tm)
            c5_test = compute_c5(h)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h.copy()
                print("  Phase6 beta={}: C5={:.7f}".format(beta, c5_test))

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("FINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
