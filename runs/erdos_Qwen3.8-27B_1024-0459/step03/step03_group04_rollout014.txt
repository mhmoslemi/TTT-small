def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to increase sum. Add to elements < 1, prioritize smallest.
            mask = h < 1.0 - 1e-14
            cnt = mask.sum()
            if cnt == 0:
                break
            add = min(diff / cnt, (1.0 - h[mask]).min())
            h[mask] += add
        else:
            # Need to decrease sum. Reduce elements > 0, prioritize largest.
            mask = h > 1e-14
            cnt = mask.sum()
            if cnt == 0:
                break
            sub = min(-diff / cnt, h[mask].min())
            h[mask] -= sub
    h = np.clip(h, 0.0, 1.0)
    return h

For shift s ≥ 0 (correlation index = n-1+s):
corr_s = sum_{i=0}^{n-1-s} h[i] * (1 - h[i+s])

∂corr_s/∂h[j] for a fixed j:
- Term where h[j] is the "first" factor: i = j, valid when 0 ≤ j ≤ n-1-s
  → contribution: (1 - h[j+s])
- Term where h[j] is the "second" factor: i+s = j → i = j-s, valid when 0 ≤ j-s ≤ n-1-s
  → i.e., s ≤ j ≤ n-1
  → contribution: h[j-s] * (-1)

So: ∂corr_s/∂h[j] = (1-h[j+s]) * [0≤j≤n-1-s] - h[j-s] * [s≤j≤n-1]

Vectorized:
- First part: for j in [0, n-1-s], grad[j] += (1-h[j+s]) = 1 - h[s:s+n-s] = 1 - h[s:n]
  → grad[0:n-s] += (1 - h[s:n])
- Second part: for j in [s, n-1], grad[j] -= h[j-s] = h[0:n-s]
  → grad[s:n] -= h[0:n-s]

For shift s < 0, let t = -s > 0 (correlation index = n-1-t):
corr_{-t} = sum_{i=0}^{n-1-t} h[i+t] * (1-h[i])

∂corr_{-t}/∂h[j]:
- h[j] as "first" factor: i+t = j → i = j-t, valid when 0 ≤ j-t ≤ n-1-t
  → t ≤ j ≤ n-1
  → contribution: (1-h[j-t])
- h[j] as "second" factor: i = j, valid when 0 ≤ j ≤ n-1-t
  → contribution: h[j+t] * (-1)

So: ∂corr_{-t}/∂h[j] = (1-h[j-t]) * [t≤j≤n-1] - h[j+t] * [0≤j≤n-1-t]

Vectorized:
- First part: for j in [t, n-1], grad[j] += (1-h[j-t]) = 1 - h[0:n-t]

- Second part: for j in [0, n-1-t-1], grad[j] -= h[j+t] = h[t:n]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def project_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = mask.sum()
            if cnt == 0:
                break
            add = min(diff / cnt, (1.0 - h[mask]).min())
            h[mask] += add
        else:
            mask = h > 1e-14
            cnt = mask.sum()
            if cnt == 0:
                break
            sub = min(-diff / cnt, h[mask].min())
            h[mask] -= sub
    h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xq = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xq, xp, src)

def find_active_set(corr, n, top_m=20, gap=0.3):
    mx = np.max(corr)
    thresh = mx - gap
    act = np.where(corr >= thresh)[0]
    if len(act) > top_m:
        order = np.argsort(corr[act])
        act = act[order[-top_m:]]
    if len(act) < 3:
        act = np.argsort(corr)[-5:]
    return act

def compute_weighted_gradient(h, corr, active_set, n, beta):
    g = np.zeros(n)
    vals = corr[active_set]
    m = np.max(vals)
    exps = np.exp(beta * (vals - m))
    w = exps / exps.sum()
    for idx, a in enumerate(active_set):
        s = a - (n - 1)
        wt = w[idx]
        if s >= 0:
            g[:n - s] += wt * (1.0 - h[s:])
            g[s:] -= wt * h[:n - s]
        else:
            t = -s
            g[t:] += wt * (1.0 - h[:n - t])
            g[:n - t] -= wt * h[t:]
    return g

def projected_gradient_step(h, n, beta, max_iter=200):

    best_local = h.copy()
    c5_local, corr_local = compute_c5(h)
    total_improved = False
    
    for it in range(max_iter):
        active = find_active_set(corr_local, n, top_m=15, gap=0.2)
        g = compute_weighted_gradient(h, corr_local, active, n, beta)
        gmax = np.max(np.abs(g))
        if gmax < 1e-14:
            break
        
        improved = False
        for scale in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]:
            alpha = scale / gmax
            h_new = h - alpha * g
            h_new = project_h(h_new, n)
            c5_new, corr_new = compute_c5(h_new)
            if c5_new < c5_local - 1e-15:
                h = h_new
                c5_local = c5_new
                corr_local = corr_new
                best_local = h_new.copy()
                improved = True
                total_improved = True
                break
        
        if not improved:
            break
    
    return best_local if total_improved else None

def fourier_perturb(h, n, rng, scale=0.02):
    h_copy = h.copy()
    H = np.fft.rfft(h_copy)
    # Perturb a random subset of frequency components
    nfreq = len(H)
    n_perturb = max(3, nfreq // 8)
    idx = rng.choice(range(1, nfreq), size=n_perturb, replace=False)
    for i in idx:
        H[i] += scale * (rng.randn() + 1j * rng.randn())
        if n - i < nfreq:
            H[n - i] = np.conj(H[i])
    h_new = np.fft.irfft(H, n=n)
    return project_h(h_new, n)

def slsqp_smooth(h, n, beta, active_set, maxiter=200):
    """SLSQP on smoothed active-set objective."""
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        vals = c[active_set]
        mx = np.max(vals)
        exps = np.exp(beta * (vals - mx))
        return (mx + np.log(exps.sum()) / beta) * (2.0 / n)
    
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        vals = c[active_set]
        mx = np.max(vals)
        exps = np.exp(beta * (vals - mx))
        w = exps / exps.sum()
        g = np.zeros(n)
        for idx, a in enumerate(active_set):
            s = a - (n - 1)
            wt = w[idx]
            if s >= 0:
                g[:n - s] += wt * (1.0 - x[s:])
                g[s:] -= wt * x[:n - s]
            else:
                t = -s
                g[t:] += wt * (1.0 - x[:n - t])
                g[:n - t] -= wt * x[t:]
        g *= (2.0 / n)
        return g
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    constraints = [{'type': 'eq', 'fun': con_fun}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_h(res.x, n)
    except Exception:
        return h

                # Quick optimize at new resolution
                h = best_h.copy()
                _, corr_r = compute_c5(h)
                active_r = find_active_set(corr_r, best_n, top_m=15, gap=0.3)
                for beta_r in [100, 500]:
                    h_new = projected_gradient_step(h, best_n, beta_r, max_iter=50)
                    if h_new is not None:
                        c5_r, _ = compute_c5(h_new)
                        if c5_r < best_c5 - 1e-14:
                            best_c5 = c5_r
                            best_h = h_new.copy()
    
    # Phase 6: Very fine final steps
    if remaining() > 15:
        n = best_n
        h = best_h.copy()
        for _ in range(3):
            if remaining() < 10:
                break
            c5_cur, corr_cur = compute_c5(h)
            active = find_active_set(corr_cur, n, top_m=10, gap=0.05)
            h_new = slsqp_smooth(h, n, 5000, active, maxiter=100)
            c5_new, _ = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
    
    # Final evaluation
    best_h = project_h(best_h, best_n)
    final_c5, _ = compute_c5(best_h)
    
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
