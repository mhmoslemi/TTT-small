def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)

            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            cnt = int(mask.sum())
            if cnt == 0: break
            h[mask] += diff / cnt
        else:

            if cnt == 0: break
            h[mask] += diff / cnt
