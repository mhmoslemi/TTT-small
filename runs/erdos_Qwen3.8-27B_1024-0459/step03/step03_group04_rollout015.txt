def obj_and_grad(h, n, beta, active_indices):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr[active_indices]
    m = np.max(vals)
    ex = np.exp(beta * (vals - m))
    w = ex / ex.sum()
    obj = (m + np.log(ex.sum()) / beta) * dx
    
    # Gradient
    grad = np.zeros(n)
    for idx, k in enumerate(active_indices):
        wi = w[idx]
        if wi < 1e-10:
            continue
        s = k - (n - 1)  # actual shift
        if s >= 0:
            # (1-h[j+s]) for j in [0, n-s)
            if n - s > 0:
                grad[:n-s] += wi * (1.0 - h[s:n])
            # -h[j-s] for j in [s, n)
            if n - s > 0:
                grad[s:] -= wi * h[:n-s]
        else:
            m2 = -s  # positive
            # (1-h[j-m2]) for j in [m2, n)
            if n - m2 > 0:
                grad[m2:] += wi * (1.0 - h[:n-m2])
            # -h[j+m2] for j in [0, n-m2)
            if n - m2 > 0:
                grad[:n-m2] -= wi * h[m2:n]
    
    grad *= dx
    return obj, grad

One more idea: instead of just smooth max, I can use a "sum of top-k" objective, which is more robust:
f(h) = dx * sum of top-k values of corr

This can be written as:
f(h) = dx * min_t { t + sum_k max(corr[k] - t, 0) / (k-1) ... }

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = 0
        v = i
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return h

def make_balanced_random(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    half = n // 2
    h[idx[:half]] = 1.0
    if n % 2 == 1:

    return h
