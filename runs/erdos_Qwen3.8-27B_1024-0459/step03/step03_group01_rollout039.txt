    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_c5 = cur_max * dx

    start = time.time()
    stall = 0
    max_stall = 150
    temp = initial_temp if initial_temp and initial_temp > 0 else 0.0
    if temp > 0:
        min_temp = 0.05 * cur_max
        cool_rate = 0.9995

    while remaining() > 5 and (time.time() - start) < time_limit:
        k_max = int(np.argmax(corr))
        offset = k_max - (n - 1)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_delta = 0
        best_swap = None
        best_new_corr = None

        # Targeted candidates based on worst shift
        if offset > 0:
            bad_i = np.array([i for i in range(n - offset) if h[i] == 1 and h[i + offset] == 0])
        elif offset < 0:
            d = -offset
            bad_i = np.array([i for i in range(d, n) if h[i] == 1 and h[i - d] == 0])
        else:
            bad_i = np.array([], dtype=int)

        # Try targeted swaps
        n_bad = len(bad_i)
        if n_bad > 0:
            n_try = min(n_bad, 30)
            chosen = rng.choice(bad_i, size=n_try, replace=False) if n_try < n_bad else bad_i
            for i in chosen:
                # Priority 1: j = i + offset (complementary position)
                if offset > 0 and i + offset < n and h[i + offset] == 0:
                    j = i + offset
                    h[i] = 0; h[j] = 1
                    nc = compute_corr(h)
                    nm = float(np.max(nc))
                    h[i] = 1; h[j] = 0
                    if nm < cur_max - 1e-12:
                        if nm - cur_max < best_delta:
                            best_delta = nm - cur_max
                            best_swap = (i, j)
                            best_new_corr = nc
                # Priority 2: j = i - offset
                if offset < 0 and i + offset >= 0 and h[i + offset] == 0:
                    j = i + offset
                    h[i] = 0; h[j] = 1
                    nc = compute_corr(h)
                    nm = float(np.max(nc))
                    h[i] = 1; h[j] = 0
                    if nm < cur_max - 1e-12:
                        if nm - cur_max < best_delta:
                            best_delta = nm - cur_max
                            best_swap = (i, j)
                            best_new_corr = nc
                # Priority 3: random zeros
                n_z = min(4, len(zeros))
                zc = rng.choice(zeros, size=n_z, replace=False)
                for j in zc:
                    if j == i:
                        continue
                    h[i] = 0; h[j] = 1
                    nc = compute_corr(h)
                    nm = float(np.max(nc))
                    h[i] = 1; h[j] = 0
                    if nm < cur_max - 1e-12:
                        if nm - cur_max < best_delta:
                            best_delta = nm - cur_max
                            best_swap = (i, j)
                            best_new_corr = nc
        else:
            # No targeted candidates, try random
            n_try = 20
            for _ in range(n_try):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0; h[j] = 1
                nc = compute_corr(h)
                nm = float(np.max(nc))
                h[i] = 1; h[j] = 0
                if nm < cur_max - 1e-12:
                    if nm - cur_max < best_delta:
                        best_delta = nm - cur_max
                        best_swap = (i, j)
                        best_new_corr = nc

        if best_swap is not None:
            i, j = best_swap
            h[i] = 0; h[j] = 1
            corr = best_new_corr
            cur_max = float(np.max(corr))
            stall = 0
            if cur_max * dx < best_c5:
                best_c5 = cur_max * dx
                best_h = h.copy()
            if temp > 0:
                temp = max(min_temp, temp * cool_rate)
        else:
            stall += 1
            if stall >= max_stall:
                if temp > 0:
                    # Simulated annealing move
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i] = 0; h[j] = 1
                    nc = compute_corr(h)
                    nm = float(np.max(nc))
                    h[i] = 1; h[j] = 0
                    delta = nm - cur_max
                    if delta < 0 or rng.rand() < np.exp(-delta / max(temp, 1e-10)):
                        h[i] = 0; h[j] = 1
                        corr = nc
                        cur_max = nm
                        stall = 0
                    temp = max(min_temp, temp * cool_rate)
                else:
                    # Aggressive perturbation: multiple random swaps
                    n_pert = rng.randint(2, 6)
                    for _ in range(n_pert):
                        ones = np.where(h == 1)[0]
                        zeros = np.where(h == 0)[0]
                        if len(ones) == 0 or len(zeros) == 0:
                            break
                        i = ones[rng.randint(len(ones))]
                        j = zeros[rng.randint(len(zeros))]
                        h[i] = 0; h[j] = 1
                    corr = compute_corr(h)
                    cur_max = float(np.max(corr))
                    stall = 0

    return best_h, best_c5


def intensive_local_search(h, n, rng, time_limit):
    """More intensive search with multiple worst-shift targeting."""
    h = h.copy()
    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_c5 = cur_max * dx
    start = time.time()
    stall = 0

    while remaining() > 5 and (time.time() - start) < time_limit:
        # Try to reduce top 3 shifts
        top3 = np.argsort(-corr)[:3]
        found = False
        for k_idx in top3:
            if found:
                break
            k_val = int(k_idx)
            offset = k_val - (n - 1)
            if offset == 0:
                continue
            if offset > 0:
                bad_i = np.array([i for i in range(n - offset) if h[i] == 1 and h[i + offset] == 0])
            else:
                d = -offset
                bad_i = np.array([i for i in range(d, n) if h[i] == 1 and h[i - d] == 0])
            if len(bad_i) == 0:
                continue
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            n_try = min(len(bad_i), 20)
            chosen = rng.choice(bad_i, size=n_try, replace=False) if n_try < len(bad_i) else bad_i
            for i in chosen:
                # Try complementary and a few random
                candidate_js = []
                if offset > 0 and i + offset < n:
                    candidate_js.append(i + offset)
                if offset < 0 and i + offset >= 0:
                    candidate_js.append(i + offset)
                nz = min(3, len(zeros))
                candidate_js.extend(rng.choice(zeros, size=nz, replace=False).tolist())

                for j in candidate_js:
                    if j == i or h[j] != 0:
                        continue
                    h[i] = 0; h[j] = 1
                    nc = compute_corr(h)
                    nm = float(np.max(nc))
                    h[i] = 1; h[j] = 0
                    if nm < cur_max - 1e-12:
                        h[i] = 0; h[j] = 1
                        corr = nc
                        cur_max = nm
                        if cur_max * dx < best_c5:
                            best_c5 = cur_max * dx
                            best_h = h.copy()
                        stall = 0
                        found = True
                        break
                if found:
                    break

        if not found:
            stall += 1
            if stall > 100:
                # Perturb
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                for _ in range(rng.randint(1, 4)):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i] = 0; h[j] = 1
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                corr = compute_corr(h)
                cur_max = float(np.max(corr))
                stall = 0
                if cur_max * dx > best_c5 * 1.5:
                    h = best_h.copy()
                    corr = compute_corr(h)
                    cur_max = float(np.max(corr))

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 320
        parent_h = thue_morse(parent_n, seed)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Binary local search from parent
    h_bin = to_binary(best_h, best_n, rng)
    c5_start = (time.time() - _START)

    if remaining() > 30:
        tl = min(120, remaining() - 20)
        h_b, c5_b = binary_local_search(h_bin, best_n, rng, tl)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = best_n

    # Intensive search
    if remaining() > 30:
        tl = min(150, remaining() - 20)
        h_i, c5_i = intensive_local_search(best_h, best_n, rng, tl)
        if c5_i < best_c5:
            best_c5 = c5_i
            best_h = h_i.copy()

    # Phase 2: Multi-resolution
    n_candidates = [256, 280, 320, 352, 384, 416, 448, 480, 512]
    n_candidates = [nn for nn in n_candidates if nn != best_n and nn >= 128 and nn <= 600]
    rng.shuffle(n_candidates)

    for n_t in n_candidates:
        if remaining() < 40:
            break

        h_interp = interp_to(best_h, best_n, n_t, rng)
        c5_interp = compute_c5(h_interp)

        # Local search at this resolution
        tl = min(80, remaining() - 20)
        h_b, c5_b = binary_local_search(h_interp, n_t, rng, tl)
        if c5_b < c5_interp:
            c5_interp = c5_b
            h_interp = h_b

        # Intensive
        if remaining() > 30:
            tl2 = min(60, remaining() - 15)
            h_i, c5_i = intensive_local_search(h_interp, n_t, rng, tl2)
            if c5_i < c5_interp:
                c5_interp = c5_i
                h_interp = h_i

        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n_t

    # Phase 3: Multi-start diversification at best_n
    best_h_bin = to_binary(best_h, best_n, rng)
    n_starts = 3
    for s in range(n_starts):
        if remaining() < 40:
            break

        if s == 0:
            h_start = best_h_bin.copy()
            # Perturb a bit
            ones = np.where(h_start == 1)[0]
            zeros = np.where(h_start == 0)[0]
            for _ in range(rng.randint(3, 8)):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_start[i] = 0; h_start[j] = 1
                ones = np.where(h_start == 1)[0]
                zeros = np.where(h_start == 0)[0]
        elif s == 1:
            h_start = thue_morse(best_n, seed + s * 100)
        else:
            h_start = np.zeros(best_n)
            idx = rng.choice(best_n, best_n // 2, replace=False)
            h_start[idx] = 1.0

        tl = min(100, remaining() - 15)
        h_b, c5_b = binary_local_search(h_start, best_n, rng, tl)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = best_n

    # Phase 4: Final intensive search
    if remaining() > 20:
        best_h_bin = to_binary(best_h, best_n, rng)
        tl = min(remaining() - 10, 200)
        h_i, c5_i = intensive_local_search(best_h_bin, best_n, rng, tl)
        if c5_i < best_c5:
            best_c5 = c5_i
            best_h = h_i.copy()

    # Phase 5: Simulated annealing restart
    if remaining() > 30:
        best_h_bin = to_binary(best_h, best_n, rng)
        corr = compute_corr(best_h_bin)
        cur_max = float(np.max(corr))
        tl = min(remaining() - 15, 150)
        h_sa, c5_sa = binary_local_search(best_h_bin, best_n, rng, tl,
                                          use_sa=True, initial_temp=cur_max * 0.5)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Final: one more intensive pass
    if remaining() > 10:
        best_h_bin = to_binary(best_h, best_n, rng)
        tl = max(5, remaining() - 5)
        h_f, c5_f = intensive_local_search(best_h_bin, best_n, rng, tl)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Validate
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
