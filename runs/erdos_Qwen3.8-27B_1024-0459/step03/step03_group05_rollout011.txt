def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            if diff > 0:
                idx = np.argsort(h)
                for i in idx:
                    room = 1.0 - h[i]
                    if room >= diff - 1e-15:
                        h[i] += diff
                        diff = 0.0
                        break
                    else:
                        h[i] += room
                        diff -= room
            else:
                idx = np.argsort(-h)
                for i in idx:
                    room = h[i]

                        h[i] += diff
                        diff = 0.0
                        break
                    else:

                        diff += room

def smooth_obj(h, target, beta, top_k, pen_weight):
    n = len(h)

    return val

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = np.argsort(h)
                for i in idx:
                    room = 1.0 - h[i]
                    if room >= diff - 1e-15:
                        h[i] += diff
                        diff = 0.0
                        break
                    else:
                        h[i] += room
                        diff -= room
            else:
                idx = np.argsort(-h)
                for i in idx:
                    room = h[i]
                    if room >= -diff - 1e-15:
                        h[i] += diff
                        diff = 0.0
                        break
                    else:
                        h[i] -= room
                        diff += room
        h = np.clip(h, 0.0, 1.0)
    return h

    if top_k is not None and 0 < top_k < len(corr):
