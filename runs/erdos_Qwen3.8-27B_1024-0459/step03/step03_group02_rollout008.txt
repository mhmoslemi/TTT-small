def lag_gradient(h, n, offset):
    grad = np.zeros(n)
    if offset >= 0:

        if npairs <= 0:
            return grad
        grad[offset:offset + npairs] += (1.0 - h[:npairs])
        grad[:npairs] -= h[offset:offset + npairs]
    else:
        abs_off = -offset
        npairs = n - abs_off
        if npairs <= 0:
            return grad
        grad[:npairs] += (1.0 - h[abs_off:abs_off + npairs])
        grad[abs_off:abs_off + npairs] -= h[:npairs]
    return grad

    h = best_h.copy()
    h, c5_this = surgical_top_lag(h, n, max_steps=3000)
    if c5_this < best_c5:
        best_c5 = c5_this
        best_h = h.copy()

    # Phase 3: Top-lag QP refinement (focus on specific lags)
    if remaining_time() > 60:
        h = best_h.copy()
        h, c5_this = top_lag_qp_refine(h, n, max_iters=200)
        if c5_this < best_c5:
            best_c5 = c5_this
            best_h = h.copy()

    # Phase 4: Try perturbation + SLSQP
    for attempt in range(8):
        if remaining_time() < 40:
            break
        noise_scale = 0.02 / (attempt + 1)
        h_pert = best_h + rng.randn(best_n) * noise_scale
        h_pert = project_feasible(h_pert, best_n)
        c5_pert = compute_c5(h_pert)
        h_try = slsqp_refine(h_pert, best_n, 30000, 300)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Phase 5: Try different resolutions
    for trial_n in [120, 140, 200, 240, 320]:
        if remaining_time() < 50:
            break
        h_interp = interp_to(best_h, best_n, trial_n)
        h_interp = project_feasible(h_interp, trial_n)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5 + 0.002:
            h_try = slsqp_refine(h_interp, trial_n, 20000, 500)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = trial_n
                # Polish at this resolution
                h_try2 = slsqp_refine(best_h, trial_n, 50000, 300)
                c5_try2 = compute_c5(h_try2)
                if c5_try2 < best_c5:
                    best_c5 = c5_try2
                    best_h = h_try2.copy()

    # Phase 6: Combine top-2/3 lags minimization
    if remaining_time() > 80:
        n_cur = best_n
        h = best_h.copy()
        corr = compute_corr(h)
        sorted_lags = np.argsort(corr)[::-1]
        # Minimize sum of top 3 with SLSQP
        for topk in [2, 3, 5]:
            if remaining_time() < 30:
                break
            k_list = sorted_lags[:topk]
            global _N_GLOBAL
            _N_GLOBAL = n_cur
            def obj_topk(x, kl=k_list):
                c = np.correlate(x, 1.0 - x, mode='full')
                return np.sum(c[kl])
            def obj_topk_grad(x, kl=k_list):
                g = np.zeros(n_cur)
                for ki in kl:
                    g += lag_gradient(x, n_cur, ki - (n_cur - 1))
                return g
            bounds = [(0.0, 1.0)] * n_cur
            cons = [{'type': 'eq', 'fun': constraint_sum}]
            try:
                res = minimize(obj_topk, h, jac=obj_topk_grad,
                              method='SLSQP', bounds=bounds, constraints=cons,
                              options={'maxiter': 200, 'ftol': 1e-15, 'iprint': 0})
                h_new = project_feasible(res.x, n_cur)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
            except Exception:
                pass

    # Phase 7: Final ultra-fine polish
    if remaining_time() > 20:
        n_cur = best_n
        h = best_h.copy()
        for beta in [50000, 100000, 200000]:
            if remaining_time() < 10:
                break
            h_new = slsqp_refine(h, n_cur, beta, 200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 8: Final surgical pass
    if remaining_time() > 10:
        h, c5_this = surgical_top_lag(best_h, best_n, max_steps=500)
        if c5_this < best_c5:
            best_c5 = c5_this
            best_h = h.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
