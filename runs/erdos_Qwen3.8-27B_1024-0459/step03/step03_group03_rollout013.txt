    # Phase 7: Perturbation escapes
    if remaining() > 40:
        for attempt in range(5):
            if remaining() < 20:
                break
            for amp in [0.05, 0.03, 0.08]:
                p = _sin_perturb(best_h, n, rng, amp)
                h2, c2 = _nesterov(p, n, 200, 0.02, 0.9)
                if c2 < best_c5:
                    best_h, best_c5 = h2, c2
            for amp in [0.04, 0.06, 0.02]:
                p = _group_perturb(best_h, n, rng, amp)
                h2, c2 = _nesterov(p, n, 150, 0.02, 0.9)
                if c2 < best_c5:
                    best_h, best_c5 = h2, c2

    # Phase 8: Final local refine
    h, c5 = _local_refine(best_h, n, rng, 200)
    if c5 < best_c5:
        best_h, best_c5 = h.copy(), c5

    # Phase 9: Final SLSQP
    if remaining() > 15:
        h2 = _slsqp(best_h, n, 8000, 400)
        c2 = _c5(h2)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2
        h, c5 = _nesterov(best_h, n, 300, 0.008, 0.94)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    # Phase 10: Final swap polish
    h, c5 = _swap_opt(best_h, n, rng, 30)
    if c5 < best_c5:
        best_h, best_c5 = h.copy(), c5

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _t0, _tb
    _t0 = time.time()
    _tb = budget_s
    rng = np.random.RandomState(seed)

    try:
        hpar = np.array(initial_h_values, dtype=np.float64)
        npar = len(hpar)
    except Exception:
        npar = 160
        hpar = np.zeros(npar, dtype=np.float64)
        hpar[:npar // 2] = 1.0

    bh = _proj(hpar, npar)
    bc5 = _c5(bh)
    bn = npar
    print("Init n={} C5={:.10f}".format(bn, bc5))

    # Optimize at parent resolution
    t_rem = _tl()
    alloc1 = min(250.0, t_rem * 0.2)
    h, c5 = _optimize_res(bh, npar, rng, alloc1)
    if c5 < bc5:
        bh, bc5, bn = h, c5, npar
        print("N={} C5={:.10f}".format(npar, c5))

    # Multi-resolution search
    res_list = [npar + 20, npar + 40, npar + 80, npar - 20, npar + 60]
    for nn in res_list:
        if _tl() < 80:
            break
        if nn < 40 or nn % 2 != 0:
            continue
        hi = _interp(bh, bn, nn)
        hi = _proj(hi, nn)
        alloc = min(300.0, _tl() * 0.3)
        h2, c2 = _optimize_res(hi, nn, rng, alloc)
        if c2 < bc5:
            bh, bc5, bn = h2, c2, nn
            print("N={} C5={:.10f}".format(nn, c2))

    # Final polish at best resolution
    if _tl() > 30:
        h, c5 = _nesterov(bh, bn, 800, 0.008, 0.94)
        if c5 < bc5:
            bh, bc5 = h.copy(), c5
    if _tl() > 25:
        h, c5 = _swap_opt(bh, bn, rng, 50)
        if c5 < bc5:
            bh, bc5 = h.copy(), c5
    if _tl() > 40:
        h2 = _slsqp(bh, bn, 10000, 500)
        c2 = _c5(h2)
        if c2 < bc5:
            bh, bc5 = h2, c2
    if _tl() > 20:
        h, c5 = _local_refine(bh, bn, rng, 200)
        if c5 < bc5:
            bh, bc5 = h.copy(), c5
    if _tl() > 15:
        h2 = _slsqp(bh, bn, 15000, 400)
        c2 = _c5(h2)
        if c2 < bc5:
            bh, bc5 = h2, c2
        h, c5 = _nesterov(bh, bn, 300, 0.005, 0.95)
        if c5 < bc5:
            bh, bc5 = h.copy(), c5

    bh = _proj(bh, bn)
    fc5 = _c5(bh)
    try:
        ec = evaluate_erdos_solution(bh)
        if ec < fc5:
            fc5 = ec
    except Exception:
        pass
    print("FINAL n={} C5={:.10f}".format(bn, fc5))
    return (bh, fc5, bn)
