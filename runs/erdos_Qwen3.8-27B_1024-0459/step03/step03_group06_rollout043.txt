- Let b(m) = 1 - h[m] (the "complement")
- Then sum_s w(s) * (1-h[m-s]) is a convolution of w and b
- And sum_s w(s) * h[m+s] is a (cross-)correlation of w and h

So the gradient can be computed in O(n log n) via FFT!

def compute_swap_matrices_v2(h, n):

    s = np.arange(-(n-1), n)  # (2n-1,)

    im_s = i_idx - s[None, :]  # (n, 2n-1)
    
    valid_ip = (ip_s >= 0) & (ip_s < n)
    valid_im = (im_s >= 0) & (im_s < n)
    
    # Clamp for safe indexing
    ip_clamped = np.clip(ip_s, 0, n-1)
    im_clamped = np.clip(im_s, 0, n-1)
    
    h_ip = h[ip_clamped]  # (n, 2n-1)
    h_im = h[im_clamped]  # (n, 2n-1)
    
    remove = np.where(valid_ip, -(1.0 - h_ip), 0.0) + np.where(valid_im, h_im, 0.0)
    add = np.where(valid_ip, (1.0 - h_ip), 0.0) + np.where(valid_im, -h_im, 0.0)
    
    return remove, add

For each i, this is O(n²) (80×319 matrix ops). For all i (80 of them), it's O(n³) = 80×80×319 ≈ 2M operations. With NumPy, this should run in < 0.1 seconds per round.

We can do 100+ rounds in a few seconds!

Now, one issue: after each swap, the `remove` and `add` matrices change (because h changed). So we need to recompute them each round. That's O(n²) per round, which is fine.

Actually wait - we only need to recompute the rows/cols that changed. When we swap h[i]: 1→0 and h[j]: 0→1, only remove[i], add[i], remove[j], add[j] change (since those are the only positions whose h value changed). The other rows stay the same.

But for simplicity, let's just recompute the whole matrix each round. O(n²) is cheap.

So the DERIVATIVE (for infinitesimal change) is:
∂C(s)/∂h[m] = (1-h[m+s]) - h[m-s]

And my "add" matrix is exactly this: add[m, s] = (1-h[m+s]) - h[m-s] = ∂C(s)/∂h[m].

So: grad = (add.T @ w) * dx + penalty_grad

This is a beautiful result! The gradient of the smooth max objective can be computed as a matrix-vector product: grad = add.T @ w, which is O(n²) and fully vectorized with NumPy.

def smooth_max_grad_fast(h, n, beta, w=None):

    m = np.max(corr)

import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0
_G_N = 0
_G_BETA = 100.0
_G_PEN = 1000.0


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _START) - 2.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def compute_swap_matrices(h, n):
    num_shifts = 2 * n - 1
    s = np.arange(-(n - 1), n)
    i_idx = np.arange(n)[:, None]
    ip_s = i_idx + s[None, :]
    im_s = i_idx - s[None, :]
    valid_ip = (ip_s >= 0) & (ip_s < n)
    valid_im = (im_s >= 0) & (im_s < n)
    ip_clamped = np.clip(ip_s, 0, n - 1)
    im_clamped = np.clip(im_s, 0, n - 1)
    h_ip = h[ip_clamped]
    h_im = h[im_clamped]
    remove = np.where(valid_ip, -(1.0 - h_ip), 0.0) + np.where(valid_im, h_im, 0.0)
    add = np.where(valid_ip, (1.0 - h_ip), 0.0) + np.where(valid_im, -h_im, 0.0)
    return remove, add


def ensure_binary_count(h, n):
    h = np.where(h > 0.5, 1.0, 0.0).copy()
    target = n // 2
    num_ones = int(np.sum(h))
    if num_ones > target:
        ones = np.where(h == 1.0)[0]
        h[ones[:num_ones - target]] = 0.0
    elif num_ones < target:
        zeros = np.where(h == 0.0)[0]
        h[zeros[:target - num_ones]] = 1.0
    return h


def binary_hill_climb(h, n, max_rounds, rng, time_limit=None):
    h = ensure_binary_count(h, n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = np.max(corr)

    for rnd in range(max_rounds):
        if remaining_time() < 3:
            break
        if time_limit is not None and (time.time() - time_limit) > _BUDGET * 0.8:
            break

        remove, add = compute_swap_matrices(h, n)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

        best_new_max = current_max
        best_i = -1
        best_j = -1

        for i in ones:
            base = corr + remove[i]
            new_maxs = np.max(base[None, :] + add[zeros, :], axis=1)
            min_idx = np.argmin(new_maxs)
            if new_maxs[min_idx] < best_new_max - 1e-15:
                best_new_max = new_maxs[min_idx]
                best_i = i
                best_j = zeros[min_idx]

        if best_i == -1:
            break

        h[best_i] = 0.0
        h[best_j] = 1.0
        corr = np.correlate(h, 1.0 - h, mode='full')
        current_max = np.max(corr)

    return h, current_max


def iterated_local_search(h_init, n, n_restarts, max_rounds, rng):
    best_h = ensure_binary_count(h_init, n).copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    best_c5_raw = np.max(corr)

    for r in range(n_restarts):
        if remaining_time() < 5:
            break
        h, c5_raw = binary_hill_climb(best_h, n, max_rounds, rng)
        if c5_raw < best_c5_raw:
            best_c5_raw = c5_raw
            best_h = h.copy()

        if remaining_time() < 5 or r == n_restarts - 1:
            break

        k = rng.randint(3, 10)
        h_pert = best_h.copy()
        for _ in range(k):
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_pert[i] = 0.0
            h_pert[j] = 1.0

        h, c5_raw = binary_hill_climb(h_pert, n, max_rounds, rng)
        if c5_raw < best_c5_raw:
            best_c5_raw = c5_raw
            best_h = h.copy()

    return best_h, best_c5_raw


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    return ensure_binary_count(h, n)


def smooth_max_obj(x):
    n = _G_N
    dx = 2.0 / n
    beta = _G_BETA
    pen = _G_PEN
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sm = (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx
    penalty = pen * (np.sum(x) - n / 2.0) ** 2
    return sm + penalty


def smooth_max_grad(x):
    n = _G_N
    dx = 2.0 / n
    beta = _G_BETA
    pen = _G_PEN
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w /= np.sum(w)
    _, add = compute_swap_matrices(x, n)
    grad = (add.T @ w) * dx
    grad += 2.0 * pen * (np.sum(x) - n / 2.0)
    return grad


def topk_smooth_obj(x, k):
    n = _G_N
    dx = 2.0 / n
    beta = 100.0
    pen = _G_PEN
    corr = np.correlate(x, 1.0 - x, mode='full')
    top_idx = np.argsort(corr)[-k:]
    tv = corr[top_idx]
    m = np.max(tv)
    sm = (m + np.log(np.sum(np.exp(beta * (tv - m)))) / beta) * dx
    penalty = pen * (np.sum(x) - n / 2.0) ** 2
    return sm + penalty


def topk_smooth_grad(x, k):
    n = _G_N
    dx = 2.0 / n
    beta = 100.0
    pen = _G_PEN
    corr = np.correlate(x, 1.0 - x, mode='full')
    top_idx = np.argsort(corr)[-k:]
    w_full = np.zeros(2 * n - 1)
    tv = corr[top_idx]
    m = np.max(tv)
    w_top = np.exp(beta * (tv - m))
    w_top /= np.sum(w_top)
    w_full[top_idx] = w_top
    _, add = compute_swap_matrices(x, n)
    grad = (add.T @ w_full) * dx
    grad += 2.0 * pen * (np.sum(x) - n / 2.0)
    return grad
