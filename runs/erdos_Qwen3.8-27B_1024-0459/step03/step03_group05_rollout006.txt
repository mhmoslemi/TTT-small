def project_feasible(h, target=None):
    h = np.asarray(h, dtype=float).copy()
    if h.size == 0:
        return h
    if target is None:
        target = h.size / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.any():
            h[mask] += diff / float(np.sum(mask))
            h = np.clip(h, 0.0, 1.0)
        else:
            break
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-12:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            room = 1.0 - h[idx]
            if room > 0.0:
                h[idx] = min(1.0, h[idx] + min(diff, room))
        else:
            idx = int(np.argmax(h))
            room = h[idx]
            if room > 0.0:
                h[idx] = max(0.0, h[idx] + max(diff, -room))
    return h


def fix_sum_exact(h):
    h = np.asarray(h, dtype=float).copy()
    target = h.size / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(8):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            room = 1.0 - h[idx]
            if room <= 0.0:
                break
            h[idx] = h[idx] + min(diff, room)
        else:
            idx = int(np.argmax(h))
            room = h[idx]
            if room <= 0.0:
                break
            h[idx] = h[idx] - min(-diff, room)
    return h


def interp_to_n(src, src_n, n):
    if int(src_n) == int(n):
        return np.asarray(src, dtype=float).copy()
    x_old = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.interp(x_new, x_old, np.asarray(src, dtype=float))
    return project_feasible(h, n / 2.0)


def counts(h):
    return np.correlate(h, 1.0 - h, mode='full')


def counts_max(h):
    return float(np.max(counts(h)))


def c5_from_counts(c, n):
    return float(np.max(c)) * (2.0 / n)


def c5_from_h(h):
    c = counts(h)
    return c, c5_from_counts(c, len(h))


def make_D(h):
    n = len(h)
    L = 2 * n - 1
    off = n - 1
    D = np.zeros((L, n), dtype=float)
    one_minus_h = 1.0 - h
    neg_h_rev = -h[::-1]
    for i in range(n):
        s1 = off - i
        D[s1:s1 + n, i] += one_minus_h
        s2 = i
        D[s2:s2 + n, i] += neg_h_rev
    return D


def make_Gv_ap(h):
    D = make_D(h)
    n = len(h)
    m = n // 2
    return D[:, :m] - D[:, m:]


def make_Gv_sym(h):
    D = make_D(h)
    n = len(h)
    q = n // 4
    p0 = np.arange(q)
    p1 = 2 * q - 1 - p0
    p2 = p0 + 2 * q
    p3 = n - 1 - p0
    return D[:, p0] - D[:, p1] + D[:, p2] - D[:, p3]


def smooth_value(c, beta, dx):
    m = float(np.max(c))
    shifted = c - m
    return float((m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx)


def softmax_weights(c, beta):
    m = float(np.max(c))
    w = np.exp(beta * (c - m))
    s = float(np.sum(w))
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros_like(c, dtype=float)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= s
    return w


def build_h_ap(u, n):
    m = n // 2
    h = np.empty(n, dtype=float)
    h[:m] = u
    h[m:] = 1.0 - u
    return h


def build_h_sym(v, n):
    q = n // 4
    h = np.empty(n, dtype=float)
    p0 = np.arange(q)
    p1 = 2 * q - 1 - p0
    p2 = p0 + 2 * q
    p3 = n - 1 - p0
    h[p0] = v
    h[p1] = 1.0 - v
    h[p2] = v
    h[p3] = 1.0 - v
    return h


def project_ap(h):
    n = len(h)
    if n % 2 != 0:
        return None, None
    h = project_feasible(h, n / 2.0)
    m = n // 2
    u = 0.5 * (h[:m] + (1.0 - h[m:]))
    u = np.clip(u, 0.0, 1.0)
    h2 = build_h_ap(u, n)
    return u, h2


def project_sym(h):
    n = len(h)
    if n % 4 != 0:
        return None, None
    h = project_feasible(h, n / 2.0)
    q = n // 4
    p0 = np.arange(q)
    p1 = 2 * q - 1 - p0
    p2 = p0 + 2 * q
    p3 = n - 1 - p0
    v = (h[p0] + (1.0 - h[p1]) + h[p2] + (1.0 - h[p3])) * 0.25
    v = np.clip(v, 0.0, 1.0)
    h2 = build_h_sym(v, n)
    return v, h2


def obj_current(params):
    typ = G['type']
    n = int(G['n'])
    dx = float(G['dx'])
    beta = float(G['beta'])
    if typ == 'ap':
        h = build_h_ap(params, n)
    else:
        h = build_h_sym(params, n)
    c = counts(h)
    return smooth_value(c, beta, dx)


def jac_current(params):
    typ = G['type']
    n = int(G['n'])
    dx = float(G['dx'])
    beta = float(G['beta'])
    if typ == 'ap':
        h = build_h_ap(params, n)
    else:
        h = build_h_sym(params, n)
    c = counts(h)
    w = softmax_weights(c, beta)
    if typ == 'ap':
        Gv = make_Gv_ap(h)
    else:
        Gv = make_Gv_sym(h)
    return (Gv.T @ w) * dx


def optimize_smooth(n, params, typ, betas, maxfun=None):
    params = np.asarray(params, dtype=float).copy()
    expected = n // 2 if typ == 'ap' else n // 4
    if params.size != expected:
        return params, np.inf
    params = np.clip(params, 0.0, 1.0)
    if typ == 'ap':
        build = build_h_ap
    else:
        build = build_h_sym

    h0 = build(params, n)
    _, best_c = c5_from_h(h0)
    if not np.isfinite(best_c):
        best_c = 1e100
    best_p = params.copy()
    if maxfun is None:
        maxfun = maxfun_for_n(n)
    bounds = [(0.0, 1.0)] * expected

    for beta in betas:
        if time_left() < 15:
            break
        G['type'] = typ
        G['n'] = int(n)
        G['dx'] = 2.0 / float(n)
        G['beta'] = float(beta)
        try:
            res = minimize(
                obj_current,
                params,
                jac=jac_current,
                method='L-BFGS-B',
                bounds=bounds,
                options={
                    'maxiter': 140,
                    'maxfun': int(maxfun),
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxls': 40
                }
            )
            if res is not None and hasattr(res, 'x'):
                cand = res.x
                if np.all(np.isfinite(cand)):
                    params = np.clip(np.asarray(cand, dtype=float), 0.0, 1.0)
        except Exception:
            pass

        h = build(params, n)
        try:
            _, c5 = c5_from_h(h)
        except Exception:
            c5 = best_c
        if np.isfinite(c5) and c5 < best_c:
            best_c = c5
            best_p = params.copy()
        else:
            params = best_p.copy()

    return best_p, best_c


def make_candidates(base, sweep, n):
    arr = [base, 0.0, 1.0]
    if sweep < 2:
        arr.extend(np.linspace(0.0, 1.0, 9)[1:-1].tolist())
    elif sweep < 4 and n <= 200:
        arr.extend(np.linspace(0.0, 1.0, 5)[1:-1].tolist())

    if sweep < 6:
        offs = [0.25, 0.12, 0.06, 0.03, 0.012]
    elif sweep < 16:
        offs = [0.10, 0.05, 0.02, 0.008, 0.003]
    else:
        offs = [0.04, 0.02, 0.008, 0.003, 0.001]

    for d in offs:
        arr.extend([base - d, base + d])

    arr = np.clip(np.asarray(arr, dtype=float), 0.0, 1.0)
    arr = np.unique(np.round(arr, 10))
    return arr


def coord_descent_ap(u, n, max_sweeps=30):
    m = n // 2
    u = np.clip(np.asarray(u, dtype=float), 0.0, 1.0).copy()
    h = build_h_ap(u, n)
    dx = 2.0 / float(n)
    cur = counts_max(h) * dx
    best_c = cur
    best_u = u.copy()
    no_improve = 0
    sweep = 0

    while sweep < max_sweeps and time_left() > 4 and no_improve < 4:
        improved = False
        if sweep % 2 == 0:
            order = np.arange(m)
        else:
            rng = np.random.RandomState(1234 + sweep)
            order = rng.permutation(m)

        for i in order:
            base = float(u[i])
            cands = make_candidates(base, sweep, n)
            best_y = base
            best_val = cur
            old0 = h[i]
            old1 = h[m + i]

            for y in cands:
                y = float(y)
                if abs(y - base) < 1e-12:
                    continue
                h[i] = y
                h[m + i] = 1.0 - y
                val = counts_max(h) * dx
                if val < best_val - 1e-14:
                    best_val = val
                    best_y = y

            h[i] = old0
            h[m + i] = old1

            if best_y != base:
                u[i] = best_y
                h[i] = best_y
                h[m + i] = 1.0 - best_y
                cur = best_val
                improved = True
                if cur < best_c - 1e-13:
                    best_c = cur
                    best_u = u.copy()

        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        sweep += 1

    best_u = np.clip(best_u, 0.0, 1.0)
    hbest = build_h_ap(best_u, n)
    return best_u, best_c, hbest

def param_h_tri(n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.minimum(x, 2.0 - x)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n / 2.0)


def param_h_sin(n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.sin(0.5 * np.pi * x) ** 2
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n / 2.0)


def vec_similar(a, b):
    if a is None or b is None or a.size != b.size or a.size == 0:
        return False
    return float(np.max(np.abs(np.asarray(a, dtype=float) - np.asarray(b, dtype=float)))) < 1e-6


def add_unique_vecs(store, vec):
    if vec is None:
        return
    vec = np.asarray(vec, dtype=float)
    if vec.size == 0:
        return
    vec = np.clip(vec, 0.0, 1.0)
    for w in store:
        if vec_similar(w, vec):
            return
    store.append(vec)


def fine_pair_search(h, n, max_trials):
    h0 = np.asarray(h, dtype=float).copy()
    h = fix_sum_exact(h0)
    dx = 2.0 / float(n)
    cur = counts_max(h) * dx
    best_c = cur
    best_h = h.copy()
    rng = np.random.RandomState(987654)

    for _ in range(max_trials):
        if time_left() < 3:
            break
        i = int(rng.randint(n))
        for _j in range(12):
            j = int(rng.randint(n))
            if i == j:
                continue
            if h[i] <= 1e-12 and h[j] <= 1e-12:
                continue
            if h[i] >= 1.0 - 1e-12 and h[j] >= 1.0 - 1e-12:
                continue

            if h[i] < h[j]:
                sign = 1.0
            else:
                sign = -1.0
            for mag in (0.004, 0.02, 0.001):
                d = sign * mag
                ni = h[i] + d
                nj = h[j] - d
                if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                    continue
                h[i] = ni
                h[j] = nj
                val = counts_max(h) * dx
                if val < best_c - 1e-13:
                    best_c = val
                    best_h = h.copy()
                    cur = val
                    break
                h[i] = ni - d
                h[j] = nj + d
            if time_left() < 3:
                break

    best_h = fix_sum_exact(best_h)
    exact = c5_from_h(best_h)[1]
    if exact < best_c:
        best_c = exact
    return best_h, best_c

    if best_h is not None and best_n == 0:
        best_n = len(best_h)

    if total_time_left() > 35 and best_h is not None:
        STAGE_END = min(time.time() + min(total_time_left() - 10, 90.0), T_START + T_BUDGET - 1.0)
        max_trials = 1200 if best_n <= 180 else 600
        h_try, c_try = fine_pair_search(best_h, best_n, max_trials)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try
            best_n = len(h_try)

    if best_h is None:
        best_h = param_h_tri(160)
        best_n = len(best_h)

    best_h = fix_sum_exact(best_h)
    final_c5 = c5_from_h(best_h)[1]
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if np.isfinite(eval_c5) and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    best_h = np.clip(best_h, 0.0, 1.0)
    return (best_h, float(best_c5), int(best_n))
