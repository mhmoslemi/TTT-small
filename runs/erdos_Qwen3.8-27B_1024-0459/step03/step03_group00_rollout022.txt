Actually, let's think about it differently. For coordinate i:
- corr[k] for k involving h[i] is linear in h[i]
- Specifically, corr[n-1+m] = ... + h[i+m]*(1-h[i]) + ... (for m = 1,...,n-1-i)
  = ... + h[i+m] - h[i+m]*h[i] + ...
  So as a function of t = h[i]: corr[n-1+m] decreases with slope -h[i+m]
  
- corr[n-1-m] = ... + h[i]*(1-h[i-m]) + ... (for m = 1,...,i)
  = ... + t*(1-h[i-m]) + ...
  So as a function of t: corr[n-1-m] increases with slope (1-h[i-m])

So we have a set of lines in t, and we want to minimize the maximum. The optimal t is where the maximum of the increasing lines equals the maximum of the decreasing lines (or at the boundary 0 or 1).

This can be solved efficiently! The optimal t* satisfies:
max_{m: i-m>=0} (base[n-1-m] + t*(1-h[i-m])) = max_{m: i+m<n} (base[n-1+m] + h[i+m] - t*h[i+m])

Where base[k] is the part of corr[k] not involving h[i].

This is a 1D problem that can be solved in O(n) time per coordinate.

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_BETA = 1.0
_G_TGT = 0.0
_G_START = 0.0
_G_BUDGET = 1300.0
_G_DX = 1.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        # Adjust the element closest to 0.5
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_smooth(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _G_BETA
    shifted = corr - m
    max_s = np.max(shifted)
    exps = np.exp(np.clip(beta * (shifted - max_s), -500, 500))
    s = np.sum(exps)
    if s < 1e-300:
        return (m + max_s) * dx
    return (m + max_s + np.log(s) / beta) * dx


def obj_raw(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(corr)) * dx


def con_sum(x):
    return np.sum(x) - _G_TGT


def con_sum_jac(x):
    return np.ones(len(x))


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt else 0.0
    return project_feasible(h, n)


def make_golden(n):
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i * gr) % 1.0 < 0.5 else 0.0
    return project_feasible(h, n)


def make_random_bin(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h


def make_sinusoidal(rng, n, freq, phase):

    h = 0.5 + 0.4 * np.sin(2.0 * np.pi * freq * t + phase)
    h = 0.5 + 0.15 * np.sin(2.0 * np.pi * 2 * freq * t + phase * 1.7)
    return project_feasible(np.clip(h, 0.0, 1.0), n)


def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)


def lbfgsb_optimize(h, n, beta_list, max_iter_per, time_limit):
    global _G_N, _G_BETA, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)

    for beta_val in beta_list:

        _G_BETA = float(beta_val)
        try:

            res = minimize(obj_smooth, h_cur, method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': mi, 'ftol': 1e-20,
                                    'gtol': 1e-14, 'iprint': 0,
                                    'maxls': 40, 'maxcor': 30})
            h_new = project_feasible(res.x, n)

        if c5_polished < best_c5 - 1e-14:
            best_c5 = c5_polished
            best_h = h_polished.copy()
            print("  Phase5: C5={:.8f}".format(best_c5))

    # Phase 6: Final coordinate descent polish
    if remaining_time() > 15:
        delta_fine = [0.01, 0.02, 0.05, 0.1]
        best_h, best_c5 = do_coord_descent(best_h, best_n, 20, delta_fine)
        print("  Final coord: C5={:.8f}".format(best_c5))

    # Phase 7: One more L-BFGS-B pass
    if remaining_time() > 15:
        n = best_n
        _G_N = n
        _G_TGT = n / 2.0
        _G_DX = 2.0 / n
        _G_BETA = 30000.0
        bounds = [(0.0, 1.0)] * n
        cons = {'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}
        try:
            res = minimize(obj_smooth, best_h, method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-20,
                                    'gtol': 1e-15, 'iprint': 0, 'maxls': 40})
            h_final = project_feasible(res.x, n)
            c5_final_test = compute_c5(h_final)
            if c5_final_test < best_c5 - 1e-14:
                best_c5 = c5_final_test
                best_h = h_final.copy()
                print("  Final L-BFGS-B: C5={:.8f}".format(best_c5))
        except Exception:
            pass

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
