        h_work_bin = h_bin.copy()

    c5_work = compute_c5(h_work_bin)
    ones_mask = h_work_bin.astype(bool)
    no_improve = 0
    max_stall = 8000

    for it in range(200000):
        if remaining() < 40:
            break
        if no_improve > max_stall:
            break

        ones_idx = np.where(ones_mask)[0]
        zeros_idx = np.where(~ones_mask)[0]
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]

        h_work_bin[i] = 0.0
        h_work_bin[j] = 1.0
        c5_test = compute_c5(h_work_bin)

        if c5_test < c5_work - 1e-15:
            c5_work = c5_test
            no_improve = 0
            ones_mask[i] = False
            ones_mask[j] = True
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_work_bin.copy()
        else:
            h_work_bin[i] = 1.0
            h_work_bin[j] = 0.0
            no_improve += 1

    print("After discrete: C5={:.8f}".format(c5_work))

    # Phase 2: L-BFGS with focused smooth max at multiple resolutions
    resolutions = sorted(set([best_n, best_n + 16, best_n + 32, best_n + 48,
                              160, 176, 192, 208, 224]) &
                          {r for r in range(120, 300, 2)})

    for n_try in resolutions:
        if remaining() < 50:
            break
        target = n_try / 2.0

        xp = (np.arange(best_n) + 0.5) * 2.0 / best_n
        xn = (np.arange(n_try) + 0.5) * 2.0 / n_try
        h0 = np.interp(xn, xp, best_h)
        h0 = project_feasible(h0, target)

        for beta in [300, 800, 2000, 5000]:
            if remaining() < 12:
                break
            tk = 3 if beta >= 2000 else 5

            def lobj(p):
                hc = np.clip(p, 0.0, 1.0)
                c = np.correlate(hc, 1.0 - hc, mode='full') * (2.0 / n_try)
                val = smooth_max_topk(c, beta, top_k=tk)
                se = np.sum(hc) - target
                val += 2000.0 * se * se
                return val

            try:
                res = minimize(lobj, h0, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * n_try,
                               options={'maxiter': 300, 'ftol': 1e-18,
                                        'gtol': 1e-14, 'maxls': 25,
                                        'maxfun': 2500})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n_try
                    h0 = h_new.copy()
                    print("  L-BFGS n={}, beta={}: C5={:.8f}".format(n_try, beta, c5_new))
                else:
                    h0 = h_new.copy()
            except Exception:
                pass

    # Phase 3: Simulated annealing on continuous values
    if remaining() > 80:
        n = best_n
        target = n / 2.0
        h = best_h.copy()
        T_start = 0.001
        T_end = 0.00001
        total_sa_steps = 80000
        no_improve_sa = 0

        for it in range(total_sa_steps):
            if remaining() < 15:
                break
            if no_improve_sa > 30000:
                break

            frac = it / total_sa_steps
            T = T_start * (T_end / T_start) ** frac

            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.choice([0.01, -0.01, 0.005, -0.005, 0.02, -0.02,
                            0.003, -0.003, 0.03, -0.03])
            if h[i] + d < -1e-9 or h[i] + d > 1.0 + 1e-9:
                continue
            if h[j] - d < -1e-9 or h[j] - d > 1.0 + 1e-9:
                continue

            h[i] += d
            h[j] -= d
            c5_t = compute_c5(h)
            delta = c5_t - best_c5

            if delta < 0:
                best_c5 = c5_t
                best_h = h.copy()
                no_improve_sa = 0
            else:
                if rng.random() < np.exp(-delta / max(T, 1e-10)):
                    no_improve_sa = 0
                else:
                    h[i] -= d
                    h[j] += d
                    no_improve_sa += 1

        print("After SA: C5={:.8f}".format(best_c5))

    # Phase 4: Focused gradient-based refinement (active shifts)
    if remaining() > 60:
        n = best_n
        target = n / 2.0
        h = best_h.copy()
        lr = 0.02
        momentum = np.zeros(n)
        decay = 0.85

        for it in range(3000):
            if remaining() < 10:
                break
            g = active_shift_gradient(h, top_k=3)
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-14:
                break
            g /= gnorm

            momentum = decay * momentum - lr * g
            h_trial = h + momentum
            h_trial = project_feasible(h_trial, target)
            c5_trial = compute_c5(h_trial)

            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
            h = h_trial.copy()

            if it % 200 == 0 and it > 0:
                lr *= 0.92

        print("After gradient: C5={:.8f}".format(best_c5))

    # Phase 5: Intensive local search
    if remaining() > 40:
        n = best_n
        target = n / 2.0
        h = best_h.copy()
        no_improve_ls = 0

        for sweep in range(400):
            if remaining() < 5:
                break
            if no_improve_ls > 80:
                break
            improved = False
            n_trials = 4000
            for _ in range(n_trials):
                if remaining() < 3:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = rng.uniform(-0.015, 0.015)
                if abs(d) < 0.001:
                    d = 0.002 if rng.random() < 0.5 else -0.002
                if h[i] + d < -1e-9 or h[i] + d > 1.0 + 1e-9:
                    continue
                if h[j] - d < -1e-9 or h[j] - d > 1.0 + 1e-9:
                    continue
                h[i] += d
                h[j] -= d
                c5_t = compute_c5(h)
                if c5_t < best_c5 - 1e-13:
                    best_c5 = c5_t
                    best_h = h.copy()
                    improved = True
                else:
                    h[i] -= d
                    h[j] += d
            if improved:
                no_improve_ls = 0
                h = best_h.copy()
            else:
                no_improve_ls += 1

        print("After local search: C5={:.8f}".format(best_c5))

    # Phase 6: SLSQP final polish
    if remaining() > 12:
        n = best_n
        target = n / 2.0
        dx = 2.0 / n
        bounds = [(0.0, 1.0)] * n

        for beta in [5000, 15000]:
            if remaining() < 6:
                break

            def sobj(x):
                xc = np.clip(x, 0.0, 1.0)
                c = np.correlate(xc, 1.0 - xc, mode='full') * dx
                top3 = np.sort(c)[-3:]
                m3 = top3.max()
                return m3 + np.log(np.sum(np.exp(beta * (top3 - m3)))) / beta

            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                     'jac': lambda x: np.ones(n)}]
            try:
                res = minimize(sobj, best_h, method='SLSQP', bounds=bounds,
                               constraints=cons, maxiter=200, ftol=1e-18)
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  SLSQP: C5={:.8f}".format(c5_new))
            except Exception:
                pass

    # Final: very fine single-coordinate adjustments
    if remaining() > 8:
        n = best_n
        target = n / 2.0
        h = best_h.copy()

        for _ in range(5000):
            if remaining() < 2:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.002, 0.002)
            if h[i] + d < 0 or h[i] + d > 1 or h[j] - d < 0 or h[j] - d > 1:
                continue
            h[i] += d
            h[j] -= d
            c5_t = compute_c5(h)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                best_h = h.copy()
            else:
                h[i] -= d
                h[j] += d

    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
