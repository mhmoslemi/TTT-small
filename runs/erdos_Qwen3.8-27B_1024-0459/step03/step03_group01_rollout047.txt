    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n

    return np.interp(x_new, xp, src_h)

    n_ones = int(np.sum(h_bin))

        add = rng.choice(zeros_idx, n // 2 - n_ones, replace=False)

def lse_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    max_c = np.max(corr)
    shifted = corr - max_c
    cl = np.clip(beta * shifted, -50, 50)
    ex = np.exp(cl)
    w = ex / np.sum(ex)

    for k in range(len(corr)):
        if w[k] > 1e-13:

def thue_morse(n, seed_val=0):
    h = np.zeros(n)
    for i in range(n):
        v = i
        c = 0
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c == 1 else 0.0
    return h


def periodic_pattern(n, period_str, seed_val=0):

    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
