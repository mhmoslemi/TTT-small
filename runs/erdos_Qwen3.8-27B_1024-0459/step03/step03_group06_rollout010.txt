_TGT = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _soft_max_val(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    max_s = np.max(shifted)
    val = m + (max_s + np.log(np.sum(np.exp(shifted - max_s)))) / beta
    return val * (2.0 / _N)


def _soft_max_weights(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    max_s = np.max(shifted)
    w = np.exp(shifted - max_s)
    w /= np.sum(w)
    return w


def obj_func(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    val = _soft_max_val(corr, _BETA)
    pen = _PEN * (np.sum(x) - _TGT) ** 2
    return val + pen


def obj_grad(x):
    n = _N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    w = _soft_max_weights(corr, _BETA)
    one_minus_x = 1.0 - x
    
    grad = np.zeros(n)
    # s = 0
    w0 = w[n - 1]
    if w0 > 1e-18:
        grad += w0 * (1.0 - 2.0 * x)
    
    for s in range(1, n):
        wp = w[n - 1 + s]
        wn = w[n - 1 - s]
        if wp > 1e-18 or wn > 1e-18:
            a = n - s
            # grad[s:] += wp*(1-x[:a]) - wn*x[:a]
            # grad[:a] += wn*(1-x[s:]) - wp*x[s:]
            tmp1 = wp * one_minus_x[:a] - wn * x[:a]

            grad[s:] += tmp1
            grad[:a] += tmp2

    grad += 2.0 * _PEN * (np.sum(x) - _TGT)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # distribute among a few points
        idx = np.argsort(np.abs(h - 0.5))[:max(1, int(abs(diff)) + 1)]
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src)

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    return h


def make_golden_ratio(n):
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if abs(frac - 0.5) < 0.25 else 0.0
    return h


def make_m_sequence(n, m_bits=8):
    # Generate an m-sequence of period 2^m_bits - 1, repeat and trim to n
    period = (1 << m_bits) - 1
    # Simple m-sequence using LFSR
    # For degree 8, use polynomial x^8 + x^4 + x^3 + x^2 + 1
    reg = [1] * m_bits
    tap_positions = [8, 4, 3, 2, 1]  # 1-indexed
    seq = []
    for _ in range(period):
        seq.append(reg[-1])
        feedback = 0
        for p in tap_positions:
            feedback ^= reg[p - 1]
        reg = reg[1:] + [feedback]
    # Repeat to fill n
    h = np.array(seq * (n // period + 2), dtype=float)[:n]
    return h


def make_sine(n, freq, phase, amp=0.3):
    t = np.arange(n) / float(n)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * t + phase)
    return np.clip(h, 0.0, 1.0)


def binary_sa(h_init, n, rng, max_iters=20000, t_start=0.003, t_end=0.00001):
    h = np.where(h_init > 0.5, 1.0, 0.0)
    target_ones = n // 2
    num_ones = int(np.sum(h))
    if num_ones > target_ones:
        ones_idx = np.where(h == 1.0)[0]
        remove = rng.choice(len(ones_idx), num_ones - target_ones, replace=False)
        for r in remove:
            h[ones_idx[r]] = 0.0
    elif num_ones < target_ones:
        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
        for a in add:
            h[zeros_idx[a]] = 1.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_c5 = best_c5
    
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        if not ones_list or not zeros_list:
            break
        
        i = rng.randint(len(ones_list))
        j = rng.randint(len(zeros_list))
        oi = ones_list[i]
        oz = zeros_list[j]
        
        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = compute_c5(h)
        
        T = t_start * (t_end / t_start) ** (float(it) / max(float(max_iters), 1.0))
        
        if new_c5 <= curr_c5 or (T > 1e-15 and rng.rand() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            ones_list[i] = oz
            zeros_list[j] = oi
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0
    
    return best_h, best_c5
