def _grad_q(q,h,n):
    dx=2.0/n
    g=np.zeros(n,float)
    # active threshold
    for idx, w in enumerate(q):
       w=float(w)
       if w < 1e-12: continue
       s=n-1-idx
       if s >= 0:
           r=n-s
           if r>0:
               g[:r] += w*(1.0-h[s:])
               g[s:] = g[s:] - w*h[:r]  # note g[s:] length n-s = r
       else:
           p=-s
           r=n-p
           if r>0:
               g[p:] += w*(1.0-h[:r])
               g[:r] -= w*h[p:]
    return g*dx

    cval = _exact(h)
    a = float(step0)
    total = 2*n - 1

    if iters <= 0:

        if _tl() < 2.0:
            break

                break
            if d > 0.0:
                room = 1.0 - h
                tot = float(room.sum())
                if tot <= 1e-15:
                    break
                h += room * (d / tot)
            else:
                tot = float(h.sum())
                if tot <= 1e-15:
                    break
                h *= (1.0 + d / tot)
            np.clip(h, 0.0, 1.0, out=h)
    d = target - float(h.sum())
    if abs(d) > 1e-9:
        h = _project_bisect(h, n)
    return _enforce_final(h)


def _interp_from(hsrc, nsrc, ndst):
    if int(ndst) == int(nsrc):
        return np.asarray(hsrc, dtype=np.float64).copy()
    xs = (np.arange(int(nsrc)) + 0.5) * (2.0 / float(int(nsrc)))
    xd = (np.arange(int(ndst)) + 0.5) * (2.0 / float(int(ndst)))
    return np.interp(xd, xs, np.asarray(hsrc, dtype=np.float64))


def _grad_q(q, h, n):
    dx = 2.0 / float(n)
    g = np.zeros(n, dtype=np.float64)
    total = len(q)
    for idx in range(total):
        w = float(q[idx])
        if w <= 1e-13:
            continue
        s = n - 1 - idx
        if s >= 0:
            r = n - s
            if r > 0:
                g[:r] += w * (1.0 - h[s:])
                g[s:] -= w * h[:r]
        else:
            p = -s
            r = n - p
            if r > 0:
                g[p:] += w * (1.0 - h[:r])
                g[:r] -= w * h[p:]
    return g * dx


def _make_q(c, beta, k):
    total = len(c)
    q = np.zeros(total, dtype=np.float64)
    b = float(beta)
    if k is None or k <= 0 or k >= total:
        mx = float(np.max(c))
        z = np.exp(b * (c - mx))
        zsum = float(np.sum(z))
        if zsum < 1e-300:
            q[int(np.argmax(c))] = 1.0
        else:
            q = z / zsum
        return q
    kk = int(k)
    if kk == 1:
        q[int(np.argmax(c))] = 1.0
        return q
    top = np.argsort(c)[-kk:]
    vals = c[top]
    mx = float(np.max(vals))
    z = np.exp(b * (vals - mx))
    zsum = float(np.sum(z))
    if zsum < 1e-300:
        q[top[int(np.argmax(vals))]] = 1.0
    else:
        q[top] = z / zsum
    return q


def _sl_obj(x):
    n = int(_SL_N)
    h = np.asarray(x, dtype=np.float64)
    if not np.all(np.isfinite(h)):
        return 1e100
    h = np.clip(h, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(c))
    b = float(_SL_BETA)
    z = np.exp(b * (c - m))
    zsum = float(np.sum(z))
    if zsum < 1e-300:
        val = m
    else:
        val = m + float(np.log(zsum)) / b
    return val * (2.0 / float(n))


def _sl_grad(x):
    n = int(_SL_N)
    h = np.asarray(x, dtype=np.float64)
    if not np.all(np.isfinite(h)):
        h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, mode="full")
    q = _make_q(c, _SL_BETA, None)
    return _grad_q(q, h, n)


def _sl_con(x):
    return float(np.sum(x)) - 0.5 * float(_SL_N)


def _sl_con_jac(x):
    return np.ones(int(_SL_N), dtype=np.float64)


def _slsqp_once(h, n, beta, maxiter):
    global _SL_N, _SL_BETA
    n = int(n)
    _SL_N = n
    _SL_BETA = float(beta)
    x0 = np.asarray(h, dtype=np.float64).copy()
    try:
        res = minimize(
            _sl_obj,
            x0,
            method="SLSQP",
            jac=_sl_grad,
            bounds=[(0.0, 1.0)] * n,
            constraints=[{"type": "eq", "fun": _sl_con, "jac": _sl_con_jac}],
            options={"maxiter": int(maxiter), "ftol": 1e-12, "maxls": 30, "iprint": 0},
        )
        x = np.asarray(res.x, dtype=np.float64)
        if np.all(np.isfinite(x)):
            return _project(x, n)
    except Exception:
        pass
    return h


def _slsqp_refine(h, n, betas, maxiter):
    h = _project(h, n)
    cval = _exact(h)
    for b in betas:
        if _tl() < 20.0:
            break
        if int(n) > 300 and int(float(b)) > 400:
            continue
        h2 = _slsqp_once(h, n, float(b), maxiter)
        c2 = _exact(h2)
        if c2 < cval - 1e-16:
            h = h2
            cval = c2
    return h, cval


def _top_grad_refine(h0, n, iters, k=1, beta=30.0, step0=0.05):
    h = _project(h0, n)
    cval = _exact(h)
    if int(iters) <= 0:
        return h, cval
    a = float(step0)
    total = 2 * int(n) - 1
    use_all = (k is None or int(k) <= 0 or int(k) >= total)
    use_k = None if use_all else int(k)
    for _ in range(int(iters)):
        if _tl() < 2.0:
            break
        c = np.correlate(h, 1.0 - h, mode="full")
        q = _make_q(c, beta, use_k)
        g = _grad_q(q, h, n)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        g = g / gn
        improved = False
        for direction in (-1.0, 1.0):
            for aa in (a, a * 2.0, a * 0.5, a * 0.25, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002):
                if aa <= 0.0:
                    continue
                h2 = _project(h + direction * aa * g, n)
                c2 = _exact(h2)
                if c2 < cval - 1e-14:
                    h = h2
                    cval = c2
                    a = min(a * 1.15, 0.5)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            a = max(a * 0.5, 1e-5)
            if a < 1e-5:
                break
    return h, cval


def _guided_transfer(h0, n, iters, rng, k=3, beta=50.0, Tstart=0.04, Tend=0.0):
    cur = _project(h0, n)
    cu = _exact(cur)
    best = cur.copy()
    bbest = cu
    if int(iters) <= 0:
        return best, bbest
    donors = None
    recipients = None
    last_grad = -1000
    it_total = int(iters)
    for its in range(it_total):
        if _tl() < 1.0:
            break
        frac = float(its) / float(it_total)
        if float(Tend) <= 0.0:
            T = float(Tstart) * max(0.0, 1.0 - frac)
        else:
            T = float(Tstart) + (float(Tend) - float(Tstart)) * frac
        if donors is None or int(its - last_grad) >= 12:
            c = np.correlate(cur, 1.0 - cur, mode="full")
            q = _make_q(c, beta, int(k))
            g = _grad_q(q, cur, n)
            don = np.where((cur > 1e-8) & (g > 1e-10))[0]
            rec = np.where((cur < 1.0 - 1e-8) & (g < -1e-10))[0]
            if len(don) == 0:
                don = np.where(cur > 1e-8)[0]
            if len(rec) == 0:
                rec = np.where(cur < 1.0 - 1e-8)[0]
            if len(don) == 0 or len(rec) == 0:
                break
            donors = don
            recipients = rec
            last_grad = its
        i = int(donors[rng.randint(len(donors))])
        j = int(recipients[rng.randint(len(recipients))])
        if i == j:
            continue
        maxd = min(float(cur[i]), 1.0 - float(cur[j]))
        if maxd <= 1e-13:
            continue
        r = float(rng.rand())
        if r < 0.45:
            delta = maxd
        elif r < 0.85:
            delta = maxd * (0.4 + 0.6 * float(rng.rand()))
        else:
            delta = maxd * (0.05 + 0.45 * float(rng.rand()))
        if delta <= 1e-14:
            continue
        nxt = cur.copy()
        nxt[i] -= delta
        nxt[j] += delta
        if nxt[i] < 0.0:
            add = -float(nxt[i])
            nxt[i] = 0.0
            nxt[j] -= add
        elif nxt[i] > 1.0:
            sub = float(nxt[i]) - 1.0
            nxt[i] = 1.0
            nxt[j] += sub
        if nxt[j] > 1.0:
            sub = float(nxt[j]) - 1.0
            nxt[j] = 1.0
            nxt[i] += sub
        elif nxt[j] < 0.0:
            add = -float(nxt[j])
            nxt[j] = 0.0
            nxt[i] -= add
        np.clip(nxt, 0.0, 1.0, out=nxt)
        d = 0.5 * float(n) - float(nxt.sum())
        if abs(d) > 1e-13:
            if abs(d) < 1e-7:
                if d > 0.0 and (1.0 - float(nxt[j])) >= d:
                    nxt[j] += d
                elif d < 0.0 and float(nxt[i]) >= -d:
                    nxt[i] -= -d
                else:
                    nxt = _project(nxt, n)
            else:
                nxt = _project(nxt, n)
        c2 = _exact(nxt)
        accept = False
        if c2 < cu - 1e-15:
            accept = True
        elif T > 1e-14:
            diff = c2 - cu
            if diff < 10.0 * T:
                accept = (float(rng.rand()) < math.exp(-diff / T))
        if accept:
            cur = nxt
            cu = c2
            last_grad = -1000
            if cu < bbest:
                bbest = cu
                best = cur.copy()
    return best, bbest


def _polish_continuous(h, n, rng, do_slq=True, slq_betas=(30, 100, 300, 800), slq_iter=70,
                       it1=1200, it2=1000, it3=1000, guided=50000):
    h = _project(h, n)
    cval = _exact(h)
    cscale = max(0.05, min(1.0, 30000.0 / (float(n) * float(n))))
    lscl = max(0.2, min(1.0, 180.0 / float(n)))
    if do_slq and _tl() > 50.0:
        h, cval = _slsqp_refine(h, n, slq_betas, int(max(20, int(slq_iter) * lscl)))
    i1 = int(max(80, int(it1) * cscale))
    i2 = int(max(80, int(it2) * cscale))
    i3 = int(max(80, int(it3) * cscale))
    gi = int(max(2000, int(guided) * cscale))
    if _tl() > 15.0:
        h, cval = _top_grad_refine(h, n, i1, 1, 30.0, 0.05)
    if _tl() > 15.0:
        h, cval = _top_grad_refine(h, n, i2, 3, 30.0, 0.05)
    if _tl() > 15.0:
        h, cval = _top_grad_refine(h, n, i3, 5, 50.0, 0.04)
    if _tl() > 25.0:
        h, cval = _guided_transfer(h, n, gi, rng, 3, 50.0, 0.05, 0.0)
    if _tl() > 12.0:
        h, cval = _top_grad_refine(h, n, max(80, i1 // 2), 3, 50.0, 0.02)
    return h, cval


def _binary_rebuild(h, n):
    ones = []
    zeros = []
    p1 = np.full(n, -1, dtype=int)
    p0 = np.full(n, -1, dtype=int)
    for i in range(int(n)):
        if h[i] == 1:
            p1[i] = len(ones)
            ones.append(i)
        else:
            p0[i] = len(zeros)
            zeros.append(i)
    return ones, zeros, p1, p0


def _binary_aux(c, M):
    band = c >= (M - 1)
    cnt = int(np.count_nonzero(band))
    s = int(np.sum(c[band]))
    return 1000 * cnt + s


def _binary_from_feature(feat, n, rng=None):
    if int(n) % 2 != 0:
        return None
    f = np.asarray(feat, dtype=np.float64)
    if len(f) != int(n):
        f = _interp_from(f, len(f), n)
    m = int(n) // 2
    order = np.argsort(-f)[:m]
    b = np.zeros(int(n), dtype=np.int16)
    b[order] = 1
    return b


def _random_binary(n, rng):
    n = int(n)
    b = np.zeros(n, dtype=np.int16)
    for i in range(n // 2):
        b[i] = 1
    rng.shuffle(b)
    return b


def _thue_binary(n, invert, rng):
    n = int(n)
    pop = np.zeros(n + 1, dtype=np.uint8)
    for i in range(1, n + 1):
        pop[i] = pop[i >> 1] + (i & 1)
    b = (pop[1:] & 1).astype(np.int16)
    if invert:
        b = 1 - b
    target = n // 2
    diff = target - int(b.sum())
    if diff != 0:
        if diff > 0:
            idx = np.flatnonzero(b == 0)
        else:
            idx = np.flatnonzero(b == 1)
        if len(idx) >= abs(diff):
            pick = idx[rng.choice(len(idx), size=abs(diff), replace=False)]
            b[pick] = 1 - b[pick]
    return b


def _binary_local_refine(b0, n, iters, rng):
    n = int(n)
    if n % 2 != 0:
        return None
    h = b0.astype(np.int16).copy()
    target = n // 2
    if int(h.sum()) != target:
        b = _binary_from_feature(h.astype(float), n, rng)
        if b is not None:
            h = b.astype(np.int16).copy()
    one = np.ones(n, dtype=np.int16) - h
    ones, zeros, p1, p0 = _binary_rebuild(h, n)
    c = np.correlate(h, one, mode="full")
    M = int(c.max())
    aux = _binary_aux(c, M)
    it = 0
    no = 0
    fail = 0
    perturb_every = 4000
    perturb_size = 5
    while it < int(iters) and _tl() > 2.0:
        it += 1
        if len(ones) == 0 or len(zeros) == 0:
            break
        ri = int(rng.randint(len(ones)))
        rj = int(rng.randint(len(zeros)))
        i = ones[ri]
        j = zeros[rj]
        h[i] = 0
        h[j] = 1
        one[i] = 1
        one[j] = 0
        c = np.correlate(h, one, mode="full")
        M2 = int(c.max())
        accept = False
        aux2 = aux
        if M2 < M:
            aux2 = _binary_aux(c, M2)
            accept = True
        elif M2 == M:
            aux2 = _binary_aux(c, M2)
            if aux2 < aux or (aux2 == aux and float(rng.rand()) < 0.03):
                accept = True
        if accept:
            last = ones[-1]
            if ri != len(ones) - 1:
                ones[ri] = last
                p1[last] = ri
            ones.pop()
            p1[i] = -1
            ones.append(j)
            p1[j] = len(ones) - 1

            last = zeros[-1]
            if rj != len(zeros) - 1:
                zeros[rj] = last
                p0[last] = rj
            zeros.pop()
            p0[j] = -1
            zeros.append(i)
            p0[i] = len(zeros) - 1

            M = M2
            aux = aux2
            no = 0
        else:
            h[i] = 1
            h[j] = 0
            one[i] = 0
            one[j] = 1
            no += 1
            if no > perturb_every:
                improved = False
                for _ in range(12):
                    if _tl() < 1.0:
                        break
                    hsave = h.copy()
                    onesave = one.copy()
                    Ms = M
                    As = aux
                    for _s in range(perturb_size):
                        oo = np.flatnonzero(h == 1)
                        zz = np.flatnonzero(h == 0)
                        if len(oo) == 0 or len(zz) == 0:
                            break
                        ii = int(oo[rng.randint(len(oo))])
                        jj = int(zz[rng.randint(len(zz))])
                        h[ii] = 0
                        h[jj] = 1
                        one[ii] = 1
                        one[jj] = 0
                    c2 = np.correlate(h, one, mode="full")
                    M3 = int(c2.max())
                    A3 = _binary_aux(c2, M3)
                    if M3 < Ms or (M3 == Ms and A3 < As):
                        M = M3
                        aux = A3
                        ones, zeros, p1, p0 = _binary_rebuild(h, n)
                        improved = True
                        no = 0
                        break
                    else:
                        h[:] = hsave
                        one[:] = onesave
                if not improved:
                    fail += 1
                    no = 0
                    if fail > 3:
                        break
    return h

                        bc = cf2

    if _tl() > 25.0:
        h2, c2 = _top_grad_refine(bh, bn, 2000, 3, 50.0, 0.01)
        if c2 < bc:
            bh = h2
            bc = c2

    if _tl() > 20.0:
        h2, c2 = _guided_transfer(bh, bn, 30000, rng, 5, 100.0, 0.02, 0.0)
        if c2 < bc:
            bh = h2
            bc = c2

    bh = _project(bh, bn)
    fc5 = _exact(bh)
    try:
        ec = float(evaluate_erdos_solution(bh))
        if np.isfinite(ec) and ec < fc5:
            fc5 = ec
    except Exception:
        pass

    print("FINAL n={0} C5={1:.10f}".format(bn, fc5))
    return (bh, fc5, int(bn))
