import numpy as np

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(40):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_max_val(corr, beta, top_k):
    n_corr = len(corr)
    if top_k is not None and top_k < n_corr:
        idx = np.argsort(corr)[-top_k:]
        cw = corr[idx]
        m = cw.max()
        sw = cw - m
        log_sum = np.log(np.sum(np.exp(beta * sw)))
        w = np.exp(beta * sw - log_sum)
        return m + log_sum / beta, idx, w
    else:
        m = np.max(corr)
        sw = corr - m
        log_sum = np.log(np.sum(np.exp(beta * sw)))
        w = np.exp(beta * sw - log_sum)
        return m + log_sum / beta, np.arange(n_corr), w


def obj_lbfgs(x, beta, top_k, pen_w, target):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    val, _, _ = smooth_max_val(corr, beta, top_k)
    dx = 2.0 / len(h)
    sum_err = np.sum(h) - target
    return val * dx + pen_w * sum_err * sum_err


def grad_lbfgs(x, beta, top_k, pen_w, target):
    h = np.clip(x, 0.0, 1.0)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val, idx, w = smooth_max_val(corr, beta, top_k)

    grad = np.zeros(n)

    grad *= dx
    if pen_w > 0:

    return grad
