    for t, k_idx in enumerate(idx):

            continue

            # j appears as first factor when j <= n-1-k: contributes (1-h[j+k])
            # j appears as second factor when j >= k and j-k <= n-1-k: contributes -h[j-k]
            valid_first = (k < n)  # j ranges from 0 to n-1-k
            if valid_first:
                i_range = np.arange(n - k)
                grad[i_range] += wt * (1.0 - h[i_range + k])

                j_range = i_range + k
                grad[j_range] += wt * (-h[i_range])

            # By symmetry, same as positive lag kk but with roles swapped
            # corr at lag -k = sum_{i=0}^{n-1-kk} h[i+kk] * (1-h[i])
            i_range = np.arange(n - kk)
            grad[i_range + kk] += wt * (1.0 - h[i_range])

                           options={'maxiter': 300, 'maxfun': 5000, 'ftol': 1e-18})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  L-BFGS: C5={:.8f}".format(c5_new))
        except Exception:
            pass
    
    # Phase 5: Second round Nesterov from best (with different params)
    if remaining() > 40:
        n = best_n
        target = n / 2.0
        h2 = best_h + 0.005 * rng.randn(n)
        h2 = project_feasible(h2, target)
        h_opt2, c5_opt2 = nesterov_optimize(h2, n, target, remaining,
                                             beta_start=500.0, max_iter=400)
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2.copy()
            print("  Nesterov2: C5={:.8f}".format(c5_opt2))
    
    # Phase 6: Final fine coordinate descent
    if remaining() > 15:
        n = best_n
        target = n / 2.0
        h_final, c5_final = coordinate_descent(best_h, n, target, remaining, max_sweeps=30)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
            print("  Final CD: C5={:.8f}".format(c5_final))
    
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
