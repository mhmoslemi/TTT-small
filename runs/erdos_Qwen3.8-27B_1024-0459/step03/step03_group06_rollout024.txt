    # correlate(h, 1-h, 'full') = convolve(h, (1-h)[::-1], 'full')
    b = (1.0 - h)[::-1]
    return fftconvolve(h, b, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Adjust the most interior values
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(interior) > 0:
            h[interior] += diff / np.sum(interior)
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def interpolated_h(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def smooth_max_obj(x, n, beta, tgt):
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    # Clip to avoid overflow
    shifted = np.clip(shifted, -100.0 / beta, 0.0)
    sm = (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx
    penalty = 1000.0 * (np.sum(x) - tgt) ** 2
    return sm + penalty

def smooth_max_grad(x, n, beta, tgt):
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    shifted = np.clip(shifted, -100.0 / beta, 0.0)
    weights = np.exp(beta * shifted)
    weights /= np.sum(weights)
    
    g = np.zeros(n)
    # Vectorize: for each shift k, add weighted contribution
    # Shift k >= 0: corr index = k + n - 1
    #   g[m] += w_k * (1-h[m-k]) for m in [k, n-1]
    #   g[m] += w_k * (-h[m+k]) for m in [0, n-1-k]
    # Shift k < 0, k = -p: corr index = n - 1 - p
    #   g[m] += w_{-p} * (1-h[m+p]) for m in [0, n-1-p]
    #   g[m] += w_{-p} * (-h[m-p]) for m in [p, n-1]
    
    for k in range(n):
        w = weights[k + n - 1]
        if w < 1e-14:
            continue
        # (1-h[m-k]) for m = k, ..., n-1
        g[k:] += w * (1.0 - x[:n-k])
        # -h[m+k] for m = 0, ..., n-1-k
        g[:n-k] += -w * x[k:]
    
    for p in range(1, n):
        w = weights[n - 1 - p]
        if w < 1e-14:
            continue
        # (1-h[m+p]) for m = 0, ..., n-1-p
        g[:n-p] += w * (1.0 - x[p:])
        # -h[m-p] for m = p, ..., n-1
        g[p:] += -w * x[:n-p]
    
    g *= dx
    # Penalty gradient
    g += 2.0 * 1000.0 * (np.sum(x) - tgt)
    return g

def topk_smooth_obj(x, n, top_k, beta, tgt):
    dx = 2.0 / n
    corr = compute_corr(x)
    top_idx = np.argsort(corr)[-top_k:]
    tv = corr[top_idx]
    m = np.max(tv)
    shifted = np.clip(tv - m, -100.0/beta, 0.0)
    sm = (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx
    penalty = 1000.0 * (np.sum(x) - tgt) ** 2
    return sm + penalty

def optimize_at_resolution(h0, n, budget_left, rng, label=""):

    dx = 2.0 / n
    tgt = n / 2.0
    bounds = [(0.0, 1.0)] * n

    c5_best = compute_c5(h_best)

            return smooth_max_grad(x, n, beta, tgt)

                          bounds=bounds,
                          options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-10,
                                   'iprint': 0, 'maxls': 40, 'maxfun': 3000})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                h_best = h_new
                c5_best = c5_new
        except Exception:
            pass

            except Exception:

def binary_local_search(h, n, rng, max_iter=5000):
    """Binary swap local search with random restarts."""
    h = np.where(h > 0.5, 1.0, 0.0)
    num_ones = int(np.sum(h))
    target = n // 2
    if num_ones > target:
        ones = np.where(h == 1.0)[0]
        rem = rng.choice(len(ones), num_ones - target, replace=False)
        h[ones[rem]] = 0.0
    elif num_ones < target:
        zeros = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros), target - num_ones, replace=False)
        h[zeros[add]] = 1.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_c5 = best_c5
    
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    
    T0 = 0.005
    T_end = 0.0001
    
    for it in range(max_iter):
        if remaining_time() < 2:
            break
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        
        i = rng.randint(len(ones_list))
        j = rng.randint(len(zeros_list))
        oi, oz = ones_list[i], zeros_list[j]
        
        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = compute_c5(h)
        
        T = T0 * (T_end / T0) ** (it / max(max_iter, 1))
        if new_c5 < curr_c5 or (T > 1e-15 and rng.rand() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            ones_list[i] = oz
            zeros_list[j] = oi
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0
    
    return best_h, best_c5

    penalty = 5000.0 * (np.sum(x) - tgt) ** 2
    return sm + penalty


def topk_grad(x, n, top_k, beta, tgt):
    dx = 2.0 / n
    corr = compute_corr(x)
    top_idx = set(np.argsort(corr)[-top_k:].tolist())
    m = np.max(corr)
    shifted = np.clip(corr - m, -50.0 / beta, 0.0)
    raw = np.exp(beta * shifted)
    # Zero out non-top-k
    for j in range(2 * n - 1):
        if j not in top_idx:
            raw[j] = 0.0
    wsum = np.sum(raw)
    if wsum < 1e-300:
        raw = np.zeros(2 * n - 1)
        for j in top_idx:
            raw[j] = 1.0
        wsum = float(len(top_idx))
    weights = raw / wsum

    g = np.zeros(n)
    for k in range(n):
        w = weights[k + n - 1]
        if w < 1e-13:
            continue
        g[k:] += w * (1.0 - x[:n - k])
        g[:n - k] += -w * x[k:]
    for p in range(1, n):
        w = weights[n - 1 - p]
        if w < 1e-13:
            continue
        g[:n - p] += w * (1.0 - x[p:])
        g[p:] += -w * x[:n - p]

    g *= dx
    g += 2.0 * 5000.0 * (np.sum(x) - tgt)
    return g


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    return h


def make_random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def make_sine_struct(n, rng):
    t = np.arange(n) / n
    h = 0.5
    h += rng.uniform(0.2, 0.35) * np.sin(2 * np.pi * 3 * t + rng.rand() * 6.28)
    h += rng.uniform(0.1, 0.25) * np.sin(2 * np.pi * 7 * t + rng.rand() * 6.28)
    h += rng.uniform(0.05, 0.15) * np.sin(2 * np.pi * 11 * t + rng.rand() * 6.28)
    return np.clip(h, 0.0, 1.0)


def make_walsh(n, rng):
    h = np.zeros(n)
    # Use a few Walsh-Hadamard basis functions
    h += 0.5
    for _ in range(3):
        k = rng.randint(1, max(2, n // 2))
        vals = np.zeros(n)
        for i in range(n):
            vals[i] = 1.0 if bin(i & k).count('1') % 2 == 0 else -1.0
        h += rng.uniform(0.15, 0.3) * vals
    return np.clip(h, 0.0, 1.0)


def optimize_continuous(h0, n, rng, label=""):
    """Multi-strategy continuous optimization at resolution n."""
    tgt = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h_best = project_feasible(h0, n)
    c5_best = compute_c5(h_best)

    # Strategy A: Smooth max with increasing beta
    for beta in [15, 40, 100, 250, 500]:
        if remaining_time() < 4:
            break
        try:
            res = minimize(smooth_max_obj, h_best,
                           jac=lambda x: smooth_max_grad(x, n, beta, tgt),
                           args=None, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 250, 'ftol': 1e-16,
                                    'gtol': 1e-11, 'iprint': 0,
                                    'maxls': 30, 'maxfun': 2000})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                h_best = h_new
                c5_best = c5_new
        except Exception:
            pass

    # Strategy B: Top-k focused
    for top_k in [4, 8, 12]:
        if remaining_time() < 4:
            break
        try:
            res = minimize(lambda x: topk_obj(x, n, top_k, 80, tgt),
                           h_best,
                           jac=lambda x: topk_grad(x, n, top_k, 80, tgt),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 200, 'ftol': 1e-16,
                                    'gtol': 1e-11, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                h_best = h_new
                c5_best = c5_new
        except Exception:
            pass

    # Strategy C: Very high beta, short (focus on single max)
    if remaining_time() > 5:
        for beta in [800, 1500]:
            if remaining_time() < 3:
                break
            try:
                res = minimize(smooth_max_obj, h_best,
                               jac=lambda x: smooth_max_grad(x, n, beta, tgt),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 150, 'ftol': 1e-16,
                                        'gtol': 1e-12, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best:
                    h_best = h_new
                    c5_best = c5_new
            except Exception:
                pass

    if label:
        print("  {}: n={}, C5={:.7f}".format(label, n, c5_best), flush=True)
    return h_best, c5_best


def binary_sa(h_init, n, rng, max_iter=4000):
    h = np.where(h_init > 0.5, 1.0, 0.0)
    num_ones = int(np.sum(h))
    target = n // 2
    if num_ones > target:
        ones = np.where(h == 1.0)[0]
        rem = rng.choice(len(ones), num_ones - target, replace=False)
        h[ones[rem]] = 0.0
    elif num_ones < target:
        zeros = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros), target - num_ones, replace=False)
        h[zeros[add]] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_c5 = best_c5
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    T0, T_end = 0.004, 0.00005

    for it in range(max_iter):
        if remaining_time() < 2:
            break
        if not ones_list or not zeros_list:
            break
        i = rng.randint(len(ones_list))
        j = rng.randint(len(zeros_list))
        oi, oz = ones_list[i], zeros_list[j]
        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = compute_c5(h)
        T = T0 * (T_end / T0) ** (float(it) / max(float(max_iter), 1.0))
        if new_c5 < curr_c5 or (T > 1e-15 and rng.rand() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            ones_list[i] = oz
            zeros_list[j] = oi
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0
    return best_h, best_c5


def perturbation_restart(h, n, rng, label=""):
    """Add structured perturbation and re-optimize."""
    h_p = h.copy()
    noise_type = rng.randint(3)
    if noise_type == 0:
        mask = rng.random(n) < 0.3
        h_p[mask] += rng.uniform(-0.4, 0.4, np.sum(mask))
    elif noise_type == 1:
        # Swap blocks
        a, b = sorted(rng.choice(n, 2, replace=False))
        L = rng.randint(3, n // 4)
        seg = h_p[b:b+L].copy()
        h_p[b:b+L] = h_p[a:a+L]
        h_p[a:a+L] = seg
    else:
        for _ in range(rng.randint(3, 10)):
            i, j = rng.choice(n, 2, replace=False)
            h_p[i], h_p[j] = h_p[j], h_p[i]

    h_p = project_feasible(h_p, n)
    return optimize_continuous(h_p, n, rng, label)


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.7f}".format(parent_n, parent_c5), flush=True)
    except Exception:
        best_n = 160
        best_h = project_feasible(make_thue_morse(best_n), best_n)
        best_c5 = compute_c5(best_h)
        print("Thue-Morse init: n={}, C5={:.7f}".format(best_n, best_c5), flush=True)

    # Phase 1: Optimize parent resolution
    if remaining_time() > 40:
        h_opt, c5_opt = optimize_continuous(best_h, best_n, rng, "Parent opt")
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Multi-resolution coarse-to-fine
    levels = sorted(set([best_n, 200, 256, 320, 400, 512]))
    levels = [r for r in levels if 128 <= r <= 640]

    h_curr = best_h.copy()
    n_curr = best_n

    for n_tgt in levels:
        if remaining_time() < 25:
            break
        if n_tgt <= n_curr:
            continue

        h_interp = project_feasible(interpolated_h(h_curr, n_curr, n_tgt), n_tgt)
        n_curr = n_tgt

        starts = [h_interp]
        starts.append(project_feasible(make_thue_morse(n_tgt), n_tgt))
        starts.append(project_feasible(make_random_binary(n_tgt, rng), n_tgt))
        starts.append(project_feasible(make_sine_struct(n_tgt, rng), n_tgt))
        starts.append(project_feasible(make_walsh(n_tgt, rng), n_tgt))

        best_at_level = h_interp
        best_c5_level = compute_c5(h_interp)

        for si, h0 in enumerate(starts):
            if remaining_time() < 8:
                break
            h0 = project_feasible(h0, n_tgt)
            c5_0 = compute_c5(h0)
            if c5_0 > best_c5_level * 1.2:
                continue
            h_opt, c5_opt = optimize_continuous(h0, n_tgt, rng,
                                                "R{n} s{}".format(n_tgt, si))
            if c5_opt < best_c5_level:
                best_c5_level = c5_opt
                best_at_level = h_opt

        h_curr = best_at_level
        if best_c5_level < best_c5:
            best_c5 = best_c5_level
            best_h = h_curr.copy()
            best_n = n_tgt
            print("  Level {}: C5={:.7f}".format(n_tgt, best_c5), flush=True)

    # Phase 3: Binary SA at best resolution
    if remaining_time() > 40:
        n_sa = best_n
        print("\nBinary SA n={}".format(n_sa), flush=True)
        sa_inits = [np.where(best_h > 0.5, 1.0, 0.0)]
        for _ in range(3):
            sa_inits.append(make_random_binary(n_sa, rng))

        for sa_i, sa_h in enumerate(sa_inits):
            if remaining_time() < 10:
                break
            h_bin, c5_bin = binary_sa(sa_h, n_sa, rng, max_iter=3000)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                best_n = n_sa
                print("  SA{}: C5={:.7f}".format(sa_i, c5_bin), flush=True)
            # Polish with continuous
            if remaining_time() > 8:
                h_pol, c5_pol = optimize_continuous(h_bin, n_sa, rng, "Polish SA{}".format(sa_i))
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()
                    best_n = n_sa

    # Phase 4: Perturbation restarts at best resolution
    if remaining_time() > 30:
        n_pr = best_n
        for pr_i in range(5):
            if remaining_time() < 8:
                break
            h_pr, c5_pr = perturbation_restart(best_h, n_pr, rng, "Pert{}".format(pr_i))
            if c5_pr < best_c5:
                best_c5 = c5_pr
                best_h = h_pr.copy()
                best_n = n_pr

    # Phase 5: Try a higher resolution
    if remaining_time() > 40:
        n_hi = min(best_n * 2, 768)
        if n_hi > best_n:
            h_hi = project_feasible(interpolated_h(best_h, best_n, n_hi), n_hi)
            h_opt, c5_opt = optimize_continuous(h_hi, n_hi, rng, "HiRes n={}".format(n_hi))
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_hi
                print("  HiRes: C5={:.7f}".format(c5_opt), flush=True)

    # Phase 6: Final deep polish
    if remaining_time() > 10:
        n_fin = best_n
        # Run several rounds with different betas
        tgt = n_fin / 2.0
        bounds = [(0.0, 1.0)] * n_fin
        for beta in [600, 1000, 2000]:
            if remaining_time() < 3:
                break
            try:
                res = minimize(smooth_max_obj, best_h,
                               jac=lambda x: smooth_max_grad(x, n_fin, beta, tgt),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 400, 'ftol': 1e-16,
                                        'gtol': 1e-12, 'iprint': 0})
                h_new = project_feasible(res.x, n_fin)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

        # Top-2 focused (essentially min of max two correlations)
        if remaining_time() > 4:
            try:
                res = minimize(lambda x: topk_obj(x, n_fin, 2, 200, tgt),
                               best_h,
                               jac=lambda x: topk_grad(x, n_fin, 2, 200, tgt),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 300, 'ftol': 1e-16,
                                        'gtol': 1e-12, 'iprint': 0})
                h_new = project_feasible(res.x, n_fin)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5), flush=True)
    return (best_h, final_c5, best_n)
