```python
import time
import numpy as np
from scipy.optimize import minimize

_START_TIME = 0.0
_BUDGET_S = 1300.0

_ACT_N = 0
_ACT_IDX = np.array([], dtype=int)

_LOG_N = 0
_LOG_BETA = 1.0


def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 2.0)


def sanitize_vector(v, n):
    x = np.asarray(v, dtype=float).reshape(-1)
    if x.size != n:
        x = np.resize(x, n)
    x = np.where(np.isfinite(x), x, 0.5)
    x = np.clip(x, 0.0, 1.0)
    return x


def project_feasible(h, n):
    h = sanitize_vector(h, n).copy()
    if n == 0:
        return h

    target = 0.5 * float(n)
    diff = target - float(np.sum(h))

    if abs(diff) <= 1e-12:
        return h

    for _ in range(6):
        if diff > 0.0:
            free = h < (1.0 - 1e-12)
        else:
            free = h > 1e-12

        cnt = int(np.count_nonzero(free))
        if cnt == 0:
            break

        h[free] += diff / float(cnt)
        h = np.clip(h, 0.0, 1.0)
        diff = target - float(np.sum(h))
        if abs(diff) <= 1e-10:
            break

    if abs(diff) > 1e-10:
        if diff > 0.0:
            lo = 0.0
            hi = 1.0
        else:
            lo = -1.0
            hi = 0.0

        for _ in range(50):
            mid = 0.5 * (lo + hi)
            if diff > 0.0:
                s2 = float(np.sum(np.minimum(h + mid, 1.0)))
            else:
                s2 = float(np.sum(np.maximum(h + mid, 0.0)))

            if s2 < target:
                lo = mid
            else:
                hi = mid

        tau = 0.5 * (lo + hi)
        if diff > 0.0:
            h = np.minimum(h + tau, 1.0)
        else:
            h = np.maximum(h + tau, 0.0)
        diff = target - float(np.sum(h))

    for _ in range(min(n, 8)):
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:
            idx = int(np.argmin(h))
            add = max(diff, -h[idx])
            h[idx] += add
            diff -= add

    return h


def corr_full_fast(h):
    n = len(h)
    if n == 0:
        return np.empty(0, dtype=float)

    x = sanitize_vector(h, n)

    if n <= 300:
        return np.correlate(x, 1.0 - x, mode="full")

    need = 2 * n - 1
    L = 1
    while L < need:
        L <<= 1

    A = np.fft.rfft(x, L)
    B = np.fft.rfft(1.0 - x, L)
    c = np.fft.irfft(A * np.conj(B), L)

    if n == 1:
        return c[0:1].copy()

    neg = c[n - 1:0:-1]
    pos = c[L - n + 1:L][::-1]
    return np.concatenate((neg, c[0:1], pos))


def objective_fast_max(h):
    n = len(h)
    if n == 0:
        return 0.0

    x = sanitize_vector(h, n)

    if n <= 300:
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = float(np.max(corr))
        return m * (2.0 / float(n))

    need = 2 * n - 1
    L = 1
    while L < need:
        L <<= 1

    A = np.fft.rfft(x, L)
    B = np.fft.rfft(1.0 - x, L)
    c = np.fft.irfft(A * np.conj(B), L)

    if n == 1:
        m = float(c[0])
    else:
        m = max(
            float(np.max(c[1:n])),
            float(np.max(c[L - n + 1:L])),
            float(c[0]),
        )

    if m < 0.0 and m > -1e-12:
        m = 0.0

    return m * (2.0 / float(n))


def objective_direct(h):
    n = len(h)
    if n == 0:
        return 0.0
    x = sanitize_vector(h, n)
    corr = np.correlate(x, 1.0 - x, mode="full")
    return float(np.max(corr) * (2.0 / float(n)))


def add_lag_gradient_into(h, offset, weight, grad):
    n = len(h)
    if weight == 0.0 or offset == None:
        return

    a = abs(int(offset))
    m = n - a
    if m <= 0:
        return

    if offset >= 0:
        grad[a:a + m] += weight * (1.0 - h[:m])
        grad[:m] -= weight * h[a:a + m]
    else:
        grad[:m] += weight * (1.0 - h[a:a + m])
        grad[a:a + m] -= weight * h[:m]


def smooth_top_grad(h, n, beta, K):
    corr = corr_full_fast(h)
    nf = len(corr)
    mcorr = float(np.max(corr))
    dx = 2.0 / float(n)

    if K < 1:
        K = 1
    if K > nf:
        K = nf

    if K < nf:
        part = np.argpartition(corr, -K)[-K:]
    else:
        part = np.arange(nf)

    vals = corr[part]
    vmax = float(np.max(vals))
    e = np.exp(float(beta) * (vals - vmax))
    s = float(np.sum(e))
    if s <= 0.0:
        return np.zeros(n, dtype=float), mcorr * dx

    w = e / s
    grad = np.zeros(n, dtype=float)

    for idx, wj in zip(part, w):
        if wj > 1e-12:
            off = int(idx) - (n - 1)
            add_lag_gradient_into(h, off, float(wj), grad)

    grad *= dx
    return grad, mcorr * dx


def projected_gradient_refine(h0, n, schedule, soft_limit_s=60.0):
    if h0 is None:
        return h0, None

    t0 = time.time()
    h = project_feasible(h0, n)
    c = objective_fast_max(h)
    best_h = h.copy()
    best_c = c

    for item in schedule:
        beta, K, iters = item
        target_move = 0.18
        stall = 0

        for it in range(iters):
            if remaining_time() < 25 or (time.time() - t0) > soft_limit_s:
                break

            grad, c = smooth_top_grad(h, n, beta, K)
            gn = float(np.linalg.norm(grad))
            if gn < 1e-13:
                break

            base_step = target_move / gn
            improved = False

            for mult in (1.0, 0.5, 0.25, 0.1, 2.0, 4.0, 0.05, 0.02):
                ht = project_feasible(h - (base_step * mult) * grad, n)
                ct = objective_fast_max(ht)
                if ct < c - 1e-13:
                    h = ht
                    c = ct
                    improved = True
                    break

            if improved:
                stall = 0
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = h.copy()
                target_move = min(0.5, target_move * 1.04)
            else:
                stall += 1
                target_move *= 0.7
                if stall > 30 or target_move < 1e-4:
                    break

    return best_h, best_c


def logsum_fun(x):
    n = _LOG_N
    beta = _LOG_BETA
    corr = corr_full_fast(x)
    mx = float(np.max(corr))
    shifted = beta * (corr - mx)
    e = np.exp(shifted)
    s = float(np.sum(e))
    if s <= 0.0:
        return float(mx * (2.0 / float(n)))
    return float((mx + np.log(s) / beta) * (2.0 / float(n)))


def logsum_jac(x):
    n = _LOG_N
    beta = _LOG_BETA
    dx = 2.0 / float(n)
    corr = corr_full_fast(x)
    mx = float(np.max(corr))
    e = np.exp(beta * (corr - mx))
    s = float(np.sum(e))
    if s <= 0.0:
        return np.zeros(n, dtype=float)

    w = e / s
    grad = np.zeros(n, dtype=float)
    for idx, wj in enumerate(w):
        if wj > 1e-10:
            off = int(idx) - (n - 1)
            add_lag_gradient_into(x, off, float(wj), grad)

    grad *= dx
    return grad


def logsum_cons(x):
    return float(np.sum(x) - 0.5 * _LOG_N)


def logsum_cons_jac(x):
    return np.full(_LOG_N, 1.0)


def logsum_refine(h0, n, beta, maxiter):
    global _LOG_N, _LOG_BETA
    if h0 is None:
        return h0, None

    h = project_feasible(h0, n)
    _LOG_N = int(n)
    _LOG_BETA = float(beta)

    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": logsum_cons, "jac": logsum_cons_jac}]

    try:
        res = minimize(
            logsum_fun,
            h,
            jac=logsum_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-10, "iprint": 0},
        )
        hp = project_feasible(res.x, n)
    except Exception:
        hp = h.copy()

    return hp, objective_fast_max(hp)


def active_obj_fun(x):
    return float(x[-1])


def active_obj_jac(x):
    g = np.zeros(len(x), dtype=float)
    g[-1] = 1.0
    return g


def active_sum_fun(x):
    return float(np.sum(x[:-1]) - 0.5 * _ACT_N)


def active_sum_jac(x):
    g = np.zeros(_ACT_N + 1, dtype=float)
    g[:-1] = 1.0
    return g


def active_ineq_fun(x):
    n = _ACT_N
    K = len(_ACT_IDX)
    if K == 0:
        return np.empty(0, dtype=float)

    h = x[:-1]
    t = float(x[-1])
    corr = corr_full_fast(h)
    out = t - (corr[_ACT_IDX] * (2.0 / float(n)))
    return out.astype(float, copy=False)


def active_ineq_jac(x):
    n = _ACT_N
    K = len(_ACT_IDX)
    dx = 2.0 / float(n)
    J = np.zeros((K, n + 1), dtype=float)
    h = x[:-1]

    for r, idx in enumerate(_ACT_IDX):
        J[r, -1] = 1.0
        off = int(idx) - (n - 1)
        add_lag_gradient_into(h, off, -dx, J[r, :-1])

    return J


def active_slsqp(h0, n, active_indices, maxiter):
    global _ACT_N, _ACT_IDX

    h = project_feasible(h0, n)
    if len(active_indices) == 0:
        return h.copy()

    _ACT_N = int(n)
    _ACT_IDX = np.asarray(active_indices, dtype=int)

    c0 = objective_fast_max(h)
    if not np.isfinite(c0):
        c0 = 0.5

    x0 = np.empty(n + 1, dtype=float)
    x0[:-1] = h
    x0[-1] = min(1.0, max(1e-8, c0 + 1e-7))

    bounds = [(0.0, 1.0)] * (n + 1)
    cons = [
        {"type": "ineq", "fun": active_ineq_fun, "jac": active_ineq_jac},
        {"type": "eq", "fun": active_sum_fun, "jac": active_sum_jac},
    ]

    try:
        res = minimize(
            active_obj_fun,
            x0,
            jac=active_obj_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-9, "iprint": 0},
        )
        hp = project_feasible(res.x[:-1], n)
    except Exception:
        hp = h.copy()

    return hp


def select_active(corr, dx, cmax, tol_c, maxK):
    nf = len(corr)
    if maxK > nf:
        maxK = nf
    if maxK < 1:
        maxK = 1

    top = np.argsort(corr, kind="mergesort")[::-1]
    base = top[:maxK].astype(int)

    thresh = max(0.0, float(cmax) - float(tol_c))
    cand = np.nonzero(corr * dx >= thresh)[0].astype(int)

    if cand.size == 0:
        return base

    combined = np.unique(np.concatenate((cand, base)))
    if combined.size > maxK:
        order = combined[np.argsort(corr[combined], kind="mergesort")[::-1]]
        combined = order[:maxK]

    return combined.astype(int)


def perturb_h(h, amt, rng):
    n = len(h)
    noise = rng.normal(loc=0.0, scale=amt, size=n)
    return project_feasible(h + noise, n)


def active_refine_multi(h0, n, rng, soft_limit_s=120.0):
    if h0 is None:
        return h0, None

    t0 = time.time()
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = objective_fast_max(h)

    if n <= 200:
        combos = [
            (0.0007, 12, 4),
            (0.0018, 18, 4),
            (0.0045, 26, 3),
            (0.0003, 8, 3),
            (0.0010, 14, 3),
            (0.0080, 34, 2),
        ]
        maxiter = 130
    elif n <= 340:
        combos = [
            (0.0009, 12, 3),
            (0.0024, 18, 3),
            (0.0055, 26, 2),
            (0.0005, 8, 3),
            (0.0090, 32, 2),
        ]
        maxiter = 100
    else:
        combos = [
            (0.0012, 10, 3),
            (0.0030, 16, 2),
            (0.0070, 22, 2),
            (0.0007, 8, 2),
        ]
        maxiter = 75

    for tol, K, cycles in combos:
        if (time.time() - t0) > soft_limit_s or remaining_time() < 25:
            break

        h_work = best_h.copy()
        for cyc in range(cycles):
            if (time.time() - t0) > soft_limit_s or remaining_time() < 25:
                break

            corr = corr_full_fast(h_work)
            dx = 2.0 / float(n)
            cmax = float(np.max(corr)) * dx
            cwork = objective_fast_max(h_work)
            if cwork < best_c - 1e-14:
                best_c = cwork
                best_h = h_work.copy()

            active = select_active(corr, dx, cmax, tol, K)
            if active.size < 2 and len(corr) >= 2:
                active = np.argsort(corr, kind="mergesort")[::-1][:min(8, len(corr))]
                active = active.astype(int)

            if active.size == 0:
                break

            h_new = active_slsqp(h_work, n, active, maxiter)
            c_new = objective_fast_max(h_new)

            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()

            if np.isfinite(c_new) and c_new <= cwork + 0.003:
                h_work = h_new

        if (time.time() - t0) > soft_limit_s or remaining_time() < 25:
            break

        hp = perturb_h(best_h, 0.02, rng)
        cp = objective_fast_max(hp)
        if cp < best_c + 0.001:
            corr = corr_full_fast(hp)
            dx = 2.0 / float(n)
            cmax = float(np.max(corr)) * dx
            active = select_active(corr, dx, cmax, 0.002, min(18, 2 * n - 1))
            if active.size >= 2:
                h_pp = active_slsqp(hp, n, active, max(40, maxiter // 2))
                cp = objective_fast_max(h_pp)
                if cp < best_c - 1e-14:
                    best_c = cp
                    best_h = h_pp.copy()

    return best_h, best_c


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n, dtype=float) + 0.5) * (2.0 / float(src_n))
    xn = (np.arange(dst_n, dtype=float) + 0.5) * (2.0 / float(dst_n))
    return np.interp(xn, xp, src)


def binarize_keep_count(h, n):
    h = sanitize_vector(h, n)
    out = np.zeros(n, dtype=float)
    if n > 0:
        k = n // 2
        if k > 0:
            idx = np.argpartition(h, -k)[-k:]
            out[idx] = 1.0
    return out


def h_to_mask(h, n):
    mask = 0
    hv = sanitize_vector(h, n) > 0.5
    for i in range(n):
        if hv[i]:
            mask |= (1 << i)
    return mask


def make_limit_list(n):
    limits = [0] * n
    for d in range(1, n):
        limits[d] = (1 << (n - d)) - 1
    return limits


def binary_maxN_limited(mask, n, limits, all_mask, threshold):
    zero = all_mask ^ mask
    best = 0
    for d in range(1, n):
        lim = limits[d]
        c1 = (mask & (zero >> d) & lim).bit_count()
        if c1 >= threshold:
            return threshold
        if c1 > best:
            best = c1

        c2 = ((mask >> d) & zero & lim).bit_count()
        if c2 >= threshold:
            return threshold
        if c2 > best:
            best = c2

    return best


def binary_grad_hill(h0, n, rng, soft_limit_s=60.0):
    if h0 is None or n > 800:
        return h0, (objective_fast_max(h0) if h0 is not None else None)

    t0 = time.time()
    h = binarize_keep_count(h0, n)
    mask = h_to_mask(h, n)
    if n > 0:
        all_mask = (1 << n) - 1
    else:
        all_mask = 0
    limits = make_limit_list(n)

    cN = binary_maxN_limited(mask, n, limits, all_mask, n + 1)
    c = cN * (2.0 / float(n))
    best_h = h.copy()
    best_c = c

    no_improve = 0

    while (time.time() - t0) < soft_limit_s and remaining_time() > 25:
        grad, curc = smooth_top_grad(h, n, 80.0, min(64, 2 * n - 1))

        ones = np.nonzero(h > 0.5)[0]
        zeros = np.nonzero(h < 0.5)[0]

        if ones.size == 0 or zeros.size == 0:
            break

        topk = min(10, int(ones.size), int(zeros.size))
        one_scores = grad[ones]
        zero_scores = grad[zeros]

        top_ones = ones[np.argsort(one_scores, kind="mergesort")[-topk:]]
        top_zeros = zeros[np.argsort(zero_scores, kind="mergesort")[:topk]]

        pairs = []
        for i in top_ones:
            for j in top_zeros:
                if i != j:
                    pairs.append((int(i), int(j)))

        rtop = max(6, topk)
        for _ in range(rtop):
            i = int(ones[int(rng.randint(0, ones.size))])
            j = int(zeros[int(rng.randint(0, zeros.size))])
            if i != j:
                pairs.append((i, j))

        bestN = cN
        best_pair = None
        seen = set()

        for i, j in pairs:
            if (i, j) in seen:
                continue
            seen.add((i, j))
            nm = (mask & ~(1 << i)) | (1 << j)
            nn = binary_maxN_limited(nm, n, limits, all_mask, cN)
            if nn < bestN:
                bestN = nn
                best_pair = (i, j)

            if bestN <= 0:
                break

        if best_pair is None:
            for _ in range(90):
                if not np.isfinite(curc) and curc > best_c:
                    break
                i = int(ones[int(rng.randint(0, ones.size))])
                j = int(zeros[int(rng.randint(0, zeros.size))])
                if i == j:
                    continue

                nm = (mask & ~(1 << i)) | (1 << j)
                nn = binary_maxN_limited(nm, n, limits, all_mask, cN)
                if nn < bestN:
                    bestN = nn
                    best_pair = (i, j)
                if bestN <= 0:
                    break
        else:
            pass

        if best_pair is not None:
            i, j = best_pair
            mask = (mask & ~(1 << i)) | (1 << j)
            h[i] = 0.0
            h[j] = 1.0
            cN = bestN
            c = cN * (2.0 / float(n))
            no_improve = 0
            if c < best_c - 1e-15:
                best_c = c
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve >= 18:
                break

    h = project_feasible(h, n)
    best_h = project_feasible(best_h, n)
    return best_h, objective_fast_max(best_h)


def thue_morse_h(n):
    h = np.zeros(n, dtype=float)
    for i in range(n):
        v = i
        parity = 0
        while v:
            parity ^= (v & 1)
            v >>= 1
        if parity == 1:
            h[i] = 1.0
    return h


def default_h_values(n, seed):
    h = thue_morse_h(n)
    return project_feasible(h, n)


def candidate_sort_key(item):
    return item[1]


def add_candidate(cands, h, c, n, limit=18):
    if h is None or c is None:
        return
    cands.append((h.copy(), float(c), int(n)))
    cands.sort(key=candidate_sort_key)
    if len(cands) > limit:
        del cands[limit:]


def consider_solution(best_h, best_c, best_n, h, c_param, n_param, cands):
    if h is None:
        return best_h, best_c, best_n

    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(h.size)
    if n == 0:
        return best_h, best_c, best_n

    hf = project_feasible(h, n)
    cf = objective_fast_max(hf)
    add_candidate(cands, hf, cf, n)

    if cf < best_c - 1e-14:
        return hf.copy(), cf, n

    return best_h, best_c, best_n


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S

    _START_TIME = time.time()
    _BUDGET_S = float(max(10.0, budget_s))

    best_h = None
    best_c = float("inf")
    best_n = 0
    cands = []

    try:
        rng = np.random.RandomState(seed)

        parent_h = None
        try:
            parent_h = np.array(initial_h_values, dtype=float)
        except Exception:
            parent_h = None

        if parent_h is not None and parent_h.size > 0:
            best_h, best_c, best_n = consider_solution(
                best_h, best_c, best_n, parent_h, None, len(parent_h), cands
            )
        else:
            h0 = default_h_values(160, seed)
            best_h, best_c, best_n = consider_solution(
                best_h, best_c, best_n, h0, None, 160, cands
            )

        if best_h is None:
            h0 = default_h_values(160, seed)
            best_h, best_c, best_n = consider_solution(
                best_h, best_c, best_n, h0, None, 160, cands
            )

        # Refine the parent / incumbent.
        if remaining_time() > 70:
            h_ref, c_ref = active_refine_multi(
                best_h, best_n, rng,
                soft_limit_s=min(180.0, max(30.0, remaining_time() - 60.0))
            )
            best_h, best_c, best_n = consider_solution(
                best_h, best_c, best_n, h_ref, c_ref, best_n, cands
            )

        if remaining_time() > 50:
            if best_n <= 260:
                sched = [(25, 90, 80), (80, 90, 80), (300, 60, 70), (1200, 30, 50)]
            elif best_n <= 420:
                sched = [(40, 80, 70), (150, 60, 60), (600, 36, 50)]
            else:
                sched = [(60, 64, 55), (250, 40, 45), (1000, 24, 35)]

            h_ref, c_ref = projected_gradient_refine(
                best_h, best_n, sched,
                soft_limit_s=min(70.0, max(20.0, remaining_time() - 30.0))
            )
            best_h, best_c, best_n = consider_solution(
                best_h, best_c, best_n, h_ref, c_ref, best_n, cands
            )

        if remaining_time() > 50 and best_n <= 340:
            for beta, maxit in [(150, 130), (600, 110), (2500, 90)]:
                if remaining_time() < 25:
                    break
                h_ref, c_ref = logsum_refine(best_h, best_n, beta, maxit)
                best_h, best_c, best_n = consider_solution(
                    best_h, best_c, best_n, h_ref, c_ref, best_n, cands
                )

        # Try higher / other resolutions.
        all_res = [192, 224, 256, 288, 320, 384, 448, 512]
        if remaining_time() > 360:
            all_res.append(640)

        for r in all_res:
            if r < 160:
                continue
            if r <= best_n + 1:
                continue
            if remaining_time() < 170:
                break

            h_interp = interp_to(best_h, best_n, r)
            h_interp = project_feasible(h_interp, r)
            best_h, best_c, best_n = consider_solution(
                best_h, best_c, best_n, h_interp, None, r, cands
            )

            if remaining_time() > 120:
                soft = 70.0 if r <= 340 else 90.0
                h_ref, c_ref = active_refine_multi(
                    h_interp, r, rng,
                    soft_limit_s=min(soft, max(20.0, remaining_time() - 60.0))
                )
                best_h, best_c, best_n = consider_solution(
                    best_h, best_c, best_n, h_ref, c_ref, r, cands
                )

            if remaining_time() > 70:
                if r <= 300:
                    sched = [(40, 80, 60), (150, 60, 55), (600, 36, 45)]
                elif r <= 520:
                    sched = [(50, 64, 50), (250, 40, 45), (1000, 24, 30)]
                else:
                    sched = [(60, 56, 40), (300, 32, 35), (1200, 18, 25)]

                h_ref, c_ref = projected_gradient_refine(
                    best_h, r, sched, soft_limit_s=35.0
                )
                best_h, best_c, best_n = consider_solution(
                    best_h, best_c, best_n, h_ref, c_ref, r, cands
                )

            if remaining_time() > 50 and r <= 340:
                h_ref, c_ref = logsum_refine(best_h, r, 800, 90)
                best_h, best_c, best_n = consider_solution(
                    best_h, best_c, best_n, h_ref, c_ref, r, cands
                )

            if remaining_time() > 100 and r <= 700:
                h_bin, c_bin = binary_grad_hill(
                    best_h, r, rng, soft_limit_s=min(45.0, remaining_time() - 50.0)
                )
                best_h, best_c, best_n = consider_solution(
                    best_h, best_c, best_n, h_bin, c_bin, r, cands
                )

                if best_n == r and remaining_time() > 50:
                    h_bin, c_bin = binary_grad_hill(
                        best_h, r, rng, soft_limit_s=min(35.0, remaining_time() - 40.0)
                    )
                    best_h, best_c, best_n = consider_solution(
                        best_h, best_c, best_n, h_bin, c_bin, r, cands
                    )

                # Leave the integer/binary ridge with small gradient steps.
                if best_n == r and remaining_time() > 45:
                    if r <= 340:
                        sched = [(20, 80, 45), (80, 60, 40), (300, 32, 30)]
                    else:
                        sched = [(30, 64, 35), (120, 40, 30), (500, 20, 20)]

                    h_ref, c_ref = projected_gradient_refine(
                        best_h, r, sched, soft_limit_s=30.0
                    )
                    best_h, best_c, best_n = consider_solution(
                        best_h, best_c, best_n, h_ref, c_ref, r, cands
                    )

                    if remaining_time() > 35 and r <= 340:
                        h_ref, c_ref = active_refine_multi(
                            best_h, r, rng, soft_limit_s=30.0
                        )
                        best_h, best_c, best_n = consider_solution(
                            best_h, best_c, best_n, h_ref, c_ref, r, cands
                        )

        # Final polishing of the best seen candidate.
        if remaining_time() > 60:
            h_ref, c_ref = active_refine_multi(
                best_h, best_n, rng,
                soft_limit_s=min(80.0, max(25.0, remaining_time() - 50.0))
            )
            best_h, best_c, best_n = consider_solution(
                best_h, best_c, best_n, h_ref, c_ref, best_n, cands
            )

        if remaining_time() > 35:
            if best_n <= 340:
                sched = [(300, 48, 45), (1500, 28, 40), (5000, 18, 35)]
            else:
                sched = [(300, 40, 35), (1500, 24, 30), (5000, 14, 25)]

            h_ref, c_ref = projected_gradient_refine(
                best_h, best_n, sched, soft_limit_s=min(50.0, remaining_time() - 25.0)
            )
            best_h, best_c, best_n = consider_solution(
                best_h, best_c, best_n, h_ref, c_ref, best_n, cands
            )

        if remaining_time() > 25 and best_n <= 512:
            h_ref, c_ref = logsum_refine(best_h, best_n, 3000, 110)
            best_h, best_c, best_n = consider_solution(
                best_h, best_c, best_n, h_ref, c_ref, best_n, cands
            )

        if len(cands) == 0:
            best_h = project_feasible(best_h, best_n)
            add_candidate(cands, best_h, objective_fast_max(best_h), best_n)

        # Choose the best candidate by a final direct/authoritative evaluation.
        final_h = None
        final_c = float("inf")
        final_n = best_n

        for hh, cc, nn in cands:
            if remaining_time() < 2:
                break
            hh = project_feasible(hh, nn)
            if nn <= 4000:
                ce = objective_direct(hh)
            else:
                ce = objective_fast_max(hh) + 1e-12
            if ce < final_c - 1e-15:
                final_c = ce
                final_h = hh
                final_n = nn

        if final_h is None and best_h is not None:
            final_h = project_feasible(best_h, best_n)
            final_c = objective_direct(final_h) if best_n <= 4000 else objective_fast_max(final_h)
            final_n = best_n

        eval_func = globals().get("evaluate_erdos_solution", None)
        if eval_func is not None and final_h is not None and remaining_time() > 3:
            try:
                ev = float(eval_func(final_h))
                if np.isfinite(ev):
                    final_c = min(final_c, ev)
            except Exception:
                pass

        final_h = project_feasible(final_h, final_n)
        if final_n <= 5000:
            direct_c = objective_direct(final_h)
        else:
            direct_c = objective_fast_max(final_h)

        if np.isfinite(direct_c) and direct_c < final_c:
            final_c = direct_c

        if not np.isfinite(final_c):
            final_c = 1.0

        return final_h, float(final_c), int(final_n)

    except Exception:
        if best_h is None:
            best_n = 160
            best_h = default_h_values(best_n, seed)
        else:
            best_h = project_feasible(best_h, best_n)

        if best_n <= 4000:
            final_c = objective_direct(best_h)
        else:
            final_c = objective_fast_max(best_h)

        if not np.isfinite(final_c):
            final_c = 1.0

        return best_h, float(final_c), int(best_n)
```
