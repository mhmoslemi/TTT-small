    """
    h: array of length n
    weights: array of length 2n-1 (corresponding to shifts -(n-1) to n-1)
    Returns gradient of sum_s weights[s] * C_s with respect to h
    """

        s = s_idx - (n-1)  # actual shift
        if s >= 0:
