    return best_h_local, best_c5_local


def coord_descent_phase(h, n, rng, t0, budget_s, best_c5_in):
    target = n / 2.0
    best_c5 = best_c5_in
    best_h = h.copy()
    deltas = [0.003, 0.001, 0.005, 0.002, 0.01, 0.0015]

    while budget_s - (time.time() - t0) > 5.0:
        improved = False
        num_trials = 3000
        for _ in range(num_trials):
            if budget_s - (time.time() - t0) < 3.0:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi = h[i]
            hj = h[j]
            if hi < hj:
                di = 1
            else:
                di = -1
            for delta in deltas:
                ni = hi + di * delta
                nj = hj - di * delta
                if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                    continue
                h[i] = ni
                h[j] = nj
                c5_test = compute_c5(h)
                if c5_test < best_c5 - 1e-14:
                    best_c5 = c5_test
                    best_h = h.copy()
                    improved = True
                    break
                h[i] = hi
                h[j] = hj
        if not improved:
            break

    return best_h, best_c5


def smoothed_max_obj(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k < len(corr):
        idx = np.argsort(corr)[-top_k:]
        m = corr[idx].max()
        shifted = corr[idx] - m
        val = (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx
    else:
        m = corr.max()
        shifted = corr - m
        val = (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx
    return val


def slsqp_phase(h, n, t0, budget_s, best_c5_in):
    target = n / 2.0
    dx = 2.0 / n
    best_c5 = best_c5_in
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n

    def obj_fn(x):
        return smoothed_max_obj(x, 2000, 10)

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    for beta in [500, 1500, 3000]:
        if budget_s - (time.time() - t0) < 8.0:
            break

        def obj_fn2(x):
            return smoothed_max_obj(x, beta, 8)

        try:
            res = minimize(obj_fn2, best_h, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                           maxiter=300, ftol=1e-18, options={'iprint': 0})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_par = np.array(initial_h_values, dtype=float)
    n_par = len(h_par)
    c5_par = compute_c5(h_par)
    print("Parent: n={}, C5={:.6f}".format(n_par, c5_par))

    best_h = h_par.copy()
    best_c5 = c5_par
    best_n = n_par

    # Phase 1: LP-based optimization at parent resolution
    h1, c5_1 = lp_optimize_phase(best_h.copy(), best_n, t0, budget_s)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    print("After LP n={}: C5={:.6f}".format(best_n, best_c5))

    # Phase 2: SLSQP polish
    if budget_s - (time.time() - t0) > 30:
        h2, c5_2 = slsqp_phase(best_h.copy(), best_n, t0, budget_s, best_c5)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        print("After SLSQP n={}: C5={:.6f}".format(best_n, best_c5))

    # Phase 3: Try higher resolutions
    for n_try in [180, 200, 220, 240, 260, 300]:
        if budget_s - (time.time() - t0) < 80:
            break
        if n_try % 2 != 0:
            continue
        target = n_try / 2.0
        xp = (np.arange(best_n) + 0.5) * 2.0 / best_n
        xn = (np.arange(n_try) + 0.5) * 2.0 / n_try
        h0 = np.interp(xn, xp, best_h)
        h0 = project_feasible(h0, target)
        c5_0 = compute_c5(h0)

        h3, c5_3 = lp_optimize_phase(h0, n_try, t0, budget_s)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = n_try
            print("  n={}: C5={:.6f} (improved from interp {:.6f})".format(n_try, c5_3, c5_0))
        else:
            print("  n={}: best {:.6f}, lp gave {:.6f} (no improvement)".format(n_try, best_c5, c5_3))

    # Phase 4: Coordinate descent at best resolution
    if budget_s - (time.time() - t0) > 20:
        h4, c5_4 = coord_descent_phase(best_h.copy(), best_n, rng, t0, budget_s, best_c5)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
        print("After coord descent: C5={:.6f}".format(best_c5))

    # Phase 5: One more LP round at best resolution
    if budget_s - (time.time() - t0) > 20:
        h5, c5_5 = lp_optimize_phase(best_h.copy(), best_n, t0, budget_s)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
        print("After final LP: C5={:.6f}".format(best_c5))

    # Phase 6: Very fine coordinate moves
    if budget_s - (time.time() - t0) > 10:
        target = best_n / 2.0
        h = best_h.copy()
        fine_deltas = [0.001, 0.0005, 0.0015, 0.002]
        for sweep in range(100):
            if budget_s - (time.time() - t0) < 4:
                break
            improved = False
            for _ in range(3000):
                if budget_s - (time.time() - t0) < 3:
                    break
                i = rng.randint(best_n)
                j = rng.randint(best_n)
                if i == j:
                    continue
                hi, hj = h[i], h[j]
                di = 1 if hi < hj else -1
                for delta in fine_deltas:
                    ni, nj = hi + di * delta, hj - di * delta
                    if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                        continue
                    h[i], h[j] = ni, nj
                    c5_test = compute_c5(h)
                    if c5_test < best_c5 - 1e-15:
                        best_c5 = c5_test
                        best_h = h.copy()
                        improved = True
                        break
                    h[i], h[j] = hi, hj
            if not improved and sweep > 5:
                break

    # Final
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
