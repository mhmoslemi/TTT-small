    return np.interp(x_new, xp, src_h)

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    # Adjust to exactly n/2 ones
    num_ones = int(np.sum(h))
    target = n // 2
    if num_ones > target:
        idx = np.where(h == 1.0)[0]
        for i in range(num_ones - target):
            h[idx[i]] = 0.0
    elif num_ones < target:
        idx = np.where(h == 0.0)[0]
        for i in range(target - num_ones):
            h[idx[i]] = 1.0
    return h

def make_lfsr(n, seed=7):
    """Generate m-sequence (maximal length sequence)"""
    # Use a simple LFSR
    state = 1
    h = np.zeros(n)
    for i in range(n):
        h[i] = float(state & 1)
        # Feedback: taps at positions 0 and 3 (for 4-bit) or 0 and 4 (for 5-bit)
        # Let's use a simpler approach: xorshift
        state ^= (state << 13) & 0xFFFFFFFF
        state ^= (state >> 17)
        state ^= (state << 5) & 0xFFFFFFFF
        state &= 0xFFFFFFFF
    # Adjust to n/2 ones
    num_ones = int(np.sum(h))
    target = n // 2
    if num_ones > target:
        idx = np.where(h == 1.0)[0]
        rng_tmp = np.random.RandomState(seed)
        perm = rng_tmp.permutation(len(idx))[:num_ones - target]
        h[idx[perm]] = 0.0
    elif num_ones < target:
        idx = np.where(h == 0.0)[0]
        rng_tmp = np.random.RandomState(seed)
        perm = rng_tmp.permutation(len(idx))[:target - num_ones]
        h[idx[perm]] = 1.0
    return h

def make_sine(n, freq, phase):
    t = np.arange(n) / float(n)
    h = 0.5 + 0.3 * np.sin(2.0 * np.pi * freq * t + phase)
    return np.clip(h, 0.0, 1.0)

def binary_sa(h_init, n, iterations, rng, t_start, t_end):
    h = np.where(h_init > 0.5, 1.0, 0.0)
    num_ones = int(np.sum(h))
    target_ones = n // 2
    if num_ones > target_ones:
        ones_idx = np.where(h == 1.0)[0]
        remove = rng.choice(len(ones_idx), num_ones - target_ones, replace=False)
        h[ones_idx[remove]] = 0.0
    elif num_ones < target_ones:
        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
        h[zeros_idx[add]] = 1.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    
    for it in range(iterations):
        if remaining_time() < 5:
            break
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        
        T = t_start * (t_end / t_start) ** (float(it) / max(float(iterations), 1))
        
        # Move 1: single swap
        if rng.rand() < 0.7:
            i = rng.randint(len(ones_list))
            j = rng.randint(len(zeros_list))
            oi, oz = ones_list[i], zeros_list[j]
            h[oi], h[oz] = 0.0, 1.0
            new_c5 = compute_c5(h)
            if new_c5 < current_c5 or (T > 1e-15 and rng.rand() < np.exp(-(new_c5 - current_c5) / T)):
                current_c5 = new_c5
                ones_list[i] = oz
                zeros_list[j] = oi
                if new_c5 < best_c5:
                    best_c5, best_h = new_c5, h.copy()
            else:
                h[oi], h[oz] = 1.0, 0.0
        else:
            # Move 2: double swap (swap two 1s positions)
            if len(ones_list) < 2:
                i = rng.randint(len(ones_list))
                j = rng.randint(len(zeros_list))
                oi, oz = ones_list[i], zeros_list[j]
                h[oi], h[oz] = 0.0, 1.0
                new_c5 = compute_c5(h)
                if new_c5 < current_c5 or (T > 1e-15 and rng.rand() < np.exp(-(new_c5 - current_c5) / T)):
                    current_c5 = new_c5
                    ones_list[i] = oz
                    zeros_list[j] = oi
                    if new_c5 < best_c5:
                        best_c5, best_h = new_c5, h.copy()
                else:
                    h[oi], h[oz] = 1.0, 0.0
                continue
            
            i1, i2 = rng.choice(len(ones_list), 2, replace=False)
            j1, j2 = rng.choice(len(zeros_list), 2, replace=False)
            oi1, oi2 = ones_list[i1], ones_list[i2]
            oz1, oz2 = zeros_list[j1], zeros_list[j2]
            h[oi1], h[oi2] = 0.0, 0.0
            h[oz1], h[oz2] = 1.0, 1.0
            new_c5 = compute_c5(h)
            if new_c5 < current_c5 or (T > 1e-15 and rng.rand() < np.exp(-(new_c5 - current_c5) / T)):
                current_c5 = new_c5
                ones_list[i1], ones_list[i2] = oz1, oz2
                zeros_list[j1], zeros_list[j2] = oi1, oi2
                if new_c5 < best_c5:
                    best_c5, best_h = new_c5, h.copy()
            else:
                h[oi1], h[oi2] = 1.0, 1.0
                h[oz1], h[oz2] = 0.0, 0.0
    
    return best_h, best_c5

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        unsat = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(unsat)
        if cnt > 0:
            h[unsat] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mid = np.argsort(np.abs(h - 0.5))
        for idx in mid:
            if h[idx] + diff >= -1e-12 and h[idx] + diff <= 1.0 + 1e-12:
                h[idx] += diff
                break
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    num_ones = int(np.sum(h))
    target = n // 2
    if num_ones > target:
        idx = np.where(h == 1.0)[0]
        for i in range(num_ones - target):
            h[idx[i]] = 0.0
    elif num_ones < target:
        idx = np.where(h == 0.0)[0]
        for i in range(target - num_ones):
            h[idx[i]] = 1.0
    return h


def make_xorshift(n, seed_val):
    state = seed_val | 1
    h = np.zeros(n)
    for i in range(n):
        h[i] = float(state & 1)
        state ^= (state << 13) & 0xFFFFFFFF
        state ^= (state >> 17)
        state ^= (state << 5) & 0xFFFFFFFF
        state &= 0xFFFFFFFF
    num_ones = int(np.sum(h))
    target = n // 2
    rng_t = np.random.RandomState(seed_val)
    if num_ones > target:
        idx = np.where(h == 1.0)[0]
        perm = rng_t.permutation(len(idx))[:num_ones - target]
        h[idx[perm]] = 0.0
    elif num_ones < target:
        idx = np.where(h == 0.0)[0]
        perm = rng_t.permutation(len(idx))[:target - num_ones]
        h[idx[perm]] = 1.0
    return h


def make_sine(n, freq, phase):
    t = np.arange(n) / float(n)
    h = 0.5 + 0.3 * np.sin(2.0 * np.pi * freq * t + phase)
    return np.clip(h, 0.0, 1.0)


def multi_sine(n, freqs, phases, rng):
    t = np.arange(n) / float(n)
    h = np.full(n, 0.5)
    for f, ph in zip(freqs, phases):
        amp = rng.uniform(0.08, 0.2)
        h += amp * np.sin(2.0 * np.pi * f * t + ph)
    return np.clip(h, 0.0, 1.0)


def binary_sa(h_init, n, iterations, rng, t_start, t_end):

        h[ones_idx[remove]] = 0.0
    elif num_ones < target_ones:
        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
        h[zeros_idx[add]] = 1.0

    current_c5 = best_c5

            oz1, oz2 = zeros_list[j1], zeros_list[j2]
            h[oi1], h[oi2] = 0.0, 0.0
            h[oz1], h[oz2] = 1.0, 1.0
            new_c5 = compute_c5(h)
            if new_c5 < current_c5 or (T > 1e-15 and rng.rand() < np.exp(-(new_c5 - current_c5) / T)):
                current_c5 = new_c5
                ones_list[i1], ones_list[i2] = oz1, oz2
                zeros_list[j1], zeros_list[j2] = oi1, oi2
                if new_c5 < best_c5:
                    best_c5, best_h = new_c5, h.copy()
            else:
                h[oi1], h[oi2] = 1.0, 1.0
                h[oz1], h[oz2] = 0.0, 0.0
        else:
            i = rng.randint(len(ones_list))
            j = rng.randint(len(zeros_list))
            oi, oz = ones_list[i], zeros_list[j]
            h[oi], h[oz] = 0.0, 1.0

                if new_c5 < best_c5:

    return best_h, best_c5


def coordinate_descend(h, n, rng, sweeps):
    dx_val = 2.0 / n
    c5_cur = compute_c5(h)
    for sweep in range(sweeps):
        if remaining_time() < 3:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if remaining_time() < 2:
                break
            old_val = h[i]
            best_local = c5_cur
            best_val = old_val
            h[i] = old_val  # ensure
            candidates = [0.0, 1.0, 0.5]
            if old_val > 0.25:
                candidates.append(0.1)
            if old_val < 0.75:
                candidates.append(0.9)
            candidates.append(max(0.0, old_val - 0.15))
            candidates.append(min(1.0, old_val + 0.15))
            for nv in candidates:
                if abs(nv - old_val) < 1e-8:
                    continue
                h[i] = nv
                h_feas = project_feasible(h, n)
                c5_test = compute_c5(h_feas)
                if c5_test < best_local:
                    best_local = c5_test
                    best_val = nv
                h[i] = old_val
            if best_val != old_val:
                h[i] = best_val
                h = project_feasible(h, n)
                c5_cur = best_local
                any_imp = True
        if not any_imp:
            break
    return h, c5_cur

                if s >= 0:
                    grad[s:] += (1.0 - h[:n - s])
                    grad[:n - s] -= h[s:]
                else:
                    sp = -s
                    grad[:n - sp] += (1.0 - h[sp:])
                    grad[sp:] -= h[:n - sp]

            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                break
            grad /= gn

            improved = False
            for step in [0.08, 0.04, 0.15, 0.02, 0.1, 0.25, 0.01]:
                h_t = h - step * grad
                h_t = project_feasible(h_t, n)
                c5_t = compute_c5(h_t)
                if c5_t < best_c5:
                    best_c5, best_h, best_n = c5_t, h_t.copy(), n
                    h = h_t.copy()
                    improved = True
                    break

            if not improved:
                if remaining_time() > 15:
                    # Coordinate descent
                    h_cd, c5_cd = coordinate_descend(h, n, rng, 3)
                    if c5_cd < best_c5:
                        best_c5, best_h, best_n = c5_cd, h_cd.copy(), n
                        h = h_cd.copy()
                    else:
                        # Try L-BFGS restart
                        try:
                            res = minimize(smooth_max_obj, h, jac=smooth_max_grad, args=(1500,),
                                          method='L-BFGS-B',
                                          bounds=[(0.0, 1.0)] * n,
                                          options={'maxiter': 200, 'ftol': 1e-16, 'iprint': 0})
                            h_new = project_feasible(res.x, n)
                            c5_new = compute_c5(h_new)
                            if c5_new < best_c5:
                                best_c5, best_h = c5_new, h_new.copy()
                                h = h_new.copy()
                            else:
                                break
                        except Exception:
                            break
                else:
                    break

    # Phase 5: Final high-precision polish
    if remaining_time() > 10:
        n = best_n
        print("\n--- Final polish (n={}) ---".format(n))
        for beta in [3000, 8000, 15000]:
            if remaining_time() < 4:
                break
            try:
                res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad, args=(beta,),
                              method='L-BFGS-B',
                              bounds=[(0.0, 1.0)] * n,
                              options={'maxiter': 800, 'ftol': 1e-16, 'gtol': 1e-12,
                                       'iprint': 0, 'maxls': 200})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5, best_h = c5_new, h_new.copy()
                    print("  beta={}: C5={:.7f}".format(beta, c5_new))
            except Exception:
                pass

        # Coordinate descent final
        if remaining_time() > 8:
            h_cd, c5_cd = coordinate_descend(best_h, n, rng, 5)
            if c5_cd < best_c5:
                best_c5, best_h = c5_cd, h_cd.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
