```python
    top_idx = np.argsort(weights)[::-1][:top_lags]
    top_w = weights[top_idx]
    top_w = top_w / top_w.sum()
    
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        w = top_w[idx]
        if w < 1e-15:
            continue
        k = int(top_idx[idx]) - (n - 1)
        if k >= 0:
            # grad[k:n] += (1-h[0:n-k])
            # grad[0:n-k] += -h[k:n]
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] += -w * h[k:]
        else:
            m_abs = -k
            # grad[0:n-m_abs] += (1-h[m_abs:n])
            # grad[m_abs:n] += -h[0:n-m_abs]
            grad[:n-m_abs] += w * (1.0 - h[m_abs:])
            grad[m_abs:] += -w * h[:n-m_abs]
    
    return grad * dx, corr
```

```python
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = np.where(h < 1.0 - 1e-12)[0]
            else:
                idx = np.where(h > 1e-12)[0]
            if len(idx) == 0:
                break
            per = diff / len(idx)
            h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-15:
            for j in range(n):
                if 0 < h[j] < 1:
                    h[j] = np.clip(h[j] + diff2, 0.0, 1.0)
                    break
    return h

def smooth_max_value(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def gradient_smooth_max(h, beta, top_lags=12):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    weights = exp_s / total
    top_idx = np.argsort(weights)[::-1][:top_lags]
    top_w = weights[top_idx]
    sw = top_w.sum()
    if sw < 1e-30:
        return np.zeros(n), corr
    top_w = top_w / sw
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        w = top_w[idx]
        if w < 1e-15:
            continue
        k = int(top_idx[idx]) - (n - 1)
        if k >= 0:
            np = n - k
            if np > 0:
                grad[k:] += w * (1.0 - h[:np])
                grad[:np] += -w * h[k:]
        else:
            ma = -k
            np = n - ma
            if np > 0:
                grad[:np] += w * (1.0 - h[ma:])
                grad[ma:] += -w * h[:np]
    return grad * dx, corr

def proj_grad_optimize(h, n, beta, top_lags, max_iter, step_base):
    h_cur = h.copy()

    for it in range(max_iter):

    return h_cur, best_local

    def obj(x):
        return smooth_max_value(x, beta)

        g, _ = gradient_smooth_max(x, beta, top_lags=30)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B',
                      bounds=bounds, maxiter=maxiter,
                      options={'ftol': 1e-18, 'gtol': 1e-10, 'maxfun': maxiter * 5})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_refine(h, n, beta, maxiter):
    def obj(x):
        return smooth_max_value(x, beta)
    def obj_grad(x):
        g, _ = gradient_smooth_max(x, beta, top_lags=50)
        return g
    def con_fun(x):

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})

        return h.copy()
```
