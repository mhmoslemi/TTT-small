∂/∂h[1] = (1-h[1-1]) [if 0<=0<3: YES] - h[1+1] [if 0<=2<3: YES] = (1-h[0]) - h[2] ✓
∂/∂h[2] = (1-h[2-1]) [if 0<=1<3: YES] - h[2+1] [if 0<=3<3: NO] = (1-h[1]) - 0 = (1-h[1]) ✓

So the gradient formula is:
∂corr[j]/∂h[i] = (1-h[i-k]) * [0<=i-k<n] - h[i+k] * [0<=i+k<n]
where k = j - (n-1)

This is great! Now I can compute the gradient of the soft-max efficiently.

For the soft-max objective:
f(h) = sm(corr) * dx where sm(corr) = max(corr) + log(sum(exp(beta*(corr - max(corr)))))/beta

    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    c5 = np.max(corr) * dx
    return c5, corr

def project_sum(h, n, tol=1e-14, max_iter=500):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(max_iter):
        diff = target - np.sum(h)
        if abs(diff) < tol:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nfree = np.sum(free)
        if nfree == 0:
            # All saturated, can't project properly
            break
        h[free] += diff / nfree
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj_and_grad(h, n, beta, lam):
    """Compute soft-max objective and its gradient.
    Objective: sm(corr)*dx + lam*(sum(h)-n/2)^2
    """
    dx = 2.0 / n
    corr = compute_corr(h)
    mx = np.max(corr)
    
    # Soft max
    if beta < 1e-6:
        sm = mx
        weights = np.zeros(2*n-1)
        weights[np.argmax(corr)] = 1.0
    else:
        shifted = beta * (corr - mx)
        shifted = np.clip(shifted, -100, 100)
        exps = np.exp(shifted)
        sum_exps = np.sum(exps)
        sm = mx + np.log(sum_exps) / beta
        weights = exps / sum_exps
    
    # Gradient of soft_max wrt h
    # d(sm)/d(h[i]) = sum_j weights[j] * d(corr[j])/d(h[i])
    # d(corr[j])/d(h[i]) = (1-h[i-k]) * [0<=i-k<n] - h[i+k] * [0<=i+k<n]
    # where k = j - (n-1)
    
    grad_sm = np.zeros(n)
    for j in range(2*n-1):
        w = weights[j]
        if w < 1e-10:
            continue
        k = j - (n - 1)
        if k >= 0:
            # Positive lag: corr[j] = sum_m h[m+k]*(1-h[m]) for m=0..n-1-k
            # d/dh[i] = (1-h[i-k])[i>=k] - h[i+k][i<=n-1-k]
            # Term 1: (1-h[i-k]) for i = k, k+1, ..., n-1
            grad_sm[k:] += w * (1.0 - h[:n-k])
            # Term 2: -h[i+k] for i = 0, 1, ..., n-1-k
            grad_sm[:n-k] -= w * h[k:]
        else:
            # Negative lag: k = -|k|
            # corr[j] = sum_m h[m]*(1-h[m+|k|]) for m=0..n-1-|k|
            # d/dh[i] = (1-h[i+|k|])[i<=n-1-|k|] - h[i-|k|])[i>=|k|]
            ak = -k  # absolute value
            # Term 1: (1-h[i+ak]) for i = 0, 1, ..., n-1-ak
            grad_sm[:n-ak] += w * (1.0 - h[ak:])
            # Term 2: -h[i-ak] for i = ak, ak+1, ..., n-1
            grad_sm[ak:] -= w * h[:n-ak]
    
    grad = dx * grad_sm
    
    # Penalty gradient
    pen_grad = 2.0 * lam * (np.sum(h) - n/2.0)
    grad += pen_grad * np.ones(n)
    
    obj = sm * dx + lam * (np.sum(h) - n/2.0)**2
    return obj, grad

import time

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = _corr_full(h)
    return (np.max(corr) * dx, corr)


def _project(h, n):
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        nf = np.count_nonzero(free)
        if nf == 0:
            break
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h


def _softmax_eval(h, n, beta, lam):
    dx = 2.0 / n
    corr = _corr_full(h)
    mx = np.max(corr)
    if beta < 1e-8:
        sm = mx
    else:
        sh = beta * (corr - mx)
        sh = np.clip(sh, -80, 80)
        sm = mx + np.log(np.sum(np.exp(sh))) / beta
    pen = lam * (np.sum(h) - n / 2.0) ** 2
    return sm * dx + pen


def _softmax_grad(h, n, beta, lam):
    dx = 2.0 / n
    corr = _corr_full(h)
    mx = np.max(corr)
    nj = 2 * n - 1
    if beta < 1e-8:
        weights = np.zeros(nj)
        weights[np.argmax(corr)] = 1.0
    else:
        sh = beta * (corr - mx)
        sh = np.clip(sh, -80, 80)
        exps = np.exp(sh)
        se = np.sum(exps)
        weights = exps / se

    grad = np.zeros(n)
    center = n - 1
    for j in range(nj):
        w = weights[j]
        if w < 1e-12:
            continue
        k = j - center
        if k >= 0:
            np_ = n - k
            grad[k:] += w * (1.0 - h[:np_])
            grad[:np_] -= w * h[k:]
        else:
            ak = -k
            np_ = n - ak
            grad[:np_] += w * (1.0 - h[ak:])
            grad[ak:] -= w * h[:np_]
    grad *= dx
    pen_g = 2.0 * lam * (np.sum(h) - n / 2.0)
    grad += pen_g
    return grad

            if k >= 0:

                grad[:np_] -= h[k:]
            else:
                ak = -k

                grad[:np_] += (1.0 - h[ak:])
                grad[ak:] -= h[:np_]

        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 200
        t_arr = np.arange(best_n) / float(best_n)
        h0 = 0.5 + 0.25 * np.sin(2*np.pi*2*t_arr) + 0.15*np.sin(2*np.pi*5*t_arr + 1.0)
        best_h = _project(h0, best_n)
        best_c5, _ = _c5_value(best_h)
        print("No parent: n={}, C5={:.10f}".format(best_n, best_c5))

    # Multi-resolution optimization
    resolutions = sorted(set([best_n, max(80, best_n-20), best_n+20, best_n+40, best_n+60, best_n+80]))
    resolutions = [r for r in resolutions if 60 <= r <= 450]

    for n in resolutions:
        if _remains(t0, budget) < 80:
            break

        h = _interp(best_h, best_n, n)
        h = _project(h, n)
        c5_chk, _ = _c5_value(h)
        if c5_chk < best_c5 - 1e-12:
            best_c5 = c5_chk
            best_h = h.copy()
            best_n = n
            print("  Res n={}: C5={:.10f}".format(n, best_c5))

        # L-BFGS-B with increasing beta
        bounds = [(0.0, 1.0)] * n
        _g_n = n

        for beta in [20, 40, 70, 100, 150, 250, 400, 600, 1000, 1500, 2500]:
            if _remains(t0, budget) < 25:
                break
            _g_beta = beta
            _g_lam = 500.0
            try:
                res = minimize(_lbfgs_fun, h, jac=_lbfgs_jac, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 2000, 'ftol': 1e-20,
                                         'gtol': 1e-14, 'maxls': 300,
                                         'max_fun': 5000})
                h_new = _project(res.x, n)
                c5_new, _ = _c5_value(h_new)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n
                    h = h_new.copy()
                    print("  n={}, beta={}: C5={:.10f}".format(n, beta, best_c5))
            except Exception:
                pass

        # SLSQP polish with exact constraint
        if _remains(t0, budget) > 40:
            for beta in [200, 500, 1000, 2000, 4000]:
                if _remains(t0, budget) < 15:
                    break
                _g_beta = beta
                try:
                    res = minimize(_slsqp_fun, h, method='SLSQP',
                                   bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': _slsqp_con}],
                                   options={'maxiter': 500, 'ftol': 1e-18, 'iprint': 0})
                    h_new = _project(res.x, n)
                    c5_new, _ = _c5_value(h_new)
                    if c5_new < best_c5 - 1e-12:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        best_n = n
                        h = h_new.copy()
                        print("  SLSQP n={}, beta={}: C5={:.10f}".format(n, beta, best_c5))
                except Exception:
                    pass

    # Final local search phases at best resolution
    n = best_n
    h = best_h.copy()
    best_h_local = best_h.copy()

    # Gradient-based descent targeting top lags
    best_c5, best_h_local = _gradient_descent_top(
        h, n, rng, budget, t0, best_c5, best_h_local, max_iters=5000)
    if best_c5 < best_c5 + 1e-14:
        best_h = best_h_local.copy()

    # Random pair perturbation
    h = best_h.copy()
    best_c5, best_h_local = _local_search(
        h, n, rng, budget, t0, best_c5, best_h_local, max_iters=20000)
    best_h = best_h_local.copy()

    # Triple perturbation
    h = best_h.copy()
    best_c5, best_h_local = _triple_search(
        h, n, rng, budget, t0, best_c5, best_h_local, max_iters=10000)
    best_h = best_h_local.copy()

    # Another round of gradient descent
    h = best_h.copy()
    best_c5, best_h_local = _gradient_descent_top(
        h, n, rng, budget, t0, best_c5, best_h_local, max_iters=3000)
    best_h = best_h_local.copy()

    # Final SLSQP precision pass
    if _remains(t0, budget) > 20:
        _g_n = n
        bounds = [(0.0, 1.0)] * n
        for beta_f in [3000, 5000, 8000]:
            if _remains(t0, budget) < 8:
                break
            _g_beta = beta_f
            try:
                res = minimize(_slsqp_fun, best_h, method='SLSQP',
                               bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': _slsqp_con}],
                               options={'maxiter': 300, 'ftol': 1e-18, 'iprint': 0})
                h_new = _project(res.x, n)
                c5_new, _ = _c5_value(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n
            except Exception:
                pass

    # Final micro local search
    if _remains(t0, budget) > 10:
        h = best_h.copy()
        best_c5, best_h_local = _local_search(
            h, n, rng, budget, t0, best_c5, best_h_local, max_iters=15000)
        best_h = best_h_local.copy()
        best_n = n

    # Final projection and verification
    best_h = _project(best_h, best_n)
    final_c5, _ = _c5_value(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < best_c5:
            best_c5 = ec5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, best_c5))
    return (best_h, best_c5, best_n)
