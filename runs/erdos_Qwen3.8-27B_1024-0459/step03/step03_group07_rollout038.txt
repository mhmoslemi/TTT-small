def coord_update(h, j, corr):
    """Find optimal value for h[j] given current h and corr."""
    n = len(h)
    old = h[j]
    center = n - 1
    
    # Compute coefficients for each affected lag
    # Returns: for each affected lag fidx, the (coeff, const) such that
    # corr_new[fidx] = rest[fidx] + coeff * x + (quadratic if lag==0)
    
    # We'll build arrays for the linear part
    fidxs = []
    coeffs = []
    rests = []
    
    has_quad = False
    quad_rest = 0.0
    
    for k in range(n):
        fidx = center + k
        coeff = 0.0
        const = 0.0
        aff = False
        if k <= j and (j - k) <= (n - 1 - k):
            coeff += (1.0 - h[j - k])
            aff = True
        if j <= (n - 1 - k):
            coeff += (-h[j + k])
            const += h[j + k]
            aff = True
        if aff:
            if k == 0:
                has_quad = True
                quad_rest = corr[fidx] - old * (1.0 - old)
            else:
                fidxs.append(fidx)
                coeffs.append(coeff)
                rests.append(corr[fidx] - coeff * old - const)
    
    for m in range(1, n):
        fidx = center - m
        coeff = 0.0
        const = 0.0
        aff = False
        if j <= (n - 1 - m):
            coeff += (1.0 - h[j + m])
            aff = True
        if m <= j:
            coeff += (-h[j - m])
            const += h[j - m]
            aff = True
        if aff:
            fidxs.append(fidx)
            coeffs.append(coeff)
            rests.append(corr[fidx] - coeff * old - const)
    
    if not fidxs and not has_quad:
        return old
    
    fidxs_arr = np.array(fidxs) if fidxs else np.array([], dtype=int)
    coeffs_arr = np.array(coeffs) if coeffs else np.array([])
    rests_arr = np.array(rests) if rests else np.array([])
    
    # Max of unaffected
    all_fidx = set(range(2 * n - 1))
    affected = set(fidxs)
    if has_quad:
        affected.add(center)
    unaffected = all_fidx - affected
    if unaffected:
        max_unaff = np.max(corr[list(unaffected)])
    else:
        max_unaff = 0.0
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    
    def obj(x):
        val = max_unaff
        if has_quad:
            v = quad_rest + x - x * x
            if v > val:
                val = v
        if len(coeffs_arr) > 0:
            v = np.max(rests_arr + coeffs_arr * x)
            if v > val:
                val = v
        return val
    
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj(c)
    fd = obj(d)
    
    for _ in range(50):
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj(d)
    
    return (lo + hi) / 2.0

def coord_descent(h, n, sweeps=100, time_limit=None, t0=None, seed=42, anneal_every=20, anneal_prob=0.1):
    h = project_feasible(h, n).copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    corr = get_corr(h).copy()
    rng = np.random.RandomState(seed)
    
    for sw in range(sweeps):
        if time_limit and t0 and time.time() - t0 > time_limit:
            break
        
        # Annealing kick
        if anneal_every > 0 and sw > 0 and sw % anneal_every == 0 and rng.rand() < anneal_prob:
            n_kicks = max(1, n // 20)
            for _ in range(n_kicks):
                j = rng.randint(n)
                h[j] = rng.rand()
            h = project_feasible(h, n)
            corr = get_corr(h).copy()
            continue
        
        order = np.arange(n)
        if rng.rand() < 0.3:
            rng.shuffle(order)
        
        for j in order:
            if time_limit and t0 and time.time() - t0 > time_limit:
                break
            old = h[j]
            new_x = coord_update(h, j, corr)
            if abs(new_x - old) > 1e-10:
                _update_corr(corr, h, j, old, new_x)
                h[j] = new_x
        
        # Project
        h_old_sum = h.sum()
        h = project_feasible(h, n)
        if abs(h.sum() - h_old_sum) > 1e-12:
            # Projection changed things, recompute corr
            # Actually, this is expensive. Let me just recompute at end of sweep.
            pass
        
        # Recompute corr periodically (every 5 sweeps) to correct for drift
        if sw % 5 == 4:
            corr = get_corr(h).copy()
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    
    return best, best_c5

def random_binary_init(n, rng):
    h = np.zeros(n)
    ones_idx = rng.choice(n, n//2, replace=False)
    h[ones_idx] = 1.0
    return h
