These are not simply 1 - autocorr(s) because the integration domain changes.

            try:
                r = minimize(tk_obj, h.copy(),
                             args=(n, beta, 16),
                             method='L-BFGS-B', bounds=bounds,
                             options={'maxiter': 400, 'ftol': 1e-15,
                                      'iprint': 0})
                hn = proj(r.x, n)
                c = c5_val(hn)
                if c < bc:
                    bc = c
                    h = hn
            except Exception:
            pass
    return h, bc

def bsa(h0, n, rng, iters=5000, t0=0.003, t1=0.00001):
    h = np.where(h0 > 0.5, 1.0, 0.0)
    ones = np.where(h == 1.0)[0]
    tgt = n // 2
    if len(ones) > tgt:
        rm = rng.choice(len(ones), len(ones) - tgt, replace=False)
        h[ones[rm]] = 0.0
    elif len(ones) < tgt:
        zs = np.where(h == 0.0)[0]
        ad = rng.choice(len(zs), tgt - len(ones), replace=False)
        h[zs[ad]] = 1.0
    bh = h.copy()
    bc = c5_val(h)
    cc = bc
    ol = list(np.where(h == 1.0)[0])
    zl = list(np.where(h == 0.0)[0])
    for it in range(iters):
        if rem() < 2:
            break
        i = rng.randint(len(ol))
        j = rng.randint(len(zl))
        oi, oz = ol[i], zl[j]
        h[oi] = 0.0
        h[oz] = 1.0
        nc = c5_val(h)
        T = t0 * (t1 / t0) ** (it / max(iters, 1))
        if nc < cc or (T > 1e-16 and rng.rand() < np.exp(-(nc - cc) / T)):
            cc = nc
            ol[i] = oz
            zl[j] = oi
            if nc < bc:
                bc = nc
                bh = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0
    return bh, bc

def local_search(h0, n, rng, max_sweeps=15):
    h = proj(h0, n)
    bc = c5_val(h)
    for sw in range(max_sweeps):
        if rem() < 3:
            break
        imp = False
        order = rng.permutation(n)
        for i in order:
            if rem() < 2:
                break
            old = h[i]
            best_loc = bc
            best_val = old
            cands = [0.0, 1.0, 0.5]
            if old > 0.3:
                cands.append(0.15)
            if old < 0.7:
                cands.append(0.85)
            cands.append(max(0.0, old - 0.1))
            cands.append(min(1.0, old + 0.1))
            for nv in cands:
                if abs(nv - old) < 1e-9:
                    continue
                h[i] = nv
                hp = proj(h, n)
                c = c5_val(hp)
                if c < best_loc - 1e-12:
                    best_loc = c
                    best_val = nv
                    h[:] = hp
            if best_val != old:
                h[i] = best_val
                h = proj(h, n)
                if best_loc < bc:
                    bc = best_loc
                    imp = True
        if not imp:
            break
    return h, bc

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float('inf')
    best_n = 0

    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        pc = c5_val(ph)
        best_h, best_c, best_n = ph.copy(), pc, pn
        print("Parent: n={}, C5={:.7f}".format(pn, pc))
    except Exception:
        best_n = 160
        best_h = proj(init_tm(best_n), best_n)
        best_c = c5_val(best_h)
        print("TM: n={}, C5={:.7f}".format(best_n, best_c))

    n = best_n
    print("\n=== P1: L-BFGS at n={} ===".format(n))
    starts = [best_h.copy()]
    starts.append(proj(init_tm(n), n))
    starts.append(proj(init_golden(n), n))
    starts.append(proj(init_vdc(n), n))
    starts.append(proj(init_sine(n, 3, 0.5), n))
    starts.append(proj(init_sine(n, 5, 1.7), n))
    for _ in range(5):
        starts.append(proj(init_rand(n, rng), n))

    for si, s in enumerate(starts):
        if rem() < 18:
            break
        h, c = optimize_lbfgs(s, n, rng)
        if c < best_c:
            best_c, best_h = c, h.copy()
            print("  S{}: C5={:.7f}".format(si, best_c))

    print("\n=== P2: Binary SA + polish at n={} ===".format(n))
    sa_starts = [proj(best_h, n), proj(init_tm(n), n)]
    for _ in range(4):
        sa_starts.append(init_rand(n, rng))
    for si, s in enumerate(sa_starts):
        if rem() < 25:
            break
        hb, cb = bsa(s, n, rng, iters=4500)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()
            print("  SA{}: C5={:.7f}".format(si, best_c))
        if rem() > 15:
            h, c = optimize_lbfgs(hb, n, rng)
            if c < best_c:
                best_c, best_h = c, h.copy()
                print("  Poly{}: C5={:.7f}".format(si, best_c))

    print("\n=== P3: Coordinate descent at n={} ===".format(n))
    if rem() > 30:
        h, c = local_search(best_h, n, rng, max_sweeps=20)
        if c < best_c:
            best_c, best_h = c, h.copy()
            print("  CS: C5={:.7f}".format(best_c))
        if rem() > 15:
            h2, c2 = optimize_lbfgs(h, n, rng)
            if c2 < best_c:
                best_c, best_h = c2, h2.copy()
                print("  Poly2: C5={:.7f}".format(best_c))

    print("\n=== P4: Higher resolutions ===")
    for nt in [200, 240, 320, 400, 480, 560]:
        if rem() < 120 or nt <= best_n:
            continue
        print("  n={}".format(nt))
        hi = proj(interp_h(best_h, best_n, nt), nt)
        ht = proj(init_tm(nt), nt)
        hg = proj(init_golden(nt), nt)
        for si, s in enumerate([hi, ht, hg]):
            if rem() < 35:
                break
            h, c = optimize_lbfgs(s, nt, rng)
            if c < best_c:
                best_c, best_h, best_n = c, h.copy(), nt
                print("    n={} S{}: C5={:.7f}".format(nt, si, best_c))
        if rem() > 40:
            hb, cb = bsa(best_h if best_n == nt else hi, nt, rng, iters=3000)
            if cb < best_c:
                best_c, best_h, best_n = cb, hb.copy(), nt
                print("    SA n={}: C5={:.7f}".format(nt, best_c))

    print("\n=== P5: Final polish at n={} ===".format(best_n))
    n = best_n
    bounds_f = [(0.0, 1.0)] * n
    for beta in [500, 1200, 3000, 6000]:
        if rem() < 8:
            break
        try:
            r = minimize(sm_obj, best_h.copy(), args=(n, beta),
                         jac=sm_grad, method='L-BFGS-B',
                         bounds=bounds_f,
                         options={'maxiter': 600, 'ftol': 1e-16,
                                  'gtol': 1e-12, 'iprint': 0})
            hn = proj(r.x, n)
            c = c5_val(hn)
            if c < best_c:
                best_c, best_h = c, hn
                print("  b={}: C5={:.7f}".format(beta, best_c))
        except Exception:
            pass

    if rem() > 8:
        h, c = local_search(best_h, best_n, rng, max_sweeps=10)
        if c < best_c:
            best_c, best_h = c, h.copy()
            print("  CS: C5={:.7f}".format(best_c))
        if rem() > 5:
            try:
                bounds_f = [(0.0, 1.0)] * best_n
                r = minimize(sm_obj, best_h.copy(), args=(best_n, 3000),
                             jac=sm_grad, method='L-BFGS-B',
                             bounds=bounds_f,
                             options={'maxiter': 400, 'ftol': 1e-16,
                                      'gtol': 1e-12, 'iprint': 0})
                hn = proj(r.x, best_n)
                c = c5_val(hn)
                if c < best_c:
                    best_c, best_h = c, hn
                    print("  Final: C5={:.7f}".format(best_c))
            except Exception:
                pass

    best_h = proj(best_h, best_n)
    fc = c5_val(best_h)
    try:
        ec = evaluate_erdos_solution(best_h)
        if ec < fc:
            fc = ec
    except Exception:
        pass
    print("\nFINAL: n={}, C5={:.7f}".format(best_n, fc))
    return (best_h, fc, best_n)
