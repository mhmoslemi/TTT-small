    h = project_feasible(h, n)
    best = h.copy()
    best_c5 = compute_c5(h)
    for sw in range(sweps):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        for j in range(n):
            if time.time() - t0 > time_limit:
                break
            old = h[j]
            max_ua = np.max(np.concatenate([corr[:j], corr[j+1:]])) if n > 1 else 0.0
            a_vals = []
            b_vals = []
            rest_vals = []
            is_quad = []
            for lag in range(-(n - 1), n):
                fidx = lag + (n - 1)
                coeff = 0.0
                const = 0.0
                affects = False
                if lag >= 0:
                    if lag <= j <= n - 1:
                        coeff += (1.0 - h[j - lag])
                        affects = True
                    if 0 <= j <= n - 1 - lag:
                        coeff += (-h[j + lag])
                        const += h[j + lag]
                        affects = True
                else:
                    s = -lag
                    if 0 <= j <= n - 1 - s:
                        coeff += (1.0 - h[j + s])
                        affects = True
                    if s <= j <= n - 1:
                        coeff += (-h[j - s])
                        const += h[j - s]
                        affects = True
                if not affects:
                    if corr[fidx] > max_ua:
                        max_ua = corr[fidx]
                    continue
                if lag == 0:
                    a_vals.append(1.0)
                    b_vals.append(0.0)
                    rest_vals.append(corr[fidx] - old * (1.0 - old))
                    is_quad.append(True)
                else:
                    a_vals.append(coeff)
                    b_vals.append(const)
                    rest_vals.append(corr[fidx] - coeff * old - const)
                    is_quad.append(False)
            if not a_vals:
                continue
            a_arr = np.array(a_vals)
            b_arr = np.array(b_vals)
            r_arr = np.array(rest_vals)
            quad_mask = np.array(is_quad)
            best_x = old
            best_val = max_ua
            for x in np.linspace(0.0, 1.0, 50):
                if any(quad_mask):
                    vals = np.where(quad_mask, r_arr + x - x * x, r_arr + a_arr * x + b_arr)
                else:
                    vals = r_arr + a_arr * x + b_arr
                val = max(max_ua, np.max(vals))
                if val < best_val:
                    best_val = val
                    best_x = x
            lo = max(0.0, best_x - 0.02)
            hi = min(1.0, best_x + 0.02)
            for _ in range(30):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if any(quad_mask):
                    v1 = max(max_ua, np.max(np.where(quad_mask, r_arr + m1 - m1*m1, r_arr + a_arr*m1 + b_arr)))
                    v2 = max(max_ua, np.max(np.where(quad_mask, r_arr + m2 - m2*m2, r_arr + a_arr*m2 + b_arr)))
                else:
                    v1 = max(max_ua, np.max(r_arr + a_arr * m1 + b_arr))
                    v2 = max(max_ua, np.max(r_arr + a_arr * m2 + b_arr))
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            h[j] = (lo + hi) / 2.0
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    return best, best_c5

def interpolate_h(h_src, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_new = np.interp(xn, xp, h_src)
    return project_feasible(h_new, n_dst)

def make_structural_init(n, freq, phase, amp, rng):
    t = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + amp * np.cos(2 * np.pi * freq * t / 2.0 + phase)
    return project_feasible(np.clip(h, 0, 1), n)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return max(0.0, budget_s - (time.time() - t0) - 10.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn > 10:
            best_h = project_feasible(ph, pn)
            best_c5 = compute_c5(best_h)
            best_n = pn
    except Exception:
        pass

    if best_h is None:
        best_n = 160
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)

    resolutions = []
    base_n = best_n
    for nn in [base_n, base_n + 40, base_n + 80, base_n + 120]:
        if 80 <= nn <= 400 and nn not in resolutions:
            resolutions.append(nn)
    resolutions = sorted(set(resolutions))

    for n in resolutions:
        if time_left() < 80:
            break

        if n == best_n:
            h_init = best_h.copy()
        else:
            h_init = interpolate_h(best_h, best_n, n)

        inits = [h_init.copy()]
        for sc in [0.03, 0.06, 0.1]:
            hp = h_init + sc * rng.randn(n)
            inits.append(project_feasible(hp, n))
        t_arr = np.linspace(0, 2, n, endpoint=False)
        for fq in [1, 2, 3, 5]:
            for ph_ in [0, np.pi/2, np.pi]:
                hs = 0.7 * h_init + 0.3 * np.clip(0.5 + 0.4*np.cos(2*np.pi*fq*t_arr/2.0+ph_), 0, 1)
                inits.append(project_feasible(hs, n))

        n_inits = len(inits)
        time_per_init = max(20.0, min(200.0, time_left() / n_inits - 5.0))

        for h0 in inits:
            if time_left() < 25:
                break
            tl = time_left()
            cur_h = h0.copy()
            cur_c5 = compute_c5(cur_h)

            # Phase A: L-BFGS with smooth max
            if tl > 30:
                h1, c5_1 = lbfgs_smooth(cur_h, n, [40.0, 150.0, 500.0, 2000.0],
                                        [100.0, 1000.0], 200, t0, t0 + tl * 0.4)
                if c5_1 < cur_c5:
                    cur_h, cur_c5 = h1, c5_1

            # Phase B: Top-k projected gradient
            if tl > 40:
                h2, c5_2 = projected_topk(cur_h, n, 5, 0.01, 2000, t0, t0 + tl * 0.6)
                if c5_2 < cur_c5:
                    cur_h, cur_c5 = h2, c5_2
                if time_left() > 15:
                    h2b, c5_2b = projected_topk(cur_h, n, 3, 0.005, 1500, t0, t0 + tl * 0.75)
                    if c5_2b < cur_c5:
                        cur_h, cur_c5 = h2b, c5_2b

            # Phase C: Coordinate descent
            if time_left() > 20:
                h3, c5_3 = coord_descent_fast(cur_h, n, 5, t0, t0 + time_left())
                if c5_3 < cur_c5:
                    cur_h, cur_c5 = h3, c5_3

            # Phase D: Final L-BFGS polish
            if time_left() > 20:
                h4, c5_4 = lbfgs_smooth(cur_h, n, [2000.0, 10000.0],
                                        [5000.0], 100, t0, t0 + time_left() * 0.7)
                if c5_4 < cur_c5:
                    cur_h, cur_c5 = h4, c5_4

            # Phase E: Another coordinate descent pass
            if time_left() > 15:
                h5, c5_5 = coord_descent_fast(cur_h, n, 8, t0, t0 + time_left())
                if c5_5 < cur_c5:
                    cur_h, cur_c5 = h5, c5_5

            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
                best_n = n

    # Final intensive polish on best solution
    if time_left() > 30:
        t0p = time.time()
        h_f, c5_f = coord_descent_fast(best_h, best_n, 25, t0p, t0p + time_left() * 0.3)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f

    if time_left() > 20:
        t0p = time.time()
        h_f2, c5_f2 = projected_topk(best_h, best_n, 3, 0.002, 3000, t0p, t0p + time_left() * 0.5)
        if c5_f2 < best_c5:
            best_c5, best_h = c5_f2, h_f2

    if time_left() > 15:
        t0p = time.time()
        h_f3, c5_f3 = coord_descent_fast(best_h, best_n, 15, t0p, t0p + time_left() * 0.8)
        if c5_f3 < best_c5:
            best_c5, best_h = c5_f3, h_f3

    if time_left() > 10:
        t0p = time.time()
        h_f4, c5_f4 = lbfgs_smooth(best_h, best_n, [5000.0, 20000.0],
                                   [10000.0], 50, t0p, t0p + time_left() * 0.8)
        if c5_f4 < best_c5:
            best_c5, best_h = c5_f4, h_f4

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
