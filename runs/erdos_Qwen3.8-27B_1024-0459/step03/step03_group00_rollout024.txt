```python
        valid_j = np.arange(0, n - k_shift)  # j indices
        # h[j+k] part: for i in range(k, n): contrib (1-h[i-k])
        for i in range(k_shift, n):
            grad[i] += (1.0 - h[i - k_shift])
        # 1-h[j] part: for i in range(0, n-k): contrib -h[i+k]
        for i in range(0, n - k_shift):
            grad[i] -= h[i + k_shift]
    else:
        ks = -k_shift  # positive
        # c = sum_{j=ks}^{n-1} h[j-ks] * (1-h[j])
        # d/dh[i]: (1-h[i+ks]) if 0 <= i <= n-1-ks, minus h[i-ks] if ks <= i <= n-1
        for i in range(0, n - ks):
            grad[i] += (1.0 - h[i + ks])
        for i in range(ks, n):
            grad[i] -= h[i - ks]
    return grad
```

```python
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("No parent. n={}, C5={:.8f}".format(parent_n, best_c5))

    # Phase 1: Interpolate to higher resolution and quick SLSQP
    target_ns = [best_n, 384, 512, 640]
    target_ns = sorted(set([nn for nn in target_ns if 128 <= nn <= 768]))

    phase1_results = []

    for n in target_ns:
        if remaining_time() < 60.0:
            break
        _G_N = n
        _G_TGT = n / 2.0
        _G_DX = 2.0 / n
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]

        h0 = project_feasible(interp_to(best_h, best_n, n), n)
        c5_cur = compute_c5(h0)
        if c5_cur < best_c5 - 1e-10:
            best_c5 = c5_cur
            best_h = h0.copy()
            best_n = n

        for beta_val in [30, 100]:
            if remaining_time() < 15.0:
                break
            _G_BETA = float(beta_val)
            try:
                res = minimize(obj_val, h0, jac=obj_jac, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 200, 'ftol': 1e-12, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-12:
                    c5_cur = c5_new
                    h0 = h_new
            except Exception:
                pass

        if c5_cur < best_c5 - 1e-10:
            best_c5 = c5_cur
            best_h = h0.copy()
            best_n = n
            print("  Phase1 n={}: C5={:.8f}".format(n, best_c5))
        phase1_results.append((c5_cur, n, h0.copy()))

    # Phase 2: Deep SLSQP at best resolution
    n_work = best_n
    if remaining_time() > 600:
        if 512 in [p[1] for p in phase1_results]:
            for c5_r, n_r, h_r in phase1_results:
                if n_r == 512:
                    if c5_r < best_c5:
                        n_work = 512
                    else:
                        n_work = 512
                    break
        else:
            n_work = max(n_work, 384)

    _G_N = n_work
    _G_TGT = n_work / 2.0
    _G_DX = 2.0 / n_work
    bounds = [(0.0, 1.0)] * n_work
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]

    # Start from best interpolated to n_work
    h_work = project_feasible(interp_to(best_h, best_n, n_work), n_work)
    c5_work = compute_c5(h_work)
    if c5_work < best_c5:
        best_c5 = c5_work
        best_h = h_work.copy()
        best_n = n_work

    print("\nPhase 2: Deep SLSQP at n={}".format(n_work))
    beta_schedule = [50, 100, 200, 400, 800, 1500, 3000]
    for beta_val in beta_schedule:
        if remaining_time() < 20.0:
            break
        _G_BETA = float(beta_val)
        maxiter = 400 if beta_val <= 400 else 250
        try:
            res = minimize(obj_val, h_work, jac=obj_jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
            h_new = project_feasible(res.x, n_work)
            c5_new = compute_c5(h_new)
            if c5_new < c5_work - 1e-13:
                c5_work = c5_new
                h_work = h_new
                print("  beta={}: C5={:.8f}".format(beta_val, c5_work))
        except Exception:
            pass

    if c5_work < best_c5 - 1e-10:
        best_c5 = c5_work
        best_h = h_work.copy()
        best_n = n_work

    # Phase 2b: Try from parent directly at n_work with different init
    if remaining_time() > 100:
        h_alt = project_feasible(interp_to(parent_h, parent_n, n_work), n_work)
        # Binarize it
        h_bin = (h_alt > 0.5).astype(float)
        # Fix count
        n_ones = int(np.sum(h_bin))
        target_count = n_work // 2
        if n_ones > target_count:
            idx_remove = np.where(h_bin == 1.0)[0]
            remove_set = idx_remove[rng.choice(len(idx_remove), size=n_ones - target_count, replace=False)]
            h_bin[remove_set] = 0.0
        elif n_ones < target_count:
            idx_add = np.where(h_bin == 0.0)[0]
            add_set = idx_add[rng.choice(len(idx_add), size=target_count - n_ones, replace=False)]
            h_bin[add_set] = 1.0
        c5_bin = compute_c5(h_bin)
        if c5_bin < c5_work:
            h_work = h_bin.copy()
            c5_work = c5_bin
            print("  Binarized init gave C5={:.8f}".format(c5_bin))
        _G_BETA = 200.0
        try:
            res = minimize(obj_val, h_work, jac=obj_jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-14, 'iprint': 0})
            h_new = project_feasible(res.x, n_work)
            c5_new = compute_c5(h_new)
            if c5_new < c5_work - 1e-13:
                c5_work = c5_new
                h_work = h_new
        except Exception:
            pass
        if c5_work < best_c5 - 1e-10:
            best_c5 = c5_work
            best_h = h_work.copy()
            best_n = n_work

    # Phase 3: Momentum subgradient
    if remaining_time() > 80.0:
        print("\nPhase 3: Momentum subgradient")
        h_sg = best_h.copy()
        n_sg = best_n
        _G_N = n_sg
        _G_TGT = n_sg / 2.0
        _G_DX = 2.0 / n_sg
        best_h, best_c5 = do_momentum_subgradient(h_sg, n_sg, None)
        best_n = len(best_h)
        print("  After subgradient: n={}, C5={:.8f}".format(best_n, best_c5))

    # Phase 4: SLSQP restart from perturbed best
    if remaining_time() > 60.0:
        print("\nPhase 4: Perturbed restart")
        n_pr = best_n
        _G_N = n_pr
        _G_TGT = n_pr / 2.0
        _G_DX = 2.0 / n_pr
        bounds = [(0.0, 1.0)] * n_pr
        cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]

        for trial in range(3):
            if remaining_time() < 15.0:
                break
            noise = rng.randn(n_pr) * 0.02
            h_pert = project_feasible(best_h + noise, n_pr)
            _G_BETA = 500.0
            try:
                res = minimize(obj_val, h_pert, jac=obj_jac, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-14, 'iprint': 0})
                h_new = project_feasible(res.x, n_pr)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  Perturb {} improved: C5={:.8f}".format(trial, best_c5))
            except Exception:
                pass

    # Phase 5: Coordinate descent
    if remaining_time() > 40.0:
        print("\nPhase 5: Coordinate descent")
        best_h, best_c5 = do_coord_descent(best_h, best_n, rng, None, max_sweeps=15)
        best_n = len(best_h)
        print("  After coord descent: n={}, C5={:.8f}".format(best_n, best_c5))

    # Phase 6: Final SLSQP polish
    if remaining_time() > 20.0:
        print("\nPhase 6: Final polish")
        n_fp = best_n
        _G_N = n_fp
        _G_TGT = n_fp / 2.0
        _G_DX = 2.0 / n_fp
        bounds = [(0.0, 1.0)] * n_fp
        cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
        for beta_val in [2000, 5000, 10000]:
            if remaining_time() < 8.0:
                break
            _G_BETA = float(beta_val)
            try:
                res = minimize(obj_val, best_h, jac=obj_jac, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-15, 'iprint': 0})
                h_new = project_feasible(res.x, n_fp)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new
            except Exception:
                pass
        print("  After polish: C5={:.8f}".format(best_c5))

    # Phase 7: One more round of subgradient + coord
    if remaining_time() > 30.0:
        best_h, best_c5 = do_momentum_subgradient(best_h, best_n, None)
        best_n = len(best_h)
        if remaining_time() > 15.0:
            best_h, best_c5 = do_coord_descent(best_h, best_n, rng, None, max_sweeps=8)
            best_n = len(best_h)
        print("  After final refinement: n={}, C5={:.8f}".format(best_n, best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
            print("  evaluate_erdos_solution gave lower: {:.8f}".format(final_c5))
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
```
