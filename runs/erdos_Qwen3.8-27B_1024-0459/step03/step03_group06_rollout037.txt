def vec_gradient(h, w, dx):
    n = len(h)
    grad = np.zeros(n)
    # w has length 2n-1
    # Positive shifts: s = 0 to n-1, j = n-1+s
    # w_pos[s] = w[n-1+s]
    w_pos = w[n-1:]  # length n, w_pos[s] = w[n-1+s]
    for s in range(n):
        wt = w_pos[s]
        if wt < 1e-15:
            continue
        grad[s:] += wt * (1.0 - h[:n-s])
        grad[:n-s] -= wt * h[s:]
    # Negative shifts: sp = 1 to n-1, j = n-1-sp
    # w_neg[sp] = w[n-1-sp]
    w_neg = w[:n-1][::-1]  # length n-1, w_neg[sp-1] = w[n-1-sp]
    for sp in range(1, n):
        wt = w_neg[sp-1]
        if wt < 1e-15:
            continue
        grad[:n-sp] += wt * (1.0 - h[sp:])
        grad[sp:] -= wt * h[:n-sp]
    grad *= dx
    return grad

**Simpler approach for binary search:**
Instead of computing deltas analytically, I'll use a "batch" approach:
1. For a given h, compute the full correlation
2. For each position i, compute the "impact vector" - how changing h[i] affects each correlation

The impact of changing h[i] from 1 to 0 on correlation at shift s:
- For s ≥ 0: delta[s] = -(1-h[i-s])*[s≤i] + h[i+s]*[s<n-i] (using original h)
- For s < 0 (sp=-s): delta[-sp] = -(1-h[i-sp])*[sp≤i] + h[i+sp]*[sp≤n-1-i]

The impact of changing h[j] from 0 to 1:
- For s ≥ 0: delta[s] = (1-h[j-s])*[s≤j] - h[j+s]*[s<n-j]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        unsat = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(unsat))
        if cnt > 0:
            h[unsat] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            cands = np.where(h < 1.0 - abs(diff) + 1e-15)[0]
        else:
            cands = np.where(h > abs(diff) - 1e-15)[0]
        if len(cands) > 0:
            h[cands[0]] += diff
        else:
            idx = int(np.argmin(np.abs(h - 0.5)))
            h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, beta, n, tgt):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = beta * (corr - m)
    obj = (m + np.log(np.sum(np.exp(s))) / beta) * dx
    penalty = 1000.0 * (np.sum(h) - tgt) ** 2
    return obj + penalty

def smooth_max_grad(h, beta, n, tgt):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = beta * (corr - m)
    w = np.exp(s)
    w /= np.sum(w)
    grad = np.zeros(n)
    # Positive shifts s=0..n-1 at index n-1+s
    for s_val in range(n):
        wt = w[n - 1 + s_val]
        if wt < 1e-16:
            continue
        grad[s_val:] += wt * (1.0 - h[:n - s_val])
        grad[:n - s_val] -= wt * h[s_val:]
    # Negative shifts sp=1..n-1 at index n-1-sp
    for sp in range(1, n):
        wt = w[n - 1 - sp]
        if wt < 1e-16:
            continue
        grad[:n - sp] += wt * (1.0 - h[sp:])
        grad[sp:] -= wt * h[:n - sp]
    grad *= dx
    grad += 2.0 * 1000.0 * (np.sum(h) - tgt)
    return grad

def topk_obj_grad(h, k, beta, n, tgt):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-k:]
    tv = corr[top_idx]
    m = np.max(tv)
    s = beta * (tv - m)
    obj = (m + np.log(np.sum(np.exp(s))) / beta) * dx
    penalty = 1000.0 * (np.sum(h) - tgt) ** 2
    obj += penalty

    tw = np.exp(s)
    tw /= np.sum(tw)
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        j = top_idx[idx]
        wt = tw[idx]
        if wt < 1e-16:
            continue
        s_val = j - (n - 1)
        if s_val >= 0:
            grad[s_val:] += wt * (1.0 - h[:n - s_val])
            grad[:n - s_val] -= wt * h[s_val:]
        else:
            sp = -s_val
            grad[:n - sp] += wt * (1.0 - h[sp:])
            grad[sp:] -= wt * h[:n - sp]
    grad *= dx
    grad += 2.0 * 1000.0 * (np.sum(h) - tgt)
    return obj + penalty, grad

def subgradient_step(h, n, top_k=5, step_scale=0.02):
    """Take a subgradient step targeting the top-k correlations."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for j in top_idx:
        s_val = j - (n - 1)
        if s_val >= 0:
            grad[s_val:] += (1.0 - h[:n - s_val])
            grad[:n - s_val] -= h[s_val:]
        else:
            sp = -s_val
            grad[:n - sp] += (1.0 - h[sp:])
            grad[sp:] -= h[:n - sp]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-14:
        return h
    step = step_scale * dx / gnorm
    h_new = h - step * grad
    return project_feasible(h_new, n)

def coordinate_descent(h, n, sweeps=5, rng=None):
    """Continuous coordinate descent."""
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    tgt = n / 2.0
    curr_c5 = compute_c5(h)
    best_c5 = curr_c5
    best_h = h.copy()

    for sweep in range(sweeps):
        if _remaining() < 5:
            break
        order = rng.permutation(n) if rng is not None else np.arange(n)
        for i in order:
            if _remaining() < 3:
                break
            old = h[i]
            old_c5 = compute_c5(h)
            candidates = [0.0, 1.0, 0.5, max(0.0, old - 0.15), min(1.0, old + 0.15)]
            best_val = old
            best_c5_local = old_c5
            for nv in candidates:
                if abs(nv - old) < 1e-9:
                    continue
                h[i] = nv
                h_feas = project_feasible(h, n)
                c5_try = compute_c5(h_feas)
                if c5_try < best_c5_local:
                    best_c5_local = c5_try
                    best_val = nv
                    h = h_feas.copy()
            h[i] = best_val
            h = project_feasible(h, n)
            c5_now = compute_c5(h)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
    return best_h, best_c5

        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.7f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 200
        best_h = make_thue_morse(best_n)
        best_c5 = compute_c5(best_h)
        print("No parent. Thue-Morse: n={}, C5={:.7f}".format(best_n, best_c5))

    # Phase 1: Continuous optimization at multiple resolutions
    resolutions = [200, 280, 360, 480, 640]
    for n in resolutions:
        if _remaining() < 30:
            break
        print("\n=== Continuous opt n={} ===".format(n))

        inits = [interp_h(best_h, n)]
        inits.append(make_thue_morse(n))
        for _ in range(2):
            inits.append(make_random_bin(n, rng))
        inits.append(make_sine(n, 3, rng.rand() * 6.0))
        inits.append(make_sine(n, 5, rng.rand() * 6.0))
        inits.append(make_cos_sums(n, [2, 5, 8], [0.15, 0.1, 0.08],
                                   [rng.rand() * 6, rng.rand() * 6, rng.rand() * 6], rng))

        for h0 in inits:
            if _remaining() < 10:
                break
            h0 = project_feasible(h0, n)
            c5_0 = compute_c5(h0)
            if c5_0 > best_c5 * 1.15:
                continue

            # Progressive beta refinement
            for beta in [20, 50, 100, 300]:
                if _remaining() < 8:
                    break
                h_new = run_lbfgs(h0, n, beta, 200, _remaining)
                c5_new = compute_c5(h_new)
                if c5_new < c5_0:
                    h0 = h_new
                    c5_0 = c5_new

            # Top-k focused
            if _remaining() > 8:
                for k in [4, 8, 12]:
                    if _remaining() < 5:
                        break
                    for beta in [100, 300]:
                        if _remaining() < 5:
                            break
                        h_new = run_lbfgs_topk(h0, n, k, beta, 150)
                        c5_new = compute_c5(h_new)
                        if c5_new < c5_0:
                            h0 = h_new
                            c5_0 = c5_new

            if c5_0 < best_c5:
                best_c5 = c5_0
                best_h = h0.copy()
                best_n = n
                print("  Improved: n={}, C5={:.7f}".format(n, c5_0))

    # Phase 2: Binary optimization
    if _remaining() > 50:
        n_bin = best_n
        print("\n=== Binary search n={} ===".format(n_bin))

        h_bin_start = project_feasible(best_h, n_bin)
        binary_inits = [h_bin_start]
        for _ in range(3):
            binary_inits.append(make_random_bin(n_bin, rng))
        binary_inits.append(make_thue_morse(n_bin))

        for h_bi in binary_inits:
            if _remaining() < 20:
                break
            # Simulated annealing
            h_res, c5_res = binary_sa(h_bi, n_bin, 5000, rng, t0=0.002, t1=0.00001)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
                best_n = n_bin
                print("  SA: C5={:.7f}".format(c5_res))

            # Directed search from best
            if _remaining() > 20:
                h_res2, c5_res2 = directed_binary_search(h_res, n_bin, 200, rng)
                if c5_res2 < best_c5:
                    best_c5 = c5_res2
                    best_h = h_res2.copy()
                    print("  Directed: C5={:.7f}".format(c5_res2))

    # Phase 3: Higher resolution binary
    if _remaining() > 40:
        for n_try in [best_n + 40, best_n + 80, best_n + 160]:
            if n_try > 800 or _remaining() < 20:
                break
            print("\n=== Binary at n={} ===".format(n_try))
            h_interp = interp_h(best_h, n_try)
            h_bin = np.where(h_interp > 0.5, 1.0, 0.0)
            # Fix to exactly n//2 ones
            target = n_try // 2
            num_ones = int(np.sum(h_bin))
            if num_ones > target:
                ones_idx = list(np.where(h_bin == 1.0)[0])
                to_rm = rng.choice(len(ones_idx), num_ones - target, replace=False)
                for idx in to_rm:
                    h_bin[ones_idx[idx]] = 0.0
            elif num_ones < target:
                zeros_idx = list(np.where(h_bin == 0.0)[0])
                to_add = rng.choice(len(zeros_idx), target - num_ones, replace=False)
                for idx in to_add:
                    h_bin[zeros_idx[idx]] = 1.0

            h_res, c5_res = binary_sa(h_bin, n_try, 3000, rng, t0=0.0015, t1=0.00001)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
                best_n = n_try
                print("  SA n={}: C5={:.7f}".format(n_try, c5_res))

    # Phase 4: Subgradient polishing
    if _remaining() > 15:
        n = best_n
        h = best_h.copy()
        print("\n=== Subgradient polish n={} ===".format(n))
        best_local = compute_c5(h)
        for it in range(500):
            if _remaining() < 5:
                break
            h_new = subgradient_step(h, n, top_k=5, step_scale=0.03)
            c5_new = compute_c5(h_new)
            if c5_new < best_local:
                best_local = c5_new
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                # Try smaller step
                h_new2 = subgradient_step(h, n, top_k=3, step_scale=0.01)
                c5_new2 = compute_c5(h_new2)
                if c5_new2 < best_local:
                    best_local = c5_new2
                    h = h_new2
                    if c5_new2 < best_c5:
                        best_c5 = c5_new2
                        best_h = h.copy()
                else:
                    break

    # Phase 5: Final continuous polish
    if _remaining() > 10:
        n = best_n
        print("\n=== Final polish n={} ===".format(n))
        for beta in [500, 1000, 2000]:
            if _remaining() < 5:
                break
            h_new = run_lbfgs(best_h, n, beta, 300, _remaining)
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  Polish beta={}: C5={:.7f}".format(beta, c5_new))

    # Phase 6: Coordinate descent
    if _remaining() > 10:
        n = best_n
        print("\n=== Coord descent n={} ===".format(n))
        h_cd, c5_cd = coordinate_descent(best_h, n, sweeps=3, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            print("  CD: C5={:.7f}".format(c5_cd))

    # Final evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
