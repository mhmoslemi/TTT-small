            res = minimize(obj_smooth, h_cur, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxit, 'ftol': 1e-16, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-13:
                c5_cur = c5_new
                h_cur = h_new.copy()
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_cur.copy()
                print("  beta={}: C5={:.10f}".format(beta_val, best_c5))
        except Exception:
            pass

    # Also try from Thue-Morse and golden ratio initializations
    for h_init_name, h_init_fn in [("TM", make_thue_morse), ("GR", make_golden_ratio)]:
        if budget_remain() < 20.0:
            break
        h_init = project_feasible(h_init_fn(n), n)
        c5_init = compute_c5(h_init)
        if c5_init > c5_cur + 0.05:
            continue
        for beta_val in [200, 800]:
            if budget_remain() < 10.0:
                break
            _G_BETA = float(beta_val)
            try:
                res = minimize(obj_smooth, h_init, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 500, 'ftol': 1e-15, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_cur = h_new.copy()
                    c5_cur = c5_new
                    print("  {} init beta={}: C5={:.10f}".format(h_init_name, beta_val, best_c5))
            except Exception:
                pass

    # Phase 2: Fine coordinate descent with line search
    print("\n--- Phase 2: Fine Line Search (n={}) ---".format(n))
    if budget_remain() > 60.0:
        best_h, best_c5 = do_local_search_fine(
            best_h, n, budget_remain, rng, max_sweeps=150)
        print("  After fine LS: C5={:.10f}".format(best_c5))

    # Phase 3: Simulated annealing
    print("\n--- Phase 3: Simulated Annealing ---".format())
    if budget_remain() > 30.0:
        best_h, best_c5 = do_simulated_annealing(
            best_h, n, budget_remain, rng, n_iters=6000)
        print("  After SA: C5={:.10f}".format(best_c5))

    # Phase 4: More line search after SA
    if budget_remain() > 40.0:
        print("\n--- Phase 4: Post-SA Line Search ---")
        best_h, best_c5 = do_local_search_fine(
            best_h, n, budget_remain, rng, max_sweeps=100)
        print("  After post-SA LS: C5={:.10f}".format(best_c5))

    # Phase 5: Try higher resolution n=512
    if budget_remain() > 90.0:
        n_h = 512
        _G_N = n_h
        _G_TGT = n_h / 2.0
        _G_DX = 2.0 / n_h
        bounds_h = [(0.0, 1.0)] * n_h
        cons_h = [{'type': 'eq', 'fun': con_sum}]
        
        h_512 = project_feasible(interp_to(best_h, n, n_h), n_h)
        c5_512 = compute_c5(h_512)
        
        print("\n--- Phase 5: n=512 (interp C5={:.10f}) ---".format(c5_512))
        
        for beta_val in [200, 600, 1500]:
            if budget_remain() < 20.0:
                break
            _G_BETA = float(beta_val)
            try:
                res = minimize(obj_smooth, h_512, method='SLSQP',
                               bounds=bounds_h, constraints=cons_h,
                               options={'maxiter': 400, 'ftol': 1e-15, 'iprint': 0})
                h_new = project_feasible(res.x, n_h)
                c5_new = compute_c5(h_new)
                if c5_new < c5_512 - 1e-13:
                    c5_512 = c5_new
                    h_512 = h_new
                    if c5_512 < best_c5:
                        best_c5 = c5_512
                        best_h = h_512.copy()
                        best_n = n_h
                        print("  n=512 beta={}: C5={:.10f}".format(beta_val, best_c5))
            except Exception:
                pass
        
        # Quick local search at 512
        if budget_remain() > 30.0 and c5_512 < best_c5 + 0.005:
            _G_N = n_h
            _G_TGT = n_h / 2.0
            _G_DX = 2.0 / n_h
            h_512, c5_512 = do_local_search_fine(
                h_512, n_h, budget_remain, rng, max_sweeps=30)
            if c5_512 < best_c5:
                best_c5 = c5_512
                best_h = h_512.copy()
                best_n = n_h
                print("  n=512 after LS: C5={:.10f}".format(best_c5))

    # Phase 6: Final SLSQP polish at best resolution
    if budget_remain() > 12.0:
        print("\n--- Phase 6: Final Polish ---")
        n_f = best_n
        _G_N = n_f
        _G_TGT = n_f / 2.0
        _G_DX = 2.0 / n_f
        bounds_f = [(0.0, 1.0)] * n_f
        cons_f = [{'type': 'eq', 'fun': con_sum}]
        
        for beta_val in [3000, 8000]:
            if budget_remain() < 6.0:
                break
            _G_BETA = float(beta_val)
            try:
                res = minimize(obj_smooth, best_h, method='SLSQP',
                               bounds=bounds_f, constraints=cons_f,
                               options={'maxiter': 1000, 'ftol': 1e-16, 'iprint': 0})
                h_new = project_feasible(res.x, n_f)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h_new
                    print("  Final beta={}: C5={:.10f}".format(beta_val, best_c5))
            except Exception:
                pass

    # Phase 7: Ultra-fine coordinate descent (last chance)
    if budget_remain() > 10.0:
        print("\n--- Phase 7: Ultra-Fine CD ---")
        best_h, best_c5 = do_local_search_fine(
            best_h, best_n, budget_remain, rng, max_sweeps=50)
        print("  Ultra-fine: C5={:.10f}".format(best_c5))

    # Final validation and return
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
