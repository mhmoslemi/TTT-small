def _c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / n)

def _proj(h, n):
    # Project to feasible set: 0<=h<=1, sum(h)=n/2
    h = np.array(h, dtype=np.float64)
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(50):
        d = 0.5 * n - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        k = int(np.sum(m))
        if k > 0:
            h[m] += d / k
        else:
            # If all are at bounds, need to move some
            if d > 0:
                # Need to increase sum, so decrease some 1s
                idx = np.where(h == 1.0)[0]
                if len(idx) > 0:
                    for idx_i in idx:
                        need = d
                        h[idx_i] -= min(need, 1.0)
                        d -= min(need, 1.0)
                        if d < 1e-14: break
            else:
                idx = np.where(h == 0.0)[0]
                if len(idx) > 0:
                    for idx_i in idx:
                        need = -d
                        h[idx_i] += min(need, 1.0)
                        d += min(need, 1.0)
                        if d > -1e-14: break
        np.clip(h, 0.0, 1.0, out=h)
    return h

    # Convert to binary: exactly n/2 ones
    # Use the values to determine which positions get 1s
    order = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[order[:n//2]] = 1.0
    return b

    if ns == nd:
        return h.copy()
    xs = (np.arange(ns) + 0.5) * (2.0 / ns)
    xd = (np.arange(nd) + 0.5) * (2.0 / nd)
    return np.interp(xd, xs, h)

def _swap_search(h, n, rng, max_sweeps=100, time_limit=60):
    """Binary swap local search"""
    t_start = time.time()
    h = h.copy()
    bc = _c5(h)
    
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit or _tl() < 5:
            break
        ones = np.where(h > 0.999)[0]
        zeros = np.where(h < 0.001)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        improved = False
        # Try all single swaps (greedy: best first)
        best_delta = -1e-10
        best_swap = None
        
        # For efficiency, sample a subset if too many
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        if n_ones * n_zeros > 50000:
            # Sample

            for _ in range(n_tries):
                i = ones[rng.randint(n_ones)]
                j = zeros[rng.randint(n_zeros)]
                h2 = h.copy()
                h2[i] = 0.0
                h2[j] = 1.0
                c2 = _c5(h2)
                if c2 < bc + best_delta:
                    best_delta = c2 - bc
                    best_swap = (i, j)
        else:
            for i in ones:
                for j in zeros:
                    h2 = h.copy()
                    h2[i] = 0.0
                    h2[j] = 1.0
                    c2 = _c5(h2)
                    if c2 < bc + best_delta:
                        best_delta = c2 - bc
                        best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            bc += best_delta
            improved = True
        
        if not improved:

            double_improved = False
            for _ in range(2000):
                if time.time() - t_start > time_limit or _tl() < 5:
                    break
                i1, i2 = rng.choice(ones, 2, replace=False) if len(ones) >= 2 else (None, None)

                if i1 is None or j1 is None:

                h2 = h.copy()

                h2[i2] = 0.0
                h2[j1] = 1.0

                    bc = c2
                    double_improved = True
                    ones = np.where(h > 0.999)[0]
                    zeros = np.where(h < 0.001)[0]
                    break
            
            if not double_improved:
                break
    
    return h, bc

def _eval_swap(a, p, q, n):
    """Evaluate the change in max value for flipping a[p] and a[q] (p:+1→-1, q:-1→+1)"""
    # Compute ΔA(m) for all m simultaneously
    # ΔA(m) = -2*a[p+m] - 2*a[p-m] + 2*a[q+m] + 2*a[q-m]
    
    # This can be done with numpy vectorization
    m = np.arange(1, n)  # m = 1, ..., n-1
    delta_A = np.zeros(n-1)
    
    # Contribution from p (flipping -2)
    # -2*a[p+m] for m where p+m < n
    idx1 = p + m
    mask1 = idx1 < n
    delta_A[mask1] += -2 * a[idx1[mask1]]
    
    # -2*a[p-m] for m where p-m >= 0
    idx2 = p - m
    mask2 = idx2 >= 0
    delta_A[mask2] += -2 * a[idx2[mask2]]
    
    # +2*a[q+m] for m where q+m < n
    idx3 = q + m
    mask3 = idx3 < n
    delta_A[mask3] += 2 * a[idx3[mask3]]
    
    # +2*a[q-m] for m where q-m >= 0
    idx4 = q - m
    mask4 = idx4 >= 0
    delta_A[mask4] += 2 * a[idx4[mask4]]

    # value(m) = (n-m) - A(m)
    # new_value(m) = (n-m) - (A(m) + delta_A(m)) = value(m) - delta_A(m)
    return delta_A  # caller can compute new max

import numpy as np
from scipy.optimize import minimize
import time

_t0 = 0.0
_tb = 1300.0


def _tl():
    return max(0.0, _tb - (time.time() - _t0) - 3.0)


def _c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / n)


def _proj(h, n):
    h = np.array(h, dtype=np.float64).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(100):
        d = 0.5 * n - h.sum()
        if abs(d) < 1e-15:
            break
        m = (h > 1e-14) & (h < 1.0 - 1e-14)
        k = int(np.sum(m))
        if k > 0:
            h[m] += d / k
        elif d > 0:
            idx = np.where(h >= 1.0 - 1e-14)[0]
            for idx_i in idx:
                amt = min(d, 1.0)
                h[idx_i] -= amt
                d -= amt
                if d < 1e-15:
                    break
        else:
            idx = np.where(h <= 1e-14)[0]
            for idx_i in idx:
                amt = min(-d, 1.0)
                h[idx_i] += amt
                d += amt
                if d > -1e-15:
                    break
        np.clip(h, 0.0, 1.0, out=h)
    d = 0.5 * n - h.sum()
    if abs(d) > 1e-14:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += d
        np.clip(h, 0.0, 1.0, out=h)
    return h


def _to_binary(h, n):
    order = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[order[:n // 2]] = 1.0
    return b


def _interp(h, ns, nd):
    if ns == nd:
        return h.copy()
    xs = (np.arange(ns) + 0.5) * (2.0 / ns)
    xd = (np.arange(nd) + 0.5) * (2.0 / nd)
    return np.interp(xd, xs, h)


def _swap_search(h, n, rng, time_limit=30, use_anneal=False):
    t_start = time.time()
    h = h.copy()
    bc = _c5(h)
    temp = 0.01

    for sweep in range(2000):
        if time.time() - t_start > time_limit or _tl() < 5:
            break

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        n_ones = len(ones)
        n_zeros = len(zeros)
        if n_ones == 0 or n_zeros == 0:
            break

        improved = False

        n_tries = min(n_ones * n_zeros, 25000)
        best_c2 = bc
        best_swap = None

        for _ in range(n_tries):
            i = ones[rng.randint(n_ones)]
            j = zeros[rng.randint(n_zeros)]
            h2 = h.copy()
            h2[i] = 0.0
            h2[j] = 1.0
            c2 = _c5(h2)
            if c2 < best_c2:
                best_c2 = c2
                best_swap = (i, j)

        if best_swap is not None:
            if best_c2 < bc - 1e-14 or (use_anneal and temp > 1e-6 and rng.rand() < np.exp(-(bc - best_c2) / temp)):
                i, j = best_swap
                h[i] = 0.0
                h[j] = 1.0
                bc = _c5(h)
                improved = True
                temp *= 0.995

        if not improved:
            double_improved = False
            n_dtries = 5000
            for _ in range(n_dtries):
                if time.time() - t_start > time_limit or _tl() < 5:
                    break
                if n_ones < 2 or n_zeros < 2:
                    break
                i1, i2 = rng.choice(ones, 2, replace=False)
                j1, j2 = rng.choice(zeros, 2, replace=False)
                h2 = h.copy()
                h2[i1] = 0.0
                h2[i2] = 0.0
                h2[j1] = 1.0
                h2[j2] = 1.0
                c2 = _c5(h2)
                if c2 < bc - 1e-14:
                    h = h2
                    bc = c2
                    double_improved = True
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
                    n_ones = len(ones)
                    n_zeros = len(zeros)
                    break
            if not double_improved:
                break

    return h, bc


def _perturb_reopt(h, n, rng, time_limit=20):
    t_start = time.time()
    best = h.copy()
    bc = _c5(best)

    for _ in range(20):
        if time.time() - t_start > time_limit or _tl() < 5:
            break
        h2 = best.copy()
        n_flip = rng.randint(2, 8)
        ones = np.where(h2 > 0.5)[0]
        zeros = np.where(h2 < 0.5)[0]
        if len(ones) < n_flip or len(zeros) < n_flip:
            break
        flip_to_zero = rng.choice(ones, n_flip, replace=False)
        flip_to_one = rng.choice(zeros, n_flip, replace=False)
        h2[flip_to_zero] = 0.0
        h2[flip_to_one] = 1.0
        h2, c2 = _swap_search(h2, n, rng, time_limit=min(5, _tl() * 0.3))
        if c2 < bc - 1e-14:
            best = h2
            bc = c2

    return best, bc


def _softmax_obj(x, n, beta):
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = np.max(c)
    shifted = np.clip(beta * (c - mx), -500, 500)
    val = mx + np.log(np.sum(np.exp(shifted))) / beta
    return val * (2.0 / n)


def _softmax_con(x, n):
    return x.sum() - 0.5 * n


def _slsqp_opt(h0, n, beta, maxiter):
    try:
        r = minimize(_softmax_obj, h0, args=(n, beta),
                     method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _softmax_con, 'args': (n,)}],
                     options={'maxiter': maxiter, 'ftol': 1e-13, 'iprint': 0})
        return _proj(np.asarray(r.x, dtype=np.float64), n)
    except Exception:
        return _proj(h0.copy(), n)


def _grad_descent(h, n, rng, niter=500, time_limit=30):
    t_start = time.time()
    best = h.copy()
    bc = _c5(best)

    for it in range(niter):
        if time.time() - t_start > time_limit or _tl() < 5:
            break

        corr = np.correlate(best, 1.0 - best, mode='full')
        kmax = int(np.argmax(corr))
        s = kmax - (n - 1)

        g = np.zeros(n)
        for m in range(n):
            idx2 = m + s
            if 0 <= idx2 < n:
                g[m] += (1.0 - best[idx2])
            idx1 = m - s
            if 0 <= idx1 < n:
                g[m] -= best[idx1]

        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn

        moved = False
        for alpha in [0.05, 0.02, 0.01, 0.005, 0.002, 0.03, 0.008, 0.001]:
            h2 = _proj(best - alpha * g, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-15:
                best = h2
                bc = c2
                moved = True
                break

        if not moved:
            for alpha in [0.0005, 0.0001, 1e-5]:
                h2 = _proj(best - alpha * g, n)
                c2 = _c5(h2)
                if c2 < bc - 1e-15:
                    best = h2
                    bc = c2
                    moved = True
                    break

    return best, bc


def _optimize_at_resolution(h_start, n, rng):
    t_start = time.time()
    best = h_start.copy()
    bc = _c5(best)

    # Binary search phase 1
    h_bin = _to_binary(best, n)
    h_b, c_b = _swap_search(h_bin, n, rng, time_limit=min(40, _tl() * 0.25))
    if c_b < bc:
        best, bc = h_b, c_b

    # Gradient descent
    h_gd, c_gd = _grad_descent(best, n, rng, niter=400, time_limit=min(30, _tl() * 0.2))
    if c_gd < bc:
        best, bc = h_gd, c_gd

    # SLSQP polish
    for beta in [100, 500, 2000]:
        if _tl() < 20:
            break
        h_s, c_s = _slsqp_opt(best, n, beta, 250)
        if c_s < bc:
            best, bc = h_s, c_s
        # Re-binary and search again
        h_b2, c_b2 = _swap_search(_to_binary(best, n), n, rng, time_limit=min(20, _tl() * 0.15))
        if c_b2 < bc:
            best, bc = h_b2, c_b2

    # Final binary polish
    h_b, c_b = _swap_search(_to_binary(best, n), n, rng, time_limit=min(30, _tl() * 0.2))
    if c_b < bc:
        best, bc = h_b, c_b

    # Final continuous
    h_gd, c_gd = _grad_descent(best, n, rng, niter=300, time_limit=min(20, _tl() * 0.15))
    if c_gd < bc:
        best, bc = h_gd, c_gd

    return best, bc
