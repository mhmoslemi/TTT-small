                j = int(zeros[rng.randint(len(zeros))])
                delta = _swap_delta(h, i, j, n)
                new_max = float(np.max(counts + delta))
                if new_max < max_count - 1e-12:
                    h[i], h[j] = h[j], h[i]
                    counts += delta
                    c5 = new_max * (2.0 / n)
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                    moved = True
                    break
            if not moved:
                break

    return h, c5


def _sa_binary(h, n, rng, time_limit=60.0):
    t_start = time.time()
    h = _to_binary(h, n)
    counts = _counts(h).copy()
    c5 = float(np.max(counts)) * (2.0 / n)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    T0 = 2.0 / n
    Tend = 0.1 * T0
    n_iter = int(time_limit * 800)
    best_h = h.copy()
    best_c5 = c5

    it = 0
    for it in range(n_iter):
        if _tl() < 5 or (time.time() - t_start) > time_limit:
            break
        T = T0 * (Tend / T0) ** (float(it) / max(1, n_iter))
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        delta = _swap_delta(h, i, j, n)
        new_max = float(np.max(counts + delta))
        old_max = float(np.max(counts))
        dc5 = (new_max - old_max) * (2.0 / n)
        if dc5 < 0 or (rng.rand() < np.exp(-dc5 / max(T, 1e-10))):
            h[i], h[j] = h[j], h[i]
            counts += delta
            c5 = float(np.max(counts)) * (2.0 / n)
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    return best_h, best_c5


def _cont_opt(h, n, budget=30.0):
    t_start = time.time()
    h = _proj(h.copy(), n)
    c5 = _c5(h)

    def _obj(x):
        cc = np.correlate(x, 1.0 - x, mode='full')
        mx = float(np.max(cc))
        sh = cc - mx
        b = 200.0
        return (mx + np.log(np.exp(b * sh).sum()) / b) * (2.0 / n)

    def _con_fun(x):
        return x.sum() - 0.5 * n

    for beta in [100, 500]:
        if _tl() < 10 or (time.time() - t_start) > budget:
            break
        try:
            r = minimize(_obj, h.copy(), method='SLSQP',
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq', 'fun': _con_fun}],
                         options={'maxiter': 200, 'ftol': 1e-14, 'iprint': 0})
            h2 = _proj(r.x, n)
            c2 = _c5(h2)
            if c2 < c5:
                h, c5 = h2, c2
        except Exception:
            pass

    # Subgradient descent
    for _ in range(300):
        if _tl() < 5 or (time.time() - t_start) > budget:
            break
        cc = np.correlate(h, 1.0 - h, mode='full')
        mx = float(np.max(cc))
        w = np.exp(80.0 * (cc - mx))
        sw = w.sum()
        if sw < 1e-30:
            w = np.zeros_like(cc)
            w[int(np.argmax(cc))] = 1.0
            sw = 1.0
        w /= sw
        g = np.zeros(n)
        nz = np.where(w > 1e-10)[0]
        for k_idx in nz:
            k = int(k_idx) - (n - 1)
            if k >= 0:
                if k == 0:
                    g[:] += w[k_idx] * (1.0 - 2.0 * h)
                else:
                    g[k:] += w[k_idx] * (1.0 - h[:n - k])
                    g[:n - k] -= w[k_idx] * h[k:]
            else:
                p = -k
                g[:n - p] += w[k_idx] * (1.0 - h[p:])
                g[p:] -= w[k_idx] * h[:n - p]
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        moved = False
        for a in [0.04, 0.02, 0.01, 0.005, 0.002]:
            h2 = _proj(h - a * g, n)
            c2 = _c5(h2)
            if c2 < c5 - 1e-14:
                h, c5 = h2, c2
                moved = True
                break
        if not moved:
            break

    return h, c5


def _poly_init(n, rng):
    """Create various initial patterns."""
    patterns = []
    # Random
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    patterns.append(h)
    # Block: 1s in middle
    h = np.zeros(n)
    h[n // 4:3 * n // 4] = 1.0
    h = _proj(h, n)
    patterns.append(h)
    # Periodic
    for period in [3, 4, 5, 7]:
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if (i % period) < (period - 1) else 0.0
        h = _proj(h, n)
        patterns.append(h)
    return patterns


def run(seed=42, budget_s=1300, **kwargs):
    global _t0, _tb
    _t0 = time.time()
    _tb = budget_s
    rng = np.random.RandomState(seed)

    try:
        hpar = np.array(initial_h_values, dtype=np.float64)
        npar = len(hpar)
    except Exception:
        npar = 160
        hpar = np.zeros(npar, dtype=np.float64)
        hpar[:npar // 2] = 1.0

    bh = _proj(hpar.copy(), npar)
    bc5 = _c5(bh)
    bn = npar
    print("Init n={} C5={:.8f}".format(bn, bc5))

    resolutions = sorted(set([npar, npar - 4, npar + 4, npar + 8, npar - 8,
                               120, 140, 180, 200, 240, 256, 280, 320, 360, 400]),
                         key=lambda x: abs(x - npar))
    resolutions = [x for x in resolutions if 60 <= x <= 500 and x % 2 == 0]

    # Phase 1: Quick binary local search on parent at original resolution
    if _tl() > 30:
        hb, cb = _bin_ls(bh, npar, rng, max_iter=1000, budget=min(60, _tl() - 10))
        if cb < bc5:
            bh, bc5, bn = hb, cb, npar
            print("BinLS n={} C5={:.8f}".format(npar, cb))

    # Phase 2: Multi-resolution search
    for nn in resolutions:
        if _tl() < 40:
            break
        if nn < 60 or nn % 2 != 0:
            continue
        # Interpolate current best to new resolution
        h_try = _proj(_interp(bh, bn, nn), nn)
        # Quick binary local search
        if _tl() > 20:
            hb, cb = _bin_ls(h_try, nn, rng, max_iter=800, budget=min(50, _tl() - 10))
            if cb < bc5:
                bh, bc5, bn = hb, cb, nn
                print("Bin n={} C5={:.8f}".format(nn, cb))
        # Simulated annealing on best
        if _tl() > 30 and (bc5 < 0.381 or nn == npar):
            hs, cs = _sa_binary(bh, bn, rng, time_limit=min(40, _tl() - 10))
            if cs < bc5:
                bh, bc5, bn = hs, cs, bn
                print("SA n={} C5={:.8f}".format(bn, cs))

    # Phase 3: Continuous optimization on best
    if _tl() > 20:
        hc, cc = _cont_opt(bh, bn, budget=min(80, _tl() - 10))
        if cc < bc5:
            bh, bc5 = hc, cc
            print("Cont n={} C5={:.8f}".format(bn, cc))

    # Phase 4: More binary local search with more iterations
    if _tl() > 30:
        hb, cb = _bin_ls(bh, bn, rng, max_iter=3000, budget=min(120, _tl() - 10))
        if cb < bc5:
            bh, bc5 = hb, cb
            print("BinLS2 n={} C5={:.8f}".format(bn, cb))

    # Phase 5: Simulated annealing for more time
    if _tl() > 30:
        hs, cs = _sa_binary(bh, bn, rng, time_limit=min(100, _tl() - 10))
        if cs < bc5:
            bh, bc5, bn = hs, cs, bn
            print("SA2 n={} C5={:.8f}".format(bn, cs))

    # Phase 6: Final continuous polish
    if _tl() > 15:
        hc, cc = _cont_opt(bh, bn, budget=min(60, _tl() - 8))
        if cc < bc5:
            bh, bc5 = hc, cc
            print("Cont2 n={} C5={:.8f}".format(bn, cc))

    # Phase 7: One more round of binary + continuous
    if _tl() > 20:
        hb, cb = _bin_ls(bh, bn, rng, max_iter=1500, budget=min(60, _tl() - 10))
        if cb < bc5:
            bh, bc5 = hb, cb
        if _tl() > 10:
            hc, cc = _cont_opt(bh, bn, budget=min(40, _tl() - 5))
            if cc < bc5:
                bh, bc5 = hc, cc

    # Phase 8: Try a couple more resolutions if time
    for nn in [bn + 16, bn - 16, bn + 32, bn - 32]:
        if _tl() < 30:
            break
        if nn < 60 or nn > 500 or nn % 2 != 0:
            continue
        h_try = _proj(_interp(bh, bn, nn), nn)
        hb, cb = _bin_ls(h_try, nn, rng, max_iter=600, budget=min(40, _tl() - 10))
        if cb < bc5:
            bh, bc5, bn = hb, cb, nn
            print("Fin n={} C5={:.8f}".format(nn, cb))
        if _tl() > 10:
            hc, cc = _cont_opt(hb, nn, budget=min(20, _tl() - 5))
            if cc < bc5:
                bh, bc5, bn = hc, cc, nn
                print("FinC n={} C5={:.8f}".format(nn, cc))

    bh = _proj(bh, bn)
    fc5 = _c5(bh)
    try:
        ec = evaluate_erdos_solution(bh)
        if ec < fc5:
            fc5 = ec
    except Exception:
        pass
    print("FINAL n={} C5={:.8f}".format(bn, fc5))
    return (bh, fc5, bn)
