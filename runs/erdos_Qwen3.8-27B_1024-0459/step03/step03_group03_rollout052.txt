```python
import numpy as np
from scipy.optimize import minimize
import time

_t0 = 0.0
_tb = 1300.0
_ng = 0
_bv = 100.0

def _tl():
    return max(0.0, _tb - (time.time() - _t0) - 3.0)

def _c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / n)

def _proj(h, n):
    h = np.array(h, dtype=np.float64).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(50):
        d = 0.5 * n - h.sum()
        if abs(d) < 1e-15:
            break
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        k = int(np.sum(m))
        if k > 0:
            h[m] += d / k
        else:
            h[:] = 0.5
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _interp_f(h, ns, nd):
    if ns == nd:
        return h.copy()
    xs = (np.arange(ns) + 0.5) * (2.0 / ns)
    xd = (np.arange(nd) + 0.5) * (2.0 / nd)
    return np.interp(xd, xs, h)

def _scon(x):
    return x.sum() - 0.5 * _ng

def _sobj(x):
    n = _ng
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = float(np.max(c))
    b = _bv
    sh = c - mx
    return (mx + np.log(np.exp(b * sh).sum()) / b) * (2.0 / n)

def _slsqp(h0, n, beta, maxiter):
    global _ng, _bv
    _ng = n
    _bv = beta
    try:
        r = minimize(_sobj, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon}],
                     options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return _proj(np.asarray(r.x, dtype=np.float64), n)
    except Exception:
        return _proj(h0.copy(), n)

def _ggrad(h, k):
    n = len(h)
    shift = k - (n - 1)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[0:n-shift] += (1.0 - h[shift:n])
            g[shift:n] -= h[0:n-shift]
    else:
        p = -shift
        if p < n:
            g[p:n] += (1.0 - h[0:n-p])
            g[0:n-p] -= h[p:n]
    return g

def _smgrad(h, bw):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(corr))
    w = np.exp(bw * (corr - mx))
    sw = float(np.sum(w))
    if sw < 1e-30:
        return _ggrad(h, int(np.argmax(corr)))
    w /= sw
    g = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for k in nz:
        g += w[int(k)] * _ggrad(h, int(k))
    return g

def _sgdescend(h, n, nit, rng):
    best = h.copy()
    bc = _c5(best)
    stall = 0
    for it in range(nit):
        if _tl() < 3:
            break
        moved = False
        for bw in [100.0, 300.0, 800.0, 2000.0]:
            g = _smgrad(best, bw)
            gn = float(np.linalg.norm(g))
            if gn < 1e-15:
                break
            g /= gn
            for a in [0.04, 0.015, 0.008, 0.004, 0.02, 0.06, 0.01, 0.002]:
                h2 = _proj(best - a * g, n)

                for j in zeros:

                    c2 = _c5(h2)
                    if c2 < best_c - 1e-15:

                        best_j = j
                if best_j >= 0:
                    h[ones[0]], h[best_j] = h[best_j], h[ones[0]]
                    h[i], h[best_j] = h[best_j], h[i]
                    bc = best_c
                    imp = True
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]

            if imp:

        else:
            ntry = min(800, len(ones) * len(zeros))

                i = ones[rng.randint(len(ones))]

                    bc = c2
                    imp = True

    return h, bc

        idx = np.where(h == 1)[0]

        rng.shuffle(idx)

    curr = _c5(h)
    best = h.copy()
    best_c = curr
    for it in range(n_iter):
        if _tl() < 3:
            break
        t = t0 * (t1 / t0) ** (float(it) / max(n_iter - 1, 1))
        i = rng.randint(n)

        h[i], h[j] = h[j], h[i]

            if curr < best_c:
                best = h.copy()
                best_c = curr

            h[i], h[j] = h[j], h[i]
```
