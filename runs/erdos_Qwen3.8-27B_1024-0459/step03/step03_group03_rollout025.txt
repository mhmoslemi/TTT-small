            i = rng.randint(n)
            h3[i] = 1.0 - h3[i]
        h3 = _proj(h3, n)
        candidates.append(h3)
    return candidates


def run(seed=42, budget_s=1300, **kwargs):
    global _t0, _tb
    _t0 = time.time()
    _tb = budget_s
    rng = np.random.RandomState(seed)

    # Get parent solution
    try:
        hpar = np.array(initial_h_values, dtype=np.float64)
        npar = len(hpar)
    except Exception:
        npar = 160
        hpar = np.zeros(npar, dtype=np.float64)
        hpar[:npar // 2] = 1.0

    bh = _proj(hpar, npar)
    bc5 = _c5(bh)
    bn = npar
    print("Init n={} C5={:.8f}".format(bn, bc5))

    # Try main optimization at parent resolution
    h, c5 = _full_optimize(bh, npar, rng)
    if c5 < bc5:
        bh, bc5 = h, c5
        print("Opt n={} C5={:.8f}".format(npar, c5))

    # Try perturbed starts at same resolution
    if _tl() > 60:
        starts = _perturb_start(bh, npar, rng, 4)
        for s in starts[1:]:
            if _tl() < 40:
                break
            h2, c2 = _full_optimize(s, npar, rng)
            if c2 < bc5:
                bh, bc5 = h2, c2
                print("Perturb n={} C5={:.8f}".format(npar, c2))

    # Try different resolutions
    alt_ns = [npar + 4, npar - 4, npar + 8, npar - 8, npar + 6, npar + 10, npar - 6]
    for nn in alt_ns:
        if _tl() < 50:
            break
        if nn < 40 or nn % 2 != 0:
            continue
        hi = _proj(_interp_f(bh, bn, nn), nn)
        h2, c2 = _full_optimize(hi, nn, rng)
        if c2 < bc5:
            bh, bc5, bn = h2, c2, nn
            print("Res n={} C5={:.8f}".format(nn, c2))

    # Final intensive polishing at best resolution
    if _tl() > 30:
        h, c5 = _subgrad_step(bh, bn, 500, rng, 0.005)
        if c5 < bc5:
            bh, bc5 = h, c5
    if _tl() > 25:
        h, c5 = _two_swap(bh, bn, 50, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
    if _tl() > 20:
        # Very fine subgradient with tiny steps
        h, c5 = _subgrad_step(bh, bn, 400, rng, 0.002)
        if c5 < bc5:
            bh, bc5 = h, c5
    if _tl() > 15:
        for beta in [8000, 20000]:
            h2 = _lbfgs(bh, bn, beta, 500, gtol=1e-13)
            c2 = _c5(h2)
            if c2 < bc5:
                bh, bc5 = h2, c2
                print("Polish beta={} n={} C5={:.8f}".format(beta, bn, c2))
    if _tl() > 12:
        h, c5 = _subgrad_step(bh, bn, 300, rng, 0.001)
        if c5 < bc5:
            bh, bc5 = h, c5
    if _tl() > 10:
        h, c5 = _quantize_descent(bh, bn, 100, rng)
        if c5 < bc5:
            bh, bc5 = h, c5
    if _tl() > 8:
        h, c5 = _two_swap(bh, bn, 30, rng)
        if c5 < bc5:
            bh, bc5 = h, c5

    bh = _proj(bh, bn)
    fc5 = _c5(bh)
    try:
        ec = evaluate_erdos_solution(bh)
        if ec < fc5:
            fc5 = ec
    except Exception:
        pass
    print("FINAL n={} C5={:.8f}".format(bn, fc5))
    return (bh, fc5, bn)
