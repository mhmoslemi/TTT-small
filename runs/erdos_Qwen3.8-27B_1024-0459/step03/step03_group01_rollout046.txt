_BUDGET = 1300.0

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode='full')

    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)

    h = np.array(h, dtype=float).copy()

    target = n / 2.0

        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def full_scan_local_search(h, n, max_iters=200, time_limit=30, patience=20):
    """Full scan of all possible pair moves, pick the best one each iteration."""
    if np.std(h) > 0.01:
        h = (h > 0.5).astype(float).copy()

        if n_ones > n // 2:

            rem = np.random.choice(idx, n_ones - n//2, replace=False)
            h[rem] = 0
        elif n_ones < n // 2:
            idx = np.where(h == 0)[0]
            add = np.random.choice(idx, n//2 - n_ones, replace=False)
            h[add] = 1
    
    h = h.copy()
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * (2.0/n)
    start = time.time()
    no_improve = 0

        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_i, best_j = -1, -1

    target = n // 2
    cur = int(np.sum(h))
    if cur > target:
        ones_idx = np.where(h == 1)[0]
        remove = rng.choice(ones_idx, cur - target, replace=False)
        h[remove] = 0.0
    elif cur < target:
        zeros_idx = np.where(h == 0)[0]
        add = rng.choice(zeros_idx, target - cur, replace=False)
        h[add] = 1.0
    return h

def make_periodic(n, seed_val):
    """Create a periodic-like pattern."""
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    pattern_len = max(2, n // 80)
    pattern = np.zeros(pattern_len)
    n_ones_in_pat = max(1, pattern_len // 2)
    ones_pos = rng.choice(pattern_len, n_ones_in_pat, replace=False)
    pattern[ones_pos] = 1.0
    for i in range(n):
        h[i] = pattern[i % pattern_len]
    # Fix count
    target = n // 2
    cur = int(np.sum(h))
    if cur > target:
        ones_idx = np.where(h == 1)[0]
        remove = rng.choice(ones_idx, cur - target, replace=False)
        h[remove] = 0.0
    elif cur < target:
        zeros_idx = np.where(h == 0)[0]
        add = rng.choice(zeros_idx, target - cur, replace=False)
        h[add] = 1.0
    return h

def ensure_binary_count(h, n, rng):
    h = h.copy()
    target = n // 2
    cur = int(np.sum(h))
    if cur > target:
        idx = np.where(h == 1)[0]
        rem = rng.choice(idx, cur - target, replace=False)
        h[rem] = 0.0
    elif cur < target:
        idx = np.where(h == 0)[0]
        add = rng.choice(idx, target - cur, replace=False)
        h[add] = 1.0
    return h

def full_scan_step(h, corr, n, rng, mode='greedy', temp=0.0):
    """Try all possible pair moves, return the best (or SA-accepted) move."""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h, corr, False
    
    cur_max = float(np.max(corr))
    
    # For large n, sample; for small n, full scan
    total_pairs = len(ones) * len(zeros)
    
    if total_pairs <= 40000:
        # Full scan
        best_gain = 1e-10
        best_i, best_j = -1, -1
        best_new_max = cur_max
        best_new_corr = None
        
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_corr = compute_corr(h)
                new_max = float(np.max(new_corr))
                h[i] = 1.0
                h[j] = 0.0
                
                if mode == 'greedy':
                    if new_max < best_new_max - 1e-12:
                        best_new_max = new_max
                        best_i, best_j = int(i), int(j)
                        best_new_corr = new_corr
                else:
                    # Simulated annealing acceptance
                    delta = new_max - cur_max
                    if delta < -1e-12 or (temp > 0 and rng.rand() < np.exp(-delta / max(temp, 1e-6))):
                        best_i, best_j = int(i), int(j)
                        best_new_max = new_max
                        best_new_corr = new_corr
                        break
            if mode != 'greedy' and best_i >= 0:
                break
    else:
        # Random sampling
        n_samples = min(20000, total_pairs)
        best_gain = 1e-10
        best_i, best_j = -1, -1
        best_new_max = cur_max
        best_new_corr = None
        
        for _ in range(n_samples):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            h[i] = 1.0
            h[j] = 0.0
            
            if mode == 'greedy':
                if new_max < best_new_max - 1e-12:
                    best_new_max = new_max
                    best_i, best_j = int(i), int(j)
                    best_new_corr = new_corr
            else:
                delta = new_max - cur_max
                if delta < -1e-12 or (temp > 0 and rng.rand() < np.exp(-delta / max(temp, 1e-6))):
                    best_i, best_j = int(i), int(j)
                    best_new_max = new_max
                    best_new_corr = new_corr
                    break
    
    if best_i >= 0:
        h[best_i] = 0.0
        h[best_j] = 1.0
        corr = best_new_corr if best_new_corr is not None else compute_corr(h)
        return h, corr, True
    return h, corr, False

def binary_optimize(h_init, n, time_limit=60, rng=None):
    """Intensive binary optimization with full-scan + SA."""
    if rng is None:
        rng = np.random.RandomState(42)
    
    h = ensure_binary_count(h_init.copy(), n, rng)
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * (2.0 / n)
    best_h = h.copy()
    start = time.time()
    
    # Phase 1: Greedy full-scan (best improvement)
    max_iters_greedy = 300
    for it in range(max_iters_greedy):
        if time.time() - start > time_limit * 0.5 or remaining() < 10:
            break
        h, corr, improved = full_scan_step(h, corr, n, rng, mode='greedy')
        if not improved:
            break
        cur_max = float(np.max(corr))
        if cur_max * (2.0/n) < best_c5:
            best_c5 = cur_max * (2.0/n)
            best_h = h.copy()
    
    # Phase 2: Simulated annealing
    if time.time() - start < time_limit * 0.7 and remaining() > 15:
        h = best_h.copy()
        corr = compute_corr(h)
        temp = 0.5
        cooling = 0.98
        n_sa_iters = 5000
        
        for it in range(n_sa_iters):
            if time.time() - start > time_limit or remaining() < 10:
                break
            h, corr, moved = full_scan_step(h, corr, n, rng, mode='sa', temp=temp)
            if moved:
                cur_max = float(np.max(corr))
                if cur_max * (2.0/n) < best_c5:
                    best_c5 = cur_max * (2.0/n)
                    best_h = h.copy()
            temp *= cooling
            if temp < 0.05:
                break
    
    # Phase 3: Final greedy polish
    if time.time() - start < time_limit and remaining() > 8:
        h = best_h.copy()
        corr = compute_corr(h)
        for it in range(100):
            if time.time() - start > time_limit or remaining() < 8:
                break
            h, corr, improved = full_scan_step(h, corr, n, rng, mode='greedy')
            if not improved:
                break
            cur_max = float(np.max(corr))
            if cur_max * (2.0/n) < best_c5:
                best_c5 = cur_max * (2.0/n)
                best_h = h.copy()
    
    return best_h, best_c5

def slsqp_smooth_obj(x, n, beta):
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    s = np.sum(ex)
    return (m + np.log(s) / beta) * dx_val

def slsqp_smooth_grad(x, n, beta):
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    s = np.sum(ex)
    w = ex / s
    g = np.zeros(n)
    # Compute gradient using FFT for speed
    # corr[k] = sum_i h[i] * (1-h[i+k-(n-1)])
    # d(corr[k])/d(h[p]) = (1-h[p+k-(n-1)]) - h[p+k-(n-1)+k-(n-1)] ... complex
    # Use numerical-ish approach via the correlation structure
    # corr_full[k] = sum_{valid i} h[i + k - n + 1] * (1 - h[i])
    # For shift m = k - n + 1:
    #   corr[m] = sum_{i=max(0,-m)}^{min(n-1, n-1-m)} h[i+m]*(1-h[i])
    # d/dh[p]: appears as h[p] (first factor) when i+m=p → i=p-m, term (1-h[p-m])
    #           appears as h[p] (second factor) when i=p, term -h[p+m]
    # So d(corr[m])/dh[p] = (1-h[p-m])*[valid] + (-h[p+m])*[valid]
    
    # This is O(n^2) but n is at most ~400 so it's fine
    for p in range(n):
        dp = 0.0
        # p as first factor: i = p - m, valid when 0 <= p-m <= n-1 and 0 <= p <= n-1
        for m in range(-(n-1), n):
            i = p - m
            if 0 <= i < n and 0 <= p < n:
                # This contributes to corr at k = m + n - 1
                k = m + n - 1
                if 0 <= k < 2*n - 1:
                    dp += w[k] * (1.0 - h_val(h, p, m, n))
        # p as second factor: i = p, term is -h[p+m], valid when 0 <= p+m < n
        for m in range(-(n-1), n):
            p2 = p + m
            if 0 <= p2 < n and 0 <= p < n:
                k = m + n - 1
                if 0 <= k < 2*n - 1:
                    dp += w[k] * (-h[p2])
        g[p] = dp * dx_val
    return g

def h_val(h, p, m, n):
    """Get h[p-m] if valid, else 0."""
    idx = p - m
    if 0 <= idx < n:
        return h[idx]
    return 0.0

def slsqp_optimize(h, n, beta, maxiter=200):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return slsqp_smooth_obj(x, n, beta)
    
    def grad(x):
        return slsqp_smooth_grad(x, n, beta)
    
    def constr_eq(x):
        return np.sum(x) - target
    
    def constr_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': constr_eq, 'jac': constr_jac}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def active_set_refine(h, n, max_iters=100):
    """Equalize top correlations using projected gradient."""
    h = h.copy()
    best_c5 = compute_c5(h)
    dx_val = 2.0 / n
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        # Get top 3 shifts
        top_idx = np.argsort(-corr)[:3]
        top_val = corr[top_idx[0]]
        
        # Check if top 3 are well-equalized
        if corr[top_idx[0]] - corr[top_idx[-1]] < 0.5:
            break
        
        # Build gradient toward reducing max
        g = np.zeros(n)
        weights = np.exp(50 * (corr[top_idx] - top_val))
        weights = weights / np.sum(weights)
        
        for wi, ki in zip(weights, top_idx):
            k = int(ki)
            m = k - (n - 1)
            # d(corr[k])/d(h[p])
            for p in range(n):
                # p as first factor in corr: i = p - m
                i = p - m
                if 0 <= i < n and 0 <= p < n:
                    g[p] += wi * (1.0 - h[i])
                # p as second factor: i = p, h[p+m]
                p2 = p + m
                if 0 <= p2 < n:
                    g[p] += wi * (-h[p2])
        
        # Projected gradient descent
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g /= gnorm
        
        improved = False
        for step in [0.005, 0.01, 0.02, 0.001, 0.05]:
            h_trial = h - step * g
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-12:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            break
    
    return h, best_c5

def interp_binary(src_h, src_n, dst_n, rng):
    """Interpolate binary solution to new resolution using nearest."""
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    h_cont = np.interp(x_new, xp, src_h)
    h_bin = (h_cont > 0.5).astype(float)
    return ensure_binary_count(h_bin, dst_n, rng)

def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 320
        parent_h = thue_morse(parent_n, seed)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    # Round parent to binary for combinatorial optimization
    h_bin_parent = (parent_h > 0.5).astype(float)
    h_bin_parent = ensure_binary_count(h_bin_parent, parent_n, rng)
    c5_bin_parent = compute_c5(h_bin_parent)
    
    # === Phase 1: Intensive binary optimization at parent resolution ===
    if remaining() > 60:
        t_alloc = min(120, remaining() * 0.3)
        h_b, c5_b = binary_optimize(h_bin_parent, best_n, time_limit=t_alloc, rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = best_n
    
    # === Phase 2: Try multiple resolutions ===
    n_list = sorted(set([
        best_n, best_n - 2, best_n + 2, best_n + 4,
        best_n - 4, best_n + 6, best_n + 8,
        256, 288, 320, 352, 384,
        int(best_n * 0.75), int(best_n * 1.25), int(best_n * 1.5)
    ]))
    n_list = [nn for nn in n_list if 128 <= nn <= 512 and nn % 2 == 0]
    n_list = sorted(set(n_list))
    
    # Allocate time per resolution
    n_res = len(n_list)
    time_per = min(60, max(20, (remaining() - 30) / max(1, n_res)))
    
    for n_t in n_list:
        if remaining() < 20:
            break
        if n_t == best_n:
            continue
        
        # Generate initial at this resolution
        h_init = None
        for init_seed in [seed, seed + 1, seed + 2]:
            if n_t == parent_n:
                h_candidate = h_bin_parent.copy()
            else:
                # Interpolate from best
                h_candidate = interp_binary(best_h, best_n, n_t, rng)
            
            h_candidate_c5 = compute_c5(h_candidate)
            if h_init is None or h_candidate_c5 < compute_c5(h_init):
                h_init = h_candidate
            break
        
        # Also try Thue-Morse
        h_tm = thue_morse(n_t, seed + n_t)
        if compute_c5(h_tm) < compute_c5(h_init):
            h_init = h_tm
        
        # Binary optimize
        t_alloc = min(time_per, remaining() - 15)
        if t_alloc > 10:
            h_b, c5_b = binary_optimize(h_init, n_t, time_limit=t_alloc, rng=rng)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_t
    
    # === Phase 3: Multi-start binary optimization at best_n ===
    n_best = best_n
    if remaining() > 30:
        h_starts = [best_h.copy()]
        h_starts.append(thue_morse(n_best, seed + 100))
        h_starts.append(make_periodic(n_best, seed + 200))
        h_starts.append(thue_morse(n_best, seed + 300))
        
        t_per_start = min(40, max(15, (remaining() - 20) / len(h_starts)))
        
        for h_s in h_starts:
            if remaining() < 15:
                break
            h_s = ensure_binary_count(h_s, n_best, rng)
            h_b, c5_b = binary_optimize(h_s, n_best, time_limit=t_per_start, rng=rng)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_best
    
    # === Phase 4: Continuous refinement (SLSQP + active set) ===
    n = best_n
    h = best_h.copy()
    
    # SLSQP with smooth max
    if remaining() > 20:
        for beta in [50, 100, 200, 500]:
            if remaining() < 8:
                break
            h_new = slsqp_optimize(h, n, beta, maxiter=150)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
    
    # Active set refinement
    if remaining() > 10:
        h_eq, c5_eq = active_set_refine(h, n, max_iters=100)
        if c5_eq < best_c5 - 1e-12:
            best_c5 = c5_eq
            best_h = h_eq.copy()
    
    # === Phase 5: Final binary micro-optimization ===
    if remaining() > 15:
        h_final_bin = (best_h > 0.5).astype(float)
        h_final_bin = ensure_binary_count(h_final_bin, best_n, rng)
        h_b, c5_b = binary_optimize(h_final_bin, best_n, 
                                    time_limit=min(30, remaining() - 5), rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = best_n
    
    # === Phase 6: Final SLSQP polish ===
    if remaining() > 8:
        for beta in [200, 500, 1000]:
            if remaining() < 5:
                break
            h_new = slsqp_optimize(best_h, best_n, beta, maxiter=100)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
