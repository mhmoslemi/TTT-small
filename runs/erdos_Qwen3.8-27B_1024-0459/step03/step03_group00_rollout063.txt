This is the classical "minimum difference set" problem in continuous form!

For the discrete case with n points, n/2 of which are "on":
- We want to minimize max_k (number of i where h[i]=1 and h[i+k]=0)
- This equals n/2 - min_k (number of i where h[i]=1 and h[i+k]=1)
- = n/2 - max_k autocorrelation at lag k

So we want to maximize the minimum nonzero autocorrelation. This is the "flat autocorrelation" problem.

For binary sequences, the optimal flat autocorrelation is achieved by:
- m-sequences (for prime or prime-power lengths)
- Barker sequences (shorter)
- Costas arrays (different but related)

For n=256 (not prime), the best we can do is close to the theoretical bound.

The theoretical lower bound on max sidelobe for a binary sequence of length n with n/2 ones:
The average of |C(k)| for k=1,...,n-1 is at most √(n/2) by the Welch bound (roughly).

So the minimum possible max C(k) for negative lags would be around -√(n/2) to -1 or so.

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-16:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xin = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xin, xp, src_h)

def smooth_c5_and_grad(h, beta, top_k=None):
    """Compute smooth C5 value and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    
    if top_k is not None and top_k < len(corr):
        # Use only top-k terms
        top_idx = np.argsort(corr)[-top_k:]
        w_full = np.zeros(len(corr))
        shifted = corr[top_idx] - m
        w_full[top_idx] = np.exp(beta * shifted)
        w_full[top_idx] /= np.sum(w_full[top_idx])
        val = dx * (m + np.log(np.sum(np.exp(beta * shifted))) / beta)
    else:
        shifted = corr - m
        log_w = beta * shifted
        log_w -= np.max(log_w)
        w_full = np.exp(log_w)
        w_full /= np.sum(w_full)
        val = dx * (m + np.log(np.sum(np.exp(beta * shifted))) / beta)

def binary_local_search(h0, n, time_limit, rng):
    """Simulated annealing style binary local search."""
    h = project_feasible(h0, n)
    # Snap to binary
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    # Fix sum if needed
    ones_count = int(np.sum(h_bin))
    target_ones = n // 2
    if ones_count > target_ones:
        # Randomly turn off some 1s
        one_idx = np.where(h_bin == 1.0)[0]
        flip = rng.permutation(one_idx)[:ones_count - target_ones]
        h_bin[flip] = 0.0
    elif ones_count < target_ones:
        zero_idx = np.where(h_bin == 0.0)[0]
        flip = rng.permutation(zero_idx)[:target_ones - ones_count]
        h_bin[flip] = 1.0
    
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    t_start = time.time()
    
    # Start temperature
    T = 0.01
    T_min = 0.0001
    cooling = 0.995
    
    current_h = h_bin.copy()
    current_c5 = best_c5
    
    while (time.time() - t_start) < time_limit:
        # Generate neighbor
        move = rng.randint(0, 4)
        trial = current_h.copy()
        
        if move == 0:
            # Single flip (with compensation)
            if rng.rand() < 0.5:
                # Flip a 1 to 0 and a 0 to 1
                ones = np.where(trial == 1.0)[0]
                zeros = np.where(trial == 0.0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    trial[i] = 0.0
                    trial[j] = 1.0
            else:
                # Just flip one (allow temporary constraint violation, project)
                i = rng.randint(n)
                trial[i] = 1.0 - trial[i]
                trial = project_feasible(trial, n)
        elif move == 1:
            # Swap adjacent
            i = rng.randint(0, n-1)
            if trial[i] != trial[i+1]:
                trial[i], trial[i+1] = trial[i+1], trial[i]
            else:
                i = rng.randint(n)
                trial[i] = 1.0 - trial[i]
                trial = project_feasible(trial, n)
        elif move == 2:
            # Block rotation
            i = rng.randint(n-2)
            length = rng.randint(2, min(8, n-i))
            block = trial[i:i+length].copy()
            shift = rng.randint(1, length)
            trial[i:i+length] = np.concatenate([block[shift:], block[:shift]])
        else:
            # Double flip
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                trial[i] = 1.0 - trial[i]
                trial[j] = 1.0 - trial[j]
                # Check sum
                if abs(np.sum(trial) - n/2.0) > 1e-10:
                    trial = project_feasible(trial, n)
        
        c5_trial = compute_c5(trial)
        delta = c5_trial - current_c5
        
        # Metropolis criterion
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current_h = trial
            current_c5 = c5_trial
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = trial.copy()
        
        T = max(T * cooling, T_min)
    
    return best_h, best_c5

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-16:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xin = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xin, xp, src_h)


def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = beta * (corr - m)
    s_max = np.max(s)
    return dx * (m + np.exp(s_max) + np.log(np.sum(np.exp(s - s_max)) - (np.exp(s_max) - 1.0)) ) / beta


def smooth_val(x, beta):

    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')

    s_max = np.max(s)

def smooth_grad(x, beta):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = beta * (corr - m)
    s_max = np.max(s)
    log_w = s - s_max
    w = np.exp(log_w)
    w /= np.sum(w)

    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-16:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            np_ = n - lag
            grad[:np_] += wk * (1.0 - x[lag:lag + np_])
            grad[lag:lag + np_] -= wk * x[:np_]

            npc = -lag
            np_ = n - npc
            grad[npc:npc + np_] += wk * (1.0 - x[:np_])
            grad[:np_] -= wk * x[npc:npc + np_]
    return grad


def projected_gradient(x0, n, betas, max_iter_per_beta, time_limit, t0):
    h = project_feasible(x0, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n

    for beta in betas:
        if time.time() - t0 > time_limit:
            break
        lr = 1.0
        stagnant = 0
        for it in range(max_iter_per_beta):
            if time.time() - t0 > time_limit:
                break
            grad = smooth_grad(h, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break

            step = lr * dx / (gnorm + 1e-30)

        tri = cur_h.copy()

        elif move == 3:
            i = rng.randint(n)

            if i != j:
                tri[i] = 1.0 - tri[i]
                tri[j] = 1.0 - tri[j]

def slsqp_polish(h0, n, beta, maxiter, t0, time_limit):
    def objf(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = beta * (corr - m)
        sm = np.max(s)
        return (m + sm + np.log(np.sum(np.exp(s - sm)))) / beta * (2.0 / n)

    def objg(x):
        dx = 2.0 / n
        return smooth_grad(x, beta) * dx

    def con_eq(x):
        return np.sum(x) - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_eq}]
    try:
        res = minimize(objf, h0, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0, compute_c5(h0)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = 0
        v = i
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return project_feasible(h, n)


def make_golden(n):
    h = np.zeros(n)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        h[i] = 1.0 if (i * gr) % 1.0 < 0.5 else 0.0
    return project_feasible(h, n)


def make_random_bin(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h


def make_sinusoidal(rng, n, freq, phase):

    h = 0.5 + 0.48 * np.sin(2.0 * np.pi * freq * t + phase)

            print("  Final SA: C5={:.10f}".format(best_c5))

    # Final projection and validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
