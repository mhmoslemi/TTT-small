            grad[sp:] -= wj * x[:n-sp]

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            # Need to increase: prioritize elements with most room
            room = 1.0 - h
            mask = room > 1e-10
            if np.sum(mask) == 0:
                break
            # Water filling: add evenly to elements with most room
            add_per = diff / np.sum(mask)
            h[mask] = np.minimum(h[mask] + add_per, 1.0)
        else:
            # Need to decrease: prioritize elements with most value
            mask = h > 1e-10
            if np.sum(mask) == 0:
                break
            sub_per = (-diff) / np.sum(mask)
            h[mask] = np.maximum(h[mask] - sub_per, 0.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Find element with most room in the needed direction
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmax(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_sa(h_init, n, iterations, rng, t_start=0.003, t_end=0.00003):
    """Binary simulated annealing with fast evaluation"""
    h = np.where(h_init > 0.5, 1.0, 0.0)
    target_ones = n // 2
    num_ones = int(np.sum(h))
    
    # Fix number of ones
    if num_ones > target_ones:
        ones_idx = np.where(h == 1.0)[0]
        remove = rng.choice(len(ones_idx), num_ones - target_ones, replace=False)
        h[ones_idx[remove]] = 0.0
    elif num_ones < target_ones:
        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
        h[zeros_idx[add]] = 1.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    
    for it in range(iterations):
        if remaining_time() < 3:
            break
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        
        i = rng.randint(len(ones_list))
        j = rng.randint(len(zeros_list))
        oi = ones_list[i]
        oz = zeros_list[j]
        
        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = compute_c5(h)
        
        T = t_start * (t_end / t_start) ** (float(it) / max(float(iterations), 1.0))
        
        if new_c5 < current_c5 or (T > 1e-15 and rng.rand() < np.exp(-(new_c5 - current_c5) / max(T, 1e-15))):
            current_c5 = new_c5
            ones_list[i] = oz
            zeros_list[j] = oi
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0
    
    return best_h, best_c5

        bounds = [(0.0, 1.0)] * block_size

            h_try = h_fixed.copy()
            h_try[block] = x_block
            # Quick feasibility: adjust sum
            needed = n/2.0 - np.sum(h_try)
            # Distribute needed to the block (approximately)
            x_block_adj = x_block + needed / block_size
            x_block_adj = np.clip(x_block_adj, 0, 1)
            h_try[block] = x_block_adj
            return smooth_max_obj(h_try)

    bounds = [(0.0, 1.0)] * n
    h = project_feasible(h0, n)
    try:
        res = minimize(topk_smooth_obj, h, jac=lambda x: topk_smooth_grad(x, 16),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-15,
                                'gtol': 1e-9, 'iprint': 0,
                                'maxls': 80, 'maxfun': 8000})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)


def binary_sa(h_init, n, iterations, rng, t_start=0.003, t_end=0.00002):
    h = np.where(h_init > 0.5, 1.0, 0.0)
    target_ones = n // 2
    num_ones = int(np.sum(h))
    if num_ones > target_ones:
        ones_idx = np.where(h == 1.0)[0]
        remove = rng.choice(len(ones_idx), num_ones - target_ones, replace=False)
        h[ones_idx[remove]] = 0.0
    elif num_ones < target_ones:
        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
        h[zeros_idx[add]] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5

    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])

    for it in range(iterations):
        if remaining_time() < 2:
            break
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break

        i = rng.randint(len(ones_list))
        j = rng.randint(len(zeros_list))
        oi = ones_list[i]
        oz = zeros_list[j]

        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = compute_c5(h)

        T = t_start * (t_end / t_start) ** (float(it) / max(float(iterations), 1.0))

        if new_c5 < current_c5 or (T > 1e-15 and rng.rand() < np.exp(-(new_c5 - current_c5) / max(T, 1e-15))):
            current_c5 = new_c5
            ones_list[i] = oz
            zeros_list[j] = oi
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0

    return best_h, best_c5


def structured_sa(h_init, n, iterations, rng):
    """SA with structured moves: single flip, pair swap, block flip"""
    h = np.where(h_init > 0.5, 1.0, 0.0)
    target_ones = n // 2
    num_ones = int(np.sum(h))
    if num_ones > target_ones:
        ones_idx = np.where(h == 1.0)[0]
        remove = rng.choice(len(ones_idx), num_ones - target_ones, replace=False)
        h[ones_idx[remove]] = 0.0
    elif num_ones < target_ones:
        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
        h[zeros_idx[add]] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5

    for it in range(iterations):
        if remaining_time() < 2:
            break

        move_type = rng.randint(3)
        accepted = False

        if move_type == 0:
            # Single swap (1 to 0, 0 to 1)
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                continue
            oi = ones[rng.randint(len(ones))]
            oz = zeros[rng.randint(len(zeros))]
            h[oi] = 0.0
            h[oz] = 1.0
            new_c5 = compute_c5(h)
            T = 0.002 * 0.98 ** (it / 50.0)
            if new_c5 < current_c5 or (T > 1e-12 and rng.rand() < np.exp(-(new_c5 - current_c5) / T)):
                current_c5 = new_c5
                accepted = True
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[oi] = 1.0
                h[oz] = 0.0

        elif move_type == 1:
            # Pair swap: swap two 1s with two 0s
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                continue
            o1, o2 = rng.choice(len(ones), 2, replace=False)
            z1, z2 = rng.choice(len(zeros), 2, replace=False)
            oi1, oi2 = ones[o1], ones[o2]
            oz1, oz2 = zeros[z1], zeros[z2]
            h[oi1] = 0.0
            h[oi2] = 0.0
            h[oz1] = 1.0
            h[oz2] = 1.0
            new_c5 = compute_c5(h)
            T = 0.002 * 0.98 ** (it / 50.0)
            if new_c5 < current_c5 or (T > 1e-12 and rng.rand() < np.exp(-(new_c5 - current_c5) / T)):
                current_c5 = new_c5
                accepted = True
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[oi1] = 1.0
                h[oi2] = 1.0
                h[oz1] = 0.0
                h[oz2] = 0.0

        else:
            # Block flip: flip a contiguous block
            block_len = rng.randint(2, 6)
            start = rng.randint(0, n - block_len)
            block = np.arange(start, start + block_len)
            ones_in = np.sum(h[block] == 1.0)
            zeros_in = np.sum(h[block] == 0.0)
            if ones_in == 0 or zeros_in == 0:
                continue
            # Only flip if it keeps total count balanced (or do swap)
            # Flip pattern within block
            saved = h[block].copy()
            h[block] = 1.0 - h[block]
            # Check constraint
            new_sum = np.sum(h)
            if abs(new_sum - n / 2.0) > 1e-10:
                h[block] = saved
                continue
            new_c5 = compute_c5(h)
            T = 0.001 * 0.97 ** (it / 50.0)
            if new_c5 < current_c5 or (T > 1e-12 and rng.rand() < np.exp(-(new_c5 - current_c5) / T)):
                current_c5 = new_c5
                accepted = True
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[block] = saved

    return best_h, best_c5


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h


def make_golden_ratio(n):
    h = np.zeros(n)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if abs(frac - 0.5) < 0.25 else 0.0
    return h


def make_van_der_corput(n):
    h = np.zeros(n)
    for i in range(n):
        val = i + 1
        rev = 0
        for _ in range(20):
            rev = (rev << 1) | (val & 1)
            val >>= 1
        h[i] = 1.0 if (rev % 2) == 0 else 0.0
    return h


def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h


def make_sine_mix(n, freqs, amps, phases, rng):
    t = np.arange(n) / float(n)
    h = np.zeros(n)
    for f, a, p in zip(freqs, amps, phases):
        h += a * np.sin(2.0 * np.pi * f * t + p)
    h = 0.5 + h
    return np.clip(h, 0.0, 1.0)


def run(seed=42, budget_s=1300, **kwargs):
    global _N, _BETA, _TGT, _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Initialize from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.7f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 160
        best_h = make_thue_morse(best_n)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)
        print("No parent. Thue-Morse: n={}, C5={:.7f}".format(best_n, best_c5))

    # Phase 1: Refine at original resolution
    print("\n=== Phase 1: Refine n={} ===".format(best_n))
    h_ref, c5_ref = l_bfgs_continuation(best_h, best_n, [5, 20, 80, 300])
    if c5_ref < best_c5:
        best_c5 = c5_ref
        best_h = h_ref
        print("  Refined: C5={:.7f}".format(best_c5))

    # Top-k polish
    if remaining_time() > 15:
        h_tp, c5_tp = topk_polish(best_h, best_n)
        if c5_tp < best_c5:
            best_c5 = c5_tp
            best_h = h_tp
            print("  TopK polished: C5={:.7f}".format(best_c5))

    # Phase 2: Multi-resolution refinement
    for n_new in [200, 240, 320]:
        if remaining_time() < 25:
            break
        print("\n=== Phase 2: n={} ===".format(n_new))

        h_interp = interp_to(best_h, best_n, n_new)
        h_interp = project_feasible(h_interp, n_new)

        starts = [h_interp]
        starts.append(project_feasible(make_thue_morse(n_new), n_new))
        starts.append(project_feasible(make_golden_ratio(n_new), n_new))
        starts.append(project_feasible(make_van_der_corput(n_new), n_new))
        starts.append(project_feasible(make_random_binary(n_new, rng), n_new))
        starts.append(project_feasible(make_random_binary(n_new, rng), n_new))

        for h0 in starts:
            if remaining_time() < 12:
                break
            c5_0 = compute_c5(h0)
            if c5_0 > best_c5 + 0.005:
                continue

            h_new, c5_new = l_bfgs_continuation(h0, n_new, [5, 20, 80, 300])
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_new
                print("  Improved n={}: C5={:.7f}".format(n_new, best_c5))

            # Top-k polish the result
            if remaining_time() > 10 and c5_new < best_c5 + 0.002:
                h_top, c5_top = topk_polish(h_new, n_new)
                if c5_top < best_c5:
                    best_c5 = c5_top
                    best_h = h_top.copy()
                    best_n = n_new
                    print("  TopK n={}: C5={:.7f}".format(n_new, best_c5))

    # Phase 3: Binary structured SA at best resolution
    if remaining_time() > 40:
        n = best_n
        print("\n=== Phase 3: Structured SA (n={}) ===".format(n))

        h_bin_init = np.where(best_h > 0.5, 1.0, 0.0)
        h_bin_init = project_feasible(h_bin_init, n)

        sa_starts = [h_bin_init.copy()]
        for _ in range(4):
            sa_starts.append(project_feasible(make_random_binary(n, rng), n))
        sa_starts.append(project_feasible(make_thue_morse(n), n))
        sa_starts.append(project_feasible(make_golden_ratio(n), n))

        for sa_h0 in sa_starts:
            if remaining_time() < 12:
                break
            h_sa, c5_sa = structured_sa(sa_h0, n, 6000, rng)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n
                print("  SA: C5={:.7f}".format(best_c5))

            # Polish binary result with continuous optimization
            if remaining_time() > 10 and c5_sa < best_c5 + 0.003:
                h_pol, c5_pol = l_bfgs_continuation(h_sa, n, [20, 80, 300, 800])
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()
                    best_n = n
                    print("  Polished: C5={:.7f}".format(best_c5))

    # Phase 4: Interpolate to even higher resolution and refine
    if remaining_time() > 40:
        n_high = best_n * 2 if best_n <= 240 else best_n + 80
        if n_high > 480:
            n_high = 480
        print("\n=== Phase 4: High-res n={} ===".format(n_high))

        h_interp = interp_to(best_h, best_n, n_high)
        h_interp = project_feasible(h_interp, n_high)
        c5_interp = compute_c5(h_interp)
        print("  Interpolated: C5={:.7f}".format(c5_interp))

        h_hi, c5_hi = l_bfgs_continuation(h_interp, n_high, [5, 20, 80, 300, 800])
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_high
            print("  High-res: C5={:.7f}".format(best_c5))

        if remaining_time() > 15:
            h_tp2, c5_tp2 = topk_polish(best_h, best_n)
            if c5_tp2 < best_c5:
                best_c5 = c5_tp2
                best_h = h_tp2.copy()
                print("  TopK high-res: C5={:.7f}".format(best_c5))

    # Phase 5: Final aggressive polish
    if remaining_time() > 10:
        n = best_n
        print("\n=== Phase 5: Final polish (n={}) ===".format(n))
        _N = n
        _TGT = n / 2.0
        bounds = [(0.0, 1.0)] * n

        for beta_val in [500, 1000, 2000, 4000]:
            if remaining_time() < 5:
                break
            _BETA = beta_val
            try:
                res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 400, 'ftol': 1e-16,
                                        'gtol': 1e-11, 'iprint': 0,
                                        'maxls': 150, 'maxfun': 20000})
                h_f = project_feasible(res.x, n)
                c5_f = compute_c5(h_f)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_f.copy()
                    print("  beta={}: C5={:.7f}".format(beta_val, best_c5))
            except Exception:
                pass

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, best_c5))
    return (best_h, best_c5, best_n)
