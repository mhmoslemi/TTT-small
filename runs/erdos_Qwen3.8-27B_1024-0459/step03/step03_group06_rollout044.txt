                break
    return h


def interp_h(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h


def make_golden(n):
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return h


def make_vdc(n):
    h = np.zeros(n)
    for i in range(n):
        val = i + 1
        rev = 0
        for _ in range(20):
            rev = (rev << 1) | (val & 1)
            val >>= 1
        h[i] = 1.0 if (rev & 1) == 0 else 0.0
    return h


def make_random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def topk_func(x, n, top_k, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    k = min(top_k, len(corr))
    top_idx = np.argpartition(corr, -k)[-k:]
    tv = corr[top_idx]
    m = np.max(tv)
    sm = (m + np.log(np.sum(np.exp(beta * (tv - m)))) / beta) * dx
    pen = 200.0 * (np.sum(x) - n / 2.0) ** 2
    return sm + pen


def topk_gradient(x, n, top_k, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    k = min(top_k, len(corr))
    top_idx = np.argpartition(corr, -k)[-k:]
    tv = corr[top_idx]
    m = np.max(tv)
    exps = np.exp(beta * (tv - m))
    sw = np.sum(exps)
    weights = exps / sw
    grad = np.zeros(n)
    for jj in range(k):
        wj = weights[jj]
        if wj < 1e-14:
            continue
        j = top_idx[jj]
        s = j - (n - 1)
        if s >= 0:
            grad[s:] += wj * (1.0 - x[:n - s])
            grad[:n - s] += wj * (-x[s:])
        else:
            sp = -s
            grad[:n - sp] += wj * (1.0 - x[sp:])
            grad[sp:] += wj * (-x[:n - sp])
    grad *= dx
    grad += 2.0 * 200.0 * (np.sum(x) - n / 2.0)
    return grad


def coord_descent(h, n, rng, max_sweeps=15):
    h = np.array(h, dtype=float).copy()
    c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 8:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if remaining_time() < 4:
                break
            old = h[i]
            c5_old = c5
            best_v = old
            best_c5 = c5_old
            candidates = [0.0, 1.0, max(0.0, old - 0.12), min(1.0, old + 0.12)]
            if 0.2 < old < 0.8:
                candidates.append(0.5)
            for nv in candidates:
                if abs(nv - old) < 1e-9:
                    continue
                h[i] = nv
                h = project_feasible(h, n)
                c5_test = compute_c5(h)
                if c5_test < best_c5 - 1e-12:
                    best_c5 = c5_test
                    best_v = nv
            if best_v != old:
                h = project_feasible(h, n)
                c5 = best_c5
                improved = True
            else:
                h[i] = old
                h = project_feasible(h, n)
        if not improved:
            break
    return h, c5


def binary_local_search(h, n, rng, max_iters=3000):
    h = np.where(h > 0.5, 1.0, 0.0)
    num_ones = int(np.sum(h))
    target_ones = n // 2
    if num_ones > target_ones:
        ones_idx = np.where(h == 1.0)[0]
        remove = rng.choice(len(ones_idx), num_ones - target_ones, replace=False)
        h[ones_idx[remove]] = 0.0
    elif num_ones < target_ones:
        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(len(zeros_idx), target_ones - num_ones, replace=False)
        h[zeros_idx[add]] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    t_start, t_end = 0.003, 0.00003
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        if not ones_list or not zeros_list:
            break
        i = rng.randint(len(ones_list))
        j = rng.randint(len(zeros_list))
        oi, oz = ones_list[i], zeros_list[j]
        h[oi] = 0.0
        h[oz] = 1.0
        new_c5 = compute_c5(h)
        T = t_start * (t_end / t_start) ** (it / max(max_iters, 1))
        if new_c5 < cur_c5 or (T > 1e-15 and rng.rand() < np.exp(-(new_c5 - cur_c5) / T)):
            cur_c5 = new_c5
            ones_list[i] = oz
            zeros_list[j] = oi
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S, _RNG
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    _RNG = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.7f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 320
        best_h = project_feasible(make_thue_morse(best_n), best_n)
        best_c5 = compute_c5(best_h)
        print("No parent. Thue-Morse: n={}, C5={:.7f}".format(best_n, best_c5))

    resolutions = [320, 256, 400, 384]
    resolutions = sorted(set(resolutions + [best_n]), key=lambda x: -abs(x - 320))
    resolutions = [nn for nn in resolutions if 128 <= nn <= 512 and nn % 2 == 0]

    for n in resolutions[:4]:
        if remaining_time() < 60:
            break
        print("\n=== Optimization at n={} ===".format(n))

        inits = []
        h_interp = project_feasible(interp_h(best_h, best_n, n), n)
        inits.append((h_interp, "interp"))
        inits.append((project_feasible(make_thue_morse(n), n), "thue"))
        inits.append((project_feasible(make_golden(n), n), "golden"))
        inits.append((project_feasible(make_vdc(n), n), "vdc"))
        inits.append((project_feasible(make_random_binary(n, _RNG), n), "rand1"))
        inits.append((project_feasible(make_random_binary(n, _RNG), n), "rand2"))

        # Perturbed parent
        h_pert = h_interp + _RNG.randn(n) * 0.1
        inits.append((project_feasible(h_pert, n), "perturbed"))

        bounds = [(0.0, 1.0)] * n

        for h0, label in inits:
            if remaining_time() < 20:
                break
            c5_cur = compute_c5(h0)
            if c5_cur > best_c5 * 1.15 and label not in ("interp", "thue"):
                continue
            print("  Init [{}] n={}, C5={:.7f}".format(label, n, c5_cur))

            for top_k, beta in [(5, 100), (8, 80), (10, 60), (8, 200), (5, 300)]:
                if remaining_time() < 10:
                    break

                h_work = project_feasible(h0, n)

                def _obj(xx):
                    return topk_func(xx, n, top_k, beta)

                def _grad(xx):
                    return topk_gradient(xx, n, top_k, beta)

                try:
                    res = minimize(_obj, h_work, jac=_grad, method='L-BFGS-B',
                                   bounds=bounds,
                                   options={'maxiter': 400, 'ftol': 1e-15,
                                            'gtol': 1e-10, 'iprint': 0,
                                            'maxls': 30, 'maxfun': 3000})
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur - 1e-10:
                        h_work = h_new
                        c5_cur = c5_new
                except Exception:
                    pass

            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_work.copy()
                best_n = n
                print("  ** Improved: n={}, C5={:.7f}".format(n, c5_cur))

        # Coordinate descent polish on best at this n
        if n == best_n and remaining_time() > 30:
            h_cd, c5_cd = coord_descent(best_h, n, _RNG, max_sweeps=8)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                print("  CD: C5={:.7f}".format(best_c5))

    # Phase 2: Binary search at best resolution
    if remaining_time() > 50:
        n = best_n
        print("\n=== Binary SA at n={} ===".format(n))
        h_bin_init = project_feasible(best_h, n)
        bh, bc = binary_local_search(h_bin_init, n, _RNG, max_iters=3000)
        if bc < best_c5:
            best_c5 = bc
            best_h = bh.copy()
            print("  Binary SA: C5={:.7f}".format(best_c5))

        # Polish binary result with continuous
        if remaining_time() > 30:
            bounds = [(0.0, 1.0)] * n
            for top_k, beta in [(5, 150), (8, 100)]:
                if remaining_time() < 10:
                    break
                h_work = np.array(best_h, dtype=float)

                def _obj2(xx):
                    return topk_func(xx, n, top_k, beta)

                def _grad2(xx):
                    return topk_gradient(xx, n, top_k, beta)

                try:
                    res = minimize(_obj2, h_work, jac=_grad2, method='L-BFGS-B',
                                   bounds=bounds,
                                   options={'maxiter': 300, 'ftol': 1e-15,
                                            'gtol': 1e-11, 'iprint': 0})
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        print("  Polished: C5={:.7f}".format(best_c5))
                except Exception:
                    pass

    # Phase 3: Multi-restart perturbation at best resolution
    if remaining_time() > 60:
        n = best_n
        bounds = [(0.0, 1.0)] * n
        print("\n=== Perturbation restarts at n={} ===".format(n))

        for restart in range(6):
            if remaining_time() < 15:
                break
            h_pert = best_h + _RNG.randn(n) * 0.05 * (1.0 + 0.3 * restart)
            h_pert = project_feasible(h_pert, n)
            c5_pert = compute_c5(h_pert)

            for top_k, beta in [(5, 120), (8, 80)]:
                if remaining_time() < 10:
                    break

                def _obj3(xx):
                    return topk_func(xx, n, top_k, beta)

                def _grad3(xx):
                    return topk_gradient(xx, n, top_k, beta)

                try:
                    res = minimize(_obj3, h_pert, jac=_grad3, method='L-BFGS-B',
                                   bounds=bounds,
                                   options={'maxiter': 250, 'ftol': 1e-15,
                                            'gtol': 1e-10, 'iprint': 0})
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        print("  Restart {}: C5={:.7f}".format(restart, best_c5))
                except Exception:
                    pass

    # Phase 4: Final coordinate descent + intensive L-BFGS-B
    if remaining_time() > 20:
        n = best_n
        h_cd, c5_cd = coord_descent(best_h, n, _RNG, max_sweeps=10)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            print("  Final CD: C5={:.7f}".format(best_c5))

    if remaining_time() > 15:
        n = best_n
        bounds = [(0.0, 1.0)] * n
        for top_k, beta in [(5, 500), (3, 800), (5, 300)]:
            if remaining_time() < 8:
                break

            def _obj4(xx):
                return topk_func(xx, n, top_k, beta)

            def _grad4(xx):
                return topk_gradient(xx, n, top_k, beta)

            try:
                res = minimize(_obj4, best_h, jac=_grad4, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 500, 'ftol': 1e-16,
                                        'gtol': 1e-12, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  Final polish: C5={:.7f}".format(best_c5))
            except Exception:
                pass

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
