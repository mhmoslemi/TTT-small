def _active_weights(c,K,beta):
    m=c.size
    if K <=0: return np.arange(0), np.array([])
    if K >= m:
       active=np.arange(m)
    else:
       active=np.argsort(c)[-K:]
    if len(active)==0: return active,np.array([])
    if beta <= 0:
       w=np.full(len(active), 1.0/len(active), dtype=float)
    else:
       vals = float(beta)*(c[active] - float(np.max(c[active])))
       # stable; vals <=0
       ev = np.exp(vals)
       s = float(np.sum(ev))
       if s <= 0 or not math.isfinite(s):
          w=np.full(len(active),1.0/len(active))
       else:
          w=ev/s
    return active,w

def _add_shift_grad(h, grad, idx, n, w):
    s = idx - (n - 1)
    if s == 0:
        grad += float(w) * (1.0 - 2.0 * h)
    elif s > 0:
        grad[s:] += float(w) * (1.0 - h[s:])
        grad[:n-s] -= float(w) * h[:n-s]
    else:
        sp = -s
        grad[:n-sp] += float(w) * (1.0 - h[:n-sp])
        grad[sp:] -= float(w) * h[sp:]

def _make_sine_mix(n, rng):
    t = np.arange(n, dtype=float) / float(n)
    h = np.full(n, 0.5, dtype=float)
    for f in (2, 5, 9, 13):

def _active_weights(c, K, beta):
    m = c.size
    if K <= 0:
        return np.arange(0, dtype=int), np.array([], dtype=float)
    if K >= m:
        active = np.arange(m, dtype=int)
    else:

    if len(active) == 0:
        return active, np.array([], dtype=float)

    if beta <= 0:
        w = np.full(len(active), 1.0 / float(len(active)), dtype=float)
    else:
        vals = float(beta) * (c[active] - float(np.max(c[active])))
        ev = np.exp(vals)

        if s <= 0.0 or not math.isfinite(s):

def _add_shift_grad(h, grad, idx, n, w):
    s = idx - (n - 1)
    w = float(w)
    if s == 0:
        grad += w * (1.0 - 2.0 * h)
    elif s > 0:
        grad[s:] += w * (1.0 - h[s:])
        grad[:n - s] -= w * h[:n - s]
    else:
        sp = -s
        grad[:n - sp] += w * (1.0 - h[:n - sp])
        grad[sp:] -= w * h[sp:]


def _pair_diff(h, i, j):

    n = len(h)

    d[:] = 0.0

    d[n - 1 - i:2 * n - 1 - i] -= h
    d[j:j + n] -= b[::-1]
    d[n - 1 - j:2 * n - 1 - j] += h
    return d


def _pair_eval(dval, c, diff, idx1, idx2):

    if dval != 0.0:

        _WORK += _WORK2
    dd = dval * dval
    if dd != 0.0:
        _WORK[idx1] += dd
        _WORK[idx2] += dd
    return float(np.max(_WORK))

                        if hi - lo <= 1e-12:
                            continue
                        dval = rng.uniform(-1.0, 1.0) * min(0.15, 0.35 * (hi - lo))
                        dval = max(lo, min(hi, dval))
                        if abs(dval) < 1e-12:
                            continue
                        _pair_diff(h, ii, jj)
                        idx1 = n - 1 + (jj - ii)
                        idx2 = n - 1 + (ii - jj)
                        val = _pair_eval(dval, c, _DIFF, idx1, idx2)
                        if val < cmax + T:
                            new_c = c + dval * _DIFF
                            dd = dval * dval
                            new_c[idx1] += dd
                            new_c[idx2] += dd
                            h[ii] += dval
                            h[jj] -= dval
                            if h[ii] < 0.0:
                                h[ii] = 0.0
                            elif h[ii] > 1.0:
                                h[ii] = 1.0
                            if h[jj] < 0.0:
                                h[jj] = 0.0
                            elif h[jj] > 1.0:
                                h[jj] = 1.0
                            c = new_c
                            cmax = val
                            current = cmax * dx
                            stagnant = 1
                            perturbs += 1
                            if current < best_c5:
                                best_c5 = current
                                best_h = h.copy()
                            break

                if stagnant in (5, 15, 30, 50, 70) and perturbs < 80:
                    scale = 0.05 if stagnant <= 15 else 0.025
                    for _ in range(3):
                        h2 = h.copy()
                        kcount = min(10, max(2, n // 20))
                        if n > 1:
                            idxs = rng.choice(n, kcount, replace=False)
                            h2[idxs] += scale * rng.randn(kcount)
                            h2 = _project_feasible(h2)
                            c5t = _compute_c5(h2)
                            tol = 0.004 if c5t < best_c5 else 0.0012
                            if c5t < current + tol:
                                h = h2
                                c = _compute_corr(h)
                                cmax = float(np.max(c))
                                current = c5t
                                perturbs += 1
                                stagnant = 1
                                if current < best_c5:
                                    best_c5 = current
                                    best_h = h.copy()
                                break

            if stagnant > 70:
                break

    h = _project_feasible(h)
    c5f = _compute_c5(h)
    if c5f < best_c5:
        best_h = h.copy()
        best_c5 = c5f

    bh = _project_feasible(best_h)
    cb = _compute_c5(bh)
    if cb <= best_c5 + 1e-8:
        best_h = bh
        best_c5 = cb
    else:
        ch = _project_feasible(h)
        cv = _compute_c5(ch)
        if cv <= best_c5 + 1e-8:
            best_h = ch
            best_c5 = cv
        else:
            best_h = bh
            best_c5 = cb

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = float(budget_s)
    rng = np.random.RandomState(seed)

    try:
        parent_arr = np.asarray(initial_h_values, dtype=float)
    except Exception:
        parent_arr = None

    if parent_arr is not None and parent_arr.size > 0:
        parent_arr = _project_feasible(parent_arr)
        best_h = parent_arr.copy()
        best_n = int(parent_arr.size)
        best_c5 = _compute_c5(best_h)
    else:
        best_n = 160
        best_h = _project_feasible(_make_uniform(best_n))
        best_c5 = _compute_c5(best_h)

    if _remaining() > 200:
        lim = min(300.0, max(30.0, _remaining() - 40.0))
        h, c5v = _local_pair_search(best_h, best_n, lim, rng, True)
        if c5v < best_c5:
            best_h = h.copy()
            best_n = int(len(h))
            best_c5 = c5v

    if _remaining() > 140:
        lim = min(140.0, max(30.0, _remaining() - 40.0))
        h2 = best_h + 0.08 * rng.randn(best_n)
        h, c5v = _local_pair_search(h2, best_n, lim, rng, True)
        if c5v < best_c5:
            best_h = h.copy()
            best_n = int(len(h))
            best_c5 = c5v

    if _remaining() > 110:
        lim = min(90.0, max(25.0, _remaining() - 30.0))
        hb = np.where(best_h > 0.5, 1.0, 0.0)
        h, c5v = _local_pair_search(hb, best_n, lim, rng, True)
        if c5v < best_c5:
            best_h = h.copy()
            best_n = int(len(h))
            best_c5 = c5v

    target_ns = []
    for nn in (240, 320, 256, 480, best_n * 2):
        if nn > best_n and nn not in target_ns and nn <= 512:
            target_ns.append(nn)
    target_ns = target_ns[:3]

    for ti, target_n in enumerate(target_ns):
        if _remaining() < 120:
            break
        if target_n < best_n and best_n > 200:
            continue

        init1 = _interp_to(best_h, target_n)
        if _remaining() > 100:
            chunk = 300.0 if ti == 0 else 180.0
            chunk = min(chunk, max(40.0, _remaining() - 50.0))
            h, c5v = _local_pair_search(init1, target_n, chunk, rng, True)
            if c5v < best_c5:
                best_h = h.copy()
                best_n = int(len(h))
                best_c5 = c5v

        if _remaining() > 110:
            init2 = _make_sine_mix(target_n, rng) if ti == 0 else _make_binary_random(target_n, rng)
            chunk = min(120.0, max(30.0, _remaining() - 40.0))
            h, c5v = _local_pair_search(init2, target_n, chunk, rng, True)
            if c5v < best_c5:
                best_h = h.copy()
                best_n = int(len(h))
                best_c5 = c5v

        if best_n == target_n and _remaining() > 120:
            chunk = min(150.0, max(30.0, _remaining() - 40.0))
            h, c5v = _local_pair_search(best_h, target_n, chunk, rng, True)
            if c5v < best_c5:
                best_h = h.copy()
                best_n = int(len(h))
                best_c5 = c5v

    if _remaining() > 90:
        chunk = min(180.0, max(30.0, _remaining() - 20.0))
        h, c5v = _local_pair_search(best_h, best_n, chunk, rng, True)
        if c5v < best_c5:
            best_h = h.copy()
            best_n = int(len(h))
            best_c5 = c5v

    best_h = _project_feasible(best_h)
    final_c5 = _compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if math.isfinite(eval_c5) and eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, int(len(best_h)))
