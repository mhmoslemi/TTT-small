        tl4 = min(120.0, time_left() * 0.25)
        h4, c5_4 = lbfgs_optimize(best_h, best_n, time_limit=tl4, t0=t0)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    
    # Phase 5: More coordinate descent rounds with perturbation
    for rnd in range(5):
        if time_left() < 20:
            break
        if rnd == 0:
            h_start = best_h.copy()
        else:
            scale = 0.02 * (1.0 + 0.5 * rnd)
            h_start = project_feasible(best_h + scale * rng.randn(best_n), best_n)
        
        tl5 = min(40.0, time_left() * 0.2)
        h5, c5_5 = fast_cd(h_start, best_n, n_sweeps=300, time_limit=tl5, t0=t0)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
    
    # Phase 6: Final heavy polish
    if time_left() > 20:
        tl6 = time_left()
        h6, c5_6 = fast_cd(best_h, best_n, n_sweeps=500, time_limit=tl6, t0=t0)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    # Validate with evaluator
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 is not None and ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
