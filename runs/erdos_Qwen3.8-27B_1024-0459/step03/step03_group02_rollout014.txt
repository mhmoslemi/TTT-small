def project_feasible(h, n):

    target = n / 2.0

                if remaining_time() < 8:
                    break
                h_new = lbgfB_refine(h_interp, trial_n, beta, maxiter=250)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = trial_n
                    h_interp = h_new.copy()

    # Phase 5: Perturbation + re-optimize at best resolution
    n = best_n
    for restart in range(6):
        if remaining_time() < 15:
            break
        sigma = 0.015 * (1.0 + restart * 0.3)
        h_pert = best_h + rng.randn(n) * sigma
        h_pert = project_feasible(h_pert, n)
        for beta in [600, 2000]:
            if remaining_time() < 7:
                break
            h_new = lbgfB_refine(h_pert, n, beta, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_pert = h_new.copy()

    # Phase 6: Coordinate descent
    if remaining_time() > 12:
        h_cd = best_h.copy()
        n = best_n
        h_cd = coordinate_descent(h_cd, n, max_passes=8, n_grid=9)
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 7: Final high-beta L-BFGS polish
    for beta in [3000, 8000, 15000]:
        if remaining_time() < 5:
            break
        h_new = lbgfB_refine(best_h, best_n, beta, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 8: Binary/ternary local search
    if remaining_time() > 8:
        h = best_h.copy()
        n = best_n
        for sweep in range(4):
            if remaining_time() < 4:
                break
            order = rng.permutation(n)
            any_imp = False
            for i in order:
                if remaining_time() < 2:
                    break
                old_val = h[i]
                for new_val in [0.0, 1.0, 0.5, 0.25, 0.75]:
                    if abs(new_val - old_val) < 1e-8:
                        continue
                    h[i] = new_val
                    h_proj = project_feasible(h, n)
                    c5_t = compute_c5(h_proj)
                    if c5_t < best_c5 - 1e-14:
                        best_c5 = c5_t
                        best_h = h_proj.copy()
                        h = h_proj.copy()
                        any_imp = True
                    else:
                        h = best_h.copy()
            if not any_imp:
                break

    # Phase 9: Final micro L-BFGS + SLSQP
    if remaining_time() > 6:
        for beta in [5000, 12000]:
            if remaining_time() < 4:
                break
            h_new = lbgfB_refine(best_h, best_n, beta, maxiter=100)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        if remaining_time() > 5:
            h_new = slsqp_eq(best_h, best_n, 3000, maxiter=150)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 10: Final active-set squeeze
    if remaining_time() > 4:
        h_a = best_h.copy()
        h_a = active_lag_optimize(h_a, best_n, beta=2000, max_iter=100, n_top=3)
        c5_a = compute_c5(h_a)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
