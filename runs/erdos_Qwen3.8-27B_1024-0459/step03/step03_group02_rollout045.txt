        else:
            h[:] += diff / n
            h = np.clip(h, 0.0, 1.0)
    # Exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-16:
        interior = np.where((h > 0.001) & (h < 0.999))[0]
        if len(interior) > 1:
            h[interior] += diff / len(interior)
            h = np.clip(h, 0.0, 1.0)
            diff = target - np.sum(h)
        if abs(diff) > 1e-16:
            for j in range(n):
                if 0 < h[j] < 1:
                    h[j] += diff
                    h[j] = np.clip(h[j], 0.0, 1.0)
                    break
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def smooth_max(x, beta):
    """Smoothed max of correlations with given sharpness."""
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_obj(x):
    n = _N_GLOBAL
    dx = 2.0 / n
    return smooth_max(x, 2000.0) * dx

def sum_constraint(x):
    return np.sum(x) - _N_GLOBAL / 2.0

def slsqp(h, n, maxiter=500, ftol=1e-18):
    global _N_GLOBAL
    _N_GLOBAL = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    try:
        res = minimize(smooth_obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def binary_sa(h_init, n, num_iters=200000, T_start=0.8, T_end=0.05, seed=42):
    """Simulated annealing on binary solutions."""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    n_ones = int(round(np.sum(h_init)))
    n_ones = min(n_ones, n // 2)
    idx = rng.choice(n, n_ones, replace=False)
    h[idx] = 1.0
    
    corr = compute_corr(h)
    best_corr = np.max(corr)
    best_h = h.copy()
    
    # Precompute which lags matter (skip lag 0 which is always 0 for binary)
    # Focus on lags 1 to n-1
    
    current_max = best_corr
    no_improve = 0
    
    for it in range(num_iters):
        if remaining_time() < 5:
            break
        T = T_start * (T_end / T_start) ** (it / num_iters)
        
        # Choose move
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Try the flip
        h[i] = 0
        h[j] = 1
        new_corr = compute_corr(h)
        new_max = np.max(new_corr)
        delta = new_max - current_max
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current_max = new_max
            no_improve = 0
            if new_max < best_corr:
                best_corr = new_max
                best_h = h.copy()
                no_improve = 0
        else:
            h[i] = 1
            h[j] = 0
        
        no_improve += 1
        if no_improve > 50000:
            # Reset temperature
            T_start = max(T * 1.5, 0.1)
            no_improve = 0
    
    return best_h, best_corr * 2.0 / n

def pattern_init(n, pattern_type='balanced', seed=42):
    """Create various initial patterns."""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    n_ones = n // 2
    
    if pattern_type == 'block':
        h[:n_ones] = 1.0
    elif pattern_type == 'alternating':
        h[::2] = 1.0
    elif pattern_type == 'random':
        idx = rng.choice(n, n_ones, replace=False)
        h[idx] = 1.0
    elif pattern_type == 'block2':
        # Two blocks
        block_size = n_ones // 2
        h[:block_size] = 1.0
        h[n//2:n//2+block_size] = 1.0
    elif pattern_type == 'cosine':
        # Weighted by cosine
        positions = np.linspace(0, 2*np.pi, n, endpoint=False)
        weights = 1.0 + 0.5 * np.cos(positions)
        weights /= weights.sum()
        probs = weights / weights.sum() * n
        # Use Gumbel trick
        gumbel = -np.log(-np.log(rng.uniform(1-1e-10, 1, n)))
        scores = np.log(probs + 1e-10) + gumbel
        h[np.argsort(scores)[::-1][:n_ones]] = 1.0
    elif pattern_type == 'sinusoidal':
        for i in range(n):
            h[i] = 1.0 if np.sin(2*np.pi*i/n*3) > 0 else 0.0
        # Adjust to have exactly n/2 ones
        # ... just use random for now
        idx = rng.choice(n, n_ones, replace=False)
        h[:] = 0
        h[idx] = 1.0
    elif pattern_type == 'parent_round':
        return None  # Signal to use parent
    
    # Adjust to have exactly n/2 ones
    current_ones = int(np.sum(h))
    if current_ones > n_ones:
        # Remove excess
        excess = current_ones - n_ones
        ones_idx = np.where(h == 1)[0]
        remove = rng.choice(ones_idx, excess, replace=False)
        h[remove] = 0
    elif current_ones < n_ones:
        need = n_ones - current_ones
        zeros_idx = np.where(h == 0)[0]
        add = rng.choice(zeros_idx, need, replace=False)
        h[add] = 1
    
    return h

def targeted_cont_opt(h, n, max_iters=5000):
    """Targeted continuous optimization focusing on top lags."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        
        corr = compute_corr(h)
        max_corr = np.max(corr)
        dx = 2.0 / n
        
        # Get top 5 lags
        top_idx = np.argsort(corr)[::-1][:5]
        
        grad = np.zeros(n)
        for k_idx in top_idx:
            offset = k_idx - (n - 1)
            if offset >= 0:
                n_pairs = n - offset
                if n_pairs > 0:
                    grad[offset:offset + n_pairs] += (1.0 - h[:n_pairs])
                    grad[:n_pairs] -= h[offset:offset + n_pairs]
            else:
                abs_off = -offset
                n_pairs = n - abs_off
                if n_pairs > 0:
                    grad[:n_pairs] += (1.0 - h[abs_off:abs_off + n_pairs])
                    grad[abs_off:abs_off + n_pairs] -= h[:n_pairs]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        
        # Line search
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.1]:
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-15:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
        
        if not improved:
            # Try SLSQP occasionally
            if it % 200 == 100 and remaining_time() > 15:
                h_new = slsqp(h, n, maxiter=200)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
            elif it % 100 == 50:
                break
    
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
