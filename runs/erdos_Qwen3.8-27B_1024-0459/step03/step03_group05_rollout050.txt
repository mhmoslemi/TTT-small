def delta_corr(h, p, dp, q, dq):
    n = len(h)
    # k ranges from -(n-1) to n-1
    k = np.arange(-(n-1), n)  # length 2n-1
    
    # For position p with change dp:
    # term1: dp * (1-h[p-k]) where 0 <= p-k < n
    # term2: -dp * h[p+k] where 0 <= p+k < n
    
    # Vectorized:
    pk = p - k  # positions for (1-h) term
    pl = p + k  # positions for h term
    
    mask1 = (pk >= 0) & (pk < n)
    mask2 = (pl >= 0) & (pl < n)
    
    dc = np.zeros(2*n-1)
    dc[mask1] += dp * (1.0 - h[pk[mask1]])
    dc[mask2] -= dp * h[pl[mask2]]
    
    # Similarly for q
    qk = q - k
    ql = q + k
    mask3 = (qk >= 0) & (qk < n)
    mask4 = (ql >= 0) & (ql < n)
    
    dc[mask3] += dq * (1.0 - h[qk[mask3]])
    dc[mask4] -= dq * h[ql[mask4]]
    
    return dc

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def corr_values(h):
    return np.correlate(h, 1.0 - h, mode='full')


def delta_corr(h, p, dp, q, dq):
    n = len(h)
    k = np.arange(-(n - 1), n)
    dc = np.zeros(2 * n - 1)

    pk = p - k
    pl = p + k
    m1 = (pk >= 0) & (pk < n)
    m2 = (pl >= 0) & (pl < n)
    dc[m1] += dp * (1.0 - h[pk[m1]])
    dc[m2] -= dp * h[pl[m2]]

    qk = q - k
    ql = q + k
    m3 = (qk >= 0) & (qk < n)
    m4 = (ql >= 0) & (ql < n)
    dc[m3] += dq * (1.0 - h[qk[m3]])
    dc[m4] -= dq * h[ql[m4]]

    return dc


def threshold_to_01(h, rng):
    n = len(h)
    h01 = (h >= 0.5).astype(float)
    target = n / 2.0
    cnt = int(np.sum(h01))
    if cnt > target:
        idx = np.where(h01 == 1.0)[0]
        perm = rng.permutation(len(idx))
        h01[idx[perm[:cnt - int(target)]]] = 0.0
    elif cnt < target:
        idx = np.where(h01 == 0.0)[0]
        perm = rng.permutation(len(idx))
        h01[idx[perm[:int(target) - cnt]]] = 1.0
    return h01
