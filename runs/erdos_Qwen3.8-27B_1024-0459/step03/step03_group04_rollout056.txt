def solve_lp(h0, active_shifts, n):
    """Solve the linearized problem.
    f_k(h) ≈ f_k(h0) + g_k · (h - h0) ≤ t
    minimize t
    s.t. 0 ≤ h ≤ 1, sum h = n/2
    """
    dx = 2.0 / n
    c5_0, corr_0 = compute_c5(h0)
    
    # Variables: [h_0, ..., h_{n-1}, t]
    # Minimize: t (last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: g_k · h ≤ t - f_k(h0) + g_k · h0 for each active k
    # i.e., g_k · h - t ≤ -f_k(h0) + g_k · h0
    
    A_ub = []
    b_ub = []
    
    for k_idx in active_shifts:
        g_k = grad_at_shift(h0, k_idx, n) * dx
        f_k_0 = corr_0[k_idx] * dx
        # g_k · h - t ≤ -f_k_0 + g_k · h0
        row = np.zeros(n + 1)
        row[:n] = g_k
        row[-1] = -1.0
        A_ub.append(row)
        b_ub.append(-f_k_0 + g_k @ h0)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Equality: sum h = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [n / 2.0]
    
    # Bounds: 0 ≤ h ≤ 1, t ≥ 0
    bounds_lp = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    if res.success:
        h_new = res.x[:n]
        t_new = res.x[-1]

    return None, None

import numpy as np

def _grad_at_shift(h, k_idx, n):
    """Gradient of corr[k_idx] w.r.t. h for the full correlation."""
    shift = k_idx - (n - 1)
    grad = np.zeros(n)
    if shift == 0:
        grad[:] = 1.0 - 2.0 * h
    elif shift > 0:
        np_ = n - shift
        grad[:np_] += (1.0 - h[shift:shift + np_])
        grad[shift:] -= h[:np_]
    else:
        s = -shift
        np_ = n - s
        grad[s:s + np_] += (1.0 - h[:np_])
        grad[:np_] -= h[s:s + np_]
    return grad

def _compute_c5(h):
