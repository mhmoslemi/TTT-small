    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 5.0)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c == 1 else 0.0
    # Fix to have exactly n/2 ones
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        idx = np.where(h == 1)[0]
        # Remove ones from end to preserve structure
        h[idx[-(n_ones - n//2):]] = 0
    elif n_ones < n // 2:
        idx = np.where(h == 0)[0]
        h[idx[-(n//2 - n_ones):]] = 1
    return h

def m_sequence(n, seed=1):
    """Generate a pseudo-random binary sequence using LFSR-like approach."""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    # Use a structured approach: golden ratio partition
    phi = (1 + np.sqrt(5)) / 2
    for i in range(n):
        h[i] = 1.0 if (i * phi) % 1.0 < 0.5 else 0.0
    # Fix count
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        idx = np.where(h == 1)[0]
        order = rng.permutation(len(idx))
        h[idx[order[:n_ones - n//2]]] = 0
    elif n_ones < n // 2:
        idx = np.where(h == 0)[0]
        order = rng.permutation(len(idx))
        h[idx[order[:n//2 - n_ones]]] = 1
    return h

def random_binary(n, rng):
    idx = rng.choice(n, n//2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def block_structured(n, rng, block_sizes=None):
    """Create a block-structured binary sequence."""
    if block_sizes is None:
        block_sizes = [4, 6, 8, 12, 16]
    h = np.zeros(n)
    pos = 0
    while pos < n:
        bs = block_sizes[rng.randint(len(block_sizes))]
        if pos + bs > n:
            bs = n - pos
        # Within block, random 50% density
        n_in_block = bs // 2
        ones_in_block = rng.choice(bs, n_in_block, replace=False)
        h[pos:pos+bs][ones_in_block] = 1.0
        pos += bs
    # Fix count
    n_ones = int(np.sum(h))
    if n_ones != n // 2:
        diff = n//2 - n_ones
        if diff > 0:
            idx = np.where(h == 0)[0]
            h[idx[rng.choice(len(idx), diff, replace=False)]] = 1
        else:
            idx = np.where(h == 1)[0]
            h[idx[rng.choice(len(idx), -diff, replace=False)]] = 0
    return h

def binary_hill_climb(h, n, time_limit, rng, max_stall=200):
    """Greedy hill climbing with random restarts on binary h."""
    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = int(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    start = time.time()
    stall = 0
    total_iters = 0
    
    while time.time() - start < time_limit and remaining() > 10:
        stall += 1
        if stall > max_stall:
            # Random perturbation: swap 2-3 pairs
            n_perturb = rng.randint(2, 5)
            for _ in range(n_perturb):
                if ones_list and zeros_list:
                    i = ones_list[rng.randint(len(ones_list))]
                    j = zeros_list[rng.randint(len(zeros_list))]
                    h[i] = 0
                    h[j] = 1
                    ones_list.remove(i)
                    zeros_list.remove(j)
                    ones_list.append(j)
                    zeros_list.append(i)
            corr = compute_corr(h)
            cur_max = int(np.max(corr))
            stall = 0
            continue
        
        # Try a single swap
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        
        h[i] = 0
        h[j] = 1
        corr = compute_corr(h)
        new_max = int(np.max(corr))
        
        if new_max < cur_max:
            # Accept
            cur_max = new_max
            ones_list.remove(i)
            zeros_list.remove(j)
            ones_list.append(j)
            zeros_list.append(i)
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
            stall = 0
        elif new_max == cur_max:
            # Accept with some probability (wander on plateau)
            if rng.rand() < 0.3:
                ones_list.remove(i)
                zeros_list.remove(j)
                ones_list.append(j)
                zeros_list.append(i)
        else:
            # Reject, revert
            h[i] = 1
            h[j] = 0
        
        total_iters += 1
    
    return best_h, best_max

def binary_sa(h, n, time_limit, rng):
    """Simulated annealing on binary h."""
    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    
    start = time.time()
    n_total = int(time_limit / 0.0002)  # estimate iterations
    T0 = 3.0
    T1 = 0.05
    alpha = (T1 / T0) ** (1.0 / max(n_total, 1))
    T = T0
    stall = 0
    
    for _ in range(n_total):
        if time.time() - start > time_limit or remaining() < 10:
            break
        
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        
        h[i] = 0
        h[j] = 1
        corr = compute_corr(h)
        new_max = float(np.max(corr))
        
        delta = new_max - cur_max
        
        if delta < 0 or (T > 0.01 and rng.rand() < np.exp(-delta / T)):
            cur_max = new_max
            ones_list.remove(i)
            zeros_list.remove(j)
            ones_list.append(j)
            zeros_list.append(i)
            
            if new_max < best_max:
                best_max = new_max
                stall = 0
            else:
                stall += 1
            
            if stall > 300:
                # Perturb
                for _ in range(3):
                    ii = ones_list[rng.randint(len(ones_list))]
                    jj = zeros_list[rng.randint(len(zeros_list))]
                    h[ii] = 0
                    h[jj] = 1
                    ones_list.remove(ii)
                    zeros_list.remove(jj)
                    ones_list.append(jj)
                    zeros_list.append(ii)
                corr = compute_corr(h)
                cur_max = float(np.max(corr))
                stall = 0
        else:
            h[i] = 1
            h[j] = 0
        
        T *= alpha
    
    return best_h, best_max
