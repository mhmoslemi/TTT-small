def lag_gradients(x, lags):
    n=len(x)
    lags=np.asarray(lags, dtype=int)

def _detect_corr_right():

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def evaluate_c5(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n <= 0:
        return float("inf")
    corr = full_corr(h)
    if not np.all(np.isfinite(corr)):
        return float("inf")
    return float(np.max(corr) * (2.0 / n))


def project_to_box_sum(h, n):
    x = np.asarray(h, dtype=float).reshape(-1).copy()
    x = np.clip(x, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(250):
        diff = target - x.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            mask = x < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                x[mask] += diff / float(cnt)
            else:
                x += diff / float(n)
        else:
            mask = x > 1e-12
            cnt = np.count_nonzero(mask)

def lag_gradients(x, lags):
    n = len(x)
    lags = np.asarray(lags, dtype=int)
    G = np.empty((len(lags), n), dtype=float)
    center = n - 1
    for r in range(len(lags)):
        d = int(lags[r]) - center
        a = abs(d)
        L = n - a
        g = G[r]
        g.fill(0.0)
        if L <= 0:
            continue
        if d > 0:
            use_minus = _CORR_RIGHT
        else:
            use_minus = (not _CORR_RIGHT)

        if use_minus:
            g[:L] += 1.0 - x[a:]
            g[a:] += -x[:L]
        else:
            g[a:] += 1.0 - x[:L]
            g[:L] -= x[a:]
    return G

def smooth_func(x, active, beta, lam):
    n = len(x)
    dx = 2.0 / n
    corr = full_corr(x)

        val = m
    else:

        val = m + np.log(np.sum(e)) / float(beta)

def smooth_jac(x, active, beta, lam):

            if r.size > 0:
                smax = min(smax, float(np.min(r)))

        if not np.isfinite(smax) or smax <= 1e-9:
            continue

        scale = min(0.045, 0.0006 + 0.045 * float(rng.random())) * min(1.0, smax)
        xn = x + scale * g
        xn = project_to_box_sum(xn, n)
        c = evaluate_c5(xn)
        if np.isfinite(c) and c < cbest - 1e-11:
            cbest = c
            xbest = xn.copy()
    return xbest, cbest


def pair_hill_climb(h, n, rng, best_in, deadline):
    x = project_to_box_sum(h, n)
    best_c = float(best_in)
    best_x = x.copy()
    ccur = evaluate_c5(x)
    if ccur < best_c:
        best_c = ccur
        best_x = x.copy()

    deltas = (0.0005, 0.001, 0.002, 0.003, 0.005, 0.008, 0.012, 0.02)
    moves = 0
    while time.time() < deadline - 2:
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        accepted = False
        for d in deltas:
            if x[i] >= d and x[j] <= 1.0 - d:
                oi = x[i]
                oj = x[j]
                x[i] = oi - d
                x[j] = oj + d
                c = evaluate_c5(x)
                if c < best_c - 1e-12:
                    best_c = c
                    best_x = x.copy()
                    accepted = True
                    break
                else:
                    x[i] = oi
                    x[j] = oj

            if x[j] >= d and x[i] <= 1.0 - d:
                oi = x[i]
                oj = x[j]
                x[j] = oj - d
                x[i] = oi + d
                c = evaluate_c5(x)
                if c < best_c - 1e-12:
                    best_c = c
                    best_x = x.copy()
                    accepted = True
                    break
                else:
                    x[j] = oj
                    x[i] = oi

        if accepted:
            moves += 1
    return best_x, best_c


def local_optimize(h0, n, local_deadline, rng):
    x = project_to_box_sum(h0, n)
    ccur = evaluate_c5(x)
    best_x = x.copy()
    best_c = ccur

    init_deadline = min(local_deadline - 12.0, time.time() + 45.0)
    if init_deadline > time.time() + 5.0:
        x, ccur = smooth_polish(x, n, init_deadline)
        if np.isfinite(ccur) and ccur < best_c:
            best_c = ccur
            best_x = x.copy()

    allm = 2 * n - 1
    if n <= 300:
        phases = [allm]
    else:
        phases = []
        for k in (50, 90, 150, 250, 380):
            if k >= allm:
                phases.append(allm)
                break
            phases.append(k)
        if len(phases) == 0 or phases[-1] != allm:
            phases.append(allm)

    step_list = (
        0.22, 0.18, 0.145, 0.115, 0.09, 0.072, 0.058, 0.047, 0.038,
        0.031, 0.025, 0.020, 0.016, 0.013, 0.0105, 0.0085, 0.0068,
        0.0054, 0.0043, 0.0034, 0.0027, 0.0021, 0.0017, 0.0013,
        0.0010, 0.0008, 0.00062, 0.0005, 0.0004, 0.00032, 0.00026
    )

    step_i = 0
    phase = 0
    stall = 0

    while time.time() < local_deadline - 14.0:
        corr = full_corr(x)
        if not np.all(np.isfinite(corr)):
            break
        ccur = float(np.max(corr) * (2.0 / n))
        if ccur < best_c:
            best_c = ccur
            best_x = x.copy()

        max_lags = int(phases[min(phase, len(phases) - 1)])
        margin = 0.005 if phase < len(phases) - 1 else 0.0015
        lags = select_lags(corr, max_lags, ccur, margin)
        step = float(step_list[min(step_i, len(step_list) - 1)])

        delta = solve_lp_step(x, lags, step)
        improved = False

        if delta is not None:
            dinf = float(np.max(np.abs(delta))) if delta.size else 0.0
            if dinf > 1e-7:
                x2, c2, _a = line_search(x, delta, ccur)
                if c2 < ccur - 1e-11:
                    x = x2
                    ccur = c2
                    if ccur < best_c:
                        best_c = ccur
                        best_x = x.copy()
                    improved = True
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1

        if not improved:
            if stall >= 2:
                old = ccur
                if step_i + 1 < len(step_list):
                    step_i += 1
                    stall = 0
                elif phase + 1 < len(phases):
                    phase += 1
                    step_i = 0
                    stall = 0
                else:
                    pol_deadline = min(local_deadline - 14.0, time.time() + 35.0)
                    x2, c2 = smooth_polish(x, n, pol_deadline)
                    if c2 < ccur:
                        x = x2
                        ccur = c2
                    if np.isfinite(ccur) and ccur < best_c:
                        best_c = ccur
                        best_x = x.copy()

                    esc_deadline = min(local_deadline - 14.0, time.time() + 14.0)
                    x3, c3 = random_escape(x, n, rng, ccur, esc_deadline)
                    if c3 < ccur:
                        x = x3
                        ccur = c3
                    if np.isfinite(ccur) and ccur < best_c:
                        best_c = ccur
                        best_x = x.copy()

                    if ccur >= old - 1e-12:
                        break
                    stall = 0

    x = project_to_box_sum(x, n)
    ccur = evaluate_c5(x)
    if np.isfinite(ccur) and ccur < best_c:
        best_c = ccur
        best_x = x.copy()
    return best_x, best_c


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if budget <= 1.0:
        run_span = budget - 0.05
    else:
        run_span = budget - 25.0
    global_deadline = t_start + max(1.0, run_span)

    rng = np.random.RandomState(int(seed))

    raw = None
    try:
        raw = initial_h_values
    except Exception:
        raw = None

    parent_h = None
    if raw is not None:
        try:
            parent_h = np.asarray(raw, dtype=float).reshape(-1)
            if parent_h.size == 0:
                parent_h = None
        except Exception:
            parent_h = None

    if parent_h is None:
        best_n = 160
        best_h = make_binary_random(best_n, rng)
    else:
        best_n = int(len(parent_h))
        if best_n < 8:
            best_n = 160
            best_h = make_binary_random(best_n, rng)
        else:
            best_h = project_to_box_sum(parent_h, best_n)

    best_c = evaluate_c5(best_h)
    if not np.isfinite(best_c):
        best_c = 1.0

    if global_deadline - time.time() > 5.0:
        for nn0 in (best_n - 20, best_n + 20, 2 * best_n, 2 * best_n - 20):
            nn = make_even(nn0)
            if 40 <= nn <= 512:
                htry = interp_h(best_h, best_n, nn)
                ctry = evaluate_c5(htry)
                if np.isfinite(ctr) and ctry < best_c:
                    best_c = ctry
                    best_h = htry
                    best_n = nn

    rem = global_deadline - time.time()
    if rem > 50.0:
        if rem > 700.0:
            parent_alloc = min(340.0, rem - 260.0)
        elif rem > 350.0:
            parent_alloc = min(220.0, rem - 120.0)
        else:
            parent_alloc = max(25.0, rem - 25.0)
        local_deadline = time.time() + parent_alloc
        reserve = max(20.0, 0.15 * (global_deadline - time.time()))
        if local_deadline > global_deadline - reserve:
            local_deadline = global_deadline - reserve
        if local_deadline > time.time() + 8.0:
            h, c = local_optimize(best_h, best_n, local_deadline, rng)
            if np.isfinite(c) and c < best_c:
                best_c = c
                best_h = h
                best_n = int(len(h))

    base = make_even(best_n)
    reqs = []
    for r in (
        2 * base, base + 40, base - 20, base + 20,
        2 * base - 40, 2 * base + 40,
        320, 256, 192, 288, 448, 384, 224, 512, 160
    ):
        nn = make_even(r)
        if 80 <= nn <= 512:
            dup = False
            for q in reqs:
                if q == nn:
                    dup = True
                    break
            if not dup:
                reqs.append(nn)

    caps = (260.0, 135.0, 95.0, 70.0, 52.0, 42.0, 32.0, 25.0,
            20.0, 17.0, 14.0, 12.0, 10.0, 8.0)
    idx = 0
    for nn in reqs:
        rem = global_deadline - time.time()
        if rem < 55.0 or nn == best_n:
            continue
        cap = caps[min(idx, len(caps) - 1)]
        if cap > rem - 50.0:
            cap = max(18.0, rem - 50.0)
        if cap < 18.0:
            break

        local_deadline = min(global_deadline - 30.0, time.time() + cap)
        hinit = interp_h(best_h, best_n, nn)

        if idx == 1:
            p = make_thue_morse(nn)
            hinit = project_to_box_sum(0.62 * hinit + 0.38 * p, nn)
        elif idx == 3:
            p = make_golden(nn)
            hinit = project_to_box_sum(0.62 * hinit + 0.38 * p, nn)
        elif idx % 2 == 1:
            noise = rng.normal(size=nn)
            hinit = project_to_box_sum(hinit + 0.018 * noise, nn)

        if local_deadline > time.time() + 8.0:
            h, c = local_optimize(hinit, nn, local_deadline, rng)
            if np.isfinite(c) and c < best_c:
                best_c = c
                best_h = h
                best_n = int(len(h))
        idx += 1

    rem = global_deadline - time.time()
    for rr in range(3):
        if rem < 45.0:
            break
        cap = min(32.0, rem - 40.0)
        if cap < 18.0:
            break
        if rr == 0:
            hinit = make_binary_random(best_n, rng)
        elif rr == 1:
            hinit = make_thue_morse(best_n)
        else:
            hinit = interp_h(best_h, best_n, best_n)
            hinit = project_to_box_sum(hinit + 0.04 * rng.normal(size=best_n), best_n)

        local_deadline = min(global_deadline - 20.0, time.time() + cap)
        h, c = local_optimize(hinit, best_n, local_deadline, rng)
        if np.isfinite(c) and c < best_c:
            best_c = c
            best_h = h
            best_n = int(len(h))
        rem = global_deadline - time.time()

    if global_deadline - time.time() > 8.0:
        for s in (-64, -48, -32, -24, -16, -8, 8, 16, 24, 32, 48, 64, 96, 128, -128):
            nn = make_even(best_n + s)
            if 80 <= nn <= 512:
                htry = interp_h(best_h, best_n, nn)
                ctry = evaluate_c5(htry)
                if np.isfinite(ctr) and ctry < best_c:
                    best_c = ctry
                    best_h = htry
                    best_n = int(len(htry))

    if global_deadline - time.time() > 18.0:
        h, c = smooth_polish(best_h, best_n, global_deadline)
        if np.isfinite(c) and c < best_c:
            best_c = c
            best_h = h
            best_n = int(len(h))

    if global_deadline - time.time() > 15.0 and best_n <= 360:
        h, c = pair_hill_climb(best_h, best_n, rng, best_c, global_deadline)
        if np.isfinite(c) and c < best_c:
            best_c = c
            best_h = h
            best_n = int(len(h))

    best_h = project_to_box_sum(best_h, best_n)
    final_c = evaluate_c5(best_h)
    if not np.isfinite(final_c):
        final_c = best_c

    try:
        ec = evaluate_erdos_solution(best_h)
        if isinstance(ec, (tuple, list)):
            ec = ec[0]
        ec = float(ec)
        if np.isfinite(ec) and ec < final_c:
            final_c = ec
    except Exception:
        pass

    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c))
    return (best_h, final_c, best_n)
