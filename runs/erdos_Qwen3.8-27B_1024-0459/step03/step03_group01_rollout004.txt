          sel=np.array(keep,dtype=int)

def full_corr_fft(h):
    n = len(h)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    a = np.fft.rfft(h, N)
    b = 1.0 - h
    ba = np.fft.rfft(b, N)
    c = np.fft.irfft(a * np.conj(ba), N)
    res = np.empty(L, dtype=float)
    res[:n - 1] = c[N - (n - 1):N]
    res[n - 1:] = c[:n]
    return res

def project_bisect(z, target):
    z = np.asarray(z, dtype=float)
    if z.size == 0:
        return z
    lo = float(z.min()) - 1.0
    hi = float(z.max()) + 1.0
    for _ in range(48):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)


def project_feasible(h, target):

    if h.size == 0:

def fix_binary(h, n, rng):

def active_grad(h, offs, weights=None):
    h = np.asarray(h, dtype=float)
    n = len(h)
    one_h = 1.0 - h
    g = np.zeros(n, dtype=float)
    for ii, off in enumerate(offs):

            w = 1.0
        else:

            g[:-a] += w * one_h[a:]

def active_offsets(corr, n, q=64):

    L = 2 * n - 1
    m = float(np.max(corr))
    top_idx = np.argsort(-corr)
    sel = top_idx[:max(1, min(q, L))]
    thr = max(1e-6, 0.002 * n)
    near = np.where(corr >= m - thr)[0]
    sel = np.unique(np.concatenate((sel, near, np.array([n - 1], dtype=int))))

    if len(sel) > q:
        order = np.argsort(-corr[sel])
        sel = sel[order]
        if not np.any(sel == n - 1):

            for v in sel:
                v = int(v)
                if v != n - 1 and len(keep) < q:
                    keep.append(v)
            sel = np.array(keep, dtype=int)

    return np.asarray(sel - (n - 1), dtype=int)

    return (m + np.log(s) / _BETA) * (2.0 / _N)


def _slq_jac(x):

        return np.zeros(len(x), dtype=float)
    vals = active_values(x, _OFF)
    m = float(np.max(vals))
    if _BETA <= 1.0:
        w = np.zeros(len(vals), dtype=float)
        w[int(np.argmax(vals))] = 1.0
    else:

        sh = np.clip(sh, -60.0, 60.0)
        ex = np.exp(sh)
        w = ex / float(np.sum(ex))
    g = active_grad(x, _OFF, w)
    return g * (2.0 / _N)

    return float(np.sum(x) - _TGT)


def _sum_jac(x):
    return np.ones(len(x), dtype=float)


def slsqp_active(h, n, beta, maxiter, q=64):
    global _N, _TGT, _BETA, _OFF
    h0 = project_feasible(np.asarray(h, dtype=float).copy(), n / 2.0)
    _N = int(n)
    _TGT = float(n) / 2.0
    _BETA = float(beta)
    corr = full_corr(h0)
    _OFF = tuple(int(x) for x in active_offsets(corr, n, q))

    if maxiter <= 0:
        return h0

    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": _sum_con, "jac": _sum_jac}]
    try:
        res = minimize(
            _slq_obj,
            h0,
            jac=_slq_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": int(maxiter), "ftol": 1e-11, "iprint": 0},
        )
        x = res.x
    except Exception:
        return h0
    return project_feasible(x, n / 2.0)


def lp_direction(G, u=0.05):
    m = G.shape[0]
    n = G.shape[1]
    if m == 0:
        return None
    if m > 48:
        norms = np.linalg.norm(G, axis=1)
        keep = np.argsort(-norms)[:48]
        G = G[keep]
        m = G.shape[0]

    c = np.zeros(n + 1, dtype=float)
    c[-1] = 1.0
    A = np.empty((m, n + 1), dtype=float)
    A[:, :n] = G
    A[:, -1] = -1.0
    b = np.zeros(m, dtype=float)
    Aeq = np.zeros((1, n + 1), dtype=float)
    Aeq[0, :n] = 1.0
    b_eq = np.zeros(1, dtype=float)
    bounds = [(-float(u), float(u))] * n + [(-1.0, 1.0)]

    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            A_eq=Aeq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
        )
    except Exception:
        return None

    if res.success and float(res.fun) < -1e-6:
        d = np.asarray(res.x[:n], dtype=float)
        if np.all(np.isfinite(d)):
            d = d - np.mean(d)
            return d
    return None


def lp_search(h, n, time_limit, q=32, u=0.04):
    deadline = time.time() + float(time_limit)
    h = project_feasible(np.asarray(h, dtype=float).copy(), n / 2.0)
    target = n / 2.0
    best_c = compute_c5(h)
    corr = full_corr(h)
    no_acc = 0
    max_iter = 160 if n <= 700 else 80

    for _ in range(max_iter):
        if time.time() > deadline - 2.0 or remaining_time() < 4.0:
            break
        if no_acc > 28:
            break

        offs = active_offsets(corr, n, q)
        G = grad_for_offsets(h, offs)
        d = lp_direction(G, u)
        if d is None:
            g = G.mean(axis=0)
            g = g - np.mean(g)
            d = -g
            dn = np.max(np.abs(d))
            if dn > u:
                d = d * (float(u) / dn)

        if not np.all(np.isfinite(d)) or np.max(np.abs(d)) < 1e-11:
            break

        improved = False
        for alpha in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625):
            tri = project_feasible(h + alpha * d, target)
            c = compute_c5(tri)
            if c < best_c - 1e-12:
                h = tri
                best_c = c
                improved = True
                break

        if improved:
            corr = full_corr(h)
            no_acc = 0
        else:
            no_acc += 1

    return h, best_c


def binary_swap_new_corr(corr, a, b, ones, zeros, base, L):
    tmp = corr.copy()
    idx = base + a - zeros
    tmp[idx] -= 1
    idx = base + ones - b
    tmp[idx] -= 1
    tmp[base + a - b] += 1

    idx = base + ones - a
    tmp[idx] += 1
    tmp[base] -= 1

    idx = base + b - zeros
    tmp[idx] += 1
    tmp[base] -= 1

    tmp[base + b - a] += 1
    return tmp, int(np.max(tmp))


def binary_local_search(h, n, time_limit, rng=None, max_trials=24):
    if rng is None:
        rng = np.random.RandomState(12345)
    deadline = time.time() + float(time_limit)

    hb = fix_binary(h, n, rng).astype(np.int8)
    h64 = hb.astype(np.int64)
    corr = np.correlate(h64, 1 - h64, mode="full").astype(np.int64)
    L = 2 * n - 1
    base = n - 1
    cur_max = int(np.max(corr))
    best_c5 = cur_max * (2.0 / n)
    ones = np.where(hb == 1)[0].astype(np.int64)
    zeros = np.where(hb == 0)[0].astype(np.int64)

    stall = 0
    while True:
        if remaining_time() < 5.0 or time.time() > deadline - 2.0:
            break

        idx = int(np.argmax(corr))
        off = idx - base
        cands = []

        if off > 0:
            aoff = off
            mask = (hb[:-aoff] == 0) & (hb[aoff:] == 1)
            z = np.where(mask)[0].astype(np.int64)
            if z.size:
                if z.size > max_trials:
                    z = z[rng.choice(z.size, max_trials, replace=False)]
                for zz in z:
                    cands.append((int(zz + aoff), int(zz)))
        elif off < 0:
            a = -off
            mask = (hb[:-a] == 1) & (hb[a:] == 0)
            o = np.where(mask)[0].astype(np.int64)
            if o.size:
                if o.size > max_trials:
                    o = o[rng.choice(o.size, max_trials, replace=False)]
                for oo in o:
                    cands.append((int(oo), int(oo + a)))

        best_new = cur_max
        best_pair = None
        best_corr = None

        for a, b in cands:
            if hb[a] != 1 or hb[b] != 0:
                continue
            tmp, nm = binary_swap_new_corr(corr, a, b, ones, zeros, base, L)
            if nm < best_new:
                best_new = nm
                best_pair = (a, b)
                best_corr = tmp

        improved = False
        if best_pair is not None and best_new < cur_max:
            a, b = best_pair
            corr = best_corr
            hb[a] = 0
            hb[b] = 1
            cur_max = int(np.max(corr))
            improved = True
        else:
            for _ in range(6):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                a = int(ones[rng.randint(len(ones))])
                b = int(zeros[rng.randint(len(zeros))])
                tmp, nm = binary_swap_new_corr(corr, a, b, ones, zeros, base, L)
                if nm < cur_max:
                    corr = tmp
                    hb[a] = 0
                    hb[b] = 1
                    cur_max = nm
                    improved = True
                    break

        if improved:
            if cur_max * (2.0 / n) < best_c5:
                best_c5 = cur_max * (2.0 / n)
            ones = np.where(hb == 1)[0].astype(np.int64)
            zeros = np.where(hb == 0)[0].astype(np.int64)
            stall = 0
        else:
            stall += 1
            if stall > 25:
                break

    return hb.astype(float), best_c5


def improve_candidate(h, n, time_limit, rng, global_best_c):
    deadline = time.time() + float(time_limit)
    h = project_feasible(np.asarray(h, dtype=float).copy(), n / 2.0)
    best_c_local = compute_c5(h)
    best_h_local = h.copy()

    if n <= 384:
        q = 96
        maxiter_base = 100
    elif n <= 720:
        q = 64
        maxiter_base = 70
    elif n <= 1200:
        q = 36
        maxiter_base = 35
    else:
        q = 24
        maxiter_base = 0

    betas = [15, 40, 100, 250, 600, 1200]
    bi = 0
    stalled_slsqp = 0

    while bi < len(betas):
        if time.time() > deadline - 70.0 or remaining_time() < 70.0:
            break
        beta = betas[bi]
        if maxiter_base > 0:
            h_new = slsqp_active(best_h_local, n, beta, maxiter_base, q)
            c_new = compute_c5(h_new)
            if c_new < best_c_local - 1e-12:
                best_c_local = c_new
                best_h_local = h_new.copy()
                stalled_slsqp = 0
            elif c_new <= best_c_local + 1e-9 and rng.rand() < 0.35:
                best_h_local = h_new.copy()
                stalled_slsqp = 0
            else:
                stalled_slsqp += 1
        bi += 1
        if stalled_slsqp > 2:
            break

    if time.time() < deadline - 55.0 and remaining_time() > 55.0:
        t = min(90.0, max(12.0, deadline - time.time() - 55.0, remaining_time() - 55.0))
        if t > 12.0:
            h_lp, c_lp = lp_search(best_h_local, n, t, q=min(q, 32), u=0.05)
            if c_lp < best_c_local - 1e-12:
                best_c_local = c_lp
                best_h_local = h_lp.copy()

    if n <= 1024 and time.time() < deadline - 35.0 and remaining_time() > 35.0:
        t = min(80.0, max(10.0, deadline - time.time() - 35.0, remaining_time() - 35.0))
        if t > 10.0:
            h_bin, c_bin = binary_local_search(best_h_local, n, t, rng, max_trials=28)
            if c_bin < best_c_local - 1e-12:
                best_c_local = c_bin
                best_h_local = h_bin.copy()
            elif c_bin < best_c_local + 0.02 and maxiter_base > 0:
                h_pol = slsqp_active(h_bin, n, 120, max(20, maxiter_base // 2), q)
                c_pol = compute_c5(h_pol)
                if c_pol < best_c_local - 1e-12:
                    best_c_local = c_pol
                    best_h_local = h_pol.copy()

    if time.time() < deadline - 12.0 and remaining_time() > 12.0:
        t = min(35.0, max(5.0, deadline - time.time() - 8.0, remaining_time() - 8.0))
        if t > 5.0:
            h_lp, c_lp = lp_search(best_h_local, n, t, q=min(q, 24), u=0.02)
            if c_lp < best_c_local - 1e-12:
                best_c_local = c_lp
                best_h_local = h_lp.copy()

    best_h_local = project_feasible(best_h_local, n / 2.0)
    c_exact = compute_c5(best_h_local)
    if c_exact < best_c_local:
        best_c_local = c_exact

    if best_c_local < global_best_c:
        return best_h_local, best_c_local
    return best_h_local, best_c_local


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = float(budget_s)
    rng = np.random.RandomState(int(seed) & 0x7fffffff)

    best_h = None
    best_c = float("inf")
    best_n = 0

    try:
        parent = np.asarray(initial_h_values, dtype=float)
        parent_n = int(len(parent))
        if parent_n >= 2:
            parent = project_feasible(parent, parent_n / 2.0)
            pc = compute_c5_exact(parent)
            best_h = parent.copy()
            best_c = pc
            best_n = parent_n
    except Exception:
        pass

    if best_h is None:
        best_n = 320
        best_h = make_thue(best_n, seed + 7)
        best_h = project_feasible(best_h, best_n / 2.0)
        best_c = compute_c5_exact(best_h)

    phase1 = float(min(0.32 * _BUDGET, 430.0))
    if phase1 > 30.0 and remaining_time() > 30.0:
        h1, c1 = improve_candidate(best_h, best_n, phase1, rng, best_c)
        if c1 < best_c:
            best_c = c1
            best_h = h1.copy()

    base = int(best_n)
    cand = []
    for x in (base + 128, base + 256, int(base * 1.25), int(base * 1.5),
              int(base * 2), int(base * 2.5), 512, 768, 1024):
        x = int(x)
        if x % 2:
            x += 1
        if x > base and 160 <= x <= 1280:
            cand.append(x)
    cand = sorted(set(cand))
    spaced = []
    for x in cand:
        if not spaced or x > spaced[-1] * 1.12:
            spaced.append(x)
    cand = spaced[:3]

    for ii, n_t in enumerate(cand):
        if remaining_time() < 70.0:
            break
        left_jobs = len(cand) - ii
        per = min(150.0, max(35.0, (remaining_time() - 70.0) / max(1, left_jobs)))
        if per < 35.0:
            break
        h_start = interp_to(best_h, best_n, n_t)
        h_start = project_feasible(h_start, n_t / 2.0)
        h2, c2 = improve_candidate(h_start, n_t, per, rng, best_c)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
            best_n = n_t

    if remaining_time() > 100.0:
        restart_ns = [int(best_n)]
        large = min(1024, max(512, int(base * 2)))
        if large % 2:
            large += 1
        restart_ns.append(large)
        restart_ns = sorted(set(restart_ns))[:2]

        inits = {
            0: make_thue,
            1: make_vander,
            2: make_sine,
            3: make_sine,
            4: make_random_bits,
        }

        for rr, n_r in enumerate(restart_ns):
            if remaining_time() < 70.0:
                break
            for init_id in range(5):
                if remaining_time() < 50.0:
                    break
                if remaining_time() < 45.0 and init_id >= 2:
                    break
                if init_id == 2:
                    h_init = inits[init_id](n_r, 2, 0.35, seed + 11 + init_id)
                elif init_id == 3:
                    h_init = inits[init_id](n_r, 3, 1.8, seed + 17 + init_id)
                else:
                    h_init = inits[init_id](n_r, seed + 23 + init_id + 5 * n_r)
                h_init = project_feasible(h_init, n_r / 2.0)
                c_init = compute_c5(h_init)
                if c_init > best_c + 0.06:
                    continue
                t = min(55.0, max(15.0, remaining_time() - 35.0))
                if t < 15.0:
                    break
                h3, c3 = improve_candidate(h_init, n_r, t, rng, best_c)
                if c3 < best_c:
                    best_c = c3
                    best_h = h3.copy()
                    best_n = n_r

    if remaining_time() > 35.0:
        tfin = remaining_time() - 10.0
        if tfin > 20.0:
            h4, c4 = improve_candidate(best_h, best_n, tfin, rng, best_c)
            if c4 < best_c:
                best_c = c4
                best_h = h4.copy()

    best_h = project_feasible(best_h, best_n / 2.0)
    best_c = min(best_c, compute_c5_exact(best_h))

    try:
        ev = evaluate_erdos_solution(best_h)
        if ev is not None:
            best_c = min(best_c, float(ev))
    except Exception:
        pass

    return (best_h, float(best_c), int(best_n))
