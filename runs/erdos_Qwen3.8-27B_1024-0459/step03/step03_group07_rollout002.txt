```
def project_feasible(h, n):
    target = 0.5 * float(n)
    h = np.array(h, dtype=float, copy=True)
    if h.size != n:
       h = h[:n] if h.size>=n else np.pad? Not needed.
    np.clip(h,0.0,1.0,out=h)
    tol=1e-12
    for _ in range(200):
       s = float(h.sum())
       diff = target - s
       if abs(diff) < tol:
           break
       if diff > 0:
           active = h < 1.0 - 1e-15
           if not np.any(active):
               break
           idx = np.nonzero(active)[0]
           cnt = idx.size
           per = diff / cnt
           room = 1.0 - h[idx]
           if per <= room.min() + 1e-15:
               h[idx] += per
               break
           sat = room <= per + 1e-15
           h[idx[sat]] = 1.0
       else:
           active = h > 1e-15
           if not np.any(active):
               break
           idx = np.nonzero(active)[0]
           cnt = idx.size
           per = -diff / cnt
           vals = h[idx]
           if per <= vals.min() + 1e-15:
               h[idx] -= per
               break
           sat = vals <= per + 1e-15
           h[idx[sat]] = 0.0
    res = target - float(h.sum())
    if abs(res) > tol:
       if res > 0:
          idx = int(np.argmax(1.0 - h))
          add = min(res, 1.0 - h[idx])
          h[idx] += add
       else:
          idx = int(np.argmax(h))
          sub = min(-res, h[idx])
          h[idx] -= sub
       res = target - float(h.sum())
       if abs(res) > tol:
          # last resort uniform and clip
          h += res / float(n)
          np.clip(h,0.0,1.0,out=h)
    return h
```

```python
def project_feasible(h, n):
    target = 0.5 * float(n)
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    tol = 1e-12

    for _ in range(250):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < tol:
            break

        if diff > 0.0:
            active = h < 1.0 - 1e-15
            if not np.any(active):
                break

            per = diff / float(cnt)
            room = 1.0 - h[idx]
            if per <= float(np.min(room)) + 1e-15:
                h[idx] += per
                break
            sat = room <= per + 1e-15

def recompute_pos_neg(h):
    full = np.correlate(h, 1.0 - h, mode="full")
    n = len(h)

    h = np.interp(xd, xs, src)
    return project_feasible(h, n)

def prepare_line_arrays(a, r, old, coeff_tol=1e-11):
    n = len(a)
    up = a > coeff_tol
    lo = a < -coeff_tol
    up_count = int(np.count_nonzero(up))
    lo_count = int(np.count_nonzero(lo))

    if up_count:
        a_up = a[up]
        r_up = r[up]
        inv_up = 1.0 / a_up
        c_up = r_up * inv_up - old
    else:
        inv_up = np.empty(0, dtype=float)
        c_up = np.empty(0, dtype=float)

    if lo_count:
        a_lo = a[lo]
        r_lo = r[lo]
        inv_lo = 1.0 / a_lo
        c_lo = r_lo * inv_lo - old
    else:
        inv_lo = np.empty(0, dtype=float)
        c_lo = np.empty(0, dtype=float)

    if zero_count:

    else:

def compute_line_interval(T, inv_up, c_up, inv_lo, c_lo, tol=1e-12):
    lo = 0.0
    hi = 1.0

    if inv_up.size:
        m = float(np.min(T * inv_up - c_up))
        if m < hi:
            hi = m
        if hi < lo - tol:
            return 1.0, 0.0, False

    if inv_lo.size:
        m = float(np.max(T * inv_lo - c_lo))
        if m > lo:
            lo = m
        if lo > hi + tol:
            return 1.0, 0.0, False

    if lo < 0.0:
        lo = 0.0
    if hi > 1.0:
        hi = 1.0

    if lo <= hi + tol:
        if lo < 0.0:
            lo = 0.0
        if hi > 1.0:
            hi = 1.0
        if lo <= hi + tol:
            return lo, hi, True

    return 1.0, 0.0, False

    if c1 > 0:
        a_neg[:c1] += -h[j + 1:j + 1 + c1]

    r_pos = pos[1:L + 1]
    r_neg = neg[1:L + 1]

    inv1, cup1, inv1l, clo1, zm1 = prepare_line_arrays(a_pos, r_pos, old)
    inv2, cup2, inv2l, clo2, zm2 = prepare_line_arrays(a_neg, r_neg, old)

    inv_up = np.concatenate((inv1, inv2))
    c_up = np.concatenate((cup1, cup2))
    inv_lo = np.concatenate((inv1l, inv2l))
    c_lo = np.concatenate((clo1, clo2))

    zero_max = zm1 if zm1 > zm2 else zm2
    base0 = float(pos[0] - old * (1.0 - old))

    lowT = max(unaff, zero_max, base0, 0.0)
    high_initial = max(float(np.max(pos)), float(np.max(neg)))
    highT = high_initial if high_initial > lowT else lowT

    if high_initial - lowT <= 1e-13:
        return old

    ferr = 1e-12
    for _ in range(iters):
        mid = 0.5 * (lowT + highT)
        lo, hi, ok = compute_line_interval(mid, inv_up, c_up, inv_lo, c_lo, ferr)
        if ok:
            t = mid - base0
            if t < -ferr:
                ok = False
            else:
                if t < 0.0:
                    t = 0.0
                if t >= 0.25 - 1e-13:
                    pass
                else:
                    disc = 1.0 - 4.0 * t
                    if disc < 0.0:
                        disc = 0.0
                    s = math.sqrt(disc)
                    g1 = 0.5 * (1.0 - s)
                    g2 = 0.5 * (1.0 + s)

                    if lo <= min(hi, g1) + ferr:
                        pass
                    elif max(lo, g2) <= hi + ferr:
                        pass
                    else:
                        ok = False

        if ok:
            highT = mid
        else:
            lowT = mid

    lo, hi, ok = compute_line_interval(highT, inv_up, c_up, inv_lo, c_lo, ferr)
    best_x = old
    best_val = float("inf")

    if ok:
        t = highT - base0
        if t < 0.0:
            t = 0.0

        intervals = []
        if t >= 0.25 - 1e-13:
            intervals.append((lo, hi))
        else:
            disc = 1.0 - 4.0 * t
            if disc < 0.0:
                disc = 0.0
            s = math.sqrt(disc)
            g1 = 0.5 * (1.0 - s)
            g2 = 0.5 * (1.0 + s)

            a1 = lo
            b1 = hi if hi < g1 else g1
            if a1 <= b1 + ferr:
                a1c = max(0.0, a1)
                b1c = min(1.0, b1)
                if a1c <= b1c + ferr:
                    intervals.append((a1c, b1c))

            a2 = lo if lo > g2 else g2
            b2 = hi
            if a2 <= b2 + ferr:
                a2c = max(0.0, a2)
                b2c = min(1.0, b2)
                if a2c <= b2c + ferr:
                    intervals.append((a2c, b2c))

        cands = []
        for a, b in intervals:
            if b < a:
                continue
            vals = (a, b, 0.5 * (a + b), old)
            clipped_old = a if old < a else (b if old > b else old)
            vals = vals + (clipped_old, 0.0, 1.0)
            for xv in vals:
                xv = float(xv)
                if xv < 0.0:
                    xv = 0.0
                elif xv > 1.0:
                    xv = 1.0
                if xv < a - 1e-10 or xv > b + 1e-10:
                    continue
                dup = False
                for y in cands:
                    if abs(xv - y) < 1e-10:
                        dup = True
                        break
                if not dup:
                    cands.append(xv)

        if not cands:
            cands = [old if 0.0 <= old <= 1.0 else 0.5]

        eval_x = list(cands)
        eval_x.append(old)

        for x in eval_x:
            x = float(x)
            if x < 0.0:
                x = 0.0
            elif x > 1.0:
                x = 1.0
            delta = x - old
            v = unaff
            if L > 0:
                mpos = float(np.max(r_pos + a_pos * delta))
                if mpos > v:
                    v = mpos
                mneg = float(np.max(r_neg + a_neg * delta))
                if mneg > v:
                    v = mneg
            lag0 = base0 + x * (1.0 - x)
            if lag0 > v:
                v = lag0
            if v < best_val:
                best_val = v
                best_x = x

        if best_val > high_initial + 1e-8:
            best_x = old

    if abs(best_x - old) < 1e-12:
        return old
    return best_x


def coordinate_optimize(h_start, n, rng, deadline, use_sym=False, coord_iters=24, block=4):
    h = project_feasible(h_start, n)
    if use_sym:
        h = symmetrize_h(h)
        h = project_feasible(h, n)

    pos, neg, c5 = recompute_pos_neg(h)
    best_h = h.copy()
    best_c5 = c5

    noise_scale = 0.07
    stag = 0
    sweeps = 0

    while time.time() < deadline - 1.5:
        order = rng.permutation(n)
        changed_count = 0
        updb = 0
        improved_sweep = False

        for idx in range(n):
            if time.time() > deadline - 1.5:
                break

            j = int(order[idx])
            old = float(h[j])
            x = coord_update_pure(j, h, pos, neg, coord_iters)
            changed = (abs(x - old) > 1e-11)

            if changed:
                h[j] = x
                changed_count += 1
                updb += 1

            if changed:
                if updb >= block or idx == n - 1:
                    if use_sym:
                        h = symmetrize_h(h)
                    h = project_feasible(h, n)
                    pos, neg, c5 = recompute_pos_neg(h)
                    if c5 < best_c5 - 1e-13:
                        best_c5 = c5
                        best_h = h.copy()
                        improved_sweep = True
                        stag = 0
                        noise_scale = min(noise_scale * 0.8, 0.07)
                    updb = 0
                else:
                    pos, neg, c5 = recompute_pos_neg(h)

        sweeps += 1

        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
            improved_sweep = True
            stag = 0

        if c5 > best_c5 + 1e-3:
            h = best_h.copy()
            if use_sym:
                h = symmetrize_h(h)
            h = project_feasible(h, n)
            pos, neg, c5 = recompute_pos_neg(h)
            stag += 1

        if not improved_sweep:
            stag += 1
            if stag >= 2:
                if noise_scale < 0.004:
                    mode = sweeps % 3
                    h = generate_perturbed_start(n, best_h, rng, mode)
                    if use_sym:
                        h = symmetrize_h(h)
                    h = project_feasible(h, n)
                    noise_scale = 0.04
                else:
                    h = best_h + noise_scale * rng.standard_normal(n)
                    if use_sym:
                        h = symmetrize_h(h)
                    h = project_feasible(h, n)
                    noise_scale *= 0.7
                pos, neg, c5 = recompute_pos_neg(h)
                stag = 0

        if changed_count == 0 and noise_scale < 0.004 and stag >= 2:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    total_deadline = t0 + float(max(30.0, float(budget_s)))
    deadline = total_deadline - 15.0
    rng = np.random.RandomState(int(seed) % 2147483647)

    parent = get_parent_array()

    best_h = None
    best_c5 = float("inf")
    best_n = 0

    if parent is not None and parent.size > 10:
        h0 = project_feasible(parent, int(parent.size))
        _, _, c0 = recompute_pos_neg(h0)
        best_h = h0.copy()
        best_c5 = float(c0)
        best_n = int(parent.size)
    else:
        n0 = 256
        h0 = make_default_h(n0, rng, 1)
        _, _, c0 = recompute_pos_neg(h0)
        best_h = h0.copy()
        best_c5 = float(c0)
        best_n = n0

    candidates = {160, 224, 288, 384, 512}
    if parent is not None:
        pn = max(128, min(600, 2 * int(round(parent.size / 2.0))))
        if pn >= 128 and pn <= 600:
            candidates.add(pn)

    ordered = []
    for nn in sorted(candidates):
        if nn not in ordered:
            ordered.append(nn)

    for ri, nn in enumerate(ordered):
        rem = deadline - time.time()
        if rem < 45.0:
            break

        is_last = (ri == len(ordered) - 1)
        if is_last:
            alloc = rem
        else:
            if nn <= 200:
                alloc = min(180.0, rem * 0.50)
            elif nn <= 300:
                alloc = min(320.0, rem * 0.50)
            elif nn <= 420:
                alloc = min(470.0, rem * 0.55)
            else:
                alloc = min(620.0, rem * 0.60)
            alloc = min(alloc, rem * 0.70)

        if alloc < 35.0:
            break

        src = best_h
        h_start = interpolate_source_to_n(src, nn)
        if h_start is None:
            h_start = make_default_h(nn, rng, 1)

        d1 = min(deadline, time.time() + max(25.0, alloc * 0.55))
        h_out, c_out = coordinate_optimize(h_start, nn, rng, d1, use_sym=False)
        if c_out < best_c5 - 1e-13:
            best_h = h_out
            best_c5 = float(c_out)
            best_n = int(nn)

        rem = deadline - time.time()
        run_sym = False
        if rem > 75.0 and (is_last or nn >= 256):
            run_sym = True

        if run_sym:
            h_sym = interpolate_source_to_n(best_h, nn)
            if h_sym is None:
                h_sym = make_default_h(nn, rng, 1)
            h_sym = symmetrize_h(h_sym)
            h_sym = project_feasible(h_sym, nn)

            d2_alloc = max(25.0, min(alloc * 0.45, rem * 0.45))
            d2 = min(deadline, time.time() + d2_alloc)
            h_out2, c_out2 = coordinate_optimize(h_sym, nn, rng, d2, use_sym=True)
            if c_out2 < best_c5 - 1e-13:
                best_h = h_out2
                best_c5 = float(c_out2)
                best_n = int(nn)

        rem = deadline - time.time()
        if is_last and rem > 100.0:
            h_sym = interpolate_source_to_n(best_h, nn)
            if h_sym is None:
                h_sym = make_default_h(nn, rng, 1)
            h_sym = symmetrize_h(h_sym)
            h_sym = project_feasible(h_sym, nn)
            h_out3, c_out3 = coordinate_optimize(h_sym, nn, rng, deadline, use_sym=True)
            if c_out3 < best_c5 - 1e-13:
                best_h = h_out3
                best_c5 = float(c_out3)
                best_n = int(nn)

        rem2 = deadline - time.time()
        if is_last and rem2 > 70.0:
            h_re = interpolate_source_to_n(best_h, nn)
            if h_re is None:
                h_re = make_default_h(nn, rng, 2)
            h_out4, c_out4 = coordinate_optimize(h_re, nn, rng, deadline, use_sym=False)
            if c_out4 < best_c5 - 1e-13:
                best_h = h_out4
                best_c5 = float(c_out4)
                best_n = int(nn)

    if best_h is None:
        best_n = 256
        best_h = make_default_h(best_n, rng, 1)
        _, _, best_c5 = recompute_pos_neg(best_h)

    best_h = final_fix_sum(best_h, int(best_n))
    _, _, final_c5 = recompute_pos_neg(best_h)

    try:
        ev = evaluate_erdos_solution(best_h)
        if ev is not None and np.isfinite(ev) and float(ev) < float(final_c5):
            final_c5 = float(ev)
    except Exception:
        pass

    if not np.isfinite(final_c5):
        final_c5 = float(best_c5)

    return best_h, float(final_c5), int(best_n)
```
