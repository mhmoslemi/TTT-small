        grad = grad + pen_w * sv * np.ones(n)
        return obj, grad

                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_lse(h, n, beta, maxiter):
    def func(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        sh = np.clip(beta * (corr - mc), -400, 400)
        dx = 2.0 / n

        return np.sum(x) - n / 2.0

                       constraints=[{'type': 'eq', 'fun': constr}],

def lag_grad(h, n, offset):
    grad = np.zeros(n)

        np_ = n - a

            grad[a:a + np_] -= (1.0 - h[:np_])
            grad[:np_] += h[a:a + np_]
    return grad
