def solve_lp_linearized(h_old, n):
    """Solve the linearized LP at iteration."""
    dx = 2.0 / n
    
    # Compute linearization for each lag
    # Variables: h[0..n-1], M (index 0 to n, M at index n)
    # Minimize M → c = [0,...,0,1]
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize M
    
    # Inequality constraints: A_ineq @ x <= b_ineq
    # For each lag k, we have: sum_j a[k,j]*h[j] + const[k] <= M
    # → sum_j a[k,j]*h[j] - M <= -const[k]
    
    num_lags = 2 * n - 1
    A = np.zeros((num_lags, n + 1))
    b = np.zeros(num_lags)
    
    for lag in range(-(n-1), n):
        fidx = lag + (n-1)  # index in A
        A[fidx, n] = -1.0  # -M term
        
        if lag >= 0:
            # Coefficient of h[j]: (1-h_old[j+lag]) * [j <= n-1-lag] - h_old[j-lag] * [lag <= j]
            # const = sum_i h_old[i]*h_old[i+lag] for i=0 to n-1-lag
            for i in range(n - lag):
                # h[i] term: coefficient (1 - h_old[i+lag])
                A[fidx, i] += (1.0 - h_old[i + lag])
                # h[i+lag] term: coefficient -h_old[i]
                A[fidx, i + lag] += (-h_old[i])
            const = np.sum(h_old[:n-lag] * h_old[lag:])
            b[fidx] = -const
        else:
            s = -lag
            # Coefficient of h[j]: (1-h_old[j-s]) * [s <= j <= n-1] - h_old[j+s] * [j <= n-1-s]
            # const = sum_i h_old[i]*h_old[i-s] for i=s to n-1
            for i in range(s, n):
                # h[i] term: coefficient (1 - h_old[i-s])
                A[fidx, i] += (1.0 - h_old[i - s])
                # h[i-s] term: coefficient -h_old[i]
                A[fidx, i - s] += (-h_old[i])
            const = np.sum(h_old[s:] * h_old[:n-s])
            b[fidx] = -const
    
    # Equality constraint: sum(h) = n/2
    # In linprog, we use A_eq, b_eq
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 <= h[j] <= 1, M >= 0
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], res.x[n]
    return None, None
