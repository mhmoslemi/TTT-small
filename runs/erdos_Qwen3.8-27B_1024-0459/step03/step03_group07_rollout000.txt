def project_box_sum_bisect(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.clip(x, 0.0, 1.0)
    if x.size == 0: return x
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(60):
       mid = 0.5*(lo+hi)
       y = np.clip(x - mid, 0.0, 1.0)
       s = float(np.sum(y))
       if s > target:
          lo = mid
       else:
          hi = mid
    tau = 0.5*(lo+hi)
    y = np.clip(x - tau, 0.0, 1.0)
    diff = target - float(np.sum(y))
    if abs(diff) > 1e-12:
       if diff > 0:
          m = y < 1.0
          c = int(np.sum(m))
          if c: 
             y[m] += diff/c
             y = np.clip(y,0,1)
       else:
          m = y > 0.0
          c = int(np.sum(m))
          if c:
             y[m] += diff/c
             y = np.clip(y,0,1)
    return y

def correlate_counts(h):

def c5_from_h(h):
    n = len(h)
    if n <= 0:
        return float("inf")
    return float(np.max(correlate_counts(h))) * (2.0 / float(n))


def project_box_sum_bisect(x, target):
    x = np.asarray(x, dtype=float).copy()
    if x.size == 0:
        return x
    x = np.clip(x, 0.0, 1.0)
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        s = float(np.sum(y))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    y = np.clip(x - tau, 0.0, 1.0)
    diff = target - float(np.sum(y))
    if abs(diff) > 1e-12:
        if diff > 0.0:
            m = y < 1.0
            c = int(np.sum(m))
            if c > 0:
                y[m] += diff / float(c)
                y = np.clip(y, 0.0, 1.0)
        else:
            m = y > 0.0
            c = int(np.sum(m))
            if c > 0:
                y[m] += diff / float(c)
                y = np.clip(y, 0.0, 1.0)
    return y


def force_exact_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = project_box_sum_bisect(h, target)
    diff = target - float(np.sum(h))
    if abs(diff) > 0.0:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
        if h[idx] < 0.0:
            h[idx] = 0.0
        elif h[idx] > 1.0:
            h[idx] = 1.0
    return h


def interpolate_h(h_src, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * (2.0 / float(n_src))
    xn = (np.arange(n_dst) + 0.5) * (2.0 / float(n_dst))
    y = np.interp(xn, xp, h_src)
    return project_box_sum_bisect(y, 0.5 * float(n_dst))


def unit_delta_coord(j, h, n):
    d = np.zeros(2 * n - 1, dtype=float)

    a = min(j, n - 1)
    if a > 0:
        d[n:n + a] += 1.0 - h[j - a:j][::-1]

    b = n - 1 - j
    if b > 0:
        d[n:n + b] -= h[j + 1:j + b + 1]

    c = n - 1 - j
    if c > 0:
        d[n - 2:n - c - 2:-1] += 1.0 - h[j + 1:j + c + 1]

    e = j
    if e > 0:
        d[n - 2:n - e - 2:-1] -= h[j - 1:j - e - 1:-1]

    d[n - 1] = 0.0
    return d


def gradient_from_indices(h, n, idx, w):
    g = np.zeros(n, dtype=float)
    r = len(idx)
    for p in range(r):
        ii = int(idx[p])
        wt = float(w[p])
        if wt == 0.0:
            continue
        lag = ii - (n - 1)
        if lag == 0:
            g += wt * (1.0 - 2.0 * h)
        elif lag > 0:
            m = n - lag
            g[lag:n] += wt * (1.0 - h[0:m])
            g[0:m] += -wt * h[lag:n]
        else:
            s = -lag
            m = n - s
            g[0:m] += wt * (1.0 - h[s:n])
            g[s:n] += -wt * h[0:m]
    return g


def active_indices_weights(corr, k, beta):

    k = int(max(1, min(k, m)))
    maxc = float(np.max(corr))
    gap = max(0.5, 0.02 * maxc)

    cap = int(min(m, max(2 * k, 32)))

    if idx.size == 0:

    a = vals - float(np.max(vals))

def gradient_pair(g, h, n, rng, topc):
    topc = int(max(1, min(topc, n)))
    low = g.copy()
    high = g.copy()
    low[h >= 1.0 - 1e-9] = np.inf
    high[h <= 1e-9] = -np.inf

    low_ord = np.argsort(low)[:topc]
    high_ord = np.argsort(high)[::-1][:topc]

    for _ in range(max(4, 3 * topc)):
        i = int(low_ord[int(rng.randint(topc))])
        j = int(high_ord[int(rng.randint(topc))])
        if i != j and h[i] < 1.0 - 1e-9 and h[j] > 1e-9:
            return i, j

    i = int(np.argmin(low))
    j = int(np.argmax(high))
    if i == j or h[i] >= 1.0 - 1e-9 or h[j] <= 1e-9:
        for _ in range(2 * n):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i != j and h[i] < 1.0 - 1e-9 and h[j] > 1e-9:
                break
    if i == j:
        j = (j + 1) % n
    return i, j


def pair_eval_t(corr, delta, t, n, u, v):
    vals = corr + delta * t
    vals[n - 1] = corr[n - 1] + 2.0 * (v - u) * t - 2.0 * t * t
    return float(np.max(vals))


def golden_pair(corr, delta, lo, hi, n, u, v, iters):
    gr = 0.6180339887498949
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = pair_eval_t(corr, delta, c, n, u, v)
    fd = pair_eval_t(corr, delta, d, n, u, v)
    for _ in range(iters):
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = pair_eval_t(corr, delta, c, n, u, v)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = pair_eval_t(corr, delta, d, n, u, v)
    return 0.5 * (lo + hi)


def pair_line_search(corr, delta, tlo, thi, n, u, v, grid_points, refine_iters, centers):
    f0 = float(np.max(corr))
    best_t = 0.0
    best_f = f0
    if thi - tlo <= 1e-13:
        return best_t, best_f

    grid_points = int(max(11, grid_points))
    centers = int(max(1, min(centers, grid_points)))
    ts = np.linspace(float(tlo), float(thi), grid_points)

    vals = ts[:, None] * delta[None, :] + corr[None, :]
    vals[:, n - 1] = corr[n - 1] + 2.0 * (v - u) * ts - 2.0 * ts * ts
    obj = vals.max(axis=1)

    if centers < grid_points:
        cand = np.argpartition(obj, centers - 1)[:centers]
    else:
        cand = np.argsort(obj)[:grid_points]

    for qi in cand:
        qi = int(qi)
        ilo = max(0, qi - 2)
        ihi = min(grid_points - 1, qi + 2)
        lo = float(ts[ilo])
        hi = float(ts[ihi])
        if hi - lo <= 1e-13:
            continue
        t_r = golden_pair(corr, delta, lo, hi, n, u, v, refine_iters)
        f_r = pair_eval_t(corr, delta, t_r, n, u, v)
        if f_r < best_f:
            best_f = f_r
            best_t = t_r

    for t in (float(tlo), float(thi), 0.0):
        if t < float(tlo) - 1e-12 or t > float(thi) + 1e-12:
            continue
        f_t = pair_eval_t(corr, delta, t, n, u, v)
        if f_t < best_f:
            best_f = f_t
            best_t = t

    return float(best_t), float(best_f)

def smooth_max_count(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(corr))
    scaled = beta * (corr - m)
    scaled = np.clip(scaled, -50.0, 0.0)
    return m + float(np.log(np.sum(np.exp(scaled)))) / float(beta)


def sum_constraint(x, n):
    return float(np.sum(x) - 0.5 * float(n))


def slsqp_polish(h, n, deadline):
    if minimize is None:
        return h, c5_from_h(h)
    h = project_box_sum_bisect(h, 0.5 * float(n))
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": sum_constraint, "args": (n,)}]
    best_h = h.copy()
    best_c = c5_from_h(best_h)

    betas = [5.0, 20.0]
    if n <= 170:
        betas = [5.0, 20.0, 80.0]

    for b in betas:
        if time.time() >= deadline:
            break
        try:
            maxiter = 30 if n <= 160 else 15
            res = minimize(
                smooth_max_count,
                best_h,
                args=(n, b),
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": maxiter, "ftol": 1e-12, "iprint": 0}
            )
            h2 = project_box_sum_bisect(res.x, 0.5 * float(n))
            c2 = c5_from_h(h2)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
        except Exception:
            pass

    return best_h, best_c

            topc = 10

        h_res, f_res = pair_optimizer(
            seed_h,
            n_ph,
            rng,
            ph_deadline,
            1000000,
            a_k,
            beta,
            topc,
            step_limit,
            anneal
        )
        c_res = f_res * (2.0 / float(n_ph))
        if c_res < best_c5 - 1e-13:
            best_c5 = float(c_res)
            best_h = h_res.copy()
            best_n = n_ph

    now = time.time()
    if minimize is not None and best_n <= 192 and now < global_deadline - 180.0:
        sl_deadline = min(global_deadline - 60.0, now + 120.0)
        h_sls, c_sls = slsqp_polish(best_h, best_n, sl_deadline)
        if c_sls < best_c5 - 1e-13:
            best_c5 = float(c_sls)
            best_h = h_sls.copy()
            best_n = best_n

    now = time.time()
    if now < global_deadline - 10.0:
        h_fin, f_fin = pair_optimizer(
            best_h,
            best_n,
            rng,
            global_deadline - 2.0,
            1000000,
            10,
            0.0,
            4,
            0.003,
            0.0
        )
        c_fin = f_fin * (2.0 / float(best_n))
        if c_fin < best_c5 - 1e-13:
            best_c5 = float(c_fin)
            best_h = h_fin.copy()

    best_h = force_exact_sum(best_h, 0.5 * float(best_n))
    final_c5 = c5_from_h(best_h)

    try:
        ev = evaluate_erdos_solution(best_h)
        evf = float(ev)
        if np.isfinite(evf) and evf < final_c5:
            final_c5 = evf
    except Exception:
        pass

    return (best_h, float(final_c5), int(best_n))
