import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_gradient(h, n, lag_index):
    """Gradient of corr[lag_index] with respect to h."""
    grad = np.zeros(n)
    shift = lag_index - (n - 1)
    if shift >= 0:
        m = n - shift
        # corr = sum_{i=0}^{m-1} h[i] * (1 - h[i+shift])
        # d/dh[i] = (1-h[i+shift]) for i in [0, m-1]
        # d/dh[i+shift] = -h[i] for i in [0, m-1]
        grad[:m] += (1.0 - h[shift:shift+m])
        grad[shift:shift+m] -= h[:m]
    else:
        s = -shift
        m = n - s
        # corr = sum_{i=s}^{n-1} h[i] * (1 - h[i-s])
        # or equivalently sum_{j=0}^{m-1} h[j+s] * (1-h[j])
        # d/dh[j+s] = (1-h[j])
        # d/dh[j] = -h[j+s]
        grad[s:s+m] += (1.0 - h[:m])
        grad[:m] -= h[s:s+m]
    return grad

def c5_objective_and_grad(h):
    """Return (c5, gradient of c5) using exact max."""
    n = len(h)

    corr = compute_corr(h)
    max_val = np.max(corr)

    tolerance = 1e-12
    active = np.where(corr >= max_val - tolerance)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(corr))])
    # Gradient is average of active correlations' gradients
    grad = np.zeros(n)

        else:

def smooth_max_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def smooth_max_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    for k in range(len(corr)):
        grad += weights[k] * corr_gradient(x, n, k)
    return grad * dx

    return np.correlate(h, 1.0 - h, mode='full')

def corr_gradient(h, n, lag_index):
    grad = np.zeros(n)
    shift = lag_index - (n - 1)
    if shift >= 0:
        m = n - shift
        if m <= 0:
            return grad
        grad[:m] += (1.0 - h[shift:shift+m])
        grad[shift:shift+m] -= h[:m]
    else:
        s = -shift
        m = n - s
        if m <= 0:
            return grad
        grad[s:s+m] += (1.0 - h[:m])
        grad[:m] -= h[s:s+m]
    return grad
