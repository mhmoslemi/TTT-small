```python
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    if top_k is not None:
        idx = np.argsort(corr)[-top_k:]
        # Only use top_k lags
        corr_work = corr[idx]
        m_work = corr_work.max()
        sw = np.exp(beta * (corr_work - m_work))
        w = np.zeros(2*n-1)
        w[idx] = sw
        w /= w.sum()
    else:
        w = np.exp(beta * shifted)
        w /= w.sum()
```

```python
        idx = np.argsort(corr)[-top_k:]
        m = corr[idx].max()
        sw = np.exp(beta * (corr[idx] - m))
        w = np.zeros(len(corr))
        w[idx] = sw
        w /= w.sum()
        val = m + np.log(sw.sum()) / beta

        m = np.max(corr)
        shifted = corr - m
        w = np.exp(beta * shifted)
        w /= w.sum()

        val = m + np.log(np.sum(np.exp(beta * shifted))) / beta

    dx = 2.0/n
    corr = compute_corr(h)
    val, w = smooth_max_and_weights(corr, beta, top_k)

def compute_gradient(h, w, beta, top_k=None):
    n = len(h)
    dx = 2.0/n
    grad = np.zeros(n)
    for i in range(n):
        t_lo = max(0, i)
        t_hi = min(2*n-2, i+n-1)
        j_lo = t_lo - i
        j_hi = t_hi - i
        grad[i] = np.sum(w[t_lo:t_hi+1] * (1 - 2*h[j_lo:j_hi+1]))
```

```python
import numpy as np
import time
from scipy.optimize import minimize


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)


def smooth_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k is not None and top_k < len(corr):
        idx = np.argsort(corr)[-top_k:]
        cw = corr[idx]
        mw = cw.max()
        sw = np.exp(beta * (cw - mw))
        w = np.zeros(len(corr))
        w[idx] = sw
        w /= w.sum()
        val = (mw + np.log(sw.sum()) / beta) * dx
    else:
        m = np.max(corr)
        shifted = corr - m
        w = np.exp(beta * shifted)
        w /= w.sum()
        val = (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx
    grad = np.zeros(n)
    a = 1.0 - 2.0 * h
    for i in range(n):
        t_lo = i
        t_hi = min(2 * n - 2, i + n - 1)
        j_hi = t_hi - i
        grad[i] = np.dot(w[t_lo:t_hi + 1], a[0:j_hi + 1])
    grad *= dx
    return val, grad


def interp_to_resolution(h_src, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_new = np.interp(xn, xp, h_src)
    return h_new


def lbfgs_smooth(h0, n, beta, top_k, maxiter=500, ftol=1e-20):
    target = n / 2.0
    pmw = 2000.0

    def obj(params):
        h = np.clip(params, 0.0, 1.0)
        val, _ = smooth_val_grad(h, beta, top_k)
        se = np.sum(h) - target
        return val + pmw * se * se

    def obj_grad(params):
        h = np.clip(params, 0.0, 1.0)
        val, g = smooth_val_grad(h, beta, top_k)
        se = np.sum(h) - target
        g = g + 2.0 * pmw * se * np.ones(n)
        return val, g

    res = minimize(obj_grad, h0, method='L-BFGS-B',
                   jac=True,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-18,
                            'maxls': 30, 'maxfun': 10000})
    return res.x, res.nit, res.success


def proj_gradient_step(h, target, beta, top_k, lr):
    n = len(h)
    val, g = smooth_val_grad(h, beta, top_k)
    h_new = h - lr * g
    h_new = project_feasible(h_new, target)
    return h_new


def binary_local_search(h_bin, n, max_iters, rng, t0, budget_s):
    target = n / 2.0
    dx = 2.0 / n
    h = h_bin.copy()
    corr = compute_corr(h)
    cur_max = np.max(corr)
    best_corr = corr.copy()
    best_max = cur_max
    best_h = h.copy()
    ones_idx = np.where(h == 1.0)[0]
    zeros_idx = np.where(h == 0.0)[0]
    tabu = set()
    tabu_len = 30
    temp = 1.0

    for it in range(max_iters):
        elapsed = time.time() - t0
        if elapsed > budget_s - 10:
            break
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        if (i, j) in tabu or (j, i) in tabu:
            continue
        # Compute delta for swap i(1->0), j(0->1)
        delta = np.zeros(2 * n - 1)
        for t in range(2 * n - 1):
            ti = t - i
            tj = t - j
            d = 0.0
            # m=i: old=1*(1-h[ti]), new=0
            if 0 <= ti <= n - 1:
                d -= (1.0 - h[ti])
                # if ti == j, h[ti] is becoming 1, so old was 1*(1-0)=1, already handled
                # if ti == i, h[ti] is becoming 0, but m=i is already 0
                # Actually need to account for h' changes in the (1-h) factor
                hti_new = h[ti]
                if ti == i:
                    hti_new = 0.0
                elif ti == j:
                    hti_new = 1.0
                d += 0.0 * (1.0 - hti_new)  # new term is 0
            # m=j: old=0*(1-h[tj]), new=1*(1-h'[tj])
            if 0 <= tj <= n - 1:
                h_tj_new = h[tj]
                if tj == i:
                    h_tj_new = 0.0
                elif tj == j:
                    h_tj_new = 1.0
                d += (1.0 - h_tj_new)
            # m=t-i (second factor reference i), if t-i not in {i,j}
            m2 = t - i
            if 0 <= m2 <= n - 1 and m2 != i and m2 != j:
                d += h[m2]  # new: h[m2]*(1-0) - old: h[m2]*(1-1) = h[m2]*1 - 0
            # m=t-j (second factor reference j), if t-j not in {i,j}
            m3 = t - j
            if 0 <= m3 <= n - 1 and m3 != i and m3 != j:
                d -= h[m3]  # new: h[m3]*(1-1) - old: h[m3]*(1-0) = 0 - h[m3]
            delta[t] = d

        new_corr = corr + delta
        new_max = np.max(new_corr)
        # Acceptance
        if new_max <= cur_max + temp * 0.1:
            # Accept the swap
            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            cur_max = new_max
            # Update indices
            this_i = np.where(ones_idx == i)[0]
            if len(this_i) > 0:
                ones_idx = np.delete(ones_idx, this_i[0])
            this_j = np.where(zeros_idx == j)[0]
            if len(this_j) > 0:
                zeros_idx = np.delete(zeros_idx, this_j[0])
            ones_idx = np.append(ones_idx, j)
            zeros_idx = np.append(zeros_idx, i)

            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                best_corr = corr.copy()
        temp *= 0.999

        if it % tabu_len == tabu_len - 1:
            tabu.add((i, j))
            if len(tabu) > tabu_len * 2:
                # Remove oldest (simplified: clear)
                tabu.pop()

    return best_h, best_max * dx


def continuous_local_search(h, n, rng, t0, budget_s):

    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur
    no_improve = 0

    for sweep in range(200):
        elapsed = time.time() - t0
        if elapsed > budget_s - 15:
            break
        if no_improve > 30:
            break
        # Get gradient to guide moves
        val, g = smooth_val_grad(h_cur, beta=500, top_k=8)
        # Find top positive and negative gradient indices
        pos_idx = np.argsort(g)[::-1][:20]  # most negative g (decrease h)
        neg_idx = np.argsort(g)[:20]  # most positive g (increase h)

        improved = False
        for _ in range(200):
            elapsed = time.time() - t0
            if elapsed > budget_s - 15:
                break
            # Pick a decrease index and increase index
            i = pos_idx[rng.randint(len(pos_idx))]
            j = neg_idx[rng.randint(len(neg_idx))]
            if i == j:
                continue
            for delta in [0.02, 0.01, 0.005, 0.05]:
                di = -delta
                dj = delta
                if h_cur[i] + di < -1e-9 or h_cur[i] + di > 1.0 + 1e-9:
                    continue
                if h_cur[j] + dj < -1e-9 or h_cur[j] + dj > 1.0 + 1e-9:
                    continue
                h_cur[i] += di
                h_cur[j] += dj
                h_cur = project_feasible(h_cur, target)
                c5_test = compute_c5(h_cur)
                if c5_test < best_c5 - 1e-10:
                    best_c5 = c5_test
                    best_h = h_cur.copy()
                    improved = True
                    break
                h_cur[i] -= di
                h_cur[j] -= dj

        if improved:
            no_improve = 0
        else:

            h_cur = best_h.copy()
            # Add small random perturbation to escape
            if no_improve < 30:
                pert = rng.randn(n) * 0.01

            below = np.where(h_bin == 0.0)[0]
            order = np.argsort(-best_h[below])
            for _ in range(int(np.ceil(diff))):
                if _ < len(order):
                    h_bin[below[order[_]]] = 1.0
        elif diff < 0:
            above = np.where(h_bin == 1.0)[0]
            order = np.argsort(best_h[above])
            for _ in range(int(np.ceil(-diff))):
                if _ < len(order):
                    h_bin[above[order[_]]] = 0.0
        h_bin = project_feasible(h_bin, target_b)
        # Also try random binary
        rng_bin = np.zeros(n_b)
        perm = rng.permutation(n_b)
        rng_bin[perm[:n_b // 2]] = 1.0
        c5_rand = compute_c5(rng_bin)
        c5_bin = compute_c5(h_bin)
        if c5_rand < c5_bin:
            h_bin = rng_bin

        max_iters_bin = min(50000, int(time_left() * 5000))
        h_bin_best, c5_bin_best = binary_local_search(h_bin, n_b, max_iters_bin, rng, t0, time_left() + t0)
        if c5_bin_best < best_c5:
            # Try to refine: start from binary, allow small continuous moves
            h_refine = h_bin_best.copy()
            for beta in [200, 1000]:
                if time_left() < 15:
                    break
                try:
                    x_new, _, _ = lbfgs_smooth(h_refine, n_b, beta, 12, maxiter=200)
                    h_refine = project_feasible(x_new, target_b)
                    c5_r = compute_c5(h_refine)
                    if c5_r < best_c5:
                        best_c5 = c5_r
                        best_h = h_refine.copy()
                except Exception:
                    pass
            print("  Binary search: C5={:.6f}".format(best_c5))

    # Phase 6: Final SLSQP polish
    if time_left() > 15:
        n = best_n
        target = n / 2.0
        dx = 2.0 / n
        bounds = [(0.0, 1.0)] * n
        cons_sum = {'type': 'eq', 'fun': None, 'jac': None}

        def slsqp_obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(corr)
            shifted = corr - m
            return (m + np.log(np.sum(np.exp(800.0 * shifted))) / 800.0) * dx

        def slsqp_obj_grad(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(corr)
            shifted = corr - m
            w = np.exp(800.0 * shifted)
            w /= w.sum()
            val = (m + np.log(np.sum(w)) / 800.0) * dx
            # val = (m + log(sum(exp(800*shifted)))/800)*dx
            # derivative of log(sum(exp(800*s_k)))/800 = sum(exp(800*s_k)*800)/sum(exp(800*s_k)) * ds_k/dx / 800 = sum(w_k * ds_k/dx)
            # ds_k/dx_i = dc_k/dx_i = (1-2*h[s_k - i]) for valid
            # So grad_i = dx * sum_k w_k * dc_k/dx_i
            grad = np.zeros(n)
            a = 1.0 - 2.0 * x
            for i in range(n):
                t_lo = i
                t_hi = min(2 * n - 2, i + n - 1)
                j_hi = t_hi - i
                grad[i] = np.dot(w[t_lo:t_hi + 1], a[0:j_hi + 1])
            grad *= dx
            return val, grad

        def sum_func(x):
            return np.sum(x) - target

        def sum_jac(x):
            return np.ones(n)

        cons = [{'type': 'eq', 'fun': sum_func, 'jac': sum_jac}]

        for beta_unused in [0]:  # just do one round
            if time_left() < 10:
                break
            try:
                res = minimize(slsqp_obj_grad, best_h, method='SLSQP',
                               jac=True, bounds=bounds, constraints=cons,
                               maxiter=200, ftol=1e-16)
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  SLSQP: C5={:.6f}".format(best_c5))
            except Exception:
                pass

    # Final verification
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
```
