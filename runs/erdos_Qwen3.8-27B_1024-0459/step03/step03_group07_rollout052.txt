def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-13
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)

        h = np.clip(h, 0.0, 1.0)
    return h

    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 0.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

    shifted = np.clip(beta * (corr - m), -500.0, 0.0)
    w = np.exp(shifted)
    w = w / np.sum(w)

        inits.append(make_antiperiodic_init(n, rng))
        
        time_per_init = max(15.0, time_per_res / len(inits) - 5)
        
        for h0 in inits:
            if time.time() - t0 > remaining_limit - 20:
                break
            
            h0 = project_feasible(h0, n)
            c5_0 = compute_c5(h0)
            
            h1, c5_1 = coord_descent_fast(h0, n, sweeps=4, time_limit=time_per_init * 0.4, t0=t0)
            h2, c5_2 = lbfgs_with_smooth_max(h1, n,
                                            betas=[100.0, 500.0, 2000.0, 10000.0],
                                            times=[0.25, 0.25, 0.25, 0.25],
                                            t0=t0, total_limit=time.time() + time_per_init * 0.5)
            h3, c5_3 = topk_lbfgs(h2, n, k=5, max_iter=200, t0=t0, total_limit=time.time() + time_per_init * 0.5)
            h4, c5_4 = coord_descent_fast(h3, n, sweeps=6, time_limit=time_per_init * 0.7, t0=t0)
            
            candidates.append((h4, c5_4, n))
            if c5_3 < c5_4:
                candidates.append((h3, c5_3, n))
            if c5_2 < c5_4:
                candidates.append((h2, c5_2, n))
    
    if candidates:
        candidates.sort(key=lambda x: x[1])
        return candidates[0][0], candidates[0][1], candidates[0][2]
    return h_init, compute_c5(h_init), n_init

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 160
    
    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn > 10:
            best_h = ph.copy()
            best_c5 = compute_c5(ph)
            best_n = pn
    except Exception:
        best_h = np.ones(160) * 0.5
        best_n = 160
        best_c5 = compute_c5(best_h)
    
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 60.0
    
    if remaining < 30:
        best_h = project_feasible(best_h, best_n)
        return (best_h, compute_c5(best_h), best_n)
    
    # Phase 1: Multi-resolution exploration
    phase1_limit = t0 + budget_s * 0.35
    phase1_time = min(remaining * 0.4, phase1_limit - t0)
    if phase1_time > 30:
        h_p1, c5_p1, n_p1 = multi_res_optimize(best_h, best_n, phase1_time, t0, rng, t0 + phase1_time)
        if c5_p1 < best_c5:
            best_c5 = c5_p1
            best_h = h_p1.copy()
            best_n = n_p1
    
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 50.0
    
    # Phase 2: Focused L-BFGS refinement at best resolution
    phase2_limit = t0 + budget_s * 0.70
    if remaining > 50 and (phase2_limit - t0) > 30:
        n_work = best_n
        h_work = best_h.copy()
        
        # Try a few different beta schedules
        for beta_seq in [[200, 1000, 5000, 20000, 50000],
                         [500, 2000, 10000, 50000, 100000]]:
            if time.time() > phase2_limit - 10:
                break
            sub_time = (phase2_limit - time.time()) / (len(beta_seq) + 2)
            times = [sub_time] * len(beta_seq)
            h_work, c5_work = lbfgs_with_smooth_max(h_work, n_work, beta_seq, times, t0, phase2_limit)
            if c5_work < best_c5:
                best_c5 = c5_work
                best_h = h_work.copy()
                best_n = n_work
    
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 40.0
    
    # Phase 3: Top-k focused optimization
    phase3_limit = t0 + budget_s * 0.85
    if remaining > 30 and (phase3_limit - t0) > 20:
        h_work = best_h.copy()
        for k in [3, 5, 10, 20]:
            if time.time() > phase3_limit - 10:
                break
            h_work, c5_work = topk_lbfgs(h_work, best_n, k=k,
                                         max_iter=300, t0=t0,
                                         total_limit=phase3_limit)
            if c5_work < best_c5:
                best_c5 = c5_work
                best_h = h_work.copy()
    
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 10.0
    
    # Phase 4: Final coordinate descent polish
    if remaining > 15:
        h_work, c5_work = coord_descent_fast(best_h, best_n, sweeps=15,
                                             time_limit=time.time() + remaining * 0.6 - t0, t0=t0)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()
    
    remaining = budget_s - (time.time() - t0) - 5.0
    
    # Phase 5: One more L-BFGS with very high beta
    if remaining > 20:
        h_work, c5_work = lbfgs_with_smooth_max(best_h, best_n,
                                               betas=[50000.0, 100000.0],
                                               times=[remaining * 0.4, remaining * 0.4],
                                               t0=t0, total_limit=time.time() + remaining * 0.8)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()
    
    remaining = budget_s - (time.time() - t0) - 3.0
    
    # Final coordinate descent
    if remaining > 10:
        h_work, c5_work = coord_descent_fast(best_h, best_n, sweeps=10,
                                             time_limit=time.time() + remaining * 0.8 - t0, t0=t0)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()
    
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
