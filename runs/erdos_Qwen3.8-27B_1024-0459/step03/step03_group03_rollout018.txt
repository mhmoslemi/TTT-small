        npar = len(hpar)
    except Exception:
        npar = 160
        hpar = np.zeros(npar, dtype=np.float64)
        hpar[:npar // 2] = 1.0

    bh = _proj(hpar, npar)
    bc5 = _c5(bh)
    bn = npar
    print("Init n={} C5={:.8f}".format(bn, bc5))

    # Main optimization at parent resolution
    h, c5 = _optimize_single(bh, npar, rng)
    if c5 < bc5:
        bh, bc5 = h, c5
        print("Opt n={} C5={:.8f}".format(npar, c5))

    # Random restart perturbations at same resolution
    if _tl() > 60:
        for nflip in [4, 6, 8, 3, 5]:
            if _tl() < 50:
                break
            hp = _perturb_restart(bh, bn, rng, nflip)
            h2, c2 = _optimize_single(hp, bn, rng)
            if c2 < bc5:
                bh, bc5 = h2, c2
                print("Perturb({}) n={} C5={:.8f}".format(nflip, bn, c2))

    # Try different resolutions
    alts = [npar + 2, npar - 2, npar + 4, npar - 4, npar + 6, npar - 6, npar + 8]
    for nn in alts:
        if _tl() < 50:
            break
        if nn < 40 or nn % 2 != 0:
            continue
        hi = _proj(_interp(bh, bn, nn), nn)
        h2, c2 = _optimize_single(hi, nn, rng)
        if c2 < bc5:
            bh, bc5, bn = h2, c2, nn
            print("Res n={} C5={:.8f}".format(nn, c2))

    # Final polish
    if _tl() > 25:
        h, c5 = _gd_loop(bh, bn, rng, 300)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _pair_local(bh, bn, rng, 200)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _binary_search(bh, bn, rng, 80)
        if c5 < bc5:
            bh, bc5 = h, c5

    if _tl() > 15:
        h2 = _slsqp(bh, bn, 0.0005, 500)
        c2 = _c5(h2)
        if c2 < bc5:
            bh, bc5 = h2, c2
        h, c5 = _gd_loop(bh, bn, rng, 200)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _pair_local(bh, bn, rng, 150)
        if c5 < bc5:
            bh, bc5 = h, c5

    if _tl() > 10:
        h, c5 = _gd_loop(bh, bn, rng, 150)
        if c5 < bc5:
            bh, bc5 = h, c5
        h, c5 = _pair_local(bh, bn, rng, 100)
        if c5 < bc5:
            bh, bc5 = h, c5

    bh = _proj(bh, bn)
    fc5 = _c5(bh)
    try:
        ec = evaluate_erdos_solution(bh)
        if ec < fc5:
            fc5 = ec
    except Exception:
        pass
    print("FINAL n={} C5={:.8f}".format(bn, fc5))
    return (bh, fc5, bn)
