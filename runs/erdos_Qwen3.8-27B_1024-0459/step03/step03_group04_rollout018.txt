Actually, a much better local search:
1. Compute the worst shift k*
2. For this shift, the correlation is sum_i h[i](1-h[i+k*])
3. To decrease this, we want to either:
   - Decrease h[i] where h[i] is high and h[i+k*] is also high (reduces product)
   - Increase h[i] where h[i] is low and h[i+k*] is low (reduces product via 1-h)
4. The best single change: find i that maximizes the reduction

Let me think about this more carefully. If I change h[i] by δ (and compensate by changing h[j] by -δ to maintain sum):

  - If i is in the "h" position for shift k*: change ≈ δ*(1-h[i+k*])
  - If i is in the "(1-h)" position: change ≈ -δ*h[i-k*]
  - Similarly for j

  - For the worst shift k*, the affected pairs are those where moving mass helps reduce corr[k*]
  - Heuristically: decrease h at positions where h is high in the "overlapping" region, increase where h is low

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr


def project_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                mask2 = h < 1.0
            else:
                mask2 = h > 0.0
            cnt2 = np.sum(mask2)
            if cnt2 > 0:
                h[mask2] += diff / cnt2
            else:
                break
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_h(src, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xq = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xq, xp, src)


def time_left(t0, budget):
    return max(0.0, budget - (time.time() - t0))


def slsqp_smoothed_max(h, n, beta, active_idx):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr[active_idx]
    m = np.max(vals)
    if beta < 1e-6:
        return m * dx
    exps = np.exp(beta * (vals - m))
    s = np.sum(exps)
    if s < 1e-300:
        return m * dx
    return (m + np.log(s) / beta) * dx


def make_sum_con(n):
    def con(x):
        return np.sum(x) - n / 2.0
    return con


def grad_smoothed_max(h, n, beta, active_idx=None):
    corr = np.correlate(h, 1.0 - h, mode='full')
    nc = len(corr)
    if active_idx is not None:
        vals = corr[active_idx]
        m = np.max(vals)
        exps = np.exp(beta * (vals - m))
        probs_act = exps / np.sum(exps)
        probs = np.zeros(nc)
        probs[active_idx] = probs_act
    else:
        m = np.max(corr)
        exps = np.exp(beta * (corr - m))
        probs = exps / np.sum(exps)

    grad = np.zeros(n)
    for m_idx in range(nc):
        w = probs[m_idx]
        if w < 1e-12:
            continue
        s = m_idx - (n - 1)
        if s >= 0:
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n - s]
        else:
            p = -s
            grad[p:] += w * (1.0 - h[:n - p])
            grad[:n - p] -= w * h[p:]
    return grad
