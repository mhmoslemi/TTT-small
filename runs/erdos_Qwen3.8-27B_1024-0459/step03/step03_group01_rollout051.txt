def simulated_annealing_binary(h, n, time_limit, rng):
    """SA on binary sequences."""
    h = (h > 0.5).astype(float).copy()
    # Ensure feasibility
    n_ones = int(np.sum(h))
    if n_ones > n//2:
        ones = np.where(h==1)[0]
        h[rng.choice(ones, n_ones - n//2, replace=False)] = 0
    elif n_ones < n//2:
        zeros = np.where(h==0)[0]
        h[rng.choice(zeros, n//2 - n_ones, replace=False)] = 1
    
    corr = np.correlate(h, 1-h, mode='full')
    cur_max = np.max(corr)
    best = h.copy()
    best_max = cur_max
    dx = 2.0/n
    
    T_init = 5.0  # in correlation units
    T_min = 0.01
    cooling = 0.9995
    T = T_init
    
    start = time.time()
    while time.time() - start < time_limit:
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0; h[j] = 1
        new_corr = np.correlate(h, 1-h, mode='full')
        new_max = np.max(new_corr)
        
        delta = new_max - cur_max
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, T_min)):
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best = h.copy()
        else:
            h[i] = 1; h[j] = 0
        
        T = max(T * cooling, T_min)
    
    return best, best_max * dx

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c == 1 else 0.0
    return h


def random_binary(n, rng):

def ensure_binary_feasible(h, n, rng):

    elif n_ones < n // 2:
        zeros_idx = np.where(h == 0)[0]
        h[rng.choice(zeros_idx, n // 2 - n_ones, replace=False)] = 1.0
    return h

def smooth_obj(x):
    n = _N_GLOBAL
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = float(np.max(corr))
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g


def smooth_grad(x):
    n = _N_GLOBAL
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = float(np.max(corr))
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    # Only compute gradient for significant weights
    thresh = 1e-10
    sig_idx = np.where(w > thresh)[0]
    for k in sig_idx:
        g += w[k] * corr_grad_per_shift(x, n, int(k))
    return g * dx_val


def sum_con(x):
    return np.sum(x) - _TGT

    return np.ones(_N_GLOBAL)


def slsqp_smooth(h, n, beta, maxiter=500):

    _N_GLOBAL = n
    _TGT = n / 2.0
    _BETA = beta

    except Exception:
        return h


def lbfgs_smooth(h, n, beta, maxiter=2000):
    global _N_GLOBAL, _BETA, _TGT
    _N_GLOBAL = n
    _TGT = n / 2.0
    _BETA = beta

    def obj_with_proj(x):

        return smooth_obj(h_p)

    try:

def active_set_polish(h, n, max_iters=500):
    """Equalize top correlations via projected steepest descent."""
    h = h.copy()
    dx_val = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:3]
        c_top = [corr[int(t)] for t in top_idx]
        spread = max(c_top) - min(c_top)
        if spread < 1e-12:
            break

        # Gradient to reduce spread: pull top down, push bottom up
        weights = np.zeros(2 * n - 1)
        for t in top_idx:
            ti = int(t)
            if ti == int(top_idx[0]):
                weights[ti] = -1.0
            elif ti == int(top_idx[-1]) and len(top_idx) > 1:
                weights[ti] = 0.5

        g = np.zeros(n)
        for k in range(len(weights)):
            if abs(weights[k]) > 1e-12:
                g += weights[k] * corr_grad_per_shift(h, n, k)

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g /= gnorm

        improved = False
        for step in [0.02, 0.01, 0.005, 0.03, 0.001, 0.05, 0.015]:
            h_trial = project_feasible(h - step * g, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-13:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            break
    return h, best_c5
