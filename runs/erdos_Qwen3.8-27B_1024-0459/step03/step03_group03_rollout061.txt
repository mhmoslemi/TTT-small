def _c5(h):
    n = len(h)

def _proj(h, n):

def _ggrad(h, kidx, n):

        if s == 0:

        else:
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
    else:
        p = -s
        if p > 0:
            g[:n - p] += 1.0 - h[p:n]
            g[p:] -= h[:n - p]
    return g

def _slsqp_con(x):
    return x.sum() - 0.5 * _slsqp_n


def _slsqp_obj(x):
    n = _slsqp_n
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = float(np.max(c))
    b = _slsqp_beta
    sh = c - mx
    return (mx + np.log(np.exp(b * sh).sum()) / b) * (2.0 / n)


def _slsqp(h, n, beta, maxiter):
    global _slsqp_n, _slsqp_beta
    _slsqp_n = n
    _slsqp_beta = beta
    try:
        r = minimize(_slsqp_obj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _slsqp_con}],
                     options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return _proj(np.asarray(r.x, dtype=np.float64), n)
    except Exception:
        return _proj(h.copy(), n)
