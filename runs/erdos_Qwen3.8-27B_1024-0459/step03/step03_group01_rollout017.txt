                g[ao:ao + np_] -= h[:np_]
        d = g_b - g_t
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            d = -g_t
            dn = np.linalg.norm(d)
            if dn < 1e-14:
                break
        d /= dn
        improved = False
        for step in [0.02, 0.05, 0.01, 0.08, 0.005, 0.03, 0.1, 0.015]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def fractional_optimize(h, n, beta=5000, maxiter=800):
    """Fix binary values, optimize only fractional ones."""
    h = h.copy()
    frac_mask = (h > 0.02) & (h < 0.98)
    n_frac = np.count_nonzero(frac_mask)
    if n_frac < 2:
        return h, compute_c5(h)
    
    # Use full SLSQP but with very tight bounds on non-fractional
    h_orig = h.copy()
    bounds = []
    for i in range(n):
        if frac_mask[i]:
            bounds.append((0.0, 1.0))
        elif h_orig[i] <= 0.02:
            bounds.append((0.0, 0.08))
        else:
            bounds.append((0.92, 1.0))
    
    result = slsqp_smooth(h, n, beta, maxiter)
    c5 = compute_c5(result)
    
    # Project back: snap non-fractional to 0 or 1
    result2 = h.copy()
    for i in range(n):
        if h_orig[i] <= 0.02:
            result2[i] = 0.0
        elif h_orig[i] >= 0.98:
            result2[i] = 1.0
        else:
            result2[i] = result[i]
    result2 = project_feasible(result2, n)
    c5_2 = compute_c5(result2)
    
    if c5_2 < c5:
        return result2, c5_2
    return result, c5


def binary_sa(h_bin, n, time_limit=60, rng=None, T0=2.0, T1=0.01, tabu_size=200):
    """Simulated annealing on binary vectors with tabu."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            rem = rng.choice(ones_idx, n_ones - n_half, replace=False)
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            add = rng.choice(zeros_idx, n_half - n_ones, replace=False)
            h[add] = 1
    dx_v = 2.0 / n
    corr = compute_corr(h)
    cur_c5 = float(np.max(corr)) * dx_v
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    tabu = {}  # (i,j) -> step when added
    n_steps = 0
    n_gen = int(time_limit * 500)  # approximate number of iterations
    
    for gen in range(n_gen):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break
        T = T0 * (T1 / T0) ** (gen / max(n_gen, 1))
        if T < 0.001:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try multiple candidate moves
        best_move = None
        best_delta = 0
        for _ in range(8):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            key = (min(i, j), max(i, j))
            if key in tabu and n_steps - tabu[key] < tabu_size:
                continue
            # Compute delta
            h[i] = 0
            h[j] = 1
            nc = compute_corr(h)
            nc5 = float(np.max(nc)) * dx_v
            delta = nc5 - cur_c5
            h[i] = 1
            h[j] = 0
            if best_move is None or delta < best_delta:
                best_delta = delta
                best_move = (i, j)
        
        if best_move is None:
            continue
        
        i, j = best_move
        accept = False
        if best_delta <= 0:
            accept = True
        elif rng.rand() < np.exp(-best_delta / max(T, 1e-10)):
            accept = True
        
        if accept:
            h[i] = 0
            h[j] = 1
            tabu[(min(i, j), max(i, j))] = n_steps
            cur_c5 += best_delta
            n_steps += 1
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    
    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 320
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Progressive SLSQP on parent resolution
    for beta in [50, 100, 300, 800, 2000, 5000, 10000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n
            h = h_new.copy()

    # Phase 2: Top-shifts equalization
    if remaining() > 10:
        h_eq, c5_eq = top_shifts_focus(h, n, top_k=5, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 3: Fractional optimization (fix binary, optimize fractional)
    if remaining() > 15:
        for beta in [3000, 8000, 15000]:
            if remaining() < 8:
                break
            h_frac, c5_frac = fractional_optimize(h, n, beta=beta, maxiter=600)
            if c5_frac < best_c5:
                best_c5 = c5_frac
                best_h = h_frac.copy()
                h = h_frac.copy()

    # Phase 4: Try other resolutions
    n_candidates = sorted(set([
        best_n, best_n - 4, best_n + 4, best_n + 8, best_n + 16,
        int(best_n * 1.2), int(best_n * 1.4), int(best_n * 1.6),
        int(best_n * 1.8), int(best_n * 2.0),
    ]))
    n_candidates = [nn for nn in n_candidates if 128 <= nn <= 640 and nn % 2 == 0]
    # Remove already done
    n_candidates = [nn for nn in n_candidates if nn != best_n]
    n_candidates.sort(key=lambda x: abs(x - best_n))

    for n_t in n_candidates[:6]:
        if remaining() < 20:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)
        
        h_opt = h_feas
        for beta in [200, 1000, 5000]:
            if remaining() < 8:
                break
            h_opt = slsqp_smooth(h_opt, n_t, beta, maxiter=400)
        c5_opt = compute_c5(h_opt)
        
        h_opt, c5_opt = top_shifts_focus(h_opt, n_t, top_k=4, max_iters=200)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 5: Binary simulated annealing on best solution
    if remaining() > 30:
        h_bin_init = (best_h > 0.5).astype(float)
        h_bin_init = project_feasible(h_bin_init, best_n)
        # Snap to binary
        n_ones = int(np.round(np.sum(h_bin_init)))
        if n_ones != best_n // 2:
            sort_idx = np.argsort(-h_bin_init)
            h_bin = np.zeros(best_n)
            h_bin[sort_idx[:best_n // 2]] = 1.0
        else:
            h_bin = (h_bin_init > 0.5).astype(float)
        
        sa_time = min(120, max(10, int(remaining() * 0.4)))
        h_b, c5_b = binary_sa(h_bin, best_n, time_limit=sa_time, rng=rng)
        
        # Also try SA starting from Thue-Morse
        h_tm = np.zeros(best_n)
        for i in range(best_n):
            c = bin(i).count('1') % 2
            h_tm[i] = 1.0 if c == 1 else 0.0
        n_ones_tm = int(np.sum(h_tm))
        if n_ones_tm != best_n // 2:
            if n_ones_tm > best_n // 2:
                ones_idx = np.where(h_tm == 1)[0]
                h_tm[ones_idx[rng.choice(len(ones_idx), n_ones_tm - best_n // 2, replace=False)]] = 0
            else:
                zeros_idx = np.where(h_tm == 0)[0]
                h_tm[zeros_idx[rng.choice(len(zeros_idx), best_n // 2 - n_ones_tm, replace=False)]] = 1
        
        sa_time2 = min(60, max(5, int(remaining() * 0.2)))
        if sa_time2 > 3:
            h_tm_b, c5_tm = binary_sa(h_tm, best_n, time_limit=sa_time2, rng=np.random.RandomState(seed + 1))
            # Convert binary to continuous for comparison
            h_tm_c = h_tm_b.copy()
            for beta in [5000, 10000]:
                if remaining() < 5:
                    break
                h_tm_c = slsqp_smooth(h_tm_c, best_n, beta, maxiter=300)
            c5_tm_c = compute_c5(h_tm_c)
            if c5_tm_c < best_c5:
                best_c5 = c5_tm_c
                best_h = h_tm_c.copy()
                best_n = best_n
        
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = best_n

    # Phase 6: Final intensive refinement
    n = best_n
    h = best_h.copy()
    
    for beta in [5000, 10000, 20000, 40000]:
        if remaining() < 5:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final top-shifts equalize
    if remaining() > 8:
        h_eq, c5_eq = top_shifts_focus(h, n, top_k=6, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Final fractional optimization
    if remaining() > 8:
        h_f, c5_f = fractional_optimize(best_h, n, beta=20000, maxiter=500)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final polish with very high beta
    if remaining() > 5:
        h_new = slsqp_smooth(best_h, n, 50000, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5:
        # Don't update if projection made it worse; use pre-projection value
        pass
    else:
        best_c5 = final_c5

    # Try evaluate_erdos_solution for validation
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
