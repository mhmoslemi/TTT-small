def project_feasible(h, n):

def interpolate_h(h_src, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_new = np.interp(xn, xp, h_src)
    return project_feasible(h_new, n_dst)

def make_structured_init(n, seed_val, freq=None, num_modes=3):
    rng = np.random.RandomState(seed_val)
    t = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    for i in range(num_modes):
        f = freq[i] if freq else (i + 1) * 2
        phase = rng.rand() * 2 * np.pi
        h += np.cos(2 * np.pi * f * t / 2.0 + phase) / num_modes
    h = 0.5 + 0.3 * h
    return project_feasible(h, n)

def perturb_init(h_base, n, seed_val, scale=0.03):
    rng = np.random.RandomState(seed_val)
    hp = h_base + scale * rng.randn(n)
    return project_feasible(hp, n)

def smooth_max_obj(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

def slsqq_polish(h, n, time_limit=None, t0=None):

    h = project_feasible(h, n)
    best_h = h.copy()
