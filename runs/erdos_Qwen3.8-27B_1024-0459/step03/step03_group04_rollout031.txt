    _, corr_cur = compute_c5_corr(h)
    active = find_active_lags(corr_cur, alpha=0.6)
    
    prev_obj = None
    stall_count = 0
    
    for outer in range(200):
        if time_left(t0, budget) < 30:
            break
        
        # Update active set every 10 iterations
        if outer % 10 == 0:
            _, corr_cur = compute_c5_corr(h)
            active = find_active_lags(corr_cur, alpha=0.5)
        
        # Compute gradient
        # Use beta=50 for the soft max
        beta_sm = 50.0
        grad = grad_soft(h, active, beta_sm, 0.0)  # No penalty, project instead
        
        # Adam update
        t_adam += 1
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad ** 2
        m_hat = m_v / (1 - beta1 ** t_adam)
        v_hat = v_v / (1 - beta2 ** t_adam)
        
        # Adaptive learning rate
        lr = lr_base / (1.0 + 0.01 * outer)
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)
        
        h_new = h + step
        h_new = project_h(h_new, n)
        
        c5_new, _ = compute_c5_corr(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            stall_count = 0
        else:
            h = project_h(h - step * 0.5, n)  # Half step
            c5_half, _ = compute_c5_corr(h)
            if c5_half < best_c5:
                best_c5 = c5_half
                best_h = h.copy()
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 20:
                    break
        
        if outer % 20 == 0:
            print("  PGD outer {}: C5={:.8f}, n_active={}".format(outer, best_c5, len(active)))
    
    # Phase 3: SLSQP refinement with soft max
    if time_left(t0, budget) > 60:
        n = best_n
        h = best_h.copy()
        bounds = [(0.0, 1.0)] * n
        
        for beta_sl in [30, 80, 200, 500]:
            if time_left(t0, budget) < 15:
                break
            _, corr_sl = compute_c5_corr(h)
            active_sl = find_active_lags(corr_sl, alpha=0.6)
            
            def obj_slsqp(x):
                dx_s = 2.0 / n
                corr_s = np.correlate(x, 1.0 - x, mode='full')
                vals = corr_s[active_sl]
                m_s = np.max(vals)
                exps_s = np.exp(beta_sl * (vals - m_s))
                return (m_s + np.log(np.sum(exps_s)) / beta_sl) * dx_s
            
            def con_sum(x):
                return np.sum(x) - n / 2.0
            
            try:
                res = minimize(obj_slsqp, h, method='SLSQP',
                               bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': con_sum}],
                               options={'maxiter': 500, 'ftol': 1e-18, 'iprint': 0})
                h_new = project_h(res.x, n)
                c5_new, _ = compute_c5_corr(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                    print("  SLSQP beta={}: C5={:.8f}".format(beta_sl, best_c5))
            except Exception:
                pass
    
    # Phase 4: Fine greedy coordinate descent
    if time_left(t0, budget) > 30:
        n = best_n
        h = best_h.copy()
        dx = 2.0 / n
        target_sum = n / 2.0
        
        # Precompute which lags each coordinate affects
        # Coordinate j affects lags where shift = a-(n-1) and j is in range
        # For efficiency, just recompute full correlation
        
        for cd_iter in range(50000):
            if time_left(t0, budget) < 5:
                break
            i = rng.randint(n)
            # Try small perturbations
            for delta in [0.001, 0.003, 0.005, 0.01, 0.02, -0.001, -0.003, -0.005, -0.01, -0.02]:
                if h[i] + delta < 0 or h[i] + delta > 1:
                    continue
                h[i] += delta
                # Compensate: distribute -delta over other coordinates
                h_new = project_h(h, n)
                c5_cd, _ = compute_c5_corr(h_new)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_new.copy()
                    h = h_new.copy()
                    break
                else:
                    h[i] -= delta
    
    # Phase 5: L-BFGS-B with penalty (multiple betas)
    if time_left(t0, budget) > 40:
        n = best_n
        h = best_h.copy()
        bounds = [(0.0, 1.0)] * n
        dx = 2.0 / n
        
        for beta_lb in [50, 150, 400]:
            if time_left(t0, budget) < 10:
                break
            _, corr_lb = compute_c5_corr(h)
            active_lb = find_active_lags(corr_lb, alpha=0.7)
            
            def obj_lb(x):
                corr_l = np.correlate(x, 1.0 - x, mode='full')
                vals = corr_l[active_lb]
                m_l = np.max(vals)
                exps_l = np.exp(beta_lb * (vals - m_l))
                sme = m_l + np.log(np.sum(exps_l)) / beta_lb
                pen = 1000.0 * (np.sum(x) - n / 2.0) ** 2
                return sme * dx + pen
            
            try:
                res = minimize(obj_lb, h, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 1000, 'ftol': 1e-18,
                                         'gtol': 1e-14, 'maxls': 200})
                h_new = project_h(res.x, n)
                c5_new, _ = compute_c5_corr(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                    print("  L-BFGS-B beta={}: C5={:.8f}".format(beta_lb, best_c5))
            except Exception:
                pass
    
    # Phase 6: Two-point exchange (move mass from one position to another)
    if time_left(t0, budget) > 20:
        n = best_n
        h = best_h.copy()
        deltas = [0.005, 0.01, 0.02, 0.05]
        for ex_iter in range(20000):
            if time_left(t0, budget) < 3:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in deltas:
                if h[i] >= d and h[j] <= 1.0 - d:
                    h_tr = h.copy()
                    h_tr[i] -= d
                    h_tr[j] += d
                    c5_tr, _ = compute_c5_corr(h_tr)
                    if c5_tr < best_c5:
                        best_c5 = c5_tr
                        best_h = h_tr.copy()
                        h = h_tr.copy()
                        break
                if h[j] >= d and h[i] <= 1.0 - d:
                    h_tr = h.copy()
