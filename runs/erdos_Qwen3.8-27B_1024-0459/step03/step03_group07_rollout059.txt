def subgradient_optimize(h_init, n, iters=2000, time_limit=None, t0=None):
    h = project_feasible(h_init, n)
    best = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    lr = 0.01
    momentum = np.zeros(n)
    mu = 0.9
    stagn = 0
    
    for it in range(iters):
        if time_limit and t0 and time.time() - t0 > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        
        # Subgradient: average gradient of top-k correlations
        # For top-k, use a soft weight
        k = min(5, 2*n-1)
        top_idx = np.argsort(corr)[-k:]
        
        grad = np.zeros(n)
        for lidx in top_idx:
            L = lidx - (n-1)
            w = 1.0 / k
            if L == 0:
                # d/dh[j] of h[j]*(1-h[j]) = 1 - 2*h[j]
                grad += w * (1.0 - 2.0 * h)
            elif L > 0:
                # corr[L] = sum_{i=0}^{n-1-L} h[i] * (1-h[i+L])
                # d/dh[j]: (1-h[j+L]) if j <= n-1-L, plus -h[j-L] if j >= L
                mask1 = np.zeros(n)
                mask1[:n-L] = 1.0 - h[L:]
                mask2 = np.zeros(n)
                mask2[L:] = -h[:n-L]
                grad += w * (mask1 + mask2)
            else:
                s = -L
                mask1 = np.zeros(n)
                mask1[s:] = 1.0 - h[:n-s]
                mask2 = np.zeros(n)
                mask2[:n-s] = -h[s:]
                grad += w * (mask1 + mask2)
        
        # Momentum update
        momentum = mu * momentum - lr * grad
        h_new = h + momentum
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best = h_new.copy()
            h = h_new
            stagn = 0
            lr = min(lr * 1.01, 0.05)
        else:
            h = h_new  # still move (momentum)
            stagn += 1
            if stagn > 50:
                # Restart with smaller lr
                lr = max(0.001, lr * 0.5)
                momentum = np.zeros(n)
                stagn = 0
                if stagn > 200:
                    break
    
    return best, best_c5

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1][:max(1, int(np.ceil(abs(diff))))]
            h[idx] += diff / len(idx)
        else:
            idx = np.argsort(h)[::-1][:max(1, int(np.ceil(abs(diff))))]
            h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -4.0, 0.0)
    return (m + np.log(1.0 + np.sum(np.exp(sh))) / beta) * (2.0 / n)

def topk_mean_obj(x, n, k):
    corr = np.correlate(x, 1.0 - x, mode='full')
    k = min(k, len(corr))
    top = np.sort(corr)[-k:]
    return np.mean(top) * (2.0 / n)

    best = h.copy()

            break

                break
            old = h[j]

                    quad_base = corr[fi] - old * (1.0 - old)
                    continue
                coeff = 0.0

                    if j <= n - 1 - L:
                        coeff += (1.0 - h[j + L])
                    if j >= L:
                        coeff += (-h[j - L])
                else:
                    s = -L
                    if j >= s:
                        coeff += (1.0 - h[j - s])
                    if j <= n - 1 - s:
                        coeff += (-h[j + s])

                    base = corr[fi] - coeff * old
                    a_list.append((base, coeff))
                else:

def slsqp_optimize(h_init, n, time_limit=None, t0=None):

    best = h.copy()
    best_c5 = compute_c5(h)

        return np.sum(x) - n / 2.0

            break

            return smooth_max_obj(x, n, beta)

                           options={'maxiter': maxit, 'ftol': 1e-18, 'iprint': 0})

            c5 = compute_c5(h)

def subgradient_step(h, n, lr, topk=5):
    corr = np.correlate(h, 1.0 - h, mode='full')
    k = min(topk, 2 * n - 1)
    top_idx = np.argsort(corr)[-k:]
    grad = np.zeros(n)
    for lidx in top_idx:
        L = lidx - (n - 1)
        w = 1.0 / k
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            g = np.zeros(n)
            g[:n - L] += (1.0 - h[L:])
            if L < n:
                g[L:] += (-h[:n - L])
            grad += w * g
        else:
            s = -L
            g = np.zeros(n)
            g[s:] += (1.0 - h[:n - s])
            if s < n:
                g[:n - s] += (-h[s:])
            grad += w * g
    return h - lr * grad

def momentum_subgrad(h_init, n, iters=3000, time_limit=None, t0=None):

    best = h.copy()
    best_c5 = compute_c5(h)

        h_new = subgradient_step(h, n, lr, topk=5)
        mom = mu * mom - lr * (h - h_new)
        h_candidate = project_feasible(h + mom, n)

    return best, best_c5
