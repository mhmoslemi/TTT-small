def coord_descent(h, n, max_passes=200, step_size=0.05):

        if remaining() < 5:
            break

        for i in range(n):
            if remaining() < 5:
                break
            orig = h[i]
            for delta in [step_size, -step_size, step_size*2, -step_size*2, step_size*0.5, -step_size*0.5]:
                h[i] = orig + delta
                h[i] = np.clip(h[i], 0, 1)
                # Fix sum constraint approximately
                h = project_feasible(h, n)
                c = compute_c5(h)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    improved = True
                    break
                h[i] = orig
            # Restore
            if not (np.abs(h[i] - orig) < 1e-15):
                h[i] = orig
                h = project_feasible(h, n)
        if not improved:
            break
    return h, best_c5

def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-14:
            g += w[k] * corr_grad_per_shift(x, n, k)
    return g * dx_val

            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_t

    # Phase 2: Two-phase continuous refinement at best n and neighbors
    n_candidates = sorted(set([best_n - 20, best_n - 10, best_n, best_n + 10, best_n + 20,
                               300, 320, 340, 360]))
    n_candidates = [n for n in n_candidates if n % 2 == 0 and 100 <= n <= 600]

    for n_t in n_candidates:
        if remaining() < 30:
            break
        if n_t == best_n:
            h_init = best_h.copy()
        else:
            h_init = interp_to(best_h, best_n, n_t)
            h_init = project_feasible(h_init, n_t)

        t_limit = min(80, max(20, remaining() / (len(n_candidates) + 1) * 1.5))
        h_ref, c5_ref = two_phase_refine(h_init, n_t, time_limit=t_limit)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            best_n = n_t

    # Phase 3: Intensive refinement on best solution
    n = best_n
    h = best_h.copy()

    # SLSQP with very high beta
    for beta in [10000, 30000, 100000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Top-k gradient descent
    if remaining() > 10:
        h_td, c5_td = topk_gradient_descent(h, n, max_iters=500, top_k=5)
        if c5_td < best_c5:
            best_c5 = c5_td
            best_h = h_td.copy()
            h = h_td.copy()

    # Phase 4: Binary SA from current best (final push)
    if remaining() > 30:
        h_bin = make_binary(best_h, n, rng)
        t_sa = min(80, max(15, remaining() * 0.4))
        for sa_seed in [seed + 5000, seed + 6000, seed + 7000]:
            if remaining() < 15:
                break
            h_sa, c5_sa = sa_binary(h_bin, n, time_limit=max(5, t_sa / 3), seed=sa_seed)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n

    # Phase 5: Final polish
    if remaining() > 10:
        n = best_n
        for beta in [50000, 150000]:
            if remaining() < 5:
                break
            h_new = slsqp_smooth(best_h, n, beta, maxiter=500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    if remaining() > 5:
        h_td, c5_td = topk_gradient_descent(best_h, best_n, max_iters=200, top_k=3)
        if c5_td < best_c5:
            best_c5 = c5_td
            best_h = h_td.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
