    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def project_binary(h, n, rng):
    h = np.array(h, dtype=float).copy()
    h = (h > 0.5).astype(float)
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        ones_idx = np.where(h == 1)[0]
        remove = rng.choice(ones_idx, n_ones - n // 2, replace=False)
        h[remove] = 0.0
    elif n_ones < n // 2:
        zeros_idx = np.where(h == 0)[0]
        add = rng.choice(zeros_idx, n // 2 - n_ones, replace=False)
        h[add] = 1.0
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') & 1
        h[i] = float(c)
    return h


def rudin_shapiro(n):
    a = np.zeros(n + 1)
    p = np.zeros(n + 1)
    a[0] = 1
    p[0] = 1
    h = np.zeros(n)
    for i in range(n):
        if i == 0:
            h[i] = 1.0
        else:
            pass
    # Standard RU and VS sequences
    def _ru_vs(n):
        ru = np.zeros(n)
        vs = np.zeros(n)
        ru[0] = 1
        vs[0] = 0
        for i in range(1, n):
            if i % 2 == 0:
                ru[i] = (ru[i//2] + vs[i//2]) % 2
                vs[i] = (ru[i//2] - vs[i//2]) % 2
            else:
                j = (i - 1) // 2
                ru[i] = (ru[j] - vs[j]) % 2
                vs[i] = (ru[j] + vs[j]) % 2
        return ru.astype(float), vs.astype(float)
    ru, vs = _ru_vs(n)
    return ru


def sa_local_search(h, n, rng, time_limit=60.0, T_init=0.15, T_min=0.001, alpha=0.9995):
    """Simulated annealing on binary landscape."""
    h = project_binary(h, n, rng).copy()
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_c5 = cur_max * 2.0 / n
    start = time.time()
    T = T_init
    stall_count = 0
    for it in range(2000000):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break
        T *= alpha
        if T < T_min:
            T = T_min
        k_max = int(np.argmax(corr))
        offset = k_max - (n - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Biased sampling toward worst shift contributors
        improved = False
        for trial in range(60):
            if offset >= 0:
                cnt = n - offset
                ci = np.array([i for i in ones if i + offset < n and h[i + offset] == 0])
                cj = np.array([j for j in zeros if 0 <= j - offset < n and h[j - offset] == 1])
            elif offset < 0:
                ao = -offset
                ci = np.array([i for i in ones if i - ao >= 0 and h[i - ao] == 0])
                cj = np.array([j for j in zeros if j + ao < n and h[j + ao] == 1])
            else:
                ci = ones
                cj = zeros
            if len(ci) > 0:
                i = ci[rng.randint(len(ci))]
            else:
                i = ones[rng.randint(len(ones))]
            if len(cj) > 0:
                j = cj[rng.randint(len(cj))]
            else:
                j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            delta = new_max - cur_max
            if delta < -1e-12 or rng.rand() < np.exp(-delta / max(T, 1e-9)):
                corr = new_corr
                cur_max = new_max
                c5_now = cur_max * 2.0 / n
                if c5_now < best_c5:
                    best_c5 = c5_now
                    best_h = h.copy()
                    stall_count = 0
                else:
                    stall_count += 1
                improved = True
                break
            else:
                h[i] = 1.0
                h[j] = 0.0
        if not improved:
            stall_count += 1
            if stall_count > 200:
                # Perturbation
                for _ in range(3):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i] = 0.0
                    h[j] = 1.0
                h = project_binary(h, n, rng)
                corr = compute_corr(h)
                cur_max = float(np.max(corr))
                stall_count = 0
    return best_h, best_c5


def greedy_local_search(h, n, rng, time_limit=60.0, max_iters=80000):
    """Aggressive greedy local search with worst-shift focus."""
    h = project_binary(h, n, rng).copy()
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_c5 = cur_max * 2.0 / n
    start = time.time()
    stall = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break
        k_max = int(np.argmax(corr))
        offset = k_max - (n - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        found = False
        n_try = 300
        for _ in range(n_try):
            if offset >= 0:
                ci = np.array([i for i in ones if i + offset < n and h[i + offset] == 0])
            elif offset < 0:
                ao = -offset
                ci = np.array([i for i in ones if i - ao >= 0 and h[i - ao] == 0])
            else:
                ci = ones
            if len(ci) > 0 and rng.rand() < 0.75:
                i = ci[rng.randint(len(ci))]
            else:
                i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            if new_max < cur_max - 1e-12:
                corr = new_corr
                cur_max = new_max
                c5_now = cur_max * 2.0 / n
                if c5_now < best_c5:
                    best_c5 = c5_now
                    best_h = h.copy()
                found = True
                break
            else:
                h[i] = 1.0
                h[j] = 0.0
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 100:
                for _ in range(2):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i] = 0.0
                    h[j] = 1.0
                corr = compute_corr(h)
                cur_max = float(np.max(corr))
                stall = 0
    return best_h, best_c5


_N_GLOBAL = 0
_BETA_GLOBAL = 1.0


def _sum_fun(x):
    return np.sum(x) - _N_GLOBAL / 2.0


def _sum_jac(x):
    return np.ones(_N_GLOBAL)


def _corr_grad_one(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g


def _smooth_obj(x):
    n = _N_GLOBAL
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA_GLOBAL
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def _smooth_grad(x):
    n = _N_GLOBAL
    dx_val = 2.0 / n
    beta = _BETA_GLOBAL
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-14:
            g += w[k] * _corr_grad_one(x, n, k)
    return g * dx_val


def slsqp_refine(h, n, beta, maxiter=300):
    global _N_GLOBAL, _BETA_GLOBAL
    _N_GLOBAL = n
    _BETA_GLOBAL = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 320
        parent_h = thue_morse(parent_n)
        parent_h = project_binary(parent_h, parent_n, rng)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Intensive binary local search on parent's n
    n = best_n
    h_bin = project_binary(best_h, n, rng)
    c5_START = remaining()

    # Greedy search (long)
    t_greedy = min(80.0, c5_START * 0.25)
    h_g, c5_g = greedy_local_search(h_bin, n, rng, time_limit=t_greedy)
    if c5_g < best_c5:
        best_c5 = c5_g
        best_h = h_g.copy()
        best_n = n

    # SA search
    if remaining() > 30:
        t_sa = min(60.0, remaining() * 0.2)
        h_s, c5_s = sa_local_search(h_g, n, rng, time_limit=t_sa)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
            best_n = n

    # Phase 2: Try different n values with different starts
    n_candidates = []
    base_ns = [280, 290, 300, 310, 320, 322, 324, 326, 328, 330,
               340, 350, 360, 380, 400, 420, 440, 460]
    for nn in base_ns:
        if nn % 2 == 0 and 80 <= nn <= 500:
            n_candidates.append(nn)
    # Deduplicate and sort by proximity to best_n first
    n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - best_n))
    # Ensure best_n is included
    if best_n not in n_candidates:
        n_candidates.insert(0, best_n)

    for n_t in n_candidates:
        if remaining() < 20:
            break
        if n_t == best_n:
            # Already done above, but try another SA round
            if remaining() > 40:
                h_s, c5_s = sa_local_search(best_h, n_t, rng,
                                            time_limit=min(30, remaining() * 0.15),
                                            T_init=0.2, alpha=0.999)
                if c5_s < best_c5:
                    best_c5 = c5_s
                    best_h = h_s.copy()
                    best_n = n_t
            continue

        # Multiple starts for this n
        starts = []
        # From best solution interpolated
        h_interp = interp_to(best_h, best_n, n_t)
        h_interp_bin = project_binary(h_interp, n_t, rng)
        starts.append(h_interp_bin)
        # Thue-Morse
        h_tm = project_binary(thue_morse(n_t), n_t, rng)
        starts.append(h_tm)
        # Rudin-Shapiro
        h_rs = project_binary(rudin_shapiro(n_t), n_t, rng)
        starts.append(h_rs)
        # Random
        h_rand = np.random.RandomState(seed + n_t).rand(n_t)
        h_rand_bin = project_binary(h_rand, n_t, rng)
        starts.append(h_rand_bin)

        for h_start in starts:
            if remaining() < 15:
                break
            # Quick greedy
            h_g2, c5_g2 = greedy_local_search(h_start, n_t, rng,
                                              time_limit=min(20, remaining() * 0.1))
            if c5_g2 < best_c5:
                best_c5 = c5_g2
                best_h = h_g2.copy()
                best_n = n_t
                # Follow up with SA
                if remaining() > 20:
                    h_s2, c5_s2 = sa_local_search(h_g2, n_t, rng,
                                                  time_limit=min(25, remaining() * 0.12))
                    if c5_s2 < best_c5:
                        best_c5 = c5_s2
                        best_h = h_s2.copy()
                        best_n = n_t

    # Phase 3: Continuous SLSQP polish on the best binary solution
    n = best_n
    h = best_h.copy()
    # Ensure feasibility
    h = project_feasible(h, n)

    for beta in [200, 500, 1500, 4000, 10000]:
        if remaining() < 10:
            break
        h_new = slsqp_refine(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: More binary local search after continuous polish
    if remaining() > 30:
        h_bin2 = project_binary(best_h, n, rng)
        h_g3, c5_g3 = greedy_local_search(h_bin2, n, rng,
                                          time_limit=min(40, remaining() * 0.25))
        if c5_g3 < best_c5:
            best_c5 = c5_g3
            best_h = h_g3.copy()

        # Another SLSQP round
        if remaining() > 15:
            h_poly = project_feasible(best_h, n)
            for beta in [3000, 8000]:
                if remaining() < 8:
                    break
                h_new2 = slsqp_refine(h_poly, n, beta, maxiter=500)
                c5_new2 = compute_c5(h_new2)
                if c5_new2 < best_c5:
                    best_c5 = c5_new2
                    best_h = h_new2.copy()
                    h_poly = h_new2.copy()

    # Phase 5: Final aggressive SA at multiple temperatures
    if remaining() > 25:
        for T_init in [0.3, 0.15, 0.08]:
            if remaining() < 12:
                break
            h_t = project_binary(best_h, n, rng)
            h_f, c5_f = sa_local_search(h_t, n, rng,
                                        time_limit=min(20, remaining() * 0.2),
                                        T_init=T_init, alpha=0.9992)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()
            # Try continuous on this
            if c5_f < best_c5 + 1e-6 and remaining() > 10:
                h_cp = project_feasible(best_h, n)
                h_cp2 = slsqp_refine(h_cp, n, 5000, maxiter=300)
                c5_cp = compute_c5(h_cp2)
                if c5_cp < best_c5:
                    best_c5 = c5_cp
                    best_h = h_cp2.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_h = np.clip(best_h, 0.0, 1.0)
    # Ensure exact sum
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-12:
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            best_h[mask] += diff / np.sum(mask)
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
