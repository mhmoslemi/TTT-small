```
import numpy as np
import time
import math

def _time_left(start, limit):
    return max(0.0, float(limit) - (time.time() - start))

def _next_pow2(m):
    p=1
    while p<m:
        p <<=1
    return p

def _c5_corr(h):
    n=len(h)
    if n==0: return 1.0
    dx=2.0/float(n)
    corr=np.correlate(h, 1.0-h, mode='full')
    return float(np.max(corr))*dx

def _c5_fft(h):
    n=len(h)
    if n==0: return 1.0
    dx=2.0/float(n)
    need=2*n-1
    N=_next_pow2(need)
    a=np.zeros(N, dtype=float)
    b=np.zeros(N, dtype=float)
    a[:n]=h
    b[:n]=1.0-h
    A=np.fft.rfft(a)
    B=np.fft.rfft(b)
    cross=np.fft.irfft(A*np.conj(B), N)
    if n==1:
       return float(cross[0])*dx
    m1=float(np.max(cross[:n]))
    m2=float(np.max(cross[-(n-1):]))
    return max(m1,m2)*dx
```

```
import numpy as np
```

```python
def _time_left(start, limit):
    return max(0.0, float(limit) - (time.time() - float(start)))


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _c5_corr(h):
    n = len(h)
    if n == 0:
        return 1.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def _c5_fft(h):
    n = len(h)
    if n == 0:
        return 1.0
    dx = 2.0 / float(n)
    need = 2 * n - 1
    N = _next_pow2(need)
    A = np.fft.rfft(h, N)
    B = np.fft.rfft(1.0 - h, N)
    cross = np.fft.irfft(A * np.conj(B), N)
    if n == 1:
        return float(cross[0]) * dx
    m1 = float(np.max(cross[:n]))
    m2 = float(np.max(cross[-(n - 1):]))
    return max(m1, m2) * dx

            h, amp, c2, c3, c = _optimize_fourier(nkey, start, limit)
            if h is not None and c < best_c - 1e-12:
                best_c = c
                best_h = h
                best_n = len(h)
        except Exception:
            break

    final_h = best_h
    final_n = len(final_h)
    final_c = _c5_corr(final_h)

    target = 0.38085
    if final_c > target + 1e-9 and _time_left(start, limit) > 30.0:
        safe_amps = [0.33, 0.35, 0.37, 0.39, 0.42, 0.46, 0.50, 0.58, 0.68, 0.80]
        for n_safe in [2048, 4096]:
            if final_c <= target + 1e-9:
                break
            for amp in safe_amps:
                if _time_left(start, limit) < 10.0:
                    break
                try:
                    h_try = _make_cosine(n_safe, amp)
                    c_try = _c5_corr(h_try)
                    if c_try < final_c:
                        final_h = h_try
                        final_n = n_safe
                        final_c = c_try
                except Exception:
                    pass

    try:
        val = evaluate_erdos_solution(final_h)
        if isinstance(val, (int, float, np.floating)):
            val = float(val)
            if val < final_c:
                final_c = val
    except Exception:
        pass

    final_h = np.asarray(final_h, dtype=float)
    final_n = int(len(final_h))
    if _time_left(start, limit) > 5.0:
        try:
            final_h = _project_feasible(final_h, final_n)
            c_chk = _c5_corr(final_h)
            if c_chk < final_c:
                final_c = c_chk
        except Exception:
            pass

    return final_h, float(final_c), final_n
```
