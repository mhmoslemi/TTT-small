    scores = [compute_c5(v) for v in pop]
    best_idx = int(np.argmin(scores))

    start = time.time()
    gen = 0
    no_improve = 0

    while time.time() - start < time_limit and remaining() > 10 and gen < 300:
        gen += 1
        # Sort by score
        order = np.argsort(scores)
        pop = [pop[i] for i in order]
        scores = [scores[i] for i in order]

        if scores[0] < best_idx and False:
            pass
        if scores[0] < min(scores):
            pass

        new_pop = []
        # Elitism: keep top 5
        for i in range(min(5, len(pop))):
            new_pop.append(pop[i].copy())

        while len(new_pop) < pop_size:
            # Tournament selection
            t_size = 5
            cand = rng.randint(0, min(t_size * 3, len(pop)), size=t_size)
            winner1 = cand[np.argmin([scores[c] for c in cand])]
            winner2 = cand[np.argmin([scores[c] for c in cand])]

            # Crossover (preserve weight)
            p1, p2 = pop[winner1], pop[winner2]
            mask = rng.rand(n) < 0.5
            child = np.zeros(n)
            child[mask] = p1[mask]
            child[~mask] = p2[~mask]
            # Repair: ensure exactly n_ones ones
            n_c_ones = int(np.sum(child))
            if n_c_ones > n_ones:
                ones_idx = np.where(child == 1)[0]
                excess = n_c_ones - n_ones
                flip = rng.choice(ones_idx, excess, replace=False)
                child[flip] = 0
            elif n_c_ones < n_ones:
                zero_idx = np.where(child == 0)[0]
                deficit = n_ones - n_c_ones
                flip = rng.choice(zero_idx, deficit, replace=False)
                child[flip] = 1

            # Mutation
            mu_rate = 0.03
            mut_positions = rng.rand(n) < mu_rate
            n_mut = int(np.sum(mut_positions))
            if n_mut > 0 and n_mut < n_ones:
                # Flip pairs to preserve weight
                ones_in_mut = np.where(mut_positions & (child == 1))[0]
                zeros_in_mut = np.where(mut_positions & (child == 0))[0]
                nm = min(len(ones_in_mut), len(zeros_in_mut))
                if nm > 0:
                    for m in range(nm):
                        child[ones_in_mut[m]], child[zeros_in_mut[m]] = \
                            child[zeros_in_mut[m]], child[ones_in_mut[m]]

            new_pop.append(child)

        pop = new_pop[:pop_size]
        scores = [compute_c5(v) for v in pop]
        best_in_gen = min(scores)

        if best_in_gen < scores[0] + 1e-12:
            pass
        # Track improvement
        prev_best = scores[0]
        scores_sorted = sorted(scores)
        if scores_sorted[0] < prev_best - 1e-12:
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 80:
            break

    best_sol = pop[np.argmin(scores)]
    best_val = min(scores)
    return best_sol, best_val


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 320

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 320
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Binary local search at parent resolution
    n = best_n
    h_bin = (best_h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones != n // 2:
        if n_ones > n // 2:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[rng.choice(len(idx), n_ones - n // 2, replace=False)]] = 0
        else:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[rng.choice(len(idx), n // 2 - n_ones, replace=False)]] = 1

    h_b, c5_b = binary_local_search(h_bin, n, time_limit=min(120, max(20, remaining() - 100)), rng=rng)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
        best_n = n

    # Phase 2: SLSQP continuous refinement at parent resolution
    h_cont = best_h.copy()
    for beta in [2000, 5000, 15000, 50000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h_cont, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n
            h_cont = h_new.copy()

    # Top-k SLSQP
    for k in [3, 5, 8]:
        if remaining() < 8:
            break
        h_new = slsqp_topk(h_cont, n, k, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n
            h_cont = h_new.copy()

    # Active set equalization
    if remaining() > 10:
        h_eq, c5_eq = active_set_equalize(h_cont, n, max_iters=500)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            best_n = n
            h_cont = h_eq.copy()

    # Phase 3: Try other resolutions
    n_candidates = []
    for mult in [0.75, 1.0, 1.25, 1.5, 2.0]:
        nn = int(n * mult)
        nn = mm if (mm := (nn // 2) * 2) >= 100 else (nn // 2) * 2
        if 100 <= nn <= 800:
            n_candidates.append(nn)
    # Add some specific values
    n_candidates.extend([240, 320, 400, 480, 560, 640])
    n_candidates = sorted(set(x for x in n_candidates if x >= 100 and x <= 800 and x % 2 == 0))

    for n_t in n_candidates:
        if remaining() < 20:
            break
        if n_t == best_n:
            continue

        # Interpolate best solution to new resolution
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)

        # Binary version
        h_bin_t = (h_feas > 0.5).astype(float)
        n_ones = int(np.sum(h_bin_t))
        if n_ones != n_t // 2:
            if n_ones > n_t // 2:
                idx = np.where(h_bin_t == 1)[0]
                h_bin_t[idx[rng.choice(len(idx), n_ones - n_t // 2, replace=False)]] = 0
            else:
                idx = np.where(h_bin_t == 0)[0]
                h_bin_t[idx[rng.choice(len(idx), n_t // 2 - n_ones, replace=False)]] = 1

        # Binary local search (shorter for larger n)
        tl = min(30, max(10, remaining() // 4)) if n_t <= 480 else min(15, max(8, remaining() // 6))
        h_b_t, c5_b_t = binary_local_search(h_bin_t, n_t, time_limit=tl, rng=rng)

        # SLSQP
        h_s_t = h_b_t.copy()
        for beta in [3000, 10000]:
            if remaining() < 6:
                break
            h_s_t = slsqp_smooth(h_s_t, n_t, beta, maxiter=300)
        c5_s_t = compute_c5(h_s_t)

        # Top-k
        if remaining() > 5:
            h_k_t = slsqp_topk(h_s_t, n_t, 5, maxiter=200)
            c5_k_t = compute_c5(h_k_t)
            if c5_k_t < c5_s_t:
                h_s_t = h_k_t
                c5_s_t = c5_k_t

        # Active set
        if remaining() > 5:
            h_s_t, c5_s_t = active_set_equalize(h_s_t, n_t, max_iters=200)

        if c5_s_t < best_c5:
            best_c5 = c5_s_t
            best_h = h_s_t.copy()
            best_n = n_t
        if c5_b_t < best_c5:
            best_c5 = c5_b_t
            best_h = h_b_t.copy()
            best_n = n_t

    # Phase 4: GA at best resolution (if time allows)
    if remaining() > 40:
        n_ga = best_n
        h_bin_ga = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin_ga))
        if n_ones != n_ga // 2:
            if n_ones > n_ga // 2:
                idx = np.where(h_bin_ga == 1)[0]
                h_bin_ga[idx[rng.choice(len(idx), n_ones - n_ga // 2, replace=False)]] = 0
            else:
                idx = np.where(h_bin_ga == 0)[0]
                h_bin_ga[idx[rng.choice(len(idx), n_ga // 2 - n_ones, replace=False)]] = 1

        h_ga, c5_ga = ga_binary(n_ga, time_limit=min(60, max(15, remaining() - 30)),
                                rng=rng, parent_bin=h_bin_ga)
        if c5_ga < best_c5:
            best_c5 = c5_ga
            best_h = h_ga.copy()
            best_n = n_ga

        # SLSQP on GA result
        if remaining() > 10:
            h_sa = slsqp_smooth(h_ga, n_ga, 10000, maxiter=400)
            c5_sa = compute_c5(h_sa)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_ga

    # Phase 5: Final intensive refinement
    n = best_n
    h = best_h.copy()

    # Multiple SLSQP passes with increasing beta
    for beta in [5000, 15000, 40000, 100000]:
        if remaining() < 5:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Top-k refinement
    for k in [3, 4, 6]:
        if remaining() < 5:
            break
        h_new = slsqp_topk(h, n, k, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Active set
    if remaining() > 6:
        h_eq, c5_eq = active_set_equalize(h, n, max_iters=600)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Binary fine-tune on best solution
    if remaining() > 8:
        h_bin_final = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin_final))
        if n_ones != n // 2:
            if n_ones > n // 2:
                idx = np.where(h_bin_final == 1)[0]
                h_bin_final[idx[rng.choice(len(idx), n_ones - n // 2, replace=False)]] = 0
            else:
                idx = np.where(h_bin_final == 0)[0]
                h_bin_final[idx[rng.choice(len(idx), n // 2 - n_ones, replace=False)]] = 1
        h_bf, c5_bf = binary_local_search(h_bin_final, n,
                                          time_limit=min(30, max(5, remaining() - 5)), rng=rng)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()
            best_n = n

        # SLSQP on the binary result
        if remaining() > 5:
            h_sf = slsqp_smooth(h_bf, n, 20000, maxiter=400)
            c5_sf = compute_c5(h_sf)
            if c5_sf < best_c5:
                best_c5 = c5_sf
                best_h = h_sf.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    else:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        best_c5 = min(best_c5, eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
