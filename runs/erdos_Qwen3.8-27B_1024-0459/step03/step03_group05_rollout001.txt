def project_box_sum(x, target):
    x = np.asarray(x, dtype=float).copy()
    if x.size == 0:
        return x
    n = x.size
    S = float(target)
    if S <= 0.0:
        return np.zeros(n, dtype=float)
    if S >= float(n):
        return np.ones(n, dtype=float)

    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if np.sum(y) < S:
            hi = mid
        else:
            lo = mid

    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(30):
        diff = S - float(np.sum(y))
        if abs(diff) < 1e-14:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            y[mask] += diff / cnt
            y = np.clip(y, 0.0, 1.0)
        else:
            if diff > 0.0:
                idx = int(np.argmin(y))
                room = 1.0 - y[idx]
                y[idx] += min(diff, room)
            else:
                idx = int(np.argmax(y))
                room = y[idx]
                y[idx] -= min(-diff, room)
    return y


def ensure_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    S = float(target)
    diff = S - float(np.sum(h))
    if abs(diff) < 1e-15:
        return h

    mask = (h > 1e-12) & (h < 1.0 - 1e-12)
    cnt = int(np.count_nonzero(mask))
    if cnt > 0:
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    else:
        if diff > 0.0:
            order = np.argsort(h)
        else:
            order = np.argsort(h)[::-1]
        rem = diff
        for idx in order:
            if rem > 0.0:
                step = min(rem, 1.0 - h[idx])
            else:
                step = max(rem, -h[idx])
            if step == 0.0:
                continue
            h[idx] += step
            rem -= step
            if abs(rem) < 1e-15:
                break

    diff = S - float(np.sum(h))
    if abs(diff) > 1e-14:
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return 0.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def ab_from_h(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    A = corr[:n][::-1].copy()
    B = corr[n - 1:].copy()
    s0 = 0.5 * (A[0] + B[0])
    A[0] = s0
    B[0] = s0
    count = max(float(np.max(A)), float(np.max(B)))
    return A, B, count


def top_lags(A, B, num=30):
    vals = np.maximum(A, B)
    if vals.size <= num:
        return np.argsort(vals)[::-1].copy()
    part = np.argpartition(vals, -num)
    idx = part[-num:]
    idx = idx[np.argsort(vals[idx])[::-1]]
    return idx.copy()


def resample_h(old, n_new, mode):
    old = np.asarray(old, dtype=float)
    n_old = len(old)
    n_new = int(n_new)
    if n_new <= 0:
        return np.empty(0, dtype=float)
    if n_old <= 0:
        return np.zeros(n_new, dtype=float)
    if n_old == n_new:
        return old.copy()

    if mode == "step":
        idx = np.floor((np.arange(n_new) + 0.5) * n_old / n_new).astype(int)
        idx = np.clip(idx, 0, n_old - 1)
        return old[idx].astype(float)

    xp = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(xn, xp, old)


def evenize(n):

def eval_line(A0, lin0, Apos, Bpos, LA, LB, cross_idx, d, tmpA, tmpB):
    dd = d * d
    f = A0 + lin0 * d - 2.0 * dd
    m = LA.size
    if m:
        np.multiply(LA, d, out=tmpA)
        tmpA += Apos
        if cross_idx >= 0:
            tmpA[cross_idx] += dd
        t = tmpA.max()
        if t > f:
            f = t

        np.multiply(LB, d, out=tmpB)
        tmpB += Bpos
        if cross_idx >= 0:
            tmpB[cross_idx] += dd
        t = tmpB.max()
        if t > f:
            f = t
    return float(f)


def line_min(A, B, h, i, j, LA, LB, cross_idx, lo, hi, tmpA, tmpB,

def prepare_pair(i, j, h, LA, LB, n):
    LA.fill(0.0)
    LB.fill(0.0)

    p = i
    dmax = n - 1 - p
    if dmax > 0:
        seg = h[p + 1:p + 1 + dmax]
        LA[:dmax] += 1.0 - seg
        LB[:dmax] -= seg

    dmax2 = p
    if dmax2 > 0:
        rev = h[:p][::-1]
        LA[:dmax2] -= rev
        LB[:dmax2] += 1.0 - rev

    p = j
    dmax = n - 1 - p
    if dmax > 0:
        seg = h[p + 1:p + 1 + dmax]
        LA[:dmax] -= 1.0 - seg
        LB[:dmax] += seg

    dmax2 = p
    if dmax2 > 0:
        rev = h[:p][::-1]
        LA[:dmax2] += rev
        LB[:dmax2] -= 1.0 - rev

    return abs(i - j) - 1


def try_line_pair(h, n, A, B, LA, LB, tmpA, tmpB, i, j,
                  grid_fracs, gold_iters, current_count):
    a = h[i]
    b = h[j]

    low = max(-a, b - 1.0)
    high = min(1.0 - a, b)

    eps = 1e-12
    low += eps
    high -= eps

    if high - low < 1e-10:
        return 0.0, current_count

    cross_idx = abs(i - j) - 1
    prepare_pair(i, j, h, LA, LB, n)
    d, f = line_min(A, B, h, i, j, LA, LB, cross_idx, low, high,
                    tmpA, tmpB, grid_fracs, gold_iters, current_count)
    return d, f


def accept_pair(h, A, B, i, j, d, LA, LB, cross_idx, tmpA, tmpB):
    a = h[i]
    b = h[j]
    dd = d * d

    A[0] = A[0] + 2.0 * (b - a) * d - 2.0 * dd

    m = LA.size
    if m:
        np.multiply(LA, d, out=tmpA)
        tmpA += A[1:]
        if cross_idx >= 0:
            tmpA[cross_idx] += dd
        A[1:] = tmpA

        np.multiply(LB, d, out=tmpB)
        tmpB += B[1:]
        if cross_idx >= 0:
            tmpB[cross_idx] += dd
        B[1:] = tmpB

    h[i] = a + d
    h[j] = b - d

    if h[i] < 0.0 and h[i] > -1e-10:
        h[i] = 0.0
    if h[i] > 1.0 and h[i] < 1.0 + 1e-10:
        h[i] = 1.0
    if h[j] < 0.0 and h[j] > -1e-10:
        h[j] = 0.0
    if h[j] > 1.0 and h[j] < 1.0 + 1e-10:
        h[j] = 1.0

    global_rng = np.random.RandomState(seed)

    for idx, n_phase in enumerate(phase_ns):
        remaining = budget_s - (time.time() - t0)
        if remaining < 80.0:
            break

        frac = fractions[idx] if idx < len(fractions) else 0.05
        cap = caps[idx] if idx < len(caps) else 80.0
        phase_time = min(frac * remaining, cap, remaining - 15.0)
        if phase_time < 30.0:
            if remaining < 140.0:
                break
            phase_time = min(30.0, remaining - 15.0)
        if phase_time <= 0.0:
            break

        if n_phase == best_n:
            cand = best_h.copy()
            c_init = best_c
        else:
            cand, c_init = choose_candidate(best_h, best_n, n_phase)
            if c_init < best_c:
                best_c = c_init
                best_h = cand.copy()
                best_n = n_phase

        max_trials = estimate_max_trials(phase_time, n_phase)
        h_out, _cnt_out = run_pair_descent(
            cand, n_phase, max_trials, phase_time * 0.97,
            seed + 17 * idx + n_phase
        )

        h_out = ensure_sum(h_out, 0.5 * float(n_phase))
        c_out = compute_c5(h_out)
        if c_out < best_c:
            best_c = c_out
            best_h = h_out.copy()
            best_n = n_phase

    remaining = budget_s - (time.time() - t0)
    if remaining > 140.0:
        mut_ns = [best_n]
        nn2 = evenize(2 * best_n)
        if nn2 <= 2048 and nn2 not in mut_ns:
            mut_ns.append(nn2)

        for mi, mn in enumerate(mut_ns):
            remaining = budget_s - (time.time() - t0)
            if remaining < 110.0:
                break

            for sigma in (0.05, 0.10, 0.20, 0.35):
                remaining = budget_s - (time.time() - t0)
                if remaining < 90.0:
                    break

                target = 0.5 * float(mn)
                if mn == best_n:
                    base_h = best_h
                else:
                    base_h = project_box_sum(resample_h(best_h, mn, "linear"), target)
                    cb = compute_c5(base_h)
                    if cb < best_c:
                        best_c = cb
                        best_h = base_h.copy()
                        best_n = mn

                noise = global_rng.normal(0.0, float(sigma), size=mn)
                hm = project_box_sum(base_h + noise, target)
                cm = compute_c5(hm)

                if cm < best_c + 0.004:
                    t_mut = min(remaining * 0.12, 120.0, remaining - 20.0)
                    if t_mut <= 20.0:
                        continue
                    max_mul = estimate_max_trials(t_mut, mn)
                    hm2, _ = run_pair_descent(
                        hm, mn, max_mul, t_mut * 0.97,
                        seed + 1000 * (mi + 1) + int(sigma * 1000)
                    )
                    hm2 = ensure_sum(hm2, target)
                    cm2 = compute_c5(hm2)
                    if cm2 < best_c:
                        best_c = cm2
                        best_h = hm2.copy()
                        best_n = mn

    remaining = budget_s - (time.time() - t0)
    best_h = ensure_sum(best_h, 0.5 * float(best_n))

    if remaining > 25.0:
        final_time = min(remaining - 8.0, 60.0)
        if final_time > 10.0:
            max_final = estimate_max_trials(final_time, best_n)
            hp, _ = run_pair_descent(
                best_h, best_n, max_final, final_time * 0.98,
                seed + 99999
            )
            hp = ensure_sum(hp, 0.5 * float(best_n))
            cp = compute_c5(hp)
            if cp < best_c:
                best_c = cp
                best_h = hp.copy()
                best_n = len(hp)

    best_h = ensure_sum(best_h, 0.5 * float(best_n))
    final_c = compute_c5(best_h)

    try:
        ev = evaluate_erdos_solution(best_h)
        ev = float(ev)
        if np.isfinite(ev) and ev < final_c:
            final_c = ev
    except Exception:
        pass

    best_h = ensure_sum(best_h, 0.5 * float(best_n))
    return (best_h, float(final_c), int(best_n))
