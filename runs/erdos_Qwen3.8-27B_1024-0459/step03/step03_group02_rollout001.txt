def project_feasible(x, target):
    x = np.array(x, dtype=float).copy()
    n = x.size
    if n == 0:
        return x
    target = float(target)

    x = np.clip(x, 0.0, 1.0)
    for _ in range(30):
        s = float(np.sum(x))
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            mask = x < (1.0 - 1e-14)
        else:
            mask = x > 1e-14
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            x[mask] += diff / cnt
        else:
            x[:] += diff / n
        x = np.clip(x, 0.0, 1.0)

    diff = target - float(np.sum(x))
    if abs(diff) > 1e-12:
        if diff > 0.0:
            i = int(np.argmax(1.0 - x))
            add = min(diff, 1.0 - x[i])
            x[i] += add
            diff -= add
            if abs(diff) > 1e-13:
                mask = (1.0 - x) > 1e-10
                if np.count_nonzero(mask) > 0:
                    x[mask] += diff / max(1, np.count_nonzero(mask))
        else:
            i = int(np.argmin(x))
            add = max(diff, -x[i])
            x[i] += add
            diff -= add
            if abs(diff) > 1e-13:
                mask = x > 1e-10
                if np.count_nonzero(mask) > 0:
                    x[mask] += diff / max(1, np.count_nonzero(mask))
        x = np.clip(x, 0.0, 1.0)

    diff = target - float(np.sum(x))
    if abs(diff) > 0.0:
        if diff > 0.0:
            i = int(np.argmax(1.0 - x))
            x[i] = min(1.0, x[i] + diff)
        else:
            i = int(np.argmin(x))
            x[i] = max(0.0, x[i] + diff)

    return x


def blend_feasible(x, y, t, target):
    if t <= 0.0:
        return x.copy()
    if t >= 1.0:
        return y.copy()
    z = (1.0 - t) * x + t * y
    np.clip(z, 0.0, 1.0, out=z)

    diff = target - float(np.sum(z))
    if abs(diff) > 1e-14:
        if diff > 0.0:
            idx = np.nonzero((1.0 - z) > 1e-12)[0]
            if idx.size:
                z[idx] += diff / idx.size
            else:
                z[:] += diff / max(1, z.size)
        else:
            idx = np.nonzero(z > 1e-12)[0]
            if idx.size:
                z[idx] += diff / idx.size
            else:
                z[:] += diff / max(1, z.size)
        z = np.clip(z, 0.0, 1.0)

    diff = target - float(np.sum(z))
    if abs(diff) > 0.0:
        if diff > 0.0:
            i = int(np.argmax(1.0 - z))
            z[i] = min(1.0, z[i] + diff)
        else:
            i = int(np.argmin(z))
            z[i] = max(0.0, z[i] + diff)

    return z


def build_h_sym(g):

def symmetrize_full(h):

def compute_corr_direct(h):
    n = h.size
    if n == 0:
        return np.array([], dtype=float)
    if n == 1:
        return np.array([h[0] * (1.0 - h[0])], dtype=float)
    return np.correlate(h, 1.0 - h, mode="full")


def compute_corr_fft(h):
    n = h.size
    if n == 0:
        return np.array([], dtype=float)
    if n == 1:
        return np.array([h[0] * (1.0 - h[0])], dtype=float)

    need = 2 * n - 1
    L = 1
    while L < need:
        L <<= 1

    fa = np.fft.rfft(h, L)
    fb = np.fft.rfft(1.0 - h, L)
    r = np.fft.irfft(fa * np.conj(fb), n=L)

    out = np.empty(need, dtype=float)
    out[: n - 1] = r[L - (n - 1): L]
    out[n - 1] = r[0]
    out[n:] = r[1:n]

    np.maximum(out, 0.0, out=out)
    return out


def compute_corr_auto(h):
    n = h.size
    if n <= 320:
        return compute_corr_direct(h)
    return compute_corr_fft(h)


def compute_c5_direct(h):

    if n == 0:
        return float("inf")
    corr = compute_corr_direct(h)
    return float(np.max(corr) * (2.0 / n))


def compute_c5_auto(h):

    if n == 0:
        return float("inf")

    return float(np.max(corr) * (2.0 / n))


def active_top(corr, dx, top_count=80, peak_tol=0.015, beta=60.0):
    q = corr * dx
    mval = float(np.max(q))
    mask = q >= (mval - peak_tol)
    idx = np.where(mask)[0]

    if idx.size == 0:
        idx = np.array([int(np.argmax(q))])

    if idx.size > top_count:
        qsub = q[idx]
        order = np.argsort(qsub)
        idx = idx[order[-top_count:]]

    qq = q[idx]
    z = beta * (qq - mval)
    z -= z.max()
    w = np.exp(z)
    sw = float(np.sum(w))
    if sw <= 1e-300:
        w = np.zeros_like(qq)
        if w.size == 0:
            return idx, w
        w[0] = 1.0
    else:
        w /= sw
    return idx, w


def weighted_gradient(h, idx, w):
    n = h.size
    grad = np.zeros(n, dtype=float)
    if n == 0:
        return grad

    for ii in range(idx.size):
        off = int(idx[ii]) - (n - 1)
        wt = float(w[ii])
        if wt == 0.0:
            continue

        if off == 0:
            grad += wt * (1.0 - 2.0 * h)
        elif off > 0:
            p = n - off
            if p <= 0:
                continue
            grad[off:off + p] += wt * (1.0 - h[:p])
            grad[:p] -= wt * h[off:off + p]
        else:
            e = -off
            p = n - e
            if p <= 0:
                continue
            grad[:p] += wt * (1.0 - h[e:e + p])
            grad[e:e + p] -= wt * h[:p]

    return grad


def vertex_from_scores(scores, target):
    n = scores.size
    v = np.zeros(n, dtype=float)
    target = float(target)

    if n == 0:
        return v
    if target <= 0.0:
        return v
    if target >= n:
        v.fill(1.0)
        return v

    if abs(target - round(target)) <= 1e-12:
        k = int(round(target))
        if k > 0:
            idx = np.argpartition(scores, k - 1)[:k]
            v[idx] = 1.0
    else:
        k = int(math.floor(target))
        frac = target - k
        if k > 0:
            idx = np.argpartition(scores, k - 1)[:k]
            v[idx] = 1.0
        if frac > 1e-12:
            rem = np.nonzero(v < 0.5)[0]
            if rem.size:
                j = rem[int(np.argmin(scores[rem]))]
                v[j] = min(1.0, max(0.0, frac))

    return project_feasible(v, target)


def fw_line_search_full(h, v, f0, n):

    best_a = 0.0
    best_f = f0
    best_h = h

        h2 = blend_feasible(h, v, a, target)
        c2 = compute_c5_auto(h2)
        if c2 < best_f - 1e-14:

            best_a = a
            best_h = h2

    if best_a > 0.0:
        refine = (best_a - 0.1, best_a - 0.05, best_a + 0.05, best_a + 0.1)
        for a in refine:

            h2 = blend_feasible(h, v, a, target)
            c2 = compute_c5_auto(h2)
            if c2 < best_f - 1e-14:

                best_a = a
                best_h = h2

    return best_h, best_f, best_a

def fw_full_step(h, n, top_count=80, peak_tol=0.015):
    dx = 2.0 / n
    corr = compute_corr_auto(h)
    f0 = float(np.max(corr)) * dx

    idx, w = active_top(corr, dx, top_count=top_count, peak_tol=peak_tol)
    grad = weighted_gradient(h, idx, w)
    grad -= float(np.mean(grad))

    target = n / 2.0
    v = vertex_from_scores(grad, target)
    if np.max(np.abs(v - h)) < 1e-13:
        return h, f0

    h1, f1, _ = fw_line_search_full(h, v, f0, n)
    if f1 < f0 - 1e-13:
        return h1, f1

    v2 = vertex_from_scores(-grad, target)
    if np.max(np.abs(v2 - h)) > 1e-13:
        h2, f2, _ = fw_line_search_full(h, v2, f0, n)
        if f2 < f0 - 1e-13:
            return h2, f2

    return h, f0


def fw_sym_step(g, m, top_count=80, peak_tol=0.015):
    n = 2 * m
    h = build_h_sym(g)
    dx = 2.0 / n
    corr = compute_corr_auto(h)
    f0 = float(np.max(corr)) * dx

    idx, w = active_top(corr, dx, top_count=top_count, peak_tol=peak_tol)
    grad_h = weighted_gradient(h, idx, w)
    s = grad_h[:m] + grad_h[m:][::-1]
    s -= float(np.mean(s))

    target = m / 2.0
    v = vertex_from_scores(s, target)

        return g, f0

    g1, f1, _ = fw_line_search_sym(g, v, f0, m)

def _slsqp_x_to_h(x):

        return x
    m = _SL_M
    h = np.empty(2 * m, dtype=float)
    h[:m] = x
    h[m:] = x[::-1]
    return h


def _slsqp_target():
    if _SL_MODE == 0:
        return _SL_N / 2.0
    return _SL_M / 2.0


def _slsqp_fun(x):
    h = _slsqp_x_to_h(x)
    n = _SL_N
    dx = 2.0 / n
    corr = compute_corr_auto(h)
    q = corr * dx
    mval = float(np.max(q))
    b = _SL_BETA
    z = b * (q - mval)
    z -= z.max()
    s = float(np.sum(np.exp(z)))
    if s <= 0.0:
        return mval
    return float(mval + math.log(s) / b)


def _slsqp_sum(x):
    return float(np.sum(x) - _slsqp_target())


def _slsqp_jac(x):
    h = _slsqp_x_to_h(x)
    n = _SL_N
    dx = 2.0 / n
    corr = compute_corr_auto(h)
    q = corr * dx
    mval = float(np.max(q))
    b = _SL_BETA
    z = b * (q - mval)
    z -= z.max()
    w = np.exp(z)
    sw = float(np.sum(w))
    if sw <= 1e-300:
        w = np.zeros_like(q)
        if w.size:
            w[int(np.argmax(q))] = 1.0
        else:
            return np.zeros(h.size, dtype=float)
    else:
        w /= sw

    if n <= 700:
        idx = np.nonzero(w > 1e-12)[0]
        if idx.size == 0:
            idx = np.array([int(np.argmax(q))])
        widx = w[idx]
        s = float(np.sum(widx))
        if s <= 1e-300:
            widx = np.zeros_like(widx)
            if widx.size:
                widx[0] = 1.0
        else:
            widx /= s
    else:
        idx, widx = active_top(corr, dx, top_count=200, peak_tol=0.02, beta=60.0)

    grad_h = weighted_gradient(h, idx, widx) * dx

    if _SL_MODE == 1:
        m = _SL_M
        return grad_h[:m] + grad_h[m:][::-1]
    return grad_h


def slsqp_smooth(x, n, m, mode, beta, maxiter):
    global _SL_N, _SL_M, _SL_BETA, _SL_MODE

    if mode == 0:
        _SL_N = n
        _SL_M = n
        _SL_MODE = 0
        target = n / 2.0
    else:
        _SL_N = n
        _SL_M = m
        _SL_MODE = 1
        target = m / 2.0

    _SL_BETA = float(beta)
    x0 = project_feasible(np.asarray(x, dtype=float).ravel(), target)
    bounds = [(0.0, 1.0)] * len(x0)
    cons = [{"type": "eq", "fun": _slsqp_sum}]

    try:
        res = minimize(
            _slsqp_fun,
            x0,
            jac=_slsqp_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-12, "iprint": 0},
        )
        if res is not None and hasattr(res, "x") and res.x is not None:
            x1 = project_feasible(np.asarray(res.x, dtype=float).ravel(), target)
        else:
            x1 = x0
    except Exception:
        x1 = x0

    return x1


def polish_gradient_sym(g, m, until_time, beta=30.0, base=0.04):

def make_valley_g(m, kind):
    x = (np.arange(m, dtype=float) + 0.5) / float(m)
    if kind == 0:
        g = 1.0 - x
    elif kind == 1:
        g = 0.5 * (1.0 + np.cos(np.pi * x))
    elif kind == 2:
        g = np.power(1.0 - x, 0.5)
    elif kind == 3:
        g = np.power(1.0 - x, 2.0)
    elif kind == 4:
        g = 1.0 / (1.0 + np.exp(4.0 * (x - 0.5)))
    elif kind == 5:
        g = 1.0 / (1.0 + np.exp(8.0 * (x - 0.5)))
    elif kind == 6:
        g = 1.0 / (1.0 + np.exp(16.0 * (x - 0.5)))
    elif kind == 7:
        g = 1.0 - np.exp(-((x - 0.5) / 0.18) ** 2)
    elif kind == 8:
        g = 1.0 - 0.9 * np.exp(-((x - 0.5) / 0.30) ** 2)
    else:
        g = (1.0 - x) + 0.15 * np.sin(4.0 * np.pi * x)
    return project_feasible(g, m / 2.0)


def make_random_g(m):

    base = 1.0 - x
    noise = 0.18 * _RNG.normal(size=m)

def interp_g(src, src_m, dst_m):
    if src_m == dst_m:
        return src.copy()
    xp = (np.arange(src_m, dtype=float) + 0.5) * (1.0 / src_m)
    xq = (np.arange(dst_m, dtype=float) + 0.5) * (1.0 / dst_m)
    out = np.interp(xq, xp, src)
    return project_feasible(out, dst_m / 2.0)


def consider_solution(h, best_h, best_c, best_n, best_g, best_sym_c, best_sym_n):
    if h is None:
        return best_h, best_c, best_n, best_g, best_sym_c, best_sym_n

    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return best_h, best_c, best_n, best_g, best_sym_c, best_sym_n

    h = project_feasible(h, n / 2.0)
    c = compute_c5_auto(h)
    if c < best_c - 1e-15:
        best_h = h.copy()
        best_c = c
        best_n = n

    if n % 2 == 0:
        hs = symmetrize_full(h)
        hs = project_feasible(hs, n / 2.0)
        cs = compute_c5_auto(hs)
        m = n // 2
        if cs < best_sym_c - 1e-15:
            best_sym_c = cs
            best_sym_n = n
            best_g = hs[:m].copy()

    return best_h, best_c, best_n, best_g, best_sym_c, best_sym_n


def optimize_full_h(h, n, time_limit):

    h = project_feasible(h, target)

            if stall > 70:
                break

    if n <= 160:
        sl_betas = (20.0, 50.0, 100.0, 200.0, 500.0)
        sl_iter = 110
    elif n <= 256:
        sl_betas = (30.0, 100.0, 300.0)
        sl_iter = 50
    else:
        sl_betas = ()
        sl_iter = 0

    if sl_betas:
        xcur = best_h.copy()
        for beta in sl_betas:
            if time.time() > limit - 35.0:
                break
            hnew = slsqp_smooth(xcur, n, n, 0, beta, sl_iter)
            cnew = compute_c5_auto(hnew)
            if cnew < best_c - 1e-14:
                best_h = hnew.copy()
                best_c = cnew
            if cnew > best_c + 0.01:
                xcur = best_h.copy()
            else:
                xcur = hnew.copy()

    if time.time() < limit - 12.0:
        for base in (0.04, 0.01):
            hnew, cnew = polish_gradient_full(best_h, n, limit, beta=40.0, base=base)
            if cnew < best_c - 1e-14:
                best_h = hnew.copy()
                best_c = cnew

    if time.time() < limit - 4.0:
        hcur = best_h.copy()
        for _ in range(60):
            if time.time() >= limit:
                break
            hnew, fnew = fw_full_step(hcur, n)
            if fnew < best_c - 1e-13:
                best_h = hnew.copy()
                best_c = fnew
                hcur = hnew.copy()
            else:
                break

    return best_h, best_c


def optimize_sym_g(g, m, time_limit):
    target = m / 2.0
    g = project_feasible(g, target)
    best_g = g.copy()
    best_c = compute_c5_auto(build_h_sym(g))

    t0 = time.time()
    limit = min(t0 + time_limit, _START_TIME + _BUDGET_S - 8.0)
    gcur = g.copy()

    fw_limit = min(limit, t0 + 0.38 * time_limit)
    stall = 0
    while time.time() < fw_limit:
        gnew, fnew = fw_sym_step(gcur, m)
        if fnew < best_c - 1e-12:
            best_g = gnew.copy()
            best_c = fnew
            gcur = gnew.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break

    if m <= 80:
        sl_betas = (20.0, 50.0, 100.0, 200.0, 500.0)
        sl_iter = 90
    elif m <= 128:
        sl_betas = (25.0, 75.0, 200.0, 500.0)
        sl_iter = 55
    else:
        sl_betas = ()
        sl_iter = 0

    if sl_betas:
        xcur = best_g.copy()
        for beta in sl_betas:
            if time.time() > limit - 35.0:
                break
            gnew = slsqp_smooth(xcur, 2 * m, m, 1, beta, sl_iter)
            cnew = compute_c5_auto(build_h_sym(gnew))
            if cnew < best_c - 1e-14:
                best_g = gnew.copy()
                best_c = cnew
            if cnew > best_c + 0.01:
                xcur = best_g.copy()
            else:
                xcur = gnew.copy()

    if time.time() < limit - 12.0:
        for base in (0.05, 0.02, 0.008):
            gnew, cnew = polish_gradient_sym(best_g, m, limit, beta=35.0, base=base)
            if cnew < best_c - 1e-14:
                best_g = gnew.copy()
                best_c = cnew

    if time.time() < limit - 4.0:
        gcur = best_g.copy()
        for _ in range(80):
            if time.time() >= limit:
                break
            gnew, fnew = fw_sym_step(gcur, m)
            if fnew < best_c - 1e-13:
                best_g = gnew.copy()
                best_c = fnew
                gcur = gnew.copy()
            else:
                break

    return best_g, best_c


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S, _RNG

    _START_TIME = time.time()
    _BUDGET_S = float(max(10.0, budget_s))
    _RNG = np.random.RandomState(int(seed) % (2**31 - 1))

    best_h = None
    best_c = float("inf")
    best_n = 0

    best_g = None
    best_sym_c = float("inf")
    best_sym_n = 0

    try:
        parent = np.asarray(initial_h_values, dtype=float)
        if parent.size > 0:
            h0 = project_feasible(parent, parent.size / 2.0)
            best_h, best_c, best_n, best_g, best_sym_c, best_sym_n = consider_solution(
                h0, best_h, best_c, best_n, best_g, best_sym_c, best_sym_n
            )
            best_h, best_c, best_n, best_g, best_sym_c, best_sym_n = consider_solution(
                1.0 - h0, best_h, best_c, best_n, best_g, best_sym_c, best_sym_n
            )
    except Exception:
        pass

    if best_h is None:
        m = 80
        g = make_valley_g(m, 1)
        h = build_h_sym(g)
        best_h, best_c, best_n, best_g, best_sym_c, best_sym_n = consider_solution(
            h, best_h, best_c, best_n, best_g, best_sym_c, best_sym_n
        )

    for n in (160, 256, 512):
        m = n // 2
        for kind in range(9):
            g = make_valley_g(m, kind)
            h = build_h_sym(g)
            best_h, best_c, best_n, best_g, best_sym_c, best_sym_n = consider_solution(
                h, best_h, best_c, best_n, best_g, best_sym_c, best_sym_n
            )
            best_h, best_c, best_n, best_g, best_sym_c, best_sym_n = consider_solution(
                1.0 - h, best_h, best_c, best_n, best_g, best_sym_c, best_sym_n
            )

        for _ in range(3):
            g = make_random_g(m)
            h = build_h_sym(g)
            best_h, best_c, best_n, best_g, best_sym_c, best_sym_n = consider_solution(
                h, best_h, best_c, best_n, best_g, best_sym_c, best_sym_n
            )

    if best_h is not None and best_n <= 320 and remaining_time() > 190.0:
        t_full = min(160.0, remaining_time() - 30.0)
        if t_full > 30.0:
            hnew, cnew = optimize_full_h(best_h, best_n, t_full)
            best_h, best_c, best_n, best_g, best_sym_c, best_sym_n = consider_solution(
                hnew, best_h, best_c, best_n, best_g, best_sym_c, best_sym_n
            )

    start_m = 80
    if best_g is not None and best_sym_n % 2 == 0 and best_sym_n > 0:
        src_m = best_sym_n // 2
        gcur = interp_g(best_g, src_m, start_m)
    else:
        gcur = make_valley_g(start_m, 1)

    sym_alloc = {160: 90.0, 256: 190.0, 512: 330.0, 1024: 270.0}

    for n in (160, 256, 512, 1024):
        m = n // 2
        if gcur.size != m:
            gcur = interp_g(gcur, gcur.size, m)

        alloc = min(sym_alloc.get(n, 80.0), remaining_time() - 30.0)
        if alloc <= 25.0:
            continue

        gnew, cnew = optimize_sym_g(gcur, m, alloc)
        hnew = build_h_sym(gnew)
        best_h, best_c, best_n, best_g, best_sym_c, best_sym_n = consider_solution(
            hnew, best_h, best_c, best_n, best_g, best_sym_c, best_sym_n
        )
        best_h, best_c, best_n, best_g, best_sym_c, best_sym_n = consider_solution(
            1.0 - hnew, best_h, best_c, best_n, best_g, best_sym_c, best_sym_n
        )
        gcur = gnew.copy()

        if remaining_time() <= 40.0:
            break

    if best_h is not None and remaining_time() > 35.0:
        n = best_n
        if n % 2 == 0:
            hsym = symmetrize_full(best_h)
            hsym = project_feasible(hsym, n / 2.0)
            if compute_c5_auto(hsym) <= best_c + 5e-7:
                m = n // 2
                gfin = hsym[:m].copy()
                until = _START_TIME + _BUDGET_S - 5.0
                g2, c2 = polish_gradient_sym(gfin, m, until, beta=60.0, base=0.015)
                h2 = build_h_sym(g2)
                best_h, best_c, best_n, best_g, best_sym_c, best_sym_n = consider_solution(
                    h2, best_h, best_c, best_n, best_g, best_sym_c, best_sym_n
                )
                best_h, best_c, best_n, best_g, best_sym_c, best_sym_n = consider_solution(
                    1.0 - h2, best_h, best_c, best_n, best_g, best_sym_c, best_sym_n
                )
        else:
            until = _START_TIME + _BUDGET_S - 5.0
            h2, c2 = polish_gradient_full(best_h, n, until, beta=80.0, base=0.01)
            best_h, best_c, best_n, best_g, best_sym_c, best_sym_n = consider_solution(
                h2, best_h, best_c, best_n, best_g, best_sym_c, best_sym_n
            )

    if best_h is None:
        m = 80
        best_h = build_h_sym(make_valley_g(m, 1))
        best_n = m * 2

    best_h = project_feasible(np.asarray(best_h, dtype=float), best_n / 2.0)
    final_c = compute_c5_direct(best_h)

    try:
        eval_c = evaluate_erdos_solution(best_h)
        eval_c = float(eval_c)
        if math.isfinite(eval_c) and eval_c < final_c:
            final_c = eval_c
    except Exception:
        pass

    return best_h, float(final_c), int(best_n)
