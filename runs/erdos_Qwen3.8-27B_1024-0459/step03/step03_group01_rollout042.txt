def compute_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n

def slsqp_smooth(h, n, beta, maxiter=400):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    
    def smooth_obj(x):
        dx_val = 2.0 / n
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -60, 60)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_val
    
    def smooth_grad(x):
        dx_val = 2.0 / n
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -60, 60)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] > 1e-14:
                g += w[k] * _corr_grad(x, n, k)
        return g * dx_val
    
    def sum_con(x):
        return np.sum(x) - _TGT
    
    def sum_con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

def _corr_grad(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g

    if n_ones > target:
        idx = np.where(h == 1.0)[0]
        to_flip = rng.choice(idx, n_ones - target, replace=False)
        h[to_flip] = 0.0
    elif n_ones < target:
        idx = np.where(h == 0.0)[0]
        to_flip = rng.choice(idx, target - n_ones, replace=False)
        h[to_flip] = 1.0
    return h


def sa_search(h, n, rng, time_limit=30.0, t0=0.5, t1=0.0003, cooling=0.9997, n_cands=35):
    h = fix_binary_count(h, n, rng)
    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx_val
    best_h = h.copy()

    start = time.time()
    t = t0
    stagnation = 0

    while time.time() - start < time_limit and remaining() > 5:
        t *= cooling
        if t < t1:
            t = t1

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        k_max = int(np.argmax(corr))
        offset = k_max - (n - 1)

        best_delta = float('inf')
        best_i = -1
        best_j = -1

        for _ in range(n_cands):
            if offset >= 0 and offset < n:
                valid = ones[ones + offset < n]
                if len(valid) > 0:
                    contrib = valid[h[valid + offset] == 0.0]
                else:
                    contrib = np.array([], dtype=int)
                if len(contrib) > 0 and rng.rand() < 0.65:
                    i = contrib[rng.randint(len(contrib))]
                else:
                    i = ones[rng.randint(len(ones))]
            elif offset < 0 and offset > -n:
                ao = -offset
                valid = ones[ones >= ao]
                if len(valid) > 0:
                    contrib = valid[h[valid - ao] == 0.0]
                else:
                    contrib = np.array([], dtype=int)
                if len(contrib) > 0 and rng.rand() < 0.65:
                    i = contrib[rng.randint(len(contrib))]
                else:
                    i = ones[rng.randint(len(ones))]
            else:
                i = ones[rng.randint(len(ones))]

            j = zeros[rng.randint(len(zeros))]

            h[i] = 0.0
            h[j] = 1.0
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            h[i] = 1.0
            h[j] = 0.0

            delta = new_max - cur_max
            if delta < best_delta:
                best_delta = delta
                best_i = i
                best_j = j

        if best_i < 0:
            break

        if best_delta <= 0 or rng.rand() < np.exp(-best_delta / max(t, 1e-12)):
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = compute_corr(h)
            cur_max = float(np.max(corr))
            if cur_max * dx_val < best_c5 - 1e-14:
                best_c5 = cur_max * dx_val
                best_h = h.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1

        if stagnation > 5000:
            break

    return best_h, best_c5


def greedy_search(h, n, rng, time_limit=30.0, n_cands=80):
    h = fix_binary_count(h, n, rng)
    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx_val
    best_h = h.copy()

    start = time.time()
    stall = 0

    while time.time() - start < time_limit and remaining() > 3:
        k_max = int(np.argmax(corr))
        offset = k_max - (n - 1)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

        if len(ones) == 0 or len(zeros) == 0:
            break

        found = False
        for _ in range(n_cands):
            if offset >= 0 and offset < n:
                valid = ones[ones + offset < n]
                if len(valid) > 0:
                    contrib = valid[h[valid + offset] == 0.0]
                else:
                    contrib = np.array([], dtype=int)
                if len(contrib) > 0 and rng.rand() < 0.75:
                    i = contrib[rng.randint(len(contrib))]
                else:
                    i = ones[rng.randint(len(ones))]
            elif offset < 0 and offset > -n:
                ao = -offset
                valid = ones[ones >= ao]
                if len(valid) > 0:
                    contrib = valid[h[valid - ao] == 0.0]
                else:
                    contrib = np.array([], dtype=int)
                if len(contrib) > 0 and rng.rand() < 0.75:
                    i = contrib[rng.randint(len(contrib))]
                else:
                    i = ones[rng.randint(len(ones))]
            else:
                i = ones[rng.randint(len(ones))]

            j = zeros[rng.randint(len(zeros))]

            h[i] = 0.0
            h[j] = 1.0
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            h[i] = 1.0
            h[j] = 0.0

            if new_max < cur_max - 1e-14:
                h[i] = 0.0
                h[j] = 1.0
                corr = new_corr
                cur_max = new_max
                if cur_max * dx_val < best_c5:
                    best_c5 = cur_max * dx_val
                    best_h = h.copy()
                found = True
                break

        if found:
            stall = 0
        else:
            stall += 1
            if stall > 80:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                corr = compute_corr(h)
                cur_max = float(np.max(corr))
                stall = 0

    return best_h, best_c5


def _slsqp_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def _corr_grad_one(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g


def _slsqp_grad(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-14:
            g += w[k] * _corr_grad_one(x, n, k)
    return g * dx_val


def _slsqp_sum_con(x):
    return np.sum(x) - _TGT


def _slsqp_sum_jac(x):
    return np.ones(_N)


def slsqp_polish(h, n, beta, maxiter=300):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_sum_con, 'jac': _slsqp_sum_jac}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def itls_search(h, n, rng, time_limit=60.0):
    """Iterated local search: greedy + random perturbation + greedy."""
    h = fix_binary_count(h, n, rng)
    dx_val = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()

    start = time.time()
    while time.time() - start < time_limit and remaining() > 5:
        h_g, c5_g = greedy_search(h, n, rng, time_limit=min(8, remaining() * 0.2), n_cands=100)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
            h = h_g.copy()

        # Perturbation: random k-swap
        k_swap = rng.randint(2, 6)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < k_swap or len(zeros) < k_swap:
            break
        flip_ones = rng.choice(ones, k_swap, replace=False)
        flip_zeros = rng.choice(zeros, k_swap, replace=False)
        h[flip_ones] = 0.0
        h[flip_zeros] = 1.0

    # Final greedy
    h_g, c5_g = greedy_search(h, n, rng, time_limit=min(10, remaining() * 0.2), n_cands=120)
    if c5_g < best_c5:
        best_c5 = c5_g
        best_h = h_g.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 320
        parent_h = make_thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Intensive refinement of parent
    n = best_n
    h_bin = fix_binary_count(best_h, n, rng)

    # SA round 1
    h_sa, c5_sa = sa_search(h_bin, n, rng, time_limit=min(100, remaining() * 0.12),
                            t0=0.8, t1=0.0002, cooling=0.9996, n_cands=40)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()
        best_n = n

    # Greedy polish
    h_g, c5_g = greedy_search(h_sa, n, rng, time_limit=min(40, remaining() * 0.06), n_cands=120)
    if c5_g < best_c5:
        best_c5 = c5_g
        best_h = h_g.copy()
        best_n = n

    # Phase 2: Explore multiple resolutions with diverse starts
    candidate_ns = set()
    for base in [192, 224, 256, 288, 320, 352, 384, 416, 448, 480, 512, 544, 576, 608]:
        if 160 <= base <= 640:
            candidate_ns.add(base)
    candidate_ns.add(best_n)
    candidate_ns = sorted(candidate_ns)
    # Sort by proximity to current best resolution
    candidate_ns.sort(key=lambda x: (abs(x - best_n), x))

    for n_t in candidate_ns:
        if remaining() < 25:
            break
        if n_t == best_n:
            continue

        # Build diverse initializations
        inits = []
        inits.append(make_thue_morse(n_t))
        inits.append(make_random_binary(n_t, rng))
        inits.append(make_random_binary(n_t, rng))

        # Interpolated from current best
        xp = (np.arange(best_n) + 0.5) * 2.0 / best_n
        x_new = (np.arange(n_t) + 0.5) * 2.0 / n_t
        h_interp = np.interp(x_new, xp, best_h)
        h_interp_bin = (h_interp > 0.5).astype(float)
        inits.append(fix_binary_count(h_interp_bin, n_t, rng))

        best_at_n = float('inf')
        best_h_at_n = None

        for h_init in inits:
            if remaining() < 12:
                break
            # Quick greedy to assess
            h_q, c5_q = greedy_search(h_init.copy(), n_t, rng,
                                      time_limit=min(4, max(1, remaining() * 0.02)), n_cands=60)
            if c5_q < best_at_n:
                best_at_n = c5_q
                best_h_at_n = h_q
            if c5_q < best_c5:
                # Promising! Do SA
                h_sa, c5_sa = sa_search(h_q, n_t, rng,
                                        time_limit=min(25, max(5, remaining() * 0.08)),
                                        t0=0.5, t1=0.0003, cooling=0.9997, n_cands=40)
                if c5_sa < best_at_n:
                    best_at_n = c5_sa
                    best_h_at_n = h_sa
                # Greedy after SA
                h_g2, c5_g2 = greedy_search(h_sa, n_t, rng,
                                            time_limit=min(10, max(3, remaining() * 0.04)),
                                            n_cands=100)
                if c5_g2 < best_at_n:
                    best_at_n = c5_g2
                    best_h_at_n = h_g2

        if best_at_n < best_c5 and best_h_at_n is not None:
            best_c5 = best_at_n
            best_h = best_h_at_n.copy()
            best_n = n_t

        if remaining() < 100:
            break

    # Phase 3: Intensive refinement on best found
    n = best_n
    h = fix_binary_count(best_h, n, rng)

    # Multiple SA with different temperatures
    for t0, t1, cool in [(1.5, 0.0001, 0.9998), (0.5, 0.0002, 0.9997),
                         (0.2, 0.0002, 0.9996), (0.05, 0.0001, 0.9995)]:
        if remaining() < 15:
            break
        h_sa, c5_sa = sa_search(h, n, rng, time_limit=min(60, remaining() * 0.12),
                                t0=t0, t1=t1, cooling=cool, n_cands=50)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n
            h = h_sa.copy()

    # ITLS on best
    if remaining() > 30:
        h_itls, c5_itls = itls_search(h, n, rng, time_limit=min(60, remaining() * 0.2))
        if c5_itls < best_c5:
            best_c5 = c5_itls
            best_h = h_itls.copy()
            best_n = n
            h = h_itls.copy()

    # Final aggressive greedy
    if remaining() > 10:
        h_g, c5_g = greedy_search(h, n, rng, time_limit=min(40, remaining() * 0.2), n_cands=200)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
            best_n = n

    # Continuous polishing
    if remaining() > 20:
        n = best_n
        h_cont = best_h.copy()
        for beta in [200, 800, 3000, 10000]:
            if remaining() < 8:
                break
            h_new = slsqp_polish(h_cont, n, beta, maxiter=250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_cont = h_new.copy()

        # Re-binary and final push
        h_bin2 = fix_binary_count(best_h, n, rng)
        h_g, c5_g = greedy_search(h_bin2, n, rng, time_limit=min(15, remaining() * 0.3), n_cands=150)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
            best_n = n

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
