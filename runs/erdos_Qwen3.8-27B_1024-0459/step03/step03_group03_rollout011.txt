import numpy as np
from scipy.optimize import minimize
import time

_t0 = 0.0
_tb = 1300.0
_ng = 0
_bv = 100.0


def _tl():
    return max(0.0, _tb - (time.time() - _t0) - 3.0)


def _c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * (2.0 / n)


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _proj(h, n):
    h = np.array(h, dtype=np.float64)
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(30):
        d = 0.5 * n - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        k = int(np.sum(m))
        if k > 0:
            h[m] += d / k
        else:
            h[:] = 0.5
        np.clip(h, 0.0, 1.0, out=h)
    return h


def _interp(h, ns, nd):
    xs = (np.arange(ns) + 0.5) * (2.0 / ns)
    xd = (np.arange(nd) + 0.5) * (2.0 / nd)
    return np.interp(xd, xs, h).astype(np.float64)


def _exhaustive_1swap(h, n, max_passes=300):
    best = h.copy()
    bc = _c5(best)
    for p in range(max_passes):
        if _tl() < 3:
            break
        hi = np.where(best > 0.5)[0]
        lo = np.where(best < 0.5)[0]
        if len(hi) == 0 or len(lo) == 0:
            break
        improved = False
        for i in hi:
            if best[i] < 0.5:
                continue
            for j in lo:
                if best[j] >= 0.5:
                    continue
                if i == j:
                    continue
                h2 = best.copy()
                tmp = h2[i]
                h2[i] = h2[j]
                h2[j] = tmp
                h2 = _proj(h2, n)
                c2 = _c5(h2)
                if c2 < bc - 1e-15:
                    best = h2
                    bc = c2
                    improved = True
                    hi = np.where(best > 0.5)[0]
                    lo = np.where(best < 0.5)[0]
                    break
        if not improved:
            break
    return best, bc


def _random_2swap(h, n, nit, rng):
    best = h.copy()
    bc = _c5(best)
    for _ in range(nit):
        if _tl() < 2:
            break
        hi = np.where(best > 0.5)[0]
        lo = np.where(best < 0.5)[0]
        if len(hi) < 2 or len(lo) < 2:
            break
        i1 = int(hi[rng.randint(len(hi))])
        i2 = int(hi[rng.randint(len(hi))])
        j1 = int(lo[rng.randint(len(lo))])
        j2 = int(lo[rng.randint(len(lo))])
        if i1 == i2 or j1 == j2:
            continue
        if i1 == j1 or i1 == j2 or i2 == j1 or i2 == j2:
            continue
        h2 = best.copy()
        h2[i1], h2[i2], h2[j1], h2[j2] = h2[j1], h2[j2], h2[i1], h2[i2]
        h2 = _proj(h2, n)
        c2 = _c5(h2)
        if c2 < bc - 1e-15:
            best = h2
            bc = c2
    return best, bc


def _block_shift(h, n, rng, nit):
    best = h.copy()
    bc = _c5(best)
    for _ in range(nit):
        if _tl() < 2:
            break
        length = rng.randint(1, min(6, n // 4))
        start = rng.randint(0, n - length)
        shift = rng.choice([-2, -1, 1, 2])
        new_start = start + shift
        if new_start < 0 or new_start + length > n:
            continue
        h2 = best.copy()
        block = h2[start:start + length].copy()
        h2[start:start + length] = 0.0
        h2[new_start:new_start + length] = block
        h2 = _proj(h2, n)
        c2 = _c5(h2)
        if c2 < bc - 1e-15:
            best = h2
            bc = c2
    return best, bc


def _scon(x):
    return x.sum() - 0.5 * _ng


def _sobj(x):
    n = _ng
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = float(np.max(c))
    b = _bv
    sh = c - mx
    return (mx + np.log(np.exp(b * sh).sum()) / b) * (2.0 / n)


def _slsqp(h0, n, beta, maxiter):
    global _ng, _bv
    _ng = n
    _bv = beta
    try:
        r = minimize(_sobj, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon}],
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return _proj(np.asarray(r.x, dtype=np.float64), n)
    except Exception:
        return _proj(h0.copy(), n)


def _exact_grad(h, kmax):
    n = len(h)
    g = np.zeros(n)
    s = kmax - (n - 1)
    if s >= 0:
        for i in range(n):
            dc = 0.0
            if i < n - s:
                dc += (1.0 - h[i + s])
            if s <= i:
                dc -= h[i - s]
            g[i] = dc
    else:
        p = -s
        for i in range(n):
            dc = 0.0
            if i >= p:
                dc += (1.0 - h[i - p])
            if i < n - p:
                dc -= h[i + p]
            g[i] = dc
    return g


def _grad_descent(h, n, maxiter):
    best = h.copy()
    bc = _c5(best)
    for _ in range(maxiter):
        if _tl() < 3:
            break
        corr = _corr(best)
        mx = float(np.max(corr))
        kmax = int(np.argmax(corr))
        g = _exact_grad(best, kmax)
        g -= g.mean()
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            # Try soft-max gradient
            b = 200.0
            w = np.exp(b * (corr - mx))
            sw = float(w.sum())
            if sw < 1e-30:
                break
            w /= sw
            g = np.zeros(n)
            nz = np.where(w > 1e-8)[0]
            for k in nz:
                g += w[int(k)] * _exact_grad(best, int(k))
            g -= g.mean()
            gn = float(np.linalg.norm(g))
            if gn < 1e-15:
                break
        g /= gn
        moved = False
        for a in [0.04, 0.02, 0.01, 0.005, 0.002, 0.06, 0.03, 0.015]:
            h2 = _proj(best - a * g, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-15:
                best = h2
                bc = c2
                moved = True
                break
        if not moved:
            break
    return best, bc


def _frac_cd(h, n, nit, rng):
    best = h.copy()
    bc = _c5(best)
    for _ in range(nit):
        if _tl() < 2:
            break
        idx = int(rng.randint(n))
        jdx = int((idx + 1) % n)
        old = best[idx]
        improved = False
        for delta in [0.01, -0.01, 0.02, -0.02, 0.005, -0.005, 0.015, -0.015]:
            nv = old + delta
            if nv < 0.0 or nv > 1.0:
                continue
            h2 = best.copy()
            h2[idx] = nv
            h2[jdx] -= delta
            h2 = _proj(h2, n)
            c2 = _c5(h2)
            if c2 < bc - 1e-15:
                best = h2
                bc = c2
                improved = True
                break
        if not improved:
            continue
    return best, bc


def _kick_reopt(h, n, rng, n_kicks=3):
    best = h.copy()
    bc = _c5(best)
    for _ in range(n_kicks):
        if _tl() < 20:
