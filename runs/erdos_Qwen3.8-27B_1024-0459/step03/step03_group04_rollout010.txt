    lo, hi = -2.0, 3.0

    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return h

def interp_rescale(src, src_n, dst_n, target):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) / src_n * 2.0
    xq = (np.arange(dst_n) + 0.5) / dst_n * 2.0
    vals = np.interp(xq, xp, src)
    return project_feasible(vals, dst_n / 2.0)

def logsumexp_max(corr, beta):
    m = np.max(corr)
    if beta < 1e-6:
        return m
    scaled = beta * (corr - m)
    return m + np.log(np.sum(np.exp(scaled))) / beta

            best_h = h_try.copy()
            best_n = n_try
            print("  Interp n={}: C5={:.8f}".format(n_try, c5_try))
    
    # Phase 2: Generate diverse initializations
    inits = [(best_h.copy(), best_n, "parent")]
    
    for n_try in [160, 200, 240, 320]:
        tm = make_thue_morse(n_try)
        tm = project_feasible(tm, n_try)
        c5_tm, _ = compute_c5(tm)
        inits.append((tm, n_try, "thue_morse_{}".format(n_try)))
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = tm.copy()
            best_n = n_try
            print("  Thue-Morse n={}: C5={:.8f}".format(n_try, c5_tm))
        
        rb = make_random_binary(n_try, np.random)
        rb = project_feasible(rb, n_try)
        inits.append((rb, n_try, "random_{}".format(n_try)))
    
    # Sine-based
    n_s = best_n
    t_arr = np.arange(n_s) / float(n_s)
    for freq in [5, 7, 11]:
        for ph in [0.0, np.pi / 4, np.pi / 2, 3 * np.pi / 4]:
            hs = 0.5 + 0.45 * np.sin(2.0 * np.pi * freq * t_arr + ph)
            hs = project_feasible(hs, n_s)
            inits.append((hs, n_s, "sine_f{}_p{}".format(freq, int(ph * 4))))
    
    # Phase 3: Binary SA on best resolution
    if time.time() - t0 < budget * 0.7:
        for (h_init, n_init, name) in inits[:8]:
            if time.time() - t0 > budget * 0.35:
                break
            t_phase_start = time.time()
            sa_time = min(budget * 0.04, 60.0)
            h_sa, c5_sa = binary_sa(h_init, n_init, t0, t0 + sa_time, max_iters=8000)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_init
                print("  SA {} n={}: C5={:.8f}".format(name, n_init, best_c5))
    
    # Phase 4: Main gradient optimization on top candidates
    # Sort inits by C5, take top ones
    scored_inits = []
    for (h_init, n_init, name) in inits:
        c5_i, _ = compute_c5(h_init)
        scored_inits.append((c5_i, h_init, n_init, name))
    scored_inits.sort(key=lambda x: x[0])
    # Also include SA results
    scored_inits.append((best_c5, best_h.copy(), best_n, "best_so_far"))
    
    # Deduplicate: keep at most 5 different (n, c5) pairs
    seen = set()
    top_inits = []
    for c5_i, h_i, n_i, name_i in scored_inits:
        key = (n_i, round(c5_i, 6))
        if key not in seen:
            seen.add(key)
            top_inits.append((c5_i, h_i, n_i, name_i))
        if len(top_inits) >= 5:
            break
    
    grad_time_limit = time.time() + (budget * 0.55)
    for c5_i, h_i, n_i, name_i in top_inits:
        if time.time() > grad_time_limit:
            break
        if c5_i >= best_c5 + 0.005:
            continue  # Too far behind
        t_grad_start = time.time()
        grad_budget = min(grad_time_limit - t_grad_start, 400.0)
        if grad_budget < 30:
            break
        h_opt, c5_opt = projected_gradient_optimize(h_i, n_i, t_grad_start, t_grad_start + grad_budget, max_iters=30000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_i
            print("  Grad {} n={}: C5={:.8f}".format(name_i, n_i, best_c5))
    
    # Phase 5: Coordinate descent polish
    if time.time() - t0 < budget * 0.85:
        cd_time = min(budget * 0.12, 150.0)
        t_cd = time.time()
        best_h, best_c5 = coordinate_descent_polish(best_h, best_n, t0, t0 + cd_time, max_iters=100000)
        print("  After CD: n={}, C5={:.8f}".format(best_n, best_c5))
    
    # Phase 6: SLSQP polish
    if time.time() - t0 < budget * 0.92:
        sl_time = min(budget * 0.06, 60.0)
        best_h, best_c5 = slsqp_polish(best_h, best_n, t0, t0 + sl_time)
        print("  After SLSQP: n={}, C5={:.8f}".format(best_n, best_c5))
    
    # Phase 7: Try interpolation to nearby resolutions
    if time.time() - t0 < budget * 0.96:
        for dn in [-10, -5, 5, 10, 20, -20, 30, -30, 40, -40]:
            nt = best_n + dn
            if nt < 80 or nt > 500:
                continue
            ht = interp_rescale(best_h, best_n, nt)
            ct, _ = compute_c5(ht)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()
                best_n = nt
                print("  Res n={}: C5={:.8f}".format(nt, best_c5))
    
    # Phase 8: Final micro-polish
    if time.time() - t0 < budget * 0.98:
        h_final = best_h.copy()
        n_final = best_n
        for d in [0.0005, 0.001, 0.002, 0.005]:
            for _ in range(200):
                if time.time() - t0 > budget * 0.98:
                    break
                i = np.random.randint(n_final)
                j = np.random.randint(n_final)
                if i == j:
                    continue
                if h_final[i] >= d and h_final[j] <= 1.0 - d:
                    h_final[i] -= d
                    h_final[j] += d
                    c5_t, _ = compute_c5(h_final)
                    if c5_t < best_c5:
                        best_c5 = c5_t
                        best_h = h_final.copy()
                    else:
                        h_final[i] += d
                        h_final[j] -= d
                if h_final[j] >= d and h_final[i] <= 1.0 - d:
                    h_final[j] -= d
                    h_final[i] += d
                    c5_t, _ = compute_c5(h_final)
                    if c5_t < best_c5:
                        best_c5 = c5_t
                        best_h = h_final.copy()
                    else:
                        h_final[j] += d
                        h_final[i] -= d
        best_h = project_feasible(best_h, best_n)
    
    # Final evaluation
    final_c5, _ = compute_c5(best_h)
    if final_c5 > best_c5:
        best_c5 = final_c5
    
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < best_c5:
            best_c5 = ec5
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.8f}".format(best_n, best_c5))
    return (best_h, best_c5, best_n)
