def subgradient_step(h, n, top_k=5):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    top_indices = np.argsort(corr)[-top_k:]
    top_indices = top_indices[np.argsort(corr[top_indices])[::-1]]
    grad = np.zeros(n)
    for k_idx in top_indices:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            m = n - offset
            if m <= 0:
                continue
            grad[offset:offset + m] += (1.0 - h[:m])
            grad[:m] -= h[offset:offset + m]
        else:
            a = -offset
            m = n - a
            if m <= 0:
                continue
            grad[:m] += (1.0 - h[a:a + m])
            grad[a:a + m] -= h[:m]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return np.zeros(n), False
    return grad / gnorm, True

            print("    After coord: C5={:.8f}".format(c5_cur))

        if c5_cur < best_c5 - 1e-10:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n

    # Phase 3: Binary + continue
    print("\n=== Phase 3: Binary refinement ===")
    if remaining_time() > 25.0:
        n = best_n
        h_bin, c5_bin = binary_search(best_h, n, rng, max_iters=150)
        print("  Binary C5: {:.8f}".format(c5_bin))

        if c5_bin < best_c5:
            h_start = h_bin
            c5_start = c5_bin
        else:
            h_start = best_h.copy()
            c5_start = best_c5

        # Continuous refinement from best of (binary, continuous)
        for (b, tk, mi) in [(100, 6, 300), (400, 10, 400), (1500, 15, 400)]:
            if remaining_time() < 10.0:
                break
            h_new, c5_new = run_slsqp_topk(h_start, n, b, tk, mi)
            if c5_new < c5_start - 1e-13:
                h_start = h_new
                c5_start = c5_new

        if c5_start < best_c5 - 1e-10:
            best_c5 = c5_start
            best_h = h_start.copy()

    # Phase 4: Higher resolution attempt
    print("\n=== Phase 4: Higher resolution ===")
    if remaining_time() > 50.0:
        n_high = best_n + 128
        if n_high % 2 != 0:
            n_high += 1
        n_high = min(n_high, 768)

        if n_high > best_n:
            print("  Trying n={}".format(n_high))
            h_high = project_feasible(interp_to(best_h, best_n, n_high), n_high)
            c5_high = compute_c5(h_high)

            for (b, tk, mi) in [(50, 5, 300), (200, 8, 400), (600, 12, 400), (1500, 15, 400)]:
                if remaining_time() < 12.0:
                    break
                h_high, c5_high = run_slsqp_topk(h_high, n_high, b, tk, mi)

            if remaining_time() > 20.0:
                h_high, c5_high = subgradient_phase(h_high, n_high, rng, max_iters=60)

            if remaining_time() > 10.0:
                h_high, c5_high = coord_descent(h_high, n_high, rng, n_sweeps=15)

            if c5_high < best_c5 - 1e-10:
                best_c5 = c5_high
                best_h = h_high.copy()
                best_n = n_high
                print("  High-res improved: n={}, C5={:.8f}".format(best_n, best_c5))

    # Phase 5: Final fine-tuning
    print("\n=== Phase 5: Final polish ===")
    if remaining_time() > 15.0:
        best_h, best_c5 = coord_descent(best_h, best_n, rng, n_sweeps=20)

    if remaining_time() > 10.0:
        h_new, c5_new = run_slsqp_topk(best_h, best_n, 3000, 10, 500)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new

    if remaining_time() > 8.0:
        best_h, best_c5 = coord_descent(best_h, best_n, rng, n_sweeps=10)

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
