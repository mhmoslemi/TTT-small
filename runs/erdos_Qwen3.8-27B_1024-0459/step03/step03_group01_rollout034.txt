The gradient of corr[k] w.r.t. h is:
For shift off = k - (n-1):
- If off >= 0: g[i] += (1-h[i+off]) for i in 0..n-1-off; g[i+off] -= h[i] for i in 0..n-1-off
  Wait: corr[off] = sum_i h[i]*(1-h[i+off]). d/dh[i] = (1-h[i+off]) - h[i]*d(h[i+off])/dh[i]... 

Actually, corr[off] = sum_{i=0}^{n-1-off} h[i] * (1-h[i+off]).
d(corr[off])/dh[j] = (1-h[j+off]) for j in 0..n-1-off (from the h[j] term)
                   + sum_{i: i+off=j} h[i]*(-1) for j in off..n-1 (from the -(h[j]) term)
                   = (1-h[j+off]) - (h[j-off] if j>=off else 0) for j in 0..n-1

More carefully:
d/dh[j] [sum_{i=0}^{n-1-off} h[i]*(1-h[i+off])]
= (1-h[j+off]) * [j <= n-1-off] - [j >= off] * h[j-off]

Where [condition] is 1 if true, 0 otherwise.

So: g[j] = (1-h[j+off]) * 1{j ≤ n-1-off} - h[j-off] * 1{j ≥ off}

For off > 0:
- j = 0 to n-1-off: g[j] = (1-h[j+off]) - (0 since j < off for j < off, else h[j-off])
  Actually: for j in [0, n-1-off): contributes (1-h[j+off])
  For j in [off, n-1]: contributes -h[j-off]
  For j in [0, off): only first term applies (1-h[j+off])
  For j in [off, n-1-off]: both terms: (1-h[j+off]) - h[j-off]
  For j in [n-off, n-1]: only second term: -h[j-off]

```python
def corr_gradient(h, n, shift):
    """Gradient of correlation at given shift (0-indexed in correlate output)."""
    g = np.zeros(n)
    off = shift - (n - 1)
    if off >= 0:
        npairs = n - off
        for i in range(npairs):
            g[i] += (1.0 - h[i + off])
            g[i + off] -= h[i]
    else:
        ao = -off
        npairs = n - ao
        for i in range(npairs):
            g[i + ao] += (1.0 - h[i])
            g[i] -= h[i + ao]
    return g
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def corr_grad_single(h, n, shift_idx):
    g = np.zeros(n)
    off = shift_idx - (n - 1)
    if off >= 0:
        npairs = n - off
        g[:npairs] += (1.0 - h[off:off + npairs])
        g[off:off + npairs] -= h[:npairs]
    else:
        ao = -off
        npairs = n - ao
        g[ao:ao + npairs] += (1.0 - h[:npairs])
        g[:npairs] -= h[ao:ao + npairs]
    return g


def slsqp_obj(x):
    n = _SLSQP_N
    beta = _SLSQP_BETA
    dx = 2.0 / n
    corr = compute_corr(x)
    m = float(np.max(corr))
    shifted = corr - m
    cl = np.clip(beta * shifted, -50.0, 50.0)
    ex = np.exp(cl)
    s = np.sum(ex)
    return (m + np.log(s) / beta) * dx


def slsqp_grad(x):
    n = _SLSQP_N
    beta = _SLSQP_BETA
    dx = 2.0 / n
    corr = compute_corr(x)
    m = float(np.max(corr))
    shifted = corr - m
    cl = np.clip(beta * shifted, -50.0, 50.0)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk > 1e-14:
            g += wk * corr_grad_single(x, n, k)
    return g * dx


def slsqp_con_fun(x):
    return np.sum(x) - _SLSQP_TGT


def slsqp_con_jac(x):
    return np.ones(_SLSQP_N)


def slsqp_run(h, n, beta, maxiter=400):
    global _SLSQP_N, _SLSQP_BETA, _SLSQP_TGT
    _SLSQP_N = n
    _SLSQP_BETA = beta
    _SLSQP_TGT = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con_fun, 'jac': slsqp_con_jac}]
    try:
        res = minimize(slsqp_obj, h, jac=slsqp_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

    if np.min(h) < 0.01 and np.max(h) > 0.99:
        h = project_feasible(h, n)

    for beta in [100, 300, 1000, 3000, 10000]:
        if _BUDGET - (time.time() - _START) < 80:
            break
        h_new = slsqp_run(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Active set
    if _BUDGET - (time.time() - _START) > 50:
        h_eq, c5_eq = active_set_refine(h, n, max_iters=500,
                                        time_limit=min(40, _BUDGET - (time.time() - _START) - 60))
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 4: Subgradient continuous optimization
    if _BUDGET - (time.time() - _START) > 100:
        sub_limit = min(200, _BUDGET - (time.time() - _START) - 80)
        h_sg, c5_sg = subgradient_optimize(h, n, sub_limit, rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 5: Try different resolutions
    if _BUDGET - (time.time() - _START) > 150:
        n_tries = []
        for factor in [0.5, 0.75, 1.5, 2.0]:
            nt = int(n_main * factor)
            nt -= nt % 2
            if 80 <= nt <= 640:
                n_tries.append(nt)
        n_tries = sorted(set(n_tries))

        for n_t in n_tries:
            if _BUDGET - (time.time() - _START) < 80:
                break
            h_interp = interp_to(best_h, best_n, n_t)
            h_feas = project_feasible(h_interp, n_t)
            c5_int = compute_c5(h_feas)

            # SLSQP at this resolution
            h_opt = h_feas
            for beta in [300, 1500, 5000]:
                if _BUDGET - (time.time() - _START) < 30:
                    break
                h_opt = slsqp_run(h_opt, n_t, beta, maxiter=300)
            c5_opt = compute_c5(h_opt)

            # Binary ILS at this resolution
            if _BUDGET - (time.time() - _START) > 60:
                h_bin = to_binary(h_opt, n_t, rng)
                h_b, c5_b = binary_ils(h_bin, n_t, rng,
                                       min(40, int(_BUDGET - (time.time() - _START) - 40)))
                if c5_b < c5_opt:
                    c5_opt = c5_b
                    h_opt = h_b

            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_t

    # Phase 6: Final intensive refinement on best
    n = best_n
    h = best_h.copy()
    for beta in [5000, 15000, 50000]:
        if _BUDGET - (time.time() - _START) < 30:
            break
        h_new = slsqp_run(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if _BUDGET - (time.time() - _START) > 30:
        h_eq, c5_eq = active_set_refine(h, n, max_iters=800,
                                        time_limit=min(25, int(_BUDGET - (time.time() - _START) - 20)))
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Final binary fine-tune
    if _BUDGET - (time.time() - _START) > 40:
        h_bin = to_binary(best_h, best_n, rng)
        h_b, c5_b = binary_ils(h_bin, best_n, rng,
                               min(35, int(_BUDGET - (time.time() - _START) - 15)))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Final SLSQP polish
    if _BUDGET - (time.time() - _START) > 20:
        n = best_n
        for beta in [10000, 30000]:
            if _BUDGET - (time.time() - _START) < 10:
                break
            h_new = slsqp_run(best_h, n, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
```
