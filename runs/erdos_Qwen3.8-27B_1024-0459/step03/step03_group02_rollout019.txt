def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):

def project_feasible(h, n):

def smooth_max_grad(x):

def lag_gradient(h, n, offset):
    """Gradient of Σ_i h[i](1-h[i+shift]) w.r.t. h[shift>=0]"""
    grad = np.zeros(n)
    if offset >= 0:
        n_pairs = n - offset
        if n_pairs <= 0:
            return grad
        # corr = sum_{i=0}^{n-1-offset} h[i]*(1-h[i+offset])
        d/dh[i] = (1-h[i+offset]) for i in [0, n-1-offset]
        d/dh[i+offset] = -h[i] for i in [0, n-1-offset]
        grad[:n_pairs] += (1.0 - h[offset:offset + n_pairs])
        grad[offset:offset + n_pairs] -= h[:n_pairs]
    else:
        abs_off = -offset
        n_pairs = n - abs_off
        if n_pairs <= 0:
            return grad
        # corr = sum_{i=abs_off}^{n-1} h[i]*(1-h[i-abs_off])
        # = sum_{j=0}^{n-1-abs_off} h[j+abs_off]*(1-h[j])
        d/dh[j+abs_off] = (1-h[j])
        d/dh[j] = -h[j+abs_off]
        grad[abs_off:abs_off + n_pairs] += (1.0 - h[:n_pairs])
        grad[:n_pairs] -= h[abs_off:abs_off + n_pairs]
    return grad

def top_k_subgrad(h, n, k_top):
    """Subgradient step focusing on top-k lags"""
    corr = compute_corr(h)
    max_corr = np.max(corr)
    # Find lags within a small gap of max
    threshold = max_corr - 0.1
    active = np.where(corr >= threshold)[0]
    if len(active) > k_top:
        # Keep only top k
        top_indices = np.argsort(corr)[::-1][:k_top]
        active = top_indices
    
    grad = np.zeros(n)
    for k_idx in active:
        offset = k_idx - (n - 1)
        grad += lag_gradient(h, n, offset)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return None
    return grad / gnorm
