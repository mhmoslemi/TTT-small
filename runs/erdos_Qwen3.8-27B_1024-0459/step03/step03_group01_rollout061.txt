    g = np.zeros(n)

        g[off:off + npairs] += (1.0 - x[:npairs])

        g[:npairs] += (1.0 - x[ao:ao + npairs])

import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-10:
            g += w[k] * corr_grad_per_shift(x, n, k)
    return g * dx_val


def sum_con(x):
    return np.sum(x) - _TGT


def sum_con_jac(x):
    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=300):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def ensure_binary_feasible(h, n, rng):
    """Ensure h is binary with exactly n/2 ones."""
    h = np.round(h).astype(float)
    target = n // 2
    n_ones = int(np.sum(h > 0.5))
    if n_ones > target:
        idx = np.where(h > 0.5)[0]
        rem = rng.choice(idx, n_ones - target, replace=False)
        h[rem] = 0.0
    elif n_ones < target:
        idx = np.where(h < 0.5)[0]
        add = rng.choice(idx, target - n_ones, replace=False)
        h[add] = 1.0
    return h


def sa_binary(h, n, time_limit, rng, T_init=3.0, T_final=0.05, multi_move_prob=0.25):
    """Simulated annealing on binary h."""
    h = ensure_binary_feasible(h, n, rng)
    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    best_c5 = cur_max * dx
    
    start = time.time()
    total_time = min(time_limit, max(remaining() - 5, 1.0))
    
    while time.time() - start < total_time and remaining() > 5:
        ones_idx = np.where(h > 0.5)[0]
        zeros_idx = np.where(h < 0.5)[0]
        if len(ones_idx) < 2 or len(zeros_idx) < 2:
            break
        
        frac = (time.time() - start) / total_time
        T = T_init * (T_final / T_init) ** min(frac, 1.0)
        
        if rng.random() < multi_move_prob and len(ones_idx) >= 2 and len(zeros_idx) >= 2:
            # 2-pair swap
            i1, i2 = rng.choice(ones_idx, 2, replace=False)
            j1, j2 = rng.choice(zeros_idx, 2, replace=False)
            h[i1] = 0.0; h[i2] = 0.0
            h[j1] = 1.0; h[j2] = 1.0
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            delta = new_max - cur_max
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                cur_max = new_max
                if new_max < best_max - 1e-12:
                    best_max = new_max
                    best_c5 = best_max * dx
                    best_h = h.copy()
            else:
                h[i1] = 1.0; h[i2] = 1.0
                h[j1] = 0.0; h[j2] = 0.0
        else:
            # Single pair swap
            i = ones_idx[rng.randint(len(ones_idx))]
            j = zeros_idx[rng.randint(len(zeros_idx))]
            h[i] = 0.0
            h[j] = 1.0
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            delta = new_max - cur_max
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):

                    best_c5 = best_max * dx
                    best_h = h.copy()
            else:

                h[j] = 0.0

def thue_morse_binary(n, rng):
    """Thue-Morse sequence of length n."""
    h = np.zeros(n)
    for i in range(n):
        h[i] = float(bin(i).count('1') % 2)

def random_binary(n, rng):
    """Random binary sequence with n/2 ones."""
    idx = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h


def perturbation_binary(h, n, rng, n_flips=3):
    """Random perturbation of binary h."""
    h = h.copy()
    ones_idx = np.where(h > 0.5)[0]
    zeros_idx = np.where(h < 0.5)[0]

        h[j] = 1.0
        ones_idx = np.where(h > 0.5)[0]
        zeros_idx = np.where(h < 0.5)[0]
    return h

Alternative to SLSQP: **projected gradient descent** on the max correlation.

The gradient of max_k corr_k is the gradient of the active correlation(s). If multiple correlations tie for max, it's the average of their gradients (or we can pick the worst one).

```python
def projected_gradient_step(h, n, step_size, rng):
    corr = compute_corr(h)
    max_val = np.max(corr)
    # Find top correlations
    top_idx = np.where(np.abs(corr - max_val) < 0.5)[0]  # All near-max
    g = np.zeros(n)
    for k in top_idx:
        g += corr_grad_per_shift(h, n, k)
    g /= len(top_idx)
    
    # Projected gradient step
    h_new = h - step_size * g
    h_new = project_feasible(h_new, n)
    return h_new
```

    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def ensure_binary(h, n, rng):
    h = np.round(h).astype(float)
    target = n // 2
    n_ones = int(np.sum(h > 0.5))
    if n_ones > target:
        idx = np.where(h > 0.5)[0]
        rem = rng.choice(idx, n_ones - target, replace=False)
        h[rem] = 0.0
    elif n_ones < target:
        idx = np.where(h <= 0.5)[0]
        add = rng.choice(idx, target - n_ones, replace=False)
        h[add] = 1.0
    return h


def thue_morse_seq(n, rng):
    h = np.zeros(n)
    # Use vectorized popcount
    for i in range(n):
        h[i] = float(bin(i).count('1') & 1)
    return ensure_binary(h, n, rng)


def random_binary(n, rng):
    idx = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h


def perturb_binary(h, n, rng, n_flips=3):
    h = h.copy()
    for _ in range(n_flips):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
    return h


def sa_binary(h, n, time_limit, rng, T_init=3.0, T_final=0.05, multi_prob=0.2):
    h = ensure_binary(h, n, rng)
    dx = 2.0 / n
    cur_max = float(np.max(compute_corr(h)))
    best_h = h.copy()
    best_max = cur_max
    best_c5 = cur_max * dx

    start = time.time()
    total_time = min(time_limit, max(remaining() - 5.0, 1.0))
    if total_time < 1.0:
        return best_h, best_c5

    while time.time() - start < total_time and remaining() > 5.0:
        frac = (time.time() - start) / max(total_time, 1e-9)
        T = T_init * (T_final / T_init) ** min(frac, 1.0)

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break

        use_multi = (multi_prob > 0 and rng.random() < multi_prob
                     and len(ones) >= 2 and len(zeros) >= 2)

        if use_multi:
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            if i1 == i2 or j1 == j2:
                # Fall through to single
                use_multi = False
            elif i1 == j1 or i1 == j2 or i2 == j1 or i2 == j2:
                use_multi = False

        if use_multi:
            h[i1] = 0.0; h[i2] = 0.0
            h[j1] = 1.0; h[j2] = 1.0
            nm = float(np.max(compute_corr(h)))
            d = nm - cur_max
            if d <= 0 or rng.random() < np.exp(-d / max(T, 1e-10)):
                cur_max = nm
                if nm < best_max - 1e-12:
                    best_max = nm
                    best_c5 = best_max * dx
                    best_h = h.copy()
            else:
                h[i1] = 1.0; h[i2] = 1.0
                h[j1] = 0.0; h[j2] = 0.0
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            nm = float(np.max(compute_corr(h)))
            d = nm - cur_max
            if d <= 0 or rng.random() < np.exp(-d / max(T, 1e-10)):
                cur_max = nm
                if nm < best_max - 1e-12:
                    best_max = nm
                    best_c5 = best_max * dx
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 320
        parent_h = thue_morse_seq(parent_n, np.random.RandomState(0))
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: SLSQP on parent (40s)
    if remaining() > 40:
        for beta in [500, 2000]:
            if remaining() < 15:
                break
            h_try = slsqp_smooth(best_h, best_n, beta, maxiter=250)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Phase 2: SA on parent resolution (200s)
    if remaining() > 50:
        h_bin = ensure_binary(best_h, best_n, rng)
        # Multiple SA runs with different params
        for T_i in [5.0, 3.0, 1.5, 0.8]:
            if remaining() < 20:
                break
            t_limit = min(50, max(remaining() - 10, 5))
            h_sa, c5_sa = sa_binary(h_bin, best_n, t_limit, rng,
                                    T_init=T_i, T_final=0.03,
                                    multi_prob=0.25)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
            h_bin = h_sa.copy()  # Continue from last SA result

    # Phase 3: Multi-resolution (400s)
    resolutions = [n for n in [192, 224, 256, 288, 352, 384, 416, 448, 480, 512]
                   if n % 2 == 0 and 160 <= n <= 560]
    resolutions = [n for n in resolutions if n != best_n]
    # Sort by distance from best_n to explore nearby first
    resolutions.sort(key=lambda n: abs(n - best_n))

    for n_t in resolutions:
        if remaining() < 25:
            break
        
        # Candidates
        h_interp = project_feasible(interp_to(best_h, best_n, n_t), n_t)
        h_tm = thue_morse_seq(n_t, np.random.RandomState(rng.randint(100000)))
        h_rand = random_binary(n_t, np.random.RandomState(rng.randint(100000)))
        
        local_best = None
        local_c5 = float('inf')
        
        for cand in [h_interp, h_tm, h_rand]:
            if remaining() < 12:
                break
            
            # Quick SLSQP
            h_opt = slsqp_smooth(cand.copy(), n_t, 1000, maxiter=150)
            c5_opt = compute_c5(h_opt)
            
            # SA on binary
            h_bin = ensure_binary(h_opt, n_t, rng)
            t_sa = min(30, max(remaining() - 10, 5))
            h_sa, c5_sa = sa_binary(h_bin, n_t, t_sa, rng,
                                    T_init=2.5, T_final=0.03,
                                    multi_prob=0.25)
            
            c5_total = min(c5_opt, c5_sa)
            h_total = h_sa if c5_sa <= c5_opt else h_opt
            
            if c5_total < local_c5:
                local_c5 = c5_total
                local_best = h_total.copy()
        
        if local_best is not None and local_c5 < best_c5:
            best_c5 = local_c5
            best_h = local_best.copy()
            best_n = n_t

                                    multi_prob=0.3)
