        best_swap = None
        
        for lag in worst_idx:
            offset = int(lag) - center
            if offset == 0:
                continue
            if offset > 0:
                for i in range(n - offset):
                    if h[i] == 1 and h[i + offset] == 0:
                        h[i], h[i+offset] = h[i+offset], h[i]
                        c5_new = compute_c5(h)
                        h[i], h[i+offset] = h[i+offset], h[i]
                        if c5_new < best_c5 - 1e-12:
                            best_c5 = c5_new
                            best_swap = (i, i + offset)
            else:
                aoff = -offset
                for i in range(aoff, n):
                    if h[i] == 1 and h[i - aoff] == 0:
                        h[i], h[i-aoff] = h[i-aoff], h[i]
                        c5_new = compute_c5(h)
                        h[i], h[i-aoff] = h[i-aoff], h[i]
                        if c5_new < best_c5 - 1e-12:
                            best_c5 = c5_new
                            best_swap = (i, i - aoff)
        
        # Also try random swaps
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) > 0 and len(zeros) > 0:
            for _ in range(20):
                i = int(rng.choice(ones))
                j = int(rng.choice(zeros))
                h[i], h[j] = h[j], h[i]
                c5_new = compute_c5(h)
                h[i], h[j] = h[j], h[i]
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_swap = (i, j)
        
        if best_swap is None:
            break
        
        i, j = best_swap
        h[i], h[j] = h[j], h[i]
        c5_best = best_c5
    
    return h, c5_best
