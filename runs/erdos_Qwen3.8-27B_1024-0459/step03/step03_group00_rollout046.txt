            if c5_t < c5_best - 1e-12:
                c5_best = c5_t
                h_best = h.copy()
                any_imp = True
            else:
                h = h_best.copy()

        # Try structured moves: move "mass" from high-correlation regions
        if budget_fn() < 3.0:
            break
        for lag in top_lags[:2]:
            offset = lag - (n - 1)
            if offset >= 0:
                region_a = slice(0, n - offset)
                region_b = slice(offset, n)
            else:
                ao = -offset
                region_a = slice(0, n - ao)
                region_b = slice(ao, n)

            h_a = h[region_a].copy()
            h_b = h[region_b].copy()
            pair_vals = h_a * h_b  # These products contribute to overlap
            # Try reducing the largest products
            top_prod_idx = np.argsort(pair_vals)[-3:]
            for ti in top_prod_idx:
                if budget_fn() < 1.5:
                    break
                # Decrease h_a[ti] or increase h_b[ti]
                for direction in [-1, 1]:
                    h_trial = h.copy()
                    ia = region_a.start + ti if hasattr(region_a, 'start') else ti
                    ib = region_b.start + ti if hasattr(region_b, 'start') else ti
                    delta = 0.05 * direction
                    h_trial[ia] = np.clip(h_trial[ia] - delta, 0, 1)
                    h_trial[ib] = np.clip(h_trial[ib] + delta, 0, 1)
                    h_trial = project_feasible(h_trial, n)
                    c5_t = compute_c5(h_trial)
                    if c5_t < c5_best - 1e-12:
                        c5_best = c5_t
                        h_best = h_trial.copy()
                        h = h_trial.copy()
                        any_imp = True
                        break

        if not any_imp and rnd > 10:
            break

    return h_best, c5_best


def do_slsqp_stage(h, n, beta_schedule, budget_fn, maxiter_per_stage=300):
    """Multi-stage SLSQP with increasing beta."""
    c5_best = compute_c5(h)
    h_cur = h.copy()
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum}]

    for beta in beta_schedule:
        if budget_fn() < 8.0:
            break
        _G_BETA = float(beta)
        mi = maxiter_per_stage if beta < 500 else int(maxiter_per_stage * 0.6)
        try:
            res = minimize(obj_smooth, h_cur, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': mi, 'ftol': 1e-14, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-14:
                c5_best = c5_new
                h_cur = h_new
        except Exception:
            pass

    return h_cur, c5_best


def do_topk_targeted(h, n, budget_fn, beta_schedule, maxiter=200):
    """SLSQP using only top-k lag approximation for faster, more targeted optimization."""
    c5_best = compute_c5(h)
    h_cur = h.copy()
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum}]

    for beta in beta_schedule:
        if budget_fn() < 8.0:
            break
        _G_BETA = float(beta)
        try:
            res = minimize(obj_soft_max, h_cur, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})

            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-14:

            pass

    return h_cur, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    # Get parent solution
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        # Create a Thue-Morse based initial
        parent_h = np.zeros(parent_n)
        for i in range(parent_n):
            cnt = 0
            v = i
            while v:
                cnt ^= (v & 1)
                v >>= 1
            parent_h[i] = 1.0 if cnt == 1 else 0.0
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        print("No parent. Thue-Morse: n={}, C5={:.10f}".format(parent_n, parent_c5))

    best_h = parent_h.copy()
    best_c5 = parent_c5
    best_n = parent_n

    # === Phase 1: Quick SLSQP polish at parent resolution ===
    print("\n=== Phase 1: SLSQP polish at n={} ===".format(best_n))
    h1, c5_1 = do_slsqp_stage(best_h, best_n, [50, 150, 400, 1000], get_time_remaining, maxiter_per_stage=250)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    print("  After SLSQP: C5={:.10f}".format(best_c5))

    # === Phase 2: Bottleneck subgradient ===
    print("\n=== Phase 2: Bottleneck subgradient ===")
    h2, c5_2 = do_bottleneck_subgradient(best_h, best_n, get_time_remaining, rng, n_iter=400)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    print("  After subgradient: C5={:.10f}".format(best_c5))

    # === Phase 3: Try higher resolution n=512 ===
    if get_time_remaining() > 120.0:
        print("\n=== Phase 3: Higher resolution n=512 ===")
        n_hi = 512
        h_interp = project_feasible(interp_to(best_h, best_n, n_hi), n_hi)
        c5_hi = compute_c5(h_interp)
        print("  Interpolated: C5={:.10f}".format(c5_hi))

        h3, c5_3 = do_slsqp_stage(h_interp, n_hi, [30, 100, 300, 800, 2000], get_time_remaining, maxiter_per_stage=200)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = n_hi
        print("  After SLSQP @512: C5={:.10f}".format(best_c5))

        if get_time_remaining() > 30.0:
            h3b, c5_3b = do_bottleneck_subgradient(best_h, best_n, get_time_remaining, rng, n_iter=200)
            if c5_3b < best_c5:
                best_c5 = c5_3b
                best_h = h3b.copy()
            print("  After subgrad @512: C5={:.10f}".format(best_c5))

    # === Phase 4: Top-k targeted optimization ===
    if get_time_remaining() > 60.0:
        print("\n=== Phase 4: Top-k targeted ===")
        h4, c5_4 = do_topk_targeted(best_h, best_n, get_time_remaining, [100, 300, 800, 2000], maxiter=200)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
        print("  After topk: C5={:.10f}".format(best_c5))

    # === Phase 5: Swap-based local search ===
    if get_time_remaining() > 30.0:
        print("\n=== Phase 5: Swap local search ===")
        h5, c5_5 = do_swap_local_search(best_h, best_n, get_time_remaining, rng, n_rounds=60)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
        print("  After swaps: C5={:.10f}".format(best_c5))

    # === Phase 6: Coordinate descent on continuous values ===
    if get_time_remaining() > 20.0:
        print("\n=== Phase 6: Coordinate descent ===")
        c5_cd = best_c5
        h_cd = best_h.copy()
        n_cd = best_n
        for sweep in range(15):
            if get_time_remaining() < 3.0:
                break
            any_imp = False
            order = rng.permutation(n_cd)
            for i in order:
                if get_time_remaining() < 1.5:
                    break
                old = h_cd[i]
                # Try a few values
                for nv in [0.0, 1.0, 0.25, 0.5, 0.75, old + 0.1, old - 0.1]:
                    nv = np.clip(nv, 0.0, 1.0)
                    if abs(nv - old) < 1e-8:
                        continue
                    h_cd[i] = nv
                    h_cd = project_feasible(h_cd, n_cd)
                    c5_t = compute_c5(h_cd)
                    if c5_t < c5_cd - 1e-13:
                        c5_cd = c5_t
                        any_imp = True
                        break
                    else:
                        h_cd[i] = old
                        h_cd = project_feasible(h_cd, n_cd)
            if not any_imp:
                break
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        print("  After coord desc: C5={:.10f}".format(best_c5))

    # === Phase 7: Final SLSQP with very high beta ===
    if get_time_remaining() > 15.0:
        print("\n=== Phase 7: Final polish ===")
        h7, c5_7 = do_slsqp_stage(best_h, best_n, [3000, 8000, 15000], get_time_remaining, maxiter_per_stage=400)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()
        print("  Final SLSQP: C5={:.10f}".format(best_c5))

        # One more subgradient pass
        if get_time_remaining() > 10.0:
            h7b, c5_7b = do_bottleneck_subgradient(best_h, best_n, get_time_remaining, rng, n_iter=200)
            if c5_7b < best_c5:
                best_c5 = c5_7b
                best_h = h7b.copy()
            print("  Final subgrad: C5={:.10f}".format(best_c5))

    # === Phase 8: Try n=384 as alternative ===
    if get_time_remaining() > 80.0:
        print("\n=== Phase 8: Alternative n=384 ===")
        n_alt = 384
        h_alt = project_feasible(interp_to(best_h, best_n, n_alt), n_alt)
        c5_alt = compute_c5(h_alt)
        h8, c5_8 = do_slsqp_stage(h_alt, n_alt, [50, 200, 600, 1500], get_time_remaining, maxiter_per_stage=200)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()
            best_n = n_alt
        print("  n=384 result: C5={:.10f}".format(best_c5))

    # === Phase 9: Aggressive local refinement ===
    if get_time_remaining() > 15.0:
        print("\n=== Phase 9: Final local refinement ===")
        h9, c5_9 = do_swap_local_search(best_h, best_n, get_time_remaining, rng, n_rounds=40)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()
        h9b, c5_9b = do_bottleneck_subgradient(best_h, best_n, get_time_remaining, rng, n_iter=150)
        if c5_9b < best_c5:
            best_c5 = c5_9b
            best_h = h9b.copy()
        print("  Phase 9: C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if isinstance(eval_c5, (int, float)) and eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
