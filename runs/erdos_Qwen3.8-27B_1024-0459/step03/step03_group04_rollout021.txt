def compute_c5(h):
    """Compute C5 value for given h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def project_to_feasible(h, n):
    """Project h onto [0,1]^n with sum = n/2."""
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        # Adjust elements that can be adjusted
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(interior) > 0:
            h[interior] += diff / np.sum(interior)
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_local_search(h, n, max_rounds=100, improve_thresh=1e-12):
    """Local search on near-binary values. Try adjusting individual elements."""
    best_c5, _ = compute_c5(h)
    h = h.copy()
    
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            for delta in [-0.1, -0.05, -0.02, -0.01, 0.01, 0.02, 0.05, 0.1]:
                h_tr = h.copy()
                h_tr[i] += delta
                # Project to maintain constraints
                h_tr = project_to_feasible(h_tr, n)
                c5_tr, _ = compute_c5(h_tr)
                if c5_tr < best_c5 - improve_thresh:
                    best_c5 = c5_tr
                    h = h_tr
                    improved = True
                    break
        if not improved:

def compute_c5_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def project_h(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(interior)
        if cnt > 0:
            h[interior] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(src, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xq = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xq, xp, src)
