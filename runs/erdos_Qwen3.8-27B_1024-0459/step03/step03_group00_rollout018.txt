                        best_c5_local = c
                        best_j = j
                        best_delta = delta
            if best_j >= 0 and best_c5_local < c5_best - 1e-12:
                h[i] += best_delta
                h[best_j] -= best_delta
                c5_best = best_c5_local
                h_best = h.copy()
                improved = True
        if not improved and sweep > 3:
            break

    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX, _G_PENALTY
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = make_thue_morse(parent_n)
        parent_h = project_feasible(parent_h, parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("No parent. Thue-Morse: n={}, C5={:.8f}".format(parent_n, best_c5))

    # Phase 1: Quick resolution sweep
    n_candidates = [96, 128, 160, 192, 224, 256, 288, 320, 352, 384]
    if best_n not in n_candidates:
        n_candidates.append(best_n)
        n_candidates.sort()

    results = []
    print("\n=== Phase 1: Resolution sweep ===")
    for n in n_candidates:
        if remaining_time() < 20.0:
            break
        print("  n={}".format(n), end=" ")
        h_interp = project_feasible(interp_to(best_h, best_n, n), n)

        inits = [h_interp]
        h_tm = project_feasible(make_thue_morse(n), n)
        inits.append(h_tm)
        h_gr = project_feasible(make_golden_ratio(n), n)
        inits.append(h_gr)
        h_rb = project_feasible(make_random_bin(rng, n), n)
        inits.append(h_rb)

        beta_list_p1 = [50, 200, 800]
        maxiter_p1 = [200, 200, 300]

        best_local_h = None
        best_local_c5 = float('inf')
        for h0 in inits:
            if remaining_time() < 10.0:
                break
            h_r, c5_r = optimize_at_resolution(h0, n, beta_list_p1, maxiter_p1, rng, remaining_time)
            if c5_r < best_local_c5:
                best_local_c5 = c5_r
                best_local_h = h_r.copy()

        results.append((best_local_c5, n, best_local_h.copy()))
        print("  best C5={:.8f}".format(best_local_c5))

        if best_local_c5 < best_c5 - 1e-10:
            best_c5 = best_local_c5
            best_h = best_local_h.copy()
            best_n = n
            print("  ** NEW BEST: n={}, C5={:.8f} **".format(n, best_c5))

    # Sort and select top resolutions for Phase 2
    results.sort(key=lambda x: x[0])
    top_ns = []
    for _, nn, _ in results:
        if nn not in top_ns:
            top_ns.append(nn)
        if len(top_ns) >= 4:
            break
    if best_n not in top_ns:
        top_ns.append(best_n)

    print("\n=== Phase 2: Focused optimization on n={} ===".format(top_ns))
    for n in top_ns:
        if remaining_time() < 30.0:
            break
        print("  n={}".format(n), end=" ")

        # Best starting point for this n
        h_start = project_feasible(interp_to(best_h, best_n, n), n)
        c5_start = compute_c5(h_start)
        for c5_r, n_r, h_r in results:
            if n_r == n and c5_r < c5_start:
                h_start = h_r.copy()
                c5_start = c5_r
                break

        # Multi-stage beta annealing
        beta_schedule = [50, 100, 300, 800, 2000, 5000, 10000, 20000]
        maxiter_schedule = [300, 300, 300, 300, 250, 200, 150, 100]

        h_cur = h_start.copy()
        c5_cur = c5_start

        for beta, maxit in zip(beta_schedule, maxiter_schedule):
            if remaining_time() < 10.0:
                break
            _G_N = n
            _G_TGT = n / 2.0
            _G_DX = 2.0 / n
            _G_BETA = float(beta)
            _G_PENALTY = 5000.0
            bounds = [(0.0, 1.0)] * n
            try:
                res = minimize(smooth_obj_and_grad, h_cur, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-10, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-13:
                    c5_cur = c5_new
                    h_cur = h_new
            except Exception:
                pass

        # SLSQP final pass with high beta and equality constraint
        if remaining_time() > 10.0:
            _G_N = n
            _G_TGT = n / 2.0
            _G_DX = 2.0 / n
            _G_BETA = 15000.0
            _G_PENALTY = 0.0
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - _G_TGT,
                     'jac': lambda x: np.ones(n)}]
            try:
                res = minimize(smooth_obj_and_grad, h_cur, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-15, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-13:
                    c5_cur = c5_new
                    h_cur = h_new
            except Exception:
                pass

        print("  C5={:.8f}".format(c5_cur))
        if c5_cur < best_c5 - 1e-10:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n
            print("  ** NEW BEST: n={}, C5={:.8f} **".format(n, best_c5))

    # Phase 3: Perturbation restarts
    print("\n=== Phase 3: Perturbation restarts ===")
    if remaining_time() > 40.0:
        n_use = best_n
        h_sig, c5_sig = perturb_and_reoptimize(best_h, n_use, rng, remaining_time)
        if c5_sig < best_c5 - 1e-10:
            best_c5 = c5_sig
            best_h = h_sig.copy()
            best_n = n_use
            print("  Improved to C5={:.8f}".format(best_c5))

        # Try at a different resolution
        n_alt = n_use + 32 if n_use < 400 else n_use - 32
        if 64 <= n_alt <= 512 and remaining_time() > 20.0:
            h_alt = project_feasible(interp_to(best_h, best_n, n_alt), n_alt)
            h_alt, c5_alt = optimize_at_resolution(h_alt, n_alt,
                                                   [100, 500, 2000, 8000],
                                                   [200, 200, 200, 150],
                                                   rng, remaining_time)
            if c5_alt < best_c5 - 1e-10:
                best_c5 = c5_alt
                best_h = h_alt.copy()
                best_n = n_alt
                print("  Alt resolution n={} gave C5={:.8f}".format(n_alt, best_c5))

    # Phase 4: Pair-swap local search
    print("\n=== Phase 4: Pair-swap local search ===")
    if remaining_time() > 15.0:
        best_h, best_c5 = pair_swap_local(best_h, best_n, rng, remaining_time)
        best_n = len(best_h)
        print("  After pair swap: C5={:.8f}".format(best_c5))

    # Phase 5: Final high-beta polish
    print("\n=== Phase 5: Final polish ===")
    if remaining_time() > 10.0:
        n = best_n
        _G_N = n
        _G_TGT = n / 2.0
        _G_DX = 2.0 / n
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - _G_TGT,
                 'jac': lambda x: np.ones(n)}]
        for beta_val in [20000, 50000, 100000]:
            if remaining_time() < 5.0:
                break
            _G_BETA = float(beta_val)
            _G_PENALTY = 0.0
            try:
                res = minimize(smooth_obj_and_grad, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 200, 'ftol': 1e-16, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h_new
            except Exception:
                pass
        print("  After polish: C5={:.8f}".format(best_c5))

    # Phase 6: One more perturbation round if time
    if remaining_time() > 20.0:
        print("\n=== Phase 6: Second perturbation round ===")
        best_h, best_c5 = perturb_and_reoptimize(best_h, best_n, rng, remaining_time)
        best_n = len(best_h)
        print("  C5={:.8f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
