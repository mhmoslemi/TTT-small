This counts positions where h[j]=1 and h[j-k]=0, i.e., h[j]=1 and h[j+(-k)]=0.

So the full set of correlations counts both 0→1 and 1→0 transitions at different shifts. The max over all shifts captures the worst-case misalignment.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        # Distribute adjustment to interior points
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        # Find the point closest to 0.5 and adjust
        dist = np.abs(h - 0.5)
        idx = np.argmin(dist)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        # Re-fix if needed
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-14:
            for i in range(n):
                if i != idx:
                    h[i] = np.clip(h[i] + diff2, 0.0, 1.0)
                    break
    return h

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xnew = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xnew, xp, src)

def shift_gradient(h, n, offset):
    """Compute gradient of single shift correlation."""
    grad = np.zeros(n)
    if offset >= 0:
        n_pairs = n - offset
        if n_pairs <= 0:
            return grad
        h_left = h[:n_pairs]
        h_right = h[offset:offset+n_pairs]
        grad[:n_pairs] += (1.0 - h_right)
        grad[offset:offset+n_pairs] -= h_left
    else:
        abs_off = -offset
        n_pairs = n - abs_off
        if n_pairs <= 0:
            return grad
        h_left = h[:n_pairs]
        h_right = h[abs_off:abs_off+n_pairs]
        grad[abs_off:abs_off+n_pairs] += (1.0 - h_left)
        grad[:n_pairs] -= h_right
    return grad

def topk_gradient(h, n, top_k=3):
    dx = 2.0 / n
    corr = compute_corr_full(h)
    top_idx = np.argsort(corr)[-top_k:]
    top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
    
    grad = np.zeros(n)
    for k_idx in top_idx:
        offset = int(k_idx) - (n - 1)
        grad += shift_gradient(h, n, offset)
    return grad * dx

def bb_gradient_descent(h, n, budget_fn, top_k=3, max_iter=20000, verbose=False):

    c5_cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur

    g_prev = None
    bb_alpha = 0.01
    bb_beta = 0.01

            break
