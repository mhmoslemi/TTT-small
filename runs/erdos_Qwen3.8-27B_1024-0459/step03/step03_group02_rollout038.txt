def lag_gradient(h, n, offset):
    """Gradient of corr[lag] with respect to h"""
    grad = np.zeros(n)
    if offset >= 0:
        n_pairs = n - offset
        if n_pairs <= 0:
            return grad
        # corr = sum_{i=0}^{n-1-offset} h[i+offset] * (1 - h[i])
        # d/dh[i] for i < n_pairs: -(h[i+offset])
        # d/dh[i+offset] for i < n_pairs: (1 - h[i])
        grad[offset:offset+n_pairs] += (1.0 - h[:n_pairs])
        grad[:n_pairs] -= h[offset:offset+n_pairs]
    else:
        abs_off = -offset
        n_pairs = n - abs_off
        if n_pairs <= 0:
            return grad
        # corr = sum_{i=abs_off}^{n-1} h[i] * (1 - h[i-abs_off])
        grad[:n_pairs] += (1.0 - h[abs_off:abs_off+n_pairs])
        grad[abs_off:abs_off+n_pairs] -= h[:n_pairs]
    return grad

import numpy as np
from scipy.optimize import minimize

_START_TIME = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            break
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-13:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

def remaining_time():

def smooth_max_lse(corr, beta):
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_obj(x):
    n = _N_GLOBAL
    beta = _BETA_GLOBAL
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max_lse(corr, beta) * dx

def constraint_sum(x):
    return np.sum(x) - _N_GLOBAL / 2.0

def constraint_sum_jac(x):
    return np.ones(_N_GLOBAL)

def lag_gradient(h, n, L):
    """Correct gradient of corr at lag L with respect to h."""
    grad = np.zeros(n)
    if L >= 0:
        n_pairs = n - L
        if n_pairs <= 0:
            return grad
        grad[:n_pairs] += (1.0 - h[L:L + n_pairs])
        grad[L:] -= h[:n_pairs]
    else:
        Lp = -L
        n_pairs = n - Lp
        if n_pairs <= 0:
            return grad
        grad[Lp:] += (1.0 - h[:n_pairs])
        grad[:n_pairs] -= h[Lp:Lp + n_pairs]
    return grad

def slsqp_optimize(h, n, beta, maxiter):

    _BETA_GLOBAL = beta
    bounds = [(0.0, 1.0)] * n

        res = minimize(smooth_max_obj, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def trf_optimize(h, n, beta, maxiter):

def interp_to(src_h, src_n, dst_n):
