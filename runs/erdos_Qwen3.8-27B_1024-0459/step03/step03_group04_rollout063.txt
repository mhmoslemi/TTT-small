def coord_descent(h, n, dx, max_sweeps=200, n_trials=30):

    c5 = np.max(corr) * dx
    h = project_feasible(h, n)
    S = h.sum()
    target = n / 2.0

        improved = False
        for i in range(n):
            h_old = h[i]
            if h_old <= 0 or h_old >= 1:
                continue  # at bound, can only move in one direction
            
            # Compute coefficient vector for changing h[i]
            coeff = np.zeros(2*n - 1)
            # Positive lags
            for k in range(min(i, n-1) + 1):
                coeff[n-1+k] += (1.0 - h[i-k])
            for k in range(min(n-1-i, n-1) + 1):
                coeff[n-1+k] += (-h[i+k])
            # Negative lags
            for ao in range(1, min(n-1-i, n-1) + 1):
                coeff[n-1-ao] += (1.0 - h[i+ao])
            for ao in range(1, min(i, n-1) + 1):
                coeff[n-1-ao] += (-h[i-ao])
            
            # Try trial values
            best_val = h_old
            best_obj = c5
            
            # Generate trial values: current, 0, 1, and grid
            trials = [0.0, 1.0] + list(np.linspace(0, 1, n_trials))
            # Also try values near current
            trials += list(h_old + np.array([-0.1,-0.05,-0.02,-0.01,0.01,0.02,0.05,0.1]))
            
            for hv in trials:
                hv = max(0.0, min(1.0, hv))
                if abs(hv - h_old) < 1e-10:
                    continue
                delta = hv - h_old
                corr_new = corr + delta * coeff
                c5_new = np.max(corr_new) * dx
                # Penalty for sum constraint
                S_new = S + delta
                pen = 100.0 * (S_new - target)**2
                obj = c5_new + pen
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_val = hv
            
            if best_val != h_old:
                delta = best_val - h_old
                h[i] = best_val
                S += delta
                corr += delta * coeff
                c5 = np.max(corr) * dx
                improved = True
        
        # Periodic projection to maintain sum constraint
        if (sweep+1) % 10 == 0:
            h = project_feasible(h, n)
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5 = np.max(corr) * dx
            S = h.sum()
        
        if not improved:
            break
    
    return h, c5

        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5, _ = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 160
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5, _ = compute_c5(best_h)
        print("Default: n={}, C5={:.8f}".format(best_n, best_c5))

    # Phase 1: Coordinate descent - main workhorse
    n = best_n
    dx = 2.0 / n
    h_cur = best_h.copy()
    h_cd, c5_cd = coord_descent(h_cur, n, dx, max_sweeps=1000, n_trials=40,
                                penalty_lam=500.0, seed=seed)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()
        print("CD1: n={}, C5={:.8f}".format(n, best_c5))

    # Phase 2: SLSQP polish at multiple betas
    if time_left(t0, budget) > 60:
        n = best_n
        for beta in [100, 500, 2000, 8000]:
            if time_left(t0, budget) < 25:
                break
            h_try = slsqp_run(best_h, n, beta, maxiter=400)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
                print("  SLSQP beta={}: C5={:.8f}".format(beta, best_c5))

    # Phase 3: More coordinate descent after SLSQP
    if time_left(t0, budget) > 40:
        n = best_n
        dx = 2.0 / n
        h_cd, c5_cd = coord_descent(best_h, n, dx, max_sweeps=1000, n_trials=50,
                                    penalty_lam=1000.0, seed=seed + 1)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            print("CD2: C5={:.8f}".format(best_c5))

    # Phase 4: Try different resolutions
    if time_left(t0, budget) > 80:
        for delta_n in [10, 20, -10, -20, 30, 40, 50, -30, -40, 60, 80, -50, 100, 120, 160]:
            if time_left(t0, budget) < 35:
                break
            nt = best_n + delta_n
            if nt < 80 or nt > 400:
                continue
            ht = interp_h(best_h, best_n, nt)
            ht = project_feasible(ht, nt)
            ct, _ = compute_c5(ht)
            if ct < best_c5 + 0.001:
                dt = 2.0 / nt
                ht2, ct2 = coord_descent(ht, nt, dt, max_sweeps=300, n_trials=30,
                                         penalty_lam=500.0, seed=seed + delta_n)
                ct2, _ = compute_c5(ht2)
                if ct2 < best_c5:
                    best_c5 = ct2
                    best_h = ht2.copy()
                    best_n = nt
                    print("  Res n={}: C5={:.8f}".format(nt, best_c5))

    # Phase 5: L-BFGS-B with penalty at current best
    if time_left(t0, budget) > 50:
        n = best_n
        for beta in [200, 1000, 5000]:
            if time_left(t0, budget) < 15:
                break
            h_try = lbfgs_run(best_h, n, beta, lam=500.0, maxiter=800)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
                print("  LBFGS beta={}: C5={:.8f}".format(beta, best_c5))

    # Phase 6: Final coordinate descent with fine trials
    if time_left(t0, budget) > 30:
        n = best_n
        dx = 2.0 / n
        h_cd, c5_cd = coord_descent(best_h, n, dx, max_sweeps=2000, n_trials=80,
                                    penalty_lam=2000.0, seed=seed + 100)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            print("CD3: C5={:.8f}".format(best_c5))

    # Phase 7: SLSQP final polish
    if time_left(t0, budget) > 20:
        n = best_n
        for beta in [3000, 10000, 30000]:
            if time_left(t0, budget) < 10:
                break
            h_try = slsqp_run(best_h, n, beta, maxiter=300)
            c5_try, _ = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
                print("  Final SLSQP beta={}: C5={:.8f}".format(beta, best_c5))

    # Phase 8: Very fine coordinate descent (near-convergence)
    if time_left(t0, budget) > 15:
        n = best_n
        dx = 2.0 / n
        h_cd, c5_cd = coord_descent(best_h, n, dx, max_sweeps=3000, n_trials=100,
                                    penalty_lam=5000.0, seed=seed + 200)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            print("CD4: C5={:.8f}".format(best_c5))

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5, _ = compute_c5(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
