```python
    n = len(h)
    grad = np.zeros(n)
    # Positive shifts
    for s in range(n):
        w = weights[s + n - 1]  # correlation index
        if w < 1e-15: continue
        # (1-h[m-s]) for m in [s, n-1]
        grad[s:] += w * (1 - h[:n-s])
        # -h[m+s] for m in [0, n-1-s]
        grad[:n-s] += w * (-h[s:])
    # Negative shifts
    for p in range(1, n):
        w = weights[n - 1 - p]  # correlation index
        if w < 1e-15: continue
        # (1-h[m+p]) for m in [0, n-1-p]
        grad[:n-p] += w * (1 - h[p:])
        # -h[m-p] for m in [p, n-1]
        grad[p:] += w * (-h[:n-p])
    return grad
```

```python
def subgradient_step(h, n, active_eps=0.01, step_size=0.01):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    active = np.where(corr > m - active_eps)[0]
    
    grad = np.zeros(n)
    for j in active:
        s = j - (n - 1)
        if s >= 0:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] += (-h[s:])
        else:
            p = -s
            grad[:n-p] += (1.0 - h[p:])
            grad[p:] += (-h[:n-p])
    
    grad /= len(active)
    return h - step_size * grad, m
```

from scipy.optimize import minimize
import time

_G_N = 0
_G_BETA = 100.0
_G_TGT = 0.0
_G_START = 0.0
_G_BUDGET = 1300.0

def _remaining():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 3.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(free)
        if cnt > 0:
            h[free] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xq = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xq, xp, src)
