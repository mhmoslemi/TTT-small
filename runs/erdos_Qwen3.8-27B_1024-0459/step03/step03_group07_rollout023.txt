    m = np.max(top)

            f_vals = np.maximum(np.maximum(max_lin, quad_val), max_unaffected)

            best_f = f_vals.min()

            # Since f(x) might not be convex (due to the concave quad term),
            # use a fine grid around best_x

            hi = min(1.0, best_x + 0.005)
            x_fine = np.linspace(lo, hi, 1000)

            f_fine = np.maximum(np.maximum(max_lin_f, quad_fine), max_unaffected)
            best_x = x_fine[f_fine.argmin()]

        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(corr) * (2.0/n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h

def grad_descent_adaptive(h, n, max_iters=2000, time_limit=None, t0=None):
    """Adaptive gradient descent on top-k smooth max"""
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    k = 5

            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        grad = compute_grad_k(h, n, k)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        grad_norm = grad / gnorm
        
        h_new = h - lr * grad_norm * gnorm  # = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            patience = 0
            lr = min(lr * 1.05, 0.02)
        else:
            patience += 1
            lr *= 0.9
            if patience > max_patience:
                # Perturbation
                scale = 0.03 + 0.02 * np.random.rand()
                h = project_feasible(h + scale * np.random.randn(n), n)
                lr = 0.005
                patience = 0
    
    return best_h
