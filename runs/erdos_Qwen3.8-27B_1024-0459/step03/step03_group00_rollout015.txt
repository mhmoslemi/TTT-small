import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_P = 20.0
_G_START = 0.0
_G_BUDGET = 1300.0
_G_DX = 1.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 4.0)


def obj_pnorm(x):
    n = _G_N
    p = _G_P
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    n_corr = len(corr)
    if p < 2.0:
        return float(np.max(corr)) * dx
    corr_clipped = np.maximum(corr, 0.0)
    val = (np.sum(corr_clipped ** p) / n_corr) ** (1.0 / p) * dx
    return val


def obj_pnorm_grad(x):
    n = _G_N
    p = _G_P
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    n_corr = len(corr)

    if p < 2.0:
        top_idx = int(np.argmax(corr))
        s = abs(top_idx - (n - 1))
        grad = np.zeros(n)
        if s < n:
            grad[s:] += (1.0 - x[:n - s])
            grad[:n - s] -= x[s:]
        return grad * dx

    corr_pos = np.maximum(corr, 1e-30)
    corr_p_minus_1 = corr_pos ** (p - 1)
    norm_base = np.sum(corr_pos ** p) / n_corr
    norm_val = norm_base ** (1.0 / p)
    if norm_val < 1e-30:
        return np.zeros(n)
    norm_coeff = norm_val ** (1 - p) * (1.0 / n_corr) * dx
    weights = corr_p_minus_1 * norm_coeff

    grad = np.zeros(n)
    grad[:] += weights[n - 1] * (1.0 - 2.0 * x)
    for s in range(1, n):
        w = weights[n - 1 - s] + weights[n - 1 + s]
        if abs(w) < 1e-20:
            continue
        grad[s:] += w * (1.0 - x[:n - s])
        grad[:n - s] -= w * x[s:]
    return grad


def lbbg_pnorm_step(h, n, p, maxiter=200, penalty_mu=5000.0):
    global _G_N, _G_P, _G_DX
    _G_N = n
    _G_P = float(p)
    _G_DX = 2.0 / n
    target = n / 2.0

    def obj_pen(x):
        val = obj_pnorm(x)
        cv = np.sum(x) - target
        return val + penalty_mu * cv * cv

    def grad_pen(x):
        g = obj_pnorm_grad(x)
        cv = np.sum(x) - target
        g += 2.0 * penalty_mu * cv
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h, jac=grad_pen, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-8,
                                               'ftol': 1e-14, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)


def slsqp_pnorm_step(h, n, p, maxiter=200):
    global _G_N, _G_P, _G_DX
    _G_N = n
    _G_P = float(p)
    _G_DX = 2.0 / n
    target = n / 2.0

    def con_sum(x):
        return np.sum(x) - target

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum}]
    try:
        res = minimize(obj_pnorm, h, jac=obj_pnorm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h


def make_golden_ratio(n):
    h = np.zeros(n)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return h


def make_random_bin(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h


def make_sinusoidal_mix(rng, n, f1, f2, f3):
    t = np.arange(n) / float(n)
    h = 0.5 + 0.3 * np.sin(2.0 * np.pi * f1 * t + rng.rand() * 6.28) \
                + 0.3 * np.sin(2.0 * np.pi * f2 * t + rng.rand() * 6.28) \
                + 0.2 * np.cos(2.0 * np.pi * f3 * t + rng.rand() * 6.28)
    return np.clip(h, 0.0, 1.0)


def make_paperfolding(n):
    h = np.zeros(n)
    h[0] = 0.0
    for i in range(1, n):
        # Paperfolding (dragon curve) sequence
        highest = 1 << (i.bit_length() - 1)
        pos = i - highest
        parity = highest & 1
        if pos & 1:
            h[i] = 1.0 - h[pos]
        else:
            prev = i - 1
            highest2 = 1 << (prev.bit_length() - 1)
            h[i] = 1.0 - h[prev - highest2] if prev >= highest2 else (1.0 - h[pos]) if pos > 0 else (1.0 - h[0])
    return h


def binary_swap_search(h, n, budget_check, rng, max_rounds=5000):
    c5_best = compute_c5(h)
    h_best = h.copy()
    ones = np.where(h_best > 0.5)[0].tolist()
    zeros = np.where(h_best < 0.5)[0].tolist()

    for rnd in range(max_rounds):
        if budget_check() < 3.0:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        oi = rng.randint(len(ones))
        zi = rng.randint(len(zeros))
        o_idx = ones[oi]
        z_idx = zeros[zi]

        h_best[o_idx] = 0.0
        h_best[z_idx] = 1.0
        c5_test = compute_c5(h_best)
        if c5_test < c5_best - 1e-12:
            c5_best = c5_test
            ones[oi] = z_idx
            zeros[zi] = o_idx
        else:
            h_best[o_idx] = 1.0
            h_best[z_idx] = 0.0

    return h_best, c5_best


def coord_descent_binary(h, n, budget_check, rng, max_sweeps=20):
    c5_best = compute_c5(h)
    h_best = h.copy()

    for sweep in range(max_sweeps):
        if budget_check() < 4.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if budget_check() < 2.0:
                break
            old = h_best[i]
            new_val = 1.0 - old
            h_best[i] = new_val
            h_proj = project_feasible(h_best, n)
            c5_test = compute_c5(h_proj)
            if c5_test < c5_best - 1e-12:
                c5_best = c5_test
                h_best = h_proj.copy()
                any_imp = True
            else:
                h_best[i] = old
        if not any_imp and sweep > 2:
            break

    return h_best, c5_best


def fractional_local_search(h, n, budget_check, rng, max_sweeps=30):
