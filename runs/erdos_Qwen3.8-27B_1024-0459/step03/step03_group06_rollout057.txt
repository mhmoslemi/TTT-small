            new_c5 = compute_c5(h)
            if new_c5 < current_c5 or (T > 1e-15 and rng.rand() < np.exp(-(new_c5 - current_c5) / T)):
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                current_c5 = new_c5
                ones.remove(i1)
                ones.remove(i2)
                ones.extend([j1, j2])
                ones.sort()
                zeros.remove(j1)
                zeros.remove(j2)
                zeros.extend([i1, i2])
                zeros.sort()
            else:
                h[i1] = 1.0
                h[i2] = 1.0
                h[j1] = 0.0
                h[j2] = 0.0
    return best_h, best_c5


def continuous_refine(h_init, n, max_beta=800):
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = float(np.max(corr))
        sm = (m + np.log(np.sum(np.exp((beta_global[0] * (corr - m))))) / beta_global[0]) * dx
        pen = 2000.0 * (np.sum(x) - target) ** 2
        return sm + pen

    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = float(np.max(corr))
        w = np.exp(beta_global[0] * (corr - m))
        w /= np.sum(w)
        g = np.zeros(n)
        for j in range(2 * n - 1):
            wj = w[j]
            if wj < 1e-14:
                continue
            s = j - (n - 1)
            if s >= 0:
                for mi in range(s, n):
                    g[mi] += wj * (1.0 - x[mi - s])
                for mi in range(n - s):
                    g[mi] += wj * (-x[mi + s])
            else:
                sp = -s
                for mi in range(n - sp):
                    g[mi] += wj * (1.0 - x[mi + sp])
                for mi in range(sp, n):
                    g[mi] += wj * (-x[mi - sp])
        g *= dx
        g += 4000.0 * (np.sum(x) - target)
        return g

    bounds = [(0.0, 1.0)] * n
    h0 = project_feasible(h_init, n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    beta_global = [200.0]
    for bv in [200, 500, 800]:
        if remaining_time() < 5:
            break
        beta_global[0] = float(bv)
        try:
            res = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 400, 'ftol': 1e-15, 'gtol': 1e-10, 'iprint': 0, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h0 = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5


def ils_binary(h_best, n, rng, num_restarts=20):
    best_h = h_best.copy()
    best_c5 = compute_c5(h_best)
    for r in range(num_restarts):
        if remaining_time() < 10:
            break
        h_pert = best_h.copy()
        k_flip = rng.randint(2, 6)
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        flip_o = rng.choice(ones, min(k_flip, len(ones)), replace=False)
        flip_z = rng.choice(zeros, min(k_flip, len(zeros)), replace=False)
        nf = min(len(flip_o), len(flip_z))
        h_pert[flip_o[:nf]] = 0.0
        h_pert[flip_z[:nf]] = 1.0
        h_opt, c5_opt = binary_steepest(h_pert, n, max_passes=100)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    return best_h, best_c5


def subgradient_step(h, n, step_size, top_k=10):
    dx = 2.0 / n
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-top_k:]
    g = np.zeros(n)
    for j in top_idx:
        s = j - (n - 1)
        if s >= 0:
            for mi in range(s, n):
                g[mi] += (1.0 - h[mi - s])
            for mi in range(n - s):
                g[mi] += (-h[mi + s])
        else:
            sp = -s
            for mi in range(n - sp):
                g[mi] += (1.0 - h[mi + sp])
            for mi in range(sp, n):
                g[mi] += (-h[mi - sp])
    gnorm = np.linalg.norm(g)
    if gnorm < 1e-14:
        return h.copy()
    g /= (gnorm * top_k)
    h_new = h - step_size * g
    return project_feasible(h_new, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.7f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 160
        best_h = make_thue_morse(best_n)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)
        print("No parent. Thue-Morse: n={}, C5={:.7f}".format(best_n, best_c5))

    # Phase 1: Binary steepest descent at multiple resolutions
    n_list = sorted(set([best_n, 120, 140, 160, 180, 200, 220, 240]))
    n_list = [nn for nn in n_list if 80 <= nn <= 260]

    for n in n_list:
        if remaining_time() < 40:
            break
        print("\n--- Steepest n={} ---".format(n))
        h_interp = project_binary(interp_to(best_h, best_n, n))
        inits = [h_interp]
        inits.append(project_binary(make_thue_morse(n)))
        inits.append(project_binary(make_golden(n)))
        inits.append(project_binary(make_vdc(n)))
        for _ in range(2):
            hr = np.zeros(n)
            idx = rng.permutation(n)
            hr[idx[:n // 2]] = 1.0
            inits.append(hr)

        for h0 in inits:
            if remaining_time() < 10:
                break
            max_pass = 400 if n <= 180 else 200
            h_s, c5_s = binary_steepest(h0, n, max_passes=max_pass)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
                best_n = n
                print("  Steepest: n={}, C5={:.7f}".format(n, c5_s))

    # Phase 2: ILS on best binary
    if remaining_time() > 60:
        n = best_n
        print("\n--- ILS n={} ---".format(n))
        h_bin = project_binary(best_h)
        h_ils, c5_ils = ils_binary(h_bin, n, rng, num_restarts=30)
        if c5_ils < best_c5:
            best_c5 = c5_ils
            best_h = h_ils.copy()
            print("  ILS: n={}, C5={:.7f}".format(n, c5_ils))

    # Phase 3: SA from multiple starts at best resolution
    if remaining_time() > 60:
        n = best_n
        print("\n--- SA n={} ---".format(n))
        h_bin = project_binary(best_h)
        h_sa, c5_sa = binary_sa(h_bin, n, rng, 15000, t_start=0.003, t_end=0.000002)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            print("  SA: n={}, C5={:.7f}".format(n, c5_sa))

        # Second SA with different schedule
        if remaining_time() > 30:
            h_sa2, c5_sa2 = binary_sa(project_binary(best_h), n, rng, 10000, t_start=0.008, t_end=0.00001)
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2.copy()
                print("  SA2: n={}, C5={:.7f}".format(n, c5_sa2))

    # Phase 4: Continuous refinement
    if remaining_time() > 40:
        n = best_n
        print("\n--- Continuous refine n={} ---".format(n))
        h_cont, c5_cont = continuous_refine(best_h, n)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()
            print("  Cont: n={}, C5={:.7f}".format(n, c5_cont))

        # Try at neighboring resolutions
        for nn in [n + 20, n - 20, n + 40]:
            if remaining_time() < 20:
                break
            if nn < 80 or nn > 320:
                continue
            h_int = project_feasible(interp_to(best_h, best_n, nn), nn)
            h_c2, c5_c2 = continuous_refine(h_int, nn)
            if c5_c2 < best_c5:
                best_c5 = c5_c2
                best_h = h_c2.copy()
                best_n = nn
                print("  Cont: n={}, C5={:.7f}".format(nn, c5_c2))

    # Phase 5: Binary steepest from continuous solution
    if remaining_time() > 30:
        n = best_n
        h_bin2 = project_binary(best_h)
        h_s2, c5_s2 = binary_steepest(h_bin2, n, max_passes=200)
        if c5_s2 < best_c5:
            best_c5 = c5_s2
            best_h = h_s2.copy()
            print("  Steepest2: n={}, C5={:.7f}".format(n, c5_s2))

    # Phase 6: Subgradient + projection iterations
    if remaining_time() > 20:
        n = best_n
        print("\n--- Subgradient n={} ---".format(n))
        h_sub = best_h.copy()
        best_sub_c5 = best_c5
        for it in range(300):
            if remaining_time() < 4:
                break
            step = 0.05 * (0.97 ** (it // 30))
            h_new = subgradient_step(h_sub, n, step, top_k=12)
            c5_new = compute_c5(h_new)
            if c5_new < best_sub_c5:
                best_sub_c5 = c5_new
                best_h = h_new.copy()
                best_c5 = c5_new
                best_n = n
                h_sub = h_new.copy()
            else:
                h_sub = h_sub * 0.99 + best_h * 0.01
                h_sub = project_feasible(h_sub, n)

    # Phase 7: Final polish - continuous
    if remaining_time() > 15:
        n = best_n
        print("\n--- Final polish n={} ---".format(n))
        h_fin, c5_fin = continuous_refine(best_h, n)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            print("  Final: n={}, C5={:.7f}".format(n, c5_fin))

    # Phase 8: One more steepest on best
    if remaining_time() > 10:
        n = best_n
        h_bin3 = project_binary(best_h)
        h_s3, c5_s3 = binary_steepest(h_bin3, n, max_passes=100)
        if c5_s3 < best_c5:
            best_c5 = c5_s3
            best_h = h_s3.copy()
            print("  Final Steepest: n={}, C5={:.7f}".format(n, c5_s3))

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
