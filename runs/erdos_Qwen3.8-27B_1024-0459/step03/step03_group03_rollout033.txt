    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while time remaining:
        best_swap = None
        best_C = None
        best_emax = best
        
        for i in ones:
            for j in zeros:
                h[i] = 0; h[j] = 1
                C_new = np.correlate(h, 1-h, mode='full')
                emax = C_new.max() * (2.0/n)
                if emax < best_emax:
                    best_emax = emax
                    best_swap = (i, j)
                    best_C = C_new.copy()

        C = best_C
        best = best_emax

def _binary_sa(h, n, n_iter, T0, Tf, rng, verbose=False):
    """Simulated annealing on binary vectors."""
    h = h.copy()
    c = _c5(h)
    best_h = h.copy()
    best_c = c
    
    for it in range(n_iter):
        if _tl() < 5:
            break
        T = T0 * (Tf / T0) ** (it / max(1, n_iter - 1))
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c2 = _c5(h)
        dc = c2 - c
        if dc < 0 or rng.rand() < np.exp(-dc / max(T, 1e-10)):
            c = c2
            if c2 < best_c:
                best_c = c2
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def _beatty_init(n, alpha):
    """Beatty sequence initialization."""
    h = np.zeros(n, dtype=np.float64)
    for i in range(n):
        if (i * alpha) % 1.0 < 0.5:
            h[i] = 1.0
    # Adjust to have exactly n/2 ones
    count = int(np.sum(h))
    if count > n // 2:
        # Remove some ones
        ones = np.where(h > 0.5)[0]
        rng = np.random.RandomState(0)
        for _ in range(count - n // 2):
            idx = rng.choice(ones)
            h[idx] = 0.0
            ones = np.where(h > 0.5)[0]
    elif count < n // 2:
        zeros = np.where(h < 0.5)[0]
        rng = np.random.RandomState(0)
        for _ in range(n // 2 - count):
            idx = rng.choice(zeros)
            h[idx] = 1.0
            zeros = np.where(h < 0.5)[0]
    return h

_ng = 0
_bv = 100.0

def _scon(x):
    return x.sum() - 0.5 * _ng

def _sobj(x):
    n = _ng
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = float(np.max(c))
    b = _bv
    sh = c - mx
    return (mx + np.log(np.exp(b * sh).sum()) / b) * (2.0 / n)

def _slsqp(h0, n, beta, maxiter):
    global _ng, _bv
    _ng = n
    _bv = beta
    try:
        r = minimize(_sobj, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon}],
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return _proj(np.asarray(r.x, dtype=np.float64), n)
    except Exception:
        return _proj(h0.copy(), n)

def _ggrad(h, kmax):
    n = len(h)
    s = kmax - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
    else:
        p = -s
        if p > 0:
            g[:n - p] += 1.0 - h[p:n]
            g[p:] -= h[:n - p]
    return g

def _smgrad(h, bw):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(np.max(corr))
    w = np.exp(bw * (corr - mx))
    sw = float(np.sum(w))
    if sw < 1e-30:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
        sw = 1.0
    w /= sw
    g = np.zeros(n)
    nz = np.where(w > 1e-10)[0]
    for k in nz:
        g += w[int(k)] * _ggrad(h, int(k))
    return g

def _sgloop(h, n, nit, rng):

    bc = _c5(best)
    for _ in range(nit):
        if _tl() < 3:
