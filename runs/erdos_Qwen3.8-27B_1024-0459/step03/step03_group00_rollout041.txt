def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n, target=None):
    if target is None:
        target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_of_corrs(h, top_indices, n):
    """Compute subgradient of max of top correlation values."""
    dx = 2.0 / n
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k_idx in top_indices:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            n_pairs = n - offset
            if n_pairs <= 0:
                continue
            # f = sum_{i=0}^{n_pairs-1} h[i] * (1 - h[i+offset])
            # d/dh[j] = (1-h[j+offset]) for j in [0, n_pairs-1]
            #           -h[j-offset] for j in [offset, n-1]
            grad[:n_pairs] += (1.0 - h[offset:offset+n_pairs])
            grad[offset:offset+n_pairs] -= h[:n_pairs]
        else:
            abs_off = -offset
            n_pairs = n - abs_off
            if n_pairs <= 0:

            # f = sum_{i=0}^{n_pairs-1} h[i+abs_off] * (1 - h[i])

            #           -h[j+abs_off] for j in [0, n_pairs-1]
            grad[abs_off:] += (1.0 - h[:n_pairs])
            grad[:n_pairs] -= h[abs_off:abs_off+n_pairs]
    grad /= len(top_indices)

            else:
                h = h_best.copy()
    
    return h_best, c5_best


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1') & 1
        h[i] = float(cnt)
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET, _G_N, _G_BETA, _G_TGT, _G_DX
    
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)
    
    def budget_remain():
        return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 256
        best_h = make_thue_morse(best_n)
        best_c5 = compute_c5(best_h)
        print("No parent. Thue-Morse: n={}, C5={:.10f}".format(best_n, best_c5))
    
    # Phase 1: Intensive SLSQP at parent resolution
    print("\n=== Phase 1: SLSQP at n={} ===".format(best_n))
    beta_schedule = [100, 300, 1000, 3000, 10000, 30000]
    iter_schedule = [400, 400, 500, 500, 600, 600]
    best_h, best_c5 = do_slsqp_phase(best_h, best_n, beta_schedule, iter_schedule, budget_remain)
    print("  After SLSQP: C5={:.10f}".format(best_c5))
    
    # Phase 2: Subgradient refinement
    if budget_remain() > 20.0:
        print("\n=== Phase 2: Subgradient at n={} ===".format(best_n))
        best_h, best_c5 = do_subgrad_phase(best_h, best_n, budget_remain, top_k=5, n_iter=400)
        print("  After subgrad: C5={:.10f}".format(best_c5))
    
    # Phase 3: Try higher resolution
    for n_try in [320, 384, 512]:
        if budget_remain() < 60.0:
            break
        print("\n=== Phase 3: n={} ===".format(n_try))
        h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        c5_interp = compute_c5(h_interp)
        
        # SLSQP at higher resolution
        beta_sched = [50, 200, 1000, 5000, 20000]
        iter_sched = [300, 400, 400, 500, 500]
        h_try, c5_try = do_slsqp_phase(h_interp, n_try, beta_sched, iter_sched, budget_remain)
        
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try
            print("  Improved! n={}, C5={:.10f}".format(n_try, best_c5))
        else:
            print("  No improvement at n={}: C5={:.10f} vs best {:.10f}".format(
                n_try, c5_try, best_c5))
    
    # Phase 4: Coordinate descent polish
    if budget_remain() > 15.0:
        print("\n=== Phase 4: Coordinate Descent ===")
        best_h, best_c5 = do_coordinate_descent(best_h, best_n, budget_remain, n_sweeps=30)
        print("  After coord descent: C5={:.10f}".format(best_c5))
    
    # Phase 5: Pairwise swaps
    if budget_remain() > 10.0:
        print("\n=== Phase 5: Pairwise Swaps ===")
        best_h, best_c5 = do_pairwise_swap(best_h, best_n, budget_remain, n_rounds=15)
        print("  After pairwise: C5={:.10f}".format(best_c5))
    
    # Phase 6: Final SLSQP polish with very high beta
    if budget_remain() > 15.0:
        print("\n=== Phase 6: Final Polish ===")
        beta_sched = [50000, 100000]
        iter_sched = [800, 800]
        best_h, best_c5 = do_slsqp_phase(best_h, best_n, beta_sched, iter_sched, budget_remain)
        print("  After final polish: C5={:.10f}".format(best_c5))
    
    # Phase 7: Try binary annealing at best_n and nearby
    if budget_remain() > 30.0:
        print("\n=== Phase 7: Binary Annealing ===")
        for n_ann in [best_n, best_n + 64, best_n + 128]:
            if budget_remain() < 15.0:
                break
            if n_ann % 2 != 0:
                n_ann += 1
            h_bin, c5_bin = do_binary_anneal(n_ann, budget_remain, seed=seed)
            if c5_bin < best_c5 - 1e-12:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                best_n = n_ann
                print("  Binary anneal improved! n={}, C5={:.10f}".format(n_ann, best_c5))
            else:
                print("  Binary n={}: C5={:.10f} (no improvement)".format(n_ann, c5_bin))
    
    # Phase 8: Final coordinate descent
    if budget_remain() > 10.0:
        print("\n=== Phase 8: Final Coord Descent ===")
        best_h, best_c5 = do_coordinate_descent(best_h, best_n, budget_remain, n_sweeps=20)
        print("  Final: C5={:.10f}".format(best_c5))
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if isinstance(eval_c5, (int, float)) and eval_c5 > 0:
            if eval_c5 < final_c5:
                final_c5 = float(eval_c5)
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
