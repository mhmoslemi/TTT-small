def compute_gradient(h, n, w):
    """w is the softmax weight array of length 2n-1, indexed by r = 0..2n-2.
    L = r - (n-1)."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for L in range(-(n-1), n):
        idx = L + (n-1)
        wl = w[idx]
        if wl < 1e-18:
            continue
        if L >= 0:
            # (1-h[j+L]) valid for j = 0..n-1-L
            grad[:n-L] += wl * (1.0 - h[L:])
            # h[j-L] valid for j = L..n-1
            grad[L:] -= wl * h[:n-L]
        else:

            # (1-h[j+L]) = (1-h[j-La]) valid for j = La..n-1
            grad[La:] += wl * (1.0 - h[:n-La])
            # h[j-L] = h[j+La] valid for j = 0..n-1-La
            grad[:n-La] -= wl * h[La:]
    grad *= dx
    return grad

        else:
            La = -L
            grad[La:] += wl * (1.0 - h[:n-La])
            grad[:n-La] -= wl * h[La:]
    grad *= dx
    return grad

def smooth_max_and_grad(h, beta, topK=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topK is not None and topK < len(corr):
        top_idx = np.argsort(corr)[::-1][:topK]
        top_vals = corr[top_idx].copy()
        other_vals = np.zeros(len(corr))
        other_vals[np.argsort(corr)[:len(corr)-topK]] = np.min(corr)
        w_full = np.zeros(len(corr))
        m = np.max(top_vals)
        top_
        exp_top = np.exp(beta * (top_vals - m))
        p_top = exp_top / np.sum(exp_top)
        w_full[top_idx] = p_top
        # Add small weight to others to avoid zero
        rest_weight = 1.0 / max(len(corr) - topK, 1)
        mask_rest = np.ones(len(corr), dtype=bool)
        mask_rest[top_idx] = False
        w_full[mask_rest] = rest_weight / max(np.sum(mask_rest), 1)
        # Renormalize
        w_full /= np.sum(w_full)
        smooth_val = m + np.log(np.sum(exp_top)) / beta
    else:
        m = np.max(corr)
        shifted = beta * (corr - m)
        exp_vals = np.exp(shifted)
        w_full = exp_vals / np.sum(exp_vals)
        smooth_val = m + np.log(np.sum(exp_vals)) / beta
    smooth_val *= dx
    grad = compute_gradient_direct(h, n, w_full)
    return smooth_val, grad, corr

def fista_optimize(h_init, n, beta, max_iter=5000, topK=None, tol=1e-13):
    h = h_init.copy()
    h_prev = h.copy()
    t = 1.0
    # Estimate step size
    val0, grad0, _ = smooth_max_and_grad(h, beta, topK)
    gnorm = np.linalg.norm(grad0)
    if gnorm < 1e-15:
        return h
    step = 1.0 / (max(beta, 1.0) * gnorm / n + 1e-6)
    step = max(step, 0.1)
    step = min(step, 100.0)
    
    best_h = h.copy()
    best_val = val0
    stagnant = 0
    
    for it in range(max_iter):
        if remaining_time() < 3:
            break
        val, grad, _ = smooth_max_and_grad(h, beta, topK)
        
        h_unt = h - step * grad
        h_new = project_feasible(h_unt, n)
        
        # Backtracking if needed (check every 10 iterations)
        if it % 10 == 0 and it > 0:
            val_new, _, _ = smooth_max_and_grad(h_new, beta, topK)
            if val_new > val + 1e-10:
                step *= 0.5
                h_unt = h - step * grad
                h_new = project_feasible(h_unt, n)
            elif it % 100 == 0:
                step *= 1.1
                step = min(step, 200.0)
        
        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        h_next = h_new + ((t - 1.0) / t_new) * (h_new - h_prev)
        h_next = np.clip(h_next, 0.0, 1.0)
        t = t_new
        h_prev = h_new
        h = h_next
        
        # Track best (using true C5, not smoothed)
        c5_true = compute_c5(h_new)
        if c5_true < best_val:
            best_val = c5_true
            best_h = h_new.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 2000:
            break
    
    # Final evaluation with true objective
    c5_best_h = compute_c5(best_h)
    c5_h = compute_c5(h_prev)
    if c5_h < c5_best_h:
        best_h = h_prev.copy()
        best_val = c5_h
    return best_h

def active_set_minimax(h_init, n, max_rounds=30, topK=3):
    """Minimize the max of the top-K correlation values."""
    h = h_init.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    step = 5.0 / n
    
    for rnd in range(max_rounds):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[::-1][:topK]
        top_vals = corr[top_idx]
        max_val = np.max(top_vals)
        
        # Subgradient from top K lags (weighted by proximity to max)
        weights = np.zeros(len(corr))
        for i, idx in enumerate(top_idx):
            weights[idx] = np.exp(50.0 * (top_vals[i] - max_val))
        weights /= np.sum(weights)
        
        grad = compute_gradient_direct(h, n, weights)
        
        # Line search
        alpha = step
        improved = False
        for _ in range(20):
            h_new = project_feasible(h - alpha * grad, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                h = h_new
                best_c5 = c5_new
                improved = True
                break
            alpha *= 0.5
            if alpha < 1e-10:
                break
        
        if not improved:
            # Try small random perturbations of active lags
            for _ in range(50):
                if remaining_time() < 2:
                    break
                h_pert = h.copy()
                mask = (h_pert > 0.02) & (h_pert < 0.98)
                if np.sum(mask) < 3:
                    break
                idxs = np.where(mask)[0][:5]
                for i in idxs:
                    h_pert[i] += np.random.uniform(-0.02, 0.02)
                h_pert = project_feasible(h_pert, n)
                c5_p = compute_c5(h_pert)
                if c5_p < best_c5 - 1e-15:
                    h = h_pert
                    best_c5 = c5_p
                    improved = True
                    break
            if not improved:
                step *= 0.8
                if step < 1e-8:
                    break
    
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def pair_transfer(h, n, max_sweeps=5, rng=None):
    """Local combinatorial search: transfer mass between pairs."""
    if rng is None:
        rng = np.random.RandomState(0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    window = min(15, n // 4)
    
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        any_imp = False
        order = rng.permutation(n)
        for ii in range(len(order)):
            if remaining_time() < 1:
                break
            i = int(order[ii])
            hi = best_h[i]
            if hi < 0.05 or hi > 0.95:
                continue
            for j in range(max(0, i-window), min(n, i+window+1)):
                if j == i:
                    continue
                hj = best_h[j]
                if abs(hi - hj) < 0.03:
                    continue
                for frac in [0.3, 0.6, 0.9]:
                    amt = (hi - hj) * frac
                    if abs(amt) < 0.005:
                        continue
                    h_trial = best_h.copy()
                    h_trial[i] -= amt
                    h_trial[j] += amt
                    h_trial = project_feasible(h_trial, n)
                    c5_t = compute_c5(h_trial)
                    if c5_t < best_c5 - 1e-14:
                        best_c5 = c5_t
                        best_h = h_trial.copy()
                        any_imp = True
                        break
        if not any_imp:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
        best_c5 = compute_c5(best_h)
    except Exception:
        best_n = 256
        best_h = project_feasible(np.full(best_n, 0.5), best_n)
        best_c5 = compute_c5(best_h)
    
    # Phase 1: FISTA at current resolution with increasing beta
    print(f"Initial: n={best_n}, C5={best_c5:.6f}")
    for beta in [50, 100, 200, 500, 1000, 2000, 5000, 10000]:
        if remaining_time() < 15:
            break
        h_opt = fista_optimize(best_h, best_n, beta, max_iter=4000, topK=8 if beta < 500 else None)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            print(f"  FISTA beta={beta}: C5={best_c5:.6f}")
    
    # Phase 2: Active-set minimax refinement
    if remaining_time() > 20:
        h_mm, c5_mm = active_set_minimax(best_h, best_n, max_rounds=40, topK=3)
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm.copy()
            print(f"  Minimax: C5={best_c5:.6f}")
    
    # Phase 3: Multi-resolution escalation
    trial_ns = [320, 384, 448, 512, 576, 640, 768, 896, 1024]
    for trial_n in trial_ns:
        if remaining_time() < 25:
            break
        h_interp = interp_to(best_h, best_n, trial_n)
        h_interp = project_feasible(h_interp, trial_n)
        c5_interp = compute_c5(h_interp)
        
        if c5_interp < best_c5 + 0.004:
            improved = False
            for beta in [200, 1000, 3000]:
                if remaining_time() < 8:
                    break
                h_opt = fista_optimize(h_interp, trial_n, beta, max_iter=3000, topK=8)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = trial_n
                    h_interp = h_opt.copy()
                    improved = True
                    print(f"  Res n={trial_n}, beta={beta}: C5={best_c5:.6f}")
            
            if improved and remaining_time() > 15:
                h_mm, c5_mm = active_set_minimax(h_interp, trial_n, max_rounds=20, topK=3)
                if c5_mm < best_c5:
                    best_c5 = c5_mm
                    best_h = h_mm.copy()
                    best_n = trial_n
                    h_interp = h_mm.copy()
                    print(f"  Res n={trial_n} minimax: C5={best_c5:.6f}")
    
    # Phase 4: Multi-start exploration
    for restart in range(5):
        if remaining_time() < 35:
            break
        # Different perturbation strategies
        if restart == 0:
            h_p = best_h.copy()
            mask = (h_p > 0.05) & (h_p < 0.95)
            h_p[mask] += rng.normal(0, 0.04, np.sum(mask))
        elif restart == 1:
            # Flip some bits
            h_p = best_h.copy()
            nflip = max(3, best_n // 20)
            idxs = rng.choice(best_n, size=nflip, replace=False)
            for i in idxs:
                h_p[i] = 1.0 - h_p[i]
        elif restart == 2:
            # Scale toward 0.5
            h_p = 0.7 * best_h + 0.3 * 0.5
        elif restart == 3:
            # Circular shift
            shift = rng.randint(1, best_n // 4 + 1)
            h_p = np.roll(best_h, shift)
        else:
            # Mirror
            h_p = best_h[::-1].copy()
        
        h_p = project_feasible(h_p, best_n)
        c5_p = compute_c5(h_p)
        
        if c5_p < best_c5 + 0.01:
            h_opt = fista_optimize(h_p, best_n, 500, max_iter=3000, topK=8)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                print(f"  Restart {restart}: C5={best_c5:.6f}")
    
    # Phase 5: Combinatorial refinement
    if remaining_time() > 15:
        h_comb, c5_comb = pair_transfer(best_h, best_n, max_sweeps=6, rng=rng)
        if c5_comb < best_c5:
            best_c5 = c5_comb
            best_h = h_comb.copy()
            print(f"  Combinatorial: C5={best_c5:.6f}")
    
    # Phase 6: Final FISTA polish with very high beta
    if remaining_time() > 10:
        for beta in [10000, 30000, 100000]:
            if remaining_time() < 4:
                break
            h_opt = fista_optimize(best_h, best_n, beta, max_iter=2000, topK=3)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                print(f"  Polish beta={beta}: C5={best_c5:.6f}")
    
    # Phase 7: Final active-set pass
    if remaining_time() > 8:
        h_mm, c5_mm = active_set_minimax(best_h, best_n, max_rounds=20, topK=2)
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm.copy()
    
    # Phase 8: Final combinatorial
    if remaining_time() > 5:
        h_comb, c5_comb = pair_transfer(best_h, best_n, max_sweeps=3, rng=rng)
        if c5_comb < best_c5:
            best_c5 = c5_comb
            best_h = h_comb.copy()
    
    # Final projection
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    print(f"Final: n={best_n}, C5={final_c5:.8f}")
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
