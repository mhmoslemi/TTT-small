The full correlation is:

So for a = h, v = 1-h:

Let me define the lag as L = k - (n-1), so L ranges from -(n-1) to (n-1).

c = sum_i h[i] * (1 - h[i + L])

where the sum is over valid i (0 ≤ i < n and 0 ≤ i+L < n).

If L ≥ 0: valid i is 0 ≤ i < n-L
If L < 0: valid i is -L ≤ i < n

- From first sum: 1 if 0 ≤ j < n-L, else 0
- From second sum:

- Total: [j < n-L] * (1 - h[j+L]) - [L ≤ j < n] * h[j-L]

For L < 0 (let m = -L > 0):

dc/d(h[j]):
- From first sum: 1 if m ≤ j < n, else 0
- From second sum:
  - h[j] appears as h[i] when i=j (if m ≤ j < n): contributes -h[j-m]
  - h[j] appears as h[i-m] when i=j+m (if m ≤ j+m < n, i.e., 0 ≤ j < n-m): contributes -h[j+m]
- Total: [m ≤ j < n] * (1 - h[j-m]) - [0 ≤ j < n-m] * h[j+m]

def subgrad_topk(h, n, top_k):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_indices = np.argsort(corr)[-top_k:]
    
    grad = np.zeros(n)
    for k_idx in top_indices:
        L = int(k_idx) - (n - 1)  # lag
        if L >= 0:
            n_valid = n - L
            # [j < n_valid] * (1 - h[j+L]) - [L <= j < n] * h[j-L]
            for j in range(n_valid):
                grad[j] += (1.0 - h[j + L])
            for j in range(L, n):
                grad[j] -= h[j - L]
        else:
            m = -L
            # [m <= j < n] * (1 - h[j-m]) - [0 <= j < n-m] * h[j+m]
            for j in range(m, n):
                grad[j] += (1.0 - h[j - m])
            for j in range(n - m):
                grad[j] -= h[j + m]
    
    return grad, top_indices, corr

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argsort(h)
            for i in idx:
                add = min(1.0 - h[i], diff)
                if add > 0:
                    h[i] += add
                    diff -= add
                if diff < 1e-12:
                    break
        else:
            idx = np.argsort(h)[::-1]
            for i in idx:
                rem = min(h[i], -diff)
                if rem > 0:
                    h[i] -= rem
                    diff += rem
                if diff > -1e-12:
                    break
    return h


def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 3.0)


def subgrad_topk(h, n, top_k):
    """Compute subgradient of max of top-k correlations."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    top_indices = np.argsort(corr)[-top_k:]
    
    grad = np.zeros(n)
    for k_idx in top_indices:
        L = int(k_idx) - (n - 1)
        if L >= 0:
            n_valid = n - L
            if n_valid <= 0:
                continue
            # dc/d(h[j]) for j in [0, n_valid): += (1 - h[j+L])
            grad[:n_valid] += (1.0 - h[L:L + n_valid])
            # dc/d(h[j]) for j in [L, n): -= h[j-L]
            grad[L:] -= h[:n_valid]
        else:
            m = -L
            n_valid = n - m
            if n_valid <= 0:
                continue
            # dc/d(h[j]) for j in [m, n): += (1 - h[j-m])
            grad[m:] += (1.0 - h[:n_valid])
            # dc/d(h[j]) for j in [0, n-m): -= h[j+m]
            grad[:n_valid] -= h[m:m + n_valid]
    
    return grad, top_indices, corr


def subgrad_optimize(h, n, max_iters=500, base_step=0.1, topk_schedule=None):

    h_best = h.copy()

        topk_schedule = []

            break

            tk = 3 + it // 50

def con_sum(x):
    return np.sum(x) - _G_TGT


def obj_smooth(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(_G_BETA * shifted, -500, 500))
    return (m + np.log(np.sum(exps)) / _G_BETA) * dx


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (np.binary_repr(i).count('1') % 2 == 1) else 0.0
    return h
