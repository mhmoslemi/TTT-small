```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def solve_lp(h_old, n, solver='ECOS'):
    """Solve the linearized LP for the current h_old."""
    dx = 2.0 / n
    m = 2 * n - 1  # number of lags
    
    # Build A and b
    A = np.zeros((m, n))
    b = np.zeros(m)
    
    h_o = h_old
    
    for j in range(n):  # positive lag j (including 0)
        k = n - 1 + j  # index in correlation
        for i in range(n - j):
            A[k, i + j] += (1.0 - h_o[i])
            A[k, i] -= h_o[i + j]
            b[k] += h_o[i] * h_o[i + j]
    
    for j in range(1, n):  # negative lag -j
        k = n - 1 - j  # index in correlation
        for i in range(n - j):
            A[k, i] += (1.0 - h_o[i + j])
            A[k, i + j] -= h_o[i]
            b[k] += h_o[i] * h_o[i + j]
    
    # LP: minimize t
    # s.t. dx*(A@h + b) <= t  =>  dx*A@h - t <= -dx*b
    #      sum(h) = n/2
    #      0 <= h <= 1
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0,
        dx * (A @ h_var + b) <= t_var
    ]
    
    objective = cp.Minimize(t_var)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
        if h_var.value is not None:
            h_new = np.array(h_var.value).flatten()
            h_new = project_feasible(h_new, n)
            return h_new, float(t_var.value) if t_var.value is not None else float('inf')
    except Exception:
        pass
    
    return h_old.copy(), float('inf')
```

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def build_AB(h_old, n):
    """Build linearization matrices A, b for the correlation.
    corr_lin[k] = A[k,:] @ h + b[k]
    """
    dx_factor = 1.0  # not used here, applied outside
    m = 2 * n - 1
    A = np.zeros((m, n))
    b = np.zeros(m)
    h_o = h_old
    
    # Positive lags j = 0, 1, ..., n-1 (correlation index n-1+j)
    for j in range(n):
        k = n - 1 + j
        # corr[n-1+j] = sum_{i=0}^{n-1-j} h[i+j]*(1-h[i])
        # Linearized: sum_{i=0}^{n-1-j} [(1-h_old[i])*h[i+j] - h_old[i+j]*h[i] + h_old[i]*h_old[i+j]]
        if n - j > 0:
            idx_i = np.arange(n - j)  # i = 0, ..., n-1-j
            idx_p = idx_i + j        # i+j = j, ..., n-1
            A[k, idx_p] += (1.0 - h_o[idx_i])
            A[k, idx_i] -= h_o[idx_p]
            b[k] += np.dot(h_o[idx_i], h_o[idx_p])
    
    # Negative lags j = 1, 2, ..., n-1 (correlation index n-1-j)
    for j in range(1, n):
        k = n - 1 - j
        # corr[n-1-j] = sum_{i=0}^{n-1-j} h[i]*(1-h[i+j])
        # Linearized: sum_{i=0}^{n-1-j} [(1-h_old[i+j])*h[i] - h_old[i]*h[i+j] + h_old[i]*h_old[i+j]]
        if n - j > 0:
            idx_i = np.arange(n - j)
            idx_p = idx_i + j
            A[k, idx_i] += (1.0 - h_o[idx_p])
            A[k, idx_p] -= h_o[idx_i]
            b[k] += np.dot(h_o[idx_i], h_o[idx_p])
    
    return A, b

def solve_lp(h_old, n, solver='ECOS'):
    """Solve the linearized LP."""
    dx = 2.0 / n
    A, b = build_AB(h_old, n)
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0,
        dx * (A @ h_var + b) <= t_var
    ]
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    try:
        prob.solve(solver=solver)
        if h_var.value is not None:
            h_new = np.array(h_var.value).flatten()
            h_new = project_feasible(h_new, n)
            return h_new
    except Exception:
        pass
    
    # Try other solver
    try:
        prob.solve(solver='SCS')
        if h_var.value is not None:
            h_new = np.array(h_var.value).flatten()
            h_new = project_feasible(h_new, n)
            return h_new
    except Exception:
        pass
    
    return h_old.copy()

def lp_iteration(h, n, n_iters=50, tlimit=None):
    """Iterative LP linearization."""
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    h_best = h_cur.copy()
    c5_best = c5_cur
    
    for it in range(n_iters):
        if tlimit is not None and tlimit() < 2.0:
            break
        
        h_new = solve_lp(h_cur, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < c5_cur - 1e-14:
            h_cur = h_new
            c5_cur = c5_new
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h_cur.copy()
        else:
            # Damped update
            for alpha in [0.5, 0.3, 0.1]:
                h_d = (1 - alpha) * h_cur + alpha * h_new
                h_d = project_feasible(h_d, n)
                c5_d = compute_c5(h_d)
                if c5_d < c5_cur - 1e-15:
                    h_cur = h_d
                    c5_cur = c5_d
                    if c5_cur < c5_best:
                        c5_best = c5_cur
                        h_best = h_cur.copy()
                    break
            else:
                # No improvement, try to perturb
                break
    
    return h_best, c5_best

# Global variables for SLSQP objective
_G_BETA = 1.0
_G_DX = 1.0
_G_N = 0
_G_TGT = 0.0

def obj_smax(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = corr - m
    b = _G_BETA
    e = np.exp(np.clip(b * s, -400, 400))
    return float((m + np.log(np.sum(e)) / b) * _G_DX)

def slsqp_smax(h0, n, beta, maxiter=300):
    global _G_BETA, _G_DX, _G_N, _G_TGT
    _G_BETA = float(beta)
    _G_DX = 2.0 / n
    _G_N = n
    _G_TGT = n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    
    try:
        res = minimize(obj_smax, h0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def coord_descent(h, n, max_sweeps=20, rng=None, tlimit=None):
    c5 = compute_c5(h)
    h_best = h.copy()
    if rng is None:
        rng = np.random.RandomState(42)
    
    for sweep in range(max_sweeps):
        if tlimit is not None and tlimit() < 2.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if tlimit is not None and tlimit() < 1.0:
                break
            v0 = h_best[i]
            # Try setting to 0 or 1
            for v in [0.0, 1.0]:
                if abs(v - v0) < 1e-9:
                    continue
                trial = h_best.copy()
                trial[i] = v
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 5e-10:
                    c5 = c
                    h_best = trial.copy()
                    improved = True
            # Try small perturbations
            for d in [0.1, -0.1, 0.05, -0.05, 0.2, -0.2]:
                nv = v0 + d
                if nv < 0 or nv > 1:
                    continue
                trial = h_best.copy()
                trial[i] = nv
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 5e-10:
                    c5 = c
                    h_best = trial.copy()
                    improved = True
        if not improved and sweep >= 3:
            break
    
    return h_best, c5

def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c else 0.0
    return h
```
