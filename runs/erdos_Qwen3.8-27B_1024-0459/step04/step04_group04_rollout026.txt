def corr_grad_per_shift(x, n, k):
    """Gradient of corr[k] with respect to x."""
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        if npairs <= 0:
            return g
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        if npairs <= 0:
            return g
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g

def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        if npairs <= 0:
            return g
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        if npairs <= 0:
            return g
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g

        d = d / d_norm

        improved = False
        for ts in [0.05, 0.02, 0.01, 0.005, 0.1]:
            h_trial = h + ts * d
            h_trial = project_feasible(h_trial, n)
            c5_new = compute_c5(h_trial)
            if c5_new < best_c5 - 1e-15:
                h = h_trial
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                break

        if not improved:
            continue

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 360
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Binary optimization - focus on primary n and neighbors
    # Primary resolution
    n_primary = best_n if best_n % 2 == 0 else best_n + 1
    n_list = [n_primary, n_primary - 20, n_primary + 20,
              n_primary - 40, n_primary + 40,
              n_primary + 80, n_primary + 160]
    n_list = sorted(set([n for n in n_list if n % 2 == 0 and 160 <= n <= 800]))

    # Remove duplicates and sort
    n_list = sorted(set(n_list))

    t_per_n = min(80, max(20, 400.0 / len(n_list)))

    for n_t in n_list:
        if remaining() < 20:
            break

        if n_t == best_n:
            h_interp = best_h.copy()
        else:
            h_interp = interp_to(best_h, best_n, n_t)

        h_bin = make_binary(h_interp, n_t, rng)

        # SA phase
        t_sa = min(t_per_n * 0.5, remaining() * 0.3)
        if t_sa > 5:
            h_sa, c5_sa = sa_binary_enhanced(h_bin, n_t, time_limit=t_sa, seed=seed + n_t)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_t

        # ILS phase
        t_ils = min(t_per_n * 0.5, remaining() * 0.3)
        if t_ils > 5:
            h_ils, c5_ils = ilsa_binary(h_bin, n_t, time_limit=t_ils, seed=seed + 10000 + n_t)
            if c5_ils < best_c5:
                best_c5 = c5_ils
                best_h = h_ils.copy()
                best_n = n_t

        # Update best_n for interpolation reference
        if best_n != n_t:
            # Keep best_h at its native resolution

    # Phase 2: From best binary solution, try continuous refinement at several n values
    n_cont_list = [best_n, best_n - 10, best_n + 10, best_n + 20, best_n - 20]
    n_cont_list = sorted(set([n for n in n_cont_list if n % 2 == 0 and 160 <= n <= 800]))

    for n_t in n_cont_list:
        if remaining() < 40:
            break

        if n_t == best_n:
            h_init = best_h.copy()
        else:
            h_init = interp_to(best_h, best_n, n_t)
            h_init = project_feasible(h_init, n_t)
            # Also try binary at this n
            h_bin = make_binary(h_init, n_t, rng)
            # Quick SA
            if remaining() > 20:
                h_sa, c5_sa = sa_binary_enhanced(h_bin, n_t, time_limit=20, seed=seed + 20000 + n_t)
                if c5_sa < compute_c5(h_init):
                    h_init = h_sa

        # SLSQP smooth optimization
        t_slq = min(60, max(15, remaining() * 0.4))
        h_ref = h_init.copy()
        for beta in [500, 2000, 8000, 30000]:
            if remaining() < 8:
                break
            if time.time() - _START > _BUDGET * 0.4:
                break
            h_new = slsqp_smooth(h_ref, n_t, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_t
                h_ref = h_new.copy()

    # Phase 3: Intensive continuous refinement at best_n
    n = best_n
    h = best_h.copy()

    # Top-k subgradient
    if remaining() > 20:
        t_sub = min(80, max(20, remaining() * 0.35))
        h_sub, c5_sub = topk_subgradient(h, n, max_iters=3000, top_k=8, time_limit=t_sub)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()
            h = h_sub.copy()

    # More SLSQP at higher beta
    for beta in [50000, 150000, 500000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Fine subgradient polish
    if remaining() > 15:
        t_fine = min(60, max(10, remaining() * 0.4))
        h_fine, c5_fine = fine_subgradient_polish(h, n, time_limit=t_fine)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()
            h = h_fine.copy()

    # Random direction search
    if remaining() > 15:
        t_rand = min(40, max(5, remaining() * 0.3))
        h_rand, c5_rand = random_direction_search(h, n, n_trials=1000, time_limit=t_rand)
        if c5_rand < best_c5:
            best_c5 = c5_rand
            best_h = h_rand.copy()
            h = h_rand.copy()

    # Phase 4: Try binary local search on rounded version, then re-refine
    if remaining() > 30:
        h_bin = make_binary(best_h, n, rng)
        h_ils, c5_ils = ilsa_binary(h_bin, n, time_limit=min(30, remaining() * 0.4), seed=seed + 50000)
        if c5_ils < best_c5:
            best_c5 = c5_ils
            best_h = h_ils.copy()
            h = h_ils.copy()
            # Re-refine continuously
            if remaining() > 20:
                for beta in [10000, 100000]:
                    if remaining() < 5:
                        break
                    h_new = slsqp_smooth(h, n, beta, maxiter=300)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h = h_new.copy()

    # Phase 5: Final ultra-fine polish
    if remaining() > 10:
        h = best_h.copy()
        for _ in range(3):
            if remaining() < 5:
                break
            h_fine, c5_fine = fine_subgradient_polish(h, n, time_limit=min(20, remaining() * 0.4))
            if c5_fine < best_c5:
                best_c5 = c5_fine
                best_h = h_fine.copy()
                h = h_fine.copy()
            h_sub, c5_sub = topk_subgradient(h, n, max_iters=500, top_k=5, time_limit=min(10, remaining() * 0.3))
            if c5_sub < best_c5:
                best_c5 = c5_sub
                best_h = h_sub.copy()
                h = h_sub.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
