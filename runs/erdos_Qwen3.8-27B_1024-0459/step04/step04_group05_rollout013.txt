from scipy.optimize import minimize
import time

_G_N = 0
_G_BETA = 1.0
_G_TGT = 0.0
_G_DX = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    """Proper projection onto {x: 0<=x_i<=1, sum(x)=n/2}"""
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    # Bisection for lambda
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine adjustment for exact sum
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-14:
        mask = (h_proj > 1e-12) & (h_proj < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h_proj[mask] += diff / np.sum(mask)
        else:
            h_proj[:] += diff / n
        h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def grad_corr_shift(h, n, d):
    """Gradient of correlation at shift d (where d is the offset)"""
    g = np.zeros(n)
    if d >= 0:
        # f_d = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
        # df_d/dh[j] = (1-h[j+d]) for j in [0, n-1-d]
        #              - h[j-d] for j in [d, n-1]
        g[:n-d] += (1.0 - h[d:n])
        g[d:] -= h[:n-d]
    else:
        # f_d = sum_{i=-d}^{n-1} h[i]*(1-h[i+d]) where d<0
        # Let d' = -d
        dp = -d
        # f = sum_{i=dp}^{n-1} h[i]*(1-h[i-dp])
        # df/dh[j] = (1-h[j-dp]) for j in [dp, n-1]
        #            - h[j+dp] for j in [0, n-1-dp]
        g[dp:] += (1.0 - h[:n-dp])
        g[:n-dp] -= h[dp:]
    return g

def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)

def obj_lse(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _G_BETA
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    return (m + np.log(np.sum(exps)) / beta) * dx

def con_sum(x):
    return np.sum(x) - _G_TGT

def do_slsqp(h_start, n, beta_list, budget_fn):
    global _G_N, _G_BETA, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    for beta in beta_list:
        if budget_fn() < 6.0:

        _G_BETA = float(beta)
        maxiter = 300

                           bounds=bounds, constraints=cons,

            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:

        except Exception:

    return h_cur, c5_cur

    """Newton method to equalize top-k correlations"""
    h = h.copy()
    c5_best = compute_c5(h)
    corr = compute_corr(h)
    
    for it in range(n_iters):
        if budget_fn() < 3.0:

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_BETA = 1.0
_G_TGT = 0.0
_G_DX = 1.0
_G_START = 0.0
_G_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    lo, hi = -float(h).max() - 2.0, float(h).max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-13:
        mask = (h_proj > 1e-12) & (h_proj < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h_proj[mask] += diff / cnt
        else:
            h_proj[:] += diff / n
        h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj


def grad_corr_shift(h, n, d):
    g = np.zeros(n)
    if d >= 0:
        g[:n - d] += (1.0 - h[d:n])
        g[d:] -= h[:n - d]
    else:
        dp = -d
        g[dp:] += (1.0 - h[:n - dp])
        g[:n - dp] -= h[dp:]
    return g


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 4.0)


def obj_lse(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _G_BETA
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    return (m + np.log(np.sum(exps)) / beta) * dx


def con_sum(x):
    return np.sum(x) - _G_TGT


def con_sum_jac(x):
    return np.ones(_G_N)


def do_slsqp(h_start, n, beta_list, maxiter_override=None):
    global _G_N, _G_BETA, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    for beta in beta_list:
        if remaining_time() < 5.0:
            break
        _G_BETA = float(beta)
        mi = 400
        if maxiter_override:
            mi = maxiter_override
        try:
            res = minimize(obj_lse, h_cur, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': mi, 'ftol': 1e-16, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, c5_cur


def newton_equalize(h, n, top_k, n_iters):
    h = h.copy()
    c5_best = compute_c5(h)
    no_improve = 0
    dx = 2.0 / n

    for it in range(n_iters):
        if remaining_time() < 3.0:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[-top_k:]
        top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
        f_top = corr[top_idx]
        c5_cur = f_top[0] * dx

        if c5_cur < c5_best - 1e-15:
            c5_best = c5_cur

        f_bar = np.mean(f_top)
        f_dev = f_top - f_bar
        f_dev_norm = np.linalg.norm(f_dev)

        if f_dev_norm < 1e-12:
            grad = np.zeros(n)
            for idx in top_idx:
                d = int(idx) - (n - 1)
                grad += grad_corr_shift(h, n, d)
        else:
            G = np.zeros((top_k, n))
            for j in range(top_k):
                d = int(top_idx[j]) - (n - 1)
                G[j] = grad_corr_shift(h, n, d)
            GtG = G @ G.T
            GtG += 1e-10 * np.eye(top_k)
            try:
                direction_weights = np.linalg.solve(GtG, f_dev)
            except np.linalg.LinAlgError:
                direction_weights = f_dev / f_dev_norm
            grad = G.T @ direction_weights

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm

        improved = False
        steps = [0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        best_trial_h = None
        best_trial_c5 = c5_cur
        for step in steps:
            h_trial = project_feasible(h - step * grad, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_trial_c5:
                best_trial_c5 = c5_trial
                best_trial_h = h_trial

        if best_trial_h is not None and best_trial_c5 < c5_best - 1e-14:
            h = best_trial_h
            c5_best = best_trial_c5
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                break

    return h, c5_best


def subgrad_broad(h, n, n_iters, noise_scale):
    rng_sg = np.random.RandomState(12345)
    h = h.copy()
    c5_best = compute_c5(h)
    no_improve = 0
    dx = 2.0 / n

    for it in range(n_iters):
        if remaining_time() < 3.0:
            break
        corr = compute_corr(h)
        max_val = np.max(corr)
        threshold = max_val - 1.0
        active = np.where(corr >= threshold)[0]

        grad = np.zeros(n)
        for idx in active:
            d = int(idx) - (n - 1)
            grad += grad_corr_shift(h, n, d)
        grad *= dx

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm

        improved = False
        base = 0.04
        for step in [base, base * 0.5, base * 0.25, base * 2.0, base * 0.1, base * 4.0]:
            h_trial = project_feasible(h - step * grad, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-13:
                c5_best = c5_trial
                h = h_trial
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            if no_improve > 60:
                for _ in range(8):
                    noise = rng_sg.randn(n) * noise_scale
                    h_trial = project_feasible(h + noise, n)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < c5_best - 1e-13:
                        c5_best = c5_trial
                        h = h_trial
                        no_improve = 0
                        break
                if no_improve > 150:
                    break

    return h, c5_best


def coord_descent_binary(h, n, n_sweeps):
    rng_cd = np.random.RandomState(67890)
    h = h.copy()
    c5_best = compute_c5(h)
    for sweep in range(n_sweeps):
        if remaining_time() < 4.0:
            break
        any_imp = False
        order = rng_cd.permutation(n)
        for i in order:
            if remaining_time() < 2.0:
                break
            old_val = h[i]
            for new_val in [0.0, 1.0]:
                if abs(new_val - old_val) < 1e-8:
                    continue
                h[i] = new_val
                h = project_feasible(h, n)
                c5_test = compute_c5(h)
                if c5_test < c5_best - 1e-13:
                    c5_best = c5_test
                    any_imp = True
                else:
                    h[i] = old_val
                    h = project_feasible(h, n)
        if not any_imp and sweep >= 2:
            break
    return h, c5_best


def do_local_binary_search(h, n, rng, n_trials=2000):
    h = h.copy()
    c5_best = compute_c5(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]

    for trial in range(n_trials):
        if remaining_time() < 2.0:
            break
        if len(ones) < 2 or len(zeros) < 2:
            break
        i1 = ones[rng.randint(len(ones))]
        i2 = zeros[rng.randint(len(zeros))]
        h[i1] = 0.0
        h[i2] = 1.0
        h = project_feasible(h, n)
        c5_trial = compute_c5(h)
        if c5_trial < c5_best - 1e-14:
            c5_best = c5_trial
        else:
            h[i1] = 1.0
            h[i2] = 0.0
            h = project_feasible(h, n)

    return h, c5_best


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h


def make_sturmian(n):
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return h
