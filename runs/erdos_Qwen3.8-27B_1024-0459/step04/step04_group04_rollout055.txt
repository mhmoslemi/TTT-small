def slsqp_smooth(h, n, beta, top_k=None, maxiter=300):
    dx_val = 2.0 / n
    target = n / 2.0

    def obj(x):
        corr = compute_corr(x)
        if top_k is not None and top_k < len(corr):
            idx = np.argpartition(corr, -top_k)[-top_k:]
            ct = corr[idx]
        else:
            ct = corr
        m = np.max(ct)
        sh = np.clip(beta * (ct - m), -60, 60)
        return (m + np.log(np.sum(np.exp(sh))) / beta) * dx_val

    def grad(x):
        corr = compute_corr(x)
        if top_k is not None and top_k < len(corr):
            idx = np.argpartition(corr, -top_k)[-top_k:]
            ct = corr[idx]
            m = np.max(ct)
            sh = np.clip(beta * (ct - m), -60, 60)
            ex = np.exp(sh)
            w = ex / np.sum(ex)
            g = np.zeros(n)
            for wi, k in zip(w, idx):
                if wi > 1e-10:
                    g += wi * _corr_grad_single(x, n, int(k))
            return g * dx_val
        else:
            m = np.max(corr)
            sh = np.clip(beta * (corr - m), -60, 60)
            ex = np.exp(sh)
            w = ex / np.sum(ex)
            g = np.zeros(n)
            for k in range(len(corr)):
                if w[k] > 1e-10:
                    g += w[k] * _corr_grad_single(x, n, k)
            return g * dx_val

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def free_var_slsqp(h, n, beta, maxiter=500, tol=0.05):
    """Optimize only the interior (free) variables."""
    free_mask = (h > tol) & (h < 1.0 - tol)
    n_free = int(np.sum(free_mask))
    if n_free == 0:
        return h
    if n_free >= n:
        return slsqp_smooth(h, n, beta, maxiter=maxiter)

    fixed_sum = np.sum(h[~free_mask])
    target_free = n / 2.0 - fixed_sum
    free_h = h[free_mask].copy()

    dx_val = 2.0 / n

    def obj(x):
        h_full = h.copy()
        h_full[free_mask] = np.clip(x, 0, 1)
        corr = compute_corr(h_full)
        ct = corr  # use all shifts for safety
        m = np.max(ct)
        sh = np.clip(beta * (ct - m), -60, 60)
        return (m + np.log(np.sum(np.exp(sh))) / beta) * dx_val

    def grad(x):
        h_full = h.copy()
        h_full[free_mask] = np.clip(x, 0, 1)
        corr = compute_corr(h_full)
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -60, 60)
        ex = np.exp(sh)
        w = ex / np.sum(ex)
        g_full = np.zeros(n)
        for k in range(len(corr)):
            if w[k] > 1e-10:
                g_full += w[k] * _corr_grad_single(h_full, n, k)
        return g_full[free_mask] * dx_val

    bounds = [(0.0, 1.0)] * n_free

             'jac': lambda x: np.ones(n_free)}]

def topk_gradient_descent(h, n, max_iters=2000, top_k=5, lr_init=1e-3):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_corr_max = np.max(compute_corr(h))
    dx = 2.0 / n
    lr = lr_init

    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        m = float(np.max(corr))

        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))

        g = np.zeros(n)
        for k in top_idx:
            g += _corr_grad_single(h, n, int(k))
        g /= len(top_idx)

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        d = -g / gn
        improved = False
        for step in [lr, lr*0.5, lr*0.25, lr*0.1, lr*0.05]:
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-15:
                h = h_trial
                best_c5 = c_new
                best_corr_max = np.max(compute_corr(h))
                improved = True
                break

        if not improved:
            lr *= 0.8
            if lr < 1e-12:
                break

    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 360
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Intensive binary SA at parent resolution
    n = best_n
    for sa_seed in [seed, seed+100, seed+200, seed+300, seed+400]:
        if remaining() < 40:
            break
        h_bin = make_binary(best_h, n, rng)
        t_sa = min(40, max(15, remaining() / 5))
        h_sa, c5_sa = sa_binary(h_bin, n, time_limit=t_sa, seed=sa_seed, multi_swap=True)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n

    # Phase 2: Subgradient refinement at parent resolution
    if remaining() > 30:
        h_sub, c5_sub = subgradient_refine(best_h, n, max_iters=30000, lr_init=5e-4, top_k=5)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()

    # Phase 3: Free-variable SLSQP at parent resolution
    if remaining() > 60:
        for beta in [500, 2000, 10000]:
            if remaining() < 20:
                break
            h_fv = free_var_slsqp(best_h, n, beta=beta, maxiter=400, tol=0.03)
            c5_fv = compute_c5(h_fv)
            if c5_fv < best_c5:
                best_c5 = c5_fv
                best_h = h_fv.copy()

    # Phase 4: Full SLSQP at parent resolution
    if remaining() > 60:
        for beta in [1000, 5000, 20000]:
            if remaining() < 25:
                break
            h_sl = slsqp_smooth(best_h, n, beta=beta, maxiter=300)
            c5_sl = compute_c5(h_sl)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()

    # Phase 5: Top-k gradient descent
    if remaining() > 30:
        h_td, c5_td = topk_gradient_descent(best_h, n, max_iters=3000, top_k=3, lr_init=1e-3)
        if c5_td < best_c5:
            best_c5 = c5_td
            best_h = h_td.copy()

    # Phase 6: Upsample to higher resolution and refine
    n2 = n * 2  # e.g., 720
    if remaining() > 120:
        h2 = interp_to(best_h, best_n, n2)
        h2 = project_feasible(h2, n2)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n2

        # Subgradient at higher res
        if remaining() > 60:
            h_sub2, c5_sub2 = subgradient_refine(best_h, best_n, max_iters=20000, lr_init=3e-4, top_k=5)
            if c5_sub2 < best_c5:
                best_c5 = c5_sub2
                best_h = h_sub2.copy()

        # Free-var SLSQP at higher res
        if remaining() > 60:
            for beta in [1000, 5000]:
                if remaining() < 30:
                    break
                h_fv2 = free_var_slsqp(best_h, best_n, beta=beta, maxiter=300, tol=0.05)
                c5_fv2 = compute_c5(h_fv2)
                if c5_fv2 < best_c5:
                    best_c5 = c5_fv2
                    best_h = h_fv2.copy()

        # Top-k at higher res
        if remaining() > 30:
            h_td2, c5_td2 = topk_gradient_descent(best_h, best_n, max_iters=3000, top_k=5, lr_init=5e-4)
            if c5_td2 < best_c5:
                best_c5 = c5_td2
                best_h = h_td2.copy()

    # Phase 7: Binary SA at higher resolution (if we're at higher res)
    if remaining() > 60:
        n_sa = best_n
        h_bin = make_binary(best_h, n_sa, rng)
        for sa_seed in [seed+500, seed+600, seed+700]:
            if remaining() < 30:
                break
            h_sa, c5_sa = sa_binary(h_bin, n_sa, time_limit=min(30, remaining()/3), seed=sa_seed, multi_swap=True)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_sa

    # Phase 8: Final intensive refinement
    n_final = best_n
    h_final = best_h.copy()

    # Multiple rounds of subgradient + SLSQP
    for round_i in range(5):
        if remaining() < 15:
            break
        h_sub, c5_sub = subgradient_refine(h_final, n_final, max_iters=15000, lr_init=2e-4, top_k=3)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()
            h_final = h_sub.copy()

        if remaining() < 10:
            break
        h_fv = free_var_slsqp(h_final, n_final, beta=10000, maxiter=200, tol=0.02)
        c5_fv = compute_c5(h_fv)
        if c5_fv < best_c5:
            best_c5 = c5_fv
            best_h = h_fv.copy()
            h_final = h_fv.copy()

    # Phase 9: Final top-k polish
    if remaining() > 10:
        h_td, c5_td = topk_gradient_descent(best_h, best_n, max_iters=5000, top_k=3, lr_init=5e-4)
        if c5_td < best_c5:
            best_c5 = c5_td
            best_h = h_td.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
