def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Use bisection on tau for exact projection
    current_sum = np.sum(h)
    if abs(current_sum - target) < 1e-14:
        return h
    # Find tau such that sum(clip(h - tau, 0, 1)) = target
    lo, hi = -1.0, 1.0
    if current_sum > target:
        # Need to decrease: tau > 0
        lo, hi = 0.0, 2.0
    else:
        # Need to increase: tau < 0
        lo, hi = -2.0, 0.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        h_t = np.clip(h - tau, 0.0, 1.0)
        s = np.sum(h_t)
        if s > target:
            lo = tau
        else:
            hi = tau
    tau = 0.5 * (lo + hi)
    h = np.clip(h - tau, 0.0, 1.0)
    # Final fix for floating point
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]

    corr = compute_corr(h)

        lags_to_use = top_idx

            h_trial = project_feasible(h_best + v, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-13:
                c5_best = c5_trial
                h_best = h_trial.copy()
                improved = True
                break
            alpha *= 0.5
        
        if improved:
            patience = 0
            step = min(alpha * 1.2, 0.3)
        else:
            patience += 1
            step *= 0.5
            v *= 0.3
            if patience > 20:
                break
    
    return h_best, c5_best


def slsqp_smooth(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(_G_BETA * (corr - m), -600, 600)
    exps = np.exp(shifted)
    return (m + np.log(np.sum(exps)) / _G_BETA) * dx


def slsqp_con(x):
    return np.sum(x) - _G_TGT


def slsqp_polish(h, n, max_iters=300):
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con}]
    
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    
    for beta in [100, 500, 2000, 5000, 10000]:
        if remaining_time() < 5.0:
            break
        _G_BETA = float(beta)
        try:
            res = minimize(slsqp_smooth, h_best, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iters, 'ftol': 1e-15, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-14:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass
    
    return h_best, c5_best


def local_search_2point(h, n, max_sweeps=5, rng=None):
    """Try 2-point swaps and changes."""
    if rng is None:
        rng = np.random.RandomState(0)
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    
    for sweep in range(max_sweeps):
        if remaining_time() < 3.0:
            break
        any_imp = False
        for _ in range(200):
            if remaining_time() < 2.0:
                break
            i, j = rng.choice(n, size=2, replace=False)
            for move in range(3):
                h_trial = h_best.copy()
                if move == 0:
                    # Swap
                    h_trial[i], h_trial[j] = h_best[j], h_best[i]
                elif move == 1:
                    # Set both to 0.5
                    h_trial[i] = h_best[i] * 0.5 + 0.5 * (1.0 - h_best[i])
                    h_trial[j] = h_best[j] * 0.5 + 0.5 * (1.0 - h_best[j])
                    h_trial[i] = 0.5
                    h_trial[j] = 0.5
                    # Adjust sum
                else:
                    # Flip both
                    h_trial[i] = 1.0 - h_best[i]
                    h_trial[j] = 1.0 - h_best[j]
                
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < c5_best - 1e-13:
                    c5_best = c5_t
                    h_best = h_trial.copy()
                    any_imp = True
                    break
        if not any_imp and sweep > 1:
            break
    
    return h_best, c5_best


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h


def make_structured_random(rng, n):
    """Random block-based construction."""
    n_blocks = rng.randint(4, 12)
    block_size = n // n_blocks
    h = np.zeros(n)
    for b in range(n_blocks):
        start = b * block_size
        end = min(start + block_size, n)
        if rng.rand() > 0.5:
            h[start:end] = 1.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = project_feasible(make_thue_morse(parent_n), parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("Thue-Morse: n={}, C5={:.10f}".format(parent_n, best_c5))
    
    # Phase 1: FISTA with adaptive beta at parent resolution
    print("\n--- Phase 1: FISTA (n={}) ---".format(best_n))
    _G_N = best_n
    _G_TGT = best_n / 2.0
    _G_DX = 2.0 / best_n
    h_fista, c5_fista = fista_optimize(best_h, best_n, beta_start=30.0, beta_end=8000.0, 
                                        max_iter=1500, n_betas=6)
    if c5_fista < best_c5:
        best_c5 = c5_fista
        best_h = h_fista
    print("  FISTA: C5={:.10f}".format(best_c5))
    
    # Phase 2: SLSQP polish
    print("\n--- Phase 2: SLSQP polish ---")
    h_slsqp, c5_slsqp = slsqp_polish(best_h, best_n, max_iters=250)
    if c5_slsqp < best_c5:
        best_c5 = c5_slsqp
        best_h = h_slsqp
    print("  SLSQP: C5={:.10f}".format(best_c5))
    
    # Phase 3: Subgradient focus
    print("\n--- Phase 3: Subgradient focus ---")
    h_sg, c5_sg = subgrad_focused(best_h, best_n, max_iter=400, top_k=5, step0=0.08)
    if c5_sg < best_c5:
        best_c5 = c5_sg
        best_h = h_sg
    print("  Subgrad: C5={:.10f}".format(best_c5))
    
    # Phase 4: Multi-start with diverse initializations
    print("\n--- Phase 4: Multi-start ---")
    n = best_n
    init_list = []
    
    # Init 1: Current best perturbed
    init_list.append(best_h + 0.02 * rng.randn(n))
    
    # Init 2: Thue-Morse
    init_list.append(make_thue_morse(n))
    
    # Init 3: Structured random
    init_list.append(make_structured_random(rng, n))
    
    # Init 4: Golden ratio
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h_gr = np.zeros(n)
    for i in range(n):
        h_gr[i] = 1.0 if ((i * phi) % 1.0) < 0.5 else 0.0
    init_list.append(h_gr)
    
    # Init 5: Complementary Thue-Morse
    init_list.append(1.0 - make_thue_morse(n))
    
    for idx, init in enumerate(init_list):
        if remaining_time() < 20.0:
            break
        h_init = project_feasible(init, n)
        _G_N = n
        _G_TGT = n / 2.0
        _G_DX = 2.0 / n
        
        # Quick FISTA
        h_try, c5_try = fista_optimize(h_init, n, beta_start=50.0, beta_end=5000.0,
                                         max_iter=800, n_betas=4)
        # SLSQP
        h_try, c5_try = slsqp_polish(h_try, n, max_iters=150)
        
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            print("  Init {}: C5={:.10f}".format(idx, best_c5))
    
    print("  Best after multi-start: C5={:.10f}".format(best_c5))
    
    # Phase 5: Second FISTA round on best
    if remaining_time() > 60.0:
        print("\n--- Phase 5: FISTA round 2 ---")
        _G_N = best_n
        _G_TGT = best_n / 2.0
        _G_DX = 2.0 / best_n
        h_f2, c5_f2 = fista_optimize(best_h, best_n, beta_start=200.0, beta_end=15000.0,
                                      max_iter=1200, n_betas=5)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2
        print("  FISTA2: C5={:.10f}".format(best_c5))
    
    # Phase 6: Try different resolutions
    if remaining_time() > 60.0:
        print("\n--- Phase 6: Resolution search ---")
        for n_try in [200, 288, 320, 384]:
            if remaining_time() < 25.0:
                break
            h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
            _G_N = n_try
            _G_TGT = n_try / 2.0
            _G_DX = 2.0 / n_try
            
            h_try, c5_try = fista_optimize(h_interp, n_try, beta_start=50.0, beta_end=5000.0,
                                             max_iter=1000, n_betas=5)
            h_try, c5_try = slsqp_polish(h_try, n_try, max_iters=200)
            
            if c5_try < best_c5 - 1e-10:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
                print("  n={}: C5={:.10f} (new best)".format(n_try, best_c5))
            else:
                print("  n={}: C5={:.10f}".format(n_try, c5_try))
    
    # Phase 7: Local search
    if remaining_time() > 15.0:
        print("\n--- Phase 7: Local search ---")
        best_h, best_c5 = local_search_2point(best_h, best_n, max_sweeps=8, rng=rng)
        print("  Local: C5={:.10f}".format(best_c5))
    
    # Phase 8: Final SLSQP with high beta
    if remaining_time() > 10.0:
        print("\n--- Phase 8: Final polish ---")
        _G_N = best_n
        _G_TGT = best_n / 2.0
        _G_DX = 2.0 / best_n
        bounds = [(0.0, 1.0)] * best_n
        cons = [{'type': 'eq', 'fun': slsqp_con}]
        
        for beta in [3000, 8000, 15000, 25000]:
            if remaining_time() < 4.0:
                break
            _G_BETA = float(beta)
            try:
                res = minimize(slsqp_smooth, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-16, 'iprint': 0})
                h_new = project_feasible(res.x, best_n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new
            except Exception:
                pass
        print("  Polish: C5={:.10f}".format(best_c5))
    
    # Phase 9: One more subgradient round
    if remaining_time() > 8.0:
        best_h, best_c5 = subgrad_focused(best_h, best_n, max_iter=200, top_k=3, step0=0.05)
        print("  Final subgrad: C5={:.10f}".format(best_c5))
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
