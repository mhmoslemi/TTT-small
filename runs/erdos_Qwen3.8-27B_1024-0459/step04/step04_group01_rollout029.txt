    """Project h to be feasible: 0<=h<=1, sum(h)=n/2"""
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0

        if s > target:

        else:

def compute_swap_delta(h, i, j, n):
    """Compute the change in corr array when h[i]: 1->0 and h[j]: 0->1."""
    nc = 2 * n - 1
    delta = np.zeros(nc)
    center = n - 1
    
    # Effect of h[i]: 1 -> 0
    # As 'a' (the first argument): removes h[i]*(1-h[i-d]) = (1-h[i-d]) for valid d
    # i-d must be in [0, n-1], so d in [i-(n-1), i], clipped to [-(n-1), n-1]
    d_lo = max(-center, i - (n-1))
    d_hi = min(center, i)
    if d_lo <= d_hi:
        d_range = np.arange(d_lo, d_hi + 1)
        b_idx = i - d_range  # these are the 'b' indices
        delta[center + d_range] -= (1.0 - h[b_idx])
    
    # As 'b' (the second argument): the term was h[a]*(1-h[i]) = 0, now h[a]*(1-0) = h[a]
    # a = i + d must be in [0, n-1], so d in [-i, n-1-i], clipped to [-(n-1), n-1]
    d_lo = max(-center, -i)
    d_hi = min(center, n-1-i)
    if d_lo <= d_hi:
        d_range = np.arange(d_lo, d_hi + 1)
        a_idx = i + d_range
        delta[center + d_range] += h[a_idx]
    
    # Effect of h[j]: 0 -> 1
    # As 'a': adds (1-h[j-d]) for valid d
    d_lo = max(-center, j - (n-1))
    d_hi = min(center, j)
    if d_lo <= d_hi:
        d_range = np.arange(d_lo, d_hi + 1)
        b_idx = j - d_range
        delta[center + d_range] += (1.0 - h[b_idx])
    
    # As 'b': the term was h[a]*(1-h[j]) = h[a], now h[a]*(1-1) = 0
    d_lo = max(-center, -j)
    d_hi = min(center, n-1-j)
    if d_lo <= d_hi:
        d_range = np.arange(d_lo, d_hi + 1)
        a_idx = j + d_range
        delta[center + d_range] -= h[a_idx]
    
    return delta

            zeros_arr = np.array(zeros_list)

            cur_max = float(np.max(corr))
            stall = 0
            temp = min(temp * 5.0, 5.0)
