def make_smooth_obj(n, beta):
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        exps = np.exp(np.clip(beta * shifted, -500, 500))
        return (m + np.log(np.sum(exps)) / beta) * dx

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        weights = np.exp(np.clip(beta * shifted, -500, 500))
        weights = weights / weights.sum()
        n_corr = len(corr)
        grad = np.zeros(n)
        for k_idx in range(n_corr):
            w = weights[k_idx]
            if w < 1e-12:
                continue
            offset = k_idx - (n - 1)
            if offset >= 0:
                nv = n - offset
                if nv <= 0:
                    continue
                grad[offset:offset + nv] += w * (1.0 - x[:nv])
                grad[:nv] -= w * x[offset:offset + nv]
            else:
                moff = -offset
                nv = n - moff
                if nv <= 0:
                    continue
                grad[:nv] += w * (1.0 - x[moff:moff + nv])
                grad[moff:moff + nv] -= w * x[:nv]
        return grad * dx

    return obj, obj_grad

        return np.sum(x) - target

def interp_to(src_h, src_n, dst_n):

    return np.interp(x_new, xp, src_h)

        print("Thue-Morse: n={}, C5={:.10f}".format(parent_n, best_c5))

    # Phase 1: Binary SA at parent resolution
    t1 = min(350.0, remaining_time() * 0.30)
    print("\n--- Phase 1: Binary SA at n={} ({}s) ---".format(best_n, t1))
    h_sa, c5_sa, iters = sa_binary(best_h, best_n, time_limit=t1, t0=0.02, t_final=0.0003)
    print("  SA: {} iters, C5={:.10f}".format(iters, c5_sa))
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa
    else:
        # Keep parent, continue from there
        pass

    # Phase 2: SLSQP polish at parent resolution
    t2 = min(100.0, remaining_time() * 0.10)
    print("\n--- Phase 2: SLSQP polish at n={} ({}s) ---".format(best_n, t2))
    h_pol, c5_pol = slsqp_polish(best_h, best_n, time_limit=t2,
                                  betas=(200, 800, 3000, 10000))
    if c5_pol < best_c5:
        best_c5 = c5_pol
        best_h = h_pol
        print("  SLSQP improved: C5={:.10f}".format(best_c5))
    else:
        print("  SLSQP: C5={:.10f} (no improvement)".format(c5_pol))

    # Phase 3: Subgradient at parent resolution
    t3 = min(80.0, remaining_time() * 0.08)
    print("\n--- Phase 3: Subgradient at n={} ({}s) ---".format(best_n, t3))
    h_sg, c5_sg = subgrad_polish(best_h, best_n, time_limit=t3, step0=0.08, top_k=10)
    if c5_sg < best_c5:
        best_c5 = c5_sg
        best_h = h_sg
        print("  Subgrad improved: C5={:.10f}".format(best_c5))
    else:
        print("  Subgrad: C5={:.10f}".format(c5_sg))

    # Phase 4: Move to higher resolution n=512
    t4_total = min(450.0, remaining_time() * 0.35)
    if t4_total > 60.0:
        n_h = 512
        print("\n--- Phase 4: n={} ({}s) ---".format(n_h, t4_total))
        h_h = project_feasible(interp_to(best_h, best_n, n_h), n_h)
        c5_h = compute_c5(h_h)
        print("  Interpolated: C5={:.10f}".format(c5_h))

        # 4a: Binary SA at high res
        t4a = min(t4_total * 0.5, 250.0)
        h_h, c5_h, iters_h = sa_binary(h_h, n_h, time_limit=t4a, t0=0.015, t_final=0.0002)
        print("  SA n={}: {} iters, C5={:.10f}".format(n_h, iters_h, c5_h))

        # 4b: SLSQP at high res
        t4b = min(t4_total * 0.35, 120.0)
        h_h, c5_h = slsqp_polish(h_h, n_h, time_limit=t4b,
                                  betas=(100, 500, 2000, 8000))
        print("  SLSQP n={}: C5={:.10f}".format(n_h, c5_h))

        # 4c: Subgradient at high res
        t4c = min(t4_total * 0.15, 50.0)
        h_h, c5_h = subgrad_polish(h_h, n_h, time_limit=t4c, step0=0.06, top_k=12)
        print("  Subgrad n={}: C5={:.10f}".format(n_h, c5_h))

        if c5_h < best_c5:
            best_c5 = c5_h
            best_h = h_h.copy()
            best_n = n_h
            print("  New best from n={}: C5={:.10f}".format(n_h, best_c5))
        else:
            print("  n={} did not beat best ({:.10f})".format(n_h, best_c5))

    # Phase 5: Try n=384 as intermediate
    t5 = min(120.0, remaining_time() * 0.10)
    if t5 > 30.0 and best_n < 512:
        n_m = 384
        print("\n--- Phase 5: n={} ({}s) ---".format(n_m, t5))
        h_m = project_feasible(interp_to(best_h, best_n, n_m), n_m)
        c5_m = compute_c5(h_m)
        h_m, c5_m, _ = sa_binary(h_m, n_m, time_limit=t5 * 0.5, t0=0.015, t_final=0.0003)
        h_m, c5_m = slsqp_polish(h_m, n_m, time_limit=t5 * 0.4, betas=(200, 1000, 5000))
        if c5_m < best_c5:
            best_c5 = c5_m
            best_h = h_m.copy()
            best_n = n_m
            print("  New best from n={}: C5={:.10f}".format(n_m, best_c5))
        else:
            print("  n={}: C5={:.10f}".format(n_m, c5_m))

    # Phase 6: More SA at best resolution (reheat)
    t6 = min(150.0, remaining_time() * 0.15)
    if t6 > 20.0:
        print("\n--- Phase 6: Reheat SA at n={} ({}s) ---".format(best_n, t6))
        # Try binary SA again with different params
        h_r, c5_r, iters_r = sa_binary(best_h, best_n, time_limit=t6 * 0.6,
                                        t0=0.01, t_final=0.0001)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r
        # Continuous SA
        h_rc, c5_rc, _ = sa_continuous(best_h, best_n, time_limit=t6 * 0.3,
                                        t0=0.008, t_final=0.0005)
        if c5_rc < best_c5:
            best_c5 = c5_rc
            best_h = h_rc
        # Final SLSQP
        h_f, c5_f = slsqp_polish(best_h, best_n, time_limit=min(30.0, t6 * 0.1),
                                  betas=(5000, 20000))
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
        print("  Phase 6 best: C5={:.10f}".format(best_c5))

    # Phase 7: Coordinate descent
    t7 = min(40.0, remaining_time() * 0.04)
    if t7 > 10.0:
        print("\n--- Phase 7: Coordinate descent ({}s) ---".format(t7))
        best_h, best_c5 = coordinate_refine(best_h, best_n, time_limit=t7,
                                             rng_local=np.random.RandomState(seed + 7))
        print("  C5={:.10f}".format(best_c5))

    # Phase 8: Final SLSQP with very high beta
    t8 = min(40.0, remaining_time() * 0.04)
    if t8 > 8.0:
        print("\n--- Phase 8: Final SLSQP ({}s) ---".format(t8))
        best_h, best_c5 = slsqp_polish(best_h, best_n, time_limit=t8,
                                        betas=(10000, 30000, 100000))
        print("  C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
