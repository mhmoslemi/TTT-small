def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    # Final exact adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

    if sn == dn:
        return src.copy()

def corr_gradient(h, k, n):
    """Analytic gradient of corr[k] w.r.t. h for np.correlate(h, 1-h, mode='full')."""
    # corr[k] = sum_i h[i+k-(n-1)] * (1-h[i]) for valid i
    # Let shift = k - (n-1)
    # corr[k] = sum_{i: 0<=i<n, 0<=i+shift<n} h[i+shift] * (1-h[i])
    # d/dh[m] corr[k] = (1-h[m-shift]) * [valid] - h[m+shift] * [valid]
    shift = k - (n - 1)
    grad = np.zeros(n)
    # Term 1: (1-h[m-shift]) where m-shift is in [0,n-1] and m is in [0,n-1]
    # i.e., m in [0,n-1] and m-shift in [0,n-1] => m in [max(0,shift), min(n-1, n-1+shift)]
    lo1 = max(0, shift)
    hi1 = min(n, n + shift)  # exclusive
    if lo1 < hi1:
        grad[lo1:hi1] += (1.0 - h[lo1 - shift:hi1 - shift])
    # Term 2: -h[m+shift] where m+shift is in [0,n-1] and m is in [0,n-1]
    # i.e., m in [0,n-1] and m+shift in [0,n-1] => m in [max(0,-shift), min(n-1, n-1-shift)]
    lo2 = max(0, -shift)
    hi2 = min(n, n - shift)  # exclusive
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2 + shift:hi2 + shift]
    return grad

def smooth_max_grad(h, n, top_m, beta):
    """Compute gradient of smooth max (softmax) over top_m lags."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top_m values
    total_lags = 2 * n - 1
    if top_m >= total_lags:
        top_indices = np.arange(total_lags)
    else:
        # Find top m
        top_indices = np.argsort(corr)[-top_m:]
    
    c_top = corr[top_indices]
    c_max = c_top.max()
    # Softmax weights
    exps = np.exp(beta * (c_top - c_max))
    w = exps / exps.sum()
    
    # Gradient is weighted sum of individual gradients
    grad = np.zeros(n)
    for idx in range(len(top_indices)):
        grad += w[idx] * corr_gradient(h, top_indices[idx], n)
    
    smooth_val = (c_max + np.log(exps.sum()) / beta) * (2.0 / n)
    return smooth_val, grad

        # Nesterov momentum
        momentum = (t - 1) / (t + 2)
        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        # Gradient at y
        _, grad = smooth_max_grad(y, n, top_m, beta)
        
        # Line search (backtracking)
        lr = base_lr

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def corr_grad(h, idx, n):
    """Analytical gradient of np.correlate(h, 1-h, 'full')[idx] w.r.t. h."""
    k = idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:n] += (1.0 - h[0:n - k])
        grad[0:n - k] -= h[k:n]
    else:
        kk = -k
        grad[0:n - kk] += (1.0 - h[kk:n])
        grad[kk:n] -= h[0:n - kk]
    return grad

def smooth_max_and_grad(h, n, top_m, beta):
    """Smooth max (softmax) over top_m lags and its gradient."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    total = 2 * n - 1
    if top_m >= total:
        top_idx = np.arange(total)
    else:
        top_idx = np.argsort(corr)[-top_m:]
    c_top = corr[top_idx]
    c_max = c_top.max()
    exps = np.exp(beta * (c_top - c_max))
    w = exps / exps.sum()
    smooth_val = (c_max + np.log(exps.sum()) / beta) * (2.0 / n)
    grad = np.zeros(n)
    for i in range(len(top_idx)):

    grad *= (2.0 / n)

    """SLSQP with analytic gradient of smooth max over top_m lags."""
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        top_idx = np.argsort(corr)[-top_m:]
        c_top = corr[top_idx]
        cmax = c_top.max()
        exps = np.exp(beta * (c_top - cmax))
        return float((cmax + np.log(exps.sum()) / beta) * (2.0 / n))

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        top_idx = np.argsort(corr)[-top_m:]
        c_top = corr[top_idx]
        cmax = c_top.max()
        exps = np.exp(beta * (c_top - cmax))
        w = exps / exps.sum()
        g = np.zeros(n)
        for i in range(len(top_idx)):
            g += w[i] * corr_grad(x, top_idx[i], n)
        g *= (2.0 / n)
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def coord_descent_fast(h, n, max_sweeps, rng, tlimit_fn):
    """Fast coordinate descent with binary moves and small perturbations."""
    c5 = compute_c5(h)
    best_h = h.copy()
    for sweep in range(max_sweeps):
        if tlimit_fn() < 2.0:
            break
        improved = False
        for i in range(n):
            if tlimit_fn() < 1.0:
                break
            v0 = best_h[i]
            for v in [0.0, 1.0]:
                if abs(v - v0) < 1e-10:
                    continue
                trial = best_h.copy()
                trial[i] = v
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 1e-11:
                    c5 = c
                    best_h = trial.copy()
                    improved = True
                    break
            if improved and best_h[i] != v0:
                continue
            for d in [0.1, -0.1, 0.05, -0.05, 0.2, -0.2]:
                trial = best_h.copy()
                trial[i] = np.clip(trial[i] + d, 0.0, 1.0)
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 1e-11:
                    c5 = c
                    best_h = trial.copy()
                    improved = True
                    break
        if not improved and sweep >= 3:
            break
    return best_h, c5

def fixed_coords_optimize(h, n, freeze_thresh, tlimit_fn):
    """Fix coordinates near 0 or 1, optimize the rest."""
    free_mask = (h > freeze_thresh) & (h < 1.0 - freeze_thresh)
    n_free = int(np.sum(free_mask))
    if n_free < 3 or n_free > n - 3:
        return h, compute_c5(h)
    indices = np.where(free_mask)[0]
    h_fixed = h.copy()
    fixed_sum = np.sum(h[~free_mask])
    target_free = n / 2.0 - fixed_sum
    h_free = h[free_mask].copy()
    
    def obj_free(xf):
        hx = h_fixed.copy()
        hx[free_mask] = xf
        corr = np.correlate(hx, 1.0 - hx, mode='full')
        return float(np.max(corr) * (2.0 / n))
    
    h_new = h_fixed.copy()
    try:
        res = minimize(obj_free, h_free, method='Nelder-Mead',
                       options={'maxiter': n_free * 50, 'xatol': 1e-12, 'fatol': 1e-14})
        h_new[free_mask] = res.x
        h_new = project_feasible(h_new, n)
    except Exception:
        pass
    c5_new = compute_c5(h_new)
    return h_new, c5_new

def make_binary_patterns(n, rng):
    """Generate various structured binary initializations."""
    patterns = []
    # Thue-Morse
    tm = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        tm[i] = 1.0 if c else 0.0
    sums = np.sum(tm)
    if abs(sums - n/2) > 0.5:
        idx = rng.permutation(n)
        if sums > n/2:
            tm[idx[:int(sums - n/2)]] = 0.0
        else:
            tm[idx[:int(n/2 - sums)]] = 1.0
    patterns.append(project_feasible(tm, n))
    
    # Alternating blocks
    for block in [4, 8, 16, 32]:
        h = np.zeros(n)
        pos = 0
        flip = 0
        while pos < n:
            if flip % 2 == 0:
                h[pos:pos+block] = 1.0
            pos += block
            flip += 1
        h = project_feasible(h, n)
        patterns.append(h)
    
    # Sine-modulated
    t = np.arange(n) / n
    for f in [2, 3, 4, 5]:
        h = 0.5 + 0.4 * np.sin(2 * np.pi * f * t)
        h = project_feasible(h, n)
        patterns.append(h)
    
    # Random binary
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n//2]] = 1.0
    patterns.append(project_feasible(h, n))
    
    return patterns

def run(seed=42, budget_s=1300, **kwargs):
    _start = time.time()
    _budget = float(budget_s)
    
    def tlimit():
        return max(0.0, _budget - (time.time() - _start) - 5.0)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        best_h = project_feasible(make_binary_patterns(parent_n, rng)[0], parent_n)
        best_c5 = compute_c5(best_h)
        best_n = parent_n
        print("No parent. n={}, C5={:.10f}".format(parent_n, best_c5))

    # Phase 1: Accelerated gradient at parent resolution
    if tlimit() > 80:
        print("\n=== Phase 1: Accel grad n={} ===".format(best_n))
        n = best_n
        h_cur = best_h.copy()
        for round_i in range(4):
            if tlimit() < 20:
                break
            h_new, c5_new = accel_grad(h_cur, n, max_iter=5000, top_m=5,
                                        beta0=200.0, tlimit_fn=tlimit)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_cur = h_new.copy()
                print("  R{}: C5={:.10f}".format(round_i, best_c5))
            else:
                break

    # Phase 2: SLSQP polish with increasing beta
    if tlimit() > 40:
        print("\n=== Phase 2: SLSQP n={} ===".format(best_n))
        n = best_n
        h_cur = best_h.copy()
        for bv in [100, 300, 800, 2000, 5000]:
            if tlimit() < 8:
                break
            h_new = slsqp_analytic(h_cur, n, bv, maxiter=400, top_m=8)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-13:
                best_c5 = c_new
                best_h = h_new
                h_cur = h_new
                print("  b={}: C5={:.10f}".format(bv, best_c5))

    # Phase 3: Multi-resolution search
    if tlimit() > 120:
        print("\n=== Phase 3: Multi-res ===")
        n_list = [256, 320, 384, 448, 512]
        n_list = [x for x in n_list if x != best_n and 128 <= x <= 512]
        for n2 in n_list:
            if tlimit() < 40:
                break
            # Interpolate from current best
            h2 = project_feasible(interp_h(best_h, len(best_h), n2), n2)
            # Also try structured init
            pats = make_binary_patterns(n2, rng)
            h2 = min([h2] + pats[:3], key=lambda x: compute_c5(x))
            c5_2 = compute_c5(h2)
            print("  n={} init: C5={:.10f}".format(n2, c5_2))
            
            if tlimit() > 25:
                h2, c5_2 = accel_grad(h2, n2, max_iter=4000, top_m=5,
                                       beta0=200.0, tlimit_fn=tlimit)
                print("  n={} grad: C5={:.10f}".format(n2, c5_2))
            
            if tlimit() > 12:
                h2 = slsqp_analytic(h2, n2, 500, maxiter=300, top_m=8)
                c5_2 = compute_c5(h2)
                print("  n={} slsqp: C5={:.10f}".format(n2, c5_2))
            
            if c5_2 < best_c5 - 1e-10:
                best_c5 = c5_2
                best_h = h2
                best_n = n2
                print("  n={} IMPROVED: C5={:.10f}".format(n2, best_c5))

    # Phase 4: Active-set equalization
    if tlimit() > 50:
        print("\n=== Phase 4: Active-set n={} ===".format(best_n))
        n = best_n
        h_cur = best_h.copy()
        c5_cur = best_c5
        for rnd in range(30):
            if tlimit() < 5:
                break
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            top_k = np.argsort(corr)[-6:]
            spread = corr[top_k].max() - corr[top_k].min()
            
            # Focus optimization on top lags
            h_new = slsqp_analytic(h_cur, n, 3000, maxiter=250, top_m=6)
            c_new = compute_c5(h_new)
            if c_new < c5_cur - 1e-13:
                c5_cur = c_new
                h_cur = h_new
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_cur.copy()
                    print("  R{}: C5={:.10f} spr={:.6f}".format(rnd, best_c5, spread))
            else:
                if rnd > 8:
                    break

    # Phase 5: Random restarts + re-optimize
    if tlimit() > 60:
        print("\n=== Phase 5: Restarts n={} ===".format(best_n))
        n = best_n
        for att in range(6):
            if tlimit() < 12:
                break
            h_pert = best_h.copy()
            n_pert = rng.randint(5, 20)
            idxs = rng.choice(n, size=n_pert, replace=False)
            for idx in idxs:
                h_pert[idx] = np.clip(h_pert[idx] + rng.randn() * 0.4, 0.0, 1.0)
            h_pert = project_feasible(h_pert, n)
            c5_p = compute_c5(h_pert)
            
            if c5_p < best_c5 * 1.02:
                h_cur = h_pert.copy()
                h_cur, c5_cur = accel_grad(h_cur, n, max_iter=2000, top_m=5,
                                            beta0=300.0, tlimit_fn=tlimit)
                if tlimit() > 8:
                    h_cur = slsqp_analytic(h_cur, n, 1000, maxiter=200, top_m=8)
                    c5_cur = compute_c5(h_cur)
                if c5_cur < best_c5 - 1e-10:
                    best_c5 = c5_cur
                    best_h = h_cur
                    print("  Att{}: C5={:.10f}".format(att, best_c5))

    # Phase 6: Coordinate descent
    if tlimit() > 20:
        print("\n=== Phase 6: Coord descent n={} ===".format(best_n))
        n = best_n
        best_h, best_c5 = coord_descent_fast(best_h, n, max_sweeps=60, rng=rng, tlimit_fn=tlimit)
        print("  C5={:.10f}".format(best_c5))

    # Phase 7: Final ultra-fine SLSQP
    if tlimit() > 15:
        print("\n=== Phase 7: Final polish n={} ===".format(best_n))
        n = best_n
        for bv in [8000, 15000, 25000]:
            if tlimit() < 5:
                break
            h_new = slsqp_analytic(best_h, n, bv, maxiter=500, top_m=10)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new
                print("  b={}: C5={:.10f}".format(bv, best_c5))

    # Phase 8: One more accelerated round
    if tlimit() > 10:
        n = best_n
        h_new, c5_new = accel_grad(best_h, n, max_iter=2000, top_m=4,
                                    beta0=500.0, tlimit_fn=tlimit)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
