def slsqp_run(h0, n, beta, maxiter, focus_lag=-1):
    """SLSQP with log-sum-exp objective"""
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = corr - m
        e = np.exp(np.clip(beta * s, -400, 400))
        val = (m + np.log(np.sum(e)) / beta) * dx
        if focus_lag >= 0:
            val += 0.01 * corr[focus_lag] * dx
        return float(val)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h


def lbfgsb_run(h0, n, beta, maxiter, pen=100.0):
    """L-BFGS-B with penalty for sum constraint"""
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = corr - m
        e = np.exp(np.clip(beta * s, -400, 400))
        val = (m + np.log(np.sum(e)) / beta) * dx
        val += pen * (np.sum(x) - target) ** 2
        return float(val)
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'iprint': 0, 'maxls': 50})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def compute_c5(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = max(int(np.sum(mask)), 1)
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def make_tm(n):

        v = i
        while v:

    return h

def slsqp_run(h0, n, beta, maxiter, focus_lag=None):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = corr - m
        e = np.exp(np.clip(beta * s, -400, 400))
        val = (m + np.log(np.sum(e)) / beta) * dx
        if focus_lag is not None:
            val += 0.02 * corr[focus_lag] * dx
        return float(val)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

            for lr in [0.15, 0.08, 0.04]:
                if tlimit() < 15:
                    break
                h_new, c_new = multigrad_descent(h_cur, best_n, lr, 3000, top_k, rng, tlimit)
                if c_new < compute_c5(h_cur) - 1e-13:
                    h_cur = h_new
                if compute_c5(h_cur) < best_c5 - 1e-13:
                    best_c5 = compute_c5(h_cur)
                    best_h = h_cur.copy()
        print("  P1 done: C5={:.8f}".format(best_c5))

    # Phase 2: Single-grad descent (finer)
    if tlimit() > 150:
        print("\n=== P2: single-grad n={} ===".format(best_n))
        h_cur = best_h.copy()
        for lr in [0.05, 0.02, 0.01, 0.005]:
            if tlimit() < 10:
                break
            h_new, c_new = single_grad_descent(h_cur, best_n, lr, 4000, rng, tlimit)
            if c_new < compute_c5(h_cur) - 1e-13:
                h_cur = h_new
            if compute_c5(h_cur) < best_c5 - 1e-13:
                best_c5 = compute_c5(h_cur)
                best_h = h_cur.copy()
        print("  P2 done: C5={:.8f}".format(best_c5))

    # Phase 3: SLSQP multi-beta
    if tlimit() > 150:
        print("\n=== P3: SLSQP n={} ===".format(best_n))
        h_cur = best_h.copy()
        for bv in [50, 100, 200, 500, 1000, 2000, 5000]:
            if tlimit() < 12:
                break
            h_new = slsqp_run(h_cur, best_n, bv, maxiter=500)
            c_new = compute_c5(h_new)
            if c_new < compute_c5(h_cur) - 1e-14:
                h_cur = h_new
            if compute_c5(h_cur) < best_c5 - 1e-14:
                best_c5 = compute_c5(h_cur)
                best_h = h_cur.copy()
        print("  P3 done: C5={:.8f}".format(best_c5))

    # Phase 4: Coordinate descent
    if tlimit() > 80:
        print("\n=== P4: coord descent n={} ===".format(best_n))
        best_h, best_c5 = coord_descent(best_h, best_n, max_sweeps=40, rng=rng, tlimit=tlimit)
        print("  P4 done: C5={:.8f}".format(best_c5))

    # Phase 5: Try n=512
    if tlimit() > 250:
        print("\n=== P5: n=512 ===")
        n512 = 512
        h512 = project_feasible(interp_h(best_h, best_n, n512), n512)
        c5512 = compute_c5(h512)
        print("  n=512 init: C5={:.8f}".format(c5512))
        h_cur = h512.copy()
        for top_k in [3, 8, 16]:
            if tlimit() < 30:
                break
            for lr in [0.1, 0.05]:
                if tlimit() < 20:
                    break
                h_new, c_new = multigrad_descent(h_cur, n512, lr, 2000, top_k, rng, tlimit)
                if c_new < compute_c5(h_cur) - 1e-13:
                    h_cur = h_new
        if tlimit() > 30:
            for lr in [0.03, 0.01]:
                if tlimit() < 15:
                    break
                h_new, c_new = single_grad_descent(h_cur, n512, lr, 3000, rng, tlimit)
                if c_new < compute_c5(h_cur) - 1e-13:
                    h_cur = h_new
        if tlimit() > 20:
            for bv in [200, 1000]:
                if tlimit() < 12:
                    break
                h_new = slsqp_run(h_cur, n512, bv, maxiter=300)
                if compute_c5(h_new) < compute_c5(h_cur) - 1e-14:
                    h_cur = h_new
        c5512 = compute_c5(h_cur)
        print("  n=512 result: C5={:.8f}".format(c5512))
        if c5512 < best_c5 - 1e-10:
            best_c5 = c5512
            best_h = h_cur.copy()
            best_n = n512
            print("  Improved to n=512!")

    # Phase 6: Focused iteration on active lags
    if tlimit() > 60:
        print("\n=== P6: focused n={} ===".format(best_n))
        h_cur = best_h.copy()
        for rnd in range(20):
            if tlimit() < 5:
                break
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            k_max = int(np.argmax(corr))
            h_new = slsqp_run(h_cur, best_n, 2000, maxiter=300, focus_lag=k_max)
            c_new = compute_c5(h_new)
            if c_new < compute_c5(h_cur) - 1e-14:
                h_cur = h_new
            else:
                break
        if compute_c5(h_cur) < best_c5 - 1e-14:
            best_c5 = compute_c5(h_cur)
            best_h = h_cur.copy()
        print("  P6 done: C5={:.8f}".format(best_c5))

    # Phase 7: More coordinate descent
    if tlimit() > 50:
        print("\n=== P7: coord2 n={} ===".format(best_n))
        best_h, best_c5 = coord_descent(best_h, best_n, max_sweeps=30, rng=rng, tlimit=tlimit)
        print("  P7 done: C5={:.8f}".format(best_c5))

    # Phase 8: Perturbation + re-optimize
    if tlimit() > 60:
        print("\n=== P8: perturb n={} ===".format(best_n))
        for attempt in range(8):
            if tlimit() < 12:
                break
            h_pert = best_h.copy()
            n_pert = rng.randint(2, 7)
            idxs = rng.choice(best_n, size=n_pert, replace=False)
            for idx in idxs:
                h_pert[idx] = np.clip(h_pert[idx] + rng.randn() * 0.15, 0.0, 1.0)
            h_pert = project_feasible(h_pert, best_n)
            c5_p = compute_c5(h_pert)
            if c5_p < best_c5 * 1.02:
                h_cur = h_pert.copy()
                h_new, c_new = multigrad_descent(h_cur, best_n, 0.05, 2000, 5, rng, tlimit)
                if c_new < compute_c5(h_cur) - 1e-14:
                    h_cur = h_new
                if tlimit() > 8:
                    h_new = slsqp_run(h_cur, best_n, 1000, maxiter=200)
                    if compute_c5(h_new) < compute_c5(h_cur) - 1e-14:
                        h_cur = h_new
                if compute_c5(h_cur) < best_c5 - 1e-9:
                    best_c5 = compute_c5(h_cur)
                    best_h = h_cur.copy()
                    print("  Perturb {} improved: C5={:.8f}".format(attempt, best_c5))

    # Phase 9: Final SLSQP polish with high beta
    if tlimit() > 20:
        print("\n=== P9: final polish n={} ===".format(best_n))
        for bv in [3000, 8000, 15000]:
            if tlimit() < 6:
                break
            h_new = slsqp_run(best_h, best_n, bv, maxiter=600)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new
        print("  P9 done: C5={:.8f}".format(best_c5))

    # Phase 10: Final ultra-fine coord descent
    if tlimit() > 10:
        best_h, best_c5 = coord_descent(best_h, best_n, max_sweeps=20, rng=rng, tlimit=tlimit)
        print("  Final coord: C5={:.8f}".format(best_c5))

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
