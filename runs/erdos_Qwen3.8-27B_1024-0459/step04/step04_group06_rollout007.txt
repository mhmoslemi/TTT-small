import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_BETA = 1.0
_G_TGT = 0.0
_G_START = 0.0
_G_BUDGET = 1300.0
_G_DX = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-15)[0]
        else:
            idx = np.where(h > 1e-15)[0]
        if len(idx) > 0:
            h[idx[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)

def lse_obj(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(_G_BETA * shifted, -700, 700))
    s = np.sum(exps)
    if s < 1e-300:
        s = 1e-300
    return (m + np.log(s) / _G_BETA) * dx

def lse_grad(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(_G_BETA * shifted, -700, 700))
    s = np.sum(exps)
    if s < 1e-300:
        s = 1e-300
    softmax = exps / s
    # Only process significant entries
    thresh = 1e-15
    mask = softmax > thresh
    if not np.any(mask):
        mask = softmax >= np.max(softmax)
    top_idx = np.where(mask)[0]
    
    grad = np.zeros(n)
    for k_idx in top_idx:
        w = softmax[k_idx]
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            nv = n - offset
            if nv <= 0:
                continue
            grad[offset:] += w * (1.0 - x[:nv])
            grad[:nv] -= w * x[offset:offset + nv]
        else:
            m2 = -offset
            nv = n - m2
            if nv <= 0:
                continue
            grad[:nv] += w * (1.0 - x[m2:m2 + nv])
            grad[m2:] -= w * x[:nv]
    
    return grad * dx

def con_sum(x):
    return np.sum(x) - _G_TGT

def slsqp_optimize(h, n, beta, maxiter=300, ftol=1e-14):
    _G_N = n
    _G_TGT = float(n) / 2.0
    _G_DX = 2.0 / n
    _G_BETA = float(beta)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum}]
    try:
        res = minimize(lse_obj, h, method='SLSQP',
                       jac=lse_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception as e:
        return h

def subgrad_step(h, n, top_k=5):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_indices = np.argsort(corr)[-top_k:]
    top_indices = top_indices[np.argsort(corr[top_indices])[::-1]]
    grad = np.zeros(n)
    for k_idx in top_indices:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            nv = n - offset
            if nv <= 0:
                continue
            grad[offset:] += (1.0 - h[:nv])
            grad[:nv] -= h[offset:offset + nv]
        else:
            m2 = -offset
            nv = n - m2
            if nv <= 0:
                continue
            grad[:nv] += (1.0 - h[m2:m2 + nv])
            grad[m2:] -= h[:nv]
    return grad, corr

def subgrad_optimize(h, n, max_iter=200, step0=0.1, top_k=5):
    c5_best = compute_c5(h)
    h_best = h.copy()
    v = np.zeros(n)
    step = step0
    patience = 0
    for it in range(max_iter):
        if remaining_time() < 5.0:
            break
        grad, corr = subgrad_step(h, n, top_k=top_k)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad = grad / gn
        alpha = step
        improved = False
        for _ in range(15):
            v = 0.9 * v - alpha * grad
            h_trial = h + v
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-13:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
            alpha *= 0.5
