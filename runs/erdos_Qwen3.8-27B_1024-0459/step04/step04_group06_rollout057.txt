    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    for it in range(n_iters):
        if remaining_time() < 3.0:
            break
        T = t_start * (t_end / t_start) ** (float(it) / float(max(1, n_iters - 1)))
        i_idx = rng.randint(len(ones))
        j_idx = rng.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        h[i] = 0.0
        h[j] = 1.0
        new_mc = max_corr_val(h)
        delta = new_mc - cur_mc
        if delta < -1e-12 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_mc = new_mc
            if new_mc < best_mc:
                best_mc = new_mc
                best_h = h.copy()
            ones[i_idx] = j
            zeros[j_idx] = i
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_mc


def coordinate_fine_tune(h, n, rng, deltas, max_sweeps=8):
    h_work = h.copy()
    best_mc = max_corr_val(h_work)
    best_h = h_work.copy()
    for sweep in range(max_sweeps):
        if remaining_time() < 3.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for idx in order:
            if remaining_time() < 1.0:
                break
            for d in deltas:
                h_trial = h_work.copy()
                h_trial[idx] = np.clip(h_trial[idx] + d, 0.0, 1.0)
                h_trial = project_feasible(h_trial, n)
                mc = max_corr_val(h_trial)
                if mc < best_mc - 1e-13:
                    best_mc = mc
                    h_work = h_trial
                    best_h = h_trial.copy()
                    any_imp = True
                    break
        if not any_imp and sweep > 1:
            break
    return best_h, best_mc


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 256
        best_h = project_feasible(make_thue_morse(256), 256)
        best_c5 = compute_c5(best_h)
        print("Thue-Morse: n={}, C5={:.10f}".format(best_n, best_c5))

    # Phase 1: PCD at parent resolution
    n = best_n
    print("\n--- Phase 1: PCD at n={} ---".format(n))
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    best_h, mc_best = paired_coord_descent(best_h, n, rng, max_sweeps=200, pairs_per_sweep=80, ternary_iters=35)
    best_c5 = mc_best * (2.0 / n)
    print("  After PCD: C5={:.10f}".format(best_c5))

    # Phase 2: SLSQP polish at parent resolution
    print("\n--- Phase 2: SLSQP polish n={} ---".format(n))
    best_h, best_c5 = slsqp_polish(best_h, n, betas=[200, 500, 1000, 2000], maxiter=250)
    print("  After SLSQP: C5={:.10f}".format(best_c5))

    # Phase 3: PCD again (refined)
    if remaining_time() > 60.0:
        print("\n--- Phase 3: PCD round 2 ---".format())
        best_h, mc_best = paired_coord_descent(best_h, n, rng, max_sweeps=150, pairs_per_sweep=100, ternary_iters=40)
        c5_val = mc_best * (2.0 / n)
        if c5_val < best_c5:
            best_c5 = c5_val
        print("  C5={:.10f}".format(best_c5))

    # Phase 4: Try n=384
    if remaining_time() > 120.0:
        print("\n--- Phase 4: n=384 ---".format())
        n_try = 384
        h_try = project_feasible(interp_to(best_h, n, n_try), n_try)
        _G_N = n_try
        _G_TGT = n_try / 2.0
        _G_DX = 2.0 / n_try
        h_try, mc_try = paired_coord_descent(h_try, n_try, rng, max_sweeps=150, pairs_per_sweep=70, ternary_iters=35)
        c5_try = mc_try * (2.0 / n_try)
        h_try, c5_try2 = slsqp_polish(h_try, n_try, betas=[300, 800, 2000], maxiter=200)
        c5_try = min(c5_try, c5_try2)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try
            print("  n=384: C5={:.10f} (improved)".format(c5_try))
        else:
            print("  n=384: C5={:.10f}".format(c5_try))

    # Phase 5: Try n=512
    if remaining_time() > 100.0:
        print("\n--- Phase 5: n=512 ---".format())
        n_try = 512
        h_try = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        _G_N = n_try
        _G_TGT = n_try / 2.0
        _G_DX = 2.0 / n_try
        h_try, mc_try = paired_coord_descent(h_try, n_try, rng, max_sweeps=120, pairs_per_sweep=60, ternary_iters=30)
        c5_try = mc_try * (2.0 / n_try)
        h_try, c5_try2 = slsqp_polish(h_try, n_try, betas=[200, 600, 1500], maxiter=200)
        c5_try = min(c5_try, c5_try2)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try
            print("  n=512: C5={:.10f} (improved)".format(c5_try))
        else:
            print("  n=512: C5={:.10f}".format(c5_try))

    # Phase 6: Binary SA at best_n (or n=256)
    if remaining_time() > 80.0:
        print("\n--- Phase 6: Binary SA ---".format())
        n_sa = best_n
        h_bin = np.zeros(n_sa)
        h_sorted = np.sort(best_h)[::-1]
        h_bin[:n_sa // 2] = 1.0
        # Also try Thue-Morse
        h_tm = make_thue_morse(n_sa)
        for h_init in [h_bin, h_tm]:
            if remaining_time() < 20.0:
                break
            h_bin_best, mc_bin = binary_sa(h_init, n_sa, rng, n_iters=20000, t_start=0.8, t_end=0.01)
            c5_bin = mc_bin * (2.0 / n_sa)
            if c5_bin < best_c5 - 1e-12:
                best_c5 = c5_bin
                best_h = h_bin_best.copy()
                best_n = n_sa
                print("  Binary SA: C5={:.10f} (improved)".format(c5_bin))
            else:
                print("  Binary SA: C5={:.10f}".format(c5_bin))

    # Phase 7: PCD with different pairing strategy (adjacent pairs)
    if remaining_time() > 50.0:
        print("\n--- Phase 7: Adjacent-pair PCD ---".format())
        n = best_n
        h_work = best_h.copy()
        mc = max_corr_val(h_work)
        best_mc_phase7 = mc
        best_h7 = h_work.copy()
        for sweep in range(100):
            if remaining_time() < 3.0:
                break
            improved = False
            for j in range(0, n - 1, 2):
                if remaining_time() < 2.0:
                    break
                i = j + 1
                S = h_work[j] + h_work[i]
                lo = max(0.0, S - 1.0)
                hi = min(1.0, S)
                if hi - lo < 1e-12:
                    continue
                for _ in range(30):
                    m1 = lo + (hi - lo) * 0.381966
                    m2 = hi - (hi - lo) * 0.381966
                    h_work[j], h_work[i] = m1, S - m1
                    f1 = max_corr_val(h_work)
                    h_work[j], h_work[i] = m2, S - m2
                    f2 = max_corr_val(h_work)
                    if f1 < f2:
                        hi = m2
                    else:
                        lo = m1
                alpha = (lo + hi) * 0.5
                h_work[j] = alpha
                h_work[i] = S - alpha
                mc_new = max_corr_val(h_work)
                if mc_new < best_mc_phase7 - 1e-13:
                    best_mc_phase7 = mc_new
                    best_h7 = h_work.copy()
                    improved = True
            if not improved and sweep > 10:
                break
        c5_7 = best_mc_phase7 * (2.0 / n)
        if c5_7 < best_c5 - 1e-13:
            best_c5 = c5_7
            best_h = best_h7.copy()
        print("  C5={:.10f}".format(best_c5))

    # Phase 8: Final SLSQP with very high beta
    if remaining_time() > 20.0:
        print("\n--- Phase 8: Final SLSQP ---".format())
        n = best_n
        best_h, best_c5 = slsqp_polish(best_h, n, betas=[3000, 8000, 15000, 30000], maxiter=400)
        print("  C5={:.10f}".format(best_c5))

    # Phase 9: Fine coordinate tuning
    if remaining_time() > 10.0:
        print("\n--- Phase 9: Fine tuning ---".format())
        n = best_n
        _G_N = n
        _G_TGT = n / 2.0
        _G_DX = 2.0 / n
        best_h, mc_ft = coordinate_fine_tune(best_h, n, rng,
                                              deltas=[0.01, -0.01, 0.03, -0.03, 0.05, -0.05],
                                              max_sweeps=6)
        c5_ft = mc_ft * (2.0 / n)
        if c5_ft < best_c5:
            best_c5 = c5_ft
        print("  C5={:.10f}".format(best_c5))

    # Phase 10: One more PCD pass
    if remaining_time() > 20.0:
        print("\n--- Phase 10: Final PCD ---".format())
        n = best_n
        best_h, mc_final = paired_coord_descent(best_h, n, rng, max_sweeps=100, pairs_per_sweep=100, ternary_iters=35)
        c5_final = mc_final * (2.0 / n)
        if c5_final < best_c5:
            best_c5 = c5_final
        print("  C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < best_c5:
                best_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, best_c5))
    return (best_h, best_c5, best_n)
