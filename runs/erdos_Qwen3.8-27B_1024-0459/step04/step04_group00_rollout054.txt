def coordinate_descent(h, n, max_iters=1000):
    """Optimize pairs of coordinates."""
    h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(42)
    for it in range(max_iters):
        if remaining() < 3:
            break
        # Pick a random pair
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            j = (j + 1) % n
        # Optimize pair (h[i], h[j]) keeping sum fixed
        s = h[i] + h[j]
        # h[i] in [max(0, s-1), min(1, s)], h[j] = s - h[i]
        lo = max(0.0, s - 1.0)
        hi = min(1.0, s)
        if lo >= hi:
            continue
        # Golden section search on h[i]
        gr = (np.sqrt(5) - 1) / 2
        a, b = lo, hi
        c = b - gr * (b - a)
        d2 = a + gr * (b - a)
        h[i], h[j] = c, s - c
        fc = compute_c5(h)
        h[i], h[j] = d2, s - d2
        fd = compute_c5(h)
        for _ in range(30):
            if b - a < 1e-10:
                break
            if fc < fd:
                b = d2
                d2 = c
                fd = fc
                c = b - gr * (b - a)
                h[i], h[j] = c, s - c
                fc = compute_c5(h)
            else:
                a = c
                c = d2
                fc = fd
                d2 = a + gr * (b - a)
                h[i], h[j] = d2, s - d2
                fd = compute_c5(h)
        # Set to best
        h[i], h[j] = (c if fc < fd else d2), (s - (c if fc < fd else d2))
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
    return h, best_c5

def project_feasible(h, n):

def _sm_obj(h, beta, lam=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    val = (m + np.log(np.sum(ex)) / beta) * dx
    if lam > 0:
        s = np.sum(h) - n / 2.0
        val += lam * s * s
    return val


def _sm_grad(h, beta, lam=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Subgradient + L-BFGS cycles
    for cyc in range(8):
        if remaining() < 12:
            break
        h_sg, c5_sg = subgradient_equalize(h, n, num_top=6, step_size=0.003, max_iters=200)
        h_new = lbfgs_opt(h_sg, n, 15000, lam=100.0, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Extended local search
    if remaining() > 30:
        h_ls, c5_ls = local_pair_search(h, n, max_iters=8000, rng=np.random.RandomState(seed + 7))
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            h = h_ls.copy()

    # Phase 6: Snap-and-optimize
    if remaining() > 25:
        h_snap = h.copy()
        for i in range(n):
            if h_snap[i] < 0.15:
                h_snap[i] = 0.0
            elif h_snap[i] > 0.85:
                h_snap[i] = 1.0
        h_snap = project_feasible(h_snap, n)
        h_new = lbfgs_opt(h_snap, n, 10000, lam=200.0, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 7: Final micro-optimization
    if remaining() > 10:
        h_ls2, c5_ls2 = local_pair_search(h, n, max_iters=5000, rng=np.random.RandomState(seed + 13))
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()

    if remaining() > 8:
        h_new = lbfgs_opt(best_h, best_n, 100000, lam=1000.0, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
