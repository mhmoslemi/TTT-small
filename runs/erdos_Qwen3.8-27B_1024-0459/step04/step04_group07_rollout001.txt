            j_hi = min(n - 1, n - 1 - mm)
            if j_lo <= j_hi:
                grad[j_lo:j_hi + 1] += (1.0 / 3.0) * (1.0 - x[j_lo + mm:j_hi + mm + 1])
            j_lo2 = max(0, mm)
            j_hi2 = min(n - 1, n - 1 + mm)
            if j_lo2 <= j_hi2:
                grad[j_lo2:j_hi2 + 1] -= (1.0 / 3.0) * x[j_lo2 - mm:j_hi2 - mm + 1]
        grad *= dx
        s = np.sum(x) - n / 2.0
        obj_f = float(np.max(corr) * dx) + 100.0 * s * s
        grad += 200.0 * s
        return obj_f, grad
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_f_grad, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0.copy(), n)
    return h

def subgradient_steps(h, n, n_steps, step_init, rng):
    c5 = compute_c5(h)
    h_cur = h.copy()
    step = step_init
    for it in range(n_steps):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_max = int(np.argmax(corr))
        m = k_max - (n - 1)
        dx = 2.0 / n
        # Subgradient of overlap at k_max
        sg = np.zeros(n)
        j_lo = max(0, -m)
        j_hi = min(n - 1, n - 1 - m)
        if j_lo <= j_hi:
            sg[j_lo:j_hi + 1] += (1.0 - h_cur[j_lo + m:j_hi + m + 1])
        j_lo2 = max(0, m)
        j_hi2 = min(n - 1, n - 1 + m)
        if j_lo2 <= j_hi2:
            sg[j_lo2:j_hi2 + 1] -= h_cur[j_lo2 - m:j_hi2 - m + 1]
        sg *= dx
        # Also add a small gradient from log-sum-exp for spread
        _, sg2 = smooth_obj_and_grad(h_cur, 100.0)
        sg = 0.7 * sg + 0.3 * sg2
        # Step
        h_trial = h_cur - step * sg
        h_trial = project_feasible(h_trial, n)
        c_new = compute_c5(h_trial)
        if c_new < c5 - 1e-12:
            h_cur = h_trial
            c5 = c_new
            step *= 1.05
        else:
            step *= 0.7
            if step < 1e-5:
                break
    return h_cur, c5

def coord_descent(h, n, max_sweeps, rng, tlimit):
    dx = 2.0 / n
    c5 = compute_c5(h)
    h_best = h.copy()
    for sweep in range(max_sweeps):
        if tlimit() < 2.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if tlimit() < 1.0:
                break
            v0 = h_best[i]
            # Try binary values
            for v in [0.0, 1.0]:
                if abs(v - v0) < 1e-9:
                    continue
                trial = h_best.copy()
                trial[i] = v
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 1e-10:
                    c5 = c
                    h_best = trial.copy()
                    improved = True
            # Try small perturbations
            for d in [0.2, -0.2, 0.1, -0.1, 0.05, -0.05, 0.3, -0.3]:
                if tlimit() < 1.0:
                    break
                trial = h_best.copy()
                trial[i] = np.clip(trial[i] + d, 0.0, 1.0)
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 1e-10:
                    c5 = c
                    h_best = trial.copy()
                    improved = True
        if not improved and sweep >= 3:
            break
    return h_best, c5

def active_lag_focused(h, n, rng, tlimit, max_rounds=30):
    c5 = compute_c5(h)
    h_best = h.copy()
    for rnd in range(max_rounds):
        if tlimit() < 5.0:
            break
        h_new = slsqp_focused(h_best, n, None, maxiter=150)
        c_new = compute_c5(h_new)
        if c_new < c5 - 1e-12:
            c5 = c_new
            h_best = h_new
        else:
            break
    return h_best, c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    budget = float(budget_s)
    rng = np.random.RandomState(seed)

    def tlimit():
        return max(0.0, budget - (time.time() - start_time) - 5.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = project_feasible(thue_morse(parent_n), parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("No parent. TM: n={}, C5={:.8f}".format(parent_n, best_c5))

    # Phase 1: Interpolate to n=512 and optimize with L-BFGS-B
    n_target = 512
    if tlimit() > 200:
        print("\n=== Phase 1: Interpolate to n={} ===".format(n_target))
        h_512 = project_feasible(interp_h(best_h, best_n, n_target), n_target)
        c5_512 = compute_c5(h_512)
        print("  After interp: C5={:.8f}".format(c5_512))

        # Progressive L-BFGS-B optimization
        beta_schedule = [10, 30, 60, 100, 200, 400, 800, 1500, 3000, 5000]
        h_cur = h_512.copy()
        c5_cur = c5_512

        for bv in beta_schedule:
            if tlimit() < 20:
                break
            h_new = optimize_lbfgs(h_cur, n_target, bv, maxiter=300, pen_coeff=50.0)
            c_new = compute_c5(h_new)
            if c_new < c5_cur - 1e-12:
                c5_cur = c_new
                h_cur = h_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n_target
            if tlimit() < 60:
                break

        print("  After L-BFGS-B multi-beta: C5={:.8f}".format(best_c5))

        # Subgradient refinement
        if tlimit() > 30:
            h_cur, c5_cur = subgradient_steps(h_cur, n_target, 200, 0.02, rng)
            if c5_cur < best_c5 - 1e-10:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n_target
            print("  After subgradient: C5={:.8f}".format(best_c5))

        # Focused SLSQP
        if tlimit() > 30:
            h_cur, c5_cur = active_lag_focused(h_cur, n_target, rng, tlimit, max_rounds=20)
            if c5_cur < best_c5 - 1e-10:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n_target
            print("  After focused SLSQP: C5={:.8f}".format(best_c5))

    # Phase 2: Try n=1024
    if tlimit() > 120:
        print("\n=== Phase 2: n=1024 ===")
        n2 = 1024
        h_1024 = project_feasible(interp_h(best_h, best_n, n2), n2)
        c5_1024 = compute_c5(h_1024)
        print("  After interp: C5={:.8f}".format(c5_1024))

        h_cur = h_1024.copy()
        c5_cur = c5_1024
        beta_schedule2 = [20, 60, 150, 400, 1000, 2000]
        for bv in beta_schedule2:
            if tlimit() < 15:
                break
            h_new = optimize_lbfgs(h_cur, n2, bv, maxiter=200, pen_coeff=30.0)
            c_new = compute_c5(h_new)
            if c_new < c5_cur - 1e-12:
                c5_cur = c_new
                h_cur = h_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n2
            if tlimit() < 40:
                break

        print("  After n=1024 L-BFGS-B: C5={:.8f}".format(best_c5))

        if tlimit() > 20:
            h_cur, c5_cur = subgradient_steps(h_cur, n2, 150, 0.015, rng)
            if c5_cur < best_c5 - 1e-10:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n2
            print("  After n=1024 subgradient: C5={:.8f}".format(best_c5))

    # Phase 3: Coordinate descent at best resolution
    if tlimit() > 30:
        print("\n=== Phase 3: Coord descent n={} ===".format(best_n))
        best_h, best_c5 = coord_descent(best_h, best_n, max_sweeps=50, rng=rng, tlimit=tlimit)
        print("  After coord descent: C5={:.8f}".format(best_c5))

    # Phase 4: Multiple restarts with perturbation
    if tlimit() > 60:
        print("\n=== Phase 4: Perturbation restarts ===")
        n = best_n
        for attempt in range(10):
            if tlimit() < 15:
                break
            h_pert = best_h.copy()
            n_pert = rng.randint(5, 20)
            idxs = rng.choice(n, size=n_pert, replace=False)
            for idx in idxs:
                h_pert[idx] = np.clip(h_pert[idx] + rng.randn() * 0.25, 0.0, 1.0)
            h_pert = project_feasible(h_pert, n)
            c5_p = compute_c5(h_pert)
            if c5_p < best_c5 * 1.08:
                h_cur = h_pert.copy()
                c5_cur = c5_p
                for bv in [200, 1000, 3000]:
                    if tlimit() < 8:
                        break
                    h_new = optimize_lbfgs(h_cur, n, bv, maxiter=150, pen_coeff=50.0)
                    c_new = compute_c5(h_new)
                    if c_new < c5_cur - 1e-12:
                        c5_cur = c_new
                        h_cur = h_new
                if c5_cur < best_c5 - 1e-10:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n
                    print("  Restart {} improved: C5={:.8f}".format(attempt, best_c5))

    # Phase 5: High-beta L-BFGS-B polish at best resolution
    if tlimit() > 20:
        print("\n=== Phase 5: High-beta polish ===")
        n = best_n
        h_cur = best_h.copy()
        c5_cur = best_c5
        for bv in [5000, 10000, 20000]:
            if tlimit() < 8:
                break
            h_new = optimize_lbfgs(h_cur, n, bv, maxiter=400, pen_coeff=100.0)
            c_new = compute_c5(h_new)
            if c_new < c5_cur - 1e-13:
                c5_cur = c_new
                h_cur = h_new
            else:
                break
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        print("  After high-beta: C5={:.8f}".format(best_c5))

    # Phase 6: Focused SLSQP final
    if tlimit() > 15:
        print("\n=== Phase 6: Final focused ===")
        n = best_n
        best_h, best_c5 = active_lag_focused(best_h, n, rng, tlimit, max_rounds=15)
        print("  After final focused: C5={:.8f}".format(best_c5))

    # Phase 7: Final coord descent
    if tlimit() > 10:
        n = best_n
        best_h, best_c5 = coord_descent(best_h, n, max_sweeps=40, rng=rng, tlimit=tlimit)
        print("  After final coord: C5={:.8f}".format(best_c5))

    # Phase 8: Ultra-fine subgradient
    if tlimit() > 5:
        n = best_n
        best_h, best_c5 = subgradient_steps(best_h, n, 300, 0.005, rng)
        print("  After ultra subgradient: C5={:.8f}".format(best_c5))

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
