    w = np.exp(shifted) / exp_sum  # weights for each lag

    val = dx * (1.0/beta) * (np.log(exp_sum) + beta * max_c)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        for _ in range(10):
            mask = (h > 1e-10) & (h < 1.0 - 1e-10)
            if np.sum(mask) > 0:
                h[mask] += diff / np.sum(mask)
            else:
                idx = int(np.argmin(np.abs(h - 0.5)))
                h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
            h = np.clip(h, 0.0, 1.0)
            diff = target - np.sum(h)
            if abs(diff) < 1e-14:
                break
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def smooth_max_value_and_weights(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    exp_sum = np.sum(exp_shifted)
    w = exp_shifted / exp_sum
    val = dx * (1.0 / beta) * (np.log(exp_sum) + max_shifted + beta * max_c)
    return val, w, corr

def compute_smooth_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    val, w, corr = smooth_max_value_and_weights(h, beta)
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = lags[k]
        if lag >= 0:
            p_start1 = lag
            p_end1 = min(n, n + lag)
            if p_start1 < p_end1:
                grad[p_start1:p_end1] += wk * (1.0 - h[p_start1 - lag:p_end1 - lag])
            p_start2 = 0
            p_end2 = n - lag
            if p_start2 < p_end2:
                grad[p_start2:p_end2] -= wk * h[p_start2 + lag:p_end2 + lag]
        else:
            m = -lag
            p_start1 = 0
            p_end1 = min(n, n - m)
            if p_start1 < p_end1:
                grad[p_start1:p_end1] += wk * (1.0 - h[p_start1 + m:p_end1 + m])
            p_start2 = m
            p_end2 = n
            if p_start2 < p_end2:
                grad[p_start2:p_end2] -= wk * h[p_start2 - m:p_end2 - m]
    grad *= dx
    return val, grad

def slsqp_with_grad(h_init, n, beta, maxiter=300):
    n_local = n
    dx_local = 2.0 / n_local
    best_x = h_init.copy()

    def obj(x):
        v, _ = smooth_max_value_and_weights(x, beta)

        return v, g

def find_optimal_delta(c, d, delta_lo, delta_hi, n_iter=50):
    if delta_hi - delta_lo < 1e-15:
        return (delta_lo + delta_hi) / 2.0
    lo = delta_lo
    hi = delta_hi
    for _ in range(n_iter):
        mid = (lo + hi) / 2.0
        vals = c + d * mid
        max_val = np.max(vals)
        active = vals >= max_val - 1e-12
        d_active = d[active]
        if len(d_active) == 0:
            break
        if np.max(d_active) < -1e-14:
            hi = mid
        elif np.min(d_active) > 1e-14:
            lo = mid
        else:
            break
    return (lo + hi) / 2.0
