import numpy as np
from scipy.optimize import minimize
import time

_G_START = 0.0
_G_BUDGET = 1300.0
_G_N = 0
_G_BETA = 1.0
_G_TGT = 0.0
_G_DX = 1.0


def time_remaining():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 3.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def con_sum_fun(x):
    return np.sum(x) - _G_TGT


def con_sum_jac(x):
    return np.ones(_G_N)


def obj_lse(x):
    n = _G_N
    dx = _G_DX
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _G_BETA
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    return (m + np.log(np.sum(exps)) / beta) * dx


def obj_lse_grad(x):
    n = _G_N
    dx = _G_DX
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _G_BETA
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    se = np.sum(exps)
    weights = exps / se
    grad = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        if s >= 0:
            np = n - s
            if np <= 0:
                continue
            grad[s:] += (1.0 - x[:np])
            grad[:np] -= x[s:s + np]
        else:
            a = -s
            np = n - a
            if np <= 0:
                continue
            grad[:np] += (1.0 - x[a:a + np])
            grad[a:] -= x[:np]
    grad *= dx
    lse_grad = (np.zeros(2 * n - 1) + weights * beta * np.exp(np.clip(beta * shifted, -500, 500)) / se) * dx
    # Actually the gradient of the log-sum-exp term
    # d/dx [log(sum(exp(beta*(corr_k - m))))] where m is fixed
    # = sum_k (beta * exp(beta*(corr_k-m)) / sum(exp(beta*(corr_k-m)))) * d(corr_k)/dx
    lse_grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        if s >= 0:
            np_ = n - s
            if np_ <= 0:
                continue
            coeff = weights[k] * beta * dx
            lse_grad[s:] += coeff * (1.0 - x[:np_])
            lse_grad[:np_] -= coeff * x[s:s + np_]
        else:
            a = -s
            np_ = n - a
            if np_ <= 0:
                continue
            coeff = weights[k] * beta * dx
            lse_grad[:np_] += coeff * (1.0 - x[a:a + np_])
            lse_grad[a:] -= coeff * x[:np_]
    total_grad = np.zeros(n)
    # Gradient of max term (subgradient) + gradient of lse term
    # For the max term: subgradient is the gradient of the argmax correlation
    top_k = int(np.argmax(corr))
    s = top_k - (n - 1)
    if s >= 0:
        np_ = n - s
        if np_ > 0:
            total_grad[s:] += (1.0 - x[:np_]) * dx
            total_grad[:np_] -= x[s:s + np_] * dx
    else:
        a = -s
        np_ = n - a
        if np_ > 0:
            total_grad[:np_] += (1.0 - x[a:a + np_]) * dx
            total_grad[a:] -= x[:np_] * dx
    total_grad += lse_grad
    return total_grad


def active_subgrad(h, n, top_k=5):
    dx = 2.0 / n
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-top_k:]
    top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
    grad = np.zeros(n)
    for k in top_idx:
        s = int(k) - (n - 1)
        if s >= 0:
            np_ = n - s
            if np_ <= 0:
                continue
            grad[s:] += (1.0 - h[:np_])
            grad[:np_] -= h[s:s + np_]
        else:
            a = -s
            np_ = n - a
            if np_ <= 0:
                continue
            grad[:np_] += (1.0 - h[a:a + np_])
            grad[a:] -= h[:np_]
    return grad * dx, corr


def do_slsqp(h_start, n, beta_list, maxiter_list=None):

    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum_fun, 'jac': con_sum_jac}]
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    for i, beta in enumerate(beta_list):
        if time_remaining() < 8.0:
            break
        _G_BETA = float(beta)
        if maxiter_list is not None:
            mi = maxiter_list[i]
        else:
            if beta < 200:
                mi = 500
            elif beta < 2000:
                mi = 400
            elif beta < 10000:
                mi = 300
            else:
                mi = 200
        mi = min(mi, max(20, int(time_remaining() * 0.3 / 0.5)))
        try:
            if beta < 5000:
                res = minimize(obj_lse, h_cur, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': mi, 'ftol': 1e-18, 'iprint': 0})

def do_active_grad(h_start, n, max_iters=5000, top_k=5):
    h = h_start.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    no_improve = 0
    for it in range(max_iters):
        if time_remaining() < 3.0:
            break
        grad, corr = active_subgrad(h, n, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        direction = -grad / gnorm
        alpha = 0.05
        improved = False
        for _ in range(30):
            h_trial = project_feasible(h + alpha * direction, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-14:
                c5_best = c5_trial
                h = h_trial
                h_best = h.copy()
                improved = True
                no_improve = 0
                break
            alpha *= 0.8
        if not improved:
            no_improve += 1
            # Try very tiny steps
            for tiny_alpha in [0.001, 0.0005, 0.0001, 0.00005]:
                h_trial = project_feasible(h + tiny_alpha * direction, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < c5_best - 1e-15:
                    c5_best = c5_trial
                    h = h_trial
                    h_best = h.copy()
                    no_improve = 0
                    break

                c5_pert = compute_c5(h_pert)

                    h = h_pert
                    c5_best = c5_pert
                    h_best = h.copy()

    return h_best, c5_best


def do_binary_local_search(h_start, n, max_sweeps=50, seed=0):
    """Local search on binary solutions: flip pairs to maintain sum constraint."""
    rng = np.random.RandomState(seed)
    # Start from nearest binary
    h_bin = (h_start >= 0.5).astype(float)
    # Ensure exactly n/2 ones
    num_ones = int(np.sum(h_bin))
    target = n // 2
    if num_ones > target:
        ones_idx = np.where(h_bin == 1.0)[0]
        remove = rng.choice(ones_idx, size=num_ones - target, replace=False)
        h_bin[remove] = 0.0
    elif num_ones < target:
        zeros_idx = np.where(h_bin == 0.0)[0]
        add = rng.choice(zeros_idx, size=target - num_ones, replace=False)
        h_bin[add] = 1.0
    c5_best = compute_c5(h_bin)
    h_best = h_bin.copy()
    
    for sweep in range(max_sweeps):
        if time_remaining() < 3.0:
            break
        improved = False
        ones_idx = np.where(h_best == 1.0)[0]
        zeros_idx = np.where(h_best == 0.0)[0]
        # Try random swaps
        n_tries = min(n, 200)
        for _ in range(n_tries):
            if time_remaining() < 1.5:
                break
            a = int(rng.choice(ones_idx))
            b = int(rng.choice(zeros_idx))
            h_try = h_best.copy()
            h_try[a] = 0.0
            h_try[b] = 1.0
            c5_try = compute_c5(h_try)
            if c5_try < c5_best - 1e-14:
                c5_best = c5_try
                h_best = h_try
                improved = True
                ones_idx = np.where(h_best == 1.0)[0]
                zeros_idx = np.where(h_best == 0.0)[0]
        if not improved and sweep > 10:
            break
    return h_best, c5_best


def make_thue_morse(n):
    h = np.zeros(n)

def make_golden_ratio(n):

        h[i] = 1.0 if frac < 0.5 else 0.0
    return h
