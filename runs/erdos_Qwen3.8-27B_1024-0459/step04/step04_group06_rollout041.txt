# Smooth objective for SLSQP
def _make_smooth_obj(n, beta):
    dx = 2.0 / n
    def smooth_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        exps = np.exp(np.clip(beta * shifted, -500, 500))
        return (m + np.log(np.sum(exps)) / beta) * dx
    return smooth_obj

def _gradient_smooth(n, beta):
    dx = 2.0 / n
    def grad_smooth(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        exps = np.exp(np.clip(beta * shifted, -500, 500))
        weights = exps / np.sum(exps)
        
        grad = np.zeros(n)
        for k in range(len(corr)):
            w = weights[k] * dx
            if w < 1e-15:
                continue
            offset = k - (n - 1)
            if offset >= 0:
                n_valid = n - offset
                if n_valid <= 0:
                    continue
                grad[offset:offset + n_valid] += w * (1.0 - x[:n_valid])

                    continue
                grad[:n_valid] += w * (1.0 - x[mm:mm + n_valid])

def compute_grad_c5(h, n, top_k=8):
    """Compute subgradient of max_k corr[k]*dx by averaging over top-k terms."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_indices = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for k_idx in top_indices:

        if offset >= 0:
            n_valid = n - offset
            if n_valid <= 0:
                continue
            grad[offset:offset + n_valid] += (1.0 - h[:n_valid])
            grad[:n_valid] -= h[offset:offset + n_valid]

def smooth_grad(h, n, beta):
    """Analytical gradient of the smooth approximation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-18:

        offset = k - (n - 1)

        if offset >= 0:
            n_valid = n - offset
            if n_valid <= 0:

            grad[:n_valid] -= dw * h[offset:offset + n_valid]

            grad[:n_valid] += dw * (1.0 - h[mm:mm + n_valid])

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
