        if len(nb_idx) > 3:
            flip_idx = rng.choice(nb_idx, size=min(3, len(nb_idx)), replace=False)
            h_f[flip_idx] = 1.0 - h_f[flip_idx]
            h_f = project_feasible(h_f, n)
            c_f = compute_c5(h_f)
            if c_f < c_p:
                h_p = h_f
                c_p = c_f
        
        if c_p < c5 * 1.03:
            # Re-optimize from perturbed point
            topK = np.argsort(compute_corr_val(h_p))[-8:]
            h_new = slsqp_topk(h_p, n, topK, dx, 500.0, maxiter=200)
            c_new = compute_c5(h_new)
            if c_new < c5 - 1e-12:
                c5 = c_new
                h_best = h_new
    return h_best, c5

def compute_corr_val(h):
    return np.correlate(h, 1.0 - h, mode='full')

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    budget = float(budget_s)
    
    def tlimit():
        return max(0.0, budget - (time.time() - t_start) - 5.0)
    
    rng = np.random.RandomState(seed)
    
    # Initialize
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
    except Exception:
        parent_n = 256
        parent_h = make_tm(parent_n)
    
    parent_h = project_feasible(parent_h, parent_n)
    best_h = parent_h.copy()
    best_c5 = compute_c5(parent_h)
    best_n = parent_n
    print("Parent: n={}, C5={:.10f}".format(parent_n, best_c5))
    
    # Phase 1: Intensive top-K active-lag optimization at parent resolution
    if tlimit() > 150:
        print("\n=== Phase 1: Active-lag iteration n={} ===".format(best_n))
        n = best_n
        h_cur = best_h.copy()
        c5_cur = best_c5
        
        # Multiple rounds of top-K optimization with increasing K
        k_schedule = [3, 4, 5, 6, 8, 10, 12, 15, 20, 25, 30, 40, 50]
        for K in k_schedule:
            if tlimit() < 10:
                break
            corr, dx = get_corr(h_cur)
            topK = np.argsort(corr)[-K:]
            for beta in [200.0, 1000.0, 5000.0]:
                if tlimit() < 5:
                    break
                h_new = slsqp_topk(h_cur, n, topK, dx, beta, maxiter=400)
                c_new = compute_c5(h_new)
                if c_new < c5_cur - 1e-14:
                    c5_cur = c_new
                    h_cur = h_new
        print("  After top-K schedule: C5={:.10f}".format(c5_cur))
        
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        else:
            # Keep the slightly better one
            c5_final_check = compute_c5(h_cur)
            if c5_final_check < best_c5:
                best_c5 = c5_final_check
                best_h = h_cur.copy()
        
        # Full smooth max optimization
        if tlimit() > 30:
            h_cur = best_h.copy()
            for beta in [500, 2000, 10000]:
                if tlimit() < 5:
                    break
                h_new = slsqp_full_smooth(h_cur, n, 2.0/n, beta, maxiter=500)
                c_new = compute_c5(h_new)
                if c_new < best_c5 - 1e-14:
                    best_c5 = c_new
                    best_h = h_new.copy()
                    h_cur = h_new
            print("  After full smooth: C5={:.10f}".format(best_c5))
    
    # Phase 2: Active lag focused iteration
    if tlimit() > 100:
        print("\n=== Phase 2: Focused active-lag iteration n={} ===".format(best_n))
        h_cur, c5_cur = active_lag_iteration(best_h, best_n, rng, tlimit, max_rounds=25)
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        print("  After focused: C5={:.10f}".format(best_c5))
    
    # Phase 3: Try other resolutions
    if tlimit() > 120:
        print("\n=== Phase 3: Other resolutions ===")
        n_tries = [256]
        if best_n != 320:
            n_tries.append(320)
        if best_n != 384:
            n_tries.append(384)
        if best_n != 128:
            n_tries.append(128)
        if best_n != 512:
            n_tries.append(512)
        
        for n_try in n_tries:
            if tlimit() < 50:
                break
            h0 = project_feasible(interp_h(best_h, best_n, n_try), n_try)
            c5_0 = compute_c5(h0)
            
            # Also try Thue-Morse at this resolution
            h_tm = project_feasible(make_tm(n_try), n_try)
            c5_tm = compute_c5(h_tm)
            
            # Start from better initial
            if c5_tm < c5_0:
                h_cur = h_tm
                c5_cur = c5_tm
            else:
                h_cur = h0
                c5_cur = c5_0
            
            # Optimize
            if n_try <= 320:
                for beta in [300, 1500, 5000]:
                    if tlimit() < 10:
                        break
                    h_new = slsqp_full_smooth(h_cur, n_try, 2.0/n_try, beta, maxiter=400)
                    c_new = compute_c5(h_new)
                    if c_new < c5_cur - 1e-14:
                        c5_cur = c_new
                        h_cur = h_new
            else:
                # L-BFGS-B for larger n
                bounds = [(0.0, 1.0)] * n_try
                target = n_try / 2.0
                dx_t = 2.0 / n_try
                beta_lb = 1000.0
                def obj_lb(x):
                    corr = np.correlate(x, 1.0 - x, mode='full')
                    m = np.max(corr)
                    s = corr - m
                    e = np.exp(np.clip(1000.0 * s, -400, 400))
                    return float((m + np.log(np.sum(e)) / 1000.0)) * dx_t + 50.0 * (np.sum(x) - target)**2
                try:
                    res = minimize(obj_lb, h_cur, method='L-BFGS-B', bounds=bounds,
                                   options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-10, 'iprint': 0})
                    h_new = project_feasible(res.x, n_try)
                    c_new = compute_c5(h_new)
                    if c_new < c5_cur - 1e-14:
                        c5_cur = c_new
                        h_cur = h_new
                except Exception:
                    pass
                # Follow up with SLSQP
                if tlimit() > 20:
                    h_new = slsqp_full_smooth(h_cur, n_try, 2.0/n_try, 2000, maxiter=300)
                    c_new = compute_c5(h_new)
                    if c_new < c5_cur - 1e-14:
                        c5_cur = c_new
                        h_cur = h_new
            
            if c5_cur < best_c5 - 1e-10:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n_try
                print("  n={} IMPROVED: C5={:.10f}".format(n_try, c5_cur))
            else:
                print("  n={} best: C5={:.10f}".format(n_try, c5_cur))
    
    # Phase 4: Pairwise coordinate descent
    if tlimit() > 40:
        print("\n=== Phase 4: Pairwise coord descent n={} ===".format(best_n))
        h_cur, c5_cur = pairwise_coord_descent(best_h, best_n, rng, tlimit, max_sweeps=50)
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        print("  After coord descent: C5={:.10f}".format(best_c5))
    
    # Phase 5: Perturbation + re-optimization
    if tlimit() > 80:
        print("\n=== Phase 5: Perturbation escapes ===")
        h_cur, c5_cur = perturb_and_optimize(best_h, best_n, rng, tlimit, n_pert=10)
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        print("  After perturbations: C5={:.10f}".format(best_c5))
    
    # Phase 6: Another round of active-lag iteration
    if tlimit() > 40:
        print("\n=== Phase 6: Second active-lag pass ===")
        h_cur, c5_cur = active_lag_iteration(best_h, best_n, rng, tlimit, max_rounds=20)
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        print("  After second focused: C5={:.10f}".format(best_c5))
    
    # Phase 7: Final coordinate descent
    if tlimit() > 20:
        best_h, best_c5 = pairwise_coord_descent(best_h, best_n, rng, tlimit, max_sweeps=30)
        print("  After final coord: C5={:.10f}".format(best_c5))
    
    # Phase 8: Ultra-fine SLSQP polish with very high beta
    if tlimit() > 15:
        print("\n=== Phase 8: Ultra-fine polish ===")
        h_cur = best_h.copy()
        for beta in [5000, 20000, 50000]:
            if tlimit() < 5:
                break
            h_new = slsqp_full_smooth(h_cur, best_n, 2.0/best_n, beta, maxiter=600)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new
                h_cur = h_new
        print("  After ultra-fine: C5={:.10f}".format(best_c5))
    
    # Final: targeted single-lag optimization
    if tlimit() > 10:
        corr, dx = get_corr(best_h)
        k_max = int(np.argmax(corr))
        # Try to reduce this specific lag
        h_cur = best_h.copy()
        for _ in range(5):
            if tlimit() < 3:
                break
            individual = np.array([k_max])
            h_new = slsqp_topk(h_cur, best_n, individual, dx, 10000.0, maxiter=500)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new.copy()
                h_cur = h_new
            else:
                break
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
