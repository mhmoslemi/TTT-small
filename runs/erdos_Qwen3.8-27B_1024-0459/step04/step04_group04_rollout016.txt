def project_feasible(h, n):

        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr)) * dx_val

                no_improve = 0
        top_idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k in top_idx:
            g += corr_grad_per_shift(h, n, int(k))
        g /= top_k
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        d = -g / g_norm
        h_new = project_feasible(h + alpha * d, n)
        c_new = compute_c5(h_new)
        if c_new < c5_val:
            h = h_new
            alpha = min(alpha * 1.05, 0.5)
        else:
            h_half = project_feasible(h + 0.5 * alpha * d, n)
            c_half = compute_c5(h_half)
            if c_half < c5_val:
                h = h_half
                alpha *= 0.8
            else:
                alpha *= decay
                if alpha < 1e-9:
                    break
    return best_h, best_c5


def targeted_binary_search(h, n, max_sweeps=40, n_tries=300, top_k_target=5, seed=42):
    rng = np.random.RandomState(seed)
    h = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[rng.choice(len(idx), n_ones - target, replace=False)] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[rng.choice(len(idx), target - n_ones, replace=False)] = 1
    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(corr[np.argmax(corr)]) * dx_val
    best_max = cur_max
    best_h = h.copy()
    for sweep in range(max_sweeps):
        if remaining() < 10:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for _ in range(n_tries):
            if remaining() < 5:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr)) * dx_val
            if new_max < cur_max - 1e-15:
                cur_max = new_max
                improved = True
                if new_max < best_max:
                    best_max = new_max
                    best_h = h.copy()
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
            else:
                h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return best_h, best_max


def sa_binary(h, n, time_limit=50, seed=42):
    rng = np.random.RandomState(seed)
    h = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[rng.choice(len(idx), n_ones - target, replace=False)] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[rng.choice(len(idx), target - n_ones, replace=False)] = 1
    dx_val = 2.0 / n
    cur_max = float(np.max(compute_corr(h))) * dx_val
    best_max = cur_max
    best_h = h.copy()
    T = 5.0
    alpha_cool = 0.997
    start = time.time()
    stall = 0
    while time.time() - start < time_limit and T > 0.05 and remaining() > 10:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_max = float(np.max(compute_corr(h))) * dx_val
        delta = new_max - cur_max
        if delta < 0 or (T > 0.05 and rng.rand() < np.exp(-delta / T)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
        T *= alpha_cool
        if stall > 3000:
            T = max(T, 1.5)
            stall = 0
    return best_h, best_max


def multi_restart_continuous(h_start, n, budget_remaining, seed):
    rng = np.random.RandomState(seed)
    best_h = h_start.copy()
    best_c5 = compute_c5(h_start)
    n_restarts = min(4, max(2, int(budget_remaining / 40)))
    for r in range(n_restarts):
        if remaining() < 20:
            break
        if r == 0:
            h_work = h_start.copy()
        else:
            pert = rng.randn(n) * 0.05
            h_work = project_feasible(h_start + pert, n)
        for beta in [500, 2000, 10000, 50000]:
            if remaining() < 10:
                break
            h_new = slsqp_run(h_work, n, beta, maxiter=250, topk=25)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_work = h_new.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT, _TOPK
    _START = time.time()
    _BUDGET = budget_s
    _TOPK = 30
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 360
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n

    # Phase 1: Multi-restart continuous optimization at parent n
    h_opt, c5_opt = multi_restart_continuous(best_h, n, remaining(), seed)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()

    # Phase 2: Subgradient descent (adaptive step, top-k=10 for sharpness)
    if remaining() > 30:
        h_sg, c5_sg = subgradient_descent(best_h, n, max_iters=8000, top_k=10,
                                          alpha_init=0.06, decay=0.996)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 3: High-beta SLSQP polish
    for beta in [30000, 80000, 200000]:
        if remaining() < 10:
            break
        h_new = slsqp_run(best_h, n, beta, maxiter=400, topk=20)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 4: Subgradient with smaller top-k (focus on single worst shift)
    if remaining() > 20:
        h_sg2, c5_sg2 = subgradient_descent(best_h, n, max_iters=5000, top_k=5,
                                            alpha_init=0.03, decay=0.994)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()

    # Phase 5: Binary local search (for integer-bound solutions)
    if remaining() > 20:
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        target = n // 2
        if n_ones != target:
            if n_ones > target:
                idx = np.where(h_bin == 1)[0]
                h_bin[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
            else:
                idx = np.where(h_bin == 0)[0]
                h_bin[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1
        h_bl, c5_bl = targeted_binary_search(h_bin, n, max_sweeps=25, n_tries=200,
                                             top_k_target=5, seed=seed)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()

    # Phase 6: Try n=720 (higher resolution)
    if remaining() > 60:
        n2 = 2 * n
        if n2 <= 1200:
            h_hi = interp_to(best_h, n, n2)
            h_hi = project_feasible(h_hi, n2)
            h_hi = slsqp_run(h_hi, n2, 1000, maxiter=200, topk=25)
            c5_hi = compute_c5(h_hi)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi.copy()
                best_n = n2
            if remaining() > 30:
                h_hi2, c5_hi2 = multi_restart_continuous(h_hi, n2, remaining(), seed + 999)
                if c5_hi2 < best_c5:
                    best_c5 = c5_hi2
                    best_h = h_hi2.copy()
                    best_n = n2

    # Phase 7: Try nearby n values (n±2, n±4) for continuous
    for dn in [-4, -2, 2, 4]:
        if remaining() < 30:
            break
        n_alt = best_n + dn
        if n_alt < 100 or n_alt > 1200 or n_alt % 2 != 0:
            continue
        h_alt = interp_to(best_h, best_n, n_alt)
        h_alt = project_feasible(h_alt, n_alt)
        h_alt = slsqp_run(h_alt, n_alt, 3000, maxiter=300, topk=25)
        c5_alt = compute_c5(h_alt)
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt.copy()
            best_n = n_alt

    # Phase 8: Final polish on best
    if remaining() > 15:
        h_new = slsqp_run(best_h, best_n, 100000, maxiter=500, topk=15)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if remaining() > 10:
        h_sg3, c5_sg3 = subgradient_descent(best_h, best_n, max_iters=3000, top_k=8,
                                            alpha_init=0.03, decay=0.995)
        if c5_sg3 < best_c5:
            best_c5 = c5_sg3
            best_h = h_sg3.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
