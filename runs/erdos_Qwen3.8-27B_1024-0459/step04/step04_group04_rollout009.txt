    return np.interp(x_new, xp, src_h)

def smooth_obj_fn(x, n, beta):
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -70, 70)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val

def smooth_grad_fn(x, n, beta):
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -70, 70)
    ex = np.exp(cl)
    wsum = np.sum(ex)
    w = ex / wsum
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            npairs = n - off
            g[:npairs] += w[k] * (1.0 - x[off:off + npairs])
            g[off:off + npairs] -= w[k] * x[:npairs]
        else:
            ao = -off
            npairs = n - ao
            g[ao:ao + npairs] += w[k] * (1.0 - x[:npairs])
            g[:npairs] -= w[k] * x[ao:ao + npairs]
    return g * dx_val

def make_sum_con(n):
    def cfun(x):
        return np.sum(x) - n / 2.0
    def cjac(x):
        return np.ones(n)
    return cfun, cjac

def slsqp_optimize(h, n, beta, maxiter=500):
    cfun, cjac = make_sum_con(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(smooth_obj_fn, h, args=(n, beta),
                       jac=lambda x, nn, bb: smooth_grad_fn(x, nn, bb),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

def cd_minimax(h, n, n_sweeps=20, top_k=5, time_limit=None):
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    for sweep in range(n_sweeps):
        if remaining() < 5:
            break
        if time_limit is not None and time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        m = float(np.max(corr))
        # Active set: top_k shifts, excluding k=n-1 (off=0, quadratic)
        valid_k = [k for k in range(2 * n - 1) if k != n - 1]
        if len(valid_k) < top_k:
            valid_k = list(range(2 * n - 1))
        top_vals = np.zeros(2 * n - 1)
        top_vals[:] = -1e18
        top_vals[valid_k] = corr[valid_k]
        active = np.argsort(-top_vals)[:top_k]
        active = [int(k) for k in active if corr[k] > 0.01 * m]
        if len(active) == 0:
            active = [int(k) for k in np.argsort(-corr)[:top_k] if k != n - 1]

        improved = False
        for i in range(n):
            if remaining() < 2:
                break
            a_vals = []
            b_vals = []
            for k in active:
                off = k - (n - 1)
                a = 0.0
                if off >= 0:
                    if i <= n - 1 - off:
                        a += (1.0 - h[i + off])
                    if off <= i <= n - 1:
                        a -= h[i - off]
                else:
                    ao = -off
                    if ao <= i <= n - 1:
                        a += (1.0 - h[i - ao])
                    if 0 <= i <= n - 1 - ao:
                        a -= h[i + ao]
                b = corr[k] - a * h[i]
                a_vals.append(a)
                b_vals.append(b)

            a_arr = np.array(a_vals)
            b_arr = np.array(b_vals)
            cur_val = float(np.max(a_arr * h[i] + b_arr))
            best_x = h[i]
            best_val = cur_val

            candidates = [0.0, 1.0]
            for p in range(len(a_vals)):
                for q in range(p + 1, len(a_vals)):
                    da = a_arr[p] - a_arr[q]
                    if abs(da) > 1e-12:
                        x_int = -(b_arr[p] - b_arr[q]) / da
                        if -0.01 <= x_int <= 1.01:
                            x_int = max(0.0, min(1.0, x_int))
                            candidates.append(x_int)

            for x_c in candidates:
                val = float(np.max(a_arr * x_c + b_arr))
                if val < best_val - 1e-10:
                    best_val = val
                    best_x = x_c

            if abs(best_x - h[i]) > 1e-8:
                h[i] = best_x
                improved = True

        h = project_feasible(h, n)
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
        if not improved:
            break
    return h, best_c5

def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin > 0.5))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin > 0.5)[0]
        h_bin[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0.0
    elif n_ones < target:
        idx = np.where(h_bin < 0.5)[0]
        h_bin[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1.0
    return h_bin

def sa_binary(h, n, time_limit=60, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    target = n // 2
    n_ones = int(np.sum(h > 0.5))
    if n_ones > target:
        idx = np.where(h > 0.5)[0]
        h[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0.0
    elif n_ones < target:
        idx = np.where(h < 0.5)[0]
        h[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1.0

    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()
    T = 12.0
    alpha = 0.9988
    start = time.time()
    stall = 0
    max_stall = 10000

    while time.time() - start < time_limit and T > 0.05:
        if remaining() < 10:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        delta = new_max - cur_max
        if delta < 0 or (T > 0.05 and rng.rand() < np.exp(-delta / T)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
        T *= alpha
        if stall > max_stall:
            T = max(T, 4.0)
            stall = 0
    return best_h, best_max * dx_val

def pair_cd(h, n, n_sweeps=5, top_k=4):
    """Pair coordinate descent: adjust pairs (h_i, h_j) to reduce max."""
    h = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(n_sweeps):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        m = float(np.max(corr))
        active = np.argsort(-corr)[:top_k]
        active = [int(k) for k in active if k != n - 1]
        if len(active) == 0:
            break

        for i in range(n - 1):
            if remaining() < 2:
                break
            j = i + 1
            # Try swapping contributions: transfer from h_i to h_j and vice versa
            corr_new = h[i] - 0.01
            h_trial = h.copy()
            h_trial[i] = max(0.0, h[i] - 0.02)
            h_trial[j] = min(1.0, h[j] + 0.02)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-10:
                h = h_trial
                best_c5 = c5_trial
            else:
                h_trial2 = h.copy()
                h_trial2[i] = min(1.0, h[i] + 0.02)
                h_trial2[j] = max(0.0, h[j] - 0.02)
                c5_trial2 = compute_c5(h_trial2)
                if c5_trial2 < best_c5 - 1e-10:
                    h = h_trial2
                    best_c5 = c5_trial2
        h = project_feasible(h, n)
        c5_check = compute_c5(h)
        if c5_check < best_c5:
            best_c5 = c5_check
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 360
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n0 = best_n

    # Phase 1: SLSQP with increasing beta at parent n
    h_work = best_h.copy()
    for beta in [500, 2000, 8000, 20000, 50000, 100000]:
        if remaining() < 15:
            break
        h_new = slsqp_optimize(h_work, n0, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_work = h_new.copy()

    # Phase 2: Coordinate descent minimax (new approach)
    if remaining() > 30:
        h_cd, c5_cd = cd_minimax(h_work, n0, n_sweeps=30, top_k=5, time_limit=120)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h_work = h_cd.copy()

    # Phase 3: SLSQP after CD
    for beta in [30000, 80000, 150000]:
        if remaining() < 10:
            break
        h_new = slsqp_optimize(h_work, n0, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_work = h_new.copy()

    # Phase 4: CD again with tighter focus
    if remaining() > 20:
        h_cd, c5_cd = cd_minimax(h_work, n0, n_sweeps=40, top_k=3, time_limit=100)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h_work = h_cd.copy()

    # Phase 5: Explore other n values
    n_explore = [320, 400, 480, 500, 600, 720, 800]
    for n_t in n_explore:
        if remaining() < 25:
            break
        h_t = interp_to(best_h, best_n, n_t)
        h_t = project_feasible(h_t, n_t)
        c5_t = compute_c5(h_t)

        # Only invest time if competitive
        if c5_t < best_c5 + 0.02:
            h_t = slsqp_optimize(h_t, n_t, 5000, maxiter=300)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5 + 0.01:
                if remaining() > 10:
                    h_t2, c5_t2 = cd_minimax(h_t, n_t, n_sweeps=15, top_k=5, time_limit=50)
                    if c5_t2 < c5_t:
                        h_t, c5_t = h_t2, c5_t2
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = h_t.copy()
                    best_n = n_t

    # Phase 6: Binary SA at various n
    for n_t in [400, 500, 600, 720, 800, 900]:
        if remaining() < 20:
            break
        h_bin = make_binary(interp_to(best_h, best_n, n_t), n_t, rng)
        t_sa = min(25, max(8, remaining() / 3.0))
        for sa_seed in [seed + n_t * 3, seed + 9999 + n_t]:
            if remaining() < 10:
                break
            h_sa, c5_sa = sa_binary(h_bin, n_t, time_limit=t_sa, seed=sa_seed)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_t

    # Phase 7: Final intensive refinement at best_n
    n_best = best_n
    h_best = best_h.copy()

    for beta in [50000, 100000, 200000, 500000]:
        if remaining() < 8:
            break
        h_new = slsqp_optimize(h_best, n_best, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_best = h_new.copy()

    if remaining() > 15:
        h_cd, c5_cd = cd_minimax(h_best, n_best, n_sweeps=50, top_k=3, time_limit=120)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h_best = h_cd.copy()

    if remaining() > 10:
        h_cd, c5_cd = cd_minimax(h_best, n_best, n_sweeps=30, top_k=5, time_limit=60)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Final SLSQP polish
    if remaining() > 8:
        h_new = slsqp_optimize(best_h, best_n, 200000, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
