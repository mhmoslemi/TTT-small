    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp  # O(n)
    
    obj = (m + np.log(sum_exp) / beta) * dx
    obj += penalty * (np.sum(h) - n/2.0)**2
    
    # Gradient: O(n * num_weighted_lags)
    # For each lag i (0 to 2n-2), weight is weights[i], lag = i-(n-1)
    grad = np.zeros(n)
    for i in range(2*n-1):
        if weights[i] < 1e-12:
            continue
        lag = i - (n-1)
        w = weights[i]
        if lag >= 0:
            # d(corr)/dh[m] = (1-h[m-lag])*[lag<=m<=n-1] - h[m+lag]*[m<=n-1-lag]
