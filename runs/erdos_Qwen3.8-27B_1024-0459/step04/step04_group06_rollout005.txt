    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h


def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 3.0)


def subgrad_compute(h, n, top_k=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    top_indices = np.argsort(corr)[-top_k:]
    top_indices = top_indices[np.argsort(corr[top_indices])[::-1]]
    grad = np.zeros(n)
    for k_idx in top_indices:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            n_valid = n - offset
            if n_valid <= 0:
                continue
            grad[offset:offset + n_valid] += (1.0 - h[:n_valid])
            grad[:n_valid] -= h[offset:offset + n_valid]
        else:
            m = -offset
            n_valid = n - m
            if n_valid <= 0:
                continue
            grad[:n_valid] += (1.0 - h[m:m + n_valid])
            grad[m:m + n_valid] -= h[:n_valid]
    return grad, top_indices, corr


def subgrad_momentum(h, n, max_iter=500, step0=0.12, momentum=0.85):
    c5_best = compute_c5(h)
    h_best = h.copy()
    v = np.zeros(n)
    step = step0
    patience = 0
    for it in range(max_iter):
        if remaining_time() < 5.0:
            break
        grad, top_ids, corr = subgrad_compute(h, n, top_k=5)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad = grad / gn
        alpha = step
        improved = False
        for _ in range(12):
            v = momentum * v - alpha * grad
            h_trial = h + v
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-13:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
            alpha *= 0.5
        if improved:
            patience = 0
            step = min(alpha * 1.3, 0.4)
        else:
            patience += 1
            step *= 0.6
            v *= 0.5
            if patience > 15:
                break
    return h_best, c5_best


def smooth_max_val(x):
    n = _G_N
    dx = _G_DX
    beta = _G_BETA
    lam = _G_LAM
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -600, 600))
    val = (m + np.log(np.sum(exps)) / beta) + lam * (np.sum(x) - _G_TGT) ** 2
    return val


def smooth_max_grad(x):
    n = _G_N
    dx = _G_DX
    beta = _G_BETA
    lam = _G_LAM
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -600, 600))
    w_sum = np.sum(exps)
    weights = exps / w_sum
    grad = np.zeros(n)
    n_corr = len(corr)
    # Only use top weights for efficiency
    thresh = 1e-10
    for i in range(n_corr):
        w = weights[i]
        if w < thresh:
            continue
        offset = i - (n - 1)
        if offset >= 0:
            n_valid = n - offset
            if n_valid <= 0:
                continue
            grad[offset:offset + n_valid] += w * dx * (1.0 - x[:n_valid])
            grad[:n_valid] -= w * dx * x[offset:offset + n_valid]
        else:
            mi = -offset
            n_valid = n - mi
            if n_valid <= 0:
                continue
            grad[:n_valid] += w * dx * (1.0 - x[mi:mi + n_valid])
            grad[mi:mi + n_valid] -= w * dx * x[:n_valid]
    grad += 2.0 * lam * (np.sum(x) - _G_TGT)
    return grad


def lbbg_optimize(h, n, beta_val, lam_val, max_iter=300):
    global _G_N, _G_TGT, _G_DX, _G_BETA, _G_LAM
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    _G_BETA = float(beta_val)
    _G_LAM = float(lam_val)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_val, h, jac=smooth_max_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-16,
                                               'gtol': 1e-10, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        return h_new, c5_new
    except Exception:
        return h.copy(), compute_c5(h)


def do_subgrad_adaptive(h, n, max_iter=400):
    c5_best = float(compute_c5(h))
    h_best = h.copy()
    v = np.zeros(n)
    step = 0.1
    patience = 0
    for it in range(max_iter):
        if remaining_time() < 4.0:
            break
        frac = it / max(max_iter - 1, 1)
        top_k = int(3 + 25 * frac)
        grad, top_ids, corr = subgrad_compute(h, n, top_k=top_k)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad = grad / gn
        alpha = step
        improved = False
        for _ in range(10):
            v = 0.85 * v - alpha * grad
            h_trial = h + v
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-13:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
            alpha *= 0.5
        if improved:
            patience = 0
            step = min(alpha * 1.2, 0.3)
        else:
            patience += 1
            step *= 0.55
            v *= 0.4
            if patience > 12:
                break
    return h_best, c5_best


def perturb_h(h, n, rng, mode=0):
    h_trial = h.copy()
    if mode == 0:
        start = rng.randint(0, max(1, n - 10))
        L = rng.randint(4, min(20, n // 4))
        h_trial[start:start + L] = 1.0 - h_trial[start:start + L]
    elif mode == 1:
        s1 = rng.randint(0, n // 3)
        s2 = s1 + rng.randint(n // 4, max(1, n - n // 4))
        L = rng.randint(3, min(10, n // 12))
        a = h_trial[s1:s1 + L].copy()
        h_trial[s1:s1 + L] = h_trial[s2:s2 + L]
        h_trial[s2:s2 + L] = a
    elif mode == 2:
        h_trial += 0.03 * rng.randn(n)
    elif mode == 3:
        subset = rng.choice(n, size=max(1, n // 6), replace=False)
        h_trial[subset] = rng.rand(len(subset))
    elif mode == 4:
        freq = rng.randint(2, max(3, n // 5))
        phase = rng.rand() * 2 * np.pi
        t = np.arange(n) / float(n)
        h_trial += 0.04 * np.sin(2.0 * np.pi * freq * t + phase)
    elif mode == 5:
        idx = rng.choice(n, size=max(1, n // 10), replace=False)
        h_trial[idx] = 1.0 - h_trial[idx]
    elif mode == 6:
        # Gaussian bump
        center = rng.randint(0, n)
        width = rng.randint(5, max(6, n // 8))
        t = np.arange(n)
        h_trial += 0.05 * np.exp(-((t - center) ** 2) / (2 * width ** 2))
    elif mode == 7:
        # Random walk perturbation
        walk = np.cumsum(0.02 * rng.randn(n))
        walk -= walk.mean()
        h_trial += 0.3 * walk
    h_trial = project_feasible(h_trial, n)
    return h_trial


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    h = project_feasible(h, n)
    return h


def coordinate_descent_fast(h, n, max_sweeps=8, rng=None):
    c5_best = float(compute_c5(h))
    h_best = h.copy()
    if rng is None:
        rng = np.random.RandomState(0)
    for sweep in range(max_sweeps):
        if remaining_time() < 3.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if remaining_time() < 2.0:
                break
            old = h_best[i]
            # Try binary values
            for new_v in [0.0, 1.0]:
                if abs(new_v - old) < 1e-8:
                    continue
                h_trial = h_best.copy()
                h_trial[i] = new_v
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < c5_best - 1e-12:
                    c5_best = c5_t
                    h_best = h_trial.copy()
                    any_imp = True
                    old = h_best[i]
            # Try small continuous perturbations
            for delta in [0.03, -0.03, 0.08, -0.08]:
                nv = np.clip(old + delta, 0.0, 1.0)
                if abs(nv - old) < 1e-8:
                    continue
                h_trial = h_best.copy()
                h_trial[i] = nv
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < c5_best - 1e-12:
                    c5_best = c5_t
                    h_best = h_trial.copy()
                    any_imp = True
                    old = h_best[i]
        if not any_imp and sweep > 1:
            break
    return h_best, c5_best


def simulated_annealing(h, n, rng, num_rounds=200, t0=0.01, t1=0.001):
    c5_best = compute_c5(h)
    h_best = h.copy()
    h_curr = h.copy()
    c5_curr = c5_best
    for r in range(num_rounds):
        if remaining_time() < 3.0:
            break
        temp = t0 * (t1 / t0) ** (r / max(num_rounds - 1, 1))
        h_trial = h_curr.copy()
        # Flip a small random set
        n_flip = rng.randint(2, max(3, n // 20))
        idx = rng.choice(n, size=n_flip, replace=False)
        h_trial[idx] = 1.0 - h_trial[idx]
        h_trial = project_feasible(h_trial, n)
        c5_trial = compute_c5(h_trial)
        delta = c5_trial - c5_curr
        if delta < 0 or rng.rand() < np.exp(-delta / max(temp, 1e-10)):
            h_curr = h_trial
            c5_curr = c5_trial
            if c5_curr < c5_best:
                c5_best = c5_curr
                h_best = h_curr.copy()
    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Initialize from parent or Thue-Morse
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = make_thue_morse(parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("Thue-Morse: n={}, C5={:.10f}".format(parent_n, best_c5))

    # Phase 1: Interpolate to n=512 and quick subgradient
    n_primary = 512
    print("\n--- Phase 1: Interpolate to n={} + subgradient ---".format(n_primary))
    h512 = project_feasible(interp_to(best_h, best_n, n_primary), n_primary)
    c5_512 = compute_c5(h512)
    print("  Interpolated C5={:.10f}".format(c5_512))

    _G_N = n_primary
    _G_TGT = n_primary / 2.0
    _G_DX = 2.0 / n_primary
    h512, c5_512 = subgrad_momentum(h512, n_primary, max_iter=200, step0=0.1)
    print("  After subgrad C5={:.10f}".format(c5_512))

    if c5_512 < best_c5:
        best_c5 = c5_512
        best_h = h512.copy()
        best_n = n_primary

    # Phase 2: L-BFGS-B with smooth max (multiple betas)
    print("\n--- Phase 2: L-BFGS-B smooth max ---")
    for beta_val in [50, 200, 800, 3000]:
        if remaining_time() < 15.0:
            break
        h512, c5_512 = lbbg_optimize(h512, n_primary, beta_val, 500.0, max_iter=400)
        print("  beta={}: C5={:.10f}".format(beta_val, c5_512))
        if c5_512 < best_c5:
            best_c5 = c5_512
            best_h = h512.copy()
            best_n = n_primary

    # Phase 3: Adaptive subgradient (increasing top-k)
    print("\n--- Phase 3: Adaptive subgradient ---")
    h512, c5_512 = do_subgrad_adaptive(h512, n_primary, max_iter=300)
    print("  C5={:.10f}".format(c5_512))
    if c5_512 < best_c5:
        best_c5 = c5_512
        best_h = h512.copy()
        best_n = n_primary

    # Phase 4: L-BFGS-B again
    if remaining_time() > 60:
        print("\n--- Phase 4: L-BFGS-B round 2 ---")
        for beta_val in [500, 2000, 8000]:
            if remaining_time() < 15.0:
                break
            h512, c5_512 = lbbg_optimize(h512, n_primary, beta_val, 2000.0, max_iter=300)
            print("  beta={}: C5={:.10f}".format(beta_val, c5_512))
            if c5_512 < best_c5:
                best_c5 = c5_512
                best_h = h512.copy()
                best_n = n_primary

    # Phase 5: Perturbation + L-BFGS-B refinement
    print("\n--- Phase 5: Perturbation exploration ---")
    for p in range(20):
        if remaining_time() < 20.0:
            break
        mode = p % 8
        h_pert = perturb_h(h512, n_primary, rng, mode)
        c5_pert = compute_c5(h_pert)
        # Quick L-BFGS-B refinement
        h_ref, c5_ref = lbbg_optimize(h_pert, n_primary, 300.0, 1000.0, max_iter=200)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            best_n = n_primary
            h512 = h_ref.copy()
            print("  Pert {} (mode {}): C5={:.10f}".format(p, mode, c5_ref))
        elif c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
            best_n = n_primary

    # Phase 6: Simulated annealing
    if remaining_time() > 80:
        print("\n--- Phase 6: Simulated annealing ---")
        h512, c5_512 = simulated_annealing(h512, n_primary, rng, num_rounds=300)
        print("  SA C5={:.10f}".format(c5_512))
        if c5_512 < best_c5:
            best_c5 = c5_512
            best_h = h512.copy()
            best_n = n_primary
        # L-BFGS-B after SA
        h512, c5_512 = lbbg_optimize(h512, n_primary, 500.0, 2000.0, max_iter=300)
        if c5_512 < best_c5:
            best_c5 = c5_512
            best_h = h512.copy()
            best_n = n_primary
            print("  After L-BFGS-B: C5={:.10f}".format(c5_512))

    # Phase 7: Try n=384 (might be easier to optimize)
    if remaining_time() > 60:
        print("\n--- Phase 7: Try n=384 ---")
        n_alt = 384
        h_alt = project_feasible(interp_to(best_h, best_n, n_alt), n_alt)
        c5_alt = compute_c5(h_alt)
        _G_N = n_alt
        _G_TGT = n_alt / 2.0
        _G_DX = 2.0 / n_alt
        h_alt, c5_alt = subgrad_momentum(h_alt, n_alt, max_iter=150, step0=0.1)
        h_alt, c5_alt = lbbg_optimize(h_alt, n_alt, 500.0, 1000.0, max_iter=300)
        print("  n=384: C5={:.10f}".format(c5_alt))
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt.copy()
            best_n = n_alt

    # Phase 8: Try n=1024 (higher resolution)
    if remaining_time() > 100:
        print("\n--- Phase 8: Try n=1024 ---")
        n_hi = 1024
        h_hi = project_feasible(interp_to(best_h, best_n, n_hi), n_hi)
        c5_hi = compute_c5(h_hi)
        _G_N = n_hi
        _G_TGT = n_hi / 2.0
        _G_DX = 2.0 / n_hi
        h_hi, c5_hi = subgrad_momentum(h_hi, n_hi, max_iter=150, step0=0.08)
        h_hi, c5_hi = lbbg_optimize(h_hi, n_hi, 200.0, 500.0, max_iter=300)
        h_hi, c5_hi = lbbg_optimize(h_hi, n_hi, 1000.0, 2000.0, max_iter=200)
        print("  n=1024: C5={:.10f}".format(c5_hi))
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_hi

    # Phase 9: Coordinate descent polish
    if remaining_time() > 30:
        print("\n--- Phase 9: Coordinate descent ---")
        best_h, best_c5 = coordinate_descent_fast(best_h, best_n, max_sweeps=6, rng=rng)
        print("  CD C5={:.10f}".format(best_c5))

    # Phase 10: Final L-BFGS-B polish
    if remaining_time() > 15:
        print("\n--- Phase 10: Final L-BFGS-B ---")
        _G_N = best_n
        _G_TGT = best_n / 2.0
        _G_DX = 2.0 / best_n
        for beta_val in [3000, 10000]:
            if remaining_time() < 8.0:
                break
            best_h, best_c5 = lbbg_optimize(best_h, best_n, beta_val, 5000.0, max_iter=300)
            print("  beta={}: C5={:.10f}".format(beta_val, best_c5))

    # Final tiny coordinate descent
    if remaining_time() > 8:
        best_h, best_c5 = coordinate_descent_fast(best_h, best_n, max_sweeps=3, rng=rng)

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            print("  Evaluated C5={:.10f}".format(eval_c5))
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
