            g[off:off + npairs] += (1.0 - x[:npairs])

    else:
        ao = -off
        npairs = n - ao
        if npairs > 0:
            g[:npairs] += (1.0 - x[ao:ao + npairs])
            g[ao:ao + npairs] -= x[:npairs]
    return g


def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-14:
            g += w[k] * corr_grad_per_shift(x, n, k)
    return g * dx_val


def sum_con(x):
    return np.sum(x) - _TGT


def sum_con_jac(x):
    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=400):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def swap_delta(h, n, i, j, top_shifts):
    """Compute change in corr at top_shifts when swapping 1 at i to 0 and 0 at j to 1."""
    dcorr = np.zeros(len(top_shifts))
    nc = 2 * n - 1
    for idx in top_shifts:
        d = idx - (n - 1)
        delta = 0.0
        # Remove i as first element: was 1*(1-h[i+d])
        if 0 <= i + d < n:
            delta -= (1.0 - h[i + d])
        # Add i as second element: was h[i-d]*(1-1)=0, now h[i-d]*(1-0)=h[i-d]
        if 0 <= i - d < n:
            delta += h[i - d]
        # Add j as first element: was 0, now 1*(1-h[j+d])
        if 0 <= j + d < n:
            delta += (1.0 - h[j + d])
        # Remove j as second element: was h[j-d]*(1-0)=h[j-d], now h[j-d]*(1-1)=0
        if 0 <= j - d < n:
            delta -= h[j - d]
        dcorr[idx] = delta
    return dcorr


def binary_directed_search(h, n, time_limit, rng, top_k_shifts=15):
    """Directed binary swap search focusing on worst shifts."""
    h = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h))
    half = n // 2
    if n_ones > half:
        ones_idx = np.where(h == 1)[0]
        to_remove = rng.choice(ones_idx, n_ones - half, replace=False)
        h[to_remove] = 0.0
    elif n_ones < half:
        zeros_idx = np.where(h == 0)[0]
        to_add = rng.choice(zeros_idx, half - n_ones, replace=False)
        h[to_add] = 1.0

    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx
    best_h = h.copy()
    start = time.time()
    stall = 0
    max_stall = 200

    while time.time() - start < time_limit and remaining() > 10:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Identify top shifts
        top_idx = np.argsort(-corr)[:top_k_shifts]
        cur_max_val = float(corr[top_idx[0]])

        # Generate swap candidates
        n_try = 80
        best_improvement = 0.0
        best_swap = None
        best_new_corr = None

        for _ in range(n_try):
            # 65%: target worst shift contributors
            if rng.rand() < 0.65:
                k_worst = int(top_idx[rng.randint(min(5, len(top_idx)))])
                off = k_worst - (n - 1)
                if off > 0:
                    cands = np.array([i for i in ones if i + off < n and h[i + off] == 0])
                    if len(cands) > 0:
                        i = int(cands[rng.randint(len(cands))])
                        j = int(zeros[rng.randint(len(zeros))])
                    else:
                        i = int(ones[rng.randint(len(ones))])
                        j = int(zeros[rng.randint(len(zeros))])
                elif off < 0:
                    m = -off
                    cands = np.array([i for i in ones if i >= m and h[i - m] == 0])
                    if len(cands) > 0:
                        i = int(cands[rng.randint(len(cands))])
                        j = int(zeros[rng.randint(len(zeros))])
                    else:
                        i = int(ones[rng.randint(len(ones))])
                        j = int(zeros[rng.randint(len(zeros))])
                else:
                    i = int(ones[rng.randint(len(ones))])
                    j = int(zeros[rng.randint(len(zeros))])
            else:
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])

            if i == j:
                continue

            # Compute delta at top shifts
            dcorr = swap_delta(h, n, i, j, top_idx)
            new_vals = corr[top_idx] + dcorr
            new_max_top = float(np.max(new_vals))

            # Check if this reduces the max
            improvement = cur_max_val - new_max_top
            if improvement > best_improvement + 1e-12:
                # Verify by computing full corr
                h[i] = 0.0
                h[j] = 1.0
                full_corr = compute_corr(h)
                full_max = float(np.max(full_corr))
                h[i] = 1.0
                h[j] = 0.0
                real_improvement = cur_max_val - full_max
                if real_improvement > best_improvement:
                    best_improvement = real_improvement
                    best_swap = (i, j)
                    best_new_corr = full_corr

        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            corr = best_new_corr
            cur_max = float(np.max(corr))
            c5_new = cur_max * dx
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                # Random perturbation to escape
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                h[i] = 0.0
                h[j] = 1.0
                corr = compute_corr(h)
                cur_max = float(np.max(corr))
                stall = 0
                max_stall = int(max_stall * 1.3)

    return best_h, best_c5


def single_coord_descent(h, n, time_limit=60, rng=None):
    """Optimize one coordinate at a time, adjusting to maintain sum constraint."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    start = time.time()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    n_corr = 2 * n - 1

    # Precompute slopes for each coordinate
    while time.time() - start < time_limit and remaining() > 10:
        improved = False
        order = rng.permutation(n)
        for idx in range(n):
            i = int(order[idx])
            # Weights for adjusting h[i]: need to maintain sum
            # Change h[i] by delta, distribute -delta to a partner
            # Partner: coordinate with most "room"
            if h[i] > 0.5:
                # Try decreasing h[i]
                partner = np.argmin(h)
            else:
                partner = np.argmax(h)

            if i == partner:
                partner = (i + 1) % n

            s = h[i] + h[partner]
            if h[i] > 0.5:
                lo_t, hi_t = max(0.0, s - 1.0), h[i]
            else:
                lo_t, hi_t = h[i], min(1.0, s)

            if hi_t - lo_t < 1e-12:
                continue

            # Binary search for optimal t
            sl_i = np.zeros(n_corr)
            sl_p = np.zeros(n_corr)
            for k_idx in range(n_corr):
                d = k_idx - (n - 1)
                si = 0.0
                sp = 0.0
                if d >= 0:
                    if d <= i < n:
                        si += (1.0 - h[i - d])
                    if 0 <= i <= n - 1 - d:
                        si -= h[i + d]
                    if d <= partner < n:
                        sp += (1.0 - h[partner - d])
                    if 0 <= partner <= n - 1 - d:
                        sp -= h[partner + d]
                else:
                    m = -d
                    if 0 <= i <= n - 1 - m:
                        si += (1.0 - h[i + m])
                    if i >= m:
                        si -= h[i - m]
                    if 0 <= partner <= n - 1 - m:
                        sp += (1.0 - h[partner + m])
                    if partner >= m:
                        sp -= h[partner - m]
                sl_i[k_idx] = si
                sl_p[k_idx] = sp

            diff_sl = sl_i - sl_p
            intercepts = corr - h[i] * diff_sl

            # Binary search for t that minimizes max(intercepts + t*diff_sl)
            lo, hi = lo_t, hi_t
            for _ in range(40):
                mid = (lo + hi) / 2.0
                vals = diff_sl * mid + intercepts
                max_v = vals.max()
                active = vals >= max_v - 1e-12
                if len(active) == 0:
                    break
                s_max = diff_sl[active].max()
                s_min = diff_sl[active].min()
                if s_max < -1e-13:
                    lo = mid
                elif s_min > 1e-13:
                    hi = mid
                else:
                    break
            t_opt = (lo + hi) / 2.0

            if abs(t_opt - h[i]) > 1e-10:
                new_h = h.copy()
                new_h[i] = t_opt
                new_h[partner] = s - t_opt
                new_h = np.clip(new_h, 0.0, 1.0)
                new_h = project_feasible(new_h, n)
                new_c5 = compute_c5(new_h)
                if new_c5 < best_c5:
                    h = new_h
                    corr = compute_corr(h)
                    best_c5 = new_c5
                    improved = True

        if not improved:
            break

    return h, best_c5


def multi_shift_equalize(h, n, time_limit=60, top_k=5):
    """Equalize top-K shifts via direction finding."""
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    corr = compute_corr(h)

    while time.time() - start < time_limit and remaining() > 5:
        top = np.argsort(-corr)[:top_k]
        c_max = corr[top[0]]
        c_min = corr[top[-1]]
        if c_max - c_min < 1e-10:
            break

        # Direction: reduce contributions to max shifts, help min shifts
        g = np.zeros(n)
        for kk in top:
            gk = corr_grad_per_shift(h, n, int(kk))
            if kk == top[0]:
                g -= gk
            else:
                g -= 0.3 * gk

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g /= gnorm

        improved = False
        for step in [0.001, 0.002, 0.005, 0.01, 0.02, 0.0005, 0.0002, 0.03]:
            h_trial = h - step * g
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-13:
                h = h_trial
                best_c5 = c_new
                corr = compute_corr(h)
                improved = True
                break

        if not improved:
            break

    return h, best_c5


def perturb_and_refine(h, n, rng, budget_remaining):
    """Perturb solution randomly, then refine. Returns best found."""
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for _ in range(3):
        if remaining() < 20:
            break
        # Random perturbation
        h_pert = h.copy()
        mask = rng.rand(n) < 0.1
        h_pert[mask] += rng.uniform(-0.3, 0.3, np.sum(mask))
        h_pert = project_feasible(h_pert, n)

        # Quick SLSQP
        h_opt = slsqp_smooth(h_pert, n, 2000, maxiter=100)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 312
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Gentle SLSQP to start from parent (careful not to worsen)
    for beta in [200, 500, 1000, 3000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        # Don't accept worse solutions

    # Phase 2: Directed binary search (main workhorse)
    if remaining() > 30:
        h_bin = (best_h > 0.5).astype(float)
        t_lim = min(180, remaining() - 20)
        h_bl, c5_bl = binary_directed_search(h_bin, n, time_limit=t_lim, rng=rng, top_k_shifts=15)
        if c5_bl < best_c5 - 1e-12:
            best_c5 = c5_bl
            best_h = h_bl.copy()
            h = h_bl.copy()

    # Phase 3: SLSQP on the binary solution
    if remaining() > 15:
        h_cont = h.copy()
        for beta in [3000, 8000, 20000]:
            if remaining() < 8:
                break
            h_new = slsqp_smooth(h_cont, n, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
                h_cont = h.copy()

    # Phase 4: Single coordinate descent
    if remaining() > 20:
        t_lim = min(120, remaining() - 15)
        h_sc, c5_sc = single_coord_descent(h, n, time_limit=t_lim, rng=rng)
        if c5_sc < best_c5 - 1e-12:
            best_c5 = c5_sc
            best_h = h_sc.copy()
            h = h_sc.copy()

    # Phase 5: Multi-shift equalization
    if remaining() > 15:
        t_lim = min(60, remaining() - 10)
        h_eq, c5_eq = multi_shift_equalize(h, n, time_limit=t_lim, top_k=5)
        if c5_eq < best_c5 - 1e-12:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 6: More binary search (second round)
    if remaining() > 30:
        h_bin = (best_h > 0.5).astype(float)
        t_lim = min(120, remaining() - 20)
        h_bl, c5_bl = binary_directed_search(h_bin, n, time_limit=t_lim, rng=rng, top_k_shifts=20)
        if c5_bl < best_c5 - 1e-12:
            best_c5 = c5_bl
            best_h = h_bl.copy()
            h = h_bl.copy()

    # Phase 7: SLSQP polish
    if remaining() > 15:
        for beta in [5000, 15000, 40000]:
            if remaining() < 6:
                break
            h_new = slsqp_smooth(h, n, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 8: Try nearby n values
    n_candidates = [n + 2, n - 2, n + 4, n - 4, n + 6, n - 6, n + 8]
    n_candidates = [nn for nn in n_candidates if nn >= 200 and nn <= 600 and nn % 2 == 0 and nn != n]
    # Deduplicate
    seen = set()
    n_candidates = [x for x in n_candidates if x not in seen and not seen.add(x)]

    for n_t in n_candidates:
        if remaining() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)

        # Quick SLSQP
        h_opt = h_feas
        for beta in [1000, 5000]:
            if remaining() < 8:
                break
            h_opt = slsqp_smooth(h_opt, n_t, beta, maxiter=150)

        # Binary search
        h_bin = (h_opt > 0.5).astype(float)
        h_b, c5_b = binary_directed_search(h_bin, n_t, time_limit=min(30, remaining() - 15), rng=rng, top_k_shifts=10)
        c5_opt = min(compute_c5(h_b), c5_b)

        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_b.copy()
            best_n = n_t
            n = n_t
            h = h_b.copy()

    # Phase 9: Final intensive refinement
    n = best_n
    h = best_h.copy()

    if remaining() > 20:
        h_bin = (h > 0.5).astype(float)
        t_lim = min(80, remaining() - 20)
        h_bl, c5_bl = binary_directed_search(h_bin, n, time_limit=t_lim, rng=rng, top_k_shifts=25)
        if c5_bl < best_c5 - 1e-12:
            best_c5 = c5_bl
            best_h = h_bl.copy()
            h = h_bl.copy()

    # Final SLSQP
    if remaining() > 10:
        h_new = slsqp_smooth(h, n, 30000, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final micro binary search
    if remaining() > 15:
        h_bin = (best_h > 0.5).astype(float)
        t_lim = min(30, remaining() - 8)
        h_bl, c5_bl = binary_directed_search(h_bin, best_n, time_limit=t_lim, rng=rng, top_k_shifts=30)
        if c5_bl < best_c5 - 1e-12:
            best_c5 = c5_bl
            best_h = h_bl.copy()

    # Validate and finalize
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
