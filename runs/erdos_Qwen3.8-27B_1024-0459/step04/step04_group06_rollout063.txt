import numpy as np
from scipy.optimize import minimize
import time

_G_START = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            step = diff / cnt
            h[mask] += step
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(h)[::-1]
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0 and h[i] < 1.0:
                amt = min(diff, 1.0 - h[i])
                h[i] += amt
                diff -= amt
                if abs(diff) < 1e-13:
                    break
            elif diff < 0 and h[i] > 0.0:
                amt = max(diff, -h[i])
                h[i] += amt
                diff -= amt
                if abs(diff) < 1e-13:
                    break
    return h


def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 3.0)


def softmax_obj_grad(h, n, beta, mu):
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, 'full')
    max_corr = float(np.max(corr))
    exp_bc = np.exp(beta * (corr - max_corr))
    sum_exp = float(np.sum(exp_bc))
    F_corr = (dx / beta) * (beta * max_corr + np.log(sum_exp))
    sm = exp_bc / sum_exp
    grad_sum = np.zeros(n)
    for k in range(2 * n - 1):
        w = sm[k]
        if w < 1e-30:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad_sum[:n - s] += w * omh[s:n]
            grad_sum[s:n] -= w * h[:n - s]
        else:
            t = -s
            grad_sum[t:n] += w * omh[0:n - t]
            grad_sum[0:n - t] -= w * h[t:n]
    grad = dx * grad_sum
    s_val = np.sum(h) - n / 2.0
    F = F_corr + mu * s_val * s_val
    grad = grad + 2.0 * mu * s_val
    return F, grad


def lbfgs_step(h, n, beta, mu, maxiter):
    def obj_func(x):
        val, _ = softmax_obj_grad(x, n, beta, mu)
        return val
    def grad_func(x):
        _, g = softmax_obj_grad(x, n, beta, mu)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h, jac=grad_func, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15,
                                'gtol': 1e-12, 'maxls': 30, 'maxcor': 30,
                                'iprint': 0})
        return res.x
    except Exception:
        return h.copy()


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def power_sum_obj_grad(h, n, p, mu):
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, 'full')
    corr_c = np.maximum(corr, 1e-30)
    max_c = float(np.max(corr_c))
    u = corr_c / max_c
    sum_up = np.sum(u ** p)
    F_corr = dx * max_c * sum_up ** (1.0 / p)
    weights = (u ** (p - 1)) / sum_up
    grad_sum = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-30:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad_sum[:n - s] += w * omh[s:n]
            grad_sum[s:n] -= w * h[:n - s]
        else:
            t = -s
            grad_sum[t:n] += w * omh[0:n - t]
            grad_sum[0:n - t] -= w * h[t:n]
    grad = dx * max_c * (sum_up ** (1.0 / p - 1.0)) * grad_sum
    s_val = np.sum(h) - n / 2.0
    F = F_corr + mu * s_val * s_val
    grad = grad + 2.0 * mu * s_val
    return F, grad

    # Phase 8: Final high-beta + power-sum polish
    print("\n--- Phase 8: Final polish ---")
    n = best_n
    for beta in [20000, 50000, 100000]:
        if remaining_time() < 8.0:
            break
        h_opt = lbfgs_step(best_h, n, beta, 200.0, maxiter=400)
        h_opt = project_feasible(h_opt, n)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            print("  beta={}: C5={:.10f}".format(beta, best_c5))
    for p in [50, 100, 200]:
        if remaining_time() < 8.0:
            break
        h_opt = power_sum_lbfgs(best_h, n, p, 200.0, maxiter=300)
        h_opt = project_feasible(h_opt, n)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            print("  p={}: C5={:.10f}".format(p, best_c5))

    # Phase 9: Coordinate descent
    if remaining_time() > 12.0:
        print("\n--- Phase 9: Coordinate descent ---")
        best_h, best_c5 = coordinate_descent(best_h, best_n, rng, max_sweeps=10)
        print("  C5={:.10f}".format(best_c5))

    # Phase 10: Tiny final adjustments
    if remaining_time() > 8.0:
        for beta in [3000, 15000]:
            if remaining_time() < 5.0:
                break
            h_opt = lbfgs_step(best_h, best_n, beta, 500.0, maxiter=200)
            h_opt = project_feasible(h_opt, best_n)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5 - 1e-14:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        if remaining_time() > 5.0:
            best_h, best_c5 = coordinate_descent(best_h, best_n, rng, max_sweeps=4)

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
