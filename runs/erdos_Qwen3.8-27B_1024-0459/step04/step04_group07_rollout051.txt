import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

Wait, but I need different constraints for different k. With the lambda-free approach, I'd need to set a global _G_ACTIVE_K before each constraint evaluation. But SLSQP evaluates all constraints in sequence, so I can't easily switch.

Alternative: define a single constraint function that checks ALL active lags at once, returning a vector. SLSQP supports vector-valued constraints.

Actually, looking at scipy docs: for SLSQP, each constraint dict can have 'fun' return a scalar or array. If it returns an array, each element must be ≥ 0 for 'ineq'.

So I can have ONE inequality constraint that returns a vector:
[ t - corr[k1]*dx, t - corr[k2]*dx, ..., t - corr[km]*dx ]

And its Jacobian is a (m × (n+1)) matrix.

import numpy as np
from scipy.optimize import minimize
import time

# Global state
_G_NV = 0
_G_BETA = 1.0
_G_TGT = 0.0
_G_START = 0.0
_G_BUDGET = 1300.0
_G_DX = 1.0
_G_PEN = 0.0
_G_ACTIVE_LAGS = None

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def grad_corr_k(h, k):
    n = len(h)
    offset = k - (n - 1)
    g = np.zeros(n)
    j_start = max(0, offset)
    j_end = min(n, offset + n)
    if j_start < j_end:
        g[j_start:j_end] += (1.0 - h[j_start - offset: j_end - offset])
    j_start2 = max(0, -offset)
    j_end2 = min(n, n - offset)
    if j_start2 < j_end2:
        g[j_start2:j_end2] -= h[j_start2 + offset: j_end2 + offset]
    return g

def obj_lse(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = corr - m
    e = np.exp(np.clip(_G_BETA * s, -500, 500))
    val = (m + np.log(np.sum(e)) / _G_BETA) * _G_DX
    if _G_PEN > 0:
        val += _G_PEN * (np.sum(x) - _G_TGT) ** 2
    return float(val)

def grad_lse(x):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = corr - m
    p = np.exp(np.clip(_G_BETA * s, -500, 500))
    psum = np.sum(p)
    p = p / psum
    offsets = np.arange(2 * n - 1) - (n - 1)
    js = np.arange(n)
    idx1 = js[:, None] - offsets[None, :]
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = js[:, None] + offsets[None, :]
    valid2 = (idx2 >= 0) & (idx2 < n)
    x1 = x[np.clip(idx1, 0, n - 1)]
    x2 = x[np.clip(idx2, 0, n - 1)]
    term1 = (1.0 - x1) * valid1
    term2 = x2 * valid2
    grad = np.sum((term1 - term2) * p[None, :], axis=1)
    grad *= _G_DX / _G_BETA
    if _G_PEN > 0:
        grad += 2.0 * _G_PEN * (np.sum(x) - _G_TGT)
    return grad

def cons_eq_fun(x):
    return float(np.sum(x[:_G_NV]) - _G_TGT)

def cons_eq_jac(x):
    g = np.zeros(_G_NV)
    g[:] = 1.0
    return g

def cons_eq_fun_t(x):
    return float(np.sum(x[:_G_NV]) - _G_TGT)

def cons_eq_jac_t(x):
    g = np.zeros(_G_NV + 1)
    g[:_G_NV] = 1.0
    return g

def act_obj(x):
    return float(x[-1])

def act_obj_grad(x):
    g = np.zeros(_G_NV + 1)
    g[-1] = 1.0
    return g

def act_ineq_fun(x):
    h = x[:_G_NV]
    c = np.correlate(h, 1.0 - h, mode='full')
    return x[-1] - c[_G_ACTIVE_LAGS] * _G_DX

def act_ineq_jac(x):
    h = x[:_G_NV]
    m = len(_G_ACTIVE_LAGS)
    J = np.zeros((m, _G_NV + 1))
    for idx, k in enumerate(_G_ACTIVE_LAGS):
        J[idx, :_G_NV] = -grad_corr_k(h, k) * _G_DX
        J[idx, _G_NV] = 1.0
    return J

def slsqp_lse(h0, n, beta, maxiter):
    global _G_NV, _G_BETA, _G_TGT, _G_DX, _G_PEN
    _G_NV = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    _G_BETA = float(beta)
    _G_PEN = 0.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    try:
        res = minimize(obj_lse, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=grad_lse,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def slsqp_active(h0, n, active_lags, maxiter):
    global _G_NV, _G_DX, _G_TGT, _G_ACTIVE_LAGS, _G_PEN
    _G_NV = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    _G_PEN = 0.0
    _G_ACTIVE_LAGS = np.array(active_lags, dtype=int)
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    t0 = float(np.max(corr[active_lags])) * (2.0 / n) + 0.005
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [
        {'type': 'eq', 'fun': cons_eq_fun_t, 'jac': cons_eq_jac_t},
        {'type': 'ineq', 'fun': act_ineq_fun, 'jac': act_ineq_jac}
    ]
    try:
        res = minimize(act_obj, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=act_obj_grad,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h = project_feasible(res.x[:n], n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def lbfgsb_lse(h0, n, beta, maxiter, pen):
    global _G_NV, _G_BETA, _G_TGT, _G_DX, _G_PEN
    _G_NV = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    _G_BETA = float(beta)
    _G_PEN = float(pen)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_lse, h0, method='L-BFGS-B', bounds=bounds,
                       jac=grad_lse,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def active_set_iteration(h0, n, max_rounds, tlimit):
    global _G_NV, _G_DX, _G_TGT, _G_PEN
    c5 = compute_c5(h0)
    h_best = h0.copy()
    for rnd in range(max_rounds):
        if tlimit() < 5.0:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode='full')
        n_corr = len(corr)
        # Find top lags
        top_k = min(5, n_corr)
        active = np.argsort(corr)[::-1][:top_k]
        # Also include lags within 1% of max
        thresh = corr[active[0]] * 0.995
        extra = np.where(corr >= thresh)[0]
        active = np.unique(np.concatenate([active, extra]))
        active = active[:10]  # cap at 10 constraints
        h_new = slsqp_active(h_best, n, active, maxiter=250)
        c_new = compute_c5(h_new)
        if c_new < c5 - 1e-12:
            c5 = c_new
            h_best = h_new
        if tlimit() < 10.0:
            break
    return h_best, c5

def subgradient_descent(h0, n, max_iter, tlimit, step_init=0.01):
    c5 = compute_c5(h0)
    h_best = h0.copy()
    step = step_init
    for it in range(max_iter):
        if tlimit() < 3.0:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode='full')
        k_worst = int(np.argmax(corr))
        g = grad_corr_k(h_best, k_worst)
        h_trial = h_best - step * g
        h_trial = project_feasible(h_trial, n)
        c_trial = compute_c5(h_trial)
        if c_trial < c5 - 1e-12:
            c5 = c_trial
            h_best = h_trial
            step *= 1.05
        else:
            step *= 0.7
        if step < 1e-8:
            break
    return h_best, c5

def coord_descent(h0, n, max_sweeps, rng, tlimit):
    c5 = compute_c5(h0)
    h_best = h0.copy()
    deltas = [0.0, 1.0, -1.0, 0.1, -0.1, 0.05, -0.05, 0.2, -0.2, 0.02, -0.02]
    for sweep in range(max_sweeps):
        if tlimit() < 2.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if tlimit() < 1.0:
                break
            v0 = h_best[i]
            for d in deltas:
                nv = v0 + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                if abs(nv - v0) < 1e-12:
                    continue
                trial = h_best.copy()
                trial[i] = nv
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 1e-11:
                    c5 = c
                    h_best = trial
                    improved = True
                    break
        if not improved and sweep >= 3:
            break
    return h_best, c5

def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1')
        h[i] = 1.0 if c % 2 == 0 else 0.0
    return project_feasible(h, n)

def make_rand_bin(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return project_feasible(h, n)
