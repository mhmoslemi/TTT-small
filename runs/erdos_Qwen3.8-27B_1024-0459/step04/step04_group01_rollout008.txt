def _fista_grad(h, n, beta):

    corr = compute_corr(h)
    m = np.max(corr)

    s = np.sum(ex)
    obj = (m + np.log(s) / beta) * dx

    # Phase 5: Pair coordinate descent (main workhorse for fine tuning)
    if remaining_time() > 20:
        h_pc, c5_pc = coord_descent_optimized(h, n, max_passes=400,
                                               time_limit=min(60, remaining_time() - 15),
                                               rng=rng)
        if c5_pc < best_c5:
            best_c5 = c5_pc
            best_h = h_pc.copy()
            h = h_pc.copy()

    # Phase 6: SLSQP polish after coord descent
    for beta in [20000, 50000, 100000]:
        if remaining_time() < 5:
            break
        h_new = slsqp_lse(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 7: Another round of active set
    if remaining_time() > 12:
        corr = compute_corr(h)
        sorted_idx = np.argsort(-corr)
        c_max = corr[sorted_idx[0]]
        active_idx = [int(k) for k in sorted_idx[:5]]
        if len(active_idx) >= 2:
            h_as = slsqp_active_set(h, n, active_idx, maxiter=600)
            c5_as = compute_c5(h_as)
            if c5_as < best_c5:
                best_c5 = c5_as
                best_h = h_as.copy()
                h = h_as.copy()

    # Phase 8: Try nearby resolutions
    n_tries = []
    for dn in [-8, -4, 4, 8, 16, 24, 32, 48]:
        nt = best_n + dn
        if 100 <= nt <= 700 and nt % 2 == 0 and nt != best_n:
            n_tries.append(nt)
    # Also try round numbers
    for nt in [256, 288, 320, 336, 360, 384, 400, 448, 512, 576]:
        if 100 <= nt <= 700 and nt % 2 == 0 and nt != best_n and nt not in n_tries:
            n_tries.append(nt)
    n_tries = sorted(set(n_tries))

    for n_t in n_tries:
        if remaining_time() < 35:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        h_opt = h_feas
        for beta in [2000, 10000, 30000]:
            if remaining_time() < 8:
                break
            h_opt = slsqp_lse(h_opt, n_t, beta, maxiter=200)
        if remaining_time() > 12:
            h_opt, _ = coord_descent_optimized(h_opt, n_t, max_passes=150,
                                                time_limit=min(20, remaining_time() - 10),
                                                rng=rng)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 9: Final intensive refinement on best solution
    n = best_n
    h = best_h.copy()

    # Final SLSQP with very high beta
    for beta in [30000, 80000, 150000]:
        if remaining_time() < 4:
            break
        h_new = slsqp_lse(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final coord descent
    if remaining_time() > 12:
        h_pc, c5_pc = coord_descent_optimized(h, n, max_passes=300,
                                               time_limit=min(35, max(5, remaining_time() - 8)),
                                               rng=rng)
        if c5_pc < best_c5:
            best_c5 = c5_pc
            best_h = h_pc.copy()

    # Final active set
    if remaining_time() > 10:
        corr = compute_corr(best_h)
        sorted_idx = np.argsort(-corr)
        active_idx = [int(k) for k in sorted_idx[:4]]
        if len(active_idx) >= 2:
            h_as = slsqp_active_set(best_h, best_n, active_idx, maxiter=400)
            c5_as = compute_c5(h_as)
            if c5_as < best_c5:
                best_c5 = c5_as
                best_h = h_as.copy()

    # Final binary local search attempt
    if remaining_time() > 15:
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        if n_ones != best_n // 2:
            if n_ones > best_n // 2:
                idx = np.where(h_bin == 1.0)[0]
                h_bin[idx[rng.choice(len(idx), n_ones - best_n // 2, replace=False)]] = 0.0
            else:
                idx = np.where(h_bin == 0.0)[0]
                h_bin[idx[rng.choice(len(idx), best_n // 2 - n_ones, replace=False)]] = 1.0
        h_b, c5_b = binary_local_search(h_bin, best_n,
                                         time_limit=min(20, max(3, remaining_time() - 6)),
                                         rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Multi-start perturbation at the end
    if remaining_time() > 25:
        h_ms, c5_ms = multi_start_perturb(best_h, best_n, rng,
                                           n_starts=3,
                                           time_per_start=min(15, (remaining_time() - 10) // 3))
        if c5_ms < best_c5:
            best_c5 = c5_ms
            best_h = h_ms.copy()

        # SLSQP polish on multi-start result
        if remaining_time() > 8:
            h_pol = slsqp_lse(best_h, best_n, 50000, maxiter=400)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Final validation and polish
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-12:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
