def nesterov_solve(h0, n, beta, lr_init, maxiter, tlimit, momentum=0.9):
    h = h0.copy().astype(float)
    v = np.zeros(n)
    dx = 2.0 / n
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    for it in range(maxiter):
        if tlimit() < 2.0:
            break
        
        # Nesterov lookahead
        h_prev = h.copy()
        h = h + momentum * v  # Wait, standard Nesterov:
        # h_{k+1} = h_k + (momentum - 1)*(h_k - h_{k-1}) - lr * grad(f, h_k)
        # Or equivalently with velocity: v_{k+1} = momentum * v_k - lr * grad; h_{k+1} = h_k + v_{k+1}
        # Or Nesterov: h_look = h + momentum * v; grad at h_look; v_new = momentum*v - lr*grad; h_new = h + v_new
        
        h_look = h + momentum * v
        h_look = project_feasible(h_look, n)
        
        grad = grad_objective(h_look, beta, dx)
        v = momentum * v - lr_init * grad
        h = h + v
        h = project_feasible(h, n)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def softmax_weights(corr, beta):
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    s = np.sum(e)
    if s < 1e-300:
        return np.full(len(corr), 1.0 / len(corr))
    return e / s

    grad[k] = dx * (S[k] - 2*T[k])
    where S[k]=sum(p[k:k+n]), T[k]=dot(p[k:k+n], h)
    """

        p = softmax_weights(corr, beta)

    p_ext[:2 * n - 1] = p

    h = h0.copy().astype(float)
    v = np.zeros(n)
    dx = 2.0 / n
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for it in range(maxiter):
        if tlimit() < 2.0:
            break
        # Nesterov lookahead
        h_look = h + momentum * v
        h_look = project_feasible(h_look, n)
        corr = compute_corr(h_look)
        p = softmax_weights(corr, beta)
        grad = dx * (np.zeros(n) - 2.0 * np.zeros(n))
        # Compute gradient
        p_ext = np.zeros(2 * n)
        p_ext[:2 * n - 1] = p
        prefix = np.zeros(2 * n + 1)
        np.cumsum(p_ext, out=prefix[1:])
        S = prefix[n:n + n] - prefix[0:n]
        T = np.zeros(n)
        for k in range(n):
            T[k] = np.dot(p_ext[k:k + n], h_look)
        grad = dx * (S - 2.0 * T)
        # Decay learning rate
        lr = lr_init / (1.0 + 0.001 * it)
        v = momentum * v - lr * grad
        h = h + v
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def binary_2flip(h, n, tlimit, max_moves=200000):
    """Local search: swap pairs (increase one, decrease another) to reduce C5."""
    h = h.copy().astype(float)
    c5 = compute_c5(h)
    dx = 2.0 / n
    moves = 0
    Stagnant = 0
    for sweep in range(200):
        if tlimit() < 2.0 or moves > max_moves:
            break
        improved = False
        # Sample random pairs
        for _ in range(2000):
            if tlimit() < 2.0:
                break
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            if h[i] < 0.01 or h[j] > 0.99:
                continue
            # Try increasing h[i] by delta, decreasing h[j] by delta
            delta = min(0.25, 1.0 - h[i], h[j])
            trial = h.copy()
            trial[i] += delta
            trial[j] -= delta
            trial = project_feasible(trial, n)
            c = compute_c5(trial)
            if c < c5 - 1e-10:
                h = trial
                c5 = c
                improved = True
                Stagnant = 0
            moves += 1
        # Also try decreasing/increasing (reverse)
        for _ in range(2000):
            if tlimit() < 2.0:
                break
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            if h[i] > 0.99 or h[j] < 0.01:
                continue
            delta = min(0.25, h[i], 1.0 - h[j])
            trial = h.copy()
            trial[i] -= delta
            trial[j] += delta
            trial = project_feasible(trial, n)
            c = compute_c5(trial)
            if c < c5 - 1e-10:
                h = trial
                c5 = c
                improved = True
                Stagnant = 0
            moves += 1
        if not improved:
            Stagnant += 1
            if Stagnant >= 3:
                break
    return h, c5

def pnorm_solve(h0, n, p_power, lr, maxiter, tlimit, momentum=0.85):
    """Optimize using p-norm of correlations as smooth max surrogate."""
    h = h0.copy().astype(float)
    v = np.zeros(n)
    dx = 2.0 / n
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for it in range(maxiter):
        if tlimit() < 2.0:
            break
        h_look = h + momentum * v
        h_look = project_feasible(h_look, n)
        corr = compute_corr(h_look)
        # p-norm approximate gradient
        # f = (sum corr^p)^(1/p), df/dcorr_j = corr_j^(p-1)/f^p * ... 
        # Simplify: gradient of corr_j^(p-1) normalized
        cp = np.abs(corr) ** (p_power - 1)
        cp = cp / np.sum(cp)  # normalize to act like softmax
        p = cp
        p_ext = np.zeros(2 * n)
        p_ext[:2 * n - 1] = p
        prefix = np.zeros(2 * n + 1)
        np.cumsum(p_ext, out=prefix[1:])
        S = prefix[n:n + n] - prefix[0:n]
        T = np.zeros(n)
        for k in range(n):
            T[k] = np.dot(p_ext[k:k + n], h_look)
        grad = dx * (S - 2.0 * T)
        lr_t = lr / (1.0 + 0.005 * it)
        v = momentum * v - lr_t * grad
        h = h + v
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def slsqp_polish(h0, n, beta, maxiter):
    global _G_BETA_POL
    _G_BETA_POL = float(beta)
    dx = 2.0 / n
    def obj_fn(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        e = np.exp(_G_BETA_POL * (corr - m))
        val = (m + np.log(np.sum(e)) / _G_BETA_POL) * dx
        return val
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        e = np.exp(_G_BETA_POL * (corr - m))
        p = e / np.sum(e)
        p_ext = np.zeros(2 * len(x))
        p_ext[:2 * len(x) - 1] = p
        prefix = np.zeros(2 * len(x) + 1)
        np.cumsum(p_ext, out=prefix[1:])
        n_loc = len(x)
        S = prefix[n_loc:n_loc + n_loc] - prefix[0:n_loc]
        T = np.zeros(n_loc)
        for k in range(n_loc):
            T[k] = np.dot(p_ext[k:k + n_loc], x)
        return dx * (S - 2.0 * T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'jac': lambda x: np.ones(n), 'fun': lambda x: np.sum(x) - n / 2.0}]
    try:
        res = minimize(obj_fn, h0, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

_G_BETA_POL = 1.0

def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c else 0.0
    return h

def make_rand_bin(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _G_BETA_POL
    _START = time.time()
    _BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    def tlimit():
        return max(0.0, _BUDGET - (time.time() - _START) - 5.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Initialize from parent or fallback
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = project_feasible(make_tm(parent_n), parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("No parent. TM: n={}, C5={:.8f}".format(parent_n, best_c5))

    n = best_n
    dx = 2.0 / n

    # Phase 1: Nesterov ACGM with beta continuation
    if tlimit() > 60:
        print("\n=== Phase 1: Nesterov ACGM (n={}) ===".format(n))
        h_cur = best_h.copy()
        c5_cur = best_c5
        beta_schedule = [20, 50, 100, 200, 400, 800, 1500, 3000, 6000, 12000]
        for bv in beta_schedule:
            if tlimit() < 10:
                break
            lr_init = 0.0005 * n / bv  # scale with problem
            maxit = min(3000, int(tlimit() * 50))
            h_new, c5_new = nesterov_solve(h_cur, n, bv, lr_init, maxit, tlimit, momentum=0.92)
            if c5_new < c5_cur - 1e-12:
                c5_cur = c5_new
                h_cur = h_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n
        print("  After Nesterov: C5={:.8f}".format(best_c5))

    # Phase 2: SLSQP polish at high beta
    if tlimit() > 40:
        print("\n=== Phase 2: SLSQP polish ===")
        for bv in [2000, 8000, 20000]:
            if tlimit() < 10:
                break
            h_new = slsqp_polish(best_h, best_n, bv, maxiter=400)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-13:
                best_c5 = c_new
                best_h = h_new
                print("  SLSQP beta={}: C5={:.8f}".format(bv, best_c5))
        n = best_n

    # Phase 3: Binary 2-flip local search
    if tlimit() > 30:
        print("\n=== Phase 3: 2-flip local search ===")
        h_cur = best_h.copy()
        h_cur, c5_cur = binary_2flip(h_cur, best_n, tlimit, max_moves=100000)
        if c5_cur < best_c5 - 1e-10:
            best_c5 = c5_cur
            best_h = h_cur
            print("  After 2-flip: C5={:.8f}".format(best_c5))

    # Phase 4: Multi-restart with perturbations
    if tlimit() > 80:
        print("\n=== Phase 4: Multi-restart ===")
        for restart in range(12):
            if tlimit() < 20:
                break
            h_pert = best_h.copy()
            n_pert = rng.randint(5, 25)
            idxs = rng.choice(best_n, size=n_pert, replace=False)
            for idx in idxs:
                h_pert[idx] = np.clip(h_pert[idx] + rng.randn() * 0.2, 0.0, 1.0)
            h_pert = project_feasible(h_pert, best_n)
            c5_p = compute_c5(h_pert)
            if c5_p < best_c5 * 1.02:
                # Nesterov from perturbed start
                for bv in [200, 1000, 5000]:
                    if tlimit() < 5:
                        break
                    lr_init = 0.0003 * best_n / bv
                    h_new, c5_new = nesterov_solve(h_pert, best_n, bv, lr_init, 1500, tlimit, momentum=0.9)
                    if c5_new < c5_p - 1e-12:
                        c5_p = c5_new
                        h_pert = h_new
                # SLSQP polish
                if tlimit() > 10 and c5_p < best_c5 + 0.001:
                    h_new = slsqp_polish(h_pert, best_n, 5000, maxiter=300)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_p:
                        c5_p = c5_new
                        h_pert = h_new
                if c5_p < best_c5 - 1e-10:
                    best_c5 = c5_p
                    best_h = h_pert.copy()
                    best_n = best_n
                    print("  Restart {} improved: C5={:.8f}".format(restart, best_c5))

    # Phase 5: p-norm optimization
    if tlimit() > 50:
        print("\n=== Phase 5: p-norm optimization ===")
        h_cur = best_h.copy()
        c5_cur = best_c5
        for p_pow in [8, 16, 32, 64, 128]:
            if tlimit() < 10:
                break
            lr = 0.0002 * best_n
            h_new, c5_new = pnorm_solve(h_cur, best_n, p_pow, lr, 2000, tlimit, momentum=0.85)
            if c5_new < c5_cur - 1e-12:
                c5_cur = c5_new
                h_cur = h_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
        print("  After p-norm: C5={:.8f}".format(best_c5))

    # Phase 6: Try higher resolution
    if tlimit() > 60:
        print("\n=== Phase 6: Higher resolution n=512 ===")
        n_hi = 512
        h_hi = project_feasible(interp_h(best_h, best_n, n_hi), n_hi)
        c5_hi = compute_c5(h_hi)
        # Also try TM at 512
        h_tm512 = project_feasible(make_tm(n_hi), n_hi)
        c5_tm = compute_c5(h_tm512)
        print("  Interp: C5={:.8f}, TM: C5={:.8f}".format(c5_hi, c5_tm))
        h_start = h_hi if c5_hi < c5_tm else h_tm512
        c5_start = min(c5_hi, c5_tm)
        for bv in [100, 500, 2000, 8000]:
            if tlimit() < 10:
                break
            lr_init = 0.0004 * n_hi / bv
            h_new, c5_new = nesterov_solve(h_start, n_hi, bv, lr_init, 2000, tlimit, momentum=0.92)
            if c5_new < c5_start - 1e-12:
                c5_start = c5_new
                h_start = h_new
        if tlimit() > 15:
            h_new = slsqp_polish(h_start, n_hi, 5000, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < c5_start:
                c5_start = c5_new
                h_start = h_new
        if c5_start < best_c5 - 1e-10:
            best_c5 = c5_start
            best_h = h_start.copy()
            best_n = n_hi
            print("  n=512 improved: C5={:.8f}".format(best_c5))
        else:
            print("  n=512 best: C5={:.8f} (vs {:.8f})".format(c5_start, best_c5))

    # Phase 7: Final combined refinement
    if tlimit() > 30:
        print("\n=== Phase 7: Final refinement ===")
        n = best_n
        # Nesterov at very high beta
        lr_init = 0.0002 * n / 10000.0
        h_new, c5_new = nesterov_solve(best_h, n, 10000, lr_init, 3000, tlimit, momentum=0.95)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new
        # SLSQP final
        if tlimit() > 10:
            h_new = slsqp_polish(best_h, n, 20000, maxiter=500)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-13:
                best_c5 = c_new
                best_h = h_new
        # 2-flip final
        if tlimit() > 10:
            h_new, c5_new = binary_2flip(best_h, best_n, tlimit, max_moves=50000)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new
        print("  Final: C5={:.8f}".format(best_c5))

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nRESULT: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
