def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)

                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def thue_morse_init(n):
    """Generate Thue-Morse based binary initialization."""
    h = np.zeros(n)
    for i in range(n):
        # Count bits
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return project_feasible(h, n)

def subgradient_step(h, n, step_size, n_active=8):
    """One subgradient descent step on the max constraint."""
    corr = compute_corr(h)
    max_corr = np.max(corr)
    threshold = max_corr - 0.35
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        idx = active[np.argsort(corr[active])[::-1][:n_active]]
        active = idx

    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]

    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return h
    grad /= gnorm
    h_new = h - step_size * grad
    return project_feasible(h_new, n)

        h = best_h.copy()
        h, c5 = fast_coord_descent(h, n, max_sweeps=100)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Multiple perturbations + coordinate descent
    if remaining_time() > 60:
        n_perturb = 8
        for p in range(n_perturb):
            if remaining_time() < 30:
                break
            h_pert = best_h.copy()
            # Random perturbation
            noise_scale = 0.15 / (1 + p * 0.3)
            mask = rng.random(n) < 0.3
            h_pert[mask] += rng.randn(np.sum(mask)) * noise_scale
            h_pert = project_feasible(h_pert, n)
            c5_pert = compute_c5(h_pert)

            # Coordinate descent from perturbed
            h_new, c5_new = fast_coord_descent(h_pert, n, max_sweeps=80)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

            # Quick SLSQP on the best perturbation result
            if remaining_time() > 40:
                h_slq = slsqp_refine(h_new, n, 5000, 300)
                c5_slq = compute_c5(h_slq)
                if c5_slq < best_c5:
                    best_c5 = c5_slq
                    best_h = h_slq.copy()

    # Phase 5: Try Thue-Morse and other structured inits
    if remaining_time() > 60:
        h_tm = thue_morse_init(n)
        c5_tm = compute_c5(h_tm)
        if c5_tm < best_c5 + 0.05:  # Only if not too far
            h_new, c5_new = fast_coord_descent(h_tm, n, max_sweeps=100)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                if remaining_time() > 30:
                    h_slq = slsqp_refine(h_new, n, 10000, 400)
                    c5_slq = compute_c5(h_slq)
                    if c5_slq < best_c5:
                        best_c5 = c5_slq
                        best_h = h_slq.copy()
                        h_new, c5_new = fast_coord_descent(h_new, n, max_sweeps=50)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_new.copy()

    # Phase 6: Try higher resolution
    if remaining_time() > 120:
        high_n = 2 * n  # 1280
        if high_n <= 2000:
            h_high = interp_to(best_h, n, high_n)
            h_high = project_feasible(h_high, high_n)
            c5_high = compute_c5(h_high)

            h_new, c5_new = fast_coord_descent(h_high, high_n, max_sweeps=60)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = high_n

            if remaining_time() > 60:
                h_slq = slsqp_refine(h_new, high_n, 3000, 300)
                c5_slq = compute_c5(h_slq)
                if c5_slq < best_c5:
                    best_c5 = c5_slq
                    best_h = h_slq.copy()
                    best_n = high_n
                    h_new, c5_new = fast_coord_descent(h_new, high_n, max_sweeps=40)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        best_n = high_n

    # Phase 7: Subgradient + coordinate descent alternating
    if remaining_time() > 40:
        n_cur = best_n
        h = best_h.copy()
        for it in range(500):
            if remaining_time() < 5:
                break
            step = 0.02 / (1.0 + it * 0.01)
            h_sg = subgradient_step(h, n_cur, step, n_active=6)
            c5_sg = compute_c5(h_sg)
            if c5_sg < best_c5 - 1e-15:
                best_c5 = c5_sg
                best_h = h_sg.copy()
                h = h_sg.copy()
            else:
                # Try coordinate descent from current
                if it % 50 == 25 and remaining_time() > 20:
                    h_new, c5_new = fast_coord_descent(h, n_cur, max_sweeps=20)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h = h_new.copy()
                    else:
                        break

    # Phase 8: Final SLSQP polish
    if remaining_time() > 10:
        n_cur = best_n
        h = best_h.copy()
        for beta in [10000, 50000, 200000]:
            if remaining_time() < 5:
                break
            h_new = slsqp_refine(h, n_cur, beta, 400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 9: Final coordinate descent
    if remaining_time() > 10:
        n_cur = best_n
        h_new, c5_new = fast_coord_descent(best_h, n_cur, max_sweeps=80)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 10: One more SLSQP
    if remaining_time() > 8:
        n_cur = best_n
        h_new = slsqp_refine(best_h, n_cur, 100000, 300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final projection and evaluation
    n_final = best_n
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, n_final)
