def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val

def corr_lag_grad(x, n, lag):
    """Gradient of R_lag w.r.t. x, where R_lag is the correlation at the given lag (0-indexed in full output, lag = m-(n-1))."""
    g = np.zeros(n)
    if lag >= 0:
        npairs = n - lag
        # x[t] * (1-x[t+lag]) for t=0..n-1-lag
        # d/dx[p]: from x[t] term (t=p): (1-x[p+lag]) for p in [0, n-1-lag]
        #          from (1-x[t+lag]) term (t+lag=p, t=p-lag): -x[p-lag] for p in [lag, n-1]
        g[:npairs] += (1.0 - x[lag:lag+npairs])
        g[lag:] -= x[:npairs]
    else:
        k = -lag
        npairs = n - k
        # x[t] * (1-x[t-k]) for t=k..n-1
        # Equivalently: x[t+k] * (1-x[t]) for t=0..n-1-k
        # d/dx[p]: from x[t+k] term (t+k=p, t=p-k): (1-x[p-k]) for p in [k, n-1]
        #          from (1-x[t]) term (t=p): -x[p+k] for p in [0, n-1-k]
        g[k:] += (1.0 - x[:npairs])
        g[:npairs] -= x[k:k+npairs]
    return g

def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    total = np.sum(ex)
    w = ex / total
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-14:
            lag = k - (n - 1)
            g += w[k] * corr_lag_grad(x, n, lag)
    return g * dx_val

def sum_con(x):
    return np.sum(x) - _TGT

def sum_con_jac(x):
    return np.ones(_N)

def slsqp_smooth(h, n, beta, maxiter=500):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

def topk_subgrad(h, n, max_iters=300, top_k=5, lr=0.3):

        if gn < 1e-14:

        d = -g / gn

        if not improved:

        if remaining() < 10:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:top_m]
        for k in top_idx:
            if remaining() < 5:
                break
            lag = int(k) - (n - 1)
            for it in range(iters_per_lag):
                if remaining() < 5:
                    break
                g = corr_lag_grad(h, n, lag) * dx_val
                gn = np.linalg.norm(g)
                if gn < 1e-14:
                    break
                d = -g / gn
                improved = False
                for step in [0.3, 0.15, 0.07, 0.5, 0.03]:
                    h_trial = h + step * d
                    h_trial = project_feasible(h_trial, n)
                    c_new = compute_c5(h_trial)
                    if c_new < best_c5 - 1e-12:
                        h = h_trial
                        best_c5 = c_new
                        best_h = h.copy()
                        improved = True
                        break
                if not improved:
                    break
    return best_h, best_c5

def perturb_and_refine(h, n, seed, rng, iters=3):
    """Add random perturbations and re-optimize."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(iters):
        if remaining() < 30:
            break
        noise_scale = 0.05
        perturbed = h + rng.randn(n) * noise_scale
        perturbed = project_feasible(perturbed, n)
        # Quick SLSQP
        h_new = slsqp_smooth(perturbed, n, 2000, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_c5 = float('inf')
    best_h = None
    best_n = 0

    # Load parent
    parent_h = None
    parent_n = 0
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_c5 = parent_c5
        best_h = parent_h.copy()
        best_n = parent_n
    except Exception:
        parent_n = 360
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n//2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_c5 = parent_c5
        best_h = parent_h.copy()
        best_n = parent_n

    # Phase 1: Binary SA at multiple resolutions
    # Track best binary solutions for continuous refinement
    binary_solutions = []  # list of (c5, h, n)
    
    # First, binarize the parent at its own resolution
    h_bin_parent = make_binary(parent_h, parent_n, rng)
    binary_solutions.append((compute_c5(h_bin_parent), h_bin_parent.copy(), parent_n))

    n_list = sorted(set([parent_n, 380, 400, 420, 440, 460, 480, 500]))
    n_list = [n for n in n_list if n % 2 == 0 and 200 <= n <= 600]
    
    # Allocate time for binary phase: ~60% of budget
    binary_time_budget = budget_s * 0.55
    n_per_n = max(1, len(n_list))
    t_per_n = binary_time_budget / n_per_n
    t_per_sa = t_per_n / 2.5  # ~2-3 SA runs per n
    
    for n_t in n_list:
        if remaining() < binary_time_budget * 0.1 + 20:
            break
        t_remaining_for_this = min(t_per_n, remaining() - 30)
        if t_remaining_for_this < 15:
            break
        
        # Interpolate parent to this n
        h_interp = interp_to(parent_h, parent_n, n_t)
        h_interp = project_feasible(h_interp, n_t)
        h_bin1 = make_binary(h_interp, n_t, rng)
        
        # Also try Thue-Morse
        h_bin2 = thue_morse_init(n_t, rng)
        
        # Run SA on both initializations
        t_sa_each = min(t_per_sa, t_remaining_for_this / 2.5)
        
        for init_h in [h_bin1, h_bin2]:
            if remaining() < 30 or t_sa_each < 5:
                break
            h_sa, c5_sa = sa_binary(init_h, n_t, time_limit=t_sa_each, seed=seed + n_t)
            binary_solutions.append((c5_sa, h_sa.copy(), n_t))
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_t
        
        # Additional SA restart from current best at this n if time allows
        if t_remaining_for_this > 2 * t_per_sa + 10 and remaining() > 30:
            h_best_at_n = interp_to(best_h, best_n, n_t)
            h_bin3 = make_binary(h_best_at_n, n_t, rng)
            t_extra = min(20, t_remaining_for_this - 2 * t_per_sa)
            if t_extra > 5:
                h_sa, c5_sa = sa_binary(h_bin3, n_t, time_limit=t_extra, seed=seed + 10000 + n_t)
                binary_solutions.append((c5_sa, h_sa.copy(), n_t))
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
                    best_n = n_t

    # Sort binary solutions by quality
    binary_solutions.sort(key=lambda x: x[0])
    
    # Phase 2: Continuous refinement on top binary solutions
    # Use ~35% of remaining budget
    cont_time = min(budget_s * 0.35, remaining() - 20)
    
    # Take top 3 distinct (n, solution) pairs
    used_ns = set()
    top_solutions = []
    for c5_b, h_b, n_b in binary_solutions:
        if n_b not in used_ns and len(top_solutions) < 3:
            top_solutions.append((h_b, n_b))
            used_ns.add(n_b)
    
    # Also ensure we include the current best
    if best_h is not None and best_n not in used_ns:
        top_solutions.append((best_h, best_n))
        used_ns.add(best_n)
    
    t_per_cont = max(30, cont_time / max(1, len(top_solutions)))
    
    for h_init, n_init in top_solutions:
        if remaining() < 40:
            break
        h_work = h_init.copy()
        h_work = project_feasible(h_work, n_init)
        c5_work = compute_c5(h_work)
        
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()
            best_n = n_init
        
        # SLSQP with increasing beta
        for beta in [300, 1000, 3000, 8000, 20000]:
            if remaining() < 40:
                break
            t_sub = min(t_per_cont / 5.0, remaining() - 40)
            if t_sub < 5:
                break
            h_new = slsqp_smooth(h_work, n_init, beta, maxiter=600)
            c5_new = compute_c5(h_new)
            if c5_new < c5_work:
                c5_work = c5_new
                h_work = h_new.copy()
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_work.copy()
                    best_n = n_init
        
        # Top-k subgradient
        if remaining() > 40:
            h_td, c5_td = topk_subgrad(h_work, n_init, max_iters=400, top_k=5, lr=0.4)
            if c5_td < c5_work:
                c5_work = c5_td
                h_work = h_td.copy()
                if c5_td < best_c5:
                    best_c5 = c5_td
                    best_h = h_work.copy()
                    best_n = n_init
        
        # Cycle lags
        if remaining() > 40:
            h_cyc, c5_cyc = cycle_lag_min(h_work, n_init, top_m=5, iters_per_lag=150)
            if c5_cyc < c5_work:
                c5_work = c5_cyc
                h_work = h_cyc.copy()
                if c5_cyc < best_c5:
                    best_c5 = c5_cyc
                    best_h = h_work.copy()
                    best_n = n_init
        
        # Perturb and refine
        if remaining() > 50:
            h_pert, c5_pert = perturb_and_refine(h_work, n_init, seed + 50000 + n_init, rng, iters=3)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                best_n = n_init
    
    # Phase 3: Final intensive refinement on the best solution
    if remaining() > 30:
        n = best_n
        h = best_h.copy()
        
        # More SLSQP
        for beta in [5000, 15000, 40000]:
            if remaining() < 15:
                break
            h_new = slsqp_smooth(h, n, beta, maxiter=800)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        
        # Top-k with more iterations
        if remaining() > 20:
            h_td, c5_td = topk_subgrad(h, n, max_iters=600, top_k=3, lr=0.3)
            if c5_td < best_c5:
                best_c5 = c5_td
                best_h = h_td.copy()
                h = h_td.copy()
        
        # Cycle
        if remaining() > 20:
            h_cyc, c5_cyc = cycle_lag_min(h, n, top_m=7, iters_per_lag=200)
            if c5_cyc < best_c5:
                best_c5 = c5_cyc
                best_h = h_cyc.copy()
    
    # Phase 4: Final SA push from best continuous solution
    if remaining() > 40:
        h_bin_final = make_binary(best_h, best_n, rng)
        t_sa = min(50, remaining() * 0.4)
        for sa_seed in [seed + 20000, seed + 30000]:
            if remaining() < 20:
                break
            h_sa, c5_sa = sa_binary(h_bin_final, best_n, time_limit=t_sa / 2, seed=sa_seed)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
            # Also try continuous refinement from SA result
            if remaining() > 20:
                h_cont = slsqp_smooth(h_sa, best_n, 5000, maxiter=400)
                c5_cont = compute_c5(h_cont)
                if c5_cont < best_c5:
                    best_c5 = c5_cont
                    best_h = h_cont.copy()
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
