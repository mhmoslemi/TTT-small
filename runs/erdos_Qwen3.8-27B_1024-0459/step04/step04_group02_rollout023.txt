def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

    """Compute log-sum-exp approximation of max and its gradient w.r.t. corr."""
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = m + np.log(wsum) / beta
    return val, weights

    grad = np.zeros(n)

            for off in range(-(n - 1), n):
                k = n - 1 + off
                if off == 0:
                    a[k] += 1.0
                else:
                    if 0 <= i + off < n:
                        a[k] += (1.0 - best[i + off])
                    if 0 <= i - off < n:
                        a[k] += (-best[i - off])

            base = best_corr - a * t0

            def obj_at_t(t):
                vals = base + a * t
                vals[n - 1] -= t * t  # quadratic correction for lag 0
                return np.max(vals) * dx

            # Candidate evaluation
            candidates = [0.0, 1.0, 0.5, t0, t0 * 0.5, t0 + 0.05, t0 - 0.05]
            candidates = [max(0.0, min(1.0, c)) for c in candidates]

            best_t = t0
            best_val = best_c5
            for t in candidates:
                if abs(t - t0) < 1e-10:
                    continue
                c = obj_at_t(t)
                if c < best_val - 1e-13:
                    best_val = c
                    best_t = t

            if abs(best_t - t0) > 1e-8:
                # Golden section
                lo = max(0.0, t0 - 0.3)
                hi = min(1.0, t0 + 0.3)
                if best_t <= lo:
                    lo, hi = 0.0, max(t0, 1e-6)
                elif best_t >= hi:
                    lo, hi = min(t0, 1.0 - 1e-6), 1.0
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1, f2 = obj_at_t(c1), obj_at_t(c2)
                for _ in range(35):
                    if f1 < f2:
                        hi = c2; c2 = c1; f2 = f1
                        c1 = hi - gr * (hi - lo); f1 = obj_at_t(c1)
                    else:
                        lo = c1; c1 = c2; f1 = f2
                        c2 = lo + gr * (hi - lo); f2 = obj_at_t(c2)
                t_star = (lo + hi) / 2.0
                c_star = obj_at_t(t_star)
                if c_star < best_c5 - 1e-13:
                    best[i] = t_star
                    best_corr = compute_corr(best)
                    best_c5 = compute_c5(best)
                    improved = True

        if not improved:
            break

    return best, best_c5

def subgrad_refine(h, n, max_iters=500, rng=None):
    if rng is None:
        rng = np.random.RandomState(7)
    best = h.copy()
    best_c5 = compute_c5(best)
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        corr = compute_corr(best)
        m = np.max(corr)
        active = np.where(corr >= m - 0.5)[0]
        if len(active) > 10:
            active = active[np.argsort(corr[active])[::-1][:10]]
        grad = np.zeros(n)
        for k_idx in active:
            offset = int(k_idx) - (n - 1)
            if offset >= 0:
                npairs = n - offset
                if npairs > 0:
                    grad[:npairs] += (1.0 - best[offset:offset + npairs])
                    grad[offset:offset + npairs] -= best[:npairs]
            else:
                aoff = -offset
                npairs = n - aoff
                if npairs > 0:
                    grad[:npairs] += (1.0 - best[aoff:aoff + npairs])
                    grad[aoff:aoff + npairs] -= best[:npairs]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        grad = grad - np.mean(grad)  # project to sum=const subspace
        gn2 = np.linalg.norm(grad)
        if gn2 < 1e-15:
            break
        grad = grad / gn2

        improved = False
        base_step = 0.02 / (1.0 + it * 0.002)
        for sm in [1.0, 0.5, 2.0, 0.25, 0.1, 0.75, 3.0]:
            step = base_step * sm
            h_trial = best - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-15:
                best_c5 = c5_trial
                best = h_trial.copy()
                improved = True
                break
        if not improved:
            if it > 100:
                break

    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _start_time, _budget_s
    _start_time = time.time()
    _budget_s = budget_s
    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
    except Exception:
        parent_n = 640
        parent_h = np.zeros(parent_n)
        for i in range(parent_n):
            v = i
            cnt = 0
            while v:
                cnt ^= (v & 1)
                v >>= 1
            parent_h[i] = 1.0 if cnt == 1 else 0.0
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)

    best_h = parent_h.copy()
    best_c5 = parent_c5
    best_n = parent_n

    # Phase 1: SLSQP at parent resolution
    h = parent_h.copy()
    n = parent_n
    for beta in [200, 800, 3000, 10000]:
        if remaining_time() < 60:
            break
        h_new = slsqp_refine(h, n, beta, 400)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-16:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                best_n = n

    # Phase 2: Try higher resolutions
    resolutions = []
    for rn in [int(parent_n * 1.125), int(parent_n * 1.25), int(parent_n * 1.5),
               int(parent_n * 2), 720, 800, 960, 1120]:
        rn = int(rn)
        if rn % 2 == 0 and 320 <= rn <= 1280 and rn != parent_n:
            if rn not in resolutions:
                resolutions.append(rn)
    resolutions.sort()

    for rn in resolutions:
        if remaining_time() < 80:
            break
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)

        # SLSQP rounds at this resolution
        for beta in [300, 1500, 6000, 20000]:
            if remaining_time() < 40:
                break
            h_new = slsqp_refine(h_r, rn, beta, 500)
            c5_new = compute_c5(h_new)
            if c5_new < c5_r - 1e-15:
                h_r = h_new
                c5_r = c5_new
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = rn

    # Phase 3: Binary local search at best resolution
    if remaining_time() > 60:
        h_bin = (best_h > 0.5).astype(float)
        # Ensure sum constraint
        n_b = len(h_bin)
        target_ones = n_b // 2
        current_ones = int(np.sum(h_bin))
        if current_ones > target_ones:
            one_idx = np.where(h_bin == 1)[0]
            for idx in rng.choice(one_idx, current_ones - target_ones, replace=False):
                h_bin[idx] = 0.0
        elif current_ones < target_ones:
            zero_idx = np.where(h_bin == 0)[0]
            for idx in rng.choice(zero_idx, target_ones - current_ones, replace=False):
                h_bin[idx] = 1.0

        h_bin, c5_bin = binary_local_search(h_bin, best_n, 80000, rng)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            best_n = best_n

        # Use binary solution to initialize continuous optimization
        if c5_bin < 0.39:
            h_cont = best_h.copy()
            # Small perturbation to allow continuous optimization
            h_cont = np.clip(h_cont + 0.01 * (rng.randn(best_n) - 0.5), 0.0, 1.0)
            h_cont = project_feasible(h_cont, best_n)
            for beta in [1000, 5000, 20000]:
                if remaining_time() < 30:
                    break
                h_new = slsqp_refine(h_cont, best_n, beta, 400)
                c5_new = compute_c5(h_new)
                if c5_new < compute_c5(h_cont) - 1e-15:
                    h_cont = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_cont.copy()
                    best_n = best_n

    # Phase 4: Coordinate descent
    if remaining_time() > 30:
        h_new, c5_new = coord_descent(best_h, best_n, 80, rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 5: Subgradient refine
    if remaining_time() > 30:
        h_new, c5_new = subgrad_refine(best_h, best_n, 300, rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 6: Another round of binary search + continuous
    if remaining_time() > 60:
        h_bin2 = (best_h > 0.5).astype(float)
        n_b = len(h_bin2)
        target_ones = n_b // 2
        current_ones = int(np.sum(h_bin2))
        if current_ones > target_ones:
            one_idx = np.where(h_bin2 == 1)[0]
            for idx in rng.choice(one_idx, current_ones - target_ones, replace=False):
                h_bin2[idx] = 0.0
        elif current_ones < target_ones:
            zero_idx = np.where(h_bin2 == 0)[0]
            for idx in rng.choice(zero_idx, target_ones - current_ones, replace=False):
                h_bin2[idx] = 1.0
        h_bin2, c5_bin2 = binary_local_search(h_bin2, best_n, 50000, rng)
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()
            # Continuous polish
            for beta in [5000, 20000, 50000]:
                if remaining_time() < 20:
                    break
                h_new = slsqp_refine(best_h, best_n, beta, 300)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()

    # Phase 7: Final SLSQP polish with very high beta
    if remaining_time() > 15:
        for beta in [30000, 100000, 300000]:
            if remaining_time() < 8:
                break
            h_new = slsqp_refine(best_h, best_n, beta, 300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 8: Final coordinate descent
    if remaining_time() > 10:
        h_new, c5_new = coord_descent(best_h, best_n, 50, rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
