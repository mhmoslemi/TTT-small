def slsqp_smooth(h, n, beta, maxiter=300):
    """SLSQP with smooth max approximation."""
    N = n
    dx_v = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_v
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(N)
        for k in range(2 * N - 1):
            wk = w[k] * dx_v
            if wk < 1e-15:
                continue
            g_k = corr_grad(x, k, N)
            g += wk * g_k
        return g
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(N)
    
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

import numpy as np

import time

_T0 = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def corr_grad(h, k_idx, n):
    g = np.zeros(n)
    off = k_idx - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - h[:npairs])
        g[:npairs] -= h[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - h[ao:ao + npairs])
        g[ao:ao + npairs] -= h[:npairs]
    return g


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(120):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)

        if best_alpha > 0:
            prev_dir = d.copy()
            h = project_feasible(h + best_alpha * d, n)
            if best_obj < best_c5:
                best_c5 = best_obj
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 2
            eps *= 0.5
            if eps < 0.0005:
                break
            prev_dir *= 0.5
            if stagnant > 50:
                break

    return h, best_c5


def slsqp_smooth(h, n, beta, maxiter=300):
    N = n
    dx_v = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_v

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(N)
        for k in range(2 * N - 1):
            wk = w[k] * dx_v
            if wk < 1e-15:
                continue
            g += wk * corr_grad(x, k, N)
        return g

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(N)

    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < obj(h) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def binary_local_search(h, n, time_limit=30, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            rem = rng.choice(ones_idx, n_ones - n_half, replace=False)
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            add = rng.choice(zeros_idx, n_half - n_ones, replace=False)
            h[add] = 1

    dx_v = 2.0 / n
    corr = compute_corr(h)
    cur_c5 = float(np.max(corr)) * dx_v
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    n_stagnant = 0

    while time.time() - start < time_limit and remaining() > 5:
        if n_stagnant > 300:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_swap = None
        best_delta = 0
        for _ in range(40):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0
            h[j] = 1
            nc = compute_corr(h)
            nc5 = float(np.max(nc)) * dx_v
            delta = nc5 - cur_c5
            h[i] = 1
            h[j] = 0
            if delta < best_delta - 1e-15:
                best_delta = delta
                best_swap = (i, j)

        if best_swap is not None:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            cur_c5 += best_delta
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            improved = True

        if not improved:
            n_stagnant += 1
        else:
            n_stagnant = 0

    return best_h, best_c5


def multi_swap_search(h, n, time_limit=20, rng=None):
    """Try 2-swap moves: swap two ones with two zeros."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            rem = rng.choice(ones_idx, n_ones - n_half, replace=False)
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            add = rng.choice(zeros_idx, n_half - n_ones, replace=False)
            h[add] = 1

    dx_v = 2.0 / n
    corr = compute_corr(h)
    cur_c5 = float(np.max(corr)) * dx_v
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    stagnant = 0

    while time.time() - start < time_limit and remaining() > 5:
        if stagnant > 200:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        best_move = None
        best_delta = 0
        for _ in range(20):
            i1, i2 = rng.choice(ones, 2, replace=False)
            j1, j2 = rng.choice(zeros, 2, replace=False)
            h[i1] = 0
            h[i2] = 0
            h[j1] = 1
            h[j2] = 1
            nc = compute_corr(h)
            nc5 = float(np.max(nc)) * dx_v
            delta = nc5 - cur_c5
            h[i1] = 1
            h[i2] = 1
            h[j1] = 0
            h[j2] = 0
            if delta < best_delta - 1e-15:
                best_delta = delta
                best_move = (i1, i2, j1, j2)

        if best_move is not None:
            i1, i2, j1, j2 = best_move
            h[i1] = 0
            h[i2] = 0
            h[j1] = 1
            h[j2] = 1
            cur_c5 += best_delta
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            improved = True

        if not improved:
            stagnant += 1
        else:
            stagnant = 0

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP at moderate beta
    for beta in [200, 1000, 5000]:
        if remaining() < 15:
            break
        try:
            h_new = slsqp_smooth(h, n, beta, maxiter=250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass

    # Phase 2: Active-set descent (fast, many iterations)
    if remaining() > 20:
        t_lim = min(100, int(remaining() * 0.25))
        h_as, c5_as = active_set_descent(h, n, time_limit=t_lim, top_k=5, eps=0.005)
        if c5_as < best_c5:
            best_c5 = c5_as
            best_h = h_as.copy()

    # Another active-set pass with different params
    if remaining() > 20:
        t_lim = min(60, int(remaining() * 0.15))
        h_as2, c5_as2 = active_set_descent(best_h, n, time_limit=t_lim, top_k=8, eps=0.002)
        if c5_as2 < best_c5:
            best_c5 = c5_as2
            best_h = h_as2.copy()

    # Phase 3: SLSQP high-beta polish
    if remaining() > 20:
        for beta in [8000, 20000, 50000]:
            if remaining() < 10:
                break
            try:
                h_new = slsqp_smooth(best_h, n, beta, maxiter=200)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    # Phase 4: Active-set again after SLSQP
    if remaining() > 25:
        t_lim = min(80, int(remaining() * 0.2))
        h_as3, c5_as3 = active_set_descent(best_h, n, time_limit=t_lim, top_k=4, eps=0.001, momentum=0.2)
        if c5_as3 < best_c5:
            best_c5 = c5_as3
            best_h = h_as3.copy()

    # Phase 5: Try nearby resolutions
    n_candidates = []
    for delta in [-8, -4, 4, 8, 16, -16]:
        n_t = best_n + delta
        if n_t >= 128 and n_t <= 700 and n_t % 2 == 0:
            n_candidates.append(n_t)
    for factor in [1.2, 1.4, 1.6, 2.0]:
        n_t = int(best_n * factor)
        n_t = 2 * (n_t // 2)
        if 128 <= n_t <= 700:
            n_candidates.append(n_t)
    n_candidates = sorted(set(n_candidates))
    n_candidates.sort(key=lambda x: abs(x - best_n))

    n_tried = 0
    for n_t in n_candidates:
        if remaining() < 35 or n_tried >= 5:
            break
        n_tried += 1
        try:
            h_interp = interp_to(best_h, best_n, n_t)
            h_feas = project_feasible(h_interp, n_t)

            # Active-set at new resolution
            t_lim = min(50, int(remaining() * 0.25))
            h_opt, c5_opt = active_set_descent(h_feas, n_t, time_limit=t_lim, top_k=5)

            # SLSQP polish
            if remaining() > 15:
                h_opt = slsqp_smooth(h_opt, n_t, 3000, maxiter=150)
                c5_opt = compute_c5(h_opt)

            # Final active-set
            if remaining() > 10:
                h_opt, c5_opt = active_set_descent(h_opt, n_t, time_limit=15, top_k=5)

            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_t
        except Exception:
            pass

    # Phase 6: Binary local search
    if remaining() > 25:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project_feasible(h_bin, best_n)
        n_ones = int(np.round(np.sum(h_bin)))
        if n_ones != best_n // 2:
            sort_idx = np.argsort(-h_bin)
            h_bin = np.zeros(best_n)
            h_bin[sort_idx[:best_n // 2]] = 1.0
        else:
            h_bin = (h_bin > 0.5).astype(float)

        t_lim = min(50, int(remaining() * 0.25))
        h_b, c5_b = binary_local_search(h_bin, best_n, time_limit=t_lim, rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

        # Multi-swap search
        if remaining() > 15:
            t_lim2 = min(25, int(remaining() * 0.15))
            h_ms, c5_ms = multi_swap_search(best_h, best_n, time_limit=t_lim2, rng=np.random.RandomState(seed + 7))
            if c5_ms < best_c5:
                best_c5 = c5_ms
                best_h = h_ms.copy()

        # Thue-Morse start
        h_tm = np.zeros(best_n)
        for i in range(best_n):
            c = bin(i).count('1') % 2
            h_tm[i] = float(c)
        n_ones_tm = int(np.sum(h_tm))
        if n_ones_tm != best_n // 2:
            if n_ones_tm > best_n // 2:
                ones_idx = np.where(h_tm == 1)[0]
                to_rem = n_ones_tm - best_n // 2
                h_tm[ones_idx[:to_rem]] = 0
            else:
                zeros_idx = np.where(h_tm == 0)[0]
                to_add = best_n // 2 - n_ones_tm
                h_tm[zeros_idx[:to_add]] = 1

        if remaining() > 15:
            t_lim3 = min(20, int(remaining() * 0.1))
            h_tm_b, c5_tm = binary_local_search(h_tm, best_n, time_limit=t_lim3, rng=np.random.RandomState(seed + 3))
            if c5_tm < best_c5:
                best_c5 = c5_tm
                best_h = h_tm_b.copy()

    # Phase 7: Final refinement cycle
    n_f = best_n
    h_f = best_h.copy()

    for cycle in range(4):
        if remaining() < 12:
            break
        # SLSQP
        try:
            h_f = slsqp_smooth(h_f, n_f, 10000, maxiter=150)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()
                h_f = h_f.copy()
        except Exception:
            pass
        # Active-set
        if remaining() > 10:
            t_lim = min(30, int(remaining() * 0.2))
            h_f, c5_f = active_set_descent(h_f, n_f, time_limit=t_lim, top_k=5, eps=0.001)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()
        # SLSQP final
        if remaining() > 8:
            try:
                h_f = slsqp_smooth(best_h, n_f, 30000, maxiter=100)
                c5_f = compute_c5(h_f)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_f.copy()
            except Exception:
                pass

    # Final projection
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
