            h_new = h_mom + momentum * (y_new - h_old)
            h_new = project_feasible(h_new, n)
            obj_new = smooth_obj_grad(h_new, beta, n)[0]
            if obj_new < obj_y + 1e-18:
                h_mom = h_new
                y = y_new
                stall = 0
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            else:
                stall += 1
                if stall > 15:
                    alpha *= 0.7
                    stall = 0
                h_mom = h_old
                y = h_old
            if alpha < 1e-12:
                break
        if it >= total_iter or remaining_time() < 3:
            break
    
    return best_h, best_c5

def slsqp_smooth(h, n, beta, maxiter=300):
    global _N_G, _BETA_G
    _N_G = n
    _BETA_G = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    
    def obj_fn(x):
        h2 = np.clip(x, 0.0, 1.0)
        v, g = smooth_obj_grad(h2, beta, n)
        return v
    
    try:
        res = minimize(obj_fn, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def slsqp_minimax(h_init, n, max_rounds=10, max_slsqp_iter=200):
    global _N_G, _DX_G, _LAGS_G
    _N_G = n
    _DX_G = 2.0 / n
    h = h_init.copy()
    dx = 2.0 / n
    
    for rnd in range(max_rounds):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        max_corr = np.max(corr)
        t_init = max_corr * dx + 1e-8
        threshold = max_corr - 0.5
        active = np.where(corr >= threshold)[0]
        top3 = np.argsort(corr)[::-1][:3]
        active = np.unique(np.concatenate([active, top3]))
        active = active[np.argsort(active)]
        _LAGS_G = active
        x0 = np.concatenate([h, [t_init]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        
        def mm_obj(x):
            return x[n]
        
        def mm_ineq(x):
            h2 = x[:n]
            t = x[n]
            c = compute_corr(h2)
            return t - dx * c[_LAGS_G]
        
        def mm_eq(x):
            return np.sum(x[:_N_G]) - _N_G / 2.0
        
        cons = [
            {'type': 'ineq', 'fun': mm_ineq},
            {'type': 'eq', 'fun': mm_eq}
        ]
        try:
            res = minimize(mm_obj, x0, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_slsqp_iter, 'ftol': 1e-15, 'iprint': 0})
            if res.x is not None:
                h_new = project_feasible(res.x[:n], n)
                c5_new = compute_c5(h_new)
                c5_old = compute_c5(h)
                if c5_new < c5_old - 1e-16:
                    h = h_new
                else:
                    if rnd > 3:
                        break
        except Exception:
            break
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def binarize_optimize(h, n, threshold_low, threshold_high, rng):
    h_b = h.copy()
    h_b[h_b < threshold_low] = 0.0
    h_b[h_b > threshold_high] = 1.0
    h_b = project_feasible(h_b, n)
    c5_b = compute_c5(h_b)
    
    free_mask = (h_b > 1e-6) & (h_b < 1.0 - 1e-6)
    n_free = np.sum(free_mask)
    if n_free > 0 and n_free <= 80 and remaining_time() > 10:
        free_idx = np.where(free_mask)[0]
        fixed_vals = h_b.copy()
        
        def free_obj(free_x):
            h2 = fixed_vals.copy()
            h2[free_idx] = np.clip(free_x, 0.0, 1.0)
            h2 = project_feasible(h2, n)
            return compute_c5(h2)
        
        x0 = h_b[free_idx]
        bounds_f = [(0.0, 1.0)] * n_free
        try:
            res = minimize(free_obj, x0, method='Nelder-Mead',
                          bounds=bounds_f,
                          options={'maxiter': 500, 'xatol': 1e-8, 'fatol': 1e-10})
            if res.x is not None:
                h2 = fixed_vals.copy()
                h2[free_idx] = np.clip(res.x, 0.0, 1.0)
                h2 = project_feasible(h2, n)
                c5_2 = compute_c5(h2)
                if c5_2 < c5_b:
                    return h2, c5_2
        except Exception:
            pass
    return h_b, c5_b

def pair_transfer(h, n, max_sweeps=10, rng=None, window=40):
    if rng is None:
        rng = np.random.RandomState(0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        any_imp = False
        order = rng.permutation(n)
        for ii in range(len(order)):
            if remaining_time() < 1:
                break
            i = int(order[ii])
            hi = best_h[i]
            if hi < 0.05 or hi > 0.95:
                continue
            j_start = (i + 1) % n
            j_end = (i + 1 + window) % n
            steps = range(window) if j_start + window <= n else []
            for jj in range(window):
                j = (i + 1 + jj) % n
                hj = best_h[j]
                if abs(hi - hj) < 0.02:
                    continue
                for frac in [0.3, 0.5, 0.7]:
                    amt = (hi - hj) * frac
                    if abs(amt) < 0.005:
                        continue
                    h_trial = best_h.copy()
                    h_trial[i] -= amt
                    h_trial[j] += amt
                    h_trial = project_feasible(h_trial, n)
                    c5_t = compute_c5(h_trial)
                    if c5_t < best_c5 - 1e-15:
                        best_c5 = c5_t
                        best_h = h_trial.copy()
                        any_imp = True
                        break
            if any_imp:
                pass
        if not any_imp:
            break
    return best_h, best_c5

def flip_search(h, n, rng, max_flips=200):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(max_flips):
        if remaining_time() < 2:
            break
        i = int(rng.randint(n))
        h_trial = best_h.copy()
        h_trial[i] = 1.0 - h_trial[i]
        h_trial = project_feasible(h_trial, n)
        c5_t = compute_c5(h_trial)
        if c5_t < best_c5 - 1e-15:
            best_c5 = c5_t
            best_h = h_trial.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S, _N_G, _BETA_G, _LAGS_G, _DX_G
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 256
        best_h = np.full(best_n, 0.5)
        for i in range(0, best_n, 2):
            best_h[i] = 1.0
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)
    
    n = best_n
    h = best_h.copy()
    
    # Phase 1: Nesterov accelerated optimization with increasing beta
    if remaining_time() > 200:
        beta_sched = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
        alloc = min(300, int(remaining_time() * 0.3))
        h1, c5_1 = nesterov_optimize(h, n, beta_sched, alpha0=0.02, 
                                       max_iter_total=alloc, momentum=0.85)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1
            h = h1
    
    # Phase 2: SLSQP smooth refinement
    if remaining_time() > 100:
        for beta in [500, 2000, 8000]:
            if remaining_time() < 15:
                break
            h2 = slsqp_smooth(h, n, beta, maxiter=300)
            c5_2 = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2
                h = h2
    
    # Phase 3: Minimax SLSQP
    if remaining_time() > 80:
        h3 = slsqp_minimax(h, n, max_rounds=12, max_slsqp_iter=250)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3
            h = h3
    
    # Phase 4: Binarize + optimize free variables
    if remaining_time() > 80:
        for tl, th in [(0.2, 0.8), (0.15, 0.85), (0.1, 0.9), (0.25, 0.75), (0.3, 0.7)]:
            if remaining_time() < 15:
                break
            hb, c5_b = binarize_optimize(best_h, best_n, tl, th, rng)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = hb
                h = hb.copy()
            # Polish the binarized version
            if remaining_time() > 20:
                hp = slsqp_smooth(hb, best_n, 1000, maxiter=200)
                c5_p = compute_c5(hp)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = hp
                    h = hp.copy()
    
    # Phase 5: Multi-resolution exploration
    trial_ns = [192, 224, 240, 288, 320, 384, 448, 512]
    for trial_n in trial_ns:
        if remaining_time() < 25:
            break
        h_interp = interp_to(best_h, best_n, trial_n)
        h_interp = project_feasible(h_interp, trial_n)
        c5_i = compute_c5(h_interp)
        
        if c5_i < best_c5 + 0.005:
            maxit = 250 if trial_n <= 320 else 150
            for beta in [300, 1500, 5000]:
                if remaining_time() < 8:
                    break
                h_new = slsqp_smooth(h_interp, trial_n, beta, maxiter=maxit)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = trial_n
                    h_interp = h_new.copy()
            
            if remaining_time() > 15 and trial_n <= 384:
                h_mm = slsqp_minimax(h_interp, trial_n, max_rounds=8, max_slsqp_iter=150)
                c5_new = compute_c5(h_mm)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_mm.copy()
                    best_n = trial_n
    
    # Phase 6: Nesterov at best resolution with higher betas
    if remaining_time() > 60:
        beta_sched2 = [5000, 10000, 20000, 50000, 100000]
        h6, c5_6 = nesterov_optimize(best_h, best_n, beta_sched2, 
                                       alpha0=0.01, max_iter_total=150, momentum=0.8)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6
    
    # Phase 7: Combinatorial refinement (pair transfers)
    if remaining_time() > 40:
        h_comb, c5_comb = pair_transfer(best_h, best_n, max_sweeps=12, rng=rng, window=50)
        if c5_comb < best_c5:
            best_c5 = c5_comb
            best_h = h_comb
        if remaining_time() > 15:
            h_f, c5_f = flip_search(best_h, best_n, rng, max_flips=200)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f
    
    # Phase 8: More pair transfers after combinatorial
    if remaining_time() > 30:
        h_comb2, c5_comb2 = pair_transfer(best_h, best_n, max_sweeps=8, rng=rng, window=30)
        if c5_comb2 < best_c5:
            best_c5 = c5_comb2
            best_h = h_comb2
        if remaining_time() > 10:
            h_pol = slsqp_smooth(best_h, best_n, 20000, maxiter=200)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol
    
    # Phase 9: Try different binarization on best
    if remaining_time() > 25:
        for tl, th in [(0.2, 0.8), (0.3, 0.7)]:
            if remaining_time() < 8:
                break
            hb, c5_b = binarize_optimize(best_h, best_n, tl, th, rng)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = hb
    
    # Phase 10: Final SLSQP minimax polish
    if remaining_time() > 15:
        h_mm = slsqp_minimax(best_h, best_n, max_rounds=8, max_slsqp_iter=200)
        c5_mm = compute_c5(h_mm)
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm
    
    # Phase 11: Final ultra-high beta Nesterov
    if remaining_time() > 10:
        h_fin, c5_fin = nesterov_optimize(best_h, best_n, [100000, 200000], 
                                           alpha0=0.005, max_iter_total=80, momentum=0.7)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin
    
    # Phase 12: Final combinatorial sweep
    if remaining_time() > 8:
        h_comb3, c5_comb3 = pair_transfer(best_h, best_n, max_sweeps=4, rng=rng, window=25)
        if c5_comb3 < best_c5:
            best_c5 = c5_comb3
            best_h = h_comb3
    
    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
