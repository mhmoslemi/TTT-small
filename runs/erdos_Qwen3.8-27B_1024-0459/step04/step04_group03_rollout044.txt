    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.minimum(shifted, 0)  # numerical stability
    w = np.exp(shifted)
    w_sum = np.sum(w)
    w /= w_sum  # softmax weights
    logsumexp = (m + np.log(w_sum) / beta)  # this is the smooth max value
    return corr, w, logsumexp

def smooth_objective_gradient(x, n, beta, mu):
    h = np.clip(x, 0, 1)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    F = (m + np.log(w_sum) / beta) * dx
    w_norm = w / w_sum  # softmax over corr
    
    # Compute gradient of corr w.r.t. h
    # For each k, s_k = k - n + 1

    grad = grad_smooth + grad_penalty

Let t = j+s, so s = t-j. Range: t from j to j+n-1, but also t < n (since j+s < n) and t ≥ 0.
So t from max(j, 0) to min(j+n-1, n-1).
B[j] = sum_{t=max(j,0)}^{min(j+n-1, n-1)} w[t-j] * h[t]

This is the cross-correlation of w and h, evaluated at lag j.
In numpy terms: this is np.correlate(h_padded, w_padded, ...) or something.

Actually, let me think of it differently.

Let me extend h to a longer array and w to a longer array, then use np.correlate.

h_ext = np.concatenate([np.zeros(n-1), h])  # length 2n-1, h starts at index n-1
w_ext = w_arr  # length 2n-1, w_arr[k] = w[k-n+1]

np.correlate(h_ext, 1 - h_ext, 'full')... no, that's not right either.

```python
def compute_smooth_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    w = np.exp(shifted)
    F = (m + np.log(np.sum(w)) / beta) * dx
    w_norm = w / np.sum(w)
    
    # w_norm has length 2n-1, indexed by k from 0 to 2n-2
    # shift s = k - n + 1
    grad = np.zeros(n)
    for j in range(n):
        # First term: sum over k where 0 <= j + (k-n+1) < n
        # i.e., n-1-j <= k <= n-1+j
        lo1 = max(0, n-1-j)
        hi1 = min(2*n-2, n-1+j)
        if lo1 <= hi1:
            s_range = np.arange(lo1, hi1+1) - (n-1)  # shifts
            j_s = j + s_range  # j+s values
            grad[j] += np.sum(w_norm[lo1:hi1+1] * (1.0 - h[j_s]))
        
        # Second term: sum over k where 0 <= j - (k-n+1) < n
        # i.e., j+1 <= k-n+1 <= j+n-1, so j+n <= k <= j+2n-2
        # But k <= 2n-2, so k from j+n to 2n-2... wait that's not right.
        # 0 <= j - s < n where s = k-n+1
        # 0 <= j - (k-n+1) < n
        # j - k + n - 1 >= 0 → k <= j + n - 1
        # j - k + n - 1 < n → k > j - 1 → k >= j
        # So k from j to j+n-1, clipped to [0, 2n-2]
        lo2 = max(0, j)
        hi2 = min(2*n-2, j+n-1)
        if lo2 <= hi2:
            s_range = np.arange(lo2, hi2+1) - (n-1)
            j_minus_s = j - s_range
            grad[j] -= np.sum(w_norm[lo2:hi2+1] * h[j_minus_s])
    
    grad *= dx
    # Add penalty gradient
    # (handled separately)
    return F, grad
```

This is O(n²) per call but with numpy vectorization inside, it should be fast for n=256.

Actually, I just realized I can compute B and C using np.correlate:

B[j] = sum_s w[s] * h[j+s] for valid s, j+s
This is the cross-correlation of h and w (reversed) at position j.

If I define w_rev[s] = w[-s] (reversed w), then:
B[j] = sum_s w[s] * h[j+s] = sum_t w_rev[t] * h[j-t] where t = -s
= cross-correlation of h and w_rev at position j

import numpy as np
from scipy.optimize import minimize, minimize as sp_minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def smooth_obj_and_grad(x, n, beta, mu):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_shifted = np.exp(shifted)
    logsumexp = m + np.log(np.sum(exp_shifted)) / beta
    F_smooth = logsumexp * dx
    
    w_norm = exp_shifted / np.sum(exp_shifted)
    
    # Gradient of smooth max
    grad = np.zeros(n)
    for j in range(n):
        # s = k - (n-1), k from 0 to 2n-2
        # Term 1: sum over k where 0 <= j+s < n → n-1-j <= k <= n-1+j
        lo1 = max(0, n - 1 - j)
        hi1 = min(2 * n - 2, n - 1 + j)
        if lo1 <= hi1:
            s_vals = np.arange(lo1, hi1 + 1) - (n - 1)
            idx_hs = j + s_vals
            grad[j] += np.dot(w_norm[lo1:hi1 + 1], (1.0 - h[idx_hs]))
        
        # Term 2: sum over k where 0 <= j-s < n → j <= k <= j+n-1
        lo2 = max(0, j)
        hi2 = min(2 * n - 2, j + n - 1)
        if lo2 <= hi2:
            s_vals = np.arange(lo2, hi2 + 1) - (n - 1)
            idx_hs = j - s_vals
            grad[j] -= np.dot(w_norm[lo2:hi2 + 1], h[idx_hs])
    
    grad *= dx
    
    # Penalty for sum constraint
    sum_h = np.sum(h)
    F_pen = mu * (sum_h - n / 2.0) ** 2
    grad_pen = 2.0 * mu * (sum_h - n / 2.0)
    
    return F_smooth + F_pen, grad + grad_pen

def lbfgs_optimize(h_init, n, beta, mu, maxiter=500):
    x0 = h_init.copy()
    bounds = [(0.0, 1.0)] * n
    
    res = minimize(
        lambda x: smooth_obj_and_grad(x, n, beta, mu)[0],
        x0,
        jac=lambda x: smooth_obj_and_grad(x, n, beta, mu)[1],
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10, 'iprint': 0}
    )
    if res.x is not None:
        return project_feasible(res.x, n)
    return h_init

def slsqp_smooth(h, n, beta, maxiter=300):
    dx = 2.0 / n
    
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx
    
    def eq(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def slsqp_minimax(h_init, n, max_rounds=15, max_slsqp_iter=200):
    h = h_init.copy()
    dx = 2.0 / n
    
    for rnd in range(max_rounds):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        max_corr = np.max(corr)
        t_init = max_corr * dx + 1e-8
        
        threshold = max_corr - 0.3
        active = np.where(corr >= threshold)[0]
        top5 = np.argsort(corr)[::-1][:5]
        active = np.unique(np.concatenate([active, top5]))
        
        _active_lags = active
        _n_g = n
        _dx_g = dx
        
        def mm_obj(x):
            return x[_n_g]
        
        def mm_ineq(x):
            h_x = x[:_n_g]
            t = x[_n_g]
            c = np.correlate(h_x, 1.0 - h_x, mode='full')
            return t - _dx_g * c[_active_lags]
        
        def mm_eq(x):
            return np.sum(x[:_n_g]) - _n_g / 2.0
        
        x0 = np.concatenate([h, [t_init]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': mm_ineq},
            {'type': 'eq', 'fun': mm_eq}
        ]
        try:
            res = minimize(mm_obj, x0, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': max_slsqp_iter, 'ftol': 1e-14, 'iprint': 0})
            if res.x is not None:
                h_new = project_feasible(res.x[:n], n)
                c5_new = compute_c5(h_new)
                c5_old = compute_c5(h)
                if c5_new < c5_old - 1e-16:
                    h = h_new
                else:
                    if rnd > 3:
                        break
        except Exception:
            break
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def coordinate_descent(h, n, max_rounds=3, improvements_per_round=10):
    h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    target = n / 2.0
    
    for r in range(max_rounds):
        if remaining_time() < 3:
            break
        n_improvements = 0
        for j in range(n):
            if remaining_time() < 1:
                break
            if n_improvements >= improvements_per_round:
                break
            h_old_j = h[j]
            # Try several values
            for trial in np.concatenate([[0.0, 1.0], np.linspace(0, 1, 20)[1:-1]]):
                if abs(trial - h_old_j) < 0.001:
                    continue
                h[j] = trial
                # Adjust another coordinate to maintain sum
                diff = target - np.sum(h)
                if abs(diff) > 0.01:
                    # Find another coordinate to adjust
                    # Skip this trial if adjustment is too large
                    h[j] = h_old_j
                    continue
                # Adjust: find nearest non-bound coordinate
                mask = (h > 0.01) & (h < 0.99) & (np.arange(n) != j)
                if not np.any(mask):
                    h[j] = h_old_j
                    continue
                adj_idx = np.argmin(np.abs(h[mask] - 0.5))
                mask_indices = np.where(mask)[0]
                h[mask_indices[adj_idx]] -= diff
                h = np.clip(h, 0, 1)
                c5_try = compute_c5(h)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    n_improvements += 1
                    break
                else:
                    h[j] = h_old_j
        if n_improvements == 0:
            break
    
    h = project_feasible(h, n)
    return h, compute_c5(h)
