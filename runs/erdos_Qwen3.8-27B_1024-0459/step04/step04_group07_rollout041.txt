    # If j+k = i (j = i-k): term = h[i] * (1-h[i-k]), derivative w.r.t. h[i] = (1-h[i-k])
    # Total slope = (1-h[i-k]) - h[i+k] for valid indices

    #   j+k = i: need i-k in [0,n-1], k = idx-(n-1), so i-idx+(n-1) in [0,n-1]

So I do need to recompute corr after each projection. But that's O(n^2) per coordinate, which is expensive.

Alternative: instead of projection, handle the sum constraint differently.

**Key idea: Use a "virtual" coordinate to handle the sum constraint.**

Parametrize h as: h[i] for i=0,...,n-2, and h[n-1] = n/2 - sum(h[0:n-1]).

Then I only optimize n-1 free coordinates, and h[n-1] is determined. But h[n-1] must be in [0,1], so sum(h[0:n-1]) must be in [n/2-1, n/2].

This is a bit awkward. Let me try a different approach.

**Alternative: Use the fact that the total sum change from one coordinate step is at most 1.**

If I change h[i] from v to t, the sum changes by (t-v). To fix this, I need to reduce the sum by (t-v) among the other n-1 coordinates. The most "natural" way is to distribute it proportionally to the "slack" of each coordinate.

For simplicity, let me just reduce the sum by adjusting the coordinate with the most "room":
- If (t-v) > 0 (sum increased), find j≠i with the largest h[j] and reduce h[j] by (t-v) (if it stays ≥ 0)
- If (t-v) < 0 (sum decreased), find j≠i with the smallest h[j] and increase h[j] by -(t-v) (if it stays ≤ 1)

This is O(n) per step and only changes one additional coordinate. The effect on corr is then:
corr += (t-v)*slope_i + (δ_j)*slope_j where δ_j is the adjustment to coordinate j.

But slope_j depends on the current h, which has changed. For small adjustments, we can approximate.

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

                    best_val = val
                    best_t = t

    # Also try a few interior points for safety
    for frac in [0.25, 0.5, 0.75]:
        t = lo + frac * (hi - lo)
        val = float(np.max(a + t * b))
        if val < best_val - 1e-12:
            best_val = val
            best_t = t

    return best_t, best_val

def exact_coord_descent(h_init, n, max_sweeps, rng, tlimit):
    """Exact coordinate descent exploiting linearity of corr in each coordinate."""
    h = project_feasible(h_init, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_raw = float(np.max(corr))
    dx = 2.0 / n

    for sweep in range(max_sweeps):
        if tlimit() < 3.0:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if tlimit() < 1.0:
                break
            v = h[i]

            # Compute slope analytically
            slope = compute_slope(h, i, n)

            # f(t) = max_k (corr[k] + (t-v)*slope[k]) for t in [0,1]
            # = max_k ((corr[k] - v*slope[k]) + t*slope[k])
            a = corr - v * slope

            best_t, best_val = find_min_convex_pl(a, slope, 0.0, 1.0, top_count=25)

            if abs(best_t - v) > 1e-10:
                h[i] = best_t
                h = project_feasible(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_new = float(np.max(corr))
                if c5_new < c5_raw - 1e-12:
                    c5_raw = c5_new
                    improved = True

        if not improved and sweep > 10:
            break

    return h, c5_raw * dx

def pair_swap_descent(h_init, n, max_sweeps, rng, tlimit):
    """Pair swap descent: change two coordinates while maintaining sum."""
    h = project_feasible(h_init, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_raw = float(np.max(corr))
    dx = 2.0 / n

    for sweep in range(max_sweeps):
        if tlimit() < 3.0:
            break
        improved = False

        # Try random pairs
        n_pairs = min(500, n * 3)
        for _ in range(n_pairs):
            if tlimit() < 1.0:
                break
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            vi, vj = h[i], h[j]

            # Compute slopes for both coordinates
            slope_i = compute_slope(h, i, n)
            slope_j = compute_slope(h, j, n)

            # Change h[i] by +d and h[j] by -d
            # new_corr = corr + d*(slope_i - slope_j)
            # Valid: 0 <= vi+d <= 1 and 0 <= vj-d <= 1
            d_lo = max(-vi, vj - 1.0)
            d_hi = min(1.0 - vi, vj)

            if d_hi - d_lo < 1e-10:
                continue

            diff_slope = slope_i - slope_j
            best_d, best_val = find_min_convex_pl(corr, diff_slope, d_lo, d_hi, top_count=20)

            if best_val < c5_raw - 1e-10:
                h[i] = vi + best_d
                h[j] = vj - best_d
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_new = float(np.max(corr))
                if c5_new < c5_raw - 1e-12:
                    c5_raw = c5_new
                    improved = True

        if not improved and sweep > 5:
            break

    return h, c5_raw * dx

def slsqp_minimize(h0, n, beta, maxiter):
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    m = float(np.max(corr))

    def obj(x):
        corr_x = np.correlate(x, 1.0 - x, mode='full')
        mx = np.max(corr_x)
        shifted = corr_x - mx
        e = np.exp(np.clip(beta * shifted, -400, 400))
        val = (mx + np.log(np.sum(e)) / beta) * (2.0 / n)
        val += 50.0 * (float(np.sum(x)) - n / 2.0) ** 2
        return float(val)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: float(np.sum(x)) - n / 2.0}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    t_budget = float(budget_s)

    def tlimit():
        return max(1.0, t_budget - (time.time() - t_start) - 5.0)

    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = project_feasible(make_tm(parent_n), parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("No parent. TM: n={}, C5={:.8f}".format(parent_n, best_c5))

    # Phase 1: Exact coordinate descent at parent resolution
    if tlimit() > 60:
        print("\n=== Phase 1: Exact coord descent n={} ===".format(best_n))
        best_h, best_c5 = exact_coord_descent(best_h, best_n,
                                              max_sweeps=800, rng=rng, tlimit=tlimit)
        best_n = len(best_h)
        print("  C5={:.8f}".format(best_c5))

    # Phase 2: Pair swap descent
    if tlimit() > 40:
        print("\n=== Phase 2: Pair swap n={} ===".format(best_n))
        best_h, best_c5 = pair_swap_descent(best_h, best_n,
                                            max_sweeps=100, rng=rng, tlimit=tlimit)
        best_n = len(best_h)
        print("  C5={:.8f}".format(best_c5))

    # Phase 3: SLSQP refinement
    if tlimit() > 50:
        print("\n=== Phase 3: SLSQP n={} ===".format(best_n))
        n = best_n
        h_cur = best_h.copy()
        for bv in [50, 200, 500, 1500, 3000, 8000]:
            if tlimit() < 10:
                break
            h_new = slsqp_minimize(h_cur, n, bv, maxiter=400)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-12:
                best_c5 = c_new
                best_h = h_new.copy()
                h_cur = h_new
                print("  beta={}: C5={:.8f}".format(bv, c_new))
        print("  Final: C5={:.8f}".format(best_c5))

    # Phase 4: More coordinate descent after SLSQP
    if tlimit() > 30:
        print("\n=== Phase 4: Coord descent after SLSQP ===")
        best_h, best_c5 = exact_coord_descent(best_h, best_n,
                                              max_sweeps=400, rng=rng, tlimit=tlimit)
        best_n = len(best_h)
        print("  C5={:.8f}".format(best_c5))

    # Phase 5: Try higher resolution
    if tlimit() > 150:
        print("\n=== Phase 5: Higher resolutions ===")
        for n_try in [512, 768, 1024]:
            if tlimit() < 80:
                break
            # Multiple initializations
            h_interp = project_feasible(interp_h(best_h, best_n, n_try), n_try)
            c5_interp = compute_c5(h_interp)
            h_tm = project_feasible(make_tm(n_try), n_try)
            c5_tm = compute_c5(h_tm)
            h_r1 = project_feasible(make_rand_bin(rng, n_try), n_try)
            c5_r1 = compute_c5(h_r1)

            inits = [(h_interp, c5_interp), (h_tm, c5_tm), (h_r1, c5_r1)]
            inits.sort(key=lambda x: x[1])
            h_cur = inits[0][0].copy()
            c5_cur = inits[0][1]
            print("  n={} best_init={:.6f}".format(n_try, c5_cur))

            # Coordinate descent
            h_cur, c5_cur = exact_coord_descent(h_cur, n_try,
                                                max_sweeps=400, rng=rng, tlimit=tlimit)
            # Pair swaps
            if tlimit() > 30:
                h_cur, c5_cur = pair_swap_descent(h_cur, n_try,
                                                  max_sweeps=50, rng=rng, tlimit=tlimit)
            # SLSQP polish
            if tlimit() > 20:
                for bv in [500, 2000, 5000]:
                    if tlimit() < 8:
                        break
                    h_new = slsqp_minimize(h_cur, n_try, bv, maxiter=300)
                    c_new = compute_c5(h_new)
                    if c_new < c5_cur - 1e-12:
                        c5_cur = c_new
                        h_cur = h_new.copy()
                h_cur, c5_cur = exact_coord_descent(h_cur, n_try,
                                                    max_sweeps=200, rng=rng, tlimit=tlimit)

            if c5_cur < best_c5 - 1e-10:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n_try
                print("  n={} IMPROVED: C5={:.8f}".format(n_try, c5_cur))
            else:
                print("  n={} best: C5={:.8f} (record: {:.8f})".format(n_try, c5_cur, best_c5))

    # Phase 6: Random restart exploration at best resolution
    if tlimit() > 60:
        print("\n=== Phase 6: Random restarts n={} ===".format(best_n))
        n = best_n
        for attempt in range(10):
            if tlimit() < 15:
                break
            # Perturb best solution
            h_pert = best_h.copy()
            n_pert = rng.randint(5, max(6, n // 10))
            idxs = rng.choice(n, size=n_pert, replace=False)
            for idx in idxs:
                h_pert[idx] = np.clip(h_pert[idx] + rng.randn() * 0.4, 0.0, 1.0)
            h_pert = project_feasible(h_pert, n)
            c5_p = compute_c5(h_pert)

            if c5_p < best_c5 * 1.02:
                h_cur, c5_cur = exact_coord_descent(h_pert, n,
                                                    max_sweeps=200, rng=rng, tlimit=tlimit)
                if tlimit() > 10:
                    h_new = slsqp_minimize(h_cur, n, 2000, maxiter=200)
                    c_new = compute_c5(h_new)
                    if c_new < c5_cur:
                        c5_cur = c_new
                        h_cur = h_new
                if c5_cur < best_c5 - 1e-10:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n
                    print("  Restart {} : C5={:.8f}".format(attempt, c5_cur))

    # Phase 7: Final intensive coordinate descent
    if tlimit() > 30:
        print("\n=== Phase 7: Final coord descent ===")
        best_h, best_c5 = exact_coord_descent(best_h, best_n,
                                              max_sweeps=1000, rng=rng, tlimit=tlimit)
        best_n = len(best_h)
        print("  C5={:.8f}".format(best_c5))

    # Phase 8: Final pair swaps
    if tlimit() > 20:
        best_h, best_c5 = pair_swap_descent(best_h, best_n,
                                            max_sweeps=100, rng=rng, tlimit=tlimit)
        best_n = len(best_h)
        print("\nAfter pair swaps: C5={:.8f}".format(best_c5))

    # Phase 9: Ultra-fine SLSQP
    if tlimit() > 15:
        for bv in [10000, 20000]:
            if tlimit() < 8:
                break
            h_new = slsqp_minimize(best_h, best_n, bv, maxiter=500)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-13:
                best_c5 = c_new
                best_h = h_new

    # Phase 10: Final coordinate descent pass
    if tlimit() > 10:
        best_h, best_c5 = exact_coord_descent(best_h, best_n,
                                              max_sweeps=300, rng=rng, tlimit=tlimit)

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
