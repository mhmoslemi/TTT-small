                if c5_j < best_c5:
                    best_c5 = c5_j
                    best_h = h_j.copy()
                    best_n = jump_n
            if remaining_time() > 20:
                h_cd, c5_cd = coord_descent(best_h, best_n, 30, rng)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
    
    # Final: one more coord descent + high beta polish
    if remaining_time() > 10:
        h_cd, c5_cd = coord_descent(best_h, best_n, 20, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        if remaining_time() > 8:
            h_new = lbfgsb_refine(best_h, best_n, 200000, 200000, 200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
