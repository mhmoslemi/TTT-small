    for _ in range(10):
        h_trial = h - alpha * grad
        h_trial = project_feasible(h_trial, n)
        c5_trial = compute_c5(h_trial)
        c5_cur = compute_c5(h)
        if c5_trial < c5_cur - 1e-13:
            return h_trial, c5_trial
        alpha *= 0.5
    
    return h, c5_cur

def homotopy_slsqp(h, n, betas=[10, 50, 100, 300, 1000, 3000, 10000], maxiter_per_beta=200):
    """Progressive homotopy with SLSQP."""
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for beta in betas:
        if remaining_time() < 10.0:
            break
        h_new, c5_new = slsqp_optimize(h_cur, n, beta, maxiter=maxiter_per_beta)
        if c5_new < c5_cur - 1e-15:
            h_cur = h_new
            c5_cur = c5_new
    return h_cur, c5_cur
