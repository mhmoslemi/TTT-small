import numpy as np
from scipy.optimize import minimize, linprog
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)

def slsqp_full(h, n, beta=5000, maxiter=500):
    """Full SLSQP optimization with smooth max approximation."""
    dx_v = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -100.0, 100.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_v
    
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -100.0, 100.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k] * dx_v
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:

                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    
    def con_fun(x):
        return np.sum(x) - target

        return np.ones(n)

def binary_local_search(h, n, time_limit=60, rng=None):
    """Local search: flip pairs of bits, optimize fractional, check improvement."""
    if rng is None:
        rng = np.random.RandomState(42)
    
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Identify fractional entries
    frac_thresh_lo = 0.05
    frac_thresh_hi = 0.95
    
    start = time.time()
    n_iter = 0
    stall = 0
    
    while time.time() - start < time_limit and remaining() > 5:
        n_iter += 1
        if n_iter % 100 == 0 and stall > 200:
            break
        
        # Get current binary pattern
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project_feasible(h_bin, n)
        
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        # Try multiple random flips
        best_flip = None
        best_flip_c5 = best_c5
        
        n_tries = min(20, max(5, int((time_limit - (time.time() - start)) * 10)))
        for _ in range(n_tries):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            
            # Try flip i: 1→0, j: 0→1
            h_try = h_bin.copy()
            h_try[i] = 0
            h_try[j] = 1
            
            # Quick check with current fractional values
            h_cont = best_h.copy()
            h_cont[i] = 0
            h_cont[j] = 1
            h_cont = project_feasible(h_cont, n)
            
            c5_try = compute_c5(h_cont)
            if c5_try < best_flip_c5:
                best_flip_c5 = c5_try
                best_flip = (i, j)
        
        if best_flip is not None:
            i, j = best_flip
            best_h[i] = 0
            best_h[j] = 1
            best_h = project_feasible(best_h, n)
            
            # Optimize fractional entries
            frac_mask = (best_h > frac_thresh_lo) & (best_h < frac_thresh_hi)
            if np.count_nonzero(frac_mask) >= 2:
                h_opt, c5_opt = optimize_fractional(best_h, n, frac_mask)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
                    stall = 0
                else:
                    best_c5 = compute_c5(best_h)
                    stall += 1
            else:
                best_c5 = compute_c5(best_h)
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5

        sh = c - m

            rem = rng.choice(idx, n_ones - n_target, replace=False)
            h[rem] = 0
        else:
            idx = np.where(h == 0)[0]
            add = rng.choice(idx, n_target - n_ones, replace=False)
            h[idx[add]] = 1
    return h


def random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP refinement (progressive beta)
    for beta in [200, 1000, 5000, 20000]:
        if remaining() < 8:
            break
        h_new = slsqp_optimize(h, n, beta=beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Active set equalization
    if remaining() > 15:
        h_eq, c5_eq = active_set_optimize(h, n, top_k=8, n_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 3: Coordinate descent on fractional entries
    if remaining() > 15:
        frac_mask = (best_h > 0.02) & (best_h < 0.98)
        if np.count_nonzero(frac_mask) >= 2:
            h_cd, c5_cd = coord_desc_fractional(best_h, n, frac_mask, n_iters=500)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Phase 4: Binary flip search at parent resolution
    if remaining() > 60:
        flip_time = min(150, int(remaining() * 0.35))
        h_flip, c5_flip = binary_flip_search(best_h, best_n, time_limit=flip_time, rng=rng)
        if c5_flip < best_c5:
            best_c5 = c5_flip
            best_h = h_flip.copy()
            best_n = best_n
        # Re-optimize after flips
        if remaining() > 10:
            h_new = slsqp_optimize(best_h, best_n, beta=10000, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 5: Try other resolutions
    n_cands = []
    base = best_n
    for delta in [-6, -4, -2, 2, 4, 6, 8, 10, 12, 16, 20, 32, 48, 64, 128]:
        nn = base + delta
        if nn >= 128 and nn <= 3000 and nn % 2 == 0:
            n_cands.append(nn)
    n_cands = [nn for nn in n_cands if nn != base]
    n_cands = sorted(set(n_cands), key=lambda x: abs(x - base))

    for n_t in n_cands[:14]:
        if remaining() < 25:
            break
        h_int = interp_to(best_h, best_n, n_t)
        h_f = project_feasible(h_int, n_t)
        c5_f = compute_c5(h_f)
        if c5_f > best_c5 + 0.01:
            continue

        h_opt = h_f
        for b in [3000, 15000]:
            if remaining() < 10:
                break
            h_opt = slsqp_optimize(h_opt, n_t, beta=b, maxiter=300)
        c5_opt = compute_c5(h_opt)

        frac_m = (h_opt > 0.02) & (h_opt < 0.98)
        if np.count_nonzero(frac_m) >= 2 and remaining() > 8:
            h_opt2, c5_opt2 = coord_desc_fractional(h_opt, n_t, frac_m, n_iters=300)
            if c5_opt2 < c5_opt:
                h_opt = h_opt2
                c5_opt = c5_opt2

        if remaining() > 20 and c5_opt < best_c5 + 0.005:
            h_fl, c5_fl = binary_flip_search(h_opt, n_t, time_limit=15, rng=rng)
            if c5_fl < c5_opt:
                h_opt = h_fl
                c5_opt = c5_fl

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 6: Multi-start (Thue-Morse + random)
    if remaining() > 80:
        starts = []
        h_tm = thue_morse_seq(best_n, rng)
        starts.append(h_tm)
        starts.append(random_binary(best_n, rng))
        # Perturbed Thue-Morse
        h_ptm = h_tm.copy()
        for _ in range(best_n // 16):
            i = rng.randint(best_n)
            j = rng.randint(best_n)
            if h_ptm[i] != h_ptm[j]:
                h_ptm[i], h_ptm[j] = h_ptm[j], h_ptm[i]
        starts.append(h_ptm)

        for h_s in starts:
            if remaining() < 25:
                break
            h_s = project_feasible(h_s, best_n)
            h_s = slsqp_optimize(h_s, best_n, beta=5000, maxiter=300)
            c5_s = compute_c5(h_s)
            frac_m = (h_s > 0.02) & (h_s < 0.98)
            if np.count_nonzero(frac_m) >= 2 and remaining() > 10:
                h_s2, c5_s2 = coord_desc_fractional(h_s, best_n, frac_m, n_iters=200)
                if c5_s2 < c5_s:
                    h_s = h_s2
                    c5_s = c5_s2
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()

    # Phase 7: Intensive final local search
    if remaining() > 40:
        ls_time = min(180, int(remaining() * 0.45))
        h_ls, c5_ls = binary_flip_search(best_h, best_n, time_limit=ls_time, rng=rng)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 8: Final SLSQP + active set polish
    if remaining() > 8:
        h_new = slsqp_optimize(best_h, best_n, beta=50000, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if remaining() > 8:
        h_eq2, c5_eq2 = active_set_optimize(best_h, best_n, top_k=12, n_iters=200)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Phase 9: Final fractional polish
    if remaining() > 5:
        frac_m = (best_h > 0.005) & (best_h < 0.995)
        if np.count_nonzero(frac_m) >= 2:
            h_f, c5_f = coord_desc_fractional(best_h, best_n, frac_m, n_iters=300)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
