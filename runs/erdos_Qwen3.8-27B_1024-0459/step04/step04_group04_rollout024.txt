import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1
    return h_bin


def binary_sa(h, n, time_limit=30, seed=42):

def compute_topk_gradient(h, n, k_list):
    """Compute weighted gradient for top-k shifts (vectorized)."""
    g = np.zeros(n)
    for k in k_list:
        d = int(k) - (n - 1)
        grad_k = np.zeros(n)
        if d >= 0:
            np_ = n - d
            if np_ > 0:
                grad_k[:np_] += (1.0 - h[d:])
                grad_k[d:] += -h[:np_]
        else:
            dp = -d
            np_ = n - dp
            if np_ > 0:
                grad_k[dp:] += (1.0 - h[:np_])
                grad_k[:np_] += -h[dp:]
        g += grad_k
    g /= len(k_list)
    return g


def subgradient_refine(h, n, time_limit=60, top_k=5):

    best_h = h.copy()
    best_c5 = compute_c5(h)

    eta = 0.3
    momentum = np.zeros(n)
    beta = 0.7
    start = time.time()
    patience = 0
    max_patience = 200

    while time.time() - start < time_limit and remaining_time() > 5:
        corr = compute_corr(h)
        m = float(np.max(corr))
        c5 = m * (2.0 / n)

        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break

        # Get top-k shifts
        top_k_idx = np.argsort(-corr)[:top_k]
        g = compute_topk_gradient(h, n, top_k_idx)

        # Momentum update
        momentum = beta * momentum - eta * g
        h_new = project_feasible(h + momentum, n)

        # Check if improvement
        c5_new = compute_c5(h_new)
        if c5_new < c5 + 1e-10:
            h = h_new
        else:
            h = h_new  # still move, but reduce step
            eta *= 0.95
            beta = 0.5

        eta = max(eta * 0.9995, 0.01)

    return best_h, best_c5


def slsqp_smooth(h, n, beta_ls, maxiter=300):
    """SLSQP with log-sum-exp smooth approximation."""
    _N_local = n
    _BETA_local = beta_ls
    dx_val = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(_BETA_local * shifted, -50, 50)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / _BETA_local) * dx_val

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(_BETA_local * shifted, -50, 50)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] > 1e-12:
                d = int(k) - (n - 1)
                if d >= 0:
                    np_ = n - d
                    if np_ > 0:
                        g[:np_] += w[k] * (1.0 - x[d:])
                        g[d:] += -w[k] * x[:np_]
                else:
                    dp = -d
                    np_ = n - dp
                    if np_ > 0:
                        g[dp:] += w[k] * (1.0 - x[:np_])
                        g[:np_] += -w[k] * x[dp:]
        return g * dx_val

    def sum_con(x):
        return np.sum(x) - n / 2.0

    def sum_con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]

    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

    # Final local search if we're at a binary solution
    if remaining_time() > 20:
        h_ls, c5_ls = two_swap_local_search(best_h, best_n, max_iters=300)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Final subgradient polish
    if remaining_time() > 10:
        h_sg, c5_sg = subgradient_refine(best_h, best_n, time_limit=min(30, remaining_time() * 0.5), top_k=5)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
