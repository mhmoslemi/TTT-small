def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    """Project h onto {x in [0,1]^n : sum(x) = n/2}"""
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    # Water-filling projection
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Adjust only free variables
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(free)
        if cnt > 0:
            h[free] += diff / cnt
        else:
            h[:] += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

        return src_h.copy()

    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

    if weights is None:

            g[:npairs] += w * (1.0 - x[ao:ao + npairs])

    best_c5 = compute_c5(h)

    step = 0.02

def slsqp_optimize(h, n, beta, maxiter=500):

             'fun': lambda x: np.sum(x) - n/2.0,

    if s < 1e-30:
        s = 1e-30
    return (m + np.log(s) / beta) * dxv


def _smooth_grad(x):
    n = _G_N
    dxv = 2.0 / n
    beta = _G_BETA
    corr = compute_corr(x)
    m = float(np.max(corr))
    shifted = corr - m
    cl = np.clip(beta * shifted, -50.0, 50.0)
    ex = np.exp(cl)
    s = np.sum(ex)
    if s < 1e-30:
        s = 1e-30
    w = ex / s
    g = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for k in active:
        g += w[k] * grad_corr(x, n, int(k))
    return g * dxv


def _con_fun(x):
    return np.sum(x) - _G_N / 2.0


def _con_jac(x):
    return np.ones(_G_N)


def slsqp(h, n, beta, maxiter=600):
    global _G_N, _G_BETA
    _G_N = n
    _G_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_smooth_obj, h.copy(), jac=_smooth_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project(res.x, n)
    except Exception:
        return h


def binary_sa(h, n, time_limit=60, seed=42):
    rng = np.random.RandomState(seed)
    h = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1
    dxv = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()
    T = 6.0
    alpha = 0.9993
    start = time.time()
    stall = 0
    while time.time() - start < time_limit and T > 0.05:
        if remaining() < 10:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        delta = new_max - cur_max
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
        T *= alpha
        if stall > 15000:
            T = max(T, 4.0)
            stall = 0
    return best_h, best_max * dxv


def refine_combo(h, n, time_limit=120):
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    # Subgradient
    h_sg, c5_sg = subgrad(h, n, iters=8000)
    if c5_sg < best_c5:
        best_c5 = c5_sg
        best_h = h_sg.copy()
    # SLSQP
    for beta in [500, 2000, 8000, 30000]:
        if time.time() - start > time_limit * 0.6 or remaining() < 15:
            break
        h_new = slsqp(best_h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    # Another subgradient
    if time.time() - start < time_limit * 0.85 and remaining() > 10:
        h_sg2, c5_sg2 = subgrad(best_h, n, iters=5000)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()
    # Final SLSQP
    if remaining() > 10:
        h_new = slsqp(best_h, n, 50000, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 360
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n

    # Phase 1: Intensive subgradient at parent n
    if remaining() > 60:
        h_sg, c5_sg = subgrad(best_h, best_n, iters=30000)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 2: SLSQP at parent n
    for beta in [300, 1000, 5000, 20000]:
        if remaining() < 30:
            break
        h_new = slsqp(best_h, best_n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: More subgradient
    if remaining() > 60:
        h_sg, c5_sg = subgrad(best_h, best_n, iters=20000)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 4: Try nearby n values
    n_neighbors = []
    for dn in [-4, -2, 2, 4, 6, 8, 10, 14, 20]:
        nn = best_n + dn
        if nn % 2 == 0 and 100 <= nn <= 1000:
            n_neighbors.append(nn)
    n_neighbors = sorted(set(n_neighbors))

    for n_t in n_neighbors:
        if remaining() < 40:
            break
        h_t = interp_to(best_h, best_n, n_t)
        h_t = project(h_t, n_t)
        # Quick subgradient
        h_sg, c5_sg = subgrad(h_t, n_t, iters=5000)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            best_n = n_t
        # SLSQP if promising
        if remaining() > 20 and c5_sg < best_c5 + 0.001:
            h_new = slsqp(h_sg, n_t, 5000, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_t

    # Phase 5: Binary SA at best n
    if remaining() > 50:
        h_bin = (best_h > 0.5).astype(float).copy()
        h_bin = project(h_bin, best_n)
        t_sa = min(80, max(20, remaining() * 0.3))
        for sa_seed in [seed, seed + 1000, seed + 2000]:
            if remaining() < 15:
                break
            h_sa, c5_sa = binary_sa(h_bin, best_n, time_limit=t_sa / 3.0, seed=sa_seed)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = best_n

    # Phase 6: Intensive refinement at best n
    n = best_n
    if remaining() > 100:
        t_lim = min(180, remaining() * 0.5)
        h_ref, c5_ref = refine_combo(best_h, n, time_limit=t_lim)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    # Phase 7: Try a couple higher n values
    for n_hi in [420, 480, 540, 600]:
        if remaining() < 60:
            break
        if n_hi == n:
            continue
        h_hi = interp_to(best_h, n, n_hi)
        h_hi = project(h_hi, n_hi)
        h_sg, c5_sg = subgrad(h_hi, n_hi, iters=4000)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            best_n = n_hi
        if remaining() > 20:
            h_new = slsqp(h_sg, n_hi, 5000, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_hi

    # Phase 8: Final polish
    n = best_n
    if remaining() > 40:
        h_sg, c5_sg = subgrad(best_h, n, iters=10000)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
    if remaining() > 20:
        for beta in [30000, 100000]:
            if remaining() < 10:
                break
            h_new = slsqp(best_h, n, beta, maxiter=500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Final validation
    best_h = project(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
