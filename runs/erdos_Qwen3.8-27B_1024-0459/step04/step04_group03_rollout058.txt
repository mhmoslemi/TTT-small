def smooth_obj_and_grad(h, n, beta, lambda_penalty=10.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    
    # Log-sum-exp for numerical stability
    max_corr = np.max(corr)
    shifted = beta * (corr - max_corr)
    log_sum = max_corr + np.log(np.sum(np.exp(shifted))) / beta
    f = log_sum * dx
    
    # Penalty for sum constraint
    sum_diff = np.sum(h) - n/2.0
    f += lambda_penalty * sum_diff**2
    
    # Gradient
    # softmax weights
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    # weights shape: (2n-1,)
    
    # grad of corr[k] w.r.t. h[j]
    # corr[k] = sum_i h[i] * (1-h[i+k-(n-1)])
    # Let lag = k - (n-1), so corr[n-1+lag] = sum_i h[i] * (1-h[i+lag]) (for valid i)
    
    grad = np.zeros(n)
    for lag in range(-(n-1), n):
        if lag >= 0:
            # corr = sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag])
            # d/dh[j] = (1-h[j+lag])*[j+lag<n] + (-h[j-lag])*[j-lag>=0] (if lag>0)
            # For lag=0: d/dh[j] = 1-2h[j]
            w = weights[n-1+lag] * beta * dx
            if lag == 0:
                grad += w * (1 - 2*h)
            else:
                # First term: j from 0 to n-1-lag, contribution (1-h[j+lag])
                idx1 = np.arange(0, n-lag)
                grad[idx1] += w * (1 - h[idx1 + lag])
                # Second term: j from lag to n-1, contribution -h[j-lag]
                idx2 = np.arange(lag, n)
                grad[idx2] += w * (-h[idx2 - lag])
        else:
            # lag < 0, let lag' = -lag > 0
            lag2 = -lag
            # corr = sum_{i=lag'}^{n-1} h[i] * (1-h[i-lag'])
            # By symmetry with the positive lag case (swapping roles)
            # d/dh[j] = (1-h[j-lag'])*[j-lag'>=0] + (-h[j+lag'])*[j+lag'<n]
            w = weights[n-1+lag] * beta * dx
            # j from lag' to n-1: contribution (1-h[j-lag'])
            idx1 = np.arange(lag2, n)
            grad[idx1] += w * (1 - h[idx1 - lag2])
            # j from 0 to n-1-lag': contribution -h[j+lag']
            idx2 = np.arange(0, n-lag2)
            grad[idx2] += w * (-h[idx2 + lag2])
    
    # Add gradient of penalty
    grad += 2 * lambda_penalty * sum_diff
    
    return f, grad

    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    f = (cmax + np.log(sum_exp) / beta) * dx
