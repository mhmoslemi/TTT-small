```python
def compute_c5(h):

def compute_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            step = diff / cnt
            h[mask] += step
        else:
            if diff > 0:
                idx = np.argmin(h)
            else:
                idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Distribute among all interior points
        mask = (h > 0.01) & (h < 0.99)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def smooth_obj(x, n, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    return (m + np.log(wsum) / beta) * dx

def smooth_grad(x, n, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return grad

def coord_descent_1d_min(h, i, n, dx, corr, max_iter=200):
    """Find optimal h[i] by 1D minimization of the max correlation."""
    t0 = h[i]
    
    # Precompute: for each lag k, the correlation as a function of t = h[i]
    # corr[k] = base[k] + a[k]*t + b[k]*t^2
    ncorr = 2 * n - 1
    a = np.zeros(ncorr)
    b = np.zeros(ncorr)
    
    for off in range(-(n-1), n):
        k = n - 1 + off
        if off == 0:
            # h[i]*(1-h[i]) term: t*(1-t) = t - t^2
            a[k] += 1.0
            b[k] += -1.0
        else:
            if 0 <= i + off < n:
                a[k] += (1.0 - h[i + off])
            if 0 <= i - off < n:
                a[k] += (-h[i - off])
    
    # base[k] = corr[k] - a[k]*t0 - b[k]*t0^2
    base = corr - a * t0 - b * t0 * t0
    
    def f(t):
        vals = base + a * t + b * t * t
        return np.max(vals) * dx
    
    # Grid search for coarse minimum
    grid = np.linspace(0.0, 1.0, 101)
    fvals = np.array([f(t) for t in grid])
    i_min = np.argmin(fvals)
    
    # Refine with golden section in the neighborhood
    lo = grid[max(0, i_min - 2)]
    hi = grid[min(len(grid)-1, i_min + 2)]
    
    if hi - lo < 1e-12:
        best_t = grid[i_min]
    else:
        # Golden section search (assuming local unimodality near the grid min)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = f(c1)
        f2 = f(c2)
        for _ in range(60):

                break
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = f(c1)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = f(c2)

    candidates = [0.0, 1.0, t0, best_t]
    best_f = min(f(t) for t in candidates)
    best_t = min(candidates, key=f)
    
    return best_t, best_f
```

```python
            b[k] += -1.0
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = np.argmin(h)
            else:
                idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        mask = (h > 0.01) & (h < 0.99)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def smooth_obj(x, n, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    return (m + np.log(wsum) / beta) * dx

def smooth_grad(x, n, beta, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return grad

def lbfgs_optimize(h_init, n, beta, maxiter=500, time_limit=120):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    penalty_coef = 500.0
    t0 = time.time()
    
    def obj(x):
        if (time.time() - t0) > time_limit:
            raise ValueError("timeout")
        base = smooth_obj(x, n, beta, dx)
        penalty = penalty_coef * (np.sum(x) - n/2.0)**2
        return base + penalty
    
    def grad(x):
        base = smooth_grad(x, n, beta, dx)
        p = 2.0 * penalty_coef * (np.sum(x) - n/2.0)
        return base + p * np.ones(n)
    
    try:
        res = minimize(obj, h_init, jac=grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-30, 'gtol': 1e-12,
                               'maxfun': 100000})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h_init.copy()

def coord_descent_1d(h, i, n, dx, corr, offs, k_indices):
    """Find optimal h[i] via vectorized 1D minimization."""
    ncorr = 2 * n - 1
    t0 = h[i]
    
    # Compute coefficients
    a = np.zeros(ncorr)
    b = np.zeros(ncorr)
    
    # off = 0
    a[n-1] += 1.0
    b[n-1] += -1.0
    
    # off > 0: k = n-1+off
    # i+off valid: 0 <= i+off < n
    valid_p = (offs > 0) & (i + offs >= 0) & (i + offs < n)
    if np.any(valid_p):
        a[k_indices[valid_p]] += (1.0 - h[i + offs[valid_p]])
    
    # i-off valid: 0 <= i-off < n
    valid_m = (offs > 0) & (i - offs >= 0) & (i - offs < n)
    if np.any(valid_m):
        a[k_indices[valid_m]] += (-h[i - offs[valid_m]])
    
    # off < 0: k = n-1+off (which is n-1-aoff where aoff=-off > 0)
    # For off < 0, the pair contributions:
    # i-off = i+aoff: if valid, contributes (1-h[i+aoff]) to k=n-1+off
    # i+off = i-aoff: if valid, contributes (-h[i-aoff]) to k=n-1+off
    valid_p2 = (offs < 0) & (i - offs >= 0) & (i - offs < n)
    if np.any(valid_p2):
        a[k_indices[valid_p2]] += (1.0 - h[i - offs[valid_p2]])
    
    valid_m2 = (offs < 0) & (i + offs >= 0) & (i + offs < n)
    if np.any(valid_m2):
        a[k_indices[valid_m2]] += (-h[i + offs[valid_m2]])
    
    # base
    base = corr - a * t0 - b * t0 * t0
    
    # Vectorized grid
    ts = np.linspace(0.0, 1.0, 81)
    vals = base[None, :] + a[None, :] * ts[:, None] + b[None, :] * (ts * ts)[:, None]
    fvals = np.max(vals, axis=1) * dx
    i_min = np.argmin(fvals)
    
    # Golden section refinement
    lo_g = ts[max(0, i_min - 1)]
    hi_g = ts[min(len(ts) - 1, i_min + 1)]
    
    def f_at(t):
        return np.max(base + a * t + b * t * t) * dx
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi_g - gr * (hi_g - lo_g)
    c2 = lo_g + gr * (hi_g - lo_g)
    f1, f2 = f_at(c1), f_at(c2)
    for _ in range(50):
        if hi_g - lo_g < 1e-13:
            break
        if f1 < f2:
            hi_g, c2, f2 = c2, c1, f1
            c1 = hi_g - gr * (hi_g - lo_g)
            f1 = f_at(c1)
        else:
            lo_g, c1, f1 = c1, c2, f2
            c2 = lo_g + gr * (hi_g - lo_g)
            f2 = f_at(c2)
    
    best_t = (lo_g + hi_g) / 2.0
    best_f = f_at(best_t)
    
    # Check special points
    for cand in [0.0, 1.0, t0]:
        fc = f_at(cand)
        if fc < best_f:
            best_f = fc
            best_t = cand
    
    return best_t, best_f

def coord_descent(h_init, n, max_sweeps=100, time_budget=200):
    h = h_init.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    corr = compute_corr(h)
    t_start = time.time()
    rng = np.random.RandomState(777)
    
    offs = np.arange(-(n-1), n)
    k_indices = np.arange(2*n - 1)
    
    for sweep in range(max_sweeps):
        if (time.time() - t_start) > time_budget:
            break
        if remaining_time() < 10:
            break
        
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if (time.time() - t_start) > time_budget:
                break
            if remaining_time() < 5:
                break
            
            t_new, c5_new = coord_descent_1d(h, i, n, dx, corr, offs, k_indices)
            
            if abs(t_new - h[i]) > 1e-12:
                h[i] = t_new
                corr = compute_corr(h)
                c5_check = np.max(corr) * dx
                if c5_check < best_c5 - 1e-15:
                    best_c5 = c5_check
                    improved = True
        
        if not improved and sweep > 10:
            break
    
    return h, best_c5
```
