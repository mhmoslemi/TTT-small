    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    return (m + np.log(np.sum(exps)) / beta) * dx

    n_h = len(h)

def ping_pong_optimize(h, n, max_rounds=20):

    target = n / 2.0

        return np.sum(x) - target

        return np.ones(n)

        if remaining_time() < 6.0:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode='full')
        top_idx = np.argsort(corr)[-4:]
        top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
        t_target = float(corr[np.max(top_idx)]) * 0.99

        def obj_pp(x):

            s = sum(c[ki] for ki in top_idx) / len(top_idx)
            # Add soft penalty for exceeding current max
            pen = 0.0
            for ki in range(len(c)):
                if c[ki] > t_target + 0.5:
                    pen += (c[ki] - t_target - 0.5) ** 2
            return (s + 10.0 * pen) * dx

            h = x
            c = np.correlate(h, 1.0 - h, mode='full')
            grad = np.zeros(n)

            for ki in range(len(c)):

                # Penalty gradient

                    nv = n - mm
                    if nv <= 0:
                        continue
                    grad[:nv] += w * (1.0 - h[mm:mm + nv])
                    grad[mm:mm + nv] -= w * h[:nv]
            return grad * dx

            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-14:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass
    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial solution
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 256
        best_h = make_thue_morse(best_n)
        best_c5 = compute_c5(best_h)
        print("Thue-Morse init: n={}, C5={:.10f}".format(best_n, best_c5))

    # Phase 1: Quick subgradient to enter basin
    print("\n--- Phase 1: Subgradient (n={}) ---".format(best_n))
    best_h, best_c5 = subgrad_optimize(best_h, best_n, max_iter=150, step0=0.1)
    print("  C5={:.10f}, time={:.0f}s".format(best_c5, time.time() - _G_START))

    # Phase 2: SLSQP with increasing beta
    print("\n--- Phase 2: SLSQP polish ---")
    n = best_n
    for beta_val in [100, 300, 1000, 3000, 10000]:
        if remaining_time() < 8.0:
            break
        h_new, c5_new = slsqp_optimize(best_h, n, beta=beta_val, maxiter=250, ftol=1e-14)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new
            print("  beta={}: C5={:.10f}".format(beta_val, best_c5))
        else:
            print("  beta={}: no improvement ({:.10f})".format(beta_val, c5_new))

    # Phase 3: Ping-pong optimization
    print("\n--- Phase 3: Ping-pong (n={}) ---".format(best_n))
    best_h, best_c5 = ping_pong_optimize(best_h, best_n, max_rounds=15)
    print("  C5={:.10f}".format(best_c5))

    # Phase 4: Coordinate descent
    if remaining_time() > 20.0:
        print("\n--- Phase 4: Aggressive coordinate descent ---")
        best_h, best_c5 = aggressive_coord_descent(best_h, best_n, rng, max_sweeps=6)
        print("  C5={:.10f}".format(best_c5))

    # Phase 5: Multi-resolution sweep
    print("\n--- Phase 5: Multi-resolution ---")
    resolutions = [192, 224, 255, 256, 257, 288, 320]
    if best_n not in resolutions:
        resolutions.insert(len(resolutions) // 2, best_n)
    # Sort: try near-best first, then expand
    resolutions = sorted(set(resolutions), key=lambda x: abs(x - best_n))

    for n_try in resolutions:
        if remaining_time() < 25.0:
            break
        if n_try == best_n:
            continue
        t0 = time.time()
        h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        c5_h = compute_c5(h_interp)
        # Subgradient
        h_interp, c5_h = subgrad_optimize(h_interp, n_try, max_iter=100, step0=0.08)
        # SLSQP
        for beta_val in [300, 1000, 5000]:
            if remaining_time() < 10.0:
                break
            h_new, c5_new = slsqp_optimize(h_interp, n_try, beta=beta_val, maxiter=200, ftol=1e-14)
            if c5_new < c5_h:
                c5_h = c5_new
                h_interp = h_new
        # Coord descent
        if remaining_time() > 8.0:
            h_interp, c5_h = aggressive_coord_descent(h_interp, n_try, rng, max_sweeps=3)

        if c5_h < best_c5 - 1e-12:
            best_c5 = c5_h
            best_h = h_interp.copy()
            best_n = n_try
            print("  n={}: C5={:.10f} (improved!)".format(n_try, c5_h))
        else:
            print("  n={}: C5={:.10f} (no imp, {:.0f}s)".format(n_try, c5_h, time.time() - t0))

    # Phase 6: Very high beta SLSQP on best
    if remaining_time() > 10.0:
        print("\n--- Phase 6: High-beta final ---")
        n = best_n
        for beta_val in [5000, 20000, 50000]:
            if remaining_time() < 5.0:
                break
            h_new, c5_new = slsqp_optimize(best_h, n, beta=beta_val, maxiter=400, ftol=1e-15)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new
                print("  beta={}: C5={:.10f}".format(beta_val, best_c5))

    # Phase 7: Final subgradient + coordinate polish
    if remaining_time() > 8.0:
        print("\n--- Phase 7: Final polish ---")
        best_h, best_c5 = subgrad_optimize(best_h, best_n, max_iter=100, step0=0.04)
        print("  After subgrad: C5={:.10f}".format(best_c5))
        if remaining_time() > 6.0:
            best_h, best_c5 = aggressive_coord_descent(best_h, best_n, rng, max_sweeps=5)
            print("  After coord: C5={:.10f}".format(best_c5))

    # Phase 8: One more SLSQP pass
    if remaining_time() > 5.0:
        h_new, c5_new = slsqp_optimize(best_h, best_n, beta=10000, maxiter=300, ftol=1e-15)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}, elapsed={:.0f}s".format(best_n, final_c5, time.time() - _G_START))
    return (best_h, final_c5, best_n)
