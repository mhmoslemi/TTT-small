def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_smooth(h0, n, beta, p, maxiter):
    """SLSQP with p-norm or log-sum-exp smoothing."""
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        if p == -1:  # log-sum-exp
            m = np.max(corr)
            s = corr - m
            return (m + np.log(np.sum(np.exp(np.clip(beta*s, -400, 400))))/beta) * dx
        else:  # p-norm
            return np.sum(corr**p)**(1.0/p) * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)

def lbfgs_pnorm(h0, n, p, maxiter, pen=100.0):

    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        val = np.sum(corr**p)**(1.0/p) * dx
        val += pen * (np.sum(x) - target)**2
        return val
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def grad_corr_k(h, k):
    """Gradient of corr[k] w.r.t. h. corr[k] = sum_j h[j]*(1-h[j+k-n+1]) for valid j."""
    n = len(h)
    grad = np.zeros(n)
    lo1 = max(0, n - 1 - k)
    hi1 = min(n - 1, 2 * n - 2 - k)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1)
        j = idx + k - n + 1
        grad[idx] += (1.0 - h[j])
    lo2 = max(0, lo1 + k - n + 1)
    hi2 = min(n - 1, hi1 + k - n + 1)
    if lo2 <= hi2:
        idx2 = np.arange(lo2, hi2 + 1)
        j2 = idx2 - k + n - 1
        grad[idx2] -= h[j2]
    return grad

def nesterov_subgrad(h0, n, max_iters, tlimit, lr0=0.5):
    """Nesterov-accelerated projected subgradient on max correlation."""
    h = h0.copy()
    v = np.zeros(n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    t = 1.0
    lr = lr0
    for it in range(max_iters):
        if tlimit() < 2.0:
            break
        t += 1.0
        # Nesterov lookahead
        beta = (1.0 + np.sqrt(1 + 4 * t * t)) / (2 * t)
        h_nest = (1 + beta) * h - beta * v
        h_nest = project_feasible(h_nest, n)
        corr = np.correlate(h_nest, 1.0 - h_nest, mode='full')
        k_star = int(np.argmax(corr))
        c5_curr = float(np.max(corr)) * (2.0 / n)
        grad = grad_corr_k(h_nest, k_star)
        # Step
        step = lr / np.sqrt(1 + it * 0.1)
        v_new = h_nest - step * grad * (2.0 / n)
        v_new = project_feasible(v_new, n)
        h = v_new
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h.copy()
    return best_h, best_c5

def slsqp_lse(h0, n, beta, maxiter=300):
    """SLSQP with log-sum-exp smoothing of the max correlation."""
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = corr - m
        e = np.exp(np.clip(beta * s, -500, 500))
        return (m + np.log(np.sum(e)) / beta) * dx

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = corr - m
        exp_s = np.exp(np.clip(beta * s, -500, 500))
        w = exp_s / np.sum(exp_s)
        # Grad of sum w[k] * corr[k] * dx
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] > 1e-8:
                g += w[k] * grad_corr_k(x, k)
        return g * dx

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=obj_grad,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        try:
            res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
            return project_feasible(res.x, n)
        except Exception:
            return project_feasible(h0, n)

def slsqp_pnorm(h0, n, p, maxiter=200):
    """SLSQP with p-norm of correlations."""
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.sum(corr ** p) ** (1.0 / p) * dx

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)

def lbfgs_pnorm(h0, n, p, maxiter=300, pen=50.0):
    """L-BFGS-B with p-norm + penalty."""
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.sum(corr ** p) ** (1.0 / p) * dx + pen * (np.sum(x) - target) ** 2

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)

def water_filling(h, n, tlimit, max_rounds=30, top_k=5):
    """Gradient steps to equalize top-k correlations."""
    h_best = h.copy()

        top_idx = np.argsort(corr)[-top_k:]
        # Only include lags within 20% of max
        thresh = corr[top_idx[-1]] - 0.2 * (corr[top_idx[-1]] - corr[top_idx[0]])
        active = [k for k in top_idx if corr[k] > thresh]
        if not active:
            active = [int(top_idx[-1])]
        grad = np.zeros(n)
        for k in active:
            grad += grad_corr_k(h_best, int(k))
        grad /= len(active)
        # Try multiple step sizes
        best_step = None
        for lr in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
            h_try = project_feasible(h_best - lr * grad * (2.0 / n), n)

            if c_try < c5 - 1e-12:

                    best_step = (h_try, c_try)
        if best_step is not None:
            h_best, c5 = best_step
    return h_best, c5

def make_tm(n):

def coord_descent(h, n, max_sweeps, rng, tlimit, continuous=True):

        if not improved and sweep >= 3:

    return h_best, c5

            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new

        if tlimit() > 10:
            h_new, c_new = water_filling(best_h, n, tlimit, max_rounds=25, top_k=8)
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new

        if tlimit() > 10:
            n = best_n
            h_new, c_new = coord_descent(best_h, n, max_sweeps=40, rng=rng, tlimit=tlimit, continuous=True)
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
