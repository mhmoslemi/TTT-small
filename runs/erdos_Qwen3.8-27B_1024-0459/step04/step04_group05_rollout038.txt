def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            rooms = 1.0 - h
        else:
            rooms = h
        total = rooms.sum()
        if total > 1e-15:
            h += (diff / total) * rooms
        else:
            idx = int(np.argmax(h if diff < 0 else 1.0 - h))
            h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_exact(h_start, n, budget_fn, top_k=5):
    """SLSQP optimizing the sum of top-k correlations (exact, not smoothed)."""
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    state = {'h': h_start.copy()}
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        # Sum of top-k correlations (smooth approximation of max)
        sorted_corr = np.sort(corr)[::-1]
        return sorted_corr[:top_k].sum() * dx
    
    def con_fun(x):
        return x.sum() - target
    
    res = minimize(obj, h_start, method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': con_fun}],
                   options={'maxiter': 300, 'ftol': 1e-14, 'iprint': 0})
    h_new = project_feasible(res.x, n)
    return h_new, compute_c5(h_new)

def slsqp_lse(h_start, n, budget_fn, beta):
    """SLSQP with log-sum-exp smoothing."""
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        exps = np.exp(np.clip(beta * shifted, -500, 500))
        return (m + np.log(np.sum(exps)) / beta) * dx
    
    res = minimize(obj, h_start, method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target}],
                   options={'maxiter': 200, 'ftol': 1e-14, 'iprint': 0})
    h_new = project_feasible(res.x, n)
    return h_new, compute_c5(h_new)

def subgradient_descent(h, n, budget_fn, max_iters=5000, top_k=3):
    """
    Subgradient method on the exact max, with Armijo backtracking.
    """
    dx = 2.0 / n
    target = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    c5_cur = best_c5
    no_improve = 0
    
    step0 = 0.05
    
    def get_subgrad(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top = np.argsort(corr)[-top_k:]
        top = top[np.argsort(corr[top])[::-1]]
        grad = np.zeros(n)
        for k_idx in top:
            offset = int(k_idx) - (n - 1)
            if offset >= 0:
                n_pairs = n - offset
                if n_pairs <= 0:
                    continue
                grad[:n_pairs] += (1.0 - h[offset:offset + n_pairs])
                grad[offset:] -= h[:n_pairs]
            else:
                d = -offset
                n_pairs = n - d
                if n_pairs <= 0:
                    continue
                grad[:n_pairs] -= h[d:d + n_pairs]
                grad[d:] += (1.0 - h[:n_pairs])
        return grad * dx / top_k
    
    for it in range(max_iters):
        if budget_fn() < 3.0:
            break
        
        grad = get_subgrad(h)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        direction = -grad / gnorm
        
        # Armijo backtracking
        alpha = step0
        improved = False
        for _ in range(10):
            h_trial = project_feasible(h + alpha * direction * gnorm, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_cur - 1e-15:
                h = h_trial
                c5_cur = c5_trial
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h.copy()
                    no_improve = 0
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve > 50:
                # Add noise
                noise = np.random.randn(n) * 0.01
                h = project_feasible(h + noise, n)
                c5_cur = compute_c5(h)
                no_improve = 0
    
    return best_h, best_c5

def threshold_and_refine(h, n, budget_fn):
    """Threshold to binary, do local search, then interpolate back for continuous refinement."""
    h_bin = (h > 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(h_bin.sum())
    target_ones = n // 2
    if n_ones > target_ones:
        # Flip some 1s to 0s
        one_indices = np.where(h_bin == 1.0)[0]
        # Flip the ones with smallest "importance" (close to 0.5 in original h)
        importance = h[one_indices]
        flip_order = np.argsort(importance)[:n_ones - target_ones]
        h_bin[one_indices[flip_order]] = 0.0
    elif n_ones < target_ones:
        zero_indices = np.where(h_bin == 0.0)[0]
        importance = 1.0 - h[zero_indices]
        flip_order = np.argsort(importance)[:target_ones - n_ones]
        h_bin[zero_indices[flip_order]] = 1.0
    
    c5_bin = compute_c5(h_bin)
    
    # Binary local search
    h_bin, c5_bin = directed_local_search(h_bin, n, budget_fn, max_iters=200000)
    
    # Return the binary solution (it's already in [0,1] and sums to n/2)
    return h_bin, c5_bin

- At lag 0: corr = 0 (always)
- At lag k: corr = #{i: h[i]=1, h[i+k]=0} for i=0..n-k-1

We want to minimize max_k #{i: h[i]=1, h[i+k]=0} * dx

The value #{i: h[i]=1, h[i+k]=0} + #{i: h[i]=1, h[i+k]=1} = #{i: h[i]=1} for i=0..n-k-1

#{i: h[i]=1, h[i+k]=0} = S_k - #{i: h[i]=1, h[i+k]=1} = S_k - A_k

where A_k is the autocorrelation at lag k.

- A_k to be large (1s should overlap with 1s at all lags)

This is a trade-off. The optimal arrangement balances these.

    dx = 2.0 / n

    return float(np.max(corr)) * dx


def compute_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            rooms = 1.0 - h
        else:
            rooms = h
        total = rooms.sum()
        if total > 1e-15:
            h += (diff / total) * rooms
        else:
            idx = int(np.argmax(h if diff < 0 else 1.0 - h))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    return h


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    return project_feasible(h, n)

    h = np.zeros(n)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        h[i] = 1.0 if (i * gr) % 1.0 < 0.5 else 0.0
    return project_feasible(h, n)

def slsqp_topk(h_start, n, top_k=5, maxiter=200):
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj_func(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        sorted_corr = np.sort(corr)[::-1]
        return sorted_corr[:top_k].sum() * dx

    def con_fun(x):
        return x.sum() - target

    try:
        res = minimize(obj_func, h_start, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new)
    except Exception:
        return h_start, compute_c5(h_start)


def slsqp_lse(h_start, n, beta, maxiter=200):
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj_func(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        exps = np.exp(np.clip(beta * shifted, -500, 500))
        return (m + np.log(np.sum(exps)) / beta) * dx

    def con_fun(x):
        return x.sum() - target

    try:
        res = minimize(obj_func, h_start, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new)
    except Exception:
        return h_start, compute_c5(h_start)

def subgradient_descent(h, n, budget_fn, max_iters=3000, top_k=3):
    """Subgradient method with adaptive step size on the exact max."""
    dx = 2.0 / n
    target = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    c5_cur = best_c5
    no_improve = 0
    step = 0.03

    for it in range(max_iters):
        if budget_fn() < 3.0:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
        grad = np.zeros(n)
        for k_idx in top_idx:
            offset = int(k_idx) - (n - 1)
            if offset >= 0:
                n_pairs = n - offset
                if n_pairs <= 0:
                    continue
                grad[:n_pairs] += (1.0 - h[offset:offset + n_pairs])
                grad[offset:] -= h[:n_pairs]
            else:
                d = -offset
                n_pairs = n - d
                if n_pairs <= 0:
                    continue
                grad[:n_pairs] -= h[d:d + n_pairs]
                grad[d:] += (1.0 - h[:n_pairs])
        grad *= dx / top_k

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        improved = False
        alpha = step
        for _ in range(8):
            h_trial = project_feasible(h - alpha * grad, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_cur - 1e-15:
                h = h_trial
                c5_cur = c5_trial
                if c5_cur < best_c5 - 1e-15:
                    best_c5 = c5_cur
                    best_h = h.copy()
                    no_improve = 0
                improved = True
                break
            alpha *= 0.5

        if not improved:
            step *= 0.9
            no_improve += 1
            if no_improve > 30:
                noise = np.random.randn(n) * 0.02
                h = project_feasible(h + noise, n)
                c5_cur = compute_c5(h)
                no_improve = 0
                step = 0.03
        else:
            step = min(0.1, step * 1.05)

    return best_h, best_c5

            h_b, c5_b = directed_swap_search(h_b, best_n, budget_fn, max_rounds=1000)
            if c5_b < best_c5 - 1e-12:
                best_c5, best_h = c5_b, h_b
                print("  {} improved: C5={:.10f}".format(name, best_c5))
            else:
                print("  {}: C5={:.10f}".format(name, c5_b))

    # Phase 7: More directed swap search at best
    if budget_fn() > 15:
        print("\n--- Phase 7: More swap search ---")
        h_bin = threshold_to_binary(best_h, best_n)
        h_bin, c5_bin = directed_swap_search(h_bin, best_n, budget_fn, max_rounds=5000)
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin, c5_bin
        print("  C5={:.10f}".format(best_c5))

    # Phase 8: Final SLSQP
    if budget_fn() > 10:
        print("\n--- Phase 8: Final SLSQP ---")
        h_new, c5_new = slsqp_topk(best_h, best_n, top_k=3, maxiter=150)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        print("  C5={:.10f}".format(best_c5))

    # Phase 9: Final binary polish
    if budget_fn() > 8:
        h_bin = threshold_to_binary(best_h, best_n)
        h_bin, c5_bin = directed_swap_search(h_bin, best_n, budget_fn, max_rounds=3000)
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin, c5_bin

    # Phase 10: ULTRA final - try a few more resolutions near the best
    if budget_fn() > 10:
        nearby = [best_n - 8, best_n - 4, best_n + 4, best_n + 8]
        for n_try in nearby:
            if n_try < 100 or n_try % 2 != 0:
                continue
            if budget_fn() < 5:
                break
            h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
            h_bin = threshold_to_binary(h_interp, n_try)
            h_bin, c5_bin = directed_swap_search(h_bin, n_try, budget_fn, max_rounds=500)
            if c5_bin < best_c5 - 1e-12:
                best_c5, best_h, best_n = c5_bin, h_bin, n_try
                print("  Nearby n={} improved: C5={:.10f}".format(n_try, best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
