```python
def binary_anneal(h_binary, n, max_iters=100000, t0=1.0, t1=0.01):
    """Simulated annealing on binary sequences.
    h_binary: numpy array of 0s and 1s with sum = n/2
    Returns: best binary solution found
    """
    h = h_binary.copy()
    best_h = h.copy()
    best_score = float(np.max(np.correlate(h, 1-h, mode='full')))
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        T = t0 * (t1/t0)**(it/max_iters)
        # Random swap: pick a random 1 and a random 0, swap them
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        # Swap
        h[i], h[j] = h[j], h[i]
        
        # Evaluate
        corr = np.correlate(h, 1-h, mode='full')
        score = float(np.max(corr))
        
        if score < best_score:
            best_score = score
            best_h = h.copy()
        elif score > best_score:
            # Accept with probability exp(-(score-best_score)/T)
            if np.random.rand() < np.exp(-(score - best_score)/T):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                continue
            else:
                h[i], h[j] = h[j], h[i]
        
        ones = np.where(h == 1)[0]
```
