            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            rooms = 1.0 - h
            order = np.argsort(rooms)[::-1]
        else:
            rooms = h
            order = np.argsort(rooms)
        for idx in order:
            if diff > 0:
                add = min(diff, rooms[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, rooms[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def time_remaining():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h


def make_golden_ratio(n):
    h = np.zeros(n)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return h


def _ones_n(x):
    return np.ones(_G_N)


def _con_fun(x):
    return np.sum(x) - _G_TGT


def _lse_objective(x):
    n = _G_N
    dx = _G_DX
    beta = _G_BETA
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -300, 300))
    s = np.sum(exps)
    return (m + np.log(s) / beta) * dx


def _lse_gradient(x):
    n = _G_N
    dx = _G_DX
    beta = _G_BETA
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -300, 300))
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-11:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs <= 0:
                continue
            grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
            grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            a = -offset
            npairs = n - a
            if npairs <= 0:
                continue
            grad[:npairs] += w * (1.0 - x[a:a + npairs])
            grad[a:a + npairs] -= w * x[:npairs]
    return grad * dx


def run_slsqp_beta(h, n, beta, maxiter):
    global _G_N, _G_BETA, _G_TGT, _G_DX
    _G_N = n
    _G_BETA = float(beta)
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _ones_n}]
    try:
        res = minimize(_lse_objective, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       jac=_lse_gradient,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return project_feasible(h, n)


def do_slsqp_sequence(h, n, betas, maxiters=None):
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for i, beta in enumerate(betas):
        if time_remaining() < 8:
            break
        if maxiters is None:
            if beta < 200:
                mi = 300
            elif beta < 2000:
                mi = 250
            else:
                mi = 200
        else:
            mi = maxiters
        h_cur = run_slsqp_beta(h_cur, n, beta, mi)
        c5_new = compute_c5(h_cur)
        if c5_new < c5_cur - 1e-15:
            c5_cur = c5_new
    return h_cur, c5_cur


def active_subgrad(h, top_k=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
    grad = np.zeros(n)
    for k_idx in top_idx:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs <= 0:
                continue
            grad[:npairs] += (1.0 - h[offset:offset + npairs])
            grad[offset:] -= h[:npairs]
        else:
            a = -offset
            npairs = n - a
            if npairs <= 0:
                continue
            grad[:npairs] += (1.0 - h[a:a + npairs])
            grad[a:] -= h[:npairs]
    return grad * dx, top_idx, corr


def nesterov_pg(h, n, max_iters, top_k, lr0, min_lr):
    h = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    v = np.zeros(n)
    mu = 1.0
    no_improve = 0
    for it in range(max_iters):
        if time_remaining() < 6:
            break
        if it % 50 == 0:
            mu = (1 + np.sqrt(1 + 4 * mu * mu)) / 2
        y = h + (mu - 1) / (mu + 1) * v
        y = np.clip(y, 0.0, 1.0)
        grad, top_idx, corr = active_subgrad(y, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        grad_n = grad / gnorm
        lr = max(lr0 / (1.0 + 0.005 * it), min_lr)
        improved = False
        for _ in range(6):
            h_new = project_feasible(h - lr * grad_n, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-16:
                v_new = h_new - h
                h = h_new
                c5_best = c5_new
                h_best = h_new.copy()
                v = v_new
                no_improve = 0
                improved = True
                break
            lr *= 0.5
        if not improved:
            no_improve += 1
            if no_improve > 40:
                h_pert = project_feasible(h + np.random.randn(n) * 0.01, n)
                c5_pert = compute_c5(h_pert)
                if c5_pert < c5_best:
                    h = h_pert
                    c5_best = c5_pert
                    h_best = h_pert.copy()
                    v = np.zeros(n)
                    no_improve = 0
                else:
                    if no_improve > 200:
                        break
                    v = np.zeros(n)
    return h_best, c5_best


def binary_2opt(h, n, rng, max_sweeps=10):
    h = h.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    for sweep in range(max_sweeps):
        if time_remaining() < 5:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if time_remaining() < 3:
                break
            old = h[i]
            if old <= 1e-7 or old >= 1.0 - 1e-7:
                continue
            for new_v in [0.0, 1.0]:
                if abs(new_v - old) < 0.3:
                    continue
                h[i] = new_v
                h = project_feasible(h, n)
                c5_test = compute_c5(h)
                if c5_test < c5_best - 1e-14:
                    c5_best = c5_test
                    h_best = h.copy()
                    any_imp = True
                else:
                    h = h_best.copy()
        if not any_imp and sweep >= 3:
            break
    return h_best, c5_best


def coord_desc(h, n, rng, max_sweeps=8):
    h = h.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    for sweep in range(max_sweeps):
        if time_remaining() < 4:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if time_remaining() < 2:
                break
            old_v = h[i]
            candidates = [0.0, 1.0, 0.1, 0.9, 0.5]
            best_trial = old_v
            best_val = c5_best
            for cv in candidates:
                if abs(cv - old_v) < 1e-6:
                    continue
                h[i] = cv
                h = project_feasible(h, n)
                c5_t = compute_c5(h)
                if c5_t < best_val - 1e-14:
                    best_val = c5_t
                    best_trial = cv
                else:
                    h[i] = old_v
                    h = h_best.copy()
            if best_trial != old_v:
                h[i] = best_trial
                h = project_feasible(h, n)
                h_best = h.copy()
                c5_best = best_val
                any_imp = True
        if not any_imp and sweep >= 2:
            break
    return h_best, c5_best


def multi_resolution(h_best, n_best, c5_best, n_target, rng):
    if time_remaining() < 30:
        return h_best, c5_best, n_best
    h_interp = project_feasible(interp_to(h_best, n_best, n_target), n_target)
    c5_interp = compute_c5(h_interp)
    h_cur, c5_cur = do_slsqp_sequence(h_interp, n_target, [50, 200, 800, 3000], maxiters=100)
    if c5_cur < c5_best - 1e-14:
        h_target, c5_target = nesterov_pg(h_cur, n_target, 500, 5, 0.03, 0.001)
        if c5_target < c5_best - 1e-14:
            return h_target, c5_target, n_target
        return h_cur, c5_cur, n_target
    else:
        if c5_interp < c5_best - 1e-14:
            return h_interp, c5_interp, n_target
        return h_best, c5_best, n_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET, _G_N, _G_BETA, _G_TGT, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
    except Exception:
        parent_h = make_thue_morse(256)
        parent_n = 256

    _G_N = parent_n
    _G_TGT = parent_n / 2.0
    _G_DX = 2.0 / parent_n

    best_h = project_feasible(parent_h, parent_n)
    best_c5 = compute_c5(best_h)
    best_n = parent_n
    print("Initial: n={}, C5={:.10f}".format(best_n, best_c5))

    # Phase 1: SLSQP with analytical gradient, increasing beta
    print("Phase 1: SLSQP low beta")
    best_h, best_c5 = do_slsqp_sequence(best_h, best_n, [30, 100, 300, 1000])
    print("  C5={:.10f}".format(best_c5))

    # Phase 2: Nesterov PG on max objective
    if time_remaining() > 20:
        print("Phase 2: Nesterov PG top-3")
        best_h, best_c5 = nesterov_pg(best_h, best_n, 1000, 3, 0.04, 0.002)
        print("  C5={:.10f}".format(best_c5))

    # Phase 3: SLSQP medium beta
    if time_remaining() > 20:
        print("Phase 3: SLSQP medium beta")
        best_h, best_c5 = do_slsqp_sequence(best_h, best_n, [3000, 8000])
        print("  C5={:.10f}".format(best_c5))

    # Phase 4: Nesterov PG top-5
    if time_remaining() > 20:
        print("Phase 4: Nesterov PG top-5")
        best_h, best_c5 = nesterov_pg(best_h, best_n, 1500, 5, 0.02, 0.001)
        print("  C5={:.10f}".format(best_c5))

    # Phase 5: SLSQP high beta
    if time_remaining() > 20:
        print("Phase 5: SLSQP high beta")
        best_h, best_c5 = do_slsqp_sequence(best_h, best_n, [15000, 40000, 80000])
        print("  C5={:.10f}".format(best_c5))

    # Phase 6: Try n=512
    if time_remaining() > 60:
        print("Phase 6: Try n=512")
        h512, c5_512, n512 = multi_resolution(best_h, best_n, best_c5, 512, rng)
        if n512 == 512 and c5_512 < best_c5:
            best_h, best_c5, best_n = h512, c5_512, 512
            print("  512 improved: C5={:.10f}".format(best_c5))
        else:
            print("  512 no improve: C5={:.10f}".format(c5_512))

    # Phase 7: Try n=384
    if time_remaining() > 40:
        print("Phase 7: Try n=384")
        h384, c5_384, n384 = multi_resolution(best_h, best_n, best_c5, 384, rng)
        if n384 == 384 and c5_384 < best_c5:
            best_h, best_c5, best_n = h384, c5_384, 384
            print("  384 improved: C5={:.10f}".format(best_c5))
        else:
            print("  384 no improve: C5={:.10f}".format(c5_384))

    # Phase 8: Nesterov PG at best resolution
    if time_remaining() > 20:
        print("Phase 8: Nesterov PG top-10")
        best_h, best_c5 = nesterov_pg(best_h, best_n, 2000, 10, 0.01, 0.0005)
        print("  C5={:.10f}".format(best_c5))

    # Phase 9: SLSQP very high beta final
    if time_remaining() > 15:
        print("Phase 9: SLSQP final")
        best_h, best_c5 = do_slsqp_sequence(best_h, best_n, [100000, 300000])
        print("  C5={:.10f}".format(best_c5))

    # Phase 10: Coordinate descent
    if time_remaining() > 15:
        print("Phase 10: Coord descent")
        best_h, best_c5 = coord_desc(best_h, best_n, rng, max_sweeps=6)
        print("  C5={:.10f}".format(best_c5))

    # Phase 11: Binary 2-opt
    if time_remaining() > 10:
        print("Phase 11: Binary 2opt")
        best_h, best_c5 = binary_2opt(best_h, best_n, rng, max_sweeps=6)
        print("  C5={:.10f}".format(best_c5))

    # Phase 12: Final SLSQP polish
    if time_remaining() > 10:
        _G_N = best_n
        _G_TGT = best_n / 2.0
        _G_DX = 2.0 / best_n
        best_h = run_slsqp_beta(best_h, best_n, 200000, 100)
        best_c5 = compute_c5(best_h)
        print("Phase 12: Final polish C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("FINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
