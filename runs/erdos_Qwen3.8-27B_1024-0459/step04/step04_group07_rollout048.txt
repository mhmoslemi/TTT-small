def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(src, sn, dn):

        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c else 0.0
    return h

    h_best = h.copy()
    c5_best = compute_c5(h)

            for i in range(n):
                grad[i] += (1.0 - 2.0 * h[i])
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        grad = grad / grad_norm * lr * dx
        h_new = h - grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5 - 1e-15:
            c5 = c5_new
            h = h_new.copy()
            h_best = h.copy()
        else:
            lr *= 0.5
            if lr < 1e-7:
                break
    return h_best, c5

def spectral_perturb(h, n, rng, strength=0.4):
    H = np.fft.rfft(h)
    phases = np.angle(H) + rng.randn(len(H)) * strength * 0.08
    mags = np.abs(H) * (1.0 + rng.randn(len(H)) * strength * 0.04)
    new_H = mags * np.exp(1j * phases)
    new_h = np.fft.irfft(new_H, n)
    return np.clip(new_h, 0.0, 1.0)

def multi_beta_slsqp(h, n, betas, maxiter_each=350, tlimit=None):
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for bv in betas:
        if tlimit is not None and tlimit() < 8.0:
            break
        h_new = slsqp_polish(h_cur, n, bv, maxiter_each)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-14:
            c5_cur = c5_new
            h_cur = h_new
        if tlimit is not None and tlimit() < 5.0:
            break
    return h_cur, c5_cur

def focused_rounds(h, n, tlimit, max_rounds=40):
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    for rnd in range(max_rounds):
        if tlimit() < 5.0:
            break
        h_new = focus_slsqp(h_best, n, 1000.0, 250)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-12:
            c5_best = c5_new
            h_best = h_new.copy()
    return h_best, c5_best

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tlimit():
        return max(0.0, float(budget_s) - (time.time() - t0) - 10.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent = np.array(initial_h_values, dtype=float).copy()
        pn = len(parent)
        parent = project_feasible(parent, pn)
        pc5 = compute_c5(parent)
        best_h = parent.copy()
        best_c5 = pc5
        best_n = pn
        print("Parent: n=%d C5=%.8f" % (pn, pc5))
    except Exception:
        pn = 256
        parent = make_tm(pn)
        parent = project_feasible(parent, pn)
        pc5 = compute_c5(parent)
        best_h = parent.copy()
        best_c5 = pc5
        best_n = pn
        print("TM init: n=%d C5=%.8f" % (pn, pc5))

    # Phase 1: SLSQP multi-beta polish at parent resolution
    if tlimit() > 60.0:
        n = best_n
        print("=== Phase 1: SLSQP polish n=%d ===" % n)
        h_cur, c5_cur = multi_beta_slsqp(best_h, n, [40, 150, 500, 1500, 4000], 350, tlimit)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        print("  After SLSQP: C5=%.8f" % best_c5)

    # Phase 2: Focused SLSQP rounds (target the active lag)
    if tlimit() > 60.0:
        n = best_n
        print("=== Phase 2: focused rounds n=%d ===" % n)
        h_cur, c5_cur = focused_rounds(best_h, n, tlimit, 30)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        print("  After focused: C5=%.8f" % best_c5)

    # Phase 3: Subgradient polish (exact max, different from SLSQP)
    if tlimit() > 30.0:
        n = best_n
        print("=== Phase 3: subgradient polish n=%d ===" % n)
        h_cur, c5_cur = subgradient_polish(best_h, n, tlimit, max_steps=300, lr=0.05)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        print("  After subgradient: C5=%.8f" % best_c5)

    # Phase 4: Pair-flip tabu search (combinatorial exploration)
    if tlimit() > 40.0:
        n = best_n
        print("=== Phase 4: pair-flip tabu n=%d ===" % n)
        h_cur, c5_cur = pair_flip_tabu(best_h, n, rng, tlimit, max_iters=80000, tabu_len=200)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        print("  After tabu: C5=%.8f" % best_c5)

    # Phase 5: Higher resolution exploration
    if tlimit() > 120.0:
        print("=== Phase 5: multi-resolution ===")
        parent_n = best_n
        for n_try in [384, 512, 640, 768, 320]:
            if tlimit() < 30.0:
                break
            if n_try == parent_n:
                continue
            h_interp = project_feasible(interp_h(best_h, parent_n, n_try), n_try)
            c5_interp = compute_c5(h_interp)
            h_tm = project_feasible(make_tm(n_try), n_try)
            c5_tm = compute_c5(h_tm)
            inits = [(h_interp, c5_interp), (h_tm, c5_tm)]
            inits.sort(key=lambda x: x[1])
            h_cur = inits[0][0].copy()
            c5_cur = inits[0][1]
            betas = [80, 300, 1000, 3000]
            h_cur, c5_cur = multi_beta_slsqp(h_cur, n_try, betas, 300, tlimit)
            if tlimit() > 15.0:
                h_cur, c5_cur = focused_rounds(h_cur, n_try, tlimit, 15)
            if c5_cur < best_c5 - 1e-10:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n_try
                print("  n=%d: C5=%.8f (improved!)" % (n_try, c5_cur))
            else:
                print("  n=%d: C5=%.8f" % (n_try, c5_cur))

    # Phase 6: Spectral perturbation + re-optimization (escape local optima)
    if tlimit() > 60.0:
        n = best_n
        print("=== Phase 6: spectral perturb n=%d ===" % n)
        for attempt in range(8):
            if tlimit() < 12.0:
                break
            h_p = spectral_perturb(best_h, n, rng, strength=0.3 + 0.1 * attempt)
            h_p = project_feasible(h_p, n)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5 * 1.18:
                h_p, c5_p = multi_beta_slsqp(h_p, n, [200, 1000, 4000], 250, tlimit)
                if tlimit() > 5.0:
                    h_p, c5_p = subgradient_polish(h_p, n, tlimit, 100, 0.03)
                if c5_p < best_c5 - 1e-11:
                    best_c5 = c5_p
                    best_h = h_p.copy()
                    print("  Spectral %d: C5=%.8f" % (attempt, c5_p))

    # Phase 7: Additional pair-flip at best resolution
    if tlimit() > 30.0:
        n = best_n
        print("=== Phase 7: second tabu pass n=%d ===" % n)
        h_cur, c5_cur = pair_flip_tabu(best_h, n, rng, tlimit, max_iters=60000, tabu_len=400)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        print("  After second tabu: C5=%.8f" % best_c5)

    # Phase 8: Ultra-fine SLSQP + subgradient polish
    if tlimit() > 20.0:
        n = best_n
        print("=== Phase 8: final polish n=%d ===" % n)
        for bv in [8000, 20000, 50000]:
            if tlimit() < 5.0:
                break
            h_new = slsqp_polish(best_h, n, bv, 500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
        h_cur, c5_cur = subgradient_polish(best_h, n, tlimit, 200, 0.01)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        print("  Final polish: C5=%.8f" % best_c5)

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 is not None and float(ec5) < final_c5:
            final_c5 = float(ec5)
    except Exception:
        pass

    global _G_FOCUS_K, _G_FOCUS_W
    _G_FOCUS_K = -1
    _G_FOCUS_W = 0.0
    print("FINAL: n=%d C5=%.8f" % (best_n, final_c5))
    return (best_h, final_c5, best_n)
