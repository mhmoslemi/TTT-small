def slsqp_optimize(h, n, beta, maxiter=600):
    dx_v = 2.0 / n
    target = n / 2.0

    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -60.0, 60.0)
        return (m + np.log(np.sum(np.exp(cl))) / beta) * dx_v

    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -60.0, 60.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)

                continue

                    continue

                g[:np_] -= wk * x[off:off + np_]

                if np_ <= 0:
                    continue
                g[:np_] += wk * (1.0 - x[ao:ao + np_])

        return np.ones(n)

def directed_sa(h, n, time_limit=60, rng=None):

            h[i] = 0
            h[j] = 1
            tabu[(min(i, j), max(i, j))] = n_steps
            cur_c5 += best_delta
            n_steps += 1
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

    return best_h, best_c5


def structured_init(n, rng):
    """Create a structured initial guess with block patterns."""
    h = np.zeros(n)
    # Random block structure
    n_blocks = rng.randint(4, 12)
    block_size = n // n_blocks
    for b in range(n_blocks):
        start = b * block_size
        end = min(start + block_size, n)
        if rng.rand() < 0.5:
            h[start:end] = rng.uniform(0.6, 1.0)
        else:
            h[start:end] = rng.uniform(0.0, 0.4)
    h = project_feasible(h, n)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP with increasing beta on parent resolution
    for beta in [100, 500, 2000, 8000, 20000, 50000]:
        if remaining() < 10:
            break
        h_new = slsqp_optimize(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n
            h = h_new.copy()

    # Phase 2: Top-k equalization
    if remaining() > 10:
        h_eq, c5_eq = top_equalize(h, n, top_k=6, max_iters=3000)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 3: Pair equalization (top 2)
    if remaining() > 10:
        h_pk, c5_pk = pair_equalize(h, n, max_iters=2000)
        if c5_pk < best_c5:
            best_c5 = c5_pk
            best_h = h_pk.copy()
            h = h_pk.copy()

    # Phase 4: Another round of SLSQP at final beta
    if remaining() > 10:
        h_new = slsqp_optimize(h, n, 80000, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Top-k equalization again (finer)
    if remaining() > 10:
        h_eq, c5_eq = top_equalize(h, n, top_k=4, max_iters=3000)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 6: Try higher resolution
    if remaining() > 60:
        n_hi = best_n * 2
        h_hi = interp_to(best_h, best_n, n_hi)
        h_hi = project_feasible(h_hi, n_hi)
        c5_hi = compute_c5(h_hi)

        for beta in [500, 3000, 15000]:
            if remaining() < 15:
                break
            h_hi_new = slsqp_optimize(h_hi, n_hi, beta, maxiter=300)
            c5_hi_new = compute_c5(h_hi_new)
            if c5_hi_new < c5_hi:
                h_hi = h_hi_new
                c5_hi = c5_hi_new

        if remaining() > 10:
            h_hi, c5_hi = top_equalize(h_hi, n_hi, top_k=5, max_iters=2000)

        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_hi

    # Phase 7: Try lower resolution (fewer points = coarser but potentially better fit)
    if remaining() > 60:
        n_lo = best_n // 2
        if n_lo >= 128:
            h_lo = interp_to(best_h, best_n, n_lo)
            h_lo = project_feasible(h_lo, n_lo)
            c5_lo = compute_c5(h_lo)

            for beta in [1000, 5000, 20000]:
                if remaining() < 10:
                    break
                h_lo_new = slsqp_optimize(h_lo, n_lo, beta, maxiter=400)
                c5_lo_new = compute_c5(h_lo_new)
                if c5_lo_new < c5_lo:
                    h_lo = h_lo_new
                    c5_lo = c5_lo_new

            if remaining() > 10:
                h_lo, c5_lo = top_equalize(h_lo, n_lo, top_k=5, max_iters=2000)

            if c5_lo < best_c5:
                best_c5 = c5_lo
                best_h = h_lo.copy()
                best_n = n_lo

    # Phase 8: Binary SA on best solution
    if remaining() > 40:
        n_cur = best_n
        h_bin = (best_h > 0.5).astype(float)
        # Ensure exact n/2 ones
        n_ones = int(np.sum(h_bin))
        if n_ones != n_cur // 2:
            sort_idx = np.argsort(-best_h)
            h_bin = np.zeros(n_cur)
            h_bin[sort_idx[:n_cur // 2]] = 1.0

        sa_time = min(80, max(10, int(remaining() * 0.3)))
        h_b, c5_b = directed_sa(h_bin, n_cur, time_limit=sa_time, rng=rng)

        # Refine the binary solution with continuous optimization
        if c5_b < best_c5 + 0.01:
            h_ref = h_b.copy()
            for beta in [5000, 20000]:
                if remaining() < 8:
                    break
                h_ref = slsqp_optimize(h_ref, n_cur, beta, maxiter=300)
            c5_ref = compute_c5(h_ref)
            if remaining() > 8:
                h_ref, c5_ref = top_equalize(h_ref, n_cur, top_k=5, max_iters=1500)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
                best_n = n_cur

        # Also try from the raw binary value
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n_cur

    # Phase 9: Multi-start with structured initializations
    if remaining() > 40:
        n_cur = best_n
        for trial in range(3):
            if remaining() < 15:
                break
            h_try = structured_init(n_cur, np.random.RandomState(seed + 100 + trial))
            # Quick SLSQP
            h_try = slsqp_optimize(h_try, n_cur, 3000, maxiter=300)
            c5_try = compute_c5(h_try)
            if remaining() > 8:
                h_try, c5_try = top_equalize(h_try, n_cur, top_k=5, max_iters=1500)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_cur

    # Phase 10: Final intensive refinement
    n = best_n
    h = best_h.copy()

    # Final SLSQP
    if remaining() > 8:
        h_new = slsqp_optimize(h, n, 100000, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final top-k equalization
    if remaining() > 8:
        h_eq, c5_eq = top_equalize(h, n, top_k=4, max_iters=4000)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Final pair equalization
    if remaining() > 5:
        h_pk, c5_pk = pair_equalize(h, n, max_iters=3000)
        if c5_pk < best_c5:
            best_c5 = c5_pk
            best_h = h_pk.copy()

    # Project to ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-12:
        # Revert to pre-projection if projection hurt
        pass
    else:
        best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
