                i = ones_list[rng.randint(len(ones_list))]
        else:
            i = ones_list[rng.randint(len(ones_list))]
        
        # Choose j
        j = zeros_list[rng.randint(len(zeros_list))]
        
        if i == j:
            continue
        
        # Swap
        h[i] = 0.0
        h[j] = 1.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_max = float(new_corr.max())
        
        delta_c5 = (new_max - cur_max) * dx
        if delta_c5 <= -1e-15 or rng.rand() < np.exp(-delta_c5 / max(T, 1e-10)):
            cur_max = new_max
            corr = new_corr
            # Update lists
            if i in ones_list:
                ones_list.remove(i)
            ones_list.append(j)
            if j in zeros_list:
                zeros_list.remove(j)
            zeros_list.append(i)
            
            c5_now = cur_max * dx
            if c5_now < best_c5 - 1e-15:
                best_c5 = c5_now
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        # Cleanup
        if n_iter % 20000 == 0:
            ones_list = np.where(h == 1)[0].tolist()
            zeros_list = np.where(h == 0)[0].tolist()
    
    return best_h, best_c5

# --- Greedy local search (no SA, pure descent) ---

def greedy_binary(h, n, time_limit, rng, max_scan=200):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(corr.max())
    dx = 2.0 / n
    best_c5 = cur_max * dx
    best_h = h.copy()
    
    ones_list = np.where(h == 1)[0].tolist()
    zeros_list = np.where(h == 0)[0].tolist()
    
    t_start = time.time()
    it = 0
    
    while time.time() - t_start < time_limit:
        if remaining() < 5:
            break
        it += 1
        
        k1 = int(np.argmax(corr))
        offset = k1 - (n - 1)
        
        # Try up to max_scan random swaps, accept first improvement
        found = False
        for attempt in range(max_scan):
            if offset > 0:
                cand = [p for p in ones_list if p >= offset and h[p - offset] == 0]
                if cand:
                    i = cand[rng.randint(len(cand))]
                else:
                    i = ones_list[rng.randint(len(ones_list))]
            elif offset < 0:
                m = -offset
                cand = [p for p in ones_list if p + m < n and h[p + m] == 0]
                if cand:
                    i = cand[rng.randint(len(cand))]
                else:
                    i = ones_list[rng.randint(len(ones_list))]
            else:
                i = ones_list[rng.randint(len(ones_list))]
            
            j = zeros_list[rng.randint(len(zeros_list))]
            if i == j:
                continue
            
            h[i] = 0.0
            h[j] = 1.0
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(new_corr.max())
            
            if new_max < cur_max - 1e-14:
                cur_max = new_max
                corr = new_corr
                ones_list.remove(i)
                ones_list.append(j)
                zeros_list.remove(j)
                zeros_list.append(i)
                c5_now = cur_max * dx
                if c5_now < best_c5:
                    best_c5 = c5_now
                    best_h = h.copy()
                found = True
                break
            else:
                h[i] = 1.0
                h[j] = 0.0
        
        if not found:
            # Perturbation: random swap to escape
            i = ones_list[rng.randint(len(ones_list))]
            j = zeros_list[rng.randint(len(zeros_list))]
            h[i] = 0.0
            h[j] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            cur_max = float(corr.max())
            ones_list.remove(i)
            ones_list.append(j)
            zeros_list.remove(j)
            zeros_list.append(i)
    
    return best_h, best_c5
