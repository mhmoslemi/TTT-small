        for step in [0.01, 0.005, 0.002]:
            h_new2 = subgradient_step(h_new2, n, step, rng)
        c2 = compute_c5(h_new2)
        
        # L-BFGS-B
        h_new3 = lbfgsb_optimize(h_best, n, 500, maxiter=150, pen=200.0, focus_lags=top_indices)
        c3 = compute_c5(h_new3)
        
        best_c = min(c1, c2, c3)
        best_h_new = h_new1 if best_c == c1 else (h_new2 if best_c == c2 else h_new3)
        
        if best_c < c5 - 1e-11:
            c5 = best_c
            h_best = best_h_new
        else:
            if rnd > 5:
                break
    
    return h_best, c5

def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET, _G_BETA, _G_TGT, _G_PEN, _G_DX, _G_N, _G_FOCUS_LAGS
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    _G_BETA = 100.0
    _G_TGT = 0.0
    _G_PEN = 0.0
    _G_DX = 0.0
    _G_N = 0
    _G_FOCUS_LAGS = None
    
    rng = np.random.RandomState(seed)
    
    def tlimit():
        return max(0.0, _G_BUDGET - (time.time() - _G_START) - 4.0)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = make_tm(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("No parent. TM: n={}, C5={:.8f}".format(parent_n, parent_c5))
    
    # Phase 1: Intensive SLSQP polish at parent resolution
    if tlimit() > 80:
        print("\n=== Phase 1: SLSQP polish n={} ===".format(best_n))
        n = best_n
        h_cur = best_h.copy()
        c5_cur = best_c5
        
        for bv in [50, 150, 400, 1000, 3000, 8000]:
            if tlimit() < 12:
                break
            h_new = slsqp_optimize(h_cur, n, bv, maxiter=350)
            c_new = compute_c5(h_new)
            if c_new < c5_cur - 1e-12:
                c5_cur = c_new
                h_cur = h_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                print("  beta={}: C5={:.8f}".format(bv, c5_cur))
        print("  Phase 1 done: C5={:.8f}".format(best_c5))
    
    # Phase 2: Active lag balancing at parent resolution
    if tlimit() > 50:
        print("\n=== Phase 2: Active lag balance n={} ===".format(best_n))
        n = best_n
        h_cur = best_h.copy()
        c5_cur = best_c5
        h_cur, c5_cur = active_lag_balance(h_cur, n, rng, tlimit, max_rounds=30)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n
        print("  Phase 2 done: C5={:.8f}".format(best_c5))
    
    # Phase 3: Paired coordinate descent
    if tlimit() > 40:
        print("\n=== Phase 3: Paired CD n={} ===".format(best_n))
        n = best_n
        best_h, best_c5 = paired_coord_descent(best_h, n, rng, tlimit, max_sweeps=50)
        best_n = len(best_h)
        print("  Phase 3 done: C5={:.8f}".format(best_c5))
    
    # Phase 4: Multi-resolution optimization
    if tlimit() > 100:
        print("\n=== Phase 4: Multi-resolution ===")
        # Try several resolutions
        res_list = [320, 384, 448, 512, 640]
        res_list = [r for r in res_list if 200 <= r <= 640 and r % 2 == 0 and r != best_n]
        
        for n_new in res_list:
            if tlimit() < 45:
                break
            
            # Upsample
            h_up = project_feasible(interp_resample(best_h, n_new), n_new)
            # Also TM and random
            h_tm = make_tm(n_new)
            h_rand = project_feasible(make_rand_bin(rng, n_new), n_new)
            
            candidates = [(h_up, compute_c5(h_up)), (h_tm, compute_c5(h_tm)), (h_rand, compute_c5(h_rand))]
            candidates.sort(key=lambda x: x[1])
            h_cur = candidates[0][0].copy()
            c5_cur = candidates[0][1]
            
            # L-BFGS-B optimization
            for bv in [50, 200, 800, 3000]:
                if tlimit() < 10:
                    break
                h_new = lbfgsb_optimize(h_cur, n_new, bv, maxiter=250, pen=300.0)
                c_new = compute_c5(h_new)
                if c_new < c5_cur - 1e-12:
                    c5_cur = c_new
                    h_cur = h_new
            
            # SLSQP polish (only for moderate n)
            if n_new <= 512 and tlimit() > 15:
                for bv in [500, 2000]:
                    if tlimit() < 8:
                        break
                    h_new = slsqp_optimize(h_cur, n_new, bv, maxiter=250)
                    c_new = compute_c5(h_new)
                    if c_new < c5_cur - 1e-12:
                        c5_cur = c_new
                        h_cur = h_new
            
            # Active lag balance
            if tlimit() > 10:
                h_cur, c5_cur = active_lag_balance(h_cur, n_new, rng, tlimit, max_rounds=10)
            
            if c5_cur < best_c5 - 1e-9:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n_new
                print("  n={} IMPROVED: C5={:.8f}".format(n_new, best_c5))
            else:
                print("  n={} best: C5={:.8f} (best overall: {:.8f})".format(n_new, c5_cur, best_c5))
    
    # Phase 5: Spectral perturbations to escape local optima
    if tlimit() > 40:
        print("\n=== Phase 5: Spectral perturbation ===")
        n = best_n
        best_h, best_c5 = spectral_perturbation(best_h, n, rng, tlimit, max_tries=15)
        best_n = len(best_h)
        print("  Phase 5 done: C5={:.8f}".format(best_c5))
    
    # Phase 6: Random perturbations + re-optimization
    if tlimit() > 50:
        print("\n=== Phase 6: Random perturbations ===")
        n = best_n
        for attempt in range(12):
            if tlimit() < 12:
                break
            h_pert = best_h.copy()
            n_pert = rng.randint(8, 25)
            idxs = rng.choice(n, size=n_pert, replace=False)
            for idx in idxs:
                h_pert[idx] = np.clip(h_pert[idx] + rng.randn() * 0.3, 0.0, 1.0)
            h_pert = project_feasible(h_pert, n)
            c5_p = compute_c5(h_pert)
            
            if c5_p < best_c5 * 1.06:
                h_cur = h_pert.copy()
                c5_cur = c5_p
                for bv in [300, 1500]:
                    if tlimit() < 6:
                        break
                    h_new = slsqp_optimize(h_cur, n, bv, maxiter=250)
                    c_new = compute_c5(h_new)
                    if c_new < c5_cur - 1e-12:
                        c5_cur = c_new
                        h_cur = h_new
                if c5_cur < best_c5 - 1e-10:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n
                    print("  Attempt {}: C5={:.8f}".format(attempt, best_c5))
    
    # Phase 7: Final active lag balance
    if tlimit() > 25:
        print("\n=== Phase 7: Final balance ===")
        n = best_n
        best_h, best_c5 = active_lag_balance(best_h, n, rng, tlimit, max_rounds=40)
        best_n = len(best_h)
        print("  Phase 7: C5={:.8f}".format(best_c5))
    
    # Phase 8: Final paired CD
    if tlimit() > 15:
        n = best_n
        best_h, best_c5 = paired_coord_descent(best_h, n, rng, tlimit, max_sweeps=40)
        best_n = len(best_h)
        print("  Phase 8: C5={:.8f}".format(best_c5))
    
    # Phase 9: Ultra-high beta SLSQP polish
    if tlimit() > 15:
        n = best_n
        for bv in [5000, 15000, 40000]:
            if tlimit() < 8:
                break
            h_new = slsqp_optimize(best_h, n, bv, maxiter=500)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-13:
                best_c5 = c_new
                best_h = h_new
                best_n = n
        print("  Phase 9: C5={:.8f}".format(best_c5))
    
    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass
    
    _G_PEN = 0.0
    _G_FOCUS_LAGS = None
    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
