def project_feasible(h, n):

        if 128 <= nn <= 768 and nn != best_n:
            n_res_list.append(nn)
    # Also try specific values
    for nn in [256, 320, 448, 512, 576, 640]:
        if 128 <= nn <= 768 and nn not in n_res_list and nn != best_n:
            n_res_list.append(nn)
    n_res_list = sorted(set(n_res_list))[:5]  # limit to 5 resolutions
    
    for n_t in n_res_list:
        if remaining() < 80:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)
        
        # Quick pipeline

        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
            best_n = n_t

    # Phase 4: Aggressive final refinement at best resolution
    n = best_n
    h = best_h.copy()
    
    # Multiple SLSQP passes

        if remaining() < 8:

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Equalize top 3
    if remaining() > 10:
        h_eq, c5_eq = equalize_top_k(h, n, top_k=3, max_iters=500)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
    
    # Equalize top 5
    if remaining() > 10:
        h_eq, c5_eq = equalize_top_k(best_h, n, top_k=5, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
    
    # Binary local search
    if remaining() > 15:
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        if n_ones != n // 2:
            sort_idx = np.argsort(-best_h)
            h_bin = np.zeros(n)
            h_bin[sort_idx[:n // 2]] = 1.0
        ls_time = min(60, max(10, int(remaining() * 0.3)))
        h_b, c5_b = binary_local_search(h_bin, n, time_limit=ls_time, rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    if remaining() > 8:
        h_new = slsqp_smooth(best_h, n, 50000, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Final equalization
    if remaining() > 5:
        h_eq, c5_eq = equalize_top_k(best_h, n, top_k=3, max_iters=200)

            best_h = h_eq.copy()
