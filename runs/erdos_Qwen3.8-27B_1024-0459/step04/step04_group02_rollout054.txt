```python
import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_BETA = 1000.0
_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(x, n, target):
    """Project x onto {0<=h<=1, sum(h)=target} via bisection."""
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def project_feasible(h, n):
    return project_box_sum(h, n, n / 2.0)

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def _g_obj(x):
    n, beta = _G_N, _G_BETA
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    max_s = np.max(shifted)
    if max_s < -700:
        shifted = np.zeros_like(shifted)
    ev = np.exp(shifted)
    ws = np.sum(ev)
    return (m + np.log(ws) / beta) * dx

def _g_grad(x):
    n, beta = _G_N, _G_BETA
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    max_s = np.max(shifted)
    if max_s < -700:
        shifted = np.zeros_like(shifted)
    ev = np.exp(shifted)
    ws = np.sum(ev)
    weights = ev / ws
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset+npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset+npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff+npairs])
                grad[aoff:aoff+npairs] -= w * x[:npairs]
    grad *= dx
    return grad

def _g_con_fun(x):
    return np.sum(x) - _G_N / 2.0

def _g_con_jac(x):
    return np.ones(_G_N)

def slsqp_optimize(h, n, beta, maxiter):
    global _G_N, _G_BETA
    _G_N = n
    _G_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_con_fun, 'jac': _g_con_jac}]
    try:
        res = minimize(_g_obj, h, jac=_g_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def _eval_1d(base, coeff, is_zero, t, t0, dx):
    dt = t - t0
    v = base + dt * coeff - dt * dt * is_zero
    return np.max(v) * dx

def coord_descent(h, n, max_sweeps, n_grid=300):
    best = h.copy()
    best_c5 = compute_c5(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    best_corr = compute_corr(best).copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    grid = np.linspace(0.0, 1.0, n_grid)
    
    offsets = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        
        for i in range(n):
            t0 = best[i]
            
            idx_plus = i + offsets
            idx_minus = i - offsets
            
            h_plus = np.zeros(ncorr)
            valid_plus = (idx_plus >= 0) & (idx_plus < n)
            h_plus[valid_plus] = best[idx_plus[valid_plus]]
            
            h_minus = np.zeros(ncorr)
            valid_minus = (idx_minus >= 0) & (idx_minus < n)
            h_minus[valid_minus] = best[idx_minus[valid_minus]]
            
            coeff = (1.0 - h_plus) - h_minus
            is_zero = (offsets == 0).astype(float)
            base = best_corr
            
            # Grid search
            dt = grid - t0
            vals = base[np.newaxis, :] + np.outer(dt, coeff) - np.outer(dt * dt, is_zero)
            obj_grid = np.max(vals, axis=1) * dx
            best_idx = np.argmin(obj_grid)
            t_best = grid[best_idx]
            c5_local = obj_grid[best_idx]
            
            if c5_local < best_c5 - 1e-14:
                # Golden section refinement
                if 0 < best_idx < n_grid - 1:
                    lo = grid[best_idx - 1]
                    hi = grid[best_idx + 1]
                else:
                    lo, hi = 0.0, 1.0
                
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = _eval_1d(base, coeff, is_zero, c1, t0, dx)
                f2 = _eval_1d(base, coeff, is_zero, c2, t0, dx)
                for _ in range(50):
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = _eval_1d(base, coeff, is_zero, c1, t0, dx)
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = _eval_1d(base, coeff, is_zero, c2, t0, dx)
                
                t_gs = (lo + hi) / 2.0
                f_gs = _eval_1d(base, coeff, is_zero, t_gs, t0, dx)
                if f_gs < c5_local:
                    t_best = t_gs
                    c5_local = f_gs
            
            if c5_local < best_c5 - 1e-14:
                dt_final = t_best - t0
                best[i] = t_best
                best_corr = best_corr + dt_final * coeff - dt_final * dt_final * is_zero
                best_c5 = c5_local
                improved = True
        
        if not improved:
            break
    
    return best, best_c5

def subgrad_descent(h, n, max_iter, n_active=10):
    """Projected subgradient descent on the exact max."""
    best = h.copy()
    best_c5 = compute_c5(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    
    for it in range(max_iter):
        if remaining_time() < 3:
            break
        
        corr = compute_corr(best)
        max_corr = np.max(corr)
        c5_curr = max_corr * dx
        
        # Active set
        threshold = max_corr - 0.5  # half-unit in corr space
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        if len(active) > n_active:
            # Keep only the top n_active
            top_idx = np.argsort(corr)[::-1][:n_active]
            active = top_idx
        
        # Compute subgradient (average of active gradients)
        grad = np.zeros(n)
        for k in active:
            offset = int(k) - (n - 1)
            if offset >= 0:
                npairs = n - offset
                if npairs > 0:
                    grad[offset:offset+npairs] += (1.0 - best[:npairs])
                    grad[:npairs] -= best[offset:offset+npairs]
            else:
                aoff = -offset
                npairs = n - aoff
                if npairs > 0:
                    grad[:npairs] += (1.0 - best[aoff:aoff+npairs])
                    grad[aoff:aoff+npairs] -= best[:npairs]
        grad /= len(active)
        
        gn = np.linalg.norm(grad)
        if gn < 1e-16:
            break
        
        # Backtracking line search
        alpha = 0.02 / (1.0 + it * 0.002)
        found = False
        for scale in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            h_trial = best - alpha * scale * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-15:
                best = h_trial
                best_c5 = c5_trial
                found = True
                break
        
        if not found:
            # Try smaller steps
            for scale in [0.01, 0.005, 0.001]:
                h_trial = best - alpha * scale * grad
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-15:
                    best = h_trial
                    best_c5 = c5_trial
                    found = True
                    break
            if not found:
                break
    
    return best, best_c5
