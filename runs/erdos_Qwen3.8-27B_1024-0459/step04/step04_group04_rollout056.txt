```python
def fft_correlate(a, b):
    n = len(a) + len(b) - 1
    fa = np.fft.fft(a, n=n*2)  # pad to avoid circular correlation
    fb = np.fft.fft(b, n=n*2)
    corr = np.fft.ifft(fa * np.conj(fb)).real
    # rearrange for 'full' mode
    return corr[-(len(b)-1):] + corr[len(a)-1:len(a)+len(b)-1]
```

```python
def steepest_descent_swap(h, n, time_limit=60):
    """Find the best swap (1→0 at i, 0→1 at j) that reduces max correlation the most."""
    dx = 2.0/n
    corr = np.correlate(h, 1-h, mode='full')
    cur_max = float(np.max(corr))
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    # For efficiency, try all swaps from each 1 to each 0
    # But that's O(n²) evaluations, each O(n) for correlation update
    # Total: O(n³) which is too slow for n=360
    
    # Instead: for each 1 position, find which shifts are worst
    # and try swapping to reduce those specific shifts
    
    start = time.time()
    best_h = h.copy()
    best_max = cur_max
    
    for iter in range(1000):
        if time.time() - start > time_limit:
            break
        if remaining() < 10:
            break
        
        corr = np.correlate(h, 1-h, mode='full')
        cur_max = float(np.max(corr))
        
        if cur_max * dx < best_max * dx:
            best_max = cur_max
            best_h = h.copy()
        
        # Find the worst shifts
        worst_shifts = np.argsort(-corr)[:5]
        
        # For each worst shift, determine which swaps would help
        # Try a limited set of swaps
        best_delta = 0
        best_i, best_j = -1, -1
        
        # Sample some candidates
        rng_sample = np.random.RandomState(iter)
        candidates_i = rng_sample.choice(ones, min(20, len(ones)), replace=False)
        candidates_j = rng_sample.choice(zeros, min(20, len(zeros)), replace=False)
        
        for i in candidates_i:
            for j in candidates_j:
                h[i], h[j] = 0, 1
                new_corr = np.correlate(h, 1-h, mode='full')
                new_max = float(np.max(new_corr))
                h[i], h[j] = 1, 0
                delta = new_max - cur_max
                if delta < best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
        
        if best_i >= 0 and best_delta < 0:
            h[best_i], h[best_j] = 0, 1
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
```

```python
def swap_delta(h, a, b, n):
    """Compute the change in correlation vector when h[a]: 1->0 and h[b]: 0->1."""
    delta = np.zeros(2*n-1)
    # For each shift s from -(n-1) to (n-1):
    # output index k = s + (n-1)
    
    for s in range(-(n-1), n):
        if s == 0:
            continue
        k = s + (n - 1)
        d = 0.0
        # Term 1: n=a, need 0 <= a-s < n
        if 0 <= a - s < n:
            d -= (1.0 - h[a - s])
        # Term 2: n=a+s, need 0 <= a+s < n
        if 0 <= a + s < n:
            d += h[a + s]
        # Term 3: n=b, need 0 <= b-s < n
        if 0 <= b - s < n:
            d += (1.0 - h[b - s])
        # Term 4: n=b+s, need 0 <= b+s < n
        if 0 <= b + s < n:
            d -= h[b + s]
        delta[k] = d
    
    return delta
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

    target = n / 2.0
```

        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_binary(h, n, rng):
    idx = np.argsort(-np.array(h, dtype=float))[:n // 2]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

        sel = rng.choice(len(idx), target - n_ones, replace=False)
