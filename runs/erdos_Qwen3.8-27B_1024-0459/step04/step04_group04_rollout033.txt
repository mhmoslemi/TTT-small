import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

_START = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1000.0
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 5.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.clip(np.interp(x_new, xp, src_h), 0.0, 1.0)


def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -50, 50)
    ex = np.exp(cl)
    s = np.sum(ex)
    if s < 1e-300:
        return m * dx_val
    return (m + np.log(s) / beta) * dx_val


def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        if npairs <= 0:
            return g
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        if npairs <= 0:
            return g
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -50, 50)
    ex = np.exp(cl)
    s = np.sum(ex)
    if s < 1e-300:
        k_max = int(np.argmax(corr))
        return corr_grad_per_shift(x, n, k_max) * dx_val
    w = ex / s
    g = np.zeros(n)
    sig = w > 1e-12
    for k in np.where(sig)[0]:
        g += w[k] * corr_grad_per_shift(x, n, int(k))
    return g * dx_val


def sum_con(x):
    return np.sum(x) - _TGT


def sum_con_jac(x):
    return np.ones(_N)


def slsqp_optimize(h, n, beta, maxiter=300):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def coordinate_descent(h, n, max_sweeps=20, tol=1e-12):
    """Cyclic coordinate descent solving exact 1D convex subproblems."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for sweep in range(max_sweeps):
        if remaining() < 3:
            break
        total_improvement = 0.0
        corr = compute_corr(h)

        for i in range(n):
            if remaining() < 1:
                break

            shifts = np.arange(-(n - 1), n, dtype=int)
            idx1 = i + shifts
            idx2 = i - shifts

            d = np.zeros(2 * n - 1)
            mask1 = (idx1 >= 0) & (idx1 < n)
            d[mask1] += (1.0 - h[idx1[mask1]])
            mask2 = (idx2 >= 0) & (idx2 < n) & (shifts != 0)
            d[mask2] -= h[idx2[mask2]]

            # 1D convex minimization: min_s max_k (corr[k] + s*d[k]) for s in [-h[i], 1-h[i]]
            lo = -h[i]
            hi = 1.0 - h[i]

            # Check if current s=0 is optimal
            active_at_0 = corr + 0.0 * d
            m0 = np.max(active_at_0)
            active_mask = active_at_0 >= m0 - 1e-14
            if np.any(active_mask):
                bd = d[active_mask]
                if np.min(bd) >= -1e-14 and np.max(bd) <= 1e-14:
                    continue  # Already optimal

            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a = lo
            b = hi
            c = b - gr * (b - a)
            dv = a + gr * (b - a)

            def obj_1d(s):
                return np.max(corr + s * d)

            fc = obj_1d(c)
            fd = obj_1d(dv)

            for _ in range(40):
                if fc < fd:
                    b = dv
                    dv = c
                    fd = fc
                    c = b - gr * (b - a)
                    fc = obj_1d(c)
                else:
                    a = c
                    c = dv
                    fc = fd
                    dv = a + gr * (b - a)
                    fd = obj_1d(dv)

            s_opt = (a + b) / 2.0
            new_val = h[i] + s_opt

            new_c5_est = np.max(corr + s_opt * d) * (2.0 / n)
            if new_c5_est < best_c5 - tol:
                h[i] = new_val
                total_improvement += best_c5 - new_c5_est

        h = project_feasible(h, n)
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()

        if total_improvement < 1e-14:
            break

    return best_h, best_c5


def subgradient_method(h, n, time_limit=60):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx_val = 2.0 / n
    start = time.time()
    h_avg = h.copy()
    n_iter = 0
    step_scale = 0.3

    while time.time() - start < time_limit and remaining() > 10:
        n_iter += 1
        corr = compute_corr(h)
        m = float(np.max(corr))
        c5 = m * dx_val
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        threshold = m * 0.995
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(corr)[-5:]

        g = np.zeros(n)
        for k in active:
            g += corr_grad_per_shift(h, n, int(k))
        g /= len(active)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        alpha = step_scale / (np.sqrt(n_iter) * g_norm)
        alpha = min(alpha, 0.05)
        h_new = project_feasible(h - alpha * g, n)
        h_avg = (n_iter * h_avg + h_new) / (n_iter + 1)
        h_avg_p = project_feasible(h_avg, n)
        c5_avg = compute_c5(h_avg_p)
        if c5_avg < best_c5:
            best_c5 = c5_avg
            best_h = h_avg_p.copy()
        h = h_new

    return best_h, best_c5

def diverse_inits(n, seed, parent_h, parent_n):
    """Generate diverse initial candidates."""
    rng = np.random.RandomState(seed)
    cands = []
    x = np.linspace(0.5 / n, 1.0 - 0.5 / n, n)

    # Parent
    cands.append(project_feasible(interp_to(parent_h, parent_n, n), n))

    # Linear ramp
    cands.append(project_feasible(0.5 + 0.4 * (2 * x - 1), n))
    # Inverted ramp
    cands.append(project_feasible(0.5 - 0.4 * (2 * x - 1), n))
    # Quadratic
    cands.append(project_feasible(0.5 + 0.35 * (2 * x - 1) ** 2 - 0.175, n))
    # Sine
    for freq in [1, 2, 3, 4]:
        cands.append(project_feasible(0.5 + 0.3 * np.sin(2 * np.pi * freq * x), n))
        cands.append(project_feasible(0.5 + 0.3 * np.cos(2 * np.pi * freq * x), n))
    # Triangle wave
    tri = 1.0 - 2.0 * np.abs(2.0 * x - 1.0)
    cands.append(project_feasible(0.5 + 0.3 * (tri - 0.5), n))

    # Random smooth
    for _ in range(5):
        n_terms = rng.randint(2, 6)
        h_r = np.zeros(n)
        for _ in range(n_terms):
            f = rng.uniform(1, 8)
            ph = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.05, 0.25)
            h_r += amp * np.sin(2 * np.pi * f * x + ph)
        cands.append(project_feasible(0.5 + h_r, n))

    return cands
