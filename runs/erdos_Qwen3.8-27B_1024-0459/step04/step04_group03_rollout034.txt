def smooth_max_slsqp(h, n, beta, maxiter=300, ftol=1e-14):
    global _N_G, _BETA_G
    _N_G = n
    _BETA_G = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _smooth_max_eq}]
    try:
        res = minimize(_smooth_max_obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def minimax_slsqp(h_init, n, max_rounds=10, max_slsqp_iter=200):
    global _N_G, _DX_G, _LAGS_G
    _N_G = n
    _DX_G = 2.0 / n
    h = h_init.copy()
    
    for rnd in range(max_rounds):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        max_corr = np.max(corr)
        
        # Identify active lags (top few + near-max)
        threshold = max_corr - 0.3
        active = np.where(corr >= threshold)[0]
        top5 = np.argsort(corr)[::-1][:5]
        active = np.unique(np.concatenate([active, top5]))
        
        _LAGS_G = active
        t_init = max_corr * _DX_G + 1e-6
        x0 = np.concatenate([h, [t_init]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        cons = [
            {'type': 'ineq', 'fun': _mm_ineq},
            {'type': 'eq', 'fun': _mm_eq}
        ]
        try:
            res = minimize(_mm_obj, x0, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_slsqp_iter, 'ftol': 1e-12, 'iprint': 0})
            if res.x is not None:
                h_new = project_feasible(res.x[:n], n)
                c5_new = compute_c5(h_new)
                c5_old = compute_c5(h)
                if c5_new < c5_old - 1e-15:
                    h = h_new
                else:
                    break
        except Exception:
            break
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

# Combinatorial local search
def local_search(h, n, max_iters=2000, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining_time() < 2:
            break
        # Try random pair transfer
        i = rng.randint(n)
        j = (i + rng.randint(1, n)) % n
        if i == j:
            continue
        for frac in [0.1, 0.25, 0.5, 0.75, 0.9]:
            amt = (best_h[i] - best_h[j]) * frac
            if abs(amt) < 0.005:
                continue
            h_trial = best_h.copy()
            h_trial[i] -= amt
            h_trial[j] += amt
            if h_trial[i] < -0.01 or h_trial[i] > 1.01 or h_trial[j] < -0.01 or h_trial[j] > 1.01:
                continue
            h_trial = project_feasible(h_trial, n)
            c5_t = compute_c5(h_trial)
            if c5_t < best_c5 - 1e-12:
                best_c5 = c5_t
                best_h = h_trial.copy()
                break
    
    return best_h, best_c5

# Binary local search (for near-binary solutions)
def binary_local_search(h, n, max_swaps=5000, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Try pair flips (0->1, 1->0) to maintain sum
    for it in range(max_swaps):
        if remaining_time() < 2:
            break
        # Find a 0-ish and a 1-ish position
        zeros = np.where(best_h < 0.3)[0]
        ones = np.where(best_h > 0.7)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        h_trial = best_h.copy()
        # Swap: move i towards 1, j towards 0
        h_trial[i] = min(1.0, h_trial[i] + (1.0 - h_trial[i]))  # set to 1
        h_trial[j] = max(0.0, h_trial[j] - h_trial[j])  # set to 0
        # Actually, let's do a softer swap
        h_trial[i] = min(1.0, best_h[i] + 0.5)
        h_trial[j] = max(0.0, best_h[j] - 0.5)
        h_trial = project_feasible(h_trial, n)
        c5_t = compute_c5(h_trial)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_trial.copy()
    
    return best_h, best_c5
