def _smooth_obj_and_grad(x, n, beta):

    shifted = np.minimum(beta * (corr - m), 0.0)

    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)

            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]
    
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return None
    grad /= gnorm
    return grad * dx

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load initial
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        h_tmp = np.zeros(best_n)
        for i in range(best_n):
            v = i
            cnt = 0
            while v:
                cnt ^= (v & 1)
                v >>= 1
            h_tmp[i] = 1.0 if cnt == 1 else 0.0
        best_h = project_feasible(h_tmp, best_n)
        best_c5 = compute_c5(best_h)

    # Phase 1: Initial SLSQP at parent resolution
    n = best_n
    h = best_h.copy()
    for beta in [200, 1000, 5000]:
        if remaining_time() < 30:
            break
        h_new = slsqp_refine(h, n, beta, 400)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-15:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()

    # Phase 2: Aggressive coordinate descent
    if remaining_time() > 60:
        h_cd, c5_cd = coord_descent_fast(h, n, max_sweeps=500, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()
        else:
            h = best_h.copy()

    # Phase 3: More SLSQP with higher beta
    for beta in [10000, 30000, 80000]:
        if remaining_time() < 30:
            break
        h_new = slsqp_refine(h, n, beta, 500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Coord descent interleaved with subgradient
    if remaining_time() > 80:
        h = best_h.copy()
        n_cur = best_n
        for it in range(3000):
            if remaining_time() < 10:
                break
            if it % 50 == 0:
                # Coordinate descent burst
                h_cd, c5_cd = coord_descent_fast(h, n_cur, max_sweeps=20, rng=rng)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                    h = h_cd.copy()
            else:
                # Subgradient step
                grad = subgrad_step(h, n_cur, n_active=6)
                if grad is None:
                    break
                improved = False
                base_step = 0.02 / (1.0 + it * 0.002)
                for sm in [1.0, 0.5, 2.0, 0.25, 0.75, 1.5]:
                    step = base_step * sm
                    h_trial = h - step * grad
                    h_trial = project_feasible(h_trial, n_cur)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < best_c5 - 1e-16:
                        best_c5 = c5_trial
                        best_h = h_trial.copy()
                        h = h_trial.copy()
                        improved = True
                        break
                if not improved and it > 200:
                    # Periodic SLSQP restart
                    if it % 200 == 100 and remaining_time() > 30:
                        h_new = slsqp_refine(h, n_cur, 20000, 200)
                        c5_new = compute_c5(h_new)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_new.copy()
                            h = h_new.copy()
                        if it > 500:
                            break

    # Phase 5: Try higher resolution n=1280
    if remaining_time() > 120:
        h_r = interp_to(best_h, best_n, 1280)
        h_r = project_feasible(h_r, 1280)
        c5_r = compute_c5(h_r)

        for beta in [500, 2000, 10000, 40000]:
            if remaining_time() < 25:
                break
            h_new = slsqp_refine(h_r, 1280, beta, 400)
            c5_new = compute_c5(h_new)
            if c5_new < c5_r - 1e-15:
                h_r = h_new
                c5_r = c5_new
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = 1280

        if remaining_time() > 40:
            h_cd, c5_cd = coord_descent_fast(h_r, 1280, max_sweeps=100, rng=rng)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = 1280

    # Phase 6: Final coordinate descent polish
    if remaining_time() > 30:
        h_cd, c5_cd = coord_descent_fast(best_h, best_n, max_sweeps=200, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 7: Final SLSQP polish
    if remaining_time() > 15:
        h_new = slsqp_refine(best_h, best_n, 100000, 300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 8: Tiny coordinate descent
    if remaining_time() > 10:
        h_cd, c5_cd = coord_descent_fast(best_h, best_n, max_sweeps=50, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
