        if remaining() > 8:
            h_opt, c5_opt = top_shift_subgrad(h_opt, n_t, top_k=4, n_iters=200)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_t

    # Phase 5: Try different initializations at best resolution
    n_best = best_n
    for init_fn, lbl in [(make_thue_morse, 'TM'), (make_random, 'RND')]:
        if remaining() < 30:
            break
        h_init = init_fn(n_best, seed + 7)
        c5_init = compute_c5(h_init)
        if c5_init < best_c5:
            best_h = h_init.copy()
            best_c5 = c5_init
        h_opt = h_init
        for beta in [100, 1000, 5000]:
            if remaining() < 10:
                break
            h_opt = lbfgsb_opt(h_opt, n_best, beta, pen=1e5, maxiter=200)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        if remaining() > 5:
            h_opt, c5_opt = top_shift_subgrad(h_opt, n_best, top_k=4, n_iters=200)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 6: High-precision final optimization at best resolution
    n = best_n
    h = best_h.copy()
    for beta in [10000, 30000, 80000, 200000]:
        if remaining() < 5:
            break
        h_new = lbfgsb_opt(h, n, beta, pen=1e5, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 7: SLSQP final
    if remaining() > 10:
        for beta in [20000, 60000]:
            if remaining() < 6:
                break
            h_new = slsqp_opt(h, n, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 8: Top-shift equalization
    if remaining() > 8:
        h_eq, c5_eq = top_shift_subgrad(h, n, top_k=6, n_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 9: Coordinate descent polish
    if remaining() > 5:
        h_cd, c5_cd = pair_coord_descent(best_h, n, n_trials=3000, seed=seed)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 10: Final L-BFGS-B polish
    if remaining() > 5:
        h_new = lbfgsb_opt(best_h, n, 100000, pen=1e6, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final projection and validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5 + 1e-12:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
