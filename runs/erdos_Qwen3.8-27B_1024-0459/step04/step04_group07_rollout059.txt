def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1')
        h[i] = 1.0 if c % 2 == 0 else 0.0
    return h

def make_rotated_tm(n, shift):

    h = np.roll(h, shift)
    return h

    h = 0.5 + 0.4 * np.sin(2.0 * np.pi * f1 * t + phase) + 0.2 * np.sin(2.0 * np.pi * f2 * t + phase * 0.7)

    h = np.zeros(n)
    for i in range(n):

            h[i] = 1.0
    return h

    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.clip(np.interp(xn, xp, src), 0.0, 1.0)

def subgradient_step(h, n, alpha, rng):
    """One step of projected subgradient descent on the max objective."""
    corr = get_corr(h)
    dx = 2.0 / n
    # Find active lags (within tolerance of max)
    cmax = np.max(corr)
    tol = 0.1 * cmax  # lags within 10% of max are "active"
    active = np.where(corr >= cmax - tol)[0]
    
    # Compute subgradient: average gradient over active lags
    grad = np.zeros(n)
    for k in active:
        # c_k = sum_i h[i] * (1-h[i+k]) for valid i
        if k < n:
            # i ranges from 0 to n-1-k
            for i in range(n - k):
                grad[i] += (1.0 - h[i + k])
                grad[i + k] += (-h[i])
        else:
            # k >= n: effective lag is k - n, but reversed
            lag = k - n
            for i in range(n - lag):
                grad[n - 1 - i] += (1.0 - h[n - 1 - i - lag])
                grad[n - 1 - i - lag] += (-h[n - 1 - i])
    
    grad /= len(active)
    
    # Projected gradient step
    h_new = h - alpha * grad
    return project_feasible(h_new, n)

Now for coordinate descent with proper line search:

If I change h[i] from v to t, the change in c[k] is:
δc[k] = (t-v) * (1-2*h[k-i]) * I(0<=k-i<n)

So the new c[k] = c[k] + (t-v) * a_k where a_k = (1-2*h[k-i]) * I(0<=k-i<n)

The objective is max_k c[k] * dx. We want to minimize max_k (c[k] + (t-v)*a_k).

This is a 1D problem: minimize max_k (A_k + B_k * t) where A_k = c[k] - v*a_k, B_k = a_k.

The optimal t is found where the active constraints are "balanced" or at a boundary (t=0 or t=1), subject to the feasibility constraint that sum(h) = n/2 (which means we need to adjust another coordinate, or we handle it via projection).

For simplicity, let's handle the sum constraint by adjusting a "reservoir" coordinate. Or we can do the coordinate descent in a way that preserves the sum: change h[i] by δ and h[j] by -δ.

Let me implement a "pair coordinate descent":
- Pick pair (i, j)
- Change h[i] += δ, h[j] -= δ
- Find optimal δ via line search

The change in c[k]:
δc[k] = δ*(1-2*h[k-i])*I(0<=k-i<n) - δ*(1-2*h[k-j])*I(0<=k-j<n)
= δ * (a_k^i - a_k^j)

So new c[k] = c[k] + δ * d_k where d_k = a_k^i - a_k^j = (1-2*h[k-i])*I(0<=k-i<n) - (1-2*h[k-j])*I(0<=k-j<n)

We want to minimize max_k (c[k] + δ*d_k) subject to 0<=h[i]+δ<=1 and 0<=h[j]-δ<=1.

This is a piecewise linear function of δ. Its minimum is at a "breakpoint" where two or more active constraints cross, or at the boundary.

To find it: the function f(δ) = max_k (c[k] + δ*d_k) is convex in δ (max of linear functions). We can find its minimum by:
1. Finding all breakpoints (where two lines cross): δ_{kl} = (c[l]-c[k])/(d[k]-d[l]) for d[k]≠d[l]
2. Checking f at each breakpoint and at the boundaries
3. Taking the minimum

But with 2n-1 = 511 lines, there are O(n²) breakpoints. For n=256, that's ~262000 breakpoints to check. Each check is O(n) to evaluate the max. Total: O(n³) per coordinate step. That's too slow.

Alternative: use the fact that we only need to check the top few lines. Or use a simpler approach:
- Just try δ = 0, ±0.01, ±0.02, ..., ±0.5 and pick the best.
- This is O(n) evaluations per coordinate, each O(n) → O(n²) per coordinate → O(n³) per full sweep. For n=256: ~16 million operations. Fast enough!

Actually even simpler: since f(δ) is convex piecewise linear, we can do a ternary search or binary search on the derivative. The derivative is the slope of the active line(s). If the slope is negative, decrease δ; if positive, increase.

Let me just do a fine grid search for δ. That's simplest and most robust.

            break
    h_best = project_feasible(h_best, n)
    c5 = compute_c5(h_best)
    return h_best, c5

def binary_flip_search(h, n, rng, tlimit, max_iter=5000):
    """Search over binary solutions by flipping pairs of bits."""
    h_bin = np.round(h)
    h_bin = np.clip(h_bin, 0.0, 1.0)
    h_bin = project_feasible(h_bin, n)
    h_bin = np.round(h_bin)
    ones = np.where(h_bin > 0.5)[0]
    zeros = np.where(h_bin <= 0.5)[0]
    if len(ones) != n // 2 or len(zeros) != n // 2:
        h_bin = make_rand_bin(rng, n)
        ones = np.where(h_bin > 0.5)[0]
        zeros = np.where(h_bin <= 0.5)[0]
    c5 = compute_c5(h_bin)
    for it in range(max_iter):
        if tlimit() < 1.0:
            break
        io = rng.randint(len(ones))
        iz = rng.randint(len(zeros))
        i = ones[io]
        j = zeros[iz]
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c_new = compute_c5(h_bin)
        if c_new < c5 - 1e-12:
            c5 = c_new
            ones[io] = j
            zeros[iz] = i
            ones = np.sort(ones)
            zeros = np.sort(zeros)
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    return h_bin, c5

def differential_evolution(h_init, n, rng, tlimit, pop_size=30, max_gens=200, F=0.8, CR=0.7):
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    pop = []
    for i in range(pop_size):
        if i == 0:
            hp = h_init.copy()
        else:
            hp = h_init + rng.randn(n) * 0.3
            hp = project_feasible(hp, n)
        pop.append((hp, compute_c5(hp)))
    pop.sort(key=lambda x: x[1])
    if pop[0][1] < best_c5:
        best_c5 = pop[0][1]
        best_h = pop[0][0].copy()
    for gen in range(max_gens):
        if tlimit() < 5.0:
            break
        pop[0] = (best_h.copy(), best_c5)
        for p in range(pop_size):
            if tlimit() < 2.0:
                break
            a, b, c = rng.choice(pop_size, 3, replace=False)
            if p in (a, b, c):
                a, b, c = max(p+1, 0), (p+2) % pop_size, (p+3) % pop_size
                if p in (a, b, c):
                    continue
            v = pop[a][0] + F * (pop[b][0] - pop[c][0])
            cross_idx = rng.rand(n) < CR
            if not np.any(cross_idx):
                cross_idx[rng.randint(n)] = True
            u = np.where(cross_idx, v, pop[p][0])
            u = project_feasible(u, n)
            c_u = compute_c5(u)
            if c_u <= pop[p][1]:
                pop[p] = (u, c_u)
            if c_u < best_c5:
                best_c5 = c_u
                best_h = u.copy()
    return best_h, best_c5

def slsqp_smooth(h0, n, beta, maxiter=300):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = corr - m
        e = np.exp(np.clip(-beta * s, -100, 400))
        val = (m + np.log(np.sum(e)) / beta) * dx
        return float(val)
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = corr - m
        e = np.exp(np.clip(-beta * s, -100, 400))
        Z = np.sum(e)
        weights = e / Z
        k_range = np.arange(2 * n - 1)
        g = np.zeros(n)
        for k_idx in range(2 * n - 1):
            w = weights[k_idx]
            if w < 1e-10:
                continue
            ki = k_range[k_idx] - np.arange(n)
            mk = (ki >= 0) & (ki < n)
            g[mk] += w * (1.0 - 2.0 * x[ki[mk]])
        return g * dx
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    budget = float(budget_s)
    rng = np.random.RandomState(seed)
    
    def tlimit():
        return max(0.0, budget - (time.time() - t_start) - 5.0)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    n_base = 256
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        n_base = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = n_base
        print("Parent: n={}, C5={:.8f}".format(n_base, parent_c5))
    except Exception:
        parent_h = make_tm(256)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = 256
        print("TM init: n=256, C5={:.8f}".format(parent_c5))
    
    # Phase 1: Quick local refinement of parent with pair coord descent
    if tlimit() > 60:
        print("\n=== Phase 1: pair coord descent on parent ===")
        h1, c5_1 = pair_coord_descent(best_h, best_n, max_sweeps=30, rng=rng, tlimit=tlimit)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
            best_n = best_n
        print("  Phase 1: C5={:.8f}".format(best_c5))
    
    # Phase 2: SLSQP smooth optimization with increasing beta
    if tlimit() > 60:
        print("\n=== Phase 2: SLSQP multi-beta ===")
        h_cur = best_h.copy()
        c5_cur = best_c5
        for bv in [50, 200, 800, 3000, 10000]:
            if tlimit() < 8:
                break
            h_new = slsqp_smooth(h_cur, best_n, beta=bv, maxiter=400)
            c_new = compute_c5(h_new)
            if c_new < c5_cur - 1e-13:
                c5_cur = c_new
                h_cur = h_new
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = len(h_cur)
        print("  Phase 2: C5={:.8f}".format(best_c5))
    
    # Phase 3: Differential evolution at base resolution
    if tlimit() > 100:
        print("\n=== Phase 3: differential evolution n={} ===".format(best_n))
        n_de = best_n
        h_de, c5_de = differential_evolution(best_h, n_de, rng, tlimit,
                                             pop_size=25, max_gens=150, F=0.7, CR=0.8)
        if c5_de < best_c5:
            best_c5 = c5_de
            best_h = h_de.copy()
            best_n = n_de
            print("  DE improved: C5={:.8f}".format(best_c5))
        else:
            print("  DE best: C5={:.8f} (vs {:.8f})".format(c5_de, best_c5))
        
        # Pair coord descent on DE result
        if c5_de < best_c5 * 1.02:
            h_de2, c5_de2 = pair_coord_descent(h_de, n_de, max_sweeps=40, rng=rng, tlimit=tlimit)
            if c5_de2 < best_c5:
                best_c5 = c5_de2
                best_h = h_de2.copy()
                best_n = n_de
            print("  After DE+PCD: C5={:.8f}".format(best_c5))
    
    # Phase 4: Binary search + continuous refinement
    if tlimit() > 60:
        print("\n=== Phase 4: binary flip search ===")
        h_bin, c5_bin = binary_flip_search(best_h, best_n, rng, tlimit, max_iter=8000)
        print("  Binary C5={:.8f}".format(c5_bin))
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            best_n = best_n
        
        # Try continuous refinement from binary
        if c5_bin < best_c5 * 1.1:
            h_ref = slsqp_smooth(h_bin, best_n, beta=500, maxiter=300)
            c_ref = compute_c5(h_ref)
            h_ref, c_ref = pair_coord_descent(h_ref, best_n, max_sweeps=30, rng=rng, tlimit=tlimit)
            if c_ref < best_c5:
                best_c5 = c_ref
                best_h = h_ref.copy()
                best_n = best_n
            print("  After binary+refine: C5={:.8f}".format(best_c5))
    
    # Phase 5: Try different resolutions
    if tlimit() > 100:
        print("\n=== Phase 5: multi-resolution ===")
        for n_try in [128, 192, 320, 384, 512]:
            if tlimit() < 30:
                break
            h_interp = project_feasible(interp_h(best_h, best_n, n_try), n_try)
            c5_interp = compute_c5(h_interp)
            
            h_tm_try = project_feasible(make_tm(n_try), n_try)
            c5_tm = compute_c5(h_tm_try)
            
            # Pick better initial
            if c5_tm < c5_interp:
                h_cur = h_tm_try
                c5_cur = c5_tm
            else:
                h_cur = h_interp
                c5_cur = c5_interp
            
            # Quick SLSQP
            if tlimit() > 20:
                h_cur = slsqp_smooth(h_cur, n_try, beta=300, maxiter=250)
                c5_cur = compute_c5(h_cur)
            
            # Pair coord descent
            if tlimit() > 15:
                h_cur, c5_cur = pair_coord_descent(h_cur, n_try, max_sweeps=25, rng=rng, tlimit=tlimit)
            
            if c5_cur < best_c5 - 1e-9:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n_try
                print("  n={} improved: C5={:.8f}".format(n_try, best_c5))
            else:
                print("  n={} best: C5={:.8f}".format(n_try, c5_cur))
    
    # Phase 6: Intensive pair coordinate descent on best
    if tlimit() > 40:
        print("\n=== Phase 6: intensive PCD ===")
        h6, c5_6 = pair_coord_descent(best_h, best_n, max_sweeps=80, rng=rng, tlimit=tlimit)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
            best_n = best_n
        print("  Phase 6: C5={:.8f}".format(best_c5))
    
    # Phase 7: SLSQP final polish
    if tlimit() > 20:
        print("\n=== Phase 7: SLSQP polish ===")
        for bv in [5000, 20000]:
            if tlimit() < 5:
                break
            h7 = slsqp_smooth(best_h, best_n, beta=bv, maxiter=500)
            c7 = compute_c5(h7)
            if c7 < best_c5 - 1e-14:
                best_c5 = c7
                best_h = h7
        print("  Phase 7: C5={:.8f}".format(best_c5))
    
    # Phase 8: Final pair coord descent
    if tlimit() > 10:
        print("\n=== Phase 8: final PCD ===")
        h8, c5_8 = pair_coord_descent(best_h, best_n, max_sweeps=50, rng=rng, tlimit=tlimit, n_grid=51)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()
            best_n = best_n
        print("  Phase 8: C5={:.8f}".format(best_c5))
    
    # Phase 9: Perturbation restarts
    if tlimit() > 40:
        print("\n=== Phase 9: perturbation restarts ===")
        for attempt in range(5):
            if tlimit() < 10:
                break
            n_pert = rng.randint(5, 20)
            h_pert = best_h.copy()
            idxs = rng.choice(best_n, size=n_pert, replace=False)
            for idx in idxs:
                h_pert[idx] = np.clip(h_pert[idx] + rng.randn() * 0.2, 0.0, 1.0)
            h_pert = project_feasible(h_pert, best_n)
            c5_p = compute_c5(h_pert)
            if c5_p < best_c5 * 1.03:
                h_pert, c5_p = pair_coord_descent(h_pert, best_n, max_sweeps=30, rng=rng, tlimit=tlimit)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_pert.copy()
                    print("  Perturb {} improved: C5={:.8f}".format(attempt, best_c5))
    
    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
