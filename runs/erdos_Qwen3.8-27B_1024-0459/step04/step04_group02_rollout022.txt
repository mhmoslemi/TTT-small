            break
    
    return best, best_c5

def project_feasible(h, n):

def compute_d_vector(j, h, n):
    """Compute the linear coefficient d[k] for changing h[j]."""
    ncorr = 2 * n - 1
    d = np.zeros(ncorr)
    # Offset 0 (k = n-1)
    d[n - 1] = 1.0
    # Offsets +s and -s for s = 1..n-1
    for s in range(1, n):
        # Offset +s (k = n-1+s): d = (1-h[j-s])*[s<=j] - h[j+s]*[j<=n-1-s]
        if s <= j:
            d[n - 1 + s] += (1.0 - h[j - s])
        if j <= n - 1 - s:
            d[n - 1 + s] -= h[j + s]
        # Offset -s (k = n-1-s): d = (1-h[j+s])*[j<=n-1-s] - h[j-s]*[s<=j]
        if j <= n - 1 - s:
            d[n - 1 - s] += (1.0 - h[j + s])
        if s <= j:
            d[n - 1 - s] -= h[j - s]
    return d

def exact_1d_minimax(base, d, n, t0):

def find_optimal_t(base, d, n, t0, V_target):
    """Find a t in [0,1] that achieves the minimax value V_target."""
    ncorr = 2 * n - 1
    k0 = n - 1
    
    # Quadratic feasible regions at V_target
    disc = 1.0 - 4.0 * (V_target - base[k0])
    if disc >= 0.0:

        t1 = (1.0 - sqrt_disc) * 0.5
        t2 = (1.0 + sqrt_disc) * 0.5
        regions = []
        if t1 >= 0.0:
            regions.append((0.0, min(1.0, t1)))
        if t2 <= 1.0:
            regions.append((max(0.0, t2), 1.0))

    idx_all = np.arange(ncorr)
    LB = 0.0
    UB = 1.0

    mask_neg = (idx_all != k0) & (d < -1e-15)
    mask_zero = (idx_all != k0) & (np.abs(d) <= 1e-15)

        LB = max(LB, float(np.max(lb_vals)))

        if np.max(base[mask_zero]) > V_target + 1e-12:
            LB = 0.0
            UB = -1.0  # infeasible

    best_t = t0

            continue

        ihi = min(UB, rhi)

                dte = abs(te - t0)
                if dte < best_dist:
                    best_dist = dte
                    best_t = te
    
    # Fallback
    if best_dist > 1e98:
        best_t = np.clip(t0, 0.0, 1.0)

def smooth_max_obj_and_grad(x, n, beta):
    """Smooth max of correlation * dx with gradient."""
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.minimum(shifted, 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def slsqp_polish(h, n, beta, maxiter):
    """SLSQP refinement with smooth max."""
    dx = 2.0 / n
    
    def obj(x):
        return smooth_max_obj_and_grad(x, n, beta)[0]
    
    def obj_grad(x):
        return smooth_max_obj_and_grad(x, n, beta)[1]
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = float(np.max(np.correlate(trial, 1.0 - trial, mode='full') * dx))
            c5_orig = float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))
            if c5_trial <= c5_orig + 1e-14:
                return trial
    except Exception:
        pass
    return h.copy()

def interp_to(src_h, src_n, dst_n):
    """Interpolate h from one resolution to another."""
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    
    # Phase 7: Try n=960 as well (different resolution)
    n3 = 960
    if remaining_time() > 60:
        h3b = interp_to(best_h, best_n, n3)
        h3b = project_feasible(h3b, n3)
        h3b, c5_3b = coord_descent_exact(h3b, n3, max_sweeps=60)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()
            best_n = n3
    
    # Phase 8: Try random binary initialization at 1280
    n4 = 1280
    if remaining_time() > 80:
        h_rand = np.zeros(n4)
        h_rand[rng.choice(n4, n4 // 2, replace=False)] = 1.0
        h_rand = project_feasible(h_rand, n4)
        h_rand, c5_rand = coord_descent_exact(h_rand, n4, max_sweeps=30)
        if c5_rand < best_c5:
            best_c5 = c5_rand
            best_h = h_rand.copy()
            best_n = n4
    
    # Phase 9: Final polish - multiple SLSQP + coord descent
    if remaining_time() > 30:
        for beta in [5000, 20000]:
            if remaining_time() < 10:
                break
            h_new = slsqp_polish(best_h, best_n, beta, 150)
            c5_new = float(np.max(np.correlate(h_new, 1.0 - h_new, mode='full') * (2.0 / best_n)))
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        
        if remaining_time() > 15:
            h9, c5_9 = coord_descent_exact(best_h, best_n, max_sweeps=30)
            if c5_9 < best_c5:
                best_c5 = c5_9
                best_h = h9.copy()
    
    # Phase 10: High precision coordinate descent (more iterations)
    if remaining_time() > 15:
        h10, c5_10 = coord_descent_exact(best_h, best_n, max_sweeps=50, tolerance=1e-14)
        if c5_10 < best_c5:
            best_c5 = c5_10
            best_h = h10.copy()
    
    # Final verification
    best_h = project_feasible(best_h, best_n)
    final_c5 = float(np.max(np.correlate(best_h, 1.0 - best_h, mode='full') * (2.0 / best_n)))
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
