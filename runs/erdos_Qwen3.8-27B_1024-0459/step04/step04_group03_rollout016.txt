    # Gradient: dx * sum_k w[k] * d(corr_k)/d(h_i)
    # corr_k = sum_j h[j] * (1 - h[j + p]) where p = n-1-k
    # d(corr_k)/d(h_i) = (1-h[i+p])*1[i+p<n] - h[i-p]*1[i-p>=0]
    
    # For efficiency, only use k where w[k] > threshold
    thresh = 1e-10
    active = w > thresh
    if not np.any(active):
        return smooth_val, np.zeros(n)
    
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        p = n - 1 - k
        wk = w[k] * dx
        if p >= 0:
            # (1-h[i+p]) for i in [0, n-1-p]
            grad[:n-p] += wk * (1.0 - h[p:])
            # -h[i-p] for i in [p, n-1]
            grad[p:] += -wk * h[:n-p]
        else:
            q = -p  # q > 0
            # (1-h[i+p]) = (1-h[i-q]) for i in [q, n-1]
            grad[q:] += wk * (1.0 - h[:n-q])
            # -h[i-p] = -h[i+q] for i in [0, n-1-q]
            grad[:n-q] += -wk * h[q:]
    
    return smooth_val, grad

def linalg_obj(x, beta, mu, n):
    """Objective for L-BFGS-B: smooth_max + mu*(sum-n/2)^2"""
    h = np.clip(x, 0.0, 1.0)
    val, _ = smooth_max_obj_grad(h, beta)
    diff = np.sum(h) - n/2.0
    val += mu * diff * diff
    return val

def linalg_grad(x, beta, mu, n):
    """Gradient for L-BFGS-B."""
    h = np.clip(x, 0.0, 1.0)
    _, g = smooth_max_obj_grad(h, beta)
    diff = np.sum(h) - n/2.0
    g += 2.0 * mu * diff * np.ones(n)
    return g

def linalg_wrapper(x, args):
    beta, mu, n = args
    return linalg_obj(x, beta, mu, n)

def linalg_grad_wrapper(x, args):
    beta, mu, n = args
    return linalg_grad(x, beta, mu, n)

def optimize_lbfgsb(h, n, beta, mu, maxiter=500):
    """Run L-BFGS-B optimization."""
    args = (beta, mu, n)
    try:
        res = fmin_l_bfgs_b(linalg_wrapper, h, fprime=linalg_grad_wrapper,
                           args=args, m=20, pgtol=1e-10,
                           maxiter=maxiter, iprint=0,
                           factr=1e7)
        return project_feasible(res[0], n)
    except Exception:
        return h.copy()

def slsqp_smooth(h, n, beta, maxiter=300):
    """SLSQP with smooth max and equality constraint."""
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        exps = np.exp(beta * (corr - m))
        s = np.sum(exps)
        return (m + np.log(s) / beta) * (2.0 / n)
    
    def eq(x):
        return np.sum(x) - n/2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        exps = np.exp(beta * (corr - m))
        s = np.sum(exps)
        return (m + np.log(s) / beta) * dx
    def eq(x):
        return np.sum(x) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def minimax_slsqp(h_init, n, max_rounds=10, max_iter=200):
    global _ACTIVE_K, _LBFGS_N
    _LBFGS_N = n
    h = h_init.copy()
    dx = 2.0 / n
    for rnd in range(max_rounds):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        max_corr = np.max(corr)
        threshold = max_corr - 0.3
        active = np.where(corr >= threshold)[0]
        top5 = np.argsort(corr)[::-1][:5]
        active = np.unique(np.concatenate([active, top5]))
        active = np.sort(active)
        _ACTIVE_K = active
        x0 = np.concatenate([h, [max_corr * dx + 1e-7]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [{'type': 'ineq', 'fun': _mm_ineq},
                {'type': 'eq', 'fun': _mm_eq}]
        c5_old = compute_c5(h)
        try:
            res = minimize(_mm_obj, x0, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-13, 'iprint': 0})
            if res.x is not None:
                h_new = project_feasible(res.x[:n], n)
                c5_new = compute_c5(h_new)
                if c5_new < c5_old - 1e-15:
                    h = h_new
        except Exception:
            break
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def projected_gradient_step(h, n, beta, lr, max_inner=50):
    """Projected gradient descent with backtracking."""
    h_curr = h.copy()
    f_curr = _smooth_max_val(h_curr, beta)
    for _ in range(max_inner):
        if remaining_time() < 1:
            break
        val, grad = _smooth_max_grad(h_curr, beta)
        diff = np.sum(h_curr) - n / 2.0
        grad += 200.0 * diff
        h_try = h_curr - lr * grad
        h_try = project_feasible(h_try, n)
        f_try = _smooth_max_val(h_try, beta)
        if f_try < f_curr - 1e-16:
            h_curr = h_try
            f_curr = f_try
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    return h_curr

def local_search(h, n, max_sweeps=4, rng=None):
    if rng is None:
        rng = np.random.RandomState(1)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        order = rng.permutation(n)
        for ii in range(len(order)):
            if remaining_time() < 1:
                break
            i = int(order[ii])
            hi = best_h[i]
            if hi < 0.06 or hi > 0.94:
                continue
            for j in range(i + 1, min(i + 20, n)):
                hj = best_h[j]
                if abs(hi - hj) < 0.02:
                    continue
                h_trial = best_h.copy()
                amt = (hi - hj) * 0.5
                if abs(amt) < 0.003:
                    continue
                h_trial[i] -= amt
                h_trial[j] += amt
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < best_c5 - 1e-14:
                    best_c5 = c5_t
                    best_h = h_trial.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S, _ACTIVE_K, _LBFGS_N
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    _ACTIVE_K = None
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 256
        best_h = np.full(best_n, 0.5)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)
    
    n = best_n
    h = best_h.copy()
    
    # Phase 1: L-BFGS-B optimization (fast, many iterations)
    for beta in [30, 100, 500, 2000, 8000]:
        if remaining_time() < 12:
            break
        mu = 500.0
        h_new = optimize_lbfgs(h, n, beta, mu, maxiter=350)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 2: Projected gradient refinement
    if remaining_time() > 30:
        h_pg = projected_gradient_step(h, n, beta=3000, lr=0.005, max_inner=200)
        c5_pg = compute_c5(h_pg)
        if c5_pg < best_c5 - 1e-15:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()
    
    # Phase 3: SLSQP smooth (higher precision)
    for br in [1000, 5000, 20000]:
        if remaining_time() < 18:
            break
        h_new = slsqp_smooth(h, n, br, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 4: Minimax SLSQP
    if remaining_time() > 30:
        h_mm = minimax_slsqp(best_h, best_n, max_rounds=10, max_iter=200)
        c5_mm = compute_c5(h_mm)
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm.copy()
        if remaining_time() > 15:
            h_pol = slsqp_smooth(best_h, best_n, 10000, maxiter=250)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
    
    # Phase 5: Multi-resolution exploration
    trial_ns = [200, 224, 288, 320, 384, 448, 512, 640]
    for trial_n in trial_ns:
        if remaining_time() < 18:
            break
        h_interp = interp_to(best_h, best_n, trial_n)
        h_interp = project_feasible(h_interp, trial_n)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5 + 0.004:
            for bt in [200, 2000, 10000]:
                if remaining_time() < 7:
                    break
                h_new = optimize_lbfgs(h_interp, trial_n, bt, 500.0, maxiter=250)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = trial_n
                    h_interp = h_new.copy()
            if remaining_time() > 15 and trial_n <= 448:
                h_new = slsqp_smooth(h_interp, trial_n, 5000, maxiter=200)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = trial_n
    
    # Phase 6: Local combinatorial search
    if remaining_time() > 20:
        h_comb, c5_comb = local_search(best_h, best_n, max_sweeps=6, rng=rng)
        if c5_comb < best_c5:
            best_c5 = c5_comb
            best_h = h_comb.copy()
        if remaining_time() > 10:
            h_pol = slsqp_smooth(best_h, best_n, 5000, maxiter=200)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
    
    # Phase 7: Perturbation restarts
    for restart in range(4):
        if remaining_time() < 25:
            break
        h_p = best_h.copy()
        mask = (h_p > 0.04) & (h_p < 0.96)
        if np.sum(mask) > 0:
            h_p[mask] += rng.normal(0, 0.04, np.sum(mask))
        # Also try a phase-like flip on a subset
        flip_mask = (h_p > 0.3) & (h_p < 0.7)
        if np.sum(flip_mask) > 2:
            idxs = np.where(flip_mask)[0]
            subset = rng.choice(idxs, size=min(8, len(idxs)), replace=False)
            h_p[subset] = 1.0 - h_p[subset]
        h_p = project_feasible(h_p, best_n)
        for beta_r in [500, 5000]:
            if remaining_time() < 8:
                break
            h_new = slsqp_smooth(h_p, best_n, beta_r, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    # Phase 8: L-BFGS-B at high beta (final push)
    if remaining_time() > 20:
        h_new = optimize_lbfgs(best_h, best_n, 20000, 2000.0, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 9: Final minimax
    if remaining_time() > 15:
        h_mm = minimax_slsqp(best_h, best_n, max_rounds=8, max_iter=180)
        c5_mm = compute_c5(h_mm)
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm.copy()
    
    # Phase 10: Final local search + ultra-polish
    if remaining_time() > 10:
        h_comb, c5_comb = local_search(best_h, best_n, max_sweeps=3, rng=rng)
        if c5_comb < best_c5:
            best_c5 = c5_comb
            best_h = h_comb.copy()
    
    if remaining_time() > 8:
        h_new = slsqp_smooth(best_h, best_n, 50000, maxiter=100)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
