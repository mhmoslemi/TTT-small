- x_{k+1} = argmin_{x in feasible} { <g_k, x> + (1/(2*t))||x - y_k||^2 }
- y_{k+1} = (1 - alpha_{k+1}) * x_{k+1} + alpha_{k+1} * s_{k+1}

```python
def nesterov_subgradient(h_init, n, budget):
    dx = 2.0 / n
    h = h_init.copy()
    y = h_init.copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    # Nesterov parameters
    T = 0.1  # initial step size
    top_m = 10  # number of top functions to average
    
    start = time.time()
    for k in range(100000):
        if time.time() - start > budget:
            break
        if remaining() < 5:
            break
        
        # Compute correlations
        corr = compute_corr(h)
        
        # Find top_m indices
        top_idx = np.argsort(-corr[:n*2-1])[:top_m]
        
        # Compute average subgradient
        g = np.zeros(n)
        for ki in top_idx:
            off = ki - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off+np_] += (1.0 - h[:np_])
                g[:np_] -= h[off:off+np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:ao+np_])
                g[ao:ao+np_] -= h[:np_]
        g /= top_m
        
        # Nesterov step
        T = 1.0 / (k + 1)  # step size
        alpha = 2.0 / (k + 2)  # Nesterov parameter
        
        # x_{k+1} = argmin <g, x> + (1/(2T))||x - y||^2 s.t. feasible
        # This is: proj_feasible(y - T*g)
        x_new = project_feasible(y - T * g, n)
        
        # s = proj_feasible(0) = constant n/2 * ones(n)
        # y_{k+1} = (1-alpha)*x_{k+1} + alpha*s
        s = np.full(n, 0.5)  # sum = n/2, in [0,1]
        y = (1 - alpha) * x_new + alpha * s
        y = project_feasible(y, n)
        
        h = x_new
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
```

def multi_pair_equalize(h, n, top_m=10, max_iters=500):
    """Equalize top-m correlations by moving in averaged gradient directions."""
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:top_m]
        c_max = corr[top_idx[0]]
        c_min = corr[top_idx[-1]]
        gap = c_max - c_min
        if gap < 1e-10:
            break
        
        # Compute direction: reduce top, increase bottom
        # Use weighted average: weight proportional to (corr_k - c_mean)
        c_mean = np.mean(corr[top_idx])
        weights = corr[top_idx] - c_mean
        
        # Direction to reduce top correlations
        g_reduce = np.zeros(n)
        g_increase = np.zeros(n)
        for i, k in enumerate(top_idx):
            if weights[i] > 0:
                g_reduce += weights[i] * grad_corr_k(h, k)
            elif weights[i] < 0:
                g_increase += (-weights[i]) * grad_corr_k(h, k)
        
        # We want to reduce the top ones and increase the bottom ones
        # Direction: -g_reduce + g_increase (scaled)
        d = g_increase - g_reduce
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        
        # Line search
        improved = False
        for step in [0.01, 0.02, 0.005, 0.05, 0.015, 0.008, 0.03]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)

                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            # Try very small step

            cn = compute_c5(ht)
            if cn < best_c5:
                h = ht

                break
    
    return h, best_c5
