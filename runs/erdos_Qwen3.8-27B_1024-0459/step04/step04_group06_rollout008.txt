def smooth_obj(x, target, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    logsum = np.log(np.sum(exps))
    return (m + logsum / beta) * dx

def smooth_grad(x, target, n, beta):
    # Numerical gradient of smooth objective
    dx = 2.0 / n
    eps = 1e-7
    grad = np.zeros(n)
    for i in range(n):
        x_plus = x.copy()
        x_plus[i] += eps
        x_minus = x.copy()
        x_minus[i] -= eps
        grad[i] = (smooth_obj(x_plus, target, n, beta) - smooth_obj(x_minus, target, n, beta)) / (2 * eps)
    return grad
