    h = h_init.copy()
    lr = lr_init
    best_h = h.copy()
    best_val = compute_c5(h)
    
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        val, grad = smooth_max_grad(h, beta)
        
        # Adaptive step: if gradient norm is very small, stop
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
        
        # Update
        h_new = h - lr * grad
        
        # Project
        h_new = project_feasible(h_new, n)
        
        # Check constraint and objective
        c5_new = compute_c5(h_new)
        if c5_new < best_val:
            best_val = c5_new
            best_h = h_new.copy()
        h = h_new
        
        # Decay learning rate
        lr *= decay
        
        # Check convergence
        if it > 100 and np.abs(val - best_val) < 1e-10:
            break
    
    return best_h, best_val

def slsqp_smooth(h, n, beta, maxiter=200):
    """SLSQP with smooth max objective."""
    def obj(x):
        h_x = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h_x, 1.0 - h_x, mode='full')
        dx = 2.0 / n
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx
    
    def eq(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq}]
    try:
        res = minimize(obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.success or res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def compute_c5(h):

def compute_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:

def slsqp_smooth(h, n, beta, maxiter=200):
    def obj(x):
        h_x = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h_x, 1.0 - h_x, mode='full')
        dx = 2.0 / n
        m = np.max(corr)

        return np.sum(x) - n / 2.0

def projected_grad_step(h, beta, lr):
    """One projected gradient descent step on smooth max."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_max = np.max(corr)
    shifted = beta * dx * (corr - m_max)
    # Numerical stability
    shifted -= np.max(shifted)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    # Compute gradient
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)
    # Only process lags with significant weight
    thresh = 1e-10
    active = np.where(weights > thresh)[0]
    for ai in active:
        l = lags[ai]
        w = weights[ai]
        # d corr[ai]/dh[i] = (1-h[i+l]) - h[i-l] (boundary safe)
        if l >= 0:
            grad[:n-l] += w * (1.0 - h[l:n])
            grad[l:n] -= w * h[0:n-l]
        else:
            lo = -l
            grad[lo:n] += w * (1.0 - h[0:n+lo])
            grad[0:lo] -= w * h[lo:n]
    grad *= dx
    
    # Update and project
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_feasible(h_new, n)
    return h_new

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
