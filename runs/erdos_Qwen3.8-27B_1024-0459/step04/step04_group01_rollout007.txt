            h[j] = 1.0
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            if new_max < cur_max - 1e-12:
                corr = new_corr
                cur_max = new_max
                best_c5 = cur_max * dx_val
                best_h = h.copy()
                found = True
                break
            else:
                h[i] = 1.0
                h[j] = 0.0

        if found:
            stall = 0
        else:
            stall += 1
            if stall > 300:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                if i != j:
                    h[i] = 0.0
                    h[j] = 1.0
                    corr = compute_corr(h)
                    cur_max = float(np.max(corr))
                    stall = 0

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 320
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP with increasing smoothness parameter
    for beta in [100, 300, 1000, 3000, 8000, 20000]:
        if remaining() < 8:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Equalize-and-descent (targeted at top shifts)
    if remaining() > 30:
        tl = min(80, max(10, remaining() - 50))
        h_eq, c5_eq = equalize_and_descent(h, n, time_limit=tl, top_k=4, rng=rng)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 3: SLSQP polish after equalization
    for beta in [5000, 15000, 50000]:
        if remaining() < 6:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=350)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: More equalize-and-descent
    if remaining() > 20:
        tl = min(50, max(10, remaining() - 40))
        h_eq, c5_eq = equalize_and_descent(h, n, time_limit=tl, top_k=5, rng=rng)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 5: Binary local search
    if remaining() > 20:
        h_bin = (best_h > 0.5).astype(float)
        tl = min(40, max(10, remaining() - 30))
        h_b, c5_b = binary_local_search(h_bin, best_n, time_limit=tl, rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 6: Try different resolutions
    n_candidates = sorted(set([
        best_n + 4, best_n + 8, best_n + 16, best_n + 32,
        best_n + 64, best_n + 128, max(200, best_n - 8)
    ]))
    n_candidates = [nn for nn in n_candidates if 200 <= nn <= 1500 and nn != best_n]

    for n_t in n_candidates:
        if remaining() < 35:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)

        h_opt = h_feas
        for beta in [300, 1500, 6000]:
            if remaining() < 8:
                break
            h_opt = slsqp_opt(h_opt, n_t, beta, maxiter=300)

        if remaining() > 20:
            tl = min(25, max(8, remaining() - 20))
            h_opt, _ = equalize_and_descent(h_opt, n_t, time_limit=tl, top_k=4, rng=rng)

        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 7: Simulated annealing on best binary
    if remaining() > 20:
        h_bin = (best_h > 0.5).astype(float)
        tl = min(35, max(10, remaining() - 25))
        h_sa, c5_sa = binary_sa(h_bin, best_n, time_limit=tl, rng=rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 8: Final intensive SLSQP + equalize
    h = best_h.copy()
    n = best_n
    for beta in [10000, 30000, 80000]:
        if remaining() < 5:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining() > 15:
        tl = min(30, max(5, remaining() - 15))
        h_eq, c5_eq = equalize_and_descent(h, n, time_limit=tl, top_k=6, rng=rng)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 9: Final binary polish
    if remaining() > 15:
        h_bin = (best_h > 0.5).astype(float)
        tl = min(25, max(5, remaining() - 10))
        h_b, c5_b = binary_local_search(h_bin, best_n, time_limit=tl, rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 10: Very final SLSQP
    if remaining() > 8:
        h_new = slsqp_opt(best_h, best_n, 50000, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Validate and return
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
