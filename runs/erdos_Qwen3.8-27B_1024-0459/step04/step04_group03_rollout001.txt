import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        # Distribute among all non-bound points
        for _ in range(10):
            mask = (h > 1e-8) & (h < 1.0 - 1e-8)
            if np.sum(mask) > 0:
                h[mask] += diff / np.sum(mask)
            else:
                # Find any point and adjust
                idx = np.argmin(np.abs(h - 0.5))
                h[idx] = np.clip(h[idx] + diff, 0, 1)
            h = np.clip(h, 0.0, 1.0)
            diff = target - np.sum(h)
            if abs(diff) < 1e-14:
                break
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)

def compute_smooth_grad(h, n, beta):
    """Compute gradient of (dx/beta)*log(sum(exp(beta*corr)))"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(beta * (corr - max_c))
    w = exp_c / np.sum(exp_c)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        wk = w[k]
        if wk < 1e-15:
            continue
        if lag == 0:
            # ∂c/∂h[j] = 1 - 2*h[j]
            grad += wk * (1.0 - 2.0 * h)
        else:
            # Term 1: (1-h[j+lag]) for j where 0 <= j+lag < n
            # j ranges from max(0, -lag) to min(n-1, n-1-lag)
            j_start = max(0, -lag)
            j_end = min(n, n - lag)
            if j_end > j_start:
                grad[j_start:j_end] += wk * (1.0 - h[j_start + lag:j_end + lag])
            
            # Term 2: -h[j-lag] for j where 0 <= j-lag < n
            # j ranges from max(0, lag) to min(n-1, n-1+lag)
            j_start2 = max(0, lag)
            j_end2 = min(n, n + lag)
            if j_end2 > j_start2:
                grad[j_start2:j_end2] += wk * (-h[j_start2 - lag:j_end2 - lag])
    
    grad *= dx
    return grad

def frank_wolfe(h, n, beta, max_iter=200):
    """Frank-Wolfe on the smooth log-sum-exp objective."""
    dx = 2.0 / n
    h = h.copy()
    k_half = n // 2
    best_c5 = compute_c5(h)
    
    for t in range(max_iter):
        if remaining_time() < 3:
            break
        grad = compute_smooth_grad(h, n, beta)
        # Find FW vertex: n/2 indices with smallest gradient
        idx = np.argsort(grad)[:k_half]
        v = np.zeros(n)
        v[idx] = 1.0
        # FW step size
        gamma = 2.0 / (t + 2)
        h_new = (1 - gamma) * h + gamma * v
        h_new = np.clip(h_new, 0, 1)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            h = h_new
            best_c5 = c5_new
        else:
            # Try smaller step
            gamma = 1.0 / (t + 2)
            h_new = (1 - gamma) * h + gamma * v
            h_new = np.clip(h_new, 0, 1)
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                h = h_new
                best_c5 = c5_new
            # Check termination
            gap = np.max(grad) - np.min(grad)
            if gap < 1e-8:
                break
    return h, best_c5

def coordinate_descent(h, n, max_sweeps=200):
    """Coordinate descent: optimize each h[j] in turn."""
    dx = 2.0 / n

                h[j] = old_val  # restore

def slsqp_smooth(h, n, beta, maxiter=500):
    """SLSQP on smooth log-sum-exp objective."""
    dx = 2.0 / n
    
    def sm_obj(x):
        h_clip = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h_clip, 1.0 - h_clip, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx
    
    def sm_eq(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sm_eq}]
    try:
        res = minimize(sm_obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and res.x is not h:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def interp_to(src_h, src_n, dst_n):

    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)
