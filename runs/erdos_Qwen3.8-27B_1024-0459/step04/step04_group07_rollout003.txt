    grad = np.zeros(n)
    for k in top_idx:
        # k is index in full correlation (0 to 2n-2)
        # full[k] = sum_n h[n + k_offset] * (1-h[n]) where k_offset depends on k
        # Actually for np.correlate(h, 1-h, mode='full'):
        # result[k] for k in [0, 2n-2]
        # result[k] = sum of h[min(n+k, 2n-1-i)] ... this is getting complex
        # Let me just use the simplest approach: for lag l (where l = k - (n-1))
        # c_l = sum_i h[i+l]*(1-h[i]) for valid i
        
        l = int(k) - (n - 1)  # actual lag
        for i in range(max(0, -l), min(n, n - l)):
            # h[i+l]*(1-h[i]) contributes to grad[i] and grad[i+l]
            if 0 <= i < n and 0 <= i + l < n:
                grad[i] += -(1.0 - h[i + l])  # d/dh[i] of h[i+l]*(1-h[i]) = -h[i+l]
                # Wait, that's not right either.
                # c_l = sum_i h[i+l]*(1-h[i])
                # d/dh[j] c_l: if j = i (coefficient is -1*h[i+l] = -h[j+l])... 
                # Actually: d/dh[j] [h[i+l]*(1-h[i])] = (1-h[i]) * d(h[i+l])/dh[j] + h[i+l] * d(1-h[i])/dh[j]
                # = (1-h[i]) * delta(i+l, j) - h[i+l] * delta(i, j)
                # For j = i: contribution is -h[i+l] = -h[j+l]
                # For j = i+l: contribution is (1-h[i]) = (1-h[j-l])
    
    # This is getting complex. Let me use a simpler approach:
    # Just compute the gradient numerically or use the correlation structure directly.
    
    # Simpler: the gradient of c_l w.r.t. h is:
    # (1-h shifted by -l) - (h shifted by l), with zeros where out of range
    for k in top_idx:
        l = int(k) - (n - 1)
        g_k = np.zeros(n)
        if l >= 0:
            # c_l = sum_{i=0}^{n-1-l} h[i+l]*(1-h[i])
            for i in range(n - l):
                g_k[i] += -h[i + l]        # d/dh[i]
                g_k[i + l] += (1.0 - h[i])  # d/dh[i+l]
        else:
            l_abs = -l
            for i in range(n - l_abs):
                g_k[i + l_abs] += -h[i]     # d/dh[i+l_abs]
                g_k[i] += (1.0 - h[i + l_abs])  # d/dh[i]
        grad += g_k
    
    h_new = h - lr * grad
    return project_feasible(h_new, n)

def multi_beta_slsqp(h0, n, betas, topk=0, base_iter=300, tlimit=None):
    """Run SLSQP with increasing beta values."""
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    for bv in betas:
        if tlimit is not None and tlimit() < 10:
            break
        h_new = slsqp_run(h_cur, n, bv, maxiter=base_iter, topk=topk)
        c_new = compute_c5(h_new)
        if c_new < c5_cur - 1e-13:
            c5_cur = c_new
            h_cur = h_new
    return h_cur, c5_cur

def subgrad_polish(h0, n, tlimit, max_iters=200, topk=3):
    """Subgradient method for the active lags."""
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    lr = 0.01 / np.sqrt(n)  # initial learning rate
    for it in range(max_iters):
        if tlimit() < 3.0:
            break
        h_new = subgradient_step(h_cur, n, lr, topk=topk)
        c_new = compute_c5(h_new)
        if c_new < c5_cur - 1e-12:
            c5_cur = c_new
            h_cur = h_new
            lr = min(lr * 1.01, 0.05 / np.sqrt(n))
        else:
            lr *= 0.9
        if lr < 1e-10:
            break
    return h_cur, c5_cur

def focused_multi(h, n, rng, tlimit, max_rounds=40, topk=3):
    """Iteratively identify top-k active lags and focus optimization on them."""
    c5 = compute_c5(h)
    h_best = h.copy()
    for rnd in range(max_rounds):
        if tlimit() < 5.0:
            break
        corr = get_corr(h_best)
        top_idx = np.argsort(corr)[::-1][:topk]
        
        # SLSQP with focus on top-k lags
        h_new = slsqp_run(h_best, n, beta=500.0, maxiter=250, topk=topk)
        c_new = compute_c5(h_new)
        
        # Also try subgradient
        h_sg = subgradient_step(h_best, n, 0.005, topk=topk)
        h_sg = slsqp_run(h_sg, n, beta=1000.0, maxiter=200, topk=topk)
        c_sg = compute_c5(h_sg)
        
        if c_sg < c_new:
            h_new, c_new = h_sg, c_sg
        
        if c_new < c5 - 1e-12:
            c5 = c_new
            h_best = h_new
    
    return h_best, c5

def coord_descent(h, n, max_sweeps, rng, tlimit):
    c5 = compute_c5(h)
    h_best = h.copy()
    for sweep in range(max_sweeps):
        if tlimit() < 2.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if tlimit() < 1.0:
                break
            v0 = h_best[i]
            # Try binary values
            for v in [0.0, 1.0]:
                if abs(v - v0) < 1e-9:
                    continue
                trial = h_best.copy()
                trial[i] = v
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 1e-10:
                    c5 = c
                    h_best = trial.copy()
                    improved = True
            # Try continuous perturbations
            for d in [0.1, -0.1, 0.05, -0.05, 0.2, -0.2, 0.02, -0.02]:
                trial = h_best.copy()
                trial[i] = np.clip(trial[i] + d, 0.0, 1.0)
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 1e-10:
                    c5 = c
                    h_best = trial.copy()
                    improved = True
        if not improved and sweep >= 2:
            break
    return h_best, c5

def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX, _G_PEN, _G_FOCUS, _G_TOPK
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    _G_PEN = 0.0
    _G_FOCUS = -1
    _G_TOPK = 0
    rng = np.random.RandomState(seed)
    
    def tlimit():
        return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = project_feasible(make_tm(parent_n), parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("No parent. TM: n={}, C5={:.8f}".format(parent_n, best_c5))
    
    # Phase 1: Multi-beta SLSQP at parent resolution with top-k focus
    if tlimit() > 100:
        print("\n=== Phase 1: n={} multi-beta with topk ===".format(best_n))
        n = best_n
        h_cur = best_h.copy()
        
        # Try both full and top-k variants
        for topk in [0, 3, 5]:
            if tlimit() < 15:
                break
            betas = [50, 100, 200, 500, 1000, 2000, 5000]
            h_new, c_new = multi_beta_slsqp(h_cur, n, betas, topk=topk, 
                                             base_iter=300, tlimit=tlimit)
            if c_new < best_c5 - 1e-12:
                best_c5 = c_new
                best_h = h_new.copy()
                h_cur = h_new.copy()
        
        print("  After multi-beta: C5={:.8f}".format(best_c5))
    
    # Phase 2: Subgradient polish
    if tlimit() > 50:
        print("\n=== Phase 2: subgradient polish ===")
        best_n = len(best_h)
        best_h, best_c5 = subgrad_polish(best_h, best_n, tlimit, max_iters=100, topk=3)
        print("  After subgrad: C5={:.8f}".format(best_c5))
    
    # Phase 3: Focused multi-lag iteration
    if tlimit() > 60:
        print("\n=== Phase 3: focused multi-lag ===")
        best_n = len(best_h)
        best_h, best_c5 = focused_multi(best_h, best_n, rng, tlimit, max_rounds=30, topk=3)
        print("  After focused: C5={:.8f}".format(best_c5))
    
    # Phase 4: Try other resolutions with diverse inits
    if tlimit() > 100:
        print("\n=== Phase 4: other resolutions ===")
        n_tries = []
        n_base = best_n
        for candidate in [n_base, n_base - 32, n_base + 32, 192, 288, 320, 384, 448]:
            if 128 <= candidate <= 512 and candidate % 2 == 0 and candidate not in n_tries:
                n_tries.append(candidate)
        n_tries.sort()
        
        for n in n_tries:
            if tlimit() < 50:
                break
            if n == best_n:
                # Already optimized, skip or do one more round
                continue
            
            # Multiple initializations
            h_interp = project_feasible(interp_h(best_h, best_n, n), n)
            h_tm = project_feasible(make_tm(n), n)
            h_rand = project_feasible(make_rand_bin(rng, n), n)
            h_four = project_feasible(make_fourier(rng, n, 5), n)
            h_wave = project_feasible(make_wavelet(rng, n, 4), n)
            
            inits = [h_interp, h_tm, h_rand, h_four, h_wave]
            inits.sort(key=lambda x: compute_c5(x))
            
            h_cur = inits[0].copy()
            c5_cur = compute_c5(h_cur)
            
            # Optimize
            for topk in [0, 5]:
                if tlimit() < 10:
                    break
                betas = [100, 300, 800, 2000, 5000]
                h_new, c_new = multi_beta_slsqp(h_cur, n, betas, topk=topk,
                                                 base_iter=250, tlimit=tlimit)
                if c_new < c5_cur - 1e-12:
                    c5_cur = c_new
                    h_cur = h_new
            
            # Subgradient polish
            if tlimit() > 15:
                h_cur, c5_cur = subgrad_polish(h_cur, n, tlimit, max_iters=30, topk=3)
            
            if c5_cur < best_c5 - 1e-9:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n
                print("  n={} improved: C5={:.8f}".format(n, best_c5))
            else:
                print("  n={} best: C5={:.8f}".format(n, c5_cur))
    
    # Phase 5: Coordinate descent
    if tlimit() > 30:
        print("\n=== Phase 5: coordinate descent ===")
        best_n = len(best_h)
        best_h, best_c5 = coord_descent(best_h, best_n, max_sweeps=30, rng=rng, tlimit=tlimit)
        print("  After coord: C5={:.8f}".format(best_c5))
    
    # Phase 6: Perturbation escapes
    if tlimit() > 60:
        print("\n=== Phase 6: perturbation ===")
        n = best_n
        for attempt in range(6):
            if tlimit() < 15:
                break
            h_pert = best_h.copy()
            n_pert = rng.randint(2, 8)
            idxs = rng.choice(n, size=n_pert, replace=False)
            for idx in idxs:
                h_pert[idx] = np.clip(h_pert[idx] + rng.randn() * 0.2, 0.0, 1.0)
            h_pert = project_feasible(h_pert, n)
            c5_p = compute_c5(h_pert)
            
            if c5_p < best_c5 * 1.03:
                betas = [500, 2000, 5000]
                h_cur, c_new = multi_beta_slsqp(h_pert, n, betas, topk=3,
                                                 base_iter=200, tlimit=tlimit)
                if c_new < best_c5 - 1e-10:
                    best_c5 = c_new
                    best_h = h_cur.copy()
                    print("  Perturb {} improved: C5={:.8f}".format(attempt, best_c5))
    
    # Phase 7: Final intensive polish
    if tlimit() > 20:
        print("\n=== Phase 7: final polish ===")
        best_n = len(best_h)
        for bv in [3000, 8000, 15000]:
            if tlimit() < 5:
                break
            h_new = slsqp_run(best_h, best_n, bv, maxiter=400, topk=3)
            c_new = compute_c5(h_new)
