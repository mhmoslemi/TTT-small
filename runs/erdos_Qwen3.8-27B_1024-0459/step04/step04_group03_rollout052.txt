def lbfgs_optimize(h_init, n, beta, max_lam_iters=12, max_inner=0):
    global _N_G, _BETA_G, _LAM_G
    _N_G = n
    _BETA_G = beta
    h = h_init.copy()
    lam = 0.0
    penalty_mu = 100.0
    for li in range(max_lam_iters):
        if remaining_time() < 5:
            break
        _LAM_G = lam
        bounds = [(0.0, 1.0)] * n
        res = minimize(_smooth_obj_grad, h, method='L-BFGS-B',
                       jac=True, bounds=bounds,
                       options={'maxiter': max_inner if max_inner > 0 else int(min(300, 25000 / n)),
                                'ftol': 1e-18, 'gtol': 1e-12, 'iprint': 0,
                                'maxls': 50, 'maxfun': 5000})
        if res.x is not None:
            h = np.clip(res.x, 0.0, 1.0)
        h = project_feasible(h, n)
        constraint_viol = abs(np.sum(h) - n / 2.0)
        if constraint_viol < 1e-10:
            break
        lam = lam + penalty_mu * (np.sum(h) - n / 2.0)
        if abs(lam) > 10.0:
            lam = np.clip(lam, -10.0, 10.0)
        penalty_mu *= 2.0
        if penalty_mu > 1e6:
            penalty_mu = 1e6
    return h

def coord_descent_step(h, n, max_sweeps=20):
    """Coordinate descent exploiting 1D linear structure of each subproblem."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if remaining_time() < 2:
            break
        improved = False
        for i in range(n):
            if remaining_time() < 1:
                break
            hi = best_h[i]
            if hi < 1e-6 or hi > 1.0 - 1e-6:
                continue
            # Compute A'_p and B_p for each correlation index p
            # c_p(t) = A'_p + t * B_p
            shifts = np.arange(2 * n - 1) - (n - 1)
            # For each p: B_p = (1-h[i-shift_p])*valid1 - h[i+shift_p]*valid2
            # valid1: 0 <= i-shift_p < n
            # valid2: 0 <= i+shift_p < n
            shifts_arr = shifts.astype(float)
            valid1 = (i - shifts_arr >= 0) & (i - shifts_arr < n)
            valid2 = (i + shifts_arr >= 0) & (i + shifts_arr < n)
            # h values at shifted positions
            idx1 = np.where(valid1, i - shifts_arr.astype(int), 0)
            idx2 = np.where(valid2, i + shifts_arr.astype(int), 0)
            h_at_1 = np.where(valid1, best_h[idx1], 0.0)
            h_at_2 = np.where(valid2, best_h[idx2], 0.0)
            B_p = (1.0 - h_at_1) * valid1 - h_at_2 * valid2
            # A'_p = c_p(hi) - hi * B_p
            c_p = np.correlate(best_h, 1.0 - best_h, mode='full')
            A_p = c_p - hi * B_p
            # Now find min_{t in [0,1]} max_p (A_p + t*B_p)
            # Using subgradient bisection
            lo, hi_t = 0.0, 1.0
            for _ in range(40):
                mid = 0.5 * (lo + hi_t)
                vals = A_p + mid * B_p
                max_val = np.max(vals)
                # Subgradient: B_p values where A_p + mid*B_p is near max
                mask_top = vals > max_val - 1e-10
                if mask_top.any():
                    bgs = B_p[mask_top]
                    if np.max(bgs) < -1e-10:
                        lo = mid
                    elif np.min(bgs) > 1e-10:
                        hi_t = mid
                    else:
                        break
                else:
                    break
            t_opt = 0.5 * (lo + hi_t)
            # Evaluate at t_opt and a few nearby points
            candidates = [0.0, 1.0, t_opt, max(0.0, t_opt - 0.05), min(1.0, t_opt + 0.05)]
            best_t = hi
            best_obj = np.inf
            for t_c in candidates:
                t_c = np.clip(t_c, 0.0, 1.0)
                vals_c = A_p + t_c * B_p
                obj_c = np.max(vals_c)
                if obj_c < best_obj:
                    best_obj = obj_c
                    best_t = t_c
            if abs(best_t - hi) < 1e-10:
                continue
            best_h[i] = best_t
            # Quick check
            c5_new = compute_c5(best_h)
            if c5_new < best_c5:
                best_c5 = c5_new
                improved = True
            else:
                best_h[i] = hi
        if not improved:
            break
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def slsqp_smooth(h, n, beta, maxiter=200):
    global _N_G, _BETA_G
    _N_G = n
    _BETA_G = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    try:
        res = minimize(_smooth_obj_grad, h, method='SLSQP',
                       bounds=bounds, constraints=cons, jac=True,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def minimax_slsqp(h_init, n, max_rounds=10):
    global _N_G, _BETA_G
    _N_G = n
    _BETA_G = 100.0
    h = h_init.copy()
    dx = 2.0 / n
    for rnd in range(max_rounds):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        max_corr = np.max(corr)
        threshold = max_corr * 0.6
        active = np.where(corr >= threshold)[0]
        top_k = np.argsort(corr)[::-1][:5]
        active = np.unique(np.concatenate([active, top_k]))
        n_act = len(active)
        
        def mm_obj(x):
            h_x = x[:n]
            t = x[n]
            c = np.correlate(h_x, 1.0 - h_x, mode='full')
            act = c[active]
            m = np.maximum(t, np.max(act))
            shift = np.clip(20.0 * (act - m), -50, 0)
            return m * dx + np.log(np.sum(np.exp(shift))) / 20.0 * dx
        
        def mm_obj_grad(x):
            h_x = x[:n]
            t = x[n]
            c = np.correlate(h_x, 1.0 - h_x, mode='full')
            act = c[active]
            m = np.maximum(t, np.max(act))
            shift = np.clip(20.0 * (act - m), -50, 0)
            exp_s = np.exp(shift)
            sum_e = np.sum(exp_s)
            obj = m * dx + np.log(sum_e) / 20.0 * dx
            # Gradient w.r.t. t
            gt = dx / max(m, 1e-15)
            # Gradient w.r.t. h
            weights = exp_s / sum_e
            dh = np.zeros(n)
            for ai in active:
                s = ai - (n - 1)
                wp = weights[ai] / 20.0
                if s == 0:
                    dh += wp * (1.0 - 2.0 * h_x)
                elif s > 0:
                    dh[s:] += wp * (1.0 - h_x[:n - s])
                    dh[:n - s] -= wp * h_x[s:]
                else:
                    sp = -s
                    dh[:n - sp] += wp * (1.0 - h_x[sp:])
                    dh[sp:] -= wp * h_x[:n - sp]
            # Also gradient from m term
            if t > max(np.max(act) - 1e-10, 0):
                pass
            grad = np.zeros(n + 1)
            grad[:n] = dh * dx
            grad[n] = gt
            return obj, grad
        
        bounds = [(0.0, 1.0)] * n + [(0.0, dx * 2.0)]
        x0 = np.concatenate([h, [max_corr * dx + 1e-4]])
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - n / 2.0}]
        try:
            res = minimize(mm_obj_grad, x0, method='L-BFGS-B',
                           jac=True, bounds=bounds,
                           options={'maxiter': 200, 'ftol': 1e-16, 'iprint': 0, 'maxfun': 2000})
            if res.x is not None:
                h_new = project_feasible(res.x[:n], n)
                c5_new = compute_c5(h_new)
                c5_old = compute_c5(h)
                if c5_new < c5_old:
                    h = h_new
        except Exception:
            break
    return h

def binary_structure_search(h_best, n_best, rng):
    """Try different binary base structures."""
    best_h = h_best.copy()
    best_c5 = compute_c5(h_best)
    dx = 2.0 / n_best
    # Generate random binary patterns with n/2 ones
    for trial in range(8):
        if remaining_time() < 10:
            break
        binary = np.zeros(n_best)
        ones_pos = rng.choice(n_best, size=n_best // 2, replace=False)
        binary[ones_pos] = 1.0
        # Smooth it slightly
        h_trial = binary * 0.8 + 0.1
        h_trial = project_feasible(h_trial, n_best)
        c5_t = compute_c5(h_trial)
        if c5_t < 1.0:
            # Quick L-BFGS polish
            _tmp_n, _tmp_b = _N_G, _BETA_G
            h_opt = lbfgs_optimize(h_trial, n_best, beta=200, max_lam_iters=5)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            _N_G, _BETA_G = _tmp_n, _tmp_b
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S, _N_G, _BETA_G, _LAM_G
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 256
        best_h = np.full(best_n, 0.5)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: L-BFGS-B with escalating beta (fast analytic gradient)
    for beta in [50, 150, 400, 1000, 3000, 8000, 20000]:
        if remaining_time() < 8:
            break
        h_new = lbfgs_optimize(h, n, beta, max_lam_iters=10)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Coordinate descent (exploits 1D linear structure)
    if remaining_time() > 20:
        h_cd, c5_cd = coord_descent_step(best_h, best_n, max_sweeps=30)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        # L-BFGS polish after CD
        if remaining_time() > 8:
            h_pol = lbfgs_optimize(best_h, best_n, beta=5000, max_lam_iters=5)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 3: Minimax SLSQP (focus on top lags)
    if remaining_time() > 25:
        h_mm = minimax_slsqp(best_h, best_n, max_rounds=8)
        c5_mm = compute_c5(h_mm)
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm.copy()
        if remaining_time() > 10:
            h_pol = lbfgs_optimize(best_h, best_n, beta=10000, max_lam_iters=5)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 4: Multi-resolution exploration
    trial_ns = [192, 224, 256, 288, 320, 384, 448, 512]
    for trial_n in trial_ns:
        if remaining_time() < 15:
            break
        h_interp = interp_to(best_h, best_n, trial_n)
        h_interp = project_feasible(h_interp, trial_n)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5 + 0.003:
            for beta in [200, 1000, 5000]:
                if remaining_time() < 5:
                    break
                h_new = lbfgs_optimize(h_interp, trial_n, beta, max_lam_iters=8)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = trial_n
                    h_interp = h_new.copy()
            # Coordinate descent at this resolution
            if remaining_time() > 10 and trial_n <= 384:
                h_cd2, c5_cd2 = coord_descent_step(h_interp, trial_n, max_sweeps=10)
                if c5_cd2 < best_c5:
                    best_c5 = c5_cd2
                    best_h = h_cd2.copy()
                    best_n = trial_n

    # Phase 5: More coordinate descent on best
    if remaining_time() > 20:
        h_cd3, c5_cd3 = coord_descent_step(best_h, best_n, max_sweeps=40)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3.copy()
        if remaining_time() > 8:
            h_pol = lbfgs_optimize(best_h, best_n, beta=20000, max_lam_iters=5)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 6: Binary structure exploration
    if remaining_time() > 30:
        h_bs, c5_bs = binary_structure_search(best_h, best_n, rng)
        if c5_bs < best_c5:
            best_c5 = c5_bs
            best_h = h_bs.copy()

    # Phase 7: Perturbation + re-optimize
    for restart in range(3):
        if remaining_time() < 25:
            break
        h_p = best_h.copy()
        mask = (h_p > 0.05) & (h_p < 0.95)
        if np.sum(mask) > 0:
            h_p[mask] += rng.normal(0, 0.03, np.sum(mask))
        # Flip a few random bits
        n_flip = min(3, n // 10)
        idx_flip = rng.choice(n, size=n_flip, replace=False)
        for i in idx_flip:
            h_p[i] = 1.0 - h_p[i]
        h_p = project_feasible(h_p, n)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5 + 0.01:
            h_p2 = lbfgs_optimize(h_p, n, beta=2000, max_lam_iters=8)
            c5_p2 = compute_c5(h_p2)
            if c5_p2 < best_c5:
                best_c5 = c5_p2
                best_h = h_p2.copy()
            if remaining_time() > 10:
                h_cd4, c5_cd4 = coord_descent_step(h_p2, n, max_sweeps=10)
                if c5_cd4 < best_c5:
                    best_c5 = c5_cd4
                    best_h = h_cd4.copy()

    # Phase 8: SLSQP refinement (different from L-BFGS-B)
    if remaining_time() > 15:
        h_sl = slsqp_smooth(best_h, best_n, beta=5000, maxiter=200)
        c5_sl = compute_c5(h_sl)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()

    # Phase 9: Final ultra-precision L-BFGS
    if remaining_time() > 10:
        for beta in [30000, 50000, 100000]:
            if remaining_time() < 4:
                break
            h_new = lbfgs_optimize(best_h, best_n, beta, max_lam_iters=5)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 10: Final coordinate descent + binarization
    if remaining_time() > 10:
        h_cd5, c5_cd5 = coord_descent_step(best_h, best_n, max_sweeps=20)
        if c5_cd5 < best_c5:
            best_c5 = c5_cd5
            best_h = h_cd5.copy()
        # Try gentle binarization
        if remaining_time() > 8:
            h_bin = best_h.copy()
            h_bin[h_bin < 0.08] = 0.0
            h_bin[h_bin > 0.92] = 1.0
            h_bin = project_feasible(h_bin, best_n)
            c5_bin = compute_c5(h_bin)
            if c5_bin < best_c5 + 0.002:
                h_b2 = lbfgs_optimize(h_bin, best_n, beta=2000, max_lam_iters=5)
                c5_b2 = compute_c5(h_b2)
                if c5_b2 < best_c5:
                    best_c5 = c5_b2
                    best_h = h_b2.copy()

    # Phase 11: Final minimax
    if remaining_time() > 12:
        h_mm2 = minimax_slsqp(best_h, best_n, max_rounds=5)
        c5_mm2 = compute_c5(h_mm2)
        if c5_mm2 < best_c5:
            best_c5 = c5_mm2
            best_h = h_mm2.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
