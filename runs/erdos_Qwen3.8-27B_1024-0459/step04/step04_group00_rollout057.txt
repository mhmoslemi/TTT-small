        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]

    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = step_init

            break
        corr = compute_corr(h)
        c5_cur = float(np.max(corr)) * dx_v

            best_h = h.copy()

def steepest_descent(h, n, top_k=10, max_iters=5000, init_step=0.05):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = init_step
    
    for it in range(max_iters):
        if remaining() < 1:
            break
        corr = compute_corr(h)
        c5_cur = float(np.max(corr)) * (2.0/n)
        
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
        
        # Top-k
        idx = np.argsort(-corr)[:top_k]
        # Softmax weights
        c_top = corr[idx] - corr[idx[0]]
        w = np.exp(c_top * 100)
        w /= np.sum(w)
        
        # Gradient
        g = np.zeros(n)
        for wt, ki in zip(w, idx):
            d = ki - (n - 1)
            g += wt * grad_offset(h, d, n)
        
        # Project to sum-preserving
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        
        # Line search
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1, alpha*0.05]:
            h_new = project_feasible(h - a * g, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                h = h_new
                alpha = min(a * 1.5, 0.5)
                improved = True
                break
        
        if not improved:
            alpha *= 0.5
            if alpha < 1e-7:
                break
    
    return best_h, best_c5

        g[a:] -= h[:np_]
    return g

def steepest_descent(h, n, top_k=10, max_iters=5000, init_step=0.05, beta=100):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = init_step
    n_improve = 0
    
    for it in range(max_iters):
        if remaining() < 1:
            break
        corr = compute_corr(h)
        dx_v = 2.0 / n
        c5_cur = float(np.max(corr)) * dx_v
        
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            n_improve = 0
        else:
            n_improve += 1
        
        if n_improve > 500 and alpha < 1e-6:
            break
        
        # Top-k with softmax weights
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_top = corr[idx] - c_max
        w = np.exp(-beta * (-c_top))  # = exp(beta * c_top)
        # Wait, c_top ≤ 0, so -c_top ≥ 0, exp(-beta*(-c_top)) = exp(beta*c_top) ≤ 1
        # For the max element, c_top = 0, w = 1. For others, w < 1.
        w = np.exp(beta * c_top)
        w /= np.sum(w)
        
        # Gradient
        g = np.zeros(n)
        for wt, ki in zip(w, idx):
            d = ki - (n - 1)
            g += wt * grad_offset(h, d, n)
        
        # Project to sum-preserving subspace
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        
        # Line search
        improved = False
        for a in [alpha, alpha*0.5, alpha*0.25, alpha*0.1]:
            h_new = project_feasible(h - a * g, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                h = h_new
                alpha = min(a * 1.3, 0.3)
                improved = True
                break
        
        if not improved:
            alpha *= 0.5
            if alpha < 1e-7:
                break
    
    return best_h, best_c5

def slsqp_smooth(h, n, beta=5000, maxiter=500):
    """SLSQP with smooth max approximation."""
    dx_v = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50, 50)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_v
    
    def obj_grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50, 50)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for ki in range(2 * n - 1):
            wk = w[ki] * dx_v
            if wk < 1e-15:
                continue
            d = ki - (n - 1)
            g += wk * grad_offset(x, d, n)
        return g
    
    def con_eq(x):
        return np.sum(x) - target
    
    def con_eq_jac(x):
        return np.ones(n)
    
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def pairwise_local_search(h, n, max_iters=5000, rng=None):
    """Targeted pairwise swap optimization."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx_v = 2.0 / n
    n_ones = np.where(h > 0.5)[0]
    n_zeros = np.where(h < 0.5)[0]
    
    for it in range(max_iters):
        if remaining() < 1:
            break
        if len(n_ones) == 0 or len(n_zeros) == 0:
            break
        
        # Pick a random pair
        i = n_ones[rng.randint(len(n_ones))]
        j = n_zeros[rng.randint(len(n_zeros))]
        
        # Swap: h[i] -= eps, h[j] += eps
        h[i] -= 1.0 / n
        h[j] += 1.0 / n
        h = np.clip(h, 0, 1)
        
        c5_new = compute_c5(h)
        if c5_new > best_c5 + 1e-13:
            # Revert
            h[i] += 1.0 / n
            h[j] -= 1.0 / n
            h = np.clip(h, 0, 1)
        else:
            best_c5 = c5_new
            best_h = h.copy()
            # Update masks
            if h[i] < 0.5:
                n_ones = np.delete(n_ones, np.where(n_ones == i)[0][0])
                n_zeros = np.insert(n_zeros, np.searchsorted(n_zeros, i), i)
            if h[j] >= 0.5:
                n_zeros = np.delete(n_zeros, np.where(n_zeros == j)[0][0])
                n_ones = np.insert(n_ones, np.searchsorted(n_ones, j), j)
    
    return best_h, best_c5
