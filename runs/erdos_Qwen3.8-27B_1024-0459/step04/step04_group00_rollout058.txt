def lbfgs_smooth(h, n, beta, maxiter=2000):
    h = project_sum(h, n)

    def obj(x):
        o, _ = smax_values(x, n, beta)
        return o

    def grad(x):
        return smax_gradient(x, n, beta)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 40})
        x_opt = project_sum(res.x, n)
        return x_opt

def projected_gradient(h, n, beta, lr=0.001, maxiter=20000):

def active_set_balance(h, n, top_k=5, max_iters=800):

    best_c5 = compute_c5(h)
    rng_local = np.random.RandomState(12345)

    for rnd in range(max_rounds):
        if remaining() < 3:
            break
        improved_this_round = False
        # Sample a random subset
        if rnd % 3 == 0:
            order = np.random.permutation(n)
        else:
            order = rng_local.permutation(n)

        for j in order[:min(n, 200)]:
            old = h[j]
            if old < 0.01 or old > 0.99:
                # Try pushing to boundary
                deltas = [0.05, -0.05, 0.1, -0.1]
            else:
                deltas = [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]
            for dv in deltas:
                h[j] = old + dv
                if h[j] < 0.0 or h[j] > 1.0:
                    h[j] = old
                    continue
                cn = compute_c5(h)
                if cn < best_c5 - 1e-14:
                    best_c5 = cn
                    improved_this_round = True
                    break
                h[j] = old
        if not improved_this_round and rnd > 100:
            break

    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def perturb_reopt(h, n, rng, n_perturb=8, beta=10000):
    best = h.copy()
    best_c5 = compute_c5(h)
    for i in range(n_perturb):
        if remaining() < 15:
            break
        scale = 0.005 + 0.01 * (i % 3)
        h_p = h + rng.randn(n) * scale
        h_p = project_sum(h_p, n)
        h_opt = lbfgs_smooth(h_p, n, beta, maxiter=800)
        h_opt, c5_opt = active_set_balance(h_opt, n, top_k=5, max_iters=200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best = h_opt.copy()
    return best, best_c5


def binary_sa(h_bin, n, time_limit=60, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            oi = np.where(h == 1)[0]
            rem = rng.choice(oi, n_ones - n_half, replace=False)
            h[rem] = 0
        else:
            zi = np.where(h == 0)[0]
            add = rng.choice(zi, n_half - n_ones, replace=False)
            h[add] = 1
    dx_v = 2.0 / n
    corr = compute_corr(h)
    cur_c5 = float(np.max(corr)) * dx_v
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    T = 0.5
    for gen in range(50000):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break
        T *= 0.9997
        if T < 0.005:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0
        h[j] = 1
        nc = compute_corr(h)
        nc5 = float(np.max(nc)) * dx_v
        delta = nc5 - cur_c5
        if delta <= 0 or rng.rand() < np.exp(-delta / T):
            cur_c5 = nc5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 384
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_h = project_sum(best_h, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: L-BFGS-B at parent resolution, increasing beta
    for beta in [50, 100, 300, 1000, 3000, 8000, 20000, 50000]:
        if remaining() < 15:
            break
        h_new = lbfgs_smooth(h, n, beta, maxiter=3000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Multi-top balance
    if remaining() > 10:
        h_mb, c5_mb = multi_top_balance(h, n, top_k=8, max_iters=500)
        if c5_mb < best_c5:
            best_c5 = c5_mb
            best_h = h_mb.copy()
            h = h_mb.copy()

    if remaining() > 10:
        h_eq, c5_eq = active_set_balance(h, n, top_k=5, max_iters=800)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 3: Try nearby and larger resolutions
    n_candidates = []
    for delta in [-2, -4, -6, -8, 2, 4, 6, 8, 12, 16, 24, 32, 48, 64, 96, 128, 192]:
        nc = best_n + delta
        if 128 <= nc <= 1200 and nc % 2 == 0:
            n_candidates.append(nc)
    n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - best_n))

    for n_t in n_candidates[:10]:
        if remaining() < 25:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_sum(h_interp, n_t)

        h_opt = h_feas
        for beta in [500, 2000, 8000, 30000]:
            if remaining() < 10:
                break
            h_opt = lbfgs_smooth(h_opt, n_t, beta, maxiter=1500)

        h_opt, c5_opt = multi_top_balance(h_opt, n_t, top_k=6, max_iters=400)
        h_opt, c5_opt = active_set_balance(h_opt, n_t, top_k=5, max_iters=300)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 4: Coordinate refinement on best
    n = best_n
    h = best_h.copy()
    if remaining() > 20:
        h_cd, c5_cd = coord_refine(h, n, max_rounds=8000)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 5: Perturbation + re-optimize
    if remaining() > 40:
        h_p, c5_p = perturb_reopt(best_h, best_n, rng, n_perturb=10, beta=15000)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Phase 6: Binary SA at best resolution
    if remaining() > 40:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project_sum(h_bin, best_n)
        sa_time = min(80, max(10, int(remaining() * 0.25)))
        h_b, c5_b = binary_sa(h_b if False else h_bin, best_n, time_limit=sa_time, rng=rng)
        # Refine the binary solution with continuous optimization
        h_b_cont = project_sum(h_b, best_n)
        for beta in [5000, 20000]:
            if remaining() < 8:
                break
            h_b_cont = lbfgs_smooth(h_b_cont, best_n, beta, maxiter=1000)
        c5_b_cont = compute_c5(h_b_cont)
        if c5_b_cont < best_c5:
            best_c5 = c5_b_cont
            best_h = h_b_cont.copy()

    # Phase 7: Final intensive refinement
    n = best_n
    h = best_h.copy()
    for beta in [30000, 80000, 150000, 300000]:
        if remaining() < 10:
            break
        h_new = lbfgs_smooth(h, n, beta, maxiter=4000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 8: Final active set + multi-top
    if remaining() > 10:
        h_eq, c5_eq = active_set_balance(best_h, best_n, top_k=4, max_iters=1000)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    if remaining() > 10:
        h_mb, c5_mb = multi_top_balance(best_h, best_n, top_k=10, max_iters=800)
        if c5_mb < best_c5:
            best_c5 = c5_mb
            best_h = h_mb.copy()

    # Phase 9: Final coordinate refinement
    if remaining() > 15:
        h_cd, c5_cd = coord_refine(best_h, best_n, max_rounds=15000)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Ensure feasibility
    best_h = project_sum(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
