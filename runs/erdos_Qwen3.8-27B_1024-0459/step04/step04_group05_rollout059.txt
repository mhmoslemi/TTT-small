def get_gradient_for_lag(h, lag_offset, n):
    """Gradient of correlation at given lag with respect to h"""
    n_corr = 2 * n - 1
    # corr_full[k] = sum_i h[i] * (1 - h[i + k - (n-1)])
    # Let s = k - (n-1) be the shift

        # corr = sum h[i]*(1-h[i])
        # d/dh[j] = (1-h[j]) - h[j] = 1 - 2h[j]
        grad[:] = 1.0 - 2.0 * h
    elif s > 0:

        grad[:n_pairs] += (1.0 - h[s:n])
        grad[s:] -= h[:n_pairs]
    else:
        # s < 0, let s' = -s
        s2 = -s
        # corr = sum_{i=s2}^{n-1} h[i]*(1-h[i-s2])
        # = sum_{j=0}^{n-s2-1} h[j+s2]*(1-h[j])

        grad[:n_pairs] -= h[s2:n]
        grad[s2:] += (1.0 - h[:n_pairs])
    return grad

        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)

    return h

    """Gradient of corr[k_idx] with respect to h.
    corr[k] = sum_i h[i] * (1 - h[i + k - (n-1)])

    return max(0.0, _bt - (time.time() - _gs) - 3.0)

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full'))) * 2.0/n

def _proj(h, n):
    h = np.clip(np.array(h, dtype=float).copy(), 0, 1)
    t = n/2.0
    for _ in range(80):
        d = t - h.sum()
        if abs(d) < 1e-14: break
        m = (h > 1e-10) & (h < 1-1e-10)
        c = m.sum()
        if c: h[m] += d/c
        else: h += d/n
        h = np.clip(h, 0, 1)
    d = t - h.sum()
    if abs(d) > 1e-12:
        i = int(np.argmin(h)) if d > 0 else int(np.argmax(h))
        h[i] = float(np.clip(h[i]+d, 0, 1))
    return h

def _interp(src, sn, dn):
    if sn == dn: return src.astype(float)
    xp = (np.arange(sn)+0.5)*2.0/sn
    xn = (np.arange(dn)+0.5)*2.0/dn
    return np.interp(xn, xp, src)

def _cg(h, k, n):
    """gradient of corr[k] wrt h"""
    s = k-(n-1)
    g = np.zeros(n)
    if s == 0: g[:] = 1-2*h
    elif s > 0:
        m = n-s
        if m > 0:
            g[:m] += 1-h[s:s+m]
            g[s:] -= h[:m]
    else:
        m = n+s
        if m > 0:
            g[:m] -= h[-s:-s+m]
            g[-s:] += 1-h[:m]
    return g

def _lp_solve(h, n, active):
    """Solve outer approximation LP"""
    n_act = len(active)
    c = np.zeros(n+1); c[-1] = 1
    A = np.zeros((n_act, n+1))
    b = np.zeros(n_act)
    corr = np.correlate(h, 1-h, mode='full')
    for i, k in enumerate(active):
        g = _cg(h, k, n)
        A[i,:n] = g
        A[i,-1] = -1
        b[i] = g@h - corr[k]
    Ae = np.zeros((1,n+1)); Ae[0,:n] = 1
    be = [n/2.0]
    bd = [(0,1)]*n + [(None,None)]
    try:
        r = linprog(c, A_ub=A, b_ub=b, A_eq=Ae, b_eq=be, bounds=bd, method='highs')
        if r.status == 0:
            return _proj(r.x[:n], n), r.x[-1]
    except: pass
    return None, None
