def compute_circular_corr(h):
    """Compute circular correlation of h with (1-h) using FFT"""
    n = len(h)
    # circular_corr[k] = sum_i h[i] * (1-h[(i+k) % n])
    # = sum_i h[i] - sum_i h[i] * h[(i+k) % n]

            h[mask] += diff / cnt
        else:
            h[:] += diff / n

    diff = target - np.sum(h)

import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 100.0
_TGT = 0.0
_DX = 1.0


def time_remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def compute_exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_circular_corr(h):
    n = len(h)
    H = np.fft.fft(h)
    autocorr = np.real(np.fft.ifft(H * np.conj(H)))
    return n / 2.0 - autocorr


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
            for i in idx:
                amt = min(diff, 1.0 - h[i])
                if amt <= 1e-14:
                    break
                h[i] += amt
                diff -= amt
        else:
            idx = np.argsort(h)
            for i in idx:
                amt = min(-diff, h[i])
                if amt <= 1e-14:
                    break
                h[i] -= amt
                diff += amt
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def obj_lse(circ_mode):
    n = _N
    dx = _DX
    beta = _BETA
    x = _X_BUF
    if circ_mode:
        corr = compute_circular_corr(x)
    else:
        corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    s = np.sum(exps)
    if s < 1e-300:
        s = 1e-300
    return (m + np.log(s) / beta) * dx

_CIRC_MODE = False

    global _X_BUF
    _X_BUF = x
    return obj_lse(_CIRC_MODE)


def con_sum_fn(x):
    return np.sum(x) - _TGT


def con_sum_jac_fn(x):
    return np.ones(_N)


def do_slsqp(h0, n, beta_list, maxiter_per_beta, use_fft):
    global _N, _BETA, _TGT, _DX, _CIRC_MODE, _X_BUF
    _N = n
    _TGT = n / 2.0
    _DX = 2.0 / n
    _CIRC_MODE = use_fft
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum_fn, 'jac': con_sum_jac_fn}]

    h_cur = h0.copy()
    c5_cur = compute_exact_c5(h_cur)

    for beta in beta_list:
        if time_remaining() < 5.0:
            break
        _BETA = float(beta)
        maxit = maxiter_per_beta
        try:
            res = minimize(_obj_wrapper, h_cur, method='SLSQP',
                          bounds=bounds, constraints=cons,

            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                h_cur = h_new
        except Exception:

def do_subgrad(h0, n, max_iters=200, top_k=3, rng=None):
    h = h0.copy()
    c5_best = compute_exact_c5(h)
    h_best = h.copy()
    dx = 2.0 / n
    no_improve = 0

    for it in range(max_iters):
        if time_remaining() < 3.0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]

        grad = np.zeros(n)
        for k_idx in top_idx:
            offset = int(k_idx) - (n - 1)
            if offset >= 0:
                np2 = n - offset
                if np2 <= 0:
                    continue
                grad[:np2] += (1.0 - h[offset:offset + np2])
                grad[offset:] -= h[:np2]
            else:
                abs_off = -offset
                np2 = n - abs_off
                if np2 <= 0:
                    continue
                grad[:np2] -= h[abs_off:abs_off + np2]
                grad[abs_off:] += (1.0 - h[:np2])
        grad *= dx

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        grad /= gnorm

        improved = False
        steps = [0.002, 0.004, 0.007, 0.01, 0.015, 0.025, 0.04, 0.07]
        for step in steps:
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_exact_c5(h_trial)
            if c5_trial < c5_best - 1e-14:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                no_improve = 0
                break
        if not improved:
            no_improve += 1
            if no_improve > 20 and rng is not None:
                h_noise = project_feasible(h + rng.randn(n) * 0.025, n)
                c5_noise = compute_exact_c5(h_noise)
                if c5_noise < c5_best:
                    c5_best = c5_noise
                    h_best = h_noise.copy()
                    h = h_noise.copy()
                    no_improve = 0
                elif no_improve > 40:
                    break
    return h_best, c5_best
