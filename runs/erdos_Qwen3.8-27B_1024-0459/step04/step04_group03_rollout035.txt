def subgradient_step(h, n, alpha):
    """One subgradient step for the max correlation objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    
    # Find active set (correlations close to max)
    tol = 1e-6
    active = np.where(corr >= max_val - tol)[0]
    
    # Compute sub-gradient
    grad = np.zeros(n)
    l_arr = np.arange(2*n-1) - (n-1)
    
    for m in active:
        l = l_arr[m]
        # d/dh[j] c[m] = I(0<=j-l<n) - h[j+l]*I(0<=j+l<n) - h[j-l]*I(0<=j-l<n)
        j_start = max(0, l)
        j_end = min(n, n + l)
        if j_end > j_start:
            grad[j_start:j_end] += 1.0
            grad[j_start:j_end] -= h[j_start-l:j_end-l]
        
        j_start2 = max(0, -l)
        j_end2 = min(n, n - l)
        if j_end2 > j_start2:
            grad[j_start2:j_end2] -= h[j_start2+l:j_end2+l]
    
    grad /= len(active)
    grad *= dx
    
    # Update
    h_new = h - alpha * grad
    # Project
    h_new = np.clip(h_new, 0.0, 1.0)
    # Adjust sum
    target = n / 2.0
    diff = target - np.sum(h_new)
    mask = (h_new > 1e-10) & (h_new < 1.0 - 1e-10)
    if np.sum(mask) > 0:
        h_new[mask] += diff / np.sum(mask)
        h_new = np.clip(h_new, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h_new)
    if abs(diff) > 1e-10:
        idx = np.argmin(np.abs(h_new - 0.5))
        h_new[idx] = np.clip(h_new[idx] + diff, 0.0, 1.0)
    
    return h_new

def projected_subgradient(h, n, max_iter=1000, alpha_init=0.01, alpha_decay=0.999):
    """Projected subgradient method."""
    h = h.copy()
    alpha = alpha_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    
    for it in range(max_iter):
        if remaining_time() < 2:
            break
        
        h = subgradient_step(h, n, alpha)
        c5 = compute_c5(h)
        
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        alpha *= alpha_decay
        
        if no_improve > 200:
            break
    
    return best_h, best_c5

def coordinate_descent(h, n, max_sweeps=50, tol=1e-8):
    """Coordinate descent for the min-max overlap problem."""
    h = h.copy()
    dx = 2.0 / n
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    
    l_arr = np.arange(2*n-1) - (n-1)  # l_k for each k
    
    for sweep in range(max_sweeps):
        if remaining_time() < 2:
            break
        
        improved = False
        for i in range(n):
            if remaining_time() < 1:
                    break
            
            # Compute A_k and B_k for coordinate i
            A = np.zeros(2*n-1)
            for m in range(2*n-1):
                l = l_arr[m]
                a = 0.0
                if 0 <= i - l < n:
                    a += (1.0 - h[i-l])
                if 0 <= i + l < n:
                    a -= h[i+l]
                A[m] = a
            
            # B_k = c_k - h[i]*A_k
            corr = np.correlate(h, 1.0 - h, mode='full')
            B = corr - h[i] * A
            
            # Minimize max_k (A_k * t + B_k) over t in [0,1]
            # Using ternary search
            def f(t):
                return np.max(A * t + B)
            
            # Ternary search
            lo, hi = 0.0, 1.0
            for _ in range(50):  # ~50 iterations gives precision ~ 1/3^50 ≈ 1e-24
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if f(m1) < f(m2):
                    hi = m2
                else:
                    lo = m1
            t_opt = (lo + hi) / 2.0
            
            # Check if update improves
            h[i] = t_opt
            # Project (fix sum)
            h = np.clip(h, 0.0, 1.0)
            diff = target_sum - np.sum(h)
            if abs(diff) > 1e-12:
                # Adjust: find the coordinate with most "room"
                mask = (h > 1e-10) & (h < 1.0 - 1e-10)
                if np.sum(mask) > 0:
                    h[mask] += diff / np.sum(mask)
                    h = np.clip(h, 0.0, 1.0)
            
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                improved = True
        
        if not improved:
            break
    
    return h, best_c5

But actually, I can vectorize the gradient computation! Instead of looping over m, I can compute it for all m at once.

Let me think... For each j, the gradient is:
grad[j] = dx * Σ_m w_m * [(1-h[j-l_m])*I(0<=j-l_m<n) - h[j+l_m]*I(0<=j+l_m<n)]

For the first term: (1-h[j-l_m]) * I(0<=j-l_m<n)
This is non-zero when 0<=j-l_m<n, i.e., l_m <= j < l_m+n.
For a fixed j, the valid m's are those with l_m in [j-n+1, j], i.e., m-(n-1) in [j-n+1, j], i.e., m in [1, j+n-1].

For the second term: -h[j+l_m] * I(0<=j+l_m<n)
This is non-zero when 0<=j+l_m<n, i.e., -j <= l_m < n-j.
For a fixed j, the valid m's are those with l_m in [-j, n-j-1], i.e., m in [n-1-j, n-1+n-j-1] = [n-1-j, 2n-j-2].

This is still complex to vectorize directly. Let me just use the loop approach since n is small enough.

import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_N_G = 0
_BETA_G = 1.0
_L_ARR_G = None

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        order = np.argsort(np.abs(h - 0.5))
        for idx in order[:20]:
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
            diff = target - np.sum(h)
            if abs(diff) < 1e-14:
                break
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def _sm_obj(x):
    n = _N_G
    beta = _BETA_G
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    shifted = beta * (corr - max_corr)
    if np.max(shifted) > 500:
        return max_corr * dx
    return (max_corr + np.log(np.sum(np.exp(shifted))) / beta) * dx

def _sm_grad(x):
    n = _N_G
    beta = _BETA_G
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    shifted = beta * (corr - max_corr)
    if np.max(shifted) > 500:
        weights = np.zeros_like(corr)
        max_idx = np.where(corr >= max_corr - 1e-10)[0]
        weights[max_idx] = 1.0 / len(max_idx)
    else:
        exp_s = np.exp(shifted)
        weights = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    l_arr = _L_ARR_G
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        l = l_arr[m]
        j_lo = max(0, l)
        j_hi = min(n, n + l)
        if j_hi > j_lo:
            grad[j_lo:j_hi] += w * (1.0 - h[j_lo - l:j_hi - l])
        j_lo2 = max(0, -l)
        j_hi2 = min(n, n - l)
        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] -= w * h[j_lo2 + l:j_hi2 + l]
    return grad * dx

def _sm_eq(x):
    return np.sum(x) - _N_G / 2.0

def slsqp_opt(h, n, beta, maxiter=500):
    global _N_G, _BETA_G, _L_ARR_G
    _N_G = n
    _BETA_G = beta
    _L_ARR_G = np.arange(2 * n - 1) - (n - 1)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sm_eq, 'jac': lambda x: np.ones(n) / n}]
    try:
        res = minimize(_sm_obj, h, method='SLSQP', jac=_sm_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and res.fun < float('inf'):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def coordinate_descent(h, n, max_sweeps=30):
    """Coordinate descent exploiting piecewise-linear structure of max of linear functions."""
    h = h.copy()
    dx = 2.0 / n
    target_sum = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    l_arr = np.arange(2 * n - 1) - (n - 1)

    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        sweep_improved = False
        for i in range(n):
            if remaining_time() < 1:
                break
            old_val = h[i]
            # Compute A[m] = dc[m]/dh[i]
            A = np.zeros(2 * n - 1)
            for m in range(2 * n - 1):
                l = l_arr[m]
                a = 0.0
                if 0 <= i - l < n:
                    a += (1.0 - h[i - l])
                if 0 <= i + l < n:
                    a -= h[i + l]
                A[m] = a
            B = corr - old_val * A
            # Minimize max_m(A[m]*t + B[m]) on [0,1]
            f0 = np.max(B)
            f1 = np.max(A + B)
            # Find crossing points of top lines
            top_lines = np.argsort(np.maximum(A * 0 + B, A * 1 + B))[-20:]
            candidates = [0.0, 1.0, old_val]
            for a1 in range(len(top_lines)):
                for a2 in range(a1 + 1, len(top_lines)):
                    m1, m2 = top_lines[a1], top_lines[a2]
                    if abs(A[m1] - A[m2]) > 1e-12:
                        tc = (B[m1] - B[m2]) / (A[m2] - A[m1])
                        if 0.0 <= tc <= 1.0:
                            candidates.append(tc)
            best_t = old_val
            best_f = np.max(A * old_val + B)
            for t in candidates:
                t = float(np.clip(t, 0.0, 1.0))
                ft = np.max(A * t + B)
                if ft < best_f - 1e-13:
                    best_f = ft
                    best_t = t
            # Refine with ternary search around best_t
            if best_t != old_val:
                lo = max(0.0, best_t - 0.1)
                hi = min(1.0, best_t + 0.1)
                for _ in range(40):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    if np.max(A * m1 + B) < np.max(A * m2 + B):
                        hi = m2
                    else:
                        lo = m1
                best_t = (lo + hi) / 2.0
                best_f = np.max(A * best_t + B)
            h[i] = best_t
            corr += (best_t - old_val) * A
            # Fix sum
            h = np.clip(h, 0.0, 1.0)
            diff = target_sum - np.sum(h)
            if abs(diff) > 1e-12:
                mask = (h > 1e-10) & (h < 1.0 - 1e-10)
                cnt = np.sum(mask)
                if cnt > 0:
                    adj = diff / cnt
                    h[mask] += adj
                    h = np.clip(h, 0.0, 1.0)
                    corr = np.correlate(h, 1.0 - h, mode='full')
            new_c5 = np.max(corr) * dx
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                sweep_improved = True
        if not sweep_improved:
            break
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def subgradient_polish(h, n, iters=500, alpha_init=0.005):
    """Projected subgradient method for exact max."""
    h = h.copy()
    dx = 2.0 / n
    target_sum = n / 2.0
    l_arr = np.arange(2 * n - 1) - (n - 1)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha = alpha_init
    stall = 0

    for it in range(iters):
        if remaining_time() < 2:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        tol = 1e-4
        active = np.where(corr >= max_val - tol)[0]
        grad = np.zeros(n)
        for m in active:
            l = l_arr[m]
            j_lo = max(0, l)
            j_hi = min(n, n + l)
            if j_hi > j_lo:
                grad[j_lo:j_hi] += (1.0 - h[j_lo - l:j_hi - l])
            j_lo2 = max(0, -l)
            j_hi2 = min(n, n - l)
            if j_hi2 > j_lo2:
                grad[j_lo2:j_hi2] -= h[j_lo2 + l:j_hi2 + l]
        grad /= len(active)
        grad *= dx
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target_sum - np.sum(h_new)
        mask = (h_new > 1e-10) & (h_new < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h_new[mask] += diff / cnt
            h_new = np.clip(h_new, 0.0, 1.0)
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        alpha *= 0.998
        if stall > 150:
            break
    return best_h, best_c5

def binary_search(h, n, rng, trials=10):
    """Try binary perturbations and re-optimize."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(trials):
        if remaining_time() < 10:
            break
        h_b = h.copy()
        # Randomly flip some values
        n_flip = rng.randint(2, max(3, n // 16))
        idx = rng.choice(n, size=n_flip, replace=False)
        for i in idx:
            h_b[i] = 1.0 - h_b[i]
        h_b = project_feasible(h_b, n)
        c5_b = compute_c5(h_b)
        if c5_b < best_c5 + 0.01:
            h_b = slsqp_opt(h_b, n, 500, maxiter=200)

            h_b2, c5_b3 = coordinate_descent(h_b, n, max_sweeps=5)
            if c5_b3 < best_c5:
                best_c5 = c5_b3
                best_h = h_b2.copy()
    return best_h, best_c5
