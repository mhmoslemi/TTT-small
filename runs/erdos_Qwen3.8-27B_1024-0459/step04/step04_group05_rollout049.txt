Let me also add a novel idea: **Lagrange multiplier method for the max constraint**.

We want: minimize max_k f_k(h) subject to g(h) = sum(h) - n/2 = 0, 0 ≤ h ≤ 1.

KKT conditions: ∇f_{k*}(h) + λ∇g(h) + μ(h) = 0 at the optimum, where k* is an active constraint.

If only one lag is active, this is a standard constrained optimization. If multiple are active, it's a max of linear functions (well, of quadratic functions in h).

Actually, I realize that f_k(h) = sum_i h[i](1-h[i+k]) is QUADRATIC in h (not linear). So the problem is non-convex in general. The log-sum-exp trick makes it smooth but doesn't fix non-convexity.

However, if we fix the problem as: minimize max_k [sum_i h[i] - h[i]*h[i+k]]
= n/2 - min_k [sum_i h[i]*h[i+k]] (since sum_i h[i] = n/2 is fixed)

Wait! That's a key insight!

max_k sum_i h[i]*(1-h[i+k]) = max_k [sum_i h[i] - sum_i h[i]*h[i+k]]

But sum_i h[i] depends on the lag! For lag k, the sum is only over the overlapping indices.

Hmm, let me be more careful. For lag k (positive), the correlation is:
c_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])

This is NOT simply sum(h) - sum(h[i]*h[i+k]) because the sums are over different ranges.

For lag 0: c_0 = sum_i h[i]*(1-h[i]) = sum(h) - sum(h²) = n/2 - sum(h²)

For lag k > 0: c_k = sum_{i=0}^{n-1-k} [h[i] - h[i]*h[i+k]]

So c_k = (sum of h over first n-k elements) - (sum of h[i]*h[i+k] over i=0 to n-1-k)

The first term is NOT n/2 (it's the partial sum). So the simplification doesn't quite work.

However, for large n and small k, the partial sum ≈ (n-k)/n * n/2 = (n-k)/2, and the max over k will likely be at small k where most of the sum is included.

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_DX = 1.0
_G_TGT = 0.0
_G_BETA = 1.0
_G_START = 0.0
_G_BUDGET = 1300.0

def fft_max_corr(h):
    """Compute max of np.correlate(h, 1-h, 'full') using FFT."""
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    H = np.fft.rfft(h, N)
    B = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(B), N)
    return float(np.max(corr))

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return fft_max_corr(h) * dx

def compute_corr_fft(h):
    """Get all correlation values using FFT."""
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    H = np.fft.rfft(h, N)
    B = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(B), N)
    # Only the first 2n-1 values (and the last n-1) are relevant
    # But since extra values are ~0, we can just return the whole thing
    return corr

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        # Find the element with most room
        rooms_up = 1.0 - h
        rooms_dn = h
        if diff > 0:
            idx = int(np.argmax(rooms_up))
        else:
            idx = int(np.argmin(rooms_dn))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def subgrad_max_corr(h, n, top_k=5):
    """Compute subgradient of max_k corr_k at h, using top_k active lags."""
    corr = compute_corr_fft(h)  # length N (power of 2)
    # The relevant values: corr[0], corr[1], ..., corr[n-1], corr[N-n+1], ..., corr[N-1]
    # Actually for our purpose, just take the whole corr array (extras are ~0)
    # Find top_k indices
    top_idx = np.argsort(corr)[-top_k:]
    top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
    
    grad = np.zeros(n)
    for k_idx in top_idx:
        # k_idx is the index in the FFT output
        # We need to figure out what lag this corresponds to
        if k_idx <= n - 1:
            lag = k_idx  # positive or zero lag
        else:
            lag = k_idx - N  # negative lag (N - (n-1) to N-1 maps to -(n-1) to -1)
        
        # For lag >= 0: corr[lag] = sum_i h[i] * (1-h[i+lag]) (over valid i)
        # d/dh[j] = (1-h[j+lag]) if j+lag < n, plus h[j-lag] if j-lag >= 0
        # Wait, this is the gradient of sum_i h[i]*(1-h[i+lag])
        # = sum_i (1-h[i+lag]) * dh[i]/dh[j] - sum_i h[i] * dh[i+lag]/dh[j]
        # = (1-h[j+lag]) [if j+lag < n] - h[j-lag] [if j-lag >= 0]
        
        # Actually for lag=0: sum_i h[i]*(1-h[i]), d/dh[j] = 1-2*h[j]
        
        if lag >= 0:
            n_pairs = n - lag
            if n_pairs <= 0:
                continue
            # d/dh[j] of sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
            # = (1-h[j+lag]) for j in [0, n-1-lag] (from first factor)
            #   - h[j-lag] for j in [lag, n-1] (from second factor)
            grad[:n_pairs] += (1.0 - h[lag:lag+n_pairs])
            grad[lag:] -= h[:n_pairs]
        else:
            a = -lag
            n_pairs = n - a
            if n_pairs <= 0:
                continue
            # corr[N-a] = sum_i h[i+a] * (1-h[i]) (by symmetry of the correlation)
            # Actually let me re-derive. For negative lag -a:
            # (a⋆b)[-a] = sum_i a[i-a]*b[i] = sum_{i=a}^{n-1} a[i-a]*b[i]
            # = sum_{j=0}^{n-1-a} a[j]*b[j+a] where j=i-a
            # = sum_{j=0}^{n-1-a} h[j]*(1-h[j+a])
            # d/dh[j] = (1-h[j+a]) for j in [0, n-1-a]
            #          - h[j-a] for j in [a, n-1]
            grad[:n_pairs] += (1.0 - h[a:a+n_pairs])
            grad[a:] -= h[:n_pairs]
    
    return grad * (2.0 / n)  # multiply by dx

def obj_lse(x, n, beta):
    """Log-sum-exp smooth approximation to max correlation."""
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    H = np.fft.rfft(x, N)
    B = np.fft.rfft(1.0 - x, N)
    corr = np.fft.irfft(H * np.conj(B), N)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    return (m + np.log(np.sum(exps)) / beta) * (2.0 / n)

def do_slsqp(h_start, n, beta_list, budget_fn, label=""):
    global _G_N, _G_TGT
    _G_N = n
    _G_TGT = n / 2.0
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
        
    def obj_fn(x):
        return obj_lse(x, n, _G_BETA)
        
    def con_fun(x):
        return np.sum(x) - n / 2.0
        
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    for beta in beta_list:
        if budget_fn() < 10.0:
            break
        _G_BETA = float(beta)
        maxiter = min(300, max(50, int(budget_fn() * 0.3)))
        try:
            res = minimize(obj_fn, h_cur, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                h_cur = h_new
            if label and beta % 1000 == 0:
                print("    beta={}: C5={:.10f}".format(beta, c5_cur))
        except Exception:
            pass
    return h_cur, c5_cur

def do_subgrad(h_start, n, budget_fn, rng, max_iters=200, top_k=5):
    h = h_start.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    no_improve = 0
    
    for it in range(max_iters):
        if budget_fn() < 3.0:
            break
        grad = subgrad_max_corr(h, n, top_k=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        
        improved = False
        # Adaptive step sizes
        base_step = 0.1 / (1.0 + 0.01 * it)
        steps = [base_step * s for s in [0.1, 0.3, 1.0, 3.0, 10.0]] + [0.05, 0.02, 0.01, 0.5]
        
        for step in steps:
            h_trial = project_feasible(h - step * grad, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-14:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                no_improve = 0
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 40:
                # Add noise to escape
                noise = rng.randn(n) * 0.03
                h_trial = project_feasible(h + noise, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < c5_best:
                    c5_best = c5_trial
                    h_best = h_trial.copy()
                    h = h_trial.copy()
                no_improve = 0
                if no_improve > 80:
                    break
    
    return h_best, c5_best

                else:
