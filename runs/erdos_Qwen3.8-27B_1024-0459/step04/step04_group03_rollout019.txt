   ```python
   def swap_delta(h, p, q, n):
       """h[p]=0, h[q]=1. Swap them. Return delta for all lags."""
       d = np.zeros(n)
       for k in range(n):
           max_i = n - 1 - k
           changed = set()
           if 0 <= p <= max_i: changed.add(p)
           if 0 <= q <= max_i: changed.add(q)
           if 0 <= (p-k) and (p-k) <= max_i: changed.add(p-k)
           if 0 <= (q-k) and (q-k) <= max_i: changed.add(q-k)
           
           delta = 0
           for i in changed:
               # old term
               old_idx1 = i + k
               old = h[old_idx1] * (1 - h[i])
               # new term (apply swap)
               new_h_old_idx1 = 0 if old_idx1 == q else (1 if old_idx1 == p else h[old_idx1])
               new_h_i = 0 if i == q else (1 if i == p else h[i])
               new = new_h_old_idx1 * (1 - new_h_i)
               delta += new - old
           d[k] = delta
       return d
   ```

```python
def remaining_time():

def project_feasible(h, n):

        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            idx = int(np.argmin(np.abs(h - 0.5)))
            h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):

def binary_perturb_restart(h_bin, n, rng, n_flips=3):

def slsqp_smooth(h_init, n, beta, maxiter=300):

        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)
    def eq(x):
        return np.sum(x) - n / 2.0
    cons = [{'type': 'eq', 'fun': eq}]
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h_init.copy()
```
