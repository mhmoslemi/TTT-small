def grad_single_lag(h, offset, n):
    """Gradient of c_k for a specific lag k (offset)."""
    dx = 2.0 / n

    if offset >= 0:

            return grad * dx

        grad[offset:] -= h[:n_pairs]

            return grad * dx
        grad[:n_pairs] -= h[abs_off:abs_off + n_pairs]

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_BETA = 1.0
_G_TGT = 0.0
_G_START = 0.0
_G_BUDGET = 1300.0
_G_DX = 1.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        for _ in range(20):
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                room = 1.0 - h
            else:
                room = h
            idx = int(np.argmax(room))
            if diff > 0:
                adj = min(diff, room[idx])
            else:
                adj = max(diff, -h[idx])
            if abs(adj) < 1e-15:
                break
            h[idx] += adj
            diff -= adj
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h


def grad_single_lag(h, offset, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    if offset >= 0:
        n_pairs = n - offset
        if n_pairs <= 0:
            return grad
        grad[:n_pairs] += (1.0 - h[offset:offset + n_pairs])
        grad[offset:] -= h[:n_pairs]
    else:
        abs_off = -offset
        n_pairs = n - abs_off
        if n_pairs <= 0:
            return grad
        grad[:n_pairs] -= h[abs_off:abs_off + n_pairs]
        grad[abs_off:] += (1.0 - h[:n_pairs])
    return grad * dx


def con_sum_fun(x):
    return np.sum(x) - _G_TGT


def con_sum_jac(x):
    return np.ones(len(x))


def obj_lse_fun(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(_G_BETA * shifted, -500, 500))
    return (m + np.log(np.sum(exps)) / _G_BETA) * _G_DX


def proj_grad_exact(h, n, max_iters, budget_threshold):
    h = h.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    no_improve = 0
    for it in range(max_iters):
        if remaining_time() < budget_threshold:
            break
        corr = compute_corr(h)
        c5_cur = float(np.max(corr)) * (2.0 / n)
        if c5_cur < c5_best - 1e-14:
            c5_best = c5_cur
            h_best = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 60:
            break
        max_val = np.max(corr)
        active_mask = corr >= max_val - max(0.01, max_val * 1e-5)
        active_idx = np.where(active_mask)[0]
        if len(active_idx) > 3:
            top3 = np.argsort(corr)[-3:]
            active_idx = top3
        grad = np.zeros(n)
        for idx in active_idx:
            offset = int(idx) - (n - 1)
            grad += grad_single_lag(h, offset, n)
        grad /= len(active_idx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        grad /= gnorm
        improved = False
        steps = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.003, 0.0005, 0.3, 0.5]
        for step in steps:
            h_trial = project_feasible(h - step * grad, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-14:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break

                    c5_best = c5_trial
                    h_best = h_trial.copy()
                    h = h_trial.copy()
                    improved = True
                    no_improve = 0
                    break
            if not improved:
                no_improve += 5
    return h_best, c5_best


def do_slsqp(h_start, n, beta_list, maxiter_cap):
    global _G_N, _G_BETA, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': con_sum_fun, 'jac': con_sum_jac}]
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    for beta in beta_list:
        if remaining_time() < 8.0:
            break
        _G_BETA = float(beta)
        maxiter = min(maxiter_cap, max(30, int(remaining_time() * 0.25 / 0.5)))
        try:

                           bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                h_cur = h_new
        except Exception:

    return h_cur, c5_cur

                    h = h_p.copy()
                    any_imp = True
                else:
                    h[i] = old_val

            break

            h[i] = m1
            h1 = project_feasible(h.copy(), n)
            c1 = compute_c5(h1)
            h[i] = m2
            h2 = project_feasible(h.copy(), n)
            c2 = compute_c5(h2)
            if c1 <= c2:
                b = m2
            else:
                a = m1
        h[i] = old_val
        mid = (a + b) / 2.0
        h[i] = mid
        h_mid = project_feasible(h, n)
        c_mid = compute_c5(h_mid)
        if c_mid < c5_best - 1e-14:
            c5_best = c_mid
            h_best = h_mid.copy()
