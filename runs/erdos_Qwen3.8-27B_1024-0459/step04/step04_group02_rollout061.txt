def smooth_grad(corr, h, n, beta):
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)  # avoid overflow
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    dx = 2.0 / n
    grad = np.zeros(n)
    for off in range(-(n-1), n):
        k = n - 1 + off
        wk = w[k] * dx
        if abs(wk) < 1e-15:
            continue
        if off == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif off > 0:
            grad[:n-off] += wk * (1.0 - h[off:])
            grad[off:] -= wk * h[:n-off]
        else:
            aoff = -off
            grad[aoff:] += wk * (1.0 - h[:n-aoff])
            grad[:n-aoff] -= wk * h[aoff:]
    return grad

    # Phase 4: Nesterov at best resolution
    n_cur = best_n
    h_cur = best_h.copy()
    if remaining_time() > 60:
        for beta_ns in [200, 1000, 5000]:
            if remaining_time() < 30:
                break
            h_ns, c5_ns = nesterov_smooth(h_cur, n_cur, beta_ns, 3000)
            if c5_ns < best_c5:
                best_c5 = c5_ns
                best_h = h_ns.copy()
                best_n = n_cur
                h_cur = h_ns.copy()
            if remaining_time() < 15:
                break

    # Phase 5: More coordinate descent on best
    if remaining_time() > 20:
        h_cd, c5_cd = coord_descent(best_h, best_n, 50)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 6: Subgradient + SLSQP interleaved
    h = best_h.copy()
    n = best_n
    for it in range(500):
        if remaining_time() < 10:
            break
        grad = subgrad_step(h, n, 6)
        if grad is None:
            break
        improved = False
        for sm in [0.5, 1.0, 0.25, 2.0, 0.1]:
            step = 0.03 * sm / (1.0 + it * 0.005)
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-15:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
        if not improved:
            if it % 100 == 50 and remaining_time() > 30:
                h_new = slsqp_refine(h, n, 5000, 300)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                if it > 200:
                    break

    # Phase 7: Try different resolution (n*1.5 or 3n/2)
    n_alt = int(best_n * 1.5)
    if n_alt % 2 == 0 and 128 <= n_alt <= 2560 and n_alt != best_n and remaining_time() > 80:
        h_alt = interp_to(best_h, best_n, n_alt)
        h_alt = project_feasible(h_alt, n_alt)
        c5_alt = compute_c5(h_alt)
        for beta in [500, 3000]:
            if remaining_time() < 20:
                break
            h_new = slsqp_refine(h_alt, n_alt, beta, 400)
            c5_new = compute_c5(h_new)
            if c5_new < c5_alt:
                h_alt = h_new
                c5_alt = c5_new
            if c5_alt < best_c5:
                best_c5 = c5_alt
                best_h = h_alt.copy()
                best_n = n_alt
        if remaining_time() > 20:
            h_cd, c5_cd = coord_descent(h_alt, n_alt, 20)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n_alt

    # Phase 8: Random restarts at moderate resolution
    if remaining_time() > 60:
        for restart in range(3):
            if remaining_time() < 30:
                break
            n_r = max(320, best_n)
            if n_r % 2 == 0 and n_r <= 1600:
                h_rand = np.zeros(n_r)
                h_rand[rng.choice(n_r, n_r // 2, replace=False)] = 1.0
                h_rand = project_feasible(h_rand, n_r)
                c5_rand = compute_c5(h_rand)
                if c5_rand < 0.45:
                    for beta in [500, 2000]:
                        if remaining_time() < 15:
                            break
                        h_new = slsqp_refine(h_rand, n_r, beta, 400)
                        c5_new = compute_c5(h_new)
                        if c5_new < c5_rand:
                            h_rand = h_new
                            c5_rand = c5_new
                        if c5_rand < best_c5:
                            best_c5 = c5_rand
                            best_h = h_rand.copy()
                            best_n = n_r

    # Phase 9: Final high-precision SLSQP
    if remaining_time() > 15:
        h = best_h.copy()
        n = best_n
        for beta in [10000, 50000]:
            if remaining_time() < 8:
                break
            h_new = slsqp_refine(h, n, beta, 500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 10: Final coordinate descent polish
    if remaining_time() > 10:
        h_cd, c5_cd = coord_descent(best_h, best_n, 40)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 11: Micro subgradient
    if remaining_time() > 5:
        h = best_h.copy()
        n = best_n
        for it in range(300):
            if remaining_time() < 2:
                break
            grad = subgrad_step(h, n, 4)
            if grad is None:
                break
            improved = False
            for step in [0.001, 0.003, 0.0005, 0.005, 0.002]:
                h_trial = h - step * grad
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-16:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h = h_trial.copy()
                    improved = True
                    break
            if not improved:
                break

    # Phase 12: One more coord descent
    if remaining_time() > 5:
        h_cd, c5_cd = coord_descent(best_h, best_n, 20)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
