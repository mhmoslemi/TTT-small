            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0; h[j] = 1.0

        new_corr = compute_corr(h)
        new_cost = float(np.max(new_corr))
        delta = new_cost - cur_cost

        if delta <= 1e-14 or (T > 1e-4 and rng.rand() < np.exp(-delta / T)):
            cur_cost = new_cost
            if new_cost < best_cost - 1e-14:
                best_cost = new_cost
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            if use_double:
                h[i1] = 1.0; h[i2] = 1.0
                h[j1] = 0.0; h[j2] = 0.0
            else:
                h[i] = 1.0; h[j] = 0.0
            stall += 1

        T *= alpha
        if stall > max_stall:
            T = max(T, T0 * 0.4)
            stall = 0
            # Perturb: 3 random swaps
            for _ in range(3):
                ones = np.nonzero(h)[0]
                zeros = np.nonzero(1.0 - h)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    oi = ones[rng.randint(len(ones))]
                    oz = zeros[rng.randint(len(zeros))]
                    h[oi] = 0.0; h[oz] = 1.0
            corr = compute_corr(h)
            cur_cost = float(np.max(corr))

    return best_h, best_cost * dx


def sa_ilS_restart(h, n, rng, time_limit=200, n_restarts=3):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    tper = time_limit / n_restarts

    for r in range(n_restarts):
        if time.time() - start > time_limit:
            break
        if remaining() < 20:
            break
        h_cur = best_h.copy()
        # Perturb for restart (except first)
        if r > 0:
            n_perturb = n // 20
            ones = np.nonzero(h_cur)[0]
            zeros = np.nonzero(1.0 - h_cur)[0]
            for _ in range(n_perturb):
                oi = ones[rng.randint(len(ones))]
                oz = zeros[rng.randint(len(zeros))]
                h_cur[oi] = 0.0; h_cur[oz] = 1.0
        h_sa, c5_sa = sa_mixed(h_cur, n, rng, T0=4.0 + r, alpha=0.9998,
                               T_min=0.003, time_limit=tper, double_prob=0.3)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    return best_h, best_c5


def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-14:
            g += w[k] * corr_grad_per_shift(x, n, k)
    return g * dx_val


def sum_con(x):
    return np.sum(x) - _TGT


def sum_con_jac(x):
    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=300):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def topk_gradient_descent(h, n, max_iters=300, top_k=5):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        m = float(np.max(corr))
        if m * (2.0 / n) >= best_c5 + 1e-8:
            break
        top_idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k in top_idx:
            g += corr_grad_per_shift(h, n, int(k))
        g /= len(top_idx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        improved = False
        for step in [0.03, 0.01, 0.005, 0.05, 0.02, 0.008, 0.015]:
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-12:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 360
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Intensive binary SA at n=360 (primary target)
    n_main = 360
    h_bin = make_binary(best_h, n_main, rng)
    c5_bin = compute_c5(h_bin)

    # Multiple SA rounds at n_main
    sa_time_each = min(150, max(80, remaining() * 0.35))
    for sa_round in range(4):
        if remaining() < 30:
            break
        h_init = h_bin.copy()
        if sa_round > 0:
            # Perturb previous best
            n_pert = max(3, n_main // 40)
            ones = np.nonzero(h_init)[0]
            zeros = np.nonzero(1.0 - h_init)[0]
            for _ in range(n_pert):
                oi = ones[rng.randint(len(ones))]
                oz = zeros[rng.randint(len(zeros))]
                h_init[oi] = 0.0; h_init[oz] = 1.0
        h_sa, c5_sa = sa_ilS_restart(h_init, n_main, rng,
                                     time_limit=sa_time_each, n_restarts=3)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n_main
            h_bin = h_sa.copy()

    # Phase 2: Try nearby n values
    n_neighbors = [356, 358, 362, 364]
    for n_t in n_neighbors:
        if remaining() < 40:
            break
        if n_t % 2 != 0:
            continue
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = make_binary(h_interp, n_t, rng)
        c5_t = compute_c5(h_bin_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_bin_t.copy()
            best_n = n_t

        # SA at this n
        h_sa, c5_sa = sa_ilS_restart(h_bin_t, n_t, rng,
                                     time_limit=min(80, remaining() * 0.2), n_restarts=2)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n_t

    # Phase 3: Continuous SLSQP refinement
    n_ref = best_n
    h_cont = best_h.copy()
    h_cont = project_feasible(h_cont, n_ref)

    for beta in [500, 2000, 8000]:
        if remaining() < 20:
            break
        h_new = slsqp_smooth(h_cont, n_ref, beta, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cont = h_new.copy()

    # Phase 4: Top-k gradient descent
    if remaining() > 20:
        h_td, c5_td = topk_gradient_descent(best_h, best_n, max_iters=200, top_k=5)
        if c5_td < best_c5:
            best_c5 = c5_td
            best_h = h_td.copy()

    # Phase 5: More SLSQP with higher beta
    for beta in [20000, 80000]:
        if remaining() < 15:
            break
        h_new = slsqp_smooth(best_h, best_n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 6: Final binary SA push on best solution
    if remaining() > 40:
        h_bin_final = make_binary(best_h, best_n, rng)
        t_rem = remaining() * 0.5
        h_sa, c5_sa = sa_mixed(h_bin_final, best_n, rng, T0=3.0,
                               alpha=0.9999, T_min=0.002,
                               time_limit=t_rem, double_prob=0.35)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = best_n

    # Phase 7: Double-swap focused SA (more aggressive)
    if remaining() > 30:
        h_bin_2 = make_binary(best_h, best_n, rng)
        t_rem = remaining() * 0.4
        h_sa, c5_sa = sa_mixed(h_bin_2, best_n, rng, T0=6.0,
                               alpha=0.9997, T_min=0.01,
                               time_limit=t_rem, double_prob=0.6)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 8: Final SLSQP polish
    if remaining() > 10:
        h_new = slsqp_smooth(best_h, best_n, 50000, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
