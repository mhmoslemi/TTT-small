import numpy as np
from scipy.optimize import fmin_l_bfgs_b
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = np.argmin(h)
            else:
                idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def smooth_obj(x, n, beta, penalty):
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_sum = np.sum(np.exp(shifted))
    smooth_max = m + np.log(exp_sum) / beta
    penalty_term = penalty * (np.sum(x) - n / 2.0) ** 2
    return smooth_max * dx + penalty_term

def smooth_grad(x, n, beta, penalty):
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    
    grad = np.zeros(n)
    # Positive lags o = 0, 1, ..., n-1
    for o in range(n):
        w = weights[n - 1 + o]
        if w < 1e-14:
            continue
        npairs = n - o
        if npairs <= 0:
            continue
        grad[:npairs] += w * (1.0 - x[o:o + npairs])
        grad[o:o + npairs] -= w * x[:npairs]
    # Negative lags o = 1, 2, ..., n-1
    for o in range(1, n):
        w = weights[n - 1 - o]
        if w < 1e-14:
            continue
        npairs = n - o
        if npairs <= 0:
            continue
        grad[:npairs] -= w * x[o:o + npairs]
        grad[o:o + npairs] += w * (1.0 - x[:npairs])
    
    grad *= dx
    # Add constraint gradient
    grad += 2.0 * penalty * (np.sum(x) - n / 2.0)
    return grad

def lbfgsb_optimize(h0, n, betas, penalty=1000.0, maxiter=300):
    h = h0.copy()
    for beta in betas:
        if remaining_time() < 5:
            break
        res = fmin_l_bfgs_b(smooth_obj, h, fprime=smooth_grad,
                            args=(n, beta, penalty),
                            pgtol=1e-12, maxiter=maxiter, iprint=0)
        h_new = project_feasible(res, n)
        if compute_c5(h_new) <= compute_c5(h) + 1e-12:
            h = h_new
        # Update h if it improved
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def coord_descent(h, n, max_sweeps=50):

    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        for i in range(n):
            if remaining_time() < 2:
                break
            if best[i] <= 0 or best[i] >= 1:
                continue
            
            # Compute the effect of changing h[i] to t
            # For each lag k, the change is linear or quadratic in t
            t0 = best[i]
            ncorr = 2 * n - 1
            a = np.zeros(ncorr)
            b = np.zeros(ncorr)  # only for lag 0
            
            for off in range(-(n-1), n):
                k = n - 1 + off
                if off == 0:
                    a[k] += 1.0
                    b[k] += -1.0
                else:
                    if 0 <= i + off < n:
                        a[k] += (1.0 - best[i + off])
                    if 0 <= i - off < n:
                        a[k] += (-best[i - off])
            
            t0 = best[i]
            base = best_corr - a * t0 - b * t0 * t0
            
            def obj_at(t):
                vals = base + a * t + b * t * t
                return np.max(vals) * dx
            
            # Try candidates
            candidates = [0.0, 1.0, t0, 0.5, t0 * 2, 1.0 - t0]
            candidates = [max(0.0, min(1.0, c)) for c in candidates]
            
            best_t = t0
            best_val = best_c5

                if abs(t - t0) < 1e-10:
                    continue
                c = obj_at(t)
                if c < best_val - 1e-15:
                    best_val = c
                    best_t = t
            
            # Golden section if improvement
            if best_t != t0:
                lo = max(0.0, t0 - 0.5)
                hi = min(1.0, t0 + 0.5)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = obj_at(c1)
                f2 = obj_at(c2)
                for _ in range(30):
                    if f1 < f2:
                        hi = c2; c2 = c1; f2 = f1
                        c1 = hi - gr * (hi - lo); f1 = obj_at(c1)
                    else:
                        lo = c1; c1 = c2; f1 = f2
                        c2 = lo + gr * (hi - lo); f2 = obj_at(c2)
                t_mid = (lo + hi) / 2.0
                c_mid = obj_at(t_mid)
                if c_mid < best_val:
                    best_val = c_mid

                best_corr = compute_corr(best)
                best_c5 = compute_c5(best)
                improved = True
        
        if not improved:

    return best, best_c5

def binary_search(h_init, n, rng, max_iters=5000):
    # Start from a binary pattern and search via swaps
    h = np.zeros(n)
    ones_count = n // 2
    h[rng.choice(n, ones_count, replace=False)] = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    temperature = 0.01
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        # Random swap: pick a 1 and a 0, swap them
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            temperature = max(0.001, temperature * 0.99)
        else:
            # Revert if not improving (or accept with probability for SA)
            if rng.rand() < np.exp(-(new_c5 - best_c5) / temperature):
                temperature = max(0.001, temperature * 0.99)
            else:
                h[i] = 1.0
                h[j] = 0.0
                temperature = max(0.001, temperature * 0.95)
    
    return best_h, best_c5
