```python
def binary_local_search(h, n, max_passes=50, time_limit=60):
    h = h.astype(float).copy()
    start = time.time()
    c = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    cur_c5 = np.max(c) * (2.0/n)
    offsets = np.arange(-(n-1), n)  # length 2n-1
    
    for p in range(max_passes):
        if time.time() - start > time_limit or remaining() < 10:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_swap = None
        best_new_c5 = cur_c5
        
        # Evaluate all swaps using vectorized approach
        # For memory efficiency, do it in chunks
        for idx in range(len(ones)):
            i = ones[idx]
            # Compute delta for flipping i from 1 to 0
            ip = i + offsets
            im = i - offsets
            valid_p = (ip >= 0) & (ip < n)
            valid_m = (im >= 0) & (im < n)
            delta_i = np.zeros(2*n-1)
            delta_i[valid_p] = -(1.0 - h[ip[valid_p]])
            delta_i[valid_m] += h[im[valid_m]]
            
            for jdx in range(len(zeros)):
                j = zeros[jdx]
                jp = j + offsets
                jm = j - offsets
                valid_pj = (jp >= 0) & (jp < n)
                valid_mj = (jm >= 0) & (jm < n)
                delta_j = np.zeros(2*n-1)
                delta_j[valid_pj] = (1.0 - h[jp[valid_pj]])
                delta_j[valid_mj] -= h[jm[valid_mj]]

                new_max = np.max(new_c)
```

```python
def compute_flip_deltas(h, n, positions, direction):
    """Compute delta vectors for flipping each position in `positions` by `direction`.
    direction=1 means 1→0, direction=-1 means 0→1."""
    offsets = np.arange(-(n-1), n)
    n_off = len(offsets)
    d = np.zeros((len(positions), n_off))
    for idx, pos in enumerate(positions):
        p_plus = pos + offsets
        p_minus = pos - offsets
        valid_p = (p_plus >= 0) & (p_plus < n)
        valid_m = (p_minus >= 0) & (p_minus < n)
        if direction == 1:  # 1→0: δa = -1, δb = +1
            d[idx, valid_p] = -(1.0 - h[p_plus[valid_p]])
            d[idx, valid_m] += h[p_minus[valid_m]]
        else:  # 0→1: δa = +1, δb = -1
            d[idx, valid_p] = (1.0 - h[p_plus[valid_p]])
            d[idx, valid_m] -= h[p_minus[valid_m]]
    return d
```

    h = project_feasible(h, n)
    dx_v = 2.0 / n
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_v
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.success:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h
