def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = max(1, int(np.sum(mask)))
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        flex = np.minimum(h, 1.0 - h)
        order = np.argsort(flex)[::-1]
        for idx in order:
            if diff > 0:
                adj = min(diff, 1.0 - h[idx])
            else:
                adj = max(diff, -h[idx])
            h[idx] += adj
            diff -= adj
            if abs(diff) < 1e-14:
                break
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_h(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    xn = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(xn, xp, src)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    return h


def make_golden(n):
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i * gr) % 1.0 < 0.5 else 0.0
    return h


def make_logistic(n, r=3.99, x0=0.4):
    h = np.zeros(n)
    x = x0
    for i in range(n):
        h[i] = 1.0 if x > 0.5 else 0.0
        x = r * x * (1.0 - x)
    return h


def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)


def lse_objective(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = beta * (corr - m)
    s = np.clip(s, -500.0, 500.0)
    val = (m + np.log(np.sum(np.exp(s))) / beta) * dx
    return val


def lse_gradient(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = beta * (corr - m)
    s = np.clip(s, -500.0, 500.0)
    exps = np.exp(s)
    w = exps / np.sum(exps)

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        lo1 = max(0, d)
        hi1 = min(n, n + d)
        if hi1 > lo1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1 - d:hi1 - d])
        lo2 = max(0, -d)
        hi2 = min(n, n - d)
        if hi2 > lo2:
            grad[lo2:hi2] -= wk * h[lo2 + d:hi2 + d]
    return grad * dx


def do_slsqp(h0, n, beta, budget_fn, maxiter=None):
    if maxiter is None:
        if beta < 500:
            maxiter = 500
        elif beta < 3000:
            maxiter = 350
        else:
            maxiter = 250
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        return lse_objective(x, n, beta)

    def grad(x):
        return lse_gradient(x, n, beta)

    def con_fun(x):
        return np.sum(x) - target

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

    try:
        res = minimize(obj, h0, method='SLSQP', jac=grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0,
                                'eps': 1e-8})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0.copy(), compute_c5(h0)


def do_subgrad(h0, n, budget_fn, rng, max_iters=300):
    h = h0.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    no_improve = 0

    for t in range(max_iters):
        if budget_fn() < 3.0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        top_k = 4
        top_idx = np.argsort(corr)[-top_k:]
        top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]

        grad = np.zeros(n)
        dx = 2.0 / n
        for k_idx in top_idx:
            d = int(k_idx) - (n - 1)
            lo1 = max(0, d)
            hi1 = min(n, n + d)
            if hi1 > lo1:
                grad[lo1:hi1] += (1.0 - h[lo1 - d:hi1 - d])
            lo2 = max(0, -d)
            hi2 = min(n, n - d)
            if hi2 > lo2:
                grad[lo2:hi2] -= h[lo2 + d:hi2 + d]
        grad *= dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        grad /= gnorm

        improved = False
        base_step = 0.015 / (1.0 + 0.005 * t)
        steps = [base_step * s for s in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 4.0, 0.15]]
        for step in steps:
            h_trial = project_feasible(h - step * grad, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-13:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                for _ in range(8):
                    noise = rng.randn(n) * 0.03
                    h_trial = project_feasible(h + noise, n)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < c5_best - 1e-13:
                        c5_best = c5_trial
                        h_best = h_trial.copy()
                        h = h_trial.copy()
                        no_improve = 0
                        improved = True
                        break
                if not improved and no_improve > 40:
                    break

    return h_best, c5_best


def do_coord_descent(h0, n, budget_fn, max_sweeps=8):
    h = h0.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)

    for sweep in range(max_sweeps):
        if budget_fn() < 2.0:
            break
        any_imp = False
        order = np.random.RandomState(42 + sweep).permutation(n)
        for i in order:
            if budget_fn() < 1.0:
                break
            old = h[i]
            for new_val in [0.0, 1.0]:
                if abs(new_val - old) < 1e-8:
                    continue
                h[i] = new_val
                h = project_feasible(h, n)
                c5_test = compute_c5(h)
                if c5_test < c5_best - 1e-14:
                    c5_best = c5_test
                    h_best = h.copy()
                    any_imp = True
                else:
                    h = h_best.copy()
        if not any_imp and sweep >= 2:
            break
    return h_best, c5_best


def do_slsqp_sequence(h0, n, budget_fn, beta_list):
    h = h0.copy()
    c5 = compute_c5(h)
    for beta in beta_list:
        if budget_fn() < 8.0:
            break
        h_new, c5_new = do_slsqp(h, n, beta, budget_fn)
        if c5_new < c5 - 1e-15:
            c5 = c5_new
            h = h_new
    return h, c5


def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = project_feasible(make_thue_morse(parent_n), parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("No parent, using TM: n={}, C5={:.10f}".format(parent_n, best_c5))

    # Phase 1: SLSQP at parent resolution with increasing beta
    print("\n--- Phase 1: SLSQP sequence at n={} ---".format(best_n))
    best_h, best_c5 = do_slsqp_sequence(best_h, best_n, remaining_time,
                                         [20, 50, 150, 400, 1000])
    print("  C5={:.10f}".format(best_c5))

    # Phase 2: Subgradient refinement
    if remaining_time() > 20:
        print("\n--- Phase 2: Subgradient ---")
        best_h, best_c5 = do_subgrad(best_h, best_n, remaining_time, rng, max_iters=250)
        print("  C5={:.10f}".format(best_c5))

    # Phase 3: High-beta SLSQP
    if remaining_time() > 15:
        print("\n--- Phase 3: High-beta SLSQP ---")
        best_h, best_c5 = do_slsqp_sequence(best_h, best_n, remaining_time,
                                             [3000, 8000, 20000, 50000])
        print("  C5={:.10f}".format(best_c5))

    # Phase 4: Try other initializations at same n
    if remaining_time() > 60:
        print("\n--- Phase 4: Alternative starts at n={} ---".format(best_n))
        inits = [make_thue_morse(best_n), make_golden(best_n), make_logistic(best_n)]
        for _ in range(3):
            inits.append(project_feasible(rng.rand(best_n), best_n))
        for h_init in inits:
            if remaining_time() < 20:
                break
            h_init = project_feasible(h_init, best_n)
            c5_init = compute_c5(h_init)
            if c5_init >= best_c5 + 0.01:
                continue
            print("  Starting: C5={:.10f}".format(c5_init))
            h_new, c5_new = do_slsqp_sequence(h_init, best_n, remaining_time,
                                               [50, 200, 800, 3000])
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  Improved to C5={:.10f}".format(best_c5))

    # Phase 5: Subgradient again
    if remaining_time() > 15:
        print("\n--- Phase 5: More subgradient ---")
        best_h, best_c5 = do_subgrad(best_h, best_n, remaining_time, rng, max_iters=200)
        print("  C5={:.10f}".format(best_c5))

    # Phase 6: Interpolate to higher resolution and optimize
    if remaining_time() > 80:
        print("\n--- Phase 6: Higher resolution ---")
        for n_try in [512, 384, 768]:
            if remaining_time() < 50:
                break
            h_interp = project_feasible(interp_h(best_h, best_n, n_try), n_try)
            c5_interp = compute_c5(h_interp)
            print("  n={} interp: C5={:.10f}".format(n_try, c5_interp))
            h_new, c5_new = do_slsqp_sequence(h_interp, n_try, remaining_time,
                                               [50, 200, 800, 3000])
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_try
                print("  n={} improved: C5={:.10f}".format(n_try, best_c5))
            else:
                print("  n={} no improve: C5={:.10f}".format(n_try, c5_new))

    # Phase 7: If we moved to new resolution, do more optimization there
    if remaining_time() > 40 and best_n != 256:
        print("\n--- Phase 7: Optimize at new resolution n={} ---".format(best_n))
        best_h, best_c5 = do_subgrad(best_h, best_n, remaining_time, rng, max_iters=200)
        print("  Subgrad: C5={:.10f}".format(best_c5))
        if remaining_time() > 20:
            best_h, best_c5 = do_slsqp_sequence(best_h, best_n, remaining_time,
                                                 [5000, 15000, 50000])
            print("  SLSQP: C5={:.10f}".format(best_c5))

    # Phase 8: Perturbed restarts
    if remaining_time() > 40:
        print("\n--- Phase 8: Perturbed restarts ---")
        for r in range(4):
            if remaining_time() < 20:
                break
            noise_scale = 0.02 + 0.01 * r
            h_pert = project_feasible(best_h + rng.randn(best_n) * noise_scale, best_n)
            c5_pert = compute_c5(h_pert)
            h_cur = h_pert if c5_pert < best_c5 else best_h.copy()
            c5_cur = min(c5_pert, best_c5)
            h_cur, c5_cur = do_slsqp_sequence(h_cur, best_n, remaining_time,
                                              [200, 800, 3000])
            if c5_cur < best_c5 - 1e-13:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                print("  Restart {}: C5={:.10f}".format(r, best_c5))

    # Phase 9: Final subgradient
    if remaining_time() > 10:
        print("\n--- Phase 9: Final subgrad ---")
        best_h, best_c5 = do_subgrad(best_h, best_n, remaining_time, rng, max_iters=150)
        print("  C5={:.10f}".format(best_c5))

    # Phase 10: Final SLSQP polish
    if remaining_time() > 10:
        print("\n--- Phase 10: Final SLSQP ---")
        best_h, best_c5 = do_slsqp_sequence(best_h, best_n, remaining_time,
                                             [10000, 30000, 100000])
        print("  C5={:.10f}".format(best_c5))

    # Phase 11: Coordinate descent
    if remaining_time() > 8:
        print("\n--- Phase 11: Coord descent ---")
        best_h, best_c5 = do_coord_descent(best_h, best_n, remaining_time, max_sweeps=6)
        print("  C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    # Verify constraint
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-10, \
        "Sum constraint violated: {} vs {}".format(np.sum(best_h), best_n / 2.0)
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12), "Bounds violated"

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
