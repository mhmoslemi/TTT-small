    """Use penalty method to enforce C5 <= C_target."""
    global _G_N, _G_C_TGT, _G_TGT, _G_DX
    _G_N = n; _G_TGT = n/2.0; _G_DX = 2.0/n; _G_C_TGT = C_target
    bd = [(0.0, 1.0)]*n
    cn = [{'type':'eq', 'fun': con_sum, 'jac': lambda x: np.ones(n)}]
    hc = h.copy()
    c5c = compute_c5(hc)
    t0 = time.time()
    # Try multiple penalty weights
    for pw in [1, 10, 100, 1000]:
        if time.time() - t0 > max_time or time_left() < 10:
            break
        _G_C_TGT = C_target
        # Modify penalty obj to include weight
        # Actually, let me just use the raw penalty
        mi = max(30, min(300, int((max_time - (time.time()-t0)) * 3)))
        try:
            r = minimize(obj_penalty, hc, method='SLSQP', bounds=bd,
                        constraints=cn, options={'maxiter': mi, 'ftol': 1e-18, 'iprint': 0})
            hn = project_feasible(r.x, n)
            c5n = compute_c5(hn)
            if c5n < c5c - 1e-15:
                hc = hn; c5c = c5n
        except:
            pass
    return hc, c5c

def subgrad_phase(h, n, max_iters=500, max_time=60):
    gset_n = n  # use global
    global _G_N, _G_DX
    _G_N = n; _G_DX = 2.0/n
    hc = h.copy()
    c5b = compute_c5(hc)
    hb = hc.copy()
    step = 0.04
    ni = 0
    t0 = time.time()
    for it in range(max_iters):
        if time.time() - t0 > max_time or time_left() < 5:
            break
        g = subgrad_max(hc, 3)
        improved = False
        for s in [step, step*0.5, step*2.0, step*0.25, 0.01, 0.005, 0.1, 0.02]:
            ht = project_feasible(hc - s*g, n)
            c5t = compute_c5(ht)
            if c5t < c5b - 1e-14:
                hb = ht.copy(); hc = ht.copy(); c5b = c5t
                step = min(s*1.1, 0.15)
                ni = 0
                improved = True
                break
        if not improved:
            ni += 1
            step *= 0.9
            if ni > 50:
                break
    return hb, c5b

def coord_phase(h, n, sweeps=8, max_time=60, seed=42):
    global _G_N, _G_DX
    _G_N = n; _G_DX = 2.0/n
    hc = h.copy()
    c5b = compute_c5(hc)
    hb = hc.copy()
    t0 = time.time()
    rng = np.random.RandomState(seed)
    for sw in range(sweeps):
        if time.time() - t0 > max_time or time_left() < 5:
            break
        imp = False
        for i in rng.permutation(n):
            if time.time() - t0 > max_time or time_left() < 3:
                break
            old = hc[i]
            for nv in [0.0, 1.0]:
                if abs(nv - old) < 1e-7: continue
                hc[i] = nv
                hc = project_feasible(hc, n)
                c5t = compute_c5(hc)
                if c5t < c5b - 1e-14:
                    c5b = c5t; hb = hc.copy(); imp = True
                else:
                    hc = hb.copy()
            hc = hb.copy()
        if not imp:
            break
    return hb, c5b
