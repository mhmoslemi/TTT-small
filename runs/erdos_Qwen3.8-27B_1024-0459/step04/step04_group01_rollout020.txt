def compute_c5(h):

def compute_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def to_binary(h, n):
    h = h.copy()
    thresholds = np.sort(h)[::-1]
    h[thresholds >= 0.5] = 1.0
    h[thresholds < 0.5] = 0.0
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        ones_order = np.argsort(-h)
        for idx in range(n // 2, n):
            h[ones_order[idx]] = 0.0
        for idx in range(n // 2):
            h[ones_order[idx]] = 1.0
    elif n_ones < n // 2:
        zeros_order = np.argsort(h)
        for idx in range(n - n // 2, n):
            h[zeros_order[idx]] = 1.0
        for idx in range(n - n // 2):
            h[zeros_order[idx]] = 0.0
    return h

def interp_to(src_h, src_n, dst_n):

    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def slsqp_smooth(h, n, beta, maxiter=300):
    dx_val = 2.0 / n
    tgt = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = beta * (corr - m)
        cl = np.clip(shifted, -50, 50)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_val

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = beta * (corr - m)
        cl = np.clip(shifted, -50, 50)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] > 1e-14:
                d = k - (n - 1)
                if d >= 0:
                    npairs = n - d
                    np.add.at(g, range(d, d + npairs), (1.0 - x[:npairs]) * w[k])
                    np.add.at(g, range(npairs), -x[d:d + npairs] * w[k])
                else:
                    ao = -d
                    npairs = n - ao
                    np.add.at(g, range(npairs), (1.0 - x[ao:ao + npairs]) * w[k])
                    np.add.at(g, range(ao, ao + npairs), -x[:npairs] * w[k])
        return g * dx_val

    def sum_fun(x):
        return np.sum(x) - tgt

    def sum_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

    # Phase 8: Continuous refinement (SLSQP) then re-binarize
    if remaining() > 15:
        h_cont = best_h.copy().astype(float)
        for beta in [2000, 8000, 30000]:
            if remaining() < 8:
                break
            h_new = slsqp_smooth(h_cont, best_n, beta, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_new_bin = to_binary(h_new, best_n)
            c5_bin_new = compute_c5(h_new_bin)
            if c5_bin_new < best_c5:
                best_c5 = c5_bin_new
                best_h = h_new_bin.copy()
            h_cont = h_new

    # Phase 9: Final tabu polish
    if remaining() > 10:
        h_final_bin = to_binary(best_h, best_n)
        rng6 = np.random.RandomState(seed + 555)
        t_limit = min(30, remaining() - 5)
        h_pol, c5_pol = tabu_search_binary(h_final_bin, best_n, t_limit, rng6,
                                           tabu_len=50, n_trials_per_iter=120, kick_every=60)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Validate
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-12:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
