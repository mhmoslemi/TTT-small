_START = 0.0
_BUDGET = 1300.0
_N = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 5.0)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1
    return h_bin

def sa_binary(h, n, time_limit, seed):
    """Simulated annealing on binary sequences with adaptive moves."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    target = n // 2
    
    n_ones = int(np.sum(h))
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1

    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()

    T = 6.0
    alpha = 0.9992
    start = time.time()
    stall = 0
    max_stall = 10000

    while time.time() - start < time_limit and T > 0.03:
        if remaining() < 5:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Choose move type
        r = rng.rand()
        if r < 0.7:
            # Single swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            move_info = ('swap', i, j)
        elif r < 0.85:
            # Double swap
            if len(ones) < 2 or len(zeros) < 2:
                continue
            i1, i2 = rng.choice(len(ones), 2, replace=False)
            j1, j2 = rng.choice(len(zeros), 2, replace=False)
            oi1, oi2 = ones[i1], ones[i2]
            oj1, oj2 = zeros[j1], zeros[j2]
            h[oi1] = 0.0
            h[oi2] = 0.0
            h[oj1] = 1.0
            h[oj2] = 1.0
            move_info = ('dswap', oi1, oi2, oj1, oj2)
        else:
            # Block move (move a run of 1s to a different location)
            block_sizes = [1, 2, 3]
            bs = block_sizes[rng.randint(len(block_sizes))]
            # Find all positions where we have a block of bs consecutive 1s
            # followed by a 0 (or end)
            candidates = []
            for pos in range(n - bs + 1):
                if all(h[pos+k] == 1 for k in range(bs)):
                    if pos + bs < n and h[pos+bs] == 0:
                        candidates.append(pos)
            if not candidates:
                # Fallback to single swap
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                move_info = ('swap', i, j)
            else:
                src = candidates[rng.randint(len(candidates))]
                # Find target: bs consecutive 0s
                target_candidates = []
                for pos in range(n - bs + 1):
                    if all(h[pos+k] == 0 for k in range(bs)):
                        if pos == src:
                            continue
                        target_candidates.append(pos)
                if not target_candidates:
                    h[i] = 0.0  # This won't work, let's just skip
                    continue
                dst = target_candidates[rng.randint(len(target_candidates))]
                h[src:src+bs] = 0.0
                h[dst:dst+bs] = 1.0
                move_info = ('block', src, dst, bs)

        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        delta = new_max - cur_max
        
        if delta < 0 or (T > 0.03 and rng.rand() < np.exp(-delta / T)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            # Revert
            mtype = move_info[0]
            if mtype == 'swap':
                h[move_info[1]] = 1.0
                h[move_info[2]] = 0.0
            elif mtype == 'dswap':
                h[move_info[1]] = 1.0
                h[move_info[2]] = 1.0
                h[move_info[3]] = 0.0
                h[move_info[4]] = 0.0
            elif mtype == 'block':
                src, dst, bs = move_info[1], move_info[2], move_info[3]
                h[dst:dst+bs] = 0.0
                h[src:src+bs] = 1.0
            stall += 1
        
        T *= alpha
        if stall > max_stall:
            T = max(T, 4.0)
            stall = 0

    return best_h, best_max * dx_val

def coordinate_descent(h, n, max_rounds=50):
    """Coordinate descent: try changing each coordinate to 0 or 1."""
    h = h.copy()
    target = n / 2.0
    best_c5 = compute_c5(h)
    
    for rnd in range(max_rounds):
        if remaining() < 5:
            break
        improved = False
        order = np.random.permutation(n)  # Hmm, need rng...
        # Actually let's use a fixed order or deterministic
        for i in range(n):
            if remaining() < 3:
                break
            old_val = h[i]
            # Try setting to 0
            h[i] = 0.0
            # Project to maintain sum
            diff = target - np.sum(h)
            if abs(diff) < 1e-10:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    improved = True
                else:
                    h[i] = old_val
            else:
                h[i] = old_val
            # Try setting to 1
            h[i] = 1.0
            diff = target - np.sum(h)
            if abs(diff) < 1e-10:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    improved = True
                else:
                    h[i] = old_val
            else:
                h[i] = old_val
        if not improved:
            break
    
    return h, best_c5

def slsqp_smooth(h, n, beta, maxiter=300):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    
    def smooth_obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -60, 60)
        ex = np.exp(cl)
        s = np.sum(ex)
        return (m + np.log(s) / beta) * (2.0 / n)
    
    def corr_grad_per_shift(x, k):
        g = np.zeros(n)
        off = k - (n - 1)
        if off >= 0:
            npairs = n - off
            g[off:off + npairs] += (1.0 - x[:npairs])
            g[:npairs] -= x[off:off + npairs]
        else:
            ao = -off
            npairs = n - ao
            g[:npairs] += (1.0 - x[ao:ao + npairs])
            g[ao:ao + npairs] -= x[:npairs]
        return g
    
    def smooth_grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -60, 60)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] > 1e-12:
                g += w[k] * corr_grad_per_shift(x, int(k))
        return g * (2.0 / n)
    
    def sum_con(x):
        return np.sum(x) - _TGT
    
    def sum_con_jac(x):
        return np.ones(_N)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h
