def smooth_max_obj(x, n, beta):
    corr = compute_corr(x)
    m = float(np.max(corr))
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)


def smooth_max_grad(x, n, beta):
    corr = compute_corr(x)
    m = float(np.max(corr))
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)

            h_t = lbfgs_smooth(h_t, n_t, beta=5000, maxiter=300)
            c5_t = compute_c5(h_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t
            h = h_t.copy()

    # Phase 7: Reduced-space optimization (optimize only fractional coords)
    if remaining() > 15:
        h_red = reduced_space_opt(best_h, best_n, frac_thresh=0.05, maxiter=400)
        c5_red = compute_c5(h_red)
        if c5_red < best_c5:
            best_c5 = c5_red
            best_h = h_red.copy()
            h = h_red.copy()

    # Phase 8: Final intensive active-set with many iterations
    for tk in [10, 20, 50]:
        if remaining() < 3:
            break
        h_new, c5_new = active_set_descend(h, best_n, top_k=tk, n_iter=10000, step_scale=0.008, mom=0.92)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 9: L-BFGS-B final polish
    if remaining() > 15:
        h_final = lbfgs_smooth(h, best_n, beta=8000, maxiter=600)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 10: Second reduced-space pass
    if remaining() > 10:
        h_red2 = reduced_space_opt(best_h, best_n, frac_thresh=0.03, maxiter=300)
        c5_red2 = compute_c5(h_red2)
        if c5_red2 < best_c5:
            best_c5 = c5_red2
            best_h = h_red2.copy()

    # Phase 11: Final active-set micro-refinement
    if remaining() > 5:
        h_micro, c5_micro = active_set_descend(best_h, best_n, top_k=15, n_iter=5000, step_scale=0.003, mom=0.95)
        if c5_micro < best_c5:
            best_c5 = c5_micro
            best_h = h_micro.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        best_c5 = final_c5
    else:
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
