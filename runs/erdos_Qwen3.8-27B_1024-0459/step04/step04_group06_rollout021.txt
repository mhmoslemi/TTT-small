    return best_h, best_c5


def lag_targeting(h, n, num_rounds=30, rng=None):
    """Specifically target the worst lag by adjusting h to reduce it"""
    global _G_N, _G_DX
    _G_N = n
    _G_DX = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    if rng is None:
        rng = np.random.RandomState(123)
    for rd in range(num_rounds):
        if remaining_time() < 4.0:
            break
        corr = compute_corr(best_h)
        dx = _G_DX
        worst_k = int(np.argmax(corr))
        offset = worst_k - (n - 1)
        if offset >= 0:
            nv = n - offset
            if nv <= 0:
                continue
            pair_idx_a = np.arange(nv)
            pair_idx_b = pair_idx_a + offset
        else:
            m2 = -offset
            nv = n - m2
            if nv <= 0:
                continue
            pair_idx_b = np.arange(nv)
            pair_idx_a = pair_idx_b + m2
        # For this lag, we want to reduce sum h[a]*(1-h[b])
        # Try flipping individual pairs
        for _ in range(5):
            idx = rng.randint(0, nv)
            ia = pair_idx_a[idx]
            ib = pair_idx_b[idx]
            # Option 1: decrease h[ia] (reduces h[ia]*(1-h[ib]))
            if best_h[ia] > 0.1:
                h_trial = best_h.copy()
                h_trial[ia] = np.clip(h_trial[ia] - 0.15, 0.0, 1.0)
                h_trial[ib] = np.clip(h_trial[ib] + 0.15, 0.0, 1.0)
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < best_c5 - 1e-13:
                    best_c5 = c5_t
                    best_h = h_trial.copy()
            # Option 2: increase h[ib]
            if best_h[ib] < 0.9:
                h_trial = best_h.copy()
                h_trial[ib] = np.clip(h_trial[ib] + 0.15, 0.0, 1.0)
                h_trial[ia] = np.clip(h_trial[ia] - 0.15, 0.0, 1.0)
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < best_c5 - 1e-13:
                    best_c5 = c5_t
                    best_h = h_trial.copy()
        # Also try random small perturbations
        if rd % 3 == 0:
            h_trial = best_h + 0.02 * rng.randn(n)
            h_trial = project_feasible(h_trial, n)
            c5_t = compute_c5(h_trial)
            if c5_t < best_c5 - 1e-13:
                best_c5 = c5_t
                best_h = h_trial.copy()
    return best_h, best_c5


def multi_perturb(h, n, rng, num_pert=20):
    """Diverse perturbations with SLSQP polish on promising ones"""
    global _G_N, _G_TGT, _G_DX, _G_BETA
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun}]
    for p in range(num_pert):
        if remaining_time() < 10.0:
            break
        pt = p % 8
        h_trial = best_h.copy()
        if pt == 0:
            # Flip a block
            start = rng.randint(0, max(1, n - 16))
            L = rng.randint(4, min(16, n // 5))
            h_trial[start:start + L] = 1.0 - h_trial[start:start + L]
        elif pt == 1:
            # Swap two blocks
            s1 = rng.randint(0, n // 3)
            s2 = s1 + rng.randint(n // 5, n - n // 5)
            L = rng.randint(3, min(8, n // 15))
            a = h_trial[s1:s1 + L].copy()
            h_trial[s1:s1 + L] = h_trial[s2:s2 + L]
            h_trial[s2:s2 + L] = a
        elif pt == 2:
            # Gaussian noise
            h_trial += 0.05 * rng.randn(n)
        elif pt == 3:
            # Sinusoidal modulation
            freq = rng.randint(1, n // 4)
            phase = rng.rand() * 2 * np.pi
            t = np.arange(n) / float(n)
            h_trial += 0.06 * np.sin(2.0 * np.pi * freq * t + phase)
        elif pt == 4:
            # Random subset reassignment
            subset = rng.choice(n, size=n // 8, replace=False)
            h_trial[subset] = rng.rand(len(subset))
        elif pt == 5:
            # Mirror a section
            L = rng.randint(n // 6, n // 3)
            start = rng.randint(0, n - L)
            h_trial[start:start + L] = h_trial[start_L[::-1]]
        elif pt == 6:
            # Threshold shift
            thresh = rng.uniform(0.3, 0.7)
            h_trial = (h_trial > thresh).astype(float) * 0.5 + h_trial * 0.5
        else:
            # Random few points flip
            idx = rng.choice(n, size=rng.randint(5, 20), replace=False)
            h_trial[idx] = 1.0 - h_trial[idx]
        h_trial = project_feasible(h_trial, n)
        c5_t = compute_c5(h_trial)
        if c5_t < best_c5 - 1e-12:
            best_c5 = c5_t
            best_h = h_trial.copy()
        # SLSQP polish on this perturbation
        if remaining_time() < 8.0:
            break
        _G_BETA = 500.0
        try:
            res = minimize(obj_lse, h_trial, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-13, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = np.binary_repr(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h


def coordinate_descent_short(h, n, max_sweeps=6, rng=None):
    """Quick coordinate descent on binary-ish values"""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    if rng is None:
        rng = np.random.RandomState(7)
    for sweep in range(max_sweeps):
        if remaining_time() < 3.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if remaining_time() < 2.0:
                break
            old = best_h[i]
            for new_v in [0.0, 1.0]:
                if abs(new_v - old) < 1e-8:
                    continue
                h_trial = best_h.copy()
                h_trial[i] = new_v
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < best_c5 - 1e-13:
                    best_c5 = c5_t
                    best_h = h_trial.copy()
                    any_imp = True
            for delta in [0.03, -0.03, 0.08, -0.08]:
                h_trial = best_h.copy()
                h_trial[i] = np.clip(h_trial[i] + delta, 0.0, 1.0)
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < best_c5 - 1e-13:
                    best_c5 = c5_t
                    best_h = h_trial.copy()
                    any_imp = True
        if not any_imp and sweep > 1:
            break
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Initialize from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 256
        best_h = project_feasible(make_thue_morse(best_n), best_n)
        best_c5 = compute_c5(best_h)
        print("Thue-Morse init: n={}, C5={:.10f}".format(best_n, best_c5))

    n = best_n
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n

    # Phase 1: Rapid SLSQP polish (increasing beta)
    print("\n--- Phase 1: SLSQP polish ---")
    best_h, best_c5 = slsqp_round(best_h, n, [100, 300, 800, 2000, 5000, 10000], maxiter=350, ftol=1e-14)

    best_h, best_c5 = lag_targeting(best_h, n, num_rounds=40, rng=rng)

    # Phase 4: Another SLSQP round
    print("\n--- Phase 4: SLSQP round 2 ---")
    best_h, best_c5 = slsqp_round(best_h, n, [800, 2000, 5000, 15000, 30000], maxiter=400, ftol=1e-15)
    print("  After SLSQP2: C5={:.10f}".format(best_c5))

    # Phase 5: Multi-perturbation exploration
    print("\n--- Phase 5: Multi-perturb ---")
    best_h, best_c5 = multi_perturb(best_h, n, rng, num_pert=20)
    print("  After perturb: C5={:.10f}".format(best_c5))

    # Phase 6: p-norm optimization
    if remaining_time() > 30.0:
        print("\n--- Phase 6: p-norm ---")
        best_h, best_c5 = pnorm_round(best_h, n, [20, 50, 100, 200], maxiter=250, ftol=1e-14)
        print("  After p-norm: C5={:.10f}".format(best_c5))

    # Phase 7: Try different resolution
    if remaining_time() > 60.0:
        print("\n--- Phase 7: Resolution search ---")
        for n_try in [200, 256, 320, 400]:
            if remaining_time() < 25.0:
                break
            if n_try == n:
                continue
            h_interp = project_feasible(interp_to(best_h, n, n_try), n_try)
            c5_h = compute_c5(h_interp)
            # Quick optimization at this resolution
            _G_N = n_try
            _G_TGT = n_try / 2.0
            _G_DX = 2.0 / n_try
            h_interp, c5_h = slsqp_round(h_interp, n_try, [500, 2000, 5000], maxiter=200, ftol=1e-14)
            h_interp, c5_h = gradient_descent(h_interp, n_try, max_iter=100, step0=0.03, momentum=0.8)
            h_interp, c5_h = lag_targeting(h_interp, n_try, num_rounds=15, rng=rng)
            _G_N = n
            _G_TGT = n / 2.0
            _G_DX = 2.0 / n
            print("  n={}: C5={:.10f}".format(n_try, c5_h))
            if c5_h < best_c5 - 1e-10:
                best_c5 = c5_h
                best_h = h_interp.copy()
                best_n = n_try
                n = best_n
                _G_N = n
                _G_TGT = n / 2.0
                _G_DX = 2.0 / n
                print("  ** New best!**")

    # Phase 8: Coordinate descent
    if remaining_time() > 15.0:

        best_h, best_c5 = coordinate_descent_short(best_h, best_n, max_sweeps=8, rng=rng)

        _G_N = n
        _G_TGT = n / 2.0
        _G_DX = 2.0 / n
        best_h, best_c5 = slsqp_round(best_h, n, [5000, 10000, 25000, 50000], maxiter=500, ftol=1e-15)
        print("  Final SLSQP: C5={:.10f}".format(best_c5))

        n = best_n

        _G_TGT = n / 2.0
        _G_DX = 2.0 / n

                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new
            except Exception:
                pass
        print("  Final polish: C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
