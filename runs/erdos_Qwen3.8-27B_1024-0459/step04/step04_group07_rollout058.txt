    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)

def subgradient_step(h, n, top_k=3, alpha=0.01):
    """One subgradient step on max-corr."""
    corr = get_corr(h)
    m = np.max(corr)
    active = np.where(corr >= m - 1e-6)[0][:top_k]
    
    # Compute average gradient of active lags
    grad = np.zeros(n)
    for k in active:
        n_min = max(0, -k)
        n_max = min(n - 1, n - 1 - k)
        for i in range(n):
            val = 0.0
            nn = i - k
            if n_min <= nn <= n_max:
                val += (1.0 - h[nn])
            if n_min <= i <= n_max:
                ki = k + i
                if 0 <= ki < n:
                    val -= h[ki]
            grad[i] += val
    grad /= len(active)
    
    # Projected step: h_new = h - alpha * grad, then project
    h_new = h - alpha * grad
    h_new = project_feasible(h_new, n)
    return h_new

def coord_descent_simple(h, n, max_sweeps=20, tlimit=lambda: 999):
    """Simple coordinate descent: try setting each h[i] to 0 or 1."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if tlimit() < 2:
            break
        improved = False
        for i in range(n):
            for v in [0.0, 1.0]:
                if abs(v - best_h[i]) < 1e-9:
                    continue
                trial = best_h.copy()
                trial[i] = v
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    best_h = trial
                    improved = True
            if not improved and tlimit() < 1:
                break
        if not improved:
            break
    return best_h, best_c5
