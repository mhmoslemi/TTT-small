def project_feasible(h, n):

import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(interior):
            h[interior] += diff / np.sum(interior)
        else:
            if diff > 0:
                idx = np.argmin(h)
            else:
                idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)

def smooth_obj_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    ncorr = 2 * n - 1
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-12:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def slsqp_optimize(h, n, beta, maxiter=400):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        return smooth_obj_grad(x, n, beta)[0]
    
    def obj_grad(x):
        return smooth_obj_grad(x, n, beta)[1]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            trial_c5 = compute_c5(trial)
            if trial_c5 <= compute_c5(h) + 1e-14:
                return trial
    except Exception:
        pass
    return h.copy()

import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):

        if abs(diff) < 1e-13:
            break
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(interior):
            h[interior] += diff / np.sum(interior)
        else:
            if diff > 0:

            else:
                idx = int(np.argmax(h))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

def smooth_obj_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    ncorr = 2 * n - 1
    for k in range(ncorr):

        if w < 1e-12:

        pass
    return h.copy()

def interp_up(src_h, dst_n):
    src_n = len(src_h)
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def vectorized_coord_descent(h, n, max_sweeps=50):
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    rng_local = np.random.RandomState(12345)

    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        interior = np.where((best > 1e-8) & (best < 1.0 - 1e-8))[0]
        if len(interior) == 0:
            break

        # Bias toward active lags
        max_c = np.max(best_corr)
        active_lags = np.where(best_corr >= max_c * 0.85)[0]
        active_idx = set()
        for k in active_lags[:30]:
            off = int(k) - (n - 1)
            if off >= 0:
                active_idx.update(range(max(0, off), min(n, n - off + off)))
            else:
                aoff = -off
                active_idx.update(range(0, n - aoff))
                active_idx.update(range(aoff, n))

        order = interior.copy()
        # Put active indices first
        for j in range(len(order) - 1, -1, -1):
            if order[j] in active_idx:
                order[j], order[j % len(order)] = order[j % len(order)], order[j]
        order = order[rng_local.permutation(len(order))]

        for p in order:
            if remaining_time() < 1:
                break
            t0 = best[p]

            # Compute a (linear coefficients) vectorized
            a = np.zeros(ncorr)

            # Type 1: (1-h[p-L]) for L where 0 <= p-L < n
            lo1 = max(-(n - 1), p - n + 1)
            hi1 = min(n - 1, p)
            if lo1 <= hi1:
                offs = np.arange(lo1, hi1 + 1)
                a[n - 1 + offs] += (1.0 - best[p - offs])

            # Type 2: -h[p+L] for L where 0 <= p+L < n
            lo2 = max(-(n - 1), -p)
            hi2 = min(n - 1, n - 1 - p)
            if lo2 <= hi2:
                offs = np.arange(lo2, hi2 + 1)
                a[n - 1 + offs] += (-best[p + offs])

            # Override offset 0 (quadratic case)
            a[n - 1] = 1.0
            b = np.zeros(ncorr)
            b[n - 1] = -1.0

            # Compute base
            base = best_corr - a * t0
            base[n - 1] = best_corr[n - 1] - t0 + t0 * t0

            # Find optimal t in [0,1] minimizing max_k(base[k] + a[k]*t + b[k]*t^2)
            # Coarse grid
            t_grid = np.linspace(0.0, 1.0, 30)
            vals = base[None, :] + a[None, :] * t_grid[:, None] + b[None, :] * t_grid[:, None] ** 2
            obj_grid = np.max(vals, axis=1)
            t_coarse = t_grid[np.argmin(obj_grid)]

            # Golden section refine
            hw = 0.03
            lo = max(0.0, t_coarse - hw)
            hi = min(1.0, t_coarse + hw)

            if hi - lo > 1e-10:
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)

                def f_t(t):
                    v = base + a * t + b * t * t
                    return np.max(v)

                f1 = f_t(c1)
                f2 = f_t(c2)
                for _ in range(35):
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = f_t(c1)
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = f_t(c2)
                t_new = (lo + hi) / 2.0
            else:
                t_new = t_coarse

            # Check improvement
            if abs(t_new - t0) > 1e-10:
                new_vals = base + a * t_new + b * t_new * t_new
                c5_cand = np.max(new_vals) * dx
                if c5_cand < best_c5 - 1e-15:
                    best[p] = t_new
                    best_corr = new_vals
                    best_c5 = c5_cand
                    improved = True

        if not improved:
            break

    return best, best_c5

def subgradient_polish(h, n, max_iters=300):
    best = h.copy()
    best_c5 = compute_c5(best)
    step = 0.04
    stalled = 0

    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(best)
        max_corr = np.max(corr)
        # Soft active set
        threshold = max_corr - max_corr * 0.15
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        grad = np.zeros(n)
        for k in active:
            off = int(k) - (n - 1)
            if off >= 0:
                npairs = n - off
                if npairs > 0:
                    grad[off:off + npairs] += (1.0 - best[:npairs])
                    grad[:npairs] -= best[off:off + npairs]
            else:
                aoff = -off
                npairs = n - aoff
                if npairs > 0:
                    grad[:npairs] += (1.0 - best[aoff:aoff + npairs])
                    grad[aoff:aoff + npairs] -= best[:npairs]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm

        improved = False
        for scale in [1.0, 0.5, 2.0, 0.25, 0.1]:
            s = step * scale
            h_trial = best - s * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-16:
                best_c5 = c5_trial
                best = h_trial.copy()
                improved = True
                step = min(step * 1.15, 0.08)
                stalled = 0
                break
        if not improved:
            step *= 0.6
            stalled += 1
            if stalled > 30:
                break
            if stalled == 20:
                # Try SLSQP
                h_new = slsqp_optimize(best, n, 5000, 200)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best = h_new.copy()
                    step = 0.02
                    stalled = 0

    return best, best_c5

def bimodal_push(h, n, strength=0.3):
    """Try pushing values toward 0 or 1 and check if C5 improves."""
    h_new = h.copy()
    for i in range(n):
        if h_new[i] < 0.5:
            h_new[i] = max(0.0, h_new[i] - strength * h_new[i])
        else:
            h_new[i] = min(1.0, h_new[i] + strength * (1.0 - h_new[i]))
    h_new = project_feasible(h_new, n)
    c5_new = compute_c5(h_new)
    c5_orig = compute_c5(h)
    if c5_new < c5_orig:
        return h_new, c5_new
    return h, c5_orig
