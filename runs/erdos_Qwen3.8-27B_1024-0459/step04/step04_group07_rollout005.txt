_G_BETA = 1.0
_G_DX = 1.0
_G_PEN = 0.0
_G_TGT = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idxs = np.argsort(1.0 - h)
        else:
            idxs = np.argsort(h)
        for idx in idxs:
            trial = h.copy()
            delta = np.clip(diff, -(h[idx]), 1.0 - h[idx])
            trial[idx] += delta
            if abs(trial[idx] - h[idx]) > 1e-15:
                h = trial
                break
    return h

def obj_lse(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    b = _G_BETA
    s = corr - m

def lse_lbfgsb(h0, n, beta, maxiter, pen=500.0):

    _G_PEN = float(pen)
    _G_TGT = n / 2.0
    bounds = [(0.0, 1.0)] * n

        res = minimize(obj_lse, h0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,

    return h

def slsqp_run(h0, n, beta, maxiter=500):

def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c else 0.0
    s = float(np.sum(h))
    target = n // 2
    if int(round(s)) != target:
        diff = target - int(round(s))
        if diff > 0:
            zeros = np.where(h == 0)[0]
            for i in range(min(diff, len(zeros))):
                h[zeros[i]] = 1.0
        elif diff < 0:
            ones = np.where(h == 1)[0]
            for i in range(min(-diff, len(ones))):
                h[ones[i]] = 0.0
    return h
