def project_feasible(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Adjust values that are not at bounds
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:

            if diff > 0:
                room = 1.0 - h
                idx = np.argsort(room)[::-1]
            else:

                idx = np.argsort(room)
            rem = diff
            for i in idx:

                h[i] += step
                rem -= step
                if abs(rem) < 1e-14:
                    break

            grad[:npairs] -= wk * x[ao:ao + npairs]
    return grad * _G_DX


def do_lbfgs(h0, n, betas, budget_fn, max_iter=500):
    global _G_N, _G_BETA, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    for beta in betas:
        if budget_fn() < 5.0:
            break
        _G_BETA = float(beta)
        mi = max(50, min(max_iter, int(budget_fn() * 2)))
        try:
            res = minimize(lse_func, h_cur, jac=lse_grad,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': mi, 'ftol': 1e-18,
                                    'gtol': 1e-12, 'maxfun': mi * 5,
                                    'iprint': 0, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-16:
                c5_cur = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, c5_cur


def do_proj_subgrad(h0, n, budget_fn, n_iters=5000, top_k=5):
    dx = 2.0 / n
    h = h0.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    step = 0.05
    no_improve = 0
    for it in range(n_iters):
        if budget_fn() < 3.0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        top_idx = np.argsort(corr)[-top_k:][::-1]
        grad = np.zeros(n)
        for k in top_idx:
            grad += _corr_grad_one(h, k)
        grad *= dx / top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-18:
            break
        grad /= gn
        improved = False
        for s in [step, step * 0.5, step * 2.0, step * 0.1]:
            h_t = project_feasible(h - s * grad, n)
            c5_t = compute_c5(h_t)
            if c5_t < c5_best - 1e-14:
                c5_best = c5_t
                h_best = h_t.copy()
                h = h_t.copy()
                improved = True
                no_improve = 0
                break
        if not improved:
            no_improve += 1
            step *= 0.8
            if no_improve > 50:
                noise = np.random.randn(n) * 0.01
                h_t = project_feasible(h + noise, n)
                c5_t = compute_c5(h_t)
                if c5_t < c5_best - 1e-14:
                    c5_best = c5_t
                    h_best = h_t.copy()
                    h = h_t.copy()
                    no_improve = 0
                step = 0.03
        else:
            step = min(step * 1.05, 0.15)
    return h_best, c5_best


def do_binary_cd(h0, n, budget_fn, max_sweeps=30):
    h = h0.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    rng_local = np.random.RandomState(12345)
    for sweep in range(max_sweeps):
        if budget_fn() < 2.0:
            break
        improved = False
        order = rng_local.permutation(n)
        for i in order:
            if budget_fn() < 1.0:
                break
            old = h[i]
            for nv in [0.0, 1.0]:
                if abs(nv - old) < 1e-7:
                    continue
                h[i] = nv
                h = project_feasible(h, n)
                c5_t = compute_c5(h)
                if c5_t < c5_best - 1e-14:
                    c5_best = c5_t
                    h_best = h.copy()
                    improved = True
                else:
                    h[i] = old
                    h = project_feasible(h, n)
        if not improved and sweep > 3:
            break
    return h_best, c5_best


def do_sim_anneal_bin(h0, n, budget_fn, n_steps=20000, T_init=0.3, T_final=0.01):
    rho = (np.log(T_final) - np.log(T_init)) / n_steps
    T = T_init
    h = h0.copy()
    h_bin = (h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, n)
    c5_cur = compute_c5(h_bin)
    c5_best = c5_cur
    h_best = h_bin.copy()
    rng_sa = np.random.RandomState(777)
    for step in range(n_steps):
        if budget_fn() < 3.0:
            break
        if step % 100 == 0:
            T = T_init * np.exp(rho * step)
        i = rng_sa.randint(n)
        old_v = h_bin[i]
        h_bin[i] = 1.0 - old_v
        h_bin = project_feasible(h_bin, n)
        c5_t = compute_c5(h_bin)
        delta = c5_t - c5_cur
        if delta < 0 or rng_sa.rand() < np.exp(-delta / T):
            c5_cur = c5_t
            if c5_cur < c5_best - 1e-15:
                c5_best = c5_cur
                h_best = h_bin.copy()
        else:
            h_bin[i] = old_v
            h_bin = project_feasible(h_bin, n)
    return h_best, c5_best


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    return project_feasible(h, n)


def make_gb(n, alpha):
    h = np.zeros(n)
    state = 0.5
    for i in range(n):
        state = (state + alpha) % 1.0
        h[i] = 1.0 if state >= 0.5 else 0.0
    counted = int(np.sum(h))
    if counted > n // 2:
        ones = np.where(h == 1.0)[0]
        h[ones[:counted - n // 2]] = 0.0
    elif counted < n // 2:
        zeros = np.where(h == 0.0)[0]
        h[zeros[:n // 2 - counted]] = 1.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET, _G_N, _G_BETA, _G_TGT, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    def budget_fn():
        return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        best_h = parent_h.copy()
        best_c5 = compute_c5(best_h)
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, best_c5))
    except Exception:
        best_n = 256
        best_h = make_thue_morse(best_n)
        best_c5 = compute_c5(best_h)
        print("Thue-Morse init: n={}, C5={:.10f}".format(best_n, best_c5))

    # Phase 1: L-BFGS at starting resolution
    if budget_fn() > 10:
        print("P1: L-BFGS n={} betas 20-2000".format(best_n))
        best_h, best_c5 = do_lbfgs(best_h, best_n, [20, 100, 500, 2000], budget_fn, max_iter=800)
        print("  C5={:.10f}".format(best_c5))

    # Phase 2: Projected subgradient
    if budget_fn() > 10:
        print("P2: Subgrad n={}".format(best_n))
        best_h, best_c5 = do_proj_subgrad(best_h, best_n, budget_fn, n_iters=4000, top_k=5)
        print("  C5={:.10f}".format(best_c5))

    # Phase 3: High-beta L-BFGS
    if budget_fn() > 10:
        print("P3: L-BFGS high beta n={}".format(best_n))
        best_h, best_c5 = do_lbfgs(best_h, best_n, [5000, 15000, 40000], budget_fn, max_iter=400)
        print("  C5={:.10f}".format(best_c5))

    # Phase 4: Binary coordinate descent
    if budget_fn() > 8:
        print("P4: Binary CD n={}".format(best_n))
        best_h, best_c5 = do_binary_cd(best_h, best_n, budget_fn, max_sweeps=20)
        print("  C5={:.10f}".format(best_c5))

    # Phase 5: Multi-resolution
    res_list = []
    for nr in [512, 384, 320, 288, 448, 1024, 640, 768]:
        if nr != best_n:
            res_list.append(nr)
    for n_try in res_list:
        if budget_fn() < 25:
            break
        h_int = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        c5_int = compute_c5(h_int)
        if budget_fn() < 15:
            break
        h_new, c5_new = do_lbfgs(h_int, n_try, [50, 200, 1000, 5000], budget_fn, max_iter=400)
        if budget_fn() > 8:
            h_new, c5_new = do_proj_subgrad(h_new, n_try, budget_fn, n_iters=2000, top_k=5)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n_try
            print("  n={} IMPROVED: C5={:.10f}".format(n_try, best_c5))
        else:
            print("  n={} {:.10f} (best {:.10f})".format(n_try, c5_new, best_c5))

    # Phase 6: Simulated annealing on binary
    if budget_fn() > 30:
        print("P6: Sim anneal n={}".format(best_n))
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project_feasible(h_bin, best_n)
        c5_bin = compute_c5(h_bin)
        h_sa, c5_sa = do_sim_anneal_bin(h_bin, best_n, budget_fn, n_steps=15000)
        if c5_sa < best_c5 - 1e-13:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            print("  SA improved: C5={:.10f}".format(best_c5))
        else:
            print("  SA: {:.10f} (best {:.10f})".format(c5_sa, best_c5))

    # Phase 7: Multi-start
    if budget_fn() > 40:
        print("P7: Multi-start")
        starts = []
        starts.append(make_thue_morse(best_n))
        gr = (1.0 + np.sqrt(5.0)) / 2.0
        starts.append(make_gb(best_n, gr - 1.0))
        starts.append(make_gb(best_n, 0.618))
        for _ in range(4):
            starts.append(make_gb(best_n, rng.uniform(0.3, 0.7)))
        for si, hs in enumerate(starts):
            if budget_fn() < 15:
                break
            c5_s = compute_c5(hs)
            h_opt, c5_opt = do_lbfgs(hs, best_n, [100, 500, 3000], budget_fn, max_iter=300)
            if c5_opt < best_c5 - 1e-13:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                print("  start{}: C5={:.10f}".format(si, best_c5))

    # Phase 8: Final polish cycle
    if budget_fn() > 15:
        print("P8: Final polish")
        best_h, best_c5 = do_lbfgs(best_h, best_n, [10000, 30000, 80000], budget_fn, max_iter=200)
        print("  L-BFGS: C5={:.10f}".format(best_c5))
    if budget_fn() > 8:
        best_h, best_c5 = do_proj_subgrad(best_h, best_n, budget_fn, n_iters=2000, top_k=3)
        print("  Subgrad: C5={:.10f}".format(best_c5))
    if budget_fn() > 5:
        best_h, best_c5 = do_binary_cd(best_h, best_n, budget_fn, max_sweeps=15)
        print("  CD: C5={:.10f}".format(best_c5))
    if budget_fn() > 5:
        best_h, best_c5 = do_lbfgs(best_h, best_n, [50000, 100000, 200000], budget_fn, max_iter=100)
        print("  L-BFGS final: C5={:.10f}".format(best_c5))

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("FINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
