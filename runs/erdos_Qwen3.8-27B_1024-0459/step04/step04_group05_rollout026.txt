    if method == 'random':
        h = rng.rand(n)
        h = project_feasible(h, n)
    elif method == 'thue_morse':
        h = np.zeros(n)
        for i in range(n):
            cnt = bin(i).count('1')
            h[i] = 1.0 if cnt % 2 == 1 else 0.0
        h = project_feasible(h, n)
    elif method == 'golden':
        gr = (1.0 + np.sqrt(5.0)) / 2.0
        h = np.zeros(n)
        for i in range(n):
            frac = (i * gr) % 1.0
            h[i] = 1.0 if frac < 0.5 else 0.0
        h = project_feasible(h, n)
    else:  # block
        h = np.zeros(n)
        block_size = max(1, n // 16)
        for b in range(16):
            if b % 2 == 0:
                start = b * block_size
                end = min(start + block_size, n)
                h[start:end] = 1.0
        h = project_feasible(h, n)
    
    return h


def do_polish(h_start, n, budget_check):
    """Final polishing with very high beta SLSQP + active gradient"""
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    
    if budget_check() > 20:
        h_cur, c5_cur = do_slsqp(h_cur, n, [5000, 20000, 50000], 300)
    
    if budget_check() > 10:
        h_cur, c5_cur = do_active_gradient(h_cur, n, max_iters=200)
    
    if budget_check() > 10:
        h_cur, c5_cur = do_slsqp(h_cur, n, [100000], 200)
    
    return h_cur, c5_cur


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    np.random.seed(seed)
    
    def budget_check():
        return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = project_feasible(make_diverse_start(parent_n, 0), parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(best_h)
        best_n = parent_n
        print("No parent. Start: n={}, C5={:.10f}".format(parent_n, best_c5))
    
    # Phase 1: Quick SLSQP at parent resolution
    print("\n--- Phase 1: SLSQP n={} ---".format(best_n))
    best_h, best_c5 = do_slsqp(best_h, best_n, [50, 200, 800, 2000, 5000], 400)
    print("  C5={:.10f}".format(best_c5))
    
    # Phase 2: Active gradient at parent resolution
    if budget_check() > 15:
        print("--- Phase 2: Active grad n={} ---".format(best_n))
        best_h, best_c5 = do_active_gradient(best_h, best_n, max_iters=300)
        print("  C5={:.10f}".format(best_c5))
    
    # Phase 3: Try n=512
    if budget_check() > 100:
        print("\n--- Phase 3: n=512 ---")
        n512 = 512
        h512 = project_feasible(interp_to(best_h, best_n, n512), n512)
        c5_512 = compute_c5(h512)
        print("  Interpolated: C5={:.10f}".format(c5_512))
        
        h512, c5_512 = do_slsqp(h512, n512, [100, 500, 2000, 5000, 15000], 500)
        print("  After SLSQP: C5={:.10f}".format(c5_512))
        
        if c5_512 < best_c5 - 1e-14:
            best_c5 = c5_512
            best_h = h512.copy()
            best_n = n512
        
        if budget_check() > 30:
            h512, c5_512 = do_active_gradient(h512, n512, max_iters=400)
            print("  After active grad: C5={:.10f}".format(c5_512))
            if c5_512 < best_c5 - 1e-14:
                best_c5 = c5_512
                best_h = h512.copy()
                best_n = n512
    
    # Phase 4: Try n=384 (between 256 and 512)
    if budget_check() > 80:
        print("\n--- Phase 4: n=384 ---")
        n384 = 384
        h384 = project_feasible(interp_to(best_h, best_n, n384), n384)
        c5_384 = compute_c5(h384)
        h384, c5_384 = do_slsqp(h384, n384, [200, 1000, 5000], 400)
        if c5_384 < best_c5 - 1e-14:
            best_c5 = c5_384
            best_h = h384.copy()
            best_n = n384
            print("  n=384 improved: C5={:.10f}".format(best_c5))
        else:
            print("  n=384: C5={:.10f} (best={:.10f})".format(c5_384, best_c5))
    
    # Phase 5: Try n=768
    if budget_check() > 150:
        print("\n--- Phase 5: n=768 ---")
        n768 = 768
        h768 = project_feasible(interp_to(best_h, best_n, n768), n768)
        c5_768 = compute_c5(h768)
        print("  Interpolated: C5={:.10f}".format(c5_768))
        h768, c5_768 = do_slsqp(h768, n768, [100, 500, 2000], 300)
        print("  After SLSQP: C5={:.10f}".format(c5_768))
        if c5_768 < best_c5 - 1e-14:
            best_c5 = c5_768
            best_h = h768.copy()
            best_n = n768
        
        if budget_check() > 30:
            h768, c5_768 = do_active_gradient(h768, n768, max_iters=300)
            if c5_768 < best_c5 - 1e-14:
                best_c5 = c5_768
                best_h = h768.copy()
                best_n = n768
    
    # Phase 6: Binary relaxation at best resolution
    if budget_check() > 15:
        print("\n--- Phase 6: Binary relax n={} ---".format(best_n))
        best_h, best_c5 = do_binary_relax(best_h, best_n, budget_check)
        print("  C5={:.10f}".format(best_c5))
    
    # Phase 7: Multi-start diversification
    if budget_check() > 40:
        print("\n--- Phase 7: Multi-start ---")
        for s in range(3):
            if budget_check() < 15:
                break
            h_div = make_diverse_start(best_n, seed + 100 + s)
            # Blend with current best
            h_blend = 0.5 * best_h + 0.5 * h_div
            h_blend = project_feasible(h_blend, best_n)
            c5_blend = compute_c5(h_blend)
            
            h_try = h_blend if c5_blend < best_c5 else best_h.copy()
            c5_try = min(c5_blend, best_c5)
            
            h_try, c5_try = do_slsqp(h_try, best_n, [500, 2000, 8000], 300)
            if budget_check() > 5:
                h_try, c5_try = do_active_gradient(h_try, best_n, max_iters=150)
            
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                print("  Start {} improved: C5={:.10f}".format(s, best_c5))
    
    # Phase 8: Final polishing
    if budget_check() > 10:
        print("\n--- Phase 8: Polish ---")
        best_h, best_c5 = do_polish(best_h, best_n, budget_check)
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
