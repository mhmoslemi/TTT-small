def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(src_h, src_n, dst_n):

def slsqp_optimize(h, n, beta, maxiter=300):
    tgt = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_obj(x, n, beta, tgt)
    
    def grad(x):
        return smooth_grad(x, n, beta, tgt)
    
    def con_fun(x):
        return np.sum(x) - tgt
    
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def lgbfgs_optimize(h, n, maxiter=2000, penalty_coef=1000.0):
    """L-BFGS-B with penalty for the equality constraint."""
    tgt = n / 2.0
    
    def obj_pen(x):
        dx_val = 2.0 / n
        corr = compute_corr(x)
        m = np.max(corr)
        pen = penalty_coef * (np.sum(x) - tgt) ** 2
        return m * dx_val + pen
    
    def grad_pen(x):
        dx_val = 2.0 / n
        corr = compute_corr(x)
        m_idx = np.argmax(corr)
        # Subgradient at the max
        g = np.zeros(n)
        nn = n - 1
        k = m_idx
        off = k - nn
        if off >= 0:
            npairs = n - off
            g[off:off + npairs] += (1.0 - x[:npairs])
            g[:npairs] -= x[off:off + npairs]
        else:
            ao = -off
            npairs = n - ao
            g[:npairs] += (1.0 - x[ao:ao + npairs])
            g[ao:ao + npairs] -= x[:npairs]
        g *= dx_val
        # Penalty gradient
        g += 2.0 * penalty_coef * (np.sum(x) - tgt)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h, jac=grad_pen, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h


def topk_descent(h, n, max_iters=300, top_k=5):
    """Focus gradient descent on the top-k most restrictive lags."""
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):

        corr = compute_corr(h)

        g = np.zeros(n)
        nn = n - 1
        for k in top_idx:
            off = int(k) - nn
            if off >= 0:
                npairs = n - off
                g[off:off + npairs] += (1.0 - h[:npairs])
                g[:npairs] -= h[off:off + npairs]
            else:
                ao = -off
                npairs = n - ao
                g[:npairs] += (1.0 - h[ao:ao + npairs])
                g[ao:ao + npairs] -= h[:npairs]
        g /= top_k
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        improved = False
        for step in [0.05, 0.02, 0.01, 0.1, 0.005, 0.03]:

        t_alloc = min(35, max(10, remaining() * 0.2))
        h_sa, c5_sa = sa_smart_binary(h_bin_hi, n_hi, time_limit=t_alloc, seed=seed + 20000 + n_hi)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n_hi
    
    # Phase 6: Final intensive refinement
    if remaining() > 30:
        n_cur = best_n
        # Binary SA one more time
        h_bin_final = make_binary(best_h, n_cur, rng)
        t_alloc = min(30, max(10, remaining() * 0.3))
        h_sa, c5_sa = sa_smart_binary(h_bin_final, n_cur, time_limit=t_alloc, seed=seed + 30000)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        
        # Continuous polish
        if remaining() > 15:
            h_cont = project_feasible(best_h, n_cur)
            t_alloc = min(20, max(5, remaining() * 0.4))
            h_ref, c5_ref = continuous_refine(h_cont, n_cur, time_limit=t_alloc)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
    
    # Final top-k on the absolute best
    if remaining() > 10:
        h_td, c5_td = topk_descent(best_h, best_n, max_iters=100, top_k=3)
        if c5_td < best_c5:
            best_c5 = c5_td
            best_h = h_td.copy()
    
    # Final projection and validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
