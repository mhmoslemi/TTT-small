def smooth_obj(x):

    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m

def solve_1d_minmax(a, b, lo, hi):
    """Find t in [lo,hi] minimizing max_i (a[i]*t + b[i])."""
    if lo >= hi:
        return lo
    for _ in range(55):

        vals = a * mid + b
        max_val = vals.max()
        active = vals >= max_val - 1e-14

        if n_active == 0:

def coord_descent_1d(h, n, max_passes=50000, time_limit=200, rng=None):
    """Single-coordinate descent: for each k, find optimal h[k] given others fixed."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = np.array(h, dtype=float).copy()
    start = time.time()
    dx_val = 2.0 / n
    nc = 2 * n - 1
    corr = compute_corr(h)
    best_c5 = np.max(corr) * dx_val
    lam = 0.0
    stall = 0

    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break

        order = np.arange(n)
        if p % 7 == 6:
            order = rng.permutation(n)

        for k in range(n):
            idx = int(order[k])
            # Compute slope of each corr[i] w.r.t. h[idx]
            slopes = np.zeros(nc)
            for shift_d in range(-(n - 1), n):
                si = shift_d + (n - 1)
                s = 0.0
                kd = idx + shift_d
                if 0 <= kd < n:
                    s += (1.0 - h[kd])
                kdm = idx - shift_d
                if 0 <= kdm < n:
                    s -= h[kdm]
                slopes[si] = s

            intercepts = corr - h[idx] * slopes
            # Add Lagrange term: minimize max(slopes*t + intercepts) + lam*t
            a_vec = slopes + lam
            t_opt = solve_1d_minmax(a_vec, intercepts, 0.0, 1.0)

            if abs(t_opt - h[idx]) > 1e-10:
                # Update correlation efficiently
                corr += (t_opt - h[idx]) * slopes
                h[idx] = t_opt

        # Project to feasible
        h = project_feasible(h, n)
        corr = compute_corr(h)
        c5_new = np.max(corr) * dx_val
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            stall = 0
        else:
            stall += 1

        # Update lambda for sum constraint
        diff = np.sum(h) - n / 2.0
        lam += 0.05 * diff
        lam = np.clip(lam, -5.0, 5.0)

        if stall > 200:
            # Perturb to escape
            h = project_feasible(h + 0.01 * (rng.random(n) - 0.5), n)
            corr = compute_corr(h)
            best_c5 = min(best_c5, np.max(corr) * dx_val)
            stall = 0

    return h, best_c5


def binary_local_search(v, n, time_limit=300, rng=None, max_iters=500000):

        rng = np.random.RandomState(42)
    v = (v > 0.5).astype(float).copy()

    target_ones = n // 2
    if n_ones > target_ones:

        rem = rng.choice(ones_idx, n_ones - target_ones, replace=False)
        v[rem] = 0.0
    elif n_ones < target_ones:
        zeros_idx = np.where(v == 0)[0]
        add = rng.choice(zeros_idx, target_ones - n_ones, replace=False)

    dx_val = 2.0 / n
    corr = compute_corr(v)

    best_v = v.copy()
    start = time.time()
    stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break

        ones = np.where(v == 1)[0]
        zeros = np.where(v == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        k_max = int(np.argmax(corr))
        offset = k_max - (n - 1)
        found_improve = False

                else:
                    i = ones[rng.randint(len(ones))]
            elif offset < 0:

                if len(bad) > 0 and rng.rand() < 0.8:

                else:
                    i = ones[rng.randint(len(ones))]
            else:

            v[i] = 0.0

                cur_max = new_max
                best_c5 = cur_max * dx_val

            if i1 == i2:

            else:

                v[j1] = 1.0

                new_max = float(np.max(new_corr))
                if new_max < cur_max - 1e-12:
                    corr = new_corr
                    cur_max = new_max

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Binary local search (most promising for discrete improvements)
    if remaining() > 20:
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        if n_ones != best_n // 2:
            if n_ones > best_n // 2:
                idx = np.where(h_bin == 1)[0]
                h_bin[idx[rng.choice(len(idx), n_ones - best_n // 2, replace=False)]] = 0
            else:
                idx = np.where(h_bin == 0)[0]
                h_bin[idx[rng.choice(len(idx), best_n // 2 - n_ones, replace=False)]] = 1
        tl_bin = min(350, remaining() - 15)
        h_b, c5_b = binary_local_search(h_bin, best_n, time_limit=tl_bin, rng=rng, max_iters=800000)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 5: More coordinate descent on the refined solution
    if remaining() > 20:
        h_cd2, c5_cd2 = coord_descent_1d(best_h, best_n, max_passes=80000,
                                         time_limit=min(200, remaining() - 15), rng=rng)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    # Phase 6: Binary search again with different seed
    if remaining() > 30:
        h_bin2 = (best_h > 0.5).astype(float)
        n_ones2 = int(np.sum(h_bin2))
        if n_ones2 != best_n // 2:
            if n_ones2 > best_n // 2:
                idx = np.where(h_bin2 == 1)[0]
                h_bin2[idx[rng.choice(len(idx), n_ones2 - best_n // 2, replace=False)]] = 0
            else:
                idx = np.where(h_bin2 == 0)[0]
                h_bin2[idx[rng.choice(len(idx), best_n // 2 - n_ones2, replace=False)]] = 1
        rng2 = np.random.RandomState(seed + 1)
        tl_bin2 = min(200, remaining() - 20)
        h_b2, c5_b2 = binary_local_search(h_bin2, best_n, time_limit=tl_bin2, rng=rng2, max_iters=500000)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

    # Phase 7: Try different resolutions
    if remaining() > 40:
        n_candidates = sorted(set([
            best_n, best_n + 2, best_n - 2, best_n + 4, best_n + 8,
            320, 324, 328, 336, 344, 352
        ]))
        n_candidates = [nn for nn in n_candidates if 200 <= nn <= 600 and nn % 2 == 0]
        n_candidates = list(dict.fromkeys(n_candidates))  # dedup preserving order

        for n_t in n_candidates:
            if remaining() < 35:
                break
            if n_t == best_n:
                continue
            h_interp = interp_to(best_h, best_n, n_t)
            h_feas = project_feasible(h_interp, n_t)

            # Quick SLSQP
            h_opt = slsqp_smooth(h_feas, n_t, 3000, maxiter=200)

            # Quick coord descent
            if remaining() > 20:
                h_opt, _ = coord_descent_1d(h_opt, n_t, max_passes=20000,
                                            time_limit=min(30, remaining() - 15), rng=rng)

            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_t

    # Phase 8: Final SLSQP polish
    if remaining() > 8:
        h_new = slsqp_smooth(best_h, best_n, 20000, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 9: Final coordinate descent
    if remaining() > 10:
        h_cd3, c5_cd3 = coord_descent_1d(best_h, best_n, max_passes=50000,
                                         time_limit=min(60, remaining() - 8), rng=rng)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3.copy()

    # Phase 10: Final binary search
    if remaining() > 12:
        h_bin_f = (best_h > 0.5).astype(float)
        n_ones_f = int(np.sum(h_bin_f))
        if n_ones_f != best_n // 2:
            if n_ones_f > best_n // 2:
                idx = np.where(h_bin_f == 1)[0]
                h_bin_f[idx[rng.choice(len(idx), n_ones_f - best_n // 2, replace=False)]] = 0
            else:
                idx = np.where(h_bin_f == 0)[0]
                h_bin_f[idx[rng.choice(len(idx), best_n // 2 - n_ones_f, replace=False)]] = 1
        h_bf, c5_bf = binary_local_search(h_bin_f, best_n,
                                          time_limit=min(30, remaining() - 5), rng=rng, max_iters=200000)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

    # Validate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
