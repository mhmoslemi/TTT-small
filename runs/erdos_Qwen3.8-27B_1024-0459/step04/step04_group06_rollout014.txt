    c5_best = compute_c5(h)
    h_best = h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum}]

    for p in range(num_pert):
        if remaining_time() < 8.0:
            break
        trial = h_best.copy()
        pt = p % 5
        if pt == 0:
            start = rng.randint(0, max(1, n - 16))
            L = rng.randint(4, min(16, n // 5))
            trial[start:start + L] = 1.0 - trial[start:start + L]
        elif pt == 1:
            trial += 0.04 * rng.randn(n)
        elif pt == 2:
            s1 = rng.randint(0, n // 4)
            s2 = s1 + rng.randint(n // 5, n - n // 5)
            L = rng.randint(3, min(8, n // 15))
            a = trial[s1:s1 + L].copy()
            trial[s1:s1 + L] = trial[s2:s2 + L]
            trial[s2:s2 + L] = a
        elif pt == 3:
            freq = rng.randint(2, n // 4)
            phase = rng.rand() * 2 * np.pi
            t = np.arange(n) / float(n)
            trial += 0.05 * np.sin(2.0 * np.pi * freq * t + phase)
        else:
            idx = rng.choice(n, size=n // 8, replace=False)
            trial[idx] = 1.0 - trial[idx]

        trial = project_feasible(trial, n)
        c5_t = compute_c5(trial)
        if c5_t < c5_best - 1e-10:
            h_best = trial.copy()
            c5_best = c5_t

        # Quick SLSQP from this perturbation
        if remaining_time() > 10.0:
            _G_BETA = 500.0
            try:
                res = minimize(obj_smooth, trial, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 60, 'ftol': 1e-13, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-12:
                    c5_best = c5_new
                    h_best = h_new
            except Exception:
                pass
    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Try to use parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = project_feasible(make_thue_morse(parent_n), parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("Thue-Morse: n={}, C5={:.10f}".format(parent_n, best_c5))

    # Phase 1: Quick warmup at parent resolution
    print("\n--- Phase 1: Warmup at n={} ---".format(parent_n))
    _G_N = parent_n
    _G_TGT = parent_n / 2.0
    _G_DX = 2.0 / parent_n
    best_h, best_c5 = accelerated_subgrad(best_h, parent_n, max_iter=200,
                                           top_k_schedule=[1, 1, 2, 3, 5],
                                           step0=0.1, momentum=0.9)
    print("  C5={:.10f}".format(best_c5))

    # Phase 2: Move to n=512 for more freedom
    print("\n--- Phase 2: Interpolate to n=512 ---")
    n512 = 512
    h512 = project_feasible(interp_to(best_h, best_n, n512), n512)
    c5_512 = compute_c5(h512)
    print("  Interpolated C5={:.10f}".format(c5_512))

    if c5_512 < best_c5 + 0.01:  # Proceed if not too bad
        best_h = h512
        best_c5 = min(best_c5, c5_512)
        best_n = n512
    else:
        print("  Interpolation too lossy, keeping n={}".format(best_n))
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    # Phase 3: Main optimization at n=512
    print("\n--- Phase 3: Main accelerated subgradient at n={} ---".format(best_n))
    _G_N = best_n
    _G_TGT = best_n / 2.0
    _G_DX = 2.0 / best_n
    best_h, best_c5 = accelerated_subgrad(
        best_h, best_n, max_iter=800,
        top_k_schedule=[1, 1, 1, 2, 2, 2, 3, 3, 5, 5, 8, 8, 12, 15, 20, 30],
        step0=0.12, momentum=0.9)
    print("  C5={:.10f}".format(best_c5))

    # Phase 4: SLSQP polish
    print("\n--- Phase 4: SLSQP polish ---")
    best_h, best_c5 = slsqp_polish(best_h, best_n, betas=[150, 400, 1000, 3000], maxiter=250)
    print("  C5={:.10f}".format(best_c5))

    # Phase 5: More subgradient with refined schedule
    print("\n--- Phase 5: Focused subgradient round 2 ---")
    best_h, best_c5 = accelerated_subgrad(
        best_h, best_n, max_iter=500,
        top_k_schedule=[1, 1, 2, 2, 3, 3, 5, 5, 8, 10],
        step0=0.08, momentum=0.85)
    print("  C5={:.10f}".format(best_c5))

    # Phase 6: SLSQP again
    if remaining_time() > 30.0:
        print("\n--- Phase 6: SLSQP round 2 ---")
        best_h, best_c5 = slsqp_polish(best_h, best_n, betas=[500, 1500, 5000], maxiter=200)
        print("  C5={:.10f}".format(best_c5))

    # Phase 7: Binary local search
    if remaining_time() > 40.0:
        print("\n--- Phase 7: Binary local search ---")
        best_h, best_c5 = binary_local_search(best_h, best_n, num_swaps=400, rng=rng)
        print("  C5={:.10f}".format(best_c5))

    # Phase 8: Perturbation exploration
    if remaining_time() > 40.0:
        print("\n--- Phase 8: Perturbation exploration ---")
        best_h, best_c5 = perturbation_round(best_h, best_n, rng, num_pert=12)
        print("  C5={:.10f}".format(best_c5))

    # Phase 9: Subgradient + SLSQP cycle
    if remaining_time() > 50.0:
        print("\n--- Phase 9: Cycle 2 ---")
        best_h, best_c5 = accelerated_subgrad(
            best_h, best_n, max_iter=300,
            top_k_schedule=[1, 1, 2, 3, 3, 5, 5, 8],
            step0=0.06, momentum=0.9)
        best_h, best_c5 = slsqp_polish(best_h, best_n, betas=[1000, 3000, 8000], maxiter=200)
        print("  C5={:.10f}".format(best_c5))

    # Phase 10: Try n=1024 if time allows
    if remaining_time() > 120.0:
        print("\n--- Phase 10: Attempt n=1024 ---")
        n1024 = 1024
        h1024 = project_feasible(interp_to(best_h, best_n, n1024), n1024)
        c5_1024 = compute_c5(h1024)
        print("  Interpolated C5={:.10f}".format(c5_1024))
        _G_N = n1024
        _G_TGT = n1024 / 2.0
        _G_DX = 2.0 / n1024
        h1024, c5_1024 = accelerated_subgrad(
            h1024, n1024, max_iter=300,
            top_k_schedule=[1, 1, 2, 3, 5, 8],
            step0=0.08, momentum=0.9)
        if remaining_time() > 60.0:
            h1024, c5_1024 = slsqp_polish(h1024, n1024, betas=[200, 800, 2000], maxiter=150)
        if c5_1024 < best_c5 - 1e-10:
            best_c5 = c5_1024
            best_h = h1024.copy()
            best_n = n1024
            print("  Improved! n=1024: C5={:.10f}".format(c5_1024))
        else:
            print("  n=1024: C5={:.10f} (no improvement, keeping n={})".format(c5_1024, best_n))

    # Phase 11: Also try n=384 as alternative
    if remaining_time() > 60.0 and best_n != 384:
        print("\n--- Phase 11: Try n=384 ---")
        n384 = 384
        h384 = project_feasible(interp_to(best_h, best_n, n384), n384)
        _G_N = n384
        _G_TGT = n384 / 2.0
        _G_DX = 2.0 / n384
        c5_384 = compute_c5(h384)
        h384, c5_384 = accelerated_subgrad(
            h384, n384, max_iter=300,
            top_k_schedule=[1, 1, 2, 3, 5, 8, 10],
            step0=0.1, momentum=0.9)
        if remaining_time() > 30.0:
            h384, c5_384 = slsqp_polish(h384, n384, betas=[300, 800, 2000], maxiter=150)
        if c5_384 < best_c5 - 1e-10:
            best_c5 = c5_384
            best_h = h384.copy()
            best_n = n384
            print("  n=384: C5={:.10f}".format(c5_384))
        else:
            print("  n=384: C5={:.10f} (no improvement)".format(c5_384))

    # Phase 12: Coordinate descent
    if remaining_time() > 15.0:
        print("\n--- Phase 12: Coordinate descent ---")
        _G_N = best_n
        _G_TGT = best_n / 2.0
        _G_DX = 2.0 / best_n
        best_h, best_c5 = coord_descent(best_h, best_n, max_sweeps=12, rng=rng)
        print("  C5={:.10f}".format(best_c5))

    # Phase 13: Final SLSQP with very high beta
    if remaining_time() > 12.0:
        print("\n--- Phase 13: Final high-beta SLSQP ---")
        _G_N = best_n
        _G_TGT = best_n / 2.0
        _G_DX = 2.0 / best_n
        best_h, best_c5 = slsqp_polish(best_h, best_n, betas=[5000, 10000, 20000], maxiter=200)
        print("  C5={:.10f}".format(best_c5))

    # Phase 14: Tiny final subgradient
    if remaining_time() > 8.0:
        print("\n--- Phase 14: Final micro-optimization ---")
        best_h, best_c5 = accelerated_subgrad(
            best_h, best_n, max_iter=100,
            top_k_schedule=[1, 1, 2, 3],
            step0=0.03, momentum=0.8)
        best_h, best_c5 = coord_descent(best_h, best_n, max_sweeps=4, rng=rng)
        print("  C5={:.10f}".format(best_c5))

    # Final validation and return
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
