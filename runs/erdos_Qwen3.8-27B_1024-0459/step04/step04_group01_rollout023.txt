```python
def compute_jacobian(h, n):
    nc = 2 * n - 1
    J = np.zeros((n, nc))
    # e = 0, k = n-1
    J[:, n-1] = 1.0 - 2.0 * h
    # Positive shifts e = 1, ..., n-1
    for e in range(1, n):
        k = n - 1 + e
        J[:n-e, k] += (1.0 - h[e:])       # (1-h[j+e]) for j = 0..n-1-e
        J[e:, k] -= h[:-e]                  # -h[j-e] for j = e..n-1
    # Negative shifts m = 1, ..., n-1, k = n-1-m
    for m in range(1, n):
        k = n - 1 - m
        J[m:, k] += (1.0 - h[:-m])         # (1-h[j-m]) for j = m..n-1
        J[:n-m, k] -= h[m:]                 # -h[j+m] for j = 0..n-1-m
    return J
```

```python
        worst_shifts = sorted_idx[:top_k_shifts]

        for k_idx in worst_shifts:
            e = int(k_idx) - (n - 1)  # shift value
            if e == 0:

                        h[i] = t_opt
                        h[j] = s_val - t_opt
            else:
                m = -e
                # Pairs (i+m, i) for i = 0, ..., n-1-m, i.e., (j, j-m) for j = m, ..., n-1
```

```python
from scipy.optimize import minimize
import time

_G_TGT = 0.0


def _remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 5.0)


def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_t = np.clip(h - mid, 0.0, 1.0)
        s = h_t.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(60):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _compute_jacobian(h, n):
    nc = 2 * n - 1
    J = np.zeros((n, nc))
    J[:, n - 1] = 1.0 - 2.0 * h
    for e in range(1, n):
        k = n - 1 + e
        J[:n - e, k] += (1.0 - h[e:])
        J[e:, k] -= h[:-e]
    for m in range(1, n):
        k = n - 1 - m
        J[m:, k] += (1.0 - h[:-m])
        J[:n - m, k] -= h[m:]
    return J


def _solve_1d_max(slopes, intercepts, lo, hi):
    if lo >= hi:
        return lo
    for _ in range(45):
        mid = (lo + hi) / 2.0
        vals = slopes * mid + intercepts
        max_val = vals.max()
        active = vals >= max_val - 1e-12
        smax = slopes[active].max()
        smin = slopes[active].min()
        if smax < -1e-14:
            lo = mid
        elif smin > 1e-14:
            hi = mid
        else:
            break
    return (lo + hi) / 2.0


def _interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xq = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xq, xp, src_h)


def _slsqp_fun(x):
    n = _G_N
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    shifted = corr - m
    cl = np.clip(_G_BETA * shifted, -60, 60)
    return (m + np.log(np.sum(np.exp(cl))) / _G_BETA) * (2.0 / n)


def _slsqp_grad(x):
    n = _G_N
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = corr.max()
    shifted = corr - m
    cl = np.clip(_G_BETA * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / ex.sum()
    J = _compute_jacobian(x, n)
    return (w @ J) * (2.0 / n)


def _slsqp_con_fun(x):
    return x.sum() - _G_TGT


def _slsqp_con_jac(x):
    return np.ones(_G_N)


def _slsqp_refine(h, n, beta, maxiter):
    global _G_N, _G_BETA, _G_TGT
    _G_N = n
    _G_BETA = float(beta)
    _G_TGT = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con_fun, 'jac': _slsqp_con_jac}]
    try:
        res = minimize(_slsqp_fun, h, jac=_slsqp_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        return _project_feasible(res.x, n)
    except Exception:
        return h

        sorted_idx = np.argsort(-corr)

                t_opt = _solve_1d_max(diff, intercepts, lo_t, hi_t)
                delta = t_opt - h[a]
                if abs(delta) > 1e-14:
                    corr += delta * diff
                    h[a] = t_opt
                    h[b] = s_val - t_opt
        else:
            done_pairs = set()
            for k_idx in worst_shifts:
                e = int(k_idx) - (n - 1)
                abs_e = abs(e)
                if abs_e == 0 or abs_e >= n:
                    continue
                for i in range(n - abs_e):
                    j = i + abs_e
                    key = (i, j)
                    if key in done_pairs:
                        continue
                    done_pairs.add(key)
                    s_val = h[i] + h[j]
                    lo_t = max(0.0, s_val - 1.0)
                    hi_t = min(1.0, s_val)
                    if lo_t >= hi_t - 1e-14:
                        continue
                    diff = J[i] - J[j]
                    intercepts = corr - h[i] * diff
                    t_opt = _solve_1d_max(diff, intercepts, lo_t, hi_t)
                    delta = t_opt - h[i]
                    if abs(delta) > 1e-14:
                        corr += delta * diff
                        h[i] = t_opt
                        h[j] = s_val - t_opt

        h = _project_feasible(h, n)
        new_corr = _compute_corr(h)
        c5_new = float(new_corr.max() * 2.0 / n)
        if c5_new < best_c5:
            best_c5 = c5_new
        corr = new_corr

    return h, best_c5


def _momentum_subgrad(h, n, max_iters, time_limit, top_k=5):
    h = h.copy()
    start = time.time()
    best_c5 = _compute_c5(h)
    best_h = h.copy()
    momentum = np.zeros(n)
    beta_m = 0.88
    step = 0.04

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if _remaining() < 10:
            break

        corr = _compute_corr(h)
        m = corr.max()
        sorted_idx = np.argsort(-corr)
        top = sorted_idx[:top_k]
        diffs = m - corr[top]
        weights = np.exp(-diffs * 80.0)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(top_k) / top_k
        else:
            weights /= wsum

        J = _compute_jacobian(h, n)
        g = weights @ J

        momentum = beta_m * momentum + g
        h_new = h - step * momentum
        h_new = _project_feasible(h_new, n)

        c5_new = _compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            if it % 500 == 0 and it > 0:
                step = min(step * 1.015, 0.15)
        else:
            step *= 0.995

        h = h_new

    return best_h, best_c5


def _active_equalize(h, n, max_iters, top_k=5):
    h = h.copy()
    best_c5 = _compute_c5(h)
    dx_val = 2.0 / n

    for it in range(max_iters):
        if _remaining() < 5:
            break
        corr = _compute_corr(h)
        top = np.argsort(-corr)[:top_k]
        c_max = corr[top[0]]
        c_min = corr[top[-1]]
        if c_max - c_min < 1e-12:
            break

        J = _compute_jacobian(h, n)
        g_max = J[:, int(top[0])]
        g_min = J[:, int(top[-1])]
        for mid_idx in top[1:-1]:
            g_max = g_max + J[:, int(mid_idx)]
            g_min = g_min + J[:, int(mid_idx)]
        d = g_min - g_max
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-14:
            break
        d = d / dnorm

        improved = False
        for step in [0.003, 0.007, 0.015, 0.03, 0.001, 0.005, 0.06, 0.01, 0.04]:
            h_t = _project_feasible(h + step * d, n)
            c_new = _compute_c5(h_t)
            if c_new < best_c5 - 1e-14:
                h = h_t
                best_c5 = c_new
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def _binary_local_search(h, n, time_limit, rng):
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(h_bin.sum())
    target_ones = n // 2
    if n_ones > target_ones:
        ones_idx = np.where(h_bin == 1)[0]
        rem = rng.choice(ones_idx, n_ones - target_ones, replace=False)
        h_bin[rem] = 0.0
    elif n_ones < target_ones:
        zeros_idx = np.where(h_bin == 0)[0]
        add = rng.choice(zeros_idx, target_ones - n_ones, replace=False)
        h_bin[add] = 1.0

    dx_val = 2.0 / n
    corr = _compute_corr(h_bin)
    cur_max = float(corr.max())
    best_c5 = cur_max * dx_val
    best_h = h_bin.copy()
    start = time.time()
    stall = 0

    for it in range(300000):
        if time.time() - start > time_limit:
            break
        if _remaining() < 5:
            break

        k_worst = int(np.argmax(corr))
        offset = k_worst - (n - 1)
        abs_off = abs(offset)
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        found = False
        n_try = 150
        for _ in range(n_try):
            if abs_off < n:
                if offset > 0:
                    cands = np.array([i for i in ones if i + abs_off < n and h_bin[i + abs_off] == 0])
                else:
                    cands = np.array([i for i in ones if i - abs_off >= 0 and h_bin[i - abs_off] == 0])
                if len(cands) > 0 and rng.rand() < 0.7:
                    i = cands[rng.randint(len(cands))]
                else:
                    i = ones[rng.randint(len(ones))]
            else:
                i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            new_corr = _compute_corr(h_bin)
            new_max = float(new_corr.max())
            if new_max < cur_max - 1e-12:
                corr = new_corr
                cur_max = new_max
                best_c5 = cur_max * dx_val
                best_h = h_bin.copy()
                found = True
                break
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0

        if found:
            stall = 0
        else:
            stall += 1
            if stall > 80:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                corr = _compute_corr(h_bin)
                cur_max = float(corr.max())
                stall = 0

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = _compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 312
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        best_h = parent_h.copy()
        best_c5 = _compute_c5(parent_h)
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP warmup
    for beta in [200, 1000, 3000]:
        if _remaining() < 10:
            break
        h_new = _slsqp_refine(h, n, beta, maxiter=300)
        c5_new = _compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Shift-targeted coordinate descent (intensive)
    if _remaining() > 30:
        tlim = min(80, _remaining() - 20)
        h_new, c5_new = _shift_targeted_cd(h, n, max_passes=800, time_limit=tlim, top_k=3, rng=rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = best_h.copy()

    # Phase 3: Momentum subgradient
    if _remaining() > 30:
        tlim = min(50, _remaining() - 20)
        h_new, c5_new = _momentum_subgrad(h, n, max_iters=20000, time_limit=tlim, top_k=5)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = best_h.copy()

    # Phase 4: More coordinate descent
    if _remaining() > 20:
        tlim = min(60, _remaining() - 15)
        h_new, c5_new = _shift_targeted_cd(h, n, max_passes=600, time_limit=tlim, top_k=5, rng=rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = best_h.copy()

    # Phase 5: High-precision SLSQP
    for beta in [8000, 20000, 50000, 100000]:
        if _remaining() < 8:
            break
        h_new = _slsqp_refine(h, n, beta, maxiter=500)
        c5_new = _compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = best_h.copy()

    # Phase 6: Active set equalization
    if _remaining() > 10:
        h_new, c5_new = _active_equalize(h, n, max_iters=300, top_k=5)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = best_h.copy()

    # Phase 7: Another round of coord descent + SLSQP
    if _remaining() > 40:
        tlim = min(50, _remaining() - 20)
        h_new, c5_new = _shift_targeted_cd(h, n, max_passes=500, time_limit=tlim, top_k=4, rng=rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = best_h.copy()
        if _remaining() > 10:
            h_new = _slsqp_refine(h, n, 50000, maxiter=400)
            c5_new = _compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = best_h.copy()

    # Phase 8: Multi-n exploration
    n_candidates = []
    for dn in [-16, -8, -4, -2, 2, 4, 8, 16, 32]:
        nn = best_n + dn
        if 80 <= nn <= 800 and nn % 2 == 0 and nn != best_n:
            n_candidates.append(nn)
    n_candidates = sorted(set(n_candidates))

    for n_t in n_candidates:
        if _remaining() < 40:
            break
        h_interp = _interp_to(best_h, best_n, n_t)
        h_feas = _project_feasible(h_interp, n_t)
        h_opt = h_feas

        for beta in [1000, 5000, 20000]:
            if _remaining() < 8:
                break
            h_opt = _slsqp_refine(h_opt, n_t, beta, maxiter=300)

        if _remaining() > 20:
            h_opt, _ = _shift_targeted_cd(h_opt, n_t, max_passes=300,
                                           time_limit=min(30, _remaining() - 15),
                                           top_k=3, rng=rng)

        c5_opt = _compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 9: Multi-resolution restart (try different basins)
    if _remaining() > 60:
        for coarse_n in [80, 120, 160]:
            if _remaining() < 40:
                break
            h_coarse = _interp_to(best_h, best_n, coarse_n)
            h_coarse = _project_feasible(h_coarse, coarse_n)
            h_c_opt = _slsqp_refine(h_coarse, coarse_n, 2000, maxiter=200)
            h_up = _interp_to(h_c_opt, coarse_n, best_n)
            h_up = _project_feasible(h_up, best_n)
            h_up = _slsqp_refine(h_up, best_n, 10000, maxiter=300)
            if _remaining() > 20:
                h_up, _ = _shift_targeted_cd(h_up, best_n, max_passes=200,
                                              time_limit=20, top_k=3, rng=rng)
            c5_up = _compute_c5(h_up)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = h_up.copy()

    # Phase 10: Binary local search (if time permits)
    if _remaining() > 20:
        tlim = min(25, _remaining() - 10)
        h_b, c5_b = _binary_local_search(best_h, best_n, time_limit=tlim, rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 11: Final SLSQP polish
    if _remaining() > 8:
        h_new = _slsqp_refine(best_h, best_n, 80000, maxiter=600)
        c5_new = _compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 12: Final coordinate descent
    if _remaining() > 15:
        tlim = min(30, _remaining() - 8)
        h_new, c5_new = _shift_targeted_cd(best_h, best_n, max_passes=400,
                                            time_limit=tlim, top_k=5, rng=rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final SLSQP
    if _remaining() > 5:
        h_new = _slsqp_refine(best_h, best_n, 100000, maxiter=400)
        c5_new = _compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Validate
    best_h = _project_feasible(best_h, best_n)
    final_c5 = _compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
```
