        # Average gradient of top correlations

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)


def _sm_obj(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / _BETA) * dx_v


def _sm_grad(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    wc = w * dx_v
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:ao + np_])
            g[ao:ao + np_] -= wk * x[:np_]
    return g


def _sc(x):
    return np.sum(x) - _TGT


def _scj(x):
    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=500):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.success or res.fun < _sm_obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def binarize(h, n):
    h = np.array(h, dtype=float)
    n_half = n // 2
    sort_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sort_idx[:n_half]] = 1.0
    return h_bin


def thue_morse(n, rng=None):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            if rng is not None:
                rem = rng.choice(ones_idx, n_ones - n_half, replace=False)
            else:
                rem = ones_idx[:n_ones - n_half]
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            if rng is not None:
                add = rng.choice(zeros_idx, n_half - n_ones, replace=False)
            else:
                add = zeros_idx[:n_half - n_ones]
            h[add] = 1
    return h


def corr_gradient_at(h, k_off, n):
    """Gradient of correlation at a specific offset."""
    g = np.zeros(n)
    if k_off >= 0:
        np_ = n - k_off
        g[k_off:k_off + np_] += (1.0 - h[:np_])
        g[:np_] -= h[k_off:k_off + np_]
    else:
        ao = -k_off
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def top_shifts_focus(h, n, top_k=5, max_iters=300):
    h = h.copy()
    dx_v = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min_top = corr[idx[-1]]
        if c_max - c_min_top < 1e-9:
            break
        k_top = idx[0]
        k_bot = idx[-1]
        off_t = k_top - (n - 1)
        off_b = k_bot - (n - 1)
        g_t = corr_gradient_at(h, off_t, n)
        g_b = corr_gradient_at(h, off_b, n)
        d = g_b - g_t
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            d = -g_t
            dn = np.linalg.norm(d)
            if dn < 1e-14:
                break
        d /= dn
        improved = False
        for step in [0.02, 0.05, 0.01, 0.08, 0.005, 0.03, 0.1, 0.015, 0.5, 0.2]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def binary_local_search(h_bin, n, rng, time_limit=60):
    """Intensive local search on binary vectors targeting worst correlations."""
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            np.random.seed(rng.randint(1000000))
            perm = np.random.permutation(len(ones_idx))
            rem = ones_idx[perm[:n_ones - n_half]]
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            np.random.seed(rng.randint(1000000))
            perm = np.random.permutation(len(zeros_idx))
            add = zeros_idx[perm[:n_half - n_ones]]
            h[add] = 1
    dx_v = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    start = time.time()
    no_improve = 0
    max_no_improve = 5000

    while time.time() - start < time_limit and remaining() > 5:
        no_improve += 1
        if no_improve > max_no_improve:
            break

        corr = compute_corr(h)
        max_val = np.max(corr)
        # Focus on top shifts
        top_k = min(5, len(corr))
        top_idx = np.argsort(-corr)[:top_k]

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Generate candidate swaps focused on reducing worst correlations
        best_delta = 0
        best_swap = None
        n_candidates = 12

        for _ in range(n_candidates):
            # Pick a one from a "hot" region and a zero
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0
            h[j] = 1
            nc = np.max(compute_corr(h))
            delta = nc - max_val
            h[i] = 1
            h[j] = 0
            if delta < best_delta - 1e-14:
                best_delta = delta
                best_swap = (i, j)

        # Also try focused swaps near hot correlations
        for _ in range(6):
            k_hot = top_idx[rng.randint(len(top_idx))]
            off = k_hot - (n - 1)
            if off >= 0:
                # h[i]=1, h[i+off]=0 is a "fault"
                faults = np.where((h[:n-off] == 1) & (h[off:] == 0))[0]
            else:
                ao = -off
                faults = np.where((h[ao:] == 1) & (h[:n-ao] == 0))[0]

            if len(faults) == 0:
                continue
            # Pick a fault to fix
            fi = faults[rng.randint(len(faults))]
            if off >= 0:
                i = fi  # h[i]=1, h[i+off]=0

            else:

                continue
            j = j_pool[rng.randint(len(j_pool))]
            if j == i:
                continue

            h[i] = 0
            h[j] = 1
            nc = np.max(compute_corr(h))
            delta = nc - max_val
            h[i] = 1
            h[j] = 0
            if delta < best_delta - 1e-14:
                best_delta = delta
                best_swap = (i, j)

        if best_swap is not None:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            cur_max += best_delta
            no_improve = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            # No improving swap found - try a random perturbation to escape
            if no_improve > 200:
                # Flip 2-3 random pairs to escape local minimum
                n_flip = rng.randint(2, 4)
                ones_now = np.where(h == 1)[0]
                zeros_now = np.where(h == 0)[0]
                for _ in range(n_flip):

                        break
                    i = ones_now[rng.randint(len(ones_now))]
                    j = zeros_now[rng.randint(len(zeros_now))]
                    h[i] = 0
                    h[j] = 1
                    ones_now = np.where(h == 1)[0]
                    zeros_now = np.where(h == 0)[0]
                cur_max = float(np.max(compute_corr(h)))

    return best_h, best_max * dx_v


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def subgradient_max(h, n, n_iter=200):
    """Direct subgradient descent on the max correlation."""
    h = project_feasible(h, n).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.05
    for it in range(n_iter):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        m = np.max(corr)
        # Subgradient: average gradient of top correlations
        threshold = m - 0.5
        top_mask = corr >= threshold
        top_idx = np.where(top_mask)[0]
        if len(top_idx) == 0:
            top_idx = [np.argmax(corr)]

        g = np.zeros(n)
        for k in top_idx:
            off = k - (n - 1)
            g += corr_gradient_at(h, off, n)
        g /= len(top_idx)

        # Project gradient to tangent space of sum constraint
        g -= np.mean(g)
        # Also clip to [0,1] direction
        g = np.where(h > 0.99, -np.abs(g), g)
        g = np.where(h < 0.01, np.abs(g), g)

        h_new = project_feasible(h - step * g, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            step *= 0.9
            if step < 0.001:
                break

    return best_h, best_c5
