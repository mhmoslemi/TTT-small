    # ∂c_k/∂h[i] = 1[i+s_k ∈ [0,n-1]] * (1-h[i+s_k]) - 1[i-s_k ∈ [0,n-1]] * h[i-s_k]

def sqp_step(h, n, active_lags, trust_radius=0.1):
    """One SQP iteration: solve LP subproblem."""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    nlags = len(corr)
    
    # Compute gradients for active lags
    # ∂c_k/∂h[i] = (1-h[i+s_k])*1[valid] - h[i-s_k]*1[valid]
    nl_active = len(active_lags)
    
    # A matrix: nl_active * (n+1) for the LHS of the LP
    # Variables: Δ[0..n-1], t
    # Constraints: for each active k: sum_i (∂c_k/∂h[i]) * Δ[i] + (1-dx*corr[k]) * t ≤ 0
    #             Actually: c_k(h) + sum_i g_ki * Δ_i ≤ t
    #             => sum_i g_ki * Δ_i - t ≤ -c_k(h)*dx
    #             (working in the scaled objective: c_k * dx)
    
    # Also: sum(Δ) = 0
    #       -trust_radius ≤ Δ_i ≤ trust_radius
    #       0 ≤ h_i + Δ_i ≤ 1 => -h_i ≤ Δ_i ≤ 1-h_i
    
    # LP: min t
    # s.t. for each k: sum_i g_ki * Δ_i - t ≤ -c_k(h)*dx
    #      sum_i Δ_i = 0
    #      max(-h_i, -trust_radius) ≤ Δ_i ≤ min(1-h_i, trust_radius)
    
    from scipy.optimize import linprog
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x ≤ b_ub
    # For each active lag k:
    #   sum_i g_ki * Δ_i - t ≤ -dx * corr[k]
    A_ub = np.zeros((nl_active, n+1))
    b_ub = np.zeros(nl_active)
    
    shifts = active_lags - (n-1)
    for idx, k in enumerate(active_lags):
        s = shifts[idx]
        for i in range(n):
            g = 0.0
            if 0 <= i+s < n:
                g += (1.0 - h[i+s])
            if 0 <= i-s < n:
                g -= h[i-s]
            A_ub[idx, i] = g * dx
        A_ub[idx, n] = -1.0
        b_ub[idx] = -dx * corr[k]
    
    # Equality constraint: sum(Δ) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    lb = np.maximum(-h, -trust_radius)
    ub = np.minimum(1.0 - h, trust_radius)
    bounds = list(zip(lb, ub)) + [(0, None)]  # t is unbounded above
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        delta = res.x[:n]
        h_new = project_feasible(h + delta, n)
        return h_new, res.x[n]
    return h, None

    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = float(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12 and n > 0:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def corr_gradient(h, k, n):
    """Compute ∂c_k/∂h for a single lag k."""
    s = k - (n - 1)
    g = np.zeros(n)
    lo = max(0, -s)
    hi = min(n-1, n-1-s)
    if lo <= hi:
        idx = np.arange(lo, hi+1)
        g[idx] += (1.0 - h[idx + s])
    lo2 = max(0, s)
    hi2 = min(n-1, n-1+s)
    if lo2 <= hi2:
        idx2 = np.arange(lo2, hi2+1)
        g[idx2] -= h[idx2 - s]
    return g

def sqp_optimize(h, n, max_iters=300, trust_radius=0.05, n_active=10):
    """SQP with LP subproblems for the Erdos problem."""
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        nlags = len(corr)
        c5 = np.max(corr) * dx
        
        # Select active lags
        sort_idx = np.argsort(corr)[::-1]
        # Include top n_active and also the current max
        active = sort_idx[:n_active]
        # Also add a few lags near the max that are close
        threshold = corr[sort_idx[0]] - 0.3
        near_active = np.where(corr >= threshold)[0]
        active = np.unique(np.concatenate([active, near_active[:20]]))
        active = active[np.argsort(active)]
        active = active[:30]  # limit to 30 active constraints
        
        # Solve LP
        nl_a = len(active)
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        A_ub = np.zeros((nl_a, n + 1))
        b_ub = np.zeros(nl_a)
        
        for idx, k in enumerate(active):
            g = corr_gradient(h, k, n)
            A_ub[idx, :n] = dx * g
            A_ub[idx, n] = -1.0
            b_ub[idx] = -dx * corr[k]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        
        lb = np.maximum(-h, -trust_radius)
        ub = np.minimum(1.0 - h, trust_radius)
        bounds = list(zip(lb, ub)) + [(0.0, 2.0)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success:
                delta = res.x[:n]
                h_new = project_feasible(h + delta, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
                # Adaptive trust radius
                step_size = np.max(np.abs(delta))
                if c5_new < c5 - 1e-10:
                    trust_radius = min(0.2, trust_radius * 1.5)
                else:
                    trust_radius = max(0.01, trust_radius * 0.7)
            else:
                trust_radius *= 0.5
        except Exception:
            trust_radius *= 0.5
    
    return best_h, best_c5

Let me also think about the constraint structure. The constraint sum(Δ) = 0 ensures that sum(h+Δ) = sum(h) = n/2. Good.

But there's a subtlety: the linearization c_k(h) + Σ_i g_ki * Δ_i is an approximation. The actual c_k(h+Δ) = c_k(h) + Σ_i g_ki * Δ_i + 0.5 * Δ^T H_k Δ where H_k is the Hessian of c_k. The Hessian terms make the actual objective slightly different from the LP prediction.

For the "inhomogeneous" quadratic: c_k(h) = Σ_i h[i](1-h[i+s]) = Σ_i h[i] - Σ_i h[i]h[i+s] = n/2 - Σ_i h[i]h[i+s] (using sum h = n/2... wait, that's only in the continuous case. In the discrete case, sum h = n/2 is the constraint, and c_k = Σ_i h[i](1-h[i+s_k]) where the sum is over valid i).

Actually, c_k(h) is a quadratic form: c_k = Σ_i h[i] - Σ_i h[i]h[i+s_k]. The Hessian w.r.t. h is:
H_k[i,j] = -1 if j = i+s_k (i.e., the cross term)
H_k[j,i] = -1 if i = j+s_k
Diagonal: H_k[i,i] = -1 if i+s_k = i, i.e., s_k = 0 (only for the zero-lag term)

So the Hessian is sparse! For a given lag k with shift s_k, the Hessian has -1 at positions (i, i+s_k) and (i+s_k, i) for valid i.

This means the nonlinearity is at most O(δ²) where δ is the step size. With δ=0.05, the error is about 0.0025 per pair, times the number of pairs (~n), divided by... actually, the total error in c_k is at most n * δ² / 2 ≈ 256 * 0.0025 / 2 ≈ 0.32 in units of the unscaled correlation. Scaled by dx = 2/256, that's about 0.0025 in the objective. That's significant!

So I should use a smaller trust radius, like δ=0.02 or even δ=0.01. But then the LP steps are tiny and convergence is slow.

Alternative: use a **step-doubling** approach. Solve the LP with trust radius δ, get Δ. Then do a line search: try h + λΔ for λ = 1, 0.5, 0.25, ... and pick the first one that improves.
