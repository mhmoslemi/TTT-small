3. **Better cooling**: Start higher, cool slower initially, then cool faster
4. **Perturbation**: When stuck, make several random swaps to escape

For the beam search SA, each step costs O(K * n) where K is the beam width. At n=360, K=5, that's about 1800 operations per step. We can do maybe 5000-10000 steps in 60 seconds. That should be plenty.

Let me also add a **2-opt style** move: instead of swapping one 1 with one 0, try swapping two 1s with two 0s (a 2-exchange). This can escape local optima that single swaps can't.

Actually, a single swap (1↔0) is already a 1-exchange. A 2-exchange would be: take two 1s at positions i1, i2 and two 0s at positions j1, j2, and swap (set h[i1]=h[i2]=0, h[j1]=h[j2]=1). This is equivalent to two independent single swaps but evaluated together. The advantage is that the combined effect might be better than either single swap alone.

For efficiency, I'll stick with single swaps in the main loop but add occasional 2-swaps as perturbations.

import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 5.0)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        to_remove = rng.permutation(len(idx))[:n_ones - target]
        h_bin[idx[to_remove]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        to_add = rng.permutation(len(idx))[:target - n_ones]
        h_bin[idx[to_add]] = 1
    return h_bin


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def thue_morse_binary(n, rng):
    h = np.zeros(n)
    mask = np.ones(n, dtype=bool)
    while mask.any():
        idx = np.where(mask)[0]
        h[idx] = 1.0
        mask[idx] = False
        if len(idx) == 0:
            break
        # Flip between 0 and 1 on next level in alternating blocks
        h[idx] = 1.0 - h[idx]
        # Subdivide
        new_mask = np.zeros(n, dtype=bool)
        for i, pos in enumerate(idx):
            if pos + 1 < n and not mask[pos + 1]:
                new_mask[pos + 1] = True
        # Simpler: just use the Thue-Morse sequence
    # Actually generate proper Thue-Morse
    t = [0]
    while len(t) < n:
        t += [1 - x for x in t]
    h = np.array(t[:n], dtype=float)
    # Ensure exactly n/2 ones
    h = make_binary(h, n, rng)
    return h


def random_binary(n, rng):
    h = np.zeros(n)
    ones = rng.permutation(n)[:n // 2]
    h[ones] = 1.0
    return h


def block_binary(n, rng):
    h = np.zeros(n)
    n_half = n // 2
    # Create blocks of alternating 1s and 0s with varying size
    pos = 0
    is_one = rng.rand() > 0.5
    while pos < n:
        block_len = rng.randint(2, max(3, n // 20))
        end = min(pos + block_len, n)
        if is_one:
            h[pos:end] = 1.0
        pos = end
        is_one = not is_one
    # Fix to have exactly n/2 ones
    h = make_binary(h, n, rng)
    return h


def sa_directed(h, n, time_limit=60, seed=42, beam_width=5):
    """Targeted SA with beam search and directed moves."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    dx_val = 2.0 / n

    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()

    T = 4.0
    T_min = 0.03
    start = time.time()
    stall = 0
    max_stall = 4000
    perturb_count = 0

    while time.time() - start < time_limit and T > T_min and remaining() > 5:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Generate candidate moves
        candidates = []

        # Directed moves: target worst shifts
        worst_idx = np.argsort(-corr)[:3]
        for widx in worst_idx:
            off = int(widx) - (n - 1)
            if off > 0:
                # h[l]=1, h[l+off]=0: swap to reduce this shift
                valid_l = np.arange(n - off)
                pairs = [(l, l + off) for l in valid_l if h[l] == 1 and h[l + off] == 0]
                if pairs:
                    chosen = rng.choice(len(pairs), size=min(3, len(pairs)), replace=False)
                    for c in chosen:
                        candidates.append((pairs[c][0], pairs[c][1]))
            elif off < 0:
                ao = -off
                valid_l = np.arange(n - ao)
                pairs = [(l + ao, l) for l in valid_l if h[l + ao] == 1 and h[l] == 0]
                if pairs:
                    chosen = rng.choice(len(pairs), size=min(3, len(pairs)), replace=False)
                    for c in chosen:
                        candidates.append((pairs[c][0], pairs[c][1]))

        # Random moves
        for _ in range(3):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            candidates.append((int(i), int(j)))

        # Evaluate beam
        best_delta = 0
        best_move = candidates[0] if candidates else None
        for (i, j) in candidates[:beam_width]:
            if i == j or h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            delta = new_max - cur_max
            h[i], h[j] = h[j], h[i]
            if delta < best_delta:
                best_delta = delta
                best_move = (i, j)

        if best_move is None:
            break

        # Apply move with SA acceptance
        i, j = best_move
        h[i], h[j] = h[j], h[i]
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        delta = new_max - cur_max

        if delta < 0 or (T > T_min and rng.rand() < np.exp(-delta / T)):
            cur_max = new_max
            corr = new_corr
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
                perturb_count = 0
            else:
                stall += 1
        else:
            h[i], h[j] = h[j], h[i]
            stall += 1

        # Cooling
        T *= 0.9995

        # Perturbation when stuck
        if stall > max_stall:
            T = max(T, 2.0)
            stall = 0
            perturb_count += 1
            # Make several random swaps
            n_perturb = min(10, 3 * perturb_count)
            for _ in range(n_perturb):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                a = ones[rng.randint(len(ones))]
                b = zeros[rng.randint(len(zeros))]
                h[a], h[b] = 0.0, 1.0
            corr = compute_corr(h)
            cur_max = float(np.max(corr))

    return best_h, best_max * dx_val


def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-12:
            g += w[k] * corr_grad_per_shift(x, n, k)
    return g * dx_val


def sum_con(x):
    return np.sum(x) - _TGT


def sum_con_jac(x):
    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=300):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def topk_gradient_descent(h, n, max_iters=300, top_k=5):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        m = float(np.max(corr))
        if m * (2.0 / n) >= best_c5 + 1e-6:
            break
        top_idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k in top_idx:
            g += corr_grad_per_shift(h, n, int(k))
        g /= len(top_idx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        improved = False
        for step in [0.05, 0.02, 0.01, 0.1, 0.005, 0.03, 0.008]:
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-13:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            break
    return h, best_c5
