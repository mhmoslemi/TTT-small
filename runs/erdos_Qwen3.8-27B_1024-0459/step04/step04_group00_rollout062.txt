def project_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to increase: prioritize increasing h where h < 1
            candidates = np.where(h < 1.0 - 1e-12)[0]
            if len(candidates) == 0:
                break
            h[candidates] += diff / len(candidates)
        else:
            # Need to decrease: prioritize decreasing h where h > 0
            candidates = np.where(h > 1e-12)[0]
            if len(candidates) == 0:
                break
            h[candidates] += diff / len(candidates)
        h = np.clip(h, 0.0, 1.0)
    # Final check
    if abs(np.sum(h) - n/2.0) > 1e-10:
        # Force exact
        diff = n/2.0 - np.sum(h)
        if diff > 0:
            candidates = np.where(h < 1.0)[0]
            h[candidates[:max(1,int(abs(diff)))]] += 1.0
        else:
            candidates = np.where(h > 0.0)[0]
            h[candidates[:max(1,int(abs(diff)))]] -= 1.0
        h = np.clip(h, 0.0, 1.0)
    return h

    """Find t in [0,1] minimizing max_k (A[k] + B[k]*t)."""
    # f(t) = max of linear functions, convex
    # Minimum is at a breakpoint or at boundary
    
    # Find breakpoints: intersection of line k and line j
    # A[k] + B[k]*t = A[j] + B[j]*t
    # t = (A[j] - A[k]) / (B[k] - B[j])
    
    # But we only need breakpoints on the upper envelope
    # Simpler: just evaluate at many points and refine
    
    # Actually, for exact solution:
    # The upper envelope of m lines has at most m segments
    # Find the minimum of the upper envelope
    
    # Simple approach: binary search on subgradient
    # f'(t-) < 0 means we should increase t
    # f'(t+) > 0 means we should decrease t
    
    # Even simpler: since f is convex, use ternary search
    lo, hi = 0.0, 1.0
    for _ in range(100):  # enough for double precision
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = np.max(A + B * m1)
        f2 = np.max(A + B * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    t_opt = (lo + hi) / 2
    return t_opt, np.max(A + B * t_opt)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):

        if abs(diff) < 1e-14:

        if diff > 0:

    m = np.max(corr)

    ex = np.exp(cl)
    logsum = m + np.log(np.sum(ex)) / beta
    sum_dev = np.sum(h) - n / 2.0
    return logsum * dx + mu * sum_dev * sum_dev

            h[j] = 1.0
            corr1 = compute_corr(h)
            h[j] = old_val
            
            A = corr0.copy()  # value at h[j]=0
            B = corr1 - corr0  # coefficient of h[j]
            
            sum_other = np.sum(h) - old_val
            target_sum = n / 2.0
            
            # 1D minimization: min_{t in [0,1]} max_k(A_k + B_k*t)*dx + mu*(sum_other+t-target)^2
            def f1d(t):
                val = np.max(A + B * t) * dx
                dev = sum_other + t - target_sum
                return val + mu * dev * dev
            
            # Ternary search (convex function)
            lo, hi = 0.0, 1.0
            for _ in range(80):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = f1d(m1)
                f2 = f1d(m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            t_opt = (lo + hi) / 2.0
            
            if abs(t_opt - old_val) > 1e-12:
                h[j] = t_opt
                improved = True
        
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
        if not improved:
            break
    
    return project_feasible(h, n), best_c5

def active_equalize(h, n, top_k=8, max_iters=300):
    """Directly equalize top-k correlations."""
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_vals = corr[idx]
        c_max = c_vals[0]
        c_min = c_vals[-1]
        
        if c_max - c_min < 1e-11:
            # All equal, try reducing all
            g = np.zeros(n)
            for k in idx:
                off = k - (n - 1)
                if off >= 0:
                    np_ = n - off
                    g[off:off + np_] += (1.0 - h[:np_])
                    g[:np_] -= h[off:off + np_]
                else:
                    ao = -off
                    np_ = n - ao
                    g[:np_] += (1.0 - h[ao:ao + np_])
                    g[ao:ao + np_] -= wk if False else (1.0 - h[:np_]) * 0 + (1.0 - h[:np_])
                    g[:np_] -= 0
                    # Redo properly
                    g = np.zeros(n)  # reset
                    for k2 in idx:
                        off2 = k2 - (n - 1)
                        if off2 >= 0:
                            np2 = n - off2
                            g[off2:off2+np2] += (1.0 - h[:np2])
                            g[:np2] -= h[off2:off2+np2]
                        else:
                            ao2 = -off2
                            np2 = n - ao2
                            g[:np2] += (1.0 - h[ao2:ao2+np2])
                            g[ao2:ao2+np2] -= h[:np2]
                    break
        else:
            pass
        
        # Proper: compute gradient direction to equalize
        def lag_grad(k, h, n):
            g = np.zeros(n)
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += (1.0 - h[:np_])
                g[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= h[:np_]
            return g
        
        g_sum = np.zeros(n)
        for k in idx:
            g_sum += lag_grad(k, h, n)
        g_sum /= len(idx)
        
        # Project to sum=0 subspace
        g_sum -= np.mean(g_sum)
        
        dn = np.linalg.norm(g_sum)
        if dn < 1e-14:
            break
        d = -g_sum / dn
        
        improved = False
        for step in [0.01, 0.005, 0.02, 0.001, 0.05, 0.002, 0.008, 0.015]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            # Try individual top lags
            found = False
            for k in idx:
                gt = lag_grad(k, h, n)
                gt -= np.mean(gt)
                dnt = np.linalg.norm(gt)
                if dnt < 1e-14:
                    continue
                dt = -gt / dnt
                for step in [0.01, 0.005, 0.02, 0.001]:
                    ht = project_feasible(h + step * dt, n)
                    cn = compute_c5(ht)
                    if cn < best_c5 - 1e-15:
                        h = ht
                        best_c5 = cn
                        best_c5 = cn
                        improved = True
                        found = True
                        break
                if found or improved:
                    break
            if not improved:
                break
    
    return h, best_c5

def lag_grad_func(k, h, n):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:off + np_] += (1.0 - h[:np_])
        g[:np_] -= h[off:off + np_]
    else:
        ao = -off
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def perturb_and_refine(h, n, rng, n_perturbs=50):
    """Random perturbation then full re-optimization."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for p in range(n_perturbs):
        if remaining() < 30:
            break
        # Random perturbation
        noise_scale = 0.05 / (1 + p * 0.1)
        h_pert = h + noise_scale * (rng.rand(n) - 0.5)
        h_pert = project_feasible(h_pert, n)
        
        # Quick L-BFGS-B
        h_opt = lbfgs_smooth(h_pert, n, 5000, 1e8, maxiter=1000)
        c5_new = compute_c5(h_opt)
        
        # Coordinate descent polish
        h_cd, c5_cd = coord_descent(h_opt, n, max_sweeps=10, mu=1e8)
        if c5_cd < c5_new:
            c5_new = c5_cd
            h_opt = h_cd
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_opt.copy()
            h = h_opt.copy()
    
    return best_h, best_c5

def binary_sa_refine(h, n, time_limit=60, rng=None):
    """Binary simulated annealing with continuous refinement."""
    if rng is None:
        rng = np.random.RandomState(42)
    
    h_bin = np.zeros(n)
    sort_idx = np.argsort(-h)[:n//2]
    h_bin[sort_idx] = 1.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = compute_c5(h_bin)
    
    start = time.time()
    T0, T1 = 1.0, 0.05
    
    while time.time() - start < time_limit and remaining() > 10:
        T = T0 * (T1 / T0) ** ((time.time() - start) / time_limit)
        if T < 0.01:
            break
        
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        new_c5 = compute_c5(h_bin)
        delta = new_c5 - cur_c5
        
        if delta <= 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                # Refine from this binary solution
                h_ref = h_bin.astype(float).copy()
                h_ref = lbfgs_smooth(h_ref, n, 5000, 1e8, maxiter=500)
                h_ref, c5_ref = coord_descent(h_ref, n, max_sweeps=5, mu=1e8)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref.copy()
                cur_c5 = compute_c5(h_bin)
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    n = best_n
    h = best_h.copy()
    
    # Phase 1: L-BFGS-B progressive beta (fast warmup)
    for beta, mu in [(50, 1e6), (200, 1e6), (500, 1e7), (1000, 1e7)]:
        if remaining() < 15:
            break
        h_new = lbfgs_smooth(h, n, beta, mu, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 2: High-beta L-BFGS-B refinement
    for beta, mu in [(3000, 1e8), (8000, 1e8), (20000, 1e8), (50000, 1e8)]:
        if remaining() < 20:
            break
        h_new = lbfgs_smooth(h, n, beta, mu, maxiter=3000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 3: Coordinate descent (exploits 1D structure)
    if remaining() > 30:
        h_cd, c5_cd = coord_descent(h, n, max_sweeps=30, mu=1e8)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()
    
    # Phase 4: Active-set equalization
    if remaining() > 20:
        h_eq, c5_eq = active_equalize(h, n, top_k=8, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
    
    # Phase 5: More coordinate descent after equalization
    if remaining() > 30:
        h_cd2, c5_cd2 = coord_descent(h, n, max_sweeps=40, mu=1e8)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
            h = h_cd2.copy()
    
    # Phase 6: Resolution exploration
    n_cands = sorted(set([
        best_n, best_n - 4, best_n + 4, best_n + 8,
        best_n + 16, best_n - 8,
        int(best_n * 1.3), int(best_n * 1.5), int(best_n * 1.8),
    ]))
    n_cands = [nn for nn in n_cands if 128 <= nn <= 768 and nn % 2 == 0]
    n_cands = [nn for nn in n_cands if nn != best_n]
    n_cands.sort(key=lambda x: abs(x - best_n))
    
    for n_t in n_cands[:5]:
        if remaining() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)
        
        # Optimize at new resolution
        h_opt = h_feas
        for beta in [1000, 5000, 20000]:
            if remaining() < 15:
                break
            h_opt = lbfgs_smooth(h_opt, n_t, beta, 1e8, maxiter=1500)
        
        h_opt, c5_opt = coord_descent(h_opt, n_t, max_sweeps=15, mu=1e8)
        h_opt, c5_opt = active_equalize(h_opt, n_t, top_k=6, max_iters=100)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
            h = h_opt.copy()
            n = n_t
    
    # Phase 7: Random perturbation + re-optimize
    if remaining() > 60:
        h_pert, c5_pert = perturb_and_refine(h, n, rng, n_perturbs=30)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
            h = h_pert.copy()
    
    # Phase 8: Binary SA + continuous refinement
    if remaining() > 40:
        sa_time = min(90, max(10, int(remaining() * 0.3)))
        h_sa, c5_sa = binary_sa_refine(best_h, n, time_limit=sa_time, rng=rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            h = h_sa.copy()
    
    # Phase 9: Final intensive refinement at best resolution
    n = best_n
    h = best_h.copy()
    
    # High-beta L-BFGS-B
    for beta in [10000, 30000, 80000, 200000]:
        if remaining() < 5:
            break
        h_new = lbfgs_smooth(h, n, beta, 1e8, maxiter=3000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Coordinate descent final
    if remaining() > 10:
        h_cd, c5_cd = coord_descent(h, n, max_sweeps=50, mu=1e8)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()
    
    # Active equalize final
    if remaining() > 10:
        h_eq, c5_eq = active_equalize(h, n, top_k=10, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
    
    # Another coordinate descent pass
    if remaining() > 10:
        h_cd2, c5_cd2 = coord_descent(h, n, max_sweeps=30, mu=1e8)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
            h = h_cd2.copy()
    
    # Phase 10: More perturbations if time remains
    if remaining() > 30:
        h_p2, c5_p2 = perturb_and_refine(h, n, rng, n_perturbs=20)
        if c5_p2 < best_c5:
            best_c5 = c5_p2
            best_h = h_p2.copy()
            h = h_p2.copy()
    
    # Final polish: very high beta + coord descent
    if remaining() > 8:
        h_final = lbfgs_smooth(h, n, 100000, 1e8, maxiter=2000)
        c5_final = compute_c5(h_final)
        if c5_final > best_c5:
            h_final = h.copy()
            c5_final = best_c5
        h_final, c5_final2 = coord_descent(h_final, n, max_sweeps=20, mu=1e8)
        if c5_final2 < c5_final:
            h_final = h_final
            c5_final = c5_final2
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = final_c5
    
    # Validate with evaluate_erdos_solution if available
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
