def targeted_refine(h, n, budget_fn, rng):
    """Directly target the max correlations and reduce them."""
    h = h.copy()
    c5_best = compute_c5(h)
    c5_cur = c5_best
    dx = 2.0 / n
    
    for iteration in range(500):
        if budget_fn() < 5.0:
            break
        
        corr = compute_corr(h)
        max_val = np.max(corr)
        top_k = np.argsort(corr)[-3:]  # top 3 lags
        
        # Compute subgradient for the top lags
        grad = np.zeros(n)
        for k_idx in top_k:
            offset = int(k_idx) - (n - 1)
            if offset >= 0:
                # corr[k_idx] = sum_{i=0}^{n-1-offset} h[i] * (1-h[i+offset])
                # d/dh[i] = (1-h[i+offset]) for i < n-offset (as first factor)
                #           - h[i-offset] for i >= offset (as second factor's 1-h)
                n_pairs = n - offset
                for i in range(n_pairs):
                    grad[i] += (1.0 - h[i + offset])
                    grad[i + offset] -= h[i]
            else:
                abs_off = -offset
                n_pairs = n - abs_off
                for i in range(n_pairs):
                    grad[i + abs_off] += (1.0 - h[i])
                    grad[i] -= h[i + abs_off]
        
        grad *= dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Line search
        improved = False
        for alpha in [0.01, 0.005, 0.002, 0.001, 0.02, 0.05]:
            h_trial = h - alpha * grad / gnorm
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)

                c5_best = c5_trial
                c5_cur = c5_best
                improved = True
                break

            h_trial = project_feasible(h + noise, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-15:
                h = h_trial
                c5_best = c5_trial

import numpy as np
from scipy.optimize import minimize
import time

_G_START = 0.0
_G_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            break
    return h

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h > 0.5))
    target_ones = n // 2
    if n_ones < target_ones:
        zeros = np.where(h < 0.5)[0]
        for i in range(target_ones - n_ones):
            h[zeros[i]] = 1.0
    elif n_ones > target_ones:
        ones = np.where(h > 0.5)[0]
        for i in range(n_ones - target_ones):
            h[ones[i]] = 0.0
    return h

def slsqp_optimize(h_start, n, budget_fn, beta_list):
    """SLSQP with log-sum-exp smoothing."""
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    dx = 2.0 / n
    
    for beta in beta_list:
        if budget_fn() < 10.0:
            break
        
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(corr)
            shifted = corr - m
            exps = np.exp(np.clip(beta * shifted, -500, 500))
            return (m + np.log(np.sum(exps)) / beta) * dx
        
        def con_fun(x):
            return np.sum(x) - n/2.0
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_fun}]
        
        maxiter = min(300, max(20, int(budget_fn() * 0.3 / 0.5)))
        
        try:
            res = minimize(obj, h_cur, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                h_cur = h_new
                c5_cur = c5_new
        except Exception:
            pass
    
    return h_cur, c5_cur
