        g_max = corr_grad_at_shift(h, n, k_max)

        for i in range(n):
            if time.time() - start > time_limit:
                break
            if remaining() < 5:
                break

            s = h[i]
            if s < 1e-14 or s > 1.0 - 1e-14:
                s_new = s
            else:
                slope = g_max[i]
                if slope > 1e-10:
                    s_new = 0.0
                elif slope < -1e-10:
                    s_new = 1.0
                else:
                    s_new = s
            if abs(s_new - s) > 1e-14:
                h[i] = s_new

        h = project_feasible(h, n)
        corr = compute_corr(h)
        c5_new = float(np.max(corr) * 2.0 / n)
        if c5_new < best_c5:
            best_c5 = c5_new

    return h, best_c5


def multi_beta_slsqp(h, n, betas, maxiter_each=250):
    """Run SLSQP with increasing beta values."""
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    for beta in betas:
        if remaining() < 8:
            break
        h_new = slsqp_opt(h_cur, n, beta, maxiter=maxiter_each)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cur = h_new.copy()
    return best_h, best_c5


def golden_ramp_init(n):
    """Create a structured initialization based on golden ratio."""
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        x = (i + 0.5) * 2.0 / n
        val = (x * phi) % 1.0
        h[i] = 1.0 if val < 0.5 else 0.0
    return project_feasible(h, n)


def periodic_init(n, period=4):
    """Create a periodic initialization."""
    h = np.zeros(n)
    pattern = np.array([1, 1, 0, 0])[:period]
    for i in range(n):
        h[i] = pattern[i % period]
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 312
        parent_h = golden_ramp_init(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP refinement with increasing beta on parent n
    if remaining() > 50:
        h, c5 = multi_beta_slsqp(h, n, [100, 300, 1000, 3000, 8000, 20000], maxiter_each=250)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    # Phase 2: L-BFGS-B with penalty
    if remaining() > 40:
        h2 = lbfgs_opt(best_h, best_n, 5000, maxiter=300)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Targeted coordinate descent
    if remaining() > 30:
        h3, c5_3 = coord_descent_targeted(best_h, best_n, max_passes=100,
                                          time_limit=min(40, remaining() - 10), rng=rng)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Try multiple n values
    n_candidates = []
    base_n = best_n
    for delta in [-4, -2, 2, 4, 8, 12, 16, 24, 32, 48, 64, 80, 96]:
        nc = base_n + delta
        if 100 <= nc <= 800 and nc % 2 == 0 and nc != base_n:
            n_candidates.append(nc)
    n_candidates = sorted(set(n_candidates))

    # Also add some "natural" n values where binary solutions can hit target
    for nc in [320, 336, 352, 368, 384, 400, 416, 432, 448, 464, 480, 496, 512]:
        if 100 <= nc <= 800 and nc % 2 == 0 and nc != base_n:
            n_candidates.append(nc)
    n_candidates = sorted(set(n_candidates))

    # Prioritize n values where binary solutions can achieve <= 0.38085
    def n_priority(nc):
        max_corr_allowed = 0.38085 * nc / 2.0
        if int(max_corr_allowed) >= int(max_corr_allowed):
            return 0
        return nc - int(max_corr_allowed * 10000)

    n_candidates.sort(key=n_priority)

    n_tried = set()
    for n_t in n_candidates[:12]:
        if remaining() < 40:
            break
        if n_t in n_tried:
            continue
        n_tried.add(n_t)

        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)

        # SLSQP on new n
        h_opt = h_feas
        for beta in [500, 2000, 8000, 20000]:
            if remaining() < 10:
                break
            h_opt = slsqp_opt(h_opt, n_t, beta, maxiter=200)

        # Coord descent
        if remaining() > 20:
            h_opt, _ = coord_descent_targeted(h_opt, n_t, max_passes=80,
                                              time_limit=min(25, remaining() - 10), rng=rng)

        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 5: Binary search on best n
    if remaining() > 30:
        h_bin, c5_bin = targeted_binary_search(best_h, best_n,
                                               time_limit=min(60, remaining() - 10), rng=rng)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 6: SLSQP on binary result (to explore nearby continuous solutions)
    if remaining() > 20 and best_c5 > 0.38085:
        h_cont = best_h.copy()
        for beta in [5000, 15000, 40000]:
            if remaining() < 8:
                break
            h_new = slsqp_opt(h_cont, best_n, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_cont = h_new.copy()

    # Phase 7: Final intensive refinement
    n = best_n
    h = best_h.copy()
    for beta in [10000, 30000, 80000]:
        if remaining() < 6:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 8: Final binary search with more time
    if remaining() > 20 and best_c5 > 0.38084:
        h_bin2, c5_bin2 = targeted_binary_search(best_h, best_n,
                                                  time_limit=min(50, remaining() - 5), rng=rng)
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()

    # Final SLSQP polish
    if remaining() > 8:
        h_new = slsqp_opt(best_h, best_n, 50000, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Validate and finalize
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
