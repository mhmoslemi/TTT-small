        for i in range(n):
            if remaining_time() < 1:
                break
            for target_val in [0.0, 1.0]:
                trial = best_h.copy()
                diff = target_val - trial[i]
                if abs(diff) < 1e-6:
                    continue
                trial[i] = target_val
                trial = project_feasible(trial, n)
                c5_t = compute_c5(trial)
                if c5_t < best_c5 - 1e-15:
                    best_c5 = c5_t
                    best_h = trial.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def perturb_restart(h, n, rng, sigma=0.03):
    h_p = h.copy()
    mask = (h_p > 0.05) & (h_p < 0.95)
    if np.sum(mask) > 0:
        h_p[mask] += rng.normal(0, sigma, np.sum(mask))
    n_flip = max(1, n // 32)
    idx = rng.choice(n, size=n_flip, replace=False)
    for i in idx:
        h_p[i] = 1.0 - h_p[i]
    h_p = np.clip(h_p, 0.0, 1.0)
    h_p = project_feasible(h_p, n)
    return h_p

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 256
        best_h = np.full(best_n, 0.5)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP with analytical gradient, escalating beta
    for beta in [20, 50, 100, 300, 800, 2000, 5000, 10000, 20000]:
        if remaining_time() < 15:
            break
        maxit = 500 if beta < 1000 else 300
        h_new = slsqp_smooth(h, n, beta, maxiter=maxit)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-16:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: L-BFGS-B with penalty (faster iterations)
    if remaining_time() > 60:
        for beta in [500, 2000, 5000, 10000, 30000]:
            if remaining_time() < 20:
                break
            h_new = lbfgsb_smooth(h, n, beta, maxiter=3000, gamma=500.0)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-16:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 3: SLSQP polish with very high beta
    if remaining_time() > 40:
        for beta in [50000, 100000]:
            if remaining_time() < 10:
                break
            h_new = slsqp_smooth(best_h, best_n, beta, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-16:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 4: Multi-resolution
    trial_ns = [192, 224, 256, 320, 384, 448, 512, 640]
    for trial_n in trial_ns:
        if remaining_time() < 25:
            break
        if trial_n == best_n:
            continue
        h_interp = interp_to(best_h, best_n, trial_n)
        h_interp = project_feasible(h_interp, trial_n)
        c5_interp = compute_c5(h_interp)

        if c5_interp < best_c5 + 0.005:
            maxit = 400 if trial_n <= 384 else 250
            for beta in [300, 1500, 5000]:
                if remaining_time() < 8:
                    break
                h_new = slsqp_smooth(h_interp, trial_n, beta, maxiter=maxit)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = trial_n
                    h_interp = h_new.copy()

            if remaining_time() > 20 and trial_n <= 512:
                h_new = lbfgsb_smooth(h_interp, trial_n, 10000, maxiter=2000, gamma=500.0)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = trial_n

            if remaining_time() > 15 and trial_n <= 384:
                h_new = slsqp_smooth(best_h, best_n, 30000, maxiter=200)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()

    # Phase 5: Coordinate descent (fine-tuning)
    if remaining_time() > 30:
        h_cd, c5_cd = coordinate_descent(best_h, best_n, 5000, max_rounds=15)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 6: Pair swap search
    if remaining_time() > 20:
        h_pw, c5_pw = pair_swap_search(best_h, best_n, max_sweeps=6, window=50, rng=rng)
        if c5_pw < best_c5:
            best_c5 = c5_pw
            best_h = h_pw.copy()

    # Phase 7: SLSQP after combinatorial improvements
    if remaining_time() > 20:
        h_new = slsqp_smooth(best_h, best_n, 8000, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 8: Perturbation restarts
    for restart in range(5):
        if remaining_time() < 35:
            break
        h_p = perturb_restart(best_h, best_n, rng, sigma=0.02)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5 + 0.005:
            h_p2 = slsqp_smooth(h_p, best_n, 2000, maxiter=250)
            c5_p2 = compute_c5(h_p2)
            if c5_p2 < best_c5:
                best_c5 = c5_p2
                best_h = h_p2.copy()
            else:
                h_p3 = lbfgsb_smooth(h_p, best_n, 5000, maxiter=1500, gamma=300.0)
                c5_p3 = compute_c5(h_p3)
                if c5_p3 < best_c5:
                    best_c5 = c5_p3
                    best_h = h_p3.copy()

    # Phase 9: Single bit search
    if remaining_time() > 10:
        h_sb, c5_sb = single_bit_search(best_h, best_n, max_rounds=3)
        if c5_sb < best_c5:
            best_c5 = c5_sb
            best_h = h_sb.copy()

    # Phase 10: Final ultra-precision SLSQP
    if remaining_time() > 10:
        for beta in [20000, 50000, 100000]:
            if remaining_time() < 5:
                break
            h_new = slsqp_smooth(best_h, best_n, beta, maxiter=150)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 11: Final L-BFGS-B polish
    if remaining_time() > 8:
        h_new = lbfgsb_smooth(best_h, best_n, 50000, maxiter=2000, gamma=1000.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 12: Final coordinate + pair
    if remaining_time() > 8:
        h_cd, c5_cd = coordinate_descent(best_h, best_n, 50000, max_rounds=5,
                                         step_sizes=[0.005, 0.002, 0.001, 0.0005])
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if remaining_time() > 5:
        h_pw, c5_pw = pair_swap_search(best_h, best_n, max_sweeps=3, window=60, rng=rng)
        if c5_pw < best_c5:
            best_c5 = c5_pw
            best_h = h_pw.copy()

    # Final projection and verification
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
