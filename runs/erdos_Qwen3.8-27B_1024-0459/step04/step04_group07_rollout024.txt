def compute_weighted_gradient(h, w_full, n):
    """
    h: length n
    w_full: length 2n-1, softmax weights for each lag (indexed by full correlation index)
    Returns gradient of weighted sum of correlations w.r.t. h
    """
    omh = 1.0 - h
    
    # Positive lags: k = 0, 1, ..., n-1 → full corr index n-1, n, ..., 2n-2
    w_pos = w_full[n-1:]  # length n, w_pos[k] = weight for lag k
    
    # A[j] = sum_{k=0}^{n-1-j} w_pos[k] * omh[j+k]
    # = np.correlate(w_pos, omh, 'full')[n-1 + j] for j=0,...,n-1
    corr_A = np.correlate(w_pos, omh, mode='full')
    A = corr_A[n-1:2*n-1]  # length n
    
    # B[j] = sum_{k=0}^{j} w_pos[k] * h[j-k]
    # = np.convolve(w_pos, h, 'full')[j] for j=0,...,n-1
    conv_B = np.convolve(w_pos, h, mode='full')
    B = conv_B[:n]  # length n
    
    # Negative lags: k = -1, -2, ..., -(n-1) → full corr index n-2, n-3, ..., 0

    # C[j] = sum_{d=1}^{j} w_neg[d-1] * omh[j-d]
    # Let w_neg_padded = [0, w_neg[0], w_neg[1], ..., w_neg[n-2]] (length n)
    # C[j] = sum_{d=1}^{min(j,n-1)} w_neg[d-1] * omh[j-d]
    # = (w_neg_padded * omh)[j] = np.convolve(w_neg_padded, omh, 'full')[j]
    w_neg_padded = np.zeros(n)

    C = conv_C[:n]

    # = sum_{e=j+1}^{n-1} w_neg[e-j-1] * h[e]

    # D[j] = sum_{d=1}^{n-1-j} w_neg[d-1] * h[j+d]

    # = sum_{f=0}^{j-1} w_neg[n-2-f] * h[j+n-1-f]... this is getting confusing.

    # This is the same as: for the sequence w_neg (length n-1) and h (length n),
    # it's the "anti-convolution" or cross-correlation.
    # np.correlate(w_neg, h, mode='full')[m] = sum_i w_neg[i] * h[i+m-(n-2)]
    # Hmm, lengths are different (w_neg is n-1, h is n), so full corr has length 2n-2.
    # np.correlate(w_neg, h, mode='full')[m] = sum_i w_neg[i] * h[i+m-(len(h)-1)]
    # = sum_i w_neg[i] * h[i+m-n+1]
    # For m = n-1-j: sum_i w_neg[i] * h[i+j]  (valid when 0≤i≤n-2 and 0≤i+j≤n-1)
    # = sum_{i=0}^{n-2-j} w_neg[i] * h[i+j]

```python
def subgradient_optimize(h0, n, tlimit, max_iters=100000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    eta = 1.0  # step size
    beta = 30.0
    dx = 2.0 / n
    patience = 0
    patience_limit = 2000
    
    for it in range(max_iters):
        if tlimit() < 2.0:
            break
        
        # Compute correlation and weights
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        corr_clipped = np.clip(beta * corr, -400, 400)
        e = np.exp(corr_clipped - np.max(corr_clipped))
        w_full = e / np.sum(e)
        
        # Compute gradient
        grad = compute_weighted_gradient(h, w_full, n)
        
        # Gradient step
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
            eta *= 0.99
        
        h = h_new
        beta += 0.5
        
        if patience > patience_limit:
            patience = 0
            eta = max(eta, 0.5)
            # Try a random perturbation
            if tlimit() > 5:
                h = h + 0.1 * np.random.randn(n)
                h = project_feasible(h, n)
    
    return best_h, best_c5
```

Let me also think about the alternating best-response approach I mentioned earlier:

For a given active lag k*, the "best response" is:
- Fix all h values except those that appear in the k* correlation.
- Actually, for the bilinear form O(k) = sum_i h[i](1-h[i+k]), if we fix h[i+k] for all i, then O(k) is linear in h[i]: O(k) = sum_i h[i](1-h[i+k]).
- To minimize this subject to sum h = n/2 and 0≤h≤1, we set h[i]=1 where (1-h[i+k]) is smallest (i.e., h[i+k] is largest).

This gives a binary solution. We can alternate:
1. k* = argmax O(k)
2. Sort indices by h[i+k*] descending. Set h[i]=1 for the top n/2, h[i]=0 for the rest.
3. Repeat.

But this is very aggressive and might oscillate. A softer version:
1. k* = argmax O(k)
2. Compute "scores" s[i] = h[i+k*] (high score means we want h[i] to be 1)
3. h_new = soft threshold of scores (e.g., h_new[i] = clip(alpha * s[i] - gamma, 0, 1) then project to sum = n/2)
4. h = (1-t)*h + t*h_new (with small t for stability)

This is like a form of gradient descent but with a "greedy" component.
