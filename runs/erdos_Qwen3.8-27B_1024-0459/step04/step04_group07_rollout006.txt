_G_USE_QUAD = False

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_top_lags(h, m=5):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-m:][::-1]
    lags = top_idx - (n - 1)
    return lags, corr, top_idx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        adj = diff / cnt
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def obj_func(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    beta = _G_BETA
    s = corr - m
    e = np.exp(np.clip(beta * s, -400, 400))
    val = (m + np.log(np.sum(e)) / beta) * _G_DX
    if _G_PEN > 0:
        val += _G_PEN * (np.sum(h) - _G_TGT) ** 2

    return float(val)

def cons_eq(x):
    return np.sum(x) - 0.5 * len(x)

def cons_eq_jac(x):
    return np.ones(len(x), dtype=float)

def slsqp_opt(h0, n, beta, maxiter, focus_lags=None, focus_wt=0.0, use_quad=False):

def subgrad_step(h, n, m_lags=5, eta=0.05):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-m_lags:][::-1]
    lags = top_idx - (n - 1)
    grad = np.zeros(n, dtype=float)
    for l in lags:
        if l >= 0:
            nx = n - l
            if nx > 0:
                grad[0:nx] += (1.0 - h[l:n])
                grad[l:n] -= h[0:nx]
        else:
            lp = -l
            nx = n - lp
            if nx > 0:
                grad[lp:n] += (1.0 - h[0:nx])
                grad[0:nx] -= h[lp:n]
    grad /= float(m_lags)
    h_new = h - eta * grad
    h_new = project_feasible(h_new, n)
    return h_new

def subgrad_descent(h, n, max_iter, eta_init=0.05, m_lags=5, tlimit_func=None):

            break
        h_try = h_best.copy()
        for _ in range(10):
            h_try = subgrad_step(h_try, n, m_lags=m_lags, eta=eta)
        c_try = compute_c5(h_try)
        if c_try < c5_best - 1e-10:
            c5_best = c_try
            h_best = h_try.copy()
            eta = min(eta * 1.2, 0.3)
            stall = 0
        else:
            stall += 1
            if stall > 5:
                eta = eta * 0.7
                stall = 0

                break
    return h_best, c5_best

def coord_descent(h, n, max_sweeps, rng, tlimit_func):
    c5 = compute_c5(h)
    h_best = h.copy()
    deltas = [0.2, -0.2, 0.1, -0.1, 0.05, -0.05, 0.3, -0.3, 0.5, -0.5]
    for sweep in range(max_sweeps):
        if tlimit_func() < 2.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if tlimit_func() < 1.0:
                break
            v0 = h_best[i]
            best_c = c5
            best_v = v0
            for d in deltas:
                v_new = np.clip(v0 + d, 0.0, 1.0)
                if abs(v_new - v0) < 1e-9:
                    continue
                trial = h_best.copy()
                trial[i] = v_new
                trial = project_feasible(trial, n)
                c = compute_c5(trial)

                h_best = project_feasible(h_best, n)
                c5 = best_c
                improved = True

            break
    return h_best, c5

                h_pert = spectral_perturb(best_h, n, rng, strength=0.2 + 0.05 * attempt)
            else:
                h_pert = best_h.copy()
                n_pert = rng.randint(5, 20)
                idxs = rng.choice(n, size=n_pert, replace=False)
                for idx in idxs:
                    h_pert[idx] = np.clip(h_pert[idx] + rng.randn() * 0.4, 0.0, 1.0)
                h_pert = project_feasible(h_pert, n)
            c5_p = compute_c5(h_pert)

            if c5_p < best_c5 * 1.03:
                h_cur = h_pert.copy()
                c5_cur = c5_p
                for bv in [100, 500, 2000]:
                    if tlimit() < 8:
                        break
                    lags, _, _ = get_top_lags(h_cur, m=4)
                    if n <= 400:
                        h_new = slsqp_opt(h_cur, n, bv, 250, focus_lags=lags, focus_wt=0.6, use_quad=True)
                    else:
                        h_new = lbfgs_opt(h_cur, n, bv, 800, pen=200.0)
                        if tlimit() > 10:
                            lags2, _, _ = get_top_lags(h_new, m=4)
                            h_new = slsqp_opt(h_new, n, bv, 100, focus_lags=lags2, focus_wt=0.8)
                    c_new = compute_c5(h_new)
                    if c_new < c5_cur - 1e-12:
                        c5_cur = c_new
                        h_cur = h_new.copy()
                if c5_cur < best_c5 - 1e-10:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n
                    print("  Perturb {} improved: C5={:.8f}".format(attempt, best_c5))

    # Phase 5: Coordinate descent at best resolution
    if tlimit() > 40:
        print("\n=== Phase 5: coordinate descent n={} ===".format(best_n))
        best_n = len(best_h)
        best_h, best_c5 = coord_descent(best_h, best_n, max_sweeps=50, rng=rng, tlimit_func=tlimit)
        print("  After coord: C5={:.8f}".format(best_c5))

    # Phase 6: Final SLSQP polish with very high beta
    if tlimit() > 20:
        print("\n=== Phase 6: final polish n={} ===".format(best_n))
        n = best_n
        for bv in [5000, 15000, 40000, 80000]:
            if tlimit() < 8:
                break
            lags, _, _ = get_top_lags(best_h, m=5)
            h_new = slsqp_opt(best_h, n, bv, 400, focus_lags=lags, focus_wt=1.0, use_quad=True)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new.copy()

        if tlimit() > 10:
            lags, _, _ = get_top_lags(best_h, m=3)
            h_new = slsqp_opt(best_h, n, 100000, 300, focus_lags=lags, focus_wt=2.0, use_quad=False)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 7: Final coordinate descent
    if tlimit() > 10:
        best_n = len(best_h)
        best_h, best_c5 = coord_descent(best_h, best_n, max_sweeps=20, rng=rng, tlimit_func=tlimit)

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    _G_PEN = 0.0
    _G_FOCUS_LAGS = None
    _G_FOCUS_WT = 0.0
    _G_USE_QUAD = False
    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
