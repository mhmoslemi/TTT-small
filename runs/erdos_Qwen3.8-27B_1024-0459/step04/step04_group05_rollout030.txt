```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    # Project to {h: 0<=h<=1, sum(h)=n/2}
    h = np.clip(h, 0, 1)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Distribute uniformly amongst non-bound variables
        free = (h > 1e-10) & (h < 1 - 1e-10)
        if np.any(free):
            h[free] += diff / np.sum(free)
        else:
            h += diff / n
        h = np.clip(h, 0, 1)
    return h

def sla_step(h, n, tol=0.01, max_active=None):
    """Successive Linear Approximation step"""
    dx = 2.0 / n
    corr = compute_corr_array(h)  # length 2n-1
    c5_vals = corr * dx
    c5_max = np.max(c5_vals)
    
    # Identify active lags (within tol of max)
    active = np.where(c5_vals >= c5_max - tol)[0]
    if max_active is not None and len(active) > max_active:
        top_vals = np.sort(c5_vals[active])[::-1][:max_active]
        threshold = top_vals[-1]
        active = np.where(c5_vals >= threshold)[0]
    
    # Set up CVXPY problem
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0
    ]
    
    # Linearize each active constraint
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        # corr[k_idx] = sum_i h[i] * (1-h[i+offset]) over valid i
        # Linearize h[i]*(1-h[i+offset]) ≈ h[i]*(1-h[i+offset]) at h
        # = h[i] - h[i]*h[i+offset]
        # Linearization of h[i]*h[i+offset] at h*: h*[i]*h[i+offset] + h[i]*h*[i+offset] - h*[i]*h*[i+offset]
        # So linearization of h[i] - h[i]*h[i+offset]:
        # = h[i] - h*[i]*h[i+offset] - h[i]*h*[i+offset] + h*[i]*h*[i+offset]
        
        if offset >= 0:
            n_pairs = n - offset
            i_range = np.arange(n_pairs)
            # Linearized: h[i_range[j]] - h_star[i_range[j]]*h[i_range[j]+offset] - h[i_range[j]]*h_star[i_range[j]+offset] + h_star[i_range[j]]*h_star[i_range[j]+offset]
            lin = (h_var[i_range] 
                   - h[i_range] * h_var[i_range + offset] 
                   - h_var[i_range] * h[i_range + offset] 
                   + h[i_range] * h[i_range + offset])
        else:
            abs_off = -offset
            n_pairs = n - abs_off
            i_range = np.arange(n_pairs)
            # corr at negative offset: sum h[i+abs_off] * (1-h[i])
            # = h[i+abs_off] - h[i+abs_off]*h[i]
            # Linearized:
            lin = (h_var[i_range + abs_off] 
                   - h[i_range + abs_off] * h_var[i_range] 
                   - h_var[i_range + abs_off] * h[i_range] 
                   + h[i_range + abs_off] * h[i_range])
        
        constraints.append(cp.sum(lin) * dx <= t_var)
    
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.GLOP)  # or CPLEX, or CBC
        if h_var.value is not None and t_var.value is not None:
            h_new = np.clip(h_var.value, 0, 1)
            h_new = project_feasible(h_new, n)
            return h_new, t_var.value
    except Exception:
        pass
    return h, c5_max
```

```python
def do_augmented_lagrangian(h, n, budget_fn, max_outer=50):
    dx = 2.0 / n
    target = n / 2.0
    lam = 0.0
    mu = 1.0
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    
    bounds = [(0.0, 1.0)] * n
    
    for outer in range(max_outer):
        if budget_fn() < 10:
            break
        
        # Inner: minimize f(h) + mu*(sum(h)-target)^2 + lam*(sum(h)-target)
        def inner_obj(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            m = np.max(corr)
            beta = 500.0
            shifted = corr - m
            exps = np.exp(np.clip(beta * shifted, -500, 500))
            lse = (m + np.log(np.sum(exps)) / beta) * dx
            penalty = mu * (np.sum(h) - target)**2 + lam * (np.sum(h) - target)
            return lse + penalty
        
        res = minimize(inner_obj, h_cur, method='L-BFGS-B', 
                      bounds=bounds, options={'maxiter': 200, 'ftol': 1e-14})
        h_new = np.clip(res.x, 0, 1)
        
        # Update multipliers
        violation = np.sum(h_new) - target
        lam = lam + mu * violation
        if abs(violation) > 1e-6:
            mu = min(mu * 2, 1e6)
        
        # Project to feasible
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        
        if abs(violation) < 1e-8 and abs(c5_new - c5_cur) < 1e-10:
            h_cur = h_new
            c5_cur = c5_new
            break
        
        h_cur = h_new
        c5_cur = c5_new
    
    return h_cur, c5_cur
```

```python
def momentum_subgrad(h, n, budget_fn, max_iters=500, use_momentum=True):
    dx = 2.0 / n
    h_cur = h.copy()
    c5_best = compute_c5(h_cur)
    h_best = h_cur.copy()
    momentum = np.zeros(n)
    lr = 0.05
    
    for it in range(max_iters):
        if budget_fn() < 5:
            break
        corr_arr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        s = np.max(corr_arr)
        # Weighted subgradient
        active = np.where(corr_arr >= s - 0.01)[0]
        g = np.zeros(n)
        for m in active:
            offset = int(m) - (n-1)
            g += corr_grad(h_cur, n, offset)
        g /= len(active)
        g *= dx
        
        # Momentum update
        if use_momentum:
            momentum = 0.9 * momentum - lr * g
        else:
            momentum = -lr * g
        
        h_trial = h_cur + momentum
        h_trial = project_feasible(h_trial, n)
        c5_trial = compute_c5(h_trial)
        
        if c5_trial < c5_best:
            c5_best = c5_trial
            h_best = h_trial.copy()
            h_cur = h_trial
        else:
            # Adaptive step size reduction
            lr *= 0.95
            # Revert
            h_cur = h_best.copy()
    
    return h_best, c5_best
```
