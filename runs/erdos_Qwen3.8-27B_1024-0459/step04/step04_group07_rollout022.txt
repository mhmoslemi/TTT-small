    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):

        else:

    return h

    m = np.max(corr)
    # log-sum-exp for numerical stability
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 0)
    val = (m + np.log(np.sum(np.exp(shifted))) / beta) * dx
    return float(val)

    dx = 2.0 / n

    m = np.max(corr)
    shifted = 50.0 * (corr - m)
    shifted = np.clip(shifted, -500, 0)
    smooth_max = (m + np.log(np.sum(np.exp(shifted))) / 50.0)
    top_val = np.mean([corr[k] for k in top_lags])
    val = dx * (weight * top_val + (1 - weight) * smooth_max)
    return float(val)

    dx = 2.0 / n

            c5_p = compute_c5(h_pert)
            if c5_p < best_c5 * 1.03:
                h_cur = h_pert.copy()
                c5_cur = c5_p
                for bv in [500, 2000, 8000]:
                    if tlimit() < 8:
                        break
                    h_new = slsqp_run(h_cur, n, bv, maxiter=300)
                    c_new = compute_c5(h_new)
                    if c_new < c5_cur - 1e-12:
                        c5_cur = c_new
                        h_cur = h_new
                # Quick coord descent
                h_cur, c5_cur = coord_descent_fine(h_cur, n, max_sweeps=10, rng=rng,
                                                    tlimit=tlimit)
                if c5_cur < best_c5 - 1e-10:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n
                    print("  Perturb {}: C5={:.8f}".format(attempt, best_c5))

    # Phase 7: Active-lag iterative refinement
    if tlimit() > 60:
        print("\n=== Phase 7: active-lag iter n={} ===".format(best_n))
        n = best_n
        h_cur = best_h.copy()
        c5_cur = best_c5
        for rnd in range(20):
            if tlimit() < 10:
                break
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            k_active = int(np.argmax(corr))
            top_lags = get_top_lags(h_cur, n, count=3)
            thresh = c5_cur / (2.0 / n) * 0.999
            h_new = slsqp_run(h_cur, n, 5000, maxiter=350, use_topk=True,
                             top_lags=[k_active], thresh=thresh)
            c_new = compute_c5(h_new)
            if c_new < c5_cur - 1e-13:
                c5_cur = c_new
                h_cur = h_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n
        print("  Active-lag: C5={:.8f}".format(best_c5))

    # Phase 8: Final ultra-fine polish
    if tlimit() > 20:
        print("\n=== Phase 8: final polish ===")
        n = best_n
        finest_steps = [0.05, -0.05, 0.03, -0.03, 0.015, -0.015, 0.01, -0.01, 0.005, -0.005]
        best_h, best_c5 = coord_descent_fine(best_h, n, max_sweeps=40, rng=rng,
                                              tlimit=tlimit, step_sizes=finest_steps)
        # Final SLSQP
        if tlimit() > 10:
            for bv in [10000, 20000, 50000]:
                if tlimit() < 5:
                    break
                h_new = slsqp_run(best_h, n, bv, maxiter=500)
                c_new = compute_c5(h_new)
                if c_new < best_c5 - 1e-14:
                    best_c5 = c_new
                    best_h = h_new
        print("  Final: C5={:.8f}".format(best_c5))

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    _G_USE_CONST = False
    _G_TOP_LAGS = []
    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
