    # Select active lags
    top_idx = np.argsort(corr)[::-1][:n_active]
    vals = corr[top_idx] * dx
    m = vals[0]  # max
    
    # Soft max computation
    shifted = beta * (vals - m)
    shifted = np.minimum(shifted, 0.0)  # for numerical stability
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    obj = (m + np.log(wsum) / beta)
    
    # Gradient: d(corr[k])/dh for each active lag k
    grad = np.zeros(n)
    for idx in range(n_active):
        w = weights[idx] * dx
        if w < 1e-16:
            continue
        lag = top_idx[idx]
        offset = int(lag) - (n - 1)
        if offset >= 0:
            d = offset
            if d == 0:
                # corr[n-1] = sum h[i]*(1-h[i])
                # d/dh[j] = (1-h[j]) - h[j] = 1 - 2*h[j]
                grad += w * (1.0 - 2.0 * h)
            else:
                # corr[n-1+d] = sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
                # d/dh[j] = (1-h[j-d]) for j>=d, minus h[j+d] for j<n-d
                grad[d:] += w * (1.0 - h[:n-d])
                grad[:n-d] -= w * h[d:]
        else:
            d = -offset
            # corr[n-1-d] = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
            # d/dh[j] = (1-h[j+d]) for j<n-d, minus h[j-d] for j>=d
            grad[:n-d] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:n-d]
    
    return obj, grad

def projected_gradient_step(h, n, beta, n_active, lr):
    """One step of projected gradient descent."""
    obj, grad = soft_max_obj_grad(h, n, beta, n_active)
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new, obj

def proj_grad_optimize(h, n, beta_start, beta_end, n_active, n_iters, lr_start):

Let me also add a key trick: **alternating between different β values**. High β focuses on the max lag (like hard max). Low β spreads the weight across many lags (like L2 norm). Alternating can help escape local minima.

Another trick: **Lag-specific coordinate descent**. Instead of updating all coordinates, focus on the coordinates that most affect the maximum lag. If the max is at lag k, then the gradient is dominated by that lag, and the most impactful coordinates to change are those in the "overlap region" of lag k.

import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            room = (1.0 - h[mask]).min()
            adj = min(diff / cnt, room)
            if adj <= 0:
                break
            h[mask] += adj
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            room = h[mask].min()
            adj = min(-diff / cnt, room)
            if adj <= 0:
                break
            h[mask] -= adj
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def soft_max_obj_grad(h, n, beta, n_active):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    ncorr = len(corr)
    top_idx = np.argsort(corr)[::-1][:n_active]

    m = vals[0]

    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    obj = m + np.log(wsum) / beta

        elif offset > 0:

            grad[d:] += w * (1.0 - h[:n - d])
            grad[:n - d] -= w * h[d:]
        else:
            d = -offset
            grad[:n - d] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:n - d]
    return obj, grad
