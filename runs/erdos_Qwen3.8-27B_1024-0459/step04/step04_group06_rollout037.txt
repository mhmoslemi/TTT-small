        return grad * dx
    return obj, obj_grad

def slsqp_smooth(h, n, top_indices, beta, max_iter=200, ftol=1e-14):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    obj, obj_grad = make_smooth_obj(n, top_indices, beta)
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new)
    except Exception:
        return h.copy(), compute_c5(h)

def swap_local_search(h, n, max_iters=100000, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    c5_best = compute_c5(h)
    h_best = h.copy()
    Corr = compute_corr(h)
    dx = 2.0 / n
    no_improve = 0
    for it in range(max_iters):
        if _remaining_time() < 3.0:
            break
        # Pick random i, j
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if abs(h[i] - h[j]) < 1e-6:
            continue
        # Try swap
        h[i], h[j] = h[j], h[i]
        # Quick check: recompute corr efficiently
        Corr_new = compute_corr(h)
        c5_trial = float(np.max(Corr_new)) * dx
        if c5_trial < c5_best - 1e-13:
            c5_best = c5_trial
            h_best = h.copy()
            Corr = Corr_new
            no_improve = 0
        else:
            h[i], h[j] = h[j], h[i]
            no_improve += 1
            if no_improve > 5000:
                break
    return h_best, c5_best

def targeted_swap_search(h, n, max_iters=50000, rng=None):
    """Swap positions that are contributing to the max correlation."""
    if rng is None:
        rng = np.random.RandomState(123)
    c5_best = compute_c5(h)
    h_best = h.copy()
    dx = 2.0 / n
    no_improve = 0
    for it in range(max_iters):
        if _remaining_time() < 3.0:
            break
        corr = compute_corr(h)
        top_k = 5
        top_idx = np.argsort(corr)[-top_k:]
        # Find positions with high contribution to top correlations
        contrib = np.zeros(n)
        for k_idx in top_idx:
            offset = int(k_idx) - (n - 1)
            if offset >= 0:
                n_valid = n - offset
                if n_valid > 0:
                    contrib[offset:offset+n_valid] += h[:n_valid] * (1.0 - h[offset:offset+n_valid])
                    contrib[:n_valid] += h[offset:offset+n_valid] * (1.0 - h[:n_valid])
            else:
                m = -offset
                n_valid = n - m
                if n_valid > 0:
                    contrib[:n_valid] += h[m:m+n_valid] * (1.0 - h[:n_valid])
                    contrib[m:m+n_valid] += h[:n_valid] * (1.0 - h[m:m+n_valid])
        # Pick high-contribution position to change
        top_contrib_mask = contrib > np.percentile(contrib, 70)
        candidates = np.where(top_contrib_mask)[0]
        if len(candidates) == 0:
            candidates = np.arange(n)
        i = candidates[rng.randint(len(candidates))]
        j = rng.randint(n)
        if i == j:
            continue
        if abs(h[i] - h[j]) < 1e-6:
            continue
        h[i], h[j] = h[j], h[i]
        c5_trial = compute_c5(h)
        if c5_trial < c5_best - 1e-13:
            c5_best = c5_trial
            h_best = h.copy()
            no_improve = 0
        else:
            h[i], h[j] = h[j], h[i]
            no_improve += 1
            if no_improve > 10000:
                break
    return h_best, c5_best

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h

def make_golden(n):
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * phi) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return h

def make_random_bin(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def perturb_explore(h_best, n_best, rng, num_pert=10, n_target=512):
    c5_best = compute_c5(h_best)
    h_best = h_best.copy()
    n = n_target
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    for p in range(num_pert):
        if _remaining_time() < 8.0:
            break
        h_trial = h_best.copy()
        if n != n_best:
            h_trial = project_feasible(interp_to(h_best, n_best, n), n)
            h_best_n = n_best
        else:
            h_best_n = n
        pt = p % 5
        if pt == 0:
            start = rng.randint(0, max(1, n - 10))
            L = rng.randint(4, min(16, n // 4))
            h_trial[start:start + L] = 1.0 - h_trial[start:start + L]
        elif pt == 1:
            h_trial += 0.04 * rng.randn(n)
        elif pt == 2:
            subset = rng.choice(n, size=n // 5, replace=False)
            h_trial[subset] = rng.rand(len(subset))
        elif pt == 3:
            freq = rng.randint(2, n // 4)
            phase = rng.rand() * 2 * np.pi
            t = np.arange(n) / float(n)
            h_trial += 0.05 * np.sin(2.0 * np.pi * freq * t + phase)
        else:
            idx = rng.choice(n, size=n // 8, replace=False)
            h_trial[idx] = 1.0 - h_trial[idx]
        h_trial = project_feasible(h_trial, n)
        c5_t = compute_c5(h_trial)
        if c5_t < c5_best - 1e-12:
            c5_best = c5_t
            h_best = h_trial.copy()
        # Quick SLSQP polish
        top_idx = get_top_indices(h_trial, k=8)
        h_new, c5_new = slsqp_smooth(h_trial, n, top_idx, beta=300, max_iter=80, ftol=1e-13)
        if c5_new < c5_best - 1e-13:
            c5_best = c5_new
            h_best = h_new
    return h_best, c5_best, n

def coordinate_descent(h, n, max_sweeps=10, rng=None):
    c5_best = compute_c5(h)
    h_best = h.copy()
    if rng is None:
        rng = np.random.RandomState(0)
    target = n / 2.0
    for sweep in range(max_sweeps):
        if _remaining_time() < 3.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if _remaining_time() < 2.0:
                break
            old = h_best[i]
            # Try binary values
            for new_v in [0.0, 1.0]:
                if abs(new_v - old) < 1e-8:
                    continue
                # Compensate another position to maintain sum
                # Find a position to adjust
                diff = old - new_v
                h_trial = h_best.copy()
                h_trial[i] = new_v
                # Find best compensation position
                best_j = -1
                best_c5 = c5_best
                for j in rng.choice(n, size=min(20, n), replace=False):
                    if j == i:
                        continue
                    if h_best[j] + diff < 0 or h_best[j] + diff > 1:
                        continue
                    h_trial[j] = h_best[j] + diff
                    c5_t = compute_c5(h_trial)
                    if c5_t < best_c5:
                        best_c5 = c5_t
                        best_j = j
                h_trial = h_best.copy()
                if best_j >= 0:
                    h_trial[i] = new_v
                    h_trial[best_j] = h_best[best_j] + diff
                elif diff != 0:
                    h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < c5_best - 1e-13:
                    c5_best = c5_t
                    h_best = h_trial.copy()
                    any_imp = True
            if not any_imp and sweep > 3:
                break
    return h_best, c5_best

def multi_start_optimize(n, rng, max_time=120):
    """Try multiple initializations and keep the best."""
    best_h = None
    best_c5 = float('inf')
    target = n / 2.0
    inits = []
    # Thue-Morse
    inits.append(project_feasible(make_thue_morse(n), n))
    # Golden ratio
    inits.append(project_feasible(make_golden(n), n))
    # Random binary
    for _ in range(3):
        inits.append(project_feasible(make_random_bin(rng, n), n))
    for h_init in inits:
        if _remaining_time() < max(10.0, _G_BUDGET * 0.1):
            break
        # Quick subgradient
        h_opt, c5_opt = subgrad_momentum(h_init, n, max_iter=200, step0=0.1, momentum=0.85)
        # Quick SLSQP
        top_idx = get_top_indices(h_opt, k=8)
        h_opt, c5_opt = slsqp_smooth(h_opt, n, top_idx, beta=200, max_iter=100)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    return best_h, best_c5
