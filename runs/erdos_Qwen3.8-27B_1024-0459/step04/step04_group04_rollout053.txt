def structured_binary_init(n, rng, method='mod'):
    """Create a structured binary initial pattern."""
    target = n // 2
    if method == 'mod':
        # Use modular arithmetic for spread-out pattern
        p = 1
        while p < n:
            p += 1
            found = True
            for d in range(2, int(np.sqrt(p)) + 1):
                if p % d == 0:
                    found = False
                    break
            if found:
                break
        h = np.zeros(n)
        for i in range(n):
            if (i * p) % n < target:
                h[i] = 1.0
        # Fix count
        n_ones = int(np.sum(h))
        if n_ones > target:
            idx = np.where(h == 1)[0]
            h[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
        elif n_ones < target:
            idx = np.where(h == 0)[0]
            h[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1
    elif method == 'fourier':
        # Fourier-based pattern
        freqs = [1, 2, 3]
        phases = rng.uniform(0, 2 * np.pi, size=len(freqs))
        signal = np.zeros(n)
        for f, ph in zip(freqs, phases):
            signal += np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        h = (signal > np.median(signal)).astype(float)
        n_ones = int(np.sum(h))
        if n_ones > target:
            idx = np.where(h == 1)[0]
            h[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
        elif n_ones < target:
            idx = np.where(h == 0)[0]
            h[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1
    else:
        h = np.zeros(n)
        h[rng.choice(n, target, replace=False)] = 1.0
    return h

                 structured_binary_init(n2, np.random.RandomState(seed + 6000), 'mod')]
    t_per_720 = t2 / len(inits_720)

    for init_idx, h_init in enumerate(inits_720):
        if remaining() < 30:
            break
        h_sa2, c5_sa2 = sa_binary_multi_temp(h_init, n2, time_limit=t_per_720,
                                             seed=seed + 1000 * (init_idx + 1))
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
            best_n = n2

    # Continue SA on best n=720 solution
    if remaining() > 100:
        h_best720 = best_h if best_n == 720 else interp_to(best_h, best_n, 720)
        h_best720 = make_binary(h_best720, 720, rng)
        t_cont = min(150, remaining() * 0.3)
        for sa_seed in [seed + 7000, seed + 8000, seed + 9000]:
            if remaining() < 30:
                break
            h_sa2b, c5_sa2b = sa_binary(h_best720, 720, time_limit=t_cont / 3,
                                        seed=sa_seed, T_init=15.0, T_min=0.2, cool_rate=0.9997)
            if c5_sa2b < best_c5:
                best_c5 = c5_sa2b
                best_h = h_sa2b.copy()
                best_n = 720

    # ===== PHASE 3: Try n=540 =====
    if remaining() > 80:
        n3 = 540
        h_interp3 = interp_to(best_h, best_n, n3)
        h_bin3 = make_binary(h_interp3, n3, rng)
        t3 = min(100, remaining() * 0.15)
        for sa_seed in [seed + 10000, seed + 11000]:
            if remaining() < 25:
                break
            h_sa3, c5_sa3 = sa_binary(h_bin3, n3, time_limit=t3 / 2, seed=sa_seed,
                                      T_init=18.0, T_min=0.2, cool_rate=0.9996)
            if c5_sa3 < best_c5:
                best_c5 = c5_sa3
                best_h = h_sa3.copy()
                best_n = n3

    # ===== PHASE 4: Try n=900 if time permits =====
    if remaining() > 80:
        n4 = 900
        h_interp4 = interp_to(best_h, best_n, n4)
        h_bin4 = make_binary(h_interp4, n4, rng)
        t4 = min(100, remaining() * 0.15)
        for sa_seed in [seed + 12000, seed + 13000]:
            if remaining() < 25:
                break
            h_sa4, c5_sa4 = sa_binary(h_bin4, n4, time_limit=t4 / 2, seed=sa_seed,
                                      T_init=25.0, T_min=0.3, cool_rate=0.9995)
            if c5_sa4 < best_c5:
                best_c5 = c5_sa4
                best_h = h_sa4.copy()
                best_n = n4

    # ===== PHASE 5: Continuous refinement on best solution =====
    if remaining() > 40:
        n_ref = best_n
        h_ref = best_h.copy()
        if n_ref <= 600:
            # SLSQP at current resolution
            for beta in [500, 2000, 8000]:
                if remaining() < 15:
                    break
                h_new = slsqp_smooth(h_ref, n_ref, beta, maxiter=300)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n_ref
                    h_ref = h_new.copy()

            # Top-k gradient descent
            if remaining() > 15:
                h_td, c5_td = topk_gd(h_ref, n_ref, max_iters=200, top_k=5)
                if c5_td < best_c5:
                    best_c5 = c5_td
                    best_h = h_td.copy()
                    best_n = n_ref

    # ===== PHASE 6: Refine at higher resolution from continuous best =====
    if remaining() > 60 and best_n < 720:
        n_hi = 720
        h_interp_hi = interp_to(best_h, best_n, n_hi)
        h_interp_hi = project_feasible(h_interp_hi, n_hi)
        # SLSQP on interpolated
        for beta in [1000, 5000]:
            if remaining() < 15:
                break
            h_new = slsqp_smooth(h_interp_hi, n_hi, beta, maxiter=250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_hi

        # Binary SA from continuous
        if remaining() > 40:
            h_bin_hi = make_binary(best_h, n_hi, rng)
            h_sa_hi, c5_sa_hi = sa_binary(h_bin_hi, n_hi, time_limit=min(50, remaining() * 0.3),
                                          seed=seed + 20000, T_init=15.0, T_min=0.2,
                                          cool_rate=0.9997)
            if c5_sa_hi < best_c5:
                best_c5 = c5_sa_hi
                best_h = h_sa_hi.copy()
                best_n = n_hi

    # ===== PHASE 7: Final intensive SA on best =====
    if remaining() > 40:
        n_final = best_n
        h_final_bin = make_binary(best_h, n_final, rng)
        t_final = min(60, remaining() * 0.5)
        for sa_seed in [seed + 30000, seed + 31000, seed + 32000]:
            if remaining() < 15:
                break
            h_sf, c5_sf = sa_binary(h_final_bin, n_final, time_limit=t_final / 3,
                                    seed=sa_seed, T_init=12.0, T_min=0.15, cool_rate=0.9997)
            if c5_sf < best_c5:
                best_c5 = c5_sf
                best_h = h_sf.copy()
                best_n = n_final

    # ===== PHASE 8: Final polish with gradient =====
    if remaining() > 15:
        h_polish = best_h.copy()
        for beta in [3000, 10000]:
            if remaining() < 8:
                break
            hp = slsqp_smooth(h_polish, best_n, beta, maxiter=200)
            cp = compute_c5(hp)
            if cp < best_c5:
                best_c5 = cp
                best_h = hp.copy()

        if remaining() > 8:
            h_td, c5_td = topk_gd(best_h, best_n, max_iters=150, top_k=3)
            if c5_td < best_c5:
                best_c5 = c5_td
                best_h = h_td.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
