import time

_START = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Find closest to 0.5 and adjust
        i = np.argmin(np.abs(h - 0.5))
        h[i] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def corr_grad_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        m = n - off
        g[off:off+m] += (1.0 - x[:m])
        g[:m] -= x[off:off+m]
    else:
        ao = -off
        m = n - ao
        g[:m] += (1.0 - x[ao:ao+m])
        g[ao:ao+m] -= x[:m]
    return g

def focused_gradient(h, n, n_top=5, max_iter=5000):
    """Gradient descent focusing on the top n_top shifts."""
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    lr = 0.02
    stuck = 0
    
    for it in range(max_iter):
        if remaining() < 5:
            break
        
        corr = compute_corr(h)
        c5_curr = float(np.max(corr)) * dx
        
        if c5_curr < best_c5 - 1e-14:
            best_c5 = c5_curr
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > 200:
            lr *= 0.5
            stuck = 0
            if lr < 1e-8:
                break
        
        # Get top shifts
        top_idx = np.argsort(-corr)[:n_top]
        
        # Gradient: average of gradients of top shifts
        g = np.zeros(n)
        for k in top_idx:
            g += corr_grad_shift(h, n, int(k))
        g /= n_top
        g *= dx  # scale
        g -= np.mean(g)  # maintain sum
        
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        
        # Line search
        improved = False
        for factor in [1.0, 0.5, 0.25, 2.0, 0.1]:

            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-15:
                h = h_trial
                best_c5 = c_new
                best_h = h.copy()
                lr = step
                improved = True
                break
        
        if not improved:
            lr *= 0.7
    
    return best_h, best_c5

        m = n - ao
        g[:m] += (1.0 - x[ao:ao + m])
        g[ao:ao + m] -= x[:m]
    return g


def focused_gradient(h, n, n_top=5, max_iter=5000):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = 0.02
    stuck = 0

    for it in range(max_iter):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        c5_curr = float(np.max(corr)) * dx
        if c5_curr < best_c5 - 1e-14:
            best_c5 = c5_curr
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        if stuck > 300:
            lr *= 0.5
            stuck = 0
            if lr < 1e-9:
                break
        top_idx = np.argsort(-corr)[:n_top]
        g = np.zeros(n)
        for k in top_idx:
            g += corr_grad_shift(h, n, int(k))
        g /= n_top
        g *= dx
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        improved = False
        for factor in [1.0, 0.5, 0.25, 2.0, 0.1]:
            step = lr * factor
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-15:
                h = h_trial
                lr = step
                improved = True
                break
        if not improved:
            lr *= 0.7
        best_c5 = min(best_c5, compute_c5(h))
        best_h = h.copy() if best_c5 < compute_c5(best_h) else best_h
    return best_h, best_c5


def softmax_gradient(h, n, max_iter=8000):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = 0.025
    stuck = 0

    for it in range(max_iter):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        m = float(np.max(corr))
        c5_curr = m * dx
        if c5_curr < best_c5 - 1e-14:
            best_c5 = c5_curr
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        if stuck > 400:
            lr *= 0.5
            stuck = 0
            if lr < 1e-9:
                break
        beta = max(80.0, 30.0 + 0.005 * it)
        shifted = corr - m
        cl = np.clip(beta * shifted, -50, 50)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        threshold = 3.0 / len(corr)
        active = np.where(w > threshold)[0]
        g = np.zeros(n)
        for k in active:
            g += w[int(k)] * corr_grad_shift(h, n, int(k))
        g *= dx
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        improved = False
        for factor in [1.0, 0.5, 0.25, 2.0]:
            step = lr * factor
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-15:
                h = h_trial
                lr = step
                improved = True
                break
        if not improved:
            lr *= 0.7
    return best_h, best_c5


def binary_sa(h, n, time_limit=60, seed=42, T_init=4.0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1

    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()
    T = T_init
    alpha = 0.9992
    start = time.time()
    stall = 0

    while time.time() - start < time_limit and T > 0.02:
        if remaining() < 10:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        r = rng.rand()
        if r < 0.7:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            delta = new_max - cur_max
            if delta < 0 or rng.rand() < np.exp(max(-delta / T, -60)):
                cur_max = new_max
                corr = new_corr
                if new_max < best_max:
                    best_max = new_max
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                h[i], h[j] = 1.0, 0.0
                stall += 1
        else:
            i1 = ones[rng.randint(len(ones))]
            j1 = zeros[rng.randint(len(zeros))]
            i2 = ones[rng.randint(len(ones))]
            j2 = zeros[rng.randint(len(zeros))]
            if i1 != i2 and j1 != j2 and i1 != j2 and i2 != j1:
                h[i1] = 0.0
                h[i2] = 0.0
                h[j1] = 1.0
                h[j2] = 1.0
                new_corr = compute_corr(h)
                new_max = float(np.max(new_corr))
                delta = new_max - cur_max
                if delta < 0 or rng.rand() < np.exp(max(-delta / T, -60)):
                    cur_max = new_max
                    corr = new_corr
                    if new_max < best_max:
                        best_max = new_max
                        best_h = h.copy()
                        stall = 0
                    else:
                        stall += 1
                else:
                    h[i1] = 1.0
                    h[i2] = 1.0
                    h[j1] = 0.0
                    h[j2] = 0.0
                    stall += 1
        T *= alpha
        if stall > 6000:
            T = max(T, 2.0)
            stall = 0
    return best_h, best_max * dx


def cont_binary_alternate(h, n, time_limit=120, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()

    while time.time() - start < time_limit and remaining() > 15:
        phase_time = min(20, remaining() * 0.3, (time_limit - (time.time() - start)) * 0.4)
        if phase_time < 3:
            break
        h_g, c5_g = focused_gradient(h, n, n_top=5, max_iter=int(50 * phase_time))
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
        h = h_g.copy()
        if remaining() < 15:
            break
        h_bin = (h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        target = n // 2
        if n_ones > target:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
        elif n_ones < target:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1
        h_sa, c5_sa = binary_sa(h_bin, n, time_limit=min(8, remaining() * 0.2),
                                seed=int(rng.randint(1000000)), T_init=2.0)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            h = h_sa.astype(float).copy()
        else:
            h = h_g.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 360
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Focused continuous gradient at parent resolution
    if remaining() > 60:
        h_g, c5_g = focused_gradient(best_h, best_n, n_top=5, max_iter=6000)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 2: Softmax gradient at parent resolution
    if remaining() > 60:
        h_g, c5_g = softmax_gradient(best_h, best_n, max_iter=8000)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 3: Binary SA at parent resolution
    if remaining() > 50:
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        target = best_n // 2
        if n_ones > target:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
        elif n_ones < target:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1
        for s in [seed, seed + 500, seed + 1000]:
            if remaining() < 20:
                break
            h_sa, c5_sa = binary_sa(h_bin, best_n, time_limit=min(20, remaining() * 0.15), seed=s)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()

    # Phase 4: Interpolate to n=500, optimize
    n_mid = 500
    if remaining() > 120:
        h_mid = interp_to(best_h, best_n, n_mid)
        h_mid = project_feasible(h_mid, n_mid)
        h_g, c5_g = softmax_gradient(h_mid, n_mid, max_iter=6000)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
            best_n = n_mid
        if remaining() > 40:
            h_bin_mid = (best_h > 0.5).astype(float) if best_n == n_mid else (h_mid > 0.5).astype(float)
            n_ones = int(np.sum(h_bin_mid))
            target_m = n_mid // 2
            if n_ones > target_m:
                idx = np.where(h_bin_mid == 1)[0]
                h_bin_mid[idx[rng.choice(len(idx), n_ones - target_m, replace=False)]] = 0
            elif n_ones < target_m:
                idx = np.where(h_bin_mid == 0)[0]
                h_bin_mid[idx[rng.choice(len(idx), target_m - n_ones, replace=False)]] = 1
            h_sa, c5_sa = binary_sa(h_bin_mid, n_mid, time_limit=min(25, remaining() * 0.2), seed=seed + 2000)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_mid

    # Phase 5: Interpolate to n=720, optimize
    n_high = 720
    if remaining() > 150:
        h_hi = interp_to(best_h, best_n, n_high)
        h_hi = project_feasible(h_hi, n_high)
        h_g, c5_g = softmax_gradient(h_hi, n_high, max_iter=8000)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
            best_n = n_high
        if remaining() > 60:
            h_g2, c5_g2 = focused_gradient(best_h, best_n, n_top=3, max_iter=6000)
            if c5_g2 < best_c5:
                best_c5 = c5_g2
                best_h = h_g2.copy()
        if remaining() > 40:
            h_bin_hi = (best_h if best_n == n_high else h_hi > 0.5).astype(float)
            if best_n != n_high:
                h_bin_hi = (h_hi > 0.5).astype(float)
            n_ones = int(np.sum(h_bin_hi))
            target_h = n_high // 2
            if n_ones > target_h:
                idx = np.where(h_bin_hi == 1)[0]
                h_bin_hi[idx[rng.choice(len(idx), n_ones - target_h, replace=False)]] = 0
            elif n_ones < target_h:
                idx = np.where(h_bin_hi == 0)[0]
                h_bin_hi[idx[rng.choice(len(idx), target_h - n_ones, replace=False)]] = 1
            for s in [seed + 3000, seed + 3500]:
                if remaining() < 25:
                    break
                h_sa, c5_sa = binary_sa(h_bin_hi, n_high, time_limit=min(30, remaining() * 0.2), seed=s)
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
                    best_n = n_high

    # Phase 6: Continuous-binary alternate at best resolution
    if remaining() > 80:
        h_alt, c5_alt = cont_binary_alternate(best_h, best_n, time_limit=min(100, remaining() * 0.4), seed=seed + 7000)
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt.copy()

    # Phase 7: Final focused gradient polish
    if remaining() > 30:
        h_g, c5_g = focused_gradient(best_h, best_n, n_top=3, max_iter=5000)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 8: Final binary SA
    if remaining() > 20:
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        target = best_n // 2
        if n_ones > target:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
        elif n_ones < target:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1
        h_sa, c5_sa = binary_sa(h_bin, best_n, time_limit=min(15, remaining() * 0.5), seed=seed + 9999)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Final: one more gradient step
    if remaining() > 10:
        h_g, c5_g = focused_gradient(best_h, best_n, n_top=3, max_iter=2000)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
