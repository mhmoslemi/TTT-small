    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-9
        else:
            mask = h > 1e-9
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def obj_slsqp(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    b = _G_BETA
    s = corr - m
    e = np.exp(np.clip(b * s, -400, 400))
    val = (m + np.log(np.sum(e)) / b) * _G_DX
    if _G_PEN > 0:
        val += _G_PEN * (np.sum(x) - _G_TGT) ** 2
    if _G_FOCUS >= 0 and _G_FOCUS < len(corr):
        val += 0.005 * corr[_G_FOCUS] * _G_DX
    return float(val)

def slsqp_run(h0, n, beta, maxiter):
    global _G_BETA, _G_TGT, _G_DX, _G_PEN, _G_FOCUS
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    _G_BETA = float(beta)
    _G_PEN = 0.0
    _G_FOCUS = -1
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    try:
        res = minimize(obj_slsqp, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def projected_gradient_minmax(h, n, tlimit, max_iter=20000):
    """Projected subgradient method for min-max with active set tracking."""
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    
    K = 8  # number of active lags to track
    beta_base = 100.0
    beta_growth = 1.003
    lr_base = 0.008
    
    beta = beta_base
    lr = lr_base
    stall_count = 0
    
    for it in range(max_iter):
        if tlimit() < 3.0:
            break
        
        # Compute correlation
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        cmax = np.max(corr)
        
        # Get top K lags
        n_corr = len(corr)
        top_k = np.argsort(corr)[-K:][::-1]
        
        # Soft weights
        shifts = corr[top_k] - cmax
        weights = np.exp(beta * shifts)
        wsum = np.sum(weights)
        if wsum < 1e-300:
            weights = np.zeros(K)
            weights[0] = 1.0
        else:
            weights /= wsum
        
        # Compute gradient: sum of w_m * d(c[m])/d(h)
        grad = np.zeros(n)
        for idx in range(K):
            m_idx = top_k[idx]
            m = m_idx - (n - 1)  # actual lag
            w = weights[idx]
            if m >= 0:
                # c[m] = sum_{j=0}^{n-1-m} h[j]*(1-h[j+m])
                # dc/dh[j] += (1-h[j+m]) for j=0..n-1-m
                # dc/dh[j] -= h[j-m] for j=m..n-1
                grad[:n-m] += w * (1.0 - h_cur[m:])
                grad[m:] -= w * h_cur[:n-m]
            else:
                mp = -m  # positive
                # c[-mp] = sum_{j=mp}^{n-1} h[j]*(1-h[j-mp])
                # dc/dh[j] += (1-h[j-mp]) for j=mp..n-1
                # dc/dh[j] -= h[j+mp] for j=0..n-1-mp
                grad[mp:] += w * (1.0 - h_cur[:n-mp])
                grad[:n-mp] -= w * h_cur[mp:]
        
        # Scale gradient
        grad *= dx
        
        # Projected step
        h_new = h_cur - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cur = h_new
            stall_count = 0
        elif c5_new <= compute_c5(h_cur) + 1e-13:
            h_cur = h_new
            stall_count += 1
        else:
            stall_count += 1
            # Try smaller step
            h_new2 = h_cur - 0.5 * lr * grad
            h_new2 = np.clip(h_new2, 0.0, 1.0)
            h_new2 = project_feasible(h_new2, n)
            if compute_c5(h_new2) < compute_c5(h_cur):
                h_cur = h_new2
                c5_chk = compute_c5(h_cur)
                if c5_chk < best_c5:
                    best_c5 = c5_chk
                    best_h = h_cur.copy()
        
        # Adaptive parameters
        if stall_count > 30:
            lr *= 0.7
            beta *= beta_growth
            stall_count = 0
        if it % 500 == 0 and it > 0:
            beta *= 1.1
            lr = max(lr * 0.98, 0.0005)
        
        # Periodic restart from best
        if it % 3000 == 0 and it > 0:
            h_cur = best_h.copy()
    
    return best_h, best_c5

def binary_swap_search(h, n, tlimit, max_sweeps=50):
    """Binary local search: swap 1s and 0s to reduce C5."""
    h_cur = h.copy()
    # Binarize
    h_bin = np.round(h_cur)
    h_bin = project_feasible(h_bin, n)
    
    # If not binary enough, force binary
    ones = np.where(h_bin > 0.5)[0]
    zeros = np.where(h_bin < 0.5)[0]
    # Adjust to have exactly n//2 ones
    n_ones = len(ones)
    target_ones = n // 2
    if n_ones > target_ones:
        # Remove excess ones
        excess = n_ones - target_ones
        to_remove = ones[:excess]
        h_bin[to_remove] = 0.0
        to_add = zeros[target_ones - n_ones:]
        if len(to_add) > 0:
            h_bin[to_add[:excess]] = 1.0
    elif n_ones < target_ones:
        need = target_ones - n_ones
        to_add = zeros[:need]
        h_bin[to_add] = 1.0
        to_remove = ones[need:]
        if len(to_remove) > 0:
            h_bin[to_remove[:need]] = 0.0
    
    h_bin = project_feasible(h_bin, n)
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    
    # Also keep the original continuous version
    c5_cont = compute_c5(h)
    
    rng_local = np.random.RandomState(12345)
    
    for sweep in range(max_sweeps):
        if tlimit() < 5.0:
            break
        ones = np.where(h_cur > 0.5)[0]
        zeros = np.where(h_cur < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        order = rng_local.permutation(len(ones))
        for oi in order:
            if tlimit() < 3.0:
                break
            i = ones[oi]
            # Try swapping with a random zero
            for attempt in range(3):
                j = zeros[rng_local.randint(len(zeros))]
                h_try = h_cur.copy()
                h_try[i] = 0.0
                h_try[j] = 1.0
                h_try = clip_array(h_try)
                c = compute_c5(h_try)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    best_h = h_try.copy()
                    h_cur = h_try
                    ones = np.where(h_cur > 0.5)[0]
                    zeros = np.where(h_cur < 0.5)[0]
                    improved = True
                    break
    return best_h, best_c5

def clip_array(h):
    return np.clip(h, 0.0, 1.0)

def coord_descent_improved(h, n, tlimit, max_sweeps=30):
    """Improved coordinate descent with line search."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng_local = np.random.RandomState(777)
    
    for sweep in range(max_sweeps):
        if tlimit() < 3.0:
            break
        improved = False
        order = rng_local.permutation(n)
        for i in order:
            if tlimit() < 2.0:
                break
            v0 = best_h[i]
            # Try binary values
            for v in [0.0, 1.0]:
                if abs(v - v0) < 1e-8:
                    continue
                trial = best_h.copy()
                trial[i] = v
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    best_h = trial
                    improved = True
                    v0 = v
            # Try continuous values via line search
            for v in [0.1, 0.3, 0.5, 0.7, 0.9]:
                if abs(v - v0) < 0.05:
                    continue
                trial = best_h.copy()
                trial[i] = v
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    best_h = trial.copy()
                    improved = True
        if not improved and sweep >= 3:
            break
    return best_h, best_c5

def focused_slsqp(h, n, tlimit, max_rounds=20):
    """SLSQP focused on the active lag."""
    global _G_FOCUS, _G_BETA, _G_TGT, _G_DX
    best_h = h.copy()
    best_c5 = compute_c5(h)
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    _G_PEN = 0.0
    
    for rnd in range(max_rounds):
        if tlimit() < 5.0:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        k_active = int(np.argmax(corr))
        
        for focus in [k_active, k_active - 1, k_active + 1]:
            if focus < 0 or focus >= len(corr):
                continue
            _G_FOCUS = focus
            _G_BETA = 500.0
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
            bounds = [(0.0, 1.0)] * n
            try:
                res = minimize(obj_slsqp, best_h, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 150, 'ftol': 1e-14, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new)
                if c_new < best_c5 - 1e-13:
                    best_c5 = c_new
                    best_h = h_new
            except Exception:
                pass
        _G_FOCUS = -1
    return best_h, best_c5

def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c else 0.0
    return project_feasible(h, n)

def make_rand_bin(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def multi_start_optimize(h_best, c5_best, n_best, n_target, tlimit, rng, phase_time):
    """Optimize at n_target starting from multiple initializations."""
    times = tlimit()
    if times < 20:
        return h_best, c5_best, n_best
    
    # Interpolate best to target resolution
    h_interp = project_feasible(interp_h(h_best, n_best, n_target), n_target)
    c5_interp = compute_c5(h_interp)
    
    # Thue-Morse at target
    h_tm = make_tm(n_target)
    c5_tm = compute_c5(h_tm)
    
    # Random binary
    h_rb = make_rand_bin(rng, n_target)
    c5_rb = compute_c5(h_rb)
    
    # Pick best init
    inits = [(h_interp, c5_interp, 'interp'), (h_tm, c5_tm, 'tm'), (h_rb, c5_rb, 'rand')]
    inits.sort(key=lambda x: x[1])
    
    h_cur = inits[0][0].copy()
    c5_cur = inits[0][1]
    
    # Projected gradient
    h_cur, c5_cur = projected_gradient_minmax(h_cur, n_target, tlimit, max_iter=8000)
    
    if tlimit() < 15:
        pass
    else:
        # SLSQP polish
        for bv in [200, 800, 3000]:
            if tlimit() < 10:
                break
            h_new = slsqp_run(h_cur, n_target, bv, maxiter=300)
            c_new = compute_c5(h_new)
            if c_new < c5_cur - 1e-13:
                c5_cur = c_new
                h_cur = h_new
    
    if c5_cur < c5_best - 1e-12:
        return h_cur.copy(), c5_cur, n_target
    
    # Also try from second-best init
    if tlimit() > 40:
        for init_idx in range(1, len(inits)):
            if tlimit() < 20:
                break
            h_alt = inits[init_idx][0].copy()
            c5_alt = inits[init_idx][1]
            h_alt, c5_alt = projected_gradient_minmax(h_alt, n_target, tlimit, max_iter=5000)
            if tlimit() > 10:
                h_alt = slsqp_run(h_alt, n_target, 1000, maxiter=200)
                c5_alt = compute_c5(h_alt)
            if c5_alt < c5_cur - 1e-12:
                c5_cur = c5_alt
                h_cur = h_alt
    
    if c5_cur < c5_best - 1e-12:
        return h_cur.copy(), c5_cur, n_target
    
    return h_best, c5_best, n_best

def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET, _G_PEN, _G_FOCUS
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    _G_PEN = 0.0
    _G_FOCUS = -1
    
    def tlimit():
        return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)
    
    rng = np.random.RandomState(seed)
    
    # Initialize from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, best_c5))
    except Exception:
        parent_n = 256
        best_h = make_tm(parent_n)
        best_c5 = compute_c5(best_h)
        best_n = parent_n
        print("No parent. TM: n={}, C5={:.8f}".format(parent_n, best_c5))
    
    # Phase 1: Projected gradient at parent resolution
    if tlimit() > 100:
        print("\n=== Phase 1: PG minmax n={} ===".format(best_n))
        best_h, best_c5 = projected_gradient_minmax(best_h, best_n, tlimit, max_iter=15000)
        best_n = len(best_h)
        print("  After PG: C5={:.8f}".format(best_c5))
    
    # Phase 2: SLSQP multi-beta at parent resolution
    if tlimit() > 80:

        for bv in [100, 300, 1000, 3000, 8000]:

            h_new = slsqp_run(best_h, best_n, bv, maxiter=350)
            c_new = compute_c5(h_new)

                best_c5 = c_new
                best_h = h_new
        print("  After SLSQP: C5={:.8f}".format(best_c5))
    
    # Phase 3: Focused SLSQP
    if tlimit() > 40:
        print("\n=== Phase 3: Focused n={} ===".format(best_n))
        best_h, best_c5 = focused_slsqp(best_h, best_n, tlimit, max_rounds=15)
        best_n = len(best_h)
        print("  After focused: C5={:.8f}".format(best_c5))
    
    # Phase 4: Coordinate descent

        print("\n=== Phase 4: Coord descent n={} ===".format(best_n))
        best_h, best_c5 = coord_descent_improved(best_h, best_n, tlimit, max_sweeps=25)
        best_n = len(best_h)

        best_h, best_c5, best_n = multi_start_optimize(best_h, best_c5, best_n, n_high, tlimit, rng, 100)
        print("  After high res: n={}, C5={:.8f}".format(best_n, best_c5))
    
    # Phase 6: Go back to best_n and refine
    if tlimit() > 60:
        print("\n=== Phase 6: Refine n={} ===".format(best_n))
        best_h, best_c5 = projected_gradient_minmax(best_h, best_n, tlimit, max_iter=10000)
        best_n = len(best_h)
        
        if tlimit() > 30:
            best_h, best_c5 = coord_descent_improved(best_h, best_n, tlimit, max_sweeps=20)
            best_n = len(best_h)
        print("  After refine: C5={:.8f}".format(best_c5))
    
    # Phase 7: Binary search + polish
    if tlimit() > 30:
        print("\n=== Phase 7: Binary + polish n={} ===".format(best_n))
        # Try binary version
        h_bin = np.round(best_h)
        h_bin = project_feasible(h_bin, best_n)
        c5_bin = compute_c5(h_bin)
        
        if c5_bin > best_c5:
            # Binary is worse, skip
            pass
        else:
            best_h = h_bin
            best_c5 = c5_bin
        
        # Binary swap search
        h_bs, c5_bs = binary_swap_search(best_h, best_n, tlimit, max_sweeps=30)
        if c5_bs < best_c5:
            best_c5 = c5_bs
            best_h = h_bs
        print("  After binary: C5={:.8f}".format(best_c5))
    
    # Phase 8: Final polish
    if tlimit() > 15:
        print("\n=== Phase 8: Final polish ===")
        for bv in [5000, 10000]:
            if tlimit() < 8:
                break
            h_new = slsqp_run(best_h, best_n, bv, maxiter=400)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new
        
        # Final coordinate descent
        if tlimit() > 10:
            best_h, best_c5 = coord_descent_improved(best_h, best_n, tlimit, max_sweeps=15)
        
        # One more PG pass
        if tlimit() > 10:
            best_h, best_c5 = projected_gradient_minmax(best_h, best_n, tlimit, max_iter=5000)

    # Final validation

    try:

        pass
    
    _G_PEN = 0.0
    _G_FOCUS = -1
    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
