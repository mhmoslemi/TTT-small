import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_k(h, k, n):
    """Gradient of corr[k] w.r.t. h."""
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        # h[j+offset]*(1-h[j]) for j=0..npairs-1
        # d/dh[i]: (1-h[i-offset]) for i in [offset, offset+npairs-1] = [offset, n-1]
        #          -h[i+offset] for i in [0, npairs-1]
        g[offset:n] += (1.0 - h[:npairs])  # i-offset ranges over 0..npairs-1 when i ranges over offset..n-1
        g[:npairs] -= h[offset:n]  # i+offset ranges over offset..n-1 when i ranges over 0..npairs-1
    else:
        a = -offset
        npairs = n - a
        # h[j]*(1-h[j+a]) for j=0..npairs-1
        # d/dh[i]: (1-h[i+a]) for i in [0, npairs-1]
        #          -h[i-a] for i in [a, n-1]
        g[:npairs] += (1.0 - h[a:n])  # i+a ranges over a..n-1 when i ranges over 0..npairs-1
        g[a:n] -= h[:npairs]  # i-a ranges over 0..npairs-1 when i ranges over a..n-1
    return g

def cutting_plane_step(h, n, K=30, max_iter_lp=100):
    """One cutting plane iteration."""
    corr = compute_corr(h)
    c_max = np.max(corr)
    dx = 2.0 / n
    
    # Select top K lags
    idx = np.argsort(-corr)[:K]
    
    # Also include lags within 1% of max
    threshold = c_max - 0.01 * c_max
    extra = np.where(corr >= threshold)[0]
    idx = np.unique(np.concatenate([idx, extra]))
    idx = idx[:50]  # Cap at 50 constraints
    
    # Build LP
    # Variables: [h_0, h_1, ..., h_{n-1}, t]
    # Minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = []
    b_ub = []
    
    for k in idx:
        g = grad_corr_k(h, k, n)
        fk = corr[k]
        # Linearization: fk + g·(x - h) <= t
        # g·x - t <= -fk + g·h
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub.append(row)
        b_ub.append(-fk + np.dot(g, h))
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Equality constraint: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            h_new = res.x[:n]
            t_new = res.x[-1]
            return h_new, t_new, len(idx)
    except Exception:
        pass
    
    return h, c_max, 0

def slsqp_smooth(h, n, beta, maxiter=300):
    """SLSQP with smooth max approximation."""
    def sm_obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)
    
    def sm_grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    
    def sc(x):
        return np.sum(x) - n / 2.0
    
    def scj(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc, 'jac': scj}]
    
    try:
        res = minimize(sm_obj, h, jac=sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.success or res.fun < sm_obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def compute_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g

        if remaining() > 8:
            for beta in [5000, 15000, 30000]:
                if remaining() < 5:
                    break
                h = smooth_max_slsqp(h, n, beta, maxiter=500)
            c5_tmp = compute_c5(h)
            if c5_tmp < best_c5:
                best_c5 = c5_tmp
                best_h = h.copy()

        # Cutting plane
        if remaining() > 15:
            for it in range(100):
                if remaining() < 5:
                    break
                h_lp, t_lp = cutting_plane_lp(h, n, K=30, delta=0.03)
                if t_lp < compute_c5(h) - 1e-10:
                    best_val = compute_c5(h)
                    for alpha in [1.0, 0.5, 0.3, 0.7]:
                        h_try = project_feasible(h + alpha * (h_lp - h), n)
                        c_try = compute_c5(h_try)
                        if c_try < best_val:
                            best_val = c_try
                            h = h_try
                        else:
                            h = project_feasible(h + alpha * (h_lp - h), n)
                            break
                    if compute_c5(h) < best_c5:
                        best_c5 = compute_c5(h)
                        best_h = h.copy()
                elif it > 50:
                    break

    # Phase 7: Final polish
    for beta in [20000, 40000, 80000]:
        if remaining() < 5:
            break
        h = smooth_max_slsqp(h, n, beta, maxiter=600)
        c5_tmp = compute_c5(h)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h.copy()

    # Final equalize
    if remaining() > 10:
        h_eq, c5_eq = equalize_top_shifts(h, n, K=10, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Final top-K gradient descent (fine-tuned)
    if remaining() > 5:
        for it in range(50):
            if remaining() < 3:
                break
            g, c_max = top_k_gradient(best_h, best_n, K=15, beta=100.0)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            for step in [0.01, 0.005, 0.02, 0.003, 0.008]:
                h_try = project_feasible(best_h - step * g / gn, best_n)
                c_try = compute_c5(h_try)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    break
                elif it % 10 == 9:
                    best_h = project_feasible(best_h - step * g / gn, best_n)
                    best_c5 = compute_c5(best_h)

    # Ensure final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
