def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            step = diff / cnt
            # Don't overshoot
            max_step_up = 1.0 - 1e-10
            max_step_dn = 1e-10
            if step > 0:
                room = max_step_up - h[mask]
                max_s = np.max(room)
                if max_s <= 0:
                    h[mask] = max_step_up
                    continue
                if abs(step) > max_s:
                    step = max_s * 0.5
                else:
                    step = min(step, max_s * 0.9)
            else:
                room = h[mask] - max_step_dn
                max_s = np.max(room)
                if max_s <= 0:
                    h[mask] = max_step_dn
                    continue
                if abs(step) > max_s:
                    step = -max_s * 0.5
                else:
                    step = max(step, -max_s * 0.9)
            h[mask] += step
            h = np.clip(h, 0.0, 1.0)
        else:
            break
    # Final fix: adjust one element to hit exact sum
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            for i in idx:
                change = diff
                new_val = h[i] + change
                if 0.0 <= new_val <= 1.0:
                    h[i] = new_val
                    diff = 0.0
                    break
    return h

    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 3.0)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cdx = corr * dx
    max_cd = np.max(cdx)
    shifted = cdx - max_cd
    log_w = beta * np.minimum(shifted, 0.0)
    # Subtract max for numerical stability
    log_w -= np.max(log_w)
    weights = np.exp(log_w)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        # All equal, uniform weights
        weights = np.ones(2*n-1) / (2*n-1)
    else:
        weights /= wsum
    
    offsets = np.arange(-(n-1), n)
    grad = np.zeros(n)
    omh = 1.0 - h  # one minus h

        idx2 = m - offsets  # m - offset, must be in [0, n)

        s2 = np.dot(weights[mask2], h[idx2[mask2]])
        grad[m] = dx * (s1 - s2)

def simulated_annealing(h, n, rng, t_start, t_end, n_iter):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    h = h.copy()
    current_c5 = best_c5
    
    for it in range(n_iter):
        if remaining_time() < 5:
            break
        temperature = t_start * (t_end / t_start) ** (it / max(n_iter-1, 1))
        
        # Generate mutation
        h_trial = h.copy()
        mut_type = rng.randint(4)
        if mut_type == 0:
            # Flip a block
            L = rng.randint(2, 7)
            s = rng.randint(0, n - L)
            h_trial[s:s+L] = 1.0 - h_trial[s:s+L]
        elif mut_type == 1:
            # Swap two blocks
            L = rng.randint(2, min(6, n//8))
            s1 = rng.randint(0, n//2)
            s2 = rng.randint(n//2, n - L)
            h_trial[s1:s1+L], h_trial[s2:s2+L] = h_trial[s2:s2+L].copy(), h_trial[s1:s1+L].copy()
        elif mut_type == 2:
            # Gaussian perturbation
            L = rng.randint(2, 5)
            s = rng.randint(0, n - L)
            h_trial[s:s+L] += 0.1 * rng.randn(L)
        else:
            # Single point perturbation
            s = rng.randint(n)
            h_trial[s] = rng.rand()
        
        h_trial = project_feasible(h_trial, n)
        c5_trial = compute_c5(h_trial)
        
        # Metropolis acceptance
        delta = c5_trial - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(temperature, 1e-10)):
            h = h_trial
            current_c5 = c5_trial
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h.copy()
    
    return best_h, best_c5

            offset = int(k_idx) - (n - 1)
            if offset >= 0:
                nv = n - offset
                if nv <= 0:
                    continue
                grad[offset:offset + nv] += (1.0 - h[:nv])
                grad[:nv] -= h[offset:offset + nv]
            else:
                m = -offset
                nv = n - m
                if nv <= 0:
                    continue
                grad[:nv] += (1.0 - h[m:m + nv])
                grad[m:m + nv] -= h[:nv]
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        alpha = step
        improved = False
        for _ in range(12):
            v = momentum * v - alpha * grad
            h_trial = h + v
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-14:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
            alpha *= 0.5
        if improved:
            patience = 0
            step = min(alpha * 1.3, 0.3)
        else:
            patience += 1
            step *= 0.6
            v *= 0.5
            if patience > 15:
                break

    return best_h, best_c5


def simulated_annealing(h, n, rng, t_start, t_end, n_iter):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    h = h.copy()
    curr_c5 = best_c5

    for it in range(n_iter):
        if remaining_time() < 5.0:
            break
        frac = it / max(n_iter - 1, 1)
        temp = t_start * (t_end / t_start) ** frac

        h_trial = h.copy()
        mut_type = rng.randint(5)
        if mut_type == 0:
            L = rng.randint(2, 7)
            s = rng.randint(0, n - L)
            h_trial[s:s + L] = 1.0 - h_trial[s:s + L]
        elif mut_type == 1:
            L = rng.randint(2, min(7, n // 8))
            s1 = rng.randint(0, n // 2)
            s2 = rng.randint(n // 2, n - L)
            a = h_trial[s1:s1 + L].copy()
            h_trial[s1:s1 + L] = h_trial[s2:s2 + L]
            h_trial[s2:s2 + L] = a
        elif mut_type == 2:
            L = rng.randint(2, 6)
            s = rng.randint(0, n - L)
            h_trial[s:s + L] += 0.15 * rng.randn(L)
        elif mut_type == 3:
            idx = rng.randint(n)
            h_trial[idx] = rng.rand()
        else:
            L = rng.randint(3, min(10, n // 6))
            s = rng.randint(0, n - L)
            h_trial[s:s + L] = rng.rand(L)

        h_trial = project_feasible(h_trial, n)
        c5_trial = compute_c5(h_trial)

        delta = c5_trial - curr_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(temp, 1e-12)):
            h = h_trial
            curr_c5 = c5_trial
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h.copy()

    return best_h, best_c5


def slsqp_smooth(h, n, beta, maxiter=200):
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        cdx = corr * dx
        m = np.max(cdx)
        shifted = cdx - m
        exps = np.exp(np.clip(beta * shifted, -500, 500))
        return (m + np.log(np.sum(exps)) / beta) * dx

    def obj_grad(x):
        g = softmax_gradient(x, n, beta)
        return g

    def con_sum(x):
        return np.sum(x) - target

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        return h_new, c5_new
    except Exception:
        return h.copy(), compute_c5(h)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def coord_descent_small(h, n, rng, max_sweeps=8):
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for sweep in range(max_sweeps):
        if remaining_time() < 3.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if remaining_time() < 2.0:
                break
            old = best_h[i]
            candidates = [0.0, 1.0, old + 0.04, old - 0.04, old + 0.02, old - 0.02]
            for nv in candidates:
                if nv < 0 or nv > 1 or abs(nv - old) < 1e-9:
                    continue
                h_trial = best_h.copy()
                h_trial[i] = nv
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < best_c5 - 1e-13:
                    best_c5 = c5_t
                    best_h = h_trial.copy()
                    any_imp = True
                    break
        if not any_imp and sweep > 1:
            break

    return best_h, best_c5


def block_optimize(h, n, rng, n_blocks=20, block_sizes=(2, 3, 4)):
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for _ in range(n_blocks):
        if remaining_time() < 4.0:
            break
        L = block_sizes[rng.randint(len(block_sizes))]
        s = rng.randint(0, n - L)
        h_block = best_h[s:s + L].copy()

        # Local optimization of this block via small SLSQP
        def obj_block(x):
            h_full = best_h.copy()
            h_full[s:s + L] = x
            h_full = project_feasible(h_full, n)
            return compute_c5(h_full)

        best_local = h_block.copy()
        best_c5_local = obj_block(h_block)

        for _ in range(30):
            perturb = 0.05 * rng.randn(L)
            h_new_block = h_block + perturb
            h_new_block = project_feasible(h_new_block, L)
            c5_loc = obj_block(h_new_block)
            if c5_loc < best_c5_local - 1e-14:
                best_c5_local = c5_loc
                h_new2 = h_trial_block = h_new_block.copy()
                h_block = h_new_block.copy()

        h_full = best_h.copy()
        h_full[s:s + L] = h_block
        h_full = project_feasible(h_full, n)
        c5_full = compute_c5(h_full)
        if c5_full < best_c5 - 1e-13:
            best_c5 = c5_full
            best_h = h_full.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = np.zeros(parent_n)
        idx = rng.permutation(parent_n)
        parent_h[idx[:parent_n // 2]] = 1.0
        best_h = project_feasible(parent_h, parent_n)
        best_c5 = compute_c5(best_h)
        best_n = parent_n
        print("Random init: n={}, C5={:.10f}".format(parent_n, best_c5))

    n = best_n
    dx = 2.0 / n

    # Phase 1: Full-softmax gradient descent (primary optimization)
    print("\n--- Phase 1: Full-softmax gradient descent (n={}) ---".format(n))
    beta_list_1 = [30, 50, 80, 100, 150, 200, 300, 400, 500, 700]
    best_h, best_c5 = full_soft_grad_descent(best_h, n, beta_list_1, max_iter=600, lr0=0.06)
    print("  C5={:.10f}".format(best_c5))

    # Phase 2: Subgradient momentum (top-k)
    print("\n--- Phase 2: Subgradient momentum ---")
    for tk in [3, 5, 8]:
        if remaining_time() < 15.0:
            break
        best_h, best_c5 = subgrad_momentum(best_h, n, top_k=tk, max_iter=200, step0=0.08)
    print("  C5={:.10f}".format(best_c5))

    # Phase 3: Simulated annealing
    print("\n--- Phase 3: Simulated annealing ---")
    n_iter_sa = 2000
    if remaining_time() > 120.0:
        n_iter_sa = 4000
    best_h, best_c5 = simulated_annealing(best_h, n, rng, 0.008, 1e-5, n_iter_sa)
    print("  C5={:.10f}".format(best_c5))

    # Phase 4: Second round of gradient descent with higher beta
    print("\n--- Phase 4: Gradient descent round 2 (high beta) ---")
    beta_list_2 = [500, 800, 1000, 1500, 2000, 3000, 5000]
    best_h, best_c5 = full_soft_grad_descent(best_h, n, beta_list_2, max_iter=400, lr0=0.03)
    print("  C5={:.10f}".format(best_c5))

    # Phase 5: SLSQP with exact gradient
    print("\n--- Phase 5: SLSQP with exact gradient ---")
    for beta_val in [300, 800, 2000]:
        if remaining_time() < 20.0:
            break
        h_new, c5_new = slsqp_smooth(best_h, n, beta_val, maxiter=150)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new
            print("  beta={}: C5={:.10f}".format(beta_val, best_c5))
    print("  C5={:.10f}".format(best_c5))

    # Phase 6: Another simulated annealing round with different mutations
    if remaining_time() > 60.0:
        print("\n--- Phase 6: Simulated annealing round 2 ---")
        best_h, best_c5 = simulated_annealing(best_h, n, rng, 0.005, 5e-6, 3000)
        print("  C5={:.10f}".format(best_c5))

    # Phase 7: Block optimization
    if remaining_time() > 40.0:
        print("\n--- Phase 7: Block optimization ---")
        best_h, best_c5 = block_optimize(best_h, n, rng, n_blocks=30, block_sizes=(2, 3, 4, 5))
        print("  C5={:.10f}".format(best_c5))

    # Phase 8: Coordinate descent
    if remaining_time() > 20.0:
        print("\n--- Phase 8: Coordinate descent ---")
        best_h, best_c5 = coord_descent_small(best_h, n, rng, max_sweeps=10)
        print("  C5={:.10f}".format(best_c5))

    # Phase 9: Try different resolutions
    if remaining_time() > 80.0:
        print("\n--- Phase 9: Different resolutions ---")
        for n_try in [192, 320, 128]:
            if remaining_time() < 30.0:
                break
            h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
            c5_h = compute_c5(h_interp)
            print("  n={} interp: C5={:.10f}".format(n_try, c5_h))

            # Quick optimization at this resolution
            beta_list_3 = [100, 300, 800, 2000]
            h_opt, c5_h = full_soft_grad_descent(h_interp, n_try, beta_list_3, max_iter=200, lr0=0.05)

            if remaining_time() > 15.0:
                h_opt, c5_h = subgrad_momentum(h_opt, n_try, top_k=5, max_iter=100, step0=0.08)

            if remaining_time() > 10.0:
                hSA, c5_SA = simulated_annealing(h_opt, n_try, rng, 0.005, 1e-5, 1000)
                if c5_SA < c5_h:
                    h_opt, c5_h = hSA, c5_SA

            if c5_h < best_c5 - 1e-10:
                best_c5 = c5_h
                best_h = h_opt.copy()
                best_n = n_try
                print("  n={}: C5={:.10f} (improved!)".format(n_try, best_c5))
            else:
                print("  n={}: C5={:.10f} (no improvement)".format(n_try, c5_h))

    # Phase 10: Final gradient polish
    if remaining_time() > 15.0:
        print("\n--- Phase 10: Final polish ---")
        n = best_n
        beta_final = [1000, 2000, 5000, 10000]
        best_h, best_c5 = full_soft_grad_descent(best_h, n, beta_final, max_iter=300, lr0=0.02)
        print("  C5={:.10f}".format(best_c5))

    # Phase 11: Final tiny coordinate descent
    if remaining_time() > 8.0:
        best_h, best_c5 = coord_descent_small(best_h, best_n, rng, max_sweeps=5)
        print("\n--- Final coordinate descent: C5={:.10f} ---".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
