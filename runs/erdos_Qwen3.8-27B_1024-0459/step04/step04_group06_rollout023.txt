def project_feasible(h, n):

        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt

            h[:] += diff / n

def obj_smooth(x):

def subgrad_step(h, n, top_k=8):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    top_indices = np.argsort(corr)[-top_k:]
    top_indices = top_indices[np.argsort(corr[top_indices])[::-1]]
    grad = np.zeros(n)
    for k_idx in top_indices:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            n_valid = n - offset
            if n_valid <= 0:
                continue
            grad[offset:offset + n_valid] += (1.0 - h[:n_valid])
            grad[:n_valid] -= h[offset:offset + n_valid]
        else:
            m = -offset
            n_valid = n - m
            if n_valid <= 0:
                continue
            grad[:n_valid] += (1.0 - h[m:m + n_valid])
            grad[m:m + n_valid] -= h[:n_valid]
    return grad, corr

        for i in range(parent_n):
            h_tm[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
        best_h = project_feasible(h_tm, parent_n)
        best_c5 = compute_c5(best_h)
        best_n = parent_n
        print("Thue-Morse: n={}, C5={:.10f}".format(parent_n, best_c5))

    # Phase 1: Binary SA
    print("\n--- Phase 1: Binary SA ---")
    h_bin, c5_bin = binary_sa(best_h, best_n, max_iters=200000, t0=0.4, t_end=0.0003, seed=seed)
    print("  Binary SA: C5={:.10f}".format(c5_bin))
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 2: SLSQP with increasing beta
    print("\n--- Phase 2: SLSQP polish ---")
    _G_N = best_n
    _G_TGT = best_n / 2.0
    _G_DX = 2.0 / best_n
    for beta in [500, 2000, 10000, 50000]:
        if remaining_time() < 8.0:
            break
        h_new, c5_new = slsqp_optimize(best_h, best_n, beta=beta, maxiter=500, ftol=1e-15)
        if c5_new < best_c5 - 1e-16:
            best_c5 = c5_new
            best_h = h_new
            print("  beta={}: C5={:.10f}".format(beta, best_c5))

    # Phase 3: Differential Evolution (global search)
    print("\n--- Phase 3: Differential Evolution ---")
    h_de, c5_de = de_optimize(best_h, best_n, pop_size=50, maxiter=120, seed=seed + 1)
    print("  DE: C5={:.10f}".format(c5_de))
    if c5_de < best_c5:
        best_c5 = c5_de
        best_h = h_de.copy()

    # Phase 4: Polish DE result
    if c5_de < 0.39:
        for beta in [5000, 20000, 80000]:
            if remaining_time() < 6.0:
                break
            h_new, c5_new = slsqp_optimize(best_h, best_n, beta=beta, maxiter=400, ftol=1e-15)
            if c5_new < best_c5 - 1e-16:
                best_c5 = c5_new
                best_h = h_new
                print("  DE polish beta={}: C5={:.10f}".format(beta, best_c5))

    # Phase 5: Binary SA round 2 (different seed and lower temp)
    if remaining_time() > 80.0:
        print("\n--- Phase 5: Binary SA 2 ---")
        h_bin2, c5_bin2 = binary_sa(best_h, best_n, max_iters=150000, t0=0.15, t_end=0.0002, seed=seed + 100)
        print("  Binary SA 2: C5={:.10f}".format(c5_bin2))
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()
        h_new, c5_new = slsqp_optimize(best_h, best_n, beta=20000, maxiter=400, ftol=1e-15)
        if c5_new < best_c5 - 1e-16:
            best_c5 = c5_new
            best_h = h_new

    # Phase 6: Subgradient optimization
    if remaining_time() > 50.0:
        print("\n--- Phase 6: Subgradient ---")
        best_h, best_c5 = subgrad_optimize(best_h, best_n, max_iter=300, step0=0.08, top_k=8)
        print("  Subgrad: C5={:.10f}".format(best_c5))

    # Phase 7: Pairwise local search
    if remaining_time() > 30.0:
        print("\n--- Phase 7: Pairwise local search ---")
        best_h, best_c5 = pairwise_local_search(best_h, best_n, max_sweeps=12, seed=seed + 200)
        print("  Pairwise: C5={:.10f}".format(best_c5))

    # Phase 8: SLSQP final polish
    if remaining_time() > 20.0:
        print("\n--- Phase 8: SLSQP final ---")
        for beta in [30000, 100000, 300000]:
            if remaining_time() < 5.0:
                break
            h_new, c5_new = slsqp_optimize(best_h, best_n, beta=beta, maxiter=600, ftol=1e-16)
            if c5_new < best_c5 - 1e-17:
                best_c5 = c5_new
                best_h = h_new
                print("  beta={}: C5={:.10f}".format(beta, best_c5))

    # Phase 9: Try higher resolution
    if remaining_time() > 120.0:
        print("\n--- Phase 9: Higher resolution ---")
        for n_try in [384, 512, 640]:
            if remaining_time() < 40.0:
                break
            h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
            c5_h = compute_c5(h_interp)
            h_interp, c5_h = subgrad_optimize(h_interp, n_try, max_iter=200, step0=0.06, top_k=6)
            for beta in [2000, 10000, 50000]:
                if remaining_time() < 10.0:
                    break
                h_new, c5_new = slsqp_optimize(h_interp, n_try, beta=beta, maxiter=300, ftol=1e-15)
                if c5_new < c5_h:
                    c5_h = c5_new
                    h_interp = h_new
            print("  n={}: C5={:.10f}".format(n_try, c5_h))
            if c5_h < best_c5 - 1e-13:
                best_c5 = c5_h
                best_h = h_interp.copy()
                best_n = n_try
                print("  * Improved to n={}".format(n_try))

    # Phase 10: Second DE attempt at best resolution
    if remaining_time() > 80.0:
        print("\n--- Phase 10: DE round 2 ---")
        h_de2, c5_de2 = de_optimize(best_h, best_n, pop_size=40, maxiter=80, seed=seed + 500)
        print("  DE2: C5={:.10f}".format(c5_de2))
        if c5_de2 < best_c5:
            best_c5 = c5_de2
            best_h = h_de2.copy()
        h_new, c5_new = slsqp_optimize(best_h, best_n, beta=50000, maxiter=500, ftol=1e-16)
        if c5_new < best_c5 - 1e-16:
            best_c5 = c5_new
            best_h = h_new

    # Phase 11: Final subgradient + SLSQP
    if remaining_time() > 15.0:
        best_h, best_c5 = subgrad_optimize(best_h, best_n, max_iter=150, step0=0.05, top_k=5)
        h_new, c5_new = slsqp_optimize(best_h, best_n, beta=200000, maxiter=500, ftol=1e-16)
        if c5_new < best_c5 - 1e-17:
            best_c5 = c5_new
            best_h = h_new
        print("\nPre-final: C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
