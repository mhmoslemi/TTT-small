import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)

# Smooth max objective and gradient
def _sm_obj(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / _BETA) * dx_v

def _corr_grad(x, k, n):
    """Gradient of corr[k] with respect to x."""
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        if np_ > 0:
            g[off:off + np_] += (1.0 - x[:np_])
            g[:np_] -= x[off:off + np_]
    else:
        ao = -off
        np_ = n - ao
        if np_ > 0:
            g[:np_] += (1.0 - x[ao:ao + np_])
            g[ao:ao + np_] -= x[:np_]
    return g

def _sm_grad(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k] * dx_v
        if wk < 1e-15:
            continue
        g += wk * _corr_grad(x, k, n)
    return g

def slsqp_smooth(h, n, beta, maxiter=500):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.success or res.fun < _sm_obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def _sc(x):
    return np.sum(x) - _TGT

def _scj(x):
    return np.ones(_N)

# Coordinate descent
def coord_descent(h, n, max_iters=500, top_k=10):
    """Iterative coordinate descent focusing on top correlations."""
    h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    dx_v = 2.0 / n
    
    for it in range(max_iters):
        if remaining() < 2:
            break
        corr = compute_corr(h)
        sorted_idx = np.argsort(-corr)
        top_indices = sorted_idx[:top_k]
        
        improved = False
        # For each top correlation, try gradient step
        for k in top_indices:
            g = _corr_grad(h, k, n)
            # Subtract mean to maintain sum constraint
            g -= np.mean(g)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                continue
            g /= gn
            
            # Line search
            for step in [0.01, 0.02, 0.05, 0.005, 0.03, 0.08]:
                ht = project_feasible(h - step * g, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-15:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
        
        if not improved:
            break
    
    return h, best_c5

def active_set_eq(h, n, k_active=8, max_iters=200):
    """Equalize the top k correlations."""
    h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    dx_v = 2.0 / n

        if remaining() < 2:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:k_active]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        
        if c_max - c_min < 1e-10:
            break
        
        # Gradient: reduce c_max, increase c_min
        g = _corr_grad(h, idx[0], n) - _corr_grad(h, idx[-1], n)
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        
        improved = False
        for step in [0.01, 0.02, 0.05, 0.005, 0.03, 0.1, 0.015, 0.08]:
            ht = project_feasible(h + step * g, n)
            cn = compute_c5(ht)

                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:

            for step in [0.001, 0.002, 0.005]:

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def binary_sa(h_init, n, time_limit=60, seed=42):
    """Fast binary simulated annealing."""
    rng = np.random.RandomState(seed)
    h = (h_init > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            rem = rng.choice(ones_idx, n_ones - n_half, replace=False)
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            add = rng.choice(zeros_idx, n_half - n_ones, replace=False)
            h[add] = 1
    
    dx_v = 2.0 / n
    corr = compute_corr(h)
    cur_c5 = float(np.max(corr)) * dx_v
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    
    for gen in range(1000000):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break
        T = max(0.01, 2.0 * (0.005) ** (gen / 50000.0))
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Fast delta computation
        # Only correlations involving i or j change
        # corr[k] = sum h[a](1-h[a+k])
        # Changing h[i] from 1 to 0 and h[j] from 0 to 1
        # affects corr at various shifts
        
        # For efficiency, just recompute (n is small enough)
        h[i] = 0
        h[j] = 1
        nc = compute_corr(h)
        nc5 = float(np.max(nc)) * dx_v
        delta = nc5 - cur_c5
        h[i] = 1
        h[j] = 0
        
        if delta <= 0 or rng.rand() < np.exp(-delta / T):
            h[i] = 0
            h[j] = 1
            cur_c5 = nc5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    
    return best_h, best_c5

def compute_c5(h):

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)

def _corr_grad_k(x, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        if np_ > 0:
            g[off:off + np_] += (1.0 - x[:np_])
            g[:np_] -= x[off:off + np_]
    else:
        ao = -off
        np_ = n - ao
        if np_ > 0:
            g[:np_] += (1.0 - x[ao:ao + np_])
            g[ao:ao + np_] -= x[:np_]
    return g


def _sm_obj(x):

    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -80.0, 80.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / _BETA) * dx_v


def _sm_grad(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -80.0, 80.0)

def binary_pair_swap(h, n, max_rounds=50):

        if c54 < best_c5:
            best_c5 = c54
            best_h = h4.copy()

    if time.time() - start < time_budget * 0.8:
        h5 = slsqp_smooth(best_h, n, 50000, maxiter=500)
        c55 = compute_c5(h5)
        if c55 < best_c5:
            best_c5 = c55
            best_h = h5.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Progressive SLSQP at parent resolution
    for beta in [50, 200, 1000, 5000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Active-set + coordinate descent
    if remaining() > 15:
        h_eq, c5_eq = active_set_equalize(h, n, k_active=10, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    if remaining() > 10:
        h_cd, c5_cd = coord_descent(h, n, max_iters=300, top_k=12)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Phase 3: Subgradient method
    if remaining() > 30:
        h_sub, c5_sub = subgradient_opt(h, n, max_iters=2000, eta0=0.015)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()
            h = h_sub.copy()

    # Phase 4: High-beta SLSQP
    for beta in [10000, 50000, 100000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Multi-resolution search
    n_candidates = sorted(set([
        best_n - 16, best_n - 8, best_n + 8, best_n + 16, best_n + 32,
        int(best_n * 0.75), int(best_n * 1.25), int(best_n * 1.5),
        int(best_n * 2.0), 512, 640, 768, 256, 320, 448, 512
    ]))
    n_candidates = [nn for nn in n_candidates if 160 <= nn <= 1024 and nn % 2 == 0 and nn != best_n]
    n_candidates.sort(key=lambda x: abs(x - best_n))

    for n_t in n_candidates[:6]:
        if remaining() < 30:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        h_opt, c5_opt = polish_chain(h_feas, n_t, time_budget=min(50, int(remaining() * 0.25)))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 6: Binary optimisation + continuous polish
    if remaining() > 50:
        n_b = best_n
        h_bin = (best_h > 0.5).astype(float)

        # Pair swap
        h_ps, c5_ps = binary_pair_swap(h_bin, n_b, max_rounds=30)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps.copy()
            best_n = n_b

        # SA
        sa_time = min(80, max(10, int(remaining() * 0.3)))
        h_sa, c5_sa = binary_sa(h_bin, n_b, time_limit=sa_time, seed=seed)

        # Thue-Morse SA
        h_tm = thue_morse(n_b)
        sa_time2 = min(50, max(5, int(remaining() * 0.2)))
        h_tm_sa, c5_tm_sa = binary_sa(h_tm, n_b, time_limit=sa_time2, seed=seed + 100)

        # Polish each binary result
        for h_b, c5_b in [(h_ps, c5_ps), (h_sa, c5_sa), (h_tm_sa, c5_tm_sa)]:
            if remaining() < 10:
                break
            h_c, c5_c = polish_chain(h_b, n_b, time_budget=min(40, int(remaining() * 0.2)))
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()
                best_n = n_b

    # Phase 7: Perturbation + re-optimize (escape local minima)
    if remaining() > 40:
        n_p = best_n
        for attempt in range(3):
            if remaining() < 15:
                break
            h_pert = best_h.copy()
            n_perturb = max(3, n_p // 50)
            idx_pert = rng.choice(n_p, n_perturb, replace=False)
            h_pert[idx_pert] = rng.uniform(0, 1, n_perturb)
            h_pert = project_feasible(h_pert, n_p)
            h_opt, c5_opt = polish_chain(h_pert, n_p, time_budget=min(25, int(remaining() * 0.3)))
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_p

    # Phase 8: Final intensive polish
    n = best_n
    h = best_h.copy()

    for beta in [50000, 150000, 300000]:
        if remaining() < 5:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining() > 10:
        h_eq, c5_eq = active_set_equalize(best_h, n, k_active=12, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    if remaining() > 8:
        h_cd, c5_cd = coord_descent(best_h, n, max_iters=300, top_k=15)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if remaining() > 5:
        h_new = slsqp_smooth(best_h, n, 200000, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
