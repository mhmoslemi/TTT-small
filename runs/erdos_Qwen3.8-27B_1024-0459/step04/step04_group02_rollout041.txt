- Evaluate the change in C₅
- Accept if it improves, or with some probability if it doesn't (simulated annealing)

But evaluating C₅ after each swap is O(n) if we can update the correlations incrementally. When we swap h[i] and h[j] (i was 1, j was 0):
- h[i] becomes 0, h[j] becomes 1
- The change in c(L) for each L:
  - c(L) changes by: (new contribution) - (old contribution)
  - At position i: was h[i]=1, now h[i]=0
    - In c(L), h[i] appears as first factor at j=i: old contribution to c(L) was 1·(1-h[i+L]), new is 0
    - h[i] appears as second factor at j=i-L: old contribution was h[i-L]·(1-1)=0, new is h[i-L]·(1-0)=h[i-L]
    Wait, this is getting complicated. Let me just recompute c for affected lags.

Actually, when we change h[i] from 1 to 0 and h[j] from 0 to 1:
- c(L) = ∑_k h[k](1-h[k+L])
- The terms involving h[i] or h[j] change:
  - h[i] as first factor: term h[i](1-h[i+L]) goes from (1)(1-h[i+L]) to 0
    → change: -(1-h[i+L])
  - h[i] as second factor: term h[i-L](1-h[i]) goes from h[i-L](1-1)=0 to h[i-L](1-0)=h[i-L]
    → change: +h[i-L]
  - Similarly for h[j]:
  - h[j] as first factor: goes from 0 to (1-h[j+L])
    → change: +(1-h[j+L])
  - h[j] as second factor: goes from h[j-L](1-0)=h[j-L] to h[j-L](1-1)=0
    → change: -h[j-L]

So the net change in c(L) is:
Δc(L) = -(1-h[i+L]) + h[i-L] + (1-h[j+L]) - h[j-L]
(where we only include terms that are in the valid range)

This is O(1) per lag, so O(n) total to update all correlations. That's fast enough.

import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = np.argmin(h)
            else:
                idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(x, n, beta):
    """Compute smooth max objective and gradient using the softmax/log-sum-exp trick."""
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    
    grad = np.zeros(n)
    ncorr = len(corr)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        L = k - (n - 1)
        if L >= 0:
            npairs = n - L
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[L:L+npairs])
                grad[L:] -= w * x[:npairs]
        else:
            M = -L
            npairs = n - M
            if npairs > 0:
                grad[M:] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[M:M+npairs]
    grad *= dx
    return val, grad

def lbfgs_refine(h, n, beta, maxiter=200):
    """Use L-BFGS-B with penalty for the sum constraint."""
    def obj(x):
        val, _ = smooth_obj_grad(x, n, beta)
        penalty = 1000.0 * (np.sum(x) - n/2.0)**2
        return val + penalty
    
    def obj_grad(x):
        val, g = smooth_obj_grad(x, n, beta)
        g += 2000.0 * (np.sum(x) - n/2.0) * np.ones(n)
        return g

            return trial
    except Exception:
        pass
    return h.copy()

        return smooth_obj_grad(x, n, beta)[0]
    
    def obj_grad(x):
        return smooth_obj_grad(x, n, beta)[1]

        excess = n_ones - n // 2
        # Remove the ones with smallest h value (closest to 0.5 from above)
        order = np.argsort(h[ones])[:excess]
        h_bin[ones[order]] = 0.0
    elif n_ones < n // 2:

        order = np.argsort(-h[zeros])[:deficiency]
        h_bin[zeros[order]] = 1.0

    c5 = np.max(corr) * (2.0 / n)
    dx = 2.0 / n
    
    T = T0
    best_h = h_bin.copy()
    best_c5 = c5
    best_corr = corr.copy()
    
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        
        # Pick a random 1 and a random 0
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        
        # Compute change in all correlations
        # Δc(L) for each L
        L_arr = np.arange(-(n-1), n)
        # We need: -(1-h[i+L]) + h[i-L] + (1-h[j+L]) - h[j-L]
        # with proper masking for valid indices
        
        delta = np.zeros(2*n-1)
        for L in range(-(n-1), n):
            val = 0.0
            # h[i] as first factor at lag L: term is h[i]*(1-h[i+L])
            # If i+L is valid (0<=i+L<n): old contribution was (1)(1-h[i+L]), new is 0
            # Change: -(1-h[i+L])
            if 0 <= i + L < n:
                val += -(1.0 - h_bin[i+L])
            # h[i] as second factor at lag L: term is h[i-L]*(1-h[i])
            # If i-L is valid (0<=i-L<n): old was h[i-L]*(1-1)=0, new is h[i-L]*(1-0)=h[i-L]
            # Change: +h[i-L]
            if 0 <= i - L < n:
                val += h_bin[i-L]
            # h[j] as first factor at lag L: old was 0*(1-h[j+L])=0, new is 1*(1-h[j+L])
            # Change: +(1-h[j+L])
            if 0 <= j + L < n:
                val += (1.0 - h_bin[j+L])
            # h[j] as second factor at lag L: old was h[j-L]*(1-0)=h[j-L], new is h[j-L]*(1-1)=0
            # Change: -h[j-L]
            if 0 <= j - L < n:
                val += -h_bin[j-L]
            delta[L + n - 1] = val
        
        # New correlations
        new_corr = corr + delta
        new_c5 = np.max(new_corr) * dx
        
        # Accept or reject
        if new_c5 < c5 or (anneal and rng.random() < np.exp((c5 - new_c5) / T)):
            # Accept
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            corr = new_corr
            c5 = new_c5
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h_bin.copy()
                best_corr = corr.copy()
        
        # Anneal
        if it % 100 == 99:
            T *= 0.95
    
    return best_h, best_c5

def coordinate_descent_continuous(h, n, max_sweeps=20, rng=None):
    """Coordinate descent with golden section on each variable."""
    if rng is None:
        rng = np.random.RandomState(0)
    
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        interior = np.where((best > 1e-8) & (best < 1.0 - 1e-8))[0]
        if len(interior) == 0:
            break
        order = rng.permutation(interior)
        
        for i in order:
            if remaining_time() < 1:
                break
            
            # Precompute coefficients for the objective as a function of t = h[i]
            # c_k(t) = base_k + a_k * t + b_k * t^2
            a = np.zeros(ncorr)
            b = np.zeros(ncorr)
            for off in range(-(n-1), n):
                k = n - 1 + off
                if off == 0:
                    a[k] += 1.0
                    b[k] += -1.0
                else:
                    if 0 <= i + off < n:
                        a[k] += (1.0 - best[i + off])
                    if 0 <= i - off < n:
                        a[k] += (-best[i - off])
            
            t0 = best[i]
            base = best_corr - a * t0 - b * t0 * t0
            
            # Evaluate at a few candidate points
            candidates = np.array([0.0, 1.0, 0.5, t0 * 0.5, t0 * 1.5, 1.0 - t0])
            candidates = np.clip(candidates, 0.0, 1.0)
            candidates = np.unique(candidates)
            
            c5_best_local = best_c5
            t_best = t0
            
            for t in candidates:
                if abs(t - t0) < 1e-10:
                    continue
                vals = base + a * t + b * t * t
                c = np.max(vals) * dx
                if c < c5_best_local - 1e-15:
                    c5_best_local = c
                    t_best = t
            
            # Golden section refinement
            if abs(t_best - t0) > 1e-8:
                lo = max(0.0, min(t0, t_best) - 0.1)
                hi = min(1.0, max(t0, t_best) + 0.1)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                vals1 = base + a * c1 + b * c1 * c1
                vals2 = base + a * c2 + b * c2 * c2
                f1 = np.max(vals1) * dx
                f2 = np.max(vals2) * dx
                for _ in range(30):
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        vals1 = base + a * c1 + b * c1 * c1
                        f1 = np.max(vals1) * dx
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        vals2 = base + a * c2 + b * c2 * c2
                        f2 = np.max(vals2) * dx
                t_best = (lo + hi) / 2.0
                vals = base + a * t_best + b * t_best * t_best
                c_gs = np.max(vals) * dx
                if c_gs < c5_best_local:
                    c5_best_local = c_gs
            
            if c5_best_local < best_c5 - 1e-14:
                best[i] = t_best
                best_corr = compute_corr(best)
                best_c5 = compute_c5(best)
                improved = True
        
        if not improved:
            break
    
    return best, best_c5

def thue_morse(n):
    """Generate Thue-Morse-like binary sequence of length n."""
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = np.argmin(h)
            else:
                idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-13 and n > 0:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)

def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-15:
            continue
        L = k - (n - 1)
        if L >= 0:
            npairs = n - L
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[L:L+npairs])
                grad[L:] -= w * x[:npairs]
        else:
            M = -L
            npairs = n - M
            if npairs > 0:
                grad[M:] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[M:M+npairs]
    grad *= dx
    return val, grad

    return smooth_obj_and_grad(x, _global_n, _global_beta)[0]

def _smooth_grad(x):
    return smooth_obj_and_grad(x, _global_n, _global_beta)[1]

    return np.sum(x) - _global_n / 2.0

def _sum_constr_jac(x):
    return np.ones(_global_n)

def slsqp_refine(h, n, beta, maxiter=400):

    _global_n = n
    _global_beta = beta
    bounds = [(0.0, 1.0)] * n

                      bounds=bounds, constraints=cons,

        val, _ = smooth_obj_and_grad(x, n, beta)
        s = np.sum(x) - n / 2.0
        return val + pc * s * s

    def obj_grad(x):
        val, g = smooth_obj_and_grad(x, n, beta)
        s = np.sum(x) - n / 2.0
        return g + 2.0 * pc * s

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            if compute_c5(trial) <= compute_c5(h) + 1e-13:
                return trial
    except Exception:
        pass
    return h.copy()

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def vectorized_swap_delta(h, n, i, j, L_arr):
    valid1 = (i + L_arr >= 0) & (i + L_arr < n)
    valid2 = (i - L_arr >= 0) & (i - L_arr < n)
    valid3 = (j + L_arr >= 0) & (j + L_arr < n)
    valid4 = (j - L_arr >= 0) & (j - L_arr < n)
    idx1 = np.clip(i + L_arr, 0, n - 1)
    idx2 = np.clip(i - L_arr, 0, n - 1)
    idx3 = np.clip(j + L_arr, 0, n - 1)
    idx4 = np.clip(j - L_arr, 0, n - 1)
    term1 = np.where(valid1, -(1.0 - h[idx1]), 0.0)
    term2 = np.where(valid2, h[idx2], 0.0)
    term3 = np.where(valid3, (1.0 - h[idx3]), 0.0)
    term4 = np.where(valid4, -h[idx4], 0.0)
    return term1 + term2 + term3 + term4

def binary_anneal(h, n, max_iters=50000, rng=None, T0=0.004, T_factor=0.985, batch_size=15):
    if rng is None:
        rng = np.random.RandomState(0)
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target_ones = n // 2
    if n_ones > target_ones:
        ones = np.where(h_bin == 1.0)[0]
        excess = n_ones - target_ones
        order = np.argsort(h[ones])[:excess]
        h_bin[ones[order]] = 0.0
    elif n_ones < target_ones:
        zeros = np.where(h_bin == 0.0)[0]
        deficiency = target_ones - n_ones
        order = np.argsort(-h[zeros])[:deficiency]
        h_bin[zeros[order]] = 1.0
    corr = compute_corr(h_bin)
    dx = 2.0 / n
    c5 = np.max(corr) * dx
    L_arr = np.arange(-(n - 1), n)
    best_h = h_bin.copy()
    best_c5 = c5
    T = T0
    stagnant = 0

    for it in range(max_iters):
        if remaining_time() < 2:
            break
        best_new_c5 = c5
        best_i = -1
        best_j = -1
        best_delta = None
        for _ in range(batch_size):
            attempts = 0
            while attempts < 50:
                i = rng.randint(n)
                if h_bin[i] == 1.0:
                    break
                attempts += 1
            else:
                continue
            attempts = 0
            while attempts < 50:
                j = rng.randint(n)
                if h_bin[j] == 0.0:
                    break
                attempts += 1
            else:
                continue
            delta = vectorized_swap_delta(h_bin, n, i, j, L_arr)
            new_c5 = np.max(corr + delta) * dx
            if new_c5 < best_new_c5 - 1e-15:
                best_new_c5 = new_c5
                best_i, best_j = i, j
                best_delta = delta
        if best_i >= 0:
            h_bin[best_i] = 0.0
            h_bin[best_j] = 1.0
            corr = corr + best_delta
            c5 = best_new_c5
            stagnant = 0
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h_bin.copy()
        else:
            stagnant += 1
            if T > 1e-8 and stagnant > batch_size * 3:
                attempts = 0
                while attempts < 50:
                    i = rng.randint(n)
                    if h_bin[i] == 1.0:
                        break
                    attempts += 1
                if attempts < 50:
                    attempts = 0
                    while attempts < 50:
                        j = rng.randint(n)
                        if h_bin[j] == 0.0:
                            break
                        attempts += 1
                    if attempts < 50:
                        delta = vectorized_swap_delta(h_bin, n, i, j, L_arr)
                        new_c5 = np.max(corr + delta) * dx
                        if rng.random() < np.exp(min(0.0, (c5 - new_c5) / T)):
                            h_bin[i] = 0.0
                            h_bin[j] = 1.0
                            corr = corr + delta
                            c5 = new_c5
                        stagnant = 0
        if it % 80 == 79:
            T *= T_factor
    return best_h, best_c5

def coord_descent(h, n, max_sweeps=15, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    for sweep in range(max_sweeps):
        if remaining_time() < 2:
            break
        improved = False
        interior = np.where((best > 1e-8) & (best < 1.0 - 1e-8))[0]
        if len(interior) == 0:
            break
        order = rng.permutation(interior)
        for i in order:
            if remaining_time() < 1:
                break
            a = np.zeros(ncorr)
            b = np.zeros(ncorr)
            for off in range(-(n - 1), n):
                k = n - 1 + off
                if off == 0:
                    a[k] += 1.0
                    b[k] += -1.0
                else:
                    if 0 <= i + off < n:
                        a[k] += (1.0 - best[i + off])
                    if 0 <= i - off < n:
                        a[k] += (-best[i - off])
            t0 = best[i]
            base = best_corr - a * t0 - b * t0 * t0
            c5_best_local = best_c5
            t_best = t0
            candidates = [0.0, 1.0, 0.5, t0 * 0.5, t0 * 1.5, 1.0 - t0]
            for t in candidates:
                t = max(0.0, min(1.0, t))
                if abs(t - t0) < 1e-10:
                    continue
                vals = base + a * t + b * t * t
                c = np.max(vals) * dx
                if c < c5_best_local - 1e-15:
                    c5_best_local = c
                    t_best = t
            if abs(t_best - t0) > 1e-8:
                lo = max(0.0, min(t0, t_best) - 0.15)
                hi = min(1.0, max(t0, t_best) + 0.15)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                vs1 = base + a * c1 + b * c1 * c1
                vs2 = base + a * c2 + b * c2 * c2
                f1 = np.max(vs1) * dx
                f2 = np.max(vs2) * dx
                for _ in range(35):
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        vs1 = base + a * c1 + b * c1 * c1
                        f1 = np.max(vs1) * dx
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        vs2 = base + a * c2 + b * c2 * c2
                        f2 = np.max(vs2) * dx
                t_best = (lo + hi) / 2.0
                vals = base + a * t_best + b * t_best * t_best
                cg = np.max(vals) * dx
                if cg < c5_best_local:
                    c5_best_local = cg
            if c5_best_local < best_c5 - 1e-14:
                best[i] = t_best
                best_corr = compute_corr(best)
                best_c5 = compute_c5(best)
                improved = True
        if not improved:
            break
    return best, best_c5

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return h

def balanced_random(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        best_h = thue_morse(best_n)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    # Phase 1: Continuous refinement at parent resolution
    if remaining_time() > 40:
        h = best_h.copy()
        n = best_n
        for beta in [500, 2000, 8000]:
            if remaining_time() < 20:
                break
            h_new = slsqp_refine(h, n, beta, 300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 2: Multi-resolution continuous optimization
    target_resolutions = sorted(set([
        2 * best_n, 3 * best_n, 4 * best_n,
        800, 1024, 1280, 1600, 1920, 2560
    ]))
    target_resolutions = [r for r in target_resolutions if r % 2 == 0 and 128 <= r <= 2560 and r != best_n]

    for rn in target_resolutions:
        if remaining_time() < 50:
            break
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)
        for beta in [1000, 5000]:
            if remaining_time() < 25:
                break
            h_new = slsqp_refine(h_r, rn, beta, 250)
            c5_new = compute_c5(h_new)
            if c5_new < c5_r - 1e-15:
                h_r = h_new
                c5_r = c5_new
        if c5_r < best_c5 - 1e-15:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn

    # Phase 3: Binary annealing at best resolution
    if remaining_time() > 60 and best_n <= 2048:
        h_bin, c5_bin = binary_anneal(best_h, best_n, max_iters=80000, rng=rng, T0=0.003, T_factor=0.99, batch_size=12)
        if c5_bin < best_c5 - 1e-15:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            best_n = best_n

    # Phase 4: Try Thue-Morse at various resolutions
    for rn in [512, 640, 1024, 1280, 1536, 1920, 2048]:
        if remaining_time() < 50:
            break
        if rn % 2 != 0:
            continue
        h_tm = thue_morse(rn)
        h_tm = project_feasible(h_tm, rn)
        c5_tm = compute_c5(h_tm)
        if c5_tm < 0.42:
            h_bin, c5_bin = binary_anneal(h_tm, rn, max_iters=40000, rng=rng, T0=0.005, T_factor=0.99, batch_size=12)
            if c5_bin < best_c5 - 1e-15:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                best_n = rn

    # Phase 5: Try random balanced starts
    for rs in range(5):
        if remaining_time() < 50:
            break
        rn = min(1280, max(640, 2 * best_n))
        if rn % 2 != 0:
            rn += 1
        h_rand = balanced_random(rn, seed + 100 + rs * 7)
        h_rand = project_feasible(h_rand, rn)
        c5_rand = compute_c5(h_rand)
        if c5_rand < 0.45:
            h_bin, c5_bin = binary_anneal(h_rand, rn, max_iters=30000, rng=rng, T0=0.008, T_factor=0.99, batch_size=15)
            if c5_bin < best_c5 - 1e-15:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                best_n = rn

    # Phase 6: Continuous coordinate descent on best
    if remaining_time() > 25:
        h_new, c5_new = coord_descent(best_h, best_n, max_sweeps=12, rng=rng)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 7: L-BFGS-B polish
    if remaining_time() > 20:
        h_new = lbfgs_refine(best_h, best_n, 8000, 200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 8: Another round of binary anneal on the polished solution
    if remaining_time() > 40 and best_n <= 1536:
        h_bin, c5_bin = binary_anneal(best_h, best_n, max_iters=60000, rng=rng, T0=0.001, T_factor=0.995, batch_size=10)
        if c5_bin < best_c5 - 1e-15:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 9: Final coordinate descent
    if remaining_time() > 15:
        h_new, c5_new = coord_descent(best_h, best_n, max_sweeps=10, rng=rng)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 10: Final SLSQP with very high beta
    if remaining_time() > 15:
        h_new = slsqp_refine(best_h, best_n, 50000, 200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 11: Micro binary search
    if remaining_time() > 20 and best_n <= 1280:
        h_bin, c5_bin = binary_anneal(best_h, best_n, max_iters=30000, rng=rng, T0=0.0005, T_factor=0.998, batch_size=8)
        if c5_bin < best_c5 - 1e-15:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
