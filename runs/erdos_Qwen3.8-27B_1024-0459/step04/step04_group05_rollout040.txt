```python
def project_feasible(h, n):
    """Project h onto {h: 0<=h<=1, sum(h)=n/2}."""
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    
    # Use bisection on lambda
    lo = -1.0 - np.max(h)  # ensures all clip to 0
    hi = 1.0 - np.min(h)   # ensures all clip to 1
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    
    h_proj = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)
    
    # Fine-tune to get exact sum
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        # Adjust the element closest to 0.5 (most room)
        rooms = np.minimum(h_proj, 1.0 - h_proj)
        # Spread the adjustment among elements with room
        mask = rooms > 1e-10
        if np.sum(mask) > 0:
            h_proj[mask] += diff / np.sum(mask)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj
```
