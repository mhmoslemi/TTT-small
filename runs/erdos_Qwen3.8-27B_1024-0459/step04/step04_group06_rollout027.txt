            grad[m:m + nv] -= h[:nv]
    return grad, corr


def subgrad_optimize(h, n, max_iter=300, step0=0.1, momentum=0.85):
    c5_best = compute_c5(h)
    h_best = h.copy()
    v = np.zeros(n)
    step = step0
    patience = 0
    for it in range(max_iter):
        if remaining_time() < 5.0:
            break
        tk = min(5 + it // 50, 20)
        grad, corr = subgrad_step(h, n, top_k=tk)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad = grad / gn
        alpha = step
        improved = False
        for _ in range(12):
            v = momentum * v - alpha * grad
            h_trial = h + v
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-13:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
            alpha *= 0.5
        if improved:
            patience = 0
            step = min(alpha * 1.3, 0.3)
        else:
            patience += 1
            step *= 0.6
            v *= 0.4
            if patience > 15:
                break
    return h_best, c5_best


def coordinate_descent(h, n, max_sweeps=10, rng=None):
    c5_best = compute_c5(h)
    h_best = h.copy()
    if rng is None:
        rng = np.random.RandomState(0)
    for sweep in range(max_sweeps):
        if remaining_time() < 3.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if remaining_time() < 2.0:
                break
            old = h_best[i]
            candidates = [0.0, 1.0]
            for dv in [0.01, -0.01, 0.03, -0.03, 0.07, -0.07]:
                candidates.append(np.clip(old + dv, 0.0, 1.0))
            for new_v in candidates:
                if abs(new_v - old) < 1e-9:
                    continue
                h_trial = h_best.copy()
                h_trial[i] = new_v
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < c5_best - 1e-13:
                    c5_best = c5_t
                    h_best = h_trial.copy()
                    any_imp = True
        if not any_imp and sweep > 1:
            break
    return h_best, c5_best


def block_flip(h, n, rng):
    h = h.copy()
    for _ in range(5):
        L = rng.randint(3, max(4, n // 8))
        start = rng.randint(0, n - L)
        h[start:start + L] = 1.0 - h[start:start + L]
        h = project_feasible(h, n)
    return h


def block_swap(h, n, rng):
    h = h.copy()
    L = rng.randint(3, max(4, n // 10))
    s1 = rng.randint(0, n - L)
    s2 = s1 + rng.randint(L, max(L + 1, n - L))
    if s2 + L > n:
        s2 = n - L
    a = h[s1:s1 + L].copy()
    h[s1:s1 + L] = h[s2:s2 + L]
    h[s2:s2 + L] = a
    return project_feasible(h, n)


def aggressive_polish(h, n, time_limit):
    """Run SLSQP with increasing beta up to very high values"""
    c5_best = compute_c5(h)
    h_best = h.copy()
    beta_schedule = [100, 300, 1000, 3000, 10000, 30000, 100000]
    for beta in beta_schedule:
        if remaining_time() < time_limit:
            break
        h_new, c5_new = slsqp_opt(h_best, n, beta=beta, maxiter=400, ftol=1e-15)
        if c5_new < c5_best - 1e-15:
            c5_best = c5_new
            h_best = h_new
    # Also try top-sum objective
    if remaining_time() > time_limit:
        for tk in [3, 5, 8]:
            if remaining_time() < time_limit:
                break
            h_new, c5_new = slsqp_opt_topsum(h_best, n, top_k=tk, maxiter=200, ftol=1e-14)
            if c5_new < c5_best - 1e-15:
                c5_best = c5_new
                h_best = h_new
    return h_best, c5_best


def explore_from(h_base, n, rng, num_trials=8, time_per_trial=15.0):
    """Try various perturbations and re-optimize"""
    c5_best = compute_c5(h_base)
    h_best = h_base.copy()
    for t in range(num_trials):
        if remaining_time() < time_per_trial + 5:
            break
        pt = t % 7
        if pt == 0:
            h_trial = block_flip(h_best, n, rng)
        elif pt == 1:
            h_trial = block_swap(h_best, n, rng)
        elif pt == 2:
            h_trial = partial_binary(h_best, n, frac=0.3 + 0.1 * (t % 4))
        elif pt == 3:
            h_trial = h_best.copy()
            h_trial += 0.02 * rng.randn(n)
            h_trial = project_feasible(h_trial, n)
        elif pt == 4:
            h_trial = h_best.copy()
            # Flip a contiguous block
            L = rng.randint(5, max(6, n // 5))
            start = rng.randint(0, n - L)
            h_trial[start:start + L] = 1.0 - h_trial[start:start + L]
            h_trial = project_feasible(h_trial, n)
        elif pt == 5:
            # Harmonic perturbation
            h_trial = h_best.copy()
            freq = rng.randint(2, n // 4)
            phase = rng.rand() * 2 * np.pi
            t_arr = np.arange(n) / float(n)
            h_trial += 0.03 * np.sin(2.0 * np.pi * freq * t_arr + phase)
            h_trial = project_feasible(h_trial, n)
        else:
            # Random subset reassignment
            h_trial = h_best.copy()
            subset = rng.choice(n, size=n // 5, replace=False)
            h_trial[subset] = 1.0 - h_trial[subset]
            h_trial = project_feasible(h_trial, n)

        # Quick SLSQP polish from perturbed point
        h_new, c5_new = slsqp_opt(h_trial, n, beta=2000, maxiter=150, ftol=1e-13)
        h_new, c5_new = slsqp_opt(h_new, n, beta=10000, maxiter=200, ftol=1e-14)
        if c5_new < c5_best - 1e-14:
            c5_best = c5_new
            h_best = h_new
    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    # Initialize from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        h = np.zeros(parent_n)
        for i in range(parent_n):
            cnt = bin(i).count('1')
            h[i] = 1.0 if cnt % 2 == 1 else 0.0
        parent_h = project_feasible(h, parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("Fallback: n={}, C5={:.10f}".format(parent_n, best_c5))

    # Phase 1: Subgradient to get into a good basin
    print("\n--- Phase 1: Subgradient (n={}) ---".format(best_n))
    best_h, best_c5 = subgrad_optimize(best_h, best_n, max_iter=200, step0=0.1)
    print("  C5={:.10f}".format(best_c5))

    # Phase 2: Aggressive SLSQP polish at parent resolution
    print("\n--- Phase 2: Aggressive SLSQP (n={}) ---".format(best_n))
    best_h, best_c5 = aggressive_polish(best_h, best_n, time_limit=20.0)
    print("  C5={:.10f}".format(best_c5))

    # Phase 3: Explore from current best with perturbations
    print("\n--- Phase 3: Perturbation exploration (n={}) ---".format(best_n))
    if remaining_time() > 60.0:
        best_h, best_c5 = explore_from(best_h, best_n, rng, num_trials=10, time_per_trial=12.0)
        print("  C5={:.10f}".format(best_c5))

    # Phase 4: Try different resolutions
    print("\n--- Phase 4: Resolution sweep ---")
    res_list = [192, 224, 288, 320, 384, 512]
    for n_try in res_list:
        if remaining_time() < 40.0:
            break
        h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        c5_h = compute_c5(h_interp)
        # Quick subgradient
        h_interp, c5_h = subgrad_optimize(h_interp, n_try, max_iter=100, step0=0.08)
        # SLSQP polish
        h_interp, c5_h = aggressive_polish(h_interp, n_try, time_limit=10.0)
        if c5_h < best_c5 - 1e-12:
            best_c5 = c5_h
            best_h = h_interp.copy()
            best_n = n_try
            print("  n={}: C5={:.10f} (NEW BEST)".format(n_try, c5_h))
        else:
            print("  n={}: C5={:.10f}".format(n_try, c5_h))

    # Phase 5: More exploration at best resolution
    print("\n--- Phase 5: More exploration (n={}) ---".format(best_n))
    if remaining_time() > 50.0:
        best_h, best_c5 = explore_from(best_h, best_n, rng, num_trials=12, time_per_trial=10.0)
        print("  C5={:.10f}".format(best_c5))

    # Phase 6: Binary push + re-optimize (different basin)
    print("\n--- Phase 6: Binary push strategies (n={}) ---".format(best_n))
    if remaining_time() > 40.0:
        for bfrac in [0.3, 0.5, 0.7]:
            if remaining_time() < 15.0:
                break
            h_bin = partial_binary(best_h, best_n, frac=bfrac)
            h_bin, c5_bin = aggressive_polish(h_bin, best_n, time_limit=5.0)
            if c5_bin < best_c5 - 1e-14:
                best_c5 = c5_bin
                best_h = h_bin
                print("  frac={}: C5={:.10f} (NEW BEST)".format(bfrac, c5_bin))
        # Full binary push
        if remaining_time() > 15.0:
            h_bin = binary_push(best_h, best_n, strength=0.2)
            h_bin, c5_bin = aggressive_polish(h_bin, best_n, time_limit=5.0)
            if c5_bin < best_c5 - 1e-14:
                best_c5 = c5_bin
                best_h = h_bin
                print("  Full binary: C5={:.10f} (NEW BEST)".format(c5_bin))

    # Phase 7: Coordinate descent
    print("\n--- Phase 7: Coordinate descent (n={}) ---".format(best_n))
    if remaining_time() > 15.0:
        best_h, best_c5 = coordinate_descent(best_h, best_n, max_sweeps=8, rng=rng)
        print("  C5={:.10f}".format(best_c5))

    # Phase 8: Final aggressive polish
    print("\n--- Phase 8: Final polish (n={}) ---".format(best_n))
    if remaining_time() > 10.0:
        best_h, best_c5 = aggressive_polish(best_h, best_n, time_limit=5.0)
        print("  C5={:.10f}".format(best_c5))

    # Phase 9: Very final - try a few more micro-perturbations
    if remaining_time() > 8.0:
        for _ in range(5):
            if remaining_time() < 3.0:
                break
            h_trial = best_h.copy()
            h_trial += 0.005 * rng.randn(best_n)
            h_trial = project_feasible(h_trial, best_n)
            h_trial, c5_t = slsqp_opt(h_trial, best_n, beta=50000, maxiter=200, ftol=1e-15)
            if c5_t < best_c5 - 1e-15:
                best_c5 = c5_t
                best_h = h_trial

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
