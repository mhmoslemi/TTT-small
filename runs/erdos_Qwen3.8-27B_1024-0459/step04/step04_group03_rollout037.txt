```python
def active_lag_newton(h, n, max_iter=200, tol=1e-15):
    """
    Newton method on the active-set KKT conditions.
    At optimum: f_{k1}(h) = f_{k2}(h) = ... = f_{km}(h) = t (minimize t)
    with sum(h) = n/2, 0 <= h <= 1.
    
    We use a "leveling" approach: at each step, compute gradients of active
    f_k's, find a direction that reduces the maximum, and take a step.
    """
    dx = 2.0 / n
    h = h.copy()
    
    for it in range(max_iter):
        if remaining_time() < 3:
            break
        
        corr = compute_corr(h)
        c_vals = corr * dx
        max_val = np.max(c_vals)
        
        # Active set: lags within 0.05*dx of max (very tight)
        active = np.where(c_vals >= max_val - 0.05 * dx)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c_vals)])
        
        if len(active) == 1:
            # Single active lag: just minimize that one
            # Gradient of f_k w.r.t. h[j] for j in [0, n-k-1]:
            # df_k/dh[j] = (1 - h[j+k]) * dx - h[j-k+...]*dx (from the second term)
            # Actually: f_k = sum_i h[i](1-h[i+k]) * dx
            # df_k/dh[i] = (1-h[i+k])*dx if i < n-k (from first product)
            #              + (-h[i-k])*dx if i >= k (from second product, where i = j+k)
            k = active[0]
            grad = np.zeros(n)
            grad[:n-k] += (1.0 - h[k:]) * dx
            grad[k:] += (-h[:n-k]) * dx
            # Project gradient onto feasible directions (sum = 0, respecting bounds)
            # Simple: subtract mean to satisfy sum constraint
            grad -= np.mean(grad)
            # Step in negative gradient direction
            step = 0.01
            h_new = h - step * grad
            h_new = project_feasible(h_new, n)
            if compute_c5(h_new) < compute_c5(h):
                h = h_new
            else:
                break
            continue
        
        # Multiple active lags: find direction that levels them
        # Compute gradients
        grads = np.zeros((len(active), n))
        for idx, k in enumerate(active):
            g = np.zeros(n)
            g[:n-k] += (1.0 - h[k:]) * dx
            g[k:] += (-h[:n-k]) * dx
            grads[idx] = g
        
        # We want: sum_i lambda_i * grad_i = -direction
        # where lambda_i are weights (sum to 1, all >= 0)
        # And the direction should reduce the max
        
        # Simple approach: weighted average of gradients with weights proportional
        # to how much each active lag exceeds the mean
        c_active = c_vals[active]
        weights = c_active - np.mean(c_active)
        if np.max(weights) < 1e-15:
            weights = np.ones(len(active)) / len(active)
        else:
            weights = np.maximum(weights, 0)
            if np.sum(weights) < 1e-15:
                weights = np.ones(len(active)) / len(active)
            else:
                weights = weights / np.sum(weights)
        
        combined_grad = weights @ grads  # shape (n,)
        combined_grad -= np.mean(combined_grad)  # project to sum=0
        
        # Line search
        best_h = h.copy()
        best_c5 = compute_c5(h)
        for step_frac in [0.1, 0.05, 0.01, 0.005, 0.001]:
            h_trial = h - step_frac * combined_grad * (1.0/np.max(np.abs(combined_grad)) + 1e-15)
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
        
        if best_c5 < compute_c5(h) - 1e-16:
            h = best_h
        else:
            # Try smaller active set
            active = np.where(c_vals >= max_val - 0.01)[0]
            if len(active) <= 1:
                break
    
    return h
```

5. Take a step in direction d, project to feasible set
6. Repeat

The "leveling" direction can be found by solving:
min ||d||² s.t. (∇c_k - ∇c_j) · d = c_k - c_j for all k,j ∈ A, and 1·d = 0

This is a least-squares problem!

```python
def leveling_step(h, n, active_lags, eps=0.01):
    """Find a direction that levels the active constraints."""
    dx = 2.0 / n
    corr = compute_corr(h) * dx
    m = len(active_lags)
    
    if m <= 1:
        return None
    
    # Compute gradients of c_k for active k
    grads = np.zeros((m, n))
    for idx, k in enumerate(active_lags):
        g = np.zeros(n)
        g[:n-k] += (1.0 - h[k:]) * dx
        g[k:] += (-h[:n-k]) * dx
        grads[idx] = g
    
    # Differences from the first active lag
    diffs = grads[1:] - grads[0:1]  # (m-1, n)
    c_diffs = corr[active_lags[1:]] - corr[active_lags[0]]  # (m-1,)
    
    # We want: diffs @ d = -c_diffs (to make all c_k equal)
    # Subject to: 1 @ d = 0 (maintain sum constraint)
    # And: minimize ||d|| (minimum norm solution)
    
    # Set up: [diffs; 1^T] @ d = [-c_diffs; 0]
    A = np.vstack([diffs, np.ones(n)])
    b = np.concatenate([-c_diffs, [0.0]])
    
    # Minimum norm solution: d = A^T (A A^T)^{-1} b
    try:
        AAT = A @ A.T
        b_vec = b
        sol = np.linalg.solve(AAT + 1e-12 * np.eye(AAT.shape[0]), A.T @ b_vec)
        d = A.T @ sol
    except np.linalg.LinAlgError:
        return None
    
    return d
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_N_G = 0
_BETA_G = 1.0
_LAGS_G = None
_DX_G = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def _sm_obj(x):
    n = _N_G
    beta = _BETA_G
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def _sm_eq(x):
    return np.sum(x) - _N_G / 2.0

def slsqp_smooth(h, n, beta, maxiter=500):
    global _N_G, _BETA_G
    _N_G = n
    _BETA_G = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sm_eq}]
    try:
        res = minimize(_sm_obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def _mm_obj(x):
    return x[_N_G]

def _mm_ineq(x):
    n = _N_G
    dx = _DX_G
    h = x[:n]
    t = x[n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - dx * corr[_LAGS_G]

def _mm_eq(x):
    return np.sum(x[:_N_G]) - _N_G / 2.0

def slsqp_minimax(h_init, n, max_rounds=15, max_slsqp_iter=250, thresh_offset=0.3):
    global _N_G, _DX_G, _LAGS_G
    _N_G = n
    _DX_G = 2.0 / n
    h = h_init.copy()
    dx = 2.0 / n

    for rnd in range(max_rounds):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        max_corr = np.max(corr)
        threshold = max_corr - thresh_offset
        active = np.where(corr >= threshold)[0]
        top5 = np.argsort(corr)[::-1][:5]
        active = np.unique(np.concatenate([active, top5]))
        active = active[np.argsort(active)]

        _LAGS_G = active
        x0 = np.concatenate([h, [max_corr * dx + 1e-6]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': _mm_ineq},
            {'type': 'eq', 'fun': _mm_eq}
        ]
        try:
            res = minimize(_mm_obj, x0, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_slsqp_iter, 'ftol': 1e-16, 'iprint': 0})
            if res.x is not None:
                h_new = project_feasible(res.x[:n], n)
                c5_new = compute_c5(h_new)
                c5_old = compute_c5(h)
                if c5_new < c5_old - 1e-18:
                    h = h_new
                elif rnd > 3:
                    break
        except Exception:
            break
    return h

def grad_fk(h, k, n):
    """Compute gradient of c_k = sum_{i=0}^{n-k-1} h[i](1-h[i+k])*dx w.r.t. h."""
    dx = 2.0 / n
    g = np.zeros(n)
    if k == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    else:
        g[:n-k] += (1.0 - h[k:]) * dx
        g[k:] += (-h[:n-k]) * dx
    return g

def leveling_newton(h, n, max_iter=500, tol=1e-16):
    """
    Newton method to level the active lags.
    At optimum: all active c_k are equal, and the weighted gradient is parallel to 1.
    """
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iter):
        if remaining_time() < 3:
            break

        corr = compute_corr(h) * dx
        max_val = np.max(corr)

        # Active set: lags within very tight tolerance
        active = np.where(corr >= max_val - 0.02 * dx)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        if len(active) == 1:
            # Single active: gradient descent
            k = int(active[0])
            g = grad_fk(h, k, n)
            g -= np.mean(g)  # project onto sum=0
            norm_g = np.linalg.norm(g)
            if norm_g < 1e-14:
                break
            d = -g / norm_g
            # Line search
            improved = False
            for alpha in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
                h_trial = project_feasible(h + alpha * d, n)
                c5_t = compute_c5(h_trial)
                if c5_t < best_c5 - 1e-17:
                    best_c5 = c5_t
                    h = h_trial.copy()
                    improved = True
                    break
            if not improved:
                break
            continue

        m = len(active)
        # Compute gradients for active lags
        grads = np.array([grad_fk(h, int(k), n) for k in active])
        c_active = corr[active]

        # We want to find direction d such that:
        # (1) c_k(h+d) - c_k(h) = 0 for all k (first order: grads[k]·d = 0 for all k)
        #     More precisely, we want to level: grads[k]·d = -(c_k - c_mean) for all k
        # (2) 1·d = 0 (maintain feasibility)
        # (3) Minimize ||d|| (minimum norm)

        c_mean = np.mean(c_active)
        targets = -(c_active - c_mean)  # what we want grads[k]·d to be

        # Set up system: [grads; 1^T] d = [targets; 0]
        A = np.vstack([grads, np.ones(n)])
        b = np.concatenate([targets, [0.0]])

        # Minimum norm least-squares solution
        try:
            # Use SVD for stability
            U, s, Vt = np.linalg.svd(A, full_matrices=False)
            # d = V^T @ diag(1/s) @ U^T @ b (truncated SVD)
            s_inv = np.where(s > 1e-12, 1.0 / s, 0.0)
            d = (Vt.T * s_inv) @ (U.T @ b)
        except np.linalg.LinAlgError:
            d = np.linalg.lstsq(A, b, rcond=None)[0]

        # Line search
        norm_d = np.linalg.norm(d)
        if norm_d < 1e-15:
            break
        d_normalized = d / norm_d

        improved = False
        for alpha in [norm_d * 0.5, norm_d * 0.2, norm_d * 0.1, norm_d * 0.05,
                      norm_d * 0.01, 0.01, 0.005, 0.001, 0.0005, 0.0001]:
            h_trial = project_feasible(h + alpha * d_normalized, n)
            c5_t = compute_c5(h_trial)
            if c5_t < best_c5 - 1e-17:
                best_c5 = c5_t
                h = h_trial.copy()
                improved = True
                break

        if not improved:
            # Try with smaller active set
            active_tight = np.where(corr >= max_val - 0.005 * dx)[0]
            if len(active_tight) <= 1:
                break
            # Or try opposite direction
            for alpha in [0.01, 0.005, 0.001]:
                h_trial = project_feasible(h - alpha * d_normalized, n)
                c5_t = compute_c5(h_trial)
                if c5_t < best_c5 - 1e-17:
                    best_c5 = c5_t
                    h = h_trial.copy()
                    improved = True
                    break
            if not improved:
                break

    return h, compute_c5(h)

def boundary_push(h, n, threshold=1e-3):
    """Push near-boundary values to exactly 0 or 1, then re-project."""
    h_b = h.copy()
    h_b[h_b < threshold] = 0.0
    h_b[h_b > 1.0 - threshold] = 1.0
    h_b = project_feasible(h_b, n)
    return h_b

def fine_pair_search(h, n, max_rounds=8, rng=None):
    """Fine pair-transfer search with adaptive steps."""
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    best_c5 = compute_c5(h)

    for rnd in range(max_rounds):
        if remaining_time() < 2:
            break
        improved = False
        order = rng.permutation(n)
        for ii in range(len(order)):
            if remaining_time() < 1:
                break
            i = int(order[ii])
            hi = h[i]
            if hi < 0.02 or hi > 0.98:
                continue
            for j in [i+1 if i+1 < n else i-1, i-1 if i > 0 else i+1,
                      i+2 if i+2 < n else i-2, i-2 if i > 1 else i+2]:
                if j < 0 or j >= n:
                    continue
                hj = h[j]
                if abs(hi - hj) < 0.01:
                    continue
                diff = hi - hj
                for frac in [0.3, 0.5, 0.7, 0.2, 0.8]:
                    amt = diff * frac
                    if abs(amt) < 0.005:
                        continue
                    h_trial = h.copy()
                    h_trial[i] -= amt
                    h_trial[j] += amt
                    h_trial = project_feasible(h_trial, n)
                    c5_t = compute_c5(h_trial)
                    if c5_t < best_c5 - 1e-17:
                        best_c5 = c5_t
                        h = h_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if not improved and ii > 2 * n // 3:
                break
        if not improved:
            break

    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def parametric_h(params, n):
    """h(x) = 1/2 + sum of Fourier cosine/sine terms on [0,2]."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.full(n, 0.5)
    n_terms = len(params) // 2
    for k in range(n_terms):
        a = params[2 * k]
        b = params[2 * k + 1]
        h += a * np.cos(np.pi * (k + 1) * x / 2.0)
        h += b * np.sin(np.pi * (k + 1) * x / 2.0)
    return h

    _START_TIME = time.time()
    _BUDGET_S = budget_s

    except Exception:
        best_n = 256
        best_h = np.full(best_n, 0.5)
        best_c5 = 0.5

    n = best_n
    h = best_h.copy()

    # Phase 1: Coarse smooth-max SLSQP (escalating beta)
    for beta in [200, 2000, 20000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Minimax SLSQP (multiple rounds with tighter active set)
    if remaining_time() > 60:
        h_mm = slsqp_minimax(best_h, best_n, max_rounds=20, max_slsqp_iter=300, thresh_offset=0.5)
        c5_mm = compute_c5(h_mm)
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm.copy()

    if remaining_time() > 40:
        h_mm2 = slsqp_minimax(best_h, best_n, max_rounds=15, max_slsqp_iter=250, thresh_offset=0.2)
        c5_mm2 = compute_c5(h_mm2)
        if c5_mm2 < best_c5:
            best_c5 = c5_mm2
            best_h = h_mm2.copy()

    # Phase 3: Leveling Newton (the key novel component)
    if remaining_time() > 30:
        h_lv, c5_lv = leveling_newton(best_h, best_n, max_iter=500)
        if c5_lv < best_c5:
            best_c5 = c5_lv
            best_h = h_lv.copy()

    # Phase 4: High-beta polish
    if remaining_time() > 25:
        h_pol = slsqp_smooth(best_h, best_n, 100000, maxiter=400)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 5: Another leveling pass
    if remaining_time() > 20:
        h_lv2, c5_lv2 = leveling_newton(best_h, best_n, max_iter=400)
        if c5_lv2 < best_c5:
            best_c5 = c5_lv2
            best_h = h_lv2.copy()

    # Phase 6: Boundary push + re-optimize
    if remaining_time() > 40:
        for thresh in [5e-4, 2e-4, 1e-4]:
            if remaining_time() < 15:
                break
            h_bp = boundary_push(best_h, best_n, threshold=thresh)
            c5_bp = compute_c5(h_bp)
            if c5_bp < best_c5 + 0.001:
                h_bp2 = slsqp_smooth(h_bp, best_n, 50000, maxiter=300)
                c5_bp2 = compute_c5(h_bp2)
                if c5_bp2 < best_c5:
                    best_c5 = c5_bp2
                    best_h = h_bp2.copy()

    # Phase 7: Ultra-high beta SLSQP
    if remaining_time() > 20:
        for beta in [500000, 2000000]:
            if remaining_time() < 10:
                break
            h_uh = slsqp_smooth(best_h, best_n, beta, maxiter=300)
            c5_uh = compute_c5(h_uh)
            if c5_uh < best_c5:
                best_c5 = c5_uh
                best_h = h_uh.copy()

    # Phase 8: Fine pair search
    if remaining_time() > 30:
        h_fp, c5_fp = fine_pair_search(best_h, best_n, max_rounds=10, rng=rng)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp.copy()

    # Phase 9: Final leveling + polish
    if remaining_time() > 15:
        h_fl, c5_fl = leveling_newton(best_h, best_n, max_iter=300)
        if c5_fl < best_c5:
            best_c5 = c5_fl
            best_h = h_fl.copy()

    # Phase 10: Try alternative resolutions
    for trial_n in [200, 224, 288, 320]:
        if remaining_time() < 30:
            break
        h_interp = interp_to(best_h, best_n, trial_n)
        h_interp = project_feasible(h_interp, trial_n)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5 + 0.003:
            h_t = slsqp_smooth(h_interp, trial_n, 10000, maxiter=300)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = trial_n
            if remaining_time() > 15:
                h_t2, c5_t2 = leveling_newton(best_h, best_n, max_iter=200)
                if c5_t2 < best_c5:
                    best_c5 = c5_t2
                    best_h = h_t2.copy()

    # Phase 11: Final fine search at best resolution
    if remaining_time() > 15:
        h_ff, c5_ff = fine_pair_search(best_h, best_n, max_rounds=5, rng=rng)
        if c5_ff < best_c5:
            best_c5 = c5_ff
            best_h = h_ff.copy()

    # Phase 12: Ultimate polish
    if remaining_time() > 10:
        h_up = slsqp_smooth(best_h, best_n, 5000000, maxiter=200)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()

    if remaining_time() > 8:
        h_ul, c5_ul = leveling_newton(best_h, best_n, max_iter=200)
        if c5_ul < best_c5:
            best_c5 = c5_ul
            best_h = h_ul.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)

            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
```
