    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(src, sn, dn):

    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn

def obj_smooth(x, beta, dx, tgt_sum):

def slsqp_core(h0, n, beta, maxiter, tgt_sum, extra_constraints=None):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt_sum,
             'jac': lambda x: np.ones(n)}]
    if extra_constraints is not None:
        for c in extra_constraints:
            cons.append(c)
    
    def obj(x):
        corr = get_corr(x)
        m = np.max(corr)
        shifted = corr - m
        e = np.exp(np.clip(beta * shifted, -500, 500))
        return float((m + np.log(np.sum(e)) / beta) * dx)
    
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def active_set_opt(h0, n, top_k=5, tlimit=None):

    tgt = n / 2.0

        cons_extra = []
        for ki in top_idx:
            # c[ki] = sum_i h[i+shift]*(1-h[i]) where shift = ki - (n-1)
            shift = ki - (n - 1)
            # Compute the linear indices for this correlation
            if shift >= 0:
                # c = sum_{i=0}^{n-1-shift} h[i+shift]*(1-h[i])
                i_range = np.arange(0, n - shift)
                h_idx = i_range + shift
                h1_idx = i_range
            else:
                # c = sum_{i=0}^{n-1+shift} h[i]*(1-h[i-shift])
                i_range = np.arange(0, n + shift)
                h_idx = i_range
                h1_idx = i_range - shift
            
            def make_con(h_idx=h_idx, h1_idx=h1_idx, tv=top_val):
                def fun(x):
                    return np.sum(x[h_idx] * (1.0 - x[h1_idx])) - tv
                def jac(x):
                    # d/dx_j [h[a]*(1-h[b])] for j in a_idx: (1-h[b_j])
                    # for j in b_idx: -h[a_j]

                        j = h_idx[ii]
                        k = h1_idx[ii]
                        # Conventionally SLSQP wants g(x) <= 0

                    return g

def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c else 0.0
    return h

def make_sine(n, f1, f2, phase):
