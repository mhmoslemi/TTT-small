```python
        g[ao:ao + npairs] -= x[:npairs]

def topk_obj_and_grad(x, n, K):
    """Objective: sum of top-K correlations. Returns (obj, grad)."""
    dx_val = 2.0 / n
    corr = compute_corr(x)
    # Get top K
    top_idx = np.argpartition(-corr, K-1)[:K]
    top_corr = corr[top_idx]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def upsample_pc(src_h, src_n, dst_n):
    """Piecewise constant upsample: each source sample repeated dst_n/src_n times."""
    if src_n == dst_n:
        return src_h.copy()
    repeats = dst_n // src_n
    h = np.repeat(src_h, repeats)
    if len(h) < dst_n:
        h = np.concatenate([h, src_h[:dst_n - len(h)]])
    return h[:dst_n]


def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        if npairs <= 0:
            return g
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        if npairs <= 0:
            return g
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g


def topk_penalty_obj(x, n, K, penalty):
    dx_val = 2.0 / n
    corr = compute_corr(x)
    K_actual = min(K, len(corr))
    top_idx = np.argpartition(-corr, K_actual - 1)[:K_actual]
    obj = np.sum(corr[top_idx]) * dx_val
    sum_viol = np.sum(x) - n / 2.0
    obj += penalty * sum_viol * sum_viol
    return obj


def topk_penalty_grad(x, n, K, penalty):
    dx_val = 2.0 / n
    corr = compute_corr(x)
    K_actual = min(K, len(corr))
    top_idx = np.argpartition(-corr, K_actual - 1)[:K_actual]
    g = np.zeros(n)
    for k in top_idx:
        g += corr_grad_per_shift(x, n, int(k))
    g *= dx_val / K_actual
    sum_viol = np.sum(x) - n / 2.0
    g += 2.0 * penalty * sum_viol / n
    return g


def softmax_obj(x, n, beta):
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = beta * (corr - m)
    logsum = np.log(np.sum(np.exp(shifted - np.max(shifted)))) + np.max(shifted)
    obj = (m + logsum / beta) * dx_val
    sum_viol = np.sum(x) - n / 2.0
    obj += 50.0 * sum_viol * sum_viol
    return obj


def softmax_grad(x, n, beta):
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted_max = np.max(shifted)
    ex = np.exp(shifted - shifted_max)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-12:
            g += w[k] * corr_grad_per_shift(x, n, int(k))
    g *= dx_val
    sum_viol = np.sum(x) - n / 2.0
    g += 100.0 * sum_viol / n
    return g


def optimize_lbgfsb(h, n, time_limit=100, rng=None):
    """Multi-K L-BFGS-B optimization with penalty for sum constraint."""
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()

    # Schedule: large K first (smooth), then smaller K (focused)
    K_schedule = [max(20, n // 10), max(10, n // 25), max(5, n // 50), 3, 2, 1]
    penalty = 50.0

    for K in K_schedule:
        if time.time() - start > time_limit * 0.8 or remaining() < 10:
            break
        for _ in range(2):
            if time.time() - start > time_limit or remaining() < 10:
                break
            res = minimize(
                topk_penalty_obj, h, args=(n, K, penalty),
                jac=topk_penalty_grad, method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-10, 'iprint': 0}
            )
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new.copy()

    # Softmax rounds for final polishing
    for beta in [200, 1000, 5000]:
        if time.time() - start > time_limit or remaining() < 10:
            break
        res = minimize(
            softmax_obj, h, args=(n, beta),
            jac=softmax_grad, method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-9, 'iprint': 0}
        )
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new.copy()

    return best_h, best_c5


def sa_binary(h, n, time_limit=60, seed=42):
    """Simulated annealing on binary vectors."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1

    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()

    T = 5.0
    alpha = 0.999
    start = time.time()
    stall = 0
    max_stall = 8000

    while time.time() - start < time_limit and T > 0.05:
        if remaining() < 10:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        delta = new_max - cur_max
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
        T *= alpha
        if stall > max_stall:
            T = max(T, 3.0)
            stall = 0

    return best_h, best_max * dx_val


def continuous_sa(h, n, time_limit=30, seed=42):
    """Simple continuous SA: perturb a few coordinates."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    T = 0.05
    start = time.time()

    while time.time() - start < time_limit and remaining() > 10:
        h_trial = h.copy()
        n_perturb = rng.randint(3, max(4, n // 30))
        idx = rng.choice(n, n_perturb, replace=False)
        h_trial[idx] += rng.uniform(-T, T, n_perturb)
        h_trial = project_feasible(h_trial, n)
        c5_trial = compute_c5(h_trial)
        delta = c5_trial - best_c5
        if delta < 0 or rng.rand() < np.exp(-delta / (T * 0.01)):
            h = h_trial
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
        T *= 0.995
        if T < 0.005:
            T = 0.02

    return best_h, best_c5

    h_bin = (h > 0.5).astype(float).copy()
```
