def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) > 0:
            h[idx[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(x):

def obj_smooth_grad(x):
    """Return (obj, grad) for L-BFGS-B"""
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)

    val = (m + np.log(s) / _G_BETA) * dx
    
    # Gradient
    grad = np.zeros(n)
    weights = exps / s  # softmax weights over lags
    # Also the max lag contributes
    max_idx = np.argmax(corr)
    
    for k_idx in range(len(corr)):
        w = weights[k_idx] + (1.0/_G_BETA) * (1.0 if k_idx == max_idx else 0.0)

            if n_valid <= 0:
                continue
            grad[offset:offset + n_valid] += w * (1.0 - x[:n_valid])
            grad[:n_valid] -= w * x[offset:offset + n_valid]
        else:

                continue
            grad[:n_valid] += w * (1.0 - x[m_offset:m_offset + n_valid])

    return val, grad

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def make_golden_ratio(n):
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * phi) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return project_feasible(h, n)


def make_vander_corput(n):
    h = np.zeros(n)
    for i in range(n):
        # Van der Corput sequence
        x = 0.0
        d = 0.5
        k = i + 1
        while k > 0:
            x += (k % 2) * d
            k //= 2
            d /= 2
        h[i] = 1.0 if x < 0.5 else 0.0
    return project_feasible(h, n)


def make_rademacher(n):
    """Based on Walsh functions - alternate blocks."""
    h = np.zeros(n)
    for i in range(n):
        # Count trailing zeros in binary representation of i
        k = i
        tz = 0
        while k > 0 and k % 2 == 0:
            tz += 1
            k //= 2
        # This gives a Rademacher-like pattern
        h[i] = 1.0 if (i & -i).bit_length() % 3 == 0 else 0.0
    return project_feasible(h, n)

            c5_r = compute_c5(h_r)
            h_r, c5_r = subgrad_descent(h_r, best_n, max_iter=120, step0=0.07, momentum=0.75)
            h_r, c5_r = slsqp_optimize(h_r, best_n, [500, 3000], max_iter_each=120)
            h_r, c5_r = flip_local_search(h_r, best_n, rng, max_rounds=15)

            if c5_r < best_c5 - 1e-14:
                best_c5 = c5_r
                best_h = h_r.copy()
                print("  Restart {}: C5={:.10f} (NEW BEST)".format(restart, best_c5))
            else:
                print("  Restart {}: C5={:.10f}".format(restart, c5_r))

    # Phase 6: Resolution sweep
    if remaining_time() > 120.0:
        print("\n--- Phase 6: Resolution sweep ---")
        for n_try in [128, 192, 320, 384, 512, 160]:
            if remaining_time() < 30.0:
                break
            h_t = project_feasible(interp_to(best_h, best_n, n_try), n_try)
            c5_t = compute_c5(h_t)
            h_t, c5_t = subgrad_descent(h_t, n_try, max_iter=120, step0=0.08, momentum=0.75)
            h_t, c5_t = slsqp_optimize(h_t, n_try, [200, 1500], max_iter_each=120)
            h_t, c5_t = flip_local_search(h_t, n_try, rng, max_rounds=12)

            if c5_t < best_c5 - 1e-12:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_try
                print("  n={}: C5={:.10f} (NEW BEST)".format(n_try, best_c5))
            else:
                print("  n={}: C5={:.10f}".format(n_try, c5_t))

    # Phase 7: Second SA round at best resolution
    if remaining_time() > 80.0:
        print("\n--- Phase 7: SA round 2 ---")
        best_h, best_c5 = simulated_annealing(best_h, best_n, rng, max_iter=400, T0=0.005, T_end=0.0002)
        best_h, best_c5 = subgrad_descent(best_h, best_n, max_iter=80, step0=0.04, momentum=0.7)
        best_h, best_c5 = slsqp_optimize(best_h, best_n, [5000, 20000], max_iter_each=200)
        best_h, best_c5 = flip_local_search(best_h, best_n, rng, max_rounds=30)
        print("  C5={:.10f}".format(best_c5))

    # Phase 8: Coordinate refine
    if remaining_time() > 25.0:
        print("\n--- Phase 8: Coordinate refine ---")
        best_h, best_c5 = coordinate_refine(best_h, best_n, rng, max_sweeps=10)
        print("  C5={:.10f}".format(best_c5))

    # Phase 9: Final SLSQP with very high beta
    if remaining_time() > 12.0:
        print("\n--- Phase 9: Final SLSQP ---")
        best_h, best_c5 = slsqp_optimize(best_h, best_n, [10000, 30000, 50000], max_iter_each=300)
        print("  C5={:.10f}".format(best_c5))

    # Phase 10: Final intensive flip search
    if remaining_time() > 10.0:
        print("\n--- Phase 10: Final flip search ---")
        best_h, best_c5 = flip_local_search(best_h, best_n, rng, max_rounds=40)
        best_h, best_c5 = coordinate_refine(best_h, best_n, rng, max_sweeps=5)
        print("  C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
