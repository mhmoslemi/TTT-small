f_β(h) = dx * (m + (1/β) log Σ_k exp(β(corr_k - m)))

where corr_k = Σ_j h_j (1 - h_{j+k}) (for appropriate j ranges)

The gradient d(f_β)/d(h_i) = dx * Σ_k w_k * d(corr_k)/d(h_i)

where w_k = exp(β(corr_k - m)) / Σ_l exp(β(corr_l - m))

And d(corr_k)/d(h_i):
- Let offset = k - (n-1)
- If offset == 0: d/d(h_i)[h_i(1-h_i)] = 1 - 2h_i
- If offset > 0 and i+offset < n: d/d(h_i)[h_i(1-h_{i+offset})] = 1 - h_{i+offset}
- If offset < 0 and i-offset < n: d/d(h_i)[h_{i-offset}(1-h_i)] = -h_{i-offset}
- Otherwise: 0

This is O(n²) per gradient evaluation for the sum over k. For n=1280, that's about 1.6 million operations per gradient, which is fine.

But wait, I can be smarter. The gradient at position i is:
d(corr_k)/d(h_i) is nonzero only for k values where i is in the range of the correlation.

Specifically, for a fixed i:
- offset = 0: contributes (1-2h_i) to corr[n-1]
- offset > 0, i+offset < n: contributes (1-h_{i+offset}) to corr[n-1+offset]
- offset < 0, i-offset < n: contributes -h_{i-offset} to corr[n-1+offset]

from scipy.optimize import minimize
import time

_BUDGET_S = 1300.0
_N_WORK = 0
_BETA_WORK = 1000.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        interior = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if len(interior) > 0:
            h[interior] += diff / len(interior)
        else:
            idx = np.argmin(np.abs(h - 0.5))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)

def smooth_max_vals(corr, beta):
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = m + np.log(wsum) / beta
    return val, weights

def _grad_smooth_obj(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    _, weights = smooth_max_vals(corr, beta)
    grad = np.zeros(n)
    ncorr = 2 * n - 1
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset+npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset+npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff+npairs])
                grad[aoff:aoff+npairs] -= w * x[:npairs]
    grad *= dx
    return grad, corr

def _obj_smooth(x):
    n = _N_WORK
    beta = _BETA_WORK
    corr = np.correlate(x, 1.0 - x, mode='full')
    val, _ = smooth_max_vals(corr, beta)
    return val * (2.0 / n)

def _obj_grad_smooth(x):
    return _grad_smooth_obj(x, _N_WORK, _BETA_WORK)[0]

def _con_fun(x):
    return np.sum(x) - _N_WORK / 2.0

def _con_jac(x):
    return np.ones(_N_WORK)

def slsqp_opt(h, n, beta, maxiter=600):
    global _N_WORK, _BETA_WORK
    _N_WORK = n
    _BETA_WORK = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj_smooth, h, jac=_obj_grad_smooth, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-22, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):

            if compute_c5(trial) <= compute_c5(h) + 1e-12:

    except Exception:

    return h.copy()

def lbfgs_penalty(h, n, beta, mu, maxiter=800):

    _N_WORK = n
    _BETA_WORK = beta
    bounds = [(0.0, 1.0)] * n
    
    def obj_p(x):

        val, _ = smooth_max_vals(corr, beta)
        s = np.sum(x) - n / 2.0
        return val * (2.0 / n) + 0.5 * mu * s * s
    
    def grad_p(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        _, weights = smooth_max_vals(corr, beta)
        grad = np.zeros(n)
        ncorr = 2 * n - 1

            if w < 1e-14:

                    grad[:npairs] += w * (1.0 - x[aoff:aoff+npairs])
                    grad[aoff:aoff+npairs] -= w * x[:npairs]
        grad *= dx
        s = np.sum(x) - n / 2.0
        grad += mu * s
        return grad
    
    try:
        res = minimize(obj_p, h, jac=grad_p, method='L-BFGS-B',

    ncorr = 2 * n - 1
    best_corr = compute_corr(best)
    best_c5 = np.max(best_corr) * dx

    rng = np.random.RandomState(seed)

            break
        improved = False
        interior = np.where((best > 1e-8) & (best < 1.0 - 1e-8))[0]
        if len(interior) == 0:
            break
        order = rng.permutation(interior)
        
        for i in order:
            if remaining_time() < 2:
                break
            t0 = best[i]
            
            def eval_at_t(t):
                delta = t - t0
                new_corr = best_corr.copy()
                for off in range(max(1, n-i), -(i), -1) if i > 0 else range(1, n, 1):
                    pass
                # More efficient: iterate over all offsets
                for off in range(-(n-1), n):
                    k = n - 1 + off
                    if off == 0:
                        new_corr[k] = best_corr[k] + t*(1.0-t) - t0*(1.0-t0)
                    else:
                        contrib = 0.0
                        if 0 <= i + off < n:
                            contrib += delta * (1.0 - best[i+off])
                        if 0 <= i - off < n:
                            contrib += -delta * best[i-off]
                        if contrib != 0.0:
                            new_corr[k] = best_corr[k] + contrib
                return np.max(new_corr) * dx
            
            # Quick candidates

def active_set_opt(h, n, n_active=10, max_iters=200):
    """Optimize focusing on top-n_active correlations using L-BFGS-B."""
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(best)
    
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        corr = compute_corr(best)
        max_val = np.max(corr)
        threshold = max_val - 0.5  # active set: correlations within 0.5 of max of max_val
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        if len(active) > n_active:
            top = np.argsort(corr)[::-1][:n_active]
            active = top
        
        # Build objective: sum of active correlations * dx
        def obj_active(x):
            c = compute_corr(x)
            return np.sum(c[active]) * dx / len(active)
        
        def grad_active(x):
            c = compute_corr(x)
            g = np.zeros(n)
            for k in active:
                offset = k - (n - 1)
                if offset >= 0:
                    npairs = n - offset
                    if npairs > 0:
                        g[offset:offset+npairs] += (1.0 - x[:npairs])
                        g[:npairs] -= x[offset:offset+npairs]
                else:
                    aoff = -offset
                    npairs = n - aoff
                    if npairs > 0:
                        g[:npairs] += (1.0 - x[aoff:aoff+npairs])
                        g[aoff:aoff+npairs] -= x[:npairs]
            return g * dx / len(active)
        
        # Use penalty for sum constraint
        mu = 10.0
        def obj_pen(x):
            base = obj_active(x)
            s = np.sum(x) - n / 2.0
            return base + 0.5 * mu * s * s
        
        def grad_pen(x):
            return grad_active(x) + mu * (np.sum(x) - n / 2.0)
        
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj_pen, best, jac=grad_pen, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 100, 'ftol': 1e-25})
            if res.x is not None and np.all(np.isfinite(res.x)):
                trial = project_feasible(res.x, n)
                c5_t = compute_c5(trial)
                if c5_t < best_c5 - 1e-15:
                    best = trial
                    best_c5 = c5_t
        except Exception:
            break
    
    return best, best_c5
