def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        h_trial = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_trial)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, n):
    h = np.array(h, dtype=float).copy()
    idx = np.argsort(-h)

def corr_grad_per_shift(h, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        if npairs > 0:
            g[off:off + npairs] += (1.0 - h[:npairs])
            g[:npairs] -= h[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        if npairs > 0:
            g[:npairs] += (1.0 - h[ao:ao + npairs])
            g[ao:ao + npairs] -= h[:npairs]
    return g


def slsqp_smooth(h, n, beta, maxiter=400):
    dx_val = 2.0 / n

    def smooth_obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -50, 50)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_val

    def smooth_grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -50, 50)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] > 1e-15:
                g += w[k] * corr_grad_per_shift(x, n, k)
        return g * dx_val

    def sum_con(x):
        return np.sum(x) - n / 2.0

    def sum_con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

def simulated_annealing_binary(h_bin, n, time_limit=200, rng=None):

    h = h_bin.astype(float).copy()

    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx_val
    best_h = h.copy()
    start_t = time.time()

    T = 2.0
    T_min = 0.05
    cooldown = 0.9997
    n_accepted = 0
    n_rejected = 0

    while T > T_min and time.time() - start_t < time_limit and remaining() > 10:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:

        if rng.rand() < 0.7:

            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            delta = (new_max - cur_max) * dx_val

                c5_cur = cur_max * dx_val
                if c5_cur < best_c5:

            else:

                n_rejected += 1
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)

def interp_to(src_h, src_n, dst_n):

        h_d, c5_d = dual_swap_search(h_bin, best_n, time_limit=min(100, remaining() - 40), rng=rng)
        if c5_d < best_c5:
            best_c5 = c5_d
            best_h = h_d.copy()
        if c5_d < c5_bin:
            c5_bin = c5_d
            h_bin = h_d.copy()

    # Simulated annealing
    if remaining() > 40:
        h_sa, c5_sa = simulated_annealing_binary(h_bin, best_n, time_limit=min(150, remaining() - 40), rng=rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        if c5_sa < c5_bin:
            c5_bin = c5_sa
            h_bin = h_sa.copy()

    # Phase 3: More targeted greedy from the best binary
    if remaining() > 40:
        h_t2, c5_t2 = targeted_greedy_search(h_bin, best_n, time_limit=min(150, remaining() - 40), rng=np.random.RandomState(seed + 1))
        if c5_t2 < best_c5:
            best_c5 = c5_t2
            best_h = h_t2.copy()

    # Phase 4: Continuous refinement from the best binary
    if remaining() > 30:
        h_cont, c5_cont = continuous_refine(h_bin, best_n, time_limit=min(150, remaining() - 30))
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()

    # Phase 5: Re-binarize and search again (sometimes continuous optimum binarizes better)
    if remaining() > 30:
        h_bin2 = binarize(best_h, best_n)
        h_t3, c5_t3 = targeted_greedy_search(h_bin2, best_n, time_limit=min(80, remaining() - 30), rng=np.random.RandomState(seed + 2))
        if c5_t3 < best_c5:
            best_c5 = c5_t3
            best_h = h_t3.copy()
        # Continuous polish
        if remaining() > 20:
            h_c2, c5_c2 = continuous_refine(h_t3, best_n, time_limit=min(60, remaining() - 20))
            if c5_c2 < best_c5:
                best_c5 = c5_c2
                best_h = h_c2.copy()

    # Phase 6: Try nearby n values
    n_candidates = []
    for dn in [-4, -2, 2, 4, 6, 8, 12, 16]:
        nn = best_n + dn
        if 100 <= nn <= 600 and nn % 2 == 0 and nn != best_n:
            n_candidates.append(nn)

    for n_t in n_candidates[:4]:
        if remaining() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        h_b_interp = binarize(h_feas, n_t)

        # Quick SLSQP
        h_opt = slsqp_smooth(h_feas, n_t, 1000, maxiter=200)
        c5_opt = compute_c5(h_opt)

        # Binary search
        if remaining() > 20:
            h_bo, c5_bo = targeted_greedy_search(h_b_interp, n_t, time_limit=min(60, remaining() - 20), rng=rng)
            if c5_bo < c5_opt:
                h_opt = h_bo
                c5_opt = c5_bo

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 7: Final continuous polish
    if remaining() > 15:
        h_final = slsqp_smooth(best_h, best_n, 20000, maxiter=400)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 8: Final greedy from binarized
    if remaining() > 15:
        h_bf = binarize(best_h, best_n)
        h_gf, c5_gf = targeted_greedy_search(h_bf, best_n, time_limit=min(30, remaining() - 10), rng=np.random.RandomState(seed + 99))
        if c5_gf < best_c5:
            best_c5 = c5_gf
            best_h = h_gf.copy()

    # Validate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
