def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def lag_gradient(h, n, lag):
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        j2 = -lag
        if j2 < n:
            g[:n - j2] += (1.0 - h[j2:])
            g[j2:] -= h[:n - j2]
    return g


def smooth_max_obj_and_grad(h, beta, top_k=15):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = float(np.max(corr))
    shifted = np.clip(beta * (corr - m), -60, 60)
    exp_s = np.exp(shifted)
    sum_exp = float(np.sum(exp_s))
    w = exp_s / sum_exp
    obj = (m + np.log(sum_exp) / beta) * dx
    top_lags = np.argsort(-w)[:top_k]
    g = np.zeros(n)
    for k in top_lags:
        wk = w[k]
        if wk < 1e-12:
            break
        lag = int(k) - (n - 1)
        g += wk * lag_gradient(h, n, lag)
    g *= dx
    return obj, g, m * dx
