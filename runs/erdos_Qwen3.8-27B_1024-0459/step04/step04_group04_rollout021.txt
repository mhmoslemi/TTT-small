from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            total = np.sum(h)
            if total < 1e-15:
                h[:] = 0.5
            else:
                h[:] = h * (target / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1
    return h_bin

def coord_grad_for_i(h, n, i):
    """Compute d(corr[k])/d(h[i]) for all k."""
    g = np.zeros(2 * n - 1)
    # s >= 0: first factor at shift s, valid if i <= n-1-s, i.e., s <= n-1-i
    for s in range(0, n - i):
        g[n-1+s] += (1.0 - h[i+s])

    # s < 0, t = -s: second factor, valid if t <= n-1-i
    for t in range(1, n-i):

    return g

def compute_all_coord_grads(h, n):
    """Compute grad[i] = d(corr)/d(h[i]) for all i.
    Returns array of shape (n, 2n-1)."""
    corr_len = 2 * n - 1
    grads = np.zeros((n, corr_len))
    for i in range(n):
        # Positive shifts s = 0, 1, ..., n-1
        # First factor: s from 0 to n-1-i
        if n - i > 0:
            s_range = np.arange(0, n - i)
            grads[i, n-1+s_range] += (1.0 - h[i:i+(n-i)])
        # Second factor: s from 0 to i
        if i >= 0:
            s_range = np.arange(0, i+1)
            grads[i, n-1+s_range] -= h[i:i-s_range-1:-1] if i > 0 else np.array([h[0]])
            # Simpler: h[i-s] for s=0..i is h[i], h[i-1], ..., h[0]
            grads[i, n-1:n+1] -= h[i::-1]
        
        # Negative shifts, t = 1, 2, ...
        # First factor: t from 1 to min(i, n-1)
        if i > 0:
            t_max = min(i, n-1)
            t_range = np.arange(1, t_max+1)
            grads[i, n-1-t_range] += (1.0 - h[i-1:i-1-t_range-1:-1])
            # h[i-t] for t=1..t_max is h[i-1], h[i-2], ..., h[i-t_max]
            grads[i, n-1-t_range] += (1.0 - h[i-t_max:i-1][::-1])
            # Hmm, this indexing is getting confusing. Let me just use a loop for clarity.
        
        # Actually, let me just do it with explicit indexing:
        # Reset
        g = np.zeros(corr_len)
        # s >= 0, first factor: s = 0..n-1-i
        for s in range(0, n - i):
            g[n-1+s] += (1.0 - h[i+s])
        # s >= 0, second factor: s = 0..i
        for s in range(0, i+1):
            g[n-1+s] -= h[i-s]
        # s < 0, first factor: t = 1..i (t = -s)
        for t in range(1, i+1):

        # s < 0, second factor: t = 1..n-1-i
        for t in range(1, n-i):
            g[n-1-t] -= h[i+t]
        grads[i] = g
    return grads
