def compute_c5(h):

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 3.0)

def obj_smooth(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(_G_BETA * shifted, -600, 600))
    return (m + np.log(np.sum(exps)) / _G_BETA) * dx

def obj_grad(x):
    """Gradient of the smooth max objective."""
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(_G_BETA * shifted, -600, 600))
    weights = exps / np.sum(exps)
    # Gradient of sum weights[k] * corr[k]
    # corr[k] = sum_i x[i] * (1-x[i+k]) for appropriate indices
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-10:
            continue
        offset = k_idx - (n - 1)
        if offset >= 0:
            nv = n - offset
            if nv <= 0:
                continue
            # corr[k_idx] = sum_{i=0}^{nv-1} x[i+offset] * (1-x[i])
            # d/dx[j]: (1-x[j]) if j in [offset, n-1], -x[j-offset] if j in [0, nv-1]
            grad[offset:offset+nv] += weights[k_idx] * (1.0 - x[:nv])
            grad[:nv] += weights[k_idx] * (-x[offset:offset+nv])
        else:
            mm = -offset
            nv = n - mm
            if nv <= 0:
                continue
            # corr[k_idx] = sum_{i=0}^{nv-1} x[i] * (1-x[i+mm])
            grad[:nv] += weights[k_idx] * (1.0 - x[mm:mm+nv])
            grad[mm:mm+nv] += weights[k_idx] * (-x[:nv])
    return grad * dx

            grad[offset:offset+nv] += (1.0 - h[:nv])
            grad[:nv] -= h[offset:offset+nv]
        else:
            mm = -offset
            nv = n - mm
            grad[:nv] += (1.0 - h[mm:mm+nv])
            grad[mm:mm+nv] -= h[:nv]

def lbfgs_smooth(h, n, beta=500, max_iter=200):
    """L-BFGS-B on the smooth max."""
    _G_N = n
    _G_BETA = float(beta)
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
    
    try:
        res = minimize(obj_smooth, h, jac=obj_grad, method='L-BFGS-B',
                       bounds=bounds, 
                       options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-10})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def slsqp_smooth(h, n, beta=500, max_iter=300):
    """SLSQP on the smooth max."""
    _G_N = n
    _G_BETA = float(beta)
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
    
    try:
        res = minimize(obj_smooth, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def compute_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):

        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:

        else:
