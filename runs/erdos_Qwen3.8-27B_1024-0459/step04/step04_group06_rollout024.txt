    # corr[k] = sum_i h[i] * (1-h[i+k]) for offset k-(n-1)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            for i in idx:
                adj = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
                h[i] += adj
                diff -= adj
                if abs(diff) < 1e-14:
                    break
    return h


def remaining_time():

def smooth_max_obj_grad(h, beta):
    """Compute smoothed max objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    # Clip to avoid overflow
    shifted = np.clip(shifted * beta, -300, 0)
    exps = np.exp(shifted)
    Z = np.sum(exps)
    weights = exps / Z
    obj = (m + np.log(Z) / beta) * dx
    
    # Gradient computation
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for k_idx in active:
        w = weights[k_idx]
        offset = int(k_idx) - (n - 1)

            grad[:n_valid] -= w * h[m_off:m_off + n_valid]
    
    grad *= dx
    return obj, grad


def projected_gradient(h, n, max_iter=5000, beta_start=50, beta_end=2000, 
                       step0=0.01, momentum=0.85):

    c5_best = compute_c5(h)
    h_best = h.copy()
    v = np.zeros(n)
    step = step0
    
    # Phase A: Low beta (finding the basin)
    n_iter_a = max_iter // 3
    # Phase B: Medium beta
    n_iter_b = max_iter // 3
    # Phase C: High beta
    n_iter_c = max_iter - n_iter_a - n_iter_b
    
    phases = [(beta_start, 200, n_iter_a), (200, 800, n_iter_b), (800, beta_end, n_iter_c)]
    
    for beta_lo, beta_hi, n_iter in phases:
        n_done = 0
        for it in range(n_iter):
            if remaining_time() < 5.0:
                return h_best, c5_best
            beta = beta_lo + (beta_hi - beta_lo) * (n_done / max(1, n_iter))
            obj, grad = smooth_max_obj_grad(h, beta)
            
            # Momentum step
            v = momentum * v - step * grad
            h_trial = project_feasible(h + v, n)
            c5_trial = compute_c5(h_trial)
            
            if c5_trial < c5_best - 1e-14:

                h_best = h_trial.copy()
                h = h_trial.copy()

                # Backtrack
                v *= 0.3
                step *= 0.7
                if step < 1e-6:

            n_done += 1
    
    return h_best, c5_best


def subgrad_momentum_v2(h, n, max_iter=3000, step0=0.05, momentum=0.85, top_k=5):

    h_best = h.copy()

    step = step0
    patience = 0

            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        
        # Find top-k correlations
        top_idx = np.argsort(corr)[-top_k:]
        top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]

        grad = np.zeros(n)

                if n_valid <= 0:

def make_thue_morse(n):
    h = np.zeros(n)

def make_golden_ratio(n):
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        h[i] = 1.0 if (i * phi) % 1.0 < 0.5 else 0.0
    return h

    """Multiplicative weights approach to the minimax problem."""
    dx = 2.0 / n
    c5_best = compute_c5(h)
    h_best = h.copy()
    
    # Initialize distribution over shifts
    n_shifts = 2 * n - 1  # number of correlation values
    p = np.ones(n_shifts) / n_shifts
    
    for it in range(max_iter):
        if remaining_time() < 5.0:
            break
        
        # Compute all correlations
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_current = float(np.max(corr)) * dx
        
        # Update distribution: concentrate on high correlations
        # Use softmax with temperature
        temp = 5.0
        shifted_corr = corr - np.max(corr)
        p_new = np.exp(shifted_corr / temp)
        p_new /= np.sum(p_new)
        p = 0.7 * p + 0.3 * p_new

        # Objective: sum_k p_k * corr[k] * dx
        # This is bilinear in h, so gradient is well-defined

        for k_idx in range(n_shifts):
            if p[k_idx] < 1e-10:
                continue

            offset = k_idx - (n - 1)
            if offset >= 0:
                n_valid = n - offset
                if n_valid <= 0:
                    continue
                grad[:n_valid] += w * (1.0 - h[offset:offset + n_valid])
                grad[offset:offset + n_valid] -= w * h[:n_valid]
            else:
                m_off = -offset
                n_valid = n - m_off
                if n_valid <= 0:

                grad[m_off:m_off + n_valid] += w * (1.0 - h[:n_valid])

        h_trial = project_feasible(h - step * grad, n)

Here's another idea: **Pairwise optimization**. Instead of changing one variable at a time, change two variables (i, j) simultaneously. This gives more flexibility while still being tractable.

For a pair (i, j), we want to find the optimal values h[i] and h[j] given all other variables fixed. The objective as a function of (h[i], h[j]) is:
max_k corr[k] where corr[k] is linear in h[i] and h[j] (since each corr[k] involves at most one term with h[i] and at most one term with h[j]).

So for each pair, the problem is a 2D minimax problem, which can be solved analytically or numerically.

This is a powerful approach! Let me implement it:

For fixed h[i]=a, h[j]=b, the correlation at offset k is:
corr[k] = c_k + α_k * a + β_k * b

where c_k is the contribution from all other variables, and α_k, β_k are coefficients that depend on the offset.

The minimax problem is:
minimize max_k (c_k + α_k * a + β_k * b) subject to a + b = s (where s is the required sum for positions i and j to maintain the integral constraint)

Wait, the constraint is sum(h) = n/2, so if we change h[i] from old_i to a and h[j] from old_j to b, we need a + b = old_i + old_j.

So: minimize max_k (c_k + α_k * a + β_k * (s - a)) = max_k (c_k + (α_k - β_k) * a + β_k * s)

This is a 1D minimax problem! The optimal a is where the top two lines cross (or at a boundary).

This is the **Cutting Plane** or **Simplicial** approach to 1D minimax.

def pair_optimize(h, i, j, n):
    """Optimize h[i] and h[j] simultaneously, keeping sum constant."""
    s = h[i] + h[j]  # Required sum
    dx = 2.0 / n
    
    # Compute base correlation (with h[i] and h[j] set to 0)
    h_base = h.copy()
    h_base[i] = 0.0
    h_base[j] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    
    # Coefficients for h[i]
    # corr[k] involves h[m] * (1-h[m+k]) or h[m+k] * (1-h[m])
    # h[i] appears in corr[k] when:
    #   i is in the "first" position: offset = k-(n-1), and i < n - offset, i.e., offset > -1 and i < n - offset
    #   i is in the "second" position: offset < 0 and i + |offset| < n, i.e., i < n + offset (offset < 0)

    #   corr = sum_{m=0}^{n-1-d} h[m]*(1-h[m+d])
    #   h[i] appears as h[m] when m=i (if i < n-d): contributes (1-h[i+d])
    #   h[i] appears as h[m+d] when m+d=i, i.e., m=i-d (if i>=d and i-d < n-d, i.e., i < n):
    #     contributes -h[i-d]
    
    # For offset d < 0 (k_idx = n-1+d, d = -|d|):
    #   corr = sum_{m=0}^{n-1-|d|} h[m+|d|]*(1-h[m])
    #   h[i] appears as h[m+|d|] when m+|d|=i, i.e., m=i-|d| (if i>=|d| and i-|d| < n-|d|, i.e., i < n):
    #     contributes (1-h[i-|d|])
    #   h[i] appears as (1-h[m]) when m=i (if i < n-|d|, i.e., i+|d| < n):
    #     contributes -h[i+|d|]
    
    # This is getting complex. Let me just compute the coefficients numerically.

    h_test = h_base.copy()
    h_test[i] = 1.0
    corr_i = np.correlate(h_test, 1.0 - h_test, mode='full')
    alpha = corr_i - corr_base  # Coefficient for h[i]
    
    # Set h[j]=1, h[j]=0, compute corr, subtract base
    h_test = h_base.copy()
    h_test[j] = 1.0
    corr_j = np.correlate(h_test, 1.0 - h_test, mode='full')

    # We want: min_a max_k g(k, a) subject to 0 <= a <= 1 and 0 <= s-a <= 1

    def f(a):
        return np.max(c + a * d)

        m = np.max(vals)

        return np.mean(d[active])
    
    # Bisection

    b_opt = s - a_opt
    
    h[i] = a_opt
    h[j] = b_opt
    return h

    h_base = h.copy()
    h_base[i] = 0.0
    h_base[j] = 0.0

    h_test = h_base.copy()
    h_test[i] = 1.0
    alpha = np.correlate(h_test, 1.0 - h_test, mode='full') - corr_base
    
    h_test = h_base.copy()
    h_test[j] = 1.0
    beta = np.correlate(h_test, 1.0 - h_test, mode='full') - corr_base
    
    # corr[k] = corr_base[k] + a*alpha[k] + (s-a)*beta[k] = corr_base[k] + s*beta[k] + a*(alpha[k]-beta[k])
    c = corr_base + s * beta
    d = alpha - beta
    
    a_lo = max(0.0, s - 1.0)
    a_hi = min(1.0, s)
    
    if a_lo >= a_hi - 1e-12:
        return h
    
    # Find minimum of max_k (c[k] + a*d[k]) over a in [a_lo, a_hi]
    # This is a 1D convex minimization (max of linear functions is convex)
    
    # The minimum is where the subgradient contains 0
    # Subgradient at a: {sum of d[k] for k in argmax(c + a*d)}
    
    # Bisection
    lo, hi = a_lo, a_hi
    for _ in range(40):
        mid = (lo + hi) / 2
        vals = c + mid * d
        m = np.max(vals)
        active = np.where(np.abs(vals - m) < 1e-10)[0]
        sg = np.mean(d[active])
        if sg > 1e-12:
            hi = mid
        elif sg < -1e-12:
            lo = mid
        else:
            break
    
    a_opt = (lo + hi) / 2
    b_opt = s - a_opt
    
    h[i] = a_opt
    h[j] = b_opt
    return h
