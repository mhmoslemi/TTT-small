        rng = np.random.RandomState(0)
    h_bin = (h > 0.5).astype(float)

    target = n // 2

        rng.shuffle(ones_idx)
        h_bin[ones_idx[target:]] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0.0)[0]
        rng.shuffle(zeros_idx)
        h_bin[zeros_idx[:target - n_ones]] = 1.0
    c5_best = compute_c5(h_bin)
    h_best = h_bin.copy()
    for sweep in range(max_sweeps):

        any_imp = False

        zeros_idx = list(np.where(h_best == 0.0)[0])
        rng.shuffle(ones_idx)
        rng.shuffle(zeros_idx)
        n_try = min(len(ones_idx), len(zeros_idx), 200)
        for a in range(n_try):
            if remaining_time() < 5.0:

            if h_best[ones_idx[a]] != 1.0 or h_best[zeros_idx[a]] != 0.0:

def shift_perturb(h, n, rng):

    h_trial = h.copy()
    s1 = rng.randint(0, n // 3)

    h_trial[s2:s2 + L] = seg
    h_trial = project_feasible(h_trial, n)
    return h_trial


def noise_perturb(h, n, rng, sigma=0.04):
    h_trial = h + sigma * rng.randn(n)
    return project_feasible(h_trial, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    N = 1024
    _G_N = N
    _G_TGT = N / 2.0
    _G_DX = 2.0 / N
    bounds = [(0.0, 1.0)] * N
    cons = [{'type': 'eq', 'fun': con_sum}]

    # Initialize from parent or Thue-Morse
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        h = project_feasible(interp_to(parent_h, parent_n, N), N)
        print("Parent n={} interpolated to n={}".format(parent_n, N))
    except Exception:
        h = make_thue_morse(N)
        print("Using Thue-Morse init, n={}".format(N))

    best_c5 = compute_c5(h)
    best_h = h.copy()
    print("Initial C5 = {:.10f}".format(best_c5))

    # Phase 1: Projected gradient momentum
    print("\n--- Phase 1: Projected gradient (n={}) ---".format(N))
    h, c5 = projected_gradient(h, N, max_iter=400, step0=0.08, momentum=0.85)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    print("  C5 = {:.10f}".format(best_c5))

    # Phase 2: SLSQP smooth max, escalating beta
    print("\n--- Phase 2: SLSQP smooth max ---")
    for beta_val in [50, 100, 300, 500, 1000, 2000, 5000]:
        if remaining_time() < 20.0:
            break
        _G_BETA = float(beta_val)
        try:
            res = minimize(obj_smooth_val, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-14, 'iprint': 0})
            h_new = project_feasible(res.x, N)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  beta={}: C5={:.10f}".format(beta_val, best_c5))
        except Exception:
            pass
    print("  After SLSQP: C5 = {:.10f}".format(best_c5))

    # Phase 3: Binary local search
    print("\n--- Phase 3: Binary local search ---")
    h_bin, c5_bin = binary_local_search(best_h, N, max_sweeps=8, rng=rng)
    # Use binary solution as restart for SLSQP
    h_cont = h_bin + 0.01 * rng.randn(N)
    h_cont = project_feasible(h_cont, N)
    for beta_val in [300, 1000, 3000]:
        if remaining_time() < 15.0:
            break
        _G_BETA = float(beta_val)
        try:
            res = minimize(obj_smooth_val, h_cont, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-14, 'iprint': 0})
            h_new = project_feasible(res.x, N)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass
    print("  After binary+SLSQP: C5 = {:.10f}".format(best_c5))

    # Phase 4: Structured perturbation exploration
    print("\n--- Phase 4: Structured perturbation ---")
    for p in range(20):
        if remaining_time() < 20.0:
            break
        h_trial = best_h.copy()
        pt = p % 5
        if pt == 0:
            h_trial = block_flip_perturb(h_trial, N, rng)
        elif pt == 1:
            h_trial = shift_perturb(h_trial, N, rng)
        elif pt == 2:
            h_trial = noise_perturb(h_trial, N, rng, sigma=0.03)
        elif pt == 3:
            # Flip a sparse random subset
            idx = rng.choice(N, size=N // 20, replace=False)
            h_trial[idx] = 1.0 - h_trial[idx]
            h_trial = project_feasible(h_trial, N)
        else:
            # Sinusoidal perturbation
            freq = rng.randint(3, N // 8)
            phase = rng.rand() * 2 * np.pi
            t = np.arange(N) / float(N)
            h_trial += 0.03 * np.sin(2.0 * np.pi * freq * t + phase)
            h_trial = project_feasible(h_trial, N)

        # Quick gradient step
        h_trial, c5_t = projected_gradient(h_trial, N, max_iter=50, step0=0.05, momentum=0.8)

        if c5_t < best_c5 - 1e-11:
            best_c5 = c5_t
            best_h = h_trial.copy()
            print("  Perturb {}: C5={:.10f}".format(p, best_c5))

        # SLSQP from perturbed point
        if remaining_time() < 10.0:
            break
        _G_BETA = 500.0
        try:
            res = minimize(obj_smooth_val, h_trial, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 100, 'ftol': 1e-13, 'iprint': 0})
            h_new = project_feasible(res.x, N)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass
    print("  After perturbation: C5 = {:.10f}".format(best_c5))

    # Phase 5: Second round of binary + SLSQP
    if remaining_time() > 60.0:
        print("\n--- Phase 5: Binary + SLSQP round 2 ---")
        h_bin2, c5_bin2 = binary_local_search(best_h, N, max_sweeps=6, rng=rng)
        h_cont2 = h_bin2 + 0.008 * rng.randn(N)
        h_cont2 = project_feasible(h_cont2, N)
        for beta_val in [200, 800, 2000, 5000]:
            if remaining_time() < 12.0:
                break
            _G_BETA = float(beta_val)
            try:
                res = minimize(obj_smooth_val, h_cont2, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 250, 'ftol': 1e-14, 'iprint': 0})
                h_new = project_feasible(res.x, N)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-13:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass
        print("  C5 = {:.10f}".format(best_c5))

    # Phase 6: Gradient + SLSQP with gradient-provided
    if remaining_time() > 80.0:
        print("\n--- Phase 6: Gradient SLSQP ---")
        # Use objective with gradient (top-50 lags)
        for beta_val in [500, 1500, 4000]:
            if remaining_time() < 15.0:
                break
            _G_BETA = float(beta_val)
            try:
                res = minimize(obj_smooth_grad, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-14, 'iprint': 0})
                h_new = project_feasible(res.x, N)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  beta={}: C5={:.10f}".format(beta_val, best_c5))
            except Exception:
                pass

    # Phase 7: Final fine-tuning gradient
    if remaining_time() > 30.0:
        print("\n--- Phase 7: Final gradient ---")
        best_h, best_c5 = projected_gradient(best_h, N, max_iter=200, step0=0.04, momentum=0.8)
        print("  C5 = {:.10f}".format(best_c5))

    # Phase 8: Very high beta SLSQP polish
    if remaining_time() > 20.0:
        print("\n--- Phase 8: High-beta polish ---")
        for beta_val in [5000, 10000, 20000]:
            if remaining_time() < 8.0:
                break
            _G_BETA = float(beta_val)
            try:
                res = minimize(obj_smooth_val, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-15, 'iprint': 0})
                h_new = project_feasible(res.x, N)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass
        print("  C5 = {:.10f}".format(best_c5))

    # Phase 9: Tiny binary tweaks
    if remaining_time() > 10.0:
        print("\n--- Phase 9: Final binary tweaks ---")
        h_bin3, c5_bin3 = binary_local_search(best_h, N, max_sweeps=3, rng=rng)
        if c5_bin3 < best_c5:
            best_c5 = c5_bin3
            best_h = h_bin3.copy()
        # Quick SLSQP from binary
        if remaining_time() > 8.0:
            _G_BETA = 3000.0
            h_start = best_h + 0.005 * rng.randn(N)
            h_start = project_feasible(h_start, N)
            try:
                res = minimize(obj_smooth_val, h_start, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 200, 'ftol': 1e-15, 'iprint': 0})
                h_new = project_feasible(res.x, N)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass
        print("  C5 = {:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, N)
    # Ensure exact sum
    diff = N / 2.0 - np.sum(best_h)
    if abs(diff) > 1e-10:
        best_h = project_feasible(best_h, N)

    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(N, final_c5))
    return (best_h, final_c5, N)
