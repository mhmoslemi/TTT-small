def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        perm = rng.permutation(len(idx))[:n_ones - target]
        h_bin[idx[perm]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        perm = rng.permutation(len(idx))[:target - n_ones]
        h_bin[idx[perm]] = 1
    return h_bin


def gen_structured(n, rng, method=0):

    h = np.zeros(n)

        rem = n - reps * 3

            h[reps * 3:] = block[:rem]
    elif method == 1:

def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g

    return np.sum(x) - _TGT

    return np.ones(_N)


def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-14:
            g += w[k] * corr_grad_per_shift(x, n, k)
    return g * dx_val


def slsqp_smooth(h, n, beta, maxiter=400):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)


def topk_gradient_descent(h, n, max_iters=300, top_k=5):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        m = float(np.max(corr))
        if m * (2.0 / n) >= best_c5 + 1e-6:
            break
        top_idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k in top_idx:
            g += corr_grad_per_shift(h, n, int(k))
        g /= len(top_idx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        improved = False
        for step in [0.05, 0.02, 0.01, 0.1, 0.005, 0.03, 0.008, 0.15, 0.2]:
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-14:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            # Try random perturbation
            h_trial = h + 0.01 * np.random.RandomState(it).randn(n)
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-14:
                h = h_trial
                best_c5 = c_new
            else:

    return h, best_c5

            else:
                h_init = interp_to(best_h, best_n, n_t)
                h_init = project_feasible(h_init, n_t)
            h_ref, c5_ref = continuous_refine(h_init, n_t, time_limit=t_per_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
                best_n = n_t

    # Phase 3: More SA from the best solution (binary) at best_n and neighbors
    if remaining() > 40:
        sa_extra_n = [best_n - 2, best_n, best_n + 2]
        sa_extra_n = [n for n in sa_extra_n if n % 2 == 0 and n >= 100]
        t_per_extra = min(60, remaining() * 0.5 / max(1, len(sa_extra_n)))
        for n_t in sa_extra_n:
            if remaining() < 20:
                break
            h_init = interp_to(best_h, best_n, n_t)
            h_bin = make_binary(h_init, n_t, rng)
            for extra_seed in [seed + 50000 + n_t, seed + 60000 + n_t * 3]:
                if remaining() < 15:
                    break
                h_sa, c5_sa = sa_binary_improved(h_bin, n_t, time_limit=t_per_extra / 2, seed=extra_seed)
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
                    best_n = n_t

    # Phase 4: Final continuous polish
    if remaining() > 15:
        n = best_n
        h = best_h.copy()
        t_polish = remaining() * 0.6
        h_pol, c5_pol = continuous_refine(h, n, time_limit=t_polish)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
