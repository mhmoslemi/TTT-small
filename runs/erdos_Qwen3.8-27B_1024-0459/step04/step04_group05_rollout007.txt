def make_thue_morse(n):

def compute_grad_corr(h, k_offset, n):

    # k_offset is the shift in [0, 2n-2] for np.correlate full mode
    # m = k_offset - (n-1) is the actual lag
    m = k_offset - (n - 1)
    grad = np.zeros(n)

        n_pairs = n - m
        grad[:n_pairs] += (1.0 - h[m:m + n_pairs])
        grad[m:] -= h[:n_pairs]

        n_pairs = n - abs_m
        grad[:n_pairs] -= h[abs_m:abs_m + n_pairs]
        grad[abs_m:] += (1.0 - h[:n_pairs])
    return grad

    m = k_idx - (n - 1)
    grad = np.zeros(n)
    if m >= 0:
        n_pairs = n - m
        if n_pairs <= 0:
            return grad
        grad[:n_pairs] += (1.0 - h[m:m + n_pairs])
        grad[m:m + n_pairs] -= h[:n_pairs]

        grad[:n_pairs] -= h[abs_m:abs_m + n_pairs]
        grad[abs_m:abs_m + n_pairs] += (1.0 - h[:n_pairs])
    return grad

    c5_cur = compute_c5(h_start)
    h_cur = h_start.copy()
    dx = 2.0 / n

    for it in range(max_iter):
        if time_check is not None and time_check() < 4.0:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        top_idx = sorted([int(x) for x in top_idx])

        A_ub_rows = []
        b_ub_vals = []
        for k_idx in top_idx:
            q_val = corr[k_idx] * dx
            g = compute_grad_one_shift(h_cur, k_idx, n) * dx
            row = np.zeros(n + 1)
            row[:n] = g
            row[n] = -1.0
            A_ub_rows.append(row)
            b_ub_vals.append(q_val - np.dot(g, h_cur))

        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds_lp = [(0.0, 1.0)] * n + [(0.0, 1.0)]

        try:
            A_ub = np.array(A_ub_rows)
            b_ub = np.array(b_ub_vals)
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds_lp, method='highs',
                          options={'presolve': True, 'time_limit': 2.0})
            if res.status != 0:
                break
            h_lp = res.x[:n]
            best_t = c5_cur
            best_h = h_cur.copy()
            for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 0.8, 0.4, 0.6]:
                h_trial = (1.0 - alpha) * h_cur + alpha * h_lp
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_t:
                    best_t = c5_trial
                    best_h = h_trial.copy()
            if best_t < c5_cur - 1e-17:
                c5_cur = best_t
                h_cur = best_h
            else:
                break
        except Exception:
            break
    return h_cur, c5_cur


def binary_local_search(h_bin, n, rng, time_limit=30.0):
    t_start = time.time()
    c5 = compute_c5(h_bin)
    best_c5 = c5
    best_h = h_bin.copy()
    no_improve = 0
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()

    while time.time() - t_start < time_limit:
        if no_improve > 300:
            break
        found = False
        n_ones = len(ones)
        n_zeros = len(zeros)
        n_tries = min(80, max(10, n_ones * n_zeros // 200))
        for _ in range(n_tries):
            i = rng.randint(n_ones)
            j = rng.randint(n_zeros)
            oi = ones[i]
            oz = zeros[j]
            h_bin[oi] = 0.0
            h_bin[oz] = 1.0
            c5_trial = compute_c5(h_bin)
            h_bin[oi] = 1.0
            h_bin[oz] = 0.0
            if c5_trial < best_c5 - 1e-15:
                h_bin[oi] = 0.0
                h_bin[oz] = 1.0
                ones[i] = oz
                zeros[j] = oi
                best_c5 = c5_trial
                best_h = h_bin.copy()
                found = True
                no_improve = 0
                break
            else:
                no_improve += 1
        if not found:
            no_improve += 1
            if no_improve > 300:
                break
    return best_h, best_c5


def do_fista(h_start, n, time_check, max_iter=3000):
    global _G_N, _G_BETA, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    h = h_start.copy()
    h_prev = h.copy()
    t_k = 1.0
    beta = 200.0
    L = 1.0
    c5_cur = compute_c5(h)

    for it in range(max_iter):
        if time_check is not None and time_check() < 3.0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        shifted = corr - m
        weights = np.exp(np.clip(beta * shifted, -500, 0))
        wsum = np.sum(weights)
        weights /= wsum
        grad = np.zeros(n)
        top_weights = np.argsort(weights)[-min(80, len(weights)):]
        for idx in top_weights:
            if weights[idx] > 1e-8:
                grad += weights[idx] * compute_grad_one_shift(h, int(idx), n)
        grad *= (2.0 / n)

        h_new = h_prev - (1.0 / L) * grad
        h_new = project_feasible(h_new, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        h_prev = h_new + ((t_k - 1.0) / t_new) * (h_new - h)
        h = h_new.copy()
        t_k = t_new

        c5_new = compute_c5(h)
        if c5_new < c5_cur - 1e-17:
            c5_cur = c5_new
    return h, c5_cur


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    def time_check():
        return max(0.0, _G_BUDGET - (time.time() - _G_START) - 3.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = make_thue_morse(parent_n)
        best_h = project_feasible(parent_h, parent_n)
        best_c5 = compute_c5(best_h)
        best_n = parent_n
        print("TM: n={}, C5={:.10f}".format(parent_n, best_c5))

    # Phase 1: SLSQP with beta sweep at parent resolution
    print("\nPhase 1: SLSQP beta sweep n={}".format(best_n))
    best_h, best_c5 = do_slsqp(best_h, best_n, [30, 100, 300, 1000, 3000], time_check, maxiter_base=120)
    print("  C5={:.10f}".format(best_c5))

    # Phase 2: Binary local search for diverse inits
    if time_check() > 60:
        print("\nPhase 2: Binary local search")
        h_bin = make_bin_from_real(best_h, best_n)
        c5_bin = compute_c5(h_bin)
        best_bin_c5 = c5_bin
        best_bin_h = h_bin.copy()
        print("  Initial binary: C5={:.10f}".format(c5_bin))

        for trial in range(6):
            if time_check() < 15:
                break
            h_trial_bin = best_bin_h.copy()
            # Random perturbation
            n_swaps = 10 + trial * 5
            ones_list = np.where(h_trial_bin == 1)[0].tolist()
            zeros_list = np.where(h_trial_bin == 0)[0].tolist()
            for _ in range(n_swaps):
                if not ones_list or not zeros_list:
                    break
                i = rng.randint(len(ones_list))
                j = rng.randint(len(zeros_list))
                h_trial_bin[ones_list[i]] = 0.0
                h_trial_bin[zeros_list[j]] = 1.0
                ones_list[i] = zeros_list[j]
                zeros_list[j] = ones_list[i]
            h_res, c5_res = binary_local_search(h_trial_bin, best_n, rng, time_limit=max(5.0, time_check() * 0.08))
            if c5_res < best_bin_c5:
                best_bin_c5 = c5_res
                best_bin_h = h_res
                print("  Binary improved: C5={:.10f}".format(best_bin_c5))

        # Use best binary to initialize continuous
        if best_bin_c5 < best_c5:
            best_c5 = best_bin_c5
            best_h = best_bin_h.copy()
            best_n = best_n

        # Continue with best available
        best_h, best_c5 = do_slsqp(best_h, best_n, [500, 2000, 8000], time_check, maxiter_base=100)
        print("  After refine: C5={:.10f}".format(best_c5))

    # Phase 3: SQP with LP subproblem (active set)
    if time_check() > 30:
        print("\nPhase 3: SQP-LP")
        best_h, best_c5 = do_sqp_lp(best_h, best_n, top_k=5, time_check=time_check, max_iter=300)
        print("  C5={:.10f}".format(best_c5))

    # Phase 4: FISTA
    if time_check() > 30:
        print("\nPhase 4: FISTA")
        best_h, best_c5 = do_fista(best_h, best_n, time_check, max_iter=2000)
        print("  C5={:.10f}".format(best_c5))

    # Phase 5: SQP-LP again (now with better init)
    if time_check() > 20:
        print("\nPhase 5: SQP-LP round 2")
        best_h, best_c5 = do_sqp_lp(best_h, best_n, top_k=8, time_check=time_check, max_iter=300)
        print("  C5={:.10f}".format(best_c5))

    # Phase 6: Multi-resolution
    if time_check() > 80:
        print("\nPhase 6: Multi-resolution")
        for n_try in [384, 512, 768, 1024]:
            if time_check() < 40:
                break
            h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
            c5_interp = compute_c5(h_interp)
            print("  n={} interp C5={:.10f}".format(n_try, c5_interp))
            h_new, c5_new = do_slsqp(h_interp, n_try, [200, 1000, 5000], time_check, maxiter_base=60)
            if time_check() > 20:
                h_new, c5_new = do_sqp_lp(h_new, n_try, top_k=5, time_check=time_check, max_iter=100)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_try
                print("  n={} IMPROVED: C5={:.10f}".format(n_try, best_c5))
            else:
                print("  n={} no improve: C5={:.10f}".format(n_try, c5_new))

    # Phase 7: Final high-beta SLSQP polish
    if time_check() > 12:
        print("\nPhase 7: High-beta polish")
        best_h, best_c5 = do_slsqp(best_h, best_n, [10000, 30000, 100000], time_check, maxiter_base=80)
        print("  C5={:.10f}".format(best_c5))

    # Phase 8: Final SQP-LP
    if time_check() > 10:
        best_h, best_c5 = do_sqp_lp(best_h, best_n, top_k=10, time_check=time_check, max_iter=200)
        print("  After final SQP: C5={:.10f}".format(best_c5))

    # Phase 9: Tiny FISTA + SLSQP
    if time_check() > 8:
        best_h, best_c5 = do_fista(best_h, best_n, time_check, max_iter=500)
    if time_check() > 6:
        best_h, best_c5 = do_slsqp(best_h, best_n, [50000], time_check, maxiter_base=40)

    # Final
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
