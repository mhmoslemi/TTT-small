```python
def project_box_sum(v, n):
    """Project v onto {x : 0<=x<=1, sum(x) = n/2}"""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(v - mid, 0, 1)
        s = projected.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    projected = np.clip(v - (lo+hi)/2, 0, 1)
    # Fix any residual
    diff = target - projected.sum()
    if abs(diff) > 1e-12:
        mask = (projected > 0) & (projected < 1)
        if mask.sum() > 0:
            projected[mask] += diff / mask.sum()
            projected = np.clip(projected, 0, 1)
    return projected
```

```python
        else:
            mom_eff = 0.9
        velocity = mom_eff * velocity - step_init * grad
        h_new = project_box_sum(h + velocity, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new > val + 1.0:
            step_init *= 0.5
            velocity *= 0.5
            momentum *= 0.5
            h_new = h
        elif c5_new < best_c5 * 0.99:
            step_init = min(step_init * 1.1, 0.5)
        h = h_new
    return best_h, best_c5

def paired_coord_descent(h, n, max_sweeps, rng, tlimit):
    c5 = compute_c5(h)
    h_best = h.copy()
    for sweep in range(max_sweeps):
        if tlimit() < 3.0:
            break
        _, grad = lob_val_grad(h_best, 2000.0, n)
        improved = False
        order = rng.permutation(n)
        for i in order:
            if tlimit() < 1.0:
                break
            gi = grad[i]
            rest = np.delete(grad, i)
            j_candidates = []
            for j in range(n):
                if j == i:
                    continue
                j_candidates.append((abs(gi - grad[j]), j))
            j_candidates.sort(reverse=True)
            for trial_s in [0.02, 0.05, 0.1, 0.2]:
                for _, j in j_candidates[:3]:
                    trial = h_best.copy()
                    trial[i] = trial[i] + trial_s
                    trial[j] = trial[j] - trial_s
                    if trial[i] < 0 or trial[i] > 1 or trial[j] < 0 or trial[j] > 1:
                        continue
                    c = compute_c5(trial)
                    if c < c5 - 1e-10:
                        c5 = c
                        h_best = trial.copy()
                        improved = True
                        break
                if improved and c5 < c5:
                    break
            for trial_s in [-0.02, -0.05, -0.1, -0.2]:
                for _, j in j_candidates[:3]:
                    trial = h_best.copy()
                    trial[i] = trial[i] + trial_s
                    trial[j] = trial[j] - trial_s
                    if trial[i] < 0 or trial[i] > 1 or trial[j] < 0 or trial[j] > 1:
                        continue
                    c = compute_c5(trial)
                    if c < c5 - 1e-10:
                        c5 = c
                        h_best = trial.copy()
                        improved = True
                        break
        if not improved and sweep >= 3:
            break
    return h_best, c5

def focused_multi_slsqp(h, n, rng, tlimit, max_rounds=25, top_k=3):
    c5 = compute_c5(h)
    h_best = h.copy()
    global _G_CONST_N
    for rnd in range(max_rounds):
        if tlimit() < 5.0:
            break
        corr = get_corr(h_best)
        dx = 2.0 / n
        top_idx = np.argsort(corr * dx)[-top_k:]
        _G_CONST_N = n
        beta = 2000.0
        def multi_obj(x):
            corr_x = get_corr(x)
            top_vals = corr_x[top_idx] * dx
            m = np.max(top_vals)
            s = top_vals - m
            e = np.exp(np.clip(50.0 * s, -300, 0))
            lse = (m + np.log(np.sum(e)) / 50.0)
            v, g = lob_val_grad(x, beta, n)
            extra = 0.1 * lse
            return v + extra, g
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac}]
        try:
            res = minimize(multi_obj, h_best, jac=True, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 150, 'ftol': 1e-15, 'iprint': 0})
            h_new = project_box_sum(res.x, n)
            c_new = compute_c5(h_new)
            if c_new < c5 - 1e-12:
                c5 = c_new
                h_best = h_new
        except Exception:
            pass
    return h_best, c5

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c else 0.0
    return h

def make_rand_bin(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def make_flat_spectral(n, rng):
    h0 = np.full(n, 0.5)
    for harmonic in range(2, n // 2 + 1):
        amp = 0.02 / np.sqrt(harmonic)
        phase = rng.rand() * 2.0 * np.pi
        h0 += amp * np.cos(2.0 * np.pi * harmonic * np.arange(n) / n + phase)
    return np.clip(h0, 0.0, 1.0)

def run(seed=42, budget_s=1300, **kwargs):

    _G_PEN = 0.0
    _G_CONST_N = 256
    rng = np.random.RandomState(seed)

    def tlimit():
        return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = make_tm(parent_n)
        parent_h = project_box_sum(parent_h, parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("No parent. TM: n={}, C5={:.10f}".format(parent_n, best_c5))

    # Phase 1: Intensive projected gradient at parent resolution
    if tlimit() > 120:
        print("\n=== Phase 1: PG n={} ===".format(best_n))
        n = best_n

        for beta in [20, 50, 150, 400, 1000, 3000]:
            if tlimit() < 10:
                break
            h_cur, c5_cur = projected_grad(h_cur, n, beta, max_iter=2000, tlimit=tlimit)
            if c5_cur < best_c5 - 1e-12:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n

            h_new = slsqp_optimize(h_cur, n, beta, maxiter=500)

            if c_new < best_c5 - 1e-13:
                best_c5 = c_new
                best_h = h_new.copy()
                best_n = n
                h_cur = h_new
        print("  After SLSQP: C5={:.10f}".format(best_c5))

    # Phase 3: Multi-lag focused SLSQP
    if tlimit() > 60:
        print("\n=== Phase 3: Multi-lag n={} ===".format(best_n))
        n = best_n
        h_cur, c5_cur = focused_multi_slsqp(best_h.copy(), n, rng, tlimit, max_rounds=15, top_k=3)
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n
        print("  After multi-lag: C5={:.10f}".format(best_c5))

    # Phase 4: Paired coordinate descent
    if tlimit() > 40:
        print("\n=== Phase 4: Paired CD n={} ===".format(best_n))
        n = best_n
        best_h, best_c5 = paired_coord_descent(best_h, n, max_sweeps=20, rng=rng, tlimit=tlimit)
        print("  After paired CD: C5={:.10f}".format(best_c5))

                    h_cur = h_new
        if c5_512 < best_c5 - 1e-12:

            h_pert = project_box_sum(h_pert, n)

    try:

    _G_PEN = 0.0
```
