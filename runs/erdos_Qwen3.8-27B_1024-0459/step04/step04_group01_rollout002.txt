            # For each correlation index k, shift d = k - (n-1)
            for k in range(n_corr):
                d = k - (n - 1)
                s = 0.0
                if d >= 0:
                    # corr[k] = sum_{j=0}^{n-1-d} h[j]*(1-h[j+d])
                    # h[i] appears as h[j] if j=i, i in [0, n-1-d]: +1, -h[i+d]
                    if 0 <= i <= n-1-d:
                        s += 1.0
                        s -= h[i+d]
                    # h[i] appears as h[j+d] if j+d=i, j=i-d, i-d in [0, n-1-d]: -h[i-d]
                    if 0 <= i-d <= n-1-d:
                        s -= h[i-d]
                else:
                    m = -d  # m > 0
                    # corr[k] = sum_{j=m}^{n-1} h[j]*(1-h[j-m])
                    # h[i] appears as h[j] if j=i, i in [m, n-1]: +1, -h[i-m]
                    if m <= i <= n-1:
                        s += 1.0
                        s -= h[i-m]
                    # h[i] appears as h[j-m] if j-m=i, j=i+m, i+m in [m, n-1]: -h[i+m]
                    if i >= 0 and i+m <= n-1:
                        s -= h[i+m]
                slopes[k] = s
            
            # Minimize max_k (slopes[k] * t + (corr[k] - slopes[k]*h[i]))
            # = max_k (slopes[k] * (t - h[i]) + corr[k])
            # Let u = t - h[i], minimize max_k (slopes[k]*u + corr[k])
            # u in [max(0, -h[i]), min(1, 1-h[i])]
            lo_u = max(0.0, -h[i])
            hi_u = min(1.0, 1.0 - h[i])

                continue

                vals = slopes * mid_u + intercepts
                max_val = vals.max()
                active = vals >= max_val - 1e-14

                    lo_u = mid_u

                    hi_u = mid_u
                else:
                    break

            new_val = h[i] + u_opt

            # Update corr
            corr += (new_val - h[i]) * slopes
            h[i] = new_val

        h = project_feasible(h, n)
        corr = compute_corr(h)
        
        c5_new = float(np.max(corr) * dx_val)
        if c5_new < best_c5:
            best_c5 = c5_new
    
    return h, best_c5
