    # Phase 7: Higher resolution
    if remaining_time() > 100.0:
        print("\n--- Phase 7: Higher resolution ---")
        for n_try in [320, 512]:
            if remaining_time() < 50.0:
                break
            h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
            _G_N = n_try
            _G_TGT = n_try / 2.0
            _G_DX = 2.0 / n_try
            c5_h = compute_c5(h_interp)
            h_interp, c5_h = focused_subgrad(h_interp, n_try, max_iter=150, step0=0.06, top_k_init=3)
            h_interp, c5_h = slsqp_polish(h_interp, n_try, betas=[500, 2000, 10000], max_iter=150)
            h_interp, c5_h = targeted_coord_descent(h_interp, n_try, max_sweeps=6, rng=rng)
            print("  n={}: C5={:.10f}".format(n_try, c5_h))
            if c5_h < best_c5 - 1e-12:
                best_c5 = c5_h
                best_h = h_interp.copy()
                best_n = n_try

    # Phase 8: Lower resolution
    if remaining_time() > 50.0:
        print("\n--- Phase 8: Lower resolution ---")
        for n_try in [192, 160]:
            if remaining_time() < 15.0:
                break
            h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
            _G_N = n_try
            _G_TGT = n_try / 2.0
            _G_DX = 2.0 / n_try
            c5_h = compute_c5(h_interp)
            h_interp, c5_h = focused_subgrad(h_interp, n_try, max_iter=150, step0=0.1, top_k_init=3)
            h_interp, c5_h = slsqp_polish(h_interp, n_try, betas=[500, 5000, 20000], max_iter=200)
            h_interp, c5_h = targeted_coord_descent(h_interp, n_try, max_sweeps=6, rng=rng)
            print("  n={}: C5={:.10f}".format(n_try, c5_h))
            if c5_h < best_c5 - 1e-12:
                best_c5 = c5_h
                best_h = h_interp.copy()
                best_n = n_try

    # Phase 9: Final intensive polish
    if remaining_time() > 15.0:
        print("\n--- Phase 9: Final polish ---")
        n = best_n
        _G_N = n
        _G_TGT = n / 2.0
        _G_DX = 2.0 / n
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_sum}]
        for beta_val in [10000, 30000, 50000]:
            if remaining_time() < 4.0:
                break
            _G_BETA = float(beta_val)
            try:
                res = minimize(obj_smooth, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-16, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    best_h = h_new
                    print("  beta={}: C5={:.10f}".format(beta_val, best_c5))
            except Exception:
                pass
        if remaining_time() > 4.0:
            best_h, best_c5 = targeted_coord_descent(best_h, best_n, max_sweeps=4, rng=rng)
        print("  C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            final_c5 = min(final_c5, eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
