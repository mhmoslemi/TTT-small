        if x not in seen:
            seen.add(x)
            out.append(x)
    return out

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def spectral_init(n, K_modes=8, seed=42):
    """Create initial solution from spectral construction."""
    rng = np.random.RandomState(seed)
    x = np.arange(n)
    h = np.full(n, 0.5)
    # Add structured sinusoidal components
    for k in range(1, K_modes + 1):
        freq = k * n / (n + 1)  # avoid exact period
        amp = 0.15 / (1 + 0.3 * k)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * x / n + phase)
    h = project(h, n)
    return h

def alternating_lag_optimize(h, n, max_rounds=30, top_k=8):
    """Iteratively optimize to equalize top correlation lags."""
    dx = 2.0 / n
    h = h.copy()
    for rnd in range(max_rounds):
        if remaining() < 10:
            break
        corr = compute_corr(h)
        max_c = np.max(corr)
        # Find lags within 1% of max
        threshold = max_c * 0.999
        active = np.where(corr >= threshold)[0]
        if len(active) > top_k:
            active = np.argsort(corr)[::-1][:top_k]
        active = np.array(sorted(active))
        if len(active) < 2:
            active = np.argsort(corr)[::-1][:top_k]
            active = np.array(sorted(active))

        # Solve minimax over active lags with SLSQP
        def obj_a(x):
            return x[n]
        def ineq_a(x):
            h_t = x[:n]
            t = x[n]
            c = np.correlate(h_t, 1.0 - h_t, mode='full')
            return t - dx * c[active]
        def eq_a(x):
            return np.sum(x[:n]) - n / 2.0

        x0 = np.concatenate([h, [max_c * dx + 1e-8]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [{'type': 'ineq', 'fun': ineq_a}, {'type': 'eq', 'fun': eq_a}]
        try:
            res = minimize(obj_a, x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-15, 'iprint': 0})
            if res.x is not None:
                h_new = project(res.x[:n], n)
                c5_new = compute_c5(h_new)
                if c5_new < compute_c5(h) - 1e-16:
                    h = h_new
                else:
                    break
        except Exception:
            break
    return h

def local_refine(h, n, max_steps=2000, rng=None):
    """Local refinement via single-point and pair adjustments."""
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    target = n / 2.0

    for step in range(max_steps):
        if remaining() < 3:
            break
        # Single point adjustment
        i = rng.randint(n)
        hi = h[i]
        if hi < 0.01 or hi > 0.99:
            continue
        delta = rng.uniform(-0.05, 0.05)
        h_trial = h.copy()
        h_trial[i] = np.clip(hi + delta, 0, 1)
        # Restore integral
        diff = target - np.sum(h_trial)
        if abs(diff) > 1e-12:
            # Distribute to nearest non-boundary point
            candidates = [(j, h[j]) for j in range(n) if j != i and h[j] > 0.01 and h[j] < 0.99]
            if candidates:
                j = rng.choice([c[0] for c in candidates])
                h_trial[j] = np.clip(h_trial[j] + diff, 0, 1)
        if 0 <= h_trial[i] <= 1 and abs(np.sum(h_trial) - target) < 1e-10:
            c5_t = compute_c5(h_trial)
            if c5_t < best_c5 - 1e-15:
                h = h_trial
                best_c5 = c5_t

    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Get parent solution
    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
    except Exception:
        best_n = 256
        best_h = np.full(best_n, 0.5)
        best_h = project(best_h, best_n)

    best_c5 = compute_c5(best_h)
    n = best_n
    h = best_h.copy()

    # Phase 1: Progressive smooth optimization at parent resolution
    betas = [200, 1000, 5000, 20000, 80000]
    for beta in betas:
        if remaining() < 40:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Alternating lag minimax
    if remaining() > 60:
        h_mm = alternating_lag_optimize(best_h, best_n, max_rounds=25, top_k=10)
        c5_mm = compute_c5(h_mm)
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm.copy()

    # Phase 3: Higher resolution attempts
    trial_ns = [384, 512, 640]
    for trial_n in trial_ns:
        if remaining() < 80:
            break
        h_hi = interp_to(best_h, best_n, trial_n)
        h_hi = project(h_hi, trial_n)
        c5_hi = compute_c5(h_hi)

        if c5_hi < best_c5 + 0.005:
            # Smooth optimization at higher resolution
            for beta in [500, 5000, 20000]:
                if remaining() < 30:
                    break
                h_hi = slsqp_smooth(h_hi, trial_n, beta, maxiter=300)
                c5_hi = compute_c5(h_hi)
                if c5_hi < best_c5:
                    best_c5 = c5_hi
                    best_h = h_hi.copy()
                    best_n = trial_n

            # Minimax at higher resolution
            if remaining() > 40 and c5_hi < best_c5 + 0.002:
                h_mm_hi = alternating_lag_optimize(h_hi, trial_n, max_rounds=10, top_k=8)
                c5_mm_hi = compute_c5(h_mm_hi)
                if c5_mm_hi < best_c5:
                    best_c5 = c5_mm_hi
                    best_h = h_mm_hi.copy()
                    best_n = trial_n

    # Phase 4: Back to best resolution, final smooth polish
    if remaining() > 40:
        h_curr = best_h.copy()
        for beta in [50000, 200000, 500000]:
            if remaining() < 20:
                break
            h_new = slsqp_smooth(h_curr, best_n, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_curr = h_new.copy()

    # Phase 5: Minimax with larger active set
    if remaining() > 50:
        h_mm = alternating_lag_optimize(best_h, best_n, max_rounds=15, top_k=15)
        c5_mm = compute_c5(h_mm)
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm.copy()

    # Phase 6: Local refinement
    if remaining() > 30:
        h_loc, c5_loc = local_refine(best_h, best_n, max_steps=3000, rng=rng)
        if c5_loc < best_c5:
            best_c5 = c5_loc
            best_h = h_loc.copy()

    # Phase 7: Final polish with very high beta
    if remaining() > 15:
        h_new = slsqp_smooth(best_h, best_n, beta=1000000, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 8: One more minimax pass
    if remaining() > 15:
        h_mm = alternating_lag_optimize(best_h, best_n, max_rounds=10, top_k=20)
        c5_mm = compute_c5(h_mm)
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm.copy()

    # Final projection and evaluation
    best_h = project(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
