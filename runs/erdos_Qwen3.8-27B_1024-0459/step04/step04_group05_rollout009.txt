from scipy.optimize import minimize

_G_DX = 1.0

_G_START = 0.0
_G_BUDGET = 1300.0


def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        # Distribute to elements with most room
        if diff > 0:
            rooms = 1.0 - h
        else:
            rooms = h
        idx = int(np.argmax(rooms))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)


def topk_subgradient(h, n, top_k=5):
    """Compute subgradient from top-k binding shifts."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top-k indices
    top_idx = np.argsort(corr)[-top_k:]
    top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
    
    grad = np.zeros(n)
    for k_idx in top_idx:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            n_pairs = n - offset
            if n_pairs <= 0:
                continue
            grad[:n_pairs] += (1.0 - h[offset:offset + n_pairs])
            grad[offset:] -= h[:n_pairs]
        else:
            abs_off = -offset
            n_pairs = n - abs_off
            if n_pairs <= 0:
                continue
            grad[:n_pairs] -= h[abs_off:abs_off + n_pairs]
            grad[abs_off:] += (1.0 - h[:n_pairs])
    return grad * dx, top_idx, corr


def nesterov_active_opt(h_best, n, budget_fn, top_k=5, max_iters=2000):
    """Nesterov-accelerated active-set gradient descent."""
    h = h_best.copy()
    c5_best = compute_c5(h_best)
    
    # Nesterov momentum
    v = np.zeros(n)
    mu = 0.9  # momentum parameter
    
    # Adaptive step size
    step = 0.1
    no_improve = 0
    
    for it in range(max_iters):
        if remaining_time() < 3.0:
            break
        
        # Nesterov lookahead
        h_look = h - mu * v
        h_look = project_feasible(h_look, n)
        
        # Compute gradient at lookahead
        grad, top_idx, corr = topk_subgradient(h_look, n, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Update momentum
        v_new = mu * v + step * grad
        h_new = h - v_new
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        
        if c5_new < c5_best - 1e-15:
            v = v_new
            h = h_new
            c5_best = c5_new
            no_improve = 0
            # Sometimes increase step
            if it % 100 == 0:
                step = min(step * 1.1, 0.5)
        else:
            no_improve += 1
            step *= 0.9
            if step < 0.001:
                # Try random perturbation
                h_pert = h + np.random.randn(n) * 0.01
                h_pert = project_feasible(h_pert, n)
                c5_pert = compute_c5(h_pert)
                if c5_pert < c5_best:
                    h = h_pert
                    c5_best = c5_pert
                    no_improve = 0
                    step = 0.1
                elif no_improve > 50:
                    break
            v = mu * v  # Decay momentum
    
    return h, c5_best


def slsqp_smooth(h_start, n, beta, budget_fn, maxiter=200):
    """SLSQP with log-sum-exp smoothing for the max."""
    global _G_N, _G_DX, _G_TGT
    
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        exps = np.exp(np.clip(beta * shifted, -500, 500))
        return (m + np.log(np.sum(exps)) / beta) * _G_DX
    
    def con_sum(x):
        return np.sum(x) - _G_TGT
    
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': lambda x: np.ones(n)}]
    
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    
    try:
        res = minimize(obj, h_cur, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-15:
            return h_new, c5_new
    except Exception:
        pass
    return h_cur, c5_cur

        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def rem_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)


def _slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(_G_BETA_CUR * shifted, -500, 500))
    return float((m + np.log(np.sum(exps)) / _G_BETA_CUR) * _G_DX_CUR)


def _slsqp_con_fun(x):
    return float(np.sum(x) - _G_TGT_CUR)


def _slsqp_con_jac(x):
    return np.ones(_G_N_CUR)


def slsqp_step(h, n, beta, maxiter):
    global _G_N_CUR, _G_BETA_CUR, _G_TGT_CUR, _G_DX_CUR
    _G_N_CUR = n
    _G_BETA_CUR = float(beta)
    _G_TGT_CUR = n / 2.0
    _G_DX_CUR = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con_fun, 'jac': _slsqp_con_jac}]
    c5_old = compute_c5(h)
    try:
        res = minimize(_slsqp_obj, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_old - 1e-15:
            return h_new, c5_new
    except Exception:
        pass
    return h, c5_old


def active_subgrad(h, n, top_k):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
    grad = np.zeros(n)
    for k_idx in top_idx:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            np_ = n - offset
            if np_ <= 0:
                continue
            grad[:np_] += (1.0 - h[offset:offset + np_])
            grad[offset:] -= h[:np_]
        else:
            ao = -offset
            np_ = n - ao
            if np_ <= 0:
                continue
            grad[:np_] -= h[ao:ao + np_]
            grad[ao:] += (1.0 - h[:np_])
    return grad * dx


def nesterov_opt(h, n, top_k, max_iters, mu, step_init):
    h = h.copy()
    c5_best = compute_c5(h)
    v = np.zeros(n)
    step = step_init
    no_improve = 0
    patience = 300

    for it in range(max_iters):
        if rem_time() < 3.0:
            break
        h_look = project_feasible(h - mu * v, n)
        grad = active_subgrad(h_look, n, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        v = mu * v + step * grad
        h_new = project_feasible(h - v, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-15:
            c5_best = c5_new
            h = h_new
            no_improve = 0
            if it % 300 == 0 and step < 0.4:
                step = min(step * 1.03, 0.5)
        else:
            no_improve += 1
            step *= 0.95
            if step < 0.003:
                if no_improve > patience:
                    break
                rng_local = np.random.RandomState(it + 12345)
                h_pert = project_feasible(h + rng_local.randn(n) * 0.015, n)
                c5_pert = compute_c5(h_pert)
                if c5_pert < c5_best:
                    h = h_pert
                    c5_best = c5_pert
                    no_improve = 0
                    step = step_init
            v = mu * v
    return h, c5_best


def binary_push(h, n, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for thresh in [0.3, 0.4, 0.5, 0.6, 0.7]:
        h_b = (h > thresh).astype(float)
        h_b = project_feasible(h_b, n)
        c5_b = compute_c5(h_b)
        if c5_b < best_c5:
            best_h = h_b
            best_c5 = c5_b
    for _ in range(100):
        idx = rng.randint(n)
        h_try = best_h.copy()
        h_try[idx] = 1.0 - h_try[idx]
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_h = h_try
            best_c5 = c5_try
    return best_h, best_c5


def pair_transfer(h, n, rng, n_pairs=2000):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_pairs):
        if rem_time() < 2.0:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        delta = 0.05 * rng.choice([-1.0, 1.0])
        h_try = best_h.copy()
        h_try[i] += delta
        h_try[j] -= delta
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_h = h_try
            best_c5 = c5_try
    return best_h, best_c5


def coord_descent(h, n, rng, max_sweeps):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if rem_time() < 2.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if rem_time() < 1.0:
                break
            old_val = best_h[i]
            candidates = [0.0, 1.0]
            if 0.05 < old_val < 0.95:
                candidates += [0.25, 0.75, old_val * 0.5, min(old_val + 0.2, 1.0), max(old_val - 0.2, 0.0)]
            for nv in candidates:
                if abs(nv - old_val) < 1e-7:
                    continue
                best_h[i] = float(np.clip(nv, 0.0, 1.0))
                h_try = project_feasible(best_h, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    any_imp = True
                else:
                    best_h = h_try.copy() if c5_try < best_c5 else best_h
                    best_h[i] = old_val
        if not any_imp and sweep > 1:
            break
    return best_h, best_c5


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        best_h = project_feasible(parent_h, parent_n)
        best_c5 = compute_c5(best_h)
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, best_c5))
    except Exception:
        parent_n = 256
        best_h = project_feasible(make_thue_morse(parent_n), parent_n)
        best_c5 = compute_c5(best_h)
        best_n = parent_n
        print("TM: n={}, C5={:.10f}".format(parent_n, best_c5))

    # Phase 1: SLSQP warm-up
    print("\n--- P1: SLSQP warm (n={}) ---".format(best_n))
    for beta in [30, 100, 300]:
        if rem_time() < 8:
            break
        best_h, best_c5 = slsqp_step(best_h, best_n, beta, maxiter=250)
        print("  b={}: {:.10f}".format(beta, best_c5))

    # Phase 2: Nesterov focused (top_k=1 for precision)
    print("\n--- P2: Nesterov k=1 (n={}) ---".format(best_n))
    best_h, best_c5 = nesterov_opt(best_h, best_n, top_k=1, max_iters=4000, mu=0.8, step_init=0.08)
    print("  C5={:.10f}".format(best_c5))

    # Phase 3: Nesterov broader (top_k=5)
    print("\n--- P3: Nesterov k=5 (n={}) ---".format(best_n))
    best_h, best_c5 = nesterov_opt(best_h, best_n, top_k=5, max_iters=4000, mu=0.85, step_init=0.06)
    print("  C5={:.10f}".format(best_c5))

    # Phase 4: Binary push
    if rem_time() > 10:
        print("\n--- P4: Binary push ---")
        h_b, c5_b = binary_push(best_h, best_n, rng)
        if c5_b < best_c5:
            best_h, best_c5 = h_b, c5_b
        print("  C5={:.10f}".format(best_c5))

    # Phase 5: High-beta SLSQP
    if rem_time() > 15:
        print("\n--- P5: High-beta SLSQP ---")
        for beta in [1000, 5000, 20000]:
            if rem_time() < 8:
                break
            best_h, best_c5 = slsqp_step(best_h, best_n, beta, maxiter=200)
            print("  b={}: {:.10f}".format(beta, best_c5))

    # Phase 6: Nesterov fine (top_k=3, small step)
    if rem_time() > 15:
        print("\n--- P6: Nesterov fine k=3 ---")
        best_h, best_c5 = nesterov_opt(best_h, best_n, top_k=3, max_iters=5000, mu=0.9, step_init=0.03)
        print("  C5={:.10f}".format(best_c5))

    # Phase 7: Pair transfer local search
    if rem_time() > 10:
        print("\n--- P7: Pair transfer ---")
        best_h, best_c5 = pair_transfer(best_h, best_n, rng, n_pairs=3000)
        print("  C5={:.10f}".format(best_c5))

    # Phase 8: Multi-resolution
    print("\n--- P8: Multi-res ---")
    for n_try in [512, 1024]:
        if rem_time() < 35:
            break
        h_up = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        c5_up = compute_c5(h_up)
        h_new, c5_new = slsqp_step(h_up, n_try, 200, maxiter=150)
        h_new, c5_new = nesterov_opt(h_new, n_try, top_k=5, max_iters=3000, mu=0.85, step_init=0.05)
        h_new, c5_new = slsqp_step(h_new, n_try, 2000, maxiter=100)
        if c5_new < best_c5 - 1e-13:
            best_h, best_c5, best_n = h_new, c5_new, n_try
            print("  n={}: {:.10f} (NEW BEST)".format(n_try, best_c5))
        else:
            print("  n={}: {:.10f} (no improve, best={:.10f})".format(n_try, c5_new, best_c5))

    # Phase 9: Final Nesterov + SLSQP polish
    if rem_time() > 15:
        print("\n--- P9: Final polish ---")
        best_h, best_c5 = nesterov_opt(best_h, best_n, top_k=5, max_iters=3000, mu=0.9, step_init=0.02)
        best_h, best_c5 = slsqp_step(best_h, best_n, 5000, maxiter=200)
        print("  C5={:.10f}".format(best_c5))

    # Phase 10: Coordinate descent + pair transfer final
    if rem_time() > 8:
        best_h, best_c5 = coord_descent(best_h, best_n, rng, max_sweeps=5)
        best_h, best_c5 = pair_transfer(best_h, best_n, rng, n_pairs=1000)
        print("\n--- P10: Coord+pair --- C5={:.10f}".format(best_c5))

    # Phase 11: Very last SLSQP
    if rem_time() > 5:
        best_h, best_c5 = slsqp_step(best_h, best_n, 10000, maxiter=150)
        print("\n--- P11: Last SLSQP --- C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_n = len(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
