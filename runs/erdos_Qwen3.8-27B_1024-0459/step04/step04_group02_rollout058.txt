def slsqp_optimize(h, n, beta, maxiter=500):

    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_corr_obj_grad(x, n, beta)[0]
    
    def obj_grad(x):
        return smooth_corr_obj_grad(x, n, beta)[1]
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            if compute_c5(trial) <= compute_c5(h) + 1e-14:
                return trial
    except Exception:
        pass
    return h.copy()

def subgradient_step(h, n, step_size, top_k=10):
    """One subgradient step."""
    corr = compute_corr(h)
    max_corr = np.max(corr)
    active = np.where(corr >= max_corr - 0.2)[0]
    if len(active) > top_k:
        active = np.argsort(corr)[::-1][:top_k]
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]
    grad *= dx
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return h.copy()
    h_new = h - step_size * grad
    return project_feasible(h_new, n)
