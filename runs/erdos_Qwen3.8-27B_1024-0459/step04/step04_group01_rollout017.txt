def round_to_binary(h, n):
    """Round h to binary with exactly n/2 ones using greedy method."""
    h = h.copy()
    target = n // 2
    n_ones = int(np.sum(h > 0.5))
    if n_ones == target:
        return (h > 0.5).astype(float)
    
    if n_ones > target:
        # Need to remove some 1s (pick those closest to 0.5 from below)
        one_fracs = [(h[i], i) for i in range(n) if h[i] > 0.5]
        one_fracs.sort(key=lambda x: x[0])  # sort by value ascending
        result = (h > 0.5).astype(float)
        for val, idx in one_fracs[:n_ones - target]:
            result[idx] = 0.0
        return result
    else:
        # Need to add some 1s (pick those closest to 0.5 from above... i.e., highest 0s)
        zero_fracs = [(h[i], i) for i in range(n) if h[i] <= 0.5]
        zero_fracs.sort(key=lambda x: -x[0])  # sort by value descending
        result = (h > 0.5).astype(float)
        for val, idx in zero_fracs[:target - n_ones]:
            result[idx] = 1.0
        return result

And by change of variable y = x+k: ∫ h(y)(1-h(y-k)) dy, which is the same as the original with shift -k.

So the solution is invariant under h → 1-h (the C₅ value doesn't change). This means I can always assume the top correlations come in ± pairs.

import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 5.0)


def corr_grad_shift(h, n, shift):
    g = np.zeros(n)
    if shift >= 0:
        npairs = n - shift
        if npairs > 0:
            g[:npairs] += (1.0 - h[shift:shift + npairs])
            g[shift:shift + npairs] -= h[:npairs]
    else:
        s = -shift
        npairs = n - s
        if npairs > 0:
            g[s:s + npairs] += (1.0 - h[:npairs])
            g[:npairs] -= h[s:s + npairs]
    return g


def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-14:
            g += w[k] * corr_grad_shift(x, n, k - (n - 1))
    return g * dx_val


def _sum_con(x):
    return np.sum(x) - _TGT


def _sum_con_jac(x):
    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=400):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def round_to_binary(h, n):
    h = np.array(h, dtype=float)
    target = n // 2
    sorted_idx = np.argsort(-h)
    result = np.zeros(n)
    result[sorted_idx[:target]] = 1.0
    return result


def topk_weighted_correlation(corr, top_k=5):
    """Return a scalar that emphasizes the top-K correlations."""
    sorted_c = np.sort(corr)[::-1]
    weights = np.exp(np.arange(top_k - 1, -1, -1) * 0.3)
    weights /= weights.sum()
    return float(np.dot(weights, sorted_c[:top_k]))


def binary_sa_search(h_bin, n, time_limit=60, rng=None, top_k=3):

    if rng is None:

    h = np.array(h_bin, dtype=float).copy()
    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx
    best_h = h.copy()
    start = time.time()
    stall = 0

    T = 5.0
    T_min = 0.02
    cooling = 0.9965
    max_iters = 1200000

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        k_max = int(np.argmax(corr))
        offset = k_max - (n - 1)

        n_try = 25
        found = False
        accepted = False

        for _ in range(n_try):
            roll = rng.rand()

            if roll < 0.50:
                # Single swap biased toward worst shift
                if offset >= 0 and offset > 0:
                    candidates = np.array([i for i in ones if i + offset < n and h[i + offset] == 0])
                    if len(candidates) > 0 and rng.rand() < 0.75:
                        i = candidates[rng.randint(len(candidates))]
                    else:
                        i = ones[rng.randint(len(ones))]
                elif offset < 0:
                    candidates = np.array([i for i in ones if i - offset < n and h[i - offset] == 0])
                    if len(candidates) > 0 and rng.rand() < 0.75:
                        i = candidates[rng.randint(len(candidates))]
                    else:
                        i = ones[rng.randint(len(ones))]
                else:
                    i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]

                h[i] = 0.0
                h[j] = 1.0
                new_corr = compute_corr(h)
                new_max = float(np.max(new_corr))
                delta = new_max - cur_max

                if delta < -1e-12:
                    corr = new_corr
                    cur_max = new_max
                    if cur_max * dx < best_c5:
                        best_c5 = cur_max * dx
                        best_h = h.copy()
                    found = True
                    break
                elif rng.rand() < np.exp(-delta / max(T, 1e-10)):
                    corr = new_corr
                    cur_max = new_max
                    accepted = True
                else:
                    h[i] = 1.0
                    h[j] = 0.0

            elif roll < 0.80:
                # Double swap
                if len(ones) < 2 or len(zeros) < 2:
                    continue
                i1, i2 = rng.choice(ones, 2, replace=False)
                j1, j2 = rng.choice(zeros, 2, replace=False)
                h[i1] = 0.0
                h[i2] = 0.0
                h[j1] = 1.0
                h[j2] = 1.0
                new_corr = compute_corr(h)
                new_max = float(np.max(new_corr))
                delta = new_max - cur_max

                if delta < -1e-12:
                    corr = new_corr
                    cur_max = new_max
                    if cur_max * dx < best_c5:
                        best_c5 = cur_max * dx
                        best_h = h.copy()
                    found = True
                    break
                elif rng.rand() < np.exp(-delta / max(T, 1e-10)):
                    corr = new_corr
                    cur_max = new_max
                    accepted = True
                else:
                    h[i1] = 1.0
                    h[i2] = 1.0
                    h[j1] = 0.0
                    h[j2] = 0.0

            elif roll < 0.92:
                # Triple swap
                if len(ones) < 3 or len(zeros) < 3:
                    continue
                i1, i2, i3 = rng.choice(ones, 3, replace=False)
                j1, j2, j3 = rng.choice(zeros, 3, replace=False)
                h[i1] = 0.0
                h[i2] = 0.0
                h[i3] = 0.0
                h[j1] = 1.0
                h[j2] = 1.0
                h[j3] = 1.0
                new_corr = compute_corr(h)
                new_max = float(np.max(new_corr))
                delta = new_max - cur_max

                if delta < -1e-12:

                    corr = new_corr
                    cur_max = new_max
                    accepted = True
                else:
                    h[i1] = 1.0
                    h[i2] = 1.0
                    h[i3] = 1.0
                    h[j1] = 0.0
                    h[j2] = 0.0
                    h[j3] = 0.0

            else:
                # Block move: shift a run of 1s
                starts = np.where((h == 1) & ((np.arange(n) == 0) | (h[np.clip(np.arange(n) - 1, 0, n-1)] == 0)))[0]
                if len(starts) == 0:

                s = starts[rng.randint(len(starts))]

                h[j] = 1.0
                corr = compute_corr(h)

    return best_h, best_c5


def equalize_topk(h, n, max_iters=200, top_k=4):
    """Equalize top-K correlations via projected subgradient."""
    h = h.copy()
    dx_val = 2.0 / n
    best_c5 = compute_c5(h)
    step = 0.01

    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:top_k]
        c_top = corr[top_idx]
        c_mean = c_top.mean()
        c_spread = c_top.max() - c_top.min()
        if c_spread < 1e-8:
            break

        # Direction: reduce top, increase min
        g = np.zeros(n)
        for ki in top_idx:
            shift = int(ki) - (n - 1)
            gk = corr_grad_shift(h, n, shift)
            weight = -1.0 if ki == top_idx[0] else (1.0 if ki == top_idx[-1] else 0.0)
            g += weight * gk

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g /= gnorm

        improved = False
        for s in [0.005, 0.01, 0.02, 0.04, 0.008, 0.03, 0.06, 0.015]:
            h_trial = project_feasible(h + s * g, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-13:
                h = h_trial
                best_c5 = c_new
                improved = True
                break

        if not improved:
            break
    return h, best_c5


def focused_coord_descent(h, n, max_passes=200, time_limit=60, rng=None):
    """Coordinate descent pairing each point with its antipode, with time-box."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = np.array(h, dtype=float).copy()
    dx = 2.0 / n
    corr = compute_corr(h)
    best_c5 = float(np.max(corr)) * dx
    start = time.time()
    n_corr = 2 * n - 1

    for pass_num in range(max_passes):
        if time.time() - start > time_limit:
            break
        if remaining() < 8:
            break

        if pass_num % 7 == 6:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for idx in range(n):
            i = int(order[idx])
            j = (i + n // 2) % n
            if i == j:
                j = (i + 1) % n
            s = h[i] + h[j]
            lo_t = max(0.0, s - 1.0)
            hi_t = min(1.0, s)
            if lo_t >= hi_t - 1e-14:
                continue

            # Compute slope difference
            sl_i = np.zeros(n_corr)
            sl_j = np.zeros(n_corr)
            for idx_c in range(n_corr):
                d = idx_c - (n - 1)
                if d >= 0:
                    if d <= i < n:
                        sl_i[idx_c] += (1.0 - h[i - d]) if i - d >= 0 else 0
                    if i < n - d:
                        sl_i[idx_c] -= h[i + d]
                    if d <= j < n:
                        sl_j[idx_c] += (1.0 - h[j - d]) if j - d >= 0 else 0
                    if j < n - d:
                        sl_j[idx_c] -= h[j + d]
                else:
                    m = -d
                    if i < n - m:
                        sl_i[idx_c] += (1.0 - h[i + m])
                    if i >= m:
                        sl_i[idx_c] -= h[i - m]
                    if j < n - m:
                        sl_j[idx_c] += (1.0 - h[j + m])
                    if j >= m:
                        sl_j[idx_c] -= h[j - m]

            diff_sl = sl_i - sl_j
            intercepts = corr - h[i] * diff_sl

            # Bisection on subgradient
            if lo_t < hi_t:
                a, b = lo_t, hi_t
                for _ in range(40):
                    mid = (a + b) / 2.0
                    vals = diff_sl * mid + intercepts
                    active = vals >= vals.max() - 1e-13
                    if diff_sl[active].max() < -1e-14:
                        a = mid
                    elif diff_sl[active].min() > 1e-14:
                        b = mid
                    else:
                        break
                t_opt = (a + b) / 2.0
            else:
                t_opt = h[i]

            delta = t_opt - h[i]
            if abs(delta) > 1e-12:
                corr = intercepts + diff_sl * t_opt
                h[i] = t_opt
                h[j] = s - t_opt

        h = project_feasible(h, n)
        corr = compute_corr(h)
        c5_new = float(np.max(corr)) * dx
        if c5_new < best_c5:
            best_c5 = c5_new

    return h, best_c5
