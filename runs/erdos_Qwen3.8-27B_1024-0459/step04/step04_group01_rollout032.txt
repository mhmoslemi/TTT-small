```python
def binary_simulated_annealing(h, n, start_temp, end_temp, n_steps, rng, time_limit):
    h = h.copy()
    corr = compute_corr(h).astype(float)
    cur_c5 = float(np.max(corr)) * (2.0/n)
    best_c5 = cur_c5
    best_h = h.copy()
    ones = set(np.where(h == 1)[0])
    zeros = set(np.where(h == 0)[0])
    start_time = time.time()
    nc = 2*n - 1
    
    # cooling schedule
    alpha = (end_temp / start_temp) ** (1.0 / n_steps)
    
    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        T = start_temp * (alpha ** step)
        
        # Pick a random swap
        i = rng.choice(sorted(ones))
        j = rng.choice(sorted(zeros))
        
        # Compute delta
        delta = pairwise_swap_delta(h, i, j, n)
        new_corr = corr + delta
        
        cur_max = np.max(corr)
        new_max = np.max(new_corr)
        delta_obj = new_max - cur_max
        
        if delta_obj <= 0 or np.log(rng.random()) < delta_obj / T:
            # Accept
            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            ones.discard(i)
            ones.add(j)
            zeros.discard(j)
            zeros.add(i)
            
            new_c5 = new_max * (2.0/n)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    
    return best_h, best_c5
```

```python
def swap_delta(h, i, j, n):
    nc = 2*n - 1
    delta = np.zeros(nc)
    one_minus_h = 1.0 - h
    h_rev = h[n-1::-1]
    
    # h[i] 1->0, first index
    delta[(n-1-i):(2*n-1-i)] -= one_minus_h
    # h[i] 1->0, second index
    delta[i:(i+n)] += h_rev
    # h[i] 0->1... wait no, h[j] 0->1
    # h[j] 0->1, first index
    delta[(n-1-j):(2*n-1-j)] += one_minus_h
    # h[j] 0->1, second index
    delta[j:(j+n)] -= h_rev
    
    # Restore d=0 (index n-1) to 0
    delta[n-1] = 0
    
    return delta
```
