    if target is None:
        target = n / 2.0
    x = np.array(x, dtype=float)
    # Find lambda such that sum(clip(x - lambda, 0, 1)) = target
    # clip(x_i - lambda, 0, 1) = 0 if x_i <= lambda, 1 if x_i >= lambda+1, x_i - lambda otherwise
    lo, hi = x.min() - 1.0, x.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_box_sum(x, n, target=None):
    if target is None:
        target = n / 2.0
    x = np.array(x, dtype=float)
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(x - lam, 0.0, 1.0)
    # Fix tiny numerical issues
    diff = target - result.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - result
            idx = int(np.argmax(room))
            result[idx] = min(1.0, result[idx] + diff)
        else:
            room = result
            idx = int(np.argmax(room))
            result[idx] = max(0.0, result[idx] + diff)
    return result


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)


def con_sum(x):
    return np.sum(x) - _G_TGT


def obj_lse(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _G_BETA
    shifted = np.clip(corr - m, -400.0 / beta, 0.0)
    exps = np.exp(beta * shifted)
    s = np.sum(exps)
    if s < 1e-300:
        return m * dx
    return (m + np.log(s) / beta) * dx


def grad_lse(x, beta):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(corr - m, -400.0 / beta, 0.0)
    exps = np.exp(beta * shifted)
    total = np.sum(exps)
    weights = exps / total
    grad = np.zeros(n)
    nc = len(corr)
    for k_idx in range(nc):
        w = weights[k_idx]
        if w < 1e-14:
            continue
        offset = k_idx - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs <= 0:
                continue
            grad[:npairs] += w * (1.0 - x[offset:offset + npairs])
            grad[offset:] -= w * x[:npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs <= 0:
                continue
            grad[:npairs] -= w * x[aoff:aoff + npairs]
            grad[aoff:] += w * (1.0 - x[:npairs])
    grad *= dx
    return grad


def do_slsqp(h_start, n, budget_fn, beta_list, maxiter_list=None):
    global _G_N, _G_BETA, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    for i, beta in enumerate(beta_list):
        if budget_fn() < 5.0:
            break
        _G_BETA = float(beta)
        if maxiter_list is not None:
            maxiter = maxiter_list[i]
        else:
            maxiter = max(50, int(budget_fn() * 0.3 / max(0.1, 2.0 / n)))
            maxiter = min(maxiter, 500)
        try:
            res = minimize(obj_lse, h_cur, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
            h_new = project_box_sum(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, c5_cur


def do_proj_grad(h_start, n, budget_fn, beta_schedule, max_iters=500):
    global _G_N, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    h = h_start.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    prev_grad = None
    prev_x = None
    for it in range(max_iters):
        if budget_fn() < 3.0:
            break
        frac = it / max(1, max_iters)
        beta = beta_schedule[0] * (beta_schedule[-1] / beta_schedule[0]) ** frac
        fval, grad = obj_lse(h, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        # BB step size
        if prev_grad is not None:
            y = grad - prev_grad
            s = h - prev_x
            sy = np.dot(s, y)
            yy = np.dot(y, y)
            if yy > 1e-16 and sy > 1e-12:
                step = np.dot(s, s) / sy
            else:
                step = 1.0 / gnorm
        else:
            step = 1.0 / gnorm
        step = max(1e-8, min(step, 0.5))
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            h_trial = project_box_sum(h - alpha * step * grad, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-15:
                prev_x = h.copy()
                prev_grad = grad.copy()
                h = h_trial
                c5_best = c5_trial
                h_best = h.copy()
                improved = True
                break
        if not improved:
            # Try smaller noise
            if it % 20 == 19:
                noise = np.random.randn(n) * 0.01
                h_trial = project_box_sum(h + noise, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < c5_best - 1e-14:
                    h = h_trial
                    c5_best = c5_trial
                    h_best = h.copy()
    return h_best, c5_best


def do_active_set_slsqp(h_start, n, budget_fn, k_active=5, n_rounds=10):
    global _G_N, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    h = h_start.copy()
    c5_best = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': lambda x: np.ones(n)}]
    for rnd in range(n_rounds):
        if budget_fn() < 4.0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-k_active:]
        top_idx = sorted(top_idx)

        def obj_active(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return np.max(c[top_idx]) * _G_DX

        def obj_active_sum(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return np.sum(c[top_idx]) * _G_DX / len(top_idx)

        # First minimize the max of active
        try:
            res = minimize(obj_active, h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 100, 'ftol': 1e-16, 'iprint': 0})
            h_new = project_box_sum(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-15:
                c5_best = c5_new
                h = h_new
        except Exception:
            pass

        # Then minimize the mean of active
        try:
            res = minimize(obj_active_sum, h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 100, 'ftol': 1e-16, 'iprint': 0})
            h_new = project_box_sum(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-15:
                c5_best = c5_new
                h = h_new
        except Exception:
            pass
    return h, c5_best


def do_coord_binary(h_start, n, budget_fn, max_sweeps=20):
    h = h_start.copy()
    c5_best = compute_c5(h)
    rng = np.random.RandomState(12345)
    for sweep in range(max_sweeps):
        if budget_fn() < 2.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if budget_fn() < 1.0:
                break
            for new_val in [0.0, 1.0]:
                if abs(new_val - h[i]) < 1e-7:
                    continue
                old = h[i]
                h[i] = new_val
                h = project_box_sum(h, n)
                c5_test = compute_c5(h)
                if c5_test < c5_best - 1e-13:
                    c5_best = c5_test
                    any_imp = True
                else:
                    h = project_box_sum(h, n)
                    h[i] = old
                    h = project_box_sum(h, n)
        if not any_imp and sweep >= 4:
            break
    return h, c5_best


def do_coord_continuous(h_start, n, budget_fn, max_sweeps=10):
    h = h_start.copy()
    c5_best = compute_c5(h)
    rng = np.random.RandomState(999)
    for sweep in range(max_sweeps):
        if budget_fn() < 2.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if budget_fn() < 1.0:
                break
            best_val = h[i]
            best_c5 = c5_best
            for new_val in [0.0, 0.1, 0.25, 0.4, 0.5, 0.6, 0.75, 0.9, 1.0]:
                if abs(new_val - h[i]) < 1e-6:
                    continue
                old = h[i]
                h[i] = new_val
                h = project_box_sum(h, n)
                c5_test = compute_c5(h)
                if c5_test < best_c5 - 1e-13:
                    best_c5 = c5_test
                    best_val = new_val
                h = project_box_sum(h, n)
                h[i] = old
                h = project_box_sum(h, n)
            if best_val != h[i]:
                h[i] = best_val
                h = project_box_sum(h, n)
                c5_best = best_c5
                any_imp = True
        if not any_imp and sweep >= 3:
            break
    return h, c5_best


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h


def make_balanced_block(n):
    # Create a balanced sequence with block structure
    h = np.zeros(n)
    quarter = n // 4
    h[0:quarter] = 1.0
    h[n - quarter:] = 1.0
    # Add some mixing in the middle
    mid_start = quarter
    mid_end = n - quarter
    mid_len = mid_end - mid_start
    if mid_len > 0:
        for i in range(mid_len):
            if (i * 7) % 11 < 5:
                h[mid_start + i] = 1.0
    return project_box_sum(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    def budget_fn():
        return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Initialize from parent or fallback
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        best_h = project_box_sum(make_thue_morse(parent_n), parent_n)
        best_c5 = compute_c5(best_h)
        best_n = parent_n
        print("No parent. TM: n={}, C5={:.10f}".format(parent_n, best_c5))

    # Phase 1: Quick SLSQP at parent resolution
    if budget_fn() > 60:
        print("\n--- Phase 1: SLSQP n={} ---".format(best_n))
        best_h, best_c5 = do_slsqp(best_h, best_n, budget_fn,
                                    [50, 200, 800, 3000], [200, 200, 150, 100])
        print("  After SLSQP: C5={:.10f}".format(best_c5))

    # Phase 2: Projected gradient at parent resolution
    if budget_fn() > 120:
        print("\n--- Phase 2: ProjGrad n={} ---".format(best_n))
        best_h, best_c5 = do_proj_grad(best_h, best_n, budget_fn,
                                        beta_schedule=[100, 50000], max_iters=300)
        print("  After ProjGrad: C5={:.10f}".format(best_c5))

    # Phase 3: Upsample to higher resolution and optimize
    target_n = 512
    if budget_fn() > 200:
        print("\n--- Phase 3: Upsample to n={} ---".format(target_n))
        h_up = project_box_sum(interp_to(best_h, best_n, target_n), target_n)
        c5_up = compute_c5(h_up)
        print("  Interpolated: C5={:.10f}".format(c5_up))
        if c5_up < best_c5 + 0.01:
            # SLSQP at new resolution
            h_up, c5_up = do_slsqp(h_up, target_n, budget_fn,
                                    [100, 500, 2000, 8000], [150, 120, 100, 80])
            print("  After SLSQP n={}: C5={:.10f}".format(target_n, c5_up))
            # Projected gradient
            if budget_fn() > 80:
                h_up, c5_up = do_proj_grad(h_up, target_n, budget_fn,
                                            beta_schedule=[500, 30000], max_iters=250)
                print("  After ProjGrad n={}: C5={:.10f}".format(target_n, c5_up))
            # Active set
            if budget_fn() > 40:
                h_up, c5_up = do_active_set_slsqp(h_up, target_n, budget_fn,
                                                   k_active=5, n_rounds=8)
                print("  After ActiveSet n={}: C5={:.10f}".format(target_n, c5_up))
            if c5_up < best_c5 - 1e-14:
                best_c5 = c5_up
                best_h = h_up.copy()
                best_n = target_n
                print("  n={} adopted: C5={:.10f}".format(target_n, best_c5))

    # Phase 4: Try n=384
    if budget_fn() > 120:
        n_alt = 384
        print("\n--- Phase 4: n={} ---".format(n_alt))
        h_alt = project_box_sum(interp_to(best_h, best_n, n_alt), n_alt)
        c5_alt = compute_c5(h_alt)
        h_alt, c5_alt = do_slsqp(h_alt, n_alt, budget_fn,
                                  [200, 1000, 5000], [150, 120, 100])
        if budget_fn() > 40:
            h_alt, c5_alt = do_proj_grad(h_alt, n_alt, budget_fn,
                                          beta_schedule=[500, 20000], max_iters=200)
        if budget_fn() > 30:
            h_alt, c5_alt = do_active_set_slsqp(h_alt, n_alt, budget_fn,
                                                 k_active=5, n_rounds=6)
        if c5_alt < best_c5 - 1e-14:
            best_c5 = c5_alt
            best_h = h_alt.copy()
            best_n = n_alt
            print("  n={} adopted: C5={:.10f}".format(n_alt, best_c5))
        else:
            print("  n={} no improve: C5={:.10f}".format(n_alt, c5_alt))

    # Phase 5: Coordinate descent (binary)
    if budget_fn() > 60:
        print("\n--- Phase 5: CoordBinary n={} ---".format(best_n))
        best_h, best_c5 = do_coord_binary(best_h, best_n, budget_fn, max_sweeps=15)
        print("  After CoordBinary: C5={:.10f}".format(best_c5))

    # Phase 6: Coordinate descent (continuous)
    if budget_fn() > 40:
        print("\n--- Phase 6: CoordCont n={} ---".format(best_n))
        best_h, best_c5 = do_coord_continuous(best_h, best_n, budget_fn, max_sweeps=8)
        print("  After CoordCont: C5={:.10f}".format(best_c5))

    # Phase 7: High-beta SLSQP polish
    if budget_fn() > 40:
        print("\n--- Phase 7: HighBeta SLSQP ---".format(best_n))
        best_h, best_c5 = do_slsqp(best_h, best_n, budget_fn,
                                    [5000, 20000, 80000], [100, 80, 60])
        print("  After HighBeta: C5={:.10f}".format(best_c5))

    # Phase 8: Perturbed restart
    if budget_fn() > 60:
        print("\n--- Phase 8: Restarts ---".format(best_n))
        for r in range(3):
            if budget_fn() < 20:
                break
            noise_scale = 0.02 + 0.02 * r
            h_pert = project_box_sum(best_h + rng.randn(best_n) * noise_scale, best_n)
            c5_pert = compute_c5(h_pert)
            h_cur = h_pert if c5_pert < best_c5 else best_h.copy()
            c5_cur = min(c5_pert, best_c5)
            h_cur, c5_cur = do_slsqp(h_cur, best_n, budget_fn,
                                      [500, 3000, 10000], [100, 80, 60])
            if budget_fn() > 20:
                h_cur, c5_cur = do_proj_grad(h_cur, best_n, budget_fn,
                                              beta_schedule=[1000, 50000], max_iters=100)
            if c5_cur < best_c5 - 1e-14:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                print("  Restart {}: C5={:.10f}".format(r, best_c5))
            else:
                print("  Restart {}: no improve ( {:.10f})".format(r, c5_cur))

    # Phase 9: Final active set + SLSQP
    if budget_fn() > 30:
        print("\n--- Phase 9: Final ActiveSet ---".format(best_n))
        best_h, best_c5 = do_active_set_slsqp(best_h, best_n, budget_fn,
                                               k_active=7, n_rounds=10)
        print("  After ActiveSet: C5={:.10f}".format(best_c5))

    # Phase 10: Final coord + SLSQP
    if budget_fn() > 20:
        best_h, best_c5 = do_coord_binary(best_h, best_n, budget_fn, max_sweeps=8)
        if budget_fn() > 15:
            best_h, best_c5 = do_slsqp(best_h, best_n, budget_fn,
                                        [30000, 100000], [60, 40])
        if budget_fn() > 10:
            best_h, best_c5 = do_coord_binary(best_h, best_n, budget_fn, max_sweeps=5)

    # Final validation
    best_h = project_box_sum(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
