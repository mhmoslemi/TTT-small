One more consideration: the ternary search for the 1D convex problem. Since f(t) = max_m (a_m * t + b_m) is convex and piecewise linear, the minimum is either at t=0, t=1, or at a breakpoint where two lines intersect. The number of breakpoints is at most O(n) (sorted by slope).

1. Find the active set at t=0 and t=1
2. The subgradient at t is {sum of a_m for m in active(t)}
3. At the minimum, 0 is in the subgradient
4. Binary search on t: if subgradient at t is positive (all active a_m > 0), decrease t; if negative, increase t

But the subgradient is a set (interval), not a single value. For convex piecewise linear functions, the subgradient at t is [min_{m active} a_m, max_{m active} a_m] (the range of slopes of active lines).

If max_{m active at t} a_m < 0: f is decreasing at t, go right.
If min_{m active at t} a_m > 0: f is increasing at t, go left.
Otherwise: t is optimal (0 is in subgradient).

This gives O(n) per evaluation and O(log(1/ε)) iterations. Much faster than ternary search!

def compute_a_vector(h, i, n):
    """Compute a[m] = d(corr[m])/dh[i] for all lags m."""
    a = np.zeros(2 * n - 1)
    # Positive offsets: off = 0..n-1, lag index = n-1+off
    offs = np.arange(0, n)
    mask1 = offs <= i
    if np.any(mask1):
        idx1 = i - offs[mask1]
        a[n - 1 + offs[mask1]] += (1.0 - h[idx1])
    mask2 = (i + offs) <= (n - 1)
    if np.any(mask2):
        idx2 = i + offs[mask2]
        a[n - 1 + offs[mask2]] -= h[idx2]
    # Negative offsets: ao = 1..n-1, lag index = n-1-ao
    aos = np.arange(1, n)
    mask3 = (i + aos) <= (n - 1)
    if np.any(mask3):
        idx3 = i + aos[mask3]
        a[n - 1 - aos[mask3]] += (1.0 - h[idx3])
    mask4 = (i - aos) >= 0
    if np.any(mask4):

        a[n - 1 - aos[mask4]] -= h[idx4]

    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n

def slsqp_refine(h, n, beta=5000, maxiter=300):
    """Quick SLSQP refinement using smooth max."""
    N = n
    TGT = n / 2.0
    BETA = beta

    def _obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(BETA * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / BETA) * (2.0 / N)

    def _sc(x):
        return np.sum(x) - TGT

    h0 = project_feasible(h.copy(), n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc}]
    try:
        res = minimize(_obj, h0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < _obj(h0):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

        c5_slq, _ = compute_c5(h_slq)
        if c5_slq < best_c5:
            best_c5 = c5_slq
            best_h = h_slq.copy()
        # Run coord descent again after SLSQP
        h_cd2, c5_cd2 = coord_descent(best_h, best_n, max_sweeps=1500)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    # Phase 3: Try other resolutions
    n_candidates = sorted(set([
        best_n, best_n - 16, best_n - 8, best_n + 8, best_n + 16,
        int(best_n * 1.1), int(best_n * 1.2), int(best_n * 1.4),
        int(best_n * 1.6),
    ]))
    n_candidates = [nn for nn in n_candidates if 192 <= nn <= 700 and nn % 2 == 0]
    n_candidates = [nn for nn in n_candidates if nn != best_n]
    n_candidates.sort(key=lambda x: abs(x - best_n))

    for n_t in n_candidates[:5]:
        if remaining_time() < 40:
            break
        h_int = interp_to(best_h, best_n, n_t)
        h_int = project_feasible(h_int, n_t)
        c5_int, _ = compute_c5(h_int)

        # Coordinate descent at new resolution
        h_t, c5_t = coord_descent(h_int, n_t, max_sweeps=1500)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t

    # Phase 4: Binary local search on best solution
    if remaining_time() > 20:
        h_bin_init = np.zeros(best_n)
        sort_idx = np.argsort(-best_h)
        h_bin_init[sort_idx[:best_n // 2]] = 1.0
        h_bl, c5_bl = binary_local_search(h_bin_init, best_n, max_passes=30)
        # Convert back to continuous and refine
        h_bl_cont = h_bl.copy()
        h_cd3, c5_cd3 = coord_descent(h_bl_cont, best_n, max_sweeps=800)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3.copy()
        elif c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()

    # Phase 5: Multi-start with different initializations
    starts = []
    # Thue-Morse
    h_tm = np.zeros(best_n)
    for i in range(best_n):
        h_tm[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    n_ones_tm = int(np.sum(h_tm))
    if n_ones_tm != best_n // 2:
        if n_ones_tm > best_n // 2:
            ones_idx = np.where(h_tm == 1)[0]
            rem_idx = ones_idx[rng.choice(len(ones_idx), n_ones_tm - best_n // 2, replace=False)]
            h_tm[rem_idx] = 0.0
        else:
            zeros_idx = np.where(h_tm == 0)[0]
            add_idx = zeros_idx[rng.choice(len(zeros_idx), best_n // 2 - n_ones_tm, replace=False)]
            h_tm[add_idx] = 1.0
    starts.append(h_tm)

    # Random binary
    h_rand = rng.randint(0, 2, best_n).astype(float)
    n_ones_r = int(np.sum(h_rand))
    if n_ones_r != best_n // 2:
        if n_ones_r > best_n // 2:
            ones_idx = np.where(h_rand == 1)[0]
            rem_idx = ones_idx[rng.choice(len(ones_idx), n_ones_r - best_n // 2, replace=False)]
            h_rand[rem_idx] = 0.0
        else:
            zeros_idx = np.where(h_rand == 0)[0]
            add_idx = zeros_idx[rng.choice(len(zeros_idx), best_n // 2 - n_ones_r, replace=False)]
            h_rand[add_idx] = 1.0
    starts.append(h_rand)

    # Quadratic residue style
    n_p = best_n
    h_qr = np.zeros(n_p)
    if n_p > 4:
        # Use a simple modular pattern
        for i in range(n_p):
            q = (i * i) % n_p
            if q < n_p // 2:
                h_qr[i] = 1.0
    n_ones_q = int(np.sum(h_qr))
    if n_ones_q != n_p // 2:
        if n_ones_q > n_p // 2:
            ones_idx = np.where(h_qr == 1)[0]
            rem_idx = ones_idx[rng.choice(len(ones_idx), n_ones_q - n_p // 2, replace=False)]
            h_qr[rem_idx] = 0.0
        else:
            zeros_idx = np.where(h_qr == 0)[0]
            add_idx = zeros_idx[rng.choice(len(zeros_idx), n_p // 2 - n_ones_q, replace=False)]
            h_qr[add_idx] = 1.0
    starts.append(h_qr)

    for s_h in starts:
        if remaining_time() < 30:
            break
        s_h = project_feasible(s_h, best_n)
        s_h, c5_s = coord_descent(s_h, best_n, max_sweeps=800)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = s_h.copy()

    # Phase 6: Final intensive coordinate descent
    if remaining_time() > 15:
        h_final, c5_final = coord_descent(best_h, best_n, max_sweeps=5000)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 7: SLSQP + coord descent polish
    if remaining_time() > 15:
        h_pol = slsqp_refine(best_h, best_n, beta=10000, maxiter=150)
        c5_pol, _ = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
        h_pol2, c5_pol2 = coord_descent(best_h, best_n, max_sweeps=1000)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5, _ = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    # If projection made it worse, try pre-projection
    else:
        # Recompute with exact values
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_feasible(best_h, best_n)
        final_c5, _ = compute_c5(best_h)
        best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
