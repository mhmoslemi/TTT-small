  s_idx = n-1+s  (array index)
  change = 0
  if 0 <= idx-s < n:  # idx appears as first factor h[idx] in the s-shift term at position i=idx-s
    change += delta * (1 - h[idx-s])
  if 0 <= idx+s < n and s != 0:  # idx appears as second factor (1-h[idx]) in the s-shift term
    change -= h[idx+s] * delta
  if s == 0:
    change = new*(1-new) - old*(1-old)  # override
  corr_array[s_idx] += change

def _smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    ncorr = len(corr)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:

                grad[:npairs] -= w * x[offset:offset + npairs]

            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def update_corr_incremental(corr, h, idx, new_val):
    """Update correlation array in-place after changing h[idx] to new_val."""
    n = len(h)
    old_val = h[idx]
    delta = new_val - old_val
    if abs(delta) < 1e-16:
        h[idx] = new_val
        return

                change += delta * (1.0 - h[idx - s])
            if 0 <= idx + s < n:
                change -= h[idx + s] * delta
        corr[k] += change
    h[idx] = new_val

def subgrad_step(h, n, n_active=8):
    """Compute subgradient direction for the max constraint."""
    corr = compute_corr(h)
    max_corr = np.max(corr)
    active = np.where(corr >= max_corr - 0.3)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        active = active[np.argsort(corr[active])[::-1][:n_active]]
    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:

            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return None
    return grad / gnorm
