import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def corr_grad_single(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        if npairs > 0:
            g[off:off + npairs] += (1.0 - x[:npairs])
            g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        if npairs > 0:
            g[:npairs] += (1.0 - x[ao:ao + npairs])
            g[ao:ao + npairs] -= x[:npairs]
    return g


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -80, 80)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def smooth_grad(x, top_k=15):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -80, 80)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    top_idx = np.argsort(-w)[:top_k]
    g = np.zeros(n)
    for k in top_idx:
        if w[k] > 1e-14:
            g += w[k] * corr_grad_single(x, n, int(k))
    return g * dx_val


def sum_con(x):
    return np.sum(x) - _TGT


def sum_con_jac(x):
    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=400, top_k=15):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=lambda x: smooth_grad(x, top_k),
                       method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        if len(idx) >= n_ones - target:
            h_bin[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        if len(idx) >= target - n_ones:
            h_bin[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1
    return h_bin


def sa_binary_ext(h, n, time_limit=60, seed=42, two_swap=False):
    """Extended binary SA with optional 2-swap moves."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]
        if len(idx) >= n_ones - target:
            h[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        if len(idx) >= target - n_ones:
            h[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1

    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()

    T = 12.0
    alpha = 0.9992
    start = time.time()
    stall = 0
    max_stall = 8000

    while time.time() - start < time_limit and T > 0.03:
        if remaining() < 5:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        if two_swap and rng.rand() < 0.3:
            # 2-swap: move two 1s to two 0 positions
            i1, i2 = rng.choice(len(ones), 2, replace=False)
            j1, j2 = rng.choice(len(zeros), 2, replace=False)
            i1, i2 = ones[i1], ones[i2]
            j1, j2 = zeros[j1], zeros[j2]
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0

        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        delta = new_max - cur_max

        if delta < 0 or (T > 0.03 and rng.rand() < np.exp(-delta / T)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            if two_swap and rng.rand() < 0.3:
                h[i1] = 1.0
                h[i2] = 1.0
                h[j1] = 0.0
                h[j2] = 0.0
            else:
                h[i] = 1.0
                h[j] = 0.0
            stall += 1

        T *= alpha
        if stall > max_stall:
            T = max(T, 5.0)
            stall = 0

    return best_h, best_max * dx_val


def local_search_binary(h, n, max_passes=100):
    """Coordinate descent: swap 1 at i with 0 at j to reduce max corr."""
    h = h.copy()
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            idx = np.where(h == 1)[0]
            h[idx[np.arange(n_ones - target)]] = 0
        else:
            idx = np.where(h == 0)[0]
            h[idx[np.arange(target - n_ones)]] = 1

    corr = compute_corr(h)
    cur_max = int(np.max(corr))
    best_max = cur_max
    best_h = h.copy()

    for _pass in range(max_passes):
        if remaining() < 3:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        for i_pos in ones:
            for j_pos in zeros:
                if remaining() < 2:
                    break
                h[i_pos] = 0.0
                h[j_pos] = 1.0
                new_corr = compute_corr(h)
                new_max = int(np.max(new_corr))
                if new_max < best_max:
                    best_max = new_max
                    best_h = h.copy()
                    cur_max = new_max
                    improved = True
                else:
                    h[i_pos] = 1.0
                    h[j_pos] = 0.0
        if not improved:
            break

    return best_h, best_max * (2.0 / n)


def topk_descent(h, n, max_iters=300, top_k=10):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        m = float(np.max(corr))
        if m * (2.0 / n) >= best_c5 + 1e-8:
            break
        top_idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k in top_idx:
            g += corr_grad_single(h, n, int(k))
        g /= top_k
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        improved = False
        for step in [0.08, 0.04, 0.02, 0.06, 0.01, 0.03, 0.005, 0.1]:
            h_trial = project_feasible(h + step * d, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-12:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            break
    return h, best_c5
