    target = n / 2.0

        cnt = int(np.sum(mask))
        if cnt == 0:

        h = np.clip(h, 0.0, 1.0)
    # Final check

        residual = target - np.sum(h)

        rooms = 1.0 - h if residual > 0 else h.copy()

            h[i] += adj * min(abs(residual), abs(rooms[i]))

def compute_corr(h):
    return fftconvolve(h, (1.0 - h)[::-1])


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = int(np.argmax(np.minimum(h, 1.0 - h)))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    h[:] = 1.0 - h  # Thue-Morse complemented
    return h


def lse_obj(x):
    n = _G_N
    dx = _G_DX
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _G_BETA
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    return (m + np.log(np.sum(exps)) / beta) * dx


def lse_con(x):
    return np.sum(x) - _G_TGT


def do_slsqp(h_start, n, budget_fn, beta_list, maxiter_override=None):
    global _G_N, _G_BETA, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    h_cur = project_feasible(h_start, n)
    c5_cur = compute_c5(h_cur)
    for beta in beta_list:
        rem = budget_fn()
        if rem < 8.0:
            break
        _G_BETA = float(beta)
        if maxiter_override is not None:
            maxiter = maxiter_override
        else:
            if beta < 500:
                maxiter = 500
            elif beta < 5000:
                maxiter = 350
            else:
                maxiter = 250
        maxiter = min(maxiter, max(20, int(rem * 0.35 / 0.05)))
        try:
            res = minimize(lse_obj, h_cur, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})

            c5_new = compute_c5(h_new)

                h_cur = h_new

    target = n / 2.0

        if rem < 5.0:

        corr = compute_corr(h)
        m = np.max(corr)

def binary_local_search(h, n, budget_fn, rng, max_sweeps=50):

            h_bin[to_flip] = 0

            idx_zeros = np.where(h_bin == 0)[0]

            h_bin[to_flip] = 1
    c5_best = compute_c5(h_bin)
    no_imp = 0

            if budget_fn() < 2.0:

            if len(ones) == 0 or len(zeros) == 0:

            i = rng.randint(len(ones))

            if budget_fn() < 2.0:
                break

            no_imp += 1
            if no_imp >= 8:
                break
        else:

def coord_descent(h, n, budget_fn, rng, max_sweeps=10):
