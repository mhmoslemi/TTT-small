```python
import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_one_lag(h, m, n):
    """Compute gradient of corr[m] w.r.t. h."""
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        np_ = n - k
        g[k:k+np_] += (1.0 - h[:np_])
        g[:np_] -= h[k:k+np_]
    else:
        ao = -k
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao+np_])
        g[ao:ao+np_] -= h[:np_]
    return g

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            step = diff / cnt
            h[mask] += step
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)

def chase_max(h, n, time_limit=60.0, eps=0.05):
    """Subgradient method chasing the maximum correlation."""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.005
    start = time.time()
    
    while time.time() - start < time_limit and remaining() > 5:
        corr = compute_corr(h)
        max_val = np.max(corr)
        c5 = max_val * 2.0 / n
        
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        
        # Find near-max lags
        near = np.where(corr >= max_val - eps)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in near:
            grad += grad_one_lag(h, k, n)
        grad /= len(near)
        
        gn = np.linalg.norm(grad)
        if gn < 1e-14:

        grad /= gn

        h_new = project_feasible(h - alpha * grad, n)
        corr_new = compute_corr(h_new)
        
        if np.max(corr_new) < max_val - 1e-12:

            alpha = min(alpha * 1.02, 0.05)
        else:
            alpha *= 0.8
            if alpha < 0.0001:
                break
    
    return best_h, best_c5
```

```python
def binary_swap_fast(h, n, time_limit=120):
    h = (h > 0.5).astype(float)
    n_ones = int(np.sum(h))
    if n_ones != n//2:
        h = np.zeros(n)
        h[:n//2] = 1.0  # crude fix
    
    dx = 2.0/n
    best_c5 = compute_c5(h)
    start = time.time()
    
    improved = True
    while improved and time.time() - start < time_limit and remaining() > 5:
        improved = False
        corr = compute_corr(h)
        cur_max = np.max(corr)
        
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        
        # Precompute gradient matrix G[m, i] = d corr[m] / d h[i]
        G = np.zeros((2*n-1, n))
        for m in range(2*n-1):
            G[m, :] = grad_one_lag(h, m, n)
        
        # For swap (i, j): Δcorr[m] = G[m, j] - G[m, i]
        # new_corr[m] = corr[m] + G[m, j] - G[m, i]
        # Want to minimize max(new_corr)
        
        best_val = cur_max
        best_swap = None
        
        # Process in chunks to limit memory
        for i_idx, i in enumerate(ones):
            # For this i, compute Δcorr for all j in zeros
            # Δcorr[:, j] = corr + G[:, j] - G[:, i]
            # Shape: (2n-1, len(zeros))
            dc = G[:, zeros] - G[:, i:i+1]  # (2n-1, len(zeros))
            new_max = np.max(corr[:, None] + dc, axis=0)  # (len(zeros),)
            
            local_best = np.argmin(new_max)
            if new_max[local_best] < best_val - 1e-12:
                best_val = new_max[local_best]
                best_swap = (i, zeros[local_best])
        
        if best_swap is not None:
            h[best_swap[0]] = 0
            h[best_swap[1]] = 1
            new_c5 = compute_c5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
            improved = True
    
    return h, best_c5
```

After finding a good binary structure, we want to slightly perturb some entries away from 0/1 to precisely equalize the top correlations. The optimal perturbation makes the top L correlations equal (by KKT conditions of the max problem).

For this, I'll use a targeted SLSQP with very high beta, or a custom Newton-like method.

Actually, let me think about this differently. If we have a binary solution with correlations corr[0], ..., corr[2n-2], and the maximum is at a few lags, we want to find small perturbations δ[h[i]] such that the affected correlations decrease by equal amounts.

```python
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA_G * sh, -50.0, 50.0)
```

```python
def slsqp_smooth(h, n, beta, maxiter=300):

    _TGT_G = n / 2.0

def chase_max(h, n, time_limit=60.0, eps=0.05):
    """Subgradient method targeting the max correlation."""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.003
    start = time.time()
    
    while time.time() - start < time_limit and remaining() > 3:
        corr = compute_corr(h)
        max_val = np.max(corr)
        c5 = max_val * 2.0 / n
        
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        
        near = np.where(corr >= max_val - eps)[0]
        
        grad = np.zeros(n)
        for k in near:
            grad += grad_one_lag(h, int(k), n)
        grad /= len(near)
        
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        
        h_new = project_fast(h - alpha * grad, n)
        corr_new = compute_corr(h_new)
        
        if np.max(corr_new) < max_val - 1e-12:
            h = h_new
            alpha = min(alpha * 1.02, 0.03)
        else:
            alpha *= 0.85
            if alpha < 5e-5:
                break
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_N_G = 0
_BETA_G = 1.0
_TGT_G = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def grad_one_lag(h, m, n):
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        np_ = n - k
        g[k:k + np_] += (1.0 - h[:np_])
        g[:np_] -= h[k:k + np_]
    else:
        ao = -k
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def project_fast(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    mask = (h > 1e-10) & (h < 1.0 - 1e-10)
    cnt = np.count_nonzero(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    else:
        h[:] += diff / n
    return np.clip(h, 0.0, 1.0)


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)


def _sm_obj(x):
    n = _N_G
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA_G * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / _BETA_G) * dx_v


def _sm_grad(x):
    n = _N_G
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA_G * sh, -50.0, 50.0)
    ex = np.exp(cl)
    ws = np.sum(ex)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = ex[k] / ws * dx_v
        if wk < 1e-15:
            continue
        g += wk * grad_one_lag(x, k, n)
    return g


def _sm_con_fun(x):
    return np.sum(x) - _TGT_G


def _sm_con_jac(x):
    return np.ones(_N_G)


def slsqp_smooth(h, n, beta, maxiter=300):
    global _N_G, _BETA_G, _TGT_G
    _N_G = n
    _TGT_G = n / 2.0
    _BETA_G = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sm_con_fun, 'jac': _sm_con_jac}]
    try:
        obj0 = _sm_obj(h)
        res = minimize(_sm_obj, h, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < obj0:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_N_G = 0
_BETA_G = 1.0
_TGT_G = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def grad_one_lag(h, m, n):
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        np_ = n - k
        g[k:k + np_] += (1.0 - h[:np_])
        g[:np_] -= h[k:k + np_]
    else:
        ao = -k
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def project_fast(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    mask = (h > 1e-10) & (h < 1.0 - 1e-10)
    cnt = np.count_nonzero(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    else:
        h[:] += diff / n
    return np.clip(h, 0.0, 1.0)


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)


def _sm_obj(x):
    n = _N_G
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA_G * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / _BETA_G) * dx_v


def _sm_grad(x):
    n = _N_G
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA_G * sh, -50.0, 50.0)
    ex = np.exp(cl)
    ws = np.sum(ex)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = ex[k] / ws * dx_v
        if wk < 1e-15:
            continue
        g += wk * grad_one_lag(x, k, n)
    return g


def _sm_con_fun(x):
    return np.sum(x) - _TGT_G


def _sm_con_jac(x):
    return np.ones(_N_G)


def slsqp_smooth(h, n, beta, maxiter=300):
    global _N_G, _BETA_G, _TGT_G
    _N_G = n
    _TGT_G = n / 2.0
    _BETA_G = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sm_con_fun, 'jac': _sm_con_jac}]
    try:
        obj0 = _sm_obj(h)
        res = minimize(_sm_obj, h, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < obj0:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.005
    start = time.time()
    nott = 0

        corr = compute_corr(h)

            best_h = h.copy()
            nott = 0
        else:
            nott += 1

        if nott > 8000:
            break

        near = np.where(corr >= max_val - eps)[0]

            grad += grad_one_lag(h, int(k), n)
        grad /= float(len(near))

        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break

def binary_swap_opt(h, n, time_limit=120.0):
    h = (h > 0.5).astype(float)
    n_ones = int(np.sum(h))
    if n_ones != n // 2:
        if n_ones > n // 2:
            idx = np.where(h == 1)[0]
            h[idx[:n_ones - n // 2]] = 0
        else:
            idx = np.where(h == 0)[0]
            h[idx[:n // 2 - n_ones]] = 1

    best_c5 = compute_c5(h)
    start = time.time()

    improved = True
    while improved and time.time() - start < time_limit and remaining() > 3:
        improved = False
        corr = compute_corr(h)
        cur_max = float(np.max(corr))

        zeros = np.where(h == 0)[0]

        if len(ones) == 0 or len(zeros) == 0:
            break

        G = np.zeros((2 * n - 1, n))
        for m in range(2 * n - 1):
            G[m, :] = grad_one_lag(h, m, n)

        best_val = cur_max
        best_i = -1
        best_j = -1

        for i in range(len(ones)):
            dc = G[:, zeros] - G[:, ones[i]:ones[i] + 1]

def coord_polish(h, n, passes=10):
    h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    trial_vals = [0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0]

    for p in range(passes):
        if remaining() < 3:
            break
        any_improved = False

            if remaining() < 3:

            orig_vec = h.copy()
            orig = h[i]

            best_tv = orig
            for tv in trial_vals:

                h[:] = orig_vec
                h[i] = tv

                best_c5 = best_local
                any_improved = True
        if not any_improved:

def interp_to(src_h, src_n, dst_n):

        if c5_opt < best_c5:

            best_h = h_opt.copy()
            best_n = n_t

    # Phase 6: Second binary swap round at best resolution
    n = best_n
    h = best_h.copy()
    if remaining() > 50:
        cl = min(60.0, max(10.0, remaining() - 35.0))
        h_bin2, _ = binary_swap_opt(h, n, time_limit=cl)
        h_ref2 = slsqp_smooth(h_bin2, n, 20000, maxiter=300)
        c5_ref2 = compute_c5(h_ref2)
        if remaining() > 20:
            c = min(30.0, max(5.0, remaining() - 8.0))
            h_ref2, c5_ref2 = chase_max(h_ref2, n, time_limit=c, eps=0.005)
        if remaining() > 8:
            h_ref2, c5_ref2 = coord_polish(h_ref2, n, passes=6)
        if c5_ref2 < best_c5:
            best_c5 = c5_ref2
            best_h = h_ref2.copy()

    # Phase 7: Very fine chase
    if remaining() > 20:
        cl = min(50.0, max(8.0, remaining() - 5.0))
        h_ch, c5_ch = chase_max(best_h, best_n, time_limit=cl, eps=0.003)
        if c5_ch < best_c5:
            best_c5 = c5_ch
            best_h = h_ch.copy()

    # Phase 8: Final SLSQP + top-shifts + coordinate polish
    if remaining() > 15:
        h_new = slsqp_smooth(best_h, best_n, 50000, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if remaining() > 12:
        c = min(25.0, max(5.0, remaining() - 8.0))
        h_ts, c5_ts = top_shifts_focus(best_h, best_n, top_k=5, max_iters=200, time_limit=c)
        if c5_ts < best_c5:
            best_c5 = c5_ts
            best_h = h_ts.copy()

    if remaining() > 8:
        h_cp, c5_cp = coord_polish(best_h, best_n, passes=12)
        if c5_cp < best_c5:
            best_c5 = c5_cp
            best_h = h_cp.copy()

    # Final chase with tiny eps
    if remaining() > 8:
        h_ch, c5_ch = chase_max(best_h, best_n, time_limit=min(15, remaining() - 3), eps=0.001)
        if c5_ch < best_c5:
            best_c5 = c5_ch
            best_h = h_ch.copy()

    # Ensure final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        ev = evaluate_erdos_solution(best_h)
        if ev < best_c5:
            best_c5 = ev
    except Exception:
        pass

    return (best_h, best_c5, best_n)
```
