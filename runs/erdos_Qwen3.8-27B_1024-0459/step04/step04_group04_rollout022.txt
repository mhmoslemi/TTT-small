Given current binary h and top-K shifts:

    - h[i] appears as h[p=i]: contributes h[i]*(1-h[i-shift]) = 1*(1-h[i-shift]) = 1-h[i-shift]
    - After flip: 0*(1-h[i-shift]) = 0. Change: -(1-h[i-shift]) = h[i-shift] - 1
    - h[i] appears as (1-h[p-shift])=1-h[i]: when p-shift=i, p=i+shift. Contributes h[i+shift]*(1-h[i]) = h[i+shift]*0 = 0. After flip: h[i+shift]*1 = h[i+shift]. Change: +h[i+shift]

  New corr[m] = corr[m] + delta[m]
  New max = max over m of new corr[m]

  We want the (i,j) pair that minimizes the new max.

  For efficiency, we only need to track the top-K correlations. If a shift is not in the top-K, its value only matters if it becomes the new max (which is unlikely for small deltas).

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)

    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def informed_binary_sa(h, n, time_limit=60, seed=42, K=10):
    """Binary SA with informed moves using precomputed deltas."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    n_ones = int(np.sum(h))
    target = n // 2
    # Ensure exactly target ones
    if n_ones > target:
        idx = np.where(h == 1)[0]
        to_flip = idx[rng.choice(len(idx), n_ones - target, replace=False)]
        h[to_flip] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        to_flip = idx[rng.choice(len(idx), target - n_ones, replace=False)]
        h[to_flip] = 1
    
    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()
    
    T = 5.0
    alpha = 0.9995
    start = time.time()
    stall = 0
    max_stall = 3000
    top_K = K
    
    while time.time() - start < time_limit:
        if remaining() < 5:
            break
        if T < 0.05:
            break
            
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Update top-K periodically
        if stall % 100 == 0 or stall == 0:
            top_idx = np.argsort(-corr)[:top_K]
            top_corr = corr[top_idx]
        
        # Compute informed move
        if len(ones) <= 200 and len(zeros) <= 200:
            # Compute delta for each one-out and zero-in
            # delta_out[i, k] = effect of flipping h[i] from 1 to 0 on corr[k]
            # delta_in[j, k] = effect of flipping h[j] from 0 to 1 on corr[k]
            
            # For shift s (where k in corr index = n-1+s, s in [-(n-1), n-1]):
            # corr[n-1+s] = sum_p h[p] * (1-h[p-s])
            # h[i] as h[p]: p=i, term = h[i]*(1-h[i-s]) = 1*(1-h[i-s]) = 1-h[i-s]
            #   After flip: 0. Delta = -(1-h[i-s]) = h[i-s] - 1
            # h[i] as (1-h[p-s]): p-s=i, p=i+s, term = h[i+s]*(1-h[i]) = h[i+s]*0 = 0
            #   After flip: h[i+s]*1 = h[i+s]. Delta = +h[i+s]
            
            # So delta_out for position i at shift s: (h[i-s] - 1) + h[i+s] (with bounds)
            
            offsets = top_idx - (n - 1)  # actual shifts for top-K
            # For each one-position i and each shift s:
            # delta_out[i, t] where t indexes the shift
            # Need h[i - s] and h[i + s] with bounds checking
            
            one_idx = ones  # shape (n_ones,)
            zero_idx = zeros  # shape (n_zeros,)
            
            # delta_out: (n_ones, K)
            delta_out = np.zeros((len(one_idx), top_K))
            for t, s in enumerate(offsets):
                s_int = int(s)
                # h[i - s_int]: valid when 0 <= i - s_int < n
                idx1 = one_idx - s_int
                valid1 = (idx1 >= 0) & (idx1 < n)
                val1 = np.zeros(len(one_idx))
                val1[valid1] = h[idx1[valid1]]
                # h[i + s_int]: valid when 0 <= i + s_int < n
                idx2 = one_idx + s_int
                valid2 = (idx2 >= 0) & (idx2 < n)
                val2 = np.zeros(len(one_idx))
                val2[valid2] = h[idx2[valid2]]
                delta_out[:, t] = val1 - 1.0 + val2
            
            delta_in = np.zeros((len(zero_idx), top_K))
            for t, s in enumerate(offsets):
                s_int = int(s)
                idx1 = zero_idx - s_int
                valid1 = (idx1 >= 0) & (idx1 < n)
                val1 = np.zeros(len(zero_idx))
                val1[valid1] = h[idx1[valid1]]
                idx2 = zero_idx + s_int
                valid2 = (idx2 >= 0) & (idx2 < n)
                val2 = np.zeros(len(zero_idx))
                val2[valid2] = h[idx2[valid2]]
                delta_in[:, t] = (1.0 - val1) - val2
            
            # Combined: for move (i→0, j→1), new top vals = top_corr + delta_out[i,:] + delta_in[j,:]
            # delta_out shape: (n_ones, K), delta_in shape: (n_zeros, K)
            # combined: (n_ones, n_zeros, K) - might be too large!
            
            if len(one_idx) * len(zero_idx) * top_K < 5000000:
                combined = delta_out[:, None, :] + delta_in[None, :, :]
                new_vals = top_corr[None, None, :] + combined
                new_max = new_vals.max(axis=2)  # (n_ones, n_zeros)
                flat_idx = new_max.argmin()
                bi = flat_idx // len(zero_idx)
                bj = flat_idx % len(zero_idx)
            else:
                # Sample moves
                bi = rng.randint(len(one_idx))
                bj = rng.randint(len(zero_idx))
            
            i = one_idx[bi]
            j = zero_idx[bj]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        
        # Apply move
        h[i], h[j] = 0.0, 1.0
        # Update corr incrementally (only for top shifts)
        # Actually, let's just recompute for the accepted move
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        delta = new_max - cur_max
        
        if delta < -1e-15 or (T > 0.05 and rng.rand() < np.exp(-delta / T)):
            corr = new_corr
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
        
        T *= alpha
        if stall > max_stall:
            T = max(T, 3.0)
            stall = 0
    
    return best_h, best_max * dx_val

The subgradient of max_k corr[k] with respect to h is:
- If unique max at k*: subgrad = d(corr[k*])/dh
- If multiple max at {k1, k2, ...}: subgrad = average of d(corr[kt])/dh

We want to minimize max_k corr[k] subject to 0≤h≤1, sum(h)=n/2.

The projected subgradient step:

Where project clips to [0,1] and adjusts sum to n/2.

Actually, for the "exact" subgradient method, we need the subgradient of the max, which is the average of the gradients of the active (maximal) functions. If only one function is maximized, it's just that function's gradient.

The gradient of corr[k] with respect to h[i]:
corr[n-1+k] = Σ_p h[p] * (1-h[p-k])

∂/∂h[i]: 
- If i is in range of p: h[i] appears as h[p=i], contributing (1-h[i-k])
- If i-k is in range (i.e., i is in range of p-k): h[i] appears in (1-h[p-k]) when p-k=i, p=i+k, contributing -h[i+k]

So ∂corr[n-1+k]/∂h[i] = (1-h[i-k]) - h[i+k] (with boundary conditions)

def subgradient_descent(h, n, time_limit=60, K=5):
    h = h.copy()
    target = n / 2.0
    corr = compute_corr(h)
    cur_max = np.max(corr)
    best_max = cur_max
    best_h = h.copy()
    
    lr0 = 1.0
    lr_decay = 0.9999
    start = time.time()
    
    t = 0
    while time.time() - start < time_limit:
        if remaining() < 3:
            break
        t += 1
        lr = lr0 / (1 + 0.1 * t)  # decreasing step size
        
        # Find top-K shifts
        top_idx = np.argsort(-corr)[:K]
        
        # Compute subgradient (average gradient of top-K)
        g = np.zeros(n)
        for k in top_idx:
            k_int = int(k)
            s = k_int - (n - 1)
            # grad of corr at this shift
            # ∂/∂h[i] = (1-h[i-s]) - h[i+s] (with bounds)
            idx1 = np.arange(n) - s
            valid1 = (idx1 >= 0) & (idx1 < n)
            term1 = np.zeros(n)
            term1[valid1] = 1.0 - h[idx1[valid1]]
            idx2 = np.arange(n) + s
            valid2 = (idx2 >= 0) & (idx2 < n)
            term2 = np.zeros(n)
            term2[valid2] = h[idx2[valid2]]
            g += term1 - term2
        g /= K
        
        # Step
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        
        # Evaluate
        new_corr = compute_corr(h_new)
        new_max = np.max(new_corr)
        
        if new_max < cur_max:
            h = h_new
            corr = new_corr
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            # Don't update h, but decay learning rate
            pass
        
        lr0 *= lr_decay
    
    return best_h, best_max * (2.0/n)

            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1
    return h_bin


def make_binary_round(h, n, rng):
    """Round to binary with exact constraint, using fractional parts as tiebreaker."""
    target = n // 2
    sorted_frac = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_frac[:target]] = 1.0
    return h_bin


def informed_binary_sa(h, n, time_limit=60, seed=42, K=12, T0=4.0, alpha_cool=0.9997):
    """Fast binary SA with vectorized informed moves."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones != target:
        if n_ones > target:
            idx = np.where(h == 1)[0]
            h[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
        else:
            idx = np.where(h == 0)[0]
            h[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1

    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()

    T = T0
    alpha = alpha_cool
    start = time.time()
    stall = 0
    max_stall = 4000
    top_idx = np.argsort(-corr)[:K]
    top_corr = corr[top_idx].copy()
    offsets = (top_idx - (n - 1)).astype(int)
    step_count = 0

    while time.time() - start < time_limit and T > 0.03:
        if remaining() < 5:
            break

        step_count += 1

        # Update top-K every 200 steps
        if step_count % 200 == 0:
            top_idx = np.argsort(-corr)[:K]
            top_corr = corr[top_idx].copy()
            offsets = (top_idx - (n - 1)).astype(int)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Compute deltas for all one-out and zero-in positions at top-K shifts
        delta_out = np.zeros((len(ones), K))
        delta_in = np.zeros((len(zeros), K))

        for t in range(K):
            s = offsets[t]
            # delta_out[i, t] = h[i-s] - 1 + h[i+s] (with bounds)
            idx1 = ones - s
            valid1 = (idx1 >= 0) & (idx1 < n)
            v1 = np.zeros(len(ones))
            if np.any(valid1):
                v1[valid1] = h[idx1[valid1]]
            idx2 = ones + s
            valid2 = (idx2 >= 0) & (idx2 < n)
            v2 = np.zeros(len(ones))
            if np.any(valid2):
                v2[valid2] = h[idx2[valid2]]
            delta_out[:, t] = v1 - 1.0 + v2

            # delta_in[j, t] = (1 - h[j-s]) - h[j+s] (with bounds)
            idx1 = zeros - s
            valid1 = (idx1 >= 0) & (idx1 < n)
            v1 = np.zeros(len(zeros))
            if np.any(valid1):
                v1[valid1] = h[idx1[valid1]]
            idx2 = zeros + s
            valid2 = (idx2 >= 0) & (idx2 < n)
            v2 = np.zeros(len(zeros))
            if np.any(valid2):
                v2[valid2] = h[idx2[valid2]]
            delta_in[:, t] = (1.0 - v1) - v2

        # Find best move: minimize max of (top_corr + delta_out[i] + delta_in[j])
        nO = len(ones)
        nZ = len(zeros)
        if nO * nZ * K < 8000000:
            combined = delta_out[:, None, :] + delta_in[None, :, :]
            new_vals = top_corr[None, None, :] + combined
            new_max_arr = new_vals.max(axis=2)
            flat_best = new_max_arr.argmin()
            bi = int(flat_best // nZ)
            bj = int(flat_best % nZ)
        else:
            # Sample-based for very large n
            bi = rng.randint(nO)
            bj = rng.randint(nZ)

        i_pos = ones[bi]
        j_pos = zeros[bj]

        # Apply move
        h[i_pos] = 0.0
        h[j_pos] = 1.0

        # Recompute correlation (needed for accurate evaluation)
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        delta = new_max - cur_max

        if delta < -1e-15 or (T > 0.03 and rng.rand() < np.exp(max(-50, -delta / T))):
            corr = new_corr
            cur_max = new_max
            if new_max < best_max - 1e-15:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i_pos] = 1.0
            h[j_pos] = 0.0
            stall += 1

        T *= alpha
        if stall > max_stall:
            # Restart temperature
            T = max(T, 2.5)
            stall = 0
            # Maybe perturb
            if T > 1.0 and rng.rand() < 0.3:
                p1 = ones[rng.randint(len(ones))]
                p2 = zeros[rng.randint(len(zeros))]
                h[p1], h[p2] = 1.0 - h[p1], 0.0 + (1.0 - h[p2])
                h = (h > 0.5).astype(float)
                corr = compute_corr(h)
                cur_max = float(np.max(corr))

    # Project to exact binary
    target = n // 2
    cur_ones = int(np.sum(best_h > 0.5))
    if cur_ones != target:
        final = make_binary(best_h, n, rng)
        best_max = float(np.max(compute_corr(final)))
        best_h = final

    return best_h, best_max * dx_val


def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = float(np.max(corr))
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -70, 70)
    ex = np.exp(cl)
    s = np.sum(ex)
    if s < 1e-300:
        return m * dx_val
    return (m + np.log(s) / beta) * dx_val


def corr_grad_single(x, n, k_idx):
    s = k_idx - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        npairs = n - s
        g[s:s + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[s:s + npairs]
    else:
        ao = -s
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = float(np.max(corr))
    shifted = corr - m
    cl = np.clip(beta * shifted, -70, 70)
    ex = np.exp(cl)
    s = float(np.sum(ex))
    if s < 1e-300:
        top_k = int(np.argmax(corr))
        return corr_grad_single(x, n, top_k) * dx_val
    w = ex / s
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-10:
            g += w[k] * corr_grad_single(x, n, k)
    return g * dx_val


def sum_con(x):
    return np.sum(x) - _TGT


def sum_con_jac(x):
    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=300):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def coordinate_descent(h, n, max_sweeps=50, top_K=8):
    """Coordinate descent: for each i, optimize h[i] given others fixed."""
    h = h.copy()
    target = n / 2.0
    best_c5 = compute_c5(h)
    corr = compute_corr(h)

    for sweep in range(max_sweeps):
        if remaining() < 3:
            break
        improved = False
        # Focus on top-K shifts
        top_idx = np.argsort(-corr)[:top_K]
        offsets = top_idx - (n - 1)

        for i in range(n):
            if remaining() < 2:
                break
            # Compute effect of h[i] on top-K correlations
            # corr[n-1+s] involves h[i]:
            # As h[p=i]: term = h[i]*(1-h[i-s])
            # As (1-h[p-s]) with p-s=i: p=i+s, term = h[i+s]*(1-h[i])
            # So corr[k] = base_k + h[i]*(1-h[i-s]) + h[i+s]*(1-h[i])
            #            = base_k + h[i] - h[i]*h[i-s] + h[i+s] - h[i+s]*h[i]
            # For fixed other values, as function of t=h[i]:
            # corr[k] = base_k + t*(1-val_s) + val_s_plus*(1-t)
            #         = base_k + val_s_plus + t*(1 - val_s - val_s_plus)
            # where val_s = h[i-s] (if valid) and val_s_plus = h[i+s] (if valid)

            a_vals = np.zeros(top_K)  # intercept part
            b_vals = np.zeros(top_K)  # coefficient of t

            # Current corr at top shifts (before changing h[i])
            cur_top_corr = corr[top_idx].copy()

            # Remove h[i]'s contribution from each top shift
            for t in range(top_K):
                s = int(offsets[t])
                # Contribution of h[i] to corr at this shift:
                val_s = h[i - s] if (0 <= i - s < n) else 0.0
                val_sp = h[i + s] if (0 <= i + s < n) else 0.0
                # Current contribution: h[i]*(1-val_s) + val_sp*(1-h[i])
                contrib = h[i] * (1.0 - val_s) + val_sp * (1.0 - h[i])
                a_vals[t] = cur_top_corr[t] - contrib
                b_vals[t] = (1.0 - val_s) - val_sp

            # Objective: max_t (a_vals[t] + b_vals[t] * t) for t in [0,1]
            # This is a piecewise-linear convex function.
            # Find optimal t: check 0, 1, and crossings
            candidates = [0.0, 1.0]
            for t1 in range(top_K):
                for t2 in range(t1 + 1, top_K):
                    db = b_vals[t1] - b_vals[t2]
                    if abs(db) > 1e-15:
                        tc = (a_vals[t2] - a_vals[t1]) / db
                        if 0.0 <= tc <= 1.0:
                            candidates.append(tc)

            best_t = h[i]
            best_val = float('inf')
            for tc in candidates:
                v = a_vals + b_vals * tc
                mv = float(np.max(v))
                # Also need to check sum constraint feasibility
                if mv < best_val - 1e-15:
                    best_val = mv
                    best_t = tc

            if abs(best_t - h[i]) > 1e-12:
                h[i] = best_t
                improved = True
                # Re-project sum
                h = project_feasible(h, n)
                if remaining() < 2:
                    break

        if not improved:
            break
        # Update correlation periodically
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * (2.0 / n)
        if c5_val < best_c5:
            best_c5 = c5_val

    return h, best_c5


def subgradient_descent(h, n, time_limit=30, K=5):
    """Subgradient descent on the max correlation."""
    h = h.copy()
    target = n / 2.0
    corr = compute_corr(h)
    best_max = float(np.max(corr))
    best_h = h.copy()
    cur_max = best_max

    lr = 5.0
    lr_decay = 0.99995
    start = time.time()
    t = 0
    arange_n = np.arange(n)

    while time.time() - start < time_limit:
        if remaining() < 3:
            break
        t += 1

        top_idx = np.argsort(-corr)[:K]
        offsets = top_idx - (n - 1)

        g = np.zeros(n)
        for s in offsets:
            s = int(s)
            idx1 = arange_n - s
            valid1 = (idx1 >= 0) & (idx1 < n)
            term1 = np.zeros(n)
            term1[valid1] = 1.0 - h[idx1[valid1]]
            idx2 = arange_n + s
            valid2 = (idx2 >= 0) & (idx2 < n)
            term2 = np.zeros(n)
            term2[valid2] = h[idx2[valid2]]
            g += term1 - term2
        g /= K

        h_new = h - lr * g
        h_new = project_feasible(h_new, n)

        new_corr = compute_corr(h_new)
        new_max = float(np.max(new_corr))

        if new_max < cur_max - 1e-15:
            h = h_new
            corr = new_corr
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()

        lr *= lr_decay
        if lr < 0.01:
            break

    return best_h, best_max * (2.0 / n)


def two_phase_refine(h, n, time_limit=100):
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()

    # Phase A: SLSQP with increasing beta
    for beta in [200, 800, 3000]:
        if time.time() - start > time_limit * 0.4 or remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=250)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase B: Coordinate descent
    if time.time() - start < time_limit * 0.65 and remaining() > 8:
        h_cd, c5_cd = coordinate_descent(h, n, max_sweeps=40, top_K=8)
        if c5_cd < best_c5 - 1e-12:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Phase C: More SLSQP
    if time.time() - start < time_limit * 0.85 and remaining() > 6:
        for beta in [10000, 50000]:
            if remaining() < 5:
                break
            h_new = slsqp_smooth(h, n, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase D: Subgradient polish
    if remaining() > 5:
        h_sg, c5_sg = subgradient_descent(h, n, time_limit=min(20, remaining() * 0.3), K=5)
        if c5_sg < best_c5 - 1e-12:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase E: Final coordinate descent
    if remaining() > 3:
        h_cd, c5_cd = coordinate_descent(best_h, n, max_sweeps=30, top_K=6)
        if c5_cd < best_c5 - 1e-12:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 360
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Informed binary SA at multiple n values
    # Focus on n values where binary solution can beat 0.38085
    # n=360: need max_corr <= 68 for C5 = 68*2/360 = 0.37778
    # n=362: need max_corr <= 68 for C5 = 68*2/362 = 0.37569
    # n=364: need max_corr <= 69 for C5 = 69*2/364 = 0.37912
    # n=366: need max_corr <= 69 for C5 = 69*2/366 = 0.37705
    # n=370: need max_corr <= 70 for C5 = 70*2/370 = 0.37838
    # n=372: need max_corr <= 70 for C5 = 70*2/372 = 0.37634
    # n=720: need max_corr <= 137 for C5 = 137*2/720 = 0.38056
    # n=722: need max_corr <= 137 for C5 = 137*2/722 = 0.37950

    n_list = [360, 362, 364, 366, 370, 372, 374, 376, 378, 380]
    n_list = [n for n in n_list if n % 2 == 0]

    t_start = time.time()
    sa_time_budget = budget_s * 0.55  # 55% for binary SA

    for n_t in n_list:
        if remaining() < 20 or time.time() - t_start > sa_time_budget:
            break

        h_interp = interp_to(best_h, best_n, n_t)
        h_bin = make_binary_round(h_interp, n_t, rng)

        t_per_n = min(50, max(10, (sa_time_budget - (time.time() - t_start)) / max(1, len(n_list) - n_list.index(n_t))))

        # Multiple SA restarts with different strategies
        for sa_seed in [seed + n_t, seed + 10000 + n_t * 3]:
            if remaining() < 8 or time.time() - t_start > sa_time_budget:
                break
            t_sa = min(30, t_per_n / 2)
            h_sa, c5_sa = informed_binary_sa(
                h_bin, n_t, time_limit=t_sa, seed=sa_seed,
                K=12, T0=4.0, alpha_cool=0.9996
            )
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_t

        # Also try with different initial binary (random perturbation)
        if remaining() > 15 and time.time() - t_start < sa_time_budget * 0.7:
            h_pert = make_binary_round(
                h_interp + rng.randn(n_t) * 0.2, n_t, rng
            )
            t_sa = min(25, t_per_n / 2)
            h_sa, c5_sa = informed_binary_sa(
                h_pert, n_t, time_limit=t_sa, seed=seed + 20000 + n_t * 7,
                K=12, T0=5.0, alpha_cool=0.9995
            )
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_t

    # Phase 2: Continuous refinement at best n and neighbors
    print(f"After binary SA: C5={best_c5:.6f} at n={best_n}", flush=True)

    n_candidates = sorted(set([
        best_n - 2, best_n, best_n + 2, best_n + 4,
        360, 362, 364, 366, 370
    ]))
    n_candidates = [n for n in n_candidates if 100 <= n <= 500 and n % 2 == 0]
    # Remove duplicates, keep best_n first
    if best_n in n_candidates:
        n_candidates.remove(best_n)
    n_candidates = [best_n] + n_candidates

    t_cont_budget = budget_s * 0.30
    t_cont_start = time.time()

    for n_t in n_candidates[:5]:  # Limit to 5 n values
        if time.time() - t_cont_start > t_cont_budget or remaining() < 15:
            break

        if n_t == best_n:
            h_init = best_h.copy()
        else:
            h_init = interp_to(best_h, best_n, n_t)
            h_init = project_feasible(h_init, n_t)

        t_limit = min(60, max(15, (t_cont_budget - (time.time() - t_cont_start)) / max(1, 5 - n_candidates.index(n_t))))
        h_ref, c5_ref = two_phase_refine(h_init, n_t, time_limit=t_limit)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            best_n = n_t

    print(f"After continuous refine: C5={best_c5:.6f} at n={best_n}", flush=True)

    # Phase 3: Final intensive refinement on best solution
    n = best_n
    h = best_h.copy()

    # More SLSQP with high beta
    for beta in [20000, 80000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Coordinate descent
    if remaining() > 8:
        h_cd, c5_cd = coordinate_descent(h, n, max_sweeps=60, top_K=10)
        if c5_cd < best_c5 - 1e-12:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Subgradient descent
    if remaining() > 5:
        h_sg, c5_sg = subgradient_descent(h, n, time_limit=min(30, remaining() * 0.3), K=6)
        if c5_sg < best_c5 - 1e-12:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 4: One more round of binary SA from current best
    if remaining() > 20:
        h_bin = make_binary_round(best_h, n, rng)
        t_sa = min(40, remaining() * 0.3)
        h_sa, c5_sa = informed_binary_sa(
            h_bin, n, time_limit=t_sa, seed=seed + 50000,
            K=14, T0=3.0, alpha_cool=0.9995
        )
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 5: Continuous refinement of the binary result if it improved
    if remaining() > 10 and best_h is not None:
        # Check if best is binary (all values 0 or 1)
        if np.allclose(best_h, (best_h > 0.5).astype(float)):
            h_cont = project_feasible(best_h + 0.01 * rng.randn(best_n), best_n)
            h_ref, c5_ref = two_phase_refine(h_cont, best_n, time_limit=min(30, remaining() * 0.5))
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()

    # Final polish: coordinate descent
    if remaining() > 5:
        h_cd, c5_cd = coordinate_descent(best_h, best_n, max_sweeps=40, top_K=8)
        if c5_cd < best_c5 - 1e-12:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Ensure exact constraint
    s = np.sum(best_h)
    target = best_n / 2.0
    if abs(s - target) > 1e-10:
        best_h = project_feasible(best_h, best_n)
        final_c5 = compute_c5(best_h)
        best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    print(f"Final: C5={best_c5:.6f} at n={best_n}", flush=True)
    return (best_h, best_c5, best_n)
