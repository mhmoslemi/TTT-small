def g_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def g_fft_corr(h):
    n = len(h)
    fl = 2 * n - 1
    N = 1
    while N < fl:
        N <<= 1
    b = 1.0 - h
    H = np.fft.fft(h, N)
    B = np.fft.fft(b, N)
    C = np.fft.ifft(H * np.conj(B)).real
    return np.concatenate([C[N - (n - 1):N], C[0:n]])


def g_corr(h):
    n = len(h)
    if n <= 512:
        return np.correlate(h, 1.0 - h, mode='full')
    return g_fft_corr(h)


def g_proj(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    for _ in range(200):
        d = tgt - h.sum()

            break
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        c = int(m.sum())
        if c > 0:
            h[m] += d / c
        else:
            h += d / n
        h = np.clip(h, 0.0, 1.0)
    return h

        hb, c5b = g_bin_search(hb, bn, rng, 3000)
        print("  Binary C5={:.10f}".format(c5b))
        if c5b < bc5:
            bc5, bh, bn = c5b, hb, bn
        # Now optimize from the binary solution in continuous space
        if g_tl() > 10 and c5b < bc5 + 0.01:
            bh, bc5 = g_slsqp(hb, bn, [1000, 5000, 20000], 200)
            print("  Post-binary SLSQP: C5={:.10f}".format(bc5))

    # Phase 11: Perturbed restarts
    if g_tl() > 25:
        print("\nP11: Restarts n={}".format(bn))
        bh, bc5 = g_perturb_slsqp(bh, bc5, bn, rng, 5)
        print("  C5={:.10f}".format(bc5))

    # Phase 12: Try Thue-Morse and golden ratio at best n
    if g_tl() > 25:
        print("\nP12: Alternative inits n={}".format(bn))
        for name, init_fn in [("TM", g_tm), ("Golden", g_golden)]:
            if g_tl() < 10:
                break
            hi = g_proj(init_fn(bn), bn)
            c5i = g_c5(hi)
            hi, c5i = g_slsqp(hi, bn, [500, 3000, 10000, 30000], 200)
            if c5i < bc5:
                bc5, bh, bn = c5i, hi, bn
                print("  {} improved: C5={:.10f}".format(name, bc5))
            else:
                print("  {}: C5={:.10f} (no imp)".format(name, c5i))

    # Phase 13: Final polish
    if g_tl() > 8:
        bh, bc5 = g_subgrad(bh, bn, rng, 100, 3)
    if g_tl() > 8:
        bh, bc5 = g_slsqp(bh, bn, [50000, 150000, 300000, 500000], 100)
    if g_tl() > 5:
        bh, bc5 = g_lbfgsb(bh, bn, [100000, 300000], 2000)
    if g_tl() > 5:
        bh, bc5 = g_subgrad(bh, bn, rng, 50, 3)
    if g_tl() > 5:
        bh, bc5 = g_slsqp(bh, bn, [200000, 500000, 1000000], 80)

    bh = g_proj(bh, bn)
    fc5 = g_c5(bh)
    try:
        ec = evaluate_erdos_solution(bh)
        if float(ec) < fc5:
            fc5 = float(ec)
    except Exception:
        pass
    print("\nFINAL: n={}, C5={:.10f}".format(bn, fc5))
    return (bh, fc5, bn)
