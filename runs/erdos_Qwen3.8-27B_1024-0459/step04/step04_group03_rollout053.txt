def compute_gradient(h, n, w, dx):
    # w has length 2n-1, w[k] = softmax weight for corr[k]
    # grad[m] = dx * sum_k w[k] * d(corr[k])/dh[m]
    # d(corr[k])/dh[m] = (1-h[m+k-(n-1)]) * 1_{valid} - h[m+k-(n-1)-(n-1)] ... 
    
    # Let me redefine: corr[k] = sum_i h[i] * (1-h[i+k-(n-1)]) for valid i
    # where i+k-(n-1) must be in [0,n-1]
    
    # d(corr[k])/dh[m]:
    # = sum_i [δ_{i,m} * (1-h[i+k-(n-1)]) + h[i] * (-δ_{i+k-(n-1),m})]
    # = (1-h[m+k-(n-1)]) * 1_{0≤m+k-(n-1)<n} - h[m-k+(n-1)] * 1_{0≤m-k+(n-1)<n}
    
    # Let lag = k-(n-1). Then:
    # d(corr[n-1+lag])/dh[m] = (1-h[m+lag]) * 1_{0≤m+lag<n} - h[m-lag] * 1_{0≤m-lag<n}
    
    # Vectorized computation:
    m_idx = np.arange(n)  # (n,)
    lag_idx = np.arange(-(n-1), n)  # (2n-1,)
    w_arr = w  # (2n-1,)
    
    # First term: sum_lag w[lag] * (1-h[m+lag]) * 1_{0≤m+lag<n}
    # Create matrix: M1[m, l] = w[l] * (1-h[m+lag[l]]) * 1_{0≤m+lag[l]<n}
    
    # m+lag: shape (n, 2n-1)
    ml = m_idx[:, None] + lag_idx[None, :]  # (n, 2n-1)
    valid1 = (ml >= 0) & (ml < n)
    h_ml = np.zeros((n, 2n-1))
    h_ml[valid1] = h[ml[valid1]]
    term1 = w_arr[None, :] * (1 - h_ml) * valid1
    A = term1.sum(axis=1)  # (n,)
    
    # Second term: sum_lag w[lag] * h[m-lag] * 1_{0≤m-lag<n}
    mml = m_idx[:, None] - lag_idx[None, :]  # (n, 2n-1)
    valid2 = (mml >= 0) & (mml < n)
    h_mml = np.zeros((n, 2n-1))
    h_mml[valid2] = h[mml[valid2]]
    term2 = w_arr[None, :] * h_mml * valid2
    B = term2.sum(axis=1)  # (n,)
    
    grad = dx * (A - B)
    return grad

    n = _N_OPT
    beta = _BETA_OPT
    pen = _PEN_OPT
    h = x
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = corr * dx
    s_max = np.max(s)
    exp_s = np.exp(s - s_max)
    sum_exp = np.sum(exp_s)
    w_full = exp_s / sum_exp
    g_smooth = _compute_grad(h, n, w_full, dx)
    g_pen = 2.0 * pen * (np.sum(h) - target) + np.zeros(n)
    return g_smooth + g_pen

def lbfgs_optimize(h0, n, beta, penalty, maxiter=500, gtol=1e-12):
    global _N_OPT, _BETA_OPT, _PEN_OPT
    _N_OPT = n
    _BETA_OPT = beta
    _PEN_OPT = penalty
    h_init = np.clip(h0, 0.0, 1.0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_opt_func, h_init, jac=_opt_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': gtol,
                                               'ftol': 1e-18, 'iprint': 0})
        h_new = project_feasible(res.x, n)
    except Exception:
        h_new = project_feasible(h0, n)
    return h_new

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def binarize(h, n, thresh=0.15):
    h_b = h.copy()
    h_b[h_b < thresh] = 0.0
    h_b[h_b > 1.0 - thresh] = 1.0
    return project_feasible(h_b, n)

def pair_search(h_init, n, max_sweeps=10, window=40, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        if remaining_time() < 2:
            break
        improved = False
        order = rng.permutation(n)
        for ii in range(len(order)):
            if remaining_time() < 1:
                break
            i = int(order[ii])
            hi = best_h[i]
            if hi < 0.05 or hi > 0.95:
                continue
            j_max = min(i + window, n)
            for j in range(i + 1, j_max):
                hj = best_h[j]
                if abs(hi - hj) < 0.03:
                    continue
                for frac in [0.3, 0.5, 0.7]:
                    amt = (hi - hj) * frac
                    if abs(amt) < 0.005:
                        continue
                    h_trial = best_h.copy()
                    new_i = hi - amt
                    new_j = hj + amt
                    if new_i < 0 or new_i > 1 or new_j < 0 or new_j > 1:
                        continue
                    h_trial[i] = new_i
                    h_trial[j] = new_j
                    h_trial = project_feasible(h_trial, n)
                    c5_t = compute_c5(h_trial)
                    if c5_t < best_c5 - 1e-14:
                        best_c5 = c5_t
                        best_h = h_trial
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def coordinate_descent(h_init, n, max_rounds=20, step_sizes=None):
    if step_sizes is None:
        step_sizes = [0.02, 0.05, 0.1, 0.2, 0.5]
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for rnd in range(max_rounds):
        if remaining_time() < 2:
            break
        improved = False
        for i in range(n):
            if remaining_time() < 1:
                break
            for delta in step_sizes:
                for sign in [1.0, -1.0]:
                    h_trial = best_h.copy()
                    h_trial[i] = np.clip(h_trial[i] + sign * delta, 0.0, 1.0)
                    h_trial = project_feasible(h_trial, n)
                    if np.array_equal(h_trial, best_h):
                        continue
                    c5_t = compute_c5(h_trial)
                    if c5_t < best_c5 - 1e-14:
                        best_c5 = c5_t
                        best_h = h_trial
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S, _N_OPT, _BETA_OPT, _PEN_OPT
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 256
        best_h = np.full(best_n, 0.5)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at parent resolution with escalating beta
    n = best_n
    h = best_h.copy()
    print(f"Phase 1: n={n}, initial C5={best_c5:.6f}", flush=True)
    for beta in [50, 150, 400, 1000, 3000, 8000, 20000]:
        if remaining_time() < 15:
            break
        maxit = 600 if beta < 1000 else 400
        h_new = lbfgs_optimize(h, n, beta, penalty=50.0, maxiter=maxit)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            print(f"  beta={beta}: C5={best_c5:.6f}", flush=True)

    # Phase 2: Low-res warm start for structural discovery
    if remaining_time() > 200:
        for low_n in [32, 64, 128]:
            if remaining_time() < 30:
                break
            h_low = interp_to(best_h, best_n, low_n)
            h_low = project_feasible(h_low, low_n)
            for beta in [100, 500, 2000]:
                if remaining_time() < 10:
                    break
                h_low = lbfgs_optimize(h_low, low_n, beta, penalty=100.0, maxiter=400)
                c5_low_test = compute_c5(h_low)
            h_up = interp_to(h_low, low_n, best_n)
            h_up = project_feasible(h_up, best_n)
            h_up = lbfgs_optimize(h_up, best_n, 1000, penalty=50.0, maxiter=300)
            c5_up = compute_c5(h_up)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = h_up.copy()
                h = h_up.copy()
                print(f"  low_n={low_n}: C5={best_c5:.6f}", flush=True)

    # Phase 3: Multi-resolution high-res optimization
    trial_ns = [192, 224, 256, 288, 320, 384, 448, 512, 640, 768, 1024]
    for trial_n in trial_ns:
        if remaining_time() < 25:
            break
        if trial_n == best_n:
            continue
        h_t = interp_to(best_h, best_n, trial_n)
        h_t = project_feasible(h_t, trial_n)
        c5_t = compute_c5(h_t)
        if c5_t < best_c5 + 0.01:
            maxit = 500 if trial_n <= 256 else 300
            for beta in [200, 1000, 5000]:
                if remaining_time() < 10:
                    break
                h_t = lbfgs_optimize(h_t, trial_n, beta, penalty=50.0, maxiter=maxit)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = trial_n
                h = h_t.copy()
                print(f"  n={trial_n}: C5={best_c5:.6f}", flush=True)

    # Phase 4: Binarization + re-optimization
    if remaining_time() > 60:
        for thresh in [0.1, 0.15, 0.2]:
            if remaining_time() < 15:
                break
            h_b = binarize(best_h, best_n, thresh)
            c5_b = compute_c5(h_b)
            if c5_b < best_c5 + 0.005:
                for beta in [500, 2000, 10000]:
                    if remaining_time() < 8:
                        break
                    h_b = lbfgs_optimize(h_b, best_n, beta, penalty=50.0, maxiter=400)
                c5_b = compute_c5(h_b)
                if c5_b < best_c5:
                    best_c5 = c5_b
                    best_h = h_b.copy()
                    h = h_b.copy()
                    print(f"  binarize t={thresh}: C5={best_c5:.6f}", flush=True)

    # Phase 5: Local combinatorial search (pair transfers)
    if remaining_time() > 30:
        h_comb, c5_comb = pair_search(best_h, best_n, max_sweeps=15, window=50, rng=rng)
        if c5_comb < best_c5:
            best_c5 = c5_comb
            best_h = h_comb.copy()
            h = h_comb.copy()
            print(f"  pair_search: C5={best_c5:.6f}", flush=True)

    # Phase 6: Coordinate descent polish
    if remaining_time() > 30:
        h_cd, c5_cd = coordinate_descent(best_h, best_n, max_rounds=15,
                                         step_sizes=[0.01, 0.03, 0.08, 0.2])
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()
            print(f"  coord_descent: C5={best_c5:.6f}", flush=True)
        if remaining_time() > 15:
            h_pol = lbfgs_optimize(h, best_n, 20000, penalty=100.0, maxiter=300)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                h = h_pol.copy()

    # Phase 7: Perturbation restarts from best
    for restart in range(5):
        if remaining_time() < 30:
            break
        h_p = best_h.copy()
        mask = (h_p > 0.05) & (h_p < 0.95)
        n_mask = np.sum(mask)
        if n_mask > 0:
            h_p[mask] += rng.normal(0, 0.06, n_mask)
        flip_idx = rng.choice(best_n, size=min(max(3, best_n // 50), best_n), replace=False)
        for i in flip_idx:
            h_p[i] = 1.0 - h_p[i]
        h_p = project_feasible(h_p, best_n)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5 + 0.02:
            for beta in [500, 5000]:
                if remaining_time() < 10:
                    break
                h_p = lbfgs_optimize(h_p, best_n, beta, penalty=50.0, maxiter=400)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()
                h = h_p.copy()
                print(f"  restart {restart}: C5={best_c5:.6f}", flush=True)

    # Phase 8: Different n exploration from current best
    if remaining_time() > 100:
        for trial_n in [200, 240, 264, 300, 360, 480, 600, 800, 1280, 1536]:
            if remaining_time() < 20:
                break
            h_t = interp_to(best_h, best_n, trial_n)
            h_t = project_feasible(h_t, trial_n)
            maxit = 400 if trial_n <= 300 else 250
            for beta in [300, 2000, 8000]:
                if remaining_time() < 8:
                    break
                h_t = lbfgs_optimize(h_t, trial_n, beta, penalty=50.0, maxiter=maxit)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = trial_n
                print(f"  phase8 n={trial_n}: C5={best_c5:.6f}", flush=True)
            if remaining_time() > 10:
                h_t2 = binarize(h_t, trial_n, 0.12)
                h_t2 = lbfgs_optimize(h_t2, trial_n, 5000, penalty=50.0, maxiter=300)
                c5_t2 = compute_c5(h_t2)
                if c5_t2 < best_c5:
                    best_c5 = c5_t2
                    best_h = h_t2.copy()
                    best_n = trial_n
                    print(f"  phase8 bin n={trial_n}: C5={best_c5:.6f}", flush=True)

    # Phase 9: Additional pair search + L-BFGS polish
    if remaining_time() > 20:
        h_comb, c5_comb = pair_search(best_h, best_n, max_sweeps=10, window=60, rng=rng)
        if c5_comb < best_c5:
            best_c5 = c5_comb
            best_h = h_comb.copy()
            print(f"  pair2: C5={best_c5:.6f}", flush=True)
        if remaining_time() > 10:
            h_pol = lbfgs_optimize(best_h, best_n, 30000, penalty=200.0, maxiter=300)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 10: Final ultra-precise optimization
    if remaining_time() > 10:
        for beta in [50000, 100000]:
            if remaining_time() < 5:
                break
            h_new = lbfgs_optimize(best_h, best_n, beta, penalty=500.0, maxiter=200, gtol=1e-14)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 11: Tiny perturbations to escape flat regions
    if remaining_time() > 15:
        for _ in range(20):
            if remaining_time() < 3:
                break
            h_tiny = best_h.copy()
            j = int(rng.randint(best_n))
            h_tiny[j] = np.clip(h_tiny[j] + rng.uniform(-0.01, 0.01), 0.0, 1.0)
            h_tiny = project_feasible(h_tiny, best_n)
            h_tiny = lbfgs_optimize(h_tiny, best_n, 20000, penalty=100.0, maxiter=150)
            c5_tiny = compute_c5(h_tiny)
            if c5_tiny < best_c5:
                best_c5 = c5_tiny
                best_h = h_tiny.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print(f"FINAL: n={best_n}, C5={final_c5:.8f}", flush=True)
    return (best_h, final_c5, best_n)
