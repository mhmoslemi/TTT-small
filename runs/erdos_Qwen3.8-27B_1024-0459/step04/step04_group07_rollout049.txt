def ilc_binary(h0, n, duration_s, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = float(np.max(get_corr(h))) * (2.0 / n)
    t_start = time.time()
    t_end = t_start + duration_s
    iter_count = 0
    
    while time.time() < t_end - 10:
        round_time = min(15.0, (t_end - time.time()) * 0.5)
        
        # Greedy descent
        h_desc, c5_desc = greedy_descent_binary(h, n, round_time, rng, max_sweeps=200)
        
        # Perturb
        n_pert = rng.randint(3, 10)
        h_pert = h_desc.copy()
        one_pos = list(np.where(h_pert == 1.0)[0])
        zero_pos = list(np.where(h_pert == 0.0)[0])
        for _ in range(n_pert):
            i = one_pos[rng.randint(len(one_pos))]
            j = zero_pos[rng.randint(len(zero_pos))]
            h_pert[i] = 0.0
            h_pert[j] = 1.0
            one_pos[one_pos.index(i)] = j
            zero_pos[zero_pos.index(j)] = i
        
        # Short SA from perturbed
        sa_time = min(10.0, (t_end - time.time()) * 0.3)
        h_sa, c5_sa = binary_sa(h_pert, n, sa_time, rng, temp0=2.0, temp_min=0.5)
        
        # Greedy from SA result
        h_final, c5_final = greedy_descent_binary(h_sa, n, min(8.0, (t_end - time.time()) * 0.3), rng, max_sweeps=100)
        
        if c5_final < best_c5 - 1e-12:
            best_c5 = c5_final
            best_h = h_final.copy()
            h = h_final.copy()
        
        iter_count += 1
        if time.time() > t_end - 20:
            break
    
    return best_h, best_c5

def continuous_refine(h_bin, n, duration_s, rng):
    h = project_feasible(h_bin, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    t_end = t_start + duration_s
    
    beta_schedule = [30, 80, 200, 500, 1200, 3000]
    for bv in beta_schedule:
        if time.time() > t_end - 5:
            break
        h_new = slsqp_polish(best_h, n, bv, maxiter=200)
        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-12:
            best_c5 = c_new
            best_h = h_new
    
    return best_h, best_c5

def multi_start_binary(n, duration_s, rng, parent_bin=None):
    inits = []
    
    if parent_bin is not None:
        inits.append(('parent', parent_bin.copy()))
    
    tm = make_tm(n)
    inits.append(('tm', tm))
    
    golden = make_golden(n)
    inits.append(('golden', golden))
    
    for s in range(4):
        hr = np.zeros(n)
        idx = rng.permutation(n)
        hr[idx[:n // 2]] = 1.0
        inits.append(('rand{}'.format(s), hr))
    
    # Evaluate and sort
    scored = []
    for name, h in inits:
        e = float(np.max(get_corr(h)))
        scored.append((e, name, h))
    scored.sort(key=lambda x: x[0])
    
    best_h = scored[0][2].copy()
    best_c5 = scored[0][0] * (2.0 / n)
    print("  Best init: {} with C5={:.8f}".format(scored[0][1], best_c5))
    
    time_per_init = max(5.0, duration_s / len(scored))
    for e, name, h in scored:
        if time.time() > time.time() - duration_s + 10:
            break
        remaining = duration_s - (time.time() - (time.time() - duration_s))
        
        # SA
        hh, c5_sa = binary_sa(h, n, time_per_init * 0.6, rng, temp0=3.0, temp_min=0.5)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = hh.copy()
        
        # Greedy
        hh, c5_g = greedy_descent_binary(hh, n, time_per_init * 0.3, rng, max_sweeps=200)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = hh.copy()
        
        # SA from best
        hh, c5_sa2 = binary_sa(best_h, n, time_per_init * 0.1, rng, temp0=2.0, temp_min=0.5)
        hh, c5_g2 = greedy_descent_binary(hh, n, time_per_init * 0.1, rng, max_sweeps=100)
        if c5_g2 < best_c5:
            best_c5 = c5_g2
            best_h = hh.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    _G_PEN = 0.0
    rng = np.random.RandomState(seed)
    
    def tl():
        return max(0.0, _G_BUDGET - (time.time() - _G_START) - 8.0)
    
    # Get parent
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        parent_bin = to_binary(parent_h, parent_n)
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = make_tm(parent_n)
        parent_bin = parent_h.copy()
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("No parent, using TM: n={}, C5={:.8f}".format(parent_n, best_c5))
    
    # Phase 1: Binary optimization at n=256
    if tl() > 300:
        print("\n=== Phase 1: Binary SA/ILS at n=256 ({}s) ===".format(min(280, tl())))
        n1 = 256
        dur1 = min(250.0, tl() - 50)
        h_bin1, c5_bin1 = multi_start_binary(n1, dur1 * 0.6, rng, parent_bin if best_n == n1 else None)
        if c5_bin1 < best_c5:
            best_c5 = c5_bin1
            best_h = h_bin1.copy()
            best_n = n1
        print("  Binary n=256: C5={:.8f}".format(c5_bin1))
        
        # ILS from best binary
        if tl() > 100:
            h_ilc, c5_ilc = ilc_binary(h_bin1, n1, min(100.0, tl() - 30), rng)
            if c5_ilc < best_c5:
                best_c5 = c5_ilc
                best_h = h_ilc.copy()
                best_n = n1
            print("  After ILS n=256: C5={:.8f}".format(c5_ilc))
    
    # Phase 2: Binary optimization at n=512
    if tl() > 250:
        print("\n=== Phase 2: Binary optimization at n=512 ({}s) ===".format(min(220, tl())))
        n2 = 512
        dur2 = min(200.0, tl() - 50)
        
        # Upsample best binary to 512
        if best_n == 256:
            h_up = interp_h(best_h if best_c5 <= 0.5 else best_h, best_n, n2)
            h_up_bin = to_binary(h_up, n2)
        else:
            h_up_bin = make_tm(n2)
        
        h_bin2, c5_bin2 = multi_start_binary(n2, dur2 * 0.5, rng, h_up_bin)
        
        # ILS at 512
        if tl() > 60:
            h_ilc2, c5_ilc2 = ilc_binary(h_bin2, n2, min(80.0, tl() - 20), rng)
            if c5_ilc2 < c5_bin2:
                c5_bin2 = c5_ilc2
                h_bin2 = h_ilc2
        
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()
            best_n = n2
        print("  Binary n=512: C5={:.8f}".format(c5_bin2))
    
    # Phase 3: Binary optimization at n=768 (even, divisible by 2)
    if tl() > 150:
        print("\n=== Phase 3: Binary at n=768 ({}s) ===".format(min(130, tl())))
        n3 = 768
        h_up3 = interp_h(best_h, best_n, n3)
        h_up3_bin = to_binary(h_up3, n3)
        tm3 = make_tm(n3)
        
        # Quick multi-start
        inits3 = [(float(np.max(get_corr(h_up3_bin))), h_up3_bin),
                  (float(np.max(get_corr(tm3))), tm3)]
        inits3.sort(key=lambda x: x[0])
        h3 = inits3[0][1].copy()
        
        dur3 = min(100.0, tl() - 30)
        h_bin3, c5_bin3 = binary_sa(h3, n3, dur3 * 0.7, rng, temp0=3.0, temp_min=0.5)
        h_bin3, c5_bin3b = greedy_descent_binary(h_bin3, n3, dur3 * 0.3, rng, max_sweeps=200)
        c5_bin3 = min(c5_bin3, c5_bin3b)
        
        if c5_bin3 < best_c5:
            best_c5 = c5_bin3
            best_h = h_bin3.copy()
            best_n = n3
        print("  Binary n=768: C5={:.8f}".format(c5_bin3))
    
    # Phase 4: Continuous refinement at best resolution
    if tl() > 100:
        print("\n=== Phase 4: Continuous refinement n={} ({}s) ===".format(best_n, min(90, tl())))
        dur4 = min(80.0, tl() - 15)
        h_cont, c5_cont = continuous_refine(best_h, best_n, dur4, rng)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont
            best_n = len(h_cont)
        print("  Continuous n={}: C5={:.8f}".format(best_n, c5_cont))
    
    # Phase 5: Cross-resolution exploration
    if tl() > 80:
        print("\n=== Phase 5: Cross-resolution ({}s) ===".format(min(70, tl())))
        # Try: optimize at 256, upsample to 512, optimize, downsample to 256
        n_a = 256
        n_b = 512
        
        # Start from best at 256
        h_a = to_binary(best_h, n_a) if best_n != n_a else to_binary(best_h, best_n)
        if len(h_a) != n_a:
            h_a = to_binary(interp_h(best_h, best_n, n_a), n_a)
        
        c5_a = float(np.max(get_corr(h_a))) * (2.0 / n_a)
        h_a_s, c5_a_s = binary_sa(h_a, n_a, 20.0, rng, temp0=2.0, temp_min=0.5)
        h_a_s, c5_a_s = greedy_descent_binary(h_a_s, n_a, 15.0, rng, max_sweeps=200)
        
        if c5_a_s < best_c5:
            best_c5 = c5_a_s
            best_h = h_a_s.copy()
            best_n = n_a
        
        # Upsample to 512
        h_b = to_binary(interp_h(h_a_s, n_a, n_b), n_b)
        c5_b0 = float(np.max(get_corr(h_b))) * (2.0 / n_b)
        h_b_s, c5_b_s = binary_sa(h_b, n_b, 20.0, rng, temp0=2.0, temp_min=0.5)
        h_b_s, c5_b_s = greedy_descent_binary(h_b_s, n_b, 15.0, rng, max_sweeps=200)
        
        if c5_b_s < best_c5:
            best_c5 = c5_b_s
            best_h = h_b_s.copy()
            best_n = n_b
        print("  Cross-res: C5={:.8f}".format(best_c5))
    
    # Phase 6: Final continuous polish at best resolution
    if tl() > 30:
        print("\n=== Phase 6: Final polish n={} ===".format(best_n))
        dur6 = min(25.0, tl() - 5)
        h_final, c5_final = continuous_refine(best_h, best_n, dur6, rng)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
            best_n = len(h_final)
        print("  After polish: C5={:.8f}".format(c5_final))
    
    # Phase 7: Ultra-high beta SLSQP
    if tl() > 15:
        for bv in [5000, 10000]:
            if tl() < 5:
                break
            h_p = slsqp_polish(best_h, best_n, bv, maxiter=300)
            c_p = compute_c5(h_p)
            if c_p < best_c5:
                best_c5 = c_p
                best_h = h_p
    
    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass
    
    _G_PEN = 0.0
    if hasattr('_G_FOCUS', '__class__'):
        pass
    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
