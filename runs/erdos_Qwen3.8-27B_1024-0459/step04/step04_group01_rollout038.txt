def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_trial = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_trial)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

        mask = (h > 1e-12) & (h < 1.0 - 1e-12)

def binarize(h, n):
    """Convert to binary with exactly n/2 ones."""
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        # Remove excess ones (lowest values among ones)
        ones_val = h[h_bin == 1]
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(ones_val)
        h_bin[ones_idx[order[:n_ones - target]]] = 0.0
    elif n_ones < target:
        zeros_val = h[h_bin == 0]
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-zeros_val)
        h_bin[zeros_idx[order[:target - n_ones]]] = 1.0
    return h_bin


def compute_delta(h, n, i, j):
    """Compute correlation change when moving 1 from i to j (h[i]:1->0, h[j]:0->1)."""
    delta = np.zeros(2 * n - 1)
    k = np.arange(-(n - 1), n)
    ik = i + k
    ikm = i - k
    jk = j + k
    jkm = j - k
    m1 = (ik >= 0) & (ik < n)
    m2 = (ikm >= 0) & (ikm < n)
    m3 = (jk >= 0) & (jk < n)
    m4 = (jkm >= 0) & (jkm < n)
    delta[m1] += (1.0 - h[ik[m1]])
    delta[m2] += -h[ikm[m2]]
    delta[m3] += -(1.0 - h[jk[m3]])
    delta[m4] += h[jkm[m4]]
    return delta

def slsqp_smooth(h, n, beta, maxiter=400):
    N = n
    TGT = n / 2.0
    BETA = beta

    def _smooth_obj(x):
        dx_val = 2.0 / N
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(BETA * shifted, -60, 60)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / BETA) * dx_val

    def _corr_grad_per_shift(x, nn, k):
        g = np.zeros(nn)
        off = k - (nn - 1)
        if off >= 0:
            npairs = nn - off
            if npairs > 0:
                g[off:off + npairs] += (1.0 - x[:npairs])
                g[:npairs] -= x[off:off + npairs]
        else:
            ao = -off
            npairs = nn - ao
            if npairs > 0:
                g[:npairs] += (1.0 - x[ao:ao + npairs])
                g[ao:ao + npairs] -= x[:npairs]
        return g

    def _smooth_grad(x):
        dx_val = 2.0 / N
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(BETA * shifted, -60, 60)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(N)
        for k in range(len(corr)):
            if w[k] > 1e-14:
                g += w[k] * _corr_grad_per_shift(x, N, k)
        return g * dx_val

    def _sum_con(x):
        return np.sum(x) - TGT

    def _sum_con_jac(x):
        return np.ones(N)

    bounds = [(0.0, 1.0)] * N
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, N)
    except Exception:
        return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

        h_bin_t = binarize(h_feas, n_t)

        # Quick local search
        h_ls_t, c5_ls_t = local_search_binary(h_bin_t, n_t, time_limit=60, rng=rng, temperature=0.01)
        if c5_ls_t < best_c5:
            best_c5 = c5_ls_t
            best_h = h_ls_t.copy()
            best_n = n_t

            # If improved, do more refinement
            if remaining_time() > 20:
                h_ls_t2, c5_ls_t2 = local_search_binary(h_ls_t, n_t, time_limit=60, rng=rng, temperature=0.005)
                if c5_ls_t2 < best_c5:
                    best_c5 = c5_ls_t2
                    best_h = h_ls_t2.copy()

    # Phase 5: Long intensive local search at best resolution
    if remaining_time() > 30:
        h_bin_best = binarize(best_h, best_n)
        # Multiple local search runs with different temperatures
        for temp in [0.0, 0.01, 0.005]:
            if remaining_time() < 20:
                break
            h_ls, c5_ls = local_search_binary(h_bin_best, best_n,
                                               time_limit=min(120, remaining_time() - 15),
                                               rng=rng, temperature=temp)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
                h_bin_best = best_h.copy()

    # Phase 6: Final SLSQP polish
    if remaining_time() > 10:
        h_final = slsqp_smooth(best_h, best_n, 20000, maxiter=600)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 7: One more binary refinement
    if remaining_time() > 10:
        h_bin_final = binarize(best_h, best_n)
        h_ls_f, c5_ls_f = local_search_binary(h_bin_final, best_n,
                                               time_limit=min(30, remaining_time() - 5),
                                               rng=rng, temperature=0.0)
        if c5_ls_f < best_c5:
            best_c5 = c5_ls_f
            best_h = h_ls_f.copy()

    # Validate and return
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
