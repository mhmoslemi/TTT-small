def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m

    ex = np.exp(cl)

def grad_shift(x, n, abs_lag):
    """Gradient of corr at a given absolute lag"""
    n_pairs = n - abs_lag
    if n_pairs <= 0:
        return np.zeros(n)
    g = np.zeros(n)
    g[abs_lag:n] += (1.0 - x[:n_pairs])
    g[:n_pairs] -= x[abs_lag:]
    return g

def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-10:
            abs_lag = abs(k - (n - 1))
            g += w[k] * grad_shift(x, n, abs_lag)
    return g * dx_val

def sum_con(x):
    return np.sum(x) - _TGT

def sum_con_jac(x):
    return np.ones(_N)

def slsqp_smooth(h, n, beta, maxiter=500):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta

                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})

    # Top-k gradient descent
    if time.time() - start < time_limit * 0.8 and remaining() > 10:
        h_td, c5_td = topk_grad_descent(h_work, n, max_iters=400, top_k=5)
        if c5_td < best_c5:
            best_c5 = c5_td
            best_h = h_td.copy()
            h_work = h_td.copy()

    # Weighted max descent
    if time.time() - start < time_limit * 0.9 and remaining() > 10:
        h_wd, c5_wd = weighted_max_descent(h_work, n, max_iters=300, n_weights=15)
        if c5_wd < best_c5:
            best_c5 = c5_wd
            best_h = h_wd.copy()
            h_work = h_wd.copy()

    # Final SLSQP at very high beta
    if time.time() - start < time_limit and remaining() > 8:
        for beta in [100000, 300000]:
            if remaining() < 5:
                break
            h_new = slsqp_smooth(h_work, n, beta, maxiter=600)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_work = h_new.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 360
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        best_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_n = parent_n

    # Phase 1: Binary SA at multiple resolutions
    n_list = [360, 380, 400, 420, 440, 460, 340, 480]
    n_list = [n for n in n_list if n % 2 == 0 and 200 <= n <= 600]
    # Remove duplicates while preserving order
    seen = set()
    n_list_unique = []
    for n_t in n_list:
        if n_t not in seen:
            seen.add(n_t)
            n_list_unique.append(n_t)
    n_list = n_list_unique

    for n_t in n_list:
        if remaining() < 20:
            break
        if n_t == best_n and best_h is not None:
            h_interp = best_h.copy()
        else:
            h_interp = interp_to(best_h, best_n, n_t)
        h_bin = make_binary(h_interp, n_t, rng)

        t_per_sa = min(30, max(8, remaining() / (len(n_list) * 2 + 2)))
        sa_seeds = [seed + 100 * n_t, seed + 200 * n_t + 7, seed + 300 * n_t + 13]
        for sa_seed in sa_seeds:
            if remaining() < 8:
                break
            h_sa, c5_sa = sa_binary(h_bin, n_t, time_limit=t_per_sa, seed=sa_seed)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_t

    # Phase 2: Continuous refinement at promising n values
    n_cands = sorted(set([best_n, best_n - 20, best_n + 20, 360, 380, 400, 420, 440]))
    n_cands = [n for n in n_cands if n % 2 == 0 and 200 <= n <= 600]
    # Remove duplicates
    seen = set()
    n_cands_unique = []
    for n_t in n_cands:
        if n_t not in seen:
            seen.add(n_t)
            n_cands_unique.append(n_t)
    n_cands = n_cands_unique

    for n_t in n_cands:
        if remaining() < 40:
            break
        if n_t == best_n:
            h_init = best_h.copy()
        else:
            h_init = interp_to(best_h, best_n, n_t)
            h_init = project_feasible(h_init, n_t)

        t_lim = min(90, max(25, remaining() / (len(n_cands) + 1)))
        h_ref, c5_ref = multi_beta_refine(h_init, n_t, time_limit=t_lim)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            best_n = n_t

    # Phase 3: Aggressive top-k on best solution
    n = best_n
    h = best_h.copy()
    if remaining() > 30:
        h_td, c5_td = topk_grad_descent(h, n, max_iters=500, top_k=3)
        if c5_td < best_c5:
            best_c5 = c5_td
            best_h = h_td.copy()
            h = h_td.copy()

    # Phase 4: Try higher resolution continuous
    for n_hi in [best_n + 40, best_n + 80, 720, 800]:
        if remaining() < 50:
            break
        if n_hi % 2 != 0:
            n_hi += 1
        if n_hi > 1000 or n_hi < 200:
            continue
        h_hi = interp_to(best_h, best_n, n_hi)
        h_hi = project_feasible(h_hi, n_hi)
        h_ref, c5_ref = multi_beta_refine(h_hi, n_hi, time_limit=60)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            best_n = n_hi

    # Phase 5: Final binary SA from best continuous (might find lower binary)
    if remaining() > 40:
        n = best_n
        h_bin = make_binary(best_h, n, rng)
        t_sa = min(50, max(15, remaining() * 0.35))
        for sa_seed in [seed + 9000, seed + 9500, seed + 9900]:
            if remaining() < 15:
                break
            h_sa, c5_sa = sa_binary(h_bin, n, time_limit=max(5, t_sa / 3), seed=sa_seed)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n

    # Phase 6: Final SLSQP polish at very high beta
    if remaining() > 15:
        h_new = slsqp_smooth(best_h, best_n, 200000, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 7: Final top-k
    if remaining() > 8:
        h_td, c5_td = topk_grad_descent(best_h, best_n, max_iters=300, top_k=3)
        if c5_td < best_c5:
            best_c5 = c5_td
            best_h = h_td.copy()

    # Phase 8: One more SLSQP
    if remaining() > 8:
        h_new = slsqp_smooth(best_h, best_n, 500000, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    # Try evaluate_erdos_solution for verification
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
