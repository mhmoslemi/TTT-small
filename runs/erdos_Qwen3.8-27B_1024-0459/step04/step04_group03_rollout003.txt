    """Coordinate descent: optimize one h[i] at a time."""
    dx = 2.0 / n
    h = h.copy()
    
    for pass_num in range(max_passes):
        max_improvement = 0
        for i in range(n):
            # Optimize h[i] with all others fixed
            old_val = h[i]
            # Ternary search on [0, 1]
            a, b = 0.0, 1.0
            for _ in range(60):  # about 1e-18 precision
                m1 = a + (b-a)/3
                m2 = b - (b-a)/3
                h[i] = m1
                h = maintain_sum(h, n)  # adjust to maintain sum
                f1 = compute_max_corr(h, n) * dx
                h[i] = m2
                h = maintain_sum(h, n)
                f2 = compute_max_corr(h, n) * dx
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            h[i] = (a+b)/2
            h = maintain_sum(h, n)
            new_val = h[i]
            improvement = old_val - new_val  # or f-based
            max_improvement = max(max_improvement, abs(improvement))
