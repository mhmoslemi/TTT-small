def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)

    n_half = n // 2
    sort_idx = np.argsort(-h)
    b = np.zeros(n)

def project_feasible(h, n):

    target = n / 2.0

            break

        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:

        frac = np.where((h > 0.01) & (h < 0.99))[0]
        if len(frac) > 0:
            h[frac] += diff / len(frac)
            h = np.clip(h, 0, 1)

def thue_morse(n, offset=0):
    """Generate Thue-Morse like pattern."""
    t = np.zeros(n)

        t[i] = 1.0 if c % 2 == 1 else 0.0

        t[ones_idx[:n_ones - n_half]] = 0
    elif n_ones < n_half:
        zeros_idx = np.where(t == 0)[0]
        t[zeros_idx[:n_half - n_ones]] = 1
    return t

def golden_ratio_pattern(n, offset=0):
    """Generate pattern based on golden ratio (Fibonacci-like)."""
    phi = (1 + np.sqrt(5)) / 2
    t = np.zeros(n)
    for i in range(n):

        t[i] = 1.0 if frac < 0.5 else 0.0

        t[ones_idx[:n_ones - n_half]] = 0
    elif n_ones < n_half:

        t[zeros_idx[:n_half - n_ones]] = 1

def slsqp_smooth(h, n, beta, maxiter=300):
    """SLSQP with smooth max approximation."""
    global _N_G, _BETA_G, _TGT_G
    _N_G = n
    _BETA_G = beta
    _TGT_G = n / 2.0
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        dx_v = 2.0 / n
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_v
    
    def grad(x):
        dx_v = 2.0 / n
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k] * dx_v
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.success or res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def top_shifts_equalize(h, n, top_k=5, max_iters=200):
    """Equalize top-k correlations using gradient steps."""
    h = h.copy()
    dx_v = 2.0 / n
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min_top = corr[idx[-1]]
        if c_max - c_min_top < 1e-10:
            break
        
        k_top = idx[0]
        k_bot = idx[-1]
        
        def corr_grad(k):
            g = np.zeros(n)
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += (1.0 - h[:np_])
                g[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= h[:np_]
            return g
        
        g_t = corr_grad(k_top)
        g_b = corr_grad(k_bot)
        d = g_b - g_t
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        
        improved = False
        for step in [0.01, 0.03, 0.05, 0.02, 0.08, 0.005, 0.04, 0.015]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        
        if not improved:
            break
    
    return h, best_c5

corr = np.correlate(h, 1-h, mode='full')

The 'full' mode correlation of a and v is defined as:
c[k] = sum_i a[i] * v[i-k+(len(v)-1)] for k = 0, ..., len(a)+len(v)-2

Wait, actually numpy's correlate is:
correlate(a, v) = sum_n a[n] * conj(v[n-k]) for the appropriate range

For 'full' mode with arrays of length n each:
result has length 2n-1
result[k] = sum_i a[i] * v[i-(k-(n-1))] for valid i

Let me verify: result[n-1] = sum_i a[i] * v[i] (the standard inner product)
result[n-1+k] = sum_i a[i] * v[i-k] (for k≥0, valid when 0≤i-k<n, i.e., k≤i<n+k)
result[n-1-k] = sum_i a[i] * v[i+k] (for k≥0, valid when 0≤i<n, 0≤i+k<n, i.e., i<n-k)

With a = h, v = 1-h:
result[n-1] = sum h[i](1-h[i]) = 0 for binary
result[n-1+k] = sum_{i=k}^{n-1} h[i](1-h[i-k]) for k≥0
result[n-1-k] = sum_{i=0}^{n-1-k} h[i](1-h[i+k]) for k≥0

Now the gradient of result[n-1+k] with respect to h[j]:
= d/dh[j] sum_{i=k}^{n-1} h[i](1-h[i-k])

For j in [k, n-1]: h[j] appears as h[i] with i=j, contributing (1-h[j-k])
For j in [0, n-1-k]: h[j] appears as h[i-k] with i=j+k, contributing -h[j+k] (derivative of (1-h) w.r.t. h)

Wait: d/dh[j] of h[j](1-h[j-k]) = (1-h[j-k])
d/dh[j] of h[j+k](1-h[j]) = h[j+k] * (-1) = -h[j+k]

So gradient of result[n-1+k] w.r.t. h[j]:
- If j ∈ [k, n-1]: +(1-h[j-k])
- If j ∈ [0, n-1-k]: -h[j+k]
- (These ranges may overlap)

For result[n-1-k] (k≥0):
= sum_{i=0}^{n-1-k} h[i](1-h[i+k])

d/dh[j]:
- If j ∈ [0, n-1-k]: +(1-h[j+k])
- If j ∈ [k, n-1]: -h[j-k]

So the full gradient of the objective (summation over all k with weights w[k]):

For offset d = k - (n-1) where k is the index in the full correlation:
- If d ≥ 0 (k = n-1+d): gradient contribution to h[j] is:
  +(1-h[j-d]) if j ≥ d

Ah, I forgot - the correlation is NOT linear in h. It's h[i]*(1-h[i-d]) which is bilinear. So it's a QP (quadratic program), not a LP.

However! If we fix most entries to 0 or 1, then:
- h[i]*(1-h[i-d]) where both h[i] and h[i-d] are fixed: constant
- h[i]*(1-h[i-d]) where one is fixed to 0: 0
- h[i]*(1-h[i-d]) where one is fixed to 1: h[other] or (1-h[other])
- h[i]*(1-h[i-d]) where both are free: h[i] - h[i]*h[i-d] (bilinear)

If we ensure that for every pair (i, i-d) that appears in the correlation, at least one is fixed, then the problem becomes linear! This is a "vertex cover" type constraint on the pairs.

For a specific shift d, the pairs are (i, i-d) for valid i. If I fix all entries except a small set F, then for the bilinear terms to vanish, I need: for all d and valid i, either i ∉ F or (i-d) ∉ F.

This means no two free entries can be at distance d from each other for any d that appears in the correlation. But ALL distances from 1 to n-1 appear. So I can have at most 1 free entry! That's not useful.

Unless... I only consider the "active" shifts (the top few). If I only need to satisfy constraints for, say, the top 10 shifts, then I only need to avoid bilinear terms for those specific d values. For a given set of d values, the constraint is: no two free entries at distance d for d in that set.

Actually, let me reconsider. The objective is max_k corr[k] * dx. The corr[k] values are not all independent. If I fix all but m entries to 0/1, and the m free entries are x_1, ..., x_m, then:

corr[k] = A[k] + sum_j B[k,j] * x_j + sum_{j1,j2} C[k,j1,j2] * x_{j1} * x_{j2}

where A is the contribution from all-fixed pairs, B from one-free-one-fixed, and C from two-free pairs.

If m is small (say ≤ 10), the C terms involve at most m² = 100 products. The overall problem is:
min t s.t. A[k] + B[k,:]·x + x^T C_k x ≤ t for all k, 0 ≤ x ≤ 1, sum(x) = target
