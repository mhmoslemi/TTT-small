```python
def sa_binary_focused(h, n, time_limit=30, seed=42):
    """
    Focused simulated annealing for binary h.
    Key improvement: track which shifts are "hot" (near max) and bias
    the acceptance toward moves that reduce hot shifts.
    """
    rng = np.random.RandomState(seed)
    h = h.copy()
    dx_val = 2.0 / n
    
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()
    
    T = 5.0
    alpha = 0.999
    start = time.time()
    stall = 0
    max_stall = 3000
    
    while time.time() - start < time_limit and T > 0.05:
        if remaining() < 10:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        delta = new_max - cur_max
        
        if delta <= 0 or (T > 0.05 and rng.rand() < np.exp(-delta / T)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
        
        T *= alpha
        if stall > max_stall:
            T = max(T, 1.5)
            stall = 0
            # Random restart within current solution
            for _ in range(10):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                ii = ones[rng.randint(len(ones))]
                jj = zeros[rng.randint(len(zeros))]
                h[ii], h[jj] = 0.0, 1.0
            corr = compute_corr(h)
            cur_max = float(np.max(corr))
    
    return best_h, best_max * dx_val
```

    return float(np.max(corr) * dx)

def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 5.0)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                mask = (h < 1.0)
            else:
                mask = (h > 0.0)
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
            else:
                break
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def corr_grad_for_shift(x, n, k):
    g = np.zeros(n)
    lag = k - (n - 1)
    if lag >= 0:
        npairs = n - lag
        g[lag:lag+npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[lag:lag+npairs]
    else:
        ao = -lag
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao+npairs])
        g[ao:ao+npairs] -= x[:npairs]
    return g

def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val

def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    s = np.sum(ex)
    w = ex / s
    g = np.zeros(n)
    # Only compute gradients for shifts with significant weight
    threshold = 1e-10
    active = np.where(w > threshold)[0]
    for k in active:
        wk = w[k]
        lag = k - (n - 1)
        if lag >= 0:
            npairs = n - lag
            g[lag:lag+npairs] += wk * (1.0 - x[:npairs])
            g[:npairs] -= wk * x[lag:lag+npairs]
        else:
            ao = -lag
            npairs = n - ao
            g[:npairs] += wk * (1.0 - x[ao:ao+npairs])
            g[ao:ao+npairs] -= wk * x[:npairs]
    return g * dx_val

def lbbfgs_optimize(h, n, beta, maxiter=300):
    """Optimize using L-BFGS-B with log-sum-exp smooth approximation."""
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                                'maxls': 50, 'maxfun': 5000, 'disp': False})
        return project_feasible(res.x, n)
    except Exception:
        return h

def sa_binary(h, n, time_limit=30, seed=42):
    """Binary simulated annealing."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    dx_val = 2.0 / n
    
    # Make binary
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1
    h = h_bin
    
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()
    
    T = 5.0
    alpha = 0.999
    start = time.time()
    stall = 0
    
    while time.time() - start < time_limit and T > 0.05:
        if remaining() < 10:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        delta = new_max - cur_max
        
        if delta <= 0 or (T > 0.05 and rng.rand() < np.exp(-delta / T)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
        
        T *= alpha
        if stall > 5000:
            T = max(T, 1.0)
            stall = 0
    
    return best_h, best_max * dx_val

def topk_gradient_descent(h, n, max_iters=300, top_k=5):
    """Top-k subgradient descent with projection."""
    h = h.copy()
    best_c5 = compute_c5(h)
    dx_val = 2.0 / n
    
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        m = float(np.max(corr))
        if m * dx_val >= best_c5 + 1e-8:
            break
        
        # Get top-k shifts
        top_idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k in top_idx:
            g += corr_grad_for_shift(h, n, int(k))
        g /= len(top_idx)
        
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        
        improved = False
        for step in [0.1, 0.05, 0.02, 0.1, 0.01, 0.05, 0.005, 0.2, 0.03, 0.015]:
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-12:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        
        if not improved:
            # Try smaller steps
            for step in [0.002, 0.001, 0.0005]:
                h_trial = h + step * d
                h_trial = project_feasible(h_trial, n)
                c_new = compute_c5(h_trial)
                if c_new < best_c5 - 1e-12:
                    h = h_trial
                    best_c5 = c_new
                    improved = True
                    break
            if not improved:
                break
    
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except NameError:
        parent_n = 360
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    # Phase 1: L-BFGS-B optimization at multiple n values with increasing beta
    n_list = sorted(set([280, 300, 320, 340, 360, 380, 400, 420, 440]))
    n_list = [n for n in n_list if n % 2 == 0 and 100 <= n <= 500]
    
    for n_t in n_list:
        if remaining() < 60:
            break
        if n_t == best_n:
            h_init = best_h.copy()
        else:
            h_init = interp_to(best_h, best_n, n_t)
            h_init = project_feasible(h_init, n_t)
        
        # Try multiple beta values
        for beta in [500, 2000, 10000]:
            if remaining() < 20:
                break
            h_new = lbbfgs_optimize(h_init, n_t, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_t
                h_init = h_new.copy()
    
    # Phase 2: Top-k gradient descent on best
    if remaining() > 30:
        h_td, c5_td = topk_gradient_descent(best_h, best_n, max_iters=500, top_k=5
