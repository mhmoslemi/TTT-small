    # then sum_p w_seq[p] * (1-h[j+p]) = convolution of (1-h) with w_seq, evaluated at j
    # (with appropriate convention)

    # We want: sum_p w[p+n-1] * (1-h)[j+p] = sum_i (1-h)[i] * w[j - i + n - 1 + ... ]
    # Hmm, this is getting confusing.

    grad = np.zeros(n)
    one_minus_h = 1.0 - h

    # Contribution to grad[j] from first factor: w[m] * (1-h[j+p]) if 0 <= j+p < n
    # Contribution to grad[j] from second factor: -w[m] * h[j-p] if 0 <= j-p < n

    # The second factor contributes -w[p+n-1] * h shifted by p

    for p in range(-(n-1), n):

        if p >= 0:

            second[0:n-pp] = h[pp:n]
        grad += wm * (first - second)
    
    return dx * grad

from scipy.optimize import minimize
import time

_G_START = 0.0
_G_BUDGET = 1300.0

    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c else 0.0
    return h

def make_rand_bin(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def ls_obj(h, beta, n, pen):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    values = corr * dx
    max_v = np.max(values)
    s = np.sum(np.exp(np.clip(beta * (values - max_v), -500, 500)))
    obj = max_v + np.log(s) / beta
    if pen > 0:
        obj += pen * (np.sum(h) - n * 0.5) ** 2
    return obj

def ls_grad(h, beta, n, pen):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    values = corr * dx
    max_v = np.max(values)
    exp_v = np.exp(np.clip(beta * (values - max_v), -500, 500))
    w = exp_v / np.sum(exp_v)

        if wm < 1e-16:

        else:
            pp = -p
            grad[pp:n] += wm * omh[0:n-pp]
            grad[0:n-pp] -= wm * h[pp:n]

    return grad

    bounds = [(0.0, 1.0)] * n
    res = minimize(ls_obj, h0, jac=lambda x: ls_grad(x, beta, n, pen),
                   args=(), method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8, 'iprint': 0})
    return res.x

def slsqp_optimize(h0, n, beta, maxiter):
    def obj(x):
        return ls_obj(x, beta, n, 0.0)
    def obj_grad(x):
        return ls_grad(x, beta, n, 0.0)
    def constraint(x):
        return np.sum(x) - n * 0.5
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint, 'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h0, jac=obj_grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
    return res.x

    best_c5 = compute_c5(best_h)
    for rnd in range(max_rounds):

        improved = False
        offsets = list(range(0, n, block_size))
        rng.shuffle(offsets)

            end = min(off + block_size, n)
            size = end - off
            block = best_h[off:end].copy()
            block_sum = np.sum(block)

                trial[off:end] = x
                # Adjust rest to maintain sum constraint
                rest_sum = np.sum(trial) - n * 0.5

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):

        if abs(diff) < 1e-15:
            break

        else:

    grad = np.zeros(n)
    for p in range(-(n - 1), n):
        wm = w[p + n - 1]
        if wm < 1e-16:
            continue
        if p >= 0:
            grad[0:n - p] += wm * omh[p:n]
            grad[p:n] -= wm * x[0:n - p]
        else:
            pp = -p
            grad[pp:n] += wm * omh[0:n - pp]
            grad[0:n - pp] -= wm * x[pp:n]
    grad *= dx
    if pen > 0:
        grad += 2.0 * pen * (np.sum(x) - n * 0.5)
    return grad

def cons_eq(x):
    return np.sum(x) - _G_N * 0.5

def cons_eq_jac(x):
    return np.ones(_G_N)

def lbfgsb_run(h0, n, beta, pen, maxiter):
    global _G_BETA, _G_N, _G_PEN
    _G_N = n
    _G_BETA = float(beta)
    _G_PEN = float(pen)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(ls_obj, h0, jac=ls_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-7, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    _G_PEN = 0.0
    return h

def slsqp_run(h0, n, beta, maxiter):
    global _G_BETA, _G_N, _G_PEN
    _G_N = n
    _G_BETA = float(beta)
    _G_PEN = 0.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}]
    try:
        res = minimize(ls_obj, h0, jac=ls_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    _G_PEN = 0.0
    return h

def coord_descent(h, n, max_sweeps, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    deltas = [1.0, -1.0, 0.5, -0.5, 0.25, -0.25, 0.1, -0.1,
              0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    for sweep in range(max_sweeps):
        if _tlimit() < 2.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if _tlimit() < 1.0:
                break
            for d in deltas:
                v = best_h[i] + d
                if v < -0.01 or v > 1.01:
                    continue
                trial = best_h.copy()
                trial[i] = np.clip(v, 0.0, 1.0)
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    best_h = trial
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def block_optimize(h, n, block_size, max_rounds, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for rnd in range(max_rounds):
        if _tlimit() < 3.0:
            break
        improved = False
        offsets = list(range(0, n - block_size + 1, block_size))
        rng.shuffle(offsets)
        for off in offsets:
            if _tlimit() < 1.5:
                break
            end = off + block_size
            block = best_h[off:end].copy()
            global _G_N2, _G_OFF2, _G_END2, _G_BH2
            _G_N2 = n
            _G_OFF2 = off
            _G_END2 = end
            _G_BH2 = best_h
            res = minimize(block_obj, block, method='Nelder-Mead',
                           options={'maxiter': 100, 'xatol': 1e-9, 'fatol': 1e-12})
            trial = best_h.copy()
            trial[off:end] = np.clip(res.x, 0.0, 1.0)
            trial = project_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c5 - 1e-12:
                best_c5 = c
                best_h = trial
                improved = True
        if not improved:
            break
    return best_h, best_c5

_G_N2 = 0
_G_OFF2 = 0
_G_END2 = 0
_G_BH2 = None

def block_obj(x):
    trial = _G_BH2.copy()
    trial[_G_OFF2:_G_END2] = np.clip(x, 0.0, 1.0)
    trial = project_feasible(trial, _G_N2)
    return compute_c5(trial)

def focused_slsqp(h, n, beta, maxiter, focus_lags):
    global _G_BETA, _G_N, _G_PEN
    _G_N = n
    _G_BETA = float(beta)
    _G_PEN = 0.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}]
    try:
        res = minimize(focused_obj, h, jac=focused_grad, method='SLSQP',

                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        h_out = project_feasible(res.x, n)
    except Exception:
        h_out = project_feasible(h, n)
    _G_PEN = 0.0
    return h_out

_G_FOCUS_LAGS = None

def focused_obj(x):
    n = _G_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    values = corr * dx
    fl = _G_FOCUS_LAGS
    if fl is not None and len(fl) > 0:
        fl = fl[fl < len(values)]
        val = np.max(values[fl]) + 0.01 * np.mean(values)
    else:
        val = np.max(values)
    return float(val)

def focused_grad(x):
    n = _G_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    values = corr * dx
    fl = _G_FOCUS_LAGS
    if fl is not None and len(fl) > 0:
        fl = fl[fl < len(values)]
        k_max = int(np.argmax(values[fl]))
        k_active = fl[k_max]
    else:
        k_active = int(np.argmax(values))
    omh = 1.0 - x
    grad = np.zeros(n)
    p = k_active - (n - 1)
    if p >= 0:
        grad[0:n - p] += omh[p:n]
        grad[p:n] -= x[0:n - p]
    else:
        pp = -p
        grad[pp:n] += omh[0:n - pp]
        grad[0:n - pp] -= x[pp:n]
    grad *= dx
    return grad
