def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_binary(h, n):
    """Project to binary with exactly n/2 ones."""
    h = np.array(h, dtype=float).copy()
    h = np.where(h >= 0.5, 1.0, 0.0)
    num_ones = int(np.sum(h))
    target = n // 2
    if num_ones > target:
        # Need to flip some 1s to 0s
        ones = np.where(h == 1.0)[0]
        # Flip the ones with lowest values (before thresholding)
        order = np.argsort(h[ones])
        to_flip = ones[order[:num_ones - target]]
        h[to_flip] = 0.0
    elif num_ones < target:
        zeros = np.where(h == 0.0)[0]
        order = np.argsort(-h[zeros])
        to_flip = zeros[order[:target - num_ones]]
        h[to_flip] = 1.0
    return h

def project_feasible(h, n):

    target = n / 2.0

        if abs(diff) < 1e-14:

        mask = (h > 1e-10) & (h < 1.0 - 1e-10)

        if cnt > 0:

        else:

    zeros = np.where(h < 0.5)[0]
    if len(ones) == 0 or len(zeros) == 0:

    best_c5 = c5_cur
    best_i, best_j = -1, -1

        to_flip = zeros[order[:target - num_ones]]
        h[to_flip] = 1.0
    return h

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 3.0)

def full_scan_improve(h, n):
    """One pass: try all (i,j) swaps, apply the best."""
    c5_cur = compute_c5(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h, c5_cur, 0
    best_c5 = c5_cur
    best_i, best_j = -1, -1
    for i in ones:
        for j in zeros:
            h[i] = 0.0
            h[j] = 1.0
            c5_new = compute_c5(h)
            h[i] = 1.0
            h[j] = 0.0
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_i, best_j = i, j
    if best_i >= 0:
        h[best_i] = 0.0
        h[best_j] = 1.0
    return h, best_c5, (1 if best_i >= 0 else 0)

def binary_hill_climb(h, n, max_passes=100):
    """Iterate full scan until convergence."""
    c5 = compute_c5(h)
    for p in range(max_passes):
        if remaining_time() < 5.0:
            break
        h, c5_new, improved = full_scan_improve(h, n)
        if not improved:
            break
        c5 = c5_new
    return h, c5

def sa_binary(h, n, t0, t_end, max_iter, rng):
    """Simulated annealing on binary space."""
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    ones_list = np.where(h > 0.5)[0].tolist()
    zeros_list = np.where(h < 0.5)[0].tolist()
    
    for it in range(max_iter):
        if remaining_time() < 5.0:
            break
        T = t0 * (t_end / t0) ** (it / float(max_iter))
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        delta = c5_new - c5_cur
        if delta < -1e-15 or (T > 1e-12 and rng.rand() < np.exp(-delta / T)):
            c5_cur = c5_new
            ones_list.remove(i)
            ones_list.append(j)
            zeros_list.remove(j)
            zeros_list.append(i)
            if c5_cur < c5_best - 1e-15:
                c5_best = c5_cur
                h_best = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return h_best, c5_best

    dx = _G_DX

    shifted = corr - m
    exps = np.exp(np.clip(_G_BETA * shifted, -600, 600))
    return (m + np.log(np.sum(exps)) / _G_BETA) * dx

    _G_TGT = n / 2.0

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def iterated_ls(h, n, rng, num_iters=30, perturb_size=5):
    """Iterated local search: perturb + hill climb, keep best."""
    best_h = h.copy()

                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def iterated_ls(h, n, rng, num_iters, perturb_size):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(num_iters):
        if remaining_time() < 15.0:
            break
        h_pert = best_h.copy()
        for _ in range(perturb_size):
            ones = np.where(h_pert > 0.5)[0]
            zeros = np.where(h_pert < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i], h_pert[j] = 0.0, 1.0
        # Occasionally do a block swap
        if it % 3 == 2:
            s1 = rng.randint(0, n - 10)
            L = rng.randint(3, min(12, n // 8))
            s2 = (s1 + rng.randint(L + 2, n - L - 2)) % (n - L)
            a = h_pert[s1:s1 + L].copy()
            h_pert[s1:s1 + L] = h_pert[s2:s2 + L]
            h_pert[s2:s2 + L] = a
        h_pert, c5_pert = binary_hill_climb(h_pert, n, max_passes=15)
        if c5_pert < best_c5 - 1e-15:
            best_h = h_pert.copy()
            best_c5 = c5_pert
    return best_h, best_c5


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h


def make_golden_ratio(n):
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        h[i] = 1.0 if (i * phi) % 1.0 < 0.5 else 0.0
    return h


def soft_local_search(h, n, max_iter=2000, rng=None):
    """Local search with small continuous perturbations."""
    if rng is None:
        rng = np.random.RandomState(77)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.05
    for it in range(max_iter):
        if remaining_time() < 4.0:
            break
        i = rng.randint(n)
        delta = step * (rng.rand() - 0.5) * 2.0
        h_trial = best_h.copy()
        h_trial[i] = np.clip(h_trial[i] + delta, 0.0, 1.0)
        h_trial = project_feasible(h_trial, n)
        c5_t = compute_c5(h_trial)
        if c5_t < best_c5 - 1e-15:
            best_h = h_trial
            best_c5 = c5_t
        else:
            step *= 0.97
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _G_BUDGET, _G_START, _G_N, _G_TGT, _G_DX, _G_BETA
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
    except Exception:
        parent_n = 256
        parent_h = make_thue_morse(parent_n)

    # Phase 0: Generate diverse candidates and quick-optimize
    print("--- Phase 0: Diverse initializations ---")
    n_base = parent_n
    candidates = []
    # Parent
    cand1 = project_binary(parent_h, n_base)
    candidates.append(cand1)
    # Thue-Morse
    cand2 = make_thue_morse(n_base)
    cand2 = project_binary(cand2, n_base)
    candidates.append(cand2)
    # Golden ratio
    cand3 = make_golden_ratio(n_base)
    cand3 = project_binary(cand3, n_base)
    candidates.append(cand3)
    # Random
    cand4 = np.zeros(n_base)
    idx = rng.permutation(n_base)
    cand4[idx[:n_base // 2]] = 1.0
    candidates.append(cand4)

    # Quick hill climb on each
    for ci in range(len(candidates)):
        if remaining_time() < 30.0:
            break
        candidates[ci], c5_i = binary_hill_climb(candidates[ci], n_base, max_passes=10)
        print("  Candidate {}: C5={:.10f}".format(ci, c5_i))
        if c5_i < best_c5:
            best_c5 = c5_i
            best_h = candidates[ci].copy()
            best_n = n_base

    n = best_n
    print("  Best after Phase 0: C5={:.10f}".format(best_c5))

    # Phase 1: Full binary hill climb
    print("\n--- Phase 1: Full binary hill climb (n={}) ---".format(n))
    best_h, best_c5 = binary_hill_climb(best_h, n, max_passes=80)
    print("  C5={:.10f}".format(best_c5))

    # Phase 2: Simulated annealing
    if remaining_time() > 60.0:
        print("\n--- Phase 2: SA (n={}) ---".format(n))
        best_h, best_c5 = sa_binary(best_h, n, t0=0.006, t_end=1e-9, max_iter=400000, rng=rng)
        best_h, best_c5 = binary_hill_climb(best_h, n, max_passes=40)
        print("  C5={:.10f}".format(best_c5))

    # Phase 3: Iterated local search
    if remaining_time() > 60.0:
        print("\n--- Phase 3: ITEL (n={}) ---".format(n))
        for ps in [3, 5, 8, 12, 6]:
            if remaining_time() < 20.0:
                break
            h_new, c5_new = iterated_ls(best_h, n, rng, num_iters=25, perturb_size=ps)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        print("  C5={:.10f}".format(best_c5))

    # Phase 4: SLSQP continuous polish
    if remaining_time() > 30.0:
        print("\n--- Phase 4: SLSQP polish (n={}) ---".format(n))
        best_h, best_c5 = slsqp_polish(best_h, n, betas=[300, 800, 2000, 5000], max_iter_per_beta=150)
        best_h = project_feasible(best_h, n)
        c5_cont = compute_c5(best_h)
        # Also try binary projection
        best_h_bin = project_binary(best_h, n)
        c5_bin = compute_c5(best_h_bin)
        if c5_bin < c5_cont:
            best_c5 = c5_bin
            best_h = best_h_bin
        print("  C5={:.10f}".format(best_c5))

    # Phase 5: Soft local search
    if remaining_time() > 20.0:
        print("\n--- Phase 5: Soft local search ---")
        h_soft = project_feasible(best_h.copy(), n)
        h_soft, c5_soft = soft_local_search(h_soft, n, max_iter=3000, rng=rng)
        if c5_soft < best_c5:
            best_c5 = c5_soft
            best_h = h_soft
        print("  C5={:.10f}".format(best_c5))

    # Phase 6: Higher resolution n=512
    if remaining_time() > 150.0:
        print("\n--- Phase 6: n=512 ---")
        n2 = 512
        h2 = interp_to(best_h, best_n, n2)
        h2 = project_binary(h2, n2)
        c5_2 = compute_c5(h2)
        print("  Interpolated: C5={:.10f}".format(c5_2))

        h2, c5_2 = binary_hill_climb(h2, n2, max_passes=15)
        print("  After hill climb: C5={:.10f}".format(c5_2))

        if remaining_time() > 80.0:
            h2, c5_2 = sa_binary(h2, n2, t0=0.003, t_end=1e-10, max_iter=150000, rng=rng)
            h2, c5_2 = binary_hill_climb(h2, n2, max_passes=15)
            print("  After SA: C5={:.10f}".format(c5_2))

        if remaining_time() > 40.0:
            h2, c5_2 = slsqp_polish(h2, n2, betas=[500, 2000], max_iter_per_beta=100)
            h2 = project_feasible(h2, n2)
            c5_2 = compute_c5(h2)
            h2b = project_binary(h2, n2)
            c5_2b = compute_c5(h2b)
            if c5_2b < c5_2:
                h2, c5_2 = h2b, c5_2b
            print("  After SLSQP: C5={:.10f}".format(c5_2))

        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n2
            print("  ** Improved! C5={:.10f}".format(best_c5))
        else:
            print("  n=512: no improvement (C5={:.10f})".format(c5_2))

    # Phase 7: Try n=128
    if remaining_time() > 50.0 and best_n > 128:
        print("\n--- Phase 7: n=128 check ---")
        n3 = 128
        h3 = interp_to(best_h, best_n, n3)
        h3 = project_binary(h3, n3)
        c5_3 = compute_c5(h3)
        h3, c5_3 = binary_hill_climb(h3, n3, max_passes=40)
        if remaining_time() > 25.0:
            h3, c5_3 = sa_binary(h3, n3, t0=0.01, t_end=1e-9, max_iter=200000, rng=rng)
            h3, c5_3 = binary_hill_climb(h3, n3, max_passes=30)
        if remaining_time() > 12.0:
            h3, c5_3 = slsqp_polish(h3, n3, betas=[500, 2000], max_iter_per_beta=100)
            h3 = project_feasible(h3, n3)
            c5_3b = compute_c5(project_binary(h3, n3))
            if c5_3b < c5_3:
                h3, c5_3 = project_binary(h3, n3), c5_3b
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = n3
            print("  ** Improved! n=128, C5={:.10f}".format(best_c5))
        else:
            print("  n=128: C5={:.10f} (no improvement)".format(c5_3))

    # Phase 8: Second SA round with different params
    if remaining_time() > 40.0:
        print("\n--- Phase 8: Second SA round ---")
        h8, c5_8 = sa_binary(best_h, best_n, t0=0.002, t_end=1e-10, max_iter=200000, rng=rng)
        h8, c5_8 = binary_hill_climb(h8, best_n, max_passes=30)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8
            print("  Improved: C5={:.10f}".format(best_c5))
        else:
            print("  No improvement (C5={:.10f})".format(c5_8))

    # Phase 9: Final polish
    if remaining_time() > 10.0:
        print("\n--- Phase 9: Final polish ---")
        best_h, best_c5 = binary_hill_climb(best_h, best_n, max_passes=30)
        if remaining_time() > 8.0:
            best_h, best_c5 = slsqp_polish(best_h, best_n, betas=[3000, 8000, 15000], max_iter_per_beta=100)
            best_h = project_feasible(best_h, best_n)
            c5b = compute_c5(project_binary(best_h, best_n))
            if c5b < best_c5:
                best_c5 = c5b
                best_h = project_binary(best_h, best_n)
        print("  C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
