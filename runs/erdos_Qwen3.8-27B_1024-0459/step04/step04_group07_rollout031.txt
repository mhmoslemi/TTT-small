    # where W[l] = weights[l + n - 1] (so W is indexed by lag from -(n-1) to n-1)

    # Second term: for each j, it's a "partial correlation" of W with h
    
    # We can compute these using np.correlate with mode='full' and then extract the right part
    # But the "partial" part (bounds) makes it tricky.
    
    # Alternative: use the fact that for valid indices, the correlation is well-defined
    # and for invalid indices, the contribution is 0.
    
    # Let's just do it with a simple loop over lags:
    W = weights  # length 2n-1, W[m] corresponds to lag m - (n-1)
    
    for m in range(2*n - 1):

        if w == 0:

            # This contributes to grad[i] for i in [0, n-1-lag]
            # Also: -w * h[i] for i in [lag, n-1] (the second argument)

            # corr[m] = sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag])
            # d(corr[m])/dh[j]:
            #   If j is in [0, n-1-lag]: j appears as first arg → (1-h[j+lag])
            #   If j is in [lag, n-1]: j appears as second arg (i+lag=j, i=j-lag) → -h[j-lag]
            #   If j is in [lag, n-1-lag]: both → (1-h[j+lag]) - h[j-lag]

                grad[idx2] -= w * h[idx2 - lag]
        else:

            l = -lag
            # corr[m] = sum_{i=0}^{n-1-l} h[i+l] * (1-h[i])
            # d(corr[m])/dh[j]:
            #   If j is in [l, n-1]: j appears as first arg (i+l=j, i=j-l) → (1-h[j-l])
            #   If j is in [0, n-1-l]: j appears as second arg → -h[j+l]
            
            if n - 1 - l >= 0:
                idx1 = np.arange(l, n)
                grad[idx1] += w * (1 - h[idx1 - l])
                
                idx2 = np.arange(0, n - l)
                grad[idx2] -= w * h[idx2 + l]
    
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
        # Re-fix
        diff = target - np.sum(h)
        if abs(diff) > 1e-15:
            if diff > 0:
                mask = h < 1.0 - 1e-12
            else:
                mask = h > 1e-12
            cnt = int(np.sum(mask))
            if cnt > 0:
                h[mask] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def lp_objective(h, p, pen=0.0, target=None):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / len(h)
    corr = corr * dx
    val = (np.sum(corr**p)) ** (1.0/p)
    if pen > 0 and target is not None:
        val += pen * (np.sum(h) - target) ** 2
    return float(val)

def lp_gradient(h, p, pen=0.0, target=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # corr[m] for m in [0, 2n-2], corresponds to lag m - (n-1)
    
    SP = np.sum(corr**p)
    if SP < 1e-30:
        grad = np.zeros(n)
    else:
        factor = SP ** (1.0/p - 1.0)
        weights = corr ** (p-1) * factor
        
        grad = np.zeros(n)
        for m in range(2*n - 1):
            w = weights[m]
            if abs(w) < 1e-20:
                continue
            lag = m - (n - 1)
            if lag >= 0:
                # c[m] = sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag])
                # dc/dh[j] = (1-h[j+lag]) for j in [0, n-1-lag]
                #          - h[j-lag] for j in [lag, n-1]
                n1 = n - lag
                if n1 > 0:
                    idx1 = np.arange(n1)
                    grad[idx1] += w * (1.0 - h[idx1 + lag])
                    idx2 = np.arange(lag, n)
                    grad[idx2] -= w * h[idx2 - lag]
            else:
                l = -lag
                n1 = n - l
                if n1 > 0:
                    idx1 = np.arange(l, n)
                    grad[idx1] += w * (1.0 - h[idx1 - l])
                    idx2 = np.arange(n1)
                    grad[idx2] -= w * h[idx2 + l]
    
    if pen > 0 and target is not None:
        grad += 2.0 * pen * (np.sum(h) - target)
    
    return grad

def lbfgsb_lp(h0, n, p, pen, maxiter, tlimit):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return lp_objective(x, p, pen, target)
    
    def grad(x):
        return lp_gradient(x, p, pen, target)
    
    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def slsqp_smooth(h0, n, beta, maxiter, tlimit):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(corr)
        s = corr - m
        e = np.exp(np.clip(beta * s, -400, 400))
        return float(m + np.log(np.sum(e)) / beta)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

    h_best = h.copy()
    c5 = compute_c5(h_best)
    
    for sweep in range(max_sweeps):
        if tlimit() < 2.0:
            break
        corr = get_corr(h_best)
        k_active = int(np.argmax(corr))
        lag = k_active - (n - 1)

                trial = h_best.copy()
                trial[i] = np.clip(trial[i] + delta, 0.0, 1.0)
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 1e-10:
                    c5 = c
                    h_best = trial.copy()
                    improved = True

def swap_local_search(h, n, max_iters, rng, tlimit):
    """Swap-based local search: pick two indices, transfer mass from one to other."""
    h_best = h.copy()
    c5 = compute_c5(h_best)
    
    for it in range(max_iters):
        if tlimit() < 2.0:
            break
        
        # Pick a random pair
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if h_best[i] <= 0.01 or h_best[j] >= 0.99:
            continue
        
        # Try transferring mass from i to j (or vice versa)
        for delta in [0.05, 0.1, 0.02, 0.15, -0.05, -0.1, -0.02, -0.15]:
            trial = h_best.copy()
            trial[i] = np.clip(trial[i] - delta, 0.0, 1.0)
            trial[j] = np.clip(trial[j] + delta, 0.0, 1.0)
            trial = project_feasible(trial, n)
            c = compute_c5(trial)
            if c < c5 - 1e-11:
                c5 = c
                h_best = trial.copy()
                break
    
    return h_best, c5
