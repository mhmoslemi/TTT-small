```python
def corr_grad(h, n):
    """Vectorized gradient of correlation."""
    # G[k, m] = (1 - h[m+L_k]) - h[m - L_k], L_k = k - n + 1
    # Let's build this with broadcasting
    L = np.arange(2*n-1) - n + 1  # shape (2n-1,)
    m = np.arange(n)  # shape (n,)
    # indices: m + L (for first term), m - L (for second term)
    idx1 = m[None, :] + L[:, None]  # shape (2n-1, n)
    idx2 = m[None, :] - L[:, None]  # shape (2n-1, n)
    
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    
    # Clip for safe indexing
    idx1c = np.clip(idx1, 0, n-1)
    idx2c = np.clip(idx2, 0, n-1)
    
    term1 = np.where(valid1, 1.0 - h[idx1c], 1.0)  # 1 - h[m+L], h=0 outside
    term2 = np.where(valid2, h[idx2c], 0.0)  # h[m-L], 0 outside
    
    return term1 - term2

2. For those lags, set up: min t s.t. t - dx*corr[k] >= 0 for active k, sum(h) = n/2, 0 <= h <= 1
3. Solve with SLSQP (minimize t)
4. Update active set
5. Repeat

The SLSQP problem for the minimax subproblem:
- Variables: h (n vars) + t (1 var) = n+1 variables
- Objective: t (linear, gradient is e_t)
- Inequality constraints: t - dx*corr[k] >= 0 for each active k (gradient: [d(corr[k])/dh * (-dx), 1] for the t component... wait)

Actually, the constraint is t >= dx * corr[k], i.e., t - dx*corr[k] >= 0.
The gradient of (t - dx*corr[k]) w.r.t. [h, t] is [-dx * G[k,:], 1] where G[k,:] is the gradient of corr[k] w.r.t. h.

For the equality constraint: sum(h) = n/2, gradient is [ones(n), 0].

This is a linear objective with quadratic constraints. SLSQP should handle it.

But the number of active constraints can be large. For n=256, there are 511 possible lags. If the top 20 are active, that's 20 inequality constraints. SLSQP can handle that.

Let me code the minimax:

_MM_DX = 0.0
_MM_LAGS = None

        # Simpler: take top-k and anything within 10% of max

        active = sorted(active)
        if len(active) < 3:

        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': _mm_ineq, 'jac': _mm_ineq_jac},
            {'type': 'eq', 'fun': _mm_eq, 'jac': _mm_eq_grad}
        ]
        
        try:

                         method='SLSQP', bounds=bounds, constraints=cons,

One more consideration: the `corr_grad_matrix` function allocates a (2n-1) × n array. For n=256, that's 511 × 256 ≈ 130K floats ≈ 1MB. For each SLSQP iteration, this is allocated. With 500 iterations, that's 500 allocations. Should be fine, but I could optimize by pre-allocating and reusing.

Actually, for the smooth objective, I only need G.T @ weights, which is (n, 2n-1) @ (2n-1,). I can compute this without materializing the full G matrix:

G.T @ weights = for each m: sum_k weights[k] * ((1-h[m+L_k]) - h[m-L_k])

This can be written as:
= sum_k weights[k] * (1-h[m+L_k]) - sum_k weights[k] * h[m-L_k]
= sum_k weights[k] - sum_k weights[k] * h[m+L_k] - sum_k weights[k] * h[m-L_k]

The first term is sum(weights) = 1 (since weights is a probability distribution).

For the second and third terms, they're like weighted correlations. I can compute them efficiently:

Let w = weights (shape 2n-1, where index k corresponds to lag L_k = k - n + 1).

sum_k w[k] * h[m + L_k] for each m:
This is like a correlation of h with w, but with w having a specific structure.

Actually, let me just compute it as a matrix multiplication. For n=256, G is 511×256, and G.T @ w is a 256×511 @ 511×1 = 256×1 operation. This is 130K multiply-adds. Very fast.

Let me just use the matrix approach since it's clean and fast enough.

```python
import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_SM_N = 0
_SM_BETA = 1.0
_MM_N = 0
_MM_DX = 0.0
_MM_LAGS = None

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def corr_grad_matrix(h, n):
    L = np.arange(2*n-1) - n + 1
    m = np.arange(n)
    idx1 = m[None, :] + L[:, None]
    idx2 = m[None, :] - L[:, None]
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    idx1c = np.clip(idx1, 0, n-1)
    idx2c = np.clip(idx2, 0, n-1)
    term1 = np.where(valid1, 1.0 - h[idx1c], 1.0)
    term2 = np.where(valid2, h[idx2c], 0.0)
    return term1 - term2

def smooth_value_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = beta * (corr - mc)
    shifted = np.minimum(shifted, 0.0)
    exp_s = np.exp(shifted)
    denom = np.sum(exp_s)
    weights = exp_s / denom
    G = corr_grad_matrix(x, n)
    grad = dx * (G.T @ weights)
    value = (mc + np.log(denom) / beta) * dx
    return value, grad

def _sm_obj(x):
    return smooth_value_grad(x, _SM_N, _SM_BETA)[0]

def _sm_grad(x):
    return smooth_value_grad(x, _SM_N, _SM_BETA)[1]

def _sm_eq(x):
    return np.sum(x) - _SM_N / 2.0

def _sm_eq_grad(x):
    return np.ones(_SM_N)

def slsqp_analytic(h_init, n, beta, maxiter=500):
    global _SM_N, _SM_BETA
    _SM_N = n
    _SM_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sm_eq, 'jac': _sm_eq_grad}]
    try:
        res = minimize(_sm_obj, h_init, jac=_sm_grad,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h_init.copy()

def _mm_obj(x):
    return x[_MM_N]

def _mm_obj_grad(x):
    g = np.zeros(_MM_N + 1)
    g[_MM_N] = 1.0
    return g

def _mm_ineq(x):

    h = x[:n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return x[n] - _MM_DX * corr[_MM_LAGS]

def _mm_ineq_jac(x):
    n = _MM_N
    h = x[:n]
    G = corr_grad_matrix(h, n)
    k = len(_MM_LAGS)
    jac = np.zeros((k, n + 1))
    jac[:, :n] = -_MM_DX * G[_MM_LAGS, :]
    jac[:, n] = 1.0
    return jac

def _mm_eq(x):

def _mm_eq_grad(x):
    g = np.zeros(_MM_N + 1)
    g[:_MM_N] = 1.0
    return g

def slsqp_minimax(h_init, n, max_rounds=20, top_k=8):

    _MM_N = n
    _MM_DX = 2.0 / n
    h = h_init.copy()
    c5_old = compute_c5(h)
    for rnd in range(max_rounds):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        mc = np.max(corr)
        # Active set: top-k lags + any within 10% of max
        sorted_idx = np.argsort(corr)[::-1]
        cutoff = mc * 0.95
        active = [i for i in sorted_idx if corr[i] >= cutoff]
        if len(active) < top_k:
            active = sorted_idx[:top_k].tolist()
        active = active[:min(len(active), 20)]  # cap at 20 constraints
        _MM_LAGS = np.array(active, dtype=int)
        
        x0 = np.concatenate([h, [mc * _MM_DX + 1e-6]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        cons = [
            {'type': 'ineq', 'fun': _mm_ineq, 'jac': _mm_ineq_jac},
            {'type': 'eq', 'fun': _mm_eq, 'jac': _mm_eq_grad}
        ]
        try:
            res = minimize(_mm_obj, x0, jac=_mm_obj_grad,
                         method='SLSQP', bounds=bounds, constraints=cons,
                         options={'maxiter': 300, 'ftol': 1e-15, 'iprint': 0})
            if res.x is not None:
                h_new = project_feasible(res.x[:n], n)
                c5_new = compute_c5(h_new)
                if c5_new < c5_old - 1e-14:
                    h = h_new
                    c5_old = c5_new
                else:
                    if rnd > 4:
                        break
        except Exception:
            break
    return h

def subgradient_refine(h, n, n_steps=200, alpha_init=0.02, top_k=5):
    """Subgradient descent on the non-smooth max objective."""
    h = h.copy()
    dx = 2.0 / n
    c5_best = compute_c5(h)
    h_best = h.copy()
    
    for step in range(n_steps):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        mc = np.max(corr)
        # Top-K lags for stabilized gradient
        sorted_idx = np.argsort(corr)[::-1][:top_k]
        # Averaged subgradient
        G = corr_grad_matrix(h, n)
        g = dx * (G[sorted_idx, :].sum(axis=0) / len(sorted_idx))
        
        # Adaptive step size
        alpha = alpha_init / (1.0 + step * 0.005)
        h_new = h - alpha * g
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()
        h = h_new
    
    return h_best, c5_best

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def combinatorial_local_search(h, n, max_moves=500):
    """Try binary-like swaps to improve C5."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Identify near-binary elements
    ones = np.where(best_h > 0.6)[0]
    zeros = np.where(best_h < 0.4)[0]
    
    moves = 0
    improved = True
    while improved and moves < max_moves:
        if remaining_time() < 3:
            break
        improved = False
        moves += 1
        
        # Try moving a 1 to a 0 position
        for i in ones[:50]:  # limit search
            for j in zeros[:50]:
                if moves > max_moves:
                    break
                trial = best_h.copy()
                trial[i] = min(trial[i], trial[j])  # swap: make i smaller, j bigger
                trial[j] = max(trial[i], trial[j])
                # Actually, let's just swap the values
                trial2 = best_h.copy()
                trial2[i], trial2[j] = best_h[j], best_h[i]
                trial2 = project_feasible(trial2, n)
                c5_t = compute_c5(trial2)
                if c5_t < best_c5 - 1e-12:
                    best_c5 = c5_t
                    best_h = trial2
                    ones = np.where(best_h > 0.6)[0]
                    zeros = np.where(best_h < 0.4)[0]
                    improved = True
    return best_h, best_c5
