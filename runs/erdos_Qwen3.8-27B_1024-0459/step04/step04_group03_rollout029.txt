    """Compute subgradient of max _k corr_k(h) * dx with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in active_lags:
        # k is the full-correlation index (0 to 2n-2)
        # Convert to lag: if k < n, lag = k (forward); if k >= n, lag = 2n-2-k (backward)
        if k < n:
            lag = k
            for i in range(n - lag):
                # h[i] * (1-h[i+lag])
                grad[i] += (1 - h[i + lag])
                grad[i + lag] += -h[i]
        else:

            for i in range(n - lag):
                # h[i+lag] * (1-h[i])  (by symmetry)
                grad[i + lag] += (1 - h[i])
                grad[i] += -h[i + lag]
    return grad * dx

_LAGS_G = None
_DX_G = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12 and n > 0:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def _sm_obj(x):
    n = _N_G
    beta = _BETA_G
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def _sm_eq(x):
    return np.sum(x) - _N_G / 2.0

def subgrad_gradient(h, n, active_lags):
    """Compute gradient contribution from active lags."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in active_lags:
        if k < n:
            lag = k
            # corr[k] = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
            for i in range(n - lag):
                grad[i] += (1.0 - h[i + lag])
                grad[i + lag] += (-h[i])
        else:
            lag = 2 * n - 2 - k
            # By symmetry, corr[2n-2-k] = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
            for i in range(n - lag):
                grad[i + lag] += (1.0 - h[i])
                grad[i] += (-h[i + lag])
    return grad * dx

    h_best = h.copy()
    c5_best = compute_c5(h)

        top = np.argsort(corr)[::-1][:top_m]

        grad = subgrad_gradient(h, n, top)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:

        best_c5 = c5_cur
