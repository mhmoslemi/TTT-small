import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_at_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        for i in range(n - lag):
            g[i] += (1.0 - h[i + lag])
            g[i + lag] -= h[i]
    else:
        L = -lag
        for i in range(L, n):
            g[i] += (1.0 - h[i - L])
            g[i - L] -= h[i]
    return g

def compute_all_grads(h):
    """Compute gradients of all correlation lags."""
    n = len(h)
    total_lags = 2 * n - 1
    grads = np.zeros((total_lags, n))
    for m in range(total_lags):
        lag = m - (n - 1)
        grads[m] = compute_grad_at_lag(h, lag)
    return grads

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        rooms = np.minimum(h, 1.0 - h)
        idx = int(np.argmax(rooms))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def do_lp_opt(h, n, budget_fn, max_iters=300, max_active=80):
    """LP-based outer approximation."""
    dx = 2.0 / n
    c5_best = compute_c5(h)
    h_best = h.copy()
    target = n / 2.0
    no_improve = 0
    trust_radius = 5.0

        t0 = time.time()
        corr = compute_corr(h) * dx
        t_cur = float(np.max(corr))

        K = min(max_active, 2 * n - 1)
        top_idx = np.argsort(corr)[-K:]
        top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
        
        # Build LP: min t s.t. g_k^T h - t <= offset_k
        n_var = n + 1
        c_obj = np.array([0.0] * n + [1.0])
        
        A_rows = []
        b_vals = []
        for k_idx in top_idx:
            lag = int(k_idx) - (n - 1)
            g = compute_grad_at_lag(h, lag) * dx
            offset = float(g @ h) - float(corr[k_idx])
            row = np.zeros(n_var)
            row[:n] = g
            row[n] = -1.0
            A_rows.append(row)
            b_vals.append(offset)
        
        A_ub = np.array(A_rows) if A_rows else np.zeros((1, n_var))
        b_ub = np.array(b_vals) if b_vals else np.array([0.0])

                          options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})

            no_improve += 1

            max_active = max(20, max_active // 2)
    
    return h_best, c5_best

def do_proj_grad(h, n, budget_fn, max_iters=500, beta=50.0):
    """Projected gradient with backtracking."""
    dx = 2.0 / n
    c5_best = compute_c5(h)
    h_best = h.copy()
    
    for it in range(max_iters):
        if budget_fn() < 3.0:
            break
        
        corr = compute_corr(h) * dx
        m = float(np.max(corr))
        
        # Softmax weights
        shifted = corr - m
        exps = np.exp(np.clip(beta * shifted, -200, 200))
        w = exps / np.sum(exps)
        
        # Gradient
        grad_h = np.zeros(n)
        sig = np.where(w > 1e-15)[0]
        for k_idx in sig:
            lag = int(k_idx) - (n - 1)
            g = compute_grad_at_lag(h, lag)
            grad_h += w[int(k_idx)] * g
        grad_h *= dx
        
        # Backtracking
        lr = 1.0
        improved = False
        for _ in range(20):
            h_new = project_feasible(h - lr * grad_h, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-15:
                h = h_new
                c5_best = c5_new
                h_best = h_new.copy()
                improved = True
                break
            lr *= 0.5
        if not improved and lr < 1e-8:
            break
    
    return h_best, c5_best

def do_slsqp_opt(h, n, budget_fn, beta_list):
    """SLSQP with log-sum-exp smoothing."""
    N = n
    DX = 2.0 / n
    TGT = n / 2.0
    
    def obj_fun(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        exps = np.exp(np.clip(beta_cur[0] * shifted, -500, 500))
        return (m + np.log(np.sum(exps)) / beta_cur[0]) * DX
    
    def con_fun(x):
        return np.sum(x) - TGT
    
    def con_jac(x):
        return np.ones(N)
    
    bounds = [(0.0, 1.0)] * N
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    
    for beta in beta_list:
        if budget_fn() < 8.0:
            break
        beta_cur = [float(beta)]
        maxiter = 300 if beta < 1000 else 200
        maxiter = min(maxiter, max(20, int(budget_fn() / 3.0)))
        try:
            res = minimize(obj_fun, h_cur, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, c5_cur

                    any_imp = True

                    # Re-project to maintain feasibility
                    h = project_feasible(h, n)
        if not any_imp and sweep > 2:
            break

Actually, I realize I should think more carefully about what makes the LP approach different from SLSQP:

SLSQP: solves a sequence of QP subproblems, each with a quadratic approximation of the objective and constraints. Uses BFGS to approximate the Hessian. Convergence is quadratic near the optimum.

LP outer approximation: at each step, linearizes the (non-convex) constraints and solves an LP. Convergence is linear but can be more robust for non-smooth/non-convex problems. The key advantage is that the LP is solved exactly (up to numerical precision), whereas SLSQP's QP subproblems are approximate.

For our problem, the objective is non-smooth (max of linear functions) and the constraints are non-convex (bilinear). The LP approach handles the non-smoothness naturally (the max is represented as multiple linear constraints) and the non-convexity is handled by iterative re-linearization.

def compute_grad_at_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        for i in range(n - lag):
            g[i] += (1.0 - h[i + lag])
            g[i + lag] -= h[i]
    else:
        L = -lag
        for i in range(L, n):
            g[i] += (1.0 - h[i - L])
            g[i - L] -= h[i]
    return g


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):

        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:

        else:
