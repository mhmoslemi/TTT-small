def smooth_max_objective(h, n, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Smooth max via log-sum-exp
    max_val = np.max(corr)
    shifted = beta * (corr - max_val)
    return max_val + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_gradient(h, n, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    shifted = beta * (corr * dx - max_val * dx)
    p = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # Gradient: for each i, sum over lags
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        for k in range(2*n-1):
            lag = k - (n-1)  # actual lag
            # d/dh[i] corr[k] = (1-h[i-lag]) * 1[0<=i-lag<n] - h[i+lag] * 1[0<=i+lag<n]

            if 0 <= i - lag < n:
                term += (1.0 - h[i - lag])
            if 0 <= i + lag < n:
                term -= h[i + lag]
            g += p[k] * term
        grad[i] = dx * g
    return grad

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    h = np.array(h, dtype=float).copy()

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)

    # Avoid overflow
    shifted -= np.max(shifted)
    p = np.exp(shifted)
    p_sum = np.sum(p)
    p /= p_sum

    q = p  # p[k] corresponds to lag k-(n-1)

        else:
            term2 = 0.0

            break
        obj, grad = smooth_obj_and_grad(h, beta)
        
        # Projected gradient step with backtracking
        step = lr
        for bt in range(10):

            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h):

            step *= 0.5
        else:

        c5_current = compute_c5(h)
        if c5_current < best_obj - 1e-12:
            best_obj = c5_current
            best_h = h.copy()
    
    return best_h, best_obj

    h = h_bin.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    for sweep in range(max_sweeps):

            break
        improved = False
        # Try all pairs (one, zero)
        for oi in range(len(ones)):
            if remaining_time() < 2:
                break
            for zi in range(len(zeros)):
                p = ones[oi]
                q = zeros[zi]
                h[p] = 0.0
                h[q] = 1.0
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-14:

                    improved = True
                    # Update lists
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
                else:
                    h[p] = 1.0
                    h[q] = 0.0
        if not improved:
            break
    
    return h, best_c5

Effect of changing h[q] from 0 to 1:
  - Position i=q: h[q]*(1-h[q+m]) changes from 0*(1-h[q+m]) to 1*(1-h[q+m]), delta = +(1-h[q+m])
  - Position i=q-m: h[q-m]*(1-h[q]) changes from h[q-m]*(1-0) to h[q-m]*(1-1)=0, delta = -h[q-m]
  
  So delta_q = (1-h[q+m]) - h[q-m] = 1 - h[q+m] - h[q-m] (for valid indices)

Total delta for swap (p,q): delta = delta_p + delta_q = h[p-m] + h[p+m] - 1 + 1 - h[q+m] - h[q-m]
= h[p-m] + h[p+m] - h[q+m] - h[q-m]

Where all h values are the ORIGINAL h values, and we only add the terms when the index is valid.

So for each lag m, the delta is:
delta[m] = h[p-m]*1[0≤p-m<n] + h[p+m]*1[0≤p+m<n] - h[q-m]*1[0≤q-m<n] - h[q+m]*1[0≤q+m<n]

This is O(1) per lag! So computing the full delta vector is O(n).

Then the new max is max(C + delta), which is O(n).

So each trial swap costs O(n). For all pairs (O(n²/4) = 16K for n=256), total is O(n³) = 16M. In a tight numpy loop, this might take 1-2 seconds per full sweep.

Let me optimize: precompute for each position i and each lag m, the values h[i-m] and h[i+m]. This is a 2D array of size (n, 2n-1). Then the delta for a swap (p,q) is just row_p - row_q (element-wise), which is O(n).

Actually even better: for each position i, define:
A[i, m] = h[i-m]*1[valid] + h[i+m]*1[valid]

Then delta for swap (p,q) = A[p] - A[q] (element-wise over valid m values)

And new C = C + A[p] - A[q]
new max = max(C + A[p] - A[q])

This is O(n) per swap with precomputation of A (O(n²) once).

For n=256: A is 256 x 511, about 131K elements. Very manageable.

def binary_local_search_fast(h_bin, n, max_sweeps=30):

    dx = 2.0 / n
    best_max = np.max(C)
    
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()

        # A[i, m_idx] = h[i-m]*1[0≤i-m<n] + h[i+m]*1[0≤i+m<n]
        # where m = m_idx - (n-1), m_idx in 0..2n-2
        A = np.zeros((n, 2*n-1))
        for i in range(n):
            for m_idx in range(2*n-1):
                m = m_idx - (n-1)
                val = 0.0
                if 0 <= i - m < n:
                    val += h[i - m]
                if 0 <= i + m < n:
                    val += h[i + m]
                A[i, m_idx] = val
        
        best_swap = None
        best_new_max = best_max + 1  # we want to minimize
        
        # Try all (one, zero) pairs
        for oi in range(len(ones)):
            p = ones[oi]
            for zi in range(len(zeros)):
                q = zeros[zi]
                new_C = C + A[p] - A[q]
                new_max = np.max(new_C)
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_swap = (p, q)
        
        if best_swap is not None and best_new_max < best_max - 1e-10:
            p, q = best_swap
            h[p] = 0.0
            h[q] = 1.0
            C = C + A[p] - A[q]
            best_max = best_new_max
            best_c5 = best_max * dx
            # Update lists
            ones.remove(p)
            zeros.remove(q)
            ones.append(q)
            zeros.append(p)
            h[p] = 0.0
            h[q] = 1.0
        else:
            break
    
    h[ones] = 1.0
    h[zeros] = 0.0
    return h, compute_c5(h)

import numpy as np
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)

def smooth_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = beta * (corr - max_c) * dx
    shifted = shifted - np.max(shifted)
    exp_vals = np.exp(shifted)
    p = exp_vals / np.sum(exp_vals)
    obj = (max_c + np.log(np.sum(exp_vals)) / beta) * dx
    grad = np.zeros(n)
    nm1 = n - 1
    for i in range(n):
        lo1 = max(0, nm1 - i)
        hi1 = min(2*n-2, 2*n-2-i)
        if lo1 <= hi1:
            pos = np.arange(lo1, hi1+1) - nm1 + i
            term1 = np.dot(p[lo1:hi1+1], 1.0 - h[pos])
        else:
            term1 = 0.0
        lo2 = max(0, i - nm1)
        hi2 = min(2*n-2, i + nm1)
        if lo2 <= hi2:
            pos2 = i - (np.arange(lo2, hi2+1) - nm1)
            term2 = np.dot(p[lo2:hi2+1], h[pos2])
        else:
            term2 = 0.0
        grad[i] = dx * (term1 - term2)
    return obj, grad

def projected_grad_descent(h0, n, beta, max_iter=3000, lr_init=0.05):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    for it in range(max_iter):
        if remaining_time() < 3:
            break
        obj, grad = smooth_obj_and_grad(h, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        step = lr
        improved = False
        for bt in range(12):
            h_new = h - step * grad
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            if step < 1e-8:
                break
            h_new = h - step * grad
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new <= best_c5 + 1e-14:
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            if it % 100 == 50:
                lr *= 0.7
    return best_h, best_c5

def build_A_matrix(h, n):
    nm1 = n - 1
    A = np.zeros((n, 2*n-1))
    h_idx = np.arange(n)
    for i in range(n):
        m_vals = np.arange(2*n-1) - nm1
        im = i - m_vals
        ip = i + m_vals
        mask_im = (im >= 0) & (im < n)
        mask_ip = (ip >= 0) & (ip < n)
        A[i, mask_im] += h[im[mask_im]]
        A[i, mask_ip] += h[ip[mask_ip]]
    return A

def binary_local_search_fast(h_bin, n, max_sweeps=40):
    h = np.round(h_bin).clip(0, 1)
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    C = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_max = np.max(C)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h < 0.5)[0].tolist()
    for sweep in range(max_sweeps):
        if remaining_time() < 8:
            break
        A = build_A_matrix(h, n)
        best_swap = None
        best_new_max = best_max + 1.0
        A_zeros = A[np.array(zeros)]
        for oi in range(len(ones)):
            p = ones[oi]
            new_maxes = np.max(C[None, :] + A[p][None, :] - A_zeros, axis=1)
            min_idx = np.argmin(new_maxes)
            if new_maxes[min_idx] < best_new_max:
                best_new_max = new_maxes[min_idx]
                best_swap = (p, zeros[min_idx])
        if best_swap is not None and best_new_max < best_max - 1e-10:
            p, q = best_swap
            h[p] = 0.0
            h[q] = 1.0
            C = C + A[p] - A[q]
            best_max = best_new_max
            best_c5 = best_max * dx
            ones.remove(p)
            zeros.remove(q)
            ones.append(q)
            zeros.append(p)
        else:
            break
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def random_binary_init(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n//2, replace=False)
    h[ones] = 1.0
    return h
