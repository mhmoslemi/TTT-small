        k_active = int(np.argmax(corr))
        h_new = slsqp_run(h_best, n, beta=800.0, maxiter=200, focus_lag=k_active)
        c_new = compute_c5(h_new)
        if c_new < c5_best - 1e-12:
            c5_best = c_new
            h_best = h_new
    return h_best, c5_best

def multi_beta_slsqp(h, n, tlimit, betas, maxiter_each=300):
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for bv in betas:
        if tlimit() < 8.0:
            break
        h_new = slsqp_run(h_cur, n, beta=bv, maxiter=maxiter_each)
        c_new = compute_c5(h_new)
        if c_new < c5_cur - 1e-13:
            c5_cur = c_new
            h_cur = h_new
    return h_cur, c5_cur

def block_flip_search(h_bin, n, max_sweeps, rng, tlimit, block_size=2):
    h = np.array(h_bin, dtype=float).copy()
    corr = get_corr(h)
    c5_corr = float(np.max(corr))
    c5 = c5_corr * (2.0 / n)
    best_h = h.copy()
    best_c5 = c5
    for sweep in range(max_sweeps):
        if tlimit() < 2.0:
            break
        improved = False
        for _ in range(n // block_size):
            if tlimit() < 1.0:
                break
            start = rng.randint(0, n - block_size + 1)
            idxs = list(range(start, start + block_size))
            for idx in idxs:
                h[idx] = 1.0 - h[idx]
            new_corr = get_corr(h)
            new_max = float(np.max(new_corr))
            if new_max < c5_corr - 1e-12:
                corr = new_corr
                c5_corr = new_max
                c5 = c5_corr * (2.0 / n)
                best_h = h.copy()
                best_c5 = c5
                improved = True
            else:
                for idx in idxs:
                    h[idx] = 1.0 - h[idx]
        if not improved:
            break
    return best_h, c5

def round_to_binary(h, n):
    h_bin = (h > 0.5).astype(float)
    ones = int(np.sum(h_bin))
    target = n // 2
    if ones > target:
        diffs = h - h_bin
        order = np.argsort(diffs)[::-1]
        for idx in order:
            if ones == target:
                break
            h_bin[idx] = 0.0
            ones -= 1
    elif ones < target:
        diffs = h - h_bin
        order = np.argsort(diffs)
        for idx in order:
            if ones == target:
                break
            h_bin[idx] = 1.0
            ones += 1
    return h_bin

def run(seed=42, budget_s=1300, **kwargs):
    global _beta, _dx_val, _target_val, _focus_val, _pen_val
    t_start = time.time()
    total_budget = float(budget_s)
    _pen_val = 0.0
    _focus_val = -1
    rng = np.random.RandomState(seed)

    def tlimit():
        return max(0.0, total_budget - (time.time() - t_start) - 3.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent solution
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = make_thue_morse(parent_n)
        parent_h = project_feasible(parent_h, parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("No parent, using TM: n={}, C5={:.8f}".format(parent_n, best_c5))

    # Phase 1: Quick SLSQP at parent resolution
    if tlimit() > 60:
        print("\n=== Phase 1: SLSQP at n={} ===".format(best_n))
        n = best_n
        for bv in [50, 100, 300, 800, 2000]:
            if tlimit() < 5:
                break
            h_new = slsqp_run(best_h, n, beta=bv, maxiter=250)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-13:
                best_c5 = c_new
                best_h = h_new
                print("  beta={}: C5={:.8f}".format(bv, best_c5))
        print("  Phase 1 done: C5={:.8f}".format(best_c5))

    # Phase 2: Interpolate to n=512 and optimize
    if tlimit() > 100:
        print("\n=== Phase 2: n=512 optimization ===")
        n512 = 512
        h512 = interp_to(best_h, best_n, n512) if best_n != n512 else best_h.copy()
        h512 = project_feasible(h512, n512)
        c512 = compute_c5(h512)
        print("  Interpolated: C5={:.8f}".format(c512))
        
        for bv in [50, 200, 600, 1500, 4000]:
            if tlimit() < 10:
                break
            h_new = slsqp_run(h512, n512, beta=bv, maxiter=200)
            c_new = compute_c5(h_new)
            if c_new < c512 - 1e-13:
                c512 = c_new
                h512 = h_new
                print("  beta={}: C5={:.8f}".format(bv, c512))
        
        if c512 < best_c5 - 1e-9:
            best_c5 = c512
            best_h = h512.copy()
            best_n = n512
            print("  n=512 improved best! C5={:.8f}".format(best_c5))
        else:
            print("  n=512 best: C5={:.8f} (not better than {:.8f})".format(c512, best_c5))

    # Phase 3: Binary local search at n=256
    if tlimit() > 80:
        print("\n=== Phase 3: Binary search n=256 ===")
        n256 = 256
        # Generate multiple binary starts
        starts_256 = []
        # From parent (rounded)
        if best_n == 256:
            starts_256.append(round_to_binary(best_h, 256))
        else:
            h256 = interp_to(best_h, best_n, 256)
            starts_256.append(round_to_binary(h256, 256))
        # Thue-Morse
        starts_256.append(make_thue_morse(256))
        # Random
        for _ in range(3):
            starts_256.append(make_random_bin(rng, 256))
        
        for si, h_bin in enumerate(starts_256):
            if tlimit() < 10:
                break
            h_bi = np.array(h_bin, dtype=float).copy()
            h_bi = project_feasible(h_bi, 256)
            h_res, c_res = binary_local_search(h_bi, 256, max_sweeps=50, rng=rng, tlimit=tlimit)
            if c_res < best_c5 - 1e-9:
                best_c5 = c_res
                best_h = h_res.copy()
                best_n = 256
                print("  Binary start {}: C5={:.8f} (NEW BEST)".format(si, best_c5))
            else:
                print("  Binary start {}: C5={:.8f}".format(si, c_res))

    # Phase 4: Binary SA at n=256 and n=512
    if tlimit() > 80:
        print("\n=== Phase 4: Binary SA search ===")
        for n_sa in [256, 512]:
            if tlimit() < 20:
                break
            if n_sa == best_n:
                h_start = round_to_binary(best_h, n_sa)
            else:
                h_interp = interp_to(best_h, best_n, n_sa)
                h_start = round_to_binary(h_interp, n_sa)
            h_start = project_feasible(h_start, n_sa)
            h_res, c_res = binary_sa_search(h_start, n_sa, max_sweeps=100, rng=rng, tlimit=tlimit)
            if c_res < best_c5 - 1e-9:
                best_c5 = c_res
                best_h = h_res.copy()
                best_n = n_sa
                print("  SA n={}: C5={:.8f} (NEW BEST)".format(n_sa, best_c5))
            else:
                print("  SA n={}: C5={:.8f}".format(n_sa, c_res))

    # Phase 5: n=1024 attempt
    if tlimit() > 120:
        print("\n=== Phase 5: n=1024 ===")
        n1024 = 1024
        h1024 = interp_to(best_h, best_n, n1024)
        h1024 = project_feasible(h1024, n1024)
        c1024 = compute_c5(h1024)
        print("  Interpolated: C5={:.8f}".format(c1024))
        
        for bv in [50, 200, 800]:
            if tlimit() < 15:
                break
            if n1024 <= 600:
                h_new = slsqp_run(h1024, n1024, beta=bv, maxiter=150)
            else:
                h_new = lbfgs_run(h1024, n1024, beta=bv, maxiter=200, pen=30.0)

            if c_new < c1024 - 1e-13:

        if c1024 < best_c5 - 1e-9:
            best_c5 = c1024
            best_h = h1024.copy()
            best_n = n1024
            print("  n=1024 improved! C5={:.8f}".format(best_c5))
        else:
            print("  n=1024: C5={:.8f} (not better)".format(c1024))

    # Phase 6: Focused optimization at best resolution
    if tlimit() > 50:
        print("\n=== Phase 6: Focused opt n={} ===".format(best_n))
        best_n = len(best_h)

    # Phase 8: Block flip + re-optimize
    if tlimit() > 40:

        best_n = len(best_h)
        h_bin = round_to_binary(best_h, best_n)
        h_bin = project_feasible(h_bin, best_n)
        h_res, c_res = block_flip_search(h_bin, best_n, max_sweeps=20, rng=rng, tlimit=tlimit, block_size=2)

        if tlimit() > 15:

    if tlimit() > 40:

        best_n = len(best_h)
        for attempt in range(10):
            if tlimit() < 10:
