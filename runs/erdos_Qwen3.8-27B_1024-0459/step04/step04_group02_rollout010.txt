def project_sum_box(x, target_sum):
    """Project x onto {x : sum(x) = target_sum, 0 <= x_i <= 1}."""
    x = np.clip(x, 0.0, 1.0)
    # Find lambda such that sum(clip(x - lambda, 0, 1)) = target_sum
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2, 0.0, 1.0)

_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_sum_box(x, target_sum):
    """Project onto sum(x)=target, 0<=x<=1 via bisection on Lagrange multiplier."""
    x = np.asarray(x, dtype=np.float64)
    # If already feasible, return
    if x.min() >= -1e-15 and x.max() <= 1 + 1e-15 and abs(x.sum() - target_sum) < 1e-12:
        return np.clip(x, 0.0, 1.0)
    x = np.clip(x, 0.0, 1.0)
    # Bisection for lambda
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    # Tiny correction
    diff = target_sum - result.sum()
    if abs(diff) > 1e-14:
        # Find an interior point to adjust
        interior = np.where((result > 1e-12) & (result < 1.0 - 1e-12))[0]
        if len(interior) > 0:
            result[interior[0]] = np.clip(result[interior[0]] + diff, 0.0, 1.0)
    return result

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

_N_W = 0
_B_W = 0.0

def _wrapped_obj(x):
    return _smooth_obj(x, _N_W, _B_W)[0]

def _wrapped_grad(x):
    return _smooth_obj(x, _N_W, _B_W)[1]

def _wrapped_cons_eq(x):
    return x.sum() - _N_W / 2.0

def _wrapped_cons_jac(x):
    return np.ones(_N_W)

def _smooth_obj(h, n, beta):
    """Smooth approximation of max_k corr_k * dx, and its gradient."""
    dx = 2.0 / n
    corr = compute_corr_full(h)
    m = np.max(corr)
    # For very large beta, use a simpler approach
    if beta > 1e8:
        # Essentially exact max
        mask = corr >= m - 1e-10
        wts = np.zeros_like(corr)
        wts[mask] = 1.0 / mask.sum()
    else:
        shifted = np.minimum(beta * (corr - m), 0.0)
        ev = np.exp(shifted)
        ws = ev.sum()
        wts = ev / ws
        m = m + np.log(ws) / beta
    
    val = m * dx
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = wts[j] * dx
        if w < 1e-15:
            continue
        k = j - (n - 1)
        m_abs = abs(k)
        if m_abs == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            # Segment A: j_idx in [0, min(m_abs, n-m_abs))
            # Contribution: -h[j_idx + m_abs]
            endA = min(m_abs, n - m_abs)
            if endA > 0:
                grad[:endA] -= w * h[m_abs:m_abs + endA]
            # Segment B: j_idx in [m_abs, n - m_abs)
            # Contribution: (1 - h[j_idx - m_abs]) - h[j_idx + m_abs]
            if n - 2 * m_abs > 0:
                seg = n - 2 * m_abs
                grad[m_abs:n - m_abs] += w * ((1.0 - h[:seg]) - h[2*m_abs:2*m_abs + seg])
            # Segment C: j_idx in [n - m_abs, n)
            # Contribution: 1 - h[j_idx - m_abs]
            grad[n - m_abs:] += w * (1.0 - h[n - 2*m_abs:n - m_abs])
    
    return val, grad

def slsqp_optimize(h, n, beta, maxiter=300):
    """Run SLSQP on the smooth approximation."""
    global _N_W, _B_W
    _N_W = n
    _B_W = float(beta)
    
    h0 = project_sum_box(h, n / 2.0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _wrapped_cons_eq, 'jac': _wrapped_cons_jac}]
    
    try:
        res = minimize(_wrapped_obj, h0, jac=_wrapped_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-25, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_sum_box(res.x, n / 2.0)
    except Exception:
        pass
    return h0.copy()
