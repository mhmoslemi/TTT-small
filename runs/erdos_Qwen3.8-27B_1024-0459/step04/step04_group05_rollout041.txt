def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def to_binary(h, n):
    n_ones = n // 2
    sorted_idx = np.argsort(-h, kind='stable')
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n_ones]] = 1.0
    return h_bin

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

        else:

def binary_sa(h_init, n, rng, budget_fn, T_start=0.15, T_min=0.001):

    h = to_binary(h_init, n)
    c5_cur = compute_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    T = T_start
    cooling = 0.99998
    no_improve = 0
    max_patience = 30000
    it = 0
    
    while budget_fn() > 5:
        # Pick random pair
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        delta = c5_new - c5_cur

            c5_cur = c5_new

def continuous_refine(h_init, n, budget_fn, beta_list=[500, 2000, 10000]):
    """SLSQP with log-sum-exp approximation."""
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        exps = np.exp(np.clip(beta * shifted, -500, 500))
        return (m + np.log(np.sum(exps)) / beta) * dx

        return np.sum(x) - target

    # Phase 4: SA from golden ratio
    if budget_fn() > 60:
        print("\n--- P4: SA from golden ---")
        gr = (1.0 + np.sqrt(5.0)) / 2.0
        h_gr = np.zeros(parent_n)
        for ii in range(parent_n):
            frac = (ii * gr) % 1.0
            h_gr[ii] = 1.0 if frac < 0.5 else 0.0
        h_gr = to_binary(h_gr, parent_n)
        h_gr_sa, c5_gr = binary_sa(h_gr, parent_n, rng, budget_fn, T_start=0.15, T_min=0.002, max_patience=30000)
        print("  golden SA: C5={:.10f}".format(c5_gr))
        if c5_gr < best_c5:
            best_c5 = c5_gr
            best_h = h_gr_sa.copy()
            best_n = parent_n

    # Phase 5: Targeted SA from best
    if budget_fn() > 50:
        print("\n--- P5: Targeted SA ---")
        h_t, c5_t = targeted_sa(best_h, best_n, rng, budget_fn, T_start=0.08, T_min=0.001, max_patience=35000)
        print("  targeted: C5={:.10f}".format(c5_t))
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 6: Other n values
    if budget_fn() > 40:
        print("\n--- P6: Other n ---")
        for n_try in [254, 258, 260, 264, 272, 280, 288, 320]:
            if budget_fn() < 30:
                break
            h_interp = to_binary(interp_to(best_h, best_n, n_try), n_try)
            c5_init = compute_c5(h_interp)
            h_o, c5_o = binary_sa(h_interp, n_try, rng, budget_fn, T_start=0.1, T_min=0.002, max_patience=20000)
            c5_show = min(c5_o, c5_init)
            if c5_show < best_c5:
                print("  n={} IMPROVED: C5={:.10f}".format(n_try, c5_show))
                best_c5 = c5_show
                best_h = h_o.copy() if c5_o < c5_init else h_interp.copy()
                best_n = n_try
            else:
                print("  n={}: C5={:.10f} (best={:.10f})".format(n_try, c5_show, best_c5))

    # Phase 7: Continuous refinement
    if budget_fn() > 25:
        print("\n--- P7: Continuous ---")
        h_c, c5_c = continuous_refine(best_h, best_n, budget_fn, [500, 2000, 8000])
        print("  cont: C5={:.10f}".format(c5_c))
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c

    # Phase 8: Final SA polish
    if budget_fn() > 15:
        print("\n--- P8: Final SA ---")
        h_f, c5_f = binary_sa(best_h, best_n, rng, budget_fn, T_start=0.04, T_min=0.0005, max_patience=30000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 9: Final continuous polish
    if budget_fn() > 10:
        h_p, c5_p = continuous_refine(best_h, best_n, budget_fn, [5000, 20000])
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, best_c5))
    return (best_h, best_c5, best_n)
