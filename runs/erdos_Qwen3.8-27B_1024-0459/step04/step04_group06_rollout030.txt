def project_feasible(h, n):

    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    max_s = float(np.max(shifted))
    exps = np.exp(shifted - max_s)
    total = np.sum(exps)
    weights = exps / total / _G_BETA
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-15:
            continue
        offset = k_idx - (n - 1)
        if offset >= 0:
            nv = n - offset
            if nv <= 0:
                continue
            grad[:nv] += w * (1.0 - x[offset:offset + nv])
            grad[offset:offset + nv] -= w * x[:nv]
        else:
            m_off = -offset
            nv = n - m_off
            if nv <= 0:
                continue
            grad[m_off:m_off + nv] += w * (1.0 - x[:nv])
            grad[:nv] -= w * x[m_off:m_off + nv]
    return grad * dx


def con_sum_h(x):
    return np.sum(x) - _G_TGT


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def binary_snap(h, n):
    target = n / 2.0
    h_bin = (h > 0.5).astype(float)
    diff = target - np.sum(h_bin)
    if diff > 0:
        zeros = np.where(h_bin == 0)[0]
        zeros_sorted = zeros[np.argsort(-h[zeros])]
        for i in range(int(diff)):
            if i < len(zeros_sorted):
                h_bin[zeros_sorted[i]] = 1.0
    elif diff < 0:
        ones = np.where(h_bin == 1)[0]
        ones_sorted = ones[np.argsort(h[ones])]
        for i in range(int(-diff)):
            if i < len(ones_sorted):
                h_bin[ones_sorted[i]] = 0.0
    return h_bin


def local_swap_search(h_bin, n, max_rounds=200, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    c5_best = compute_c5(h_bin)
    h_best = h_bin.copy()
    ones = np.where(h_best == 1)[0]
    zeros = np.where(h_best == 0)[0]
    no_improve = 0
    for rnd in range(max_rounds):
        if remaining_time() < 5.0:
            break
        found = False
        n_try = min(80, len(ones) * len(zeros))
        for _ in range(n_try):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h_trial = h_best.copy()
            h_trial[i] = 0.0
            h_trial[j] = 1.0
            c5_t = compute_c5(h_trial)
            if c5_t < c5_best - 1e-13:
                c5_best = c5_t
                h_best = h_trial.copy()
                ones = np.where(h_best == 1)[0]
                zeros = np.where(h_best == 0)[0]
                found = True
                break
        if not found:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0
    return h_best, c5_best


def active_set_refine(h, n, max_iter=200, top_k=3):
    c5_best = compute_c5(h)
    h_best = h.copy()
    step = 0.04
    patience = 0
    for it in range(max_iter):
        if remaining_time() < 4.0:
            break
        corr = compute_corr(h)
        top_idx = get_top_k(corr, top_k)
        grad = np.zeros(n)
        for k_idx in top_idx:
            offset = int(k_idx) - (n - 1)
            if offset >= 0:
                nv = n - offset
                if nv <= 0:
                    continue
                grad[:nv] += (1.0 - h[offset:offset + nv])
                grad[offset:offset + nv] -= h[:nv]
            else:
                m = -offset
                nv = n - m
                if nv <= 0:
                    continue
                grad[m:m + nv] += (1.0 - h[:nv])
                grad[:nv] -= h[m:m + nv]
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        alpha = step
        improved = False
        for _ in range(8):
            h_trial = h - alpha * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-13:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
            alpha *= 0.5
        if improved:
            patience = 0
            step = min(alpha * 1.1, 0.12)
        else:
            patience += 1
            step *= 0.7
            if patience > 12:
                break
    return h_best, c5_best


def coordinate_descent_focused(h, n, max_sweeps=8):
    c5_best = compute_c5(h)
    h_best = h.copy()
    rng = np.random.RandomState(123)
    for sweep in range(max_sweeps):
        if remaining_time() < 3.0:
            break
        any_imp = False
        frac_idx = np.where((h_best > 0.01) & (h_best < 0.99))[0]
        if len(frac_idx) > 0:
            order = np.concatenate([frac_idx, rng.permutation(n)[len(frac_idx):]])
        else:
            order = rng.permutation(n)
        for i in order:
            if remaining_time() < 2.0:
                break
            old = h_best[i]
            for new_v in [0.0, 1.0]:
                if abs(new_v - old) < 1e-8:
                    continue
                h_trial = h_best.copy()
                h_trial[i] = new_v
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < c5_best - 1e-13:
                    c5_best = c5_t
                    h_best = h_trial.copy()
                    any_imp = True
            for delta in [0.01, -0.01, 0.04, -0.04, 0.08, -0.08]:
                new_v = np.clip(old + delta, 0.0, 1.0)
                if abs(new_v - old) < 1e-8:
                    continue
                h_trial = h_best.copy()
                h_trial[i] = new_v
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < c5_best - 1e-13:
                    c5_best = c5_t
                    h_best = h_trial.copy()
                    any_imp = True
        if not any_imp and sweep >= 2:
            break
    return h_best, c5_best


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def make_golden_ratio(n):
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * phi) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return project_feasible(h, n)


def make_periodic_tm(n):
    half = n // 2
    h_half = np.zeros(half)
    for i in range(half):
        cnt = bin(i).count('1')
        h_half[i] = 1.0 if cnt % 2 == 1 else 0.0
    h = np.tile(h_half, 2)
    return project_feasible(h, n)


def slsqp_z_optimize(z0, n, max_iter=200):
    cons = [{'type': 'eq', 'fun': con_sum_z, 'jac': lambda z: sigmoid(z)}]
    try:
        res = minimize(obj_z, z0, jac=grad_obj_z, method='SLSQP',
                       constraints=cons, options={'maxiter': max_iter,
                                                   'ftol': 1e-14, 'iprint': 0})
        return sigmoid(res.x)
    except Exception:
        return z0  # shouldn't happen since z0 is already h


def slsqp_h_optimize(h0, n, max_iter=200):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum_h}]
    try:
        res = minimize(obj_smooth_h, h0, jac=obj_grad_h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h0


def h_to_z(h):
    h = np.clip(h, 1e-6, 1.0 - 1e-6)
    return np.log(h / (1.0 - h))


def perturb_and_optimize(h_best, n, rng, num_pert=10):
    c5_best = compute_c5(h_best)
    h_best = h_best.copy()
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n

    for p in range(num_pert):
        if remaining_time() < 30.0:
            break
        h_trial = h_best.copy()
        pt = p % 5
        if pt == 0:
            start = rng.randint(0, max(1, n - 16))
            L = rng.randint(4, min(16, n // 8))
            h_trial[start:start + L] = 1.0 - h_trial[start:start + L]
        elif pt == 1:
            s1 = rng.randint(0, n // 4)
            s2 = s1 + rng.randint(n // 6, n // 2)
            L = rng.randint(3, min(8, n // 16))
            a = h_trial[s1:s1 + L].copy()
            h_trial[s1:s1 + L] = h_trial[s2:s2 + L]
            h_trial[s2:s2 + L] = a
        elif pt == 2:
            h_trial += 0.04 * rng.randn(n)
        elif pt == 3:
            freq = rng.randint(2, n // 8)
            phase = rng.rand() * 2 * np.pi
            t = np.arange(n) / float(n)
            h_trial += 0.05 * np.sin(2.0 * np.pi * freq * t + phase)
        else:
            idx = rng.choice(n, size=n // 8, replace=False)
            h_trial[idx] = 1.0 - h_trial[idx]

        h_trial = project_feasible(h_trial, n)
        c5_t = compute_c5(h_trial)
        if c5_t < c5_best - 1e-10:
            h_best = h_trial.copy()
            c5_best = c5_t

        # Quick SLSQP polish from perturbed point
        if remaining_time() < 15.0:
            continue
        _G_BETA = 300.0
        h_new = slsqp_h_optimize(h_trial, n, max_iter=100)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-12:
            c5_best = c5_new
            h_best = h_new.copy()

    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Phase 0: Get starting point
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_h = make_golden_ratio(256)
        parent_n = 256
        parent_c5 = compute_c5(parent_h)
        print("Golden ratio: n={}, C5={:.10f}".format(parent_n, parent_c5))

    # Target resolution
    N = 512

    # Generate multiple candidates
    candidates = []
    # Interpolated parent
    h_interp = project_feasible(interp_to(parent_h, parent_n, N), N)
    candidates.append(('interp', h_interp, compute_c5(h_interp)))
    # Thue-Morse
    h_tm = make_thue_morse(N)
    candidates.append(('tm', h_tm, compute_c5(h_tm)))
    # Periodic TM
    h_ptm = make_periodic_tm(N)
    candidates.append(('ptm', h_ptm, compute_c5(h_ptm)))
    # Golden ratio
    h_gr = make_golden_ratio(N)
    candidates.append(('gr', h_gr, compute_c5(h_gr)))

    # Pick the best candidate
    candidates.sort(key=lambda x: x[2])
    name, h, c5 = candidates[0]
    print("Best candidate: {} with C5={:.10f}".format(name, c5))

    # If parent is much better, use it
    for nm, hc, c5c in candidates:
        print("  {}: C5={:.10f}".format(nm, c5c))

    best_h = h.copy()
    best_c5 = c5
    best_n = N
    _G_N = N
    _G_TGT = N / 2.0
    _G_DX = 2.0 / N

    # Phase 1: Subgradient momentum
    print("\n--- Phase 1: Subgradient momentum ---")
    t0 = time.time()
    h, c5 = subgrad_momentum(h, N, max_iter=400, step0=0.1, momentum=0.85, top_k=5)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    print("  Time: {:.1f}s, C5={:.10f}".format(time.time() - t0, c5))

    # Phase 2: SLSQP in z-space (sigmoid parameterization)
    print("\n--- Phase 2: SLSQP (z-space) ---")
    z = h_to_z(h)
    for beta in [30, 100, 300, 800]:
        if remaining_time() < 20.0:
            break
        _G_BETA = float(beta)
        t0 = time.time()
        h_new = slsqp_z_optimize(z, N, max_iter=200)
        h_new = project_feasible(h_new, N)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            z = h_to_z(h)
        print("  beta={}: C5={:.10f} ({:.1f}s)".format(beta, c5_new, time.time() - t0))

    # Phase 3: SLSQP in h-space with analytical gradient
    print("\n--- Phase 3: SLSQP (h-space, analytic grad) ---")
    for beta in [200, 600, 1500]:
        if remaining_time() < 15.0:
            break
        _G_BETA = float(beta)
        t0 = time.time()
        h_new = slsqp_h_optimize(h, N, max_iter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        print("  beta={}: C5={:.10f} ({:.1f}s)".format(beta, c5_new, time.time() - t0))

    # Phase 4: Active set refinement (top-2)
    print("\n--- Phase 4: Active set (top-2) ---")
    h, c5 = active_set_refine(h, N, max_iter=250, top_k=2)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    print("  C5={:.10f}".format(c5))

    # Phase 5: Binary snap + local search
    print("\n--- Phase 5: Binary snap + swap search ---")
    h_bin = binary_snap(h, N)
    c5_bin = compute_c5(h_bin)
    print("  Binary: C5={:.10f}".format(c5_bin))
    h_bin, c5_bin = local_swap_search(h_bin, N, max_rounds=300, rng=rng)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    print("  After swaps: C5={:.10f}".format(c5_bin))

    # Continue from the better of binary and continuous
    if c5_bin < compute_c5(h):
        h = h_bin.copy()

    # Re-optimize continuously from current best
    h, c5 = subgrad_momentum(h, N, max_iter=200, step0=0.06, top_k=3)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    _G_BETA = 500.0
    h_new = slsqp_h_optimize(h, N, max_iter=150)
    c5_new = compute_c5(h_new)
    if c5_new < best_c5:
        best_c5 = c5_new
        best_h = h_new.copy()
        h = h_new

    print("  After re-optimization: C5={:.10f}".format(best_c5))

    # Phase 6: Perturbation exploration
    print("\n--- Phase 6: Perturbation exploration ---")
    h, c5 = perturb_and_optimize(best_h, N, rng, num_pert=10)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    print("  C5={:.10f}".format(c5))

    # Phase 7: Try n=1024
    if remaining_time() > 150.0:
        print("\n--- Phase 7: Higher resolution n=1024 ---")
        N2 = 1024
        h2 = project_feasible(interp_to(best_h, best_n, N2), N2)
        _G_N = N2
        _G_TGT = N2 / 2.0
        _G_DX = 2.0 / N2
        c5_2 = compute_c5(h2)
        print("  Interpolated: C5={:.10f}".format(c5_2))

        h2, c5_2 = subgrad_momentum(h2, N2, max_iter=250, step0=0.07, top_k=8)
        print("  After subgrad: C5={:.10f}".format(c5_2))

        z2 = h_to_z(h2)
        for beta in [50, 200, 600]:
            if remaining_time() < 25.0:
                break
            _G_BETA = float(beta)
            h2_new = slsqp_z_optimize(z2, N2, max_iter=150)
            h2_new = project_feasible(h2_new, N2)
            c5_2_new = compute_c5(h2_new)
            if c5_2_new < c5_2:
                c5_2 = c5_2_new
                h2 = h2_new
                z2 = h_to_z(h2)
            print("  beta={}: C5={:.10f}".format(beta, c5_2_new))

        # Active set at n=1024
        h2, c5_2 = active_set_refine(h2, N2, max_iter=150, top_k=3)
        print("  After active set: C5={:.10f}".format(c5_2))

        if c5_2 < best_c5 - 1e-10:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = N2
            print("  New best! C5={:.10f}".format(c5_2))
        else:
            print("  No improvement at n=1024")

        _G_N = best_n
        _G_TGT = best_n / 2.0
        _G_DX = 2.0 / best_n

    # Phase 8: Final coordinate descent
    if remaining_time() > 15.0:
        print("\n--- Phase 8: Coordinate descent ---")
        h = best_h.copy()
        h, c5 = coordinate_descent_focused(h, best_n, max_sweeps=10)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        print("  C5={:.10f}".format(c5))

    # Phase 9: Final high-beta polish
    if remaining_time() > 10.0:
        print("\n--- Phase 9: Final polish ---")
        _G_N = best_n
        _G_TGT = best_n / 2.0
        _G_DX = 2.0 / best_n
        h = best_h.copy()
        for beta in [2000, 5000, 10000]:
            if remaining_time() < 6.0:
                break
            _G_BETA = float(beta)
            h_new = slsqp_h_optimize(h, best_n, max_iter=250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
            print("  beta={}: C5={:.10f}".format(beta, c5_new))

    # Phase 10: One more active set pass
    if remaining_time() > 8.0:
        h = best_h.copy()
        h, c5 = active_set_refine(h, best_n, max_iter=100, top_k=2)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        print("\n  After final active set: C5={:.10f}".format(c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
