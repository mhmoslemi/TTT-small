**Key insight for fast local search**:

4. Among all hot shifts, find a swap that reduces the total "excess" (sum of C(d) for active shifts).
5. Accept if max(C) decreases or stays the same but some other metric improves.

        corr[n:] += d
        corr[:n - 1] += d[::-1]
        new_max = float(np.max(corr))

        if new_max < cur_max - 1e-12:
            h[p] = 0.0
            h[q] = 1.0
            cur_max = new_max
            stall = 0
            T = max(0.05, T * 0.9997)
            if cur_max * 2.0 / n < best_c5:
                best_c5 = cur_max * 2.0 / n
                best_h = h.copy()
        elif abs(new_max - cur_max) < 1e-12:
            if rng.rand() < 0.03:
                h[p] = 0.0
                h[q] = 1.0
                stall = 0
            else:
                corr[n:] -= d
                corr[:n - 1] -= d[::-1]
                stall += 1
        else:
            delta = new_max - cur_max
            if rng.rand() < np.exp(-delta / max(T, 0.01)):
                h[p] = 0.0
                h[q] = 1.0
                cur_max = new_max
            else:
                corr[n:] -= d
                corr[:n - 1] -= d[::-1]
            stall += 1
            T = max(0.05, T * 0.999)
            if T <= 0.05:
                T = 0.5

        if stall > 400:
            for _ in range(3):
                ones2 = np.where(h == 1)[0]
                zeros2 = np.where(h == 0)[0]
                pp = int(ones2[rng.randint(len(ones2))])
                qq = int(zeros2[rng.randint(len(zeros2))])
                dd = swap_delta_vec(h, pp, qq, n)
                corr[n:] += dd
                corr[:n - 1] += dd[::-1]
                h[pp] = 0.0
                h[qq] = 1.0
            cur_max = float(np.max(corr))
            T = 0.8
            stall = 0

    h = (h > 0.5).astype(np.float64)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h
    return best_h, best_c5

def targeted_binary_search(h0, n, tlim, seed):
    """Focus swaps on the worst shift."""
    rng = np.random.RandomState(seed)
    h = (h0 > 0.5).astype(np.float64).copy()
    s = int(np.sum(h))
    if s > n // 2:
        oi = np.where(h == 1)[0]
        h[rng.choice(oi, s - n // 2, replace=False)] = 0
    elif s < n // 2:
        zi = np.where(h == 0)[0]
        h[rng.choice(zi, n // 2 - s, replace=False)] = 1

    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * 2.0 / n
    best_h = h.copy()
    start = time.time()
    T = 0.5
    stall = 0

    while time.time() - start < tlim and remaining() > 5:
        worst_idx = int(np.argmax(corr))
        worst_d = worst_idx - (n - 1)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if rng.rand() < 0.6 and abs(worst_d) > 0 and abs(worst_d) < n:
            # Targeted: find a contributing pair at worst shift
            if worst_d > 0:
                valid_i = np.arange(0, n - worst_d)
                contrib = valid_i[(h[valid_i] == 1) & (h[valid_i + worst_d] == 0)]
            else:
                dd = -worst_d
                valid_i = np.arange(dd, n)
                contrib = valid_i[(h[valid_i] == 1) & (h[valid_i - dd] == 0)]

            if len(contrib) > 0:
                p = int(contrib[rng.randint(len(contrib))])
                if worst_d > 0:
                    partner = p + worst_d
                else:
                    partner = p - worst_d
                # Pick q != partner from zeros
                other_zeros = zeros[zeros != partner]
                if len(other_zeros) > 0:
                    q = int(other_zeros[rng.randint(len(other_zeros))])
                else:
                    q = int(zeros[rng.randint(len(zeros))])
            else:
                p = int(ones[rng.randint(len(ones))])
                q = int(zeros[rng.randint(len(zeros))])
        else:
            p = int(ones[rng.randint(len(ones))])
            q = int(zeros[rng.randint(len(zeros))])

        d = swap_delta_vec(h, p, q, n)
        corr[n:] += d
        corr[:n - 1] += d[::-1]
        new_max = float(np.max(corr))

        if new_max < cur_max - 1e-12:
            h[p] = 0.0
            h[q] = 1.0
            cur_max = new_max
            stall = 0
            T = max(0.05, T * 0.9997)
            if cur_max * 2.0 / n < best_c5:
                best_c5 = cur_max * 2.0 / n
                best_h = h.copy()
        elif abs(new_max - cur_max) < 1e-12:
            if rng.rand() < 0.05:
                h[p] = 0.0
                h[q] = 1.0
                stall = 0
            else:
                corr[n:] -= d
                corr[:n - 1] -= d[::-1]
                stall += 1
        else:
            delta = new_max - cur_max
            if rng.rand() < np.exp(-delta / max(T, 0.01)):
                h[p] = 0.0
                h[q] = 1.0
                cur_max = new_max
            else:
                corr[n:] -= d
                corr[:n - 1] -= d[::-1]
            stall += 1
            T = max(0.03, T * 0.998)
            if T <= 0.03:
                T = 0.5

        if stall > 500:
            np.random.RandomState(seed + stall)
            for _ in range(5):
                ones2 = np.where(h == 1)[0]
                zeros2 = np.where(h == 0)[0]
                pp = int(ones2[rng.randint(len(ones2))])
                qq = int(zeros2[rng.randint(len(zeros2))])
                dd = swap_delta_vec(h, pp, qq, n)
                corr[n:] += dd
                corr[:n - 1] += dd[::-1]
                h[pp] = 0.0
                h[qq] = 1.0
            cur_max = float(np.max(corr))
            T = 0.8
            stall = 0

    h = (h > 0.5).astype(np.float64)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h
    return best_h, best_c5

def smooth_obj(x):
    n = _N_G
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA_G
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val

def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        if npairs > 0:
            g[off:off + npairs] += (1.0 - x[:npairs])
            g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        if npairs > 0:
            g[:npairs] += (1.0 - x[ao:ao + npairs])
            g[ao:ao + npairs] -= x[:npairs]
    return g

def smooth_grad(x):
    n = _N_G
    dx_val = 2.0 / n
    beta = _BETA_G
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-14:
            g += w[k] * corr_grad_per_shift(x, n, k)
    return g * dx_val

def sum_con(x):
    return np.sum(x) - _TGT_G

def sum_con_jac(x):
    return np.ones(_N_G)

def slsqp_smooth(h, n, beta, maxiter=300):
    global _N_G, _BETA_G, _TGT_G
    _N_G = n
    _TGT_G = n / 2.0
    _BETA_G = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pair_coord_descent(h, n, max_passes=200, time_limit=50, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    start = time.time()
    best_c5 = compute_c5(h)
    corr = compute_corr(h)
    nc = 2 * n - 1

    for pass_num in range(max_passes):
        if time.time() - start > time_limit:
            break
        if remaining() < 10:
            break
        order = np.arange(n)
        for idx in range(n):
            i = int(order[idx])
            j = (i + n // 2) % n
            if i == j:
                j = (i + 1) % n
            s = h[i] + h[j]
            lo_t = max(0.0, s - 1.0)
            hi_t = min(1.0, s)
            if lo_t >= hi_t - 1e-14:
                continue
            sl_i = np.zeros(nc)
            sl_j = np.zeros(nc)
            for d in range(1, n):
                v = 0.0
                if d <= n - 1 - i:
                    v += h[i + d] - 1.0
                if d <= i:
                    v += h[i - d]
                sl_i[n - 1 + d] = v
                sl_i[n - 1 - d] = v
                v2 = 0.0
                if d <= n - 1 - j:
                    v2 += h[j + d] - 1.0
                if d <= j:
                    v2 += h[j - d]
                sl_j[n - 1 + d] = v2
                sl_j[n - 1 - d] = v2
            diff_sl = sl_i - sl_j
            intercepts = corr - h[i] * diff_sl
            for _ in range(40):
                lo_t2 = lo_t
                hi_t2 = hi_t
                mid = (lo_t2 + hi_t2) / 2.0
                vals = diff_sl * mid + intercepts
                max_val = vals.max()
                active = vals >= max_val - 1e-13
                if active.any():
                    smax = diff_sl[active].max()
                    smin = diff_sl[active].min()
                    if smax < -1e-14:
                        lo_t2 = mid
                    elif smin > 1e-14:
                        hi_t2 = mid
                    else:
                        break
                else:
                    break
            t_opt = (lo_t2 + hi_t2) / 2.0
            if abs(t_opt - h[i]) < 1e-14:
                continue
            delta = t_opt - h[i]
            corr += delta * diff_sl
            h[i] = t_opt
            h[j] = s - t_opt
        h = project_feasible(h, n)
        corr = compute_corr(h)
        c5_new = float(np.max(corr) * 2.0 / n)
        if c5_new < best_c5:
            best_c5 = c5_new
    return h, best_c5

def greedy_construct(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    for _ in range(n // 2):
        best_pos = -1
        best_score = float('inf')
        candidates = np.where(h == 0)[0]
        for p in candidates:
            h[p] = 1
            sc = np.max(compute_corr(h))
            h[p] = 0
            if sc < best_score:
                best_score = sc
                best_pos = p
        if best_pos >= 0:
            h[best_pos] = 1
        else:
            h[candidates[0]] = 1
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 312
        parent_h = np.zeros(parent_n)
        idx = np.arange(0, parent_n, 2)
        parent_h[idx] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Binary local search on parent n
    n0 = best_n
    h_bin = (best_h > 0.5).astype(float)
    s = int(np.sum(h_bin))
    if s != n0 // 2:
        if s > n0 // 2:
            oi = np.where(h_bin == 1)[0]
            h_bin[rng.choice(oi, s - n0 // 2, replace=False)] = 0
        else:
            zi = np.where(h_bin == 0)[0]
            h_bin[rng.choice(zi, n0 // 2 - s, replace=False)] = 1

    t1 = min(200, max(30, remaining() - 400))
    h_b1, c5_b1 = fast_binary_search(h_bin, n0, t1, seed)
    if c5_b1 < best_c5:
        best_c5 = c5_b1
        best_h = h_b1.copy()
        best_n = n0

    # Targeted search
    if remaining() > 50:
        t2 = min(150, max(20, remaining() - 300))
        h_b2, c5_b2 = targeted_binary_search(best_h, best_n, t2, seed + 100)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()
            best_n = best_n

    # Phase 2: SLSQP continuous refinement on best binary
    for beta in [500, 2000, 8000, 20000]:
        if remaining() < 15:
            break
        h_s = slsqp_smooth(best_h, best_n, beta, maxiter=300)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 3: Try other n values
    n_tries = [n0 - 4, n0 - 2, n0 + 2, n0 + 4, n0 + 8, n0 + 16,
               n0 - 8, n0 - 16, 256, 320, 384, 448, 512]
    n_tries = sorted(set([nn for nn in n_tries if 128 <= nn <= 600 and nn % 2 == 0 and nn != best_n]))

    for n_t in n_tries:
        if remaining() < 60:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        h_bin_t = (h_feas > 0.5).astype(float)
        st = int(np.sum(h_bin_t))
        if st > n_t // 2:
            oi = np.where(h_bin_t == 1)[0]
            h_bin_t[rng.choice(oi, st - n_t // 2, replace=False)] = 0
        elif st < n_t // 2:
            zi = np.where(h_bin_t == 0)[0]
            h_bin_t[rng.choice(zi, n_t // 2 - st, replace=False)] = 1

        t_alloc = min(80, max(15, remaining() - 40))
        h_bt, c5_bt = fast_binary_search(h_bin_t, n_t, t_alloc, seed + n_t)
        if c5_bt < best_c5:
            best_c5 = c5_bt
            best_h = h_bt.copy()
            best_n = n_t

        if remaining() < 20:
            break
        # SLSQP on this n
        h_st = slsqp_smooth(h_bt, n_t, 3000, maxiter=200)
        c5_st = compute_c5(h_st)
        if c5_st < best_c5:
            best_c5 = c5_st
            best_h = h_st.copy()
            best_n = n_t

    # Phase 4: More binary search on best
    if remaining() > 30:
        t4 = min(200, max(20, remaining() - 30))
        h_b4, c5_b4 = targeted_binary_search(best_h, best_n, t4, seed + 500)
        if c5_b4 < best_c5:
            best_c5 = c5_b4
            best_h = h_b4.copy()
            best_n = best_n

    # Phase 5: SLSQP polish
    for beta in [5000, 15000, 40000]:
        if remaining() < 10:
            break
        h_s = slsqp_smooth(best_h, best_n, beta, maxiter=400)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 6: Pair coordinate descent
    if remaining() > 20:
        h_pc, c5_pc = pair_coord_descent(best_h, best_n, max_passes=100,
                                         time_limit=min(40, max(10, remaining() - 15)), rng=rng)
        if c5_pc < best_c5:
            best_c5 = c5_pc
            best_h = h_pc.copy()

    # Phase 7: Final binary search
    if remaining() > 15:
        t7 = max(5, remaining() - 10)
        h_b7, c5_b7 = fast_binary_search(best_h, best_n, t7, seed + 1000)
        if c5_b7 < best_c5:
            best_c5 = c5_b7
            best_h = h_b7.copy()

    # Final SLSQP
    if remaining() > 8:
        h_s = slsqp_smooth(best_h, best_n, 30000, maxiter=300)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Validate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
