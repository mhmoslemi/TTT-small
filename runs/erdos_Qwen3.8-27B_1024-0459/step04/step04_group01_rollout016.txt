def project_feasible(h, n):

    # Bisection to find offset
    lo, hi = -2.0, 2.0
    for _ in range(60):

        h_trial = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_trial)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine adjustment
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)

```python
def compute_effects(h, n):
    """Precompute effect vectors for each position.
    effect[i][lag] = change in corr[lag] when h[i] goes from 1 to 0.
    corr = np.correlate(h, 1-h, 'full')
    corr[lag_idx] = sum_n h[n+lag] * (1-h[n]) where lag = lag_idx - (n-1)
    """
    nc = 2 * n - 1
    effects = np.zeros((n, nc))
    for i in range(n):
        # h[i] appears as h[n+lag] when n = i - lag, valid if 0 <= i-lag <= n-1
        # Contribution before: h[i]*(1-h[i-lag]) = 1*(1-h[i-lag]) for h[i]=1
        # After flip to 0: 0
        # So change: -(1-h[i-lag])
        # h[i] appears in (1-h[n]) when n=i, valid if 0 <= i+lag <= n-1
        # Contribution before: h[i+lag]*(1-h[i]) = h[i+lag]*0 = 0
        # After flip to 1... wait, we're flipping 1 to 0
        # After: h[i+lag]*(1-0) = h[i+lag]
        # So change: +h[i+lag]
        
        # For lag range: lag_idx from 0 to nc-1, lag = lag_idx - (n-1)
        tags = np.arange(nc) - (n - 1)  # lag values from -(n-1) to (n-1)
        
        # First part: -(1-h[i-lag]) where 0 <= i-lag <= n-1
        valid1 = (i - tags >= 0) & (i - tags < n)
        effects[i, valid1] = -(1.0 - h[i - tags[valid1]])
        
        # Second part: +h[i+lag] where 0 <= i+lag <= n-1
        valid2 = (i + tags >= 0) & (i + tags < n)
        effects[i, valid2] += h[i + tags[valid2]]
    
    return effects
```
