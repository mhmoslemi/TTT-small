def compute_c5(h):

    dx = 2.0 / n

def compute_corr(h):

def grad_corr_k(h, k_idx, n):
    """Gradient of the k-th correlation value w.r.t. h."""
    offset = k_idx - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        m = n - offset
        g[offset:offset+m] += (1.0 - h[:m])
        g[:m] -= h[offset:offset+m]
    else:
        a = -offset
        m = n - a
        g[:m] += (h[a:a+m] - 1.0)
        g[a:a+m] += h[:m]
    return g

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):

        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def active_set_grad(h, n, time_limit=60.0, top_k_init=5, verbose=False):
    """Active-set projected gradient for minimax correlation."""
    h = h.copy()
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    start = time.time()

    patience = 0

        active = sorted_idx[:top_k]

def softmax_slsqp(h, n, betas=[100, 500, 2000, 8000, 30000], maxiter_per=300):
    """SLSQP with softmax-smoothed max, increasing beta."""
    dx_v = 2.0 / n
    h = project_feasible(h, n)
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_v
    
    def obj_grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * dx_v
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off+np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off+np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao+np_])
                g[ao:ao+np_] -= wk * x[:np_]
        return g
    
    def constraint_fun(x):
        return np.sum(x) - n / 2.0
    
    def constraint_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    
    best = h.copy()
    best_c5 = compute_c5(h)
    
    for beta in betas:
        if remaining() < 5:
            break
        try:
            res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per, 'ftol': 1e-16, 'iprint': 0})
            if res.success:
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best = h_new.copy()
                    h = h_new.copy()
        except Exception:
            pass
    
    return best, best_c5

def pair_swap_optimize(h, n, time_limit=30.0, rng=None):

    if rng is None:

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def grad_corr_k(h, k_idx, n):
    offset = k_idx - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        m = n - offset
        g[offset:offset + m] += (1.0 - h[:m])
        g[:m] -= h[offset:offset + m]
    else:
        a = -offset
        m = n - a
        g[:m] += (h[a:a + m] - 1.0)
        g[a:a + m] += h[:m]
    return g


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)


def active_set_grad(h, n, time_limit=60.0, top_k_init=5, top_k_max=25, seed=0):

    best_h = h.copy()

    base_alpha = 0.1
    rng = np.random.RandomState(seed)

    while time.time() - start < time_limit and remaining() > 5:

        active_max = corr[active[0]]
        active_min = corr[active[-1]]

        if active_max - active_min < 1e-9 * max(1.0, abs(active_max)) and top_k < top_k_max:
            top_k = min(top_k + 2, top_k_max, 2 * n - 1)
            continue

        g = np.zeros(n)
        for ki in active:

        g /= top_k
        g -= np.mean(g)

def softmax_slsqp(h, n, betas=None, maxiter_per=250):
    if betas is None:
        betas = [100, 500, 2000, 8000, 30000, 80000]
    dx_v = 2.0 / n
    h = project_feasible(h, n)

    def _obj(x, beta):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_v

    def _obj_grad(x, beta):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * dx_v
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g

    def _cons_fun(x):
        return np.sum(x) - n / 2.0

    def _cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]

    best = h.copy()
    best_c5 = compute_c5(h)

    for beta in betas:
        if remaining() < 5:
            break
        try:
            res = minimize(lambda x: _obj(x, beta), h,
                           jac=lambda x: _obj_grad(x, beta),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per, 'ftol': 1e-16, 'iprint': 0})
            if res.success:
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best = h_new.copy()
                    h = h_new.copy()
        except Exception:
            pass

    return best, best_c5


def pair_swap_optimize(h, n, time_limit=30.0, seed=42):
    rng = np.random.RandomState(seed)
    h = project_feasible(h, n)
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    start = time.time()
    dx_v = 2.0 / n

    while time.time() - start < time_limit and remaining() > 5:
        k_max = int(np.argmax(corr))
        g = grad_corr_k(h, k_max, n)

        # Find pairs to swap: move mass from high g to low g
        # g[i] > g[j] means: decreasing h[i] and increasing h[j] reduces corr
        # Effect on corr[k_max]: -alpha*g[i] + alpha*g[j] = alpha*(g[j]-g[i]) < 0 if g[i]>g[j]
        n_try = 30
        best_strength = 0.0
        best_pair = None

        for _ in range(n_try):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            if g[i] > g[j]:
                strength = g[i] - g[j]
                if strength > best_strength:
                    best_strength = strength
                    best_pair = (i, j)

        if best_pair is None:
            # Also check a few top correlations
            for alt_k in np.argsort(-corr)[1:4]:
                g_alt = grad_corr_k(h, alt_k, n)
                for _ in range(n_try):
                    i = rng.randint(n)
                    j = rng.randint(n)
                    if i == j:
                        continue
                    if g_alt[i] > g_alt[j]:
                        strength = g_alt[i] - g_alt[j]
                        if strength > best_strength:
                            best_strength = strength
                            best_pair = (i, j)
                if best_pair is not None:
                    break

        if best_pair is None:
            continue

        i, j = best_pair
        max_alpha = min(h[i], 1.0 - h[j])
        if max_alpha < 1e-12:
            continue

        improved = False
        for frac in [0.3, 0.1, 0.05, 0.5, 0.02, 0.01]:
            alpha = frac * max_alpha
            h[i] -= alpha
            h[j] += alpha
            corr_new = compute_corr(h)
            new_max = float(np.max(corr_new))
            if new_max < cur_max - 1e-14:
                cur_max = new_max
                corr = corr_new
                improved = True
                if cur_max < best_max - 1e-14:
                    best_max = cur_max
                    best_h = h.copy()
                break
            else:
                h[i] += alpha
                h[j] -= alpha

        if not improved:
            # Try small random pair
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j and h[i] > 0.01 and h[j] < 0.99:
                alpha = 0.01
                h[i] -= alpha
                h[j] += alpha
                corr_new = compute_corr(h)
                new_max = float(np.max(corr_new))
                if new_max < cur_max - 1e-14:
                    cur_max = new_max
                    corr = corr_new
                    if cur_max < best_max - 1e-14:
                        best_max = cur_max
                        best_h = h.copy()
                else:
                    h[i] += alpha
                    h[j] -= alpha

    return best_h, best_max * dx_v


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def multi_start_optimize(h, n, time_limit=60.0, n_starts=5, seed=0):
    """Try multiple random perturbations and optimize each."""
    rng = np.random.RandomState(seed)
    best_h = project_feasible(h, n)
    best_c5 = compute_c5(best_h)
    per_start = max(3.0, time_limit / n_starts)

    for s in range(n_starts):
        if remaining() < per_start + 5:
            break
        if s == 0:
            h_s = best_h.copy()
        else:
            # Perturb: snap to binary then optimize
            h_s = (best_h > 0.5).astype(float)
            n_ones = int(np.sum(h_s))
            if n_ones != n // 2:
                idx_sorted = np.argsort(-best_h)
                h_s = np.zeros(n)
                h_s[idx_sorted[:n // 2]] = 1.0
            # Add random perturbation
            pert = rng.randn(n) * 0.1
            h_s = project_feasible(h_s + pert, n)

        h_opt, c5_opt = active_set_grad(h_s, n, time_limit=per_start,
                                         top_k_init=5, top_k_max=20, seed=seed + s)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    return best_h, best_c5

            best_h = h6.copy()
            h = h6.copy()

    if remaining() > 30:
        h7, c5_7 = softmax_slsqp(h, n, betas=[5000, 20000, 60000, 150000], maxiter_per=250)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()
            h = h7.copy()

    if remaining() > 25:
        h8, c5_8 = pair_swap_optimize(h, n, time_limit=min(20, remaining() * 0.3), seed=seed + 300)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()
            h = h8.copy()

    # Phase 7: Another multi-start round
    if remaining() > 40:
        h9, c5_9 = multi_start_optimize(h, n, time_limit=min(35, remaining() * 0.5),
                                         n_starts=5, seed=seed + 500)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()
            h = h9.copy()

    # Phase 8: Final active-set push
    if remaining() > 20:
        h10, c5_10 = active_set_grad(h, n, time_limit=min(15, remaining() * 0.4),
                                      top_k_init=10, top_k_max=30, seed=seed + 600)
        if c5_10 < best_c5:
            best_c5 = c5_10
            best_h = h10.copy()
            h = h10.copy()

    # Phase 9: Final SLSQP polish with very high beta
    if remaining() > 10:
        h11, c5_11 = softmax_slsqp(h, n, betas=[30000, 80000, 200000], maxiter_per=200)
        if c5_11 < best_c5:
            best_c5 = c5_11
            best_h = h11.copy()
            h = h11.copy()

    # Phase 10: Last few pair swaps
    if remaining() > 8:
        h12, c5_12 = pair_swap_optimize(h, n, time_limit=min(5, remaining() * 0.4), seed=seed + 700)
        if c5_12 < best_c5:
            best_c5 = c5_12
            best_h = h12.copy()

    # Final: ensure feasibility and validate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
