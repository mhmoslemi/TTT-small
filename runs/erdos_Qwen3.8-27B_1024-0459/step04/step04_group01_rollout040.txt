with the convention that h_old[x] = 0 if x is out of [0, n-1].

- The "second position" contribution at i-d=a (i=a+d): was h_old[a+d]*(1-h_old[a]) = h_old[a+d]*0 = 0. Now h_old[a+d]*1 = h_old[a+d]. Change: +h_old[a+d].
- Similarly for b.

- We want to find a swap (a, b) where h[a]=1, h[b]=0, such that max(corr + Δcorr) < M

def compute_c5(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_trial = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_trial)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(30):
        diff = target - np.sum(h)

        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)

            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)

def to_binary(h, n, rng):
    h = np.array(h, dtype=float).copy()
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h_bin == 1)[0]
        # Remove the ones with smallest h values
        order = np.argsort(h[ones_idx])
        remove = ones_idx[order[:n_ones - target]]
        h_bin[remove] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        # Add zeros with largest h values
        order = np.argsort(-h[zeros_idx])
        add = zeros_idx[order[:target - n_ones]]
        h_bin[add] = 1.0
    return h_bin

def swap_delta(h, a, b, n):
    """Compute change in corr array when h[a]:1->0 and h[b]:0->1. Returns O(n) array."""
    nc = 2 * n - 1
    delta = np.zeros(nc)
    # For each shift d (represented by index k = n-1+d):
    # corr[k] = sum_{valid i} h[i]*(1-h[i-d])
    # Change from h[a]:1->0: -(1-h[a-d]) at shift d (first position) + h[a+d] at shift -d (second position)
    # Change from h[b]:0->1: +(1-h[b-d]) at shift d (first position) - h[b+d] at shift -d (second position)
    
    # For shift d, a affects corr at:
    #   first position: shift d_a where a-d_a = some valid index, i.e., d_a = a - j for valid j
    #   Actually: corr[n-1+d] has term h[a]*(1-h[a-d]) if 0<=a-d<n
    #   And term h[a+d]*(1-h[a]) if 0<=a+d<n (this is the "second" position contribution)
    
    # Let me just compute directly using indexing:
    # corr[n-1+d] = sum_i h[i]*(1-h[i-d]) for valid i
    # If h[a] goes 1->0:
    #   For shift d where a is first: need 0<=a<n and 0<=a-d<n, contribution was (1-h[a-d]), now 0
    #   For shift d where a is second: i-d=a => i=a+d, need 0<=a+d<n and 0<=a<n
    #     contribution was h[a+d]*(1-1)=0, now h[a+d]*(1-0)=h[a+d]
    
    # For shift d (d can be negative):
    # a is "first" in corr[n-1+d] iff max(0,d) <= a <= min(n-1, n-1+d) AND 0<=a-d<n
    #   Actually simpler: a is first iff 0<=a<n (always) and 0<=a-d<n
    #   The shift d must satisfy: a-d >= 0 and a-d < n, i.e., a-n < d <= a
    #   And a must be in range [max(0,d), min(n-1,n-1+d)] which is automatic if 0<=a<n and 0<=a-d<n
    
    # Let me just iterate over all d values efficiently:
    # For a: first position affects shifts d where 0<=a-d<n, i.e., a-n+1 <= d <= a
    #         contribution change: -(1-h[a-d])
    # For a: second position affects shifts d where 0<=a+d<n, i.e., -a <= d < n-a
    #         contribution change: +h[a+d]
    # For b: first position affects shifts d where 0<=b-d<n, i.e., b-n+1 <= d <= b
    #         contribution change: +(1-h[b-d])
    # For b: second position affects shifts d where 0<=b+d<n, i.e., -b <= d < n-b
    #         contribution change: -h[b+d]
    
    # Convert to index: k = n-1+d
    # a first: d in [a-n+1, a], k in [1, a+n-1]
    d_min_a = a - n + 1
    d_max_a = a
    for d in range(d_min_a, d_max_a + 1):
        k = n - 1 + d
        j = a - d
        delta[k] -= (1.0 - h[j])
    
    # a second: d in [-a, n-a-1], k in [n-1-a, 2n-2-a]
    d_min_a2 = -a
    d_max_a2 = n - a - 1
    for d in range(d_min_a2, d_max_a2 + 1):
        k = n - 1 + d
        j = a + d
        delta[k] += h[j]
    
    # b first: d in [b-n+1, b], k in [1, b+n-1]
    d_min_b = b - n + 1
    d_max_b = b
    for d in range(d_min_b, d_max_b + 1):
        k = n - 1 + d
        j = b - d
        delta[k] += (1.0 - h[j])
    
    # b second: d in [-b, n-b-1], k in [n-1-b, 2n-2-b]
    d_min_b2 = -b
    d_max_b2 = n - b - 1
    for d in range(d_min_b2, d_max_b2 + 1):
        k = n - 1 + d
        j = b + d
        delta[k] -= h[j]
    
    return delta


def swap_delta_fast(h, a, b, n):
    """Faster vectorized version of swap_delta."""
    nc = 2 * n - 1
    delta = np.zeros(nc)
    
    # a first: d in [a-n+1, a], k = n-1+d in [1, a+n-1], j = a-d
    # j = a - (k - n + 1) = a + n - 1 - k
    k_lo = max(1, n - 1 + (a - n + 1))  # = max(1, a)
    k_hi = min(nc - 1, n - 1 + a)  # = a + n - 1
    ks = np.arange(k_lo, k_hi + 1)
    js = a + n - 1 - ks
    delta[ks] -= (1.0 - h[js])
    
    # a second: d in [-a, n-a-1], k = n-1+d in [n-1-a, 2n-2-a], j = a+d = a + k - n + 1
    k_lo2 = max(0, n - 1 - a)
    k_hi2 = min(nc - 1, 2 * n - 2 - a)
    ks2 = np.arange(k_lo2, k_hi2 + 1)
    js2 = a + ks2 - n + 1
    delta[ks2] += h[js2]
    
    # b first: d in [b-n+1, b], k in [1, b+n-1], j = b-d = b + n - 1 - k
    k_lo3 = max(1, b)
    k_hi3 = min(nc - 1, b + n - 1)
    ks3 = np.arange(k_lo3, k_hi3 + 1)
    js3 = b + n - 1 - ks3
    delta[ks3] += (1.0 - h[js3])
    
    # b second: d in [-b, n-b-1], k in [n-1-b, 2n-2-b], j = b+d = b + k - n + 1
    k_lo4 = max(0, n - 1 - b)
    k_hi4 = min(nc - 1, 2 * n - 2 - b)
    ks4 = np.arange(k_lo4, k_hi4 + 1)
    js4 = b + ks4 - n + 1
    delta[ks4] -= h[js4]
    
    return delta

def pair_coord_descent(h, n, max_passes=200, time_limit=120, rng=None):
    """Pair coordinate descent for continuous h."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    start = time.time()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1
    
    # Precompute slope masks for efficiency
    # For variable i, the slope at corr index k is:
    #   +1 if i is "first position" at shift d=k-(n-1) (i.e., 0<=i-d<n)
    #   -1 if i is "second position" (i.e., 0<=i+d<n) with coefficient (1-h[...]) etc.
    # Actually let's just compute slopes on the fly for the pair.
    
    def coord_slope_i(h, i):
        """Slope of corr[k] w.r.t. h[i] for all k."""
        nc = 2 * n - 1
        slopes = np.zeros(nc)
        d = np.arange(-(n-1), n)  # all shifts
        # i is first position: 0<=i-d<n => d in [i-n+1, i]
        mask_first = (d >= i - n + 1) & (d <= i)
        j_first = i - d
        mask_first_valid = mask_first & (j_first >= 0) & (j_first < n)
        # Contribution: (1 - h[j_first])
        contrib_first = np.zeros(nc)
        contrib_first[mask_first_valid] = 1.0 - h[j_first[mask_first_valid]]
        # i is second position: 0<=i+d<n => d in [-i, n-i-1]
        mask_second = (d >= -i) & (d <= n - i - 1)
        j_second = i + d
        mask_second_valid = mask_second & (j_second >= 0) & (j_second < n)
        # Contribution: -h[j_second]
        contrib_second = np.zeros(nc)
        contrib_second[mask_second_valid] = -h[j_second[mask_second_valid]]
        return contrib_first + contrib_second
    
    corr = compute_corr(h)
    
    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        if remaining() < 10:
            break
        
        # Random pairing
        order = rng.permutation(n)
        for idx in range(0, n - 1, 2):
            i = int(order[idx])
            j = int(order[idx + 1])
            s = h[i] + h[j]
            lo_t = max(0.0, s - 1.0)
            hi_t = min(1.0, s)
            if lo_t >= hi_t - 1e-14:
                continue
            
            sl_i = coord_slope_i(h, i)
            sl_j = coord_slope_i(h, j)

            # corr_new[k] = corr[k] - h[i]*sl_i[k] + t*sl_i[k] - h[j]*sl_j[k] + (s-t)*sl_j[k]
            # = corr[k] + t*(sl_i[k]-sl_j[k]) + (s-h[i]-h[j])*0 + ... 
            # = corr[k] + t*diff_sl[k] + (s - h[i] - h[j])*(sl_i[k]+sl_j[k])/... 
            # Wait: corr_new[k] = corr[k] + (t-h[i])*sl_i[k] + ((s-t)-h[j])*sl_j[k]
            #   = corr[k] + (t-h[i])*sl_i[k] + (s-t-h[j])*sl_j[k]
            #   = corr[k] + t*(sl_i[k]-sl_j[k]) + (s-h[j]-h[i])*sl_j[k]
            # Since s = h[i]+h[j], s-h[j]-h[i] = 0
            # So corr_new[k] = corr[k] + t*diff_sl[k]

            t_opt = (lo_t + hi_t) / 2.0
            
            if abs(t_opt - h[i]) < 1e-14:
                continue
            
            delta_t = t_opt - h[i]
            corr += delta_t * diff_sl
            h[i] = t_opt
            h[j] = s - t_opt
        
        h = project_feasible(h, n)
        corr = compute_corr(h)
        c5_new = float(np.max(corr) * 2.0 / n)
        if c5_new < best_c5:
            best_c5 = c5_new
    
    return h, best_c5


def group_cd(h, n, group_size=3, max_iters=5000, time_limit=100, rng=None):
    """Group coordinate descent: optimize groups of variables via LP."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    start = time.time()
    best_c5 = compute_c5(h)
    nc = 2 * n - 1
    dx = 2.0 / n
    
    # Precompute for each variable i, the slope vector
    # We'll compute on demand for the group
    
    def get_group_slopes(indices):
        """Get slope matrix for a group of indices. Returns (m x nc) array."""
        m = len(indices)
        slopes = np.zeros((m, nc))
        d = np.arange(-(n-1), n)
        for mi, i in enumerate(indices):
            mask_first = (d >= i - n + 1) & (d <= i)
            j_first = i - d
            mask_first_valid = mask_first & (j_first >= 0) & (j_first < n)
            contrib_first = np.zeros(nc)
            contrib_first[mask_first_valid] = 1.0 - h[j_first[mask_first_valid]]
            mask_second = (d >= -i) & (d <= n - i - 1)
            j_second = i + d
            mask_second_valid = mask_second & (j_second >= 0) & (j_second < n)
            contrib_second = np.zeros(nc)
            contrib_second[mask_second_valid] = -h[j_second[mask_second_valid]]
            slopes[mi] = contrib_first + contrib_second
        return slopes
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining() < 10:
            break
        
        # Pick a random group
        indices = rng.choice(n, group_size, replace=False)
        indices = sorted(indices.tolist())
        
        slope_mat = get_group_slopes(indices)  # (m x nc)
        # corr_new[k] = corr[k] + sum_m (h_new[indices[m]] - h[indices[m]]) * slope_mat[m,k]
        # Let x_m = h_new[indices[m]], constraint: 0<=x_m<=1, sum(x) = sum(h[indices])
        # Objective: minimize t s.t. corr[k] + sum_m (x_m - h[indices[m]]) * slope_mat[m,k] <= t for all k
        #           0 <= x_m <= 1, sum(x_m) = sum(h[indices])
        
        corr = compute_corr(h)
        current_vals = h[indices].copy()
        target_sum = np.sum(current_vals)
        m = len(indices)
        
        # LP: min t
        # s.t. sum_m slope_mat[m,k] * x_m - t <= -corr[k] + sum_m slope_mat[m,k]*current_vals[m] for all k
        #      0 <= x_m <= 1
        #      sum_m x_m = target_sum
        #      t free (but bounded)
        
        # Variables: [x_0, ..., x_{m-1}, t]
        # Objective: [0, 0, ..., 0, 1]
        c_obj = np.zeros(m + 1)
        c_obj[-1] = 1.0
        
        # Inequality constraints: A_ub @ x <= b_ub
        # For each k: sum_m slope_mat[m,k]*x_m - t <= -corr[k] + sum_m slope_mat[m,k]*current_vals[m]
        # In matrix form: [slope_mat[:,k], -1] @ var <= rhs_k
        
        # This is nc constraints, which might be too many for linprog speed
        # Use only the top 50 worst shifts
        sorted_k = np.argsort(-corr)[:80]
        
        A_ub = np.zeros((len(sorted_k), m + 1))
        b_ub = np.zeros(len(sorted_k))
        for ki, k in enumerate(sorted_k):
            A_ub[ki, :m] = slope_mat[:, k]
            A_ub[ki, m] = -1.0
            b_ub[ki] = -corr[k] + np.dot(slope_mat[:, k], current_vals)
        
        # Bounds: 0 <= x_m <= 1, t free (use large bound)
        bounds = [(0.0, 1.0)] * m + [(-10.0, 10.0)]
        
        # Equality: sum(x) = target_sum
        A_eq = np.zeros((1, m + 1))
        A_eq[0, :m] = 1.0
        b_eq = np.array([target_sum])
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.status == 0:
                new_vals = res.x[:m]
                # Check improvement
                new_corr = corr + slope_mat.T @ (new_vals - current_vals)
                new_t = res.x[m]
                # Verify against all corr (not just top 80)
                full_max = float(np.max(new_corr))
                if full_max < float(np.max(corr)) - 1e-10:
                    h[indices] = new_vals
                    best_c5 = full_max * dx
        except Exception:
            pass
        
        if it % 50 == 0:
            c5_check = compute_c5(h)
            if c5_check < best_c5:
                best_c5 = c5_check
    
    return h, best_c5


def slsqp_smooth(h, n, beta, maxiter=300):
    """SLSQP with log-sum-exp smoothing."""
    dx_val = 2.0 / n
    _N_local = n
    
    def smooth_obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -60, 60)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_val
    
    def smooth_grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -60, 60)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        # Gradient of corr w.r.t. x
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] > 1e-14:
                d = k - (n - 1)
                if d >= 0:
                    npairs = n - d
                    if npairs > 0:
                        g[d:d + npairs] += (1.0 - x[:npairs])
                        g[:npairs] -= x[d:d + npairs]
                else:
                    ao = -d
                    npairs = n - ao
                    if npairs > 0:
                        g[:npairs] += (1.0 - x[ao:ao + npairs])
                        g[ao:ao + npairs] -= x[:npairs]
        return g * dx_val
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def multi_start_binary(base_h, n, n_starts=5, time_per_start=30, rng=None):
    """Try multiple random perturbations of a binary solution."""
    if rng is None:
        rng = np.random.RandomState(42)
    best_h = base_h.copy()
    best_c5 = compute_c5(base_h)
    
    for s in range(n_starts):
        if remaining() < 15:
            break
        h_try = base_h.copy()
        # Random perturbation: flip k random pairs
        k_flips = rng.randint(2, 8)
        ones = np.where(h_try == 1)[0]
        zeros = np.where(h_try == 0)[0]
        a_sel = rng.choice(ones, min(k_flips, len(ones)), replace=False)
        b_sel = rng.choice(zeros, min(k_flips, len(zeros)), replace=False)
        h_try[a_sel] = 0.0
        h_try[b_sel] = 1.0
        
        # Local search from perturbed state
        h_res, c5_res = binary_local_search(h_try, n, time_limit=time_per_start, rng=rng)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
    
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
    except Exception:
        parent_n = 312
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0

    best_h = parent_h.copy()
    best_c5 = compute_c5(parent_h)
    best_n = parent_n

    # Phase 1: Binary local search on parent resolution
    h_bin = to_binary(parent_h, parent_n, rng)
    h_bin, c5_bin = binary_local_search(h_bin, parent_n, time_limit=120, rng=rng)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
        best_n = parent_n

    # Phase 2: Continuous refinement on best
    h_cont = best_h.astype(float)
    for beta in [500, 2000, 8000, 20000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h_cont, best_n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cont = h_new.copy()

    # Phase 3: Pair coordinate descent
    if remaining() > 30:
        h_pc, c5_pc = pair_coord_descent(h_cont, best_n, max_passes=100,
                                          time_limit=min(80, remaining() - 20), rng=rng)
        if c5_pc < best_c5:
            best_c5 = c5_pc
            best_h = h_pc.copy()
            h_cont = h_pc.copy()

    # Phase 4: Binary local search again (from continuous, rounded)
    if remaining() > 30:
        h_bin2 = to_binary(h_cont, best_n, rng)
        h_bin2, c5_bin2 = binary_local_search(h_bin2, best_n, time_limit=min(60, remaining() - 20), rng=rng)
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()
            best_n = best_n

    # Phase 5: Try different n values
    n_candidates = []
    for offset in [8, 16, 32, 48, 64, 96, 128]:
        n_candidates.append(best_n + offset)
    for offset in [8, 16, 32]:
        n_candidates.append(max(160, best_n - offset))
    # Also try some round numbers
    for nn in [320, 352, 384, 448, 480, 512, 576, 640]:
        n_candidates.append(nn)
    
    n_candidates = sorted(set(n for n in n_candidates if 160 <= n <= 800 and n % 2 == 0 and n != best_n))
    
    for n_t in n_candidates[:6]:  # Limit number of tries
        if remaining() < 40:
            break
        
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        
        # Binary local search on new n
        h_bin_t = to_binary(h_feas, n_t, rng)
        h_bin_t, c5_t = binary_local_search(h_bin_t, n_t, time_limit=min(50, remaining() - 20), rng=rng)
        
        # Continuous refinement
        if remaining() > 15:
            h_cont_t = slsqp_smooth(h_bin_t.astype(float), n_t, 3000, maxiter=200)
            c5_cont_t = compute_c5(h_cont_t)
            if c5_cont_t < c5_t:
                c5_t = c5_cont_t
                h_bin_t = h_cont_t
        
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_bin_t.copy()
            best_n = n_t

    # Phase 6: Final intensive refinement on best
    n = best_n
    h = best_h.copy()
    
    # Binary local search (focused)
    if remaining() > 40:
        h_bin_f = to_binary(h, n, rng)
        h_bin_f, c5_f = binary_local_search(h_bin_f, n, time_limit=min(100, remaining() - 30), rng=rng)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_bin_f.copy()
            h = h_bin_f.copy()
    
    # Group CD
    if remaining() > 30:
        h_g, c5_g = group_cd(h.astype(float), n, group_size=3, max_iters=2000,
                             time_limit=min(60, remaining() - 20), rng=rng)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
            h = h_g.copy()
    
    # Final SLSQP
    if remaining() > 8:
        h_final = slsqp_smooth(h, n, 15000, maxiter=300)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    # Final binary polish
    if remaining() > 15:
        h_bp = to_binary(best_h, best_n, rng)
        h_bp, c5_bp = binary_local_search(h_bp, best_n, time_limit=min(30, remaining() - 10), rng=rng)
        if c5_bp < best_c5:
            best_c5 = c5_bp
            best_h = h_bp.copy()

    # Validate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    else:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
