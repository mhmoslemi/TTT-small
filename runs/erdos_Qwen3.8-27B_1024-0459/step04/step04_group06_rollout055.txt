                best_c5 = new_c5
    return best, best_c5

def targeted_sa(h, n, rng, n_rounds=8):
    best = h.copy()
    best_c5 = compute_c5(best)
    for r in range(n_rounds):
        if tl() < 4.0:
            break
        corr = compute_corr(best)
        top_k = min(5, len(corr))
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
        h_t = best.copy()
        for k_idx in top_idx:
            lag = int(k_idx) - (n - 1)
            noise = np.zeros(n)
            for i in range(n):
                grad = 0.0
                if lag >= 0:
                    if i >= lag and i - lag < n:
                        grad += (1.0 - h_t[i - lag])
                    if i + lag < n:
                        grad -= h_t[i + lag]
                else:
                    m = -lag
                    if i - m >= 0:
                        grad -= h_t[i - m]
                    if i + m < n:
                        grad += (1.0 - h_t[i + m])
                noise[i] = -0.3 * grad
            h_t += noise * rng.uniform(0.3, 0.8)
        h_t += rng.normal(0, 0.01, n)
        h_t = project_feasible(h_t, n)
        c5_t = compute_c5(h_t)
        if c5_t < best_c5 - 1e-15:
            best = h_t.copy()
            best_c5 = c5_t
    return best, best_c5

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h

def make_random_bin(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def local_binary_search(h, n, rng, max_flips=500):
    best = h.copy()
    best_c5 = compute_c5(best)
    for _ in range(max_flips):
        if tl() < 2.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if tl() < 1.0:
                break
            h_t = best.copy()
            h_t[i] = 1.0 - best[i]
            h_t = project_feasible(h_t, n)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5 - 1e-15:
                best = h_t
                best_c5 = c5_t
                improved = True
        if not improved:
            break
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _G_T0, _G_BUDGET
    _G_T0 = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        ph = np.array(initial_h_values, dtype=float).copy()
        pn = len(ph)
    except Exception:
        pn = 256
        ph = make_thue_morse(pn)

    candidates = [ph.copy()]
    candidates.append(make_thue_morse(pn))
    candidates.append(make_random_bin(rng, pn))

    for ci, cand in enumerate(candidates):
        if tl() < 60.0:
            break
        h = project_feasible(cand, pn)
        c5 = compute_c5(h)
        print("Cand {}: n={}, C5={:.10f}".format(ci, pn, c5))

        h, c5 = slsqp_polish(h, pn, betas=[100, 500, 2000], iters=150)
        print("  After SLSQP: C5={:.10f}".format(c5))

        h, c5 = sa_optimize(h, pn, rng, T0=0.003, Tend=0.0003, n_steps=30000)
        print("  After SA-1: C5={:.10f}".format(c5))

        h, c5 = slsqp_polish(h, pn, betas=[500, 2000, 8000], iters=200)
        print("  After SLSQP-2: C5={:.10f}".format(c5))

        h, c5 = targeted_sa(h, pn, rng, n_rounds=8)
        print("  After targeted: C5={:.10f}".format(c5))

        h, c5 = sa_optimize(h, pn, rng, T0=0.001, Tend=0.0001, n_steps=30000)
        print("  After SA-2: C5={:.10f}".format(c5))

        h, c5 = slsqp_polish(h, pn, betas=[2000, 8000, 30000], iters=300)
        print("  After SLSQP-3: C5={:.10f}".format(c5))

        if ci == 0 and tl() > 120.0:
            h, c5 = local_binary_search(h, pn, rng, max_flips=300)
            print("  After binary: C5={:.10f}".format(c5))
        if ci == 0 and tl() > 60.0:
            h, c5 = sa_optimize(h, pn, rng, T0=0.0005, Tend=0.00005, n_steps=30000)
            print("  After SA-3: C5={:.10f}".format(c5))
            h, c5 = slsqp_polish(h, pn, betas=[5000, 20000, 50000], iters=300)
            print("  After SLSQP-4: C5={:.10f}".format(c5))

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = pn

    if tl() > 40.0:
        for n_try in [192, 320]:
            if tl() < 25.0:
                break
            h_t = project_feasible(interp_to(best_h, best_n, n_try), n_try)
            c5_t = compute_c5(h_t)
            print("Res n={}: init C5={:.10f}".format(n_try, c5_t))
            h_t, c5_t = slsqp_polish(h_t, n_try, betas=[200, 1000, 4000], iters=150)
            h_t, c5_t = sa_optimize(h_t, n_try, rng, T0=0.002, Tend=0.0002, n_steps=20000)
            h_t, c5_t = slsqp_polish(h_t, n_try, betas=[2000, 10000], iters=200)
            print("Res n={}: final C5={:.10f}".format(n_try, c5_t))
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_try

    if tl() > 15.0:
        best_h, best_c5 = slsqp_polish(best_h, best_n,
                                       betas=[1000, 5000, 20000, 80000], iters=400)
        print("Final SLSQP: C5={:.10f}".format(best_c5))

    if tl() > 8.0:
        best_h, best_c5 = sa_optimize(best_h, best_n, rng,
                                      T0=0.0003, Tend=0.00002, n_steps=15000)
        print("Final SA: C5={:.10f}".format(best_c5))

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    try:
        ev = evaluate_erdos_solution(best_h)
        if ev is not None:
            final_c5 = min(final_c5, float(ev))
    except Exception:
        pass

    print("RESULT: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
