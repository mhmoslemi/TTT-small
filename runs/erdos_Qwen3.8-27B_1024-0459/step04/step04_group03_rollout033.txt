        top3 = np.argsort(corr)[::-1][:3]
        active = np.unique(np.concatenate([active, top3]))
        active = np.sort(active)
        n_active = len(active)
        
        def obj(x):
            t = x[n]
            h_part = x[:n]
            c = compute_corr(h_part)
            return np.max(c[active] * dx - t)
        
        def obj2(x):
            return x[n]
        
        def ineq(x):
            h_part = x[:n]
            t = x[n]
            c = compute_corr(h_part)
            return t - c[active] * dx
        
        def eq(x):
            return np.sum(x[:n]) - n / 2.0
        
        x0 = np.concatenate([h, [corr_max * dx + 1e-6]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': ineq},
            {'type': 'eq', 'fun': eq}
        ]
        try:
            res = minimize(obj2, x0, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': max_slsqp_iter, 'ftol': 1e-14, 'iprint': 0})
            if res.x is not None:
                h_new = project_feasible(res.x[:n], n)
                c5_new = compute_c5(h_new)
                c5_old = compute_c5(h)
                if c5_new < c5_old - 1e-15:
                    h = h_new
                else:
                    if rnd > 3:
                        break
        except Exception:
            break
    return h

def multi_start_binary(n, rng, max_starts=5):
    best_h = None
    best_c5 = float('inf')
    for s in range(max_starts):
        if remaining_time() < 30:
            break
        h_rand = np.zeros(n)
        positions = rng.choice(n, size=n//2, replace=False)
        h_rand[positions] = 1.0
        h_rand = project_feasible(h_rand, n)
        c5_r = compute_c5(h_rand)
        
        h_opt = coordinate_descent(h_rand, n, max_rounds=30, max_active=25)
        c5_opt = compute_c5(h_opt)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S

    rng = np.random.RandomState(seed)

        best_n = 256

    target_ns = [512, 1024, 2048]
    for tn in target_ns:
        if remaining_time() < 60:
            break
        h_cur = interp_to(best_h, best_n, tn)
        h_cur = project_feasible(h_cur, tn)
        c5_cur = compute_c5(h_cur)
        
        # Coordinate descent with gradually tighter active set
        for active_tol in [0.0, 0.001, 0.005, 0.01]:
            if remaining_time() < 30:
                break
            h_cd = coordinate_descent(h_cur, tn, max_rounds=100,
                                      active_tol=active_tol, max_active=50)
            c5_cd = compute_c5(h_cd)
            if c5_cd < c5_cur:
                c5_cur = c5_cd
                h_cur = h_cd
        
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = tn
    
    # Phase 2: Minimax SLSQP on best solution
    if remaining_time() > 60 and best_n <= 1024:
        h_mm = slsqp_minimax_round(best_h, best_n, max_rounds=8, max_slsqp_iter=80)
        c5_mm = compute_c5(h_mm)
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm.copy()
    
    # Phase 3: Binary projection and bit-flip optimization
    if remaining_time() > 80:
        for thr in [0.45, 0.5, 0.55]:
            if remaining_time() < 30:
                break
            h_bin = binary_projection(best_h, best_n, threshold=thr)
            c5_bin = compute_c5(h_bin)
            h_bf = bit_flip_optimize(h_bin, best_n, max_rounds=20, window=30)
            c5_bf = compute_c5(h_bf)
            if c5_bf < best_c5:
                best_c5 = c5_bf
                best_h = h_bf.copy()
    
    # Phase 4: Multi-resolution coordinate descent sweep
    if remaining_time() > 100:
        for tn in [384, 448, 512, 640, 768, 1024, 1280]:
            if remaining_time() < 40:
                break
            h_interp = interp_to(best_h, best_n, tn)
            h_interp = project_feasible(h_interp, tn)
            c5_int = compute_c5(h_interp)
            
            if c5_int < best_c5 + 0.005:
                h_cd = coordinate_descent(h_interp, tn, max_rounds=80,
                                          active_tol=0.0, max_active=40)
                c5_cd = compute_c5(h_cd)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                    best_n = tn
    
    # Phase 5: Subgradient refinement
    if remaining_time() > 50 and best_n <= 1024:
        h_sg = subgradient_step(best_h, best_n, step_size=0.01,
                                active_tol=0.01, max_iter=200)
        c5_sg = compute_c5(h_sg)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    if remaining_time() > 30:
        h_final = coordinate_descent(best_h, best_n, max_rounds=100,

        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

        h_up = project_feasible(h_up, len(h_up))

        h_up_cd = coordinate_descent(h_up, tu, max_rounds=50,
                                     active_tol=0.0, max_active=40)
        c5_up = compute_c5(h_up_cd)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up_cd.copy()
            best_n = tu
    
    # Phase 8: Soft binarize + optimize
    if remaining_time() > 30:
        for eps in [0.15, 0.2, 0.25]:
            if remaining_time() < 15:
                break
            h_soft = soft_binarize(best_h, best_n, epsilon=eps)
            c5_soft = compute_c5(h_soft)
            h_ref = coordinate_descent(h_soft, best_n, max_rounds=40,
                                       active_tol=0.0, max_active=30)
            c5_ref = compute_c5(h_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()

                                 active_tol=0.0, max_active=80)

    final_c5 = compute_c5(best_h)

        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
