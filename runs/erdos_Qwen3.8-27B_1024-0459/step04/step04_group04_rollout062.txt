    g = np.zeros(n)

        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]

def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g

        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 360
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: SLSQP at parent n with increasing beta
    n0 = best_n
    h_work = best_h.copy()
    for beta in [200, 1000, 5000, 20000, 100000]:
        if remaining() < 15:
            break
        h_new = slsqp_smooth(h_work, n0, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n0
            h_work = h_new.copy()

    # Phase 2: Projected gradient
    if remaining() > 30:
        h_pg, c5_pg = projected_gradient(best_h, best_n, max_iters=400, top_k=10)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 3: Binary SA at multiple n values
    n_bin_list = [340, 350, 360, 370, 380, 390, 400, 420, 440]
    n_bin_list = [n for n in n_bin_list if n % 2 == 0]

    for n_t in n_bin_list:
        if remaining() < 20:
            break
        # Try from parent interpolation and diverse patterns
        h_parent_bin = None
        if best_n != n_t:
            h_interp = interp_to(best_h, best_n, n_t)
            h_parent_bin = (h_interp > 0.5).astype(float)
            cnt = int(np.sum(h_parent_bin))
            tgt = n_t // 2
            if cnt > tgt:
                idx = np.where(h_parent_bin == 1)[0]
                h_parent_bin[idx[rng.choice(len(idx), cnt - tgt, replace=False)]] = 0
            elif cnt < tgt:
                idx = np.where(h_parent_bin == 0)[0]
                h_parent_bin[idx[rng.choice(len(idx), tgt - cnt, replace=False)]] = 1
        else:
            h_parent_bin = (best_h > 0.5).astype(float)
            cnt = int(np.sum(h_parent_bin))
            tgt = n_t // 2
            if cnt > tgt:
                idx = np.where(h_parent_bin == 1)[0]
                h_parent_bin[idx[rng.choice(len(idx), cnt - tgt, replace=False)]] = 0
            elif cnt < tgt:
                idx = np.where(h_parent_bin == 0)[0]
                h_parent_bin[idx[rng.choice(len(idx), tgt - cnt, replace=False)]] = 1

        t_per = min(25, max(8, remaining() / (len(n_bin_list) + 2)))
        starts = [h_parent_bin]
        for pat in [2, 3, 6]:
            if remaining() > t_per:
                starts.append(make_diverse_binary(n_t, seed + pat * 111 + n_t, pat))

        for h_start in starts:
            if remaining() < 10:
                break
            h_sa, c5_sa = sa_binary(h_start, n_t, time_limit=end_of := min(t_per, remaining() - 5), seed=seed + n_t * 13)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_t

    # Phase 4: SLSQP at best_n and neighbors (continuous)
    n_cont_list = sorted(set([best_n - 20, best_n, best_n + 20, 360, 380, 400]))
    n_cont_list = [n for n in n_cont_list if n % 2 == 0 and 100 <= n <= 500]

    for n_t in n_cont_list:
        if remaining() < 20:
            break
        if n_t == best_n:
            h_init = best_h.copy()
        else:
            h_init = project_feasible(interp_to(best_h, best_n, n_t), n_t)
        for beta in [5000, 30000]:
            if remaining() < 10:
                break
            h_new = slsqp_smooth(h_init, n_t, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_t
                h_init = h_new.copy()

    # Phase 5: Coordinate descent
    if remaining() > 40:
        n_cd = best_n
        h_cd, c5_cd = coord_descent(best_h, n_cd, max_sweeps=80)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 6: More SLSQP with very high beta
    for beta in [50000, 200000, 500000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(best_h, best_n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 7: Final projected gradient
    if remaining() > 10:
        h_pg, c5_pg = projected_gradient(best_h, best_n, max_iters=300, top_k=8)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 8: Final binary SA push
    if remaining() > 20:
        n =
        best_n
        h_bin = (best_h > 0.5).astype(float)
        cnt = int(np.sum(h_bin))
        tgt = n // 2
        if cnt > tgt:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[rng.choice(len(idx), cnt - tgt, replace=False)]] = 0
        elif cnt < tgt:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[rng.choice(len(idx), tgt - cnt, replace=False)]] = 1
        for sa_s in [seed + 50000, seed + 60000]:
            if remaining() < 10:
                break
            h_sa, c5_sa = sa_binary(h_bin, n, time_limit=min(20, remaining() - 5), seed=sa_s)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()

    # Final polish: one more SLSQP + gradient
    if remaining() > 8:
        h_new = slsqp_smooth(best_h, best_n, 100000, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    if remaining() > 5:
        h_pg, c5_pg = projected_gradient(best_h, best_n, max_iters=200, top_k=5)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
