import time

_t0 = 0.0
_budget = 1300.0

def remaining():
    return max(0, _budget - (time.time() - _t0) - 3.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0, 1)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-14) & (h < 1 - 1e-14)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            j = np.argmin(h) if diff > 0 else np.argmax(h)
            h[j] = np.clip(h[j] + diff, 0, 1)
        h = np.clip(h, 0, 1)
    return h

def smooth_obj_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax for stability
    max_c = corr.max()
    shifted = beta * (corr - max_c)
    exps = np.exp(shifted)
    wsum = exps.sum()
    w = exps / wsum
    val = (max_c + np.log(wsum) / beta) * dx
    
    grad = np.zeros(n)
    # For each k in correlation:
    ncorr = 2 * n - 1
    for k in range(ncorr):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        if d == 0:
            grad += wk * (1 - 2 * h)
        else:
            ad = abs(d)
            # d corr[k]/d h[j] = (1-h[j-ad])[j>=ad] - h[j+ad][j+ad<n]
            # (1-h[j-ad]) for j >= ad:
            if ad < n:
                grad[ad:] += wk * (1 - h[:n-ad])
                # -h[j+ad] for j+ad < n, i.e., j < n-ad
                grad[:n-ad] -= wk * h[ad:]
    grad *= dx

The quadratic terms h[i](1-h[j]) can be written as h[i] - h[i]h[j]. If we introduce variables x_{ij} = h[i]h[j], we get a QP with quadratic constraints, which can be relaxed to an SDP. But for n=640, the SDP would have a 640×640 matrix, which is too large.
