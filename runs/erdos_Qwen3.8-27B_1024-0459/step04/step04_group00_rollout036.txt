   b. Identifies the top-m (m grows over time)
   c. Computes the "balanced" direction: the direction in L2 sense that most reduces the max while trying not to increase others
   d. Uses a careful line search
   e. Projects onto feasible set after each step

The "balanced direction" can be computed as:
- Let c_max = max(corr)
- Let S = {k : corr[k] = c_max} (or within epsilon)
- The subgradient of max is uniform average over S
- But we want to go further: we want a direction that not only reduces c_max but also "flattens" the top
- This is like solving: min t s.t. c_k - <grad_k, d> ≤ t for k in S, ||d|| = 1
- Which is a small QP (size |S|)

Actually, this is getting complex. Let me simplify:

**Approach: Iteratively reduce the correlation at the worst shift while being careful not to increase others too much.**

Algorithm:
1. Find k* = argmax corr[k]
2. Compute g* = grad_corr_k(h, k*, n)
3. For the proposed direction -g*, compute how it affects all other top correlations
4. If it would significantly increase others, blend with their "opposite" gradient
5. Line search + project

This is essentially a **constraint-aware gradient method**.

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:off + np_] += (1.0 - h[:np_])
        g[:np_] -= h[off:off + np_]
    else:
        ao = -off
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g

def smooth_max_and_grad(h, n, beta):
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    sh = corr - m

    ex = np.exp(cl)

        return v

        _, g = smooth_max_and_grad(x, n, beta)
        return g

        return np.ones(n)

def custom_grad_descent(h, n, max_iters=1000, top_m=10):
    """
    Custom gradient descent with adaptive step.
    Uses exact subgradient of max (uniform over active set) 
    plus a "flatten" term for the top-m correlations.
    """
    h = project_feasible(h, n).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iters):

            break
        
        corr = compute_corr(h)
        c_max = np.max(corr)
        
        # Find active set (within tolerance of max)
        tol = max(1e-10, 0.001 * c_max)
        active = np.where(corr >= c_max - tol)[0]
        if len(active) > top_m:
            active = np.argsort(-corr)[:top_m]
        elif len(active) == 0:
            active = np.argsort(-corr)[:top_m]
        
        # Subgradient of max: uniform average over active
        g = np.zeros(n)
        for k in active:
            g += grad_corr_k(h, k, n)
        g /= len(active)
        
        # Flatten term: for top-m, reduce spread
        top_idx = np.argsort(-corr)[:top_m]

        c_mean_top = np.mean(c_top)
        g_flat = np.zeros(n)
        for i, k in enumerate(top_idx):
            weight = c_top[i] - c_mean_top  # positive for above-mean, negative for below
            g_flat += weight * grad_corr_k(h, k, n)
        
        # Combined direction: -g - alpha * g_flat
        alpha = 0.5
        d = -g - alpha * g_flat
        
        # Project direction onto feasible space
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        d[~mask] = 0.0
        # Orthogonalize against the equality constraint
        free = mask
        n_free = np.count_nonzero(free)
        if n_free > 0:
            d[free] -= np.mean(d[free])
        
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        
        # Adaptive line search with backtracking
        step = 0.05
        improved = False

            c5_new = compute_c5(h_new)

                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                break

                break

                h_new = project_feasible(h + pert, n)
                # Don't update best, just move
                if compute_c5(h_new) < best_c5 + 0.001:
                    h = h_new

    return np.interp(x_new, xp, src_h)

            best_c5 = c5_pb
        best_h = h_pb if c5_pb <= compute_c5(best_h) else best_h

    # Phase 10: Final subgradient with larger active set
    if remaining_time() > 10:
        h_sg, c5_sg = subgrad_descent(best_h, n, max_iters=500, top_m=25, alpha_flat=0.5)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 11: Final SLSQP polish
    if remaining_time() > 5:
        h_new = slsqp_opt(best_h, n, 500000, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 12: Final pairwise polish
    if remaining_time() > 5:
        h_pb, c5_pb = pairwise_balance(best_h, n, max_iters=300, m_pairs=15)
        if c5_pb < best_c5:
            best_c5 = c5_pb
            best_h = h_pb.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
