        s = np.sum(ex)

But wait, I'm being wasteful. When I flip (i,j), I can compute the change in correlation more efficiently. The correlation at lag k (where k is the index in the full correlation, ranging from 0 to 2n-2) is:
- off = k - (n-1)
- If off >= 0: corr[k] = sum_{i=0}^{n-1-off} h[i] * (1-h[i+off])
- If off < 0: corr[k] = sum_{i=0}^{n-1+off} h[i] * (1-h[i-off]) = sum_{i=-off}^{n-1} h[i] * (1-h[i+off])

When I change h[i]: 1→0 and h[j]: 0→1, the change in corr[k] is:
- For the h[i] term: at lag k, i contributes h[i] * (1-h[i+off]) if i is in range. Changing h[i] from 1 to 0 removes (1-h[i+off]) from corr[k] (i.e., corr[k] decreases by (1-h[i+off])).
  Also, i appears as the "second" argument in lag k' where i = i' + off', i.e., off' = i - i'. For lag k' = off' + (n-1), the contribution is h[i'] * (1-h[i]). Changing h[i] from 1 to 0 INCREASES corr[k'] by h[i'].
  
- Similarly for h[j]: 0→1, at lag k where j contributes as first argument: adds (1-h[j+off]) to corr[k].
  At lag where j is second argument: h[i'] * (1-h[j]) changes from h[i'] to 0, so corr decreases by h[i'].

import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(h)
            for idx in order[::-1]:
                add = min(1.0 - h[idx], diff)
                h[idx] += add
                diff -= add
                if diff < 1e-12:
                    break
        else:
            order = np.argsort(-h)
            for idx in order:
                rem = min(h[idx], -diff)
                h[idx] -= rem
                diff += rem
                if diff > -1e-12:
                    break
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])
        h_bin[ones[order[:n_ones - target]]] = 0
    elif n_ones < target:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])
        h_bin[zeros[order[:target - n_ones]]] = 1
    return h_bin


def corr_grad_single(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g


def l_bfgsb_opt(h_init, n, beta, maxiter=400):
    dx_val = 2.0 / n
    target = n / 2.0

    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -50, 50)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_val

    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -50, 50)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] > 1e-13:
                g += w[k] * corr_grad_single(x, n, k)
        return g * dx_val

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h_init, jac=grad, method='L-BFGS-B',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h_init.copy()


def slsqp_opt(h_init, n, beta, maxiter=400):
    dx_val = 2.0 / n
    target = n / 2.0

    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -50, 50)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_val

    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -50, 50)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] > 1e-13:
                g += w[k] * corr_grad_single(x, n, k)
        return g * dx_val

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h_init, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h_init.copy()


def topk_gradient_descent(h, n, max_iters=400, top_k=5):
    h = h.copy()
    best_c5 = compute_c5(h)
    dx_val = 2.0 / n
    step_sizes = [0.08, 0.04, 0.02, 0.1, 0.01, 0.15, 0.005]

    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        m = float(np.max(corr))
        if m * dx_val >= best_c5 + 1e-10:
            break
        top_idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k in top_idx:
            g += corr_grad_single(h, n, int(k))
        g /= top_k
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        d = -g / gn
        improved = False
        for step in step_sizes:
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-13:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def binary_sa(h_init, n, time_limit=30, seed=42, T_init=8.0, alpha=0.9988):
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1

    dx_val = 2.0 / n
    cur_max = float(np.max(compute_corr(h)))
    best_max = cur_max
    best_h = h.copy()
    T = T_init
    start = time.time()
    stall = 0
    max_stall = 6000

    while time.time() - start < time_limit and T > 0.03:
        if remaining() < 5:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_max = float(np.max(compute_corr(h)))
        delta = new_max - cur_max
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
        if stall > max_stall:
            T = max(T, 2.5)
            stall = 0
        T *= alpha

    return best_h, best_max * dx_val


def binary_greedy_sa(h_init, n, time_limit=40, seed=42, n_samples=3000):
    """Hybrid: greedily find best flip among samples, with SA escape."""
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1

    dx_val = 2.0 / n
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    start = time.time()
    no_improve = 0

    while time.time() - start < time_limit and remaining() > 5:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        n_samp = min(n_samples, len(ones) * len(zeros))
        best_delta = 0.0
        best_i, best_j = -1, -1
        worst_delta = 0.0
        worst_i, worst_j = -1, -1

        for _ in range(n_samp):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            c_new = compute_c5(h)
            delta = c_new - cur_c5
            h[i], h[j] = 1.0, 0.0
            if delta < best_delta:
                best_delta = delta
                best_i, best_j = i, j
            if delta > worst_delta:
                worst_delta = delta
                worst_i, worst_j = i, j

        if best_i >= 0 and best_delta < -1e-12:
            h[best_i], h[best_j] = 0.0, 1.0
            cur_c5 += best_delta
            no_improve = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # SA escape: accept a random flip with probability
            T = max(0.5, 5.0 * (1.0 - (time.time() - start) / time_limit))
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            cur_c5 = compute_c5(h)
            no_improve += 1
            if no_improve > 20:
                # Revert to best
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
