        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13 and n > 0:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-13:
        for j in range(n):
            if h[j] > 1e-8 and h[j] < 1.0 - 1e-8:
                h[j] = np.clip(h[j] + diff2, 0.0, 1.0)
                break
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def smooth_max_value(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = beta * corr * dx
    x -= np.max(x)
    return np.log(np.sum(np.exp(x))) / beta * dx

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = beta * corr * dx
    x -= np.max(x)
    exp_x = np.exp(x)
    p = exp_x / np.sum(exp_x)
    
    total_p = np.sum(p)
    
    # S1[i] = sum_m p[m] * h[i+m-n+1] (with h=0 out of range)
    # S2[i] = sum_m p[m] * h[i-m+n-1] (with h=0 out of range)
    # Use FFT for efficiency
    size = 4 * n
    P = np.fft.fft(p, size=size)
    H = np.fft.fft(h, size=size)
    
    # S1[i] = sum_m p[m] * h[i+m-n+1]
    # = cross_correlation(p, h)[i-n+1] where xcorr[l] = sum_m p[m]*h[m+l]
    xcorr = np.fft.ifft(np.conj(P) * H).real
    S1 = xcorr[(np.arange(n) - n + 1) % size]
    
    # S2[i] = sum_m p[m] * h[i-m+n-1] = sum_m p[m] * h[(i+n-1)-m]
    # = convolution(p, h)[i+n-1] where conv[k] = sum_m p[m]*h[k-m]
    conv = np.fft.ifft(P * H).real
    S2 = conv[np.arange(n) + n - 1]
    
    grad = dx * (total_p - S1 - S2)
    return grad

def projected_gradient_descent(h_init, n, beta, max_iter=2000, lr_init=0.01):
    h = h_init.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_c5(h)
    
    for it in range(max_iter):
        if remaining_time() < 10:
            break
        if it % 500 == 0 and it > 0:
            val = compute_c5(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()
        
        grad = smooth_max_grad(h, beta)
        # Adaptive learning rate
        lr = lr_init / (1 + 0.01 * it)
        
        # Gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        # Check improvement
        # (Skip expensive C5 check most iterations)
        h = h_new
        
        # Periodic check
        if (it + 1) % 200 == 0:
            val = compute_c5(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()
    
    val = compute_c5(h)
    if val < best_val:
        best_val = val
        best_h = h.copy()
    
    return best_h, best_val

def coordinate_descent_pass(h, n, max_coords=None):
    """One full pass of coordinate descent."""
    h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    nc = len(corr)  # 2n-1
    
    if max_coords is None:
        max_coords = n
    
    improved = 0
    for i in range(n):
        if remaining_time() < 5:
            break
        old_val = h[i]
        
        # For coordinate i, the objective as function of t=h[i]:
        # corr[m] changes by (t - old_val) * (1 - H[i+m-n+1] - H[i-m+n-1])
        # where H[k]=h[k] if 0<=k<n else 0, and the "1" term only applies if i+m-n+1 is valid
        
        # Compute the slope for each lag
        # slope[m] = 1 - H[i+m-n+1] - H[i-m+n-1] (with special handling)
        # base[m] = corr[m] - old_val * slope[m]
        
        # For efficiency, compute the effect:
        # new_corr[m] = corr[m] + (t - old_val) * slope[m]
        # objective(t) = max_m new_corr[m] * dx
        
        # slope[m] for m = 0..2n-2:
        # The term (1 - H[i+m-n+1]) contributes 1 if i+m-n+1 is in [0,n), else 0
        # Wait no: if i+m-n+1 is NOT in [0,n), then H[i+m-n+1]=0 and the original 
        # term h[i]*(1-h[i+m-n+1]) doesn't exist (it's 0), so changing h[i] doesn't affect it.
        # Actually: the term in correllate where j=i is: h[i]*(1-h[i+m-n+1])
        # This term only exists if 0 <= i+m-n+1 < n.
        # If it exists: ∂/∂h[i] = (1-h[i+m-n+1])
        # The term where j+m-n+1=i (j=i-m+n-1): h[j]*(1-h[i])
        # This exists if 0 <= i-m+n-1 < n.
        # If it exists: ∂/∂h[i] = -h[i-m+n-1]
        # Special case: if both j=i and j+m-n+1=i (i.e., i+m-n+1=i → m=n-1):
        #   term is h[i]*(1-h[i]), derivative = 1-2*h[i]
        
        # So slope[m] = (1-H[i+m-n+1])*valid1 - H[i-m+n-1]*valid2
        # with special case m=n-1: slope = 1-2*h[i]
        
        # For the max: new_corr[m] = corr[m] + (t-old_val)*slope[m]
        # We want to find t in [0,1] minimizing max_m new_corr[m]*dx
        # = dx * max_m (corr[m] + (t-old_val)*slope[m])
        # = dx * max_m (base[m] + t*slope[m]) where base[m] = corr[m] - old_val*slope[m]
        
        # This is a convex piecewise-linear function of t.
        # Simple approach: evaluate at t=0, t=1, and at the breakpoints.
        # Breakpoints occur where two lines cross: base[m1]+t*slope[m1] = base[m2]+t*slope[m2]
        # t = (base[m2]-base[m1])/(slope[m1]-slope[m2])
        
        # For efficiency, I'll use a simplified approach:
        # Evaluate the objective at a few candidate t values and pick the best.
        
        # Actually, let's just do a golden section search since the function is convex.
        # But golden section needs ~50 evaluations, each O(n). For n=512, that's ~25K per coordinate.
        # Total per pass: 512 * 25K = 12.8M. That's fine.
        
        # Even simpler: since it's piecewise linear and convex, the minimum is where
        # the subgradient crosses 0. I'll use a simple approach:
        # 1. Compute the objective at t=old_val (current), t=0, t=1
        # 2. If current is best, skip
        # 3. Otherwise, do a few bisection steps
        
        obj_curr = np.max(corr)
        obj_0 = np.max(corr + (0 - old_val) * np.array([0]*nc))  # not helpful directly
        
        # Let me just do a simple approach: try a few values of t and pick the best
        # that also maintains feasibility (we'll project after)
        candidates = [0.0, 1.0, max(0, old_val - 0.1), max(0, old_val + 0.1),
                      max(0, old_val - 0.05), max(0, old_val + 0.05)]
        
        # Also try the "greedy" direction: decrease t if it reduces the max
        # The subgradient at t=old_val is: slope[m*] where m* = argmax(corr)
        # If subgradient > 0, decreasing t helps; if < 0, increasing t helps.
        m_star = int(np.argmax(corr))
        # Compute slope at m_star
        s1 = 0.0
        idx1 = i + m_star - n + 1
        if 0 <= idx1 < n:
            s1 = 1.0 - h[idx1]
        s2 = 0.0
        idx2 = i - m_star + n - 1
        if 0 <= idx2 < n:
            s2 = h[idx2]
        # Special case
        if m_star == n - 1:
            slope_ms = 1.0 - 2.0 * h[i]
        else:
            slope_ms = s1 - s2
        
        if slope_ms > 0:
            candidates.append(max(0.0, old_val - 0.05))
            candidates.append(max(0.0, old_val - 0.02))
        elif slope_ms < 0:
            candidates.append(min(1.0, old_val + 0.05))
            candidates.append(min(1.0, old_val + 0.02))
        
        best_t = old_val
        best_obj = obj_curr
        
        for t in candidates:
            if abs(t - old_val) < 1e-6:
                continue
            # Compute new corr
            new_corr = corr.copy()
            # For each m, new_corr[m] = corr[m] + (t - old_val) * slope[m]
            # Only need to check if max decreases
            # Slope computation:
            # For m where i+m-n+1 is valid: slope contribution from first term = 1-h[i+m-n+1]
            # For m where i-m+n-1 is valid: slope contribution from second term = -h[i-m+n-1]
            
            # Quick check: what's the new max?
            # The lags most likely to be affected are those near the current max
            # For efficiency, just compute the effect on all lags
            # slope[m] = (1-H[i+m-n+1])*valid1 - H[i-m+n-1]*valid2 (non-special)
            # or 1-2*h[i] for m=n-1
            
            # Build slope array
            # This is O(2n) per coordinate, total O(n*2n) = O(2n²) per pass
            # For n=512: 512*1024 ≈ 524K per pass. Fine.
            
            # Actually, for this approach I'll just compute the new corr directly
            # by updating the relevant terms
            new_corr = corr.copy()
            delta = t - old_val
            for m in range(nc):
                sl = 0.0
                idx_a = i + m - n + 1
                idx_b = i - m + n - 1
                if m == n - 1:
                    sl = 1.0 - 2.0 * old_val
                else:
                    if 0 <= idx_a < n:
                        sl += (1.0 - h[idx_a])
                    if 0 <= idx_b < n:
                        sl -= h[idx_b]
                new_corr[m] += delta * sl
            
            new_max = np.max(new_corr)
            if new_max < best_obj - 1e-12:
                best_obj = new_max
                best_t = t
        
        if best_t != old_val:
            h[i] = best_t
            h = project_feasible(h, n)
            # Recompute corr periodically
            if improved % 10 == 9:
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_check = np.max(corr) * dx
                if c5_check < best_c5:
                    best_c5 = c5_check
            improved += 1
        
        # Update corr every 5 iterations to save time
        if improved % 5 == 4:
            corr = np.correlate(h, 1.0 - h, mode='full')
    
    return h, best_c5

def slsqp_smooth(h, n, beta, maxiter=200):
    """SLSQP with smooth max objective."""
    dx = 2.0 / n
    def obj(x):
        hx = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hx, 1.0 - hx, mode='full')
        x_b = beta * corr * dx
        x_b -= np.max(x_b)
        return np.log(np.sum(np.exp(x_b))) / beta * dx
    
    def obj_grad(x):
        hx = np.clip(x, 0.0, 1.0)
        return smooth_max_grad(hx, beta)
    
    def eq_con(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con}]
    
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def slsqp_minimax(h_init, n, max_rounds=8, max_slsqp_iter=150):
    """Minimax SLSQP: introduce variable t, minimize t subject to corr[k]*dx <= t."""
    dx = 2.0 / n
    h = h_init.copy()
    
    for rnd in range(max_rounds):
        if remaining_time() < 8:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        threshold = max_corr * dx - 0.3 * max_corr * dx
        active = np.where(corr * dx >= threshold)[0]
        top3 = np.argsort(corr)[::-1][:3]
        active = np.unique(np.concatenate([active, top3]))
        
        def mm_obj(x):
            return x[n]
        
        def mm_ineq(x):
            hx = x[:n]
            t = x[n]
            c = np.correlate(hx, 1.0 - hx, mode='full')
            return t - dx * c[active]
        
        def mm_eq(x):
            return np.sum(x[:n]) - n / 2.0
        
        t_init = max_corr * dx + 1e-8
        x0 = np.concatenate([h, [t_init]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': mm_ineq},
            {'type': 'eq', 'fun': mm_eq}
        ]
        
        try:
            res = minimize(mm_obj, x0, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_slsqp_iter, 'ftol': 1e-13, 'iprint': 0})
            if res.x is not None:
                h_new = project_feasible(res.x[:n], n)
                c5_new = compute_c5(h_new)
                c5_old = compute_c5(h)
                if c5_new < c5_old - 1e-15:
                    h = h_new
                else:
                    if rnd > 3:
                        break
        except Exception:
            break
    
    return h

def pair_transfer(h, n, max_sweeps=5):
    """Local combinatorial search: transfer mass between pairs of coordinates."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(0)
    
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        any_imp = False
        for i in range(n):
            if remaining_time() < 3:
                break
            hi = best_h[i]
            if hi < 0.05 or hi > 0.95:
                continue
            for j in range(i+1, min(i+15, n)):
                hj = best_h[j]
                if abs(hi - hj) < 0.03:
                    continue
                for frac in [0.3, 0.5, 0.7]:
                    amt = (hi - hj) * frac
                    if abs(amt) < 0.01:
                        continue
                    h_trial = best_h.copy()
                    h_trial[i] = hi - amt
                    h_trial[j] = hj + amt
                    h_trial = project_feasible(h_trial, n)
                    c5_t = compute_c5(h_trial)
                    if c5_t < best_c5 - 1e-14:
                        best_c5 = c5_t
                        best_h = h_trial.copy()
                        hi = best_h[i]
                        any_imp = True
                        break
        if not any_imp:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 256
        best_h = np.full(best_n, 0.5)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)
    
    # Phase 1: Work at parent resolution with PGM + SLSQP
    n_work = best_n
    h_work = best_h.copy()
    
    # PGM with escalating beta
    for beta in [20, 50, 100, 300, 1000, 3000, 10000]:
        if remaining_time() < 20:
            break
        lr = 0.005 / (1 + 0.05 * np.log1p(beta))
        h_work, val = projected_gradient_descent(h_work, n_work, beta, max_iter=1500, lr_init=lr)
        if val < best_c5:
            best_c5 = val
            best_h = h_work.copy()
            best_n = n_work
    
    # SLSQP polish at parent resolution
    if remaining_time() > 30:
        h_work = slsqp_smooth(h_work, n_work, beta=2000, maxiter=300)
        val = compute_c5(h_work)
        if val < best_c5:
            best_c5 = val
            best_h = h_work.copy()
    
    # Minimax at parent resolution
    if remaining_time() > 40:
        h_mm = slsqp_minimax(h_work, n_work, max_rounds=10, max_slsqp_iter=200)
        val = compute_c5(h_mm)
        if val < best_c5:
            best_c5 = val
            best_h = h_mm.copy()
            best_n = n_work
    
    # Phase 2: Upscale to higher resolution and optimize
    for target_n in [384, 512, 640, 768, 1024]:
        if remaining_time() < 60:
            break
        if target_n == best_n:
            continue
        h_interp = interp_to(best_h, best_n, target_n)
        h_interp = project_feasible(h_interp, target_n)
        c5_interp = compute_c5(h_interp)
        
        # PGM at new resolution
        for beta in [30, 100, 500, 2000, 8000]:
            if remaining_time() < 15:
                break
            lr = 0.004 / (1 + 0.03 * np.log1p(beta))
            max_it = 1000 if target_n <= 512 else 500
            h_new, val = projected_gradient_descent(h_interp, target_n, beta, max_iter=max_it, lr_init=lr)
            if val < best_c5:
                best_c5 = val
                best_h = h_new.copy()
                best_n = target_n
                h_interp = h_new.copy()
        
        # SLSQP polish at new resolution
        if remaining_time() > 30:
            maxit = 200 if target_n <= 512 else 100
            h_new = slsqp_smooth(h_interp, target_n, beta=5000, maxiter=maxit)
            val = compute_c5(h_new)
            if val < best_c5:
                best_c5 = val
                best_h = h_new.copy()
                best_n = target_n
        
        # Minimax at new resolution
        if remaining_time() > 40 and target_n <= 640:
            h_mm = slsqp_minimax(h_interp, target_n, max_rounds=6, max_slsqp_iter=100)
            val = compute_c5(h_mm)
            if val < best_c5:
                best_c5 = val
                best_h = h_mm.copy()
                best_n = target_n
    
    # Phase 3: Coordinate descent refinement at best resolution
    if remaining_time() > 30:
        h_cd = best_h.copy()
        for pass_num in range(5):
            if remaining_time() < 10:
                break
            h_cd, cd_c5 = coordinate_descent_pass(h_cd, best_n)
            if cd_c5 < best_c5:
                best_c5 = cd_c5
                best_h = h_cd.copy()
    
    # Phase 4: Pair transfer
    if remaining_time() > 25:
        h_pt, pt_c5 = pair_transfer(best_h, best_n, max_sweeps=4)
        if pt_c5 < best_c5:
            best_c5 = pt_c5
            best_h = h_pt.copy()
    
    # Phase 5: Final SLSQP polish with very high beta
    if remaining_time() > 20:
        h_final = slsqp_smooth(best_h, best_n, beta=20000, maxiter=150)
        val = compute_c5(h_final)
        if val < best_c5:
            best_c5 = val
            best_h = h_final.copy()
    
    # Phase 6: Final minimax
    if remaining_time() > 15:
        h_mm = slsqp_minimax(best_h, best_n, max_rounds=6, max_slsqp_iter=100)
        val = compute_c5(h_mm)
        if val < best_c5:
            best_c5 = val
            best_h = h_mm.copy()
    
    # Phase 7: One more coordinate descent
    if remaining_time() > 10:
        h_cd, cd_c5 = coordinate_descent_pass(best_h, best_n)
        if cd_c5 < best_c5:
            best_c5 = cd_c5
            best_h = h_cd.copy()
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
