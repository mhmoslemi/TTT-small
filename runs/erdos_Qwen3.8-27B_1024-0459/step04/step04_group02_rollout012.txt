    res_list = [r for r in res_list if r % 2 == 0 and 480 <= r <= 2560 and r != best_n]

    for rn in res_list:
        if remaining_time() < 50:
            break
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)

        for beta in [100, 400, 1500]:
            if remaining_time() < 15:
                break
            h_new = lbfgsb_run(h_r, rn, beta, 1e5, 500)
            c5_new = compute_c5(h_new)
            if c5_new < c5_r - 1e-15:
                h_r = h_new
                c5_r = c5_new
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = rn

        if remaining_time() > 20:
            h_new, c5_new = coord_descent_single(h_r, rn, 25)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = rn

    # Phase 5: Perturbation and restart at best resolution
    n = best_n
    if remaining_time() > 80:
        for trial in range(8):
            if remaining_time() < 40:
                break
            scale = 0.03 + 0.02 * trial
            h_pert = best_h + rng.normal(0, scale, n)
            h_pert = project_feasible(h_pert, n)
            c5_pert = compute_c5(h_pert)

            if c5_pert < best_c5 + 0.03:
                for beta in [200, 800, 3000]:
                    if remaining_time() < 15:
                        break
                    h_new = lbfgsb_run(h_pert, n, beta, 1e5, 400)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_pert:
                        h_pert = h_new
                        c5_pert = c5_new
                if c5_pert < best_c5:
                    best_c5 = c5_pert
                    best_h = h_pert.copy()

    # Phase 6: Final high-precision L-BFGS-B
    if remaining_time() > 15:
        h = best_h.copy()
        n = best_n
        for beta in [5000, 20000, 50000]:
            if remaining_time() < 8:
                break
            h_new = lbfgsb_run(h, n, beta, 1e5, 400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new

    # Phase 7: Final coordinate descent
    if remaining_time() > 15:
        h_new, c5_new = coord_descent_single(best_h, best_n, 40)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 8: Fine subgradient
    if remaining_time() > 8:
        h = best_h.copy()
        n = best_n
        steps = [0.001, 0.003, 0.005, 0.002, 0.0015]
        for it in range(300):
            if remaining_time() < 2:
                break
            h, c5 = subgrad_step(h, n, steps, 5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            else:
                if it > 100:
                    break

    # Phase 9: One more L-BFGS-B polish
    if remaining_time() > 10:
        h_new = lbfgsb_run(best_h, best_n, 10000, 1e5, 500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 10: Final coordinate descent
    if remaining_time() > 8:
        h_new, c5_new = coord_descent_single(best_h, best_n, 30)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final projection
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
