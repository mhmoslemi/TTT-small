def soft_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    shifted = corr - np.max(corr)
    weights = np.exp(np.clip(beta * shifted, -600, 600))
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for m in range(2*n-1):
        if weights[m] < 1e-10:
            continue
        lag = m - (n - 1)
        if lag >= 0:
            # corr[m] = sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag])
            # d/dh[j]: 
            #   if j in [0, n-1-lag]: term from h[j]*(1-h[j+lag]) -> (1-h[j+lag])
            #   if j in [lag, n-1]: term from -(h[j-lag])*h[j] -> -h[j-lag]

                grad[j] += weights[m] * (1.0 - h[j + lag])
            for j in range(lag, n):

            for j in range(n - m_neg):
                grad[j] += weights[m] * (-h[j])
    
    return grad * dx

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

def soft_grad(h, n, beta):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = corr - np.max(corr)
    weights = np.exp(np.clip(beta * shifted, -600, 600))
    weights /= np.sum(weights)
    grad = np.zeros(n)

        if weights[m] < 1e-12:
            continue
        w = weights[m]

                grad[lag:] -= w * h[:L]
        else:

                grad[:L] -= w * h[p:]
    return grad * dx

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_BETA = 1.0
_G_TGT = 0.0
_G_START = 0.0
_G_BUDGET = 1300.0
_G_DX = 1.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h


def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 3.0)


def soft_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = corr - np.max(corr)
    weights = np.exp(np.clip(beta * shifted, -600, 600))
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if weights[m] < 1e-12:
            continue
        w = weights[m]
        lag = m - (n - 1)
        if lag >= 0:
            L = n - lag
            if L > 0:
                grad[:L] += w * (1.0 - h[:L])
                grad[lag:] -= w * h[:L]
        else:
            p = -lag
            L = n - p
            if L > 0:
                grad[p:] += w * (1.0 - h[:L])
                grad[:L] -= w * h[p:]
    return grad * dx


def gradient_optimize(h, n, beta, max_iter=1000, step0=0.05, mu=0.8):
    c5_best = compute_c5(h)
    h_best = h.copy()
    v = np.zeros(n)
    alpha = step0
    patience = 0
    for it in range(max_iter):
        if remaining_time() < 3.0:
            break
        grad = soft_grad(h, n, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad = grad / gn
        improved = False
        for _ in range(20):
            v = mu * v - alpha * grad
            h_trial = h + v
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-13:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
            alpha *= 0.5
        if improved:
            patience = 0
            alpha = min(alpha * 1.5, 0.4)
        else:
            patience += 1
            alpha *= 0.5
            v *= 0.3
            if patience > 25:
                break
    return h_best, c5_best


def con_sum(x):
    return np.sum(x) - _G_TGT


def obj_smooth(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(_G_BETA * shifted, -600, 600))
    return (m + np.log(np.sum(exps)) / _G_BETA) * dx


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    return h


def make_golden(n):
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * phi) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return h


def do_slsqp(h, n, beta_val, maxiter=300):

    _G_N = n

        c5_new = compute_c5(h_new)

        d = rng.uniform(-0.04, 0.04)

            d = -d
            if h_cur[i] + d < 0.0 or h_cur[i] + d > 1.0:
                continue
        if h_cur[j] - d < 0.0 or h_cur[j] - d > 1.0:
            continue
        h_trial = h_cur.copy()
        h_trial[i] += d
        h_trial[j] -= d
        c5_trial = compute_c5(h_trial)
        dE = c5_trial - c5_cur
        if dE < 0 or rng.rand() < np.exp(-dE / max(T, 1e-10)):
            h_cur = h_trial
            c5_cur = c5_trial
            if c5_cur < c5_best:

                h_best = h_cur.copy()

    h_best = h.copy()
    if rng is None:
        rng = np.random.RandomState(2)

        if remaining_time() < 3.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if remaining_time() < 2.0:
                break
            old = h_best[i]
            candidates = [0.0, 1.0]
            if 0.0 < old < 1.0:

                    continue
                h_trial = h_best.copy()
                h_trial[i] = new_v
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < c5_best - 1e-13:
                    c5_best = c5_t
                    h_best = h_trial.copy()
                    any_imp = True
        if not any_imp and sweep > 2:
            break
    return h_best, c5_best
