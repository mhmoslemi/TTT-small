def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 360
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        best_h = parent_h.copy()
        best_c5 = compute_c5(best_h)
        best_n = parent_n

    # Phase 1: High resolution SLSQP at n=720
    n720 = 720
    h720 = interp_to(best_h, best_n, n720)
    h720 = project_feasible(h720, n720)
    c5_720 = compute_c5(h720)
    if c5_720 < best_c5:
        best_c5 = c5_720
        best_h = h720.copy()
        best_n = n720

    h720 = h720.copy()
    for beta in [150, 400, 1000, 3000, 8000]:
        if remaining() < 35:
            break
        h_new = slsqp_smooth(h720, n720, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n720
            h720 = h_new.copy()
        else:
            c5_tmp = compute_c5(h720)
            if c5_new < c5_tmp:
                h720 = h_new.copy()

    # Phase 2: Projected gradient at n=720
    if remaining() > 80:
        h_pg, c5_pg = projected_gradient(h720, n720, beta=2000, max_iters=400, lr_init=0.2)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            best_n = n720
            h720 = h_pg.copy()

    # Phase 3: Top-k local search at n=720
    if remaining() > 30:
        h_tk, c5_tk = topk_local_search(h720, n720, max_iters=300, top_k=3)
        if c5_tk < best_c5:
            best_c5 = c5_tk
            best_h = h_tk.copy()
            best_n = n720
            h720 = h_tk.copy()

    # Phase 4: Try n=480 as alternative resolution
    if remaining() > 100:
        n480 = 480
        h480 = interp_to(best_h, best_n, n480)
        h480 = project_feasible(h480, n480)
        c5_480 = compute_c5(h480)
        if c5_480 < best_c5:
            best_c5 = c5_480
            best_h = h480.copy()
            best_n = n480

        for beta in [300, 1000, 4000]:
            if remaining() < 30:
                break
            h_new = slsqp_smooth(h480, n480, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n480
                h480 = h_new.copy()
        if remaining() > 30:
            h_tk480, c5_tk480 = topk_local_search(h480, n480, max_iters=200, top_k=3)
            if c5_tk480 < best_c5:
                best_c5 = c5_tk480
                best_h = h_tk480.copy()
                best_n = n480

    # Phase 5: Multiple restarts at n=720 with perturbations
    if remaining() > 150:
        for r in range(4):
            if remaining() < 40:
                break
            strength = 0.03 + 0.02 * r
            h_pert = perturb(best_h, best_n, rng, strength=strength)
            h_pert_720 = interp_to(h_pert, len(h_pert), n720)
            h_pert_720 = project_feasible(h_pert_720, n720)
            for beta in [800, 3000]:
                if remaining() < 25:
                    break
                h_new = slsqp_smooth(h_pert_720, n720, beta, maxiter=400)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n720
                    h_pert_720 = h_new.copy()

    # Phase 6: Try n=1080 briefly
    if remaining() > 80:
        n1080 = 1080
        h1080 = interp_to(best_h, best_n, n1080)
        h1080 = project_feasible(h1080, n1080)
        for beta in [500, 2000]:
            if remaining() < 30:
                break
            h_new = slsqp_smooth(h1080, n1080, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n1080
                h1080 = h_new.copy()

    # Phase 7: Binary SA at n=720
    if remaining() > 80:
        h_bin = make_binary(best_h, best_n, rng)
        if best_n != n720:
            h_bin = make_binary(interp_to(best_h, best_n, n720), n720, rng)
        t_sa = min(120, max(30, remaining() * 0.35))
        for sa_seed in [seed + 100, seed + 200]:
            if remaining() < 20:
                break
            h_sa, c5_sa = sa_binary(h_bin, n720, time_limit=t_sa / 2, seed=sa_seed)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n720
                h_bin = best_h.copy()

    # Phase 8: Final intensive SLSQP polish
    if remaining() > 40:
        n = best_n
        for beta in [5000, 15000, 50000]:
            if remaining() < 15:
                break
            h_new = slsqp_smooth(best_h, n, beta, maxiter=600)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 9: Final top-k polish
    if remaining() > 15:
        h_tk, c5_tk = topk_local_search(best_h, best_n, max_iters=300, top_k=3)
        if c5_tk < best_c5:
            best_c5 = c5_tk
            best_h = h_tk.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
