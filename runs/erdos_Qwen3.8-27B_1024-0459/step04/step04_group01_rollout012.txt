```python
def simulate_annealing(h, n, time_limit, rng, t_start=1.0, t_end=0.01, cooling=0.999):
    h = h.copy()
    corr = compute_corr(h)
    max_corr = np.max(corr)
    best_h = h.copy()
    best_max = max_corr
    
    # Precompute: for efficiency, track which positions are 1 and 0
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    t = t_start
    it = 0
    start_time = time.time()
    
    while time.time() - start_time < time_limit:
        # Pick a random 1 and a random 0 to swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Compute delta in max_corr
        h[i] = 0
        h[j] = 1
        new_corr = compute_corr(h)
        new_max = np.max(new_corr)
        delta = new_max - max_corr
        
        if delta < 0 or rng.rand() < np.exp(-delta / t):
            # Accept
            max_corr = new_max
            corr = new_corr
            # Update ones and zeros
            idx_i = np.where(ones == i)[0][0]
            idx_j = np.where(zeros == j)[0][0]
            ones = np.delete(ones, idx_i)
            ones = np.append(ones, j)
            zeros = np.delete(zeros, idx_j)
            zeros = np.append(zeros, i)
            
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            # Reject
            h[i] = 1
            h[j] = 0
        
        t *= cooling
        it += 1
    
    return best_h, best_max * 2.0 / n
```
