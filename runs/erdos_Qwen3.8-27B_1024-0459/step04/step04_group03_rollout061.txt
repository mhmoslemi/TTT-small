def _sl_eq(x):
    return np.sum(x) - _N_SL / 2.0

def slsqp_run(h_init, n, beta, maxiter=200):
    global _N_SL, _BETA_SL, _PEN_SL
    _N_SL = n
    _BETA_SL = beta
    _PEN_SL = 0.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sl_eq, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_sl_obj, h_init, method='SLSQP',
                       bounds=bounds, constraints=cons, jac=_sl_grad,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h_init.copy()

def binarize(h, n):
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin

def binary_local_search(h_binary, n, max_iters=8000, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = h_binary.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h < 0.5)[0].copy()
    stuck = 0
    for it in range(max_iters):
        if remaining_time() < 2:
            break
        found = False
        for attempt in range(30):
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            h[i] = 1.0
            h[j] = 0.0
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                zeros = np.delete(zeros, np.where(zeros == i))
                zeros = np.append(zeros, j)
                ones = np.delete(ones, np.where(ones == j))
                ones = np.append(ones, i)
                found = True
                stuck = 0
                break
            else:
                h[i] = 0.0
                h[j] = 1.0
        if not found:
            stuck += 1
            if stuck > 30:
                break
        else:
            stuck = 0
    return h, best_c5

def simulated_annealing_binary(n, rng, max_iters=15000, T_start=0.5, T_end=0.005):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h < 0.5)[0].copy()
    for it in range(max_iters):
        if remaining_time() < 2:
            break
        frac = it / max(max_iters - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h[i] = 1.0
        h[j] = 0.0
        c5_new = compute_c5(h)
        delta = c5_new - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = c5_new
            zeros = np.delete(zeros, np.where(zeros == i))
            zeros = np.append(zeros, j)
            ones = np.delete(ones, np.where(ones == j))
            ones = np.append(ones, i)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i] = 0.0
            h[j] = 1.0
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def thue_morse_binary(n):
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    h = tm.astype(float)
    if sum(h) != n // 2:
        sorted_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sorted_idx[:n // 2]] = 1.0
    return h

def van_der_corput_binary(n):
    scores = np.zeros(n)
    for i in range(n):
        s = 0.0
        d = 2.0
        m = i + 1
        while m > 0:
            s += (m % 2) / d
            m //= 2
            d *= 2.0
        scores[i] = s
    sorted_idx = np.argsort(scores)[::-1]
    h = np.zeros(n)
    h[sorted_idx[:n // 2]] = 1.0
    return h

def coordinate_descent(h, n, max_sweeps=10, step=0.1):
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        any_imp = False
        for i in range(n):
            for delta in [step, -step, step * 0.5, -step * 0.5]:
                new_val = h[i] + delta
                if new_val < 0 or new_val > 1:
                    continue
                h[i] = new_val
                h = project_feasible(h, n)
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    any_imp = True
                    break
                else:
                    h[i] -= delta
        if not any_imp:
            step *= 0.5
            if step < 0.01:
                break
    return h, best_c5

_N_SL = 0
_BETA_SL = 0.0
_PEN_SL = 0.0

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S, _N_SL, _BETA_SL, _PEN_SL
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        best_n = len(parent_h)
        best_h = project_feasible(parent_h, best_n)
        best_c5 = compute_c5(best_h)
    except Exception:
        best_n = 256
        best_h = np.full(best_n, 0.5)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h_work = best_h.copy()

    # Phase 1: L-BFGS-B with escalating beta
    betas = [40, 100, 300, 800, 2000, 5000, 15000, 40000]
    for beta in betas:
        if remaining_time() < 12:
            break
        pen = 200.0 if beta < 100 else 1000.0
        h_new = lbfgsb_optimize(h_work, n, beta, pen, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-16:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_work = h_new.copy()
        elif remaining_time() > 30:
            h_work = h_new.copy()

    # Phase 2: SLSQP polish (handles constraint exactly)
    if remaining_time() > 20:
        for beta in [2000, 10000, 50000]:
            if remaining_time() < 8:
                break
            h_new = slsqp_run(best_h, n, beta, maxiter=150)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-16:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 3: Binary local search
    if remaining_time() > 25:
        h_bin = binarize(best_h, n)
        c5_bin0 = compute_c5(h_bin)
        h_ls, c5_ls = binary_local_search(h_bin, n, max_iters=12000, rng=rng)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 4: Simulated annealing
    if remaining_time() > 25:
        h_sa, c5_sa = simulated_annealing_binary(n, rng, max_iters=20000)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        # Also try from structured starts
        for init_func in [thue_morse_binary, van_der_corput_binary]:
            if remaining_time() < 15:
                break
            h_struct = init_func(n)
            h_struct_ls, c5_struct = binary_local_search(h_struct, n, max_iters=6000, rng=rng)
            if c5_struct < best_c5:
                best_c5 = c5_struct
                best_h = h_struct_ls.copy()

    # Phase 5: Refine best binary with smooth optimization
    if remaining_time() > 20:
        h_smooth_init = best_h.copy()
        h_smooth_init[h_smooth_init > 0.5] = 0.98
        h_smooth_init[h_smooth_init < 0.5] = 0.02
        h_smooth_init = project_feasible(h_smooth_init, n)
        for beta in [500, 3000, 15000]:
            if remaining_time() < 5:
                break
            h_new = lbfgsb_optimize(h_smooth_init, n, beta, 2000.0, maxiter=250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_smooth_init = h_new.copy()

    # Phase 6: Multi-resolution exploration
    trial_ns = [200, 224, 240, 288, 320, 384, 448, 512, 640]
    for trial_n in trial_ns:
        if remaining_time() < 18:
            break
        h_interp = interp_to(best_h, best_n, trial_n)
        h_interp = project_feasible(h_interp, trial_n)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5 + 0.008:
            for beta in [300, 2000, 10000]:
                if remaining_time() < 5:
                    break
                maxit = 250 if trial_n <= 320 else 150
                h_new = lbfgsb_optimize(h_interp, trial_n, beta, 1000.0, maxiter=maxit)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = trial_n
                    h_interp = h_new.copy()
            if remaining_time() > 12 and trial_n <= 384:
                h_bin2 = binarize(h_interp, trial_n)
                h_lin, c5_lin = binary_local_search(h_bin2, trial_n, max_iters=5000, rng=rng)
                if c5_lin < best_c5:
                    best_c5 = c5_lin
                    best_h = h_lin.copy()
                    best_n = trial_n

    # Phase 7: Coordinate descent on best solution
    if remaining_time() > 15:
        h_cd, c5_cd = coordinate_descent(best_h, best_n, max_sweeps=8, step=0.08)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 8: Final SLSQP + L-BFGS-B polish
    if remaining_time() > 12:
        h_new = slsqp_run(best_h, best_n, 30000, maxiter=100)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    if remaining_time() > 8:
        h_new = lbfgsb_optimize(best_h, best_n, 80000, 5000.0, maxiter=150)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 9: Final binary refinement
    if remaining_time() > 10:
        h_bin = binarize(best_h, best_n)
        h_fs, c5_fs = binary_local_search(h_bin, best_n, max_iters=6000, rng=rng)
        if c5_fs < best_c5:
            best_c5 = c5_fs
            best_h = h_fs.copy()

    # Phase 10: Ultimate smooth polish
    if remaining_time() > 6:
        h_new = lbfgsb_optimize(best_h, best_n, 200000, 10000.0, maxiter=100)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
