                backtrack += 1

            if it % 100 == 0 or it == iters_per_beta - 1:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        momentum = np.zeros(n)
    
    return best_h, best_c5

def _mm_ineq(x):
    n = _N_G
    dx = _DX_G
    h = x[:n]
    t = x[n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - dx * corr[_LAGS_G]

def _mm_eq(x):
    return np.sum(x[:_N_G]) - _N_G / 2.0

def _mm_obj(x):
    return x[_N_G]

def slsqp_minimax(h_init, n, max_rounds=12, max_slsqp_iter=200):
    global _N_G, _DX_G, _LAGS_G
    _N_G = n
    _DX_G = 2.0 / n
    h = h_init.copy()
    dx = 2.0 / n

    for rnd in range(max_rounds):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        max_corr = np.max(corr)
        t_init = max_corr * dx + 1e-8

        threshold = max_corr - 0.5
        active = np.where(corr >= threshold)[0]
        top3 = np.argsort(corr)[::-1][:3]
        active = np.unique(np.concatenate([active, top3]))
        active = active[np.argsort(active)]

        _LAGS_G = active
        x0 = np.concatenate([h, [t_init]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': _mm_ineq},
            {'type': 'eq', 'fun': _mm_eq}
        ]
        try:
            res = minimize(_mm_obj, x0, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_slsqp_iter, 'ftol': 1e-14, 'iprint': 0})
            if res.x is not None:
                h_new = project_feasible(res.x[:n], n)
                c5_new = compute_c5(h_new)
                c5_old = compute_c5(h)
                if c5_new < c5_old - 1e-16:
                    h = h_new
                else:
                    if rnd > 2:
                        break
        except Exception:
            break
    return h

def _sm_obj(x):
    n = _N_G
    beta = _BETA_G
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def _sm_eq(x):
    return np.sum(x) - _N_G / 2.0

def slsqp_smooth(h, n, beta, maxiter=300):
    global _N_G, _BETA_G
    _N_G = n
    _BETA_G = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sm_eq}]
    try:
        res = minimize(_sm_obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def coordinate_descent(h, n, max_sweeps=5, window=5):
    """Optimize small groups of coordinates via SLSQP."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        for start in range(0, n - window, window):
            if remaining_time() < 2:
                break
            idx = list(range(start, start + window))
            # Save current values
            saved = best_h[idx].copy()
            # Optimize these coordinates
            h_trial = best_h.copy()
            for _ in range(3):
                # Simple: try to reduce max corr by adjusting these coords
                corr = compute_corr(h_trial)
                # Find which lags are affected by changes in idx
                max_idx = np.argmax(corr)
                # Perturb
                for i in idx:
                    for delta in [0.02, -0.02, 0.05, -0.05]:
                        h_t2 = h_trial.copy()
                        h_t2[i] = np.clip(h_trial[i] + delta, 0.0, 1.0)
                        h_t2 = project_feasible(h_t2, n)
                        c5_t = compute_c5(h_t2)
                        if c5_t < best_c5 - 1e-15:
                            h_trial = h_t2
                            best_c5 = c5_t
                            best_h = h_trial.copy()
                            improved = True
                            break
            best_h[idx] = np.clip(np.concatenate([best_h[:start], h_trial[start:start+window], best_h[start+window:]]), 0, 1)
            best_h = project_feasible(best_h, n)
        if not improved:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S, _N_G, _BETA_G, _LAGS_G, _DX_G
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = project_feasible(parent_h, parent_n)
        best_c5 = compute_c5(best_h)
        best_n = parent_n
    except Exception:
        best_n = 256
        best_h = np.full(best_n, 0.5)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: Projected gradient descent at native resolution
    beta_schedule_1 = [20, 50, 100, 200, 500, 1000, 2000, 4000, 8000]
    h_pgd, c5_pgd = pgd_solve(h, n, beta_schedule_1, iters_per_beta=400, step_init=0.05)
    if c5_pgd < best_c5:
        best_c5 = c5_pgd
        best_h = h_pgd.copy()

    # Phase 2: SLSQP minimax polish
    if remaining_time() > 30:
        h_mm = slsqp_minimax(best_h, n, max_rounds=12, max_slsqp_iter=250)
        c5_mm = compute_c5(h_mm)
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm.copy()

    # Phase 3: SLSQP smooth polish
    if remaining_time() > 15:
        for beta in [3000, 10000, 30000]:
            if remaining_time() < 5:
                break
            h_new = slsqp_smooth(best_h, n, beta, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 4: Multi-resolution exploration
    trial_ns = [320, 384, 448, 512, 640, 768]
    for trial_n in trial_ns:
        if remaining_time() < 20:
            break
        h_interp = interp_to(best_h, best_n, trial_n)
        h_interp = project_feasible(h_interp, trial_n)
        c5_interp = compute_c5(h_interp)
        
        if c5_interp < best_c5 + 0.005:
            # PGD at higher resolution
            beta_sched = [50, 200, 500, 1000, 3000]
            if trial_n > 512:
                beta_sched = [100, 500, 2000]
            iters = 200 if trial_n <= 448 else 100
            h_hr, c5_hr = pgd_solve(h_interp, trial_n, beta_sched, iters_per_beta=iters, step_init=0.03)
            
            # SLSQP minimax at higher resolution
            if remaining_time() > 15 and trial_n <= 512:
                h_hr2 = slsqp_minimax(h_hr, trial_n, max_rounds=6, max_slsqp_iter=150)
                c5_hr2 = compute_c5(h_hr2)
                if c5_hr2 < c5_hr:
                    h_hr = h_hr2
                    c5_hr = c5_hr2
            
            if c5_hr < best_c5:
                best_c5 = c5_hr
                best_h = h_hr.copy()
                best_n = trial_n

    # Phase 5: Coordinate descent refinement
    if remaining_time() > 15:
        h_cd, c5_cd = coordinate_descent(best_h, best_n, max_sweeps=4, window=4)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 6: Perturbation restarts
    for restart in range(3):
        if remaining_time() < 25:
            break
        h_p = best_h.copy()
        mask = (h_p > 0.08) & (h_p < 0.92)
        if np.sum(mask) > 0:
            h_p[mask] += rng.normal(0, 0.03, np.sum(mask))
        h_p = project_feasible(h_p, best_n)
        
        # Quick PGD
        h_p2, c5_p2 = pgd_solve(h_p, best_n, [200, 1000, 3000], iters_per_beta=150, step_init=0.03)
        if c5_p2 < best_c5:
            best_c5 = c5_p2
            best_h = h_p2.copy()

    # Phase 7: Final minimax
    if remaining_time() > 12:
        h_mm = slsqp_minimax(best_h, best_n, max_rounds=8, max_slsqp_iter=200)
        c5_mm = compute_c5(h_mm)
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm.copy()

    # Phase 8: Ultra-precision PGD polish
    if remaining_time() > 10:
        h_new, c5_new = pgd_solve(best_h, best_n, [5000, 10000, 20000], iters_per_beta=200, step_init=0.01)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final: SLSQP smooth at very high beta
    if remaining_time() > 8:
        h_new = slsqp_smooth(best_h, best_n, 50000, maxiter=150)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
