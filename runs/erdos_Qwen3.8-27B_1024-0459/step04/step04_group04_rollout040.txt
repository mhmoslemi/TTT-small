def project_box_sum(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum} using bisection."""
    h = np.array(h, dtype=float)
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target_sum
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h_proj = np.clip(h - lam, 0.0, 1.0)
    # Fix numerical precision
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-12:
        mask = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            h_proj[mask] += diff / cnt
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def corr_grad_lag(h, n, k):
    """Gradient of corr[k] with respect to h."""

def lse_smooth(h, n, beta):
    """Compute smoothed max of correlation."""

    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def lbfags_optimize(h0, n, beta, maxiter=200, mu0=1.0):
    """L-BFGS-B optimization with penalty on sum constraint."""
    target = n / 2.0

        val = lse_smooth(x, n, beta)
        penalty = mu0 * (np.sum(x) - target) ** 2
        return val + penalty

        g = lse_smooth_grad(x, n, beta)
        g += 2 * mu0 * (np.sum(x) - target)
        return g
    
    bounds = [(0.0, 1.0)] * n

                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
