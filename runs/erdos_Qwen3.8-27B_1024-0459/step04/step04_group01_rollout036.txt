def compute_influence(h, n):
    nc = 2*n - 1
    inf = np.zeros((n, nc))
    one_minus_h = 1.0 - h
    for i in range(n):
        # Role 1: k in [i, i+n-1], k != n-1
        # inf[i, k] -= (1-h[n-1-k+i])
        # Let idx = k - i, range 0 to n-1, skip idx = n-1-i
        # inf[i, i+idx] -= one_minus_h[n-1-idx]
        ks = np.arange(i, i+n)
        if n-1 in ks:
            mask = ks != (n-1)
            ks_valid = ks[mask]
            idx_vals = ks_valid - i
            inf[i, ks_valid] -= one_minus_h[n-1-idx_vals]
        else:
            idx_vals = ks - i
            inf[i, ks] -= one_minus_h[n-1-idx_vals]
        
        # Role 2: k in [n-1-i, 2n-2-i], k != n-1
        # inf[i, k] += h[k-n+1+i]
        # Let idx = k-(n-1-i), range 0 to n-1, skip idx = i
        ks = np.arange(n-1-i, 2*n-1-i)
        if n-1 in ks:

            ks_valid = ks[mask]
            idx_vals = ks_valid - (n-1-i)
            inf[i, ks_valid] += h[idx_vals]
        else:

    nc = 2*n - 1
    delta = np.zeros(nc)
    # Role 1 for i (1->0): k in [i, i+n-1], k!=n-1: delta[k] -= (1-h[n-1-k+i])

        delta[ks] -= (1 - h[n-1-ks+i])
    # Role 2 for i (1->0): k in [n-1-i, 2n-2-i], k!=n-1: delta[k] += h[k-n+1+i]
    k_lo = n-1-i
