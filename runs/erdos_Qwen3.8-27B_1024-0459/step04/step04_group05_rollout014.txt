    cons = [{'type': 'eq', 'fun': con_sum_global, 'jac': grad_sum_global}]
    try:
        res = minimize(obj_global, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def subgrad_combined(h, n, active_idx):
    dx = 2.0 / n
    corr = compute_corr(h)
    max_val = np.max(corr)
    n_act = len(active_idx)
    weights = np.zeros(n_act)
    for i in range(n_act):
        d = max_val - corr[active_idx[i]]
        weights[i] = np.exp(-60.0 * d)
    wsum = np.sum(weights)
    if wsum < 1e-30:
        weights = np.ones(n_act) / n_act
    else:
        weights /= wsum

    grad = np.zeros(n)
    for i in range(n_act):
        if weights[i] < 1e-10:
            continue
        k_idx = int(active_idx[i])
        offset = k_idx - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[:npairs] += (1.0 - h[offset:offset + npairs])
                grad[offset:] -= h[:npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] -= h[aoff:aoff + npairs]
                grad[aoff:] += (1.0 - h[:npairs])
    grad *= dx
    return grad


def do_adaptive_minimax(h_start, n, budget_fn, rng, max_outer=40):
    global _G_N
    _G_N = n
    h = h_start.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    stuck = 0

    for outer in range(max_outer):
        if budget_fn() < 4.0:
            break

        corr = compute_corr(h)
        top_k = min(10, len(corr))
        active_idx = np.argsort(corr)[-top_k:]
        active_idx = np.sort(active_idx)

        progressed = False

        for beta in [80, 300, 1200]:
            if budget_fn() < 2.0:
                break
            h_new = run_slsqp(h, n, active_idx, beta, maxiter=120)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-14:
                c5_best = c5_new
                h_best = h_new.copy()
                h = h_new.copy()
                progressed = True

        for sg_it in range(15):
            if budget_fn() < 1.5:
                break
            grad = subgrad_combined(h, n, active_idx)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-18:
                break
            grad /= gnorm
            found = False
            for step in [0.04, 0.015, 0.08, 0.005, 0.12, 0.025, 0.06]:
                h_t = h - step * grad
                h_t = project_feasible(h_t, n)
                c5_t = compute_c5(h_t)
                if c5_t < c5_best - 1e-14:
                    c5_best = c5_t
                    h_best = h_t.copy()
                    h = h_t.copy()
                    progressed = True
                    found = True
                    break
            if not found:
                h_t = project_feasible(h + rng.randn(n) * 0.015, n)
                c5_t = compute_c5(h_t)
                if c5_t < c5_best - 1e-14:
                    c5_best = c5_t
                    h_best = h_t.copy()
                    h = h_t.copy()
                    progressed = True

        if not progressed:
            stuck += 1
            if stuck > 8:
                # Larger perturbation to escape
                h_t = project_feasible(h + rng.randn(n) * 0.06, n)
                h = h_t
                stuck = 0
        else:
            stuck = 0

    return h_best, c5_best


def do_binary_round(h, n):
    h_bin = np.where(h >= 0.5, 1.0, 0.0)
    s = np.sum(h_bin)
    target = n / 2.0
    if s > target:
        order = np.argsort(h_bin - h)
        for idx in order:
            if s <= target:
                break
            if h_bin[idx] == 1.0:
                h_bin[idx] = 0.0
                s -= 1
    elif s < target:
        order = np.argsort(h - h_bin)
        for idx in order:
            if s >= target:
                break
            if h_bin[idx] == 0.0:
                h_bin[idx] = 1.0
                s += 1
    return h_bin


def do_coord_descent(h_start, n, budget_fn, rng, max_sweeps=8):
    h = h_start.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    for sweep in range(max_sweeps):
        if budget_fn() < 2.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if budget_fn() < 1.0:
                break
            old = h[i]
            for nv in [0.0, 1.0]:
                if abs(nv - old) < 1e-7:
                    continue
                h[i] = nv
                h_proj = project_feasible(h, n)
                c5_t = compute_c5(h_proj)
                if c5_t < c5_best - 1e-14:
                    c5_best = c5_t
                    h_best = h_proj.copy()
                    h = h_proj.copy()
                    any_imp = True
                else:
                    h = h_best.copy()
                    break
        if not any_imp and sweep > 2:
            break
    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_START, _G_BUDGET, _G_ACTIVE, _G_BETA
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    def budget_fn():
        return remaining_time()

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        _G_N = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        _G_N = parent_n
        best_h = make_thue_morse(parent_n)
        best_h = project_feasible(best_h, parent_n)
        best_c5 = compute_c5(best_h)
        best_n = parent_n
        print("TM: n={}, C5={:.10f}".format(parent_n, best_c5))

    # Phase 1: Adaptive minimax
    print("\n--- Phase 1: Adaptive minimax n={} ---".format(best_n))
    best_h, best_c5 = do_adaptive_minimax(best_h, best_n, budget_fn, rng, max_outer=35)
    _G_ACTIVE = None
    print("  C5={:.10f}".format(best_c5))

    # Phase 2: Binary rounding cycle
    if budget_fn() > 25:
        print("\n--- Phase 2: Binary round ---")
        h_bin = do_binary_round(best_h, best_n)
        c5_bin = compute_c5(h_bin)
        print("  Binary: C5={:.10f}".format(c5_bin))
        # Optimize from binary
        h_b_opt, c5_b_opt = do_adaptive_minimax(h_bin, best_n, budget_fn, rng, max_outer=25)
        if c5_b_opt < best_c5 - 1e-14:
            best_h = h_b_opt
            best_c5 = c5_b_opt
            print("  Binary path improved: C5={:.10f}".format(best_c5))
        # Also try mixed
        h_mix = project_feasible(0.4 * best_h + 0.6 * h_bin, best_n)
        h_m_opt, c5_m_opt = do_adaptive_minimax(h_mix, best_n, budget_fn, rng, max_outer=20)
        if c5_m_opt < best_c5 - 1e-14:
            best_h = h_m_opt
            best_c5 = c5_m_opt
            print("  Mixed improved: C5={:.10f}".format(best_c5))
        _G_ACTIVE = None
        print("  Phase 2 best: C5={:.10f}".format(best_c5))

    # Phase 3: Higher resolutions
    if budget_fn() > 50:
        print("\n--- Phase 3: Higher resolution ---")
        for n_try in [320, 384, 512]:
            if budget_fn() < 25:
                break
            h_int = interp_h(best_h, best_n, n_try)
            h_int = project_feasible(h_int, n_try)
            c5_int = compute_c5(h_int)
            h_opt, c5_opt = do_adaptive_minimax(h_int, n_try, budget_fn, rng, max_outer=25)
            _G_ACTIVE = None
            if c5_opt < best_c5 - 1e-14:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_try
                print("  n={} C5={:.10f}".format(n_try, best_c5))
            else:
                print("  n={} C5={:.10f} (best={:.10f})".format(n_try, c5_opt, best_c5))

    # Phase 4: Multi-start at best resolution
    if budget_fn() > 35:
        print("\n--- Phase 4: Multi-start ---")
        inits = [make_thue_morse(best_n), make_golden(best_n)]
        inits.append(make_random_bin(rng, best_n))
        inits.append(make_random_bin(rng, best_n))
        for init in inits:
            if budget_fn() < 15:
                break
            h_try = project_feasible(init, best_n)
            # Bias toward current best for warm start
            h_try = project_feasible(0.25 * init + 0.75 * best_h, best_n)
            h_opt, c5_opt = do_adaptive_minimax(h_try, best_n, budget_fn, rng, max_outer=20)
            _G_ACTIVE = None
            if c5_opt < best_c5 - 1e-14:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                print("  Restart improved: C5={:.10f}".format(best_c5))

    # Phase 5: Coordinate descent
    if budget_fn() > 15:
        print("\n--- Phase 5: Coord descent ---")
        best_h, best_c5 = do_coord_descent(best_h, best_n, budget_fn, rng, max_sweeps=6)
        print("  C5={:.10f}".format(best_c5))

    # Phase 6: Final adaptive minimax polish
    if budget_fn() > 10:
        print("\n--- Phase 6: Final polish ---")
        best_h, best_c5 = do_adaptive_minimax(best_h, best_n, budget_fn, rng, max_outer=25)
        _G_ACTIVE = None
        print("  C5={:.10f}".format(best_c5))

    # Phase 7: Last SLSQP with very high beta
    if budget_fn() > 8:
        corr = compute_corr(best_h)
        top_k = 6
        active = np.sort(np.argsort(corr)[-top_k:])
        h_hb = run_slsqp(best_h, best_n, active, 10000, maxiter=80)
        c5_hb = compute_c5(h_hb)
        if c5_hb < best_c5 - 1e-14:
            best_h = h_hb
            best_c5 = c5_hb
        _G_ACTIVE = None

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
