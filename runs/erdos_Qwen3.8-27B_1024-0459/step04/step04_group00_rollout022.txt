def build_F(h, n):
    s = np.arange(-(n-1), n)  # (2n-1,)
    i_idx = np.arange(n)[:, None]  # (n,1)
    p1 = i_idx - s[None, :]  # (n, 2n-1)
    p2 = i_idx + s[None, :]  # (n, 2n-1)
    v1 = (p1 >= 0) & (p1 < n)
    v2 = (p2 >= 0) & (p2 < n)
    F = np.where(v1, h[np.clip(p1, 0, n-1)], 0.0) + np.where(v2, h[np.clip(p2, 0, n-1)], 0.0)
    return F

def binary_local_search(h, n, time_limit=120):
