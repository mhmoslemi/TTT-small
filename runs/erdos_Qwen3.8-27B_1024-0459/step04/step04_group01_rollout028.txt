                x0_new[mask] += diff / mask.sum()
            x0_new = np.clip(x0_new, 0.0, 1.0)

            hx_new = h.copy()
            hx_new[free_idx] = x0_new
            c5_new = compute_c5(hx_new)

            if c5_new < best_c5 - 1e-15:
                x0 = x0_new
                h = hx_new
                best_c5 = c5_new
                best_h = h.copy()
                step_s = min(step_s * 1.02, 0.05)
            else:
                step_s *= 0.9

        h = project_feasible(best_h, n)

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
    except Exception:
        parent_n = 312
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)

    best_h = parent_h.copy()
    best_c5 = parent_c5
    best_n = parent_n

    # Phase 1: SLSQP on parent resolution with increasing beta
    for beta in [500, 2000, 8000, 20000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(best_h, best_n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Targeted subgradient
    if remaining() > 20:
        h_sg, c5_sg = targeted_subgrad(best_h, best_n, T=4000, time_limit=50, top_k=3)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 3: Pair coordinate descent
    if remaining() > 15:
        h_pc, c5_pc = pair_coord_descent(best_h, best_n, max_passes=300, time_limit=60, rng=rng)
        if c5_pc < best_c5:
            best_c5 = c5_pc
            best_h = h_pc.copy()

    # Phase 4: SLSQP polish
    for beta in [5000, 20000, 50000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(best_h, best_n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 5: Resolution sweep
    n_candidates = set()
    for dn in range(-20, 21, 2):
        nn = best_n + dn
        if 100 <= nn <= 600 and nn % 2 == 0 and nn != best_n:
            n_candidates.add(nn)
    # Add some specific values
    for nn in [196, 200, 204, 240, 248, 256, 264, 280, 288, 300,
               316, 320, 324, 336, 340, 344, 360, 384, 400, 448, 500, 512]:
        if 80 <= nn <= 600 and nn % 2 == 0 and nn != best_n:
            n_candidates.add(nn)

    n_list = sorted(n_candidates)
    for n_t in n_list:
        if remaining() < 35:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)

        h_opt = h_feas
        for beta in [2000, 10000]:
            if remaining() < 12:
                break
            h_opt = slsqp_smooth(h_opt, n_t, beta, maxiter=200)

        if remaining() > 15:
            h_opt, _ = targeted_subgrad(h_opt, n_t, T=1500, time_limit=20, top_k=3)

        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 6: Near-binary refinement on best
    if remaining() > 30:
        h_nb, c5_nb = near_binary_refine(best_h, best_n, time_limit=min(60, max(10, remaining() - 15)))
        if c5_nb < best_c5:
            best_c5 = c5_nb
            best_h = h_nb.copy()

    # Phase 7: Multi-restart with perturbations
    if remaining() > 60:
        for r in range(8):
            if remaining() < 30:
                break
            h_pert = best_h.copy()
            n_pert = rng.randint(3, 8)
            idx = rng.choice(best_n, n_pert, replace=False)
            h_pert[idx] += rng.randn(n_pert) * 0.04
            h_pert = project_feasible(h_pert, best_n)

            h_opt = slsqp_smooth(h_pert, best_n, 10000, maxiter=200)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    # Phase 8: Final intensive optimization
    n = best_n
    h = best_h.copy()

    for beta in [10000, 30000, 80000, 150000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining() > 20:
        h_sg2, c5_sg2 = targeted_subgrad(best_h, best_n, T=5000,
                                         time_limit=min(50, max(10, remaining() - 15)),
                                         top_k=3)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()

    if remaining() > 15:
        h_nb2, c5_nb2 = near_binary_refine(best_h, best_n,
                                           time_limit=min(40, max(8, remaining() - 10)))
        if c5_nb2 < best_c5:
            best_c5 = c5_nb2
            best_h = h_nb2.copy()

    # Final SLSQP polish
    if remaining() > 6:
        h_new = slsqp_smooth(best_h, best_n, 50000, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Validate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    else:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
