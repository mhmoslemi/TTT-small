        ks = np.arange(lo4, hi4 + 1)
        dr[lo4:hi4+1] -= h[j + ks - (n - 1)]
    # Correction at k=n-1 (double counting)
    dr[n - 1] -= 2.0
    return dr

def binary_local_search(h_bin, n, time_limit=60, rng=None):
    """Fast binary local search with O(n) correlation updates."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h_bin > 0.5).copy().astype(float)
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones = np.where(h == 1)[0]
            to_rm = rng.choice(ones, n_ones - n_half, replace=False)
            h[to_rm] = 0
        else:
            zeros = np.where(h == 0)[0]
            to_add = rng.choice(zeros, n_half - n_ones, replace=False)
            h[to_add] = 1
    
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n
    
    start = time.time()
    stall = 0
    max_stall = 500
    
    while time.time() - start < time_limit and remaining() > 10:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Generate candidates from top shifts
        top_k = np.argsort(-corr)[:12]
        candidates = []
        for s in top_k[:8]:
            off = s - (n - 1)
            if off >= 0:
                np_ = n - off
                contrib = np.where((h[off:off+np_] == 1) & (h[:np_] == 0))[0]
                for ci in contrib[:15]:
                    candidates.append((ci + off, ci))
            else:
                ao = -off
                np_ = n - ao
                contrib = np.where((h[:np_] == 1) & (h[ao:ao+np_] == 0))[0]
                for ci in contrib[:15]:
                    candidates.append((ci, ci + ao))
        # Random candidates
        for _ in range(30):
            candidates.append((ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))]))
        
        best_move = None
        best_new_max = cur_max + 1e-15
        
        seen = set()
        for (i, j) in candidates:
            if h[i] != 1 or h[j] != 0:
                continue
            key = (i, j)
            if key in seen:
                continue
            seen.add(key)
            
            dr = make_swap_delta(h, i, j, n)
            new_corr = corr + dr
            new_max = float(np.max(new_corr))
            
            if new_max < best_new_max:
                best_new_max = new_max
                best_move = (i, j, dr)
        
        if best_move is not None:
            i, j, dr = best_move
            h[i] = 0
            h[j] = 1
            corr = corr + dr
            cur_max = best_new_max
            stall = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            stall += 1
            if stall > max_stall:
                # Random perturbation (accept 3 random worsening moves)
                for _ in range(3):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    if h[i] == 1 and h[j] == 0:
                        dr = make_swap_delta(h, i, j, n)
                        h[i] = 0
                        h[j] = 1
                        corr = corr + dr
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                cur_max = float(np.max(corr))
                stall = 0
    
    return best_h, best_max * dx

def slsqp_smooth(h, n, beta, maxiter=500):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - _TGT,
             'jac': lambda x: np.ones(_N)}]
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(_BETA * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / _BETA) * (2.0 / n)
    
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(_BETA * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def top_shifts_equalize(h, n, top_k=5, max_iters=300):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-10:
            break
        
        # Average gradient of top-k to push them down together
        g_avg = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_avg[off:off + np_] += (1.0 - h[:np_])
                g_avg[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g_avg[:np_] += (1.0 - h[ao:ao + np_])
                g_avg[ao:ao + np_] -= h[:np_]
        
        # Also add negative gradient of bottom to pull them up
        bot_idx = np.argsort(corr)[:min(3, len(corr))]
        g_bot = np.zeros(n)
        for k in bot_idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_bot[off:off + np_] += (1.0 - h[:np_])
                g_bot[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g_bot[:np_] += (1.0 - h[ao:ao + np_])
                g_bot[ao:ao + np_] -= h[:np_]
        
        d = g_bot - g_avg
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        
        improved = False
        for step in [0.01, 0.03, 0.05, 0.02, 0.08, 0.005, 0.1, 0.015, 0.04]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    # Adjust to have exactly n/2 ones
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

def greedy_construct(n, rng):
    """Greedy: place 1s to minimize max correlation so far."""
    h = np.zeros(n)
    n_half = n // 2
    used = set()
    positions = list(range(n))
    rng.shuffle(positions)
    
    for _ in range(n_half):
        best_pos = -1
        best_max = float('inf')
        for p in positions:
            if p in used:
                continue
            h[p] = 1
            corr = compute_corr(h)
            mx = float(np.max(corr))
            h[p] = 0
            if mx < best_max:
                best_max = mx
                best_pos = p
        if best_pos >= 0:
            h[best_pos] = 1
            used.add(best_pos)
        else:
            for p in positions:
                if p not in used:
                    h[p] = 1
                    used.add(p)
                    break
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    # Phase 1: Binary local search at parent resolution
    n = best_n
    h_bin = (best_h > 0.5).astype(float)
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-best_h)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0
    
    sa_time1 = min(300, int(remaining() * 0.3))
    if sa_time1 > 20:
        h_b1, c5_b1 = binary_local_search(h_bin, n, time_limit=sa_time1, rng=rng)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()
            best_n = n
    
    # Phase 2: Try other binary starting points
    start_points = [
        ('thue_morse', thue_morse(n)),
        ('greedy', greedy_construct(n, np.random.RandomState(seed + 100))),
        ('random', None),
    ]
    for name, sp in start_points:
        if remaining() < 30:
            break
        if sp is None:
            sp = (rng.rand(n) > 0.5).astype(float)
            n_o = int(np.sum(sp))
            if n_o != n_half:
                if n_o > n_half:
                    idx = np.where(sp == 1)[0]
                    sp[idx[:n_o - n_half]] = 0
                else:
                    idx = np.where(sp == 0)[0]
                    sp[idx[:n_half - n_o]] = 1
        sp_c5 = compute_c5(sp)
        if sp_c5 < c5_b1:
            h_b2, c5_b2 = binary_local_search(sp, n, time_limit=min(120, int(remaining() * 0.15)), rng=rng)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()
                best_n = n
    
    # Phase 3: Try other resolutions with binary search
    resolutions = sorted(set([
        n, n - 16, n - 8, n + 8, n + 16, n + 32,
        int(n * 0.75), int(n * 1.25), int(n * 1.5),
    ]))
    resolutions = [r for r in resolutions if 128 <= r <= 768 and r % 2 == 0 and r != n]
    resolutions.sort(key=lambda x: abs(x - n))
    
    for n_t in resolutions[:4]:
        if remaining() < 40:
            break
        # Interpolate best continuous solution to new resolution, round to binary
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = (h_interp > 0.5).astype(float)
        nt_half = n_t // 2
        nt_ones = int(np.sum(h_bin_t))
        if nt_ones != nt_half:
            if nt_ones > nt_half:
                idx = np.where(h_bin_t == 1)[0]
                h_bin_t[idx[:nt_ones - nt_half]] = 0
            else:
                idx = np.where(h_bin_t == 0)[0]
                h_bin_t[idx[:nt_half - nt_ones]] = 1
        
        t_bi = min(80, int(remaining() * 0.08))
        if t_bi > 10:
            h_b3, c5_b3 = binary_local_search(h_bin_t, n_t, time_limit=t_bi, rng=rng)
            if c5_b3 < best_c5:
                best_c5 = c5_b3
                best_h = h_b3.copy()
                best_n = n_t
    
    # Phase 4: Continuous refinement
    n = best_n
    h = best_h.copy()
    
    for beta in [100, 500, 2000, 5000, 10000, 20000]:
        if remaining() < 15:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 5: Top-shifts equalization
    if remaining() > 20:
        h_eq, c5_eq = top_shifts_equalize(h, n, top_k=6, max_iters=500)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
    
    # Phase 6: More continuous refinement with higher beta
    for beta in [10000, 20000, 40000, 80000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 7: Try finer resolution
    if remaining() > 60:
        n_fine = n + 64
        h_fine = interp_to(best_h, n, n_fine)
        h_fine = project_feasible(h_fine, n_fine)
        for beta in [5000, 15000, 30000]:
            if remaining() < 15:
                break
            h_fine = slsqp_smooth(h_fine, n_fine, beta, maxiter=400)
        c5_fine = compute_c5(h_fine)
        if remaining() > 20:
            h_fine, c5_fine2 = top_shifts_equalize(h_fine, n_fine, top_k=6, max_iters=300)
            c5_fine = min(c5_fine, c5_fine2)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()
            best_n = n_fine
    
    # Phase 8: Final polish
    n = best_n
    h = best_h.copy()
    for beta in [20000, 50000, 100000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    if remaining() > 10:
        h_eq, c5_eq = top_shifts_equalize(h, n, top_k=8, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
