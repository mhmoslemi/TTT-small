def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h


def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 3.0)


def grad_lag(h, n, idx):
    """Gradient of np.correlate(h, 1-h, 'full')[idx] w.r.t. h."""
    grad = np.zeros(n)
    lag = idx - (n - 1)
    if lag >= 0:
        m = n - lag
        if m <= 0:
            return grad
        grad[:m] -= h[lag:n]
        grad[lag:n] += (1.0 - h[:m])
    else:
        m = -lag
        if m >= n:
            return grad
        valid = n - m
        grad[:valid] -= h[m:m+valid]
        grad[m:n] += (1.0 - h[:valid])
    return grad

                   top_k=1, anneal_temp=0.02, anneal_decay=0.999):
    """Subgradient with simulated annealing."""
    c5_best = compute_c5(h)
    h_best = h.copy()
    v = np.zeros(n)
    step = step0
    patience = 0
    temp = anneal_temp
    rng = np.random.RandomState(12345)
    
    for it in range(max_iter):
        if remaining_time() < 5.0:
            break
        
        corr = compute_corr(h)
        if top_k == 1:
            top_idx = np.array([np.argmax(corr)])
        else:
            top_idx = np.argsort(corr)[-top_k:]
        
        g = np.zeros(n)
        for ti in top_idx:
            g += grad_lag(h, n, ti)
        g /= len(top_idx)
        
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        
        alpha = step
        improved = False
        for _ in range(10):
            v = momentum * v - alpha * g
            h_trial = h + v
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            # Annealing: accept worsening with probability exp(-delta/temp)
            if c5_trial < c5_best:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
            elif temp > 1e-10 and rng.random() < np.exp(-(c5_trial - c5_best) / temp * 100):
                h = h_trial.copy()
                improved = True
                break
            alpha *= 0.5
        temp *= anneal_decay

            # h was updated to a non-best point (annealing)
            pass

            patience = 0
            step = min(alpha * 1.15, 0.6)
        else:
            patience += 1
            step *= 0.75
            v *= 0.4
            if patience > 30:
                break
    
    return h_best, c5_best

def slsqp_optimize(h, n, maxiter=300, ftol=1e-14):
    """Run SLSQP with the raw max objective."""
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return float(np.max(corr)) * dx
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h


def slsqp_smooth(h, n, beta, maxiter=300):
    """Run SLSQP with log-sum-exp smooth approximation."""
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        exps = np.exp(np.clip(beta * shifted, -600, 600))
        return (m + np.log(np.sum(exps)) / beta) * dx
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h

            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-15:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
            elif temp > 1e-12:
                log_p = -(c5_trial - c5_best) / max(temp, 1e-15) * 50.0
                if log_p > -50.0 and rng.random() < np.exp(log_p):
                    h = h_trial.copy()
                    improved = True
                    break
            alpha *= 0.5
        temp *= anneal_decay

        if improved:
            patience = 0
            step = min(alpha * 1.15, 0.6)
        else:
            patience += 1
            step *= 0.75
            v *= 0.4
            if patience > 30:
                break

    return h_best, c5_best


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h


def make_random_bin(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h


def make_golden_ratio(n):
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * phi) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return h


def con_sum(x):
    return np.sum(x) - _G_N / 2.0


def slsqp_smooth(h, n, beta, maxiter=300):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum}]
    global _G_N

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        exps = np.exp(np.clip(beta * shifted, -600, 600))
        return (m + np.log(np.sum(exps)) / beta) * dx

    old_n = _G_N
    _G_N = n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        _G_N = old_n
        return h_new
    except Exception:
        _G_N = old_n
        return h


def slsqp_raw(h, n, maxiter=200):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum}]
    global _G_N

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return float(np.max(corr)) * dx

    old_n = _G_N
    _G_N = n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        _G_N = old_n
        return h_new
    except Exception:
        _G_N = old_n
        return h


def pairwise_swap_search(h, n, max_rounds=300, rng=None):
    if rng is None:
        rng = np.random.RandomState(99)
    c5_best = compute_c5(h)
    h_best = h.copy()
    found = 0

    for _ in range(max_rounds):
        if remaining_time() < 3.0 or found > 50:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        pert_type = rng.randint(3)
        h_trial = h_best.copy()
        if pert_type == 0:
            h_trial[i], h_trial[j] = h_trial[j], h_trial[i]
        elif pert_type == 1:
            h_trial[i] = 1.0 - h_trial[i]
            h_trial[j] = 1.0 - h_trial[j]
        else:
            t = h_best[i]
            h_trial[i] = h_best[j]
            h_trial[j] = t
        h_trial = project_feasible(h_trial, n)
        c5_t = compute_c5(h_trial)
        if c5_t < c5_best - 1e-14:
            c5_best = c5_t
            h_best = h_trial.copy()
            found += 1

    return h_best, c5_best


def coordinate_descent_fast(h, n, max_sweeps=5, rng=None):
    if rng is None:
        rng = np.random.RandomState(77)
    c5_best = compute_c5(h)
    h_best = h.copy()

    for sweep in range(max_sweeps):
        if remaining_time() < 5.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if remaining_time() < 2.0:
                break
            old = h_best[i]
            for new_v in [0.0, 1.0, old + 0.1, old - 0.1, old + 0.03, old - 0.03]:
                if new_v < -0.01 or new_v > 1.01:
                    continue
                if abs(new_v - old) < 1e-8:
                    continue
                h_trial = h_best.copy()
                h_trial[i] = np.clip(new_v, 0.0, 1.0)
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < c5_best - 1e-13:
                    c5_best = c5_t
                    h_best = h_trial.copy()
                    any_imp = True
                    break
        if not any_imp and sweep > 1:
            break
    return h_best, c5_best


def polarity_search(h, n, rng, max_flips=50):
    """Try flipping subsets of coordinates to find better binary-like solutions."""
    c5_best = compute_c5(h)
    h_best = h.copy()
    for _ in range(max_flips):
        if remaining_time() < 3.0:
            break
        h_trial = h_best.copy()
        k = rng.randint(1, max(2, n // 10))
        idx = rng.choice(n, size=k, replace=False)
        h_trial[idx] = 1.0 - h_trial[idx]
        h_trial = project_feasible(h_trial, n)
        c5_t = compute_c5(h_trial)
        if c5_t < c5_best - 1e-13:
            c5_best = c5_t
            h_best = h_trial.copy()
    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_START, _G_BUDGET
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(parent_h)
        best_h = project_feasible(parent_h, best_n)
        best_c5 = compute_c5(best_h)
        _G_N = best_n
        print("Parent: n={}, C5={:.10f}".format(best_n, best_c5))
    except Exception:
        best_n = 256
        best_h = project_feasible(make_thue_morse(best_n), best_n)
        best_c5 = compute_c5(best_h)
        _G_N = best_n
        print("Init: n={}, C5={:.10f}".format(best_n, best_c5))

    # Phase 1: Aggressive single-lag subgradient
    if remaining_time() > 180:
        print("\n--- Phase 1: Subgrad single-lag ---")
        best_h, best_c5 = subgrad_anneal(best_h, best_n, max_iter=20000,
                                         step0=0.15, momentum=0.7, top_k=1,
                                         anneal_temp=0.03, anneal_decay=0.9995)
        print("  C5={:.10f}".format(best_c5))

    # Phase 2: Multi-lag subgradient
    if remaining_time() > 120:
        print("\n--- Phase 2: Subgrad multi-lag ---")
        best_h, best_c5 = subgrad_anneal(best_h, best_n, max_iter=10000,
                                         step0=0.08, momentum=0.7, top_k=3,
                                         anneal_temp=0.01, anneal_decay=0.999)
        print("  C5={:.10f}".format(best_c5))

    # Phase 3: SLSQP polish with increasing beta
    if remaining_time() > 200:
        print("\n--- Phase 3: SLSQP smooth ---")
        for beta in [50, 100, 200, 500, 1000, 2000, 5000]:
            if remaining_time() < 15.0:
                break
            _G_N = best_n
            h_new = slsqp_smooth(best_h, best_n, beta=beta, maxiter=350)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new
                print("  beta={}: C5={:.10f}".format(beta, best_c5))

    # Phase 4: Raw SLSQP
    if remaining_time() > 80:
        print("\n--- Phase 4: SLSQP raw ---")
        _G_N = best_n
        h_new = slsqp_raw(best_h, best_n, maxiter=250)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new
            print("  C5={:.10f}".format(best_c5))

    # Phase 5: Pairwise swap
    if remaining_time() > 60:
        print("\n--- Phase 5: Pairwise swap ---")
        best_h, best_c5 = pairwise_swap_search(best_h, best_n, max_rounds=500, rng=rng)
        print("  C5={:.10f}".format(best_c5))

    # Phase 6: Coordinate descent
    if remaining_time() > 40:
        print("\n--- Phase 6: Coord descent ---")
        best_h, best_c5 = coordinate_descent_fast(best_h, best_n, max_sweeps=5, rng=rng)
        print("  C5={:.10f}".format(best_c5))

    # Phase 7: Final SLSQP
    if remaining_time() > 30:
        print("\n--- Phase 7: Final SLSQP ---")
        for beta in [3000, 8000, 15000]:
            if remaining_time() < 10.0:
                break
            _G_N = best_n
            h_new = slsqp_smooth(best_h, best_n, beta=beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new
        print("  C5={:.10f}".format(best_c5))

    # Phase 8: Iterated local search with restarts
    if remaining_time() > 80:
        print("\n--- Phase 8: Iterated local search ---")
        n = best_n
        for restart in range(15):
            if remaining_time() < 40.0:
                break

            h_pert = best_h.copy()
            pert_type = restart % 5
            if pert_type == 0:
                idx = rng.choice(n, size=n // 6, replace=False)
                h_pert[idx] = 1.0 - h_pert[idx]
            elif pert_type == 1:
                h_pert += 0.04 * rng.randn(n)
            elif pert_type == 2:
                start = rng.randint(0, n // 3)
                L = rng.randint(n // 12, n // 5)
                h_pert[start:start + L] = 1.0 - h_pert[start:start + L]
            elif pert_type == 3:
                s1 = rng.randint(0, n // 4)
                s2 = rng.randint(n // 3, n - n // 8)
                L = min(rng.randint(4, n // 8), n - s2)
                if L > 0:
                    a = h_pert[s1:s1 + L].copy()
                    h_pert[s1:s1 + L] = h_pert[s2:s2 + L]
                    h_pert[s2:s2 + L] = a
            else:
                ones = np.where(h_pert > 0.5)[0]
                zeros = np.where(h_pert < 0.5)[0]
                if len(ones) > 2 and len(zeros) > 2:
                    k = min(3, len(ones), len(zeros))
                    oi = rng.choice(ones, size=k, replace=False)
                    zi = rng.choice(zeros, size=k, replace=False)
                    h_pert[oi] = 0.0
                    h_pert[zi] = 1.0

            h_pert = project_feasible(h_pert, n)
            _G_N = n
            h_pert, c5_pert = subgrad_anneal(h_pert, n, max_iter=3000,
                                              step0=0.1, momentum=0.7, top_k=1,
                                              anneal_temp=0.01, rng_seed=100 + restart)
            if remaining_time() > 15.0:
                h_new = slsqp_smooth(h_pert, n, beta=1000, maxiter=250)
                c5_new = compute_c5(h_new)
                if c5_new < c5_pert:
                    h_pert = h_new
                    c5_pert = c5_new

            if c5_pert < best_c5 - 1e-12:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                print("  restart {}: C5={:.10f}".format(restart, best_c5))

    # Phase 9: Final micro-polish
    if remaining_time() > 15:
        print("\n--- Phase 9: Micro-polish ---")
        best_h, best_c5 = subgrad_anneal(best_h, best_n, max_iter=5000,
                                         step0=0.03, momentum=0.5, top_k=1,
                                         anneal_temp=0.001, anneal_decay=0.998)
        _G_N = best_n
        h_new = slsqp_smooth(best_h, best_n, beta=10000, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        best_h, best_c5 = pairwise_swap_search(best_h, best_n, max_rounds=200, rng=rng)
        print("  C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
