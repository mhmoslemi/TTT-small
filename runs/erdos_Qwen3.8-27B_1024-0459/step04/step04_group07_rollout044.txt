import numpy as np
from scipy.optimize import minimize
import time

_G_N = 256
_G_BETA = 100.0
_G_DX = 2.0 / 256
_G_PEN = 0.0
_G_TGT = 128.0
_G_FOCUS = -1
_G_START = 0.0
_G_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)


def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    return h


def _obj(x):
    n = _G_N
    dx = _G_DX
    beta = _G_BETA
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = corr - m
    e = np.exp(np.clip(beta * s, -400, 400))
    val = (m + np.log(np.sum(e)) / beta) * dx
    if _G_PEN > 0:
        val += _G_PEN * (np.sum(x) - _G_TGT) ** 2
    if _G_FOCUS >= 0 and _G_FOCUS < len(corr):
        val += 0.005 * corr[_G_FOCUS] * dx
    return float(val)


def slsqp_run(h0, n, beta, maxiter=300, focus=-1):
    global _G_N, _G_BETA, _G_DX, _G_PEN, _G_TGT, _G_FOCUS
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    _G_BETA = float(beta)
    _G_PEN = 0.0
    _G_FOCUS = focus
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    try:
        res = minimize(_obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    _G_PEN = 0.0
    _G_FOCUS = -1
    return h


def lbfgsb_run(h0, n, beta, maxiter=300, pen=200.0, focus=-1):
    global _G_N, _G_BETA, _G_DX, _G_PEN, _G_TGT, _G_FOCUS
    _G_N = n

    _G_DX = 2.0 / n

    _G_PEN = float(pen)
    _G_FOCUS = focus
    bounds = [(0.0, 1.0)] * n

        h = project_feasible(h0, n)
    _G_PEN = 0.0
    _G_FOCUS = -1
    return h


def coord_descent(h, n, max_sweeps=50, rng=None, tlimit=None):
    if rng is None:
        rng = np.random.RandomState(0)
    c5 = compute_c5(h)
    h_best = h.copy()
    steps = [0.0, 1.0, 0.2, -0.2, 0.1, -0.1, 0.05, -0.05, 0.02, -0.02]
    fine_steps = [0.0, 1.0, 0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    for sweep in range(max_sweeps):
        if tlimit is not None and tlimit() < 2.0:
            break
        improved = False
        order = rng.permutation(n)
        use_fine = sweep >= max_sweeps // 2
        trial_steps = fine_steps if use_fine else steps
        for i in order:
            if tlimit is not None and tlimit() < 1.0:
                break
            v0 = h_best[i]
            for d in trial_steps:
                if d == 0.0:
                    v_try = 0.0 if v0 > 0.5 else 1.0
                elif d == 1.0:
                    v_try = 1.0 if v0 < 0.5 else 0.0
                else:
                    v_try = v0 + d
                if v_try < 0.0 or v_try > 1.0:
                    continue
                if abs(v_try - v0) < 1e-10:
                    continue
                trial = h_best.copy()
                trial[i] = v_try
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 1e-12:
                    c5 = c
                    h_best = trial.copy()
                    improved = True
        if not improved and sweep >= 3:
            break
    return h_best, c5
