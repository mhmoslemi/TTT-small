def pair_swap_local_search(h, n, rng, max_sweeps=20):
    """Try all pair swaps (1 at i, 0 at j) and accept if improves."""
    h = h.copy()
    # Make binary
    h[h >= 0.5] = 1.0
    h[h < 0.5] = 0.0
    # Fix sum
    s = np.sum(h)
    target = n // 2
    if s > target:
        ones = np.where(h == 1.0)[0]
        flip = rng.choice(len(ones), int(s - target), replace=False)
        h[ones[flip]] = 0.0
    elif s < target:
        zeros = np.where(h == 0.0)[0]
        flip = rng.choice(len(zeros), int(target - s), replace=False)
        h[zeros[flip]] = 1.0
    
    c5_best = compute_c5(h)
    h_best = h.copy()
    
    for sweep in range(max_sweeps):
        if remaining_time() < 3.0:
            break
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        # Try random subset of swaps
        num_tries = min(50, len(ones) * len(zeros) // 10)
        for _ in range(num_tries):
            if remaining_time() < 2.0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            c5_try = compute_c5(h)
            if c5_try < c5_best - 1e-14:
                c5_best = c5_try
                h_best = h.copy()
                improved = True
                # Update lists
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        if not improved and sweep > 3:
            break
    
    return h_best, c5_best

def compute_c5(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

def subgrad_compute(h, n, top_k=10, thresh_frac=0.95):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    center = n - 1
    threshold = np.max(corr) * thresh_frac
    active = np.where(corr >= threshold)[0]
    top_idx = np.argsort(corr)[-top_k:]
    active = np.unique(np.concatenate([active, top_idx]))
    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - center
        if offset >= 0:
            n_valid = n - offset
            if n_valid <= 0:
                continue
            grad[offset:offset + n_valid] += (1.0 - h[:n_valid])
            grad[:n_valid] -= h[offset:offset + n_valid]
        else:
            m = -offset
            n_valid = n - m
            if n_valid <= 0:
                continue
            grad[:n_valid] += (1.0 - h[m:m + n_valid])
            grad[m:m + n_valid] -= h[:n_valid]
    return grad, corr

            best_c = c5_best
            for dv in deltas:
                new_v = np.clip(h[i] + dv, 0.0, 1.0)
                if abs(new_v - h[i]) < 1e-12:
                    continue
                h_trial = h.copy()
                h_trial[i] = new_v
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < best_c - 1e-14:
                    best_c = c5_t
                    best_val = new_v
            if abs(best_val - h[i]) > 1e-12:
                h[i] = best_val
                h = project_feasible(h, n)
                c5_best = compute_c5(h)
                any_imp = True
        if not any_imp and sweep > 2:
            break
    return h, c5_best


def optimize_at_resolution(h, n, rng, time_frac=1.0):
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    c5_cur = compute_c5(h)

    # Binary local search
    h_bi = np.where(h >= 0.5, 1.0, 0.0)
    h_bi = project_feasible(h_bi, n)
    h_bi, c5_bi = binary_local_search(h_bi, n, rng, max_iter=5000, sa_beta=1200.0)
    if c5_bi < c5_cur:
        c5_cur = c5_bi
        h = h_bi.copy()

    # Subgradient
    for tk in [8, 15, 30]:
        if remaining_time() < 4.0:
            break
        h_sg, c5_sg = subgrad_optimize(h, n, max_iter=250, step0=0.08, momentum=0.85, top_k=tk)
        if c5_sg < c5_cur:
            c5_cur = c5_sg
            h = h_sg.copy()

    # SLSQP
    h_sq, c5_sq = slsqp_optimize(h, n, beta_vals=[200, 800, 3000, 10000], max_iter=250)
    if c5_sq < c5_cur:
        c5_cur = c5_sq
        h = h_sq.copy()

    # Coord descent
    if remaining_time() > 5.0:
        h_cd, c5_cd = coord_descent(h, n, rng, max_sweeps=8)
        if c5_cd < c5_cur:
            c5_cur = c5_cd
            h = h_cd.copy()

    return h, c5_cur
