def compute_corr_array(h):
    """Full correlation of h with 1-h."""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n):
    """Convert to binary with exactly n//2 ones."""
    h_bin = (h > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        # Remove excess ones (those with lowest marginal benefit)
        ones_idx = np.where(h_bin == 1)[0]
        # Remove ones that contribute most to worst shifts
        corr = np.correlate(h_bin, 1-h_bin, 'full')
        worst_shift = np.argmax(corr) - (n-1)
        # Prefer removing ones that are in bad pairs at worst shifts
        scores = np.zeros(n)
        for d in np.argsort(-corr)[:5]:
            dd = d - (n-1)
            if dd >= 0:
                for i in range(n-dd):
                    if h_bin[i] == 1 and h_bin[i+dd] == 0:
                        scores[i] += 1
            elif dd < 0:
                for i in range(-dd, n):
                    if h_bin[i] == 1 and h_bin[i+dd] == 0:
                        scores[i] += 1
        remove_order = ones_idx[np.argsort(-scores[ones_idx])]
        h_bin[remove_order[:n_ones-target]] = 0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        scores = np.zeros(n)
        corr = np.correlate(h_bin, 1-h_bin, 'full')
        for d in np.argsort(-np.abs(corr))[:5]:
            dd = d - (n-1)
            if dd > 0:
                for i in range(n-dd):
                    if h_bin[i] == 0 and h_bin[i+dd] == 1:
                        scores[i] += 1
            elif dd < 0:
                for i in range(-dd, n):
                    if h_bin[i] == 0 and h_bin[i+dd] == 1:
                        scores[i] += 1
        add_order = zeros_idx[np.argsort(-scores[zeros_idx])]
        h_bin[add_order[:target-n_ones]] = 1
    return h_bin

def slsqp_refine(h, n, maxiter=300):
    """SLSQP refinement for fractional solutions."""
    dx_val = 2.0 / n
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        beta = 5000.0
        shifted = corr - m
        cl = np.clip(beta * shifted, -50, 50)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_val

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        beta = 5000.0
        shifted = corr - m
        cl = np.clip(beta * shifted, -50, 50)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] > 1e-14:
                d = k - (n - 1)
                if d >= 0:
                    npairs = n - d
                    if npairs > 0:
                        g[d:d + npairs] += (1.0 - x[:npairs])
                        g[:npairs] -= x[d:d + npairs]
                else:
                    ao = -d
                    npairs = n - ao
                    if npairs > 0:
                        g[:npairs] += (1.0 - x[ao:ao + npairs])
                        g[ao:ao + npairs] -= x[:npairs]
        return g * dx_val

    def sum_con(x):
        return np.sum(x) - target

    constraint = {'type': 'eq', 'fun': sum_con, 'jac': lambda x: np.ones(n)}
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=[constraint],
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    budget = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, budget - (time.time() - t_start) - 3.0)

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
    except Exception:
        parent_n = 312
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Try multiple n values
    n_values = [312, 316, 320, 324, 328, 332, 336, 340, 344, 348]
    # Prioritize n values close to parent
    n_values = sorted(set(n_values), key=lambda x: abs(x - parent_n))

    time_per_n = 90  # seconds per n value (will adjust dynamically)

    for n_try in n_values:
        if remaining() < 30:
            break
        if n_try % 2 != 0:
            n_try += 1

        # Initialize for this n
        if n_try == parent_n:
            h_init = parent_h.copy()
        else:
            h_init = interp_to(parent_h, parent_n, n_try)
        h_bin = make_binary(h_init, n_try, rng)

        # Also try a random initialization for diversity
        if n_try != parent_n and rng.rand() < 0.3:
            h_rand = np.zeros(n_try)
            ones_positions = rng.choice(n_try, n_try // 2, replace=False)
            h_rand[ones_positions] = 1.0
            h_bin2 = h_rand
            # Quick local search on random init
            h_r, c5_r = local_search_1swap(h_bin2, n_try, max_iters=5000,
                                           time_limit=8, rng=np.random.RandomState(seed + n_try))
            h_b, c5_b = local_search_1swap(h_bin, n_try, max_iters=5000,
                                           time_limit=8, rng=np.random.RandomState(seed + n_try + 1))
            if c5_b < c5_r:
                h_bin = h_b
            else:
                h_bin = h_r

        # Phase 1: Intensive 1-swap local search
        t_phase = min(40, max(10, remaining() / 3))
        h1, c5_1, h_cur, corr_cur = local_search_1swap(h_bin, n_try,
                                                       max_iters=200000,
                                                       time_limit=t_phase,
                                                       rng=rng)
        c5_cur = c5_1

        # Phase 2: 2-swap local search
        if remaining() > 20:
            h2, c5_2 = local_search_2swap(h1, n_try, max_iters=80000,
                                          time_limit=min(25, max(5, remaining() / 4)), rng=rng)
            if c5_2 < c5_cur:
                c5_cur = c5_2
                h1 = h2

        # Phase 3: Simulated annealing
        if remaining() > 20:
            h3, c5_3 = simulated_annealing(h1, n_try, max_iters=60000,
                                           time_limit=min(30, max(5, remaining() / 4)),
                                           rng=rng, T_start=3.0, T_end=0.05)
            if c5_3 < c5_cur:
                c5_cur = c5_3
                h1 = h3

        # Phase 4: More 1-swap local search (polish)
        if remaining() > 10:
            h4, c5_4, _, _ = local_search_1swap(h1, n_try, max_iters=100000,
                                                 time_limit=min(20, max(5, remaining() / 3)),
                                                 rng=rng)
            if c5_4 < c5_cur:
                c5_cur = c5_4
                h1 = h4

        # Phase 5: Block flip
        if remaining() > 8:
            h5, c5_5 = block_flip_search(h1, n_try, max_iters=30000,
                                         time_limit=min(12, max(3, remaining() / 3)), rng=rng)
            if c5_5 < c5_cur:
                c5_cur = c5_5
                h1 = h5

        # Phase 6: SLSQP fractional refinement
        if remaining() > 8:
            h_cont = h1.copy()
            h_frac = slsqp_refine(h_cont, n_try, maxiter=400)
            c5_frac = compute_c5(h_frac)
            if c5_frac < c5_cur:
                c5_cur = c5_frac
                h1 = h_frac

        # Update global best
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h1.copy()
            best_n = n_try

        # Also check the binary version even if fractional was better
        h_bin_final = make_binary(h1, n_try, rng)
        c5_bin_final = compute_c5(h_bin_final)
        if c5_bin_final < best_c5:
            best_c5 = c5_bin_final
            best_h = h_bin_final.copy()
            best_n = n_try

    # Final intensive refinement on best solution
    n_final = best_n
    h_final = best_h.copy()

    # If best is fractional, also try its binary version
    h_bin_fin = (h_final > 0.5).astype(float)
    n_ones = int(h_bin_fin.sum())
    if n_ones != n_final // 2:
        h_bin_fin = make_binary(h_final, n_final, rng)
    c5_bin_fin = compute_c5(h_bin_fin)

    # Run more local search on whichever is better
    if c5_bin_fin < best_c5 + 0.001:
        h_work = h_bin_fin.copy()
        t_rem = remaining()
        if t_rem > 30:
            h_w, c5_w, _, _ = local_search_1swap(h_work, n_final,
                                                 max_iters=300000,
                                                 time_limit=int(min(50, t_rem * 0.4)),
                                                 rng=rng)
            if c5_w < best_c5:
                best_c5 = c5_w
                best_h = h_w.copy()
        if remaining() > 20:
            h_w, c5_w = local_search_2swap(h_work, n_final,
                                           max_iters=100000,
                                           time_limit=int(min(30, remaining() * 0.3)),
                                           rng=rng)
            if c5_w < best_c5:
                best_c5 = c5_w
                best_h = h_w.copy()
        if remaining() > 15:
            h_w, c5_w = simulated_annealing(h_work, n_final,
                                            max_iters=80000,
                                            time_limit=int(min(25, remaining() * 0.4)),
                                            rng=rng, T_start=2.0, T_end=0.03)
            if c5_w < best_c5:
                best_c5 = c5_w
                best_h = h_w.copy()

    # If best is fractional, polish with SLSQP
    if np.any((best_h > 0) & (best_h < 1)) and remaining() > 10:
        h_polish = slsqp_refine(best_h, best_n, maxiter=500)
        c5_polish = compute_c5(h_polish)
        if c5_polish < best_c5:
            best_c5 = c5_polish
            best_h = h_polish.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
