def project_feasible(h, n):

def smooth_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    clamped = np.clip(beta * shifted, -50.0, 50.0)
    ex = np.exp(clamped)
    logsum = np.log(np.sum(ex))
    obj = (m + logsum / beta) * dx
    w = ex / np.sum(ex) * dx / beta
    g = np.zeros(n)

        _, g = smooth_obj_and_grad(x, beta)
        return g

        return np.sum(x) - n / 2.0

        return np.ones(n)

        h_r, c5_r = binary_sa_multi(h_rand, n, time_limit=sa_time, rng=np.random.RandomState(seed + 13))
        if remaining() > 15:
            h_r_frac, c5_r_frac = fractional_refine(h_r.astype(float), n, [5000, 20000])
            if c5_r_frac < best_c5:
                best_c5 = c5_r_frac
                best_h = h_r_frac.copy()
                best_n = n
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n

    # Phase 8: Final intensive fractional refinement
    n = best_n
    h = best_h.copy()
    if remaining() > 15:
        for beta in [10000, 50000, 200000]:
            if remaining() < 8:
                break
            h_new = slsqp_run(h, n, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 9: Final projected gradient
    if remaining() > 15:
        h_pg, c5_pg = projected_gradient(best_h, best_n, beta=100000, max_iters=3000, lr=0.02)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 10: Final top-k equalize
    if remaining() > 8:
        h_eq, c5_eq = top_k_equalize(best_h, best_n, top_k=8, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5:
        best_c5 = final_c5
    else:
        best_h = project_feasible(best_h, best_n)
        best_c5 = min(best_c5, compute_c5(best_h))

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
