def subgrad_topk(h, n, top_k):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if top_k >= n_corr:

    else:
        top_indices = np.argsort(corr)[-top_k:]
        top_indices = top_indices[np.argsort(corr[top_indices])[::-1]]
    grad = np.zeros(n)

        offset = int(k_idx) - (n - 1)
        if offset >= 0:

            if n_valid <= 0:

            grad[m:m + n_valid] -= h[:n_valid]
    return grad, top_indices, corr

            h_opt, c5_opt = do_subgrad(h_opt, n, max_iter=80, step0=0.05, momentum=0.85, top_k_init=3, top_k_max=12)
            h_opt, c5_opt = do_slsqp_polish(h_opt, n, [10000, 30000], max_iter_per_beta=150)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                print("  Restart {}: C5={:.10f}".format(restart, best_c5))
            else:
                print("  Restart {}: C5={:.10f} (no improvement)".format(restart, c5_opt))

    # Phase 8: Multi-resolution
    if remaining_time() > 100.0:
        print("\n--- Phase 8: Multi-resolution ---")
        for n_try in [192, 320, 384]:
            if remaining_time() < 40.0:
                break
            h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
            c5_h = compute_c5(h_interp)
            h_interp, c5_h = do_slsqp_polish(h_interp, n_try, [100, 500, 2000, 8000], max_iter_per_beta=200)
            h_interp, c5_h = do_subgrad(h_interp, n_try, max_iter=120, step0=0.08, momentum=0.85, top_k_init=3, top_k_max=15)
            h_interp, c5_h = do_slsqp_polish(h_interp, n_try, [5000, 20000], max_iter_per_beta=200)
            if c5_h < best_c5:
                best_c5 = c5_h
                best_h = h_interp.copy()
                best_n = n_try
                print("  n={}: C5={:.10f}".format(n_try, best_c5))
            else:
                print("  n={}: C5={:.10f} (no improvement)".format(n_try, c5_h))

    # Phase 9: Final intensive polish
    if remaining_time() > 20.0:
        print("\n--- Phase 9: Final polish ---")
        best_h, best_c5 = do_slsqp_polish(best_h, best_n, [20000, 50000, 100000], max_iter_per_beta=250)
        best_h, best_c5 = swap_search(best_h, best_n, max_rounds=2000, rng=rng)
        if remaining_time() > 8.0:
            best_h, best_c5 = do_slsqp_polish(best_h, best_n, [50000, 100000, 200000], max_iter_per_beta=200)
        print("  Final: C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
