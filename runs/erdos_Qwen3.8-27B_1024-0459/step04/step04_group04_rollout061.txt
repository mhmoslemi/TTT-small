import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_sum_box(h, target):
    n = len(h)
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_p = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-13:
        mask = (h_p > 1e-10) & (h_p < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h_p[mask] += diff / cnt
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1
    return h_bin


def thue_morse_binary(n, rng):
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    return make_binary(tm, n, rng)


def golden_binary(n, rng):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    g = np.array([1.0 if (i * phi) % 1.0 < 0.5 else 0.0 for i in range(n)])
    return make_binary(g, n, rng)


def alternating_binary(n, rng):
    alt = np.array([1.0 if (i % 2 == 0) else 0.0 for i in range(n)])
    return make_binary(alt, n, rng)


def block_binary(n, block_size, rng):
    h = np.zeros(n)
    in_block = True
    pos = 0
    while pos < n:
        size = min(block_size, n - pos)
        if in_block:
            h[pos:pos+size] = 1.0
        in_block = not in_block
        pos += size
    return make_binary(h, n, rng)


def sa_binary(h, n, time_limit=60, seed=42):
    rng = np.random.RandomState(seed)
    h = make_binary(h, n, rng)
    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()

    T = 5.0
    alpha = 0.9993
    start = time.time()
    stall = 0
    max_stall = 8000

    while time.time() - start < time_limit and remaining() > 5 and T > 0.05:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        delta = new_max - cur_max
        if delta <= 0 or (T > 0.05 and rng.rand() < np.exp(-delta / T)):
            cur_max = new_max
            corr = new_corr
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
        T *= alpha
        if stall > max_stall:
            T = max(T, 3.0)
            stall = 0

    return best_h, best_max * dx_val


def guided_sa_binary(h, n, time_limit=60, seed=42):

    T = 4.0
    alpha = 0.999
    start = time.time()
    stall = 0
    max_stall = 6000

    while time.time() - start < time_limit and remaining() > 5 and T > 0.05:
        top_idx = np.argsort(-corr)[:15]

        zeros = np.where(h == 0)[0]

def smooth_obj(x):
    n = _N

    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -700, 700)
    ex = np.exp(cl)
    s_ex = np.sum(ex)
    return (m + np.log(s_ex) / beta) * dx_val


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -700, 700)
    ex = np.exp(cl)
    s_ex = np.sum(ex)
    w = ex / s_ex

    g = np.zeros(n)
    active = w > 1e-12
    for k_idx in np.where(active)[0]:
        wk = w[k_idx]
        k = int(k_idx)
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off+np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off+np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:ao+np_])
            g[ao:ao+np_] -= wk * x[:np_]
    return g * dx_val

def slsqp_smooth(h, n, beta, maxiter=300):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_sum_box(res.x, n / 2.0)
    except Exception:
        return h


def projected_grad(h, n, time_limit=60):
    """Accelerated projected gradient descent for smooth max."""
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    y_prev = h.copy()
    mu = 0.5

    for beta in [50, 200, 800, 3000]:
        if time.time() - start > time_limit:
            break
        global _N, _BETA, _TGT
        _N = n
        _TGT = n / 2.0
        _BETA = beta

        step_start = 1.0 / np.sqrt(n)
        for it in range(300):
            if time.time() - start > time_limit or remaining() < 5:
                break
            _, g = smooth_obj_grad_pair(h)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            step = step_start / (1.0 + 0.05 * it)
            h_new = h - step * g
            h_new = project_sum_box(h_new, n / 2.0)
            # Nesterov momentum
            h_new = mu * h_new + (1.0 - mu) * y_prev
            h_new = project_sum_box(h_new, n / 2.0)

            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
            y_prev = h.copy()
            h = h_new

    return best_h, best_c5
