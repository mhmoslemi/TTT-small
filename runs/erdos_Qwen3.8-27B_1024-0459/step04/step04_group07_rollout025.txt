    """Compute gradient of log-sum-exp approximation + penalty."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    v = corr - M
    bv = np.clip(beta * v, -500, 500)

    n = len(h)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c else 0.0
    return h

def make_rand_bin(rng, n):

    idx = rng.permutation(n)

def interp_to_n(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    xp = (np.arange(n) + 0.5) * 2.0 / n
    xn = (np.arange(new_n) + 0.5) * 2.0 / new_n
    return np.interp(xn, xp, h)

def lsesp_compute(h, beta, dx, penalty=0.0):
    """Compute log-sum-exp objective value and gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    v = corr - M
    bv = np.clip(beta * v, -500, 500)
    exp_bv = np.exp(bv)
    sum_exp = np.sum(exp_bv)
    obj = (M + np.log(sum_exp) / beta) * dx
    w = exp_bv / sum_exp

    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        m = k - (n - 1)
        wk = w[k]
        js = max(0, -m)
        je = min(n, n - m)
        if js < je:
            grad[js:je] += wk * (1.0 - h[js + m:je + m])
        js = max(0, m)
        je = min(n, n + m)
        if js < je:
            grad[js:je] -= wk * h[js - m:je - m]
    grad *= dx

    if penalty > 0:
        diff = np.sum(h) - n / 2.0
        obj += penalty * diff * diff
        grad += 2.0 * penalty * diff

    return obj, grad

def lbfgs_optimize(h0, n, beta, maxiter=400, penalty=500.0):
    """L-BFGS-B optimization with analytic gradient of log-sum-exp."""
    dx = 2.0 / n
    h0 = h0.copy()

    def obj_fn(x):
        val, _ = lsesp_compute(x, beta, dx, penalty)
        return val

    def grad_fn(x):
        _, g = lsesp_compute(x, beta, dx, penalty)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h0, jac=grad_fn, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def topk_optimize(h0, n, topk=3, maxiter=300):

    m = k_focus - (n - 1)

        g *= dx

    c5 = compute_c5(h)

    for sweep in range(max_sweeps):

            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if get_time_left() < 1.5:
                break
            v0 = h_best[i]
            candidates = [0.0, 1.0,
                         np.clip(v0 + 0.1, 0, 1), np.clip(v0 - 0.1, 0, 1),
                         np.clip(v0 + 0.05, 0, 1), np.clip(v0 - 0.05, 0, 1),
                         np.clip(v0 + 0.2, 0, 1), np.clip(v0 - 0.2, 0, 1),
                         np.clip(v0 + 0.3, 0, 1), np.clip(v0 - 0.3, 0, 1)]
            best_c = c5
            best_v = v0
            for v in candidates:
                if abs(v - v0) < 1e-9:
                    continue
                trial = h_best.copy()
                trial[i] = v
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-10:
                    best_c = c
                    best_v = v
            if best_v != v0:
                h_best[i] = best_v
                h_best = project_feasible(h_best, n)
                c5 = best_c
                improved = True
        if not improved and sweep >= 3:
            break
    return h_best, c5

        n = best_n
        for attempt in range(12):
            if get_time_left() < 15:
                break
            h_pert = best_h.copy()
            n_pert = rng.randint(5, 25)
            idxs = rng.choice(n, size=n_pert, replace=False)
            for idx in idxs:
                h_pert[idx] = np.clip(h_pert[idx] + rng.randn() * 0.35, 0.0, 1.0)
            h_pert = project_feasible(h_pert, n)
            c5_p = compute_c5(h_pert)

            if c5_p < best_c5 * 1.06:
                h_cur = h_pert.copy()
                c5_cur = c5_p
                for beta in [100, 300]:
                    if get_time_left() < 8:
                        break
                    h_new = lbfgs_optimize(h_cur, n, beta, maxiter=200, penalty=500.0)
                    c_new = compute_c5(h_new)
                    if c_new < c5_cur - 1e-12:
                        c5_cur = c_new
                        h_cur = h_new
                if c5_cur < best_c5 - 1e-9:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n
                    print("  Restart {}: C5={:.8f}".format(attempt, best_c5))

    # Phase 7: Final coordinate descent
    if get_time_left() > 10:
        best_h, best_c5 = coord_descent(best_h, best_n, max_sweeps=50, rng=rng)

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
