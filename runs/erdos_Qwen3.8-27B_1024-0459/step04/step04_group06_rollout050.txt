import numpy as np
from scipy.optimize import minimize
import time

_G_START = 0.0
_G_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)


def topk_exact_gradient(h, n, top_k):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_indices = np.argsort(corr)[-top_k:]
    top_indices = top_indices[np.argsort(corr[top_indices])[::-1]]
    grad = np.zeros(n)
    for k_idx in top_indices:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            n_valid = n - offset
            if n_valid <= 0:
                continue
            grad[offset:offset + n_valid] += (1.0 - h[:n_valid])
            grad[:n_valid] -= h[offset:offset + n_valid]
        else:
            m = -offset
            n_valid = n - m
            if n_valid <= 0:
                continue
            grad[:n_valid] += (1.0 - h[m:m + n_valid])
            grad[m:m + n_valid] -= h[:n_valid]
    return -grad, top_indices, corr


def smooth_obj(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    return (m + np.log(np.sum(exps)) / beta) * dx


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_random_start(rng, n):
    h = rng.rand(n)
    h = project_feasible(h, n)
    return h


def focused_gradient_descent(h, n, max_iter=2000, top_k=3, step0=0.05):
    c5_best = compute_c5(h)
    h_best = h.copy()
    step = step0
    patience = 0
    for it in range(max_iter):
        if remaining_time() < 5.0:
            break
        neg_grad, top_ids, corr = topk_exact_gradient(h, n, top_k)
        gn = np.linalg.norm(neg_grad)
        if gn < 1e-15:
            break
        direction = neg_grad / gn
        alpha = step
        improved = False
        for _ in range(15):
            h_trial = h + alpha * direction
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-14:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
            alpha *= 0.5
            if alpha < 1e-8:
                break
        if improved:
            patience = 0
            step = min(alpha * 1.5, 0.3)
        else:
            patience += 1
            step *= 0.6
            if patience > 20:
                break
    return h_best, c5_best


def slsqp_optimize(h, n, beta, max_iter=300):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_obj(x, n, beta)

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = float(np.max(corr))
        shifted = corr - m
        exps = np.exp(np.clip(beta * shifted, -500, 500))
        softmax = exps / np.sum(exps)
        grad = np.zeros(n)
        for li in range(len(corr)):
            if softmax[li] < 1e-12:
                continue
            offset = li - (n - 1)
            if offset >= 0:
                n_valid = n - offset
                if n_valid <= 0:
                    continue
                grad[offset:offset + n_valid] += (1.0 - x[:n_valid])
                grad[:n_valid] -= x[offset:offset + n_valid]
            else:
                m2 = -offset
                n_valid = n - m2
                if n_valid <= 0:
                    continue
                grad[:n_valid] += (1.0 - x[m2:m2 + n_valid])
                grad[m2:m2 + n_valid] -= x[:n_valid]
        return softmax * grad * dx * beta

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]

    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

    h_best = h.copy()
    if rng is None:
        rng = np.random.RandomState(42)
    deltas = [0.01, -0.01, 0.02, -0.02, 0.05, -0.05]
    for sweep in range(max_sweeps):
        if remaining_time() < 4.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if remaining_time() < 2.0:
                break
            for dv in [0.0, 1.0]:
                if abs(dv - h_best[i]) < 1e-8:
                    continue
                h_trial = h_best.copy()
                h_trial[i] = dv
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < c5_best - 1e-13:
                    c5_best = c5_t
                    h_best = h_trial.copy()
                    any_imp = True
            for d in deltas:
                h_trial = h_best.copy()
                h_trial[i] = np.clip(h_trial[i] + d, 0.0, 1.0)
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < c5_best - 1e-13:
                    c5_best = c5_t
                    h_best = h_trial.copy()
                    any_imp = True
        if not any_imp and sweep >= 2:
            break
    return h_best, c5_best
