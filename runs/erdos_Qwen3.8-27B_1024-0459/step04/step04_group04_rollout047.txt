Let me also think about whether the optimal solution is binary. For the continuous case with h ∈ [0,1], the objective is:
C₅ = max_k sum_i h[i]*(1-h[i+s]) * dx

For fixed k, the function sum_i h[i]*(1-h[i+s]) is concave in h (it's a sum of products). The max of concave functions is not concave, so we can't use convex optimization directly.

However, at the optimum, many values of h should be 0 or 1 (by a revelation principle type argument - if 0 < h[i] < 1, we can adjust it to reduce some shifts without increasing others... this isn't rigorously proven but empirically the best solutions seem binary).

4. Try swaps that reduce these specific contributions while not hurting other shifts too much

import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def _corr_grad_single(x, n, k):
    g = np.zeros(n)
    s = k - (n - 1)
    if s >= 0:
        a = n - s
        g[:a] += (1.0 - x[s:s + a])
        g[s:n] += (-x[:a])
    else:
        a = -s
        b = n - a
        g[a:n] += (1.0 - x[:b])
        g[:b] += (-x[a:n])
    return g


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    total = np.sum(ex)
    g = np.zeros(n)
    threshold = 1e-10
    for k in range(len(corr)):
        if ex[k] / total > threshold:
            g += (ex[k] / total) * _corr_grad_single(x, n, k)
    return g * dx_val


def sum_con(x):
    return np.sum(x) - _TGT


def sum_con_jac(x):
    return np.ones(_N)


def lbfgs_smooth(h, n, beta, maxiter=500, gtol=1e-14):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='L-BFGS-B',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': gtol, 'iprint': 0, 'maxfun': 5000})
        return project_feasible(res.x, n)
    except Exception:
        try:
            res = minimize(smooth_obj, h, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-18,
                                    'gtol': gtol, 'iprint': 0})
            return project_feasible(res.x, n)
        except Exception:
            return h


def slsqp_smooth(h, n, beta, maxiter=400):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        rng.shuffle(idx)
        h_bin[idx[:n_ones - target]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        rng.shuffle(idx)
        h_bin[idx[:target - n_ones]] = 1
    return h_bin


def sa_binary_smart(h, n, time_limit=60, seed=42):
    """Binary SA with smart restarts and focus on worst shifts."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1

    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()

    T = 10.0
    T_min = 0.05
    cool_start = 0.3
    cool_end = 0.95
    total_iters = max(2000, int(time_limit * 500))
    it = 0
    stall = 0
    stall_cap = 3000

    start = time.time()
    while it < total_iters:
        if time.time() - start > time_limit or remaining() < 5:
            break
        # Adaptive cooling
        frac = it / total_iters
        if frac < cool_start:
            T = 10.0
        elif frac < cool_end:
            T = 10.0 * (T_min / 10.0) ** ((frac - cool_start) / (cool_end - cool_start))
        else:
            T = T_min

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Smart move: sometimes pick from positions affecting worst shifts
        if frac > 0.4 and rng.rand() < 0.6:
            # Focus on worst shifts
            top_shifts = np.argsort(-corr)[:5]
            # Pick position near a worst-shift boundary
            best_s = int(top_shifts[0])
            s = best_s - (n - 1)
            if s >= 0:
                a = n - s
                contrib = h[:a] * (1.0 - h[s:s + a])
                pos = np.where(contrib == 1)[0]
                if len(pos) > 0:
                    i = pos[rng.randint(len(pos))]
                    j_cands = zeros
                else:
                    i = ones[rng.randint(len(ones))]
                    j_cands = zeros
            else:
                a = -s
                b = n - a
                contrib = h[a:a + b] * (1.0 - h[:b])
                pos = np.where(contrib == 1)[0] + a
                if len(pos) > 0:
                    i = pos[rng.randint(len(pos))]
                    j_cands = zeros
                else:
                    i = ones[rng.randint(len(ones))]
                    j_cands = zeros
            j = j_cands[rng.randint(len(j_cands))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

        if i == j:
            it += 1
            continue

        h[i], h[j] = 0.0, 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        diff = new_corr
        delta = new_max - cur_max

        if delta < 0 or rng.rand() < np.exp(-delta / max(T, T_min)):
            cur_max = new_max
            corr = new_corr
            if new_max < best_max - 1e-15:
                best_max = new_max
                best_h = h.copy()
                stall = 0
                # Restart temperature on new best
                if frac > 0.5:
                    T = max(T, 3.0)
            else:
                stall += 1
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1

        it += 1
        if stall > stall_cap:
            # Reheat
            T = max(T, 5.0)
            stall = 0

    return best_h, best_max * dx_val


def topk_descent(h, n, max_iters=500, top_k=8):
    """Top-k shift focused gradient descent with projection."""
    h = h.copy()
    best_c5 = compute_c5(h)
    dx_val = 2.0 / n

    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        m = float(np.max(corr))
        if m * dx_val >= best_c5 + 1e-8:
            h = h.__class__(best_h_stash if 'best_h_stash' in dir() else h)
            break

        # Focus on top-k worst shifts
        top_idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k in top_idx:
            g += _corr_grad_single(h, n, int(k))
        g /= len(top_idx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn

        improved = False
        for step in [0.05, 0.02, 0.01, 0.1, 0.005, 0.03, 0.008, 0.015]:
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-14:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            # Try smaller steps
            for step in [0.002, 0.001, 0.0005]:
                h_trial = h + step * d
                h_trial = project_feasible(h_trial, n)
                c_new = compute_c5(h_trial)
                if c_new < best_c5 - 1e-14:
                    h = h_trial
                    best_c5 = c_new
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def two_phase_refine(h, n, time_limit=100):
    """Multi-stage continuous refinement."""
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()

    # Phase A: L-BFGS-B with increasing beta
    for beta in [200, 500, 2000, 8000]:
        if time.time() - start > time_limit * 0.45 or remaining() < 10:
            break
        h_new = lbfgs_smooth(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase B: Top-k gradient descent
    if time.time() - start < time_limit * 0.65 and remaining() > 10:
        h_td, c5_td = topk_descent(h, n, max_iters=300, top_k=8)
        if c5_td < best_c5:
            best_c5 = c5_td
            best_h = h_td.copy()
            h = h_td.copy()

    # Phase C: More L-BFGS-B with higher beta
    if time.time() - start < time_limit * 0.85 and remaining() > 10:
        for beta in [5000, 20000, 50000]:
            if remaining() < 5:
                break
            h_new = lbfgs_smooth(h, n, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:

                h = h_new.copy()

    # Phase D: Final top-k
