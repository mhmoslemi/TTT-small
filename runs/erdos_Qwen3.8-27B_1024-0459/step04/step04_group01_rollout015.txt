def fast_binary_sa(h_bin, n, T_init, n_steps, rng, time_limit=60):
    """Simulated annealing on binary h with O(n) correlation updates."""
    h = h_bin.copy()
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5_val = cur_max
    best_h = h.copy()
    
    T = T_init
    start = time.time()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    # Precompute for fast updates
    ni = n - 1
    
    for step in range(n_steps):
        if time.time() - start > time_limit:
            break
        if remaining() < 10:
            break
        
        # Choose move
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Compute delta for all K
        # First factor i: K in [ni-i, ni+i], corr[K] += h[K-ni+i] - 1
        k1_lo, k1_hi = ni - i, ni + i
        idx1 = np.arange(k1_lo, k1_hi + 1)
        delta1 = h[idx1 - ni + i] - 1.0
        
        # First factor j: K in [ni-j, ni+j], corr[K] += 1 - h[K-ni+j]
        k2_lo, k2_hi = ni - j, ni + j
        idx2 = np.arange(k2_lo, k2_hi + 1)
        delta2 = 1.0 - h[idx2 - ni + j]
        
        # Second factor i: K in [i, i+ni], corr[K] -= h[i+ni-K]
        k3_lo, k3_hi = i, i + ni
        idx3 = np.arange(k3_lo, k3_hi + 1)
        delta3 = -h[i + ni - idx3]
        
        # Second factor j: K in [j, j+ni], corr[K] -= h[j+ni-K]

        corr_new[idx3] += delta3
