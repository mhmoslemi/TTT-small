_TGT = 0.0

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode='full')

    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):

        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _sm_obj(x):
    n = _N
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / _BETA) * (2.0 / n)

    """Optimize with explicit active set constraints."""
    # Variables: [h_0, ..., h_{n-1}, t]
    # Minimize t
    # Subject to: corr[k_i] - t*n/2 = 0 for each active shift (since corr*dx = corr*2/n = t)
    #             sum(h) = n/2
    #             0 <= h <= 1
    
    def objective(z):
        return z[-1]  # t
    
    def objective_grad(z):
        g = np.zeros(n+1)
        g[-1] = 1.0
        return g
    
    cons = []
    bounds = [(0.0, 1.0)] * n + [(-1.0, 1.0)]  # t is free
    
    # Active set constraints: corr[k_i] * 2/n = t
    def corr_k(h_vals, k):
        off = k - (n-1)
        if off >= 0:
            return np.sum(h_vals[:n-off] * (1 - h_vals[off:]))
        else:
            ao = -off
            return np.sum(h_vals[ao:] * (1 - h_vals[:n-ao]))
    
    for k in active_shifts:
        def con_active(z, k=k):
            h_vals = z[:n]
            return corr_k(h_vals, k) * 2.0 / n - z[-1]
        
        def con_active_jac(z, k=k):
            h_vals = z[:n]
            g = np.zeros(n+1)
            g[-1] = -1.0
            off = k - (n-1)
            if off >= 0:
                np_ = n - off
                g[off:off+np_] += (1.0 - h_vals[:np_]) * 2.0/n
                g[:np_] -= h_vals[off:off+np_] * 2.0/n
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h_vals[ao:ao+np_]) * 2.0/n
                g[ao:ao+np_] -= h_vals[:np_] * 2.0/n
            return g
        
        cons.append({'type': 'eq', 'fun': con_active, 'jac': con_active_jac})
    
    # Sum constraint
    def sum_con(z):
        return np.sum(z[:n]) - n/2.0
    
    def sum_con_jac(z):
        g = np.zeros(n+1)
        g[:n] = 1.0
        return g
    
    cons.append({'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac})
    
    h_init = h.copy()
    t_init = compute_c5(h)
    z0 = np.concatenate([h_init, [t_init]])
    
    try:

                       bounds=bounds, constraints=cons,

        if res.x[-1] < t_init:
