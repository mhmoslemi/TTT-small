def binary_sa(h, n, time_limit, rng):
    """Simulated annealing for binary optimization."""
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_val = np.max(corr) * dx
    best_val = cur_val
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    t_start = time.time()
    T = 1.0  # initial temperature
    cooling = 0.9999
    
    it = 0
    while time.time() - t_start < time_limit:
        it += 1
        # Pick random 1 and 0 to swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        new_val = np.max(corr) * dx

        if delta < 0 or rng.rand() < np.exp(-delta / T):
            cur_val = new_val

                best_h = h.copy()
            # Update ones/zeros
            ones = np.where(h == 1)[0]  # This is O(n) each time

        else:
            h[i] = 1
            h[j] = 0
        
        T *= cooling
        if T < 0.001:
            T = 0.001
    
    return best_h, best_val
