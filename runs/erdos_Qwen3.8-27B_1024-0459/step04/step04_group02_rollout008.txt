import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_N_WORK = 0
_BETA_WORK = 1000.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = np.argmin(h)
            else:
                idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def smooth_obj_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def _s_obj(x):
    return smooth_obj_grad(x, _N_WORK, _BETA_WORK)[0]

def _s_obj_grad(x):
    return smooth_obj_grad(x, _N_WORK, _BETA_WORK)[1]

def _s_sum(x):
    return np.sum(x) - _N_WORK / 2.0

def _s_sum_jac(x):
    return np.ones(_N_WORK)

def slsqp_refine(h, n, beta, maxiter):
    global _N_WORK, _BETA_WORK
    _N_WORK = n
    _BETA_WORK = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _s_sum, 'jac': _s_sum_jac}]
    try:
        res = minimize(_s_obj, h, jac=_s_obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-25, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            if compute_c5(trial) <= compute_c5(h) + 1e-15:
                return trial
    except Exception:
        pass
    return h.copy()

def coord_descent_1d(h, n, i, corr, dx):
    """Exact 1D minimization for coordinate i using the structure of the problem."""
    t0 = h[i]
    ncorr = 2 * n - 1
    
    # Compute A_k, B_k, C_k for each correlation term
    # g_k(t) = A_k + B_k*t + C_k*t^2
    # where the original corr[k] = base_k + A_k_contribution from h[i]
    
    # For offset 0 (k = n-1): h[i]*(1-h[i]) -> C=-1, B=1, base = corr[n-1] - h[i]*(1-h[i])
    # For offset off>0, 0<=i+off<n: h[i]*(1-h[i+off]) -> B += (1-h[i+off])
    # For offset off>0, 0<=i-off<n: (1-h[i-off])*h[i] -> B += (1-h[i-off])
    
    B = np.zeros(ncorr)
    C = np.zeros(ncorr)
    
    # Offset 0
    C[n-1] = -1.0
    B[n-1] = 1.0
    
    # Offsets where h[i] appears
    for off in range(1, n):
        k_pos = n - 1 + off  # h[i] * (1-h[i+off])
        k_neg = n - 1 - off  # (1-h[i-off]) * h[i]
        
        if 0 <= i + off < n:
            B[k_pos] += (1.0 - h[i + off])
        if 0 <= i - off < n:
            B[k_neg] += (1.0 - h[i - off])
    
    # Base: corr[k] - (B[k]*t0 + C[k]*t0^2)
    base = corr - B * t0 - C * t0 * t0
    
    # Now f(t) = max_k (base[k] + B[k]*t + C[k]*t^2) * dx
    # Find min over t in [0,1]
    
    def f_of_t(t):
        vals = base + B * t + C * t * t
        return np.max(vals) * dx
    
    # Candidate points
    candidates = [0.0, 1.0, t0]
    
    # Vertices of parabolas (C_k = -1)
    # Only k = n-1 has C = -1
    # vertex at B[n-1]/2 = 0.5
    if 0.0 <= 0.5 <= 1.0:
        candidates.append(0.5)
    
    # Intersections of the parabola (k=n-1) with active lines
    # The "active" lines are those with high base[k] + B[k]*t0
    # For efficiency, only check the top few
    top_k = np.argsort(base + B * t0 + C * t0 * t0)[::-1][:20]
    for k in top_k:
        if k == n - 1:
            continue
        # Intersection of g_{n-1}(t) and g_k(t):
        # base[n-1] + t - t^2 = base[k] + B[k]*t
        # -t^2 + (1-B[k])*t + (base[n-1]-base[k]) = 0
        # t^2 - (1-B[k])*t - (base[n-1]-base[k]) = 0
        a = 1.0
        b = -(1.0 - B[k])
        c = -(base[n-1] - base[k])
        disc = b*b - 4*a*c
        if disc >= 0:
            sq = np.sqrt(disc)
            t1 = (-b + sq) / (2*a)
            t2 = (-b - sq) / (2*a)
            if 0.0 <= t1 <= 1.0:
                candidates.append(t1)
            if 0.0 <= t2 <= 1.0:
                candidates.append(t2)
    
    # Coarse grid
    for j in range(1, 10):
        candidates.append(j / 10.0)
    
    # Evaluate all candidates
    best_t = t0
    best_val = f_of_t(t0)
    for t in candidates:
        t = max(0.0, min(1.0, t))
        v = f_of_t(t)
        if v < best_val - 1e-16:
            best_val = v
            best_t = t
    
    # Golden section refinement around best_t
    if abs(best_t - t0) > 1e-10:
        lo = max(0.0, best_t - 0.15)
        hi = min(1.0, best_t + 0.15)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = f_of_t(c1)
        f2 = f_of_t(c2)
        for _ in range(50):
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = f_of_t(c1)
            else:
                lo = c1
                c1 = c2
