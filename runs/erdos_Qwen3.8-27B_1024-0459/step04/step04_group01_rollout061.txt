```python
def corr_update_swap(corr, h, i, j, n):
    """Update correlation after swapping h[i] and h[j].
    Assumes h[i]=0, h[j]=1 (or vice versa). Returns updated corr.
    """
    # δ_i = +1 (0→1), δ_j = -1 (1→0)
    # For each lag d (represented as index in corr of length 2n-1):
    # corr[k] corresponds to lag d = k - (n-1)
    # ΔD(d) = [valid(i+d) ? (1-h[i+d]) : 0] - [valid(i-d) ? h[i-d] : 0]
    #         - [valid(j+d) ? (1-h[j+d]) : 0] + [valid(j-d) ? h[j-d] : 0]
    # where valid(x) means 0 ≤ x < n
    
    n_corr = 2 * n - 1
    for k in range(n_corr):
        d = k - (n - 1)
        delta = 0.0
        # Position i (0→1, δ=+1):
        idx = i + d
        if 0 <= idx < n:
            delta += (1.0 - h[idx])
        idx = i - d
        if 0 <= idx < n:
            delta -= h[idx]
        # Position j (1→0, δ=-1):
        idx = j + d
        if 0 <= idx < n:
            delta -= (1.0 - h[idx])
        idx = j - d
        if 0 <= idx < n:
            delta += h[idx]
        corr[k] += delta
    return corr
```

    lags = np.arange(n_corr) - (n - 1)  # shape (2n-1,)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_trial = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_trial)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():

def to_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones > n // 2:
        ones_idx = np.where(h_bin == 1)[0]
        if len(ones_idx) > n_ones - n // 2:
            remove = rng.choice(ones_idx, n_ones - n // 2, replace=False)
            h_bin[remove] = 0.0

def greedy_binary(h, n, time_limit=60, rng=None):

    h = h.copy()
    corr = compute_corr(h).copy()
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    start = time.time()

    zeros = np.where(h == 0)[0]

        k_worst = int(np.argmax(corr))

        best_delta = 0.0
        best_i = -1
        best_j = -1

        n_sample = min(len(zeros) * len(ones), 2000)
        for _ in range(n_sample):
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            delta = 0.0
            idx = i + d_worst
            if 0 <= idx < n:
                delta += (1.0 - h[idx])
            idx = i - d_worst
            if 0 <= idx < n:
                delta -= h[idx]
            idx = j + d_worst
            if 0 <= idx < n:
                delta -= (1.0 - h[idx])
            idx = j - d_worst
            if 0 <= idx < n:
                delta += h[idx]

                best_delta = delta
                best_i = i
                best_j = j

            break

def sa_binary(h, n, rng, time_limit=120):
    """Simulated annealing on binary solutions."""
    h = h.copy()
    corr = compute_corr(h).copy()
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx_val = 2.0 / n

    T_final = 0.3
    start = time.time()

    zeros = np.where(h == 0)[0]

    while time.time() - start < time_limit and remaining() > 10:
        elapsed = time.time() - start
        progress = min(1.0, elapsed / time_limit)
        T = T_init * (T_final / T_init) ** progress

        d_worst = k_worst - (n - 1)

                d1 += (1.0 - h[idx])

            if 0 <= idx < n:

            idx = j_c + d_worst

                d1 -= (1.0 - h[idx])
            idx = j_c - d_worst
            if 0 <= idx < n:
                d1 += h[idx]

def slsqp_smooth(h, n, beta, maxiter=400):
    """SLSQP with log-sum-exp smoothing."""
    n_pts = n
    dx_val = 2.0 / n_pts
    target = n_pts / 2.0
    bounds = [(0.0, 1.0)] * n_pts

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shift = corr - m
        cl = np.clip(beta * shift, -60, 60)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_val

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shift = corr - m
        cl = np.clip(beta * shift, -60, 60)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n_pts)
        nc = len(corr)
        for k in range(nc):
            if w[k] > 1e-14:
                d = k - (n_pts - 1)
                if d >= 0:
                    npairs = n_pts - d
                    g[d:d + npairs] += (1.0 - x[:npairs])
                    g[:npairs] -= x[d:d + npairs]
                else:
                    ae = -d
                    npairs = n_pts - ae
                    g[:npairs] += (1.0 - x[ae:ae + npairs])
                    g[ae:ae + npairs] -= x[:npairs]
        return g * dx_val

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n_pts)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

        if c5_greedy < best_c5:

            best_h = h_greedy.copy()
