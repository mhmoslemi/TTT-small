        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def random_balanced(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def binary_local_search(h, n, time_limit=60, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0
    dx = 2.0 / n
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve_rounds = 0
    while time.time() - start < time_limit and remaining_time() > 5:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_pairs = len(ones) * len(zeros)
        if n_pairs <= 60000:
            best_delta = 1e-10
            best_pair = None
            for i in ones:
                for j in zeros:
                    h[i] = 0.0
                    h[j] = 1.0
                    c_new = compute_c5(h)
                    h[i] = 1.0
                    h[j] = 0.0
                    delta = c_new - best_c5
                    if delta < best_delta:
                        best_delta = delta
                        best_pair = (i, j)
            if best_pair is not None:
                i, j = best_pair
                h[i] = 0.0
                h[j] = 1.0
                best_c5 = compute_c5(h)
                best_h = h.copy()
                improved = True
        else:
            n_trials = min(300, n_pairs)
            best_delta = 1e-10
            best_pair = None
            for _ in range(n_trials):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                c_new = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                delta = c_new - best_c5
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (i, j)
            if best_pair is not None:
                i, j = best_pair
                h[i] = 0.0
                h[j] = 1.0
                best_c5 = compute_c5(h)
                best_h = h.copy()
                improved = True
        if not improved:
            no_improve_rounds += 1
            if no_improve_rounds > 15:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            best_c5 = compute_c5(h)
    return best_h, best_c5


def coordinate_refine(h, n, max_sweeps=50):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved_sweep = False
        for i in range(n):
            old = h[i]
            cands = [0.0, 1.0, old]
            if 0.0 < old < 1.0:
                cands.extend([old * 0.5, old + (1.0 - old) * 0.5,
                              old - 0.25, old + 0.25])
            cands = [c for c in cands if 0.0 <= c <= 1.0]
            best_cand_c5 = compute_c5(h)
            best_cand_val = old
            for c in cands:
                h[i] = c
                h[i] = np.clip(c, 0.0, 1.0)
                h = project_feasible(h, n)
                c5_v = compute_c5(h)
                if c5_v < best_cand_c5 - 1e-12:
                    best_cand_c5 = c5_v
                    best_cand_val = c
                    improved_sweep = True
            h[i] = best_cand_val
            h = project_feasible(h, n)
            if best_cand_c5 < best_c5:
                best_c5 = best_cand_c5
        if not improved_sweep:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(384)
        best_h[:192] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Progressive SLSQP on parent resolution
    h = best_h.copy()
    for beta in [100, 300, 1000, 3000, 10000, 30000, 100000, 300000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Top-k equalization
    if remaining_time() > 10:
        h_eq, c5_eq = top_k_equalize(h, best_n, top_k=5, max_iters=600)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 3: More SLSQP after equalization
    for beta in [50000, 150000, 500000]:
        if remaining_time() < 15:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Multi-resolution search
    n_candidates = []
    for d in [-6, -4, -2, 2, 4, 6, 8, 12, 16, 24, 32, 48, 64]:
        nn = best_n + d
        if 160 <= nn <= 800 and nn % 2 == 0:
            n_candidates.append(nn)
    # Add some multiplicative changes
    for mult in [0.85, 0.9, 1.1, 1.2, 1.35, 1.5, 1.75, 2.0]:
        nn = int(best_n * mult)
        if 160 <= nn <= 800 and nn % 2 == 0:
            n_candidates.append(nn)
    n_candidates = sorted(set(n_candidates))
    n_candidates.sort(key=lambda x: abs(x - best_n))

    for n_t in n_candidates[:8]:
        if remaining_time() < 25:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)

        h_opt = h_feas
        for beta in [500, 5000, 50000]:
            if remaining_time() < 10:
                break
            h_opt = slsqp_opt(h_opt, n_t, beta, maxiter=500)
        c5_opt = compute_c5(h_opt)
        h_opt, c5_opt = top_k_equalize(h_opt, n_t, top_k=4, max_iters=300)

        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
            h = h_opt.copy()

    # Phase 5: Try alternative starting points
    alt_starts = []
    # Thue-Morse
    alt_starts.append(theta_morse(best_n))
    # Random
    alt_starts.append(random_balanced(best_n, rng))
    # Shifted parent
    h_shift = np.roll(best_h, 1)
    alt_starts.append(project_feasible(h_shift, best_n))

    for h_alt in alt_starts:
        if remaining_time() < 30:
            break
        h_opt = h_alt
        for beta in [1000, 10000, 100000]:
            if remaining_time() < 10:
                break
            h_opt = slsqp_opt(h_opt, best_n, beta, maxiter=500)
        c5_opt = compute_c5(h_opt)
        h_opt, c5_opt = top_k_equalize(h_opt, best_n, top_k=5, max_iters=300)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h = h_opt.copy()

    # Phase 6: Binary local search
    if remaining_time() > 30:
        h_bin = (best_h > 0.5).astype(float)
        n_half = best_n // 2
        n_ones = int(np.sum(h_bin))
        if n_ones != n_half:
            sort_idx = np.argsort(-h_bin)
            h_bin = np.zeros(best_n)
            h_bin[sort_idx[:n_half]] = 1.0

        sa_time = min(120, max(15, int(remaining_time() * 0.3)))
        h_b, c5_b = binary_local_search(h_bin, best_n, time_limit=sa_time, rng=rng)

        # Refine the binary result continuously
        h_b_c = h_b.copy()
        for beta in [5000, 50000, 200000]:
            if remaining_time() < 10:
                break
            h_b_c = slsqp_opt(h_b_c, best_n, beta, maxiter=500)
        c5_b_c = compute_c5(h_b_c)
        h_b_c, c5_b_c = top_k_equalize(h_b_c, best_n, top_k=5, max_iters=300)

        if c5_b_c < best_c5 - 1e-14:
            best_c5 = c5_b_c
            best_h = h_b_c.copy()
            h = h_b_c.copy()

        # Also try the raw binary value
        if c5_b < best_c5 - 1e-14:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 7: Final intensive refinement
    n = best_n
    h = best_h.copy()
    for beta in [100000, 300000, 1000000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining_time() > 8:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=6, max_iters=500)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 8: Coordinate refinement
    if remaining_time() > 10:
        h_cr, c5_cr = coordinate_refine(best_h, n, max_sweeps=30)
        if c5_cr < best_c5 - 1e-14:
            best_c5 = c5_cr
            best_h = h_cr.copy()

    # Final SLSQP polish
    if remaining_time() > 8:
        h_final = slsqp_opt(best_h, n, 500000, maxiter=500)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate with evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, n)
