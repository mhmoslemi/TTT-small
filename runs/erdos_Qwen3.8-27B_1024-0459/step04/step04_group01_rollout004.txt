        worst_idx = int(np.argmax(corr))
        
        # Try to improve
        improved = False
        # Sample candidate swaps
        n_attempts = 100
        for _ in range(n_attempts):
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            if a == b:
                continue
            
            # Compute new max
            # Quick check: will this reduce the worst?
            # Just compute delta for worst shift
            # For efficiency, compute full delta
            delta = binary_swap_delta(h, a, b, n)
            new_corr = corr + delta
            new_max = float(np.max(new_corr))
            
            if new_max < cur_max - 1e-12:
                # Accept
                h[a] = 0
                h[b] = 1
                corr = new_corr
                cur_max = new_max
                # Update ones/zeros
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1

                h[b] = 1

                cur_max = float(np.max(corr))
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                stall = 0
        
        it += 1
        
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
    
    return best_h, best_max
