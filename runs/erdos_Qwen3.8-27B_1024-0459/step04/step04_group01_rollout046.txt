def compute_c5(h):

def project_binary(h, n):
    """Project to binary with exactly n/2 ones."""
    h = np.array(h, dtype=float).copy()
    # Sort by value, threshold at middle
    sorted_idx = np.argsort(-h)
    h2 = np.zeros(n)
    h2[sorted_idx[:n//2]] = 1.0
    return h2

def project_feasible(h, n):
    """Project to [0,1] with sum = n/2."""
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        h_trial = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_trial)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(20):

        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

    """Simulated annealing for binary optimization."""
    h = h_bin.copy()
    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx_val
    best_h = h.copy()
    T = T_init
    start = time.time()
    n_corr = 2 * n - 1
    it = 0

        it += 1
        # Pick a random 1 and a random 0

_G_BETA = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        ht = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(ht)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)

def to_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    h = np.array(h, dtype=float).copy()
    target = n // 2
    one_count = int(np.sum(h > 0.5))
    if one_count == target:
        return (h > 0.5).astype(float)
    # Use threshold
    sorted_idx = np.argsort(-h)
    out = np.zeros(n)
    out[sorted_idx[:target]] = 1.0
    return out

def sa_optimize(h_bin, n, time_limit, rng, T_init=0.008, T_min=0.0005, cool=0.99998):
    """Simulated annealing for binary optimization."""
    h = h_bin.astype(float).copy()
    dx_val = 2.0 / n
    corr = compute_corr(h).astype(float)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx_val
    best_h = h.copy()
    T = T_init
    start = time.time()
    it = 0
    stall_count = 0
    best_stall = best_c5
    
    while time.time() - start < time_limit and remaining() > 5:
        it += 1
        # Find 1s and 0s
        ones_mask = h == 1.0
        zeros_mask = h == 0.0
        ones = np.where(ones_mask)[0]
        zeros = np.where(zeros_mask)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i0 = ones[rng.randint(len(ones))]
        j0 = zeros[rng.randint(len(zeros))]
        
        # Try swap
        h[i0] = 0.0
        h[j0] = 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        delta = (new_max - cur_max) * dx_val
        
        accept = False
        if delta < -1e-15:
            accept = True
        elif delta < 1e-8:
            accept = rng.rand() < 0.5  # Near-neutral, accept 50%
        else:
            accept = rng.rand() < np.exp(-delta / max(T, 1e-12))
        
        if accept:
            corr = new_corr
            cur_max = new_max
            if new_max * dx_val < best_c5:
                best_c5 = new_max * dx_val
                best_h = h.copy()
                stall_count = 0
            else:
                stall_count += 1
        else:
            h[i0] = 1.0
            h[j0] = 0.0
            stall_count += 1
        
        # Reheat if stalled
        if stall_count % 50000 == 0 and T < 0.003:
            T = min(T * 2.0, 0.008)
            stall_count = 0
        
        T = max(T * cool, T_min)
    
    return best_h, best_c5

def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        if npairs > 0:
            g[off:off+npairs] += (1.0 - x[:npairs])
            g[:npairs] -= x[off:off+npairs]
    else:
        ao = -off
        npairs = n - ao
        if npairs > 0:
            g[:npairs] += (1.0 - x[ao:ao+npairs])
            g[ao:ao+npairs] -= x[:npairs]
    return g

def smooth_obj(x):
    n = _G_N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _G_BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val

def smooth_grad(x):
    n = _G_N
    dx_val = 2.0 / n
    beta = _G_BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-14:
            g += w[k] * corr_grad_per_shift(x, n, k)
    return g * dx_val

def sum_con(x):
    return np.sum(x) - _G_TGT

def sum_con_jac(x):
    return np.ones(_G_N)

def slsqp_smooth(h, n, beta, maxiter=300):
    global _G_N, _G_BETA, _G_TGT
    _G_N = n
    _G_TGT = n / 2.0
    _G_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
