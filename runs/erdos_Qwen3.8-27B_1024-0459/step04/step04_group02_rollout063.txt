```python
def subgrad_step(h, n, n_active=5, step=0.01):
    corr = compute_corr(h)
    max_corr = np.max(corr)
    # Find active lags
    threshold = max_corr - 0.5 * (2.0 / n)
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        # Keep top n_active
        active_sorted = np.argsort(corr[active])[::-1][:n_active]
        active = active[active_sorted]
    
    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]
    
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return None
    return grad / gnorm
```

```python
def compute_grad_matrix(h, n):
    """Compute gradient matrix G where G[k, i] = dc_k/dh_i for all k, i."""
    m = 2 * n - 1
    G = np.zeros((m, n))
    
    # offset >= 0: k = n-1+offset, k ranges from n-1 to 2n-2
    for offset in range(0, n):
        k = n - 1 + offset
        npairs = n - offset
        # "first" positions: i in [0, npairs-1]
        G[k, :npairs] += (1.0 - h[offset:offset+npairs])
        # "second" positions: i in [offset, n-1]
        G[k, offset:offset+npairs] += (-h[:npairs])
    
    # offset < 0: k = n-1+offset, k ranges from 0 to n-2
    for aoff in range(1, n):
        k = n - 1 - aoff
        npairs = n - aoff
        # "first" positions: i in [aoff, n-1]
        G[k, aoff:npairs+aoff] += (1.0 - h[:npairs])
        # "second" positions: i in [0, npairs-1]
        G[k, :npairs] += (-h[aoff:npairs+aoff])
    
    return G
```

```python
def lp_minmax_step(h, n):
    """One LP-based step for min-max optimization."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_vals = corr * dx
    m = len(c_vals)
    
    G = compute_grad_matrix(h, n)
    
    # LP: min t
    # s.t. G @ delta - t <= -c  (m constraints)
    #      -delta >= -h  →  delta >= -h  →  -delta <= h  (n constraints)
    #      delta <= 1-h  (n constraints)
    #      sum(delta) = 0
    #
    # Variables: [t, delta_0, ..., delta_{n-1}]  (n+1 vars)
    
    # Objective: min t
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # Rows 0 to m-1: G @ delta - t <= -c
    A_ub = np.zeros((m + 2 * n, n + 1))
    b_ub = np.zeros(m + 2 * n)
    
    # Max constraints
    A_ub[:m, 0] = -1.0
    A_ub[:m, 1:] = G
    b_ub[:m] = -c_vals
    
    # Lower bounds: -delta_i <= h_i  →  row: [0, 0, ..., -1 (at i+1), ..., 0]
    for i in range(n):
        A_ub[m + i, i + 1] = -1.0
        b_ub[m + i] = h[i]
    
    # Upper bounds: delta_i <= 1 - h_i
    for i in range(n):
        A_ub[m + n + i, i + 1] = 1.0
        b_ub[m + n + i] = 1.0 - h[i]
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(None, None)] * (n + 1)  # t is free, delta handled by inequalities
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        delta = res.x[1:]
        t_delta = res.x[0]
        return delta, t_delta
    return None, None
```
