            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return h, compute_c5(h)
    grad = grad / gnorm
    h_new = project_feasible(h - step * grad, n)
    return h_new, compute_c5(h_new)

def binary_flip_optimize(h, n, rng, max_flips=100):
    best = h.copy()
    best_c5 = compute_c5(best)
    for _ in range(max_flips):
        if remaining_time() < 2:
            break
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        trial = best.copy()
        trial[i] = 1.0 - trial[i]
        trial[j] = 1.0 - trial[j]
        trial = np.clip(trial, 0.0, 1.0)
        trial = project_feasible(trial, n)
        c5_trial = compute_c5(trial)
        if c5_trial < best_c5 - 1e-14:
            best = trial
            best_c5 = c5_trial
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        h_tmp = np.zeros(best_n)
        for i in range(best_n):
            v = i
            cnt = 0
            while v:
                cnt ^= (v & 1)
                v >>= 1
            h_tmp[i] = 1.0 if cnt == 1 else 0.0
        best_h = project_feasible(h_tmp, best_n)
        best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B at parent resolution with increasing beta
    n = best_n
    h = best_h.copy()
    betas = [30, 80, 200, 500, 1200, 3000, 8000, 20000]
    mus = [10000.0, 5000.0, 2000.0, 1000.0, 500.0, 200.0, 100.0, 50.0]
    for idx, (beta, mu) in enumerate(zip(betas, mus)):
        if remaining_time() < 15:
            break
        maxit = 300 if n <= 800 else 200
        h_new = lbgfss_opt(h, n, beta, mu, maxit)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-15:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                best_n = n
        # Also try SLSQP for the first couple of betas (better constraint handling)
        if idx < 3 and remaining_time() > 30:
            h_s = slsqp_opt(h, n, beta, mu, 200)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
                best_n = n
                h = h_s.copy()

    # Phase 2: Coordinate descent polish at parent resolution
    if remaining_time() > 30:
        h_cd, c5_cd = coord_descent(best_h, best_n, 40, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 3: Subgradient descent with restarts
    if remaining_time() > 50:
        h_sg = best_h.copy()
        n_sg = best_n
        for it in range(3000):
            if remaining_time() < 5:
                break
            steps = [0.05, 0.02, 0.01, 0.03, 0.005, 0.08]
            improved = False
            for st in steps:
                h_trial, c5_trial = subgrad_one_step(h_sg, n_sg, 5, st)
                if c5_trial < best_c5 - 1e-16:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h_sg = h_trial.copy()
                    improved = True
                    break
            if not improved:
                if it % 300 == 200 and remaining_time() > 20:
                    h_lbg = lbgfss_opt(h_sg, n_sg, 2000, 500, 200)
                    c5_lbg = compute_c5(h_lbg)
                    if c5_lbg < best_c5:
                        best_c5 = c5_lbg
                        best_h = h_lbg.copy()
                        h_sg = h_lbg.copy()
                    if it > 600:
                        break

    # Phase 4: Try higher resolution (n=1280)
    if remaining_time() > 120:
        hi_n = 1280
        if hi_n % 2 == 0:
            h_hi = interp_to(best_h, best_n, hi_n)
            h_hi = project_feasible(h_hi, hi_n)
            c5_hi = compute_c5(h_hi)
            for beta in [100, 400, 1500, 5000]:
                if remaining_time() < 20:
                    break
                mu_v = 5000.0 if beta < 500 else 500.0
                h_new = lbgfss_opt(h_hi, hi_n, beta, mu_v, 200)
                h_new = project_feasible(h_new, hi_n)
                c5_new = compute_c5(h_new)
                if c5_new < c5_hi:
                    h_hi = h_new
                    c5_hi = c5_new
                if c5_hi < best_c5:
                    best_c5 = c5_hi
                    best_h = h_hi.copy()
                    best_n = hi_n
            if remaining_time() > 30:
                h_cd2, c5_cd2 = coord_descent(best_h, best_n, 30, rng)
                if c5_cd2 < best_c5:
                    best_c5 = c5_cd2
                    best_h = h_cd2.copy()

    # Phase 5: Try medium resolution (n=960) if n=1280 wasn't tried
    if remaining_time() > 80 and best_n < 1200:
        med_n = 960
        h_med = interp_to(best_h, best_n, med_n)
        h_med = project_feasible(h_med, med_n)
        c5_med = compute_c5(h_med)
        for beta in [200, 1000, 4000]:
            if remaining_time() < 15:
                break
            h_new = lbgfss_opt(h_med, med_n, beta, 1000.0, 250)
            c5_new = compute_c5(h_new)
            if c5_new < c5_med:
                h_med = h_new
                c5_med = c5_new
            if c5_med < best_c5:
                best_c5 = c5_med
                best_h = h_med.copy()
                best_n = med_n
        if remaining_time() > 20:
            h_cd3, c5_cd3 = coord_descent(best_h, best_n, 25, rng)
            if c5_cd3 < best_c5:
                best_c5 = c5_cd3
                best_h = h_cd3.copy()

    # Phase 6: Binary flip local search
    if remaining_time() > 30:
        h_flip, c5_flip = binary_flip_optimize(best_h, best_n, rng, 200)
        if c5_flip < best_c5:
            best_c5 = c5_flip
            best_h = h_flip.copy()

    # Phase 7: Final L-BFGS-B polish with very high beta
    if remaining_time() > 15:
        for beta in [30000, 80000]:
            if remaining_time() < 8:
                break
            h_new = lbgfss_opt(best_h, best_n, beta, 10.0, 300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 8: Final coordinate descent
    if remaining_time() > 15:
        h_final_cd, c5_final_cd = coord_descent(best_h, best_n, 30, rng)
        if c5_final_cd < best_c5:
            best_c5 = c5_final_cd
            best_h = h_final_cd.copy()

    # Phase 9: Micro subgradient
    if remaining_time() > 8:
        h_ms = best_h.copy()
        for it in range(500):
            if remaining_time() < 2:
                break
            improved = False
            for st in [0.003, 0.001, 0.01, 0.005]:
                h_trial, c5_trial = subgrad_one_step(h_ms, best_n, 3, st)
                if c5_trial < best_c5 - 1e-16:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h_ms = h_trial.copy()
                    improved = True
                    break
            if not improved:
                break

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
