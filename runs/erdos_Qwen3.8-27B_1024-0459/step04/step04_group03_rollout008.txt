    """Vectorized construction of A matrix."""
    # A[j,m] = g[m-j+n-1] if 0 <= m-j+n-1 < n else 0
    # For row j: valid m range is [max(0, j-n+1), min(n-1, j)]

    j_idx = np.arange(2*n-1)[:, None]  # shape (2n-1, 1)

    g_idx = m_idx - j_idx + n - 1  # shape (2n-1, n)

    A[valid] = g[g_idx[valid]]

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def build_A_fast(n, g):
    j_idx = np.arange(2 * n - 1)[:, None]
    m_idx = np.arange(n)[None, :]
    g_idx = m_idx - j_idx + n - 1
    valid = (g_idx >= 0) & (g_idx < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = g[g_idx[valid]]
    return A

def solve_lp(h, n, solver_list=None):
    g = 1.0 - h
    dx = 2.0 / n
    A = build_A_fast(n, g)
    T = cp.Variable()
    h_var = cp.Variable(n)
    obj = cp.Minimize(T)
    cons = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0,
        A @ h_var * dx <= T
    ]
    prob = cp.Problem(obj, cons)
    if solver_list is None:
        solver_list = [cp.CBC, cp.GLPK, cp.CVXOPT, cp.ECOS, cp.GUROBI, cp.GLPK_CC, cp.PDLP, cp.SCIP]
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if h_var.value is not None:
                h_new = np.array(h_var.value).flatten()
                h_new = project_feasible(h_new, n)
                return h_new
        except Exception:
            continue
    return h.copy()

def lp_alternative(h, n, max_steps=150, patience=15):
    h = h.copy()
    best_local = h.copy()
    best_c5_local = compute_c5(h)
    no_improve = 0
    solvers_to_try = [cp.CBC, cp.GLPK, cp.CVXOPT, cp.ECOS]
    for it in range(max_steps):
        if remaining_time() < 10:
            break
        h_new = solve_lp(h, n, solver_list=solvers_to_try)
        if np.allclose(h_new, h, atol=1e-10):
            no_improve += 1
            if no_improve > patience:
                break
            continue
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5_local - 1e-12:
            best_c5_local = c5
            best_local = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break
    return best_local, best_c5_local

def smooth_obj_grad(h, n, beta):
    dx = 2.0 / n
    corr = compute_corr(h)
    c5v = corr * dx
    m = np.max(c5v)
    shifted = beta * (c5v - m)
    shifted -= np.max(shifted)
    exp_v = np.exp(shifted)
    w = exp_v / np.sum(exp_v)
    g = 1.0 - h
    A = build_A_fast(n, g)
    grad = dx * (A.T @ w)
    obj_val = (m + np.log(np.sum(exp_v)) / beta)
    return obj_val, grad

def gradient_descent(h, n, beta, max_iter=3000, lr=0.3, patience=100):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    for it in range(max_iter):
        if remaining_time() < 5:
            break
        if (it - no_improve) >= patience and no_improve > 50:
            break
        _, grad = smooth_obj_grad(h, n, beta)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
        else:
            h = h_new
            no_improve += 1
    return best_h, best_c5

def gradient_descent_adaptive(h, n, beta, max_iter=5000):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.5
    no_improve = 0
    for it in range(max_iter):
        if remaining_time() < 5:
            break
        if no_improve > 200:
            lr *= 0.5
            no_improve = 0
            if lr < 1e-6:
                break

        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            lr = min(lr * 1.05, 2.0)
        else:
            h = h_new
            no_improve += 1
            lr *= 0.9
    return best_h, best_c5

def minimax_slsqp(h_init, n, max_rounds=10, max_slsqp_iter=200):
    dx = 2.0 / n
    h = h_init.copy()
    for rnd in range(max_rounds):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        max_corr = np.max(corr)
        t_init = max_corr * dx + 1e-6
        threshold = max_corr - 0.5
        active = np.where(corr >= threshold)[0]
        top3 = np.argsort(corr)[::-1][:5]
        active = np.unique(np.concatenate([active, top3]))
        active = active[np.argsort(active)]
        nn = n
        ddx = dx
        lags_g = active
        nn_g = nn
        ddx_g = ddx

        def _mm_obj(x):
            return x[nn_g]

        def _mm_ineq(x):
            hh = x[:nn_g]
            cc = np.correlate(hh, 1.0 - hh, mode='full')
            return x[nn_g] - ddx_g * cc[lags_g]

        def _mm_eq(x):
            return np.sum(x[:nn_g]) - nn_g / 2.0

        x0 = np.concatenate([h, [t_init]])
        bounds = [(0.0, 1.0)] * nn + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': _mm_ineq},
            {'type': 'eq', 'fun': _mm_eq}
        ]
        try:
            res = minimize(_mm_obj, x0, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_slsqp_iter, 'ftol': 1e-14, 'iprint': 0})
            if res.x is not None:
                h_new = project_feasible(res.x[:n], n)
                c5_new = compute_c5(h_new)
                c5_old = compute_c5(h)
                if c5_new < c5_old - 1e-16:
                    h = h_new
                elif rnd > 3:
                    break
        except Exception:
            break
    return h

def swap_improve(h, n, max_sweeps=4, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        for i in range(n):
            if remaining_time() < 1:
                break
            hi = best_h[i]
            if hi < 0.05 or hi > 0.95:
                continue
            for delta in [0.1, 0.2, 0.3]:
                for j in range((i + 1) % n, (i + 1 + min(30, n)) % n):
                    if j == i:
                        continue
                    hj = best_h[j]
                    h_trial = best_h.copy()
                    amt = (hi - 0.5) * delta + (0.5 - hj) * delta
                    amt = np.clip(amt, -0.3, 0.3)
                    h_trial[i] -= amt
                    h_trial[j] += amt
                    h_trial = project_feasible(h_trial, n)
                    c5_t = compute_c5(h_trial)
                    if c5_t < best_c5 - 1e-14:
                        best_c5 = c5_t
                        best_h = h_trial.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5
