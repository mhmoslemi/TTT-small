        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c else 0.0
    return h

def make_binary_random(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n//2]] = 1.0
    return h

def obj_lse(x):
    corr = np.correlate(x, 1.0-x, mode='full')
    m = np.max(corr)
    b = _G_BETA
    s = corr - m
    e = np.exp(np.clip(b*s, -400, 400))
    val = (m + np.log(np.sum(e))/b) * _G_DX
    if _G_PEN > 0:
        val += _G_PEN * (np.sum(x) - _G_TGT)**2
    if _G_FOCUS >= 0:
        val += 0.01 * corr[_G_FOCUS] * _G_DX
    return float(val)

def grad_corr_at(k, h, n):
    """Compute gradient of corr[k] w.r.t. h, where k is the index in 'full' correlation."""
    d = n - 1 - k  # effective lag
    g = np.zeros(n)
    if d >= 0:
        g[:n-d] += (1.0 - h[d:])
        if d > 0:
            g[d:] -= h[:n-d]
    else:
        dd = -d
        g[dd:] += (1.0 - h[:n-dd])
        g[:n-dd] -= h[dd:]
    return g

def lp_active_set(h, n, max_iters=30, tol=1e-10):
    """Use LP to refine h by linearizing active correlations."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        corr = np.correlate(h, 1.0-h, mode='full')
        cmax = np.max(corr)
        # Find active lags (within 5% of max)
        threshold = cmax - 0.05 * cmax
        active = np.where(corr >= threshold)[0]
        
        if len(active) == 0:
            break
        
        # Set up LP using scipy
        # Variables: h[0],...,h[n-1], t
        # min t
        # s.t. g_k · h + (corr_k - g_k·h_cur) <= t  for k in active
        #      0 <= h[i] <= 1
        #      sum(h) = n/2
        
        m = len(active)
        # Objective: minimize t → c = [0,...,0, 1]
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        # Inequality constraints: A_ub @ x <= b_ub
        # For each active k: g_k · h - t <= -(corr_k - g_k · h_cur)
        A_ub = np.zeros((m + n, n + 1))  # m correlation constraints + n upper bound
        b_ub = np.zeros(m + n)
        
        h_flat = h
        for idx, k in enumerate(active):
            g = grad_corr_at(int(k), h_flat, n)
            A_ub[idx, :n] = g
            A_ub[idx, n] = -1.0
            b_ub[idx] = -(corr[int(k)] - g @ h_flat)
        
        # Upper bounds: h[i] <= 1 → h[i] <= 1
        A_ub[m:m+n, :n] = np.eye(n)
        b_ub[m:m+n] = 1.0
        
        # Equality: sum(h) = n/2
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n/2.0])
        
        from scipy.optimize import linprog
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(0,1)]*n + [(None, None)], method='highs')
        
        if res.status != 0 or res.x is None:
            break
        
        h_new = np.clip(res.x[:n], 0, 1)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Check convergence
        if abs(c5_new - best_c5) < tol:
            break
        h = h_new
    
    return best_h, best_c5

def subgradient_polish(h, n, steps=5000, alpha0=0.005, tlimit=None):
    """Subgradient method using the active lag."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for t in range(steps):
        if tlimit is not None and tlimit() < 2.0:
            break
        corr = np.correlate(h, 1.0-h, mode='full')
        k_star = int(np.argmax(corr))
        g = grad_corr_at(k_star, h, n)
        
        alpha = alpha0 / (1.0 + 0.005 * t)
        h = h - alpha * g
        h = np.clip(h, 0, 1)
        h = project_feasible(h, n)
        
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def slsqp_lse(h0, n, beta, maxiter=400):
    global _G_BETA, _G_DX, _G_TGT, _G_PEN, _G_FOCUS
    _G_BETA = float(beta)
    _G_DX = 2.0 / n
    _G_TGT = n / 2.0
    _G_PEN = 0.0
    _G_FOCUS = -1
    bounds = [(0.0, 1.0)] * n
    cons = [
        {'type': 'eq', 'fun': _make_eq_con_fun, 'args': (n,)}
    ]
    res = minimize(obj_lse, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
    h = project_feasible(res.x, n)
    return h

def _make_eq_con_fun(x, n):
    return np.sum(x) - n / 2.0

def make_sine_init(n, freqs, amplitudes, phase, rng):
    t = np.arange(n) / n
    h = np.full(n, 0.5)
    for f, a, p in zip(freqs, amplitudes, phase):
        h += a * np.sin(2.0 * np.pi * f * t + p)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)
