```python
def compute_R(h, n):
    """Compute R[i, p] = change to corr_full[p] when h[i] goes from 1 to 0 (binary h)."""
    nc = 2 * n - 1
    R = np.zeros((n, nc))
    d = np.arange(-(n-1), n)  # d[p] = p - (n-1) = shift k
    
    for i in range(n):
        if h[i] < 0.5:
            continue
        # k = d[p]
        valid_low = (i + d >= 0) & (i + d < n)   # i+k valid
        valid_high = (i - d >= 0) & (i - d < n)   # i-k valid
        
        # R[i,p] = (h[i+k]-1)*[valid_low] + (h[i-k]-1)*[valid_high] for k≠0
        # For k=0 (p=n-1): R[i,p] = 0
        
        term1 = np.zeros(nc)
        term1[valid_low] = h[i + d[valid_low]] - 1.0
        
        term2 = np.zeros(nc)
        term2[valid_high] = h[i - d[valid_high]] - 1.0
        
        R[i, :] = term1 + term2
        R[i, n-1] = 0.0  # k=0 special case
    
    return R
```

```python
import numpy as np
from scipy.optimize import minimize
import time
```

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_trial = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_trial)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def compute_R(h, n):
    nc = 2 * n - 1
    R = np.zeros((n, nc))
    d = np.arange(-(n - 1), n)
    for i in range(n):
        valid_low = (i + d >= 0) & (i + d < n)
        valid_high = (i - d >= 0) & (i - d < n)
        term1 = np.zeros(nc)
        term1[valid_low] = h[i + d[valid_low]] - 1.0
        term2 = np.zeros(nc)
        term2[valid_high] = h[i - d[valid_high]] - 1.0
        R[i, :] = term1 + term2
        R[i, n - 1] = 0.0
    return R

def swap_delta(R, corr, i, j, n):
    """Compute the exact change to corr when swapping h[i] (1->0) and h[j] (0->1)."""
    delta = R[i] - R[j]
    p1 = n - 1 + (i - j)
    p2 = n - 1 + (j - i)
    if 0 <= p1 < 2 * n - 1:
        delta[p1] += 1.0
    if 0 <= p2 < 2 * n - 1:
        delta[p2] += 1.0
    return delta

def slsqp_smooth(h, n, beta, maxiter=300):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - _TGT,
             'jac': lambda x: np.ones(_N)}]
    
    def obj(x):
        dx_val = 2.0 / _N
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(_BETA * shifted, -60, 60)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / _BETA) * dx_val
    
    def grad(x):
        dx_val = 2.0 / _N
        beta = _BETA
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -60, 60)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(_N)
        nc = 2 * _N - 1
        d = np.arange(-( _N - 1), _N)
        for p in range(nc):
            if w[p] > 1e-14:
                k = d[p]
                if k >= 0:
                    idx_a = slice(0, _N - k)
                    idx_b = slice(k, _N)
                    g[idx_a] += w[p] * (1.0 - x[idx_b])
                    g[idx_b] -= w[p] * x[idx_a]
                else:
                    m2 = -k
                    idx_a = slice(m2, _N)
                    idx_b = slice(0, _N - m2)
                    g[idx_a] += w[p] * (1.0 - x[idx_b])
                    g[idx_b] -= w[p] * x[idx_a]
        return g * dx_val
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

def greedy_local_search(h_bin, n, time_limit=30, rng=None):
    """Best-improvement local search using R matrix."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = h_bin.copy().astype(float)
    corr = compute_corr(h)
    R = compute_R(h, n)

    best_c5 = float(np.max(corr)) * 2.0 / n
    best_h = h.copy()
    nc = 2 * n - 1
    
    improved = True
    while improved and time.time() - start < time_limit and remaining() > 5:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        cur_max = float(np.max(corr))
        best_imp = 1e-10
        best_pair = None
        
        R_ones = R[ones]  # shape (n_ones, nc)
        R_zeros = R[zeros]  # shape (n_zeros, nc)
        
        # For each one i, find the best zero j
        # score[i_z, j_z] = max(corr + R[i] - R[j])
        # We want to minimize this
        
        # Vectorized: for all pairs, compute the max
        # This is (n_ones, n_zeros, nc) which is too large for n=312
        # Instead, do it row by row (over ones)
        
        for idx_i in range(len(ones)):
            i = ones[idx_i]
            Ri = R[i]
            # For each zero j: new_max = max(corr + Ri - R[j])
            # = max(corr + Ri - R_zeros)   (broadcast)
            temp = np.max(corr[None, :] + Ri[None, :] - R_zeros, axis=1)
            j_best = temp.argmin()
            imp = cur_max - temp[j_best]
            if imp > best_imp:
                best_imp = imp
                best_pair = (i, zeros[j_best])
        
        if best_pair is not None:
            i, j = best_pair
            # Apply swap
            h[i] = 0.0
            h[j] = 1.0
            corr = corr + swap_delta(R, corr, i, j, n)
            # Recompute R (it changed)
            R = compute_R(h, n)
            best_c5 = float(np.max(corr)) * 2.0 / n
            best_h = h.copy()
            improved = True
    
    return best_h, best_c5
