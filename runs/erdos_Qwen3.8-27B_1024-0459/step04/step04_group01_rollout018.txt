import numpy as np
import time

_START = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(np.float64), 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=np.float64).copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_trial = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_trial)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def kron_initialization(n, alpha, seed=42):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    for i in range(n):
        if (i * alpha) % 1.0 < 0.5:
            h[i] = 1.0
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        ones_idx = np.where(h == 1)[0]
        remove = rng.choice(ones_idx, n_ones - n // 2, replace=False)
        h[remove] = 0.0
    elif n_ones < n // 2:
        zeros_idx = np.where(h == 0)[0]
        add = rng.choice(zeros_idx, n // 2 - n_ones, replace=False)
        h[add] = 1.0
    return h


def random_initialization(n, seed=42):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones_pos = rng.choice(n, n // 2, replace=False)
    h[ones_pos] = 1.0
    return h


def greedy_local_search(h, n, dx, max_passes=300, time_limit=60, rng=None):
    if rng is None:
        rng = np.random.RandomState(123)
    h = np.array(h, dtype=np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_c5 = cur_max * dx
    start = time.time()

    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_new_max = cur_max
        best_swap = None

        for i in ones:
            hi_save = h[i]
            h[i] = 0.0
            for j in zeros:
                h[j] = 1.0
                new_max = np.max(np.correlate(h, 1.0 - h, mode='full'))
                h[j] = 0.0
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_swap = (i, j)
            h[i] = hi_save

        if best_swap is None:
            break

        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        cur_max = best_new_max
        c5_val = cur_max * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()

    h = (h > 0.5).astype(np.float64)
    return best_h, best_c5


def simulated_annealing(h, n, dx, time_limit=30, rng=None, T_start=0.5, T_end=0.01, max_iter=50000):
    if rng is None:
        rng = np.random.RandomState(77)
    h = np.array(h, dtype=np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_c5 = cur_max * dx
    start = time.time()
    n_iter = max_iter
    end = start + time_limit

    for it in range(n_iter):
        if time.time() > end:
            break
        T = T_start * (T_end / T_start) ** (it / max(1, n_iter))

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_max = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
        delta = new_max - cur_max

        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-6)):
            cur_max = new_max
            if cur_max * dx < best_c5:
                best_c5 = cur_max * dx
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0

    h = (h > 0.5).astype(np.float64)
    return best_h, best_c5

            h[i1] = 0.0; h[i2] = 0.0

            if new_max < cur_max - 1e-15:

                    best_h = h.copy()
                found = True
                break
            else:

def continuous_polish(h, n, time_limit=15):
    """Simple projected gradient step to allow fractional values."""
    h = np.array(h, dtype=np.float64).copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    step = 0.01

    for it in range(500):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        # Subgradient: gradient of the max correlation
        active = np.where(np.abs(corr - c_max) < 1e-8)[0]
        g = np.zeros(n)
        for k in active:
            d = k - (n - 1)
            if d >= 0:
                np_pairs = n - d
                if np_pairs > 0:
                    g[d:d + np_pairs] += (1.0 - h[:np_pairs])
                    g[:np_pairs] -= h[d:d + np_pairs]
            else:
                ao = -d
                np_pairs = n - ao
                if np_pairs > 0:
                    g[:np_pairs] += (1.0 - h[ao:ao + np_pairs])
                    g[ao:ao + np_pairs] -= h[:np_pairs]

        h_trial = h - step * g * dx
        h_trial = project_feasible(h_trial, n)
        h_trial = np.clip(h_trial, 0.0, 1.0)
        c5_new = compute_c5(h_trial)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_trial.copy()
            h = h_trial.copy()
            step = min(step * 1.01, 0.05)
        else:
            step *= 0.9
            if step < 0.001:
                break

    return best_h, best_c5
