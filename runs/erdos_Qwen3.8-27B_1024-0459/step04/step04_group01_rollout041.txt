        for step in [0.01, 0.005, 0.02, 0.002, 0.05, 0.001, 0.03, 0.08]:
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-13:
                h = h_trial
                best_c5 = c_new
                best_h = h.copy()
                improved = True
                break
            if c_new < best_c5:
                h = h_trial
                best_c5 = c_new
                best_h = h.copy()
                improved = True
        if not improved:
            break

    return best_h, best_c5


def reshape_interp(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 312
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP with increasing beta
    for beta in [200, 500, 1500, 4000, 10000]:
        if remaining() < 8:
            break
        h_new = slsqp_optimize(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Pair coordinate descent (main)
    if remaining() > 15:
        h_pc, c5_pc = pair_coord_descent_v2(h, n, time_limit=min(70, remaining() - 10), rng=rng)
        if c5_pc < best_c5:
            best_c5 = c5_pc
            best_h = h_pc.copy()
            h = h_pc.copy()

    # Phase 3: Projected gradient with smooth max
    if remaining() > 15:
        h_pg, c5_pg = projected_gradient_smooth(h, n, beta=2000, time_limit=min(40, remaining() - 10), lr_start=0.03)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()

    # Phase 4: Multi-pair coordinate descent with different offsets
    if remaining() > 15:
        h_mp, c5_mp = multi_pair_coord(h, n, time_limit=min(50, remaining() - 10), rng=rng)
        if c5_mp < best_c5:
            best_c5 = c5_mp
            best_h = h_mp.copy()
            h = h_mp.copy()

    # Phase 5: SLSQP polish
    for beta in [5000, 15000, 40000]:
        if remaining() < 8:
            break
        h_new = slsqp_optimize(h, n, beta, maxiter=350)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 6: Top shift equalization
    if remaining() > 10:
        h_eq, c5_eq = top_shift_equalization(h, n, time_limit=min(25, remaining() - 8), top_k=3)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 7: More pair coordinate descent
    if remaining() > 15:
        h_pc, c5_pc = pair_coord_descent_v2(h, n, time_limit=min(50, remaining() - 10), rng=rng)
        if c5_pc < best_c5:
            best_c5 = c5_pc
            best_h = h_pc.copy()
            h = h_pc.copy()

    # Phase 8: Binary local search
    if remaining() > 15:
        h_b, c5_b = binary_targeted_search(h, n, time_limit=min(35, remaining() - 8), rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n

    # Phase 9: Try different n values
    if remaining() > 40:
        n_candidates = sorted(set([
            best_n + 4, best_n + 8, best_n + 16,
            best_n - 4, best_n - 8,
            256, 320, 384, 448, 328, 344
        ]))
        n_candidates = [nn for nn in n_candidates if 128 <= nn <= 640 and nn % 2 == 0 and nn != best_n]

        for n_t in n_candidates:
            if remaining() < 25:
                break
            h_interp = reshape_interp(best_h, best_n, n_t)
            h_feas = project_feasible(h_interp, n_t)

            h_opt = slsqp_optimize(h_feas, n_t, 1000, maxiter=200)
            if remaining() > 12:
                h_opt, _ = pair_coord_descent_v2(h_opt, n_t, time_limit=min(20, remaining() - 10), rng=rng)

            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_t

    # Phase 10: Final intensive refinement
    n = best_n
    h = best_h.copy()

    if remaining() > 10:
        h_new = slsqp_optimize(h, n, 30000, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining() > 10:
        h_pc, c5_pc = pair_coord_descent_v2(h, n, time_limit=min(30, remaining() - 8), rng=rng)
        if c5_pc < best_c5:
            best_c5 = c5_pc
            best_h = h_pc.copy()

    if remaining() > 8:
        h_eq, c5_eq = top_shift_equalization(h, n, time_limit=min(15, remaining() - 5), top_k=2)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    if remaining() > 6:
        h_new = slsqp_optimize(best_h, best_n, 50000, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Validate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
