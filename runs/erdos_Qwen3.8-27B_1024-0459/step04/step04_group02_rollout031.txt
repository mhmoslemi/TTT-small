For each coordinate i:

   - If i-k is valid (k≠0): the k-th term contributes -h[i-k]*t (linear in t)
3. So the objective is max of (piecewise linear + quadratic) functions of t.
4. We can find the exact optimum by finding the breakpoints and checking.

The breakpoints occur where two different lags cross. For each pair of lags (k1, k2), the crossing point is:
base_k1 + a_k1 * t + b_k1 * t² = base_k2 + a_k2 * t + b_k2 * t²

This gives (a_k1 - a_k2) * t + (b_k1 - b_k2) * t² = base_k2 - base_k1

If b_k1 = b_k2 (both 0 or both -1), this is linear: t = (base_k2 - base_k1) / (a_k1 - a_k2)
If b_k1 ≠ b_k2, it's quadratic.

Since b_k is -1 only for k=0 and 0 otherwise, we have:
- Pairs of non-zero lags: linear crossing
- Pairs involving lag 0: quadratic crossing

The number of breakpoints is O(m²) where m is the number of lags, which is 2n-1. So O(n²) which for n=640 is about 800,000. That's a lot per coordinate update.

1. Evaluate the objective at t=0, t=1, and t=current
2. The maximum at these points gives a lower bound
3. Find the active lag at the minimum of these
4. The optimal t must be where the objective equals this lower bound, or at a boundary

            idx = np.argmin(h) if diff > 0 else np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-14:
            idx = int(np.argmin(np.abs(h - 0.5)))
            h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def exact_coord_opt(h, n, i, corr):
    ncorr = 2 * n - 1
    dx = 2.0 / n
    t0 = h[i]
    
    a_coeff = np.zeros(ncorr)
    b_coeff = np.zeros(ncorr)
    base = corr.copy()
    
    for d in range(-(n-1), n):
        k = n - 1 + d
        case_a = (0 <= i + d < n)
        case_b = (d != 0) and (0 <= i - d < n)
        
        if d == 0:
            base[k] = corr[k] - t0 * (1.0 - t0)
            a_coeff[k] = 1.0
            b_coeff[k] = -1.0
        elif case_a and case_b:
            base[k] = corr[k] - t0 * (1.0 - h[i+d]) - h[i-d] * (1.0 - t0)
            a_coeff[k] = (1.0 - h[i+d]) - h[i-d]
        elif case_a:
            base[k] = corr[k] - t0 * (1.0 - h[i+d])
            a_coeff[k] = (1.0 - h[i+d])
        elif case_b:
            base[k] = corr[k] - h[i-d] * (1.0 - t0)
            a_coeff[k] = -h[i-d]
    
    candidates = [0.0, 1.0, t0]
    
    for t_cand in [0.0, 1.0, t0]:
        vals = base + a_coeff * t_cand + b_coeff * t_cand * t_cand
        max_v = np.max(vals)
        tol = 1e-5 * max(1.0, abs(max_v))
        active = np.where(vals >= max_v - tol)[0]
        if len(active) == 0:
            continue
        active = active[:60]
        for pi in range(len(active)):
            ak = active[pi]
            for qi in range(pi + 1, len(active)):
                bk = active[qi]
                da = a_coeff[ak] - a_coeff[bk]
                db = b_coeff[ak] - b_coeff[bk]
                dc = base[ak] - base[bk]
                if abs(db) < 1e-14:
                    if abs(da) > 1e-14:
                        tc = -dc / da
                        if -0.001 <= tc <= 1.001:
                            candidates.append(np.clip(tc, 0.0, 1.0))
                else:
                    disc = da * da - 4.0 * db * dc
                    if disc >= 0:
                        sq = np.sqrt(disc)
                        candidates.append(np.clip((-da + sq) / (2.0 * db), 0.0, 1.0))
                        candidates.append(np.clip((-da - sq) / (2.0 * db), 0.0, 1.0))
    
    best_t = t0
    best_max = np.inf
    for t_c in candidates:
        t_c = float(np.clip(t_c, 0.0, 1.0))
        vals = base + a_coeff * t_c + b_coeff * t_c * t_c
        v = np.max(vals)
        if v < best_max:
            best_max = v
            best_t = t_c
    
    return best_t, best_max * dx

def coord_descent(h, n, max_sweeps=100, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    best = h.copy()
    corr = compute_corr(best)
    best_c5 = np.max(corr) * (2.0 / n)
    
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if remaining_time() < 2:
                break
            t_new, c5_new = exact_coord_opt(best, n, i, corr)
            if abs(t_new - best[i]) > 1e-10:
                best[i] = t_new
                corr = compute_corr(best)
                new_c5 = np.max(corr) * (2.0 / n)
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    improved = True
        
        if not improved:
            break
    
    return best, best_c5

def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def _slsqp_obj(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[0]

def _slsqp_grad(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[1]

def _slsqp_cons_fun(x):
    return np.sum(x) - _N_WORK / 2.0

def _slsqp_cons_jac(x):
    return np.ones(_N_WORK)

def slsqp_refine(h, n, beta, maxiter=300):
    global _N_WORK, _BETA_WORK
    _N_WORK = n
    _BETA_WORK = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            if compute_c5(trial) <= compute_c5(h) + 1e-14:
                return trial
    except Exception:
        pass
    return h.copy()

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def binary_round(h, n, threshold=0.01):
    h2 = h.copy()
    h2[h2 < threshold] = 0.0
    h2[h2 > 1.0 - threshold] = 1.0
    h2 = project_feasible(h2, n)
    return h2

def subgrad_direction(h, n, n_active=5):
    corr = compute_corr(h)
    max_corr = np.max(corr)
    active = np.where(corr >= max_corr - 0.3)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        active = np.argsort(corr)[::-1][:n_active]
    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return None
    return grad / gnorm

def multi_start_coordinate(h, n, max_sweeps, rng, allow_neutral=False):
    best = h.copy()
    corr = compute_corr(best)
    best_c5 = np.max(corr) * (2.0 / n)
    
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if remaining_time() < 2:
                break
            t_new, c5_new = exact_coord_opt(best, n, i, corr)
            if abs(t_new - best[i]) > 1e-10:
                old_val = best[i]
                best[i] = t_new
                corr = compute_corr(best)
                new_c5 = np.max(corr) * (2.0 / n)
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    improved = True
                elif allow_neutral and new_c5 <= best_c5 + 1e-12:
                    improved = True
                    best_c5 = new_c5
                else:
                    best[i] = old_val
                    corr = compute_corr(best)
        
        if not improved:
            break
    
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        h_tmp = np.zeros(best_n)
        for i in range(best_n):
            cnt = bin(i).count('1')
            h_tmp[i] = 1.0 if cnt % 2 == 1 else 0.0
        best_h = project_feasible(h_tmp, best_n)
        best_c5 = compute_c5(best_h)
    
    # Phase 1: Exact coordinate descent at parent resolution
    n = best_n
    h = best_h.copy()
    h, c5 = multi_start_coordinate(h, n, 60, rng, allow_neutral=False)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        best_n = n
    
    # Phase 2: SLSQP polish at parent resolution
    if remaining_time() > 40:
        for beta in [500, 2000, 8000, 30000]:
            if remaining_time() < 10:
                break
            h_new = slsqp_refine(h, n, beta, 400)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h):
                h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                best_n = n
    
    # Phase 3: Coordinate descent after SLSQP
    if remaining_time() > 20:
        h2, c5_2 = multi_start_coordinate(h, n, 80, rng, allow_neutral=True)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n
    
    # Phase 4: Binary cleanup + re-optimize
    if remaining_time() > 40:
        h_round = binary_round(best_h, best_n, threshold=0.03)
        c5_round = compute_c5(h_round)
        if c5_round < best_c5 + 0.005:
            h_round, c5_round2 = multi_start_coordinate(h_round, best_n, 60, rng, allow_neutral=False)
            if c5_round2 < best_c5:
                best_c5 = c5_round2
                best_h = h_round.copy()
    
    # Phase 5: Higher resolution attempts
    tried_ns = set([best_n])
    res_list = []
    for mult in [1.5, 2.0, 2.5, 3.0, 1.333]:
        rn = int(round(best_n * mult))
        if rn % 2 != 0:
            rn += 1
        if 128 <= rn <= 2048 and rn not in tried_ns:
            res_list.append(rn)
    for rn in [800, 1280, 1600, 960]:
        if rn not in tried_ns and 128 <= rn <= 2048:
            res_list.append(rn)
    res_list = sorted(set(res_list))
    
    for rn in res_list:
        if remaining_time() < 80:
            break
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)
        
        h_r, c5_r = multi_start_coordinate(h_r, rn, 50, rng, allow_neutral=False)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn
            tried_ns.add(rn)
        
        if remaining_time() > 30 and c5_r < best_c5 + 0.002:
            for beta in [1000, 5000, 20000]:
                if remaining_time() < 15:
                    break
                h_new = slsqp_refine(h_r, rn, beta, 250)
                c5_new = compute_c5(h_new)
                if c5_new < c5_r:
                    h_r = h_new
                    c5_r = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_r.copy()
                    best_n = rn
        
        if remaining_time() > 20 and c5_r < best_c5:
            h_r2, c5_r2 = multi_start_coordinate(h_r, rn, 40, rng, allow_neutral=True)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2.copy()
                best_n = rn
    
    # Phase 6: Subgradient on the best
    if remaining_time() > 30:
        h = best_h.copy()
        n = best_n
        for it in range(500):
            if remaining_time() < 5:
                break
            grad = subgrad_direction(h, n, 5)
            if grad is None:
                break
            base_step = 0.02 / (1.0 + it * 0.005)
            improved = False
            for sm in [1.0, 0.5, 2.0, 0.25, 0.1]:
                step = base_step * sm
                h_trial = h - step * grad
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-16:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h = h_trial.copy()
                    improved = True
                    break
            if not improved:
                if it > 100 and remaining_time() > 15:
                    h_new = slsqp_refine(h, n, 5000, 200)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h = h_new.copy()
                    if it > 300:
                        break
    
    # Phase 7: Final coordinate descent
    if remaining_time() > 15:
        h = best_h.copy()
        n = best_n
        h, c5 = multi_start_coordinate(h, n, 100, rng, allow_neutral=True)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
    
    # Phase 8: Final SLSQP polish
    if remaining_time() > 10:
        for beta in [20000, 100000]:
            if remaining_time() < 5:
                break
            h_new = slsqp_refine(best_h, best_n, beta, 200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    # Phase 9: One more pass of coordinate descent
    if remaining_time() > 8:
        h = best_h.copy()
        n = best_n
        h, c5 = multi_start_coordinate(h, n, 50, rng, allow_neutral=True)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
