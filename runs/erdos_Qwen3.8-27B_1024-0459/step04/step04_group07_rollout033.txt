    """Coordinate descent with single-bit flips and pair swaps."""
    if rng is None:
        rng = np.random.RandomState(0)
    c5 = compute_c5(h)
    h_best = h.copy()
    corr = np.correlate(h_best, 1.0 - h_best, mode='full')
    
    for sweep in range(max_sweeps):
        if tlimit is not None and tlimit() < 3.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if tlimit is not None and tlimit() < 1.0:
                break

            v0 = h_best[i]
            for v in [0.0, 1.0]:

                    continue
                trial = h_best.copy()
                trial[i] = v

                    c5 = c
                    h_best = trial.copy()
                    improved = True
                    corr = np.correlate(h_best, 1.0 - h_best, mode='full')
            # Pair swap with random j
            for _ in range(3):

Wait, I made an error. Let me redo.

The objective is: f(h) = (m + log(sum(exp(beta*(corr-m)))) / beta) * dx

Let S = sum(exp(beta*(corr[k]-m)))
Let H = log(S) / beta  (the soft part)

f = (m + H) * dx

d(f)/dh[i] = dx * (dm/dh[i] + dH/dh[i])

Now, m = max_k corr[k]. If m is differentiable (which it is almost everywhere), dm/dh[i] = dc[k*]/dh[i] where k* = argmax.

But for the soft version, we don't need to compute dm/dh[i] separately because:

f = (m + log(S)/beta) * dx = -log(S) * dx / beta + m * dx + log(S) * dx / beta ... no that's circular.

Let me use a different approach. The soft max is:
soft_max = -log(-sum(-exp(corr * beta / beta)) / beta) ... no.

Actually, the standard soft max is:
soft_max_beta(c) = (1/beta) * log(sum(exp(beta * c)))

And (1/beta) * log(sum(exp(beta * c))) → max(c) as beta → ∞.

So my objective is: f(h) = soft_max_beta(corr) * dx = (1/beta) * log(sum(exp(beta * corr))) * dx

d(f)/dh[i] = (1/beta) * dx * sum_k [beta * exp(beta*corr[k]) / sum_j exp(beta*corr[j])] * dc[k]/dh[i]
= dx * sum_k [exp(beta*corr[k]) / sum_j exp(beta*corr[j])] * dc[k]/dh[i]
= dx * sum_k p[k] * dc[k]/dh[i]

where p[k] = exp(beta*corr[k]) / sum_j exp(beta*corr[j])

To avoid overflow, I use:
p[k] = exp(beta*(corr[k]-M)) / sum_j exp(beta*(corr[j]-M)) where M = max(corr)

So the gradient is:
d(f)/dh[i] = dx * sum_k p[k] * dc[k]/dh[i]

This is cleaner. Let me implement it this way.

But for n=256, computing the full gradient is O(n^2) = 65k operations, which is fast. And SLSQP will call it many times. Let me use it.

Actually, for SLSQP, providing the gradient can speed things up significantly. Let me include it.

But wait, there's a subtlety. The constraint is sum(h) = n/2, and the bounds are 0 ≤ h[i] ≤ 1. SLSQP handles these. The gradient I provide should be the gradient of the objective only (not including constraints).

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def interp_to_n(h_src, n_src, n_dst):

        return h_src.copy()

    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    return np.interp(xn, xp, h_src)

def softmax_obj_and_grad(h, beta):
    """Compute soft-max objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    shifted = beta * (corr - M)
    e = np.exp(np.clip(shifted, -500, 500))
    S = np.sum(e)
    p = e / S  # softmax weights, shape (2n-1,)
    
    obj = (M + np.log(S) / beta) * dx
    
    # Gradient
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = p[idx]
        if w < 1e-15:
            continue
        k = idx - (n - 1)
        # dc[
        # dc[idx]/dh[i] = (1-h[i-k]) * [0 <= i-k < n] - h[i] * [0 <= i+k < n]
        
        # First term: (1-h[i-k]) for i where 0 <= i-k < n
        if k >= 0:
            i_start1 = k
            i_end1 = n  # i-k < n means i < n+k, but i < n anyway
        else:
            i_start1 = 0
            i_end1 = n + k  # i-k < n means i < n+k
        
        for i in range(i_start1, i_end1):
            grad[i] += w * (1.0 - h[i - k])
        
        # Second term: -h[i] for i where 0 <= i+k < n
        if k >= 0:
            i_start2 = 0
            i_end2 = n - k
        else:
            i_start2 = -k
            i_end2 = n
        
        for i in range(i_start2, i_end2):
            grad[i] -= w * h[i]
    
    grad *= dx
    
    return obj, grad

def sa_swap(h, n, rng, tlimit, max_iters=50000, T0=0.01, T1=0.001):
    """Simulated annealing with pair swaps."""
    c5 = compute_c5(h)
    h_best = h.copy()
    h_cur = h.copy()
    c5_cur = c5
    best_c5 = c5
    
    for it in range(max_iters):
        if tlimit is not None and tlimit() < 3.0:
            break
        
        T = T0 * (T1 / T0) ** (it / max_iters)
        
        # Pick two random positions to swap
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Swap
        h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
        
        c_new = compute_c5(h_cur)
        delta = c_new - c5_cur
        
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            c5_cur = c_new
            if c_new < best_c5:
                best_c5 = c_new
                h_best = h_cur.copy()
        else:
            # Undo swap
            h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
    
    return h_best, best_c5

    """Enhanced coordinate descent with single-bit flips and pair swaps."""
    if rng is None:

    c5 = compute_c5(h)
    h_best = h.copy()

                    improved = True

    return h_best, c5
