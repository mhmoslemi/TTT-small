        order = np.argsort(h_b)  # zeros first
        # Actually, we want to flip ones to zero where h is smallest
        one_indices = np.where(h_b == 1.0)[0]
        one_indices = one_indices[np.argsort(h[one_indices])[:n_ones - target_ones]]
        h_b[one_indices] = 0.0
    elif n_ones < target_ones:
        zero_indices = np.where(h_b == 0.0)[0]
        zero_indices = zero_indices[np.argsort(-h[zero_indices])[:target_ones - n_ones]]
        h_b[zero_indices] = 1.0
    
    c5_best = compute_c5(h_b)
    h_best = h_b.copy()
    c5_cur = c5_best
    T = T0
    
    for it in range(max_iter):
        if remaining_time() < 5.0:
            break
        
        # Choose a move
        move_type = rng.randint(0, 3)
        h_trial = h_b.copy()
        
        if move_type == 0:
            # Single flip pair: flip a 1 to 0 and a 0 to 1
            ones = np.where(h_b == 1.0)[0]
            zeros = np.where(h_b == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                continue
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_trial[i] = 0.0
            h_trial[j] = 1.0
        elif move_type == 1:
            # Swap two positions
            if n < 2:
                continue
            i, j = rng.choice(n, size=2, replace=False)
            h_trial[i], h_trial[j] = h_b[j], h_b[i]
        else:
            # Block swap: swap two blocks of length 2-5
            L = rng.randint(2, 6)
            if n < 2 * L:
                continue
            s1 = rng.randint(0, n - L)
            s2 = rng.randint(0, n - L)
            if abs(s1 - s2) < L:
                s2 = (s2 + L) % (n - L)
            a = h_trial[s1:s1+L].copy()
            h_trial[s1:s1+L] = h_b[s2:s2+L]
            h_trial[s2:s2+L] = a
        
        c5_trial = compute_c5(h_trial)
        
        # Accept if better, or with probability exp(-delta/T)
        delta = c5_trial - c5_cur
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):

            c5_cur = c5_trial
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h_b.copy()
        
        T *= gamma
    
    return h_best, c5_best

def compute_c5_top5(h):
    """Compute C5 and sum of top-5 correlations (for composite objective)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c5 = float(np.max(corr))
    top5 = float(np.sum(np.sort(corr)[-5:]))
    return c5, top5

def subgrad_all(h, n, w_power=1.0, thresh=0.38):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    n_corr = len(corr)
    excess = np.maximum(corr - thresh, 0.0)
    weights = excess ** w_power
    total_w = np.sum(weights)
    if total_w < 1e-14:
        weights = np.ones(n_corr) / n_corr
    else:
        weights = weights / total_w
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        w = weights[k_idx]
        if w < 1e-15:
            continue
        offset = int(k_idx) - (n - 1)

            nv = n - offset
            if nv <= 0:

            grad[offset:offset + nv] += w * (1.0 - h[:nv])

        else:

            if nv <= 0:
                continue
            grad[:nv] += w * (1.0 - h[m:m + nv])
            grad[m:m + nv] -= w * h[:nv]
    return grad * dx, corr

        delta = c5_trial - c5_cur
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            h_b = h_trial
            c5_cur = c5_trial
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h_b.copy()
        T *= gamma
    return h_best, c5_best


def slsqp_refine(h, n, beta_list=None):
    global _G_N, _G_BETA, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    if beta_list is None:
        beta_list = [100, 500, 2000, 5000]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum}]
    c5_best = compute_c5(h)
    h_best = h.copy()
    for beta in beta_list:
        if remaining_time() < 6.0:
            break
        _G_BETA = float(beta)
        try:
            res = minimize(obj_smooth, h_best, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-14, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-14:
                c5_best = c5_new
                h_best = h_new.copy()
        except Exception:
            pass
    return h_best, c5_best


def multi_start(n, rng):
    results = []
    inits = []
    inits.append(('tm', make_thue_morse(n)))
    inits.append(('golden', make_golden(n)))
    inits.append(('rand1', (rng.rand(n) < 0.5).astype(float)))
    inits.append(('rand2', (rng.rand(n) < 0.5).astype(float)))
    h_tm_pert = make_thue_morse(n).copy()
    perm = rng.permutation(n)
    for i in range(n // 10):
        h_tm_pert[perm[i]] = 1.0 - h_tm_pert[perm[i]]
    inits.append(('tm_pert', h_tm_pert))
    for name, h_init in inits:
        if remaining_time() < 10.0:
            break
        h_init = project_feasible(h_init, n)
        c5 = compute_c5(h_init)
        h_init, c5 = subgrad_opt(h_init, n, max_iter=80, step0=0.1, thresh_init=0.45)
        results.append((c5, h_init, name))
    results.sort(key=lambda x: x[0])
    return results


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        best_h = project_feasible(make_thue_morse(parent_n), parent_n)
        best_c5 = compute_c5(best_h)
        best_n = parent_n
        print("Thue-Morse: n={}, C5={:.10f}".format(parent_n, best_c5))

    # Phase 1: Multi-start
    print("\n--- Phase 1: Multi-start (n={}) ---".format(best_n))
    _G_N = best_n
    _G_TGT = best_n / 2.0
    _G_DX = 2.0 / best_n
    results = multi_start(best_n, rng)
    for c5_r, h_r, name in results:
        print("  {}: C5={:.10f}".format(name, c5_r))
    if results[0][0] < best_c5:
        best_c5 = results[0][0]
        best_h = results[0][1].copy()

    # Phase 2: All-shift subgradient
    print("\n--- Phase 2: Subgradient all-shifts (n={}) ---".format(best_n))
    best_h, best_c5 = subgrad_opt(best_h, best_n, max_iter=500, step0=0.12, w_power=1.0, thresh_init=0.42)
    print("  C5={:.10f}".format(best_c5))

    # Phase 3: Binary SA from best solution
    print("\n--- Phase 3: Binary SA (n={}) ---".format(best_n))
    h_bin, c5_bin = binary_sa(best_h, best_n, max_iter=4000, rng=rng, T0=0.04, gamma=0.998)
    print("  SA from best: C5={:.10f}".format(c5_bin))
    h_tm_b = project_feasible(make_thue_morse(best_n), best_n)
    h_bin2, c5_bin2 = binary_sa(h_tm_b, best_n, max_iter=4000, rng=rng, T0=0.05, gamma=0.998)
    print("  SA from TM: C5={:.10f}".format(c5_bin2))
    if c5_bin < c5_bin2:
        h_sa = h_bin
        c5_sa = c5_bin
    else:
        h_sa = h_bin2
        c5_sa = c5_bin2
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()

    # Phase 4: SLSQP on best continuous
    print("\n--- Phase 4: SLSQP continuous (n={}) ---".format(best_n))
    best_h, best_c5 = slsqp_refine(best_h, best_n, beta_list=[50, 200, 500, 1000, 3000, 5000])
    print("  C5={:.10f}".format(best_c5))

    # Phase 5: SLSQP on best binary
    if c5_sa < best_c5 + 0.02:
        h_br, c5_br = slsqp_refine(h_sa, best_n, beta_list=[100, 500, 2000, 5000])
        if c5_br < best_c5:
            best_c5 = c5_br
            best_h = h_br.copy()
            print("  SA+SLSQP: C5={:.10f}".format(best_c5))

    # Phase 6: Round 2 subgradient + SA
    if remaining_time() > 60.0:
        print("\n--- Phase 6: Round 2 (n={}) ---".format(best_n))
        best_h, best_c5 = subgrad_opt(best_h, best_n, max_iter=250, step0=0.08, w_power=2.0, thresh_init=0.40)
        h_r2, c5_r2 = binary_sa(best_h, best_n, max_iter=3000, rng=rng, T0=0.02, gamma=0.999)
        if c5_r2 < best_c5:
            best_c5 = c5_r2
            best_h = h_r2.copy()
        best_h, best_c5 = slsqp_refine(best_h, best_n, beta_list=[500, 2000, 5000])
        print("  C5={:.10f}".format(best_c5))

    # Phase 7: Higher resolution
    if remaining_time() > 100.0:
        print("\n--- Phase 7: Higher resolution ---")
        for n_try in [384, 512]:
            if remaining_time() < 60.0:
                break
            h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
            _G_N = n_try
            _G_TGT = n_try / 2.0
            _G_DX = 2.0 / n_try
            c5_h = compute_c5(h_interp)
            h_interp, c5_h = subgrad_opt(h_interp, n_try, max_iter=200, step0=0.08, thresh_init=0.40)
            h_bin_h, c5_bh = binary_sa(h_interp, n_try, max_iter=2000, rng=rng, T0=0.02, gamma=0.999)
            if c5_bh < c5_h:
                c5_h = c5_bh
                h_interp = h_bin_h
            h_interp, c5_h = slsqp_refine(h_interp, n_try, beta_list=[200, 1000, 3000])
            if c5_h < best_c5 - 1e-10:
                best_c5 = c5_h
                best_h = h_interp.copy()
                best_n = n_try
                print("  n={}: C5={:.10f} (NEW BEST)".format(n_try, best_c5))
            else:
                print("  n={}: C5={:.10f}".format(n_try, c5_h))

    # Phase 8: Final polish
    if remaining_time() > 15.0:
        print("\n--- Phase 8: Final polish (n={}) ---".format(best_n))
        _G_N = best_n
        _G_TGT = best_n / 2.0
        _G_DX = 2.0 / best_n
        best_h, best_c5 = slsqp_refine(best_h, best_n, beta_list=[3000, 8000, 15000])
        h_fbin, c5_fbin = binary_sa(best_h, best_n, max_iter=1500, rng=rng, T0=0.01, gamma=0.9995)
        if c5_fbin < best_c5:
            best_c5 = c5_fbin
            best_h = h_fbin.copy()
        best_h, best_c5 = slsqp_refine(best_h, best_n, beta_list=[5000, 10000])
        print("  C5={:.10f}".format(best_c5))

    # Final
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass
    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
