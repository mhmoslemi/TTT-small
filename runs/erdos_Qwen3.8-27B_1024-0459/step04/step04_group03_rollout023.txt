corr[m] = sum_{i': 0≤i'<n, 0≤i'+m<n} h[i'+m]*(1-h[i'])

Change when h[i] goes 1→0:
- Terms where i' = i: was h[i+m]*(1-1)=0, now h[i+m]*(1-0)=h[i+m]. Change: +h[i+m] (if i+m valid)
- Terms where i'+m = i, i.e., i' = i-m: was h[i]*(1-h[i-m])=1-h[i-m], now 0. Change: -(1-h[i-m]) (if i-m valid)

Change when h[j] goes 0→1:
- Terms where i' = j: was h[j+m]*(1-0)=h[j+m], now h[j+m]*(1-1)=0. Change: -h[j+m] (if j+m valid)
- Terms where i'+m = j, i.e., i' = j-m: was h[j]*(1-h[j-m])=0, now 1-h[j-m]. Change: +(1-h[j-m]) (if j-m valid)

Total Δcorr[m]:
= +h[i+m]*I(0≤i+m<n) - (1-h[i-m])*I(0≤i-m<n) - h[j+m]*I(0≤j+m<n) + (1-h[j-m])*I(0≤j-m<n)

For binary h:
= h[i+m]*I(0≤i+m<n) - (1-h[i-m])*I(0≤i-m<n) - h[j+m]*I(0≤j+m<n) + (1-h[j-m])*I(0≤j-m<n)

This is a vector of length 2n-1 that I can compute in O(n) with numpy indexing.

Let me implement this efficiently:

```python
def compute_swap_delta(h, i, j, n):
    """Compute change in corr when h[i] goes 1->0 and h[j] goes 0->1 (binary h)."""
    m = np.arange(-(n-1), n)  # shifts
    corr_len = 2*n - 1
    delta = np.zeros(corr_len)
    
    # +h[i+m] where 0<=i+m<n
    im = i + m
    mask_im = (im >= 0) & (im < n)
    delta[mask_im] += h[im[mask_im]]
    
    # -(1-h[i-m]) where 0<=i-m<n
    im_neg = i - m
    mask_im_neg = (im_neg >= 0) & (im_neg < n)
    delta[mask_im_neg] -= (1 - h[im_neg[mask_im_neg]])
    
    # -h[j+m] where 0<=j+m<n
    jm = j + m
    mask_jm = (jm >= 0) & (jm < n)
    delta[mask_jm] -= h[jm[mask_jm]]
    
    # +(1-h[j-m]) where 0<=j-m<n
    jm_neg = j - m
    mask_jm_neg = (jm_neg >= 0) & (jm_neg < n)
    delta[mask_jm_neg] += (1 - h[jm_neg[mask_jm_neg]])
    
    return delta
```

For binary h, each term is 0 or 1, so delta[m] ∈ {-2, -1, 0, 1, 2} roughly.

Then: new_corr = corr + delta, check if max(new_corr) < max(corr).

This is O(n) per swap (just a few numpy operations on vectors of length 2n-1=511). Very fast.

import numpy as np

import time

_BUDGET_S = 1300.0
_N_G = 0
_BETA_G = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def binary_local_search(h_bin, n, max_sweeps=200, rng=None, sa=False):
    """Greedy + optional simulated annealing on binary solution."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = h_bin.copy().astype(float)
    m_all = np.arange(-(n-1), n)
    corr = compute_corr(h)
    best_max = np.max(corr)
    h_best = h.copy()
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    T_init = 2.0
    T_final = 0.01
    total_sweeps = max_sweeps
    sweep = 0
    no_improve = 0
    
    while sweep < total_sweeps:
        if remaining_time() < 5:
            break
        T = T_init * (T_final / T_init) ** (sweep / max(total_sweeps, 1))
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        for _ in range(3):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            if i == j:
                continue
            
            # Compute delta
            im = i + m_all
            mask_im = (im >= 0) & (im < n)
            im_neg = i - m_all
            mask_imn = (im_neg >= 0) & (im_neg < n)
            jm = j + m_all
            mask_jm = (jm >= 0) & (jm < n)
            jm_neg = j - m_all
            mask_jmn = (jm_neg >= 0) & (jm_neg < n)
            
            delta = np.zeros(2*n-1)
            delta[mask_im] += h[im[mask_im]]
            delta[mask_imn] -= (1.0 - h[im_neg[mask_imn]])
            delta[mask_jm] -= h[jm[mask_jm]]
            delta[mask_jmn] += (1.0 - h[jm_neg[mask_jmn]])
            
            new_corr = corr + delta
            new_max = np.max(new_corr)
            
            if new_max < best_max:
                # Accept and update
                h[i] = 0.0
                h[j] = 1.0
                corr = new_corr
                best_max = new_max
                h_best = h.copy()
                improved = True
                no_improve = 0
            elif sa and rng.random() < np.exp(-(new_max - best_max) / max(T, 1e-10)):
                h[i] = 0.0
                h[j] = 1.0
                corr = new_corr
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        sweep += 1
    
    return h_best, best_max * (2.0/n)

def projected_gradient_smooth(h, n, beta=100, max_iter=5000, alpha_init=0.002):
    """Projected gradient with log-sum-exp smoothing."""
    h = h.copy()
    alpha = alpha_init
    v = np.zeros(n)  # momentum
    gamma = 0.9  # momentum coefficient
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stall = 0
    
    for it in range(max_iter):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        # log-sum-exp for numerical stability
        max_corr = np.max(corr)
        shifted = beta * (corr - max_corr)
        log_sum_exp = max_corr + np.log(np.sum(np.exp(shifted))) / beta
        dx = 2.0 / n
        obj = log_sum_exp * dx
        
        if obj < best_c5 - 1e-15:
            best_c5 = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Weights for active lags
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            weights = np.ones(2*n-1) / (2*n-1)
            w_sum = 1.0
        weights = weights / w_sum
        
        # Only use lags with weight > threshold for gradient computation
        thresh = 0.01 / (2*n-1)
        active = np.where(weights > thresh)[0]
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for k in active:
            p = k - (n-1)
            if p == 0:
                grad += weights[k] * (1.0 - 2.0 * h)
            else:
                j_min = max(0, p)
                j_max = min(n, n + p)
                if j_min < j_max:
                    idx = np.arange(j_min, j_max)
                    grad[idx] += weights[k] * (1.0 - h[idx - p])
                j_min2 = max(0, -p)
                j_max2 = min(n, n - p)
                if j_min2 < j_max2:
                    idx = np.arange(j_min2, j_max2)
                    grad[idx] -= weights[k] * h[idx + p]
        
        grad = dx * grad
        
        # Nesterov momentum
        v = gamma * v - alpha * grad
        h_new = h + v
        h_new = project_feasible(h_new, n)
        
        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new < obj:
            h = h_new
            alpha = min(alpha * 1.05, 0.01)
        else:
            alpha *= 0.7
            v *= 0.5
            h_new2 = h - alpha * grad
            h_new2 = project_feasible(h_new2, n)
            h = h_new2
        
        if stall > 200:
            break
    
    return best_h, best_c5

def binary_to_continuous(h_bin, n, beta_seq=[50, 200, 1000, 5000], max_iter_each=2000):
    """Start from binary, then do continuous optimization."""
    h = h_bin.astype(float).copy()
    h = project_feasible(h, n)
    for beta in beta_seq:
        if remaining_time() < 10:
            break
        h, _ = projected_gradient_smooth(h, n, beta=beta, max_iter=max_iter_each)
    return h, compute_c5(h)

def slsqp_smooth(h, n, beta, maxiter=300):
    global _N_G, _BETA_G
    _N_G = n
    _BETA_G = beta
    
    def sm_obj(x):
        h_x = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h_x, 1.0 - h_x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0/n)
    
    def sm_eq(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sm_eq}]
    try:
        res = minimize(sm_obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def pair_transfer_search(h, n, max_sweeps=10, rng=None):
    """Local search: transfer mass between pairs of non-binary coordinates."""
    if rng is None:
        rng = np.random.RandomState(0)
    best_h = h.copy()
    best_c5 = compute_c5(h)

        order = rng.permutation(len(non_bin))
        for idx in range(len(order)):
            if remaining_time() < 1:

            i = int(non_bin[order[idx]])
            hi = best_h[i]
            for dj in range(1, min(20, n)):
                j = (i + dj) % n
                hj = best_h[j]

                    continue
                for frac in [0.3, 0.5, 0.7]:

                    if abs(amt) < 0.02:
                        continue

                    h_trial[i] -= amt
                    h_trial[j] += amt

                    c5_t = compute_c5(h_trial)
                    if c5_t < best_c5 - 1e-14:
                        best_c5 = c5_t
                        best_h = h_trial.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:

    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
