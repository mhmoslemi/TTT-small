        w = weights[k_idx]
        if w < 1e-12:
            continue
        offset = k_idx - (n-1)
        if offset >= 0:
            n_pairs = n - offset
            if n_pairs <= 0: continue
            grad[:n_pairs] += w * (1.0 - h[offset:offset+n_pairs])
            grad[offset:] -= w * h[:n_pairs]
        else:
            abs_off = -offset
            n_pairs = n - abs_off
            if n_pairs <= 0: continue
            grad[:n_pairs] -= w * h[abs_off:abs_off+n_pairs]
            grad[abs_off:] += w * (1.0 - h[:n_pairs])
    return grad * dx

def lse_obj(x, n, dx, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 0))
    return (m + np.log(np.sum(exps) + 1e-300) / beta) * dx

def lse_grad(x, n, dx, beta):
    return get_gradient(x, n, beta=beta)

def do_slsqp(h0, n, beta_list, time_fn, max_seconds_per_beta=60):
    dx = 2.0 / n
    tgt = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt,
             'jac': lambda x: np.ones(n)}]
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    
    for beta in beta_list:
        remaining = time_fn()
        if remaining < 5.0:
            break
        elapsed_beta = time.time()
        try:
            res = minimize(
                lambda x: lse_obj(x, n, dx, beta),
                h_cur,
                jac=lambda x: lse_grad(x, n, dx, beta),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-18, 'iprint': 0}
            )
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                h_cur = h_new.copy()
        except Exception:
            pass
        # Check time
        if time.time() - elapsed_beta > max_seconds_per_beta:
            break
    return h_cur, c5_cur

def do_active_subgrad(h0, n, time_fn, rng, max_iters=200, top_k=5):

    h_best = h.copy()
    no_improve = 0
    for it in range(max_iters):

        top_k_actual = min(len(active), top_k + len(active)//2)

        grad = np.zeros(n)

                if n_pairs <= 0: continue
                grad[:n_pairs] -= h[abs_off:abs_off+n_pairs]
                grad[abs_off:] += (1.0 - h[:n_pairs])
        grad *= dx_val / len(idx)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        grad /= gnorm

        for step in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.001]:

                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True

                        h = h_pert.copy()
                        no_improve = 0

    h = np.array(h, dtype=float).copy()

    for r in range(6):
        if time_fn() < 12:
            break
        sc = 0.004 * (1.0 + 0.4 * r)
        hp = project_feasible(best_h + rng.randn(best_n) * sc, best_n)
        cp = compute_c5(hp)
        hs = hp if cp < best_c5 else best_h.copy()
        cs = min(cp, best_c5)
        hn, cn = do_slsqp(hs, best_n, [5000, 50000], time_fn, max_iters=200)
        if cn < best_c5 - 1e-14:
            best_c5 = cn
            best_h = hn.copy()
            print("  R{:d}: C5={:.10f}".format(r, cn))

    # Phase 8: Coordinate descent
    if time_fn() > 8:
        print("P8: Coord")
        best_h, best_c5 = do_coord_flip(best_h, best_n, time_fn, rng, max_sweeps=8)
        print("  C5={:.10f}".format(best_c5))

    # Phase 9: Final subgradient
    if time_fn() > 5:
        best_h, best_c5 = do_subgrad(best_h, best_n, time_fn, rng, max_iters=80, band_frac=0.02, max_lags=12)

    # Phase 10: Final SLSQP
    if time_fn() > 5:
        best_h, best_c5 = do_slsqp(best_h, best_n, [200000, 500000], time_fn, max_iters=100)

    # Phase 11: Final tiny subgrad
    if time_fn() > 3:
        best_h, best_c5 = do_subgrad(best_h, best_n, time_fn, rng, max_iters=50, band_frac=0.01, max_lags=15)

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        ec = evaluate_erdos_solution(best_h)
        if ec < final_c5:
            final_c5 = float(ec)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
