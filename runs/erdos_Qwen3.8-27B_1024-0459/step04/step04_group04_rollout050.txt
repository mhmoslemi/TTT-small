import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 100.0
_TGT = 0.0
_PEN = 500.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        sel = rng.choice(len(idx), n_ones - target, replace=False)
        h_bin[idx[sel]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        sel = rng.choice(len(idx), target - n_ones, replace=False)
        h_bin[idx[sel]] = 1
    return h_bin


def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    s = np.sum(ex)
    if s < 1e-300:
        return np.zeros(n)
    w = ex / s
    g = np.zeros(n)
    thresh = 1e-12
    for k in range(len(corr)):
        if w[k] > thresh:
            off = k - (n - 1)
            if off >= 0:
                npairs = n - off
                g[off:off + npairs] += w[k] * (1.0 - x[:npairs])
                g[:npairs] -= w[k] * x[off:off + npairs]
            else:
                ao = -off
                npairs = n - ao
                g[:npairs] += w[k] * (1.0 - x[ao:ao + npairs])
                g[ao:ao + npairs] -= w[k] * x[:npairs]
    return g * dx_val


def pen_obj(x):
    base = smooth_obj(x)
    s = np.sum(x) - _TGT
    return base + _PEN * s * s


def pen_grad(x):
    base_g = smooth_grad(x)
    s = np.sum(x) - _TGT
    return base_g + 2.0 * _PEN * s * np.ones(_N)


def _sum_con_fun(x):
    return np.sum(x) - _TGT


def _sum_con_jac(x):
    return np.ones(_N)


def lbfgs_smooth(h, n, beta, maxiter=800, pen_coef=500.0):
    global _N, _BETA, _TGT, _PEN
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    _PEN = pen_coef
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(pen_obj, h, jac=pen_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-18, 'maxls': 100, 'maxfun': 5000})
        return project_feasible(res.x, n)
    except Exception:
        return h


def slsqp_smooth(h, n, beta, maxiter=500):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})

    except Exception:
        return h


def topk_descent(h, n, max_iters=800, top_k=8):

    best_c5 = compute_c5(h)
    dx_val = 2.0 / n
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        m = float(np.max(corr))
        if m * dx_val >= best_c5 + 1e-8:
            break
        top_idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k in top_idx:
            kk = int(k)
            off = kk - (n - 1)
            if off >= 0:
                npairs = n - off
                g[off:off + npairs] += (1.0 - h[:npairs])
                g[:npairs] -= h[off:off + npairs]
            else:
                ao = -off
                npairs = n - ao
                g[:npairs] += (1.0 - h[ao:ao + npairs])
                g[ao:ao + npairs] -= h[:npairs]
        g /= len(top_idx)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        d = -g / gnorm
        improved = False
        for step in [0.05, 0.02, 0.01, 0.1, 0.005, 0.03, 0.008, 0.001, 0.04]:
            h_trial = h + step * d

                best_c5 = c_new

    return h, best_c5


def sa_binary(h0, n, time_limit=60, seed=42):

    stall = 0
    max_stall = 8000
    pair_count = 0

    while time.time() - start < time_limit:
        if remaining() < 8:
            break
        if T < 0.05:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        if pair_count % 7 == 0 and len(ones) >= 2 and len(zeros) >= 2:

                    stall = 0
                else:
                    stall += 1
            else:

                h[i2] = 1
                h[j1] = 0
                h[j2] = 0
                stall += 1
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

            h[j] = 1

                    stall = 0

                h[j] = 0
                stall += 1

        c5_sls2 = compute_c5(h_sls2)
        if c5_sls2 < best_c5:
            best_c5 = c5_sls2
            best_h = h_sls2.copy()
            h_best = h_sls2.copy()

    # Phase 5: Binary SA final push
    if remaining() > 25:
        h_bin_f = make_binary(best_h, best_n, rng)
        t_final = min(60, max(10, remaining() * 0.5))
        for fs in [seed + 9000, seed + 9500, seed + 10000]:
            if remaining() < 10:
                break
            h_sf, c5_sf = sa_binary(h_bin_f, best_n, time_limit=t_final // 3, seed=fs)
            if c5_sf < best_c5:
                best_c5 = c5_sf
                best_h = h_sf.copy()
                h_bin_f = h_sf.copy()

    # Shift-focused local search
    if remaining() > 15:
        h_sf, c5_sf = shift_focus_local(best_h, best_n, max_rounds=100, top_k=4)
        if c5_sf < best_c5:
            best_c5 = c5_sf
            best_h = h_sf.copy()

    # Final top-k
    if remaining() > 8:
        h_tdf, c5_tdf = topk_descent(best_h, best_n, max_iters=300, top_k=3)
        if c5_tdf < best_c5:
            best_c5 = c5_tdf
            best_h = h_tdf.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
