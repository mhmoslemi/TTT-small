```python
    # This is hard to check efficiently for all pairs, so let's just try all pairs
    
    for j in range(n):
        for m_db in range(n):
            if j == m_db:
                continue
            # Feasible delta range
            d_min = max(-h[j], h[m_db] - 1)
            d_max = min(1 - h[j], h[m_db])
            if d_min >= d_max - 1e-15:
                continue
            
            b = G[j, :] - G[m_db, :]
            # f(delta) = max_k (corr[k] + delta * b[k])
            # Try delta candidates: 0, d_min, d_max, and kinks
            
            # Quick check: does this pair help reduce the max?
            # The slope at delta=0 for the max is:
            # Among active k, the max slope is max(b[active])
            # If max(b[active]) < 0, then increasing delta helps
            # If min(b[active]) > 0, then decreasing delta helps
            
            # For efficiency, just check the top few correlations
            top_idx = np.argsort(-corr)[:top_k]
            b_top = b[top_idx]
            
            if np.max(b_top) >= 0 and np.min(b_top) <= 0:
                # Mixed slopes: delta=0 might be optimal, or a small step
                pass
            
            # Compute f at a few candidate points
            candidates = [0.0, d_min, d_max]
            # Add kink points for top correlations
            kref = top_idx[0]  # current max
            for ki in range(min(top_k, 5)):
                if b[top_idx[ki]] != b[kref]:
                    dk = (corr[kref] - corr[top_idx[ki]]) / (b[top_idx[ki]] - b[kref])
                    if d_min <= dk <= d_max:
                        candidates.append(dk)
            
            best_f = np.max(corr)  # current value
            best_d = 0.0
            for dc in candidates:
                dc = np.clip(dc, d_min, d_max)
                f_val = np.max(corr + dc * b)
                if f_val < best_f - 1e-12:
                    best_f = f_val
                    best_d = dc
            
            if best_d != 0 and best_f < m - best_improvement:
                best_improvement = m - best_f
                best_pair = (j, m_db)
                best_delta = best_d
    
    if best_pair is not None:
        j, m_db = best_pair
        h[j] += best_delta
        h[m_db] -= best_delta
        # Recompute corr and G
        corr = compute_corr(h)
        G = compute_grad_matrix(h, n)
        return h, corr, G, True
    return h, corr, G, False
```

```python
def project_box_sum(h, n):
    """Project h onto {x : 0<=x<=1, sum(x)=n/2}."""
    target = n / 2.0
    # Bisection on lambda
    # x_i = clip(h_i - lambda, 0, 1)
    # Find lambda such that sum(x_i) = target
    lo = h.min() - 1
    hi = h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo+hi)/2, 0, 1)
    # Final correction
    diff = target - x.sum()
    if abs(diff) > 1e-15:
        mask = (x > 0) & (x < 1)
        if mask.sum() > 0:
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0, 1)
    return x
```

**Coordinate descent on individual variables with exact 1D solution:**

For variable h[j] (with all others fixed), the objective is:

For this, the objective becomes:

And the constraint is:

To solve this exactly, I need to find the minimum of a piecewise-linear convex function on an interval. This can be done in O(n) time (by finding the relevant kink points).

For a practical implementation, I can:
1. For each j, precompute G[j, :]
2. For each m, compute b = G[j, :] - G[m, :]
3. Find the minimum of max_k (corr[k] + δ * b[k]) on the feasible interval

The key insight for step 3: the function f(δ) = max_k (a_k + b_k * δ) is convex and piecewise linear. Its minimum occurs at a point where the subdifferential contains 0, which means:
- There exist k1, k2 such that a_k1 + b_k1 * δ* = a_k2 + b_k2 * δ* (they're both at the max)
- And b_k1 * δ* + a_k1 ≤ a_k + b_k * δ* for all k (all others are below)
- The "slope" changes from negative to positive at δ*

Practically:
- The active set at δ=0 is {k : a_k = max(a)}. Let S0 be this set.
- For δ > 0: the binding constraint is the k in S0 with the largest b_k (or the k that "catches up" first)
- For δ < 0: the binding constraint is the k in S0 with the smallest b_k

The optimal δ* is:
- If max_{k in S0} b_k ≤ 0 and min_{k in S0} b_k ≥ 0: δ* = 0
- If max_{k in S0} b_k > 0: δ* < 0 (decrease δ)
  δ* = min over k not in S0 of (a_k - max_S0) / (b_k - b_{argmax_S0 b})  [the first k to catch up]
  But also bounded below by the feasible lower bound.
- If min_{k in S0} b_k < 0: δ* > 0 (increase δ)
  Similarly.

```python
def pair_swap_optimize(h, n, max_passes=100, top_k=10):
    """Pair-swap local optimization."""
    h = h.copy()
    corr = compute_corr(h)
    G = compute_grad_matrix(h, n)
    current_max = np.max(corr)
    
    for pass_num in range(max_passes):
        if remaining() < 5:
            break
        improved = False
        
        for j in range(n):
            if remaining() < 3:
                break
            
            # Compute b for all m at once
            b_all = G[j, :] - G  # shape (n, 2n-1), b_all[m, k] = G[j,k] - G[m,k]
            
            # Candidate deltas
            d_min_arr = np.maximum(-h[j], h - 1.0)  # shape (n,)
            d_max_arr = np.minimum(1.0 - h[j], h)    # shape (n,)
            feasible = d_max_arr - d_min_arr > 1e-12
            
            # Generate candidate deltas
            # 1. 0
            # 2. d_min, d_max
            # 3. Kink points for top correlations
            
            top_idx = np.argsort(-corr)[:top_k]
            
            # Kink points: for top correlation kref, the kink with correlation ki is at:
            # delta = (corr[kref] - corr[ki]) / (b[ki] - b[kref])
            kref = top_idx[0]
            
            # Evaluate at several deltas
            delta_candidates = np.array([0.0])
            # Add kink points (vectorized over m)
            for ki in range(1, min(top_k, 6)):
                denom = b_all[:, top_idx[ki]] - b_all[:, kref]
                # where denom != 0
                valid = np.abs(denom) > 1e-15
                dk = np.zeros(n)
                dk[valid] = (corr[kref] - corr[top_idx[ki]]) / denom[valid]
                # Clip to feasible range
                dk = np.maximum(dk, d_min_arr)
                dk = np.minimum(dk, d_max_arr)
                # Only keep valid ones
                dk[~valid] = 0.0
                delta_candidates = np.vstack([delta_candidates, dk])
            
            # Also add d_min and d_max
            delta_candidates = np.vstack([delta_candidates, d_min_arr[None, :], d_max_arr[None, :]])
            
            # Evaluate f at all candidates for all m
            # f[di, m] = max_k (corr[k] + delta_candidates[di, m] * b_all[m, k])
            # This is: for each di, compute max over k of (corr + delta_candidates[di] * b_all)
            
            # Vectorized: shape (n_deltas, n, 2n-1) → max over last axis → (n_deltas, n)
            # But this might use too much memory for large n_deltas
            
            # Let's process in batches
            best_improvement = 0
            best_m = -1
            best_delta = 0
            
            n_deltas = len(delta_candidates)
            for di in range(n_deltas):
                d_val = delta_candidates[di, :]  # shape (n,)
                # f[m] = max_k (corr[k] + d_val[m] * b_all[m, k])
                # = max_k (corr[None, :] + d_val[:, None] * b_all)
                # shape (n, 2n-1) → max over axis 1 → (n,)
                
                # Only evaluate for feasible m
                if not np.any(feasible):
                    break
                
                f_vals = np.max(corr[None, :] + d_val[:, None] * b_all, axis=1)
                
                # Only consider feasible m where this delta is in range
                valid_m = feasible & (d_val >= d_min_arr - 1e-12) & (d_val <= d_max_arr + 1e-12)
                
                if not np.any(valid_m):
                    continue
                
                f_valid = f_vals[valid_m]
                improvement = current_max - f_valid
```

Let me try a different tactic: **exact direction finding**.

Given the current solution h with max correlation M at shifts S, I want to find a direction d (with sum(d) = 0) that reduces M.

The condition for d to reduce M is:
For all k in S: G[:,k].T @ d < 0 (the correlation at k decreases)

This is a system of linear inequalities. If a feasible d exists, we can take a step in that direction.

If no feasible d exists (i.e., the current point is "stuck"), we need to change the active set.

This is essentially a linear programming feasibility problem! And we can solve it with scipy or even manually.

But the direction also needs to be feasible with respect to the box constraints (0 ≤ h + αd ≤ 1 for small α > 0).

Let me formulate this:
Find d such that:
1. sum(d) = 0
2. G[:,k].T @ d ≤ -ε for k in S (reduce active correlations)
3. d[j] > 0 for j where h[j] = 0 (don't go below 0)
4. d[j] < 0 for j where h[j] = 1 (don't go above 1)
5. ||d|| = 1 (normalization)

This is a linear feasibility problem (ignoring constraint 5). We can solve it with scipy.optimize.linprog or with a simple greedy approach.

Actually, for a simpler approach: just find d that minimizes the maximum increase among active shifts:
minimize max_{k in S} (G[:,k].T @ d)
subject to sum(d) = 0, ||d||_2 ≤ 1, and box feasibility

If the optimal value is negative, we have a descent direction.

```python
from scipy.optimize import linprog

def find_descent_direction(h, n, G, corr):
    """Find a direction that reduces the max correlation."""
    M = np.max(corr)
    active = np.where(corr >= M - 1e-8)[0]
    
    # Minimize max_{k in active} (G[:,k].T @ d)
    # equivalently: minimize t subject to G[:,k].T @ d <= t for k in active
    # This is: minimize t
    # subject to: G[:,k].T @ d - t <= 0 for k in active
    #             sum(d) = 0
    #             -1 <= d[j] <= 1 for j where 0 < h[j] < 1
    #             d[j] >= 0 for j where h[j] = 0
    #             d[j] <= 0 for j where h[j] = 1
    
    # Variables: d[0], ..., d[n-1], t  (total n+1 variables)
    # Objective: minimize t → c = [0, 0, ..., 0, 1]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # G[:,k].T @ d - t <= 0 for each active k
    # [G[:,k], -1] @ [d, t] <= 0
    A_ub = np.hstack([G[:, active].T, -np.ones((len(active), 1))])
    b_ub = np.zeros(len(active))
    
    # Box constraints on d
    # d[j] >= 0 when h[j] = 0  → -d[j] <= 0
    # d[j] <= 0 when h[j] = 1  → d[j] <= 0
    # -1 <= d[j] <= 1 when 0 < h[j] < 1
    A_box = np.zeros((2*n, n))
    b_box = np.zeros(2*n)
    
    h_at_0 = np.where(h <= 1e-10)[0]
    h_at_1 = np.where(h >= 1 - 1e-10)[0]
    h_frac = np.where((h > 1e-10) & (h < 1 - 1e-10))[0]
    
    # d[j] >= 0 for h_at_0: -d[j] <= 0
    for j in h_at_0:
        A_box[j, j] = -1.0
        b_box[j] = 0.0
    # d[j] <= 0 for h_at_1: d[j] <= 0
    for j in h_at_1:
        A_box[j + n, j] = 1.0
        b_box[j + n] = 0.0
    # -1 <= d[j] <= 1 for h_frac
    for j in h_frac:
        A_box[j, j] = -1.0
        b_box[j] = 1.0
        A_box[j + n, j] = 1.0
        b_box[j + n] = 1.0
    
    # Equality constraint: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Solve LP
    res = linprog(c, A_ub=np.vstack([A_ub, A_box[:, :n+1-1]]), b_ub=np.concatenate([b_ub, b_box]),
                  A_eq=A_eq, b_eq=b_eq, bounds=[(None, None)] * n + [(None, None)],
                  method='highs')
    
    if res.status == 0 and res.fun < -1e-8:
        return res.x[:n]  # descent direction found
    return None
```
