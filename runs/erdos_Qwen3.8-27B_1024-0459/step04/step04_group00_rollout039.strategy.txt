def project_feasible(h, n):
    # Enforce 0<=h<=1 and sum(h) = n/2 (integral=1)
    target = n/2.0
    # We use iterative adjustment
    h = np.array(h, dtype=float).copy()
    # Clip
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if abs(total - target) < 1e-10:
        return h
    diff = target - total
    # Add or remove mass
    h_sorted = np.sort(h)
    if diff > 0:
        # Need to add mass
        # Choose the smallest 50%
        ind1 = int(len(h)*0.25)
        ind2 = int(len(h)*0.75)
        candidates = h_sorted[ind1:ind2].copy()
        candidates += diff / len(candidates)
        h_sorted[ind1:ind2] = candidates
    else:
        # Remove mass
        candidates = h_sorted[:int(len(h)/2)].copy()
        diff2 = -diff / len(candidates)
        candidates += diff2   # Actually remove by lowering
        candidates = np.clip(candidates, 0, 1)
        h_sorted[:int(len(h)/2)] = candidates
    # Reassign the sorted values after making sure it's bounded
    h = np.clip(h_sorted[:len(h)], 0, 1)
    # Adjust the sum if it doesn't match enough
    total = np.sum(h)
    if abs(total - target) > 1e-5:
        h = h * target / total
    return h
