                h[j] = 0.0
        if found:
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                corr = compute_corr(h)
                cur_max = float(np.max(corr))
                stall = 0
                max_stall = min(max_stall + 50, 800)
    return best_h, best_c5


def simulated_annealing_binary(h, n, time_limit, rng):
    """Simulated annealing for binary solutions."""
    h = h.copy()
    n_ones = int(np.sum(h == 1))
    target_ones = n // 2
    if n_ones > target_ones:
        ones = np.where(h == 1)[0]
        remove = rng.choice(ones, n_ones - target_ones, replace=False)
        h[remove] = 0
    elif n_ones < target_ones:
        zeros = np.where(h == 0)[0]
        add = rng.choice(zeros, target_ones - n_ones, replace=False)
        h[add] = 1
    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx_val
    best_h = h.copy()
    start = time.time()
    t0 = time_limit
    T = 0.5
    T_final = 0.01

    for it in range(50000):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break
        T = T_final + (T - T_final) * (1.0 - (time.time() - start) / t0)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        delta = new_max - cur_max
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-8)):
            corr = new_corr
            cur_max = new_max
            if new_max * dx_val < best_c5:
                best_c5 = new_max * dx_val
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5


def symmetric_init(n, rng):
    """Create symmetric binary init: h[i + n/2] = 1 - h[i]."""
    h = np.zeros(n)
    half = n // 2
    # Choose n/4 random positions in first half
    positions = rng.choice(half, half // 2, replace=False)
    h[positions] = 1.0
    h[half + positions] = 0.0
    # Fill rest of second half with 1s
    remaining_pos = [i for i in range(half) if h[i] == 0]
    for p in remaining_pos:
        h[half + p] = 1.0
    return h


def greedy_binary_init(n, rng):
    """Greedy binary construction: add ones to minimize max correlation increase."""
    h = np.zeros(n)
    target = n // 2
    corr = np.zeros(2 * n - 1)  # all zeros initially
    cur_max = 0.0
    n_corr = 2 * n - 1

    for step in range(target):
        best_p = -1
        best_max = float('inf')
        # Evaluate all candidates
        for p in range(n):
            if h[p] > 0:
                continue
            # Delta: corr[d] += (1-h[p-d]) - h[p+d] for valid indices
            d_range = np.arange(-(n - 1), n)
            contrib = np.zeros(n_corr)
            # h[p+d] term: valid when 0 <= p+d < n
            idx_plus = (p + d_range >= 0) & (p + d_range < n)
            if np.any(idx_plus):
                contrib[idx_plus] += h[p + d_range[idx_plus]]
            # (1-h[p-d]) term: valid when 0 <= p-d < n
            idx_minus = (p - d_range >= 0) & (p - d_range < n)
            if np.any(idx_minus):
                contrib[idx_minus] += (1.0 - h[p - d_range[idx_minus]])
            new_corr = corr + contrib
            new_max = float(np.max(new_corr))
            if new_max < best_max:
                best_max = new_max
                best_p = p
        # Place one at best_p
        h[best_p] = 1.0
        # Update corr
        d_range = np.arange(-(n - 1), n)
        contrib = np.zeros(n_corr)
        idx_plus = (best_p + d_range >= 0) & (best_p + d_range < n)
        if np.any(idx_plus):
            contrib[idx_plus] += h[best_p + d_range[idx_plus]]
        idx_minus = (best_p - d_range >= 0) & (best_p - d_range < n)
        if np.any(idx_minus):
            contrib[idx_minus] += (1.0 - h[best_p - d_range[idx_minus]])
        corr += contrib
        cur_max = float(np.max(corr))
    return h


def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        if npairs > 0:
            g[off:off + npairs] += (1.0 - x[:npairs])
            g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        if npairs > 0:
            g[:npairs] += (1.0 - x[ao:ao + npairs])
            g[ao:ao + npairs] -= x[:npairs]
    return g


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-14:
            g += w[k] * corr_grad_per_shift(x, n, k)
    return g * dx_val


def sum_con(x):
    return np.sum(x) - _TGT


def sum_con_jac(x):
    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=400):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def corr_slope_at_i(h, i, n):
    nc = 2 * n - 1
    slopes = np.zeros(nc)
    for idx in range(nc):
        d = idx - (n - 1)
        s = 0.0
        if d >= 0:
            if d <= i < n:
                s += (1.0 - h[i - d])
            if 0 <= i < n - d:
                s -= h[i + d]
        else:
            m = -d
            if 0 <= i < n - m:
                s += (1.0 - h[i + m])
            if i >= m:
                s -= h[i - m]
        slopes[idx] = s
    return slopes


def coord_opt_1d(slopes, intercepts, lo, hi):
    if lo >= hi:
        return lo
    for _ in range(40):
        mid = (lo + hi) / 2.0
        vals = slopes * mid + intercepts
        max_val = vals.max()
        active = vals >= max_val - 1e-13
        if slopes[active].max() < -1e-14:
            lo = mid
        elif slopes[active].min() > 1e-14:
            hi = mid
        else:
            break
    return (lo + hi) / 2.0


def pair_coord_descent(h, n, max_passes=200, time_limit=60, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    start = time.time()
    best_c5 = compute_c5(h)
    corr = compute_corr(h)
    n_corr = 2 * n - 1

    for pass_num in range(max_passes):
        if time.time() - start > time_limit:
            break
        if remaining() < 10:
            break
        if pass_num % 7 == 6:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        improved = False
        for idx in range(n):
            i = int(order[idx])
            j = (i + n // 2) % n
            if i == j:
                j = (i + 1) % n
            s = h[i] + h[j]
            lo_t = max(0.0, s - 1.0)
            hi_t = min(1.0, s)
            if lo_t >= hi_t - 1e-14:
                continue
            sl_i = corr_slope_at_i(h, i, n)
            sl_j = corr_slope_at_i(h, j, n)
            diff_sl = sl_i - sl_j
            intercepts = corr - h[i] * diff_sl
            t_opt = coord_opt_1d(diff_sl, intercepts, lo_t, hi_t)
            if abs(t_opt - h[i]) < 1e-14:
                continue
            delta = t_opt - h[i]
            corr += delta * diff_sl
            h[i] = t_opt
            h[j] = s - t_opt
            improved = True
        if improved:
            h = project_feasible(h, n)
            corr = compute_corr(h)
            c5_new = float(np.max(corr) * 2.0 / n)
            if c5_new < best_c5:
                best_c5 = c5_new
    return h, best_c5


def multi_density_local_search(h, n, time_limit, rng):
    """Local search with occasional triple moves for better mixing."""
    h = h.copy()
    n_ones = int(np.sum(h == 1))
    target_ones = n // 2
    if n_ones > target_ones:
        ones = np.where(h == 1)[0]
        remove = rng.choice(ones, n_ones - target_ones, replace=False)
        h[remove] = 0
    elif n_ones < target_ones:
        zeros = np.where(h == 0)[0]
        add = rng.choice(zeros, target_ones - n_ones, replace=False)
        h[add] = 1
    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx_val
    best_h = h.copy()
    start = time.time()
    stall = 0

    for it in range(80000):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        k_max = int(np.argmax(corr))
        offset = k_max - (n - 1)
        found = False
        n_try = 150
        for _ in range(n_try):
            # 70% biased, 30% random
            if rng.rand() < 0.7:
                if offset >= 0:
                    bz = offset
                else:
                    bz = -offset
                candidates = [i for i in ones if i + bz < n and h[i + bz] == 0]
                if candidates:
                    i = candidates[rng.randint(len(candidates))]
                else:
                    i = ones[rng.randint(len(ones))]
            else:
                i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            if new_max < cur_max - 1e-14:
                corr = new_corr
                cur_max = new_max
                best_c5 = cur_max * dx_val
                best_h = h.copy()
                found = True
                k_max = int(np.argmax(corr))
                offset = k_max - (n - 1)
                break
            else:
                h[i] = 1.0
                h[j] = 0.0
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 300:
                # Triple move: swap 3 pairs
                for _ in range(3):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i] = 0.0
                    h[j] = 1.0
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                corr = compute_corr(h)
                cur_max = float(np.max(corr))
                stall = 0
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 320
        parent_h = np.zeros(parent_n)
        half = parent_n // 2
        parent_h[:half] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Intensive binary local search on parent
    n_p = best_n
    h_bin = (best_h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin == 1))
    if n_ones != n_p // 2:
        if n_ones > n_p // 2:
            idx = np.where(h_bin == 1)[0]
            rem = rng.choice(idx, n_ones - n_p // 2, replace=False)
            h_bin[rem] = 0
        else:
            idx = np.where(h_bin == 0)[0]
            add = rng.choice(idx, n_p // 2 - n_ones, replace=False)
            h_bin[add] = 1
    if remaining() > 30:
        tl = min(50, remaining() - 20)
        h_b, c5_b = binary_local_search(h_bin, n_p, tl, rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n_p

    # Phase 2: Try symmetric binary constructions at various n
    n_sym_tries = [320, 384, 400, 448, 480, 512, 560, 640]
    for n_s in n_sym_tries:
        if remaining() < 40:
            break
        if n_s == best_n:
            continue
        h_sym = symmetric_init(n_s, rng)
        # Quick local search
        tl = min(20, max(8, remaining() // 10))
        h_s, c5_s = binary_local_search(h_sym, n_s, tl, rng)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
            best_n = n_s

    # Phase 3: Interpolate parent to new n and optimize
    n_interp_tries = sorted(set([
        max(120, best_n - 4), best_n + 4, best_n + 8,
        best_n + 16, best_n + 32, best_n + 64
    ]))
    n_interp_tries = [nn for nn in n_interp_tries if 100 <= nn <= 1000 and nn != best_n and nn % 2 == 0]

    for n_t in n_interp_tries:
        if remaining() < 35:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        h_bin_t = (h_feas > 0.5).astype(float)
        # Fix count
        n_ones_t = int(np.sum(h_bin_t == 1))
        if n_ones_t != n_t // 2:
            if n_ones_t > n_t // 2:
                idx = np.where(h_bin_t == 1)[0]
                h_bin_t[np.random.choice(idx, n_ones_t - n_t // 2, replace=False)] = 0
            else:
                idx = np.where(h_bin_t == 0)[0]
                h_bin_t[np.random.choice(idx, n_t // 2 - n_ones_t, replace=False)] = 1
        # Local search
        tl = min(25, max(10, remaining() // 8))
        h_opt_t, c5_t = binary_local_search(h_bin_t, n_t, tl, rng)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_opt_t.copy()
            best_n = n_t

    # Phase 4: SLSQP continuous refinement on best
    n = best_n
    h = best_h.copy()
    for beta in [500, 2000, 8000, 20000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=350)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Pair coordinate descent
    if remaining() > 20:
        tl = min(50, max(15, remaining() - 30))
        h_pc, c5_pc = pair_coord_descent(h, n, max_passes=300, time_limit=tl, rng=rng)
        if c5_pc < best_c5:
            best_c5 = c5_pc
            best_h = h_pc.copy()
            h = h_pc.copy()

    # Phase 6: More intensive binary search on best
    n = best_n
    h_bin = (best_h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin == 1))
    if n_ones != n // 2:
        if n_ones > n // 2:
            idx = np.where(h_bin == 1)[0]
            h_bin[rng.choice(idx, n_ones - n // 2, replace=False)] = 0
        else:
            idx = np.where(h_bin == 0)[0]
            h_bin[rng.choice(idx, n // 2 - n_ones, replace=False)] = 1
    if remaining() > 30:
        tl = min(45, remaining() - 25)
        h_b2, c5_b2 = multi_density_local_search(h_bin, n, tl, rng)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()
            best_n = n

    # Phase 7: Try a few more n values with full pipeline
    extra_ns = sorted(set([best_n - 8, best_n + 8, best_n + 16,
                           max(200, best_n - 16), best_n + 32,
                           max(150, best_n - 32)]))
    extra_ns = [nn for nn in extra_ns if 100 <= nn <= 1200 and nn % 2 == 0 and nn != best_n]
    for n_x in extra_ns[:4]:
        if remaining() < 30:
            break
        h_x = interp_to(best_h, best_n, n_x)
        h_x = project_feasible(h_x, n_x)
        h_xb = (h_x > 0.5).astype(float)
        n_ox = int(np.sum(h_xb == 1))
        if n_ox != n_x // 2:
            if n_ox > n_x // 2:
                idx = np.where(h_xb == 1)[0]
                h_xb[rng.choice(idx, n_ox - n_x // 2, replace=False)] = 0
            else:
                idx = np.where(h_xb == 0)[0]
                h_xb[rng.choice(idx, n_x // 2 - n_ox, replace=False)] = 1
        # Quick SLSQP
        if remaining() > 20:
            h_xc = slsqp_smooth(h_x, n_x, 3000, maxiter=200)
        else:
            h_xc = h_x.copy()
        # Binary refinement
        h_xb2 = (h_xc > 0.5).astype(float)
        n_ox2 = int(np.sum(h_xb2 == 1))
        if n_ox2 != n_x // 2:
            if n_ox2 > n_x // 2:
                idx = np.where(h_xb2 == 1)[0]
                h_xb2[rng.choice(idx, n_ox2 - n_x // 2, replace=False)] = 0
            else:
                idx = np.where(h_xb2 == 0)[0]
                h_xb2[rng.choice(idx, n_x // 2 - n_ox2, replace=False)] = 1
        tl = min(15, max(5, remaining() // 6))
        h_xf, c5_xf = binary_local_search(h_xb2, n_x, tl, rng)
        c5_xc = compute_c5(h_xc)
        if c5_xf < best_c5:
            best_c5 = c5_xf
            best_h = h_xf.copy()
            best_n = n_x
        elif c5_xc < best_c5:
            best_c5 = c5_xc
            best_h = h_xc.copy()
            best_n = n_x

    # Phase 8: Final polish with SLSQP + coord descent
    n = best_n
    h = best_h.copy()
    for beta in [5000, 15000, 40000]:
        if remaining() < 5:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining() > 12:
        tl = min(30, max(8, remaining() - 12))
        h_pc, c5_pc = pair_coord_descent(h, n, max_passes=200, time_limit=tl, rng=rng)
        if c5_pc < best_c5:
            best_c5 = c5_pc
            best_h = h_pc.copy()

    # Final binary local search
    if remaining() > 15:
        h_bin_f = (best_h > 0.5).astype(float)
        n_of = int(np.sum(h_bin_f == 1))
        if n_of != best_n // 2:
            if n_of > best_n // 2:
                idx = np.where(h_bin_f == 1)[0]
                h_bin_f[rng.choice(idx, n_of - best_n // 2, replace=False)] = 0
            else:
                idx = np.where(h_bin_f == 0)[0]
                h_bin_f[rng.choice(idx, best_n // 2 - n_of, replace=False)] = 1
        tl = min(20, max(5, remaining() - 10))
        h_bf, c5_bf = binary_local_search(h_bin_f, best_n, tl, rng)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

    # Validate and final check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
