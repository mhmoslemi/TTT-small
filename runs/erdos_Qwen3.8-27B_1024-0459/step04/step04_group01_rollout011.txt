def swap_delta(h, i, j, n):
    """Compute change in corr vector when h[i]:1->0, h[j]:0->1."""
    delta = np.zeros(2*n-1)
    
    # h[i] goes from 1 to 0
    # As source (h[a] where a=i):
    #   d ≥ 0, d ≤ n-1-i: corr[d] -= (1-h[i+d])
    #   d < 0, |d| ≤ i: corr[d] -= (1-h[i+d]) = (1-h[i-|d|])
    for d in range(0, n-i):  # d = 0 to n-1-i
        idx = d + (n-1)  # full correlate index
        delta[idx] -= (1.0 - h[i+d])
    for d in range(1, i+1):  # d = -1 to -i
        idx = -d + (n-1)
        delta[idx] -= (1.0 - h[i-d])
    
    # As target ((1-h[a+d]) where a+d=i, a=i-d):
    #   d ≥ 0, d ≤ i: corr[d] += h[i-d] (1-h[i] goes 0→1)
    for d in range(1, i+1):  # d = 1 to i (d=0 would be h[i]*1-h[i]=0, skipped)
        idx = d + (n-1)
        delta[idx] += h[i-d]
    #   d < 0, |d| ≤ n-1-i: corr[d] += h[i+|d|]
    for d in range(1, n-i):  # d = -1 to -(n-1-i)
        idx = -d + (n-1)
        delta[idx] += h[i+d]
    
    # h[j] goes from 0 to 1
    # As source:
    for d in range(0, n-j):  # d = 0 to n-1-j
        idx = d + (n-1)
        delta[idx] += (1.0 - h[j+d])

        delta[idx] -= h[j-d]
    for d in range(1, n-j):  # d = -1 to -(n-1-j)
        idx = -d + (n-1)
        delta[idx] -= h[j+d]
    
    return delta

import numpy as np
import time

_START = 0.0
_BUDGET = 1300.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    corr = compute_corr(h)
    return float(np.max(corr) * 2.0 / n)

def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_trial = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_trial)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def round_to_binary(h, n, rng=None):

        rng = np.random.RandomState(0)
    n_ones_target = n // 2
    # Rank by h value, threshold to get exactly n_ones_target ones
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:n_ones_target]] = 1.0
    return h_bin

def full_greedy_swap(h, n, time_limit=120, rng=None, stop_on_stall=5):
    """Exhaustive single-swap greedy: try ALL (i,j) pairs each pass."""
    h = h.copy()

    cur_max = int(np.max(corr)) if np.allclose(corr, np.round(corr)) else float(np.max(corr))

    best_max = cur_max

def simulated_annealing(h, n, time_limit=120, rng=None, T_start=2.0, T_end=0.05):

    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()

    dx = 2.0 / n
    best_h = h.copy()
    best_max = cur_max
    best_c5 = best_max * dx
    start = time.time()
    
    total_time = max(time_limit, 1.0)
    
    while remaining() > 5 and time.time() - start < time_limit:
        progress = min(1.0, (time.time() - start) / total_time)
        T = T_start * (T_end / T_start) ** progress
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

            h[j] = 1.0

            h[i2] = 0.0
            h[j1] = 1.0

                    best_c5 = best_max * dx

            else:
                # Just do a random single swap as fallback
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                new_max = float(np.max(compute_corr(h)))
                delta = new_max - cur_max
                if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
                    cur_max = new_max
                    if new_max < best_max:
                        best_max = new_max
                        best_h = h.copy()
                        best_c5 = best_max * dx
                else:
                    h[i] = 1.0
                    h[j] = 0.0
    
    return best_h, best_c5

def multi_start_optimize(n, time_limit_total=300, rng=None):
    """Try multiple starting points and return the best."""
    if rng is None:
        rng = np.random.RandomState(0)
    
    best_h = None
    best_c5 = float('inf')
    
    starting_points = []
    
    # Starting point 1: Alternating
    h_alt = np.zeros(n)
    h_alt[::2] = 1.0
    if int(np.sum(h_alt)) != n // 2:
        h_alt[1::2] = 1.0
    starting_points.append(h_alt.copy())
    
    # Starting point 2: Alternating + random perturbation
    h_pert = h_alt.copy()
    for _ in range(n // 6):
        i = rng.randint(n)
        j = rng.randint(n)
        if h_pert[i] != h_pert[j]:
            h_pert[i], h_pert[j] = h_pert[j], h_pert[i]
    starting_points.append(h_pert.copy())
    
    # Starting point 3: Random
    h_rand = np.zeros(n)
    idx = rng.permutation(n)
    h_rand[idx[:n//2]] = 1.0
    starting_points.append(h_rand.copy())
    
    # Starting point 4: Structured (blocks)
    h_block = np.zeros(n)
    block_size = max(2, n // 78)
    for b in range(0, n, block_size * 2):
        h_block[b:b+block_size] = 1.0
    # Fix count
    n_ones = int(np.sum(h_block))
    if n_ones > n // 2:
        excess = n_ones - n // 2
        ones_idx = np.where(h_block == 1)[0]
        h_block[ones_idx[:excess]] = 0.0
    elif n_ones < n // 2:
        deficit = n // 2 - n_ones
        zeros_idx = np.where(h_block == 0)[0]
        h_block[zeros_idx[:deficit]] = 1.0
    starting_points.append(h_block.copy())
    
    time_per_start = time_limit_total / len(starting_points)
    
    for si, h0 in enumerate(starting_points):
        if remaining() < 15:
            break
        t_left = min(time_per_start, remaining() - 10)
        
        # Greedy single swap
        h_g, c5_g, _ = full_greedy_swap(h0, n, time_limit=t_left * 0.4, rng=rng, stop_on_stall=3)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
        
        # Simulated annealing
        if remaining() > 10:
            t_sa = min(t_left * 0.5, remaining() - 10)
            h_sa, c5_sa = simulated_annealing(h_g, n, time_limit=t_sa, rng=rng, T_start=1.5, T_end=0.02)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
            h_g = h_sa.copy()
        
        # More greedy after SA
        if remaining() > 8:
            t_g2 = min(t_left * 0.2, remaining() - 8)
            h_g2, c5_g2, _ = full_greedy_swap(h_g, n, time_limit=t_g2, rng=rng, stop_on_stall=3)
            if c5_g2 < best_c5:
                best_c5 = c5_g2
                best_h = h_g2.copy()
    
    return best_h, best_c5

def slsqp_round_and_optimize(h_cont, n, time_limit=60, rng=None):
    """Use continuous solution to guide binary construction, then optimize."""
    if rng is None:
        rng = np.random.RandomState(42)
    h_bin = round_to_binary(h_cont, n, rng)
    h_g, c5_g, _ = full_greedy_swap(h_bin, n, time_limit=time_limit * 0.5, rng=rng, stop_on_stall=3)
    if remaining() > 10:
        h_sa, c5_sa = simulated_annealing(h_g, n, time_limit=min(time_limit * 0.4, remaining() - 10), rng=rng)
        if c5_sa < c5_g:
            h_g, c5_g = h_sa, c5_sa
    return h_g, c5_g

def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Try to load parent
    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        pass
    
    n_parent = parent_n if parent_h is not None else 312
    
    # Phase 1: Binary optimization at parent's n (or 312)
    n1 = n_parent if n_parent >= 100 else 312
    if n1 % 2 != 0:
        n1 = n1 + 1
    
    h_init = None
    if parent_h is not None:
        h_init = parent_h.copy()
    else:
        h_init = np.zeros(n1)
        h_init[::2] = 1.0
    
    # Round parent to binary and optimize
    h_bin = round_to_binary(h_init, n1, rng)
    
    # Full greedy
    t1 = min(200, remaining() - 50)
    h1, c5_1, _ = full_greedy_swap(h_bin, n1, time_limit=t1, rng=rng, stop_on_stall=5)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = n1
    
    # Simulated annealing
    if remaining() > 30:
        t2 = min(150, remaining() - 20)
        h2, c5_2 = simulated_annealing(h1, n1, time_limit=t2, rng=rng, T_start=2.0, T_end=0.03)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
    
    # More greedy
    if remaining() > 20:
        t3 = min(120, remaining() - 15)
        h3, c5_3, _ = full_greedy_swap(h2 if c5_2 < c5_1 else h1, n1, time_limit=t3, rng=rng, stop_on_stall=5)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
    
    # Phase 2: Try other n values
    n_candidates = []
    for delta in [-8, -4, 0, 4, 8, 12, 16, 24, 32, 48]:
        nn = n1 + delta
        if 200 <= nn <= 500 and nn % 2 == 0:
            n_candidates.append(nn)
    n_candidates = sorted(set(n_candidates))
    n_candidates.remove(n1) if n1 in n_candidates else None
    
    for n_try in n_candidates:
        if remaining() < 40:
            break
        # Interpolate best solution to new n
        if parent_h is not None:
            xp = (np.arange(n_parent) + 0.5) * 2.0 / n_parent
            x_new = (np.arange(n_try) + 0.5) * 2.0 / n_try
            h_interp = np.interp(x_new, xp, parent_h)
        else:
            h_interp = np.zeros(n_try)
            h_interp[::2] = 1.0
        
        h_bin_try = round_to_binary(h_interp, n_try, rng)
        
        # Quick greedy
        t_q = min(60, remaining() - 20)
        h_q, c5_q, _ = full_greedy_swap(h_bin_try, n_try, time_limit=t_q, rng=rng, stop_on_stall=3)
        if c5_q < best_c5:
            best_c5 = c5_q
            best_h = h_q.copy()
            best_n = n_try
        
        # SA if still time and looking promising
        if remaining() > 30 and c5_q < best_c5 + 0.005:
            t_sa = min(60, remaining() - 20)
            h_sa, c5_sa = simulated_annealing(h_q, n_try, time_limit=t_sa, rng=rng, T_start=1.0, T_end=0.02)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_try
            h_q = h_sa.copy()
            
            if remaining() > 15:
                h_q2, c5_q2, _ = full_greedy_swap(h_q, n_try, time_limit=min(30, remaining()-15), rng=rng, stop_on_stall=3)
                if c5_q2 < best_c5:
                    best_c5 = c5_q2
                    best_h = h_q2.copy()
                    best_n = n_try
    
    # Phase 3: Intensive multi-start at best_n
    if remaining() > 60:
        n_best = best_n
        h_best_for_multistart = best_h.copy()
        
        # Try a few perturbed versions of the best solution
        for pert_level in [2, 4, 8, 16]:
            if remaining() < 20:
                break
            h_pert = h_best_for_multistart.copy()
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            for _ in range(pert_level):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i], h_pert[j] = h_pert[j], h_pert[i]
            
            t_pert = min(80, remaining() - 15)
            h_p, c5_p, _ = full_greedy_swap(h_pert, n_best, time_limit=t_pert * 0.4, rng=rng, stop_on_stall=3)
            if remaining() > 15:
                h_p_sa, c5_p_sa = simulated_annealing(h_p, n_best, time_limit=min(40, remaining()-15), rng=rng)
                if c5_p_sa < c5_p:
                    h_p, c5_p = h_p_sa, c5_p_sa
            if remaining() > 10:
                h_p2, c5_p2, _ = full_greedy_swap(h_p, n_best, time_limit=min(30, remaining()-10), rng=rng, stop_on_stall=3)
                if c5_p2 < c5_p:
                    h_p, c5_p = h_p2, c5_p2
            
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()
    
    # Phase 4: Final polish with exhaustive greedy
    if remaining() > 15:
        t_final = min(120, remaining() - 5)
        h_f, c5_f, _ = full_greedy_swap(best_h, best_n, time_limit=t_final, rng=rng, stop_on_stall=8)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    
    # Validate and return
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
