    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g

import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1000.0
_TOPK = 30
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def corr_grad_single(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g


def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    topk = _TOPK
    idx = np.argsort(-corr)[:topk]
    shifted = corr[idx] - m
    beta = _BETA
    cl = np.clip(beta * shifted, -50.0, 50.0)
    ex = np.exp(cl)
    s = np.sum(ex)
    return (m + np.log(s) / beta) * dx_val


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    topk = _TOPK
    idx = np.argsort(-corr)[:topk]
    shifted = corr[idx] - m
    beta = _BETA
    cl = np.clip(beta * shifted, -50.0, 50.0)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for kv, wv in zip(idx, w):
        if wv > 1e-10:
            g += wv * corr_grad_single(x, n, int(kv))
    return g * dx_val


def lbfgs_refine(h, n, beta, topk, maxiter=200, penalty_w=500.0):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    _TOPK = topk

    def obj_p(x):
        base = smooth_obj(x)
        penalty = penalty_w * (np.sum(x) - n / 2.0) ** 2
        return base + penalty

    def grad_p(x):
        g = smooth_grad(x)
        g += 2.0 * penalty_w * (np.sum(x) - n / 2.0)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_p, h, jac=grad_p, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)


def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1
    return h_bin

def topk_subgradient(h, n, time_limit=60, top_k=5):
    """Subgradient method focusing on top-k worst shifts"""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx_val = 2.0 / n
    start = time.time()
    lr = 0.02

    for t in range(5000):
        if time.time() - start > time_limit or remaining() < 5:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k_val in top_idx:
            g += corr_grad_single(h, n, int(k_val))
        g /= top_k
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h_new = project_feasible(h - lr * g / gn, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        lr = max(lr * 0.998, 0.001)

    return best_h, best_c5
