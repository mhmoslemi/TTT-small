```python
def compute_swap_delta(h, n, i, j):
    """Compute change in corr when h[i]: 1->0, h[j]: 0->1"""
    # i goes from 1 to 0
    delta = np.zeros(2*n-1)  # indexed 0..2n-2, offset = idx - (n-1)
    off = np.arange(2*n-1) - (n) + 1  # offsets from -(n-1) to n-1
    
    # i as first: valid when 0 <= i+off < n
    mask1 = (i + off >= 0) & (i + off < n)
    delta[mask1] -= (1 - h[i + off[mask1]])
    # i as second: valid when 0 <= i-off < n  
    mask2 = (i - off >= 0) & (i - off < n)
    delta[mask2] += h[i - off[mask2]]
    
    # j goes from 0 to 1
    mask3 = (j + off >= 0) & (j + off < n)
    delta[mask3] += (1 - h[j + off[mask3]])
    # j as second: valid when 0 <= j-off < n
    mask4 = (j - off >= 0) & (j - off < n)
    delta[mask4] -= h[j - off[mask4]]
    
    return delta
```

```python
            h_rev[n-1-j] = 1.0
            corr += delta
            cur_max = new_max
            
            # Update tracking arrays
            pos_i = np.where(ones == i)[0]
            if len(pos_i) > 0:
                ones[pos_i[0]] = j
                pos_j = np.where(zeros == j)[0]
                if len(pos_j) > 0:
                    zeros[pos_j[0]] = i
            
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        # else: reject
        
        T *= alpha
        it += 1
        if stall > 3000:
            T = max(T, 5.0)
            stall = 0
            if remaining() < 10:
                break
    
    return best_h, best_max * (2.0/n)
```
