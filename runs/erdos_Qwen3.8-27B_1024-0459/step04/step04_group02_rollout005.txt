    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def coord_descent(h, n, max_sweeps=50, rng=None):
    """Cyclic coordinate descent with exact (ternary search) line search."""
    best = h.copy()
    best_c5 = compute_c5(best)
    dx = 2.0 / n
    nc = 2 * n - 1
    m_offsets = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        order = np.random.permutation(n) if rng is not None else np.arange(n)
        
        for i in order:
            if remaining_time() < 1:
                break
            
            old_t = best[i]
            corr = compute_corr(best)
            
            # Compute contribution of h[i] to each corr entry
            contrib = np.zeros(nc)
            # h[i] as first: pairs (i, i+m) for valid m
            valid_f = (i + m_offsets >= 0) & (i + m_offsets < n)
            if np.any(valid_f):
                idx_f = i + m_offsets[valid_f]
                contrib[valid_f] += best[i] * (1.0 - best[idx_f])
            # h[i] as second: pairs (i-m, i) for valid m≠0
            valid_s = (m_offsets != 0) & (i - m_offsets >= 0) & (i - m_offsets < n)
            if np.any(valid_s):
                idx_s = i - m_offsets[valid_s]
                contrib[valid_s] += best[idx_s] * (1.0 - best[i])
            
            base = corr - contrib
            
            # New contribution with h[i] = t:
            ncl = np.zeros(nc)  # linear coeff
            ncs = np.zeros(nc)  # constant
            ncu = np.zeros(nc)  # quadratic coeff (only for m=0)
            
            # m=0: t*(1-t) = t - t^2
            ncl[n-1] += 1.0
            ncu[n-1] += -1.0
            
            # m≠0, pair (i, i+m): t*(1-h[i+m])
            vf = (m_offsets != 0) & (i + m_offsets >= 0) & (i + m_offsets < n)
            if np.any(vf):
                ncl[vf] += (1.0 - best[i + m_offsets[vf]])
            
            # m≠0, pair (i-m, i): h[i-m]*(1-t)
            vs = (m_offsets != 0) & (i - m_offsets >= 0) & (i - m_offsets < n)
            if np.any(vs):
                ncs[vs] += best[i - m_offsets[vs]]
                ncl[vs] += (-best[i - m_offsets[vs]])
            
            # f(t) = max(base + ncs + ncl*t + ncu*t^2) * dx
            # Ternary search on [0,1]
            lo, hi = 0.0, 1.0
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                v1 = np.max(base + ncs + ncl * m1 + ncu * m1 * m1)
                v2 = np.max(base + ncs + ncl * m2 + ncu * m2 * m2)
                if v1 < v2:
                    hi = m2
