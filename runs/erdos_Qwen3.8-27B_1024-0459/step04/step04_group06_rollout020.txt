_g_target = 0.0

_g_beta = 100.0
_g_start_time = 0.0
_g_budget = 1300.0

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

def interp_to(src_h, src_n, dst_n):
