```python
def compute_delta_corr(h, n, i, j, corr):
    """Compute the change in correlation when h[i]: 1->0, h[j]: 0->1.
    h is the OLD array (h[i]=1, h[j]=0).
    Returns the delta array (same shape as corr)."""
    delta = np.zeros(2*n - 1)
    # corr[k] corresponds to shift s = k - (n-1)
    # For shift s, valid m range:
    #   s >= 0: m in [0, n-1-s], pair (m, m+s)
    #   s < 0: p=-s, m in [p, n-1], pair (m, m-p)
    
    for s in range(-(n-1), n):
        k = s + (n-1)  # index in corr
        # Determine valid m range
        if s >= 0:
            m_min, m_max = 0, n-1-s
        else:
            p = -s
            m_min, m_max = p, n-1
        
        total_delta = 0
        for m in [i, j, i-s, j-s]:
            if m_min <= m <= m_max:
                # Check that m+s is also valid
                ms = m + s
                if 0 <= ms < n:
                    old_term = h[m] * (1 - h[ms])
                    # New values
                    h_m = h[m]
                    if m == i: h_m = 0.0
                    elif m == j: h_m = 1.0
                    h_ms = h[ms]
                    if ms == i: h_ms = 0.0
                    elif ms == j: h_ms = 1.0
                    new_term = h_m * (1 - h_ms)
                    total_delta += new_term - old_term
        delta[k] = total_delta
    return delta
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)

def remaining():

def interp_to(src_h, src_n, dst_n):

        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1

    """Fast SA for binary sequences using full correlation recompute."""
    rng = np.random.RandomState(seed)

    target = n // 2

        if remaining() < 5:
            break

        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

                stall = 0
            else:

                zeros = np.where(h == 0)[0]
            corr = compute_corr(h)
            cur_max = float(np.max(corr))
            T = max(T, 2.0)
            stall = 0

def slsqp_smooth(h, n, beta, maxiter=300):
    """SLSQP with smooth max approximation."""
    global _N_GLOBAL, _BETA_GLOBAL, _TGT_GLOBAL
    _N_GLOBAL = n

    _BETA_GLOBAL = beta

        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -60, 60)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_val

        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -60, 60)
        ex = np.exp(cl)
        w = ex / np.sum(ex)

        # corr[k] = sum_m h[m] * (1-h[m+s]) where s = k-(n-1)
        # d(corr[k])/d(h[a]) = (1-h[a+s]) if a is first arg, minus h[a-s] if a is second arg (with valid indices)

        g = np.zeros(n)
        for a in range(n):

            fp = obj(x)
            x[a] -= 2*eps
            fm = obj(x)
            x[a] += eps
            g[a] = (fp - fm) / (2*eps)
        return g
    
    def con_fun(x):
        return np.sum(x) - n/2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:

                       bounds=bounds, constraints=cons,
```

I need the analytical gradient. Let me derive it.

For the smooth objective: f(x) = (m + log(sum(exp(beta*(corr-m))))/beta) * dx

where m = max(corr), corr[k] = sum_{valid m} x[m] * (1-x[m+s]).

The gradient of corr[k] with respect to x[a]:
- corr[k] uses shift s = k - (n-1)
- x[a] appears as x[m] when m=a (if valid): contributes (1-x[a+s])
- x[a] appears as x[m+s] when m+s=a, i.e., m=a-s (if valid): contributes -x[a-s] (derivative of (1-x) w.r.t. x)

Wait, more carefully:
corr[k] = sum_{m valid} x[m] * (1-x[m+s])

d(corr[k])/d(x[a]) = sum_{m valid} [δ_{m,a} * (1-x[m+s]) + x[m] * (-δ_{m+s,a})]
= (1-x[a+s]) * [a is valid as m] - x[a-s] * [a-s is valid as m, i.e., m=a-s is in range]

Where "a is valid as m" means a is in the valid range for the sum.
And "a-s is valid as m" means a-s is in the valid range.

For s ≥ 0: valid m range is [0, n-1-s]
For s < 0 (p=-s): valid m range is [p, n-1]

So:
- "a is valid as m": if s≥0: 0≤a≤n-1-s. If s<0: -s≤a≤n-1.
- "a-s is valid as m": if s≥0: 0≤a-s≤n-1-s. If s<0: -s≤a-s≤n-1.

This is what the previous program had in `corr_grad_per_shift`. Let me use that approach.

```python
    n = _N_GLOBAL
    dx_val = 2.0 / n
    beta = _BETA_GLOBAL
    corr = compute_corr(x)
    m = np.max(corr)
    scaled = beta * (corr - m)
    scaled = np.clip(scaled, -60, 60)
    exp_s = np.exp(scaled)
    w = exp_s / np.sum(exp_s)
    
    dfdx = np.zeros(n)
    for k in range(2*n - 1):
        if w[k] < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            # d(corr)/dx[a] = (1-x[a-s]) for a in [s, n-1] minus x[a+s] for a in [0, n-1-s]
            npairs = n - s
            dfdx[s:s+npairs] += w[k] * (1.0 - x[:npairs])
            dfdx[:npairs] -= w[k] * x[s:s+npairs]

            dfdx[:npairs] += w[k] * (1.0 - x[p:p+npairs])
            dfdx[p:p+npairs] -= w[k] * x[:npairs]
    
    return dfdx * dx_val
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0
_N_GLOBAL = 0
_BETA_GLOBAL = 1.0
_TGT_GLOBAL = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1
    return h_bin


def sa_fast(h_init, n, time_limit=60, seed=42, T_init=4.0, T_floor=0.02, alpha=0.9997):

    h = h_init.copy()
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]

    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1

    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()

    T = T_init
    start = time.time()
    stall = 0
    kick_count = 0

        if remaining() < 5:
            break

        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        h[i] = 0.0
        h[j] = 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))

                stall = 0

def smooth_objective(x):
    n = _N_GLOBAL
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA_GLOBAL
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -60, 60)
    ex = np.exp(shifted)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def smooth_gradient(x):
    n = _N_GLOBAL
    dx_val = 2.0 / n
    beta = _BETA_GLOBAL
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -60, 60)
    ex = np.exp(shifted)
    w = ex / np.sum(ex)

    dfdx = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            npairs = n - s
            dfdx[s:s + npairs] += w[k] * (1.0 - x[:npairs])
            dfdx[:npairs] -= w[k] * x[s:s + npairs]
        else:
            p = -s
            npairs = n - p
            dfdx[:npairs] += w[k] * (1.0 - x[p:p + npairs])
            dfdx[p:p + npairs] -= w[k] * x[:npairs]

    return dfdx * dx_val


def sum_con_fun(x):
    return np.sum(x) - _TGT_GLOBAL


def sum_con_jac(x):
    return np.ones(_N_GLOBAL)


def slsqp_refine(h, n, beta, maxiter=400):
    global _N_GLOBAL, _BETA_GLOBAL, _TGT_GLOBAL
    _N_GLOBAL = n
    _TGT_GLOBAL = n / 2.0
    _BETA_GLOBAL = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_objective, h, jac=smooth_gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def topk_gradient_descent(h, n, max_iters=300, top_k=5):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        m = float(np.max(corr))
        if m * (2.0 / n) >= best_c5 + 1e-8:
            break
        top_idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k in top_idx:
            s = int(k) - (n - 1)
            if s >= 0:
                npairs = n - s
                g[s:s + npairs] += (1.0 - h[:npairs])
                g[:npairs] -= h[s:s + npairs]
            else:
                p = -s
                npairs = n - p
                g[:npairs] += (1.0 - h[p:p + npairs])
                g[p:p + npairs] -= h[:npairs]
        g /= len(top_idx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        improved = False
        for step in [0.05, 0.02, 0.01, 0.1, 0.005, 0.03, 0.08, 0.003]:
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-13:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            break
    return h, best_c5
```
