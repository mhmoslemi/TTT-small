def smooth_obj(x, n, beta):
    dx_val = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val

def corr_grad(h, n, k):
    g = np.zeros(n)
    shift = k - (n - 1)
    if shift >= 0:
        npairs = n - shift
        g[shift:shift + npairs] += (1.0 - h[:npairs])
        g[:npairs] -= h[shift:shift + npairs]
    else:
        ao = -shift
        npairs = n - ao
        g[:npairs] += (1.0 - h[ao:ao + npairs])
        g[ao:ao + npairs] -= h[:npairs]
    return g

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)

            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float).copy()

    rng = np.random.RandomState(seed)
    h = h_init.copy()
    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones > target:
        idx = np.where(h == 1)[0]
        to_remove = n_ones - target
        perm = rng.permutation(len(idx))[:to_remove]
        h[idx[perm]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        to_add = target - n_ones
        perm = rng.permutation(len(idx))[:to_add]

    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    h_rev = np.zeros(n)
    h_rev[:] = h[::-1]
    one_minus_h_rev = 1.0 - h_rev

    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()

    dx_val = 2.0 / n
    alpha = 0.99999
    T = T0
    start = time.time()
    stall = 0
    max_stall = 8000
    it_step = 0

    # Precompute offset vectors for the incremental update
    # For position i (1->0, delta=-1):
    #   corr[i:i+n] += (h_rev - 1)  [which is -(1-h_rev)]
    #   corr[n-1-i:n-1-i+n] += h
    # For position j (0->1, delta=+1):
    #   corr[j:j+n] += (1 - h_rev)  [which is (1-h_rev)]
    #   corr[n-1-j:n-1-j+n] += -h

        if remaining() < 10:
            break

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])

        # Compute delta_c using incremental formula
        # h[i] goes 1->0 (delta=-1), h[j] goes 0->1 (delta=+1)
        # For i (delta=-1):
        #   corr[i:i+n] += delta * (1 - h_rev) = -1 * (1 - h_rev) = h_rev - 1
        #   corr[n-1-i:n-1-i+n] += -delta * h = +1 * h = h
        # For j (delta=+1):
        #   corr[j:j+n] += delta * (1 - h_rev) = +1 * (1 - h_rev)
        #   corr[n-1-j:n-1-j+n] += -delta * h = -1 * h = -h
        # At m=n-1: subtract delta_i^2 + delta_j^2 = 1 + 1 = 2

        # Apply changes
        corr[i:i+n] += (h_rev - 1.0)
        corr[n-1-i:n-1-i+n] += h
        corr[j:j+n] += one_minus_h_rev
        corr[n-1-j:n-1-j+n] -= h
        corr[n-1] -= 2.0

        # Update h
        h[i] = 0.0
        h[j] = 1.0
        h_rev[i] = 0.0  # h_rev[i] = h[n-1-i], wait no. h_rev[k] = h[n-1-k]
        # Actually h_rev = h[::-1], so h_rev[k] = h[n-1-k]
        # When h[i] changes, h_rev[n-1-i] changes
        h_rev[n-1-i] = 0.0
        one_minus_h_rev[n-1-i] = 1.0
        h_rev[n-1-j] = 1.0
        one_minus_h_rev[n-1-j] = 0.0

        new_max = float(np.max(corr))
        delta_max = new_max - cur_max

            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            # Revert
            h[i] = 1.0
            h[j] = 0.0
            h_rev[n-1-i] = 1.0
            one_minus_h_rev[n-1-i] = 0.0
            h_rev[n-1-j] = 0.0
            one_minus_h_rev[n-1-j] = 1.0
            # Revert corr
            corr[i:i+n] -= (h_rev - 1.0)
            corr[n-1-i:n-1-i+n] -= h
            corr[j:j+n] -= one_minus_h_rev
            corr[n-1-j:n-1-j+n] += h
            corr[n-1] += 2.0
            stall += 1

        if T < T1:
            T = T1
        if stall > max_stall:

            if np.max(np.abs(corr - corr_check)) > 0.01:
                # Rebuild from scratch
                corr = corr_check
                h_rev[:] = h[::-1]
                one_minus_h_rev[:] = 1.0 - h_rev
                cur_max = float(np.max(corr))

def _smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def _corr_grad_single(h, n, k):
    g = np.zeros(n)
    shift = k - (n - 1)
    if shift >= 0:
        npairs = n - shift
        g[shift:shift+npairs] += (1.0 - h[:npairs])
        g[:npairs] -= h[shift:shift+npairs]
    else:
        ao = -shift
        npairs = n - ao
        g[:npairs] += (1.0 - h[ao:ao+npairs])
        g[ao:ao+npairs] -= h[:npairs]
    return g


def _smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-10:
            g += w[k] * _corr_grad_single(x, n, k)
    return g * dx_val

        h_bin_t = make_binary(best_h, n_t, rng)
        t_alloc = min(remaining() * 0.25, 120)
        for r in range(3):
            if remaining() < 15:
                break
            h_sa, c5_sa = sa_binary_fast(h_bin_t, n_t, time_limit=t_alloc / 3,
                                         seed=seed + 5000 + r * 31, T0=0.8)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_t

    # Phase 6: Local search
    if remaining() > 20:
        h_ls, c5_ls = local_search_binary(best_h, best_n, time_limit=min(40, remaining() * 0.4),
                                          seed=seed + 9000)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 7: Continuous SLSQP polish
    if remaining() > 15:
        n = best_n
        for beta in [1000, 5000]:
            if remaining() < 8:
                break
            h_new = slsqp_opt(best_h, n, beta, maxiter=150)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n

    # Phase 8: Final local search
    if remaining() > 15:
        h_bin_f = make_binary(best_h, best_n, rng)
        h_ls, c5_ls = local_search_binary(h_bin_f, best_n, time_limit=min(30, remaining() * 0.5),
                                          seed=seed + 15000)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
