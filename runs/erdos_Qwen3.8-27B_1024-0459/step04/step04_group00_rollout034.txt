import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def smooth_obj_and_grad(h, n, beta):
    dx = 2.0 / n
    corr = compute_corr(h)
    cmax = np.max(corr)
    shifted = np.clip(beta * (corr - cmax), -50.0, 50.0)
    exps = np.exp(shifted)
    s = np.sum(exps)
    obj = (cmax + np.log(s) / beta) * dx
    w = exps / s
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            npairs = n - off

        else:
            ao = -off
            npairs = n - ao
            g[:npairs] += wk * (1.0 - h[ao:ao + npairs])
            g[ao:ao + npairs] -= wk * h[:npairs]
    return obj, g


def _sum_constraint(h):
    return np.sum(h) - len(h) / 2.0


def _sum_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h, n, beta, maxiter=400):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_constraint, 'jac': _sum_jac}]
    obj0, _ = smooth_obj_and_grad(h, n, beta)
    try:
        res = minimize(smooth_obj_and_grad, h, args=(n, beta), jac=True,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < obj0 - 1e-12:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def pair_local_search(h, n, max_sweeps=50):
    """2-opt style: try swapping small amounts between pairs of indices."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    corr = compute_corr(h)

    for sweep in range(max_sweeps):
        if remaining() < 5:
            break
        n_improved = 0
        for j in range(n):
            if remaining() < 3:
                break
            v_j = h[j]
            if v_j < 1e-6 or v_j > 1.0 - 1e-6:
                continue
            # Compute gradient contribution of h[j] to each corr
            # dc[k]/dh[j] for all k
            gc = np.zeros(2 * n - 1)
            # h[j] as first factor (h[i+k] where i+k=j, so i=j-k)
            k_min = max(0, j - (n - 1))  # not needed, let's do directly
            for k in range(2 * n - 1):
                off = k - (n - 1)
                if off >= 0:
                    npairs = n - off
                    if k - (n - 1) <= j <= k - (n - 1) + npairs - 1:
                        i = j - off
                        gc[k] += (1.0 - h[i])
                    if 0 <= j < npairs:
                        gc[k] -= h[j + off]
                else:
                    ao = -off
                    npairs = n - ao
                    if 0 <= j < npairs:
                        gc[k] += (1.0 - h[j + ao])
                    if ao <= j <= ao + npairs - 1:
                        gc[k] -= h[j - ao]

            # Find which direction (increase or decrease h[j]) reduces max corr
            # If we increase h[j] by delta: new_corr = corr + delta * gc
            # If we decrease h[j] by delta: new_corr = corr - delta * gc
            # We want to minimize max(new_corr)

            # Try both directions
            for direction in [1, -1]:
                # delta direction: h[j] changes by direction * delta
                # corr changes by direction * delta * gc
                # Find optimal delta: minimize max(corr + direction*delta*gc)
                # subject to 0 <= v_j + direction*delta <= 1
                # and sum constraint: need another variable to compensate

                # For simplicity, just try a few delta values and pair with mean
                for delta in [0.01, 0.05, 0.1, 0.2]:
                    new_delta = direction * delta
                    if v_j + new_delta < 0 or v_j + new_delta > 1:
                        continue
                    h_new = h.copy()
                    h_new[j] = v_j + new_delta
                    # Compensate: distribute -new_delta among other free variables
                    mask = (h_new > 5e-4) & (h_new < 1.0 - 5e-4)
                    mask[j] = False
                    cnt = np.count_nonzero(mask)
                    if cnt == 0:
                        continue
                    h_new[mask] -= new_delta / cnt
                    h_new = np.clip(h_new, 0, 1)
                    h_new = project_feasible(h_new, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5 - 1e-14:
                        h = h_new.copy()
                        best_c5 = c5_new
                        best_h = h.copy()
                        corr = compute_corr(h)
                        n_improved += 1
                        break
        if n_improved == 0 and sweep > 5:
            break
    return best_h, best_c5

    # Phase 2: Top-k equalization at parent resolution
    if remaining() > 20:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=5, max_iters=400, beta_smooth=150)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 3: Try multiple higher resolutions
    resolutions = [512, 640, 768]
    for n_new in resolutions:
        if remaining() < 40:
            break
        h_new = interpolate_project(best_h, best_n, n_new)
        c5_interp = compute_c5(h_new)

        # Progressive SLSQP at this resolution
        betas_phase3 = [200, 500, 1500, 5000, 15000, 40000]
        for beta in betas_phase3:
            if remaining() < 15:
                break
            h_new = slsqp_optimize(h_new, n_new, beta, maxiter=400)
        c5_opt = compute_c5(h_new)

        # Top-k equalization
        if remaining() > 15:
            h_new, c5_opt = top_k_equalize(h_new, n_new, top_k=5, max_iters=300, beta_smooth=200)

        # Final SLSQP polish
        if remaining() > 10:
            h_new = slsqp_optimize(h_new, n_new, 50000, maxiter=400)
            c5_opt = compute_c5(h_new)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_new.copy()
            best_n = n_new

    # Phase 4: Try also n=448 (between 384 and 512)
    if remaining() > 40:
        n_mid = 448
        h_mid = interpolate_project(best_h, best_n, n_mid)
        for beta in [500, 2000, 10000]:
            if remaining() < 15:
                break
            h_mid = slsqp_optimize(h_mid, n_mid, beta, maxiter=300)
        if remaining() > 15:
            h_mid, c5_mid = top_k_equalize(h_mid, n_mid, top_k=4, max_iters=300, beta_smooth=200)
        c5_mid = compute_c5(h_mid)
        if c5_mid < best_c5:
            best_c5 = c5_mid
            best_h = h_mid.copy()
            best_n = n_mid

    # Phase 5: Pair local search at best resolution
    if remaining() > 30:
        h_ps, c5_ps = pair_local_search(best_h, best_n, max_sweeps=30)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps.copy()

    # Phase 6: Noise perturbation + re-optimization (multiple attempts)
    for attempt in range(5):
        if remaining() < 30:
            break
        h_pert = noise_perturb(best_h, best_n, magnitude=0.03 + 0.01 * attempt, rng=rng)
        h_pert = slsqp_optimize(h_pert, best_n, 10000, maxiter=300)
        h_pert, c5_pert = top_k_equalize(h_pert, best_n, top_k=4, max_iters=200, beta_smooth=200)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()

    # Phase 7: Final intensive polish at best resolution
    n_final = best_n
    h_final = best_h.copy()
    for beta in [20000, 50000, 100000]:
        if remaining() < 8:
            break
        h_final = slsqp_optimize(h_final, n_final, beta, maxiter=500)
        c5_f = compute_c5(h_final)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_final.copy()

    # Final top-k equalization
    if remaining() > 10:
        h_eq, c5_eq = top_k_equalize(best_h, best_n, top_k=6, max_iters=500, beta_smooth=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate with evaluate_erdos_solution if available
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
