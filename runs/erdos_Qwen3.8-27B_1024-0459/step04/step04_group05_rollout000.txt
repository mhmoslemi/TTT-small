            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-13:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                no_improve = 0
                break
        if not improved:
            no_improve += 1
            if no_improve > 50:
                for attempt in range(8):
                    noise = rng.randn(n) * 0.03
                    h_trial = project_feasible(h + noise, n)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < c5_best - 1e-13:
                        c5_best = c5_trial
                        h_best = h_trial.copy()
                        h = h_trial.copy()
                        no_improve = 0
                        break
                if no_improve > 100:
                    break
    return h_best, c5_best


def do_coord_descent(h_best, n, budget_fn, rng, max_sweeps=20):
    h = h_best.copy()
    c5_best = compute_c5(h)
    for sweep in range(max_sweeps):
        if budget_fn() < 2.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if budget_fn() < 1.0:
                break
            old_val = h[i]
            for new_val in [0.0, 1.0, 0.5]:
                if abs(new_val - old_val) < 1e-7:
                    continue
                h[i] = new_val
                h = project_feasible(h, n)
                c5_test = compute_c5(h)
                if c5_test < c5_best - 1e-14:
                    c5_best = c5_test
                    h_best = h.copy()
                    any_imp = True
                else:
                    h = h_best.copy()
        if not any_imp and sweep > 5:
            break
    return h_best, c5_best


def round_to_binary(h, n, rng):
    """Round continuous h to binary, maintaining sum = n/2."""
    h_sorted = np.sort(h)[::-1]
    ones = np.zeros(n)
    ones[:n // 2] = 1.0
    # Use top n/2 positions
    top_idx = np.argsort(h)[::-1][:n // 2]
    ones = np.zeros(n)
    ones[top_idx] = 1.0
    return ones


def do_cyclic_sa(h_bin, n, budget_fn, rng, max_rounds=200):
    """Cyclic sweep SA: for each position, try flipping with a random partner."""
    h = h_bin.copy().astype(float)
    ones = sorted(np.where(h > 0.5)[0].tolist())
    zeros = sorted(np.where(h < 0.5)[0].tolist())
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    
    T = 0.008
    decay = 0.995
    for rnd in range(max_rounds):
        if budget_fn() < 2.0:
            break
        for a in ones[:]:
            if budget_fn() < 1.0:
                break
            if a not in ones:
                continue
            for _ in range(3):
                if not zeros:
                    break
                b = zeros[rng.integers(len(zeros))]
                h[a] = 0.0
                h[b] = 1.0
                c5_new = compute_c5(h)
                delta = c5_new - c5_cur
                if delta < 0 or (T > 1e-5 and rng.random() < np.exp(-delta / T)):
                    c5_cur = c5_new
                    ones.remove(a)
                    ones.append(b)
                    ones.sort()
                    zeros.remove(b)
                    zeros.append(a)
                    zeros.sort()
                    if c5_new < c5_best - 1e-14:
                        c5_best = c5_new
                        h_best = h.copy()
                else:
                    h[a] = 1.0
                    h[b] = 0.0
        T *= decay
    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    def budget_fn():
        return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = make_thue_morse(parent_n)
        best_h = project_feasible(parent_h, parent_n)
        best_c5 = compute_c5(best_h)
        best_n = parent_n
        print("No parent. TM: n={}, C5={:.10f}".format(parent_n, best_c5))

    # Phase 1: Quick SLSQP from parent (warm up)
    print("\n--- Phase 1: SLSQP warm-up ---")
    best_h, best_c5 = do_slsqp_phase(best_h, best_n, budget_fn, [50, 300, 1000])
    print("  C5={:.10f}".format(best_c5))

    # Phase 2: Binary local search (SA) at parent resolution
    print("\n--- Phase 2: Binary SA at n={} ---".format(best_n))
    h_bin = round_to_binary(best_h, best_n, rng)
    c5_bin = compute_c5(h_bin)
    if c5_bin > best_c5:
        # Keep continuous as best
        pass
    else:
        best_h = h_bin.copy()
        best_c5 = c5_bin
    
    h_sa, c5_sa = binary_local_search(h_bin, best_n, budget_fn, rng, mode='sa', max_iters=80000)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()
    print("  After binary SA: C5={:.10f}".format(best_c5))

    # Phase 3: Cyclic SA refinement
    if budget_fn() > 30:
        print("\n--- Phase 3: Cyclic SA ---")
        h_bin2 = round_to_binary(best_h, best_n, rng)
        h_cyc, c5_cyc = do_cyclic_sa(h_bin2, best_n, budget_fn, rng, max_rounds=100)
        if c5_cyc < best_c5:
            best_c5 = c5_cyc
            best_h = h_cyc.copy()
        print("  After cyclic SA: C5={:.10f}".format(best_c5))

    # Phase 4: SLSQP from best binary (high beta)
    if budget_fn() > 30:
        print("\n--- Phase 4: High-beta SLSQP ---")
        best_h, best_c5 = do_slsqp_phase(best_h, best_n, budget_fn, [2000, 8000, 30000, 80000])
        print("  C5={:.10f}".format(best_c5))

    # Phase 5: Multi-resolution search
    print("\n--- Phase 5: Multi-resolution ---")
    resolutions_to_try = []
    for nr in [250, 258, 260, 264, 270, 272, 280, 288, 300, 320, 384, 512]:
        if nr != best_n and nr % 2 == 0:
            resolutions_to_try.append(nr)
    
    for n_try in resolutions_to_try:
        if budget_fn() < 20.0:
            break
        h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        h_interp_bin = round_to_binary(h_interp, n_try, rng)
        c5_interp_bin = compute_c5(h_interp_bin)
        
        # Quick binary SA at this resolution
        h_bs, c5_bs = binary_local_search(h_interp_bin, n_try, budget_fn, rng, mode='sa', max_iters=20000)
        
        # SLSQP polish
        h_new, c5_new = do_slsqp_phase(h_bs if c5_bs < c5_interp_bin else h_interp_bin, n_try, budget_fn, [500, 3000, 15000])
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n_try
            print("  n={} improved: C5={:.10f}".format(n_try, best_c5))
        else:
            print("  n={} no improve: C5={:.10f} (best={:.10f})".format(n_try, c5_new, best_c5))

    # Phase 6: Subgradient with momentum
    if budget_fn() > 30:
        print("\n--- Phase 6: Subgradient ---")
        best_h, best_c5 = do_subgrad_phase(best_h, best_n, budget_fn, rng, max_iters=300)
        print("  C5={:.10f}".format(best_c5))

    # Phase 7: More binary search at best resolution
    if budget_fn() > 40:
        print("\n--- Phase 7: More binary search ---")
        h_bin3 = round_to_binary(best_h, best_n, rng)
        h_sa2, c5_sa2 = binary_local_search(h_bin3, best_n, budget_fn, rng, mode='greedy', max_iters=100000)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
        # Also try SA with different seed
        rng2 = np.random.RandomState(seed + 123)
        h_sa3, c5_sa3 = binary_local_search(h_bin3, best_n, budget_fn, rng2, mode='sa', max_iters=50000)
        if c5_sa3 < best_c5:
            best_c5 = c5_sa3
            best_h = h_sa3.copy()
        print("  C5={:.10f}".format(best_c5))

    # Phase 8: Final SLSQP + coordinate descent polish
    if budget_fn() > 20:
        print("\n--- Phase 8: Final polish ---")
        best_h, best_c5 = do_slsqp_phase(best_h, best_n, budget_fn, [50000, 200000])
        if budget_fn() > 10:
            best_h, best_c5 = do_coord_descent(best_h, best_n, budget_fn, rng, max_sweeps=15)
        if budget_fn() > 8:
            best_h, best_c5 = do_subgrad_phase(best_h, best_n, budget_fn, rng, max_iters=100)
        if budget_fn() > 5:
            best_h, best_c5 = do_slsqp_phase(best_h, best_n, budget_fn, [500000])
        print("  C5={:.10f}".format(best_c5))

    # Phase 9: Try a few more diverse starts at the same n
    if budget_fn() > 60:
        print("\n--- Phase 9: Diverse restarts ---")
        for restart_seed in [1, 7, 42, 99]:
            if budget_fn() < 25:
                break
            rng_r = np.random.RandomState(seed + restart_seed)
            h_rand = make_random_bin(rng_r, best_n)
            h_tm = make_thue_morse(best_n) if best_n % 2 == 0 else make_random_bin(rng_r, best_n)
            for h_start in [h_random if (h_rand := h_rand) is not None else h_rand, h_tm]:
                if budget_fn() < 15:
                    break
                h_try, c5_try = binary_local_search(h_start, best_n, budget_fn, rng_r, mode='sa', max_iters=30000)
                h_try, c5_try = do_slsqp_phase(h_try, best_n, budget_fn, [3000, 20000])
                if c5_try < best_c5 - 1e-13:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    print("  Restart improved: C5={:.10f}".format(best_c5))
        print("  C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
