def active_lag_subgrad(h, n, top_k=4):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
    grad = np.zeros(n)
    for k_idx in top_idx:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            n_pairs = n - offset
            if n_pairs <= 0:
                continue
            grad[:n_pairs] += (1.0 - h[offset:offset + n_pairs])
            grad[offset:] -= h[:n_pairs]
        else:
            abs_off = -offset
            n_pairs = n - abs_off
            if n_pairs <= 0:
                continue
            grad[:n_pairs] -= h[abs_off:abs_off + n_pairs]
            grad[abs_off:] += (1.0 - h[:n_pairs])
    return grad * dx, top_idx, corr

        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Also try Thue-Morse and golden ratio
    h_tm = make_thue_morse(n)
    h_tm = project_feasible(h_tm, n)
    c5_tm = compute_c5(h_tm)
    if c5_tm < best_c5:
        best_c5 = c5_tm
        best_h = h_tm.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    def budget_fn():
        return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Initialize from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = make_thue_morse(parent_n)
        parent_h = project_feasible(parent_h, parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(best_h)
        best_n = parent_n
        print("No parent. TM: n={}, C5={:.10f}".format(parent_n, best_c5))

    # Phase 1: SLSQP at parent resolution (quick convergence)
    print("\n--- Phase 1: SLSQP n={} ---".format(best_n))
    best_h, best_c5 = do_slsqp_phase(best_h, best_n, budget_fn, [50, 200, 800])
    print("  After SLSQP: C5={:.10f}".format(best_c5))

    # Phase 2: Binary swap search (key new phase)
    if budget_fn() > 60:
        print("\n--- Phase 2: Binary swap search n={} ---".format(best_n))
        bin_h, bin_c5 = binary_swap_search(best_h, best_n, budget_fn, rng,
                                           max_greedy=400, max_kicks=15)
        if bin_c5 < best_c5 - 1e-14:
            best_c5 = bin_c5
            best_h = bin_h.copy()
            print("  Binary improved: C5={:.10f}".format(best_c5))
        else:
            print("  Binary no improve: C5={:.10f} (best={:.10f})".format(bin_c5, best_c5))

    # Phase 3: SLSQP polish after binary
    if budget_fn() > 20:
        print("\n--- Phase 3: SLSQP polish ---")
        best_h, best_c5 = do_slsqp_phase(best_h, best_n, budget_fn, [2000, 5000, 15000])
        print("  After polish: C5={:.10f}".format(best_c5))

    # Phase 4: Subgradient refinement
    if budget_fn() > 20:
        print("\n--- Phase 4: Subgradient ---")
        best_h, best_c5 = do_subgrad_phase(best_h, best_n, budget_fn, rng, max_iters=200)
        print("  After subgrad: C5={:.10f}".format(best_c5))

    # Phase 5: Multi-resolution search
    print("\n--- Phase 5: Multi-resolution ---")
    resolutions_to_try = [320, 384, 448, 512, 576, 640]
    # Filter out current resolution
    resolutions_to_try = [r for r in resolutions_to_try if r != best_n]

    for n_try in resolutions_to_try:
        if budget_fn() < 30.0:
            break
        h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5:
            print("  n={} interp already better: C5={:.10f}".format(n_try, c5_interp))
            best_h = h_interp.copy()
            best_c5 = c5_interp
            best_n = n_try

        h_new, c5_new = do_slsqp_phase(h_interp, n_try, budget_fn, [200, 1000, 5000])
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n_try
            print("  n={} SLSQP improved: C5={:.10f}".format(n_try, best_c5))
        else:
            print("  n={} no improve: C5={:.10f}".format(n_try, c5_new))

        # Binary search at this resolution if time allows
        if budget_fn() > 40 and c5_new < best_c5 + 0.005:
            bin_h2, bin_c52 = binary_swap_search(h_new, n_try, budget_fn, rng,
                                                 max_greedy=200, max_kicks=8)
            if bin_c52 < best_c5 - 1e-13:
                best_c5 = bin_c52
                best_h = bin_h2.copy()
                best_n = n_try
                print("  n={} binary improved: C5={:.10f}".format(n_try, best_c5))

    # Phase 6: Try periodic inits at best resolution
    if budget_fn() > 40:
        print("\n--- Phase 6: Periodic inits n={} ---".format(best_n))
        per_h, per_c5 = try_periodic_init(best_n, budget_fn, rng)
        if per_h is not None:
            per_h2, per_c52 = do_slsqp_phase(per_h, best_n, budget_fn, [200, 1000, 5000])
            if per_c52 < best_c5 - 1e-13:
                best_c5 = per_c52
                best_h = per_h2.copy()
                print("  Periodic improved: C5={:.10f}".format(per_c52))
            else:
                print("  Periodic no improve: C5={:.10f}".format(per_c52))

    # Phase 7: More binary + continuous cycles
    if budget_fn() > 30:
        print("\n--- Phase 7: Binary+Continuous cycles ---")
        for cycle in range(3):
            if budget_fn() < 15:
                break
            bin_h3, bin_c53 = binary_swap_search(best_h, best_n, budget_fn, rng,
                                                 max_greedy=200, max_kicks=5)
            if bin_c53 < best_c5 - 1e-14:
                best_c5 = bin_c53
                best_h = bin_h3.copy()
            best_h, best_c5 = do_slsqp_phase(best_h, best_n, budget_fn, [5000, 20000])
            print("  Cycle {}: C5={:.10f}".format(cycle, best_c5))

    # Phase 8: Final subgradient + coordinate descent
    if budget_fn() > 10:
        best_h, best_c5 = do_subgrad_phase(best_h, best_n, budget_fn, rng, max_iters=100)
    if budget_fn() > 8:
        best_h, best_c5 = do_coord_descent(best_h, best_n, budget_fn, rng, max_sweeps=8)
    if budget_fn() > 8:
        best_h, best_c5 = do_slsqp_phase(best_h, best_n, budget_fn, [30000, 100000])

    # Phase 9: Final binary micro-search
    if budget_fn() > 10:
        bin_h4, bin_c54 = binary_swap_search(best_h, best_n, budget_fn, rng,
                                             max_greedy=300, max_kicks=5)
        if bin_c54 < best_c5 - 1e-14:
            best_c5 = bin_c54
            best_h = bin_h4.copy()
        # Final SLSQP
        if budget_fn() > 5:
            best_h, best_c5 = do_slsqp_phase(best_h, best_n, budget_fn, [50000])

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
