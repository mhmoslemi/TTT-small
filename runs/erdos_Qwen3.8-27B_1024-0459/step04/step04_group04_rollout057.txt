    return h

def sa_binary(h, n, time_limit=30, seed=42, T_init=3.0, T_min=0.01):
    """Simulated annealing for binary sequences with swap moves."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1

    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()

    T = T_init
    alpha = 0.9992  # Slow cooling
    start = time.time()
    stall = 0
    max_stall = 6000

    while time.time() - start < time_limit and T > T_min:
        if remaining() < 5:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        delta = new_max - cur_max
        if delta < 0 or rng.rand() < np.exp(max(-delta / T, -50)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
        T *= alpha
        if stall > max_stall:
            # Reheat
            T = max(T, T_init * 0.5)
            stall = 0
    return best_h, best_max * dx_val

def sa_binary_2swap(h, n, time_limit=20, seed=42):
    """SA with 2-swap moves (swap two 1s with two 0s)."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1

    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()

    T = 5.0
    alpha = 0.999
    start = time.time()
    stall = 0

    while time.time() - start < time_limit and T > 0.05:
        if remaining() < 5:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        # Pick 2 ones and 2 zeros
        i1, i2 = rng.choice(len(ones), 2, replace=False)
        j1, j2 = rng.choice(len(zeros), 2, replace=False)
        i1, i2 = ones[i1], ones[i2]
        j1, j2 = zeros[j1], zeros[j2]
        
        h[i1], h[i2] = 0.0, 0.0
        h[j1], h[j2] = 1.0, 1.0
        new_max = float(np.max(compute_corr(h)))
        delta = new_max - cur_max
        if delta < 0 or rng.rand() < np.exp(max(-delta / T, -50)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i1], h[i2] = 1.0, 1.0
            h[j1], h[j2] = 0.0, 0.0
            stall += 1
        T *= alpha
        if stall > 5000:
            T = max(T, 2.0)
            stall = 0
    return best_h, best_max * dx_val

def slsqp_smooth(h, n, beta, maxiter=300):
    dx_val = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -50, 50)
        ex = np.exp(cl)
        return (m + np.log(max(np.sum(ex), 1e-300)) / beta) * dx_val
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -50, 50)
        ex = np.exp(cl)
        w = ex / max(np.sum(ex), 1e-300)
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] > 1e-12:
                off = k - (n - 1)
                if off >= 0:
                    npairs = n - off
                    g[off:off+npairs] += (1.0 - x[:npairs])
                    g[:npairs] -= x[off:off+npairs]
                else:
                    ao = -off
                    npairs = n - ao
                    g[:npairs] += (1.0 - x[ao:ao+npairs])
                    g[ao:ao+npairs] -= x[:npairs]
                g *= w[k]
        return g * dx_val
    
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    
    def sum_con(x):
        return np.sum(x) - target
    def sum_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def topk_gradient_descent(h, n, max_iters=300, top_k=5):
    h = h.copy()
    best_c5 = compute_c5(h)
    dx_val = 2.0 / n
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        m = float(np.max(corr))
        if m * dx_val >= best_c5 + 1e-8:
            break
        
        top_idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k in top_idx:
            off = int(k) - (n - 1)
            if off >= 0:
                npairs = n - off
                g[off:off+npairs] += (1.0 - h[:npairs])
                g[:npairs] -= h[off:off+npairs]
            else:
                ao = -off
                npairs = n - ao
                g[:npairs] += (1.0 - h[ao:ao+npairs])
                g[ao:ao+npairs] -= h[:npairs]
        g /= len(top_idx)
        
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        
        improved = False
        for step in [0.05, 0.02, 0.01, 0.1, 0.005, 0.03, 0.008, 0.015, 0.04]:
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-12:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            break
    return h, best_c5

    global _N, _TGT, _BETA
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def lbfgsb_optimize(h, n, beta, maxiter=500):
    global _N, _TGT, _BETA
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(penalty_obj, h, jac=penalty_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def corr_grad_single(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off+npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off+npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao+npairs])
        g[ao:ao+npairs] -= x[:npairs]
    return g

def topk_gradient_descent(h, n, max_iters=400, top_k=5):
    h = h.copy()
    best_c5 = compute_c5(h)
    dx_val = 2.0 / n
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        m = float(np.max(corr))
        if m * dx_val >= best_c5 + 1e-9:
            break
        top_idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k in top_idx:
            g += corr_grad_single(h, n, int(k))
        g /= len(top_idx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        improved = False
        for step in [0.05, 0.02, 0.01, 0.1, 0.005, 0.03, 0.008, 0.015, 0.04, 0.025]:
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-12:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            break
    return h, best_c5

def sa_binary(h, n, time_limit=25, seed=42, T_init=4.0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1

    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()

    T = T_init
    alpha = 0.9993
    start = time.time()
    stall = 0
    max_stall = 8000
    reheat_count = 0
    max_reheat = 5

    while time.time() - start < time_limit and T > 0.02:
        if remaining() < 5:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        delta = new_max - cur_max
        if delta < 0 or (T > 0.02 and rng.rand() < np.exp(max(-delta / T, -50))):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
        T *= alpha
        if stall > max_stall and reheat_count < max_reheat:
            T = max(T, T_init * 0.4)
            stall = 0
            reheat_count += 1
    return best_h, best_max * dx_val

def sa_binary_2swap(h, n, time_limit=15, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1

    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()

    T = 5.0
    alpha = 0.999
    start = time.time()
    stall = 0

    while time.time() - start < time_limit and T > 0.03:
        if remaining() < 5:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i1, i2 = rng.choice(len(ones), 2, replace=False)
        j1, j2 = rng.choice(len(zeros), 2, replace=False)
        i1, i2 = ones[i1], ones[i2]
        j1, j2 = zeros[j1], zeros[j2]
        h[i1], h[i2] = 0.0, 0.0
        h[j1], h[j2] = 1.0, 1.0
        new_max = float(np.max(compute_corr(h)))
        delta = new_max - cur_max
        if delta < 0 or rng.rand() < np.exp(max(-delta / T, -50)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i1], h[i2] = 1.0, 1.0
            h[j1], h[j2] = 0.0, 0.0
            stall += 1
        T *= alpha
        if stall > 4000:
            T = max(T, 2.0)
            stall = 0
    return best_h, best_max * dx_val

def hill_climb_binary(h, n, time_limit=15, seed=99):
    rng = np.random.RandomState(seed)
    h = h.copy()
    dx_val = 2.0 / n
    best_max = float(np.max(compute_corr(h)))
    best_h = h.copy()
    start = time.time()
    stall = 0

    while time.time() - start < time_limit:
        if remaining() < 5:
            break
        if stall > 10000:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_max = float(np.max(compute_corr(h)))
        if new_max < best_max:
            best_max = new_max
            best_h = h.copy()
            stall = 0
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
    return best_h, best_max * dx_val

def coarse_to_fine(h_coarse, n_coarse, n_fine):
    xp = (np.arange(n_coarse) + 0.5) * 2.0 / n_coarse
    x_new = (np.arange(n_fine) + 0.5) * 2.0 / n_fine
    h_interp = np.interp(x_new, xp, h_coarse)
    return make_binary(h_interp, n_fine, np.random.RandomState(0))
