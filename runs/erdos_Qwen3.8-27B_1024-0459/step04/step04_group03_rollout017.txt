def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def interp_to_finite(src_h, src_n, dst_n):
    """Interpolation using cubic for smoother result."""
    if src_n == dst_n:
        return src_h.copy()
    from scipy.interpolate import interp1d
    xp = (np.arange(src_n)) * (2.0 / src_n)
    x_new = (np.arange(dst_n)) * (2.0 / dst_n)
    # Clamp to avoid extrapolation
    x_new_clamped = np.clip(x_new, xp[0], xp[-1])
    f = interp1d(xp, src_h, kind='cubic', bounds_error=False, fill_value='extrapolate')
    return np.clip(f(x_new_clamped), 0, 1)

def spectral_smooth(h, n, num_keep=64):
    """Keep only top frequency components and reconstruct."""
    spectrum = np.fft.rfft(h)
    mag = np.abs(spectrum)
    # Keep DC and top num_keep frequencies
    keep = np.zeros(len(spectrum), dtype=bool)
    keep[0] = True
    top_idx = np.argsort(mag)[::-1][:num_keep]
    keep[top_idx] = True
    spectrum_filt = spectrum.copy()
    spectrum_filt[~keep] = 0
    h_new = np.fft.irfft(spectrum_filt, n)
    return np.clip(h_new, 0, 1)

def local_binary_search(h, n, rng, max_iters=200):
    """Try flipping individual or small groups of near-boundary values."""
    best_h = h.copy()
    best_c5 = compute_c5(h)

        if len(trans_idx) < 2:
            break

        idx = trans_idx[rng.randint(len(trans_idx))]
        neighbors = [idx]
        for d in [1, -1, 2, -2]:
            ni = idx + d
            if 0 <= ni < n:
                neighbors.append(ni)

                    break

                break

def corr_gradient(h, k):
    """Gradient of correlation at lag k (full mode index) w.r.t. h."""
    n = len(h)
    shift = k - (n - 1)  # actual shift
    grad = np.zeros(n)
    if shift >= 0:
        # d(c_k)/d(h[j]) = (1-h[j-shift]) for shift≤j≤n-1, minus h[j+shift] for j≤n-1-shift
        j_range_1 = np.arange(shift, n)
        grad[j_range_1] += (1.0 - h[j_range_1 - shift])
        j_range_2 = np.arange(0, n - shift)
        grad[j_range_2] -= h[j_range_2 + shift]
    else:
        M = -shift
        # d(c_{-M})/d(h[j]) = (1-h[j+M]) for j≤n-1-M, minus h[j-M] for j≥M
        j_range_1 = np.arange(0, n - M)
        grad[j_range_1] += (1.0 - h[j_range_1 + M])
        j_range_2 = np.arange(M, n)
        grad[j_range_2] -= h[j_range_2 - M]
    return grad

        j2 = np.arange(0, n - shift)
        if len(j2) > 0:
            grad[j2] -= h[j2 + shift]
    else:
        M = -shift
        j1 = np.arange(0, n - M)
        if len(j1) > 0:
            grad[j1] += (1.0 - h[j1 + M])
        j2 = np.arange(M, n)
        if len(j2) > 0:
            grad[j2] -= h[j2 - M]
    return grad

def subgradient_polish(h, n, steps=60, base_alpha=0.015):
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for step in range(steps):
        if remaining_time() < 2:
            break
        corr = compute_corr(h)
        max_c = np.max(corr)
        # Use lags within 0.5% of max
        active = np.where(corr >= max_c * 0.9995)[0]
        if len(active) > 8:
            active = np.argsort(corr)[::-1][:8]

        grad = np.zeros(n)
        for idx in active:
            grad += corr_gradient(h, idx)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad_n = grad / gnorm

        alpha = base_alpha / (1.0 + 0.15 * step)
        for factor in [1.0, 0.5, 0.25, 0.1]:
            a = alpha * factor
            h_trial = project_feasible(h - a * grad_n, n)
            c5_t = compute_c5(h_trial)
            if c5_t < best_c5 - 1e-16:
                best_c5 = c5_t
                best_h = h_trial.copy()
                h = h_trial.copy()
                break

    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def interp_cubic(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    from scipy.interpolate import PchipInterpolator
    xp = np.arange(src_n) * (2.0 / src_n)
    x_new = np.arange(dst_n) * (2.0 / dst_n)
    fc = PchipInterpolator(xp, src_h, extrapolate=True)
    return np.clip(fc(x_new), 0.0, 1.0)

def local_pair_search(h, n, rng, max_its=500):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    improved_streak = 0

    for it in range(max_its):
        if remaining_time() < 3:
            break
        if improved_streak > 30:
            break
        i = rng.randint(n)
        hi = best_h[i]
        if hi < 0.02 or hi > 0.98:
            continue
        for d in range(1, 12):
            j = (i + d) % n
            hj = best_h[j]
            if abs(hi - hj) < 0.03:
                continue
            for frac in [0.2, 0.4, 0.6, 0.8]:
                amt = (hi - hj) * frac
                if abs(amt) < 0.005:
                    continue
                h_t = best_h.copy()
                h_t[i] -= amt
                h_t[j] += amt
                h_t = project_feasible(h_t, n)
                c5_t = compute_c5(h_t)
                if c5_t < best_c5 - 1e-15:
                    best_c5 = c5_t
                    best_h = h_t.copy()
                    improved_streak = 0
                    break
        else:
            improved_streak += 1
        continue

    return best_h, best_c5

def weighted_minimax_slsqp(h, n, max_rounds=12, max_iter=250):
    """Minimax with weighted active-set focus."""
    global _N_G, _DX_G, _LAGS_G
    _N_G = n
    _DX_G = 2.0 / n
    h = h.copy()
    dx = 2.0 / n

    for rnd in range(max_rounds):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        max_corr = np.max(corr)
        t_init = max_corr * dx + 1e-10

        # Weighted selection: focus on top lags
        ranked = np.argsort(corr)[::-1]
        top_k = min(20, len(ranked))
        active = ranked[:top_k]
        # Keep only those within 1% of max
        active = active[corr[active] >= max_corr * 0.99]
        active = active[np.argsort(active)]
        if len(active) < 3:
            active = ranked[:5]
            active = active[np.argsort(active)]

        _LAGS_G = active
        x0 = np.concatenate([h, [t_init]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': _mm_ineq},
            {'type': 'eq', 'fun': _mm_eq}
        ]
        try:
            res = minimize(_mm_obj, x0, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter, 'ftol': 1e-18, 'iprint': 0})
            if res.x is not None and res.x.size == n + 1:
                h_new = project_feasible(res.x[:n], n)
                c5_new = compute_c5(h_new)
                c5_old = compute_c5(h)
                if c5_new < c5_old - 1e-18:
                    h = h_new
                elif rnd > 5:
                    break
        except Exception:
            break
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S, _N_G, _BETA_G, _LAGS_G, _DX_G
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 256
        best_h = np.full(best_n, 0.5)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    # === Phase 1: Optimize at parent resolution (n=256) ===
    n1 = best_n
    h1 = best_h.copy()

    for beta in [50, 200, 800, 3000, 10000, 30000]:
        if remaining_time() < 12:
            break
        h1 = slsqp_smooth(h1, n1, beta, maxiter=400)
        c5_new = compute_c5(h1)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h1.copy()

    # Minimax at parent res
    if remaining_time() > 20:
        h1 = weighted_minimax_slsqp(h1, n1, max_rounds=12, max_iter=250)
        c5_new = compute_c5(h1)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h1.copy()

    # Subgradient at parent res
    if remaining_time() > 10:
        h1, c5_sg = subgradient_polish(h1, n1, steps=40, base_alpha=0.02)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h1.copy()
        if remaining_time() > 5:
            h1 = slsqp_smooth(h1, n1, 20000, maxiter=200)
            c5_new = compute_c5(h1)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h1.copy()

    # === Phase 2: Upsample to n=512 and optimize ===
    n2 = 512
    h2 = interp_to(best_h, best_n, n2)
    h2 = project_feasible(h2, n2)
    c5_2 = compute_c5(h2)

    if c5_2 < best_c5 + 0.005:
        for beta in [80, 300, 1200, 5000, 15000]:
            if remaining_time() < 12:
                break
            h2 = slsqp_smooth(h2, n2, beta, maxiter=350)
            c5_new = compute_c5(h2)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h2.copy()
                best_n = n2

        if remaining_time() > 25:
            h2 = weighted_minimax_slsqp(h2, n2, max_rounds=14, max_iter=250)
            c5_new = compute_c5(h2)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h2.copy()
                best_n = n2

        if remaining_time() > 15:
            h2, c5_sg = subgradient_polish(h2, n2, steps=40, base_alpha=0.01)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h2.copy()
                best_n = n2
            if remaining_time() > 8:
                h2 = slsqp_smooth(h2, n2, 30000, maxiter=200)
                c5_new = compute_c5(h2)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h2.copy()
                    best_n = n2

    # === Phase 3: Try n=384 (between 256 and 512) ===
    n3 = 384
    h3 = interp_cubic(best_h, best_n, n3)
    h3 = project_feasible(h3, n3)
    c5_3 = compute_c5(h3)

    if c5_3 < best_c5 + 0.003:
        for beta in [200, 1000, 4000, 12000]:
            if remaining_time() < 10:
                break
            h3 = slsqp_smooth(h3, n3, beta, maxiter=300)
            c5_new = compute_c5(h3)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h3.copy()
                best_n = n3

        if remaining_time() > 15:
            h3 = weighted_minimax_slsqp(h3, n3, max_rounds=10, max_iter=200)
            c5_new = compute_c5(h3)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h3.copy()
                best_n = n3

    # === Phase 4: Try n=768 ===
    n4 = 768
    if remaining_time() > 80:
        h4 = interp_to(best_h, best_n, n4)
        h4 = project_feasible(h4, n4)
        c5_4 = compute_c5(h4)
        if c5_4 < best_c5 + 0.004:
            for beta in [100, 500, 2000, 8000]:
                if remaining_time() < 10:
                    break
                h4 = slsqp_smooth(h4, n4, beta, maxiter=250)
                c5_new = compute_c5(h4)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h4.copy()
                    best_n = n4
            if remaining_time() > 15:
                h4 = weighted_minimax_slsqp(h4, n4, max_rounds=8, max_iter=180)
                c5_new = compute_c5(h4)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h4.copy()
                    best_n = n4

    # === Phase 5: Local combinatorial search at best resolution ===
    if remaining_time() > 25:
        h_loc, c5_loc = local_pair_search(best_h, best_n, rng, max_its=400)
        if c5_loc < best_c5:
            best_c5 = c5_loc
            best_h = h_loc.copy()
        if remaining_time() > 10:
            h_loc = slsqp_smooth(h_loc, best_n, 10000, maxiter=200)
            c5_new = compute_c5(h_loc)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_loc.copy()

    # === Phase 6: Multi-start perturbations ===
    for restart in range(3):
        if remaining_time() < 25:
            break
        h_p = best_h.copy()
        mask = (h_p > 0.03) & (h_p < 0.97)
        cnt_mask = np.sum(mask)
        if cnt_mask > 3:
            h_p[mask] += rng.normal(0, 0.025, cnt_mask)
        # Flip 2-3 random interior points
        interior = np.where((h_p > 0.15) & (h_p < 0.85))[0]
        if len(interior) > 3:
            flip_idx = rng.choice(interior, size=min(3, len(interior)), replace=False)
            for fi in flip_idx:
                h_p[fi] = 1.0 - h_p[fi]
        h_p = project_feasible(h_p, best_n)
        c5_p = compute_c5(h_p)

        if c5_p < best_c5 + 0.008:
            h_p = slsqp_smooth(h_p, best_n, 800, maxiter=250)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()
            if remaining_time() > 10:
                h_p = weighted_minimax_slsqp(h_p, best_n, max_rounds=8, max_iter=150)
                c5_p = compute_c5(h_p)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p.copy()

    # === Phase 7: Ultra-precision final polish ===
    if remaining_time() > 15:
        for beta in [50000, 150000, 400000]:
            if remaining_time() < 5:
                break
            h_new = slsqp_smooth(best_h, best_n, beta, maxiter=150)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # === Phase 8: Final minimax ===
    if remaining_time() > 15:
        h_mm = weighted_minimax_slsqp(best_h, best_n, max_rounds=10, max_iter=200)
        c5_mm = compute_c5(h_mm)
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm.copy()

    # === Phase 9: Final subgradient ===
    if remaining_time() > 8:
        h_sg, c5_sg = subgradient_polish(best_h, best_n, steps=30, base_alpha=0.005)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # === Phase 10: Final local search ===
    if remaining_time() > 8:
        h_loc, c5_loc = local_pair_search(best_h, best_n, rng, max_its=200)
        if c5_loc < best_c5:
            best_c5 = c5_loc
            best_h = h_loc.copy()
        if remaining_time() > 5:
            h_loc = slsqp_smooth(h_loc, best_n, 50000, maxiter=100)
            c5_new = compute_c5(h_loc)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_loc.copy()

    # Final projection
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, final_c5, best_n)
