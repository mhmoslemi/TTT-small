                grad[:npairs] -= w * h[aoff:]
    return val, grad

def slsqp_optimize(h, n, beta, maxiter=500):
    thresholds = []
    def obj(x):
        return smooth_max_and_grad(x, n, beta)[0]
    def obj_grad(x):
        return smooth_max_and_grad(x, n, beta)[1]
    def sum_con(x):
        return np.sum(x) - n / 2.0
    def sum_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            out = project_feasible(res.x, n)
            if compute_c5(out) <= compute_c5(h) + 1e-14:
                return out
    except Exception:
        pass
    return h.copy()

def interp_upsample(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_new = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    return np.interp(x_new, xp, h_src)

def coord_descent(h, n, max_sweeps=100):
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        # Process in random order
        order = np.random.permutation(n)
        
        for i in order:
            if remaining_time() < 2:
                break
            
            # Precompute: for each lag k, the coefficient of t (=h[i]) 
            # in corr_k = base_k + a_k * t + b_k * t^2
            a = np.zeros(ncorr)
            b = np.zeros(ncorr)  # only nonzero for k corresponding to off=0
            
            t0 = best[i]
            for off in range(-(n-1), n):
                k = n - 1 + off
                if off == 0:
                    # corr_k includes h[i]*(1-h[i]) = t*(1-t)
                    a[k] += 1.0
                    b[k] += -1.0
                else:
                    if 0 <= i + off < n:
                        a[k] += (1.0 - best[i + off])
                    if 0 <= i - off < n:
                        a[k] += (-best[i - off])
            
            base = best_corr - a * t0 - b * t0 * t0
            
            # Find best t in [0,1] minimizing max_k (base_k + a_k*t + b_k*t^2) * dx
            # This is a 1D optimization problem
            
            # Try grid + golden section
            # First, check endpoints and current value
            candidates = [0.0, 1.0, t0]
            # Add critical points of the "max" - hard to find analytically
            # Try a few additional candidates
            if t0 > 0:
                candidates.append(t0 * 0.5)
                candidates.append(t0 * 0.75)
            if t0 < 1:
                candidates.append(t0 * 1.25)
                candidates.append(t0 * 1.5)
            candidates.append(1.0 - t0)
            
            best_t = t0
            best_val = np.max(base + a * t0 + b * t0 * t0) * dx
            
            for t in candidates:
                t = max(0.0, min(1.0, t))
                if abs(t - t0) < 1e-12:
                    continue
                val = np.max(base + a * t + b * t * t) * dx
                if val < best_val - 1e-16:
                    best_val = val
                    best_t = t
            
            # Golden section refinement
            if best_t != t0:
                lo = max(0.0, min(t0, best_t) - 0.1)
                hi = min(1.0, max(t0, best_t) + 0.1)
                lo = max(0.0, lo)
                hi = min(1.0, hi)
                
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                f2 = np.max(base + a * c2 + b * c2 * c2) * dx
                
                for _ in range(50):
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = np.max(base + a * c2 + b * c2 * c2) * dx
                
                t_gs = (lo + hi) / 2.0
                f_gs = np.max(base + a * t_gs + b * t_gs * t_gs) * dx
                if f_gs < best_val - 1e-16:
                    best_val = f_gs
                    best_t = t_gs
            
            if best_val < best_c5 - 1e-15:
                best[i] = best_t
                # Recompute full correlation
                best_corr = compute_corr(best)
                best_c5 = compute_c5(best)
                improved = True
        
        if not improved:
            break
    
    return best, best_c5

def subgradient_step(h, n, step_size):
    corr = compute_corr(h)
    max_corr = np.max(corr)
    # Active set: lags within tolerance of max
    tol = 0.1  # lags within 0.1 of max are "active"
    active = np.where(corr >= max_corr - tol)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    
    grad = np.zeros(n)
    for k_idx in active:
        off = int(k_idx) - (n - 1)
        if off >= 0:
            npairs = n - off
            if npairs > 0:
                grad[:npairs] += (1.0 - h[off:off+npairs])
                grad[off:off+npairs] -= h[:npairs]
        else:
            aoff = -off
            npairs = n - aoff
            if npairs > 0:
                grad[aoff:] += (1.0 - h[:npairs])
                grad[:npairs] -= h[aoff:]
    
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return h.copy(), True  # converged
    
    h_new = h - step_size * grad / gnorm
    h_new = project_feasible(h_new, n)
    return h_new, False
