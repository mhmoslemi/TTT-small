def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        idx = int(np.argmin(np.minimum(h, 1.0 - h)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

def smooth_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * corr / dx
    max_s = np.max(scaled)
    exps = np.exp(scaled - max_s)
    ssum = np.sum(exps)
    if ssum <= 0:
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w = exps / ssum
    smooth_val = (max_s + np.log(ssum)) / beta * dx
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        s = m - (n - 1)
        wm = w[m]
        if wm < 1e-14:
            continue
        j1s = max(0, -s)
        j1e = min(n, n - s)
        if j1s < j1e:

        if j2s < j2e:

def coordinate_descent(h, n, n_passes, time_limit):

            valid2 = (i - s_vals >= 0) & (i - s_vals < n)
