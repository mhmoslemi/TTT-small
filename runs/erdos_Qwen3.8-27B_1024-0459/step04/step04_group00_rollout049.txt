def compute_c5(h):
    n = len(h)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():

def grad_corr_single(h, k, n):
    off = k - (n - 1)
    g = np.zeros(n)
    if off >= 0:
        npairs = n - off
        if npairs <= 0:
            return g
        g[off:off + npairs] += (1.0 - h[:npairs])
        g[:npairs] -= h[off:off + npairs]
    else:
        a = -off
        npairs = n - a
        if npairs <= 0:
            return g
        g[:npairs] += (1.0 - h[a:a + npairs])
        g[a:a + npairs] -= h[:npairs]
    return g


def smooth_obj(x, n, beta):
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    s = np.sum(ex)
    return (m + np.log(s) / beta) * dx_v


def smooth_grad(x, n, beta):
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    s = np.sum(ex)
    w = ex / s
    g = np.zeros(n)
    thresh = 1e-8
    for k in range(2 * n - 1):
        wk = w[k] * dx_v
        if wk < thresh:
            continue
        g += wk * grad_corr_single(x, k, n)
    return g


def sum_constraint(x):
    return np.sum(x) - len(x) / 2.0


def sum_jac(x):
    return np.ones(len(x))


def slsqp_opt(h, n, beta, maxiter=400):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac}]
    try:
        res = minimize(lambda x: smooth_obj(x, n, beta), h,
                       jac=lambda x: smooth_grad(x, n, beta),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.fun < smooth_obj(h, n, beta):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
