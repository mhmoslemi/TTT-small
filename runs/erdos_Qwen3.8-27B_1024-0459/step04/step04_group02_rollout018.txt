    return np.max(corr) * dx

For SLSQP, I'll use the smooth approximation but with a much faster gradient. The key is:
grad[i] = dx * sum_k w_k * J[k, i]

Where J[k, i] = A_i[k] (the same A vector from coordinate descent) for k ≠ center, and J[center, i] = 1 - 2*h[i].

To compute grad efficiently:
grad[i] = dx * (w_{center} * (1-2*h[i]) + sum_{k≠center} w_k * A_i[k])

The sum over k can be written as:
sum_{k=0}^{2n-2} w_k * A_i[k] = sum_m w_{m+n-1} * [(1-h[i+m])_v - h[i-m]_v]_{m≠0}

= sum_{m≠0} w_{m+n-1} * (1-h[i+m])_v - sum_{m≠0} w_{m+n-1} * h[i-m]_v

def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = np.argmin(h)
            else:
                idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-14 and n > 0:
        idx = int(np.argmin(np.abs(h - 0.5)))

def _smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    center = n - 1
    m_offsets = np.arange(-(n-1), n)
    for i in range(n):
        A = np.zeros(2*n-1)
        valid1 = (i + m_offsets >= 0) & (i + m_offsets < n)
        A[valid1] += 1.0 - x[i + m_offsets[valid1]]
        valid2 = (i - m_offsets >= 0) & (i - m_offsets < n) & (m_offsets != 0)
        A[valid2] -= x[i - m_offsets[valid2]]
        A[center] = 0.0
        grad[i] = dx * np.dot(weights, A)
        # Add center contribution
        grad[i] += dx * weights[center] * (1.0 - 2.0 * x[i])
    return val, grad

    _N_WORK = n
    _BETA_WORK = beta

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            if compute_c5(trial) <= compute_c5(h) + 1e-12:
                return trial

        pass
    return h.copy()

    if src_n == dst_n:

    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def subgradient_step(h, n, n_active=5, step=0.01):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    active = np.where(corr >= max_corr - 0.3)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:

        A = np.zeros(2*n-1)

        A[valid2] -= h[i - m_offsets[valid2]]

        for k_idx in active:

            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 7: Ultra-fine coordinate descent (narrow grid near current values)
    if remaining_time() > 10:
        h_fine = best_h.copy()
        n_fine = best_n
        dx_fine = 2.0 / n_fine
        ncorr_fine = 2 * n_fine - 1
        center_fine = n_fine - 1
        m_offsets_fine = np.arange(-(n_fine-1), n_fine)
        corr_fine = np.correlate(h_fine, 1.0 - h_fine, mode='full')
        c5_fine = np.max(corr_fine) * dx_fine
        
        for sweep in range(50):
            if remaining_time() < 3:
                break
            improved = False
            for i in range(n_fine):
                t0 = h_fine[i]
                # Narrow grid around current value
                lo = max(0.0, t0 - 0.05)
                hi = min(1.0, t0 + 0.05)
                t_grid = np.linspace(lo, hi, 50)
                
                A = np.zeros(ncorr_fine)
                valid1 = (i + m_offsets_fine >= 0) & (i + m_offsets_fine < n_fine)
                A[valid1] += 1.0 - h_fine[i + m_offsets_fine[valid1]]
                valid2 = (i - m_offsets_fine >= 0) & (i - m_offsets_fine < n_fine) & (m_offsets_fine != 0)
                A[valid2] -= h_fine[i - m_offsets_fine[valid2]]
                A[center_fine] = 0.0
                
                dt = t_grid - t0
                corr_new = corr_fine[None, :] + dt[:, None] * A[None, :]
                corr_new[:, center_fine] = corr_fine[center_fine] + dt * (1.0 - t_grid - t0)
                H = np.max(corr_new, axis=1) * dx_fine
                idx_min = np.argmin(H)
                H_min = H[idx_min]
                
                if H_min < c5_fine - 1e-14:
                    # Golden section
                    delta = (hi - lo) / 49.0
                    glo = max(0.0, t_grid[idx_min] - 3*delta)
                    ghi = min(1.0, t_grid[idx_min] + 3*delta)
                    if ghi - glo > 1e-10:
                        gr = (np.sqrt(5.0) - 1.0) / 2.0
                        c1 = ghi - gr * (ghi - glo)
                        c2 = glo + gr * (ghi - glo)
                        dt1 = c1 - t0
                        v1 = corr_fine + dt1 * A
                        v1[center_fine] = corr_fine[center_fine] + dt1 * (1.0 - c1 - t0)
                        f1 = np.max(v1) * dx_fine
                        dt2 = c2 - t0
                        v2 = corr_fine + dt2 * A
                        v2[center_fine] = corr_fine[center_fine] + dt2 * (1.0 - c2 - t0)
                        f2 = np.max(v2) * dx_fine
                        for _ in range(30):
                            if f1 < f2:
                                ghi = c2; c2 = c1; f2 = f1
                                c1 = ghi - gr * (ghi - glo)
                                dt1 = c1 - t0
                                v1 = corr_fine + dt1 * A
                                v1[center_fine] = corr_fine[center_fine] + dt1 * (1.0 - c1 - t0)
                                f1 = np.max(v1) * dx_fine
                            else:
                                glo = c1; c1 = c2; f1 = f2
                                c2 = glo + gr * (ghi - glo)
                                dt2 = c2 - t0
                                v2 = corr_fine + dt2 * A
                                v2[center_fine] = corr_fine[center_fine] + dt2 * (1.0 - c2 - t0)
                                f2 = np.max(v2) * dx_fine
                        t_gs = (glo + ghi) / 2.0
                        dt_gs = t_gs - t0
                        vals = corr_fine + dt_gs * A
                        vals[center_fine] = corr_fine[center_fine] + dt_gs * (1.0 - t_gs - t0)
                        H_gs = np.max(vals) * dx_fine
                        if H_gs < H_min:
                            t_grid[idx_min] = t_gs
                            H_min = H_gs
                    
                    if H_min < c5_fine - 1e-14:
                        h_fine[i] = t_grid[idx_min]
                        dt_inc = t_grid[idx_min] - t0
                        corr_fine += dt_inc * A
                        corr_fine[center_fine] = np.sum(h_fine * (1.0 - h_fine))
                        c5_fine = np.max(corr_fine) * dx_fine
                        improved = True
            if not improved:
                break
        
        h_fine = project_feasible(h_fine, n_fine)
        c5_fine = compute_c5(h_fine)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()
            best_n = n_fine

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
