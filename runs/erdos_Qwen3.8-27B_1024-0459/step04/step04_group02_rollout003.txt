        wsum = np.sum(exp_vals)

        return np.ones(n)

                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            if compute_c5(trial) <= compute_c5(h) + 1e-14:
                return trial
    except Exception:
        pass
    return h.copy()

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def subgradient_step(h, n, n_active=8):
    corr = compute_corr(h)
    max_corr = np.max(corr)
    # Active set: lags within 0.05 of max
    active = np.where(corr >= max_corr - 0.05)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        # Pick the top n_active
        top_idx = np.argsort(corr)[::-1][:n_active]
        active = top_idx
    
    grad = np.zeros(n)
    k0 = n - 1
    for idx in active:
        k = int(idx)
        offset = k - k0
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset+npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset+npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff+npairs])
                grad[aoff:aoff+npairs] -= h[:npairs]
    
    # Remove mean component to preserve sum constraint
    grad -= np.mean(grad)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return None, 0.0
    return grad / gnorm, gnorm

def project_sum(h, n):
    """Project h onto {sum(h) = n/2, 0 <= h <= 1} via simple method."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    mask = (h > 1e-10) & (h < 1.0 - 1e-10)
    if np.sum(mask) > 0:
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            # Add to smallest
            idx = int(np.argmin(h))
            h[idx] = np.clip(h[idx] + diff, 0, 1)
        else:
            idx = int(np.argmax(h))
            h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        h_tmp = np.zeros(best_n)
        for i in range(best_n):
            v = i
            cnt = 0
            while v:
                cnt ^= (v & 1)
                v >>= 1
            h_tmp[i] = 1.0 if cnt == 1 else 0.0
        best_h = project_feasible(h_tmp, best_n)
        best_c5 = compute_c5(best_h)
    
    # Phase 1: Thorough coordinate descent at parent resolution
    n = best_n
    h = best_h.copy()
    h, c5 = fast_coord_descent(h, n, max_sweeps=150, seed=seed)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # More coordination sweep with different seed
    if remaining_time() > 30:
        h2 = best_h.copy()
        h2, c5_2 = fast_coord_descent(h2, n, max_sweeps=80, seed=seed+1)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
    
    # Phase 2: SLSQP polish at increasing beta
    for beta in [500, 2000, 8000, 30000]:
        if remaining_time() < 25:
            break
        h_new = slsqp_smooth(best_h, best_n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 3: Try higher resolutions
    resolutions_tried = set([best_n])
    for rn in [int(best_n * 1.5), 2 * best_n, int(best_n * 2.5), 1280, 960, 1920]:
        if rn % 2 != 0 or rn < 64 or rn > 2560:
            continue
        if rn in resolutions_tried:
            continue
        resolutions_tried.add(rn)
        if remaining_time() < 60:
            break
        
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)
        
        # Coordinate descent at new resolution
        h_r, c5_r = fast_coord_descent(h_r, rn, max_sweeps=80, seed=seed+10)
        
        # SLSQP polish
        if remaining_time() > 30:
            h_r2 = slsqp_smooth(h_r, rn, 3000, maxiter=300)
            c5_r2 = compute_c5(h_r2)
            if c5_r2 < c5_r:
                h_r = h_r2
                c5_r = c5_r2
        
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn
    
    # Phase 4: Subgradient with momentum + coordinate descent interleaved
    if remaining_time() > 60:
        h = best_h.copy()
        n = best_n
        momentum = 0.0
        v = np.zeros(n)
        for it in range(500):
            if remaining_time() < 8:
                break
            grad, gnorm = subgradient_step(h, n, 10)
            if grad is None:
                break
            lr = 0.02 / (1.0 + it * 0.005)
            v = 0.9 * v - lr * grad
            h_trial = project_sum(h + v, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-16:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial.copy()
            else:
                # Revert, try smaller step
                v = 0.5 * v
                h_trial2 = project_sum(h + v, n)
                c5_trial2 = compute_c5(h_trial2)
                if c5_trial2 < best_c5:
                    best_c5 = c5_trial2
                    best_h = h_trial2.copy()
                    h = h_trial2.copy()
            
            # Every 50 iters, do a coordinate descent pass
            if it % 50 == 40 and remaining_time() > 20:
                h_cd, c5_cd = fast_coord_descent(h, n, max_sweeps=20, seed=seed+it)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                    h = h_cd.copy()
                momentum = 0.0
                v = np.zeros(n)
    
    # Phase 5: Try a different starting point (random binary-like)
    if remaining_time() > 80:
        try_n = max(best_n, 960)
        if try_n % 2 != 0:
            try_n += 1
        if try_n <= 2560:
            for attempt in range(3):
                if remaining_time() < 40:
                    break
                h_rand = np.zeros(try_n)
                positions = rng.choice(try_n, try_n // 2, replace=False)
                h_rand[positions] = 1.0
                h_rand = project_feasible(h_rand, try_n)
                c5_rand = compute_c5(h_rand)
                if c5_rand < 0.45:
                    h_rand, c5_rand = fast_coord_descent(h_rand, try_n, max_sweeps=60, seed=seed+100+attempt)
                    if remaining_time() > 20:
                        h_rand = slsqp_smooth(h_rand, try_n, 5000, maxiter=300)
                        c5_rand = compute_c5(h_rand)
                    if c5_rand < best_c5:
                        best_c5 = c5_rand
                        best_h = h_rand.copy()
                        best_n = try_n
    
    # Phase 6: Final intensive coordinate descent
    if remaining_time() > 30:
        h, c5 = fast_coord_descent(best_h, best_n, max_sweeps=200, seed=seed+200)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase 7: Final SLSQP polish
    if remaining_time() > 15:
        h_new = slsqp_smooth(best_h, best_n, 50000, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 8: One last coordinate descent
    if remaining_time() > 10:
        h, c5 = fast_coord_descent(best_h, best_n, max_sweeps=50, seed=seed+300)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final projection and verification
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    # Try the evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
