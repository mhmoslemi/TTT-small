Actually, I realize I should think more carefully about the SA moves. The key insight for this problem:

For a binary string with n/2 ones, the correlation at shift d counts the number of "1 followed by 0 at distance d" patterns. To reduce the max correlation:
1. Find the shift d* with maximum correlation
2. Identify which pairs (i, i+d*) contribute (h[i]=1, h[i+d*]=0)
3. Move some 1s away from these positions, or move 0s into them

The most effective move is to find a 1 at position i where h[i+d*]=0 (contributing to the max) and move it to position j where h[j+d*]=1 (not contributing, and the 0 at j was contributing to a different shift).

But I also need to check that this move doesn't increase other shifts' correlations too much.

Let me implement a "smart" move:
1. Find worst shift d*
2. Find all contributing (i, j=i+d*) pairs
3. For each such pair, consider moving the 1 at i to some zero position j'
4. Score the move by the change in max(corr)

This is O(n) per candidate, and there are O(n) candidates, so O(n²) per move evaluation. With n=512, that's 262K ops per move, which is too slow for many iterations.

            if not found:
                zi = rng.randint(len(zeros_arr))
                j = zeros_arr[zi]
        else:
            # Block reversal
            s = rng.randint(0, n - 1)
            length = rng.randint(2, max(3, n // 10))
            if s + length > n:
                s = n - length
            e = s + length - 1
            h[s:e+1] = h[s:e+1][::-1]
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            delta = new_max - cur_max
            accept = delta < 0 or (T > 0.01 and rng.rand() < np.exp(-max(delta, 0.0) / T))
            if not accept:
                h[s:e+1] = h[s:e+1][::-1]
                T *= cool
                n_iters += 1
                continue
            cur_max = new_max
            corr = new_corr
            ones_arr = np.where(h == 1)[0].copy().tolist()
            zeros_arr = np.where(h == 0)[0].copy().tolist()
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
            T *= cool
            n_iters += 1
            continue

        # Check tabu
        if tabu_len > 0:
            if i in tabu_i and tabu_counter - tabu_i[i] < tabu_len:
                T *= cool
                n_iters += 1
                continue
            if j in tabu_j and tabu_counter - tabu_j[j] < tabu_len:
                T *= cool
                n_iters += 1
                continue

        # Perform swap
        h[i] = 0.0
        h[j] = 1.0
        ones_arr[oi] = j
        zeros_arr[zi] = i

        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))

        delta = new_max - cur_max
        accept = False
        if delta < 0:
            accept = True
        elif T > 0.005:
            if rng.rand() < np.exp(-abs(delta) / T):
                accept = True

        if accept:
            cur_max = new_max
            corr = new_corr
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
            if tabu_len > 0:
                tabu_i[i] = tabu_counter
                tabu_j[j] = tabu_counter
        else:
            h[i] = 1.0
            h[j] = 0.0
            ones_arr[oi] = i
            zeros_arr[zi] = j

        tabu_counter += 1
        T *= cool
        n_iters += 1

    return best_h, best_max * (2.0 / n), n_iters

def greedy_first_improve(h_bin, n, time_limit, rng, n_cand=100):
    h = h_bin.copy()
    start = time.time()
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()
    stall = 0

    while time.time() - start < time_limit and remaining() > 12 and stall < 300:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        found = False
        for _ in range(n_cand):
            i_val = ones[rng.randint(len(ones))]
            j_val = zeros[rng.randint(len(zeros))]
            h[i_val] = 0.0
            h[j_val] = 1.0
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            if new_max < cur_max - 1e-12:
                corr = new_corr
                cur_max = new_max
                found = True
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
                break
            else:
                h[i_val] = 1.0
                h[j_val] = 0.0
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 50:
                # Random kick
                i_val = ones[rng.randint(len(ones))]
                j_val = zeros[rng.randint(len(zeros))]
                h[i_val] = 0.0
                h[j_val] = 1.0
                corr = compute_corr(h)
                cur_max = float(np.max(corr))
                stall = 0

    return best_h, best_max * (2.0 / n)

def _slsqp_obj(x, n, beta):
    dx_val = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -50, 50)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val

def _slsqp_grad(x, n, beta):
    dx_val = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -50, 50)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    nc = len(corr)
    g = np.zeros(n)
    for k in range(nc):
        if w[k] > 1e-15:
            d = k - (n - 1)
            if d >= 0:
                npairs = n - d
                if npairs > 0:
                    g[d:d+npairs] += w[k] * (1.0 - x[:npairs])
                    g[:npairs] -= w[k] * x[d:d+npairs]
            else:
                ao = -d
                npairs = n - ao
                if npairs > 0:
                    g[:npairs] += w[k] * (1.0 - x[ao:ao+npairs])
                    g[ao:ao+npairs] -= w[k] * x[:npairs]
    return g * dx_val

def _sum_con(x, n):
    return np.sum(x) - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_refine(h, n, beta, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: _sum_con(x, n),
             'jac': lambda x: _sum_jac(x, n)}]
    try:
        res = minimize(_slsqp_obj, h, jac=lambda x: _slsqp_grad(x, n, beta),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 312
        parent_h = np.zeros(parent_n)
        parent_h[parent_n//4 : 3*parent_n//4] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: SA on parent resolution
    h_bin_parent = to_binary(best_h, best_n, rng)
    if remaining() > 80:
        h_sa, c5_sa, _ = sa_optimize(h_bin_parent, best_n, time_limit=min(70, remaining()-10),
                                      rng=rng, T0=3.0, cool=0.999997, tabu_len=0)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = best_n

    # Phase 2: Try multiple n values with SA
    n_candidates = [256, 288, 312, 320, 352, 384, 416, 448, 480, 512, 544, 576, 640]
    n_candidates = [n for n in n_candidates if n != best_n and n >= 128 and n <= 700]
    # Sort by priority: closer to current best_n first, then larger
    n_candidates.sort(key=lambda x: (abs(x - best_n), x))

    for n_try in n_candidates[:6]:
        if remaining() < 50:
            break
        # Interpolate parent
        h_interp = interp_to(best_h, best_n, n_try)
        h_bin = to_binary(h_interp, n_try, rng)

        # SA
        t_alloc = min(55, max(20, remaining() - 15))
        h_sa, c5_sa, _ = sa_optimize(h_bin, n_try, time_limit=t_alloc,
                                      rng=rng, T0=2.0, cool=0.999997, tabu_len=30)
        # Greedy polish
        if remaining() > 20:
            h_g, c5_g = greedy_first_improve(h_sa, n_try, time_limit=min(12, remaining()-10),
                                              rng=rng, n_cand=150)
            if c5_g < c5_sa:
                h_sa = h_g
                c5_sa = c5_g

        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n_try

    # Phase 3: Multi-start on best n
    if remaining() > 60:
        for pt in range(5):
            if remaining() < 20:
                break
            h_pat = make_pattern(best_n, pt + 1, rng)
            t_alloc = min(30, max(10, remaining() - 10))
            h_sa, c5_sa, _ = sa_optimize(h_pat, best_n, time_limit=t_alloc,
                                          rng=rng, T0=4.0, cool=0.999995, tabu_len=20)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()

    # Phase 4: Intensive optimization on best
    if remaining() > 40:
        h_g, c5_g = greedy_first_improve(best_h, best_n, time_limit=min(25, remaining()-8),
                                          rng=rng, n_cand=200)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 5: One more SA round with lower T (exploitation)
    if remaining() > 35:
        h_sa, c5_sa, _ = sa_optimize(best_h, best_n, time_limit=min(25, remaining()-8),
                                      rng=rng, T0=1.0, cool=0.99999, tabu_len=50)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 6: SLSQP polish (continuous relaxation)
    if remaining() > 15:
        for beta in [1000, 5000, 20000]:
            if remaining() < 6:
                break
            h_cont = best_h.astype(float)
            h_new = slsqp_refine(h_cont, best_n, beta, maxiter=200)
            c5_new = compute_c5(h_new)
            # Also try rounding
            h_rounded = to_binary(h_new, best_n, rng)
            c5_rounded = compute_c5(h_rounded)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            if c5_rounded < best_c5:
                best_c5 = c5_rounded
                best_h = h_rounded.copy()

    # Phase 7: Final greedy
    if remaining() > 10:
        h_g, c5_g = greedy_first_improve(best_h, best_n, time_limit=min(8, remaining()-4),
                                          rng=rng, n_cand=300)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5:
        best_c5 = final_c5
    else:
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
