def compute_c5(h):

def project_feasible(h, n):

        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n

        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = project_feasible(h_new, n)
        h = h_new
        if t % 25 == 0:
            c5_cur = compute_c5(h)
            if c5_cur < c5_best - 1e-13:
                c5_best = c5_cur
                h_best = h.copy()
                patience = 0
            else:
                patience += 1
                if patience > restart_every // 25:
                    h = h_best.copy() + lr0 * 0.5 * rng.randn(n)
                    h = project_feasible(h, n)
                    m = np.zeros(n)
                    v = np.zeros(n)
                    patience = 0
    c5_end = compute_c5(h)
    if c5_end < c5_best:
        c5_best = c5_end
        h_best = h.copy()
    return h_best, c5_best
