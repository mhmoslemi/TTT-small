```python
def swap_delta(h, a, b, n):
    """Compute change in all correlations when h[a]:1->0, h[b]:0->1"""
    # For each offset o, compute the change
    # corr[o_index] where o_index = o + (n-1)
    # Valid i range for offset o: max(0,-o) to min(n-1, n-1-o)
    
    dcorr = np.zeros(2*n-1)
    
    # For h[a] going 1->0 (delta = -1):
    # As first factor (i+o = a): valid if max(0,-o) <= a-o <= min(n-1, n-1-o)
    #   i.e., max(o, -o+(0)) ... let me just check bounds
    # a-o >= 0 AND a-o <= n-1 AND a >= 0 AND a <= n-1 (already true)
    # a-o >= 0 => o <= a
    # a-o <= n-1 => o >= a-(n-1)
    # Also need: max(0,-o) <= a-o, which is automatic if a>=0 and o<=a
    # And: a-o <= min(n-1, n-1-o) => a-o <= n-1-o => a <= n-1 (true) and a-o <= n-1 (true if o >= a-n+1)
    
    # Simpler: a-o is valid index in [0, n-1] AND a-o is in the valid i range
    # Valid i range: i_lo = max(0, -o), i_hi = min(n-1, n-1-o)
    # a-o >= i_lo AND a-o <= i_hi
    
    # Let me just use direct index checks:
    # For offset o, position p = a (as the "h" value):
    #   p appears as h[i+o] if i = p-o is valid: 0<=p-o<n and p-o in [i_lo, i_hi]
    #   p appears as h[i] if i = p is valid: 0<=p<n and p in [i_lo, i_hi]
    
    # I'll iterate over offsets efficiently:
    # For position a:
    #   As first factor: valid offsets o where 0 <= a-o <= n-1 and max(0,-o) <= a-o <= min(n-1, n-1-o)
    #     Simplification: a-o >= 0 => o <= a; a-o <= n-1 => o >= a-n+1
    #     And we need the pair (a-o, a) to be valid: both in [0,n-1], which is satisfied.
    #     Additional constraint from i range: the pair is valid iff both indices are in [0,n-1].
    #     So valid o range for a as first factor: o in [a-(n-1), a] intersect [-(n-1), n-1]
    #   As second factor: valid offsets o where 0 <= a+o <= n-1 and a in [i_lo, i_hi]
    #     a+o >= 0 => o >= -a; a+o <= n-1 => o <= n-1-a
    #     a in [i_lo, i_hi]: a >= max(0,-o) and a <= min(n-1, n-1-o)
    #       a >= max(0,-o): if o>=0, always true; if o<0, a >= -o => o >= -a
    #       a <= min(n-1, n-1-o): if o>=0, a <= n-1-o => o <= n-1-a; if o<0, a <= n-1 (true)
    
    # This is getting complex. Let me just iterate over a limited range of o for each position.
    
    # For position a as first factor (h[a] is the "h" in h[i+o]*(1-h[i])):
    #   i = a-o, need 0 <= i < n and the pair (i, i+o) = (a-o, a) valid
    #   o ranges from a-(n-1) to a (since i = a-o in [0, n-1])
    #   But also need o in [-(n-1), n-1]
    o_lo = max(a - (n-1), -(n-1))
    o_hi = min(a, n-1)
    for o in range(o_lo, o_hi+1):
        i = a - o
        # Check validity: 0 <= i < n and 0 <= i+o < n (both satisfied by construction)
        # The pair contributes h[a]*(1-h[i]) to corr
        # Change: -1 * (1-h[i]) = -(1-h[i])
        dcorr[o + (n-1)] += -(1.0 - h[i])
    
    # For position a as second factor (h[a] is the "h" in (1-h[i]) part):
    #   i = a, need 0 <= i+o < n
    #   o ranges from -a to n-1-a
    o_lo2 = max(-a, -(n-1))
    o_hi2 = min(n-1-a, n-1)
    for o in range(o_lo2, o_hi2+1):
        j = a + o
        # Change: +1 * h[j] (since (1-h[a]) goes from 0 to 1, so h[j]*(1-h[a]) goes from 0 to h[j])
        # Wait: old h[a]=1, so (1-h[a])=0, contribution was h[j]*0=0
        # new h[a]=0, so (1-h[a])=1, contribution is h[j]*1=h[j]
        # Change: h[j] - 0 = h[j]
        dcorr[o + (n-1)] += h[j]
    
    # For position b as first factor (h[b] is the "h"):
    #   i = b-o, o from b-(n-1) to b
    o_lo3 = max(b - (n-1), -(n-1))
    o_hi3 = min(b, n-1)
    for o in range(o_lo3, o_hi3+1):
        i = b - o
        # old h[b]=0, contribution was 0
        # new h[b]=1, contribution is 1*(1-h[i]) = 1-h[i]
        dcorr[o + (n-1)] += (1.0 - h[i])
    
    # For position b as second factor:
    #   i = b, j = b+o, o from -b to n-1-b
    o_lo4 = max(-b, -(n-1))
    o_hi4 = min(n-1-b, n-1)
    for o in range(o_lo4, o_hi4+1):
        j = b + o
        # old h[b]=0, (1-h[b])=1, contribution was h[j]*1 = h[j]
        # new h[b]=1, (1-h[b])=0, contribution is 0
        dcorr[o + (n-1)] += -h[j]
    
    return dcorr
```

```python
def smooth_obj(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / _BETA) * dx_v


def smooth_grad(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    wc = w * dx_v
    n1 = n - 1
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-15:
            continue
        off = k - n1
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:ao + np_])
            g[ao:ao + np_] -= wk * x[:np_]
    return g

    return np.sum(x) - _TGT

    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=400):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.success or res.fun < smooth_obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

        g_t = np.zeros(n)
        g_b = np.zeros(n)
        for k, g in [(k_top, g_t), (k_bot, g_b)]:
            off = k - n1
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += (1.0 - h[:np_])
                g[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= h[:np_]

        d = g_b - g_t
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            d = -g_t
            dn = np.linalg.norm(d)
            if dn < 1e-14:
                break
        d /= dn

        improved = False
        for step in [0.01, 0.03, 0.05, 0.02, 0.08, 0.005, 0.1]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    # Adjust to have exactly n//2 ones
    n_ones = int(np.sum(h))
    n_target = n // 2
    if n_ones > n_target:
        ones = np.where(h == 1)[0]
        h[ones[n_target:]] = 0
    elif n_ones < n_target:
        zeros = np.where(h == 0)[0]
        h[zeros[:n_target - n_ones]] = 1
    return h


def make_random_balanced(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = make_random_balanced(parent_n, rng)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n

    # Phase 1: Quick SLSQP warmup on parent (10% budget)
    warmup_time = budget_s * 0.08
    if remaining() > warmup_time:
        for beta in [200, 1000, 5000]:
            if remaining() < warmup_time / 3:
                break
            h_new = slsqp_smooth(best_h, n, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 2: Binary swift search from parent (15% budget)
    if remaining() > budget_s * 0.1:
        h_bin = (best_h > 0.5).astype(float)
        sa_time = min(budget_s * 0.15, max(30, remaining() * 0.2))
        h_b, c5_b = binary_swift_search(h_bin, n, sa_time, rng, n_candidates=20)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 3: Binary search from Thue-Morse (10% budget)
    if remaining() > budget_s * 0.05:
        h_tm = make_thue_morse(n)
        tm_time = min(budget_s * 0.1, max(20, remaining() * 0.15))
        h_tm_b, c5_tm = binary_swift_search(h_tm, n, tm_time, rng, n_candidates=20)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm_b.copy()

    # Phase 4: Binary search from random (5% budget)
    if remaining() > budget_s * 0.03:
        h_rand = make_random_balanced(n, rng)
        rand_time = min(budget_s * 0.05, max(15, remaining() * 0.1))
        h_r_b, c5_r = binary_swift_search(h_rand, n, rand_time, rng, n_candidates=16)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r_b.copy()

    # Phase 5: Continuous refinement on best binary (25% budget)
    if remaining() > 20:
        h_ref = best_h.copy()
        # Progressive SLSQP
        for beta in [500, 2000, 8000, 20000, 50000]:
            if remaining() < 10:
                break
            h_ref = slsqp_smooth(h_ref, n, beta, maxiter=400)
            c5_ref = compute_c5(h_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()

        # Top-shifts equalization
        if remaining() > 10:
            h_eq, c5_eq = top_shifts_equalize(h_ref, n, top_k=5, max_iters=300)
            if c5_eq < best_c5:
                best_c5 = c5_eq
                best_h = h_eq.copy()
                h_ref = h_eq.copy()

    # Phase 6: Try higher resolution (15% budget)
    n_high_candidates = [n + 32, n + 64, n + 128, n + 192]
    n_high_candidates = [nn for nn in n_high_candidates if 200 <= nn <= 1024 and nn % 2 == 0]
    n_high_candidates = sorted(set(n_high_candidates))

    for n_t in n_high_candidates[:3]:
        if remaining() < 30:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)

        # Quick SLSQP
        for beta in [500, 5000]:
            if remaining() < 15:
                break
            h_feas = slsqp_smooth(h_feas, n_t, beta, maxiter=300)
        c5_hi = compute_c5(h_feas)

        # Binary search at this resolution
        if remaining() > 15:
            h_bin_hi = (h_feas > 0.5).astype(float)
            h_bin_opt, c5_bin = binary_swift_search(h_bin_hi, n_t,
                                                     min(20, int(remaining() * 0.3)),
                                                     rng, n_candidates=16)
            if c5_bin < c5_hi:
                h_feas = h_bin_opt
                c5_hi = c5_bin

        # Final SLSQP
        if remaining() > 10:
            h_feas = slsqp_smooth(h_feas, n_t, 10000, maxiter=300)
            c5_hi = compute_c5(h_feas)

        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_feas.copy()
            best_n = n_t

    # Phase 7: More intensive binary search on best (10% budget)
    if remaining() > 20:
        n_cur = best_n
        h_cur = (best_h > 0.5).astype(float)
        # Ensure balance
        ones_count = int(np.sum(h_cur))
        if ones_count != n_cur // 2:
            sort_idx = np.argsort(-best_h)
            h_cur = np.zeros(n_cur)
            h_cur[sort_idx[:n_cur // 2]] = 1.0

        bs_time = min(budget_s * 0.1, max(30, int(remaining() * 0.4)))
        h_b2, c5_b2 = binary_swift_search(h_cur, n_cur, bs_time, rng,
                                           n_candidates=24, reheat_every=800)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

    # Phase 8: Final continuous polish (10% budget)
    if remaining() > 10:
        n_final = best_n
        h_final = best_h.copy()
        for beta in [10000, 30000, 80000]:
            if remaining() < 5:
                break
            h_final = slsqp_smooth(h_final, n_final, beta, maxiter=500)
            c5_f = compute_c5(h_final)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_final.copy()

        if remaining() > 5:
            h_eq, c5_eq = top_shifts_equalize(best_h, best_n, top_k=6, max_iters=300)
            if c5_eq < best_c5:
                best_c5 = c5_eq
                best_h = h_eq.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        best_c5 = final_c5

    # Validate with evaluate_erdos_solution if available
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
```
