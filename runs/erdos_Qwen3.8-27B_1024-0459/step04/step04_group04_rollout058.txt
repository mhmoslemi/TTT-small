    stall = 0
    max_stall = 2000
    iters = 0

        if remaining() < 5:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Choose multi_move swaps
        n_swap = min(multi_move, len(ones), len(zeros))
        i_idx = ones[rng.choice(len(ones), n_swap, replace=False)]
        j_idx = zeros[rng.choice(len(zeros), n_swap, replace=False)]

        old_corr = corr.copy()
        for a, b in zip(i_idx, j_idx):
            # h[a]: 1→0, h[b]: 0→1
            # Δcorr[lag] = -(1-h[a+lag]) + h[a-lag] + (1-h[b+lag]) - h[b-lag]
            # where out-of-range terms are 0
            
            # Term 1: -(1-h[a+lag]) for lag in [-a, n-1-a]
            for lag in range(-a, n - a):
                k = lag + n - 1
                other = a + lag
                if 0 <= other < n:
                    corr[k] -= (1.0 - h[other])
            
            # Term 2: +h[a-lag] for lag in [a-(n-1), a]
            for lag in range(a - n + 1, a + 1):
                k = lag + n - 1
                other = a - lag
                if 0 <= other < n:
                    corr[k] += h[other]
            
            # Term 3: +(1-h[b+lag]) for lag in [-b, n-1-b]
            for lag in range(-b, n - b):
                k = lag + n - 1
                other = b + lag
                if 0 <= other < n:
                    corr[k] += (1.0 - h[other])
            
            # Term 4: -h[b-lag] for lag in [b-(n-1), b]
            for lag in range(b - n + 1, b + 1):
                k = lag + n - 1
                other = b - lag
                if 0 <= other < n:
                    corr[k] -= h[other]
        
        new_max = float(np.max(corr))
        delta = new_max - cur_max
        
        if delta < 0 or rng.rand() < np.exp(-max(0, delta) / T):
            cur_max = new_max
            # Apply swap to h
            for a, b in zip(i_idx, j_idx):
                h[a], h[b] = 0.0, 1.0
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            corr = old_corr
            stall += 1
        
        T *= alpha
        if stall > max_stall:
            T = max(T, 1.0)
            stall = 0
        iters += 1
    
    return best_h, best_max * dx_val

import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0
_N_G = 0
_BETA_G = 1.0
_TGT_G = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        perm = rng.permutation(len(idx))[:n_ones - target]
        h_bin[idx[perm]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        perm = rng.permutation(len(idx))[:target - n_ones]
        h_bin[idx[perm]] = 1
    return h_bin


def corr_delta_single(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    lags1 = np.arange(-i, n - i)
    ks1 = lags1 + n - 1
    others1 = i + lags1
    delta[ks1] -= (1.0 - h[others1])
    lags2 = np.arange(i - n + 1, i + 1)
    ks2 = lags2 + n - 1
    others2 = i - lags2
    delta[ks2] += h[others2]
    lags3 = np.arange(-j, n - j)
    ks3 = lags3 + n - 1
    others3 = j + lags3
    delta[ks3] += (1.0 - h[others3])
    lags4 = np.arange(j - n + 1, j + 1)
    ks4 = lags4 + n - 1
    others4 = j - lags4
    delta[ks4] -= h[others4]
    return delta


def binary_sa_fast(h, n, time_limit, seed, multi_move=1, T_init=4.0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        h = make_binary(h, n, rng)

    dx_val = 2.0 / n
    corr = compute_corr(h).astype(float)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()

    T = T_init
    alpha = 0.9997
    start = time.time()
    stall = 0
    max_stall = 5000

    while time.time() - start < time_limit and T > 0.005:
        if remaining() < 3:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        mm = min(multi_move, len(ones), len(zeros))
        if mm == 0:
            break
        i_list = ones[rng.choice(len(ones), mm, replace=False)]
        j_list = zeros[rng.choice(len(zeros), mm, replace=False)]

        delta = np.zeros(2 * n - 1)
        for a, b in zip(i_list, j_list):
            delta += corr_delta_single(h, int(a), int(b), n)

        new_corr = corr + delta
        new_max = float(np.max(new_corr))
        delta_obj = new_max - cur_max

        accept = delta_obj < 0 or (T > 0.005 and rng.rand() < np.exp(-delta_obj / T))

        if accept:
            corr = new_corr
            cur_max = new_max
            for a, b in zip(i_list, j_list):
                h[a] = 0.0
                h[b] = 1.0
            if new_max < best_max - 1e-15:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        T *= alpha
        if stall > max_stall:
            # Kick: randomly swap a few more pairs
            kick_n = min(3, len(ones), len(zeros))
            if kick_n > 0:
                ki = ones[rng.choice(len(ones), kick_n, replace=False)]
                kj = zeros[rng.choice(len(zeros), kick_n, replace=False)]
                kd = np.zeros(2 * n - 1)
                for a, b in zip(ki, kj):
                    kd += corr_delta_single(h, int(a), int(b), n)
                corr = corr + kd
                cur_max = float(np.max(corr))
                for a, b in zip(ki, kj):
                    h[a] = 0.0
                    h[b] = 1.0
            T = max(T, 1.5)
            stall = 0

    return best_h, best_max * dx_val


def smooth_obj(x):
    n = _N_G
    dx_val = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _BETA_G
    shifted = corr - m
    cl = np.clip(beta * shifted, -50, 50)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g


def smooth_grad(x):
    n = _N_G
    dx_val = 2.0 / n
    beta = _BETA_G
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -50, 50)
    ex = np.exp(cl)
    s = np.sum(ex)
    w = ex / s
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-12:
            g += w[k] * corr_grad_per_shift(x, n, k)
    return g * dx_val


def _sum_con(x):
    return np.sum(x) - _TGT_G


def _sum_con_jac(x):
    return np.ones(_N_G)


def slsqp_opt(h, n, beta, maxiter=300):
    global _N_G, _BETA_G, _TGT_G
    _N_G = n
    _TGT_G = n / 2.0
    _BETA_G = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def topk_grad(h, n, max_iters=300, top_k=5):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 4:
            break
        corr = compute_corr(h)
        m = float(np.max(corr))
        if m * (2.0 / n) >= best_c5 + 1e-7:
            break
        top_idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k in top_idx:
            g += corr_grad_per_shift(h, n, int(k))
        g /= len(top_idx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        improved = False
        for step in [0.05, 0.02, 0.01, 0.08, 0.005, 0.03]:
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-13:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            break
    return h, best_c5
