                    h_trial[i] = np.clip(h_best[i] + delta, 0.0, 1.0)
                    h_trial = project_feasible(h_trial, n)
                    c5_t = compute_c5(h_trial)
                    if c5_t < c5_best - 1e-13:
                        c5_best = c5_t
                        h_best = h_trial.copy()
                        any_imp = True
                        break
        if not any_imp and sweep > 3:
            break
    return h_best, c5_best


def adaptive_topk_subgrad(h, n, max_iter=300):
    """Subgradient with adaptive top_k that increases as we get closer to optimal."""
    c5_best = compute_c5(h)
    h_best = h.copy()
    mom = np.zeros(n)
    step = 0.12
    for it in range(max_iter):
        if remaining_time() < 4.0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_val = np.max(corr)
        # Adaptive top_k: start small, increase as correlations get more equal
        threshold = top_val - 0.02
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(corr)[::-1][:3]
        elif len(active) > 25:
            active = active[np.argsort(corr[active])[::-1][:25]]

        grad = np.zeros(n)
        for k_idx in active:
            grad += corr_gradient(h, int(k_idx))
        grad /= len(active)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad = grad / gn

        alpha = step
        improved = False
        for _ in range(12):
            mom = 0.8 * mom - alpha * grad
            h_trial = project_feasible(h + mom, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-13:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
            alpha *= 0.5
        if improved:
            step = min(alpha * 1.15, 0.35)
        else:
            step *= 0.65
            mom *= 0.5
            if step < 0.005:
                break
    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Phase 0: Get initial point
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = project_feasible(make_thue_morse(parent_n), parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("Thue-Morse: n={}, C5={:.10f}".format(parent_n, best_c5))

    # Phase 1: Quick optimization at parent resolution
    print("\n--- Phase 1: Quick opt at n={} ---".format(best_n))
    _G_N = best_n
    _G_TGT = best_n / 2.0
    _G_DX = 2.0 / best_n
    best_h, best_c5 = subgrad_momentum_v2(best_h, best_n, max_iter=200, step0=0.12)
    best_h, best_c5 = slsqp_polish(best_h, best_n, betas=[200, 1000, 5000], maxiter=200)
    best_h, best_c5 = active_set_refine(best_h, best_n, max_iter=80)
    print("  C5={:.10f}".format(best_c5))

    # Phase 2: Multi-resolution exploration
    print("\n--- Phase 2: Multi-resolution ---")
    n_candidates = [192, 224, 256, 288, 320, 384, 448, 512, 640]
    # Filter to those different from parent
    n_candidates = [n for n in n_candidates if n != parent_n]
    # Sort by distance from parent to try close ones first
    n_candidates.sort(key=lambda x: abs(x - parent_n))

    for n_try in n_candidates:
        if remaining_time() < 30.0:
            break
        h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        c5_h = compute_c5(h_interp)
        print("  n={}: interp C5={:.10f}".format(n_try, c5_h))

        # Subgradient
        h_interp, c5_h = subgrad_momentum_v2(h_interp, n_try, max_iter=150, step0=0.1)

        # SLSQP
        h_interp, c5_h = slsqp_polish(h_interp, n_try, betas=[300, 1500, 6000], maxiter=150)

        # Active set
        h_interp, c5_h = active_set_refine(h_interp, n_try, max_iter=50)

        print("  n={}: C5={:.10f}".format(n_try, c5_h))

        if c5_h < best_c5 - 1e-10:
            best_c5 = c5_h
            best_h = h_interp.copy()
            best_n = n_try
            print("  ** NEW BEST: n={}, C5={:.10f} **".format(best_n, best_c5))

    # Phase 3: Intensive optimization at best resolution
    print("\n--- Phase 3: Intensive at n={} ---".format(best_n))
    _G_N = best_n
    _G_TGT = best_n / 2.0
    _G_DX = 2.0 / best_n

    # Adaptive subgradient
    best_h, best_c5 = adaptive_topk_subgrad(best_h, best_n, max_iter=300)
    print("  After adaptive subgrad: C5={:.10f}".format(best_c5))

    # Top-k equalization
    best_h, best_c5 = topk_equalize(best_h, best_n, k=3, max_iter=150)
    best_h, best_c5 = topk_equalize(best_h, best_n, k=5, max_iter=150)
    best_h, best_c5 = topk_equalize(best_h, best_n, k=8, max_iter=100)
    print("  After topk equalize: C5={:.10f}".format(best_c5))

    # More SLSQP
    best_h, best_c5 = slsqp_polish(best_h, best_n, betas=[1000, 5000, 20000], maxiter=300)
    print("  After SLSQP: C5={:.10f}".format(best_c5))

    # Active set again
    best_h, best_c5 = active_set_refine(best_h, best_n, max_iter=100)
    print("  After active set: C5={:.10f}".format(best_c5))

    # Phase 4: Perturbation exploration
    print("\n--- Phase 4: Perturbation ---")
    best_h, best_c5 = do_perturb(best_h, best_n, rng, num_pert=20)
    print("  C5={:.10f}".format(best_c5))

    # Phase 5: Another round of intensive optimization
    if remaining_time() > 60.0:
        print("\n--- Phase 5: Round 2 intensive ---")
        best_h, best_c5 = adaptive_topk_subgrad(best_h, best_n, max_iter=250)
        best_h, best_c5 = slsqp_polish(best_h, best_n, betas=[2000, 10000, 30000], maxiter=300)
        best_h, best_c5 = active_set_refine(best_h, best_n, max_iter=100)
        best_h, best_c5 = topk_equalize(best_h, best_n, k=5, max_iter=100)
        print("  C5={:.10f}".format(best_c5))

    # Phase 6: Try a few more resolutions if time permits
    if remaining_time() > 60.0:
        print("\n--- Phase 6: More resolutions ---")
        # Try resolutions near the best
        nearby_ns = sorted(set([best_n - 32, best_n - 16, best_n + 16, best_n + 32,
                                 best_n - 64, best_n + 64]))
        nearby_ns = [n for n in nearby_ns if n >= 96]
        for n_try in nearby_ns[:4]:
            if remaining_time() < 25.0:
                break
            h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
            c5_h = compute_c5(h_interp)
            h_interp, c5_h = subgrad_momentum_v2(h_interp, n_try, max_iter=100, step0=0.08)
            h_interp, c5_h = slsqp_polish(h_interp, n_try, betas=[500, 3000], maxiter=150)
            h_interp, c5_h = active_set_refine(h_interp, n_try, max_iter=40)
            if c5_h < best_c5 - 1e-10:
                best_c5 = c5_h
                best_h = h_interp.copy()
                best_n = n_try
                print("  ** New best at n={}: C5={:.10f} **".format(n_try, best_c5))
            else:
                print("  n={}: C5={:.10f}".format(n_try, c5_h))

    # Phase 7: Coordinate descent
    if remaining_time() > 15.0:
        print("\n--- Phase 7: Coordinate descent ---")
        best_h, best_c5 = coordinate_descent_v2(best_h, best_n, max_sweeps=8, rng=rng)
        print("  C5={:.10f}".format(best_c5))

    # Phase 8: Final polish
    if remaining_time() > 8.0:
        print("\n--- Phase 8: Final polish ---")
        _G_N = best_n
        _G_TGT = best_n / 2.0
        _G_DX = 2.0 / best_n
        best_h, best_c5 = slsqp_polish(best_h, best_n, betas=[5000, 20000, 50000], maxiter=400)
        best_h, best_c5 = active_set_refine(best_h, best_n, max_iter=80)
        best_h, best_c5 = topk_equalize(best_h, best_n, k=3, max_iter=80)
        print("  C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
