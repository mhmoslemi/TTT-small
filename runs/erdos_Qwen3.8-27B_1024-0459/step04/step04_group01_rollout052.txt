        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 312
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Quick SLSQP refinement (30s)
    for beta in [200, 800, 3000]:
        if remaining() < 5:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Round to binary and do aggressive local search (300s)
    h_bin = round_to_binary(best_h, best_n)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    tl = min(150, max(10, remaining() * 0.35))
    h_bs, c5_bs = binary_local_search(h_bin, best_n, time_limit=tl, rng=rng)
    if c5_bs < best_c5:
        best_c5 = c5_bs
        best_h = h_bs.copy()

    # Phase 3: Simulated annealing on best binary (120s)
    if remaining() > 30:
        h_bin2 = round_to_binary(best_h, best_n)
        tl2 = min(100, max(10, remaining() * 0.2))
        h_sa, c5_sa = simulated_annealing_binary(h_bin2, best_n, time_limit=tl2, rng=rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 4: More binary local search with different seed (150s)
    if remaining() > 30:
        rng2 = np.random.RandomState(seed + 123)
        h_bin3 = round_to_binary(best_h, best_n)
        tl3 = min(120, max(10, remaining() * 0.2))
        h_bs2, c5_bs2 = binary_local_search(h_bin3, best_n, time_limit=tl3, rng=rng2)
        if c5_bs2 < best_c5:
            best_c5 = c5_bs2
            best_h = h_bs2.copy()

    # Phase 5: Multi-shift coordinate descent on continuous (120s)
    if remaining() > 20:
        h_cont = best_h.copy()
        tl4 = min(80, max(10, remaining() * 0.2))
        h_mc, c5_mc = multi_shift_coord_descent(h_cont, best_n, max_passes=300,
                                                 time_limit=tl4, rng=rng)
        if c5_mc < best_c5:
            best_c5 = c5_mc
            best_h = h_mc.copy()

    # Phase 6: Try different n values (200s)
    n_candidates = sorted(set([
        best_n - 8, best_n - 4, best_n + 4, best_n + 8, best_n + 16,
        best_n + 32, best_n + 64,
        256, 288, 320, 352, 384, 448, 512, 576, 640
    ]))
    n_candidates = [nn for nn in n_candidates
                    if 128 <= nn <= 800 and nn % 2 == 0 and nn != best_n
                    and remaining() - 10 > 0]

    for n_t in n_candidates:
        if remaining() < 40:
            break
        t_per_n = min(45, max(20, remaining() * 0.25))

        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)

        # SLSQP
        h_opt = h_feas
        for beta in [500, 2000]:
            if remaining() < 10:
                break
            h_opt = slsqp_smooth(h_opt, n_t, beta, maxiter=250)

        # Binary round + local search
        h_b = round_to_binary(h_opt, n_t)
        if remaining() > 20:
            h_b, _ = binary_local_search(h_b, n_t, time_limit=min(25,
                                                                   max(5, remaining() * 0.2)), rng=rng)
            c5_b = compute_c5(h_b)
        else:
            c5_b = compute_c5(h_b)

        c5_cont = compute_c5(h_opt)
        if c5_b < c5_cont:
            h_opt = h_b
            c5_opt = c5_b
        else:
            c5_opt = c5_cont

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 7: Final intensive refinement on best solution
    n = best_n
    h = best_h.copy()

    # SLSQP polish
    for beta in [3000, 10000, 30000]:
        if remaining() < 5:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final binary search
    if remaining() > 15:
        h_bin_final = round_to_binary(best_h, best_n)
        tl_final = min(40, max(5, remaining() * 0.3))
        h_bf, c5_bf = binary_local_search(h_bin_final, best_n, time_limit=tl_final, rng=rng)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

    # Final continuous polish
    if remaining() > 8:
        h_new = slsqp_smooth(best_h, best_n, 50000, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final multi-shift desc
    if remaining() > 8:
        h_mc, c5_mc = multi_shift_coord_descent(best_h, best_n, max_passes=100,
                                                 time_limit=min(15, remaining() * 0.4), rng=rng)
        if c5_mc < best_c5:
            best_c5 = c5_mc
            best_h = h_mc.copy()

    # Validate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
