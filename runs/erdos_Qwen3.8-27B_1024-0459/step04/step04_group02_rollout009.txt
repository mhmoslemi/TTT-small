Actually, let me reconsider the coordinate descent. The key issue is that for each coordinate i, the objective as a function of t = h[i] is:
max_k (base[k] + a[k]*t + b[k]*t^2) * dx

where b[k] is non-zero only for k corresponding to lag 0 (the self-correlation).

For lag != 0, it's linear in t. For lag 0, it's quadratic (t - t^2).

The minimum of this piecewise function over t ∈ [0,1] can be found by:
1. Identifying which k achieves the max for each t
2. The breakpoints are where two lines cross: base[k1] + a[k1]*t = base[k2] + a[k2]*t → t = (base[k2]-base[k1])/(a[k1]-a[k2])
3. At each interval, the piecewise function is a quadratic (if k=0 is active) or linear
4. Minimize over each interval

This is exact but O(n^2) to find all breakpoints. For n=640, that's manageable.

import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_N_WORK = 0
_BETA_WORK = 1000.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = np.argmin(h)
            else:
                idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def smooth_obj(x, n, beta, mu):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = exp_vals.sum()
    smooth_max = (m + np.log(wsum) / beta) * dx
    s = np.sum(x) - n / 2.0
    penalty = mu * s * s
    return smooth_max + penalty

def smooth_obj_grad(x, n, beta, mu):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = exp_vals.sum()
    weights = exp_vals / wsum

    grad = np.zeros(n)
    nc = len(corr)
    for k in range(nc):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset+npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset+npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff+npairs])
                grad[aoff:aoff+npairs] -= w * x[:npairs]
    grad *= dx
    s = np.sum(x) - n / 2.0
    grad += 2.0 * mu * s
    return grad

def lbfgs_opt(h, n, beta, mu, maxiter=200):
    def obj(x):
        return smooth_obj(x, n, beta, mu)
    def obj_grad(x):
        return smooth_obj_grad(x, n, beta, mu)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-12, 'ftol': 1e-18})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xq = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xq, xp, src)

def coord_descent(h, n, max_sweeps=50):
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best).astype(float)
    dx = 2.0 / n
    nc = 2 * n - 1

    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        interior = np.where((best > 1e-10) & (best < 1.0 - 1e-10))[0]
        if len(interior) == 0:
            break
        order = np.random.permutation(interior)

        for i in order:
            if remaining_time() < 1.5:
                break
            t0 = best[i]
            a = np.zeros(nc)
            b = np.zeros(nc)
            for off in range(-(n-1), n):
                k = n - 1 + off
                if off == 0:
                    a[k] += 1.0
                    b[k] += -1.0
                else:
                    if 0 <= i + off < n:
                        a[k] += (1.0 - best[i + off])
                    if 0 <= i - off < n:
                        a[k] += (-best[i - off])
            base = best_corr - a * t0 - b * t0 * t0

            def f_of_t(t):
                return np.max(base + a * t + b * t * t) * dx

            cands = [0.0, 1.0, t0, t0 + 0.15, t0 - 0.15, t0 + 0.3, t0 - 0.3,
                     max(0.0, t0 - 0.5), min(1.0, t0 + 0.5), 1.0 - t0]
            cands = [max(0.0, min(1.0, c)) for c in cands]
            cands.append(t0)

            best_t = t0
            best_c = best_c5
            for t in cands:
                if abs(t - t0) < 1e-12:
                    continue
                c = f_of_t(t)
                if c < best_c - 1e-15:
                    best_c = c
                    best_t = t

            if best_t != t0:
                hi = min(1.0, best_t + 0.25)
                lo = max(0.0, best_t - 0.25)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = f_of_t(c1)
                f2 = f_of_t(c2)
                for _ in range(60):
                    if f1 < f2:
                        hi = c2; c2 = c1; f2 = f1
                        c1 = hi - gr * (hi - lo); f1 = f_of_t(c1)
                    else:
                        lo = c1; c1 = c2; f1 = f2
                        c2 = lo + gr * (hi - lo); f2 = f_of_t(c2)
                tc = (lo + hi) / 2.0
                fc = f_of_t(tc)
                if fc < best_c:
                    best_c = fc
                    best_t = tc

            if best_c < best_c5 - 1e-14:
                best[i] = best_t
                best_corr += a * (best_t - t0) + b * (best_t * best_t - t0 * t0)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return best, best_c5

def subgrad_step(h, n, top_k, n_steps):
    best = h.copy()
    best_c5 = compute_c5(best)
    for it in range(n_steps):
        if remaining_time() < 3:
            break
        corr = compute_corr(best)
        max_c = corr.max()
        active = np.where(corr >= max_c - 0.05)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        if len(active) > top_k:
            top_idx = np.argsort(corr)[::-1][:top_k]
            active = top_idx
        grad = np.zeros(n)
        for k in active:
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                if npairs > 0:
                    grad[offset:offset+npairs] += (1.0 - best[:npairs])
                    grad[:npairs] -= best[offset:offset+npairs]
            else:
                ao = -offset
                npairs = n - ao
                if npairs > 0:
                    grad[:npairs] += (1.0 - best[ao:ao+npairs])
                    grad[ao:ao+npairs] -= best[:npairs]
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        found = False
        for step in [0.02, 0.01, 0.005, 0.04, 0.002, 0.008]:
            ht = best - step * grad
            ht = project_feasible(ht, n)
            ct = compute_c5(ht)
            if ct < best_c5 - 1e-15:
                best = ht
                best_c5 = ct
                found = True
                break
        if not found:
            break
    return best, best_c5
