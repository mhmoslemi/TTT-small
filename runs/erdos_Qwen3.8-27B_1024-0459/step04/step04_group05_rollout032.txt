def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    # Ensure exactly n/2 ones
    num_ones = int(np.sum(h))
    if num_ones != n // 2:
        # Adjust
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if num_ones > n // 2:
            excess = num_ones - n // 2
            for i in range(excess):
                h[ones[i]] = 0
        else:
            deficit = n // 2 - num_ones
            for i in range(deficit):
                h[zeros[i]] = 1
    return h

def make_golden_ratio(n):
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    num_ones = int(np.sum(h))
    if num_ones != n // 2:
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if num_ones > n // 2:

            for i in range(excess):

        h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

# ---------- helpers ----------

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _tm(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') & 1 else 0
    return h

def _golden(n):
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    idx = np.arange(n)
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if ((i * gr) % 1.0) < 0.5 else 0.0
    return h

def _fix_binary(h, n):
    h = h.copy()
    num = int(np.round(np.sum(h)))
    if num == n // 2:
        h[h < 0.5] = 0.0
        h[h >= 0.5] = 1.0
        return h
    sorted_idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[sorted_idx[:n // 2]] = 1.0
    return b

def _proj(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(m):
            h[m] += d / np.count_nonzero(m)
        else:
            h += d / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _interp(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xq = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xq, xp, src)

def _greedy_init(n, rng, seed_extra=0):
    """Greedy: start empty, add 1s choosing position that minimizes max corr increase."""
    h = np.zeros(n)

            break

            pass
        else:
            h[pos] = 0.0
        if int(np.sum(h)) >= target_ones:
            break

    return h

def _sa_binary(h_init, n, budget_fn, rng, t_start=0.08, t_end=0.001, max_iters=None):
    """Simulated annealing on binary vectors with pair flips."""
    h = h_init.copy()
    c5_cur = _c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    
    if max_iters is None:
        max_iters = 2000000
    
    it = 0
    stuck = 0
    total = max_iters
    
    while it < total and budget_fn() > 10:
        frac = min(1.0, it / total)
        temp = t_start * (t_end / t_start) ** frac
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        c5_new = _c5(h)
        
        if c5_new <= c5_cur:
            c5_cur = c5_new
            stuck = 0
            if c5_new < c5_best - 1e-15:
                c5_best = c5_new
                h_best = h.copy()
        else:
            delta = c5_new - c5_cur
            if delta < temp * 2 and rng.rand() < np.exp(-delta / temp):
                c5_cur = c5_new
                stuck += 1
            else:
                h[i] = 1
                h[j] = 0
                stuck += 1
        
        it += 1

                h[a], h[b] = h[b], h[a]
            c5_cur = _c5(h)
            stuck = 0
            temp = max(temp, t_start * 0.3)

def _local_search_binary(h_init, n, budget_fn, max_sweeps=200):
    """Best-improvement local search: try all pair flips, take the best."""
    h = h_init.copy()
    c5_cur = _c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    
    for sweep in range(max_sweeps):
        if budget_fn() < 5:
            break

        zeros = np.where(h == 0)[0]

        best_i = -1
        best_j = -1
        
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                c5_try = _c5(h)
                h[i] = 1
                h[j] = 0
                delta = c5_try - c5_cur
                if delta < best_delta:
                    best_delta = delta
                    best_i = i
                    best_j = j

            h[best_i] = 0
            h[best_j] = 1
            c5_cur += best_delta
            if c5_cur < c5_best - 1e-15:
                c5_best = c5_cur
                h_best = h.copy()
        else:
            break
    
    return h_best, c5_best

_G_N = 256
_G_BETA = 100.0
_G_TGT = 128.0

def _obj_slsqp(x):
    n = _G_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _G_BETA
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -400, 400))
    return (m + np.log(np.sum(exps)) / beta) * dx

def _con_slsqp(x):
    return np.sum(x) - _G_TGT

def _slsqp(h_init, n, budget_fn, beta_list=None):
    global _G_N, _G_BETA, _G_TGT
    if beta_list is None:
        beta_list = [50, 200, 1000, 5000]
    
    _G_N = n
    _G_TGT = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_slsqp, 'jac': lambda x: np.ones(n)}]
    
    h_cur = h_init.copy()
    c5_cur = _c5(h_cur)
    
    for beta in beta_list:
        if budget_fn() < 10:
            break
        _G_BETA = beta
        maxiter = min(500, max(30, int(budget_fn() * 0.15 / 0.5)))
        try:
            res = minimize(_obj_slsqp, h_cur, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
            h_new = _proj(res.x, n)
            c5_new = _c5(h_new)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, c5_cur
