import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def solve_1d_minmax(C, D2):

def coordinate_descent(h, n, max_sweeps=500, lam_lr=0.1, lam_init=0.0):
    """Coordinate descent with exact 1D subproblem solving."""
    h = h.copy().astype(float)
    lam = lam_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    n_lags = 2 * n - 1
    lags = np.arange(-(n - 1), n)
    
    for sweep in range(max_sweeps):

            break

            D = np.zeros(n_lags)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    n_lags = 2 * n - 1
    lags = np.arange(n_lags) - (n - 1)
    
    for sweep in range(max_sweeps):
        if remaining() < 5:
            break
        
        full = compute_corr(h)
        
        for i in range(n):
            hi = h[i]
            # For each lag l (index j in 0..2n-2, l = j-(n-1)):
            # valid1: 0 <= i-l < n
            # valid2: 0 <= i+l < n
            
            # Compute D and C adjustments vectorized
            # l = j - (n-1)
            # valid1: 0 <= i - (j-(n-1)) < n => n-1 <= j <= i+n-1 (and j in 0..2n-2)
            # valid2: 0 <= i + (j-(n-1)) < n => n-1-i <= j <= 2n-2-i (and j in 0..2n-2)
            
            v1_lo = max(0, n - 1)
            v1_hi = min(2 * n - 2, i + n - 1)
            v2_lo = max(0, n - 1 - i)
            v2_hi = min(2 * n - 2, 2 * n - 2 - i)
            
            C = full.copy()
            D = np.zeros(n_lags)
            
            if v1_lo <= v1_hi:
                idx1 = np.arange(v1_lo, v1_hi + 1)
                l1 = idx1 - (n - 1)
                tidx1 = i - l1  # = i - idx1 + n - 1
                D[idx1] += (1.0 - h[tidx1])
                C[idx1] -= hi * (1.0 - h[tidx1])
            
            if v2_lo <= v2_hi:
                idx2 = np.arange(v2_lo, v2_hi + 1)
                l2 = idx2 - (n - 1)
                tidx2 = i + l2  # = i + idx2 - n + 1
                D[idx2] -= h[tidx2]
                C[idx2] += h[tidx2] * hi
            
            D2 = D + lam
            t_new = solve_1d_minmax(C, D2)
            h[i] = t_new
        
        h = project_feasible(h, n)
        s = np.sum(h) - n / 2.0
        lam += lam_lr * s
        lam = np.clip(lam, -5.0, 5.0)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5


def sa_binary_improved(h, n, time_limit=60, seed=42):
    """Improved binary SA with adjacent swaps and block moves."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    n_ones = int(np.round(np.sum(h)))
    target = n // 2
    h_bin = (h > 0.5).astype(float)
    
    n_ones = int(np.sum(h_bin))
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1
    
    dx_val = 2.0 / n
    corr = compute_corr(h_bin)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h_bin.copy()
    
    T = 6.0
    alpha = 0.999
    start = time.time()
    stall = 0
    max_stall = 8000
    
    while time.time() - start < time_limit and T > 0.05:
        if remaining() < 8:
            break
        
        r = rng.rand()
        if r < 0.5:
            # Swap 1 and 0 (random positions)
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bin[i], h_bin[j] = 0.0, 1.0
            new_corr = compute_corr(h_bin)
            new_max = float(np.max(new_corr))
            delta = new_max - cur_max
            if delta < 0 or (T > 0.05 and rng.rand() < np.exp(-delta / T)):
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_h = h_bin.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                h_bin[i], h_bin[j] = 1.0, 0.0
                stall += 1
        elif r < 0.8:
            # Adjacent swap (move a 1 by 1 or 2 positions)
            ones = np.where(h_bin == 1)[0]
            if len(ones) == 0:
                break
            i = ones[rng.randint(len(ones))]
            move = rng.randint(1, 4) * rng.choice([-1, 1])
            j = i + move
            if 0 <= j < n and h_bin[j] == 0:
                h_bin[i], h_bin[j] = 0.0, 1.0
                new_corr = compute_corr(h_bin)
                new_max = float(np.max(new_corr))
                delta = new_max - cur_max
                if delta < 0 or (T > 0.05 and rng.rand() < np.exp(-delta / T)):
                    cur_max = new_max
                    if new_max < best_max:
                        best_max = new_max
                        best_h = h_bin.copy()
                        stall = 0
                    else:
                        stall += 1
                else:
                    h_bin[i], h_bin[j] = 1.0, 0.0
                    stall += 1
        else:
            # Block move: move a contiguous block of 1s
            ones = np.where(h_bin == 1)[0]
            if len(ones) == 0:
                break
            i = ones[rng.randint(len(ones))]
            blen = rng.randint(1, 4)
            if i + blen >= n:
                continue
            block = h_bin[i:i+blen]
            if np.any(block == 0):
                continue
            j = rng.randint(0, n - blen)
            if j == i:
                continue
            if np.any(h_bin[j:j+blen] == 1):
                continue
            h_bin[i:i+blen] = 0
            h_bin[j:j+blen] = 1
            new_corr = compute_corr(h_bin)
            new_max = float(np.max(new_corr))
            delta = new_max - cur_max
            if delta < 0 or (T > 0.05 and rng.rand() < np.exp(-delta / T)):
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_h = h_bin.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                h_bin[j:j+blen] = 0
                h_bin[i:i+blen] = 1
                stall += 1
        
        T *= alpha
        if stall > max_stall:
            T = max(T, 3.0)
            stall = 0
    
    return best_h, best_max * dx_val


def slsqp_refine(h, n, beta=5000, maxiter=300):
    """SLSQP with smoothed max objective."""
    global _N, _BETA, _TGT
    
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    
    def smooth_obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -50, 50)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)
    
    def smooth_grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -50, 50)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k in range(len(corr)):
            wk = w[k]
            if wk < 1e-14:
                continue
            off = k - (n - 1)
            if off >= 0:
                npairs = n - off
                g[off:off + npairs] += (1.0 - x[:npairs])
                g[:npairs] -= x[off:off + npairs]
            else:
                ao = -off
                npairs = n - ao
                g[:npairs] += (1.0 - x[ao:ao + npairs])
                g[ao:ao + npairs] -= x[:npairs]
            g *= wk  # This is wrong, need to accumulate
        # Redo gradient properly
        g = np.zeros(n)
        for k in range(len(corr)):
            wk = w[k]
            if wk < 1e-14:
                continue
            off = k - (n - 1)
            if off >= 0:
                npairs = n - off
                g[off:off + npairs] += wk * (1.0 - x[:npairs])
                g[:npairs] -= wk * x[off:off + npairs]
            else:
                ao = -off
                npairs = n - ao
                g[:npairs] += wk * (1.0 - x[ao:ao + npairs])
                g[ao:ao + npairs] -= wk * x[:npairs]
        return g * (2.0 / n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1
    return h_bin


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 360
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    n = best_n
    
    # Phase 1: Coordinate descent at parent resolution (main improvement)
    # Run multiple rounds with different lambda learning rates
    for lam_lr in [0.3, 0.1, 0.05, 0.02]:
        if remaining() < 30:
            break
        h_cd, c5_cd = coordinate_descent_fast(best_h, n, max_sweeps=300, lam_lr=lam_lr)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Phase 2: More coordinate descent with periodic perturbation
    if remaining() > 20:
        h_cd, c5_cd = coordinate_descent(best_h, n, max_sweeps=500, lam_lr=0.05)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Phase 3: SLSQP refinement
    if remaining() > 20:
        for beta in [2000, 10000, 50000]:
            if remaining() < 10:
                break
            h_new = slsqp_refine(best_h, n, beta=beta, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    # Phase 4: More coordinate descent after SLSQP
    if remaining() > 30:
        for lam_lr in [0.1, 0.05, 0.02, 0.01]:
            if remaining() < 15:
                break
            h_cd, c5_cd = coordinate_descent_fast(best_h, n, max_sweeps=400, lam_lr=lam_lr)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
    
    # Phase 5: Binary SA at parent and nearby resolutions
    n_list = sorted(set([n - 20, n - 10, n, n + 10, n + 20, n + 40]))
    n_list = [x for x in n_list if x % 2 == 0 and 200 <= x <= 800]
    
    for n_t in n_list:
        if remaining() < 25:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin = make_binary(h_interp, n_t, rng)
        
        t_per = min(40, max(10, remaining() / (len(n_list) + 1)))
        for sa_seed in [seed + n_t, seed + 1000 + n_t * 3]:
            if remaining() < 10:
                break
            h_sa, c5_sa = sa_binary_improved(h_bin, n_t, time_limit=max(5, t_per / 2), seed=sa_seed)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_t
    
    # Phase 6: Coordinate descent at best resolution
    if remaining() > 30:
        n = best_n
        for lam_lr in [0.2, 0.1, 0.05, 0.03, 0.01]:
            if remaining() < 10:
                break
            h_cd, c5_cd = coordinate_descent_fast(best_h, n, max_sweeps=500, lam_lr=lam_lr)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
    
    # Phase 7: Try higher resolutions
    for n_high in [n + 60, n + 120, n + 200]:
        if remaining() < 40 or n_high > 1200:
            break
        if n_high % 2 != 0:
            n_high += 1
        h_interp = interp_to(best_h, best_n, n_high)
        h_interp = project_feasible(h_interp, n_high)
        h_cd, c5_cd = coordinate_descent_fast(h_interp, n_high, max_sweeps=300, lam_lr=0.1)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n_high
    
    # Phase 8: Final intensive coordinate descent
    n = best_n
    if remaining() > 20:
        max_sw = int(min(2000, max(100, remaining() * 3)))
        for lam_lr in [0.05, 0.02, 0.01, 0.005]:
            if remaining() < 8:
                break
            h_cd, c5_cd = coordinate_descent_fast(best_h, n, max_sweeps=max_sw // 4, lam_lr=lam_lr)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
    
    # Phase 9: Final SLSQP polish
    if remaining() > 15:
        for beta in [20000, 100000]:
            if remaining() < 8:
                break
            h_new = slsqp_refine(best_h, best_n, beta=beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    # Phase 10: Final coordinate descent
    if remaining() > 10:
        h_cd, c5_cd = coordinate_descent_fast(best_h, best_n, max_sweeps=500, lam_lr=0.02)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
