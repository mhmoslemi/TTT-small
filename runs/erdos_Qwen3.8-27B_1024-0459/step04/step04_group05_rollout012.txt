def project_feasible(h, n):

    target = n / 2.0

def grad_corr(h, offset):
    """Gradient of f_offset w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    if offset >= 0:
        n_pairs = n - offset
        if n_pairs <= 0:
            return grad
        grad[:n_pairs] += (1.0 - h[offset:offset+n_pairs])
        grad[offset:] -= h[:n_pairs]
    else:
        abs_off = -offset
        n_pairs = n - abs_off
        if n_pairs <= 0:
            return grad
        grad[:n_pairs] -= h[abs_off:abs_off+n_pairs]
        grad[abs_off:] += (1.0 - h[:n_pairs])
    return grad

    # Phase 3: Flip search
    if budget_fn() > 20:
        print("\n--- Phase 3: Flip search ---")
        best_h, best_c5 = do_flip_search(best_h, best_n, budget_fn, rng, max_iters=4000)
        print("  C5={:.10f}".format(best_c5))

    # Phase 4: Pair swap
    if budget_fn() > 20:
        print("\n--- Phase 4: Pair swap ---")
        best_h, best_c5 = do_pair_swap(best_h, best_n, budget_fn, rng, max_iters=3000)
        print("  C5={:.10f}".format(best_c5))

    # Phase 5: High-beta SLSQP
    if budget_fn() > 30:
        print("\n--- Phase 5: High-beta SLSQP ---")
        best_h, best_c5 = do_slsqp_multi(best_h, best_n, budget_fn, [8000, 20000])
        print("  C5={:.10f}".format(best_c5))

    # Phase 6: More subgradient after SLSQP
    if budget_fn() > 20:
        print("\n--- Phase 6: Subgradient round 2 ---")
        best_h, best_c5 = do_subgrad_refined(best_h, best_n, budget_fn, max_iters=3000, top_k=7)
        print("  C5={:.10f}".format(best_c5))

    # Phase 7: Try alternative starts at same n
    if budget_fn() > 40:
        print("\n--- Phase 7: Alt starts ---")
        alt_starts = [make_thue_morse(best_n), make_golden_ratio(best_n), make_random_bin(rng, best_n)]
        for alt in alt_starts:
            if budget_fn() < 15:
                break
            c5_alt = compute_c5(alt)
            h_t, c5_t = do_slsqp_multi(alt, best_n, budget_fn, [500, 2000])
            h_t, c5_t = do_subgrad_refined(h_t, best_n, budget_fn, max_iters=1000, top_k=5)
            if c5_t < best_c5 - 1e-13:
                best_c5 = c5_t
                best_h = h_t
                print("  Alt improved: C5={:.10f}".format(best_c5))

    # Phase 8: Higher resolution
    if budget_fn() > 50:
        print("\n--- Phase 8: Higher resolution ---")
        for n_try in [384, 512]:
            if budget_fn() < 30:
                break
            h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
            c5_interp = compute_c5(h_interp)
            h_new, c5_new = do_slsqp_multi(h_interp, n_try, budget_fn, [300, 1500, 5000])
            h_new, c5_new = do_subgrad_refined(h_new, n_try, budget_fn, max_iters=1000, top_k=5)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new
                best_n = n_try
                print("  n={} improved: C5={:.10f}".format(n_try, best_c5))
            else:
                print("  n={} no improve: C5={:.10f}".format(n_try, c5_new))

    # Phase 9: Final polish at best resolution
    if budget_fn() > 20:
        print("\n--- Phase 9: Final polish ---")
        best_h, best_c5 = do_subgrad_refined(best_h, best_n, budget_fn, max_iters=3000, top_k=5)
        best_h, best_c5 = do_slsqp_multi(best_h, best_n, budget_fn, [15000, 50000])
        best_h, best_c5 = do_flip_search(best_h, best_n, budget_fn, rng, max_iters=2000)
        best_h, best_c5 = do_coord_descent(best_h, best_n, budget_fn, rng, max_sweeps=5)
        print("  Final polish C5={:.10f}".format(best_c5))

    # Phase 10: Ultra-high beta SLSQP if time remains
    if budget_fn() > 15:
        best_h, best_c5 = do_slsqp_multi(best_h, best_n, budget_fn, [100000])
        best_h, best_c5 = do_subgrad_refined(best_h, best_n, budget_fn, max_iters=1000, top_k=3)

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
