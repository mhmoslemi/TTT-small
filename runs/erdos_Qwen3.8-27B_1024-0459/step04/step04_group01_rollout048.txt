def bisect_min_max(slopes, intercepts, lo, hi):
    if lo >= hi:
        return lo
    for _ in range(40):
        mid = (lo + hi) / 2.0
        vals = slopes * mid + intercepts
        max_val = vals.max()
        active = vals >= max_val - 1e-12
        if slopes[active].max() < -1e-14:
            lo = mid
        elif slopes[active].min() > 1e-14:
            hi = mid
        else:
            break
    return (lo + hi) / 2.0

def bisect_min_max(slopes, intercepts, lo, hi):
    if lo >= hi:
        return lo
    for _ in range(45):
        mid = (lo + hi) / 2.0
        vals = slopes * mid + intercepts
        max_val = vals.max()
        active = vals >= max_val - 1e-12
        s_act = slopes[active]
        if s_act.size == 0:
            break
        if s_act.max() < -1e-14:
            lo = mid
        elif s_act.min() > 1e-14:
            hi = mid
        else:
            break
    return (lo + hi) / 2.0

               max(80, best_n - 8), max(80, best_n - 4),
               384, 448, 512, 576, 640, 704]:
        if 80 <= nn <= 900 and nn % 2 == 0 and nn != best_n and nn not in n_tries:
            n_tries.append(nn)
    n_tries.sort(key=lambda x: abs(x - best_n))
    n_tries = n_tries[:5]

    for n_t in n_tries:
        if remaining() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)

        h_opt = h_feas
        for beta in [500, 3000, 15000]:
            if remaining() < 10:
                break
            h_opt = slsqp_smooth(h_opt, n_t, beta, maxiter=250)

        if remaining() > 20:
            h_opt, _ = pair_coord_descent(h_opt, n_t, max_passes=300,
                                          time_limit=min(35, remaining() - 10), rng=rng)

        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 8: Final intensive refinement on best solution
    n = best_n
    h = best_h.copy()

    # Final SLSQP with very high beta
    for beta in [30000, 80000, 150000]:
        if remaining() < 6:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final coordinate descent (long)
    if remaining() > 15:
        h_pc, c5_pc = pair_coord_descent(h, n, max_passes=600,
                                         time_limit=min(60, max(5, remaining() - 12)), rng=rng)
        if c5_pc < best_c5:
            best_c5 = c5_pc
            best_h = h_pc.copy()

    # Final SLSQP polish
    if remaining() > 8:
        h_new = slsqp_smooth(best_h, best_n, 100000, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final active set
    if remaining() > 8:
        h_eq, c5_eq = active_set_equalize(best_h, best_n, max_iters=200, top_k=5)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Final micro coordinate descent
    if remaining() > 10:
        h_pc, c5_pc = pair_coord_descent(best_h, best_n, max_passes=400,
                                         time_limit=min(25, max(3, remaining() - 6)), rng=rng)
        if c5_pc < best_c5:
            best_c5 = c5_pc
            best_h = h_pc.copy()

    # Validate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
