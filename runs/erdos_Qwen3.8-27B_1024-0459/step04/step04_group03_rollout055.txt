import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_N_G = 0
_BETA_G = 1.0
_LAGS_G = None
_DX_G = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        free = (h > 0) & (h < 1.0)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
        if abs(np.sum(h) - target) > 1e-11:
            idx = int(np.argmin(np.abs(h - 0.5)))
            h[idx] = np.clip(h[idx] + (target - np.sum(h)), 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)

def _sm_obj(x):
    n = _N_G
    beta = _BETA_G
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def _sm_eq(x):
    return np.sum(x) - _N_G / 2.0

    return x[_N_G]

    dx = _DX_G
    h = x[:n]
    t = x[n]
    corr = np.correlate(h, 1.0 - h, mode='full')

def _mm_eq(x):

def slsqp_smooth(h, n, beta, maxiter=300):

    _N_G = n
    _BETA_G = beta
    bounds = [(0.0, 1.0)] * n

    try:

                      bounds=bounds, constraints=cons,

        if res.x is not None:

            active = np.where(corr >= threshold)[0]

def coordinate_descent(h_init, n, max_sweeps=10, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h = h_init.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        order = rng.permutation(n)
        for i in order:
            if remaining_time() < 1:
                break
            corr = compute_corr(h)
            n_full = 2 * n - 1
            m_offsets = np.arange(n_full) - (n - 1)
            i_pm = i + m_offsets
            i_mm = i - m_offsets
            
            a = np.ones(n_full)
            vp = (i_pm >= 0) & (i_pm < n)
            vm = (i_mm >= 0) & (i_mm < n)
            a[vp] -= h[i_pm[vp]]
            a[vm] -= h[i_mm[vm]]
            
            b = corr - a * h[i]
            
            # Ternary search for min of max(a*t + b) over t in [0,1]
            lo_t, hi_t = 0.0, 1.0
            for _ in range(40):
                t1 = lo_t + (hi_t - lo_t) / 3.0
                t2 = hi_t - (hi_t - lo_t) / 3.0
                f1 = np.max(a * t1 + b)
                f2 = np.max(a * t2 + b)
                if f1 < f2:
                    hi_t = t2
                else:
                    lo_t = t1
            h[i] = (lo_t + hi_t) / 2.0
        
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
