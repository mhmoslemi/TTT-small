import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_N_G = 0
_BETA_G = 1.0
_DX_G = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-14:
            for j in range(n):
                if h[j] > 1e-8 and h[j] < 1.0 - 1e-8:
                    h[j] = np.clip(h[j] + diff2, 0.0, 1.0)
                    break
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def _sm_obj(x):
    n = _N_G
    beta = _BETA_G
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def _sm_eq(x):
    return np.sum(x) - _N_G / 2.0

def slsqp_smooth(h, n, beta, maxiter=300, ftol=1e-16):
    global _N_G, _BETA_G
    _N_G = n
    _BETA_G = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sm_eq}]
    try:
        res = minimize(_sm_obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def _grad_weighted(h, weights, n):
    """Compute gradient of weighted sum of correlation components."""
    dx = 2.0 / n
    grad = np.zeros(n)
    max_w = np.max(weights)
    if max_w < 1e-15:
        return grad
    significant = weights > max_w * 0.005
    active = np.where(significant)[0]
    for k in active:
        w = weights[int(k)]
        lag = int(k) - (n - 1)
        if lag >= 0:
            m1 = np.arange(lag, n)
            m2 = np.arange(0, n - lag)
            if len(m1) > 0:
                grad[m1] += w * (1.0 - h[m1 - lag])
            if len(m2) > 0:
                grad[m2] += w * (-h[m2 + lag])
        else:
            la = -lag
            m1 = np.arange(0, n - la)
            m2 = np.arange(la, n)
            if len(m1) > 0:
                grad[m1] += w * (1.0 - h[m1 + la])
            if len(m2) > 0:
                grad[m2] += w * (-h[m2 - la])
    return grad * dx

def subgradient_minimax(h_init, n, max_iter=800, lr_init=0.005):
    """Subgradient descent with line search for minimax."""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if remaining_time() < 5:
            break
        if it % 200 == 199 and compute_c5(h) >= best_c5 + 0.002:
            h = best_h.copy()
        
        corr = compute_corr(h)
        m = np.max(corr)
        c5 = m * (2.0 / n)
        
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
        
        beta = 200.0
        shifted = beta * (corr - m)
        weights = np.exp(shifted)
        weights /= np.sum(weights)
        
        grad = _grad_weighted(h, weights, n)
        grad -= np.mean(grad)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        lr = lr_init / (1.0 + 0.005 * it)
        direction = -grad / gnorm
        
        # Line search
        best_step = 0.0
        best_val = c5
        for step_factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            step = lr * step_factor
            h_trial = h + step * direction
            h_trial = project_feasible(h_trial, n)
            val = compute_c5(h_trial)
            if val < best_val:
                best_val = val
                best_step = step
        
        h = h + best_step * direction
        h = project_feasible(h, n)
    
    return best_h, best_c5

        h_sg, c5_sg = subgradient_minimax(best_h, best_n, max_iter=500, lr_init=0.002)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
    
    # Phase 8: Final SLSQP with very high beta
    if remaining_time() > 20:
        for beta in [30000, 80000]:
            if remaining_time() < 10:
                break
            h_new = slsqp_smooth(best_h, best_n, beta, maxiter=250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    # Phase 9: Final coordinate descent
    if remaining_time() > 10:
        h_cd, c5_cd = coordinate_descent(best_h, best_n, max_sweeps=6)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Phase 10: Ultimate polish
    if remaining_time() > 8:
        h_new = slsqp_smooth(best_h, best_n, 100000, maxiter=150)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
