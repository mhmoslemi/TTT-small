- At each C, check: does there exist h with 0≤h≤1, sum(h)=n/2, and c_k ≤ C*n/2 for all k?
- This is a system of bilinear inequalities, which is non-convex

So bisection doesn't directly work.

**Alternative:Alternating direction / Penalty method:**
- Minimize Σ_k max(0, c_k - t)² + penalty * (sum(h) - n/2)²
- Use coordinate descent on h (fixing t) and on t (fixing h)

2. **Coordinate descent** for local refinement
3. **Multi-resolution** (start at low res, refine at high res)
4. **Binary projection** periodically

    # corr[n-1+m] = sum_i h[i+m]*(1-h[i]) with valid indices

        g = compute_gradient_corr(h, n, m)
        grad += w[idx] * g
    return grad * dx, np.max(corr) * dx

            h[:] += diff / n

def grad_lag(h, n, m):
    grad = np.zeros(n)
    if m >= 0:
        if m < n:
            pad = n - m
            grad[m:m+pad] += (1.0 - h[:pad])
            grad[:pad] += -h[m:m+pad]
    else:
        k = -m
        if k < n:
            pad = n - k
            grad[:pad] += (1.0 - h[k:k+pad])
            grad[k:k+pad] += -h[:pad]
    return grad

def grad_smooth(h, n, beta):
    corr = compute_corr(h)
    shifted = beta * (corr - np.max(corr))
    w = np.exp(shifted)
    ws = np.sum(w)
    if ws < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    thresh = 1e-6

    dx = 2.0 / n
    for idx in active:
        m = idx - (n - 1)
        grad += w[idx] * grad_lag(h, n, m)
    return grad * dx
