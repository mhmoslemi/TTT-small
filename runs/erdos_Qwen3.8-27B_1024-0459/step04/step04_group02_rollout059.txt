            else:

    weights = exp_vals / wsum

            if np.any(mask_second_neg):
                idx_sn = offsets_pos[mask_second_neg]
                coeff[:n - 1][mask_second_neg] -= h[i - idx_sn]

            base = corr - coeff * old_t - quad * old_t * old_t

            # Try simple candidates first
            t_new = old_t
            c5_local = best_c5
            candidates = [0.0, 1.0, 1.0 - old_t, 0.5]
            best_candidate_t = old_t
            best_candidate_val = np.max(base + coeff * old_t + quad * old_t * old_t) * dx

            for t_cand in candidates:
                if abs(t_cand - old_t) < 1e-10:
                    continue
                vals = base + coeff * t_cand + quad * t_cand * t_cand
                v = np.max(vals) * dx
                if v < best_candidate_val - 1e-15:
                    best_candidate_val = v
                    best_candidate_t = t_cand

            # Golden section in a neighborhood
            lo = max(0.0, old_t - 0.4)
            hi = min(1.0, old_t + 0.4)
            if hi - lo > 1e-8:
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                vals1 = base + coeff * c1 + quad * c1 * c1
                f1 = np.max(vals1)
                vals2 = base + coeff * c2 + quad * c2 * c2
                f2 = np.max(vals2)
                for _ in range(35):
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        vals1 = base + coeff * c1 + quad * c1 * c1
                        f1 = np.max(vals1)
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        vals2 = base + coeff * c2 + quad * c2 * c2
                        f2 = np.max(vals2)
                t_gs = (lo + hi) / 2.0
                vals_gs = base + coeff * t_gs + quad * t_gs * t_gs
                f_gs = np.max(vals_gs) * dx
                if f_gs < best_candidate_val - 1e-15:
                    best_candidate_val = f_gs
                    best_candidate_t = t_gs

            if abs(best_candidate_t - old_t) > 1e-10:
                h[i] = best_candidate_t
                delta = coeff * (best_candidate_t - old_t) + quad * (best_candidate_t * best_candidate_t - old_t * old_t)
                corr = corr + delta
                new_c5 = np.max(corr) * dx
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    improved_this_sweep += 1

        if improved_this_sweep == 0:
            break
        if sweep % 5 == 4:
            h = project_feasible(h, n)
            corr = compute_corr(h)

    h = project_feasible(h, n)
    corr_final = compute_corr(h)
    final_c5 = np.max(corr_final) * dx
    return h, final_c5

def subgradient_h(h, n, n_active):
    corr = compute_corr(h)
    max_corr = np.max(corr)
    active = np.where(corr >= max_corr - 0.5)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        active = active[np.argsort(corr[active])[::-1][:n_active]]
    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return None
    return grad / gnorm

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    # Load initial values
    n_target = 1280
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
    except Exception:
        parent_n = 256
        parent_h = np.zeros(parent_n)
        parent_h[rng.choice(parent_n, parent_n // 2, replace=False)] = 1.0
        parent_h = project_feasible(parent_h, parent_n)

    # Interpolate to target resolution
    h = interp_to(parent_h, parent_n, n_target)
    h = project_feasible(h, n_target)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    best_n = n_target

    # Phase 1: SLSQP at moderate beta
    for beta in [200, 500, 1500]:
        if remaining_time() < 20:
            break
        h_new = slsqp_refine(h, n_target, beta, 300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Primary coordinate descent
    if remaining_time() > 30:
        n_sweeps = min(200, int(remaining_time() * 0.4 / 0.5))
        n_sweeps = max(n_sweeps, 30)
        h_new, c5_new = coord_descent(h, n_target, n_sweeps, rng)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: SLSQP at higher beta
    for beta in [5000, 20000]:
        if remaining_time() < 15:
            break
        h_new = slsqp_refine(h, n_target, beta, 400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: More coordinate descent
    if remaining_time() > 20:
        n_sweeps = min(150, int(remaining_time() * 0.35 / 0.5))
        n_sweeps = max(n_sweeps, 20)
        h_new, c5_new = coord_descent(h, n_target, n_sweeps, rng)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Subgradient with small steps
    if remaining_time() > 20:
        h = best_h.copy()
        for it in range(500):
            if remaining_time() < 5:
                break
            grad = subgradient_h(h, n_target, 8)
            if grad is None:
                break
            base_step = 0.01 / (1.0 + it * 0.01)
            improved = False
            for sm in [1.0, 0.5, 2.0, 0.25]:
                step = base_step * sm
                h_trial = h - step * grad
                h_trial = project_feasible(h_trial, n_target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-16:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h = h_trial.copy()
                    improved = True
                    break
            if not improved:
                if it > 100:
                    break

    # Phase 6: Final SLSQP polish
    for beta in [50000, 100000]:
        if remaining_time() < 8:
            break
        h_new = slsqp_refine(best_h, n_target, beta, 300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 7: Final coordinate descent
    if remaining_time() > 10:
        n_sweeps = min(80, int(remaining_time() * 0.8 / 0.5))
        h_new, c5_new = coord_descent(best_h, n_target, n_sweeps, rng)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 8: Try higher resolution if time allows
    if remaining_time() > 120:
        n_high = 2560
        h_high = interp_to(best_h, n_target, n_high)
        h_high = project_feasible(h_high, n_high)
        c5_high = compute_c5(h_high)
        if c5_high < 0.45:
            for beta in [500, 2000]:
                if remaining_time() < 30:
                    break
                h_high = slsqp_refine(h_high, n_high, beta, 300)
                c5_high = compute_c5(h_high)
            if remaining_time() > 30:
                n_sw = min(50, int(remaining_time() * 0.6 / 1.0))
                n_sw = max(n_sw, 10)
                h_high2, c5_high2 = coord_descent(h_high, n_high, n_sw, rng)
                if c5_high2 < c5_high:
                    h_high = h_high2
                    c5_high = c5_high2
            if c5_high < best_c5:
                best_c5 = c5_high
                best_h = h_high.copy()
                best_n = n_high

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
