import numpy as np
from scipy.optimize import minimize
import time

_G_START = 0.0
_G_BUDGET = 1300.0

def time_left():
    return max(0.0, _G_BUDGET - (time.time() - _G_START))

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    A = np.zeros(N)
    B = np.zeros(N)
    A[:n] = h
    B[:n] = 1.0 - h
    C = np.real(np.fft.ifft(np.fft.fft(A) * np.conj(np.fft.fft(B))))
    full = np.empty(2 * n - 1)
    full[:n] = C[n-1::-1]
    full[n:] = C[N-1:N-n:-1]
    return float(np.max(full)) * dx, full * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Find the element with most room
        if diff > 0:
            rooms = 1.0 - h
        else:
            rooms = h.copy()
        # Distribute proportionally to room
        total_room = np.sum(rooms)
        if total_room > 0:
            h += diff * rooms / total_room
        else:
            idx = int(np.argmax(rooms))
            h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def obj_lse(h, beta, top_k=None):
    """Log-sum-exp objective focusing on top_k correlations."""
    n = len(h)
    dx = 2.0 / n
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    A = np.zeros(N)
    B = np.zeros(N)
    A[:n] = h
    B[:n] = 1.0 - h
    C = np.real(np.fft.ifft(np.fft.fft(A) * np.conj(np.fft.fft(B))))
    full = np.empty(2 * n - 1)
    full[:n] = C[n-1::-1]
    full[n:] = C[N-1:N-n:-1]
    full *= dx
    
    if top_k is not None and top_k < len(full):
        idx = np.argpartition(full, -top_k)[-top_k:]
        vals = full[idx]
    else:
        vals = full
    
    m = np.max(vals)
    shifted = vals - m
    exps = np.exp(np.clip(beta * shifted, -500, 0))
    return m + np.log(np.sum(exps)) / beta

def make_binary_pattern(n, seed, pattern_type):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    if pattern_type == 'thue_morse':
        for i in range(n):
            h[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    elif pattern_type == 'golden':
        gr = (1.0 + np.sqrt(5.0)) / 2.0
        for i in range(n):
            h[i] = 1.0 if (i * gr) % 1.0 < 0.5 else 0.0
    elif pattern_type == 'block':
        # Alternating blocks
        block = max(1, n // 10)
        for i in range(n):
            h[i] = 1.0 if (i // block) % 2 == 0 else 0.0
    else:
        # Random binary
        idx = rng.permutation(n)
        h[idx[:n//2]] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1300, **kwargs):
