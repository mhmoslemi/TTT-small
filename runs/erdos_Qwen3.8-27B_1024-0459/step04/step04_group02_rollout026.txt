import numpy as np
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(interior)
        if cnt > 0:
            h[interior] += diff / cnt
        else:
            idx = np.argmin(h) if diff > 0 else np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        # Spread the remaining difference
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = 0
        v = i
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return project_feasible(h, n)

def make_quadratic_residue(n):
    """Create a sequence based on quadratic residues near size n."""
    # Find a prime p close to n
    p = n
    while p < 4 * n:
        if is_prime(p):
            break
        p += 1
    # Create QR sequence of length p
    qr = np.zeros(p)
    for i in range(1, p):
        if pow(i, (p-1)//2, p) == 1:
            qr[i] = 1.0
    # Pad or trim to length n
    if p >= n:
        h = qr[:n].copy()
    else:
        h = np.zeros(n)
        h[:p] = qr
        # Fill the rest to make sum = n/2
    return project_feasible(h, n)

def is_prime(p):
    if p < 2:
        return False
    if p < 4:
        return True
    if p % 2 == 0 or p % 3 == 0:
        return False
    i = 5
    while i * i <= p:
        if p % i == 0 or p % (i + 2) == 0:
            return False
        i += 6
    return True

def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

    vals = corr + (t - t0) * coeff
    vals = vals.copy()  # Avoid modifying original
    vals[n - 1] = self_base + t - t * t
    return np.max(vals) * dx
