        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps.copy()

    # Phase 2: SLSQP with increasing beta at parent resolution
    for beta in [200, 500, 1000, 3000, 8000]:
        if remaining() < 15:
            break
        h_new = slsqp_optimize(best_h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: Peak shave again after SLSQP
    if remaining() > 30:
        h_ps, c5_ps = peak_shave(best_h, n, max_iters=3000, top_k=8)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps.copy()

    # Phase 4: Try nearby resolutions
    for n_try in [n - 2, n + 2, n + 4, n - 4, n + 6]:
        if remaining() < 40:
            break
        if n_try < 100 or n_try > 800 or n_try % 2 != 0:
            continue
        h_init = interp_to(best_h, best_n, n_try)
        h_init = project_feasible(h_init, n_try)
        for beta in [1000, 5000]:
            if remaining() < 15:
                break
            h_new = slsqp_optimize(h_init, n_try, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_try
                h_init = h_new.copy()

    # Phase 5: More SLSQP at best resolution with higher beta
    n_best = best_n
    for beta in [15000, 30000, 60000]:
        if remaining() < 15:
            break
        h_new = slsqp_optimize(best_h, n_best, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 6: Peak shaving intensive
    if remaining() > 30:
        h_ps, c5_ps = peak_shave(best_h, n_best, max_iters=5000, top_k=6)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps.copy()

    # Phase 7: SLSQP + peak shave alternation
    for rnd in range(4):
        if remaining() < 15:
            break
        beta = 20000 * (rnd + 1)
        h_new = slsqp_optimize(best_h, n_best, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if remaining() < 10:
            break
        h_ps, c5_ps = peak_shave(best_h, n_best, max_iters=2000, top_k=8)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps.copy()

    # Phase 8: Binary targeted SA
    if remaining() > 50:
        h_bin = (best_h > 0.5).astype(float)
        target_cnt = n_best // 2
        n_ones = int(np.sum(h_bin))
        if n_ones > target_cnt:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[:n_ones - target_cnt]] = 0
        elif n_ones < target_cnt:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[:target_cnt - n_ones]] = 1
        t_sa = min(100, max(10, remaining() * 0.3))
        h_b, c5_b = binary_targeted_sa(h_bin, n_best, time_limit=t_sa, seed=seed)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 9: Final polish - alternate SLSQP and peak shave
    for beta in [50000, 100000, 200000]:
        if remaining() < 8:
            break
        h_new = slsqp_optimize(best_h, n_best, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    if remaining() > 10:
        h_ps, c5_ps = peak_shave(best_h, n_best, max_iters=3000, top_k=5)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps.copy()

    # Final validation
    best_h = project_feasible(best_h, n_best)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, n_best)
