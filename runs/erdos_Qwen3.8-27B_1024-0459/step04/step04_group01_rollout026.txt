    g = np.zeros(n)

def pair_coord_descent(h, n, max_passes=300, time_limit=60, rng=None):
    """Pair coordinate descent for continuous h."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    start = time.time()
    best_c5 = compute_c5(h)
    corr = compute_corr(h)
    n_corr = 2 * n - 1

    def corr_slope_at_i(h, i, n):
        nc = 2 * n - 1
        slopes = np.zeros(nc)
        for idx in range(nc):
            d = idx - (n - 1)
            s = 0.0
            if d >= 0:
                if d <= i <= n - 1 and i - d >= 0:
                    s += (1.0 - h[i - d])
                if 0 <= i <= n - 1 - d:
                    s -= h[i + d]
            else:
                m = -d
                if 0 <= i <= n - 1 - m:
                    s += (1.0 - h[i + m])
                if i >= m:
                    s -= h[i - m]
            slopes[idx] = s
        return slopes

    def coord_opt_1d(slopes, intercepts, lo, hi):
        if lo >= hi - 1e-14:
            return lo
        for _ in range(40):
            mid = (lo + hi) / 2.0
            vals = slopes * mid + intercepts
            max_val = vals.max()
            active = vals >= max_val - 1e-13
            if slopes[active].max() < -1e-14:
                lo = mid
            elif slopes[active].min() > 1e-14:
                hi = mid
            else:
                break
        return (lo + hi) / 2.0

    for pass_num in range(max_passes):
        if time.time() - start > time_limit:
            break
        if remaining() < 10:
            break
        order = np.arange(n)
        for idx in range(n):
            i = int(order[idx])
            j = (i + n // 2) % n
            if i == j:
                j = (i + 1) % n
            s = h[i] + h[j]
            lo_t = max(0.0, s - 1.0)
            hi_t = min(1.0, s)
            if lo_t >= hi_t - 1e-14:
                continue
            sl_i = corr_slope_at_i(h, i, n)
            sl_j = corr_slope_at_i(h, j, n)
            diff_sl = sl_i - sl_j
            intercepts = corr - h[i] * diff_sl
            t_opt = coord_opt_1d(diff_sl, intercepts, lo_t, hi_t)
            if abs(t_opt - h[i]) < 1e-14:
                continue
            delta = t_opt - h[i]
            corr += delta * diff_sl
            h[i] = t_opt
            h[j] = s - t_opt
        h = project_feasible(h, n)
        corr = compute_corr(h)
        c5_new = float(np.max(corr) * 2.0 / n)
        if c5_new < best_c5:
            best_c5 = c5_new

    return h, best_c5


def subgradient_momentum(h, n, max_iters=5000, time_limit=60, top_k=5):

    best_c5 = compute_c5(h)
    momentum_vec = np.zeros(n)
    beta_mom = 0.8
    step_size = 0.01

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining() < 10:
            break
        corr = compute_corr(h)
        sorted_idx = np.argsort(-corr)
        top = sorted_idx[:top_k]
        weights = np.exp((corr[top] - corr[top[0]]) * 100.0)
        weights /= weights.sum()
        grad = np.zeros(n)
        for w, k in zip(weights, top):
            grad += w * corr_grad_per_shift(h, n, int(k))
        momentum_vec = beta_mom * momentum_vec + grad
        h_trial = h - step_size * momentum_vec
        h = project_feasible(h_trial, n)
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            step_size = min(step_size * 1.01, 0.05)
        else:
            step_size *= 0.97
            if step_size < 0.001:
                step_size = 0.001

    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 312
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP on continuous parent
    for beta in [200, 800, 3000, 10000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Directed binary search (main workhorse)
    if remaining() > 30:
        h_bin = (best_h > 0.5).astype(float)
        h_b, c5_b = directed_binary_search(h_bin, n, time_limit=min(120, remaining() - 20), rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 3: Multi-lag binary search
    if remaining() > 30:
        h_b2, c5_b2 = multi_lag_binary_search(best_h, n, time_limit=min(80, remaining() - 20), rng=rng)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

    # Phase 4: SLSQP polish after binary search
    for beta in [5000, 15000, 40000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(best_h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 5: More binary search after continuous refinement
    if remaining() > 30:
        h_bin2 = (best_h > 0.5).astype(float)
        h_b3, c5_b3 = directed_binary_search(h_bin2, n, time_limit=min(100, remaining() - 20), rng=rng)
        if c5_b3 < best_c5:
            best_c5 = c5_b3
            best_h = h_b3.copy()

    # Phase 6: Try nearby resolutions
    n_tries = []
    for delta in [-8, -4, -2, 2, 4, 8, 16, 32]:
        nt = n + delta
        if nt >= 100 and nt <= 800 and nt % 2 == 0 and nt != n:
            n_tries.append(nt)
    n_tries = sorted(set(n_tries))

    for n_t in n_tries:
        if remaining() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)

        # Quick SLSQP
        h_opt = h_feas
        for beta in [1000, 5000]:
            if remaining() < 10:
                break
            h_opt = slsqp_smooth(h_opt, n_t, beta, maxiter=200)

        # Binary search
        if remaining() > 20:
            h_bin_t = (h_opt > 0.5).astype(float)
            h_bt, c5_bt = directed_binary_search(h_bin_t, n_t, time_limit=min(40, remaining() - 15), rng=rng)
            if c5_bt < compute_c5(h_opt):
                h_opt = h_bt

        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 7: Higher resolution attempts
    for n_hi in [512, 640, 768]:
        if remaining() < 50:
            break
        h_interp = interp_to(best_h, best_n, n_hi)
        h_feas = project_feasible(h_interp, n_hi)
        h_opt = slsqp_smooth(h_feas, n_hi, 3000, maxiter=200)
        if remaining() > 25:
            h_bin_hi = (h_opt > 0.5).astype(float)
            h_bhi, c5_bhi = directed_binary_search(h_bin_hi, n_hi, time_limit=min(30, remaining() - 15), rng=rng)
            if c5_bhi < compute_c5(h_opt):
                h_opt = h_bhi
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_hi

    # Phase 8: Final intensive refinement
    n = best_n
    h = best_h.copy()

    # Final SLSQP
    for beta in [10000, 30000, 80000]:
        if remaining() < 5:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final binary search
    if remaining() > 15:
        h_bin_f = (best_h > 0.5).astype(float)
        h_bf, c5_bf = directed_binary_search(h_bin_f, n, time_limit=min(30, max(10, remaining() - 8)), rng=rng)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

    # Final multi-lag binary
    if remaining() > 12:
        h_bf2, c5_bf2 = multi_lag_binary_search(best_h, n, time_limit=min(20, max(5, remaining() - 6)), rng=rng)
        if c5_bf2 < best_c5:
            best_c5 = c5_bf2
            best_h = h_bf2.copy()

    # Final polish
    if remaining() > 6:
        h_new = slsqp_smooth(best_h, best_n, 50000, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Validate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
