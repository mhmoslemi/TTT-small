        T = T_init * (T_final / T_init) ** (it / max(total_iters, 1))
        if T < 0.001:
            T = 0.001
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(ones) == 0:
            break
        
        # Generate candidates: mix of targeted and random
        best_delta = float('inf')
        best_i, best_j = -1, -1
        best_corr_new = None
        
        n_candidates = 30 + int(50 * T / T_init)
        
        for _ in range(n_candidates):
            # 70% targeted, 30% random
            if rng.rand() < 0.7:
                # Target: pick i from zeros near active correlation peaks
                active_idx = np.where(corr > 0.95 * cur_max)[0]
                if len(active_idx) > 0:
                    ak = active_idx[rng.randint(len(active_idx))]
                    off = ak - (n - 1)
                    if off >= 0:
                        # corr at offset off: h[p+off](1-h[p])
                        # Positions contributing: p in [0, n-1-off]
                        # i should be a 0 at position p, j should be a 1 at position p+off
                        valid_p = np.where((h[:n-off] == 0) & (h[off:] == 1))[0]
                        if len(valid_p) > 0:
                            p = valid_p[rng.randint(len(valid_p))]
                            i_cand = p
                            j_cand = p + off
                        else:
                            i_cand = zeros[rng.randint(len(zeros))]
                            j_cand = ones[rng.randint(len(ones))]
                    else:
                        ao = -off
                        valid_p = np.where((h[:n-ao] == 1) & (h[ao:] == 0))[0]
                        if len(valid_p) > 0:
                            p = valid_p[rng.randint(len(valid_p))]
                            i_cand = p + ao  # the 0 position
                            j_cand = p  # the 1 position
                        else:
                            i_cand = zeros[rng.randint(len(zeros))]
                            j_cand = ones[rng.randint(len(ones))]
                else:
                    i_cand = zeros[rng.randint(len(zeros))]
                    j_cand = ones[rng.randint(len(ones))]
            else:
                i_cand = zeros[rng.randint(len(zeros))]
                j_cand = ones[rng.randint(len(ones))]
            
            # Evaluate
            h[i_cand] = 1.0
            h[j_cand] = 0.0
            corr_new = compute_corr(h)
            mx = float(np.max(corr_new))
            h[i_cand] = 0.0
            h[j_cand] = 1.0
            delta = mx - cur_max
            
            if delta < best_delta:
                best_delta = delta
                best_i, best_j = i_cand, j_cand
                best_corr_new = corr_new
        
        if best_i < 0:
            continue
        
        # Accept/reject
        if best_delta <= 0:
            # Accept (improvement)
            h[best_i] = 1.0
            h[best_j] = 0.0
            cur_max = best_delta + cur_max  # approximate
            corr = best_corr_new
            cur_max = float(np.max(corr))
            if cur_max * (2.0/n) < best_c5:
                best_c5 = cur_max * (2.0/n)
                best_h = h.copy()
        else:
            # Accept with probability exp(-delta/T)
            if rng.rand() < np.exp(-best_delta / max(T, 1e-10)):
                h[best_i] = 1.0
                h[best_j] = 0.0
                corr = best_corr_new
                cur_max = float(np.max(corr))
    
    return best_h, best_c5

def slp_optimize(h, n, max_iter=100):
    """Successive Linear Programming."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx_v = 2.0 / n
    
    for it in range(max_iter):
        if remaining() < 5:
            break
        
        corr = compute_corr(h)
        c_max = float(np.max(corr))
        
        # Compute gradients
        grads = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            off = k - (n-1)
            if off >= 0:
                np_ = n - off
                grads[k, off:off+np_] += (1-h[:np_])
                grads[k, :np_] -= h[off:off+np_]
            else:
                ao = -off
                np_ = n - ao
                grads[k, :np_] += (1-h[ao:ao+np_])
                grads[k, ao:ao+np_] -= h[:np_]
        
        # Solve LP
        x = cp.Variable(n)
        t = cp.Variable()
        constraints = [x >= 0, x <= 1, cp.sum(x) == n/2]
        
        b = corr - grads @ h
        constraints.append(grads @ x - t <= b)
        
        prob = cp.Problem(cp.Minimize(t), constraints)
        
        x_lp = None
        for solver_name in ['ECOS', 'SCS', 'CBC', 'GLPK']:
            try:
                prob.solve(solver=cp.ECOS if solver_name == 'ECOS' else 
                           cp.SCS if solver_name == 'SCS' else
                           cp.CBC if solver_name == 'CBC' else cp.GLPK)
                if x.value is not None:
                    x_lp = np.array(x.value).flatten()
                    break
            except Exception:
                continue
        
        if x_lp is None:
            break
        
        x_lp = np.clip(x_lp, 0, 1)
        direction = x_lp - h
        
        # Line search
        best_alpha = 0.0
        best_trial_c5 = best_c5
        for alpha in [1.0, 0.5, 0.3, 0.2, 0.1, 0.7, 0.4, 0.15, 0.05]:
            h_trial = project_feasible(h + alpha * direction, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_trial_c5:
                best_trial_c5 = c5_trial
                best_alpha = alpha
        
        if best_alpha > 0 and best_trial_c5 < best_c5 - 1e-12:
            h = project_feasible(h + best_alpha * direction, n)
            best_c5 = best_trial_c5
            best_h = h.copy()
        else:
            break
    
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
