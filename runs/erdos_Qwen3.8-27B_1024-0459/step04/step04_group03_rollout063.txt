def project_box_sum(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum} using water-filling."""
    h = np.array(h, dtype=float).copy()
    # Water-filling: find t such that sum(clip(h-t, 0, 1)) = target_sum
    # clip(h-t, 0, 1) = max(0, min(1, h-t))
    # This is equivalent to finding t where the "water level" gives the right sum.
    
    # Sort h values
    sorted_h = np.sort(h)
    n = len(h)

    # For each element h_i:
    #   if h_i - t <= 0: contributes 0
    #   if 0 < h_i - t < 1: contributes h_i - t
    #   if h_i - t >= 1: contributes 1
    # So: sum = sum_{h_i>t} min(1, h_i - t)
    #     = count_{h_i >= t+1} * 1 + sum_{t < h_i < t+1} (h_i - t)
    
    # Bisection on t
    lo = h.min() - 1.5
    hi = h.max() + 0.5
    
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    t = (lo + hi) / 2
    result = np.clip(h - t, 0.0, 1.0)
    
    # Fix numerical issues

        else:

        result = np.clip(result, 0.0, 1.0)
    
    return result

    log_sum = np.log(np.sum(np.exp(shifted)))

- Maintain momentum: v = 0
- Each step:

  - v = (h_new - y) / (1 - lr * L) (or just v = h_new - y)
  - h = h_new

    return max(0.0, _BUDGET - (time.time() - _START) - 2.0)

    for _ in range(64):
