import time

    dx = 2.0 / n

    target = n / 2.0

            break

        else:

        h = np.clip(h, 0.0, 1.0)

            idx = np.argsort(h)[1:]  # most full

            diff -= np.sign(diff) * min(abs(adj), abs(diff))

                break

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def subgrad_topk(h, n, top_k=10):

    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]

    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n), float(np.max(corr)) * dx
    w = w / w_sum
    
    grad = np.zeros(n)
    for k_idx, weight in zip(top_idx, w):
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            n_valid = n - offset
            if n_valid <= 0:
                continue
            # For i in [offset, n): grad[i] += weight * (1-h[i-offset])
            grad[offset:] += weight * (1.0 - h[:n_valid])
            # For i in [0, n_valid): grad[i] -= weight * h[i+offset]
            grad[:n_valid] -= weight * h[offset:offset+n_valid]
        else:

            if n_valid <= 0:

            grad[:n_valid] += weight * (1.0 - h[m:m+n_valid])
            # For i in [m, n): grad[i] -= weight * h[i-m]
            grad[m:] -= weight * h[:n_valid]
    
    return grad * dx, float(np.max(corr)) * dx

    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    c5_best = compute_c5(h)
    h_best = h.copy()

            break

        t += 1

        v = beta2 * v + (1 - beta2) * g**2

        v_hat = v / (1 - beta2**t)

            h = h_new.copy()
        # else: keep h unchanged (or could try with smaller step)
    
    return h_best, c5_best

_G_START = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(h)
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            adj = np.sign(diff) * min(abs(diff), abs((1.0 if diff > 0 else 0.0) - h[i]) + 1e-15)
            h[i] += adj
            diff -= adj
            if abs(diff) < 1e-14:
                break
    return h

                accepted = True
                break
            alpha *= 0.5

        if not accepted:
            v *= 0.5
            step *= 0.7

        h_prev = h
        g_prev = g

    return h_best, c5_best


def adam_opt(h, n, max_iter=500, top_k=10, lr=0.005, beta1=0.9, beta2=0.999,
             eps=1e-8, lr_decay=0.999):
    h = h.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    cur_lr = lr
    stall = 0

    for it in range(max_iter):
        if remaining_time() < 4.0:
            break
        g, _ = subgrad_topk(h, n, top_k)
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-15:
            c5_best = c5_new
            h_best = h_new.copy()
            h = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                cur_lr *= 0.5
                stall = 0
        cur_lr *= lr_decay

    return h_best, c5_best


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def coord_descent_pairs(h, n, max_sweeps=20, rng=None):
    c5_best = compute_c5(h)
    h_best = h.copy()
    if rng is None:
        rng = np.random.RandomState(0)
    deltas = [0.01, 0.02, 0.05, -0.01, -0.02, -0.05, 0.1, -0.1]

    for sweep in range(max_sweeps):
        if remaining_time() < 3.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if remaining_time() < 2.0:
                break
            j = (i + 1) % n
            if j == i:
                j = (i + 2) % n
            for d in deltas:
                h_trial = h_best.copy()
                h_trial[i] = np.clip(h_trial[i] + d, 0.0, 1.0)
                h_trial[j] = np.clip(h_trial[j] - d, 0.0, 1.0)
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < c5_best - 1e-14:
                    c5_best = c5_t
                    h_best = h_trial.copy()
                    any_imp = True
                    break
            old = h_best[i]
            for new_v in [0.0, 1.0]:
                if abs(new_v - old) < 1e-8:
                    continue
                h_trial = h_best.copy()
                h_trial[i] = new_v
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < c5_best - 1e-14:
                    c5_best = c5_t
                    h_best = h_trial.copy()
                    any_imp = True
        if not any_imp and sweep > 3:
            break
    return h_best, c5_best


def simulated_annealing(h, n, rng, n_iter=3000, T0=0.003, gamma=0.999):
    c5_best = compute_c5(h)
    h_best = h.copy()
    h_cur = h.copy()
    c5_cur = c5_best
    T = T0

    for it in range(n_iter):
        if remaining_time() < 3.0:
            break
        T = T0 * (gamma ** it)
        pert = rng.randn(n) * (0.02 + 0.03 * T / T0)
        h_trial = h_cur + pert
        h_trial = project_feasible(h_trial, n)
        c5_trial = compute_c5(h_trial)
        delta = c5_trial - c5_cur
        if delta < 0 or (T > 1e-8 and rng.rand() < np.exp(-delta / T)):
            h_cur = h_trial
            c5_cur = c5_trial
            if c5_cur < c5_best - 1e-15:
                c5_best = c5_cur
                h_best = h_trial.copy()

    return h_best, c5_best


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h


def slsqp_polish(h, n, maxiter=200):
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return float(np.max(corr)) * dx

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        max_idx = np.argmax(corr)
        offset = int(max_idx) - (n - 1)
        grad = np.zeros(n)
        if offset >= 0:
            n_valid = n - offset
            if n_valid > 0:
                grad[offset:] += (1.0 - x[:n_valid])
                grad[:n_valid] -= x[offset:offset + n_valid]
        else:
            m = -offset
            n_valid = n - m
            if n_valid > 0:
                grad[:n_valid] += (1.0 - x[m:m + n_valid])
                grad[m:] -= x[:n_valid]
        return grad * dx

    def con_fun(x):
        return np.sum(x) - target

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h


def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(parent_h)
        best_h = project_feasible(parent_h, best_n)
        best_c5 = compute_c5(best_h)
        print("Parent: n={}, C5={:.10f}".format(best_n, best_c5))
    except Exception:
        best_n = 256
        best_h = project_feasible(make_thue_morse(best_n), best_n)
        best_c5 = compute_c5(best_h)
        print("Thue-Morse: n={}, C5={:.10f}".format(best_n, best_c5))

    n = best_n

    # Phase 1: Nesterov subgradient
    if remaining_time() > 120:
        print("\n--- Phase 1: Nesterov subgradient ---")
        best_h, best_c5 = nesterov_opt(best_h, n, max_iter=250, top_k_start=15,
                                       top_k_end=3, lr0=0.04)
        print("  C5={:.10f}".format(best_c5))

    # Phase 2: Adam optimization
    if remaining_time() > 100:
        print("\n--- Phase 2: Adam optimization ---")
        best_h, best_c5 = adam_opt(best_h, n, max_iter=600, top_k=8,
                                   lr=0.004, lr_decay=0.9995)
        print("  C5={:.10f}".format(best_c5))

    # Phase 3: Another Nesterov round with different params
    if remaining_time() > 80:
        print("\n--- Phase 3: Nesterov round 2 ---")
        best_h, best_c5 = nesterov_opt(best_h, n, max_iter=200, top_k_start=8,
                                       top_k_end=2, lr0=0.02,
                                       momentum_start=0.7, momentum_end=0.95)
        print("  C5={:.10f}".format(best_c5))

    # Phase 4: Multi-resolution exploration
    print("\n--- Phase 4: Multi-resolution ---")
    for n_try in [192, 224, 288, 320]:
        if remaining_time() < 60:
            break
        h_interp = interp_to(best_h, best_n, n_try)
        h_interp = project_feasible(h_interp, n_try)
        c5_h = compute_c5(h_interp)
        h_interp, c5_h = adam_opt(h_interp, n_try, max_iter=400, top_k=8,
                                  lr=0.004, lr_decay=0.9995)
        if remaining_time() > 30:
            h_interp, c5_h = nesterov_opt(h_interp, n_try, max_iter=150,
                                          top_k_start=8, top_k_end=3, lr0=0.03)
        if c5_h < best_c5 - 1e-10:
            best_c5 = c5_h
            best_h = h_interp.copy()
            best_n = n_try
            n = n_try
            print("  n={}: C5={:.10f} (NEW BEST)".format(n_try, c5_h))
        else:
            print("  n={}: C5={:.10f}".format(n_try, c5_h))

    # Phase 5: Coordinate descent with pair swaps
    if remaining_time() > 30:
        print("\n--- Phase 5: Coordinate descent ---")
        best_h, best_c5 = coord_descent_pairs(best_h, best_n, max_sweeps=15, rng=rng)
        print("  C5={:.10f}".format(best_c5))

    # Phase 6: Simulated annealing
    if remaining_time() > 80:
        print("\n--- Phase 6: Simulated annealing ---")
        best_h, best_c5 = simulated_annealing(best_h, best_n, rng,
                                              n_iter=3000, T0=0.002, gamma=0.9995)
        print("  C5={:.10f}".format(best_c5))

    # Phase 7: Post-AN optimization
    if remaining_time() > 40:
        print("\n--- Phase 7: Post-AN optimization ---")
        best_h, best_c5 = adam_opt(best_h, best_n, max_iter=400, top_k=6,
                                   lr=0.002, lr_decay=0.9997)
        best_h, best_c5 = coord_descent_pairs(best_h, best_n, max_sweeps=10, rng=rng)
        print("  C5={:.10f}".format(best_c5))

    # Phase 8: SLSQP polish with raw objective
    if remaining_time() > 20:
        print("\n--- Phase 8: SLSQP polish ---")
        h_slsqp = slsqp_polish(best_h, best_n, maxiter=150)
        c5_slsqp = compute_c5(h_slsqp)
        if c5_slsqp < best_c5 - 1e-14:
            best_c5 = c5_slsqp
            best_h = h_slsqp
            print("  SLSQP: C5={:.10f}".format(c5_slsqp))
        else:
            print("  SLSQP: C5={:.10f} (no improvement)".format(c5_slsqp))

    # Phase 9: Final coordinate descent
    if remaining_time() > 10:
        print("\n--- Phase 9: Final coordinate descent ---")
        best_h, best_c5 = coord_descent_pairs(best_h, best_n, max_sweeps=8, rng=rng)
        print("  C5={:.10f}".format(best_c5))

    # Phase 10: One more Adam pass with very small lr
    if remaining_time() > 8:
        best_h, best_c5 = adam_opt(best_h, best_n, max_iter=300, top_k=4,
                                   lr=0.001, lr_decay=0.9999)
        print("  Final Adam: C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
