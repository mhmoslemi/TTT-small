def subgrad_one(h, n, shift):
    """Subgradient of corr at the lag with given shift."""
    sg = np.zeros(n)
    if shift >= 0:

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project(h, n):
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.sum(free)
        if cnt > 0:
            h[free] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        d2 = target - np.sum(h)
        if abs(d2) > 1e-14:
            for j in range(n):
                if 1e-12 < h[j] < 1.0 - 1e-12:
                    h[j] = np.clip(h[j] + d2, 0.0, 1.0)
                    break
    return h


def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    s = (corr - mc) * dx * beta
    exps = np.exp(s)
    Z = np.sum(exps)
    val = mc * dx + np.log(Z) / beta
    probs = exps / Z
    conv = np.convolve(1.0 - 2.0 * h, probs, 'full')
    grad = dx * conv[n - 1:2 * n - 1]
    return val, grad


def pgd_optimize(h, n, beta, max_iter, lr_init):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    val, grad = smooth_obj_grad(h, beta)
    best_val = val
    lr = lr_init
    stall = 0
    for it in range(max_iter):
        if remaining() < 1.0:
            break
        improved = False
        for bt in range(15):
            step = project(h - lr * grad, n)
            v, g = smooth_obj_grad(step, beta)
            if v < val - 1e-17:
                improved = True
                break
            lr *= 0.5
        if not improved:
            stall += 1
            if stall > 30:
                break
            lr *= 0.7
            continue
        h = step
        val, grad = v, g
        lr = min(lr * 1.1, 0.3)
        stall = 0
        if val < best_val - 1e-14:
            best_val = val
            best_h = h.copy()
        if (it + 1) % 30 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5


def subgrad_optimize(h, n, max_iter=300, lr_init=0.015):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    for it in range(max_iter):
        if remaining() < 0.5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)
        sg = np.zeros(n)
        if shift >= 0:
            sg[shift:] += 1.0 - h[:n - shift]
            sg[:n - shift] -= h[shift:]
        else:
            s = -shift
            sg[:n - s] += 1.0 - h[s:]
            sg[s:] -= h[:n - s]
        decay = 1.0 / (1.0 + 0.002 * it)
        step = project(h - lr_init * decay * sg, n)
        c5_new = compute_c5(step)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = step.copy()
        h = step
    return best_h, best_c5


def interp(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    xp = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(xn, xp, h_old)
    return project(h_new, n_new)


def pair_transfer(h, n, rng, max_rounds=8):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if remaining() < 2.0:
            break
        any_imp = False
        indices = rng.permutation(n)
        checked = 0
        for ii in indices:
            if remaining() < 1.0:
                break
            i = int(ii)
            hi = best_h[i]
            if hi < 0.05 or hi > 0.95:
                continue
            for j in range(i + 1, min(i + 40, n)):
                hj = best_h[j]
                if abs(hi - hj) < 0.03:
                    continue
                for frac in (0.3, 0.5, 0.7):
                    amt = (hi - hj) * frac
                    if abs(amt) < 0.02:
                        continue
                    h_t = best_h.copy()
                    h_t[i] -= amt
                    h_t[j] += amt
                    h_t = project(h_t, n)
                    c5_t = compute_c5(h_t)
                    if c5_t < best_c5 - 1e-15:
                        best_c5 = c5_t
                        best_h = h_t.copy()
                        any_imp = True
                        break
                checked += 1
                if checked > 5000:
                    break
            if not any_imp and checked > n // 4:
                break
        if not any_imp:
            break
    return best_h, best_c5


def perturb_cleanup(h, n, rng, sigma):
    h_p = h.copy()
    mask = (h_p > 0.03) & (h_p < 0.97)
    n_mask = int(np.sum(mask))
    if n_mask > 0:
        h_p[mask] += rng.normal(0, sigma, n_mask)
    n_flip = min(n_mask // 3, n // 4)
    if n_flip > 0:
        flip_idx = rng.choice(n, size=n_flip, replace=False)
        for i in flip_idx:
            h_p[int(i)] = 1.0 - h_p[int(i)]
    return project(h_p, n)


def topk_subgrad(h, n, n_active=8, max_iter=200, lr=0.005):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    for it in range(max_iter):
        if remaining() < 0.5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_active]
        weights = corr[top_idx]
        w_sum = weights.sum()
        if w_sum < 1e-15:
            break
        w = (weights / w_sum) ** 1.5
        w = w / w.sum()
        sg = np.zeros(n)
        for w_i, k_i in zip(w, top_idx):
            shift = int(k_i) - (n - 1)
            if shift >= 0:
                sg[shift:] += w_i * (1.0 - h[:n - shift])
                sg[:n - shift] -= w_i * h[shift:]
            else:
                s = -shift
                sg[:n - s] += w_i * (1.0 - h[s:])
                sg[s:] -= w_i * h[:n - s]
        step = project(h - lr * sg, n)
        c5_new = compute_c5(step)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = step.copy()
        h = step
        lr *= 0.997
    return best_h, best_c5


def _sm_xx(n_val, beta_val):
    return n_val, beta_val


def slsqp_polish(h, n, beta=5000, maxiter=200):
    global _T0, _BUDGET

    def _obj(x):
        nn = n
        bb = beta
        dx = 2.0 / nn
        hh = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hh, 1.0 - hh, mode='full')
        mc = np.max(corr)
        s = (corr - mc) * dx * bb
        return (mc * dx + np.log(np.sum(np.exp(s))) / bb)

    def _eq(x):
        return np.sum(x) - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq}]
    try:
        res = minimize(_obj, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            return project(res.x, n)
    except Exception:
        pass
    return h.copy()


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=np.float64)
        parent_n = len(parent_h)
        best_h = parent_h.copy()
        best_c5 = compute_c5(best_h)
        best_n = parent_n
    except Exception:
        best_n = 256
        best_h = np.full(best_n, 0.5)
        best_h = project(best_h, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: PGD at current resolution with escalating beta
    for beta in [50, 150, 400, 1000, 3000, 8000]:
        if remaining() < 15:
            break
        h_new, c5_new = pgd_optimize(h, n, beta, max_iter=300, lr_init=0.1)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Subgradient polish
    if remaining() > 20:
        h_new, c5_new = subgrad_optimize(h, n, max_iter=250, lr_init=0.012)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: SLSQP polish
    if remaining() > 25:
        h_new = slsqp_polish(best_h, best_n, beta=5000, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 4: Multi-resolution
    for target_n in [512, 1024]:
        if remaining() < 60:
            break
        h_interp = interp(best_h, best_n, target_n)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5 + 0.005:
            for beta in [100, 500, 2000]:
                if remaining() < 10:
                    break
                h_new, c5_new = pgd_optimize(h_interp, target_n, beta,
                                             max_iter=200, lr_init=0.08)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = target_n
                    h_interp = h_new.copy()
            if remaining() > 20 and target_n <= 1024:
                h_new, c5_new = subgrad_optimize(h_interp, target_n,
                                                 max_iter=200, lr_init=0.01)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = target_n

    # Phase 5: Top-k subgradient
    if remaining() > 20:
        h_new, c5_new = topk_subgrad(best_h, best_n, n_active=8,
                                     max_iter=200, lr=0.004)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 6: Pair transfer
    if remaining() > 15:
        h_new, c5_new = pair_transfer(best_h, best_n, rng, max_rounds=6)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if remaining() > 10:
            h_new, c5_new = pgd_optimize(best_h, best_n, 2000,
                                         max_iter=150, lr_init=0.05)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 7: Random restarts
    for restart in range(3):
        if remaining() < 35:
            break
        h_p = perturb_cleanup(best_h, best_n, rng, sigma=0.03)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5 + 0.01:
            h_new, c5_new = pgd_optimize(h_p, best_n, 500,
                                         max_iter=200, lr_init=0.08)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            if remaining() > 15:
                h_new, c5_new = subgrad_optimize(best_h, best_n,
                                                 max_iter=150, lr_init=0.01)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()

    # Phase 8: Try other resolutions from best solution
    if remaining() > 40:
        for try_n in [200, 220, 240, 280, 320, 360, 400]:
            if remaining() < 20:
                break
            if try_n == best_n:
                continue
            h_i = interp(best_h, best_n, try_n)
            c5_i = compute_c5(h_i)
            if c5_i < best_c5 + 0.003:
                h_new, c5_new = pgd_optimize(h_i, try_n, 1000,
                                             max_iter=150, lr_init=0.06)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = try_n
                if remaining() > 15:
                    h_new, c5_new = subgrad_optimize(h_new, try_n,
                                                     max_iter=100, lr_init=0.008)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        best_n = try_n

    # Phase 9: Final SLSQP + subgrad + pair
    if remaining() > 15:
        h_new = slsqp_polish(best_h, best_n, beta=10000, maxiter=150)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if remaining() > 12:
        h_new, c5_new = topk_subgrad(best_h, best_n, n_active=6,
                                     max_iter=150, lr=0.003)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if remaining() > 10:
        h_new, c5_new = pair_transfer(best_h, best_n, rng, max_rounds=4)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 10: Ultra-fine PGD
    if remaining() > 8:
        h_new, c5_new = pgd_optimize(best_h, best_n, 20000,
                                     max_iter=100, lr_init=0.02)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final projection and evaluation
    best_h = project(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
