Δcorr[k] = delta * (dc(k)/dh[i] - dc(k)/dh[j])

_BETA = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)


def _sm_obj(x):
    n = _N
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _BETA
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)


def _sm_grad(x):
    n = _N
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _BETA
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    sx = np.sum(ex)
    w = ex / sx
    dx_v = 2.0 / n
    g = np.zeros(n)

def slsqp_opt(h, n, beta, maxiter=500, ftol=1e-16):

                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})

    return h


def _grad_one_k(x, k, n):
    """Compute d(corr[k])/dx for a given k."""
    off = k - (n - 1)
    g = np.zeros(n)
    if off >= 0:
        np_ = n - off
        g[off:off + np_] += (1.0 - x[:np_])

    """Fast pairwise coordinate descent using precomputed gradients."""
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    dx_v = 2.0 / n
    best_c5 = compute_c5(h)

    for p in range(n_passes):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        for _ in range(n_pairs_per_pass):
            if remaining() < 3:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            # Compute d_k = dc(k)/dh[i] - dc(k)/dh[j]
            # For efficiency, compute gradient contributions
            di = np.zeros(2 * n - 1)
            dj = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                off = k - (n - 1)
                if off >= 0:
                    np_ = n - off
                    # d/dh[m] of sum_{r=0}^{np_-1} h[r+off]*(1-h[r])
                    # h[i] appears in h[r+off] when r+off=i -> r=i-off (if 0<=i-off<np_)
                    # h[i] appears in h[r] when r=i (if 0<=i<np_)
                    gi = 0.0
                    if 0 <= i - off < np_:
                        gi += (1.0 - h[i - off])
                    if 0 <= i < np_:
                        gi -= h[i + off]
                    di[k] = gi
                    gj = 0.0
                    if 0 <= j - off < np_:
                        gj += (1.0 - h[j - off])
                    if 0 <= j < np_:
                        gj -= h[j + off]
                    dj[k] = gj
                else:
                    ao = -off
                    np_ = n - ao
                    gi = 0.0
                    if 0 <= i < np_:
                        gi += (1.0 - h[i + ao])
                    if 0 <= i - ao < np_:
                        gi -= h[i - ao]
                    di[k] = gi
                    gj = 0.0
                    if 0 <= j < np_:
                        gj += (1.0 - h[j + ao])
                    if 0 <= j - ao < np_:
                        gj -= h[j - ao]
                    dj[k] = gj

            d = di - dj  # d(corr[k])/dh[i] - d(corr[k])/dh[j]

            # Feasible range for delta: h[i]+delta in [0,1], h[j]-delta in [0,1]
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_hi - d_lo < 1e-12:
                continue

            # 1D convex minimax: min_delta max_k (corr[k] + delta*d_k)
            # Ternary search
            a, b = d_lo, d_hi
            for _ in range(30):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                f1 = np.max(corr + m1 * d)
                f2 = np.max(corr + m2 * d)
                if f1 < f2:
                    b = m2
                else:
                    a = m1

            delta_opt = (a + b) / 2.0
            # Check improvement
            new_c5 = np.max(corr + delta_opt * d) * dx_v
            if new_c5 < best_c5 - 1e-14:
                h[i] += delta_opt
                h[j] -= delta_opt
                h = np.clip(h, 0.0, 1.0)
                # Re-project for sum constraint (should be maintained)
                h = project_feasible(h, n)
                best_c5 = compute_c5(h)
                corr = compute_corr(h)

    return h, best_c5
