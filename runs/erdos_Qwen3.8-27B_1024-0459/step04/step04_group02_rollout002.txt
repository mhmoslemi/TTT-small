def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        # Distribute adjustment across interior points
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(interior):
            h[interior] += diff / np.sum(interior)
        else:
            # Edge case: move one point
            if diff > 0:
                idx = np.argmin(h)
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + diff)
        h = np.clip(h, 0.0, 1.0)
    # Final correction

    if abs(diff) > 1e-15:
        # Spread tiny adjustment
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, beta, n_active=None):
    """Compute smooth max of corr and its gradient w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    nc = len(corr)
    
    if n_active is not None and n_active < nc:
        # Only consider top n_active + some margin
        top = np.argsort(corr)[::-1][:n_active * 2]
        m = np.max(corr[top])
        idx = np.where(corr >= m - 1.0 / beta)[0]
        if len(idx) == 0:
            idx = top[:n_active]
    else:
        idx = np.arange(nc)
        m = np.max(corr)
    
    shifted = beta * (corr[idx] - m)
    shifted = np.minimum(shifted, 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    
    # Gradient
    grad = np.zeros(n)
    for j, k in enumerate(idx):
        w = weights[j]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                # corr[k] = sum_{i=0}^{npairs-1} h[i] * (1 - h[i+offset])
                # d/dh[i] = (1-h[i+offset]) for i in [0, npairs)
                # d/dh[i+offset] = -h[i] for i in [0, npairs)
                grad[:npairs] += w * (1.0 - h[offset:offset+npairs])
                grad[offset:offset+npairs] -= w * h[:npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                # corr[k] = sum_{i=0}^{npairs-1} h[i+aoff] * (1 - h[i])
                # d/dh[i] = -h[i+aoff]
                # d/dh[i+aoff] = (1-h[i])
                grad[:npairs] -= w * h[aoff:aoff+npairs]
                grad[aoff:aoff+npairs] += w * (1.0 - h[:npairs])
    
    grad *= dx
    return val, grad

def slsqp_optimize(h, n, beta, maxiter, n_active=None):
    """SLSQP optimization with smooth max objective."""
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    
    def obj(x):
        v, _ = smooth_max_and_grad(x, beta, n_active)
        return v
    
    def obj_grad(x):
        _, g = smooth_max_and_grad(x, beta, n_active)
        return g
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,

    except Exception:
        pass
    return h.copy()

            # - Terms where j+off = i, i.e., j = i-off (negative: h[j]*(1-h[i]))

            b_coeff = np.zeros(len(active))  # quadratic coefficient

                off = k - (n - 1)
                base = corr[k]

                # h[i] appears as "h[j+off]" when j=i-off and 0<=i-off<n and off!=0
