            if c5_new < c5_best - 1e-13:
                c5_best = c5_new
                h_best = h_new.copy()
        except Exception:
            pass

    return h_best, c5_best


def coordinate_descent(h, n, max_sweeps=10, rng=None):
    c5_best = float(compute_c5(h))
    h_best = h.copy()
    if rng is None:
        rng = np.random.RandomState(0)
    for sweep in range(max_sweeps):
        if remaining_time() < 3.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if remaining_time() < 2.0:
                break
            old = h_best[i]
            for new_v in [0.0, 1.0]:
                if abs(new_v - old) < 1e-9:
                    continue
                h_trial = h_best.copy()
                h_trial[i] = new_v
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < c5_best - 1e-13:
                    c5_best = c5_t
                    h_best = h_trial.copy()
                    any_imp = True
            if not any_imp:
                for delta in [0.01, -0.01, 0.03, -0.03, 0.05, -0.05]:
                    h_trial = h_best.copy()
                    h_trial[i] = np.clip(h_best[i] + delta, 0.0, 1.0)
                    h_trial = project_feasible(h_trial, n)
                    c5_t = compute_c5(h_trial)
                    if c5_t < c5_best - 1e-13:
                        c5_best = c5_t
                        h_best = h_trial.copy()
                        any_imp = True
                        break
        if not any_imp and sweep > 3:
            break
    return h_best, c5_best


def alternating_opt(h, n, max_rounds=50):
    c5_best = compute_c5(h)
    h_best = h.copy()
    corr = compute_corr(h)
    for rnd in range(max_rounds):
        if remaining_time() < 3.0:
            break
        m = np.max(corr)
        active = np.where(corr >= m - 0.5)[0]
        top_k = min(len(active), 15)
        grad, _, _ = subgrad_compute(h_best, n, top_k=top_k)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad = grad / gn
        best_step = 0.0
        best_c5 = c5_best
        for s in [0.05, 0.03, 0.02, 0.01, 0.005, 0.002]:
            h_trial = project_feasible(h_best - s * gn * grad, n)
            c5_t = compute_c5(h_trial)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                best_step = s
                best_h = h_trial
        if best_step > 0:
            h_best = best_h
            c5_best = best_c5
            corr = compute_corr(h_best)
        else:
            break
    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX

    best_c5 = float('inf')

        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = project_feasible(make_thue_morse(parent_n), parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("Thue-Morse init: n={}, C5={:.10f}".format(parent_n, best_c5))

    # Phase 1: SLSQP polish at parent resolution with escalating beta
    print("\n--- Phase 1: SLSQP polish (n={}) ---".format(best_n))
    _G_N = best_n
    _G_TGT = best_n / 2.0
    _G_DX = 2.0 / best_n
    best_h, best_c5 = slsqp_polish(best_h, best_n, [100, 300, 1000, 3000, 10000], max_iter_each=200)
    print("  After SLSQP: C5={:.10f}".format(best_c5))

    # Phase 2: Subgradient with momentum
    print("\n--- Phase 2: Subgradient momentum ---")
    best_h, best_c5 = subgrad_momentum(best_h, best_n, max_iter=300, step0=0.08, momentum=0.85)
    print("  After subgrad: C5={:.10f}".format(best_c5))

    # Phase 3: Alternating optimization
    print("\n--- Phase 3: Alternating opt ---")
    best_h, best_c5 = alternating_opt(best_h, best_n, max_rounds=60)
    print("  After alternating: C5={:.10f}".format(best_c5))

    # Phase 4: SLSQP again after subgrad changes
    print("\n--- Phase 4: SLSQP round 2 ---")
    _G_N = best_n
    _G_TGT = best_n / 2.0
    _G_DX = 2.0 / best_n
    best_h, best_c5 = slsqp_polish(best_h, best_n, [500, 2000, 8000, 20000], max_iter_each=150)
    print("  After SLSQP2: C5={:.10f}".format(best_c5))

    # Phase 5: Perturbation exploration
    print("\n--- Phase 5: Perturb explore ---")
    best_h, best_c5 = do_perturb_explore(best_h, best_n, rng, num_pert=18)
    print("  After perturb: C5={:.10f}".format(best_c5))

    # Phase 6: Try higher resolution n=512
    if remaining_time() > 120.0:
        print("\n--- Phase 6: Higher resolution n=512 ---")
        n_try = 512
        h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        _G_N = n_try
        _G_TGT = n_try / 2.0
        _G_DX = 2.0 / n_try
        c5_h = compute_c5(h_interp)
        h_interp, c5_h = subgrad_momentum(h_interp, n_try, max_iter=200, step0=0.06, momentum=0.8)
        h_interp, c5_h = lbfgs_polish(h_interp, n_try, [100, 500, 2000], max_iter_each=150)
        if remaining_time() > 20.0:
            h_interp, c5_h = slsqp_polish(h_interp, n_try, [500, 2000], max_iter_each=100)
        if c5_h < best_c5 - 1e-10:
            best_c5 = c5_h
            best_h = h_interp.copy()
            best_n = n_try
            print("  n=512: C5={:.10f} [NEW BEST]".format(best_c5))
        else:
            print("  n=512: C5={:.10f} (no improvement)".format(c5_h))

    # Phase 7: Try n=1024 if time allows
    if remaining_time() > 100.0:
        print("\n--- Phase 7: Higher resolution n=1024 ---")
        n_try = 1024
        h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        _G_N = n_try
        _G_TGT = n_try / 2.0
        _G_DX = 2.0 / n_try
        c5_h = compute_c5(h_interp)
        h_interp, c5_h = subgrad_momentum(h_interp, n_try, max_iter=150, step0=0.05, momentum=0.75)
        h_interp, c5_h = lbfgs_polish(h_interp, n_try, [50, 200, 800], max_iter_each=100)
        if c5_h < best_c5 - 1e-10:
            best_c5 = c5_h
            best_h = h_interp.copy()
            best_n = n_try
            print("  n=1024: C5={:.10f} [NEW BEST]".format(best_c5))
        else:
            print("  n=1024: C5={:.10f} (no improvement)".format(c5_h))

    # Phase 8: Go back to best_n and do final intensive optimization
    print("\n--- Phase 8: Final intensive optimization (n={}) ---".format(best_n))
    n = best_n
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n

    # Subgradient
    best_h, best_c5 = subgrad_momentum(best_h, n, max_iter=200, step0=0.05, momentum=0.8)
    print("  After subgrad: C5={:.10f}".format(best_c5))

    # Alternating
    best_h, best_c5 = alternating_opt(best_h, n, max_rounds=40)
    print("  After alt: C5={:.10f}".format(best_c5))

    # SLSQP with very high beta
    best_h, best_c5 = slsqp_polish(best_h, n, [1000, 5000, 20000, 50000], max_iter_each=200)
    print("  After SLSQP: C5={:.10f}".format(best_c5))

    # Phase 9: Coordinate descent
    if remaining_time() > 15.0:
        print("\n--- Phase 9: Coordinate descent ---")
        best_h, best_c5 = coordinate_descent(best_h, best_n, max_sweeps=8, rng=rng)
        print("  After coord: C5={:.10f}".format(best_c5))

    # Phase 10: One more perturbation + optimize cycle
    if remaining_time() > 20.0:
        print("\n--- Phase 10: Final perturb + optimize ---")
        _G_N = best_n
        _G_TGT = best_n / 2.0
        _G_DX = 2.0 / best_n
        best_h, best_c5 = do_perturb_explore(best_h, best_n, rng, num_pert=10)
        best_h, best_c5 = slsqp_polish(best_h, best_n, [2000, 10000, 30000], max_iter_each=150)
        best_h, best_c5 = subgrad_momentum(best_h, best_n, max_iter=100, step0=0.04, momentum=0.8)
        print("  Final: C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    # Ensure sum constraint is met exactly
    diff = best_n / 2.0 - np.sum(best_h)
    if abs(diff) > 1e-15:
        best_h = project_feasible(best_h, best_n)
        final_c5 = compute_c5(best_h)

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
