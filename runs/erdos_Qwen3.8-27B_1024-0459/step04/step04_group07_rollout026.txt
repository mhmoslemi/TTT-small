    H = np.fft.fft(src)
    freqs_src = np.fft.fftfreq(sn)
    freqs_dn = np.fft.fftfreq(dn)
    H_new = np.zeros(dn, dtype=complex)
    for j in range(dn):
        idx = int(round(freqs_dn[j] * sn)) % sn
        H_new[j] = H[idx]
    result = np.fft.ifft(H_new).real
    return result

def linear_interp(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def power_sum_obj(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = (np.sum(corr ** p)) ** (1.0 / p) * dx
    return val

def softmax_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = corr - m
    e = np.exp(np.clip(beta * s, -500, 500))
    val = (m + np.log(np.sum(e)) / beta) * dx
    return val

def slsqp_optimize(h0, n, beta, maxiter=300):
    def obj(x):
        n2 = len(x)
        dx = 2.0 / n2
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = corr - m
        e = np.exp(np.clip(beta * s, -500, 500))
        return float((m + np.log(np.sum(e)) / beta) * dx)
    
    def obj_grad(x):
        n2 = len(x)
        dx = 2.0 / n2
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = corr - m
        e = np.exp(np.clip(beta * s, -500, 500))
        p = e / np.sum(e)
        # Compute gradient via convolution
        # corr[k] = sum_j x[j+k]*(1-x[j]) for lag k
        # d(corr[k])/d(x[i]) = (1-x[i-k]) - x[i+k] (with boundary: x[j]=0 outside [0,n-1])
        # grad[i] = dx/beta * sum_k p[k] * ((1-x[i-k]) - x[i+k])
        # The sum_k p[k]*(1-x[i-k]) is a correlation of p and (1-x)
        # The sum_k p[k]*x[i+k] is a correlation of p and x
        
        # p is indexed from 0 to 2n-2, corresponding to lags -(n-1) to (n-1)
        # We need: sum_k p[k] * (1-x[i-k]) where k ranges over valid lags
        # This equals: (p correlated with (1-x)) at position i
        # In numpy: np.correlate(p, 1-x, mode='same') gives approximately what we want
        # But we need to be careful with the indexing.
        
        # Let's use a different approach: for each i, the contribution is
        # sum_{k: i-k in [0,n-1]} p[k] * (1-x[i-k]) - sum_{k: i+k in [0,n-1]} p[k] * x[i+k]
        
        # First term: let j = i-k, so k = i-j, sum_{j=0}^{n-1} p[i-j] * (1-x[j])
        # This is (p reversed) convolved with (1-x), evaluated at position i... 
        # Actually it's a cross-correlation.
        
        # Let me just compute it directly for n <= 512 (O(n^2) is fine)
        grad = np.zeros(n2)
        x_ext = np.zeros(2 * n2 - 1)
        x_ext[n2 - 1:n2 - 1 + n2] = x  # center x in extended array
        p_ext = p  # p[k] for k = -(n2-1), ..., n2-1, stored as p[0..2n2-2]
        
        for i in range(n2):
            # sum_k p[k] * (1 - x[i-k]) where i-k in [0, n2-1]
            # k ranges from i-(n2-1) to i, and p index = k + n2 - 1
            k_min = max(0, i - (n2 - 1))
            k_max = min(i, 2 * n2 - 2)
            if k_min <= k_max:
                ks = np.arange(k_min, k_max + 1)
                js = i - (ks - (n2 - 1))  # = i - k + (n2-1) ... wait
                # Actually p is indexed 0 to 2n-2, where index q corresponds to lag q-(n-1)
                # So p[q] has lag = q - (n2-1)
                # We need: sum over lags k where i-k is valid: p[q] * (1-x[i-k]) where q = k+(n2-1)
                # k = q - (n2-1), so i-k = i - q + (n2-1)
                # Valid when 0 <= i - q + (n2-1) <= n2-1, i.e., 1 <= q <= i + n2 - 1... 
                # Let me just do it numerically for small n.
                pass
        
        # For simplicity and correctness, use finite differences for the gradient
        # when n is small enough, or skip gradient entirely.
        return None  # Use numerical gradient
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def lbfgsb_optimize(h0, n, beta, maxiter=500, pen=100.0):
    def obj(x):
        n2 = len(x)
        dx = 2.0 / n2
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = corr - m
        e = np.exp(np.clip(beta * s, -500, 500))
        val = float((m + np.log(np.sum(e)) / beta) * dx)
        if pen > 0:
            val += pen * (np.sum(x) - n2 / 2.0) ** 2
        return val
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9, 'maxfun': 5000, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def coord_descent_focus(h, n, max_sweeps, rng, tlimit, target_c5=None):
    """Coordinate descent that focuses on the active lag."""
    c5 = compute_c5(h)
    h_best = h.copy()
    
    for sweep in range(max_sweeps):
        if tlimit() < 2.0:
            break
        # Find active lag
        corr = np.correlate(h_best, 1.0 - h_best, mode='full')
        k_active = int(np.argmax(corr))
        # Get the indices that contribute to the active lag
        lag = k_active - (n - 1)
        
        improved = False
        order = rng.permutation(n)
        
        for i in order:
            if tlimit() < 1.0:
                break
            v0 = h_best[i]
            
            # Try binary moves
            for v in [0.0, 1.0]:
                if abs(v - v0) < 1e-9:
                    continue
                trial = h_best.copy()
                trial[i] = v
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 1e-10:
                    c5 = c
                    h_best = trial.copy()
                    improved = True
                    break
            
            # Try small perturbations
            for d in [0.1, -0.1, 0.05, -0.05, 0.2, -0.2]:
                if tlimit() < 1.0:
                    break
                trial = h_best.copy()
                trial[i] = np.clip(trial[i] + d, 0.0, 1.0)
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 1e-10:
                    c5 = c
                    h_best = trial.copy()
                    improved = True
                    break
        
        if not improved and sweep >= 3:
            break
    
    return h_best, c5

def simulated_annealing_bin(h0, n, rng, tlimit, max_iters=50000):
    """SA on binary sequences to find good binary solutions."""
    # Start from threshold of h0
    h_bin = (h0 > 0.5).astype(float)
    h_bin = project_feasible(h_bin, n)
    # Make it truly binary
    h_bin = np.zeros(n)
    top = np.argsort(h0)[::-1][:n // 2]
    h_bin[top] = 1.0
    c5_cur = compute_c5(h_bin)
    h_best = h_bin.copy()
    c5_best = c5_cur
    
    T = 0.05
    T_min = 0.001
    cooling = 0.9997
    
    for it in range(max_iters):
        if tlimit() < 2.0:
            break
        # Pick a random index to flip
        i = rng.randint(n)
        trial = h_best.copy()
        if trial[i] > 0.5:
            # Flip to 0, need to turn on another
            zeros = np.where(trial < 0.5)[0]
            if len(zeros) == 0:
                continue
            j = rng.choice(zeros)
            trial[i] = 0.0
            trial[j] = 1.0
        else:
            ones = np.where(trial > 0.5)[0]
            if len(ones) == 0:
                continue
            j = rng.choice(ones)
            trial[i] = 1.0
            trial[j] = 0.0
        
        c5_trial = compute_c5(trial)
        delta = c5_trial - c5_best
        
        if delta < 0 or (T > T_min and rng.rand() < np.exp(-delta / T)):
            h_best = trial
            c5_cur = c5_trial
            if c5_cur < c5_best:
                c5_best = c5_cur
        T *= cooling
        
        if it % 5000 == 0:
            print("    SA iter {}: T={:.5f}, best={:.8f}".format(it, T, c5_best))
    
    return h_best, c5_best

def sigsig_optimize(h0, n, tlimit, max_rounds=20):
    """Optimize using sigmoid parameterization."""
    # Map h to sigmoid parameters: h[i] = 1/(1+exp(-z[i]))
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    
    for rnd in range(max_rounds):
        if tlimit() < 5.0:
            break
        # Convert to z (clip to avoid overflow)
        h_clipped = np.clip(h_cur, 0.01, 0.99)
        z0 = np.log(h_clipped / (1.0 - h_clipped))
        
        def obj_z(z):
            x = 1.0 / (1.0 + np.exp(-np.clip(z, -30, 30)))
            x = project_feasible(x, n)
            dx = 2.0 / n
            corr = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(corr)
            beta = 500.0
            s = corr - m
            e = np.exp(np.clip(beta * s, -500, 500))
            return float((m + np.log(np.sum(e)) / beta) * dx)
        
        try:
            res = minimize(obj_z, z0, method='L-BFGS-B',
                          options={'maxiter': 200, 'ftol': 1e-14, 'maxfun': 3000, 'iprint': 0})
            z_new = res.x
            h_new = 1.0 / (1.0 + np.exp(-np.clip(z_new, -30, 30)))
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-12:
                c5_cur = c5_new
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    
    def tlimit():
        return max(0.0, float(budget_s) - (time.time() - t_start) - 5.0)
    
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Initialize from parent or Thue-Morse
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = project_feasible(make_tm(parent_n), parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("No parent, using TM: n={}, C5={:.8f}".format(parent_n, parent_c5))
    
    # Phase 1: Optimize at parent resolution with L-BFGS-B (fast)
    if tlimit() > 150:
        print("\n=== Phase 1: L-BFGS-B at n={} ===".format(best_n))
        n = best_n
        h_cur = best_h.copy()
        
        for beta in [50, 100, 200, 500, 1000, 2000, 5000]:
            if tlimit() < 10:
                break
            h_new = lbfgsb_optimize(h_cur, n, beta, maxiter=1000, pen=500.0)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-12:
                best_c5 = c_new
                best_h = h_new.copy()
                best_n = n
                h_cur = h_new
                print("  beta={}: C5={:.8f}".format(beta, best_c5))
            else:
                h_cur = h_new  # Still use as starting point
    
    # Phase 2: SLSQP refinement at parent resolution
    if tlimit() > 100:
        print("\n=== Phase 2: SLSQP at n={} ===".format(best_n))
        n = best_n
        h_cur = best_h.copy()
        
        for beta in [1000, 3000, 5000, 10000]:
            if tlimit() < 10:
                break
            h_new = slsqp_optimize(h_cur, n, beta, maxiter=500)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-12:
                best_c5 = c_new
                best_h = h_new.copy()
                best_n = n
                h_cur = h_new
                print("  SLSQP beta={}: C5={:.8f}".format(beta, best_c5))
            else:
                h_cur = h_new
    
    # Phase 3: Try higher resolution n=512
    if tlimit() > 200:
        print("\n=== Phase 3: n=512 ===")
        n512 = 512
        
        # Multiple initializations for n=512
        inits_512 = []
        
        # Spectral interpolation
        h_spec = spectral_interp(best_h, best_n, n512)
        h_spec = project_feasible(h_spec, n512)
        inits_512.append(h_spec)
        
        # Linear interpolation
        h_lin = linear_interp(best_h, best_n, n512)
        h_lin = project_feasible(h_lin, n512)
        inits_512.append(h_lin)
        
        # Thue-Morse at 512
        h_tm512 = make_tm(n512)
        inits_512.append(h_tm512)
        
        # Random binary
        h_rb = make_rand_bin(rng, n512)
        inits_512.append(h_rb)
        
        # Sort by C5
        inits_scored = [(h, compute_c5(h)) for h in inits_512]
        inits_scored.sort(key=lambda x: x[1])
        
        for h_init, c_init in inits_scored[:3]:
            if tlimit() < 30:
                break
            h_cur = h_init.copy()
            c5_cur = c_init
            print("  Init C5={:.8f}".format(c5_cur))

                if c_new < c5_cur - 1e-12:

                    h_cur = h_new
            
            if c5_cur < best_c5 - 1e-9:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n512
                print("  n=512 improved: C5={:.8f}".format(best_c5))

                        c_new = compute_c5(h_new)
                        if c_new < best_c5 - 1e-9:
                            best_c5 = c_new
                            best_h = h_new.copy()
                            best_n = n_sa
                            print("  SA+opt n={} improved: C5={:.8f}".format(n_sa, best_c5))
    
    # Phase 6: Sigmoid parameterization
    if tlimit() > 60:

            h_pert = project_feasible(h_pert, n)
            c5_p = compute_c5(h_pert)
            
            if c5_p < best_c5 * 1.03:
                for beta in [500, 2000, 5000]:

                    h_new = lbfgsb_optimize(h_pert, n, beta, maxiter=400, pen=500.0)
                    c_new = compute_c5(h_new)

                        best_n = n

    if tlimit() > 30:

        n = best_n
