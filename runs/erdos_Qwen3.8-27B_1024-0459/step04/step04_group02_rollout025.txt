    weights = exp_vals / wsum
    
    val = (m + np.log(wsum) / beta) * dx
    
    grad = np.zeros(n)
    ncorr = 2 * n - 1
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def slsqp_refine(h, n, beta, maxiter=200):
    def obj(x):
        return smooth_objective_and_grad(x, n, beta)[0]
    def grad(x):
        return smooth_objective_and_grad(x, n, beta)[1]
    def cfun(x):
        return np.sum(x) - n / 2.0
    def cjac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            return trial
    except Exception:
        pass
    return h.copy()

def lbbf_refine(h, n, beta, maxiter=300, mu=1000.0):
    """L-BFGS-B with penalty for the sum constraint."""
    def obj(x):
        base_val, base_grad = smooth_objective_and_grad(x, n, beta)
        penalty = mu * (np.sum(x) - n / 2.0) ** 2
        return base_val + penalty
    def grad(x):
        _, base_grad = smooth_objective_and_grad(x, n, beta)
        penalty_grad = 2 * mu * (np.sum(x) - n / 2.0) * np.ones(n)
        return base_grad + penalty_grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            return trial
    except Exception:
        pass
    return h.copy()

```python
def compute_linear_coeffs(best, i, n):
    ncorr = 2 * n - 1
    lin = np.zeros(ncorr)
    lin[n-1] = 1.0
    
    # Positive offsets (k > n-1)
    if i < n - 1:
        lin[n : n + (n-1-i)] += (1.0 - best[i+1 : n])
    if i > 0:
        lin[n : n + i] += (-best[0 : i][::-1])
    
    # Negative offsets (k < n-1)
    if i > 0:
        lin[n-1-i : n-1] += (1.0 - best[0 : i])
    if i < n - 1:
        lin[i : n-1] += (-best[i+1 : n][::-1])
    
    return lin
```
