import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_N_WORK = 0
_BETA_WORK = 1000.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Distribute evenly among non-boundary elements
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            # All at boundary; adjust the "closest to middle" one
            idx = int(np.argmin(np.abs(h - 0.5)))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        # Spread over all elements proportionally
        for j in range(n):
            adj = diff / (n - j)
            h[j] = np.clip(h[j] + adj, 0.0, 1.0)
            diff = target - np.sum(h)
            if abs(diff) < 1e-14:
                break
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def _smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def slsqp_refine(h, n, beta, maxiter):

    _N_WORK = n
    _BETA_WORK = beta

    try:

            if obj < best_c5 - 1e-15:
                # Verify with full update
                # Update corr incrementally
                old_corr = corr.copy()  # expensive!
                # Actually, let's just do the full update at the end
                best_c5 = obj
                best_t = tc
        
        if best_t != t0:
            # Apply the change
            delta = best_t - t0
            if abs(delta) > 1e-14:
                # Update corr incrementally
                for off in range(-(n-1), n):
                    k = n - 1 + off
                    if off == 0:
                        corr[k] += best_t*(1-best_t) - t0*(1-t0)
                    else:
                        if 0 <= i+off < n:
                            corr[k] += delta * (1 - h[i+off])
                        if 0 <= i-off < n:
                            corr[k] += -delta * h[i-off]
                h[i] = best_t
                c5 = np.max(corr) * (2.0/n)
    
    return h, c5, corr

Actually, the incremental update for changing h[i] from t0 to t:
- For each off, corr[n-1+off] changes by:
  - If off==0: t*(1-t) - t0*(1-t0)
  - Else: delta * [(1-h[i+off])*I(0≤i+off<n) - h[i-off]*I(0≤i-off<n)]

The term in brackets depends on off. Let me precompute it:
- For off > 0: (1-h[i+off]) if i+off < n, else 0
- For off < 0: -h[i-off] if i-off ≥ 0, else 0
- For off = 0: special (quadratic)

So the update vector is:
update[off] = delta * (1-h[i+off]) for off > 0, i+off < n
update[off] = delta * (-h[i-off]) for off < 0, i-off ≥ 0
update[0] = t*(1-t) - t0*(1-t0)

def make_update_vec(i, t0, t, h, n):
    delta = t - t0
    if abs(delta) < 1e-15:
        return None
    vec = np.zeros(2*n-1)
    nminus = n - 1
    # Positive offsets
    pos_start = max(0, n - i)  # off > 0 means i+off < n, so off < n-i
    # off ranges from 1 to min(n-1, n-1-i)

    if max_pos_off >= 1:
        vec[nminus+1:nminus+1+max_pos_off] = delta * (1.0 - h[i+1:i+1+max_pos_off])
    # Negative offsets
    max_neg_off = min(n-1, i)

        # That's vec[n-1-max_neg_off : n-1] = -delta * h[i-max_neg_off : i]
    # Zero offset
    if True:  # off=0 always valid
