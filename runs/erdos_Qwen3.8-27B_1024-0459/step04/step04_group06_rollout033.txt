        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)

                best_h = h_interp.copy()
                n = n_try
                print("  Improved at n={}: C5={:.10f}".format(n_try, c5_h))
            else:
                print("  n={}: C5={:.10f} (no improvement)".format(n_try, c5_h))

    # Phase 8: Coordinate descent
    print("\n--- Phase 8: Coordinate descent ---")
    if remaining_time() > 10:
        best_h, best_c5 = coordinate_descent_quick(best_h, n, max_sweeps=10, rng=rng)
        print("  C5={:.10f}".format(best_c5))

    # Phase 9: Final SLSQP with very high beta
    print("\n--- Phase 9: Final high-beta polish ---")
    for beta_val in [10000, 30000, 50000]:
        if remaining_time() < 5:
            break
        h_new, c5_new = slsqp_smooth(best_h, n, beta_val, maxiter=500)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new

    # Phase 10: Tiny subgradient final
    if remaining_time() > 8:
        best_h, best_c5 = subgrad_optimize(best_h, n, max_iter=100, step0=0.03)

    # Final validation
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(n, final_c5))
    return (best_h, final_c5, n)
