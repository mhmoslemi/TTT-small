The function f(h) = max_s C_s(h) is the max of bilinear functions. Its subgradient at h is the gradient of the active C_s. I can use:

1. **Mirror descent**: Update h in a way that respects [0,1] constraints naturally.
2. **FISTA-style**: Fast iterative shrinkage for the subgradient.
3. **Frank-Wolfe (conditional gradient)**: Exploit the linear structure of each C_s.

Actually, Frank-Wolfe might work well here! The feasible set is the intersection of [0,1]^n and {sum = n/2}, which has a simple Frank-Wolfe direction computation.

Wait, Frank-Wolfe is for convex problems. Our problem is non-convex (max of bilinear functions). But we can still use a conditional gradient step as a heuristic.

At point h, the Frank-Wolfe direction is:
v = argmin_{z ∈ [0,1]^n, sum(z)=n/2} ∇f(h)^T z

where ∇f(h) is a subgradient. The solution is: set z[i] = 1 for the top n/2 indices of (-∇f(h)), z[i] = 0 for the rest (assuming ∇f(h) has distinct entries).

Then: h_new = (1-γ)h + γ*v, where γ is a step size.

This is O(n log n) per step (for the sorting). Very fast!

Let me also think about a key insight: **the problem has a lot of symmetry**. If h is optimal, then the "complement" 1-h is also optimal (since C_s(1-h) = C_s(h) by symmetry of the correlation). Also, if h is optimal, its reverse is also optimal (C_s(h_reversed) = C_{-s}(h)).

This means at the optimum, we might have C_s = C_{-s} (the correlation is even). So we can focus on s ≥ 0.

Also, **shift invariance**: If h is defined on [0,2] (not cyclic), shifting h changes the valid ranges for different lags. So shift invariance doesn't directly apply.

Let me write the code now. I'll aim for a clean, efficient implementation.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def get_subgradient(h, n):
    """Compute subgradient of max_s C_s at h. Returns (gradient, active_lag)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    s_star = int(np.argmax(corr))
    s = s_star - (n - 1)
    j_lo = max(0, -s)
    j_hi = min(n - 1, n - 1 - s)
    grad = np.zeros(n)
    for j in range(j_lo, j_hi + 1):
        grad[j] += (1.0 - h[j + s])
        grad[j + s] -= h[j]
    return grad, s_star, corr[s_star]

def smooth_max_obj(x, n, beta, target, pen=0.0):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = corr - m
    e = np.exp(np.clip(beta * s, -400, 400))
    val = (m + np.log(np.sum(e)) / beta) * (2.0 / n)
    if pen > 0:
        val += pen * (np.sum(x) - target) ** 2
    return float(val)

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def slsqp_opt(h0, n, beta, maxiter=300):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_max_obj(x, n, beta, target, pen=0.0)
    def con_fun(x):
        return np.sum(x) - target
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)

def subgradient_method(h0, n, tlimit, max_iter=50000):
    """Projected subgradient method with decreasing step size."""
    h = h0.copy()
    c5 = compute_c5(h)
    h_best = h.copy()
    c5_best = c5
    # Initial step size
    eta0 = 0.5
    for t in range(1, max_iter + 1):
        if tlimit() < 2.0:
            break
        grad, s_star, corr_val = get_subgradient(h, n)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        # Polyak-like step size
        eta = eta0 / np.sqrt(t)
        h_new = h - eta * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-12:
            c5_best = c5_new
            h_best = h_new.copy()
        h = h_new
        # Early stop if no improvement for long
        if t % 1000 == 0 and c5_new >= c5_best * (1 - 1e-10):
            # Check convergence
            if t > 5000:
                break
    return h_best, c5_best

def frank_wolfe_step(h, n, grad):
    """Compute Frank-Wolfe direction: binary vector with n/2 ones at indices of -grad."""
    # minimize grad^T z subject to z ∈ {0,1}^n, sum(z) = n/2
    # Solution: set z=1 where grad is smallest
    idx = np.argsort(grad)[:n // 2]
    v = np.zeros(n)
    v[idx] = 1.0
    return v

def fw_subgradient(h0, n, tlimit, max_iter=20000):
    """Frank-Wolfe + subgradient hybrid."""
    h = h0.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    # Momentum
    h_mom = h.copy()
    mu = 0.5  # momentum parameter
    for t in range(1, max_iter + 1):
        if tlimit() < 2.0:
            break
        grad, s_star, corr_val = get_subgradient(h, n)
        v = frank_wolfe_step(h, n, grad)
        gamma = 2.0 / (t + 2)  # FW step size
        h_new = (1 - gamma) * h + gamma * v
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-12:
            c5_best = c5_new
            h_best = h_new.copy()
        h_mom = (1 - mu) * h_mom + mu * h_new
        h = h_mom.copy()
        if t > 10000 and c5_new >= c5_best * (1 - 1e-12):
            break
    return h_best, c5_best

def coord_descent(h, n, max_sweeps, rng, tlimit):
    c5 = compute_c5(h)
    h_best = h.copy()
    for sweep in range(max_sweeps):
        if tlimit() < 2.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if tlimit() < 1.0:
                break
            v0 = h_best[i]
            # Try moving towards 0, 1, and intermediate values
            for v in [0.0, 1.0]:
                if abs(v - v0) < 1e-9:
                    continue
                trial = h_best.copy()
                trial[i] = v
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 5e-9:
                    c5 = c
                    h_best = trial.copy()
                    improved = True
            # Try small perturbations
            for d in [0.1, -0.1, 0.05, -0.05, 0.2, -0.2, 0.02, -0.02]:
                val = v0 + d
                if val < 0.0 or val > 1.0:
                    continue
                trial = h_best.copy()
                trial[i] = val
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 5e-9:
                    c5 = c
                    h_best = trial.copy()
                    improved = True
        if not improved and sweep >= 3:
            break
    return h_best, c5

def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c else 0.0
    return project_feasible(h, n)

def make_rand_bin(rng, n):

    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h
