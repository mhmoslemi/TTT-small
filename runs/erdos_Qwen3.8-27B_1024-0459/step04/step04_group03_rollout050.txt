def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_value(corr, beta):
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted -= np.max(shifted)
    return max_c + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(corr, beta):
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted -= np.max(shifted)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    return weights

def compute_grad_h(corr, h, n, weights):
    dx = 2.0 / n
    grad = np.zeros(n, dtype=float)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-16:
            continue
        s = m - n + 1
        if s >= 0:
            j1_lo, j1_hi = s, n
            if j1_lo < j1_hi:
                grad[j1_lo:j1_hi] += w * (1.0 - h[j1_lo - s : j1_hi - s])
            j2_lo, j2_hi = 0, n - s
            if j2_lo < j2_hi:
                grad[j2_lo:j2_hi] -= w * h[j2_lo + s : j2_hi + s]
        else:
            ms = -s
            j1_lo, j1_hi = 0, n - ms
            if j1_lo < j1_hi:

            j2_lo, j2_hi = ms, n
            if j2_lo < j2_hi:

def slsqq_smooth_max(h_init, n, beta, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return smooth_max_value(corr, beta) * dx
    def eq_con(x):
        return np.sum(x) - n / 2.0

        if res.x is not None:

    except Exception:
        pass
    return h_init.copy()
