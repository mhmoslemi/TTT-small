            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                print("  Multi-start {}: C5={:.10f}".format(attempt, best_c5))

    # Phase 7: Try complement symmetry
    if remaining_time() > 30.0:
        h_comp = 1.0 - best_h
        h_comp = project_feasible(h_comp, best_n)
        c5_comp = compute_c5(h_comp)
        # Try optimizing the complement too
        h_comp, c5_comp = slsqp_multi2(h_comp, best_n, [500, 5000], maxiter=200)
        if c5_comp < best_c5 - 1e-12:
            best_c5 = c5_comp
            best_h = h_comp.copy()
            print("Complement improved: C5={:.10f}".format(best_c5))

    # Phase 8: Higher resolution
    if remaining_time() > 150.0:
        best_n_orig = best_n
        for n_try in [320, 384, 512]:
            if remaining_time() < 60.0:
                break
            h_h = project_feasible(interp_to(best_h, best_n_orig, n_try), n_try)
            _G_N = n_try
            _G_TGT = n_try / 2.0
            _G_DX = 2.0 / n_try
            c5_h = compute_c5(h_h)
            h_h, c5_h = subgrad_loop(h_h, n_try, max_iter=200, top_k=5, step0=0.08)
            h_h, c5_h = slsqp_multi2(h_h, n_try, [100, 500, 2000, 10000], maxiter=250)
            h_h, c5_h = focused_local_search(h_h, n_try, rng, max_ops=3000)
            if c5_h < best_c5 - 1e-10:
                best_c5 = c5_h
                best_h = h_h.copy()
                best_n = n_try
                print("  Higher res n={}: C5={:.10f}".format(n_try, best_c5))
            else:
                print("  Higher res n={}: C5={:.10f}".format(n_try, c5_h))

    # Phase 9: Also try lower resolution (sometimes helps)
    if remaining_time() > 60.0 and best_n > 256:
        for n_try in [192, 256]:
            if remaining_time() < 30.0:
                break
            h_h = project_feasible(interp_to(best_h, best_n, n_try), n_try)
            _G_N = n_try
            _G_TGT = n_try / 2.0
            _G_DX = 2.0 / n_try
            c5_h = compute_c5(h_h)
            h_h, c5_h = subgrad_loop(h_h, n_try, max_iter=150, top_k=4, step0=0.1)
            h_h, c5_h = slsqp_multi2(h_h, n_try, [500, 5000, 50000], maxiter=300)
            if c5_h < best_c5 - 1e-10:
                best_c5 = c5_h
                best_h = h_h.copy()
                best_n = n_try
                print("  Lower res n={}: C5={:.10f}".format(n_try, c5_h))

    # Phase 10: Final intensive local search
    if remaining_time() > 20.0:
        best_h, best_c5 = focused_local_search(best_h, best_n, rng, max_ops=15000)
        print("Phase 10 (final LS): C5={:.10f}".format(best_c5))

    # Phase 11: Final SLSQP with extreme beta
    if remaining_time() > 15.0:
        best_h, best_c5 = slsqp_multi2(best_h, best_n, [50000, 200000, 500000], maxiter=500)
        print("Phase 11 (final SLSQP): C5={:.10f}".format(best_c5))

    # Phase 12: Tiny final adjustments
    if remaining_time() > 8.0:
        best_h, best_c5 = focused_local_search(best_h, best_n, rng, max_ops=5000)

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
