def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2} using binary search on Lagrange multiplier."""
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    
    # Binary search for t such that sum(clip(h + t, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    h = np.clip(h + t, 0, 1)
    # Fix any remaining tiny error
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        h[np.argmin(np.abs(h - 0.5))] += diff
        h = np.clip(h, 0, 1)
    return h

import numpy as np
from scipy.optimize import minimize, lbfgsb
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2} via binary search on shift."""
    h = np.array(h, dtype=float)
    target = n / 2.0
    # Binary search for t: sum(clip(h+t, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Tiny residual fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        # Adjust the element closest to 0.5
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def remaining(budget_s, t0):
    return max(0.0, budget_s - (time.time() - t0) - 2.0)

def smooth_max_obj_grad(h, beta, target_sum):
    """Returns (objective_value, gradient) for smooth max + penalty."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(shifted)
    sum_ex = np.sum(ex)
    w = ex / sum_ex
    val = (m + np.log(sum_ex) / beta) * dx
    
    # Penalty for sum constraint
    s = np.sum(h)
    penalty = 1000.0 * (s - target_sum) ** 2
    val_pen = val + penalty
    
    # Gradient of smooth max
    g = np.zeros(n)
    for off in range(0, n):
        k = n - 1 + off
        wk = w[k] * dx
        if wk > 1e-15:
            np_ = n - off
            g[off:n] += wk * (1.0 - h[0:np_])
            g[0:np_] -= wk * h[off:n]
    for a in range(1, n):
        k = n - 1 - a
        wk = w[k] * dx
        if wk > 1e-15:
            np_ = n - a
            g[0:np_] += wk * (1.0 - h[a:n])
            g[a:n] -= wk * h[0:np_]
    
    # Gradient of penalty
    g += 2.0 * 1000.0 * (s - target_sum)
    
    return val_pen, g

def slsqp_opt(h0, n, beta, maxiter=300, penalty=500.0):
    """SLSQP with smooth max + penalty."""
    target = n / 2.0
    h = project_feasible(h0.copy(), n)
    
    def obj(x):
        n_ = len(x)
        dx_ = 2.0 / n_
        corr_ = compute_corr(x)
        m_ = np.max(corr_)
        sh_ = np.clip(beta * (corr_ - m_), -50.0, 50.0)
        ex_ = np.exp(sh_)
        val = (m_ + np.log(np.sum(ex_)) / beta) * dx_
        s_ = np.sum(x)
        val += penalty * (s_ - n_ / 2.0) ** 2
        return val
    
    def grad(x):
        n_ = len(x)
        dx_ = 2.0 / n_
        corr_ = compute_corr(x)
        m_ = np.max(corr_)
        sh_ = np.clip(beta * (corr_ - m_), -50.0, 50.0)
        ex_ = np.exp(sh_)
        w_ = ex_ / np.sum(ex_)
        g_ = np.zeros(n_)
        for off in range(0, n_):
            k_ = n_ - 1 + off
            wk_ = w_[k_] * dx_
            if wk_ > 1e-15:
                np_ = n_ - off
                g_[off:n_] += wk_ * (1.0 - x[0:np_])
                g_[0:np_] -= wk_ * x[off:n_]
        for a_ in range(1, n_):
            k_ = n_ - 1 - a_
            wk_ = w_[k_] * dx_
            if wk_ > 1e-15:
                np_ = n_ - a_
                g_[0:np_] += wk_ * (1.0 - x[a_:n_])
                g_[a_:n_] -= wk_ * x[0:np_]
        s_ = np.sum(x)
        g_ += 2.0 * penalty * (s_ - n_ / 2.0)
        return g_
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-16})
        h_new = project_feasible(res.x, n)
        if compute_c5(h_new) < compute_c5(h):
            return h_new
    except Exception:
        pass
    return h
