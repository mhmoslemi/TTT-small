import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

_G_N = 0
_G_BETA = 1.0
_G_TGT = 0.0
_G_START = 0.0
_G_BUDGET = 1300.0
_G_DX = 1.0
_G_PEN = 1e4


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        rooms = np.minimum(h, 1.0 - h)
        idx = int(np.argmax(rooms))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 8.0)


def obj_lse_pen(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _G_BETA
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 0))
    lse_val = (m + np.log(np.sum(exps)) / beta) * dx
    sum_pen = _G_PEN * (np.sum(x) - _G_TGT) ** 2
    return lse_val + sum_pen


def obj_lse_pen_grad(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _G_BETA
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 0))
    probs = exps / np.sum(exps)
    nfull = len(corr)
    n_minus_1 = n - 1
    grad_corr = np.zeros((nfull, n))
    for k_idx in range(nfull):
        offset = k_idx - n_minus_1
        if offset >= 0:
            npair = n - offset
            if npair <= 0:
                continue
            grad_corr[k_idx, :npair] = (1.0 - x[offset:offset + npair])
            grad_corr[k_idx, offset:] = -x[:npair]
        else:
            abs_off = -offset
            npair = n - abs_off
            if npair <= 0:
                continue
            grad_corr[k_idx, :npair] = -x[abs_off:abs_off + npair]
            grad_corr[k_idx, abs_off:] = (1.0 - x[:npair])
    grad_lse = (probs @ grad_corr) * dx
    sum_term = 2.0 * _G_PEN * (np.sum(x) - _G_TGT)
    grad = grad_lse + sum_term * np.ones(n)
    val = (m + np.log(np.sum(exps)) / beta) * dx + _G_PEN * (np.sum(x) - _G_TGT) ** 2
    return val, grad


def obj_slsqp(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _G_BETA
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 0))
    return (m + np.log(np.sum(exps)) / beta) * dx


def con_sum(x):
    return np.sum(x) - _G_TGT


def compute_subgrad_h(h, n, top_k):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
    grad = np.zeros(n)
    n_minus_1 = n - 1
    for k_idx in top_idx:
        offset = int(k_idx) - n_minus_1
        if offset >= 0:
            npair = n - offset
            if npair <= 0:
                continue
            grad[:npair] += (1.0 - h[offset:offset + npair])
            grad[offset:] -= h[:npair]
        else:
            abs_off = -offset
            npair = n - abs_off
            if npair <= 0:
                continue
            grad[:npair] -= h[abs_off:abs_off + npair]
            grad[abs_off:] += (1.0 - h[:npair])
    return grad * dx, top_idx, corr


def do_lbfgsb_phase(h_start, n, budget_fn, beta_list, pen=1e4):
    global _G_N, _G_BETA, _G_TGT, _G_DX, _G_PEN
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    _G_PEN = pen
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for beta in beta_list:
        if budget_fn() < 8.0:
            break
        _G_BETA = float(beta)
        try:
            maxiter = 500
            est_time = budget_fn() * 0.3
            maxiter = min(maxiter, max(50, int(est_time / 0.5)))
            res = minimize(obj_lse_pen_grad, h_cur, jac=True, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, c5_cur


def do_slsqp_phase(h_start, n, budget_fn, beta_list):
    global _G_N, _G_BETA, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    for beta in beta_list:
        if budget_fn() < 6.0:
            break
        _G_BETA = float(beta)
        try:
            maxiter = 300
            est_time = budget_fn() * 0.35
            maxiter = min(maxiter, max(30, int(est_time / 0.4)))
            res = minimize(obj_slsqp, h_cur, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, c5_cur


def do_nesterov_subgrad(h_best, n, budget_fn, rng, max_iters=500, top_k_start=3, top_k_max=15):
    h = h_best.copy()
    c5_best = compute_c5(h)
    t = 1.0
    x_prev = h.copy()
    no_improve = 0
    base_step = 0.04
    for it in range(max_iters):
        if budget_fn() < 3.0:
            break
        frac = it / max(max_iters - 1, 1)
        top_k = top_k_start + int(frac * (top_k_max - top_k_start))
        top_k = max(2, min(top_k, top_k_max))
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        beta_m = (t - 1.0) / t_new
        y = np.clip(h + beta_m * (h - x_prev), 0.0, 1.0)
        grad, top_idx, corr = compute_subgrad_h(y, n, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        c5_y = float(np.max(corr)) * (2.0 / n)
        improved = False
        step = base_step / (1.0 + 0.005 * it)
        for _ls in range(15):
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-14:
                c5_best = c5_trial
                h_best = h_trial.copy()
                improved = True
                break
            step *= 0.65
        if improved:
            no_improve = 0
            x_prev = h.copy()
            h = h_best.copy()
        else:
            no_improve += 1
            x_prev = h.copy()
            if no_improve > 50 and rng.random() < 0.4:
                ns = 0.04 / (1.0 + 0.05 * no_improve)
                h_pert = project_feasible(h + rng.randn(n) * ns, n)
                c5_pert = compute_c5(h_pert)
                if c5_pert < c5_best:
                    c5_best = c5_pert
                    h_best = h_pert.copy()
                    h = h_pert.copy()
                    x_prev = h.copy()
                    no_improve = 0
        t = t_new
    return h_best, c5_best


def do_coord_block(h_best, n, budget_fn, rng, block_size=4, max_sweeps=20):
    h = h_best.copy()
    c5_best = compute_c5(h)
    for sweep in range(max_sweeps):
        if budget_fn() < 3.0:
            break
        any_imp = False
        idx = rng.permutation(n)
        for b in range(0, n, block_size):
            if budget_fn() < 2.0:
                break
            sel = idx[b:b + block_size]
            if len(sel) < 2:
                continue
            old_vals = h[sel].copy()
            for _attempt in range(3):
                pert = rng.randn(len(sel)) * 0.1
                new_vals = np.clip(old_vals + pert, 0.0, 1.0)
                h[sel] = new_vals
                h = project_feasible(h, n)
                c5_test = compute_c5(h)
                if c5_test < c5_best - 1e-14:
                    c5_best = c5_test
                    h_best = h.copy()
                    any_imp = True
                else:
                    h = h_best.copy()
        if not any_imp and sweep > 4:
            break
    return h_best, c5_best


def make_thue_morse(n):
    h = np.zeros(n)
    half = n // 2
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)
