        g[ao:ao+npairs] -= x[:npairs]
    return g

def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g

def lbfgs_optimize(h, n, time_limit=120, betas=(100, 300, 800, 2000, 5000, 15000)):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx_val = 2.0 / n
    target = n / 2.0
    start = time.time()

    for beta in betas:
        if time.time() - start > time_limit or remaining() < 10:
            break

        def obj_and_grad(x):
            corr = compute_corr(x)
            m = np.max(corr)
            shifted = corr - m
            cl = np.clip(beta * shifted, -50, 50)
            ex = np.exp(cl)
            s = np.sum(ex)
            if s < 1e-300:
                return m * dx_val, np.zeros(n)
            w = ex / s
            smooth_val = (m + np.log(s) / beta) * dx_val

            g = np.zeros(n)
            for k in range(len(corr)):
                if w[k] > 1e-12:
                    g += w[k] * corr_grad_per_shift(x, n, int(k))
            g *= dx_val
            return smooth_val, g

        # Add sum constraint via penalty
        lam = 1e4
        def obj_pen(x):
            v, g = obj_and_grad(x)
            sp = lam * (np.sum(x) - target) ** 2
            g = g + 2 * lam * (np.sum(x) - target) * np.ones(n)
            return v + sp, g

        res = minimize(obj_pen, h, jac=True, method='L-BFGS-B',
                       bounds=[(0, 1)] * n,
                       options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-12,
                                'factr': 1e6, 'pgtol': 1e-14})

        if c5_new < best_c5:

            best_h = h_new.copy()
            h = h_new.copy()

    return best_h, best_c5

        # Targeted SA at n=720 (binary)
        if remaining() > 40:
            h_bin720 = _make_binary(best_h, n_up, rng)
            for sa_seed in [seed + 300, seed + 400]:
                if remaining() < 20:
                    break
                t_lsta = min(50, max(15, remaining() * 0.15))
                h_sa, c5_sa = targeted_binary_sa(h_bin720, n_up, time_limit=t_lsta, seed=sa_seed)
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
                    best_n = n_up

    # === Phase 5: Try n=600 as well ===
    n_mid = 600
    if remaining() > 100:
        h_mid = interp_to(best_h, best_n, n_mid)
        h_mid = project_feasible(h_mid, n_mid)
        t_lim = min(100, max(30, remaining() * 0.25))
        h_opt, c5_opt = lbfgs_optimize(h_mid, n_mid, time_limit=t_lim,
                                       betas=(200, 800, 3000, 10000))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_mid

        if remaining() > 30:
            h_bin600 = _make_binary(best_h, n_mid, rng)
            t_lsta = min(40, max(15, remaining() * 0.2))
            h_sa, c5_sa = targeted_binary_sa(h_bin600, n_mid, time_limit=t_lsta, seed=seed + 500)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_mid

    # === Phase 6: Top-k gradient descent on best ===
    if remaining() > 30:
        t_lim = min(60, remaining() * 0.3)
        h_td, c5_td = topk_descent(best_h, best_n, max_iters=400, top_k=5, time_limit=t_lim)
        if c5_td < best_c5:
            best_c5 = c5_td
            best_h = h_td.copy()

    # === Phase 7: SLSQP final polish ===
    if remaining() > 30:
        for beta in [10000, 50000]:
            if remaining() < 10:
                break
            t_lim = min(60, max(15, remaining() * 0.2))
            h_opt, c5_opt = slsqp_optimize(best_h, best_n, time_limit=t_lim, beta=beta)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # === Phase 8: Final targeted SA ===
    if remaining() > 25:
        h_bin = _make_binary(best_h, best_n, rng)
        t_lim = min(40, max(10, remaining() * 0.5))
        for sa_seed in [seed + 700, seed + 800, seed + 900]:
            if remaining() < 10:
                break
            h_sa, c5_sa = targeted_binary_sa(h_bin, best_n, time_limit=max(10, t_lim / 3),
                                             seed=sa_seed)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = best_n

    # === Phase 9: Final top-k polish ===
    if remaining() > 10:
        h_td, c5_td = topk_descent(best_h, best_n, max_iters=200, top_k=3,
                                   time_limit=min(30, remaining() * 0.5))
        if c5_td < best_c5:
            best_c5 = c5_td
            best_h = h_td.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
