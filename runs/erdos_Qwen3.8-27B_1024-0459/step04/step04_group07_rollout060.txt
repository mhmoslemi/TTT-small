def subgradient_step(h, n, step_size, rng):
    """Take a subgradient step for the max correlation problem."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_active = int(np.argmax(corr))
    s_val = k_active - (n - 1)
    
    # Compute gradient of corr at active lag
    grad = np.zeros(n)
    if s_val >= 0:
        js = np.arange(s_val, n)
        grad[js] = 1.0 - h[js - s_val]
        if s_val > 0:
            js2 = np.arange(0, n - s_val)
            grad[js2] -= h[js2 + s_val]
        else:
            grad -= h
    else:
        neg = -s_val
        js = np.arange(0, n - neg)
        grad[js] = 1.0 - h[js + neg]
        js2 = np.arange(neg, n)
        grad[js2] -= h[js2 - neg]
    
    # Add small noise to avoid getting stuck
    if rng is not None:
        grad += rng.randn(n) * 0.01
    
    # Line search
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for alpha in [step_size, step_size * 0.5, step_size * 0.1, step_size * 0.01]:
        trial = h - alpha * grad
        trial = project_feasible(trial, n)
        c = compute_c5(trial)
        if c < best_c5:
            best_h = trial
            best_c5 = c
    return best_h, best_c5
