        grad = subgrad_step(h, n, 5)
        if grad is None:
            break
        base_step = 0.04 / (1.0 + it * 0.005)
        improved = False
        for sm in [1.0, 0.5, 2.0, 0.25, 0.75, 0.1]:
            step = base_step * sm
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-16:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
        if not improved:
            if it % 100 == 50 and remaining_time() > 30:
                h_new = slsqp_optimize(h, n, 50000, 300)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                if it > 300:
                    break

    # Phase 6: Try higher resolutions
    higher_res = set()
    for mult in [1.2, 1.35, 1.5, 2.0]:
        rn = int(best_n * mult)
        if rn % 2 != 0:
            rn += 1
        if 64 <= rn <= 1600:
            higher_res.add(rn)
    for extra in [128, 256]:
        rn = best_n + extra
        if rn % 2 != 0:
            rn += 1
        if 64 <= rn <= 1600:
            higher_res.add(rn)
    higher_res = sorted(higher_res)

    for rn in higher_res:
        if remaining_time() < 60:
            break
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)

        for beta in [3000, 15000, 80000]:
            if remaining_time() < 20:
                break
            h_r = slsqp_optimize(h_r, rn, beta, 300)

        if remaining_time() > 20:
            h_r, c5_r = coord_descent(h_r, rn, n_sweeps=25)

        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn

    # Phase 7: Random perturbations around best
    n = best_n
    h = best_h.copy()
    n_perturb = max(10, n // 30)
    for pert in range(30):
        if remaining_time() < 25:
            break
        h_pert = h.copy()
        idx = rng.choice(n, n_pert, replace=False)
        h_pert[idx] += rng.uniform(-0.15, 0.15, n_pert)
        h_pert = project_feasible(h_pert, n)
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5 + 0.005:
            h_new = slsqp_optimize(h_pert, n, 30000, 250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 8: Final intensive coordinate descent
    if remaining_time() > 15:
        h_new, c5_new = coord_descent(best_h, best_n, n_sweeps=40)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 9: Final SLSQP polish at very high beta
    if remaining_time() > 8:
        h_new = slsqp_optimize(best_h, best_n, 1000000, 200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 10: One final coordinate descent
    if remaining_time() > 5:
        h_new, c5_new = coord_descent(best_h, best_n, n_sweeps=20)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    # Ensure sum constraint is met exactly
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        if np.sum(free) > 0:
            best_h[free] += diff / np.sum(free)
            best_h = np.clip(best_h, 0.0, 1.0)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
