    return np.correlate(h, 1.0 - h, mode='full')

def corr_gradient(h):
    """Returns gradient matrix G where G[m, i] = d(corr[m])/d(h[i])"""
    n = len(h)
    nc = 2 * n - 1
    G = np.zeros((nc, n))
    for m in range(nc):
        s = m - (n - 1)  # shift
        # d(corr[m])/d(h[i]) = (1-h[i+s]) - h[i-s] with bounds
        for i in range(n):
            val = 0.0
            js = i + s
            if 0 <= js < n:
                val += (1.0 - h[js])
            js2 = i - s
            if 0 <= js2 < n:
                val -= h[js2]
            G[m, i] = val
    return G
```

    nc = 2 * n - 1
    grads = np.zeros((nc, n))
    for m in range(nc):
        s = m - (n - 1)
        if s >= 0:
            # (1-h[i+s]) for i in [0, n-1-s]
            grads[m, :n-s] += (1.0 - h[s:n])
            # -h[i-s] for i in [s, n-1]
            grads[m, s:] -= h[:n-s]
        else:

            # (1-h[i-t]) for i in [t, n-1]
            grads[m, t:] += (1.0 - h[:n-t])
            # -h[i+t] for i in [0, n-1-t]
            grads[m, :n-t] -= h[t:n]
    return grads
```

def make_rand_bin(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def active_lag_opt(h, n, rng, tlimit, max_rounds=30):
    c5 = compute_c5(h)
    h_best = h.copy()
    for rnd in range(max_rounds):
        if tlimit() < 3.0:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode='full')
        top_k = np.argsort(corr)[-3:]
        for k_act in top_k:
            if tlimit() < 2.0:
                break
            _G_init(n, 500.0, 0.0)
            bounds = [(0.0, 1.0)] * n
            eq_cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                       'jac': lambda x: np.ones(n)}
            try:
                res = minimize(smooth_max_obj, h_best, jac=smooth_max_grad,
                               method='SLSQP', bounds=bounds,
                               constraints=[eq_cons],
                               options={'maxiter': 150, 'ftol': 1e-15, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new)
                if c_new < c5 - 1e-12:
                    c5 = c_new
                    h_best = h_new
            except Exception:
                pass
        if tlimit() < 5.0:
            break
    return h_best, c5

def coord_descent(h, n, max_sweeps, rng, tlimit):
    c5 = compute_c5(h)
    h_best = h.copy()
    deltas = [0.2, -0.2, 0.1, -0.1, 0.05, -0.05, 0.3, -0.3, 0.0, 1.0]
    for sweep in range(max_sweeps):
        if tlimit() < 3.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if tlimit() < 1.5:
                break
            v0 = h_best[i]
            for d in deltas:
                trial = h_best.copy()
                trial[i] = np.clip(trial[i] + d, 0.0, 1.0)
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 1e-10:
                    c5 = c
                    h_best = trial.copy()
                    improved = True
        if not improved and sweep >= 2:
            break
    return h_best, c5

def binary_local_search(h, n, rng, tlimit, max_iters=5000):
    h_b = np.where(h > 0.5, 1.0, 0.0)
    h_b = project_feasible(h_b, n)
    c5_b = compute_c5(h_b)
    h_orig_c5 = compute_c5(h)
    if c5_b > h_orig_c5:
        return h, h_orig_c5
    best_h = h_b.copy()
    best_c5 = c5_b
    T = 0.001
    for it in range(max_iters):
        if tlimit() < 2.0:
            break
        T *= 0.999
        move_type = rng.randint(3)
        trial = best_h.copy()
        if move_type == 0:
            i = rng.randint(n)
            trial[i] = 1.0 - trial[i]
        elif move_type == 1:
            i = rng.randint(n - 1)
            if trial[i] != trial[i + 1]:
                trial[i], trial[i + 1] = trial[i + 1], trial[i]
        else:
            i = rng.randint(n - 2)
            if i + 2 < n and trial[i] == 1.0 and trial[i + 1] == 0.0 and trial[i + 2] == 0.0:
                trial[i] = 0.0
                trial[i + 2] = 1.0
            elif i + 2 < n and trial[i] == 0.0 and trial[i + 1] == 1.0 and trial[i + 2] == 1.0:
                trial[i] = 1.0
                trial[i + 2] = 0.0
            else:
                continue
        trial = project_feasible(trial, n)
        c_trial = compute_c5(trial)
        delta = c_trial - best_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-8)):
            best_h = trial
            best_c5 = c_trial
    if best_c5 < h_orig_c5:
        return best_h, best_c5
    return h, h_orig_c5

def multi_beta_slsqp(h, n, tlimit, betas=None):
    if betas is None:
        betas = [30, 60, 100, 200, 400, 800, 1500, 3000, 5000, 8000, 12000]
    c5 = compute_c5(h)
    h_best = h.copy()
    for bv in betas:
        if tlimit() < 10:
            break
        h_new = slsqp_opt(h_best, n, bv, maxiter=350)
        c_new = compute_c5(h_new)
        if c_new < c5 - 1e-13:
            c5 = c_new
            h_best = h_new
    return h_best, c5

def multi_beta_lbfgs(h, n, tlimit, betas=None):
    if betas is None:
        betas = [50, 100, 200, 500, 1000, 2000, 4000]
    c5 = compute_c5(h)
    h_best = h.copy()
    for bv in betas:
        if tlimit() < 10:
            break
        h_new = lbfgsb_opt(h_best, n, bv, maxiter=400, pen=500.0)
        c_new = compute_c5(h_new)
        if c_new < c5 - 1e-13:
            c5 = c_new
            h_best = h_new
        # Follow up with SLSQP polish for this n if small enough
        if n <= 320 and tlimit() > 15:
            h_new2 = slsqp_opt(h_best, n, bv, maxiter=200)
            c_new2 = compute_c5(h_new2)
            if c_new2 < c5 - 1e-13:
                c5 = c_new2
                h_best = h_new2
    return h_best, c5

def run(seed=42, budget_s=1300, **kwargs):
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    def tlimit():
        return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = make_tm(parent_n)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("No parent. TM: n={}, C5={:.8f}".format(parent_n, parent_c5))

    # Phase 1: SLSQP at parent resolution
    if tlimit() > 200:
        print("\n=== Phase 1: SLSQP n={} ===".format(best_n))
        n1 = best_n
        h1 = best_h.copy()
        h1, c5_1 = multi_beta_slsqp(h1, n1, tlimit,
                                    betas=[20, 50, 100, 200, 500, 1000, 2000, 4000, 8000, 15000])
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1
            best_n = n1
        print("  Phase 1 done: C5={:.8f}".format(best_c5))

    # Phase 2: Active-lag focused optimization
    if tlimit() > 100:
        print("\n=== Phase 2: Active-lag focus n={} ===".format(best_n))
        h2, c5_2 = active_lag_opt(best_h, best_n, rng, tlimit, max_rounds=20)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
        print("  Phase 2 done: C5={:.8f}".format(best_c5))

    # Phase 3: Higher resolution L-BFGS-B
    for n_try in [512, 1024]:
        if tlimit() < 120:
            break
        print("\n=== Phase 3: L-BFGS-B n={} ===".format(n_try))
        h3 = project_feasible(interp_h(best_h, best_n, n_try), n_try)
        c5_3 = compute_c5(h3)
        # Also try a fresh initialization
        h3_b = project_feasible(make_tm(n_try), n_try)
        c5_3b = compute_c5(h3_b)
        h3_r = project_feasible(make_rand_bin(rng, n_try), n_try)
        c5_3r = compute_c5(h3_r)
        # Pick best init
        h_start = h3
        c5_start = c5_3
        if c5_3b < c5_start:
            h_start = h3_b
            c5_start = c5_3b
        if c5_3r < c5_start:
            h_start = h3_r
            c5_start = c5_3r
        h3, c5_3 = multi_beta_lbfgs(h_start, n_try, tlimit,
                                     betas=[30, 80, 200, 500, 1000, 2000, 4000])
        if c5_3 < best_c5 - 1e-10:
            best_c5 = c5_3
            best_h = h3
            best_n = n_try
            print("  n={} improved: C5={:.8f}".format(n_try, best_c5))
        else:
            print("  n={} best: C5={:.8f}".format(n_try, c5_3))

    # Phase 4: Back-interpolate to parent n and re-optimize
    if tlimit() > 80 and best_n > 256:
        print("\n=== Phase 4: Down-interp to n={} ===".format(256))
        h4 = project_feasible(interp_h(best_h, best_n, 256), 256)
        c5_4 = compute_c5(h4)
        h4, c5_4 = multi_beta_slsqp(h4, 256, tlimit,
                                     betas=[100, 500, 1000, 3000, 8000, 15000, 25000])
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4
            best_n = 256
        # Also try at 384
        if tlimit() > 40:
            h4b = project_feasible(interp_h(best_h, best_n, 384), 384)
            c5_4b = compute_c5(h4b)
            h4b, c5_4b = multi_beta_lbfgs(h4b, 384, tlimit,
                                           betas=[50, 200, 500, 1000, 3000])
            if c5_4b < best_c5:
                best_c5 = c5_4b
                best_h = h4b
                best_n = 384
        print("  Phase 4 done: C5={:.8f} (n={})".format(best_c5, best_n))

    # Phase 5: Perturbation escapes
    if tlimit() > 100:
        print("\n=== Phase 5: Perturbation escapes ===")
        n5 = best_n
        for attempt in range(12):
            if tlimit() < 20:
                break
            h_p = best_h.copy()
            n_pert = rng.randint(5, max(6, n5 // 10))
            idxs = rng.choice(n5, size=n_pert, replace=False)
            for idx in idxs:
                h_p[idx] = np.clip(h_p[idx] + rng.randn() * 0.4, 0.0, 1.0)
            h_p = project_feasible(h_p, n5)
            c5_p = compute_c5(h_p)
            if c5_p > best_c5 * 1.08:
                continue
            if n5 <= 320:
                h_p, c5_p = multi_beta_slsqp(h_p, n5, tlimit,
                                              betas=[100, 500, 2000, 8000])
            else:
                h_p, c5_p = multi_beta_lbfgs(h_p, n5, tlimit,
                                              betas=[100, 500, 2000, 5000])
            if c5_p < best_c5 - 1e-10:
                best_c5 = c5_p
                best_h = h_p
                best_n = n5
                print("  Perturb {}: C5={:.8f}".format(attempt, best_c5))

    # Phase 6: Coordinate descent
    if tlimit() > 60:
        print("\n=== Phase 6: Coordinate descent n={} ===".format(best_n))
        best_h, best_c5 = coord_descent(best_h, best_n, max_sweeps=50, rng=rng, tlimit=tlimit)
        print("  Coord descent: C5={:.8f}".format(best_c5))

    # Phase 7: Binary local search
    if tlimit() > 30:
        print("\n=== Phase 7: Binary local search ===")
        h_b, c5_b = binary_local_search(best_h, best_n, rng, tlimit, max_iters=8000)
        if c5_b < best_c5:
            best_h = h_b
            best_c5 = c5_b
            print("  Binary improved: C5={:.8f}".format(best_c5))
        else:
            print("  Binary: no improvement (C5={:.8f})".format(c5_b))

    # Phase 8: Final SLSQP polish
    if tlimit() > 15:
        print("\n=== Phase 8: Final polish ===")
        n8 = best_n
        for bv in [5000, 10000, 20000, 40000]:
            if tlimit() < 5:
                break
            if n8 <= 320:
                h_new = slsqp_opt(best_h, n8, bv, maxiter=500)
            else:
                h_new = lbfgsb_opt(best_h, n8, bv, maxiter=300, pen=1000.0)
                if n8 <= 512 and tlimit() > 10:
                    h_new = slsqp_opt(h_new, n8, bv, maxiter=300)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new
        print("  Final polish: C5={:.8f}".format(best_c5))

    # Phase 9: One more round of active-lag + coord
    if tlimit() > 20:
        best_h, best_c5 = active_lag_opt(best_h, best_n, rng, tlimit, max_rounds=10)
        if tlimit() > 15:
            best_h, best_c5 = coord_descent(best_h, best_n, max_sweeps=20, rng=rng, tlimit=tlimit)
        print("  Phase 9: C5={:.8f}".format(best_c5))

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
