def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    # Project onto {x: 0<=x<=1, sum(x)=target}
    # Bisection on lambda
    lo = -max(h.max(), 0) - 1.0
    hi = 1.0 - min(h.min(), 1.0) + 1.0
    lo = -2.0
    hi = 2.0
    for _ in range(100):

        x = np.clip(h - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine adjustment
    diff = target - h.sum()
    mask = (h > 1e-12) & (h < 1.0 - 1e-12)
    if mask.sum() > 0 and abs(diff) > 1e-12:
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

```python
def compute_delta(i, j, h, N, offset):
    """Delta at a specific shift offset."""
    affected_p = set()
    if 0 <= i - offset < N: affected_p.add(i - offset)
    if 0 <= j - offset < N: affected_p.add(j - offset)
    if 0 <= i < N and 0 <= i + offset < N: affected_p.add(i)
    if 0 <= j < N and 0 <= j + offset < N: affected_p.add(j)
    
    delta = 0.0
    for p in affected_p:
        po = p + offset
        if 0 <= po < N:
            # h_old[p+offset] * (1-h_old[p])
            old_h_po = h[po]
            old_h_p = h[p]
            # Apply flips
            if po == i: new_h_po = 1.0
            elif po == j: new_h_po = 0.0
            else: new_h_po = h[po]
            if p == i: new_h_p = 1.0
            elif p == j: new_h_p = 0.0
            else: new_h_p = h[p]
            delta += new_h_po * (1 - new_h_p) - old_h_po * (1 - old_h_p)
    return delta
```
