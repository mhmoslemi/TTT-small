  For lag < 0: j in [0, n-1-|lag|] = [0, n+lag-1], grad[j] += w[k]*(1-x[j+|lag|]) = w[k]*(1-x[j-lag])

- Second term: subtract w[k]*x[j+lag] from grad[j]
  0 <= j+lag < n → -lag <= j < n-lag
  For lag >= 0: j in [0, n-lag-1], grad[j] -= w[k]*x[j+lag]
  For lag < 0: j in [|lag|, n-1], grad[j] -= w[k]*x[j+lag] = w[k]*x[j-|lag|]

For lag < 0 (let p = -lag > 0):

  grad[p:n] -= w[k] * x[0:n-p]

        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def top_lag_gradient_descent(h, n, tlimit, max_iters=2000, lr=0.001):
    """Focus on reducing the top correlation lag(s) using gradient steps."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if tlimit() < 2.0:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        # Find top 3 lags
        top3 = np.argsort(corr)[::-1][:3]
        # Subgradient of max at these lags
        # For each top lag, compute gradient of that correlation
        n_half = n - 1
        grad = np.zeros(n)
        for k in top3:
            lag = k - n_half
            if lag >= 0:
                grad[lag:n] += (1.0 - best_h[0:n-lag])
                grad[0:n-lag] -= best_h[lag:n]
            else:
                p = -lag
                grad[0:n-p] += (1.0 - best_h[p:n])
                grad[p:n] -= best_h[0:n-p]
        
        # Normalized gradient step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        grad_norm = grad / gnorm
        
        # Adaptive step size
        step = lr * 2.0 / n
        trial = best_h - step * grad_norm
        trial = np.clip(trial, 0.0, 1.0)
        trial = project_feasible(trial, n)
        c_new = compute_c5(trial)
        
        if c_new < best_c5 - 1e-12:
            best_c5 = c_new
            best_h = trial
            # Try to increase step
            lr = min(lr * 1.2, 0.05)
        else:
            lr = max(lr * 0.7, 0.0001)
    
    return best_h, best_c5

def coord_descent_top(h, n, rng, tlimit, max_sweeps=50):
    """Coordinate descent focusing on reducing top lags."""
    h_best = h.copy()
    c5 = compute_c5(h_best)
    
    for sweep in range(max_sweeps):
        if tlimit() < 2.0:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode='full')
        top_k = int(np.argmax(corr))
        improved = False
        
        order = rng.permutation(n)
        for i in order:
            if tlimit() < 1.0:
                break
            v0 = h_best[i]
            # Try pushing towards 0 or 1 (binary)
            for target_v in [0.0, 1.0]:
                if abs(target_v - v0) < 1e-9:
                    continue
                trial = h_best.copy()
                trial[i] = target_v
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 1e-10:
                    c5 = c
                    h_best = trial.copy()
                    improved = True
            # Try small perturbations
            for d in [0.1, -0.1, 0.05, -0.05, 0.2, -0.2]:
                trial = h_best.copy()
                trial[i] = np.clip(trial[i] + d, 0.0, 1.0)
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 1e-10:
                    c5 = c
                    h_best = trial.copy()
                    improved = True
        if not improved and sweep >= 3:
            break
    
    return h_best, c5
