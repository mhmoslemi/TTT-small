        # f = sum_{i=0}^{n-1-offset} h[i]*(1-h[i+offset])
        n_pairs = n - offset
        # d/dh[j] of sum_i h[i](1-h[i+offset]):
        # j < n-offset: contributes (1-h[j+offset])
        # j >= offset: contributes -h[j-offset] (from i=j-offset term where h[i+offset]=h[j])
        for j in range(n_pairs):
            grad[j] += (1.0 - h[j + offset])
        for j in range(offset, n):
            grad[j] += -h[j - offset]
    else:
        abs_off = -offset
        n_pairs = n - abs_off
        # f = sum_{i=0}^{n-1-abs_off} h[i+abs_off]*(1-h[i])
        for j in range(abs_off, n):
            grad[j] += (1.0 - h[j - abs_off])
        for j in range(n_pairs):
            grad[j] += -h[j + abs_off]
    return grad


def active_set_gradient(h, n, top_k=5):
    """Compute weighted gradient using top-k active shifts"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    # Weight by proximity to max
    top_vals = corr[top_idx]
    weights = top_vals / top_vals.sum()
    grad = np.zeros(n)
    for i, idx in enumerate(top_idx):
        grad += weights[i] * grad_single_shift(h, n, int(idx))
    return grad, top_idx, corr

    c5_best = compute_c5(h)
    no_imp = 0
    for it in range(max_iters):
        if budget_fn() < 4.0:
            break
        grad, top_idx, corr = active_grad(h_best, n, top_k=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        gdir = grad / gnorm
        improved = False
        steps = [0.003, 0.006, 0.012, 0.025, 0.05, 0.009, 0.018, 0.035, 0.07]
        for s in steps:
            h_t = h_best - s * gdir
            h_t = project_feasible(h_t, n)
            c5_t = compute_c5(h_t)
            if c5_t < c5_best - 1e-15:
                c5_best = c5_t
                h_best = h_t.copy()
                improved = True
                no_imp = 0
                break
        if not improved:
            no_imp += 1
            if no_imp > 60:
                break
    return h_best, c5_best


def do_coord_desc(h, n, max_sweeps=15):
    """Coordinate descent trying 0, 1, and current*0.5+0.25 for each coordinate."""
    h_best = h.copy()
    c5_best = compute_c5(h)
    for sw in range(max_sweeps):
        if budget_fn() < 2.0:
            break
        any_imp = False
        for i in range(n):
            if budget_fn() < 1.0:
                return h_best, c5_best
            ov = h_best[i]
            cands = []
            cands.append(0.0)
            cands.append(1.0)
            mid = 0.5 * (ov + 0.0)
            cands.append(mid)
            for cv in cands:
                if abs(cv - ov) < 1e-9:
                    continue
                h_t = h_best.copy()
                h_t[i] = cv
                h_t = project_feasible(h_t, n)
                c5_t = compute_c5(h_t)
                if c5_t < c5_best - 1e-15:
                    c5_best = c5_t
                    h_best = h_t.copy()
                    any_imp = True
                    break
        if not any_imp and sw > 4:
            break
    return h_best, c5_best


def do_subgrad(h, n, rng, max_iters=300):
    h_best = h.copy()
    c5_best = compute_c5(h)
    no_imp = 0
    for it in range(max_iters):
        if budget_fn() < 3.0:
            break
        grad, _, _ = active_grad(h_best, n, top_k=3)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-17:
            break
        gdir = grad / gnorm
        improved = False
        steps = [0.004, 0.008, 0.015, 0.03, 0.006, 0.012, 0.05, 0.01]
        for s in steps:
            h_t = h_best - s * gdir
            h_t = project_feasible(h_t, n)
            c5_t = compute_c5(h_t)
            if c5_t < c5_best - 1e-15:
                c5_best = c5_t
                h_best = h_t.copy()
                improved = True
                no_imp = 0
                break
        if not improved:
            no_imp += 1
            if no_imp > 50:
                for _ in range(8):
                    noise = rng.randn(n) * 0.015
                    h_t = project_feasible(h_best + noise, n)
                    c5_t = compute_c5(h_t)
                    if c5_t < c5_best - 1e-15:
                        c5_best = c5_t
                        h_best = h_t.copy()
                        no_imp = 0
                        improved = True
                        break
                if no_imp > 100:
                    break
    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = np.zeros(parent_n)
        for i in range(parent_n):
            cnt = bin(i).count('1')
            parent_h[i] = 1.0 if cnt % 2 == 1 else 0.0
        best_h = project_feasible(parent_h, parent_n)
        best_c5 = compute_c5(best_h)
        best_n = parent_n
        print("No parent. TM n={}, C5={:.10f}".format(parent_n, best_c5))

    # Phase 1: SLSQP low-to-mid beta (restricted)
    print("\n[1] SLSQP restricted betas=[20,60,150,400,1000]")
    best_h, best_c5 = do_slsqp(best_h, best_n, [20, 60, 150, 400, 1000], use_restricted=True)
    print("  C5={:.10f}".format(best_c5))

    # Phase 2: Active-set gradient
    if budget_fn() > 25:
        print("\n[2] Active-set gradient (top5, 500 iters)")
        best_h, best_c5 = do_active_grad(best_h, best_n, max_iters=500, top_k=5)
        print("  C5={:.10f}".format(best_c5))

    # Phase 3: SLSQP high beta (full for precision)
    if budget_fn() > 25:
        print("\n[3] SLSQP full betas=[3000,10000,30000]")
        best_h, best_c5 = do_slsqp(best_h, best_n, [3000, 10000, 30000], use_restricted=False)
        print("  C5={:.10f}".format(best_c5))

    # Phase 4: Active-set gradient round 2
    if budget_fn() > 20:
        print("\n[4] Active-set gradient round 2")
        best_h, best_c5 = do_active_grad(best_h, best_n, max_iters=400, top_k=4)
        print("  C5={:.10f}".format(best_c5))

    # Phase 5: Subgradient
    if budget_fn() > 15:
        print("\n[5] Subgradient")
        best_h, best_c5 = do_subgrad(best_h, best_n, rng, max_iters=250)
        print("  C5={:.10f}".format(best_c5))

    # Phase 6: Multi-resolution
    print("\n[6] Multi-resolution")
    tried_ns = set()
    tried_ns.add(best_n)
    for n_try in [192, 224, 320, 384, 448, 512, 576]:
        if budget_fn() < 35.0:
            break
        if n_try in tried_ns:
            continue
        tried_ns.add(n_try)
        h_i = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        h_new, c5_new = do_slsqp(h_i, n_try, [100, 500, 2000], use_restricted=True)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n_try
            print("  n={} IMPROVED: C5={:.10f}".format(n_try, best_c5))
        else:
            print("  n={} {:.10f}".format(n_try, c5_new))

    # Phase 7: Final polish at best n
    if budget_fn() > 20:
        print("\n[7] Final SLSQP + active grad at n={}".format(best_n))
        best_h, best_c5 = do_slsqp(best_h, best_n, [5000, 20000, 80000], use_restricted=False)
        print("  After SLSQP: C5={:.10f}".format(best_c5))
        if budget_fn() > 15:
            best_h, best_c5 = do_active_grad(best_h, best_n, max_iters=300, top_k=3)
            print("  After active: C5={:.10f}".format(best_c5))

    # Phase 8: Subgradient fine
    if budget_fn() > 12:
        print("\n[8] Final subgrad")
        best_h, best_c5 = do_subgrad(best_h, best_n, rng, max_iters=200)
        print("  C5={:.10f}".format(best_c5))

    # Phase 9: Coordinate descent
    if budget_fn() > 10:
        print("\n[9] Coord descent")
        best_h, best_c5 = do_coord_desc(best_h, best_n, max_sweeps=8)
        print("  C5={:.10f}".format(best_c5))

    # Phase 10: Tiny final adjustments
    if budget_fn() > 8:
        best_h, best_c5 = do_active_grad(best_h, best_n, max_iters=100, top_k=3)
    if budget_fn() > 5:
        best_h, best_c5 = do_slsqp(best_h, best_n, [50000], use_restricted=False)

    # Final projection and validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
