    return h_best, c5_best


def slsqp_smooth(h, n, beta, maxiter=300):
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        exps = np.exp(np.clip(beta * shifted, -500, 500))
        return (m + np.log(np.sum(exps)) / beta) * dx

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new)
    except Exception:
        return h.copy(), compute_c5(h)


def periodic_search(h_best, n_best, c5_best, periods):
    for p in periods:
        if remaining_time() < 20.0:
            break
        if n_best % p != 0:
            continue
        m = n_best // p
        h_full = h_best.copy()
        base = h_full[:p * m].reshape(m, p).mean(axis=0)
        base = np.clip(base, 0.0, 1.0)
        target_base = p / 2.0
        base = project_feasible(base, p) if p >= 2 else base

        def obj_period(x):
            h_pern = np.tile(x, m)
            h_pern = project_feasible(h_pern, n_best)
            return compute_c5(h_pern)

        bounds = [(0.0, 1.0)] * p
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_base}]
        try:
            res = minimize(obj_period, base, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 200, 'ftol': 1e-14, 'iprint': 0})
            base_new = project_feasible(res.x, p)
            h_new = np.tile(base_new, m)
            h_new = project_feasible(h_new, n_best)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-12:
                c5_best = c5_new
                h_best = h_new.copy()
        except Exception:
            pass
    return h_best, c5_best


def coord_descent_fast(h, n, max_sweeps=10, rng=None):
    if rng is None:
        rng = np.random.RandomState(99)
    c5_best = compute_c5(h)
    h_best = h.copy()
    corr = compute_corr(h_best)
    dx = 2.0 / n
    k_vals = np.arange(-(n - 1), n)

    for sweep in range(max_sweeps):
        if remaining_time() < 4.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if remaining_time() < 2.0:
                break
            old = h_best[i]
            for new_val in [0.0, 1.0]:
                if abs(new_val - old) < 1e-8:
                    continue
                delta = new_val - old
                ik = i + k_vals
                imk = i - k_vals
                d = np.zeros(2 * n - 1)
                m1 = (ik >= 0) & (ik < n)
                m2 = (imk >= 0) & (imk < n)
                d[m1] += delta * (1.0 - h_best[ik[m1]])
                d[m2] += delta * (-h_best[imk[m2]])
                new_corr = corr + d
                new_c5 = float(np.max(new_corr)) * dx
                if new_c5 < c5_best - 1e-13:
                    h_best[i] = new_val
                    const_adj = (n / 2.0 - np.sum(h_best))
                    if abs(const_adj) > 1e-10:
                        h_best = project_feasible(h_best, n)
                        corr = compute_corr(h_best)
                    else:
                        corr = new_corr
                    c5_best = new_c5
                    any_imp = True
        if not any_imp and sweep > 2:
            break
    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Initialization from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 256
        best_h = make_thue_morse(best_n)
        best_c5 = compute_c5(best_h)
        print("Thue-Morse init: n={}, C5={:.10f}".format(best_n, best_c5))

    # Phase 1: Nesterov top-k gradient
    print("\n--- P1: Nesterov gradient ---")
    best_h, best_c5 = nesterov_topk_gradient(best_h, best_n, max_iter=350, top_k_init=3, top_k_max=12)
    print("  C5={:.10f}".format(best_c5))

    # Phase 2: SLSQP multi-beta
    print("\n--- P2: SLSQP ---")
    for beta_val in [100, 500, 2000, 8000]:
        if remaining_time() < 15.0:
            break
        h_new, c5_new = slsqp_smooth(best_h, best_n, beta_val, maxiter=300)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new
            print("  beta={}: C5={:.10f}".format(beta_val, best_c5))
    print("  C5={:.10f}".format(best_c5))

    # Phase 3: Binary pair-swap (NEW)
    print("\n--- P3: Binary pair-swap ---")
    h_bin, c5_bin = binary_pair_swap(best_h, best_n, max_sweeps=250, rng=rng)
    print("  binary C5={:.10f}".format(c5_bin))
    if c5_bin < best_c5 - 1e-12:
        best_c5 = c5_bin
        best_h = h_bin.copy()
        print("  IMPROVED to {:.10f}".format(best_c5))

    # Phase 4: Re-optimize from best (could be binary or continuous)
    print("\n--- P4: Re-optimize ---")
    best_h, best_c5 = nesterov_topk_gradient(best_h, best_n, max_iter=200, top_k_init=2, top_k_max=8)
    for beta_val in [500, 3000, 10000]:
        if remaining_time() < 12.0:
            break
        h_new, c5_new = slsqp_smooth(best_h, best_n, beta_val, maxiter=250)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new
    print("  C5={:.10f}".format(best_c5))

    # Phase 5: Periodic structure search (NEW)
    print("\n--- P5: Periodic search ---")
    periods_to_try = [8, 16, 32, 64, 128]
    best_h, best_c5 = periodic_search(best_h, best_n, best_c5, periods_to_try)
    print("  C5={:.10f}".format(best_c5))

    # Phase 6: Another binary swap round from current best
    print("\n--- P6: Binary swap round 2 ---")
    h_bin2, c5_bin2 = binary_pair_swap(best_h, best_n, max_sweeps=150, rng=rng)
    print("  binary C5={:.10f}".format(c5_bin2))
    if c5_bin2 < best_c5 - 1e-12:
        best_c5 = c5_bin2
        best_h = h_bin2.copy()

    # Phase 7: Multi-resolution
    print("\n--- P7: Multi-resolution ---")
    for n_try in [128, 384, 512, 640]:
        if remaining_time() < 40.0:
            break
        h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        c5_h = compute_c5(h_interp)
        h_interp, c5_h = nesterov_topk_gradient(h_interp, n_try, max_iter=100, top_k_init=3, top_k_max=10)
        for bv in [500, 3000]:
            if remaining_time() < 10.0:
                break
            h_new, c5_new = slsqp_smooth(h_interp, n_try, bv, maxiter=200)
            if c5_new < c5_h - 1e-13:
                c5_h = c5_new
                h_interp = h_new
        if c5_h < best_c5 - 1e-10:
            best_c5 = c5_h
            best_h = h_interp.copy()
            best_n = n_try
            print("  n={}: C5={:.10f} (IMPROVED)".format(n_try, c5_h))
        else:
            print("  n={}: C5={:.10f}".format(n_try, c5_h))

    # Phase 8: Final gradient + SLSQP
    print("\n--- P8: Final optimization ---")
    best_h, best_c5 = nesterov_topk_gradient(best_h, best_n, max_iter=150, top_k_init=2, top_k_max=6)
    for beta_val in [5000, 20000]:
        if remaining_time() < 8.0:
            break
        h_new, c5_new = slsqp_smooth(best_h, best_n, beta_val, maxiter=400)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new
    print("  C5={:.10f}".format(best_c5))

    # Phase 9: Fast coordinate descent
    print("\n--- P9: Coord descent ---")
    best_h, best_c5 = coord_descent_fast(best_h, best_n, max_sweeps=12, rng=rng)
    print("  C5={:.10f}".format(best_c5))

    # Phase 10: Tiny final polish
    if remaining_time() > 8.0:
        h_new, c5_new = slsqp_smooth(best_h, best_n, 50000, maxiter=200)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new
        best_h, best_c5 = coord_descent_fast(best_h, best_n, max_sweeps=5, rng=rng)
        print("\n--- P10: Final ---")
        print("  C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-6, "Sum constraint violated: {:.6f}".format(np.sum(best_h))
    assert np.all(best_h >= 0) and np.all(best_h <= 1), "Bounds violated"

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
