def compute_c5(h):
    n = len(h)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    rng = np.random.RandomState(0)
    if n_ones > n_half:
        ones = np.where(h == 1)[0]
        rm = rng.choice(ones, n_ones - n_half, replace=False)
        h[rm] = 0
    elif n_ones < n_half:
        zeros = np.where(h == 0)[0]
        add = rng.choice(zeros, n_half - n_ones, replace=False)
        h[add] = 1
    return h


def _sm_obj(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / _BETA) * dx_v


def _sm_grad_sum(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    wc = w * dx_v
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            npairs = n - off
            g[off:off + npairs] += wk * (1.0 - x[:npairs])
            g[:npairs] -= wk * x[off:off + npairs]
        else:
            ao = -off
            npairs = n - ao
            g[:npairs] += wk * (1.0 - x[ao:ao + npairs])
            g[ao:ao + npairs] -= wk * x[:npairs]
    return g


def _sc(x):

def slsqp_smooth(h, n, beta, maxiter=500):

    _N = n
    _TGT = n / 2.0
    _BETA = beta

            c5_int = c5_wf2

        if c5_int < best_c5:
            best_c5 = c5_int
            best_h = h_feas.copy()
            best_n = n_t

    # Phase 6: Perturb and re-optimize (escape local optima)
    if remaining() > 40:
        for perturb_count in [32, 64, 128]:
            if remaining() < 15:
                break
            h_pert = best_h.copy()
            idx = rng.choice(n, size=min(perturb_count, n // 4), replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
            h_pert = project_feasible(h_pert, n)

            h_opt = slsqp_smooth(h_pert, n, 3000, maxiter=300)
            if remaining() > 10:
                h_opt, c5_opt = water_filling(h_opt, n, time_limit=min(30, int(remaining() * 0.2)))
            else:
                c5_opt = compute_c5(h_opt)

            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 7: Final intensive refinement
    n = best_n
    h = best_h.copy()
    for beta in [10000, 30000, 80000]:
        if remaining() < 5:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining() > 10:
        h_wf, c5_wf_final = water_filling(h, n, time_limit=min(60, int(remaining() * 0.5)))
        if c5_wf_final < best_c5:
            best_c5 = c5_wf_final
            best_h = h_wf_final.copy()

    if remaining() > 10:
        h_bin_final = (best_h > 0.5).astype(float)
        h_bin_final = project_feasible(h_bin_final, n)
        n_ones_f = int(np.round(np.sum(h_bin_final)))
        if n_ones_f != n // 2:
            sort_idx = np.argsort(-h_bin_final)
            h_bin_final = np.zeros(n)
            h_bin_final[sort_idx[:n // 2]] = 1.0
        h_bf, c5_bf = binary_greedy(h_bin_final, n, time_limit=min(40, int(remaining() * 0.5)))
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()
            if remaining() > 5:
                h_final = slsqp_smooth(best_h, n, 50000, maxiter=200)
                c5_final = compute_c5(h_final)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
