            c5_cur = c_new
        else:
            h_new, c_new = paired_descent(best_h, n, max_rounds=15, rng=rng, tlimit=tlimit)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
                h_cur = h_new.copy()
                c5_cur = c_new
        print("  After paired: C5={:.10f}".format(best_c5))
    
    # Phase 5: SLSQP polish after subgradient/paired
    if tlimit() > 40:
        print("\n=== Phase 5: SLSQP polish ===")
        for bv in [50000, 100000, 200000]:
            if tlimit() < 10:
                break
            h_new = slsqp_smooth(h_cur, n, beta=bv, maxiter=3000)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new.copy()
                h_cur = h_new.copy()
                print("  beta={}: C5={:.10f}".format(bv, c5_cur))
    
    # Phase 6: Try n=512
    if tlimit() > 120:
        print("\n=== Phase 6: Try n=512 ===")
        n2 = 512
        h0_512 = project_feasible(interp_h(best_h, best_n, n2), n2)
        c5_512 = compute_c5(h0_512)
        print("  Interpolated n=512: C5={:.10f}".format(c5_512))
        
        h_cur512 = h0_512.copy()
        c5_cur512 = c5_512
        for bv in [100, 500, 2000, 10000]:
            if tlimit() < 15:
                break
            h_new512 = lbfgsb_penalty(h_cur512, n2, beta=bv, pen=200.0, maxiter=1500)
            c_new512 = compute_c5(h_new512)
            if c_new512 < c5_cur512 - 1e-14:
                c5_cur512 = c_new512
                h_cur512 = h_new512
        for bv in [5000, 20000]:
            if tlimit() < 12:
                break
            h_new512 = slsqp_smooth(h_cur512, n2, beta=bv, maxiter=1000)
            c_new512 = compute_c5(h_new512)
            if c_new512 < c5_cur512 - 1e-14:
                c5_cur512 = c_new512
                h_cur512 = h_new512
        print("  n=512 optimized: C5={:.10f}".format(c5_cur512))
        
        if c5_cur512 < best_c5 - 1e-10:
            best_c5 = c5_cur512
            best_h = h_cur512.copy()
            best_n = n2
            h_cur = h_cur512.copy()
            n = n2
            print("  IMPROVED! n=512: C5={:.10f}".format(best_c5))
    
    # Phase 7: Multi-start escape
    if tlimit() > 80:
        print("\n=== Phase 7: Multi-start ===")
        h_new, c_new = multi_start_slsqp(best_h, len(best_h), rng, tlimit, max_starts=8)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
        print("  After multi-start: C5={:.10f}".format(best_c5))
    
    # Phase 8: Final subgradient + SLSQP
    if tlimit() > 30:
        print("\n=== Phase 8: Final refinement ===")
        n_final = len(best_h)
        h_r, c_r = subgradient_refine(best_h, n_final, max_iters=5000, alpha0=0.01, alpha_decay=0.9998)
        if c_r < best_c5:
            best_h = h_r.copy()
            best_c5 = c_r
        h_new = slsqp_smooth(best_h, n_final, beta=200000, maxiter=5000)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c_new
        print("  After final: C5={:.10f}".format(best_c5))
    
    # Phase 9: Final paired descent
    if tlimit() > 20:
        h_new, c_new = paired_descent(best_h, len(best_h), max_rounds=10, rng=rng, tlimit=tlimit)
        if c_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c_new
    
    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
