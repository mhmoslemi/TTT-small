```python
def fft_cross_correlate(a, v):
    """Compute cross-correlation using FFT."""
    n = len(a)
    m = len(v)
    size = n + m - 1
    # Next power of 2
    size_pow2 = int(2 ** np.ceil(np.log2(size)))
    
    fa = np.fft.rfft(a, size_pow2)
    fv = np.fft.rfft(v, size_pow2)
    
    # Cross-correlation
    corr = np.fft.irfft(fa * np.conj(fv), size_pow2)
    
    # The first n+m-1 elements give the circular cross-correlation
    # but we need to shift it to match np.correlate output
    # np.correlate(a, v, 'full')[k] corresponds to cross-corr at lag k-(m-1)
    # The FFT gives cross-corr at lags 0, 1, ..., size_pow2-1
    # We need lags -(m-1), ..., n-1
    # So we take elements [size_pow2-(m-1):] and [0:n]
    
    result = np.zeros(n + m - 1)
    # Lags 0 to n-1
    result[:n] = corr[:n]
    # Lags -(m-1) to -1 (i.e., the last m-1 elements of the circular correlation)
    result[n:] = corr[size_pow2-m+1:]
    
    return result
```

```python
    conv = np.fft.irfft(fa * fv, size)
```

```python
def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            rooms = 1.0 - h
        else:
            rooms = h
        order = np.argsort(-rooms)
        for i in order:
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                add = min(diff, rooms[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, rooms[i])
                h[i] -= sub
                diff += sub
    return h

    beta = _G_BETA
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    lse_val = (m + np.log(np.sum(exps)) / beta) * dx
    pen = _G_GAMMA * (np.sum(h) - _G_TGT) ** 2
    return lse_val + pen


def grad_fun(h):
    n = _G_N
    dx = _G_DX
    corr = compute_corr(h)
    beta = _G_BETA
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exps = np.exp(shifted)
    total = np.sum(exps)
    softmax = exps / total
    grad = np.zeros(n)
    active = np.where(softmax > 1e-8)[0]
    for k in active:
        w = softmax[k]
        offset = int(k) - (n - 1)
        if offset > 0:
            d = offset
            grad[d:] += w * (1.0 - h[:n - d])
            grad[:n - d] -= w * h[d:]
        elif offset < 0:
            d = -offset
            grad[:n - d] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:n - d]
        else:
            grad += w * (1.0 - 2.0 * h)
    grad *= dx
    grad += 2.0 * _G_GAMMA * (np.sum(h) - _G_TGT)
    return grad


def obj_only(h):
    n = _G_N
    dx = _G_DX
    corr = compute_corr(h)
    m = np.max(corr)
    beta = _G_BETA
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    return (m + np.log(np.sum(exps)) / beta) * dx


def grad_only(h):
    n = _G_N
    dx = _G_DX
    corr = compute_corr(h)
    beta = _G_BETA
    max_c = np.max(corr)
    shifted = np.clip(beta * (corr - max_c), -500, 0)
    exps = np.exp(shifted)
    total = np.sum(exps)
    softmax = exps / total
    grad = np.zeros(n)
    active = np.where(softmax > 1e-8)[0]
    for k in active:
        w = softmax[k]
        offset = int(k) - (n - 1)
        if offset > 0:
            d = offset
            grad[d:] += w * (1.0 - h[:n - d])
            grad[:n - d] -= w * h[d:]
        elif offset < 0:
            d = -offset
            grad[:n - d] += w * (1.0 - h[d:])
            grad[d:] -= w * h[:n - d]
        else:
            grad += w * (1.0 - 2.0 * h)
    return grad * dx


def con_sum_fun(x):
    return np.sum(x) - _G_TGT


def grad_con_sum(x):
    return np.ones(_G_N)


def lbfgsb_optimize(h0, n, beta, gamma, budget_fn):
    global _G_N, _G_BETA, _G_TGT, _G_DX, _G_GAMMA
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    _G_BETA = beta
    _G_GAMMA = gamma
    bounds = [(0.0, 1.0)] * n
    h = h0.copy()
    c5_cur = compute_c5(h)
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-12,
                                'maxfun': 50000, 'disp': False})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
    except Exception:
        pass
    return h, c5_cur


def slsqp_optimize(h0, n, beta, budget_fn):
    global _G_N, _G_BETA, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    _G_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum_fun, 'jac': grad_con_sum}]
    h = h0.copy()
    c5_cur = compute_c5(h)
    try:
        res = minimize(obj_only, h, jac=grad_only, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 1000, 'ftol': 1e-18, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
    except Exception:
        pass
    return h, c5_cur


def binary_local_search(h, n, rng, budget_fn, max_iters=2000):
    h = np.round(h)
    h = h.astype(float)
    h = project_feasible(h, n)
    c5 = compute_c5(h)
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    no_improve = 0
    for it in range(max_iters):
        if budget_fn() < 5.0:
            break
        if not ones or not zeros:
            break
        n_tries = 50
        found = False
        for _ in range(n_tries):
            i_idx = rng.randint(len(ones))
            j_idx = rng.randint(len(zeros))
            i = ones[i_idx]
            j = zeros[j_idx]
            h[i] = 0.0
            h[j] = 1.0
            c5_new = compute_c5(h)
            if c5_new < c5 - 1e-14:
                c5 = c5_new
                ones = list(np.where(h == 1.0)[0])
                zeros = list(np.where(h == 0.0)[0])
                found = True
                no_improve = 0
                break
            else:
                h[i] = 1.0
                h[j] = 0.0
        if not found:
            no_improve += 1
            if no_improve > 30:
                break
    return h, c5


def subgradient_step(h, n, budget_fn, max_iters=100):
    global _G_N, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    h = h.copy()
    c5_best = compute_c5(h)
    no_improve = 0
    for it in range(max_iters):
        if budget_fn() < 5.0:
            break
        corr = compute_corr(h)
        max_val = np.max(corr)
        active = np.where(corr > max_val - 0.05)[0]
        top_k = min(5, len(active))
        top_idx = active[np.argsort(corr[active])[-top_k:]]
        # Compute subgradient of max using active set
        _G_BETA = 100000.0
        grad = grads_top_k(h, top_idx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        grad = grad / gnorm
        improved = False
        steps = [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.03, 0.08]
        for step in steps:
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-13:
                c5_best = c5_trial
                h = h_trial.copy()
                improved = True
                no_improve = 0
                break
        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
    return h, c5_best


def grads_top_k(h, top_idx):
    n = _G_N
    grad = np.zeros(n)
    for k in top_idx:
        offset = int(k) - (n - 1)
        if offset > 0:
            d = offset
            grad[d:] += (1.0 - h[:n - d])
            grad[:n - d] -= h[d:]
        elif offset < 0:
            d = -offset
            grad[:n - d] += (1.0 - h[d:])
            grad[d:] -= h[:n - d]
        else:
            grad += (1.0 - 2.0 * h)
    return grad * _G_DX


def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = make_thue_morse(parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("No parent. TM: n={}, C5={}.format(parent_n, best_c5)".format(parent_n, best_c5))

    if best_c5 <= 0.38085:
        print("Already at target! C5={:.10f}".format(best_c5))
        return (best_h, best_c5, best_n)

    # Phase 1: L-BFGS-B optimization at parent resolution
    print("\n--- Phase 1: L-BFGS-B at n={} ---".format(best_n))
    for beta in [50, 200, 1000, 5000]:
        if remaining_time() < 30:
            break
        h_new, c5_new = lbfgsb_optimize(best_h, best_n, beta, 500.0, remaining_time)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            print("  beta={}: C5={:.10f}".format(beta, best_c5))
        else:
            print("  beta={}: no improve ({:.10f})".format(beta, c5_new))

    # Phase 2: SLSQP refinement
    if remaining_time() > 30:
        print("\n--- Phase 2: SLSQP at n={} ---".format(best_n))
        for beta in [500, 2000, 10000]:
            if remaining_time() < 20:
                break
            h_new, c5_new = slsqp_optimize(best_h, best_n, beta, remaining_time)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                print("  beta={}: C5={:.10f}".format(beta, best_c5))

    # Phase 3: Try other binary patterns at same resolution
    print("\n--- Phase 3: Binary patterns at n={} ---".format(best_n))
    patterns = [
        make_thue_morse(best_n),
        make_paperfolding(best_n),
        make_random_binary(rng, best_n),
        make_random_binary(rng, best_n),
    ]
    for pat in patterns:
        if remaining_time() < 30:
            break
        c5_pat = compute_c5(pat)
        if c5_pat < best_c5:
            best_c5 = c5_pat
            best_h = pat.copy()
            print("  Pattern C5={:.10f}".format(best_c5))
        # L-BFGS-B from this pattern
        h_new, c5_new = lbfgsb_optimize(pat, best_n, 200, 500.0, remaining_time)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            print("  Pattern+opt C5={:.10f}".format(best_c5))

    # Phase 4: Binary local search
    if remaining_time() > 20:
        print("\n--- Phase 4: Binary local search ---")
        h_bin = np.round(best_h)
        h_bin = project_feasible(h_bin.astype(float), best_n)
        h_bls, c5_bls = binary_local_search(h_bin, best_n, rng, remaining_time, max_iters=3000)
        if c5_bls < best_c5:
            best_c5 = c5_bls
            best_h = h_bls
            print("  Binary LS C5={:.10f}".format(best_c5))

    # Phase 5: Subgradient refinement
    if remaining_time() > 30:
        print("\n--- Phase 5: Subgradient ---")
        h_new, c5_new = subgradient_step(best_h, best_n, remaining_time, max_iters=200)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            print("  Subgrad C5={:.10f}".format(best_c5))

    # Phase 6: Multi-resolution (double)
    n_double = best_n * 2
    if remaining_time() > 100:
        print("\n--- Phase 6: Resolution upgrade to n={} ---".format(n_double))
        h_interp = project_feasible(interpolate(best_h, best_n, n_double), n_double)
        c5_interp = compute_c5(h_interp)
        print("  Interpolated: C5={:.10f}".format(c5_interp))
        for beta in [100, 500, 2000, 10000]:
            if remaining_time() < 30:
                break
            h_new, c5_new = lbfgsb_optimize(h_interp, n_double, beta, 500.0, remaining_time)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                best_n = n_double
                print("  n={}, beta={}: C5={:.10f}".format(n_double, beta, best_c5))
        # SLSQP at new resolution
        if remaining_time() > 30:
            h_new, c5_new = slsqp_optimize(best_h, best_n, 5000, remaining_time)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                print("  n={}, SLSQP: C5={:.10f}".format(best_n, best_c5))

    # Phase 7: Another resolution upgrade
    if best_n < 1024 and remaining_time() > 150:
        n_double = best_n * 2
        print("\n--- Phase 7: Resolution upgrade to n={} ---".format(n_double))
        h_interp = project_feasible(interpolate(best_h, best_n, n_double), n_double)
        c5_interp = compute_c5(h_interp)
        print("  Interpolated: C5={:.10f}".format(c5_interp))
        for beta in [200, 1000, 5000]:
            if remaining_time() < 30:
                break
            h_new, c5_new = lbfgsb_optimize(h_interp, n_double, beta, 500.0, remaining_time)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                best_n = n_double
                print("  n={}, beta={}: C5={:.10f}".format(n_double, beta, best_c5))

    # Phase 8: Binary local search at current best resolution
    if remaining_time() > 30:
        print("\n--- Phase 8: Binary LS at n={} ---".format(best_n))
        h_bin = np.round(best_h)
        h_bin = project_feasible(h_bin.astype(float), best_n)
        h_bls, c5_bls = binary_local_search(h_bin, best_n, rng, remaining_time, max_iters=5000)
        if c5_bls < best_c5:
            best_c5 = c5_bls
            best_h = h_bls
            print("  Binary LS C5={:.10f}".format(best_c5))

    # Phase 9: Final L-BFGS-B polish
    if remaining_time() > 30:
        print("\n--- Phase 9: Final polish ---")
        for beta in [5000, 20000, 50000]:
            if remaining_time() < 15:
                break
            h_new, c5_new = lbfgsb_optimize(best_h, best_n, beta, 2000.0, remaining_time)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                print("  beta={}: C5={:.10f}".format(beta, best_c5))

    # Phase 10: Subgradient final
    if remaining_time() > 15:
        h_new, c5_new = subgradient_step(best_h, best_n, remaining_time, max_iters=100)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 11: Coordinate descent on active lags
    if remaining_time() > 20:
        print("\n--- Phase 11: Active-set coordinate optimization ---")
        h = best_h.copy()
        c5 = best_c5
        for sweep in range(3):
            if remaining_time() < 10:
                break
            corr = compute_corr(h)
            max_c = np.max(corr)
            active = np.where(corr > max_c - 0.1)[0]
            for k in active:
                if remaining_time() < 5:
                    break
                # Binary search for best single-coordinate adjustment at this lag
                offset = int(k) - (best_n - 1)
                # Try small perturbations in the gradient direction
                grad_k = np.zeros(best_n)
                if offset > 0:
                    d = offset
                    grad_k[d:] = (1.0 - h[:best_n - d])
                    grad_k[:best_n - d] -= h[d:]
                elif offset < 0:
                    d = -offset
                    grad_k[:best_n - d] = (1.0 - h[d:])
                    grad_k[d:] -= h[:best_n - d]
                else:
                    grad_k = (1.0 - 2.0 * h)
                gnorm = np.linalg.norm(grad_k)
                if gnorm < 1e-14:
                    continue
                grad_k /= gnorm
                for step in [0.001, 0.003, 0.01, 0.03]:
                    h_trial = project_feasible(h - step * grad_k, best_n)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < c5 - 1e-13:
                        c5 = c5_trial
                        h = h_trial
                        break
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            print("  Active-set C5={:.10f}".format(best_c5))

    # Final: ensure feasibility and compute exact C5
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    # Try evaluate_erdos_solution if available
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
```
