_START = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:

        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)

The full correlation c = np.correlate(h, 1-h, mode='full') has length 2n-1.
Entry at index idx (0 to 2n-2) computes:
c[idx] = sum_{i} h[i + idx - (n-1)] * (1 - h[i]) for valid i

Where "valid i" means 0 ≤ i ≤ n-1 and 0 ≤ i + idx - (n-1) ≤ n-1.

Let k = idx - (n-1). Then k ranges from -(n-1) to (n-1).
c[idx] = sum_{i} h[i+k] * (1-h[i]) for valid i

For k ≥ 0: valid i is 0 ≤ i ≤ n-1-k
  c = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])
  dc/dh[i+k] = 1-h[i] for i=0..n-1-k

    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val

def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-14:
            g += w[k] * corr_grad(x, n, k)
    return g * dx_val

    return np.sum(x) - _TGT

3. Compute the "contribution" of each position to this lag

For lag k*, the correlation is:

If I flip h[i] from 1 to 0, the change in c(k*) is:
- The term h[i+k*]*(1-h[i]) becomes h[i+k*]*1 instead of h[i+k*]*0
- So c(k*) increases by h[i+k*] (bad!)

- The term h[j+k*]*(1-h[j]) becomes h[j+k*]*0 instead of h[j+k*]*1
- So c(k*) decreases by h[j+k*] (good!)

- The term at position i (where h[i+k*] is the "first" factor): h[i+k*]*(1-h[i])
  If h[i] goes 1→0: term goes from 0 to h[i+k*]. Change: +h[i+k*]
  If h[i+k*] goes 1→0: term goes from (1-h[i])*1 to 0. Change: -(1-h[i])

def corr_grad_at(h, n, k_idx):
    g = np.zeros(n)
    off = k_idx - (n - 1)
    n_valid = n - abs(off)
    if n_valid <= 0:
        return g
    if off >= 0:
        g[off:off + n_valid] += (1.0 - h[:n_valid])
        g[:n_valid] -= h[off:off + n_valid]
    else:
        m = -off
        g[:n_valid] += (1.0 - h[m:m + n_valid])
        g[m:m + n_valid] -= h[:n_valid]
    return g


def smooth_obj(x):
    n = _N

    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val

    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-14:
            g += w[k] * corr_grad_at(x, n, k)
    return g * dx_val

            break

        m = float(np.max(corr))
        if m * dx_val >= best_c5 + 1e-8:
            break
        top_idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k in top_idx:
            g += corr_grad_at(h, n, int(k))
        g /= len(top_idx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)

    best_h = parent_h.copy()
    best_c5 = parent_c5
    best_n = parent_n

    # Phase 1: Binary hill climbing at parent resolution (key phase)
    h_bin = make_binary(parent_h, parent_n, rng)
    t1 = min(150, remaining() * 0.22)
    h_res, c5_res = binary_hill_climb(h_bin, parent_n, max_time=t1, rng=rng)
    if c5_res < best_c5:
        best_c5 = c5_res
        best_h = h_res.copy()
        best_n = parent_n

    # Phase 2: Multiple binary SA restarts at parent resolution
    t2_total = min(100, remaining() * 0.15)
    for s in [seed + 1, seed + 500, seed + 1000, seed + 2000]:
        if remaining() < 15 or time.time() > _START + budget_s - t2_total * 0.2:
            break
        t_sa = min(25, max(8, t2_total / 4))
        h_sa, c5_sa = binary_sa(best_h, parent_n, max_time=t_sa,
                                rng=np.random.RandomState(s))
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = parent_n

    # Phase 3: Try higher resolutions
    n_candidates = []
    for dn in [20, 40, 60, 80, 100, 120, 140, 160]:
        nt = parent_n + dn
        if nt % 2 == 0 and nt <= 800:
            n_candidates.append(nt)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if remaining() < 50:
            break
        h_interp = interp_to(best_h, best_n, n_try)
        h_bin2 = make_binary(h_interp, n_try, rng)

        # Binary hill climbing at new resolution
        t_hc = min(60, max(15, remaining() * 0.12))
        h_res2, c5_res2 = binary_hill_climb(h_bin2, n_try, max_time=t_hc, rng=rng)
        if c5_res2 < best_c5:
            best_c5 = c5_res2
            best_h = h_res2.copy()
            best_n = n_try

        # Quick continuous refinement
        if remaining() > 30:
            h_res3, c5_res3 = continuous_refine(best_h, best_n, max_time=min(30, remaining() * 0.1))
            if c5_res3 < best_c5:
                best_c5 = c5_res3
                best_h = h_res3.copy()

    # Phase 4: Also try random/structured inits at parent resolution
    if remaining() > 40:
        for pat in ['random', 'alternating']:
            if remaining() < 20:
                break
            h_struct = structured_init(parent_n, rng, pat)
            h_bin3 = make_binary(h_struct, parent_n, rng)
            t_hc2 = min(40, max(10, remaining() * 0.1))
            h_res4, c5_res4 = binary_hill_climb(h_bin3, parent_n, max_time=t_hc2, rng=rng)
            if c5_res4 < best_c5:
                best_c5 = c5_res4
                best_h = h_res4.copy()
                best_n = parent_n

    # Phase 5: Intensive continuous refinement at best resolution
    if remaining() > 30:
        h_res5, c5_res5 = continuous_refine(best_h, best_n, max_time=min(120, remaining() * 0.3))
        if c5_res5 < best_c5:
            best_c5 = c5_res5
            best_h = h_res5.copy()

    # Phase 6: Final binary hill climb on best solution
    if remaining() > 20:
        h_bin4 = make_binary(best_h, best_n, rng)
        t_final = min(60, remaining() * 0.3)
        h_res6, c5_res6 = binary_hill_climb(h_bin4, best_n, max_time=t_final, rng=rng)
        if c5_res6 < best_c5:
            best_c5 = c5_res6
            best_h = h_res6.copy()

    # Phase 7: Final top-k gradient polish
    if remaining() > 10:
        h_res7, c5_res7 = topk_gradient_descent(best_h, best_n, max_iters=200, top_k=3,
                                                time_limit=min(30, remaining() * 0.5))
        if c5_res7 < best_c5:
            best_c5 = c5_res7
            best_h = h_res7.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
