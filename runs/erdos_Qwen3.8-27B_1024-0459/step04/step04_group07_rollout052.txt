    return h_best, c5_best

def coord_descent_binary(h, n, rng, max_sweeps, tlimit):
    """Coordinate descent for binary: try swapping each 1 with each nearby 0."""
    dx = 2.0 / n
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    c5 = compute_c5(h)
    h_best = h.copy()
    c5_best = c5
    
    for sweep in range(max_sweeps):
        if tlimit() < 3.0:
            break
        improved = False
        ones_pos = np.where(h_best == 1.0)[0]
        zeros_pos = np.where(h_best == 0.0)[0]
        
        # Try swapping 1s with all 0s (or a random subset for large n)
        rng.shuffle(ones_pos)
        for i in ones_pos:
            if tlimit() < 2.0:
                break
            # Try swapping with nearby zeros (within window) or random zeros
            candidates = rng.choice(zeros_pos, size=min(20, len(zeros_pos)), replace=False)
            for j in candidates:
                trial = h_best.copy()
                trial[i] = 0.0
                trial[j] = 1.0
                c = compute_c5(trial)
                if c < c5_best - 1e-15:
                    h_best = trial
                    c5_best = c
                    improved = True
                    zeros_pos = np.where(h_best == 0.0)[0]
                    break
        if not improved:
            break
    
    return h_best, c5_best

def continuous_polish(h, n, rng, tlimit, max_iters=2000, lr_start=0.05):
    """Subgradient-based continuous optimization around binary solution."""
    dx = 2.0 / n
    h = project_feasible(h, n)
    c5 = compute_c5(h)
    h_best = h.copy()
    c5_best = c5
    
    for it in range(max_iters):
        if tlimit() < 3.0:
            break
        # Compute correlation and find active lag
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_active = np.argmax(corr)
        
        # Subgradient of c_k at h:
        # c_k = sum_i h[i]*(1-h[i+k]) for valid i
        # dc_k/dh[i] = (1-h[i+k]) - h[i-k] (with appropriate indexing)
        
        # Build gradient
        grad = np.zeros(n)
        lag = k_active - (n - 1)  # actual lag
        for i in range(n):
            # d/dh[i] of sum_j h[j]*(1-h[j+lag])
            # = (1 - h[i+lag]) if i+lag valid, as the "first" term
            #   - h[i-lag] if i-lag valid, as the "second" term (from h[i-lag]*(1-h[i]))
            gi = 0.0
            ip = i + lag
            if 0 <= ip < n:
                gi += (1.0 - h[ip])
            im = i - lag
            if 0 <= im < n:
                gi -= h[im]
            grad[i] = gi
        
        # Normalize and take step
        lr = lr_start * (0.999 ** it)
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad /= grad_norm
            h_new = h - lr * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new)
            if c_new < c5:
                h = h_new
                c5 = c_new
                if c5 < c5_best - 1e-15:
                    c5_best = c5
                    h_best = h.copy()
            else:
                # Halve learning rate implicitly by not updating
                lr *= 0.5
    
    return h_best, c5_best
