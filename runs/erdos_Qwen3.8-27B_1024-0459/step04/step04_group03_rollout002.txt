def l_bfgsb_optimize(h_init, n, beta, maxiter=500):
    """Optimize using L-BFGS-B with smooth max objective."""
    def obj(x):
        x_c = np.clip(x, 0.0, 1.0)
        val, _ = smooth_max_and_grad(x_c, beta)
        # Add penalty for constraint violation
        penalty = 100.0 * (np.sum(x_c) - n/2.0)**2
        return val + penalty
    
    def grad(x):
        x_c = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(x_c, beta)
        # Add penalty gradient
        g = g + 200.0 * (np.sum(x_c) - n/2.0)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_init, method='L-BFGS-B', jac=grad,
                      bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-14})
        return res.x
    except Exception:
        return h_init.copy()

def slsqp_smooth_grad(h, n, beta, maxiter=300):
    """SLSQP with analytical gradient."""
    def obj(x):
        x_c = np.clip(x, 0.0, 1.0)
        val, _ = smooth_max_and_grad(x_c, beta)
        return val
    
    def grad(x):
        x_c = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(x_c, beta)
        return g
    
    def eq(x):
        return np.sum(x) - n / 2.0
    
    def eq_grad(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_grad}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', jac=grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def binarize(h, n, threshold=0.1):
    h_b = h.copy()
    h_b[h_b < threshold] = 0.0
    h_b[h_b > 1.0 - threshold] = 1.0
    return project_feasible(h_b, n)

def flip_search(h, n, max_iter=5000):
    """Try flipping individual elements and small groups."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(123)
    
    for it in range(max_iter):
        if remaining_time() < 2:
            break
        # Try a random flip
        i = rng.randint(n)
        h_trial = best_h.copy()
        h_trial[i] = 1.0 - h_trial[i]
        h_trial = project_feasible(h_trial, n)
        c5_t = compute_c5(h_trial)
        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_h = h_trial
        elif it % 50 == 0:
            # Try a swap
            j = rng.randint(n)
            if j != i:
                h_trial2 = best_h.copy()
                tmp = h_trial2[i]
                h_trial2[i] = h_trial2[j]
                h_trial2[j] = tmp
                h_trial2 = project_feasible(h_trial2, n)
                c5_t2 = compute_c5(h_trial2)
                if c5_t2 < best_c5 - 1e-14:
                    best_c5 = c5_t2
                    best_h = h_trial2
    
    return best_h, best_c5

def swap_search(h, n, max_sweeps=20):
    """Systematic sweep of swaps for near-binary solutions."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(456)
    
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        order = rng.permutation(n)
        for i in range(min(200, len(order))):
            if remaining_time() < 1:
                break
            ii = int(order[i])
            if best_h[ii] < 0.3 or best_h[ii] > 0.7:
                continue
            # Try swapping with a few other positions
            for _ in range(5):
                jj = int(rng.randint(n))
                if jj == ii:
                    continue
                if abs(best_h[ii] - best_h[jj]) < 0.1:
                    continue
                h_trial = best_h.copy()
                tmp = h_trial[ii]
                h_trial[ii] = h_trial[jj]
                h_trial[jj] = tmp
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < best_c5 - 1e-14:
                    best_c5 = c5_t
                    best_h = h_trial
                    improved = True
                    break
        if not improved:
            break
    
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 256
        best_h = np.full(best_n, 0.5)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)
    
    n = best_n
    h = best_h.copy()
    
    # Phase 1: SLSQP with gradient, escalating beta
    for beta in [50, 150, 400, 1000, 2500, 5000, 10000]:
        if remaining_time() < 15:
            break
        h_new = slsqp_smooth_grad(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 2: L-BFGS-B with gradient
    if remaining_time() > 30:
        for beta in [500, 2000, 5000, 10000, 20000]:
            if remaining_time() < 10:
                break
            h_new = l_bfgsb_optimize(best_h, best_n, beta, maxiter=300)
            h_new = project_feasible(h_new, best_n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    # Phase 3: Binarize and local search
    if remaining_time() > 30:
        h_bin = binarize(best_h, best_n, threshold=0.15)
        c5_bin = compute_c5(h_bin)
        # Refine the binary solution
        h_ref = slsqp_smooth_grad(h_bin, best_n, 2000, maxiter=200)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
        elif c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
    
    # Phase 4: Multi-resolution
    trial_ns = [192, 224, 288, 320, 384, 448, 512, 640, 768, 1024]
    for trial_n in trial_ns:
        if remaining_time() < 20:
            break
        h_interp = interp_to(best_h, best_n, trial_n)
        h_interp = project_feasible(h_interp, trial_n)
        c5_interp = compute_c5(h_interp)
        
        if c5_interp < best_c5 + 0.005:
            maxit = 200 if trial_n <= 512 else 100
            for beta in [500, 2000, 5000]:
                if remaining_time() < 8:
                    break
                h_new = slsqp_smooth_grad(h_interp, trial_n, beta, maxiter=maxit)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = trial_n
                    h_interp = h_new.copy()
    
    # Phase 5: Flip search on near-binary
    if remaining_time() > 20:
        h_bin = binarize(best_h, best_n, threshold=0.2)
        h_flip, c5_flip = flip_search(h_bin, best_n, max_iter=8000)
        if c5_flip < best_c5:
            best_c5 = c5_flip
            best_h = h_flip.copy()
        # Also refine after flip search
        if remaining_time() > 10:
            h_ref = slsqp_smooth_grad(h_flip, best_n, 3000, maxiter=150)
            c5_ref = compute_c5(h_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
    
    # Phase 6: Swap search
    if remaining_time() > 15:
        h_bin = binarize(best_h, best_n, threshold=0.2)
        h_swap, c5_swap = swap_search(h_bin, best_n, max_sweeps=30)
        if c5_swap < best_c5:
            best_c5 = c5_swap
            best_h = h_swap.copy()
    
    # Phase 7: Final high-beta polish
    if remaining_time() > 15:
        for beta in [10000, 30000, 50000]:
            if remaining_time() < 5:
                break
            h_new = slsqp_smooth_grad(best_h, best_n, beta, maxiter=150)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
