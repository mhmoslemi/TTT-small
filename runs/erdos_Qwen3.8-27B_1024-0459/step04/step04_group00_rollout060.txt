from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break

        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def smax_objective(h, n, beta):
    dx = 2.0 / n
    corr = compute_corr_full(h)
    m = np.max(corr)
    shifted = corr - m
    clamped = np.clip(beta * shifted, -50.0, 50.0)
    ex = np.exp(clamped)
    return (m + np.log(np.sum(ex)) / beta) * dx


def smax_gradient(h, n, beta):
    dx = 2.0 / n
    corr = compute_corr_full(h)
    m = np.max(corr)
    shifted = corr - m
    clamped = np.clip(beta * shifted, -50.0, 50.0)
    ex = np.exp(clamped)
    s = np.sum(ex)
    w = ex / s
    g = np.zeros(n)
    nnm1 = n - 1
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        off = k - nnm1
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:ao + np_])
            g[ao:ao + np_] -= wk * h[:np_]
    return g


def slsqp_optimize(h, n, beta, maxiter=400):
    h = project_feasible(h.copy(), n)

    def obj(x):
        return smax_objective(x, n, beta)

    def grad(x):
        return smax_gradient(x, n, beta)

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.fun < smax_objective(h, n, beta):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

    for it in range(n_iter):
        if remaining() < 2:
            break
        corr = compute_corr_full(h)
        top_idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k in top_idx:
            off = k - nnm1
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += (1.0 - h[:np_])
                g[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= h[:np_]
        g /= len(top_idx)

        # Try multiple step sizes
        improved = False
        for step in [lr, lr * 0.5, lr * 2, lr * 0.1]:
            ht = project_feasible(h - step * g, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            lr *= 0.8
            if lr < 1e-6:
                break
    return h, best_c5


def binary_local_search(h_bin, n, time_limit=60, rng=None):
    """Local search on binary vectors with single and block moves."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            rem = rng.choice(ones_idx, n_ones - n_half, replace=False)
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            add = rng.choice(zeros_idx, n_half - n_ones, replace=False)
            h[add] = 1
    dx = 2.0 / n
    corr = compute_corr_full(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_c5 = cur_max * dx
    start = time.time()
    stall = 0

    for it in range(1000000):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        improved = False
        # Try single swaps
        for _ in range(30):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0
            h[j] = 1
            nc = compute_corr_full(h)
            nc_max = float(np.max(nc))
            h[i] = 1
            h[j] = 0

            if nc_max < cur_max - 1e-12:
                h[i] = 0
                h[j] = 1
                cur_max = nc_max
                if cur_max * dx < best_c5:
                    best_c5 = cur_max * dx
                    best_h = h.copy()
                improved = True
                stall = 0
                break
            elif nc_max <= cur_max + 1e-12 and rng.rand() < 0.1:
                h[i] = 0
                h[j] = 1
                improved = True
                break

        # Try block moves (move a small contiguous block of 1s)
        if not improved and rng.rand() < 0.3:
            block_len = rng.randint(2, 5)
            valid_starts = [i for i in range(len(ones) - block_len + 1)
                           if ones[i + block_len - 1] - ones[i] == block_len - 1]
            if valid_starts:
                si = valid_starts[rng.randint(len(valid_starts))]
                block = list(ones[si:si + block_len])
                # Find a valid destination (contiguous zeros of same length not overlapping)
                dest_candidates = []
                for z0 in zeros:
                    if all(z0 + d < n for d in range(block_len)):
                        if all(h[z0 + d] == 0 for d in range(block_len)):
                            if not any(b >= z0 and b < z0 + block_len for b in block):
                                dest_candidates.append(z0)
                if dest_candidates:
                    z0 = dest_candidates[rng.randint(len(dest_candidates))]
                    for b in block:
                        h[b] = 0
                    for d in range(block_len):
                        h[z0 + d] = 1
                    nc = compute_corr_full(h)
                    nc_max = float(np.max(nc))
                    if nc_max >= cur_max + 1e-12:
                        # Revert
                        for d in range(block_len):
                            h[z0 + d] = 0
                        for b in block:
                            h[b] = 1
                    else:
                        cur_max = nc_max
                        if cur_max * dx < best_c5:
                            best_c5 = cur_max * dx
                            best_h = h.copy()
                        improved = True
                        stall = 0

        if not improved:
            stall += 1
            if stall > 500:
                break
    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        pc = bin(i).count('1') % 2
        h[i] = 1.0 if pc == 0 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            idx = np.where(h == 1)[0]
            rm = idx[np.random.RandomState(0).choice(len(idx), n_ones - target, replace=False)]
            h[rm] = 0
        else:
            idx = np.where(h == 0)[0]
            add = idx[np.random.RandomState(0).choice(len(idx), target - n_ones, replace=False)]
            h[add] = 1
    return h


def coordinate_descent_max(h, n, max_passes=20):
    """For each coordinate, try to optimally set it to reduce the max correlation."""
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    nnm1 = n - 1

    for p in range(max_passes):
        if remaining() < 3:
            break
        improved = False
        # Only bother with fractional coordinates
        frac = np.where((h > 0.01) & (h < 0.99))[0]
        for j in frac:
            if remaining() < 2:
                break
            orig = h[j]
            # Try different values
            candidates = [0.0, orig - 0.1, orig - 0.05, orig + 0.05, orig + 0.1, 1.0]
            candidates = [max(0.0, min(1.0, c)) for c in candidates]
            best_val = orig
            best_obj = best_c5
            for cv in candidates:
                if abs(cv - orig) < 1e-8:
                    continue
                h[j] = cv
                # Adjust sum (tiny correction)
                c5_val = compute_c5(project_feasible(h, n))
                if c5_val < best_obj - 1e-14:
                    best_obj = c5_val
                    best_val = cv
            if best_val != orig:
                h[j] = best_val
                h = project_feasible(h, n)
                if compute_c5(h) < best_c5 - 1e-14:
                    best_c5 = compute_c5(h)
                    improved = True
                else:
                    h[j] = orig
                    h = project_feasible(h, n)
            else:
                h[j] = orig
        if not improved:
            break
    return h, best_c5


def sluguplicate_solve(h, n):
    """SlugDuplicate-style: solve min sum_k w_k * g_k(h) with adaptive weights."""
    return h, compute_c5(h)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP refinement with increasing beta
    for beta in [50, 100, 300, 800, 2000, 5000, 12000, 30000]:
        if remaining() < 12:
            break
        h_new = slsqp_optimize(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Projected subgradient on active set
    if remaining() > 20:
        h_pg, c5_pg = projected_subgrad(h, n, n_iter=800, lr_init=0.03, top_k=6)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()
        h_pg2, c5_pg2 = projected_subgrad(h, n, n_iter=500, lr_init=0.01, top_k=10)
        if c5_pg2 < best_c5:
            best_c5 = c5_pg2
            best_h = h_pg2.copy()
            h = h_pg2.copy()

    # Phase 3: SLSQP again after subgd
    for beta in [10000, 30000, 80000]:
        if remaining() < 10:
            break
        h_new = slsqp_optimize(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Binary local search with block moves
    if remaining() > 30:
        h_bin = (h > 0.5).astype(float)
        sort_idx = np.argsort(-h)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n // 2]] = 1.0

        sa_time = min(120, max(15, int(remaining() * 0.25)))
        h_b, c5_b = binary_local_search(h_bin, n, time_limit=sa_time, rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

        # Refine binary with SLSQP
        h_ref = h_b.copy()
        for beta in [5000, 15000, 50000]:
            if remaining() < 8:
                break
            h_ref = slsqp_optimize(h_ref, n, beta, maxiter=400)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            h = h_ref.copy()

    # Phase 5: Randomized rounding + local search (multiple restarts)
    if remaining() > 40:
        for attempt in range(3):
            if remaining() < 15:
                break
            # Random rounding of current best
            h_rr = np.zeros(n)
            for i in range(n):
                if rng.rand() < h[i]:
                    h_rr[i] = 1.0
            n_ones = int(np.sum(h_rr))
            if n_ones > n // 2:
                idx = np.where(h_rr == 1)[0]
                h_rr[idx[rng.choice(len(idx), n_ones - n // 2, replace=False)]] = 0
            elif n_ones < n // 2:
                idx = np.where(h_rr == 0)[0]
                h_rr[idx[rng.choice(len(idx), n // 2 - n_ones, replace=False)]] = 1

            # Quick binary local search
            h_rr2, c5_rr2 = binary_local_search(h_rr, n, time_limit=min(30, max(5, int(remaining() * 0.1))), rng=rng)
            if c5_rr2 < best_c5:
                best_c5 = c5_rr2
                best_h = h_rr2.copy()

            # Refine
            for beta in [10000, 40000]:
                if remaining() < 6:
                    break
                h_rr3 = slsqp_optimize(h_rr2, n, beta, maxiter=300)
            c5_rr3 = compute_c5(h_rr3)
            if c5_rr3 < best_c5:
                best_c5 = c5_rr3
                best_h = h_rr3.copy()

    # Phase 6: Try different resolutions
    if remaining() > 25:
        n_candidates = set()
        for delta in [-6, -4, -2, 2, 4, 6, 8, 12, 16, 24, 32, -8]:
            nn = best_n + delta
            if 128 <= nn <= 768 and nn % 2 == 0:
                n_candidates.add(nn)
        # Also try powers of 2 or near
        for nn in [256, 320, 384, 448, 512]:
            if 128 <= nn <= 768 and nn != best_n:
                n_candidates.add(nn)

        n_candidates = sorted(n_candidates, key=lambda x: abs(x - best_n))

        for n_t in n_candidates[:8]:
            if remaining() < 15:
                break
            h_interp = interp_to(best_h, best_n, n_t)
            h_feas = project_feasible(h_interp, n_t)
            c5_int = compute_c5(h_feas)

            if c5_int < best_c5 + 0.005:
                h_opt = h_feas
                for beta in [500, 3000, 15000]:
                    if remaining() < 8:
                        break
                    h_opt = slsqp_optimize(h_opt, n_t, beta, maxiter=350)
                c5_opt = compute_c5(h_opt)
                # Subgrad
                h_opt, c5_opt = projected_subgrad(h_opt, n_t, n_iter=300, lr_init=0.02, top_k=6)

                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_t

    # Phase 7: Final intensive refinement
    n = best_n
    h = best_h.copy()

    for beta in [20000, 50000, 100000, 200000]:
        if remaining() < 5:
            break
        h_new = slsqp_optimize(h, n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining() > 8:
        h_eq, c5_eq = projected_subgrad(h, n, n_iter=1000, lr_init=0.005, top_k=4)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    if remaining() > 5:
        h_cd, c5_cd = coordinate_descent_max(best_h, n, max_passes=10)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if remaining() > 5:
        h_new = slsqp_optimize(best_h, n, 300000, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Ensure feasibility and get final value
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    # Validate with evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
