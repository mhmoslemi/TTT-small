                for _ in range(55):
                    if v1 < v2:
                        hi = c2
                        c2 = c1
                        v2 = v1
                        c1 = hi - gr * (hi - lo)
                        v1 = np.max(base + a * c1 + b * c1 * c1) * dx
                    else:
                        lo = c1
                        c1 = c2
                        v1 = v2
                        c2 = lo + gr * (hi - lo)
                        v2 = np.max(base + a * c2 + b * c2 * c2) * dx
                t_best = (lo + hi) / 2.0
                f_best = np.max(base + a * t_best + b * t_best * t_best) * dx
            if f_best < best_c5 - 1e-15:
                best[i] = t_best
                best_c5 = f_best
                corr_cur = np.correlate(best, 1.0 - best, mode='full')
                improved += 1
        if improved == 0:
            break
    return best, compute_c5(best)

def subgrad_step(h, n, n_active=8):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    active = np.where(corr >= max_corr - 0.5)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        top_probs = corr[active] - max_corr
        order = np.argsort(-top_probs)
        active = active[order[:n_active]]
    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        if offset == 0:
            grad += (1.0 - 2.0 * h)
        elif offset > 0:
            npairs = n - offset
            grad[offset:offset + npairs] += (1.0 - h[:npairs])
            grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
            grad[aoff:aoff + npairs] -= h[:npairs]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return None
    return grad / gnorm

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        h_tmp = np.zeros(best_n)
        for i in range(best_n):
            v = i
            cnt = 0
            while v:
                cnt ^= (v & 1)
                v >>= 1
            h_tmp[i] = 1.0 if cnt == 1 else 0.0
        best_h = project_feasible(h_tmp, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: Aggressive coordinate descent at base resolution
    h, c5_new = coord_descent(h, n, 200, seed_val=seed)
    if c5_new < best_c5:
        best_c5 = c5_new
        best_h = h.copy()

    # Phase 2: SLSQP refinement with increasing beta
    for beta in [200, 500, 2000, 8000, 30000, 100000]:
        if remaining_time() < 15:
            break
        h_new = slsqp_refine(best_h, best_n, beta, 400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-16:
            best_c5 = c5_new
            best_h = h_new.copy()
            if remaining_time() < 10:
                break

    # Phase 3: Coordinate descent after SLSQP
    if remaining_time() > 15:
        h, c5_new = coord_descent(best_h, best_n, 150, seed_val=seed + 1)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 4: Subgradient descent with line search
    if remaining_time() > 30:
        h = best_h.copy()
        for it in range(500):
            if remaining_time() < 5:
                break
            grad = subgrad_step(h, best_n, 6)
            if grad is None:
                break
            base_step = 0.05 / (1.0 + it * 0.005)
            improved = False
            for sm in [1.0, 0.5, 2.0, 0.25, 0.1, 0.75, 0.05]:
                step = base_step * sm
                h_trial = h - step * grad
                h_trial = project_feasible(h_trial, best_n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-17:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h = h_trial.copy()
                    improved = True
                    break
            if not improved:
                if it % 100 == 50 and remaining_time() > 20:
                    h_new = slsqp_refine(h, best_n, 5000, 200)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h = h_new.copy()
                if it > 300:
                    break

    # Phase 5: Try higher resolution
    if remaining_time() > 60:
        for rn in [2 * best_n, int(best_n * 1.5), 1280, 960]:
            if remaining_time() < 40:
                break
            if rn % 2 != 0 or rn < 128 or rn > 2560 or rn == best_n:
                continue
            h_r = interp_to(best_h, best_n, rn)
            h_r = project_feasible(h_r, rn)
            c5_r = compute_c5(h_r)
            h_r, c5_r = coord_descent(h_r, rn, 80, seed_val=seed + 2)
            if c5_r < best_c5 - 1e-10:
                for beta in [2000, 10000]:
                    if remaining_time() < 15:
                        break
                    h_new = slsqp_refine(h_r, rn, beta, 200)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_r:
                        h_r = h_new
                        c5_r = c5_new
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r.copy()
                    best_n = rn
                    h_r, c5_r2 = coord_descent(h_r, rn, 50, seed_val=seed + 3)
                    if c5_r2 < best_c5:
                        best_c5 = c5_r2
                        best_h = h_r.copy()
                break

    # Phase 6: Final coordinate descent on best
    if remaining_time() > 10:
        h, c5_new = coord_descent(best_h, best_n, 100, seed_val=seed + 4)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 7: Final SLSQP polish
    if remaining_time() > 10:
        for beta in [50000, 200000]:
            if remaining_time() < 5:
                break
            h_new = slsqp_refine(best_h, best_n, beta, 300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 8: Micro subgradient
    if remaining_time() > 5:
        h = best_h.copy()
        for it in range(100):
            if remaining_time() < 2:
                break
            grad = subgrad_step(h, best_n, 4)
            if grad is None:
                break
            improved = False
            for step in [0.001, 0.003, 0.005, 0.002, 0.008]:
                h_trial = h - step * grad
                h_trial = project_feasible(h_trial, best_n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-17:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h = h_trial.copy()
                    improved = True
                    break
            if not improved:
                break

    # Phase 9: One more coordinate descent pass
    if remaining_time() > 5:
        h, c5_new = coord_descent(best_h, best_n, 50, seed_val=seed + 5)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
