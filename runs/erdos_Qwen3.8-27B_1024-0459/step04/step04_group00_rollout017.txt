from scipy.fft import fft, ifft
import time

_T0 = 0.0
_BUDGET = 1300.0

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        # Distribute evenly among elements not at bounds
        upper = h > 0.999999
        lower = h < 1e-9
        free = ~(upper | lower)
        n_free = np.count_nonzero(free)
        if n_free > 0:
            h[free] += diff / n_free
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)

    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

def corr_gradient(h, weights=None):
    """Compute gradient of weighted sum of correlations."""
    n = len(h)
    if weights is None:
        weights = np.ones(2*n-1) / (2*n-1)
    g = np.zeros(n)
    for k in range(2*n-1):
        wk = weights[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            npairs = n - off
            g[off:off+npairs] += wk * (1.0 - h[:npairs])
            g[:npairs] -= wk * h[off:off+npairs]
        else:
            a = -off
            npairs = n - a
            g[:npairs] += wk * (1.0 - h[a:a+npairs])
            g[a:a+npairs] -= wk * h[:npairs]
    return g

def smooth_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx

    n = len(h)
    dx = 2.0 / n

    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)

    w = ex / np.sum(ex)
    w = w * dx

        c5_new = compute_c5(h_new)

                    best_h = h_try.copy()
                    h = h_try.copy()
                    break
            else:
                h = h_new  # accept anyway to explore
        # Decay step
        step_scale *= 0.999
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(120):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def corr_grad_weighted(h, weights):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wk * h[:np_]
    return g


def smooth_obj(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(ex)) / beta) * dx


def smooth_grad(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    dx = 2.0 / len(h)
    w = w * dx
    return corr_grad_weighted(h, w)


def slsqp_run(h, n, beta, maxiter=500):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        return smooth_obj(x, beta)

    def _grad(x):
        return smooth_grad(x, beta)

    def _sc(x):
        return np.sum(x) - n / 2.0

    def _scj(x):
        return np.ones(n)

    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sc, 'jac': _scj}],
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.fun < _obj(h0) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def subgrad_topk(h, n, top_k=5, max_iters=800):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    mu = 0.85
    step = 0.05

    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:top_k]
        c_top = corr[top_idx]
        w_top = c_top - c_top.min() + 1e-10
        w_top = w_top ** 2
        w_top /= w_top.sum()
        weights = np.zeros(2 * n - 1)
        weights[top_idx] = w_top
        weights *= dx
        g = corr_grad_weighted(h, weights)
        g -= np.mean(g)
        momentum = mu * momentum - step * g
        h_new = project_feasible(h + momentum, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        elif it % 50 == 0:
            h_rollback = project_feasible(h - 0.5 * momentum, n)
            cr = compute_c5(h_rollback)
            if cr < compute_c5(h):
                h = h_rollback
        step *= 0.9995
        if it % 100 == 0:
            mu = min(0.95, mu + 0.01)
    return best_h, best_c5


def binary_exhaustive_ls(h_bin, n, max_rounds=20):
    h = (h_bin > 0.5).astype(float).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for rnd in range(max_rounds):
        if remaining() < 10:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        corr_cur = compute_corr(h)
        c_max = np.max(corr_cur)
        best_improve = 1e-15
        best_pair = None

        for ia in range(len(ones)):
            pa = ones[ia]
            h[pa] = 0.0
            for ib in range(len(zeros)):
                pb = zeros[ib]
                h[pb] = 1.0
                nc = np.max(compute_corr(h))
                h[pb] = 0.0
                imp = c_max - nc
                if imp > best_improve:
                    best_improve = imp
                    best_pair = (pa, pb)
            h[pa] = 1.0

        if best_pair is None:
            break
        pa, pb = best_pair
        h[pa] = 0.0
        h[pb] = 1.0
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def round_to_binary(h, n):
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin


def nesterov_smooth(h, n, beta, max_iters=2000):
    """Nesterov accelerated gradient on smooth max with projection."""
    h = project_feasible(h, n)
    y = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_val = 1.0

    for it in range(max_iters):
        if remaining() < 5:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        alpha = (t_val - 1.0) / t_new
        y_new = y + alpha * (y - h)
        g = smooth_grad(h, beta)
        lr = 1.0 / (beta * (it + 10))
        h_new = project_feasible(y_new - lr * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        y = h
        h = h_new
        t_val = t_new
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        best_h = parent_h.copy()
        best_n = parent_n
        best_c5 = compute_c5(best_h)
    except Exception:
        parent_n = 384
        best_n = parent_n
        best_h = np.zeros(parent_n)
        best_h[:parent_n // 2] = 1.0
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: Progressive SLSQP
    for beta in [30, 100, 300, 1000, 3000, 8000, 20000, 50000]:
        if remaining() < 15:
            break
        h_new = slsqp_run(h, n, beta, maxiter=min(500, int(remaining() // 10)))
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n
            h = h_new.copy()

    # Phase 2: Subgradient top-k equalization
    if remaining() > 30:
        for tk in [3, 5, 8, 12]:
            if remaining() < 15:
                break
            h_sg, c5_sg = subgrad_topk(h, n, top_k=tk, max_iters=600)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()
                h = h_sg.copy()

    # Phase 3: Nesterov accelerated gradient
    if remaining() > 30:
        for beta in [2000, 5000, 15000]:
            if remaining() < 15:
                break
            h_nes, c5_nes = nesterov_smooth(h, n, beta, max_iters=1500)
            if c5_nes < best_c5:
                best_c5 = c5_nes
                best_h = h_nes.copy()
                h = h_nes.copy()

    # Phase 4: Binary exhaustive local search
    if remaining() > 30:
        h_bin = round_to_binary(best_h, best_n)
        h_b, c5_b = binary_exhaustive_ls(h_bin, best_n, max_rounds=15)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = best_n

        # Also try SLSQP starting from binary solution
        if remaining() > 20:
            h_from_bin = h_b.copy()
            for beta in [5000, 20000]:
                if remaining() < 10:
                    break
                h_opt = slsqp_run(h_from_bin, best_n, beta, maxiter=400)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    h_from_bin = h_opt.copy()

    # Phase 5: Multi-resolution exploration
    n_candidates = []
    for scale in [1.25, 1.5, 2.0, 2.5, 0.75, 0.5]:
        nn = int(best_n * scale)
        if nn >= 96 and nn <= 1024 and nn % 2 == 0:
            n_candidates.append(nn)
    n_candidates = sorted(set([x for x in n_candidates if x != best_n]))
    n_candidates.sort(key=lambda x: abs(x - best_n))

    for n_t in n_candidates[:4]:
        if remaining() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_interp = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_interp)

        h_opt = h_interp
        for beta in [500, 3000, 10000, 30000]:
            if remaining() < 15:
                break
            h_opt = slsqp_run(h_opt, n_t, beta, maxiter=400)
        c5_opt = compute_c5(h_opt)

        if remaining() > 20:
            h_opt, c5_opt = subgrad_topk(h_opt, n_t, top_k=5, max_iters=400)

        # Binary LS at this resolution
        if remaining() > 20:
            h_b2 = round_to_binary(h_opt, n_t)
            h_b2, c5_b2 = binary_exhaustive_ls(h_b2, n_t, max_rounds=8)
            if c5_b2 < c5_opt:
                c5_opt = c5_b2
                h_opt = h_b2.copy()

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 6: Final intensive polish at best resolution
    n = best_n
    h = best_h.copy()
    for beta in [10000, 30000, 80000, 200000]:
        if remaining() < 8:
            break
        h_new = slsqp_run(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 7: Final subgradient + top-k
    if remaining() > 10:
        h_eq, c5_eq = subgrad_topk(h, n, top_k=7, max_iters=800)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 8: Final binary tweak
    if remaining() > 15:
        h_bin_final = round_to_binary(best_h, best_n)
        h_bf, c5_bf = binary_exhaustive_ls(h_bin_final, best_n, max_rounds=10)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()
        # SLSQP from best binary
        if remaining() > 10:
            h_pol = slsqp_run(h_bf, best_n, 50000, maxiter=400)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Final projection and validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        best_c5 = final_c5
    else:
        best_h = best_h.copy()

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
