        _G_BETA = 500.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_sum}]
        try:
            res = minimize(obj_smooth, h_trial, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 80, 'ftol': 1e-13, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-12:
                c5_best = c5_new
                h_best = h_new.copy()
        except Exception:
            pass
    
    return h_best, c5_best
