        for beta in [100, 500, 2000]:
            if remaining_time() < 30:
                break
            h_new = lbfgs9_optimize(h_r, rn, beta, maxiter=2000)
            c5_new = compute_c5(h_new)
            if c5_new < c5_r:
                h_r = h_new
                c5_r = c5_new
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = rn

        # One SLSQP pass
        if remaining_time() > 25:
            h_new = slsqp_refine(h_r, rn, 3000, 300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = rn

    # Phase 4: Coordinate descent on best
    if remaining_time() > 20:
        h_new, c5_new = coordinate_descent(best_h.copy(), best_n, 50)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 5: Nesterov on best
    if remaining_time() > 30:
        for beta in [300, 1000, 3000]:
            if remaining_time() < 15:
                break
            h_new, c5_new = nesterov_iter(best_h.copy(), best_n, beta, 500, lr_base=0.03)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 6: Try fresh binary initialization at high res
    if remaining_time() > 120:
        for try_n in [best_n, 2 * best_n, 1280]:
            if remaining_time() < 40:
                break
            if try_n != best_n and try_n > 2048:
                continue
            h_rand = binary_init(try_n, seed + try_n)
            c5_rand = compute_c5(h_rand)

            for beta in [100, 500, 2000, 5000]:
                if remaining_time() < 20:
                    break
                h_new = lbfgs9_optimize(h_rand, try_n, beta, maxiter=2000)
                c5_new = compute_c5(h_new)
                if c5_new < c5_rand:
                    h_rand = h_new
                    c5_rand = c5_new
                if c5_rand < best_c5:
                    best_c5 = c5_rand
                    best_h = h_rand.copy()
                    best_n = try_n

    # Phase 7: Thue-Morse init at high res
    if remaining_time() > 80:
        for try_n in [best_n, 1280, 2048]:
            if remaining_time() < 35:
                break
            if try_n > 2048:
                continue
            h_tm = thue_morse_init(try_n, seed)
            c5_tm = compute_c5(h_tm)

            for beta in [200, 1000, 4000]:
                if remaining_time() < 15:
                    break
                h_new = lbfgs9_optimize(h_tm, try_n, beta, maxiter=1500)
                c5_new = compute_c5(h_new)
                if c5_new < c5_tm:
                    h_tm = h_new
                    c5_tm = c5_new
                if c5_tm < best_c5:
                    best_c5 = c5_tm
                    best_h = h_tm.copy()
                    best_n = try_n

    # Phase 8: Final coordinate descent + SLSQP polish
    if remaining_time() > 15:
        h_new, c5_new = coordinate_descent(best_h.copy(), best_n, 40)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if remaining_time() > 10:
        h_new = slsqp_refine(best_h.copy(), best_n, 10000, 300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 9: Micro coordinate descent
    if remaining_time() > 8:
        h_new, c5_new = coordinate_descent(best_h.copy(), best_n, 30)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
