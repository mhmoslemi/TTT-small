```python
            m = k_idx - (n - 1)
            val = 0.0
            if 0 <= i + m < n:
                val += (1.0 - h[i + m])
            if 0 <= i - m < n:
                val -= h[i - m]
            s += p[k_idx] * val
        grad[i] = s + 2 * penalty_M * (sum_h - target)
    
    return obj, grad
```

```python
def coordinate_descent(h, n, max_sweeps=50):
    h = h.copy()
    dx = 2.0 / n
    corr_full = np.correlate(h, 1.0 - h, mode='full')
    n_lags = len(corr_full)  # = 2n-1
    
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        any_change = False
        for i in range(n):
            # Compute B_k (coefficient of h[i] in each corr_k)
            B = np.zeros(n_lags)
            for k_idx in range(n_lags):
                m = k_idx - (n - 1)
                b = 0.0
                if 0 <= i + m < n:
                    b += (1.0 - h[i + m])
                if 0 <= i - m < n:
                    b -= h[i - m]
                B[k_idx] = b
            
            # A_k = corr_k - B_k * h[i]
            A = corr_full - B * h[i]
            
            # 1D minimax: min_x max_k (A_k + B_k * x) for 0<=x<=1
            # ... (as before)
            
            new_x = best_x
            delta = new_x - h[i]
            if abs(delta) > 1e-10:
                h[i] = new_x
                corr_full += B * delta  # Update correlation
                any_change = True
        
        h = project_feasible(h, n)
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        
        if not any_change:
            break
    
    return h
```

The key insight: `corr_full += B * delta` correctly updates the correlation because corr_k is linear in h[i] (when other h are fixed).

This is O(n * n_lags) = O(n * 2n) = O(n²) per sweep for the B computation, plus O(n_lags * top_k²) for the 1D minimax. Very fast!
