def corr_delta_binary(h, i, j, n):
    """Compute delta correlation when h[i]: 1->0, h[j]: 0->1. Returns array of length 2n-1."""
    delta = np.zeros(2*n - 1)
    omh = 1.0 - h
    center = n - 1  # offset: d maps to index d + center
    
    # Term 1: -(1-h[i+d]) for d in [-i, n-1-i]
    d_lo = max(-(n-1), -i)  # d >= -(n-1) (correlation range)
    d_hi = min(n-1, n-1-i)
    if d_lo <= d_hi:
        m_lo = d_lo + center
        m_hi = d_hi + center
        delta[m_lo:m_hi+1] -= omh[i+d_lo : i+d_hi+1]
    
    # Term 2: h[i-d] for d in [i-n+1, i]
    d_lo = max(-(n-1), i-n+1)
    d_hi = min(n-1, i)
    if d_lo <= d_hi:
        m_lo = d_lo + center
        m_hi = d_hi + center
        # h[i-d] for d from d_lo to d_hi → indices i-d_hi to i-d_lo
        delta[m_lo:m_hi+1] += h[i-d_hi : i-d_lo+1][::-1]
    
    # Term 3: (1-h[j+d]) for d in [-j, n-1-j]
    d_lo = max(-(n-1), -j)
    d_hi = min(n-1, n-1-j)
    if d_lo <= d_hi:
        m_lo = d_lo + center
        m_hi = d_hi + center
        delta[m_lo:m_hi+1] += omh[j+d_lo : j+d_hi+1]
    
    # Term 4: -h[j-d] for d in [j-n+1, j]
    d_lo = max(-(n-1), j-n+1)
    d_hi = min(n-1, j)
    if d_lo <= d_hi:
        m_lo = d_lo + center
        m_hi = d_hi + center
        delta[m_lo:m_hi+1] -= h[j-d_hi : j-d_lo+1][::-1]
    
    # Corrections
    delta[center] += -2  # d=0
    di_j = j - i
    if -(n-1) <= di_j <= n-1:
        delta[di_j + center] += 1
    dj_i = i - j
    if -(n-1) <= dj_i <= n-1:
        delta[dj_i + center] += 1
    
    return delta
