def project_feasible(h, n):

def grad_corr_k(h, k, n):
    off = k - (n - 1)
    g = np.zeros(n)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - h[:npairs])
        g[:npairs] -= h[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - h[ao:ao + npairs])
        g[ao:ao + npairs] -= h[:npairs]
    return g


def grad_corr_all_weighted(h, weights, n):

def slsqp_smooth(h, n, beta, maxiter=400):
    """SLSQP with log-sum-exp smoothing."""
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        if s < 1e-300:
            return m * dx
        return (m + np.log(s) / beta) * dx
    
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        if s < 1e-300:
            w = np.zeros(2 * n - 1)
            w[np.argmax(corr)] = 1.0
        else:
            w = ex / s
        return grad_corr_all_weighted(x, w * dx, n)
    
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    
    best_res = h.copy()
    best_fun = obj(h)
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq',
                                     'fun': lambda x: np.sum(x) - target,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < best_fun:
            best_res = project_feasible(res.x, n)
            best_fun = obj(best_res)
    except Exception:
        pass
    
    return best_res


def binary_sa(h_bin, n, time_limit=60, seed=42):
    """Simulated annealing on binary vectors."""

            h[rem] = 0

        if time.time() - start > time_limit or remaining_time() < 5:
            break
        
        T = T0_temp * (T1_temp / T0_temp) ** (gen / max(n_gen, 1))
        if T < 0.001:
            T = 0.001
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        nc = compute_corr(h)

            h[j] = 0

                idx = np.where(h == 1)[0]
                h[idx[rng.choice(len(idx), n_ones - n // 2, replace=False)]] = 0
            else:
                idx = np.where(h == 0)[0]
                h[idx[rng.choice(len(idx), n // 2 - n_ones, replace=False)]] = 1
        h = project_feasible(h, n)
        best_h = h.copy()
        best_c5 = compute_c5(h)
        best_n = n
    
    # Phase 1: Coarse-to-fine SLSQP at parent resolution
    for beta in [50, 200, 800, 3000, 8000, 20000]:
        if remaining_time() < 12:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=350)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new.copy()
    
    # Phase 2: Iterative equalization with multiple top_m values
    for top_m in [5, 8, 12, 16, 20]:
        if remaining_time() < 15:
            break
        h_eq, c5_eq = iterative_equalize(h, n, max_iters=4000, top_m=top_m)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
        h = h_eq.copy()
    
    # Phase 3: Multi-resolution search
    n_candidates = []
    for delta in [-16, -12, -8, -6, -4, -2, 2, 4, 6, 8, 12, 16, 20, 24, 32]:
        nc = n + delta
        if 256 <= nc <= 700 and nc % 2 == 0 and nc != n:
            n_candidates.append(nc)
    n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - n))
    
    for n_t in n_candidates[:12]:
        if remaining_time() < 25:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        
        # Quick optimization
        for beta in [1000, 5000]:
            if remaining_time() < 10:
                break
            h_feas = slsqp_smooth(h_feas, n_t, beta, maxiter=300)
        
        h_feas, c5_tmp = iterative_equalize(h_feas, n_t, max_iters=2500, top_m=10)
        
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h_feas.copy()
            best_n = n_t
            n = n_t
            h = h_feas.copy()
    
    # Phase 4: Binary SA for global exploration
    if remaining_time() > 40:
        h_bin_init = (h > 0.5).astype(float)
        n_half = n // 2
        n_ones = int(np.sum(h_bin_init))
        if n_ones != n_half:
            sort_idx = np.argsort(-h)
            h_bin = np.zeros(n)
            h_bin[sort_idx[:n_half]] = 1.0
        else:
            h_bin = (h > 0.5).astype(float)
        
        sa_time = min(120, max(15, int(remaining_time() * 0.35)))
        h_b, c5_b = binary_sa(h_bin, n, time_limit=sa_time, seed=seed)
        
        # Refine SA result
        h_ref = h_b.copy()
        for beta in [3000, 10000]:
            if remaining_time() < 10:
                break
            h_ref = slsqp_smooth(h_ref, n, beta, maxiter=300)
        h_ref, c5_ref = iterative_equalize(h_ref, n, max_iters=3000, top_m=10)
        
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            h = h_ref.copy()
        
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    
    # Phase 5: Fractional focus optimization
    if remaining_time() > 20:
        for nf in [10, 20, 30]:
            if remaining_time() < 10:
                break
            h_frac = fractional_focus(h, n, n_frac_target=nf, beta=8000, maxiter=250)
            c5_frac = compute_c5(h_frac)
            if c5_frac < best_c5:
                best_c5 = c5_frac
                best_h = h_frac.copy()
            h = h_frac.copy()
    
    # Phase 6: Final intensive refinement
    for beta in [10000, 20000, 50000]:
        if remaining_time() < 8:
            break
        h_new = slsqp_smooth(best_h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 7: Final equalization with various parameters
    for top_m in [6, 10, 14, 20, 30]:
        if remaining_time() < 8:
            break
        h_eq, c5_eq = iterative_equalize(best_h, n, max_iters=8000, top_m=top_m)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
    
    # Phase 8: Try a couple more resolutions with the improved solution
    if remaining_time() > 20:
        for n_t in [n - 2, n + 2, n - 4, n + 4]:
            if remaining_time() < 15:
                break
            if n_t < 256 or n_t > 700 or n_t % 2 != 0:
                continue
            h_interp = interp_to(best_h, best_n, n_t)
            h_feas = project_feasible(h_interp, n_t)
            h_feas = slsqp_smooth(h_feas, n_t, 8000, maxiter=300)
            h_feas, c5_tmp = iterative_equalize(h_feas, n_t, max_iters=3000, top_m=10)
            if c5_tmp < best_c5:
                best_c5 = c5_tmp
                best_h = h_feas.copy()
                best_n = n_t
    
    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5
    
    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
