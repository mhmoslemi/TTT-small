def slsqp_lob(h0, n, beta, maxiter):
    """SLSQP with log-sum-exp objective."""
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = corr - m
        e = np.exp(np.clip(beta * s, -400, 400))
        return (m + np.log(np.sum(e)) / beta) * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def slsqp_run(h0, n, beta, maxiter):
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = corr - m
        e = np.exp(np.clip(beta * s, -400, 400))
        return float((m + np.log(np.sum(e)) / beta) * dx)

        m = np.max(corr)
        s = corr - m

        print("  After coord: C5={:.8f}".format(best_c5))

    # Phase 4: More subgradient with tighter active set
    if tlimit() > 60:
        print("\n=== Phase 4: Subgradient tight ===")
        for thresh in [0.005, 0.002, 0.001]:
            if tlimit() < 10:
                break
            h_sg, c5_sg = subgrad_descent(best_h, n, max_iters=2000, tlimit=tlimit,
                                          initial_step=0.2, active_thresh=thresh)
            if c5_sg < best_c5 - 1e-12:
                best_c5 = c5_sg
                best_h = h_sg
        print("  After subgrad tight: C5={:.8f}".format(best_c5))

    # Phase 5: SLSQP high-beta polish
    if tlimit() > 80:
        print("\n=== Phase 5: SLSQP high beta ===")
        for bv in [2000, 5000, 10000]:
            if tlimit() < 15:
                break
            h_new = slsqp_run(best_h, n, bv, maxiter=400)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-13:
                best_c5 = c_new
                best_h = h_new
        print("  After SLSQP high: C5={:.8f}".format(best_c5))

    # Phase 6: Swap descent to escape local optima
    if tlimit() > 40:
        print("\n=== Phase 6: Swap descent ===")
        h_sw, c5_sw = swap_descent(best_h, n, max_sweeps=20, rng=rng, tlimit=tlimit)
        if c5_sw < best_c5:
            best_c5 = c5_sw
            best_h = h_sw
            print("  Swap improved: C5={:.8f}".format(c5_sw))
        else:
            print("  Swap: no improvement (C5={:.8f})".format(c5_sw))

    # Phase 7: Perturbation + re-optimize
    if tlimit() > 80:
        print("\n=== Phase 7: Perturbation escapes ===")
        for attempt in range(6):
            if tlimit() < 20:
                break
            h_pert = best_h.copy()
            n_pert = rng.randint(3, 10)
            idxs = rng.choice(n, size=n_pert, replace=False)
            for idx in idxs:
                h_pert[idx] = np.clip(h_pert[idx] + rng.randn() * 0.25, 0.0, 1.0)
            h_pert = project_feasible(h_pert, n)
            c5_p = compute_c5(h_pert)
            if c5_p < best_c5 * 1.02:
                for bv in [100, 500, 2000]:
                    if tlimit() < 10:
                        break
                    h_new = slsqp_run(h_pert, n, bv, maxiter=200)
                    c_new = compute_c5(h_new)
                    if c_new < c5_p:
                        c5_p = c_new
                        h_pert = h_new
                if c5_p < best_c5 - 1e-10:
                    best_c5 = c5_p
                    best_h = h_pert.copy()
                    print("  Perturb {} improved: C5={:.8f}".format(attempt, c5_p))

    # Phase 8: Coordinate descent again
    if tlimit() > 40:
        print("\n=== Phase 8: Final coord descent ===")
        h_cd, c5_cd = coord_descent(best_h, n, max_sweeps=50, rng=rng, tlimit=tlimit)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
        print("  After final coord: C5={:.8f}".format(best_c5))

    # Phase 9: Try higher resolution (n*2)
    if tlimit() > 120:
        print("\n=== Phase 9: Higher resolution ===")
        n2 = n * 2
        h2 = project_feasible(interp_h(best_h, n, n2), n2)
        c5_2 = compute_c5(h2)
        print("  n={} initial C5={:.8f}".format(n2, c5_2))

        # Subgradient at higher res
        if tlimit() > 60:
            h_sg2, c5_sg2 = subgrad_descent(h2, n2, max_iters=2000, tlimit=tlimit,
                                            initial_step=0.3, active_thresh=0.02)
            if c5_sg2 < c5_2:
                h2 = h_sg2
                c5_2 = c5_sg2

        # SLSQP at higher res
        for bv in [20, 100, 400]:
            if tlimit() < 15:
                break
            h_new2 = slsqp_run(h2, n2, bv, maxiter=150)
            c_new2 = compute_c5(h_new2)
            h2 = h_new2
            if c_new2 < c5_2:
                c5_2 = c_new2

        # Coord descent at higher res
        if tlimit() > 30:
            h2, c5_2 = coord_descent(h2, n2, max_sweeps=30, rng=rng, tlimit=tlimit)

        if c5_2 < best_c5 - 1e-9:
            best_c5 = c5_2
            best_h = h2
            best_n = n2
            print("  Improved at n={}: C5={:.8f}".format(n2, c5_2))
        else:
            print("  n={} best: {:.8f} (no improvement over {:.8f})".format(n2, c5_2, best_c5))

    # Phase 10: Final SLSQP polish at best resolution
    if tlimit() > 20:
        print("\n=== Phase 10: Final polish ===")
        best_n = len(best_h)
        for bv in [5000, 15000]:
            if tlimit() < 8:
                break
            h_new = slsqp_run(best_h, best_n, bv, maxiter=400)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-13:
                best_c5 = c_new
                best_h = h_new
        print("  After final polish: C5={:.8f}".format(best_c5))

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
