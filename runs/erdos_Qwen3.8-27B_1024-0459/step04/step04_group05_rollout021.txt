        grad[:n_pairs] -= h[abs_off:abs_off + n_pairs]

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_BETA = 1.0
_G_TGT = 0.0
_G_START = 0.0
_G_BUDGET = 1300.0
_G_DX = 1.0
_G_M = 4
_G_P = 6

def get_budget():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        dist = np.abs(h - 0.5)
        idx = int(np.argmin(dist))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def con_fun(x):
    return np.sum(x) - _G_TGT

def con_jac(x):
    return np.ones(_G_N)

def obj_lse(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _G_BETA
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -300, 300))
    return (m + np.log(np.sum(exps)) / beta) * _G_DX

def obj_topm(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = max(_G_M, 1)
    p = max(_G_P, 2)
    top = np.sort(corr)[-m:]
    return np.mean(top ** p) * _G_DX

def subgrad_lags(h, n, lag_indices):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_idx in lag_indices:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            np_ = n - offset
            if np_ <= 0:
                continue
            grad[:np_] += (1.0 - h[offset:offset + np_])
            grad[offset:] -= h[:np_]
        else:
            a = -offset
            np_ = n - a
            if np_ <= 0:
                continue
            grad[:np_] -= h[a:a + np_]
            grad[a:] += (1.0 - h[:np_])
    return grad * dx

def do_slsqp(h_start, n, beta_list, use_topm=False):
    global _G_N, _G_BETA, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    for beta in beta_list:
        if get_budget() < 5.0:
            break
        _G_BETA = float(beta)
        try:
            if beta < 500:
                maxiter = 600
            elif beta < 5000:
                maxiter = 500
            else:
                maxiter = 400
            est_time = get_budget() * 0.35
            maxiter = min(maxiter, max(30, int(est_time / 0.05)))
            if use_topm:
                _G_M = 4
                _G_P = 6
                res = minimize(obj_topm, h_cur, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
            else:
                res = minimize(obj_lse, h_cur, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, c5_cur

def do_subgrad(h_best, n, rng, max_iters, top_k=1):
    h = h_best.copy()
    c5_best = compute_c5(h)
    no_improve = 0
    for it in range(max_iters):

            break

        k_star = int(np.argmax(corr))

            lags = [int(x) for x in top_idx]
        grad = subgrad_lags(h, n, lags)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:

        grad /= gnorm
        improved = False
        if it < 200:

def do_line_search(h_best, n, rng, n_lines):

    for _ in range(n_lines):
        if get_budget() < 2.0:

        d = rng.randn(n)
        d -= np.mean(d)
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-10:

        d /= dnorm
        best_t = 0.0
        best_c5 = c5_best

def do_binary_local(h_best, n, rng, max_iters=500):
    """Local search on nearly-binary solutions via pairwise adjustments."""
    h = h_best.copy()
    c5_best = compute_c5(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    no_improve = 0
    for it in range(max_iters):
        if get_budget() < 3.0:
            break
        if len(ones) < 50 or len(zeros) < 50:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        offset = k_star - (n - 1)
        best_gain = 1e-14
        best_a, best_b = -1, -1
        n_check = min(len(ones), 200)
        n_check_b = min(len(zeros), 200)
        idx_a = rng.choice(len(ones), size=min(n_check, len(ones)), replace=False)
        idx_b = rng.choice(len(zeros), size=min(n_check_b, len(zeros)), replace=False)
        for ia in idx_a:
            a = ones[ia]
            for ib in idx_b:
                b = zeros[ib]
                da = (1.0 - (h[(a - offset) % n] if 0 <= a - offset < n else 0.0))
                db = (h[(b + offset) % n] if 0 <= b + offset < n else 0.0)
                gain = da - db
                if gain > best_gain:
                    best_gain = gain
                    best_a, best_b = a, b
        if best_a >= 0 and best_b >= 0:

            h[best_b] += 0.05

                c5_best = c5_trial
                h_best = h.copy()
                no_improve = 0

                no_improve += 1
        else:
            no_improve += 1
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]

        c5_interp = compute_c5(h_interp)
        print("  Interp: C5={:.10f}".format(c5_interp))
        h_new, c5_new = do_slsqp(h_interp, n_try, [100, 500, 2000])
        h_new, c5_new = do_subgrad(h_new, n_try, rng, max_iters=300, top_k=1)
        h_new, c5_new = do_slsqp(h_new, n_try, [5000, 20000])
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n_try
            print("  n=512 IMPROVED: C5={:.10f}".format(best_c5))
        else:
            print("  n=512 no improve: C5={:.10f}".format(c5_new))

    # Phase 8: Try n=384
    if get_budget() > 60:
        n_try = 384
        if n_try != best_n:
            print("\n--- Phase 8: n=384 ---")
            h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
            h_new, c5_new = do_slsqp(h_interp, n_try, [200, 1000, 5000])
            h_new, c5_new = do_subgrad(h_new, n_try, rng, max_iters=200, top_k=1)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_try
                print("  n=384 IMPROVED: C5={:.10f}".format(best_c5))
            else:
                print("  n=384 no improve: C5={:.10f}".format(c5_new))

    # Phase 9: Multi-restart at best resolution
    if get_budget() > 40:
        print("\n--- Phase 9: Multi-restart ---")
        for r in range(12):
            if get_budget() < 15.0:
                break
            noise_scale = 0.001 * (1.0 + 0.5 * r)
            h_pert = project_feasible(best_h + rng.randn(best_n) * noise_scale, best_n)
            h_cur = h_pert
            c5_cur = compute_c5(h_cur)
            if c5_cur < best_c5:
                h_cur = h_pert
            else:
                h_cur = best_h.copy()
            h_cur, c5_cur = do_slsqp(h_cur, best_n, [500, 2000, 10000])
            h_cur, c5_cur = do_subgrad(h_cur, best_n, rng, max_iters=150, top_k=1)
            if c5_cur < best_c5 - 1e-14:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                print("  Restart {}: C5={:.10f}".format(r, best_c5))

    # Phase 10: Final surgical + SLSQP
    if get_budget() > 15:
        print("\n--- Phase 10: Final polish ---")
        best_h, best_c5 = do_surgical(best_h, best_n, rng, budget_target=10.0)
        best_h, best_c5 = do_slsqp(best_h, best_n, [20000, 100000])
        print("  C5={:.10f}".format(best_c5))

    # Phase 11: Tiny line search
    if get_budget() > 8:
        best_h, best_c5 = do_line_search(best_h, best_n, rng, n_lines=200)
        print("\n--- Phase 11: Line search done: C5={:.10f}".format(best_c5))

    # Phase 12: Ultra-fine subgradient
    if get_budget() > 5:
        best_h, best_c5 = do_subgrad(best_h, best_n, rng, max_iters=100, top_k=1)

    # Final
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
