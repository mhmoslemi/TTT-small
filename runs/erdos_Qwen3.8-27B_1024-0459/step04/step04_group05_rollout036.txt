import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        # Distribute among elements with most room
        rooms = np.where(h < 1.0, 1.0 - h, h)  # room to increase or decrease
        if diff > 0:
            idx = np.argsort(1.0 - h)[-min(abs(int(diff*2)), n):]
            for i in idx:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if abs(diff) < 1e-14:
                    break
        else:
            idx = np.argsort(h)[-min(abs(int(-diff*2)), n):]
            for i in idx:
                rem = min(h[i], -diff)
                h[i] -= rem
                diff += rem
                if abs(diff) < 1e-14:
                    break
    return h

def block_slsqp(h, n, block_size, budget_fn, max_rounds=5):
    """Block coordinate descent with SLSQP on each block."""
    dx = 2.0 / n
    c5_cur = compute_c5(h)
    h_best = h.copy()
    
    for rnd in range(max_rounds):
        if budget_fn() < 10:
            break
        improved = False
        offsets = list(range(0, n - block_size + 1, block_size // 2))
        rng_local = np.random.RandomState(rnd * 7919)
        rng_local.shuffle(offsets)
        
        for off in offsets:
            if budget_fn() < 5:
                break
            end = off + block_size
            block = h[off:end].copy()
            
            def obj_block(x):
                h_trial = h_best.copy()
                h_trial[off:end] = x
                h_trial = project_feasible(h_trial, n)
                corr = np.correlate(h_trial, 1.0 - h_trial, mode='full')
                return np.max(corr) * dx
            
            def obj_block_lse(x, beta=500):
                h_trial = h_best.copy()
                h_trial[off:end] = x
                h_trial = project_feasible(h_trial, n)
                corr = np.correlate(h_trial, 1.0 - h_trial, mode='full')
                m = np.max(corr)
                shifted = corr - m
                exps = np.exp(np.clip(beta * shifted, -300, 300))
                return (m + np.log(np.sum(exps)) / beta) * dx
            
            bounds = [(0.0, 1.0)] * block_size
            # No constraint needed if we project, but let's add soft constraint
            try:
                res = minimize(obj_block_lse, block, method='SLSQP',
                             bounds=bounds, options={'maxiter': 100, 'ftol': 1e-12})
                h_trial = project_feasible(res.x, n)
                # Need to fix the rest
                h_full = h_best.copy()
                h_full[off:end] = res.x
                h_full = project_feasible(h_full, n)
                c5_try = compute_c5(h_full)
                if c5_try < c5_cur - 1e-14:
                    h_best = h_full
                    c5_cur = c5_try
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return h_best, c5_cur

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    weights = np.exp(np.clip(beta * (corr - max_c), -500, 500))
    total_w = np.sum(weights)
    weights = weights / total_w
    
    # Get top K indices by weight
    top_idx = np.argsort(weights)[::-1][:K]
    
    grad = np.zeros(n)
    for k_idx in top_idx:
        wp = weights[k_idx]
        if wp < 1e-10:
            break

        # dcorr[k_idx]/dh[j] = (1-h[j+p]) - h[j-p] with bounds
