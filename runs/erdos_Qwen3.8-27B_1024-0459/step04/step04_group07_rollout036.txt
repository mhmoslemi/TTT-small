def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_main(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    b = _G_BETA
    s = corr - m

    N = _G_N
    dx = _G_DX
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    b = _G_BETA
    s = corr - m
    e = np.exp(np.clip(b * s, -400, 400))
    se = np.sum(e)
    w = e / se

        j = m_out - N + 1

            L = N - j
            if L > 0:

        else:
            aj = -j

            grad[:N-aj] -= wm * h[aj:]

    if _G_FOCUS_LAGS is not None and _G_FOCUS_W > 0:
        fw = _G_FOCUS_W * dx

            if 0 <= lag < len(corr):

                    L = N - j

                    print("  Mode {}: C5={:.8f}".format(mode, best_c5))

    # Phase 7: Perturbation escapes
    if tlimit() > 40:
        print("\n=== Phase 7: perturbation ===")
        n = best_n
        for attempt in range(10):
            if tlimit() < 10:
                break
            h_pert = best_h.copy()
            n_pert = rng.randint(2, 12)
            idxs = rng.choice(n, size=n_pert, replace=False)
            for idx in idxs:
                h_pert[idx] = np.clip(h_pert[idx] + rng.randn() * 0.4, 0.0, 1.0)
            h_pert = project_feasible(h_pert, n)
            c5_p = compute_c5(h_pert)
            if c5_p < best_c5 * 1.03:
                h_cur = h_pert.copy()
                c5_cur = c5_p
                for bv in [200, 800, 2000]:
                    if tlimit() < 5:
                        break
                    h_new = lbfgsb_grad_optimize(h_cur, n, bv, maxiter=200, pen=200.0)
                    c_new = compute_c5(h_new)
                    if c_new < c5_cur - 1e-14:
                        c5_cur = c_new
                        h_cur = h_new
                if c5_cur < best_c5 - 1e-9:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n
                    print("  Perturb {}: C5={:.8f}".format(attempt, best_c5))

    # Phase 8: Coordinate descent
    if tlimit() > 20:
        print("\n=== Phase 8: coord descent n={} ===".format(best_n))
        best_n = len(best_h)
        best_h, best_c5 = coord_descent(best_h, best_n, max_sweeps=60, rng=rng, tlimit=tlimit)
        print("  After CD: C5={:.8f}".format(best_c5))

    # Phase 9: Final driving + polish
    if tlimit() > 15:
        print("\n=== Phase 9: final polish ===")
        n = best_n
        h_cur, c5_cur = driving_optimize(best_h, n, rng, tlimit, max_rounds=20)
        if c5_cur < best_c5 - 1e-13:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        for bv in [10000, 20000, 50000]:
            if tlimit() < 4:
                break
            h_new = slsqp_optimize(best_h, n, bv, maxiter=500)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new.copy()
        print("  Final: C5={:.8f}".format(best_c5))

    # Final coordinate descent
    if tlimit() > 5:
        best_h, best_c5 = coord_descent(best_h, len(best_h), max_sweeps=20, rng=rng, tlimit=tlimit)

    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    _G_PEN = 0.0
    _G_FOCUS_LAGS = None
    _G_FOCUS_W = 0.0
    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
