import numpy as np

import time

_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)


def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def _sm_fun(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 50.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty


def _sm_grad(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += w[j] * corr_grad_single(x, ki, n)
    g *= dx
    g += 100.0 * (np.sum(x) - _TGT)
    return g


def lbfgs_pass(h, n, beta, top_k=5, maxiter=800):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    _TOPK = top_k
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_fun, h, jac=_sm_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-10, 'maxfun': maxiter * 10})
        h_new = proj(res.x, n)
        return h_new
    except Exception:
        return h


def projected_gradient_nesterov(h, n, top_k=5, max_iters=10000, step_init=0.03):
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.9
    step = step_init

    for it in range(max_iters):
        if remaining() < 3:
            break

        corr = compute_corr(h)
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        c_top = corr[idx]
        w = (c_top - np.mean(c_top)) + 0.1
        w = np.maximum(w, 0.01)
        w /= np.sum(w)
        for j, ki in enumerate(idx):
            g += w[j] * corr_grad_single(h, ki, n)

        g -= np.mean(g)
        g[(h >= 1.0 - 1e-9)] = 0.0
        g[(h <= 1e-9)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        g /= gn

        v = momentum * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 + 0.003:
            h = h_new
            if c5_new <= c5:
                step = min(step * 1.02, 0.15)
        else:
            h = best_h.copy()
            v *= 0.5
            step *= 0.7

        if it % 1000 == 0 and it > 0:
            step *= 0.85
            momentum = min(momentum + 0.005, 0.95)

    return best_h, best_c5


def top_equalize(h, n, top_k=5, max_iters=500):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)

    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-8:
            break

        g_max = corr_grad_single(h, idx[0], n)
        g_min = corr_grad_single(h, idx[-1], n)
        d = g_min - g_max
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-13:
            d = -g_max
            dn = np.linalg.norm(d)
            if dn < 1e-13:
                break
        d /= dn

        improved = False
        for scale in [5, 3, 1, 2, 0.5, 8, 1.5, 0.1, 4, 0.2]:

            h_try = proj(h + alpha * d, n)

                best_c5 = c5_try
                improved = True

def binary_local_search(h, n, max_iters=2000, top_k=3):
    """Targeted binary local search using gradient-directed swaps."""
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    corr = compute_corr(h_bin)
    cur_c5 = float(np.max(corr)) * dx

    for it in range(max_iters):
        if remaining() < 3:
            break

        corr = compute_corr(h_bin)
        idx_top = np.argsort(-corr)[:top_k]

        # Compute combined gradient direction
        g = np.zeros(n)
        for ki in idx_top:
            g += corr_grad_single(h_bin, ki, n)
        g -= np.mean(g)

        # Find best swap: flip a 1 to 0 at i, flip a 0 to 1 at j
        # Want g[i] very negative (reduce contribution) and g[j] very positive
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Score each candidate
        score_one = -g[ones]  # higher is better (more negative g)
        score_zero = g[zeros]  # higher is better (more positive g)

        # Try top candidates
        best_delta = 0
        best_swap = None
        n_try = min(12, len(ones))
        top_ones = ones[np.argsort(-score_one)[:n_try]]
        n_try_z = min(12, len(zeros))
        top_zeros = zeros[np.argsort(-score_zero)[:n_try_z]]

        for i in top_ones:
            for j in top_zeros:
                if i == j:
                    continue
                h_bin[i] = 0
                h_bin[j] = 1
                nc = compute_c5(h_bin)
                h_bin[i] = 1
                h_bin[j] = 0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)

        if best_swap is None or best_delta >= 0:
            # Try random swaps to escape local opt
            found = False
            for _ in range(6):
                i = int(ones[np.random.randint(len(ones))])
                j = int(zeros[np.random.randint(len(zeros))])
                if i == j:
                    continue
                h_bin[i] = 0
                h_bin[j] = 1
                nc = compute_c5(h_bin)
                h_bin[i] = 1
                h_bin[j] = 0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)
                    found = True
            if not found:
                break

        i, j = best_swap
        h_bin[i] = 0
        h_bin[j] = 1
        cur_c5 = compute_c5(h_bin)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_bin.copy()

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):

    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n

    return np.interp(x_new, xp, src_h)

                       bounds=bounds, constraints=cons,
