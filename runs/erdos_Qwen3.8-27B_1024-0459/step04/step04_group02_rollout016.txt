        if remaining_time() < 8:
            break
        h_new = lbgfb_refine(h, n, beta, 300)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-14:
            h = h_new
        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            best_n = n

    # Phase 2: Coordinate descent at parent resolution
    if remaining_time() > 15:
        h_cd, c5_cd = coord_descent(h, n, 35)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n
            h = h_cd.copy()
            n = best_n

    # Phase 3: Try binarization
    if remaining_time() > 10:
        h_bin, c5_bin = try_binarize(best_h, best_n)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            best_n = best_n
            h = h_bin.copy()

    # Phase 4: Higher resolution n=1280
    for target_n in [1280, 960, 800]:
        if remaining_time() < 50:
            break
        if target_n <= best_n:
            continue
        h_r = interp_to(best_h, best_n, target_n)
        h_r = project_feasible(h_r, target_n)
        c5_r = compute_c5(h_r)

        for beta in [50, 300, 1500, 6000, 20000, 60000]:
            if remaining_time() < 12:
                break
            h_new = lbgfb_refine(h_r, target_n, beta, 250)
            c5_new = compute_c5(h_new)
            if c5_new < c5_r - 1e-14:
                h_r = h_new
                c5_r = c5_new
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = target_n

        if remaining_time() > 12:
            h_cd, c5_cd = coord_descent(h_r, target_n, 25)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = target_n
                h_r = h_cd.copy()

        if remaining_time() > 10:
            h_bin2, c5_bin2 = try_binarize(h_r, target_n)
            if c5_bin2 < best_c5:
                best_c5 = c5_bin2
                best_h = h_bin2.copy()
                best_n = target_n

    # Phase 5: Subgradient descent with L-BFGS-B restarts
    if remaining_time() > 20:
        h = best_h.copy()
        n = best_n
        for it in range(1500):
            if remaining_time() < 6:
                break
            grad = subgrad_smooth(h, n)
            if grad is None:
                break
            base_step = 0.025 / (1.0 + it * 0.004)
            improved = False
            for sm in [1.0, 0.5, 2.0, 0.25, 0.75, 0.125]:
                step = base_step * sm
                h_trial = h - step * grad
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-15:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h = h_trial.copy()
                    improved = True
                    break
            if not improved:
                if it % 150 == 75 and remaining_time() > 25:
                    h_new = lbgfb_refine(h, n, 5000, 200)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h = h_new.copy()
                    if it > 400:
                        break

    # Phase 6: Multi-start coordinate descent
    if remaining_time() > 30:
        h_ms, c5_ms = multi_start_coord(best_h, best_n, rng, 4)
        if c5_ms < best_c5:
            best_c5 = c5_ms
            best_h = h_ms.copy()

    # Phase 7: Final coordinate descent polish
    if remaining_time() > 10:
        h_cd, c5_cd = coord_descent(best_h, best_n, 40)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 8: Very high beta L-BFGS-B
    if remaining_time() > 10:
        h_new = lbgfb_refine(best_h, best_n, 200000, 200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 9: Final coordinate descent
    if remaining_time() > 8:
        h_cd, c5_cd = coord_descent(best_h, best_n, 25)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 10: Micro subgradient
    if remaining_time() > 5:
        h = best_h.copy()
        n = best_n
        for it in range(300):
            if remaining_time() < 2:
                break
            grad = subgrad_smooth(h, n)
            if grad is None:
                break
            improved = False
            for step in [0.003, 0.001, 0.005, 0.002, 0.0005]:
                h_trial = h - step * grad
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-16:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h = h_trial.copy()
                    improved = True
                    break
            if not improved:
                break

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
