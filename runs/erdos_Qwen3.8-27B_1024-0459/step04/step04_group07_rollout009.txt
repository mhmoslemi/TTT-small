def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:

        h = np.clip(h, 0.0, 1.0)

            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmin(h))
            h[idx] = max(0.0, h[idx] + diff)

            grad[:n - s] += -h[s:]

            grad[s:] += 1.0 - h[:n - s]

        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 256
        best_h = make_tm(best_n)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)
        parent_n = best_n
        print("No parent. TM: n={}, C5={:.8f}".format(best_n, best_c5))

    # Phase 1: Multi-start quick optimization at n=256
    n1 = 256
    print("\n=== Phase 1: Multi-start n=256 ===")
    if remain() > 60:
        candidates = []
        # Parent interpolated
        h_p = project_feasible(interp_to_n(best_h, best_n, n1), n1)
        candidates.append(('parent', h_p, compute_c5(h_p)))
        # Thue-Morse
        h_tm = project_feasible(make_tm(n1), n1)
        candidates.append(('tm', h_tm, compute_c5(h_tm)))
        # Sine variants
        for fi in [(1, 2, 0.0, 0.35, 0.2), (1, 3, 0.5, 0.3, 0.25),
                   (2, 1, 0.0, 0.3, 0.3), (1, 4, 1.0, 0.25, 0.2),
                   (2, 3, 0.5, 0.25, 0.25), (3, 1, 0.0, 0.2, 0.3)]:
            h_s = project_feasible(make_sine(n1, fi[0], fi[1], fi[2], fi[3], fi[4]), n1)
            candidates.append(('sine_{}_{}'.format(fi[0], fi[1]), h_s, compute_c5(h_s)))
        # Random binary
        for ri in range(3):
            h_r = np.zeros(n1)
            idx = rng.permutation(n1)
            h_r[idx[:n1 // 2]] = 1.0
            h_r = project_feasible(h_r, n1)
            candidates.append(('rand_{}'.format(ri), h_r, compute_c5(h_r)))

        candidates.sort(key=lambda x: x[2])
        print("  Top candidates:")
        for name, _, c in candidates[:5]:
            print("    {}: C5={:.8f}".format(name, c))

        # Optimize top 3 candidates
        for ci in range(min(3, len(candidates))):
            if remain() < 30:
                break
            name, h_c, c_c = candidates[ci]
            print("  Optimizing {} (C5={:.8f})...".format(name, c_c))
            for bv in [100, 500, 2000]:
                if remain() < 15:
                    break
                h_c = lbfgsb_optimize(h_c, n1, bv, maxiter=200, start_time=t0, budget=budget)
            h_c = slsqp_optimize(h_c, n1, 3000, maxiter=300, start_time=t0, budget=budget)
            c_new = compute_c5(h_c)
            print("    After opt: C5={:.8f}".format(c_new))
            if c_new < best_c5 - 1e-10:
                best_c5 = c_new
                best_h = h_c.copy()
                best_n = n1

    # Phase 2: Intensive optimization at best n
    print("\n=== Phase 2: Intensive n={} ===".format(best_n))
    n2 = best_n
    h2 = best_h.copy()
    c52 = best_c5
    if remain() > 100:
        # Subgradient warm-up
        h2, c52 = subgradient_loop(h2, n2, n_iter=3000, lr0=0.002, start_time=t0, budget=budget)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2.copy()
            best_n = n2
        print("  After subgrad: C5={:.8f}".format(best_c5))

        # Multi-beta SLSQP
        for bv in [500, 2000, 8000, 20000]:
            if remain() < 20:
                break
            h2 = slsqp_optimize(h2, n2, bv, maxiter=400, start_time=t0, budget=budget)
            c_new = compute_c5(h2)
            if c_new < best_c5 - 1e-12:
                best_c5 = c_new
                best_h = h2.copy()
                best_n = n2
        print("  After multi-beta SLSQP: C5={:.8f}".format(best_c5))

        # More subgradient
        if remain() > 30:
            h2, c52 = subgradient_loop(best_h, n2, n_iter=5000, lr0=0.001, start_time=t0, budget=budget)
            if c52 < best_c5:
                best_c5 = c52
                best_h = h2.copy()
                best_n = n2
            print("  After more subgrad: C5={:.8f}".format(best_c5))

    # Phase 3: Block coordinate descent
    print("\n=== Phase 3: Block CD n={} ===".format(best_n))
    if remain() > 40:
        best_h, best_c5 = block_cd(best_h, best_n, block_size=8, n_sweeps=50,
                                   start_time=t0, budget=budget, rng=rng)
        print("  After block CD: C5={:.8f}".format(best_c5))

    # Phase 4: Try n=512
    n3 = 512
    if remain() > 120:
        print("\n=== Phase 4: n=512 ===")
        h3 = project_feasible(interp_to_n(best_h, best_n, n3), n3)
        c5_3 = compute_c5(h3)
        print("  Interpolated: C5={:.8f}".format(c5_3))
        if c5_3 < best_c5 + 0.02:
            for bv in [200, 1000, 5000]:
                if remain() < 20:
                    break
                h3 = lbfgsb_optimize(h3, n3, bv, maxiter=150, start_time=t0, budget=budget)
            h3 = slsqp_optimize(h3, n3, 5000, maxiter=200, start_time=t0, budget=budget)
            h3, c5_3 = subgradient_loop(h3, n3, n_iter=3000, lr0=0.001, start_time=t0, budget=budget)
            c5_3 = compute_c5(h3)
            print("  After n=512 opt: C5={:.8f}".format(c5_3))
            if c5_3 < best_c5 - 1e-10:
                best_c5 = c5_3
                best_h = h3.copy()
                best_n = n3

    # Phase 5: Try n=384
    n4 = 384
    if remain() > 80:
        print("\n=== Phase 5: n=384 ===")
        h4 = project_feasible(interp_to_n(best_h, best_n, n4), n4)
        c5_4 = compute_c5(h4)
        print("  Interpolated: C5={:.8f}".format(c5_4))
        if c5_4 < best_c5 + 0.02:
            for bv in [500, 2000, 8000]:
                if remain() < 15:
                    break
                h4 = lbfgsb_optimize(h4, n4, bv, maxiter=150, start_time=t0, budget=budget)
            h4 = slsqp_optimize(h4, n4, 10000, maxiter=300, start_time=t0, budget=budget)
            h4, c5_4 = subgradient_loop(h4, n4, n_iter=3000, lr0=0.001, start_time=t0, budget=budget)
            c5_4 = compute_c5(h4)
            print("  After n=384 opt: C5={:.8f}".format(c5_4))
            if c5_4 < best_c5 - 1e-10:
                best_c5 = c5_4
                best_h = h4.copy()
                best_n = n4

    # Phase 6: Active lag focus iteration
    print("\n=== Phase 6: Active lag focus ===")
    if remain() > 40:
        focused_h = best_h.copy()
        focused_c5 = best_c5
        n6 = best_n
        for rnd in range(30):
            if remain() < 5:
                break
            corr = compute_corr(focused_h)
            k_active = int(np.argmax(corr))
            c_max = corr[k_active]
            # Subgradient on active lag with small steps
            lr_seq = [0.005, 0.003, 0.002, 0.001]
            for lr in lr_seq:
                grad = vec_subgradient(focused_h, n6, k_active)
                h_new = project_feasible(focused_h - lr * grad, n6)
                c_new = compute_c5(h_new)
                if c_new < focused_c5 - 1e-13:
                    focused_h = h_new
                    focused_c5 = c_new
            if focused_c5 < best_c5 - 1e-12:
                best_c5 = focused_c5
                best_h = focused_h.copy()
                best_n = n6
        print("  After focus: C5={:.8f}".format(best_c5))

    # Phase 7: Final SLSQP polish
    print("\n=== Phase 7: Final polish ===")
    if remain() > 20:
        n7 = best_n
        for bv in [10000, 50000]:
            if remain() < 5:
                break
            best_h = slsqp_optimize(best_h, n7, bv, maxiter=500, start_time=t0, budget=budget)
        best_c5 = compute_c5(best_h)
        print("  After final SLSQP: C5={:.8f}".format(best_c5))

    # Phase 8: Final subgradient polish
    if remain() > 10:
        best_h, best_c5 = subgradient_loop(best_h, best_n, n_iter=5000, lr0=0.0005,
                                           start_time=t0, budget=budget)
        print("  After final subgrad: C5={:.8f}".format(best_c5))

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
