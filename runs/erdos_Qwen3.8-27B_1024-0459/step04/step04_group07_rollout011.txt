def compute_c5(h):

def get_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:

            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

    return h

            h_cur, c5_cur = barrier_descent(h_cur, n, rng, tlimit,
                                             t_start=t_cur, t_end=t_cur * 0.99)
            # SLSQP step
            for bv in [500, 2000]:
                if tlimit() < 10:
                    break
                h_new = slsqp_run(h_cur, n, bv, maxiter=300)
                c_new = compute_c5(h_new)
                if c_new < c5_cur - 1e-13:
                    c5_cur = c_new
                    h_cur = h_new
            if c5_cur < best_c5 - 1e-10:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n
            print("  Cycle {}: C5={:.8f}".format(cycle, best_c5))

    # Phase 6: Random restarts with targeted perturbation
    if tlimit() > 80:
        print("\n=== Phase 6: Perturbation escapes ===")
        n = best_n
        n_pert_rounds = min(10, int(tlimit() // 10))
        for attempt in range(n_pert_rounds):
            if tlimit() < 12:
                break
            h_pert = best_h.copy()
            # Perturb a random subset
            n_mod = rng.randint(5, 20)
            idxs = rng.choice(n, size=n_mod, replace=False)
            for idx in idxs:
                h_pert[idx] = np.clip(h_pert[idx] + rng.randn() * 0.2, 0.0, 1.0)
            h_pert = project_feasible(h_pert, n)
            c5_p = compute_c5(h_pert)
            
            # Quick optimize
            if c5_p < best_c5 * 1.02:
                h_cur = h_pert.copy()
                c5_cur = c5_p
                for bv in [500, 2000]:
                    if tlimit() < 8:
                        break
                    h_new = slsqp_run(h_cur, n, bv, maxiter=250)
                    c_new = compute_c5(h_new)
                    if c_new < c5_cur - 1e-13:
                        c5_cur = c_new
                        h_cur = h_new
                if c5_cur < best_c5 - 1e-10:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n
                    print("  Perturb {} -> C5={:.8f}".format(attempt, best_c5))

    # Phase 7: Final local search
    if tlimit() > 30:
        print("\n=== Phase 7: Final local search ===")
        n = best_n
        best_h, best_c5 = guided_local_search(best_h, n, rng, tlimit, max_sweeps=50)
        print("  After final LS: C5={:.8f}".format(best_c5))

    # Phase 8: Ultra-fine SLSQP
    if tlimit() > 15:
        print("\n=== Phase 8: Ultra-fine SLSQP ===")
        n = best_n
        for bv in [10000, 30000, 50000]:
            if tlimit() < 5:
                break
            h_new = slsqp_run(best_h, n, bv, maxiter=500)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new
        print("  After ultra SLSQP: C5={:.8f}".format(best_c5))

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    _G_PEN = 0.0
    _G_FOCUS = -1
    _G_T_BARRIER = 0.0
    _G_BARRIER_MULT = 0.0
    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
