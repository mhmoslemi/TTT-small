def do_subgrad_top(h, n, max_iter, top_k_list):
    """Subgradient targeting top-k correlations."""
    dx = 2.0 / n
    h_best = h.copy()
    c5_best = compute_c5(h)
    v = np.zeros(n)
    step = 0.1
    patience = 0

    for it in range(max_iter):
        if remaining_time() < 5.0:
            break
        tk = top_k_list[min(it // max(1, max_iter // len(top_k_list)), len(top_k_list) - 1)]
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[-tk:]
        top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
        grad = np.zeros(n)
        for k_idx in top_idx:
            offset = int(k_idx) - (n - 1)
            if offset >= 0:
                nv = n - offset
                if nv <= 0:
                    continue
                grad[offset:offset + nv] += (1.0 - h[:nv])
                grad[:nv] -= h[offset:offset + nv]
            else:
                m_off = -offset
                nv = n - m_off
                if nv <= 0:
                    continue
                grad[:nv] += (1.0 - h[m_off:m_off + nv])
                grad[m_off:m_off + nv] -= h[:nv]

        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        alpha = step
        improved = False
        for _ in range(10):
            v = 0.85 * v - alpha * grad
            h_trial = h + v
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-12:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
            alpha *= 0.5
        if improved:
            patience = 0
            step = min(alpha * 1.2, 0.3)
        else:
            patience += 1
            step *= 0.55
            v *= 0.4
            if patience > 15:
                break
    return h_best, c5_best

            no_imp += 1
    return h_best, c5_best


def quick_optimize(h, n, rng, time_budget=60.0):
    """Run a quick round of optimization on h at resolution n."""
    if remaining_time() < time_budget * 0.3:
        return h, compute_c5(h)
    h_best = h.copy()
    c5_best = compute_c5(h)

    # Subgradient
    h_best, c5_best = do_subgrad_top(h_best, n, 100, [3, 5, 8])

    # SLSQP
    if remaining_time() > 10.0:
        h_best, c5_best = do_slsqp_polish(h_best, n, [100, 500, 2000])

    # Greedy pairs
    if remaining_time() > 15.0:
        h_best, c5_best = do_greedy_pair_flip(h_best, n, rng, min(15.0, remaining_time() * 0.5))

    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET, _G_N, _G_TGT, _G_DX, _G_BETA
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial solution
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 256
        parent_h = project_feasible(make_thue_morse(parent_n), parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n

    n = best_n
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n

    # Phase 1: Multi-start with quick greedy (60s)
    print("Phase 1: Multi-start quick greedy")
    starts = [best_h.copy()]
    for _ in range(3):
        h_rand = best_h.copy() + rng.uniform(-0.1, 0.1, n)
        h_rand = project_feasible(h_rand, n)
        starts.append(h_rand)
    starts.append(project_feasible(make_golden_ratio(n), n))

    for idx, h_s in enumerate(starts):
        if remaining_time() < 10.0:
            break
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
        # Quick greedy pair flip
        h_s, c5_s = do_greedy_pair_flip(h_s, n, rng, min(10.0, remaining_time() * 0.8))
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
    print("  C5={:.10f}".format(best_c5))

    # Phase 2: Simulated annealing (300s)
    print("Phase 2: Simulated annealing")
    sa_time = min(300.0, remaining_time() * 0.35)
    if sa_time > 30.0:
        best_h, best_c5 = do_simulated_annealing(best_h, n, rng, sa_time)
        print("  C5={:.10f}".format(best_c5))

    # Phase 3: Greedy pair flips (120s)
    print("Phase 3: Greedy pair flips")
    gf_time = min(120.0, remaining_time() * 0.2)
    if gf_time > 10.0:
        best_h, best_c5 = do_greedy_pair_flip(best_h, n, rng, gf_time)
        print("  C5={:.10f}".format(best_c5))

    # Phase 4: Subgradient on top correlations (100s)
    print("Phase 4: Subgradient top-k")
    if remaining_time() > 15.0:
        best_h, best_c5 = do_subgrad_top(best_h, n, 400, [3, 5, 8, 12, 16])
        print("  C5={:.10f}".format(best_c5))

    # Phase 5: SLSQP multi-beta polish (120s)
    print("Phase 5: SLSQP polish")
    if remaining_time() > 10.0:
        _G_N = n
        _G_TGT = n / 2.0
        _G_DX = 2.0 / n
        best_h, best_c5 = do_slsqp_polish(best_h, n, [50, 200, 800, 3000, 10000])
        print("  C5={:.10f}".format(best_c5))

    # Phase 6: Block swap search (80s)
    print("Phase 6: Block swap search")
    bs_time = min(80.0, remaining_time() * 0.15)
    if bs_time > 5.0:
        best_h, best_c5 = do_block_swap_search(best_h, n, rng, bs_time)
        print("  C5={:.10f}".format(best_c5))

    # Phase 7: Second round SA at lower temp + greedy (100s)
    print("Phase 7: Refine SA + greedy")
    if remaining_time() > 20.0:
        sa2_time = min(60.0, remaining_time() * 0.4)
        if sa2_time > 10.0:
            best_h, best_c5 = do_simulated_annealing(best_h, n, rng, sa2_time)
        if remaining_time() > 15.0:
            best_h, best_c5 = do_greedy_pair_flip(best_h, n, rng, min(40.0, remaining_time() * 0.5))
        print("  C5={:.10f}".format(best_c5))

    # Phase 8: Try different resolutions
    print("Phase 8: Resolution search")
    for n_try in [192, 320, 256, 384, 128]:
        if remaining_time() < 25.0:
            break
        if n_try == best_n:
            continue
        h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        c5_tmp = compute_c5(h_interp)
        h_try, c5_try = quick_optimize(h_interp, n_try, rng, min(50.0, remaining_time() * 0.7))
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try
            print("  n={} improved: C5={:.10f}".format(n_try, best_c5))
        else:
            print("  n={}: C5={:.10f} (no improvement)".format(n_try, c5_try))

    # Phase 9: Final polish on best resolution
    if remaining_time() > 10.0:
        print("Phase 9: Final polish")
        n = best_n
        _G_N = n
        _G_TGT = n / 2.0
        _G_DX = 2.0 / n
        best_h, best_c5 = do_subgrad_top(best_h, n, 200, [3, 5, 8])
        best_h, best_c5 = do_slsqp_polish(best_h, n, [500, 2000, 8000, 20000])
        best_h, best_c5 = do_greedy_pair_flip(best_h, n, rng, min(30.0, max(5.0, remaining_time() * 0.5)))
        print("  C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
