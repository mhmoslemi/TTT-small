import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)

        else:

def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xq = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xq, xp, src)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_and_grad(h, n, beta, active_only=None):
    """
    Compute smooth max of correlations and its gradient.
    active_only: if given, a boolean mask of which correlations to include
    """
    corr = get_corr(h)
    dx = 2.0 / n
    
    if active_only is not None:
        # Only consider active correlations
        c_act = corr[active_only]
        m = np.max(c_act) if len(c_act) > 0 else 0
        ex = np.exp(beta * (c_act - m))
        w_act = ex / np.sum(ex)
        # Map back to full weights
        w = np.zeros(len(corr))
        w[active_only] = w_act
        m_full = np.max(corr)
    else:
        m = np.max(corr)
        ex = np.exp(beta * (corr - m))
        w = ex / np.sum(ex)
    
    # Compute gradient
    g = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for k in nz:
        wk = w[k]
        off = k - (n - 1)
        if off >= 0:
            npairs = n - off
            g[off:off+npairs] += wk * (1.0 - h[:npairs])
            g[:npairs] -= wk * h[off:off+npairs]
        else:

            npairs = n - ao
            g[:npairs] += wk * (1.0 - h[ao:ao+npairs])
            g[ao:ao+npairs] -= wk * h[:npairs]

        val = (m + np.log(np.sum(ex)) / beta) * dx
    else:
        val = (m + np.log(np.sum(ex)) / beta) * dx
    
    return val, g * dx

def lbfgs_optimize(h, n, beta, maxiter=300, penalty=1e6, active_set=None):
    """Use L-BFGS-B to minimize smooth max + penalty for sum constraint."""
    target = n / 2.0
    
    def obj(x):
        v, _ = smooth_max_and_grad(x, n, beta, active_set)
        return v + penalty * (np.sum(x) - target) ** 2
    
    def grad(x):
        _, g = smooth_max_and_grad(x, n, beta, active_set)
        g += 2 * penalty * (np.sum(x) - target) * np.ones(n)
        return g
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
    return project_feasible(res.x, n)

def topk_gradient(h, n, top_k=10, lr=0.01, maxiters=500):
    """Gradient descent using only the top-k worst correlations."""
    h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    target = n / 2.0
    
    for it in range(maxiters):
        if remaining() < 3:
            break
        corr = get_corr(h)
        # Get top-k worst correlations
        top_idx = np.argsort(-corr)[:top_k]
        mask = np.zeros(len(corr), dtype=bool)
        mask[top_idx] = True
        
        # Compute gradient using only active set
        m = np.max(corr)
        c_act = corr[mask]
        ex = np.exp(100 * (c_act - m))
        w_act = ex / np.sum(ex)
        w = np.zeros(len(corr))
        w[mask] = w_act
        
        g = np.zeros(n)
        nz = np.where(w > 1e-14)[0]
        for k in nz:
            wk = w[k]
            off = k - (n - 1)
            if off >= 0:
                npairs = n - off
                g[off:off+npairs] += wk * (1.0 - h[:npairs])
                g[:npairs] -= wk * h[off:off+npairs]
            else:
                ao = -off
                npairs = n - ao
                g[:npairs] += wk * (1.0 - h[ao:ao+npairs])
                g[ao:ao+npairs] -= wk * h[:npairs]
        g *= dx
        
        # Normalize and take step
        gn = np.linalg.norm(g)
        if gn < 1e-14:

        # Line search
        improved = False
        for step in [lr, lr*0.5, lr*0.1, lr*2, lr*0.05]:

                best_c5 = c_new
                improved = True
                break

            break
    
    return h, best_c5

            h_bin[i], h_bin[j] = 1.0, 0.0
            stall += 1
        T *= alpha
        if stall > 3000:
            T = max(T, 2.0)
            stall = 0
    
    return best_h, best_max * dx

    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except:
        parent_n = 360
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n//2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    # Phase 1: L-BFGS-B optimization at parent resolution
    n = best_n
    h = best_h.copy()
    
    # Adaptive beta L-BFGS-B
    for beta in [50, 100, 200, 500, 1000, 2000, 5000]:
        if remaining() < 20:
            break
        h_new = lbfgs_optimize(h, n, beta=beta, maxiter=200, penalty=1e6)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 2: Active set focusing
    for _ in range(5):
        if remaining() < 20:
            break
        corr = get_corr(h)
        m = np.max(corr)
        # Active set: correlations within 0.5 of max
        active = corr >= m - 0.5
        h_new = lbfgs_optimize(h, n, beta=5000, maxiter=200, penalty=1e6, active_set=active)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining() > 30:

            best_h = h_td.copy()
            h = h_td.copy()
