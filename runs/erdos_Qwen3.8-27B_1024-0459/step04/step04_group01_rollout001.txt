This is a convex function? Let's check: each f_d(h) is bilinear (not convex in general), so the max is not necessarily convex. For binary h, it's just a max of linear functions (h[i]*(1-h[j]) is either 0 or 1 for binary). For continuous h, h[i]*(1-h[j]) is not linear.

Wait, h[i]*(1-h[j]) = h[i] - h[i]*h[j]. This is bilinear, not linear. So the problem is NOT a linear program.

However, if we fix one of the variables, it becomes linear in the other. This suggests a coordinate descent approach might work well.

```python
import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_trial = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_trial)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, n, d):
    """Gradient of the correlation at lag d w.r.t. h."""
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        if m <= 0:
            return g
        g[d:d+m] += (1.0 - h[:m])
        g[:m] -= h[d:d+m]
    else:
        m = -d
        if m >= n:
            return g
        g[:n-m] += (1.0 - h[m:n])
        g[m:] -= h[:n-m]
    return g

def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 5.0)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
```

Now for the main optimization algorithms:

**Algorithm 1: Active-set projected gradient with backtracking line search**

```python
def active_gradient_descent(h, n, max_iters=5000, time_limit=60, top_k=5, mom=0.9):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx_val = 2.0 / n
    corr = compute_corr(h)
    momentum = np.zeros(n)
    step = 0.01
    n_corr = 2 * n - 1
    
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit or remaining() < 10:
            break
        
        # Find active set (top-k shifts)
        idx = np.argsort(-corr)
        top = idx[:top_k]
        weights = np.exp((corr[top] - corr[top[0]]) * 50.0)
        weights /= weights.sum()
        
        # Compute weighted gradient
        g = np.zeros(n)
        for w, k in zip(weights, top):
            d = int(k) - (n - 1)
            g += w * grad_lag(h, n, d)
        
        # Project gradient to feasible subspace (remove mean component for sum constraint)
        g -= np.mean(g)  # project onto sum=0 subspace
        
        # Update with momentum
        momentum = mom * momentum + g
        h_trial = h - step * momentum
        h_trial = project_feasible(h_trial, n)
        
        corr = compute_corr(h_trial)
        c5_new = float(np.max(corr) * dx_val)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_trial.copy()
            h = h_trial.copy()
            step = min(step * 1.01, 0.05)
        else:
            step *= 0.9
            if step < 1e-6:
                break
    
    return best_h, best_c5
```

**Algorithm 2: Pair coordinate descent with gradient-guided pairing**

The key idea: for each pair (i, j), if we change h[i] → h[i]+t and h[j] → h[j]-t (to maintain sum constraint), the correlation changes linearly in t:
corr_new[k] = corr[k] + t * (grad_lag[k][i] - grad_lag[k][j])

We want to minimize max_k corr_new[k], which is a 1D convex problem in t.

```python
def pair_coord_descent_v2(h, n, max_passes=200, time_limit=60, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    dx_val = 2.0 / n
    corr = compute_corr(h)
    n_corr = 2 * n - 1
    best_c5 = float(np.max(corr) * dx_val)
    
    start = time.time()
    for p in range(max_passes):
        if time.time() - start > time_limit or remaining() < 10:
            break
        
        # Generate pairs: adjacent, far, and random
        pairs = []
        # Adjacent pairs
        for i in range(n - 1):
            pairs.append((i, i + 1))
        # Pairs separated by n//2
        for i in range(n):
            j = (i + n // 2) % n
            if j != i:
                pairs.append((i, j))
        # Random pairs
        for _ in range(n):
            i, j = rng.randint(n), rng.randint(n)
            if i != j:
                pairs.append((i, j))
        
        rng.shuffle(pairs)
        
        for i, j in pairs:
            if time.time() - start > time_limit or remaining() < 10:
                break
            if i == j:
                continue
            
            # Constraint: h[i] + h[j] = s, h[i] in [max(0, s-1), min(1, s)]
            s = h[i] + h[j]
            lo_t = max(0.0, s - 1.0)
            hi_t = min(1.0, s)
            if hi_t - lo_t < 1e-14:
                continue
            
            # corr_new[k] = corr[k] + t * (grad_lag[k][i] - grad_lag[k][j])
            # where t = h[i] - h[i]_old
            # We need: for all k, corr[k] + t * dg[k] ≤ M (minimize M)
            # dg[k] = grad of corr[k] w.r.t. h[i] - grad of corr[k] w.r.t. h[j]
            
            # Compute dg for all k
            dg = np.zeros(n_corr)
            for k in range(n_corr):
                d = k - (n - 1)
                if d >= 0:
                    m = n - d
                    if m > 0:
                        # grad[i]: 
                        if d <= i < d + m:  # i is in [d, n-1]
                            gi = 1.0 - h[i - d]
                        else:
                            gi = 0.0
                        if i < m:  # i is in [0, n-1-d]
                            gi -= h[i + d]
                        # grad[j]:
                        if d <= j < d + m:
                            gj = 1.0 - h[j - d]
                        else:
                            gj = 0.0
                        if j < m:
                            gj -= h[j + d]
                        dg[k] = gi - gj
                    else:
                        dg[k] = 0.0
                else:
                    m = -d
                    if m < n:
                        if i < n - m:
                            gi = 1.0 - h[i + m]
                        else:
                            gi = 0.0
                        if i >= m:
                            gi -= h[i - m]
                        if j < n - m:
                            gj = 1.0 - h[j + m]
                        else:
                            gj = 0.0
                        if j >= m:
                            gj -= h[j - m]
                        dg[k] = gi - gj
                    else:
                        dg[k] = 0.0
            
            # Minimize max_k (corr[k] + t * dg[k]) over t in [lo_t, hi_t]
            # This is: minimize M subject to corr[k] + t*dg[k] ≤ M for all k
            # Equivalent to: M ≥ corr[k] + t*dg[k] for all k
            # For given t: M(t) = max_k (corr[k] + t*dg[k])
            # M(t) is piecewise linear in t (convex)
            # Find t* that minimizes M(t)
            
            # Bisection on subgradient: M'(t) = dg[argmax_k (corr[k]+t*dg[k])]
            # At optimum, 0 is in the subgradient: min_active dg ≤ 0 ≤ max_active dg
            
            t_lo, t_hi = lo_t, hi_t
            for _ in range(40):
                t_mid = (t_lo + t_hi) / 2.0
                vals = corr + t_mid * dg
                mval = np.max(vals)
                active = vals >= mval - 1e-12
                if dg[active].min() > 1e-14:
                    t_lo = t_mid  # M is increasing at t_mid, go left
                elif dg[active].max() < -1e-14:
                    t_hi = t_mid  # M is decreasing at t_mid, go right
                else:
                    break
            
            t_opt = (t_lo + t_hi) / 2.0
            
            # Clip to valid range
            t_opt = np.clip(t_opt, lo_t, hi_t)
            
            delta = t_opt - h[i]
            if abs(delta) < 1e-14:
                continue
            
            h[i] = t_opt
            h[j] = s - t_opt
            corr += delta * dg
        
        h = project_feasible(h, n)
        corr = compute_corr(h)
        c5_new = float(np.max(corr) * dx_val)
        if c5_new < best_c5:
            best_c5 = c5_new
    
    return h, best_c5
```

Wait, the pair coordinate descent recomputing dg for all k for each pair is O(n³) per pass (n pairs × n shifts × O(1) per shift). For n=312, that's 312³ ≈ 30M operations per pass. With 200 passes, that's 6 billion... too slow.

I need to optimize this. Let me precompute the gradients more efficiently.

Actually, for a given pair (i, j), the vector dg[k] = grad_lag_k[i] - grad_lag_k[j] can be computed as:
dg = grad_lag_all(h, n, i) - grad_lag_all(h, n, j)

where grad_lag_all(h, n, pos) returns the vector of partial derivatives ∂corr[k]/∂h[pos] for all k.

Let me define:
∂corr[k]/∂h[pos]:
corr[k] = Σ_n h[n+d]*(1-h[n]) where d = k-(n-1)
= Σ_n h[n+d] - h[n+d]*h[n]

∂/∂h[p]:
- If p = n+d for some valid n: (1 - h[n])
- If p = n for some valid n: -h[n+d]

So:
∂corr[k]/∂h[p] = (1-h[p-d]) * [d ≤ p ≤ n-1] - h[p+d] * [p ≤ n-1-d] for d ≥ 0
∂corr[k]/∂h[p] = (1-h[p+m]) * [p ≤ n-1-m] - h[p-m] * [m ≤ p ≤ n-1] for d < 0 (m=-d)

This can be computed for all k simultaneously:

```python
def grad_all_shifts(h, n, pos):
    """For a fixed position pos, compute ∂corr[k]/∂h[pos] for all k (all shifts)."""
    n_corr = 2 * n - 1
    result = np.zeros(n_corr)
    for k in range(n_corr):
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            val = 0.0
            if d <= pos <= n - 1:
                val += (1.0 - h[pos - d])
            if 0 <= pos <= n - 1 - d:
                val -= h[pos + d]
            result[k] = val
        else:
            m = -d
            if m >= n:
                continue
            val = 0.0
            if 0 <= pos <= n - 1 - m:
                val += (1.0 - h[pos + m])
            if m <= pos <= n - 1:
                val -= h[pos - m]
            result[k] = val
    return result
```

This is O(n) per call (one shift at a time). For a pair (i,j), we need grad_all_shifts(h,n,i) - grad_all_shifts(h,n,j), which is O(n).

So per pair: O(n), per pass: O(n²) (n pairs × O(n)), per all passes: O(n² × passes).
For n=312, passes=200: 312² × 200 ≈ 19.5M. That's fine!

- For each pair (i, j), compute the 1D minimization
- The "slower" part is computing dg = grad_all_shifts(h,n,i) - grad_all_shifts(h,n,j)

I can precompute grad_all_shifts for all positions at the start of each pass:
g_all = [[grad_all_shifts(h,n,i) for i in range(n)]]  # shape (n, 2n-1)

This is O(n²) memory and O(n²) computation per pass (each grad_all_shifts is O(n), done n times).

Then for pair (i,j): dg = g_all[i] - g_all[j], and the 1D minimization is O(n) (bisection with O(n) evaluation).

Total per pass: O(n²) precompute + O(n_pairs × n) optimization.
If n_pairs = n (one pass over all positions): O(n² + n²) = O(n²).
For 200 passes: O(200 * n²) = 200 * 312² ≈ 19.5M. Very fast!
