    corr = np.real(np.fft.ifft(H * V))[:2*n-1]
    return corr  # max of this equals max of np.correlate(h, v, 'full')
```

This is O(n log n) instead of O(n²). For n=1024, this is a big speedup.

Now, for the gradient, I still need the per-shift gradients. Let me precompute the correlation vector and then compute gradients for the top-K shifts.

Let me now write the full solution:

```python
import numpy as np
from scipy.optimize import minimize

_G_START = 0.0
_G_BUDGET = 1300.0
_G_N = 0
_G_DX = 1.0
_G_BETA = 1.0
_G_TGT = 0.0

def fft_corr_full(h):
    n = len(h)
    v = 1.0 - h
    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = np.fft.rfft(h, N)
    V = np.fft.rfft(v[::-1], N)
    corr = np.fft.irfft(H * V, N)[:2*n-1]
    return corr

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_corr_full(h)
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - target) > 1e-14:
        rooms = np.minimum(h, 1.0 - h)
        idx = int(np.argmax(rooms))
        h[idx] += target - np.sum(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_topk(h, corr, top_k=5):
    """Compute subgradient of max_k corr[k] using top-k active shifts."""
    n = len(h)
    top_idx = np.argsort(corr)[-top_k:]
    top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
    grad = np.zeros(n)
    for k_idx in top_idx:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            n_pairs = n - offset
            if n_pairs <= 0:
                continue
            grad[:n_pairs] += (1.0 - h[offset:offset + n_pairs])
            grad[offset:] -= h[:n_pairs]
        else:
            abs_off = -offset
            n_pairs = n - abs_off
            if n_pairs <= 0:
                continue
            grad[:n_pairs] -= h[abs_off:abs_off + n_pairs]
            grad[abs_off:] += (1.0 - h[:n_pairs])
    return grad

def projected_gradient_step(h, corr, alpha, n):
    """One projected gradient descent step on max correlation."""
    grad = subgrad_topk(h, corr, top_k=5)
    h_new = h - alpha * grad
    return project_feasible(h_new, n)

def backtracking_pg(h, n, max_iters=500, alpha_init=0.01):
    """Projected gradient with backtracking line search."""
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    dx = 2.0 / n
    alpha = alpha_init
    no_improve = 0
    
    for it in range(max_iters):
        corr = fft_corr_full(h_cur)
        grad = subgrad_topk(h_cur, corr, top_k=5)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        improved = False
        for _ in range(20):
            h_trial = project_feasible(h_cur - alpha * grad, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_cur - 1e-14:
                h_cur = h_trial
                c5_cur = c5_trial
                no_improve = 0
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve > 50:
                alpha = alpha_init * 0.5
                if alpha < 1e-8:
                    break
            no_improve = 0
        
        # Occasionally reset alpha
        if it > 0 and it % 100 == 0:
            alpha = alpha_init
    
    return h_cur, c5_cur

def logsumexp_obj(x):
    n = _G_N
    dx = _G_DX
    corr = fft_corr_full(x)
    m = np.max(corr)
    beta = _G_BETA
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    return (m + np.log(np.sum(exps)) / beta) * dx

def con_sum(x):
    return np.sum(x) - _G_TGT

def slsqp_optimize(h_start, n, beta_list, maxiter_per_beta=200):
    global _G_N, _G_BETA, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    for beta in beta_list:
        _G_BETA = float(beta)
        try:
            res = minimize(logsumexp_obj, h_cur, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per_beta, 'ftol': 1e-15, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, c5_cur

    T_end = 0.001

    for it in range(iterations):
        if budget_fn() < 5.0:
            break
        T = T_start * (T_end / T_start) ** (it / max(1, iterations - 1))

        n_flips = rng.randint(1, 4)

            cur_h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
    
    return best_h, best_c5
