        seq.extend([1] + [1 - s for s in reversed(seq)])
    h[:n] = seq[:n]
    return h

def do_lbfgsb_phase(h_start, n, budget_fn, beta_list):
    global _G_N, _G_BETA, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def obj_pen(x):
        base = obj_lse(x)
        pen = 50.0 * (np.sum(x) - _G_TGT) ** 2
        return base + pen

    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    for beta in beta_list:
        if budget_fn() < 4.0:
            break
        _G_BETA = float(beta)
        try:
            res = minimize(obj_pen, h_cur, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 800, 'ftol': 1e-18, 'gtol': 1e-12})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                h_cur = h_new.copy()
        except Exception:
            pass
    return h_cur, c5_cur


def do_power_mean_phase(h_start, n, budget_fn, p_list):
    global _G_N, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    for p in p_list:
        if budget_fn() < 4.0:
            break
        pf = float(p)
        try:
            res = minimize(lambda x: obj_power_mean(x, pf), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-16, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                h_cur = h_new.copy()
        except Exception:
            pass
    return h_cur, c5_cur


def do_subgrad_phase(h_best, n, budget_fn, rng, max_iters=300):
    h = h_best.copy()
    c5_best = compute_c5(h)
    no_improve = 0
    for it in range(max_iters):
        if budget_fn() < 3.0:
            break
        grad, top_idx = active_lag_subgrad(h, n, top_k=5)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        grad = grad / gnorm
        improved = False
        steps = [0.002, 0.004, 0.006, 0.01, 0.015, 0.02, 0.03, 0.05, 0.08, 0.12]
        for step in steps:
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-14:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                no_improve = 0
                break
        if not improved:
            no_improve += 1
            if no_improve > 15:
                found = False
                for attempt in range(8):
                    scale = 0.005 + 0.005 * attempt
                    noise = rng.randn(n) * scale
                    h_trial = project_feasible(h + noise, n)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < c5_best - 1e-14:
                        c5_best = c5_trial
                        h_best = h_trial.copy()
                        h = h_trial.copy()
                        no_improve = 0
                        found = True
                        break
                if not found and no_improve > 60:
                    break
    return h_best, c5_best


def do_perturbed_slsqp(h_best, n, budget_fn, rng, n_restarts=5):
    c5_best = compute_c5(h_best)
    for r in range(n_restarts):
        if budget_fn() < 12.0:
            break
        noise_scale = 0.015 + 0.01 * r
        h_pert = project_feasible(h_best + rng.randn(n) * noise_scale, n)
        c5_pert = compute_c5(h_pert)
        h_cur = h_pert if c5_pert < c5_best else h_best.copy()
        c5_cur = min(c5_pert, c5_best)
        h_cur, c5_cur = do_slsqp_phase(h_cur, n, budget_fn, [200, 800, 3000])
        if c5_cur < c5_best - 1e-14:
            c5_best = c5_cur
            h_best = h_cur.copy()
    return h_best, c5_best


def do_focused_refinement(h_best, n, budget_fn, rng, top_k=7):
    """Refine focusing on a larger set of near-active lags."""
    h = h_best.copy()
    c5_best = compute_c5(h)
    for it in range(100):
        if budget_fn() < 2.0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        # Find all lags within a small range of max
        threshold = m - 0.01 * m * (2.0 / n)
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for k_idx in active:
            grad += subgradient_for_lag(h, k_idx, n)
        grad /= len(active)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        grad /= gnorm
        improved = False
        for step in [0.003, 0.005, 0.01, 0.02, 0.04, 0.01]:
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-14:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
        if not improved:
            h_trial = project_feasible(h + rng.randn(n) * 0.005, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-14:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    def budget_fn():
        return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = make_thue_morse(parent_n)
        best_h = project_feasible(parent_h, parent_n)
        best_c5 = compute_c5(best_h)
        best_n = parent_n
        print("No parent. TM: n={}, C5={:.10f}".format(parent_n, best_c5))

    # Phase 1: Thorough SLSQP at parent resolution with fine-grained beta
    print("\n--- Phase 1: SLSQP at n={} ---".format(best_n))
    best_h, best_c5 = do_slsqp_phase(best_h, best_n, budget_fn,
                                     [10, 30, 80, 200, 500, 1200, 3000, 8000, 20000])
    print("  After SLSQP: C5={:.10f}".format(best_c5))

    # Phase 2: Power-mean softening (different landscape)
    if budget_fn() > 25:
        print("\n--- Phase 2: Power-mean ---")
        h_try, c5_try = do_power_mean_phase(best_h, best_n, budget_fn, [4, 8, 16, 32, 64])
        if c5_try < best_c5:
            best_h, best_c5 = h_try, c5_try
            print("  PM improved: C5={:.10f}".format(best_c5))
        else:
            print("  PM no improve: C5={:.10f}".format(c5_try))

    # Phase 3: Subgradient with more active constraints
    if budget_fn() > 20:
        print("\n--- Phase 3: Subgradient (top-5) ---")
        best_h, best_c5 = do_subgrad_phase(best_h, best_n, budget_fn, rng, max_iters=400)
        print("  After subgrad: C5={:.10f}".format(best_c5))

    # Phase 4: Focused refinement on near-active lags
    if budget_fn() > 15:
        print("\n--- Phase 4: Focused refinement ---")
        best_h, best_c5 = do_focused_refinement(best_h, best_n, budget_fn, rng)
        print("  After focused: C5={:.10f}".format(best_c5))

    # Phase 5: L-BFGS-B with penalty (different optimizer path)
    if budget_fn() > 20:
        print("\n--- Phase 5: L-BFGS-B ---")
        h_try, c5_try = do_lbfgsb_phase(best_h, best_n, budget_fn, [100, 400, 1500, 5000])
        if c5_try < best_c5:
            best_h, best_c5 = h_try, c5_try
            print("  LBFGS improved: C5={:.10f}".format(best_c5))
        else:
            print("  LBFGS no improve: C5={:.10f}".format(c5_try))

    # Phase 6: Systematic resolution search (fine grid around parent)
    print("\n--- Phase 6: Resolution search ---")
    tried_ns = set([best_n])
    # Fine grid around parent
    fine_res = []
    for dr in range(-12, 13, 2):
        nr = best_n + dr
        if nr >= 128 and nr % 2 == 0 and nr not in tried_ns:
            fine_res.append(nr)
    # Sort by proximity to parent
    fine_res.sort(key=lambda x: abs(x - best_n))

    for n_try in fine_res:
        if budget_fn() < 18.0:
            break
        h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        h_new, c5_new = do_slsqp_phase(h_interp, n_try, budget_fn, [100, 500, 2000, 8000])
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n_try
            tried_ns.add(n_try)
            print("  n={} IMPROVED: C5={:.10f}".format(n_try, best_c5))

    # Also try some non-local resolutions
    for n_try in [192, 224, 288, 320, 384, 448, 512, 576, 640]:
        if budget_fn() < 15.0 or n_try in tried_ns:
            continue
        h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        h_new, c5_new = do_slsqp_phase(h_interp, n_try, budget_fn, [200, 1000, 5000])
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n_try
            tried_ns.add(n_try)
            print("  n={} IMPROVED: C5={:.10f}".format(n_try, best_c5))

    # Phase 7: Perturbed restarts at best resolution
    if budget_fn() > 30:
        print("\n--- Phase 7: Perturbed restarts ---")
        best_h, best_c5 = do_perturbed_slsqp(best_h, best_n, budget_fn, rng, n_restarts=4)
        print("  After restarts: C5={:.10f}".format(best_c5))

    # Phase 8: More subgradient + focused
    if budget_fn() > 15:
        print("\n--- Phase 8: Subgrad + focused ---")
        best_h, best_c5 = do_subgrad_phase(best_h, best_n, budget_fn, rng, max_iters=200)
        best_h, best_c5 = do_focused_refinement(best_h, best_n, budget_fn, rng)
        print("  After: C5={:.10f}".format(best_c5))

    # Phase 9: Final high-beta SLSQP
    if budget_fn() > 12:
        print("\n--- Phase 9: High-beta polish ---")
        best_h, best_c5 = do_slsqp_phase(best_h, best_n, budget_fn,
                                         [30000, 80000, 200000, 500000])
        print("  After high-beta: C5={:.10f}".format(best_c5))

    # Phase 10: Re-check nearby resolutions from the new best
    if budget_fn() > 20:
        print("\n--- Phase 10: Recheck resolutions ---")
        for dr in [-4, -2, 2, 4, 6, -6]:
            n_try = best_n + dr
            if n_try < 128 or n_try in tried_ns or budget_fn() < 12:
                continue
            h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
            h_new, c5_new = do_slsqp_phase(h_interp, n_try, budget_fn, [500, 2000, 10000])
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_try
                tried_ns.add(n_try)
                print("  n={} IMPROVED: C5={:.10f}".format(n_try, best_c5))

    # Phase 11: Final tiny refinements
    if budget_fn() > 8:
        best_h, best_c5 = do_subgrad_phase(best_h, best_n, budget_fn, rng, max_iters=80)
    if budget_fn() > 6:
        best_h, best_c5 = do_slsqp_phase(best_h, best_n, budget_fn, [100000, 300000])
    if budget_fn() > 4:
        best_h, best_c5 = do_focused_refinement(best_h, best_n, budget_fn, rng)

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
