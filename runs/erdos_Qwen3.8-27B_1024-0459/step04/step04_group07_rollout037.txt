def project_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    h = np.array(h, dtype=float).copy()

def smooth_max_grad(h, beta, dx):

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_terms = np.exp(beta * (corr - m))

    for j in range(len(corr)):
        if w[j] < 1e-14:
            continue

        ww = w[j] * dx

        # d/dh[m] of corr[j]:
        # = (1-h[m-lag]) * I[m-lag in [n_lo, n_hi], 0<=m<n]
        #   - h[m+lag] * I[m in [n_lo, n_hi], 0<=m+lag<n]

        m2_hi = min(n_hi, n-1-lag, n-1)
        if m2_lo <= m2_hi:
            grad[m2_lo:m2_hi+1] += ww * (-h[m2_lo + lag : m2_hi + 1 + lag])
    
    return grad, (m + np.log(np.sum(exp_terms)) / beta) * dx

def lhfess_run(h0, n, beta, maxiter, pen):
    """L-BFGS-B with log-sum-exp smooth max."""
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        val, _ = smooth_max(x, beta, dx)
        val += pen * (np.sum(x) - target) ** 2
        return val

        g, _ = smooth_max_grad(x, beta, dx)
        g += 2 * pen * (np.sum(x) - target)
        return g
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10, 'iprint': 0, 'maxfun': 5000})

Actually, let me just look at this more carefully. From numpy docs:

In 'full' mode, k ranges from -(len(v)-1) to (len(a)-1).
The output array index for lag k is: k + (len(v) - 1) = k + (n-1).

So output[j] corresponds to lag = j - (n-1).

Valid n: 0 ≤ n < n AND 0 ≤ n + (j-(n-1)) < n

i.e., max(0, -(j-(n-1))) ≤ n ≤ min(n-1, n-1-(j-(n-1)))

Let lag = j - (n-1):
Valid n: max(0, -lag) ≤ n ≤ min(n-1, n-1-lag)

⟺ m-lag ≥ max(0,-lag) AND m-lag ≤ min(n-1,n-1-lag)
⟺ m ≥ lag (if lag≥0) or m ≥ 0 (if lag<0, trivially true)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c else 0.0
    return h

def make_rand_bin(rng, n):

    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def smooth_max_and_grad(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_terms = np.exp(beta * (corr - m))
    s = np.sum(exp_terms)
    w = exp_terms / s
    val = (m + np.log(s) / beta) * dx
    grad = np.zeros(n)
    for j in range(len(corr)):
        if w[j] < 1e-14:
            continue
        ww = w[j] * dx
        lag = j - (n - 1)
        if lag >= 0:
            m1_lo, m1_hi = lag, n - 1
        else:
            m1_lo, m1_hi = 0, n - 1 + lag
        if m1_lo <= m1_hi:
            grad[m1_lo:m1_hi + 1] += ww * (1.0 - h[m1_lo - lag:m1_hi + 1 - lag])
        if lag >= 0:
            m2_lo, m2_hi = 0, n - 1 - lag
        else:
            m2_lo, m2_hi = -lag, n - 1
        if m2_lo <= m2_hi:
            grad[m2_lo:m2_hi + 1] += ww * (-h[m2_lo + lag:m2_hi + 1 + lag])
    return val, grad

def lbfgs_smooth(h0, n, beta, pen, maxiter):
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        v, _ = smooth_max_and_grad(x, beta, dx)
        v += pen * (np.sum(x) - target) ** 2
        return v

    def grad(x):
        _, g = smooth_max_and_grad(x, beta, dx)
        g += 2.0 * pen * (np.sum(x) - target)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12,
                                'maxfun': maxiter * 3, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)
