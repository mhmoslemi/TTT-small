        m = np.max(corr)
        sh = corr - m

        ex = np.exp(cl)

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def thue_morse_seq(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 else 0.0
    return h


def ensure_balance_binary(h, n, rng):
    h = (h > 0.5).astype(float).copy()
    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones > target:
        ones = np.where(h == 1)[0]
        rem = rng.choice(ones, n_ones - target, replace=False)
        h[rem] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        add = rng.choice(zeros, target - n_ones, replace=False)
        h[add] = 1
    return h


def slsqp_smooth(h, n, beta, maxiter=400):
    dx = 2.0 / n
    h = project_feasible(h, n)
    target_sum = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k] * dx
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g

    def con_fun(x):
        return np.sum(x) - target_sum

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) < compute_c5(h):
                return h_new
    except Exception:
        pass
    return h


def binary_sa(h_init, n, time_limit, rng, T_start=4.0, T_end=0.003,
              n_cand=100, soft_beta=0.0):

    dx = 2.0 / n

    corr = np.correlate(h, 1.0 - h, mode='full')
    if soft_beta > 0:
        m = np.max(corr)
        cur_val = (m + np.log(np.sum(np.exp(soft_beta * (corr - m)))) / soft_beta) * dx
    else:
        cur_val = float(np.max(corr)) * dx
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_c5 = cur_val
    best_max = cur_max

    max_no_improve = 300

                cur_val = cur_max * dx

        h_res, c5_res = binary_sa(h_start, n, sa_time_each_local, rng_sa,
                                  T_start=3.0, T_end=0.003, n_cand=100)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
            best_n = n

        # Also try with soft-max objective
        if remaining() > 30:
            soft_beta = 50.0
            h_res2, c5_res2 = binary_sa(h_start, n, min(sa_time_each_local, 60),
                                        np.random.RandomState(seed + 2000 * idx),
                                        T_start=5.0, T_end=0.01,
                                        n_cand=80, soft_beta=soft_beta)
            c5_check = compute_c5(h_res2)
            if c5_check < best_c5:
                best_c5 = c5_check
                best_h = h_res2.copy()
                best_n = n

    # Phase 3: Greedy improvement on best
    if remaining() > 20:
        h_g, c5_g = greedy_improve(best_h, n, time_limit=min(40, int(remaining() * 0.1)))
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 4: SLSQP refinement (continuous)
    if remaining() > 30:
        for beta in [2000, 8000, 20000, 50000]:
            if remaining() < 12:
                break
            h_new = slsqp_smooth(best_h, best_n, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 5: Try alternative resolutions
    if remaining() > 40:
        candidates = [best_n - 8, best_n + 8, best_n + 16, best_n + 32]
        candidates = [c for c in candidates if 128 <= c <= 768 and c % 2 == 0 and c != best_n]
        for n_try in candidates[:3]:
            if remaining() < 25:
                break
            xp = (np.arange(best_n) + 0.5) * 2.0 / best_n
            x_new = (np.arange(n_try) + 0.5) * 2.0 / n_try
            h_interp = np.interp(x_new, xp, best_h)
            h_interp = project_feasible(h_interp, n_try)

            h_opt = slsqp_smooth(h_interp, n_try, 3000, maxiter=300)
            c5_opt = compute_c5(h_opt)

            # Binary SA at this resolution
            h_bin_t = (h_opt > 0.5).astype(float)
            h_bin_t = ensure_balance_binary(h_bin_t, n_try, rng)
            sa_t = min(40, max(10, int(remaining() * 0.1)))
            if sa_t > 5:
                h_sa, c5_sa = binary_sa(h_bin_t, n_try, sa_t,
                                        np.random.RandomState(seed + 5000),
                                        T_start=3.0, T_end=0.005, n_cand=80)
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
                    best_n = n_try
                    continue

            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_try

    # Phase 6: Final intensive binary SA on best
    if remaining() > 30:
        h_final_bin = (best_h > 0.5).astype(float)
        h_final_bin = ensure_balance_binary(h_final_bin, best_n, rng)
        final_sa_time = min(120, max(15, int(remaining() * 0.35)))
        h_f, c5_f = binary_sa(h_final_bin, best_n, final_sa_time,
                              np.random.RandomState(seed + 9999),
                              T_start=2.0, T_end=0.002, n_cand=120)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 7: Final greedy + SLSQP polish
    if remaining() > 15:
        h_g, c5_g = greedy_improve(best_h, best_n, time_limit=min(20, max(5, int(remaining() * 0.2))))
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    if remaining() > 10:
        h_new = slsqp_smooth(best_h, best_n, 30000, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
