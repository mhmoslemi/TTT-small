    # Find mu such that sum(clip(x[i] + mu, 0, 1)) = target
    
    # x[i] + mu > 1 when mu > 1 - x[i], i.e., mu > max(1 - x[i] for x[i] < 1)
    # x[i] + mu < 0 when mu < -x[i], i.e., mu < min(-x[i] for x[i] > 0)
    
    lo = -x.max() - 1  # ensures all h[i] = 0 (sum = 0 < target)
    hi = 1 - x.min() + 1  # ensures all h[i] = 1 (sum = n > target)
    
    for _ in range(60):
        mu = (lo + hi) / 2
        h = np.clip(x + mu, 0, 1)
        s = h.sum()
        if s < target:
            lo = mu
        else:
            hi = mu
    
    h = np.clip(x + (lo + hi) / 2, 0, 1)
    # Tiny correction
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        mask = (h > 1e-12) & (h < 1 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h

import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(x, n):
    target = n / 2.0
    lo = -x.max() - 1.0
    hi = 1.0 - x.min() + 1.0
    for _ in range(60):
        mu = (lo + hi) / 2.0
        h = np.clip(x + mu, 0.0, 1.0)
        s = h.sum()
        if s < target:
            lo = mu
        else:
            hi = mu
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g


def smooth_obj_grad(x, n, beta):
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60.0, 60.0)
    ex = np.exp(cl)
    s = np.sum(ex)
    val = (m + np.log(s) / beta) * dx_val
    w = ex / s
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-10:
            g += w[k] * corr_grad_per_shift(x, n, k)
    g *= dx_val
    return val, g


def nesterov_opt(h, n, beta, time_limit, top_k_full=None):
    start = time.time()
    x = h.copy().astype(float)
    y = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = 1.0
    lr = 0.5 / (beta * 2.0 / n)
    it = 0
    while time.time() - start < time_limit and remaining() > 5:
        f_y, g_y = smooth_obj_grad(y, n, beta)
        y_new = y - lr * g_y
        y_new = project_feasible(y_new, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mom = (t - 1.0) / t_new
        x_new = y_new + mom * (y_new - y)
        x_new = project_feasible(x_new, n)
        f_x, _ = smooth_obj_grad(x_new, n, beta)
        if f_x <= f_y + 1e-12 * max(1.0, abs(f_y)):
            y = y_new
            x = x_new
            t = t_new
            it += 1
            if it % 20 == 0:
                c5_exact = compute_c5(x)
                if c5_exact < best_c5:
                    best_c5 = c5_exact
                    best_h = x.copy()
        else:
            lr *= 0.95
            it += 1
    c5_final = compute_c5(x)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = x.copy()
    return best_h, best_c5


def slsqp_smooth(h, n, beta, maxiter=400):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta

    def obj(x):
        v, _ = smooth_obj_grad(x, n, beta)
        return v

    def grad(x):
        _, g = smooth_obj_grad(x, n, beta)
        return g

    def con_fun(x):
        return np.sum(x) - _TGT

    def con_jac(x):
        return np.ones(_N)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def topk_subgrad(h, n, max_iters=300, top_k=8):
    h = h.copy().astype(float)
    best_c5 = compute_c5(h)
    dx_val = 2.0 / n
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        m = float(np.max(corr))
        top_idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k in top_idx:
            g += corr_grad_per_shift(h, n, int(k))
        g /= len(top_idx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        improved = False
        for step in [0.1, 0.05, 0.02, 0.01, 0.2, 0.005, 0.03]:
            h_trial = project_feasible(h + step * d, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-14:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            for step in [0.001, 0.002, 0.0005]:
                h_trial = project_feasible(h + step * d, n)
                c_new = compute_c5(h_trial)
                if c_new < best_c5 - 1e-14:
                    h = h_trial
                    best_c5 = c_new
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def sa_binary(h_init, n, time_limit=60, seed=42):
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    target = n // 2 if n % 2 == 0 else n // 2
    n_ones = int(np.sum(h))
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1

    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()

    T = 6.0
    alpha = 0.9992
    start = time.time()
    stall = 0
    max_stall = 3000

    while time.time() - start < time_limit and T > 0.05:
        if remaining() < 8:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])

        h[i] = 0.0
        h[j] = 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        delta = new_max - cur_max

        if delta < 0 or (T > 0.05 and rng.rand() < np.exp(-delta * n / T)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1

        T *= alpha
        if stall > max_stall:
            T = max(T, 2.0)
            stall = 0
            max_stall = int(max_stall * 1.3)

    return best_h, best_max * dx_val


def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float).copy()

    target = n // 2
    if n_ones > target:

    elif n_ones < target:

    return h_bin


def structured_init(n, seed, pattern_type=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    target = n // 2
    if pattern_type == 0:
        # Spread ones evenly
        positions = np.linspace(0, n, target + 1, dtype=int)[:target]
        h[positions] = 1.0
    elif pattern_type == 1:
        # Thue-Morse like
        tm = np.zeros(n, dtype=int)
        for i in range(1, n):
            tm[i] = tm[i >> 1] ^ (i & 1)
        idx = np.where(tm == 0)[0][:target]
        h[idx] = 1.0
        if len(idx) < target:
            remaining_idx = np.where(tm == 1)[0][:target - len(idx)]
            h[remaining_idx] = 1.0
    else:
        # Random
        positions = rng.choice(n, target, replace=False)
        h[positions] = 1.0
    return h
