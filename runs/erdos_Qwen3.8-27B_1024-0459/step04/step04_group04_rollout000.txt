    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr * beta - m * beta
    cl = np.clip(shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    # Only compute for significant weights
    thresh = 1e-10
    for k in range(len(corr)):
        if w[k] > thresh:
            g += w[k] * corr_grad_for_shift(x, n, k)
    return g * dx_val

def subgradient_step(h, n, top_k=20, alpha=0.1, eta=0.99):
    """One step of subgradient descent targeting top-k correlations."""
    corr = compute_corr(h)
    # Get top-k shift indices
    top_idx = np.argsort(-corr)[:top_k]
    g = np.zeros(n)
    for k in top_idx:
        g += corr_grad_for_shift(h, n, int(k))
    g /= top_k
    g_norm = np.linalg.norm(g)
    if g_norm < 1e-14:
        return h
    h_new = h - alpha * g / g_norm
    h_new = project_feasible(h_new, n)
    return h_new

            corr_cur = compute_corr(h)
            worst_k = int(np.argmax(corr_cur))
            d = worst_k - (n - 1)
            # Try to find a swap that reduces corr at this shift
            # For shift d >= 0: positions i where h[i]=1 and h[i+d]=0 contribute
            if d >= 0:
                m = n - d
                # Good a: h[a]=1, h[a+d]=0 (removes contribution)
                good_a = [i for i in range(m) if h[i] == 1 and h[i+d] == 0]
                # Good b: h[b]=0, and flipping to 1 doesn't add at this shift
                # h[b](1-h[b+d]) goes from 0 to (1-h[b+d])
                # So we want h[b+d]=1 (no contribution added)
                # Also if b-d >= 0: h[b-d](1-h[b]) goes from h[b-d] to 0 (removes if h[b-d]=1)
                good_b = [j for j in range(n) if h[j] == 0 and (j + d >= n or h[j+d] == 1)]
            else:

                m = n - a_abs
                good_a = [i for i in range(a_abs, n) if h[i] == 1 and h[i-a_abs] == 0]
                good_b = [j for j in range(n) if h[j] == 0 and (j - a_abs < 0 or h[j-a_abs] == 1)]

                b = good_b[rng.randint(len(good_b))]

            b = zeros[rng.randint(len(zeros))]

        new_max = float(np.max(new_corr))
        delta = new_max - cur_max
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 0.01)):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:

            stall += 1

            T = max(T, 3.0)
            stall = 0

                    else:

                        if len(o) > 0:

                # Fix sum

                        h[zi[rng.choice(len(zi), target - n_ones, replace=False)]] = 1
                corr = compute_corr(h)
                cur_max = float(np.max(corr))
    
    return best_h, best_max * dx_val

import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argsort(h)
        if diff > 0:
            start = len(idx) - int(np.ceil(abs(diff)))
        else:
            start = 0
        for i in range(start, min(start + int(abs(np.ceil(abs(diff)) * 2)) + 5, n)):
            if diff > 0 and h[idx[i]] < 1.0:
                h[idx[i]] = min(1.0, h[idx[i]] + diff)
                diff = 0
            elif diff < 0 and h[idx[i]] > 0.0:
                h[idx[i]] = max(0.0, h[idx[i]] + diff)
                diff = 0
            if abs(diff) < 1e-12:
                break
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def corr_grad_for_shift(x, n, shift_idx):
    d = shift_idx - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        m = n - d
        g[:m] = (1.0 - x[d:d + m])
        g[d:d + m] -= x[:m]
    else:
        a = -d
        g[a:] = (1.0 - x[:n - a])
        g[:n - a] -= x[a:]
    return g


def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = (corr - m) * beta
    cl = np.clip(shifted, -50, 50)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = (corr - m) * beta
    cl = np.clip(shifted, -50, 50)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-12:
            g += w[k] * corr_grad_for_shift(x, n, k)
    return g * dx_val


def slsqp_optimize(h, n, beta, maxiter=300):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - _TGT,
             'jac': lambda x: np.ones(_N)}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def subgradient_descent(h, n, max_iters=2000, top_k=15, alpha_init=0.5,
                        alpha_decay=0.999, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha_init

    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        cur_max_val = float(np.max(corr) * (2.0 / n))
        if cur_max_val < best_c5:
            best_c5 = cur_max_val
            best_h = h.copy()

        top_idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k in top_idx:
            g += corr_grad_for_shift(h, n, int(k))
        g /= top_k
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        # Adaptive step size
        h_try = h - alpha * g / gn
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)

        if c_try < best_c5 + 1e-10:
            h = h_try
        else:
            # Try smaller step
            for frac in [0.5, 0.25, 0.1]:
                h_try2 = h - alpha * frac * g / gn
                h_try2 = project_feasible(h_try2, n)
                c_try2 = compute_c5(h_try2)
                if c_try2 < compute_c5(h):
                    h = h_try2
                    break
            alpha *= 0.95
        alpha = max(alpha * alpha_decay, 0.01)
    return h, best_c5


def mirror_descent(h, n, max_iters=1000, top_k=20, alpha=0.3, seed=42):
    """Multiplicative weight update for [0,1] constrained problem."""
    rng = np.random.RandomState(seed)
    h = h.copy() + 0.01 * rng.rand(n)
    h = np.clip(h, 0.01, 0.99)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        cur_c5 = float(np.max(corr) * (2.0 / n))
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        top_idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k in top_idx:
            g += corr_grad_for_shift(h, n, int(k))
        g /= top_k

        h = h * np.exp(-alpha * g)
        h = np.clip(h, 1e-6, 1.0 - 1e-6)
        h = project_feasible(h, n)
    return best_h, best_c5

        # Subgradient phase
        t_sub = t_per * 0.4
        max_it_sub = max(100, int(t_sub * 200))
        h_sg, c5_sg = subgradient_descent(h_try, n, max_iters=max_it_sub,
                                          top_k=10, alpha_init=0.3, seed=seed + s * 100)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

        # Mirror descent phase
        if remaining() > 8:
            h_md, c5_md = mirror_descent(h_sg, n, max_iters=300, top_k=15,
                                         alpha=0.2, seed=seed + s * 200)
            if c5_md < best_c5:
                best_c5 = c5_md
                best_h = h_md.copy()

    # SLSQP polish
    if remaining() > 10:
        for beta in [500, 2000, 8000]:
            if remaining() < 8:
                break
            h_sl = slsqp_optimize(best_h, n, beta, maxiter=200)
            c5_sl = compute_c5(h_sl)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()

    return best_h, best_c5


def make_structured_init(n, seed=0):
    """Create structured initial patterns."""
    rng = np.random.RandomState(seed)
    target = n // 2

    # Option 1: modular pattern
    p = max(3, n // 50)
    h = np.zeros(n)
    h[p // 2::p] = 1.0
    # Adjust to target
    n_ones = int(np.sum(h))
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 360
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Binary SA at multiple resolutions (30% of time)
    n_list = [360, 480, 600, 720, 960, 1200, 1440]
    n_list = [n for n in n_list if n % 2 == 0]
    t_phase1 = budget_s * 0.28

    for n_t in n_list:
        if remaining() < t_phase1 * 0.15:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_interp = project_feasible(h_interp, n_t)
        h_bin = (h_interp > 0.5).astype(float)
        # Fix count
        n_ones = int(np.sum(h_bin))
        target = n_t // 2
        if n_ones > target:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
        elif n_ones < target:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1

        t_sa = min(25, max(8, remaining() * 0.08))
        for sa_seed in [seed, seed + 100 + n_t]:
            if remaining() < 10:
                break
            h_sa, c5_sa = sa_binary(h_bin, n_t, time_limit=t_sa / 2, seed=sa_seed, directed=True)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_t

        # Also try structured init at this n
        if n_t >= 480 and remaining() > 20:
            h_struct = make_structured_init(n_t, seed=seed + n_t)
            h_sa2, c5_sa2 = sa_binary(h_struct, n_t, time_limit=min(12, t_sa / 3),
                                      seed=seed + 500 + n_t, directed=True)
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2.copy()
                best_n = n_t

    # Phase 2: Continuous optimization at best and nearby n (45% of time)
    n_candidates = sorted(set([
        best_n, best_n - 120, best_n + 120,
        480, 600, 720, 840, 960, 1200, 1440
    ]))
    n_candidates = [n for n in n_candidates if n % 2 == 0 and 240 <= n <= 1800]

    t_phase2 = budget_s * 0.42
    t_per_n = max(15, t_phase2 / len(n_candidates))

    for n_t in n_candidates:
        if remaining() < 20:
            break
        if n_t == best_n:
            h_init = best_h.copy()
        else:
            h_init = interp_to(best_h, best_n, n_t)
            h_init = project_feasible(h_init, n_t)

        t_alloc = min(t_per_n, remaining() * 0.5)
        h_opt, c5_opt = multi_start_continuous(h_init, n_t, time_limit=t_alloc, seed=seed + n_t)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 3: Intensive refinement on best (20% of time)
    n = best_n
    h = best_h.copy()
    t_phase3 = min(budget_s * 0.20, remaining() * 0.5)

    # Extended subgradient
    if remaining() > 10:
        h_sg, c5_sg = subgradient_descent(h, n, max_iters=3000, top_k=20,
                                          alpha_init=0.2, alpha_decay=0.9995, seed=seed + 9000)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # SLSQP with high beta
    for beta in [3000, 10000, 30000]:
        if remaining() < 8:
            break
        h_sl = slsqp_optimize(h, n, beta, maxiter=400)
        c5_sl = compute_c5(h_sl)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()
            h = h_sl.copy()

    # More subgradient with smaller top_k (focus on the worst few)
    if remaining() > 10:
        h_sg2, c5_sg2 = subgradient_descent(h, n, max_iters=2000, top_k=5,
                                            alpha_init=0.15, alpha_decay=0.999, seed=seed + 9500)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()

    # Phase 4: Binary refinement from best continuous (10% of time)
    if remaining() > 25:
        n_b = best_n
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        target = n_b // 2
        if n_ones > target:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
        elif n_ones < target:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1

        t_sa_final = min(40, remaining() * 0.4)
        for sa_seed in [seed + 7000, seed + 7500, seed + 8000]:
            if remaining() < 10:
                break
            h_sa, c5_sa = sa_binary(h_bin, n_b, time_limit=t_sa_final / 3,
                                    seed=sa_seed, directed=True)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_b

        # Re-optimize continuously from best binary
        if remaining() > 15:
            h_cont = best_h + 0.0
            h_cont = project_feasible(h_cont, best_n)
            h_opt2, c5_opt2 = multi_start_continuous(h_cont, best_n,
                                                     time_limit=min(30, remaining() * 0.4),
                                                     seed=seed + 10000)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()

    # Final: try a couple more n values if time allows
    if remaining() > 30:
        for n_try in [best_n + 60, best_n - 60, 720, 1440]:
            if remaining() < 15:
                break
            if n_try % 2 != 0 or n_try < 240 or n_try > 1800:
                continue
            h_in = interp_to(best_h, best_n, n_try)
            h_in = project_feasible(h_in, n_try)
            h_opt3, c5_opt3 = multi_start_continuous(h_in, n_try,
                                                     time_limit=min(25, remaining() * 0.4),
                                                     seed=seed + n_try)
            if c5_opt3 < best_c5:
                best_c5 = c5_opt3
                best_h = h_opt3.copy()
                best_n = n_try

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
