def slsqp_smooth(h, n, beta, maxiter=300):
    """SLSQP with log-sum-exp smoothing."""
    def _sm_obj(x):
        h_l = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h_l, 1.0 - h_l, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0/n)
    
    def _sm_eq(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sm_eq}]
    try:
        res = minimize(_sm_obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

            h_l = x[:n]
            t = x[n]
            c = np.correlate(h_l, 1.0 - h_l, mode='full')
            return t - dx * c[active]
        
        def _mm_eq(x):
            return np.sum(x[:n]) - n / 2.0
        
        x0 = np.concatenate([h, [t_init]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [{'type': 'ineq', 'fun': _mm_ineq}, {'type': 'eq', 'fun': _mm_eq}]
        try:
            res = minimize(_mm_obj, x0, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
            if res.x is not None:
                h_new = project_feasible(res.x[:n], n)
                if compute_c5(h_new) < compute_c5(h) - 1e-16:
                    h = h_new
                else:
                    if rnd > 4:
                        break
        except Exception:
            break
    
    return h

def binary_round(h, n, threshold=0.3):
    """Round near-binary values and re-project."""
    h_bin = h.copy()
    h_bin[h_bin < threshold] = 0.0
    h_bin[h_bin > 1.0 - threshold] = 1.0
    return project_feasible(h_bin, n)

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 256
        best_h = np.full(best_n, 0.5)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)
    
    # Phase 1: Coordinate descent at parent resolution
    if remaining_time() > 30:
        h_cd, c5_cd = coord_descent(best_h, best_n, max_sweeps=80000, 
                                     grid_pts=21, refine_pts=11, stop_time=30, seed=seed)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Phase 2: High-beta SLSQP polish
    if remaining_time() > 25:
        h_s = slsqp_smooth(best_h, best_n, 2000, maxiter=400)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
    
    # Phase 3: More CD after SLSQP
    if remaining_time() > 40:
        h_cd2, c5_cd2 = coord_descent(best_h, best_n, max_sweeps=60000,
                                       grid_pts=25, refine_pts=13, stop_time=30, seed=seed+1)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
    
    # Phase 4: Minimax SLSQP
    if remaining_time() > 30:
        h_mm = slsqp_minimax(best_h, best_n, max_rounds=12, max_iter=250)
        c5_mm = compute_c5(h_mm)
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm.copy()
        if remaining_time() > 20:
            h_cd3, c5_cd3 = coord_descent(best_h, best_n, max_sweeps=40000,
                                           grid_pts=21, refine_pts=11, stop_time=15, seed=seed+2)
            if c5_cd3 < best_c5:
                best_c5 = c5_cd3
                best_h = h_cd3.copy()
    
    # Phase 5: Multi-resolution exploration
    trial_ns = [best_n + 16, best_n + 32, best_n + 64, best_n - 32, best_n - 16]
    trial_ns = [tn for tn in trial_ns if 64 <= tn <= 1024]
    # Add some standard sizes
    for tn in [192, 224, 288, 320, 384, 448, 512]:
        if tn not in trial_ns and 64 <= tn <= 1024:
            trial_ns.append(tn)
    trial_ns = sorted(set(trial_ns))
    
    for trial_n in trial_ns:
        if remaining_time() < 20:
            break
        if trial_n == best_n:
            continue
        
        h_interp = interp_to(best_h, best_n, trial_n)
        h_interp = project_feasible(h_interp, trial_n)
        c5_interp = compute_c5(h_interp)
        
        if c5_interp < best_c5 + 0.005:
            max_sw = 40000 if trial_n <= 300 else 25000
            h_cd_r, c5_cd_r = coord_descent(h_interp, trial_n, max_sweeps=max_sw,
                                             grid_pts=19, refine_pts=9, stop_time=15, seed=seed+trial_n)
            if c5_cd_r < best_c5:
                best_c5 = c5_cd_r
                best_h = h_cd_r.copy()
                best_n = trial_n
            
            if remaining_time() > 12:
                h_s2 = slsqp_smooth(best_h, best_n, 3000, maxiter=200)
                c5_s2 = compute_c5(h_s2)
                if c5_s2 < best_c5:
                    best_c5 = c5_s2
                    best_h = h_s2.copy()
    
    # Phase 6: Binary structure exploration
    if remaining_time() > 25:
        h_bin = binary_round(best_h, best_n, threshold=0.35)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5 + 0.003:
            h_cd_b, c5_cd_b = coord_descent(h_bin, best_n, max_sweeps=30000,
                                             grid_pts=15, refine_pts=7, stop_time=15, seed=seed+100)
            if c5_cd_b < best_c5:
                best_c5 = c5_cd_b
                best_h = h_cd_b.copy()
    
    # Phase 7: Perturbation + CD restarts
    for restart in range(3):
        if remaining_time() < 25:
            break
        h_p = best_h.copy()
        mask_mid = (h_p > 0.1) & (h_p < 0.9)
        if np.sum(mask_mid) > 0:
            h_p[mask_mid] += rng.normal(0, 0.03, np.sum(mask_mid))
        idx = rng.choice(best_n, size=min(8, best_n), replace=False)
        for j in idx:
            h_p[j] = 1.0 - h_p[j]
        h_p = project_feasible(h_p, best_n)
        
        h_cd_p, c5_cd_p = coord_descent(h_p, best_n, max_sweeps=20000,
                                         grid_pts=15, refine_pts=7, stop_time=12, seed=seed+200+restart)
        if c5_cd_p < best_c5:
            best_c5 = c5_cd_p
            best_h = h_cd_p.copy()
    
    # Phase 8: Final ultra-precision CD
    if remaining_time() > 15:
        h_cd_f, c5_cd_f = coord_descent(best_h, best_n, max_sweeps=50000,
                                         grid_pts=31, refine_pts=15, stop_time=10, seed=seed+999)
        if c5_cd_f < best_c5:
            best_c5 = c5_cd_f
            best_h = h_cd_f.copy()
    
    # Phase 9: Final SLSQP polish
    if remaining_time() > 10:
        h_sf = slsqp_smooth(best_h, best_n, 10000, maxiter=300)
        c5_sf = compute_c5(h_sf)
        if c5_sf < best_c5:
            best_c5 = c5_sf
            best_h = h_sf.copy()
    
    # Phase 10: Final CD on the polished result
    if remaining_time() > 8:
        h_cd_ff, c5_cd_ff = coord_descent(best_h, best_n, max_sweeps=30000,
                                           grid_pts=25, refine_pts=13, stop_time=5, seed=seed+777)
        if c5_cd_ff < best_c5:
            best_c5 = c5_cd_ff
            best_h = h_cd_ff.copy()
    
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
