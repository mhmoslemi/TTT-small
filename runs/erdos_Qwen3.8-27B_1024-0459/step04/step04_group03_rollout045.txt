where lag = k - n + 1.

Note: when lag = 0, both terms are h[i], so d(corr[k])/dh[i] = 1 - 2*h[i] for the lag=0 term.

For the full gradient:
dF/dh[i] = dx * sum_k p[k] * [1 - h[i+lag_k]*I1 - h[i-lag_k]*I2]
= dx * [1 - sum_k p[k]*h[i+lag_k]*I1 - sum_k p[k]*h[i-lag_k]*I2]

and B[i] = sum_k p[k]*h[i-lag_k]*I(i-lag_k valid)

Note that A and B are related: A[i] = B[i] (by reindexing k → 2n-2-k). Actually no, p is not symmetric in general.

Anyway, let me just compute A and B directly:

A[i] = sum_{k: i+lag_k in [0,n-1]} p[k] * h[i+lag_k]

Actually, I realize I can compute this as a convolution:
A = correlate(p_shifted, h) where p_shifted is the array p with appropriate indexing.

```python
def smooth_grad(x, n, beta):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(beta * (corr - max_c))
    p = exp_c / np.sum(exp_c)
    
    A = np.zeros(n)  # A[i] = sum_k p[k] * h[i + lag_k] (valid)
    B = np.zeros(n)  # B[i] = sum_k p[k] * h[i - lag_k] (valid)
    
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-18:
            continue
        lag = k - n + 1
        if lag >= 0:
            # i + lag in [0, n-1] → i in [0, n-1-lag]
            A[:n-lag] += pk * h[lag:]
            # i - lag in [0, n-1] → i in [lag, n-1]
            B[lag:] += pk * h[:n-lag]
        else:
            il = -lag
            # i + lag = i - il in [0, n-1] → i in [il, n-1]
            A[il:] += pk * h[:n-il]
            # i - lag = i + il in [0, n-1] → i in [0, n-1-il]
            B[:n-il] += pk * h[il:]
    
    return dx * (1.0 - A - B)
```

Let me verify: when k = n-1, lag = 0.
A[:n] += p[n-1] * h[0:] → A[i] += p[n-1]*h[i]
B[0:] += p[n-1] * h[0:] → B[i] += p[n-1]*h[i]
Total: A[i] + B[i] gets 2*p[n-1]*h[i]
d(corr)/dh[i] for lag=0 term: 1 - 2*h[i] ✓

import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_N_G = 0
_BETA_G = 1.0
_LAGS_G = None
_DX_G = 0.0

def compute_c5(h, n=None):
    if n is None:
        n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-16:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        for i in range(n):
            nv = h[i] + diff
            if 0.0 <= nv <= 1.0:
                h[i] = nv
                break
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def smooth_objective(x, n, beta):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    sh = beta * (corr - mc)
    return (mc + np.log(np.sum(np.exp(sh))) / beta) * dx

def smooth_gradient(x, n, beta):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    ec = np.exp(beta * (corr - mc))
    p = ec / np.sum(ec)
    
    A = np.zeros(n)
    B = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-18:
            continue
        lag = k - n + 1
        if lag >= 0:
            A[:n - lag] += pk * h[lag:]
            B[lag:] += pk * h[:n - lag]
        else:
            il = -lag
            A[il:] += pk * h[:n - il]
            B[:n - il] += pk * h[il:]
    return dx * (1.0 - A - B)

def sd_optimize(h, n, beta_start, beta_end, max_iter, ftol=1e-14):
    """Smoothed gradient descent with adaptive beta."""
    h = h.copy()
    beta = beta_start
    n_betas = 10
    beta_schedule = np.geomspace(beta_start, beta_end, n_betas)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    
    for bi, beta in enumerate(beta_schedule):
        if remaining_time() < 5:
            break
        x = h.copy()
        for it in range(max_iter):
            if remaining_time() < 3:
                break
            obj = smooth_objective(x, n, beta)
            grad = smooth_gradient(x, n, beta)
            
            # Check convergence
            if np.max(np.abs(grad)) < ftol:
                break
            
            # Backtracking line search
            step = 1.0
            for _ in range(30):
                x_new = x - step * grad
                x_new = np.clip(x_new, 0.0, 1.0)
                x_new = project_feasible(x_new, n)
                obj_new = smooth_objective(x_new, n, beta)
                if obj_new < obj - 1e-12:
                    break
                step *= 0.5
            else:
                break
            
            if step * np.max(np.abs(grad)) < 1e-15:
                break
            x = x_new
        
        h = project_feasible(x, n)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def _mm_obj(x):
    return x[_N_G]

def _mm_ineq(x):
    n = _N_G
    dx = _DX_G
    h = x[:n]
    t = x[n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return t - dx * corr[_LAGS_G]

def _mm_eq(x):
    return np.sum(x[:_N_G]) - _N_G / 2.0

def minimax_slsqp(h_init, n, max_rounds=15, max_iter_per=200, active_frac=0.3, top_k=5):
    global _N_G, _DX_G, _LAGS_G
    _N_G = n
    _DX_G = 2.0 / n
    h = h_init.copy()
    dx = 2.0 / n
    
    for rnd in range(max_rounds):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        max_corr = np.max(corr)
        
        # Active set: top lags + lags above threshold
        thr = max_corr - (1 - active_frac) * (np.max(corr) - np.min(corr))
        active = np.where(corr >= thr)[0]
        top = np.argsort(corr)[::-1][:top_k]
        active = np.unique(np.concatenate([active, top]))
        active = np.sort(active)
        
        _LAGS_G = active
        t_init = max_corr * dx + 1e-10
        x0 = np.concatenate([h, [t_init]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': _mm_ineq},
            {'type': 'eq', 'fun': _mm_eq}
        ]
        try:
            res = minimize(_mm_obj, x0, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter_per, 'ftol': 1e-14, 'iprint': 0})
            if res.x is not None:
                h_new = project_feasible(res.x[:n], n)
                c5_new = compute_c5(h_new, n)
                c5_old = compute_c5(h, n)
                if c5_new < c5_old - 1e-16:
                    h = h_new
                elif rnd > 3:
                    break
        except Exception:
            break
    
    return h

def _sm_eq(x):
    return np.sum(x) - _N_G / 2.0

def slsqp_smooth(h, n, beta, maxiter=300):
    global _N_G, _BETA_G
    _N_G = n
    _BETA_G = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sm_eq}]
    try:
        res = minimize(lambda x: smooth_objective(x, n, beta), h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def local_search(h, n, max_rounds=5, rng=None):
    """Pair transfer local search."""
    if rng is None:
        rng = np.random.RandomState(0)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    
    for rnd in range(max_rounds):
        if remaining_time() < 3:
            break
        improved = False
        order = rng.permutation(n)
        for ii in order:
            i = int(ii)
            hi = best_h[i]
            if hi < 0.02 or hi > 0.98:
                continue
            for j in range(n):
                if j == i:
                    continue
                hj = best_h[j]
                diff = hi - hj
                if abs(diff) < 0.02:
                    continue
                for frac in [0.3, 0.5, 0.7]:
                    amt = diff * frac
                    if abs(amt) < 0.005:
                        continue
                    trial = best_h.copy()
                    trial[i] -= amt
                    trial[j] += amt
                    trial = np.clip(trial, 0.0, 1.0)
                    trial = project_feasible(trial, n)
                    c5_t = compute_c5(trial, n)
                    if c5_t < best_c5 - 1e-14:
                        best_c5 = c5_t
                        best_h = trial.copy()
                        hi = best_h[i]
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5
