    target = n / 2.0

def active_set_lp_direction(h, n, top_k=10):
    """Find direction using LP on active set."""
    corr = compute_corr(h)
    top_idx = np.argsort(-corr)[:top_k]

    grads = np.array([corr_grad_single(h, n, int(k)) for k in top_idx])
    # (top_k, n)

    # Variables: d_0, ..., d_{n-1}, z (total n+1)
    # Objective: [0, 0, ..., 0, 1] (minimize z)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: grads @ d - z <= 0
    A_ub = np.hstack([grads, -np.ones((top_k, 1))])
    b_ub = np.zeros(top_k)
    
    # Bounds: -1 <= d_i <= 1, z unbounded below (but effectively bounded by constraints)
    bounds = [(-1, 1)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status == 0:

    g = np.zeros(n)
    if d >= 0:
        g[d:] += (1.0 - h[:n - d])

    else:
        m = -d
        g[m:] += (1.0 - h[:n - m])
        g[:n - m] -= h[m:]

def smooth_max_obj_grad(h, n, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50, 50)
    ex = np.exp(shifted)
    s = np.sum(ex)
    w = ex / s
    obj = (m + np.log(s) / beta) * (2.0 / n)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-12:
            g += w[k] * corr_grad_single(h, n, k)
    g *= 2.0 / n
    return obj, g


def slsqp_optimize(h, n, beta, maxiter=300):

def top_k_subgradient(h, n, top_k=20, max_iters=3000, time_limit=60):
    h = h.copy()
    start = time.time()
    best_c5 = compute_c5(h)
    step_size = 0.05
    momentum = np.zeros(n)
    beta_mom = 0.9
    beta_ann = 0.9995

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break

        corr = compute_corr(h)
        sorted_idx = np.argsort(-corr)
        top = sorted_idx[:top_k]
        c_max = corr[top[0]]
        span = corr[top[0]] - corr[top[-1]]

        if span < 1e-10:
            weights = np.ones(top_k) / top_k
        else:
            weights = np.exp(200.0 * (corr[top] - c_max))
            weights /= weights.sum()

        g = np.zeros(n)
        for w, k in zip(weights, top):
            g += w * corr_grad_single(h, n, int(k))

        momentum = beta_mom * momentum + g
        h_trial = project_feasible(h - step_size * momentum, n)
        c5_trial = compute_c5(h_trial)

            if step_size < 0.001:
                step_size = 0.01
                momentum = g

        beta_mom = min(beta_mom * beta_ann, 0.98)

    return h, best_c5

                stall = 0

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def refine_continuous(h, n, time_budget=120):
    """Multi-stage continuous refinement."""
    start = time.time()
    best_c5 = compute_c5(h)
    h = h.copy()

    # Stage 1: Top-K subgradient (broad)
    t_rem = time_budget - (time.time() - start)
    if t_rem > 15:
        h, c5 = top_k_subgradient(h, n, top_k=20, max_iters=2000,
                                  time_limit=min(40, t_rem))
        if c5 < best_c5:
            best_c5 = c5

    # Stage 2: Multi-active descent (targeted)
    t_rem = time_budget - (time.time() - start)
    if t_rem > 10:
        h, c5 = multi_active_descent(h, n, max_iters=300,
                                     time_limit=min(30, t_rem), n_active=8)
        if c5 < best_c5:
            best_c5 = c5
        elif c5 < best_c5 + 1e-6:
            h, c5 = multi_active_descent(h, n, max_iters=200,
                                         time_limit=min(20, t_rem), n_active=5)
            if c5 < best_c5:
                best_c5 = c5

    # Stage 3: SLSQP polish
    t_rem = time_budget - (time.time() - start)
    if t_rem > 10:
        for beta in [500, 2000, 8000, 20000]:
            t_rem = time_budget - (time.time() - start)
            if t_rem < 5:
                break
            h_new = slsqp_optimize(h, n, beta, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h = h_new

    # Stage 4: Final active descent
    t_rem = time_budget - (time.time() - start)
    if t_rem > 5:
        h, c5 = multi_active_descent(h, n, max_iters=200,
                                     time_limit=min(15, t_rem), n_active=15)
        if c5 < best_c5:
            best_c5 = c5

    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 312
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # === Phase 1: Intensive continuous refinement on parent ===
    if remaining() > 60:
        h_ref, c5_ref = refine_continuous(best_h, best_n, time_budget=min(200, remaining() - 30))
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref
            best_n = best_n

    # === Phase 2: Try nearby n values with continuous optimization ===
    n_candidates = []
    base_n = best_n
    for dn in [-8, -6, -4, -2, 2, 4, 6, 8, 12, 16, 20, 24, 32, 40]:
        nn = base_n + dn
        if 100 <= nn <= 600 and nn % 2 == 0 and nn != base_n:
            n_candidates.append(nn)

    # Sort by potential (smaller n tends to give better C5 per unit corr)
    n_candidates.sort(key=lambda x: abs(x - base_n))

    for n_t in n_candidates:
        if remaining() < 40:
            break
        t_alloc = min(80, remaining() - 10)
        if t_alloc < 20:
            break

        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)

        h_opt, c5_opt = refine_continuous(h_feas, n_t, time_budget=t_alloc)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # === Phase 3: Binary local search at best_n and nearby ===
    if remaining() > 60:
        n_try_list = [best_n]
        for dn in [-4, -2, 2, 4]:
            nn = best_n + dn
            if nn >= 100 and nn % 2 == 0 and nn != best_n:
                n_try_list.append(nn)

        for n_t in n_try_list:
            if remaining() < 20:
                break
            h_bin, c5_bin = binary_local_search(best_h, n_t,
                                                 time_limit=min(40, remaining() - 10),
                                                 rng=rng)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                best_n = n_t
            elif c5_bin < best_c5 + 0.01:
                # Refine binary solution continuously
                h_cont = h_bin + 0.01 * rng.randn(n_t)
                h_cont = project_feasible(h_cont, n_t)
                h_ref, c5_ref = refine_continuous(h_cont, n_t,
                                                   time_budget=min(40, remaining() - 10))
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref.copy()
                    best_n = n_t

    # === Phase 4: Binary simulated annealing for diversity ===
    if remaining() > 40:
        h_sa, c5_sa = binary_sa(best_h, best_n,
                                 time_limit=min(30, remaining() - 10), rng=rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = best_n

    # === Phase 5: Final intensive polish on best solution ===
    if remaining() > 20:
        h_final, c5_final = refine_continuous(best_h, best_n,
                                               time_budget=min(100, remaining() - 10))
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # === Phase 6: Very final SLSQP with high beta ===
    if remaining() > 8:
        h_new = slsqp_optimize(best_h, best_n, 50000, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Final active descent
    if remaining() > 6:
        h_ad, c5_ad = multi_active_descent(best_h, best_n, max_iters=100,
                                            time_limit=min(5, remaining() - 3), n_active=20)
        if c5_ad < best_c5:
            best_c5 = c5_ad
            best_h = h_ad

    # Validate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
