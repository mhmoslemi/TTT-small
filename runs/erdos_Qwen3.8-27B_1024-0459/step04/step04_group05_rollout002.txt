_G_BETA = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)

            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:

    val = (c_max + np.log(np.sum(exps)) / beta) * dx

    return val + penalty


def softmax_grad(x):
    n = _G_N
    dx = 2.0 / n
    beta = _G_BETA
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = corr - c_max
    exps = np.exp(np.clip(beta * shifted, -600, 600))
    s = np.sum(exps)
    probs = exps / s

    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for k in range(n_lags):
        pk = probs[k]

        offset = k - (n - 1)

                dk[:npairs] -= x[aoff:aoff + npairs]
        grad += pk * dk

    grad = grad * dx
    grad += 2e8 * (np.sum(x) - n / 2.0)
    return grad
