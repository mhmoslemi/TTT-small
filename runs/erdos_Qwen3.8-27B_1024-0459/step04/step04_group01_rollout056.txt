def project_feasible(h, n):

    h = np.array(h, dtype=float).copy()

        else:

def corr_grad_per_shift(h, n, k):

        elif not improved:
            # Try a small random perturbation to escape
            rng_local = np.random.RandomState(it * 7919 + 3)
            pert = rng_local.randn(n) * 0.002
            pert -= np.mean(pert)
            h_trial = project_feasible(h + pert, n)
            c_try = compute_c5(h_trial)
            if c_try < best_c5:
                h = h_trial
                best_c5 = c_try
            else:
                if it > 100:
                    break

    return h, best_c5


def grad_rebalance(h, n, max_iters=5000, time_limit=60):
    """Projected gradient with adaptive step, focusing on top shifts."""
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    step_size = 0.01
    momentum = np.zeros(n)
    beta_mom = 0.7

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining() < 10:
            break

        corr = compute_corr(h)
        top_n = min(8, 2 * n - 1)
        top_idx = np.argsort(-corr)[:top_n]
        # Softmax weights
        c_top = corr[top_idx]
        w = np.exp((c_top - c_top[0]) * 100.0)
        w /= w.sum()

        g = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            if wi > 1e-8:
                g += wi * corr_grad_per_shift(h, n, int(ki))
        g -= np.mean(g)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g /= gnorm

        momentum = beta_mom * momentum + g
        mnorm = np.linalg.norm(momentum)
        if mnorm > 0:
            momentum /= mnorm

        h_trial = project_feasible(h - step_size * momentum, n)
        c_new = compute_c5(h_trial)
        if c_new < best_c5:
            best_c5 = c_new
            h = h_trial
            step_size = min(step_size * 1.01, 0.05)
        else:
            step_size *= 0.9
            if step_size < 1e-5:
                break

    return h, best_c5


def binary_local_search(h, n, time_limit=60, seed=42):
    """Binary swap local search focused on worst shifts."""
    rng = np.random.RandomState(seed)
    h_bin = (h > 0.5).astype(float).copy()
    target_ones = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones > target_ones:
        ones_idx = np.where(h_bin == 1)[0]
        remove = rng.choice(ones_idx, n_ones - target_ones, replace=False)
        h_bin[remove] = 0.0
    elif n_ones < target_ones:
        zeros_idx = np.where(h_bin == 0)[0]
        add = rng.choice(zeros_idx, target_ones - n_ones, replace=False)
        h_bin[add] = 1.0

    dx_val = 2.0 / n
    corr = compute_corr(h_bin)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx_val
    best_h = h_bin.copy()
    start = time.time()
    stall = 0

    while time.time() - start < time_limit and remaining() > 5:
        corr = compute_corr(h_bin)
        k_max = int(np.argmax(corr))
        offset = k_max - (n - 1)
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        found = False
        n_try = 300
        for _ in range(n_try):
            if len(ones) == 0 or len(zeros) == 0:
                break
            if offset >= 0:
                candidates = np.array([i for i in ones if i + offset < n and h_bin[i + offset] == 0])
                if len(candidates) > 0 and rng.rand() < 0.8:
                    i = candidates[rng.randint(len(candidates))]
                else:
                    i = ones[rng.randint(len(ones))]
            elif offset < 0:
                candidates = np.array([i for i in ones if i - offset < n and h_bin[i - offset] == 0])
                if len(candidates) > 0 and rng.rand() < 0.8:
                    i = candidates[rng.randint(len(candidates))]
                else:
                    i = ones[rng.randint(len(ones))]
            else:
                i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            new_corr = compute_corr(h_bin)
            new_max = float(np.max(new_corr))
            if new_max < cur_max - 1e-12:
                corr = new_corr
                cur_max = new_max
                best_c5 = cur_max * dx_val
                best_h = h_bin.copy()
                found = True
                break
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0

        if found:
            stall = 0
        else:
            stall += 1
            if stall > 200:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                corr = compute_corr(h_bin)
                cur_max = float(np.max(corr))
                stall = 0

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 312
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP with increasing beta on parent n
    for beta in [500, 2000, 10000, 50000]:
        if remaining() < 15:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Active set equalization
    if remaining() > 30:
        h_eq, c5_eq = active_set_cross(h, n, top_k=3, max_iters=2000, time_limit=80)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 3: More SLSQP
    for beta in [50000, 200000, 500000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Gradient rebalance
    if remaining() > 30:
        h_gr, c5_gr = grad_rebalance(h, n, max_iters=5000, time_limit=40)
        if c5_gr < best_c5:
            best_c5 = c5_gr
            best_h = h_gr.copy()
            h = h_gr.copy()

    # Phase 5: Try higher resolution (key improvement)
    high_n_options = []
    # Double resolution
    high_n_options.append(n * 2)
    # Other near options
    for factor in [3, 4]:
        high_n_options.append(n * factor)
    # Slightly perturbed
    for delta in [4, 8, 16, 32]:
        high_n_options.append(n + delta)
        high_n_options.append(max(128, n - delta))
    high_n_options = sorted(set(nn for nn in high_n_options if 128 <= nn <= 1200 and nn != n))

    for n_t in high_n_options:
        if remaining() < 50:
            break
        # Interpolate best solution
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)

        # Optimize at new resolution
        h_opt = h_feas
        for beta in [2000, 10000, 50000]:
            if remaining() < 15:
                break
            h_opt = slsqp_smooth(h_opt, n_t, beta, maxiter=400)

        # Active set
        if remaining() > 20:
            h_opt, _ = active_set_cross(h_opt, n_t, top_k=3, max_iters=500, time_limit=30)

        # Final SLSQP
        if remaining() > 10:
            h_opt = slsqp_smooth(h_opt, n_t, 100000, maxiter=300)

        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 6: Intensive final refinement on best solution
    n = best_n
    h = best_h.copy()

    # Very high beta SLSQP
    for beta in [200000, 500000, 1000000, 2000000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Active set with top 5
    if remaining() > 20:
        h_eq, c5_eq = active_set_cross(h, n, top_k=5, max_iters=3000, time_limit=50)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Top 2
    if remaining() > 15:
        h_eq, c5_eq = active_set_cross(h, n, top_k=2, max_iters=2000, time_limit=30)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 7: Binary local search (can sometimes beat continuous for specific n)
    if remaining() > 20:
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        target = best_n // 2
        if n_ones != target:
            if n_ones > target:
                idx = np.where(h_bin == 1)[0]
                h_bin[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
            else:
                idx = np.where(h_bin == 0)[0]
                h_bin[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1
        h_b, c5_b = binary_local_search(h_bin, best_n, time_limit=min(30, max(5, remaining() - 15)), seed=seed)
        if c5_b < best_c5:
            # Binary solution is better - use it but also try continuous refinement from it
            best_c5 = c5_b
            best_h = h_b.copy()
            # Continuous refinement from binary
            h_cont = slsqp_smooth(h_b, best_n, 50000, maxiter=300)
            c5_cont = compute_c5(h_cont)
            if c5_cont < best_c5:
                best_c5 = c5_cont
                best_h = h_cont.copy()

    # Phase 8: Final polish
    if remaining() > 8:
        h_new = slsqp_smooth(best_h, best_n, 1000000, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if remaining() > 5:
        h_eq, c5_eq = active_set_cross(best_h, best_n, top_k=3, max_iters=500, time_limit=int(remaining() - 5))
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Validate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
