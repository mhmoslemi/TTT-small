    h = np.zeros(n)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def smooth_random_init(n, rng):
    h = np.zeros(n)
    # Create a few random blocks
    n_blocks = rng.randint(3, 8)
    positions = sorted(rng.choice(n, size=n_blocks, replace=False))
    widths = [rng.randint(5, n // 4) for _ in range(n_blocks)]
    for pos, w in zip(positions, widths):
        h[pos:pos + w] = 1.0
    h = project_feasible(h, n)
    # Add some noise
    h += rng.normal(0, 0.1, n)
    h = project_feasible(h, n)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 256
        best_h = np.full(best_n, 0.5)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)
    
    n = best_n
    h = best_h.copy()
    
    # Phase 1: Quick FISTA polish at parent resolution
    if remaining_time() > 30:
        for beta in [50, 200, 1000]:
            if remaining_time() < 10:
                break
            h_f, c5_f = fista_optimize(h, n, beta, max_iter=500, lr=0.5, time_limit_s=30)
            if c5_f < best_c5 - 1e-15:
                best_c5 = c5_f
                best_h = h_f.copy()
                h = h_f.copy()
    
    # Phase 2: SLSQP smooth at multiple betas
    if remaining_time() > 50:
        for beta in [200, 1000, 5000]:
            if remaining_time() < 15:
                break
            h_s = slsqp_smooth(h, n, beta, maxiter=300)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5 - 1e-15:
                best_c5 = c5_s
                best_h = h_s.copy()
                h = h_s.copy()
    
    # Phase 3: Minimax SLSQP
    if remaining_time() > 60:
        h_mm = slsqp_minimax(h, n, max_rounds=15, max_slsqp_iter=200)
        c5_mm = compute_c5(h_mm)
        if c5_mm < best_c5 - 1e-15:
            best_c5 = c5_mm
            best_h = h_mm.copy()
            h = h_mm.copy()
    
    # Phase 4: Upsample to higher resolutions
    trial_ns = [320, 384, 448, 512, 640]
    for trial_n in trial_ns:
        if remaining_time() < 30:
            break
        h_interp = interp_to(best_h, best_n, trial_n)
        h_interp = project_feasible(h_interp, trial_n)
        
        # FISTA at new resolution
        for beta in [100, 500, 2000]:
            if remaining_time() < 10:
                break
            h_f, c5_f = fista_optimize(h_interp, trial_n, beta, max_iter=400, lr=0.4, time_limit_s=40)
            if c5_f < best_c5 - 1e-15:
                best_c5 = c5_f
                best_h = h_f.copy()
                best_n = trial_n
                h_interp = h_f.copy()
        
        if remaining_time() > 15 and trial_n <= 512:
            h_mm2 = slsqp_minimax(h_interp, trial_n, max_rounds=8, max_slsqp_iter=150)
            c5_mm2 = compute_c5(h_mm2)
            if c5_mm2 < best_c5 - 1e-15:
                best_c5 = c5_mm2
                best_h = h_mm2.copy()
                best_n = trial_n
    
    # Phase 5: Combinatorial search at best resolution
    if remaining_time() > 40:
        rng2 = np.random.RandomState(seed + 1)
        h_comb, c5_comb = combinatorial_search(best_h, best_n, max_sweeps=30, rng=rng2)
        if c5_comb < best_c5 - 1e-15:
            best_c5 = c5_comb
            best_h = h_comb.copy()
    
    # Phase 6: Multi-start exploration
    for trial_n in [best_n, best_n + 16 if best_n < 400 else best_n - 16, 512]:
        if remaining_time() < 40:
            break
        if trial_n < 100:
            continue
        for start_type in ['binary', 'smooth']:
            if remaining_time() < 15:
                break
            if start_type == 'binary':
                h_start = random_binary_init(trial_n, rng)
            else:
                h_start = smooth_random_init(trial_n, rng)
            
            c5_start = compute_c5(h_start)
            if c5_start > 0.45:
                continue
            
            for beta in [200, 1000]:
                if remaining_time() < 8:
                    break
                h_f, c5_f = fista_optimize(h_start, trial_n, beta, max_iter=300, lr=0.5, time_limit_s=25)
                if c5_f < best_c5 - 1e-15:
                    best_c5 = c5_f
                    best_h = h_f.copy()
                    best_n = trial_n
    
    # Phase 7: Final SLSQP polish
    if remaining_time() > 20:
        for beta in [5000, 20000]:
            if remaining_time() < 8:
                break
            h_p = slsqp_smooth(best_h, best_n, beta, maxiter=200)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5 - 1e-15:
                best_c5 = c5_p
                best_h = h_p.copy()
    
    # Phase 8: Final minimax
    if remaining_time() > 15:
        h_mm3 = slsqp_minimax(best_h, best_n, max_rounds=10, max_slsqp_iter=150)
        c5_mm3 = compute_c5(h_mm3)
        if c5_mm3 < best_c5 - 1e-15:
            best_c5 = c5_mm3
            best_h = h_mm3.copy()
    
    # Phase 9: Final combinatorial
    if remaining_time() > 10:
        rng3 = np.random.RandomState(seed + 2)
        h_comb2, c5_comb2 = combinatorial_search(best_h, best_n, max_sweeps=15, rng=rng3)
        if c5_comb2 < best_c5 - 1e-15:
            best_c5 = c5_comb2
            best_h = h_comb2.copy()
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
