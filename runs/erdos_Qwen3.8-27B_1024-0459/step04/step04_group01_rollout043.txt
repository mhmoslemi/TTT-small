```python
def compute_swap_delta(h, i, j, n):
    d = np.arange(2*n-1) - (n-1)  # lags: -(n-1), ..., n-1
    # i+d: positions i+d_k for each k
    ip = i + d  # array of length 2n-1
    im = i - d
    jp = j + d
    jm = j - d
    
    delta = np.zeros(2*n-1)
    
    # Term 1: -(1-h[i+d]) for valid i+d
    mask_ip = (ip >= 0) & (ip < n)
    delta[mask_ip] -= (1 - h[ip[mask_ip]])
    
    # Term 2: +h[i-d] for valid i-d
    mask_im = (im >= 0) & (im < n)
    delta[mask_im] += h[im[mask_im]]
    
    # Term 3: +(1-h[j+d]) for valid j+d
    mask_jp = (jp >= 0) & (jp < n)
    delta[mask_jp] += (1 - h[jp[mask_jp]])
    
    # Term 4: -h[j-d] for valid j-d
    mask_jm = (jm >= 0) & (jm < n)
    delta[mask_jm] -= h[jm[mask_jm]]
    
    return delta
```

```python
        # Targeted: find swaps that reduce the worst lag
        if worst_d > 0:
            cand_i = [i for i in ones if i + worst_d < n and h[i + worst_d] == 0]
        elif worst_d < 0:
            da = -worst_d
            cand_i = [i for i in ones if i - da >= 0 and h[i - da] == 0]
        else:
            cand_i = []
        
        # Try targeted swaps (biased toward fixing worst lag)
        n_target = min(len(cand_i), 80)
        if n_target > 0:
            if n_target < len(cand_i):
                chosen_i = rng.choice(cand_i, n_target, replace=False)
            else:
                chosen_i = np.array(cand_i)
            
            for i in chosen_i:
                i = int(i)
                # Choose j: prefer the position that would create "11" at worst lag
                if worst_d > 0 and i + worst_d < n and h[i + worst_d] == 0:
                    j = i + worst_d
                elif worst_d < 0 and i + worst_d >= 0 and h[i + worst_d] == 0:
                    j = i + worst_d
                else:
                    j = int(zeros[rng.randint(len(zeros))])
                
                if h[j] == 1:
                    continue
                
                delta = swap_delta(h, i, j, n)
                new_max = float(np.max(corr + delta))
                if new_max < cur_max - 1e-15:
                    h[i] = 0.0
                    h[j] = 1.0
                    corr = corr + delta
                    cur_max = new_max
                    improved = True
                    stall = 0
                    break
        
        # Try random swaps
        if not improved:
            n_rand = 60
            for _ in range(n_rand):
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                delta = swap_delta(h, i, j, n)
                new_max = float(np.max(corr + delta))
                if new_max < cur_max - 1e-15:
                    h[i] = 0.0
                    h[j] = 1.0
                    corr = corr + delta
                    cur_max = new_max
                    improved = True
                    stall = 0
                    break
        
        if improved:
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            stall += 1
            # Perturb after long stall
            if stall % 500 == 0:
                n_perturb = rng.randint(2, 6)
                for _ in range(n_perturb):
                    i = int(ones[rng.randint(len(ones))])
                    j = int(zeros[rng.randint(len(zeros))])
                    h[i] = 0.0
                    h[j] = 1.0
                corr = np.correlate(h, 1.0 - h, mode='full')
                cur_max = float(np.max(corr))
    
    return best_h, best_max * dx


def simulated_annealing(h, n, time_limit=60, rng=None, T_init=0.5, T_min=0.01):
    """Simulated annealing for binary h."""
    if rng is None:
        rng = np.random.RandomState(42)
    
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()
    dx = 2.0 / n
    
    start = time.time()
    elapsed_total = time_limit
    
    while time.time() - start < time_limit:
        if remaining() < 5:
            break
        
        elapsed = time.time() - start
        T = T_init * (T_min / T_init) ** (elapsed / elapsed_total)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        
        delta = swap_delta(h, i, j, n)
        new_max = float(np.max(corr + delta))
        diff = new_max - cur_max
        
        if diff < 0 or rng.rand() < np.exp(-diff / max(T, 1e-10)):
            h[i] = 0.0
            h[j] = 1.0
            corr = corr + delta
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
    
    return best_h, best_max * dx


def greedy_init(n, rng):
    """Greedy construction: add ones one at a time to minimize max corr."""
    h = np.zeros(n)
    # Start with n/2 ones placed greedily
    target = n // 2
    used = np.zeros(n, dtype=bool)
    
    # Initialize with a random spread
    positions = rng.permutation(n)
    for p in positions[:target]:
        h[p] = 1.0
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(corr))
    
    # Local search to improve
    for _ in range(2000):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_improve = 0
        best_swap = None
        for _ in range(30):
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            delta = swap_delta(h, i, j, n)
            new_max = float(np.max(corr + delta))
            improve = cur_max - new_max
            if improve > best_improve:
                best_improve = improve
                best_swap = (i, j, delta)
        if best_swap is None:
            break
        i, j, delta = best_swap
        h[i] = 0.0
        h[j] = 1.0
        corr = corr + delta
        cur_max -= best_improve
    
    return h, cur_max * (2.0 / n)


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
    except Exception:
        parent_n = 312
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
    
    # Phase 1: Intensive binary local search at parent resolution
    n = parent_n
    h_bin = project_to_binary(parent_h, n, rng)
    c5_init = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = min(compute_c5(parent_h), c5_init)
    best_n = n
    
    if best_c5 > c5_init:
        best_c5 = c5_init
        best_h = h_bin.copy()
    
    # Run local search with time allocation
    t1 = min(300, max(30, remaining() * 0.35))
    h_ls, c5_ls = local_search(h_bin, n, time_limit=t1, rng=rng)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls.copy()
        best_n = n
    
    # Run SA from the locally searched solution
    if remaining() > 30:
        t2 = min(120, max(15, remaining() * 0.2))
        h_sa, c5_sa = simulated_annealing(h_ls, n, time_limit=t2, rng=rng, T_init=0.3, T_min=0.01)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n
        # Restart SA from best
        if remaining() > 20:
            t3 = min(80, max(10, remaining() * 0.15))
            h_sa2, c5_sa2 = simulated_annealing(best_h, n, time_limit=t3, rng=rng, T_init=0.2, T_min=0.005)
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2.copy()
                best_n = n
    
    # More local search from best
    if remaining() > 20:
        t4 = min(200, max(10, remaining() * 0.35))
        h_ls2, c5_ls2 = local_search(best_h, n, time_limit=t4, rng=rng)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()
            best_n = n
    
    # Phase 2: Try other n values
    n_tries = [320, 336, 384, 448, 512, 624]
    for n_t in n_tries:
        if remaining() < 40:
            break
        # Interpolate best solution to new n
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = project_to_binary(h_interp, n_t, rng)
        c5_t = compute_c5(h_bin_t)
        
        # Local search at new n
        t_limit = min(80, max(10, remaining() * 0.25))
        h_opt, c5_opt = local_search(h_bin_t, n_t, time_limit=t_limit, rng=rng)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
        
        # Also try a fresh random init at this n
        if remaining() > 30 and n_t != 312:
            h_rand = np.zeros(n_t)
            pos = rng.permutation(n_t)
            h_rand[pos[:n_t // 2]] = 1.0
            t_limit2 = min(50, max(10, remaining() * 0.2))
            h_opt2, c5_opt2 = local_search(h_rand, n_t, time_limit=t_limit2, rng=rng)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
                best_n = n_t
    
    # Phase 3: Final intensive pass on the best solution
    n = best_n
    h = best_h.copy()
    if remaining() > 15:
        t5 = min(150, max(5, remaining() * 0.6))
        h_final, c5_final = local_search(h, n, time_limit=t5, rng=rng)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
            best_n = n
    
    # Phase 4: Continuous refinement (SLSQP on the best binary solution)
    if remaining() > 20 and best_n <= 400:
        try:
            from scipy.optimize import minimize
            
            n_c = best_n
            h_c = best_h.astype(float).copy()
            tgt = n_c / 2.0
            
            def _obj(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                m = np.max(corr)
                beta = 5000.0
                shifted = corr - m
                cl = np.clip(beta * shifted, -50, 50)
                return (m + np.log(np.sum(np.exp(cl))) / beta) * (2.0 / n_c)
            
            def _grad(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                m = np.max(corr)
                beta = 5000.0
                shifted = corr - m
                cl = np.clip(beta * shifted, -50, 50)
                ex = np.exp(cl)
                w = ex / np.sum(ex)
                g = np.zeros(n_c)
                d = np.arange(2 * n_c - 1) - (n_c - 1)
                for k_idx in range(2 * n_c - 1):
                    if w[k_idx] < 1e-14:
                        continue
                    dd = d[k_idx]
                    ip = np.arange(n_c) + dd
                    im = np.arange(n_c) - dd
                    mask_p = (ip >= 0) & (ip < n_c)
                    mask_m = (im >= 0) & (im < n_c)
                    g[mask_p] += w[k_idx] * (1.0 - x[ip[mask_p]])
                    g[mask_m] += w[k_idx] * (-x[im[mask_m]])
                return g * (2.0 / n_c)
            
            bounds = [(0.0, 1.0)] * n_c
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt,
                     'jac': lambda x: np.ones(n_c)}]
            
            res = minimize(_obj, h_c, jac=_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-16, 'disp': 0})
            
            h_res = np.clip(res.x, 0, 1)
            # Fix sum constraint
            diff = tgt - np.sum(h_res)
            if abs(diff) > 1e-10:
                mask = (h_res > 1e-8) & (h_res < 1.0 - 1e-8)
                if np.sum(mask) > 0:
                    h_res[mask] += diff / np.sum(mask)
                else:
                    h_res += diff / n_c
                h_res = np.clip(h_res, 0, 1)
            
            c5_cont = compute_c5(h_res)
            if c5_cont < best_c5:
                best_c5 = c5_cont
                best_h = h_res.copy()
                best_n = n_c
        except Exception:
            pass
    
    # Final validation
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
```
