            # Choose partner j (prefer one with value near 0.5, far from i)
            # For speed, just pick j = (i + n//2) % n or random
            j = rng.randint(n)
            while j == i:
                j = rng.randint(n)
            
            s = h_best[i] + h_best[j]
            if s < 1e-10 or s > 2 - 1e-10:

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_TGT = 0.0
_G_DX = 1.0
_G_START = 0.0
_G_BUDGET = 1300.0
_G_BETA = 1.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            adj = diff / cnt
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
        else:
            h[:] = np.clip(h[:] + diff / n, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx[0]] += diff
            h[idx[0]] = np.clip(h[idx[0]], 0.0, 1.0)
    return h


def remaining_time():

def subgrad_compute(h, n, top_k=5):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    top_indices = np.argsort(corr)[-top_k:]
    top_indices = top_indices[np.argsort(corr[top_indices])[::-1]]
    grad = np.zeros(n)
    for k_idx in top_indices:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            n_valid = n - offset
            if n_valid <= 0:
                continue
            grad[offset:offset + n_valid] += (1.0 - h[:n_valid])
            grad[:n_valid] -= h[offset:offset + n_valid]
        else:
            m = -offset
            n_valid = n - m
            if n_valid <= 0:
                continue
            grad[:n_valid] += (1.0 - h[m:m + n_valid])
            grad[m:m + n_valid] -= h[:n_valid]
    return grad, top_indices, corr


def subgrad_momentum(h, n, max_iter, step0, deadline):
    c5_best = compute_c5(h)
    h_best = h.copy()
    v = np.zeros(n)
    step = step0
    patience = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad, top_ids, corr = subgrad_compute(h, n, top_k=5)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad = grad / gn
        alpha = step
        improved = False
        for _ in range(14):
            v = 0.85 * v - alpha * grad
            h_trial = project_feasible(h + v, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-13:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
            alpha *= 0.5
        if improved:
            patience = 0
            step = min(alpha * 1.3, 0.4)
        else:
            patience += 1
            step *= 0.6
            v *= 0.4
            if patience > 18:
                break
    return h_best, c5_best


def obj_smooth(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(_G_BETA * shifted, -600, 600))
    return (m + np.log(np.sum(exps)) / _G_BETA) * dx


def con_sum(x):
    return np.sum(x) - _G_TGT


def slsqp_polish(h, n, betas, deadline):
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum}]
    for beta_val in betas:
        if time.time() > deadline:
            break
        _G_BETA = float(beta_val)
        try:
            res = minimize(obj_smooth, h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-14, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            c5_cur = compute_c5(h)
            if c5_new < c5_cur - 1e-13:
                h = h_new
        except Exception:
            pass
    return h


def coordinate_descent(h, n, max_sweeps, rng, deadline):
    c5_best = compute_c5(h)
    h_best = h.copy()
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline:
                return h_best, c5_best
            # Try multiple partners
            for trial_partner in range(3):
                if trial_partner == 0:
                    j = (i + n // 2) % n
                elif trial_partner == 1:
                    j = (i + n // 3) % n
                else:
                    j = rng.randint(n)
                if j == i:
                    continue
                s = h_best[i] + h_best[j]
                if s < 1e-12 or s > 2.0 - 1e-12:
                    continue
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                if hi - lo < 1e-6:
                    continue
                # Candidate values including current and boundary
                old_i = h_best[i]
                candidates = [old_i]
                # Add perturbation points
                for d in [0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.15, -0.15]:
                    cv = np.clip(old_i + d, lo, hi)
                    candidates.append(cv)
                # Add golden section and midpoint
                candidates.append((lo + hi) * 0.5)
                gr = (1.0 + 5.0 ** 0.5) / 2.0
                candidates.append(lo + (hi - lo) / gr)
                candidates.append(hi - (hi - lo) / gr)
                candidates.append(lo)
                candidates.append(hi)
                best_t = old_i
                best_c5_local = c5_best
                for t in candidates:
                    t = float(np.clip(t, lo, hi))
                    if abs(t - old_i) < 1e-10:
                        continue
                    h_trial = h_best.copy()
                    h_trial[i] = t
                    h_trial[j] = s - t
                    c5_t = compute_c5(h_trial)
                    if c5_t < best_c5_local - 1e-13:
                        best_c5_local = c5_t
                        best_t = t
                if best_t != old_i:
                    h_best[i] = best_t
                    h_best[j] = s - best_t
                    c5_best = best_c5_local
                    any_imp = True
                    break  # Move to next i
        if not any_imp:
            break
    return h_best, c5_best


def sa_binary(h, n, max_iters, rng, deadline):
    # Simulated annealing on the binary projection
    c5_cur = compute_c5(h)
    h_bin = (h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, n)
    c5_bin = compute_c5(h_bin)
    h_best = h.copy()
    c5_best = c5_cur
    T = 0.02
    T_min = 0.001
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = max(T_min, T * 0.997)
        # Pick two indices to swap (one 0→1, one 1→0)
        zeros = np.where(h_bin < 0.5)[0]
        ones = np.where(h_bin > 0.5)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h_bin[i] = 1.0
        h_bin[j] = 0.0
        c5_try = compute_c5(h_bin)
        delta = c5_try - c5_bin
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            c5_bin = c5_try
            if c5_try < c5_best - 1e-13:
                c5_best = c5_try
                h_best = h_bin.copy()
        else:
            h_bin[i] = 0.0
            h_bin[j] = 1.0
    # After SA, do continuous optimization starting from best binary
    h_cont = h_best.copy()
    h_cont, c5_cont = subgrad_momentum(h_cont, n, max_iter=100, step0=0.05,
                                        deadline=deadline)
    if c5_cont < c5_best:
        c5_best = c5_cont
        h_best = h_cont
    return h_best, c5_best


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h


def make_golden(n):
    h = np.zeros(n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * phi) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return h
