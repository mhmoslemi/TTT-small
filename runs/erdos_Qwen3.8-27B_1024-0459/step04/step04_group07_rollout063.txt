def objective_and_grad(h, beta, dx, pen=0.0, target_sum=None):
    n = len(h)
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')
    nk = 2 * n - 1
    
    # Soft-max value
    scaled = beta * dx * corr
    max_s = scaled.max()
    exp_s = np.exp(scaled - max_s)
    sum_exp = exp_s.sum()
    soft_val = (max_s + np.log(sum_exp)) / beta * dx
    
    # Soft-max probabilities
    p = exp_s / sum_exp
    
    # Gradient
    grad = np.zeros(n)
    for m in range(n):
        # term1: sum_j (1-h[j]) * p[j+n-1-m]
        p_idx1 = np.arange(n) + (n - 1 - m)
        term1 = np.dot(1.0 - h, p[p_idx1])
        
        # term2: -sum_i h[i] * p[m-i+n-1]
        p_idx2 = (m + n - 1) - np.arange(n)
        term2 = -np.dot(h, p[p_idx2])
        
        grad[m] = dx * (term1 + term2)
    
    val = soft_val
    if pen > 0 and target_sum is not None:
        s = h.sum() - target_sum
        val += pen * s * s
        grad += 2 * pen * s * np.ones(n)
    
    return val, grad

Wait, I need to be careful with the soft-max computation. The soft-max of {beta*dx*corr_k} is:
(1/beta) * log(sum_k exp(beta * dx * corr_k))

For numerical stability:
= (1/beta) * (beta*dx*max(corr) + log(sum_k exp(beta*dx*(corr_k - max(corr)))))
= dx*max(corr) + (1/beta)*log(sum_k exp(beta*dx*(corr_k - max(corr))))

The second term goes to 0 as beta → ∞, so the soft-max → dx*max(corr).

For the gradient:
d/dh[m] [(1/beta) * log(sum_k exp(beta*dx*corr_k))]
= (1/beta) * (1/sum_k exp(...)) * sum_k exp(...) * beta*dx * d(corr_k)/dh[m]
= dx * sum_k p_k * d(corr_k)/dh[m]
where p_k = exp(beta*dx*corr_k) / sum_j exp(beta*dx*corr_j)

This is correct.

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-15
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def softobj_grad(h, beta, dx, pen=0.0):
    n = len(h)
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')
    nk = 2 * n - 1
    
    scaled = beta * dx * corr
    max_s = scaled.max()
    exp_s = np.exp(scaled - max_s)
    sum_exp = exp_s.sum()
    soft_val = (max_s + np.log(sum_exp)) / (beta * dx) * dx  # = (max_s + log(sum_exp)) / beta
    # Actually: (1/beta)*log(sum exp(beta*dx*corr_k))
    # = (1/beta)*(beta*dx*max(corr) + log(sum exp(beta*dx*(corr-max(corr)))))
    # = dx*max(corr) + (1/beta)*log(sum exp(beta*dx*(corr-max(corr))))
    # The exp_s here is exp(scaled - max_s) = exp(beta*dx*(corr - max(corr)))
    # So soft_val = dx*max(corr) + log(sum_exp)/beta
    # But for the gradient, I just need p_k = exp_s/sum_exp

        term1 = np.dot(1.0 - h, p[p_idx1])
        p_idx2 = (m + n - 1) - np.arange(n)
        term2 = -np.dot(h, p[p_idx2])
        grad[m] = dx * (term1 + term2)
    
    val = soft_val
    if pen > 0:
        s = h.sum() - n / 2.0
        val += pen * s * s
        grad = grad + 2.0 * pen * s
    
    return val, grad

def lbfgsb_optimize(h0, n, beta, maxiter, pen=1000.0):
    dx = 2.0 / n
    
    def fun(x):
        v, _ = softobj_grad(x, beta, dx, pen)
        return v
    
    def jac(x):
        _, g = softobj_grad(x, beta, dx, pen)
        return g
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(fun, h0, jac=jac, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10, 'iprint': 0})
    return project_feasible(res.x, n)

def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1')
        h[i] = 1.0 if c % 2 == 0 else 0.0
    return h

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def slsqp_polish(h0, n, beta, maxiter=300):
