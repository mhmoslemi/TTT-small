def compute_c5(h):

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)

def _obj(x):

    sm = m + np.log(np.sum(ex)) / _BETA

def _grad(x):

def thue_morse(n, seed=0):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        rng = np.random.RandomState(seed)
        if n_ones > n_half:
            ones = np.where(h == 1)[0]
            perm = rng.permutation(len(ones))
            for idx in perm[:n_ones - n_half]:
                h[ones[idx]] = 0
        else:
            zeros = np.where(h == 0)[0]
            perm = rng.permutation(len(zeros))
            for idx in perm[:n_half - n_ones]:
                h[zeros[idx]] = 1
    return h
