    while time.time() - start_t < time_limit:
        if _T0 > 0 and remaining_time() < 5:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:8]
        max_corr = corr[top_idx[0]]

        # Find swaps that reduce the top correlations
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        improved = False
        for _ in range(200):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            nc = compute_corr(h)
            nv = float(np.max(nc))
            h[i] = 1.0
            h[j] = 0.0
            if nv * dx < best_val - 1e-14:
                h[i] = 0.0
                h[j] = 1.0
                best_val = nv * dx
                improved = True
                break

        if not improved:
            # Try fractional adjustment of worst elements
            h_frac = h.copy()
            alpha = 0.05
            for k in top_idx[:3]:
                off = k - (n - 1)
                if off >= 0:
                    np_ = n - off
                    # Reduce h where h[i+off]=1, h[i]=0
                    for pos in range(np_):
                        if h_frac[pos + off] > 0.5 and h_frac[pos] < 0.5:
                            h_frac[pos + off] -= alpha
                            h_frac[pos] += alpha
                            break
                else:
                    ao = -off
                    np_ = n - ao
                    for pos in range(np_):
                        if h_frac[pos] > 0.5 and h_frac[pos + ao] < 0.5:
                            h_frac[pos] -= alpha
                            h_frac[pos + ao] += alpha
                            break
            h_frac = project_feasible(h_frac, n)
            cv = compute_c5(h_frac)
            if cv < best_val - 1e-14:
                h = h_frac
                best_val = cv
                improved = True

        if not improved:
            break

    return h, best_val


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N_G, _BETA_G, _TGT_G
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = make_random_balanced(parent_n, rng)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n

    # Phase 1: SLSQP on parent resolution with progressive beta
    h = best_h.copy()
    for beta in [100, 500, 2000, 8000, 20000]:
        if remaining_time() < 15:
            break
        h_new = slsqp_optimize(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Binary local search on the best solution
    if remaining_time() > 30:
        h_bin = (h > 0.5).astype(float)
        h_bl, c5_bl = binary_local_search(h_bin, n, rng, time_limit=min(60, int(remaining_time() * 0.2)),
                                          tabu_size=100, T_init=0.5, T_final=0.01, K=400)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()

        # SLSQP from binary result
        if remaining_time() > 15:
            h_new = slsqp_optimize(h_bl, n, 10000, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 3: Multi-start with structured initializations
    starts = [
        make_thue_morse(n, rng),
        make_periodic(n, 4, 2, rng),
        make_periodic(n, 6, 3, rng),
        make_periodic(n, 3, 1, rng),
        make_random_balanced(n, rng),
    ]

    for si, s in enumerate(starts):
        if remaining_time() < 20:
            break
        # Quick SLSQP
        h_s = slsqp_optimize(s, n, 2000, maxiter=200)
        c5_s = compute_c5(h_s)
        # Binary local search
        h_s_bin = (h_s > 0.5).astype(float)
        tl = min(30, int(remaining_time() * 0.1))
        h_sb, c5_sb = binary_local_search(h_s_bin, n, rng, time_limit=tl,
                                          tabu_size=60, T_init=0.3, T_final=0.005, K=250)
        # SLSQP from binary
        h_sf = slsqp_optimize(h_sb, n, 8000, maxiter=200)
        c5_sf = compute_c5(h_sf)

        if c5_sf < best_c5:
            best_c5 = c5_sf
            best_h = h_sf.copy()
        if c5_sb < best_c5:
            best_c5 = c5_sb
            best_h = h_sb.copy()

    # Phase 4: Targeted refinement
    if remaining_time() > 20:
        h_t, c5_t = targeted_refine(best_h, n, rng, time_limit=min(40, int(remaining_time() * 0.2)))
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
        # SLSQP from targeted result
        if remaining_time() > 10:
            h_new = slsqp_optimize(h_t, n, 15000, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 5: Multi-resolution exploration
    n_candidates = []
    for factor in [0.75, 0.85, 1.0, 1.15, 1.3, 1.5, 1.75, 2.0]:
        nn = int(n * factor)
        nn = (nn // 2) * 2  # ensure even
        if 128 <= nn <= 1200:
            n_candidates.append(nn)
    n_candidates = sorted(set(n_candidates))
    n_candidates = [nn for nn in n_candidates if nn != n]
    n_candidates.sort(key=lambda x: abs(x - n))

    for n_t in n_candidates[:5]:
        if remaining_time() < 25:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)

        # SLSQP at new resolution
        h_opt = slsqp_optimize(h_feas, n_t, 3000, maxiter=300)
        c5_opt = compute_c5(h_opt)

        # Binary local search
        if remaining_time() > 15:
            h_bin = (h_opt > 0.5).astype(float)
            tl = min(25, int(remaining_time() * 0.25))
            h_bl, c5_bl = binary_local_search(h_bin, n_t, rng, time_limit=tl,
                                              tabu_size=80, T_init=0.4, T_final=0.01, K=300)
            # SLSQP from binary
            h_sf = slsqp_optimize(h_bl, n_t, 10000, maxiter=200)
            c5_sf = compute_c5(h_sf)

            if c5_sf < best_c5:
                best_c5 = c5_sf
                best_h = h_sf.copy()
                best_n = n_t
            if c5_bl < best_c5:
                best_c5 = c5_bl
                best_h = h_bl.copy()
                best_n = n_t
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_t

    # Phase 6: Final intensive refinement
    n = best_n
    h = best_h.copy()

    for beta in [8000, 20000, 40000, 80000]:
        if remaining_time() < 8:
            break
        h_new = slsqp_optimize(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final binary search
    if remaining_time() > 20:
        h_bin = (h > 0.5).astype(float)
        tl = min(60, int(remaining_time() * 0.3))
        h_bl, c5_bl = binary_local_search(h_bin, n, rng, time_limit=tl,
                                          tabu_size=120, T_init=0.3, T_final=0.003, K=500)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()
        # SLSQP polish
        if remaining_time() > 10:
            h_new = slsqp_optimize(h_bl, n, 50000, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Final targeted refine
    if remaining_time() > 15:
        h_t, c5_t = targeted_refine(best_h, n, rng, time_limit=min(30, int(remaining_time() * 0.4)))
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
        if remaining_time() > 8:
            h_new = slsqp_optimize(h_t, n, 100000, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    else:
        # Keep the better value but ensure feasibility
        best_h = project_feasible(best_h, best_n)

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
