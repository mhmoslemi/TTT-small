    nc = 2 * n - 1
    b = np.zeros(nc)
    for k in range(nc):

    """Solve min_t max_k (a[k] + t*b[k]) for t in [lo, hi]."""
    if lo >= hi:
        return (lo + hi) / 2.0
    for _ in range(60):

        vals = a + mid * b
        max_val = vals.max()
        active = vals >= max_val - 1e-15
        if active.sum() == 0:
            break
        b_active = b[active]
        if b_active.max() < -1e-15:
            # all decreasing, increase t
            lo = mid
        elif b_active.min() > 1e-15:
            # all increasing, decrease t
            hi = mid
        else:
            # 0 in subgradient
            break
    return (lo + hi) / 2.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        ht = np.clip(h - mid, 0.0, 1.0)
        if np.sum(ht) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        if npairs > 0:
            g[off:off + npairs] += (1.0 - x[:npairs])
            g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        if npairs > 0:
            g[:npairs] += (1.0 - x[ao:ao + npairs])
            g[ao:ao + npairs] -= x[:npairs]
    return g

def slsqp_smooth(h, n, beta, maxiter=500):

    _N = n
    _TGT = n / 2.0

def compute_B(h, n):
    """Compute the (2n-1) x n derivative matrix B where B[k,i] = d corr[k]/d h[i]."""
    nc = 2 * n - 1
    B = np.zeros((nc, n))
    for d in range(n):
        k = n - 1 + d
        if d == 0:
            B[k, :] = (1.0 - h) - h
        else:
            B[k, d:] += (1.0 - h[:n - d])
            B[k, :n - d] -= h[d:n]
    for m in range(1, n):
        k = n - 1 - m
        B[k, :n - m] += (1.0 - h[m:n])
        B[k, m:] -= h[:n - m]
    return B


def solve_1d_max(b, a, lo, hi):
    """Solve min_t max_k (a[k] + t*b[k]) for t in [lo, hi] via bisection."""
    if lo >= hi:
        return (lo + hi) / 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        vals = a + mid * b
        max_val = vals.max()
        active = vals >= max_val - 1e-14
        if not active.any():
            break
        ba = b[active]
        if ba.max() < -1e-15:
            lo = mid
        elif ba.min() > 1e-15:
            hi = mid
        else:
            break
    return (lo + hi) / 2.0

def active_set_reduce(h, n, max_iters=200, top_k=3):
    """Reduce max by pushing against the top-k active shifts."""
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        top = np.argsort(-corr)[:top_k]
        c_max = corr[top[0]]
        if c_max < 1e-10:
            break
        # Direction: reduce the max shifts
        g_max = corr_grad_per_shift(h, n, int(top[0]))
        d = -g_max
        for k in top[1:]:
            d -= 0.5 * corr_grad_per_shift(h, n, int(k))
        # Project direction to maintain sum
        d -= np.mean(d)
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-14:
            break
        d /= dnorm
        improved = False

            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
