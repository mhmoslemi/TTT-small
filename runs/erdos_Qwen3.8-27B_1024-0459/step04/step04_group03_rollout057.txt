def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):

def remaining_time():

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

def lse_gradient(h, beta, dx):
    """Gradient of dx * LSE(beta * corr) / beta w.r.t. h"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exps = np.exp(beta * (corr - max_c))
    s = np.sum(exps)
    P = exps / s  # softmax weights
    
    # Gradient: dx * sum_k P[k] * dcorr[k]/dh[j]
    # dcorr[k]/dh[j] = (1-h[j+m])*1_{0<=j+m<n} - h[j-m]*1_{0<=j-m<n} where m=k-(n-1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if P[k] < 1e-15:
            continue
        m = k - (n - 1)
        w = P[k]
        # term1: j+m valid
        js = max(0, -m)
        je = min(n, n - m)
        if je > js:
            idx = np.arange(js, je)
            grad[idx] += w * (1.0 - h[idx + m])
        # term2: j-m valid
        js2 = max(0, m)
        je2 = min(n, n + m)
        if je2 > js2:
            idx = np.arange(js2, je2)
            grad[idx] -= w * h[idx - m]
    
    return dx * grad

def lse_objective(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    return float(dx / beta * np.log(np.sum(np.exp(beta * (corr - max_c)))) + dx * max_c)

def lbfgsb_optimize(h0, n, beta, dx, maxiter=500, tol=1e-12):

    def grad(x):

    # Let's just use penalty
    
    def obj_pen(x):

def structured_binary_search(n, rng):
    """Try structured 0-1 patterns."""
    target = n // 2
    
    candidates = []
    
    # Alternating blocks of various sizes
    for block in [1, 2, 3, 4, 5, 6, 8]:
        h = np.zeros(n)
        for i in range(0, n, 2 * block):
            h[i:i+block] = 1.0
        candidates.append(h)
    
    # Symmetric patterns
    for frac in np.arange(0.2, 0.5, 0.05):
        h = np.zeros(n)
        size = int(frac * n)
        h[:size] = 1.0
        h[n-size:] = 1.0
        if np.sum(h) > target:
            h[n-size-1:n-size+2] = 0.0  # adjust
        # Better: just take first target elements
        h = np.zeros(n)
        idx = rng.choice(n, size=target, replace=False)
        h[idx] = 1.0
        candidates.append(h)
    
    # Golden ratio spacing
    for gamma in [0.618, 0.7, 0.382]:
        h = np.zeros(n)
        for i in range(target):
            pos = int((i * gamma * n) % n)
            h[pos] = 1.0
        if np.sum(h) != target:
            # Fix
            h = np.zeros(n)
            used = set()
            for i in range(target):
                pos = int((i * gamma * n) % n)
                while pos in used:
                    pos = (pos + 1) % n
                h[pos] = 1.0
                used.add(pos)
        candidates.append(h)
    
    # Best candidate
    best_h = None
    best_c5 = float('inf')

        if np.sum(h) == target:
            c5 = compute_c5(h)

        h = np.zeros(n)
        h[:target] = 1.0
        best_h = h
        best_c5 = compute_c5(h)

def project_feasible(h, n):

def lse_grad(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exps = np.exp(beta * (corr - max_c))
    s = np.sum(exps)
    P = exps / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if P[k] < 1e-16:
            continue
        m = k - (n - 1)
        w = P[k]
        js = max(0, -m)
        je = min(n, n - m)
        if je > js:
            idx = np.arange(js, je)
            grad[idx] += w * (1.0 - h[idx + m])
        js2 = max(0, m)
        je2 = min(n, n + m)
        if je2 > js2:
            idx = np.arange(js2, je2)
            grad[idx] -= w * h[idx - m]
    return dx * grad

def lse_val(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    return float(dx / beta * (max_c * beta + np.log(np.sum(np.exp(beta * (corr - max_c))))))

    target = n / 2.0

        h = np.clip(x, 0.0, 1.0)
        base = lse_val(h, beta, dx)
        pen = pen_coeff * (np.sum(h) - target) ** 2
        return base + pen
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        g = lse_grad(h, beta, dx)
        g += 2.0 * pen_coeff * (np.sum(h) - target)
        g = g * ((h > 0) & (h < 1))
        return g
    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'iprint': 0})

                best_h = h_new.copy()
    # Phase 9: Ultimate coord descent
    if remaining_time() > 10:
        h_cd2, c5_cd2 = coord_descent(best_h, best_n, max_sweeps=3, window=60)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
    # Final
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    return (best_h, final_c5, best_n)
