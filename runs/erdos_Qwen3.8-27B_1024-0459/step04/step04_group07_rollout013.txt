The cross-correlation with complement: c(k) = Σ_{i=0}^{N-1-k} t(i+k)(1-t(i)) = Σ t(i+k) - Σ t(i+k)t(i)

For Thue-Morse of length N, Σ t(i) = N/2 (by symmetry).
And Σ t(i+k)t(i) is related to the autocorrelation.

c(k) = (N-k)/2 - Σ t(i+k)t(i) for 0 ≤ k < N
= (N-k)/2 - Σ t(i+k)t(i)

For the Thue-Morse, Σ t(i+k)t(i) = (N-k)/4 + R(k)/4 where R(k) is the (autocorrelation-related) term.

So c(k) = (N-k)/2 - (N-k)/4 - R(k)/4 = (N-k)/4 - R(k)/4

For N=512, k=0: c(0) = 0 (binary, t(i)(1-t(i))=0)
For small k: c(k) ≈ N/4 = 128 (minus corrections)

So the max correlation would be around 128 for random-looking sequences, but Thue-Morse has cancellations in R(k) that reduce this.

def compute_c5(h):

def compute_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h > 0.5).astype(float)
    target = n // 2
    cnt = int(np.sum(b))
    if cnt > target:
        ones = np.where(b == 1.0)[0]
        order = np.argsort(h[ones])
        for i in range(cnt - target):
            b[ones[order[i]]] = 0.0
    elif cnt < target:
        zeros = np.where(b == 0.0)[0]
        order = np.argsort(-h[zeros])
        for i in range(target - cnt):
            b[zeros[order[i]]] = 1.0
    return b

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def make_tm(n):

def make_rand_bin(rng, n):

    idx = rng.permutation(n)

def logsumexp_val(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = corr - m
    e = np.exp(np.clip(beta * s, -400, 400))
    return float((m + np.log(np.sum(e)) / beta) * dx)

def slsqp_optimize(h0, n, beta, maxiter=300, ftol=1e-14, focus_lag=-1):
    dx = 2.0 / n

    def obj(x):
        corr_x = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr_x)
        s = corr_x - m
        e = np.exp(np.clip(beta * s, -400, 400))
        val = (m + np.log(np.sum(e)) / beta) * dx
        if focus_lag >= 0 and focus_lag < len(corr_x):
            val += 0.008 * corr_x[focus_lag] * dx
        return float(val)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def lbfgs_optimize(h0, n, beta, maxiter=500, penalty=200.0):
    dx = 2.0 / n

    def obj(x):
        corr_x = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr_x)
        s = corr_x - m
        e = np.exp(np.clip(beta * s, -400, 400))
        val = (m + np.log(np.sum(e)) / beta) * dx
        val += penalty * (np.sum(x) - n / 2.0) ** 2
        return float(val)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def directed_binary_search(h_bin, n, rng, tlimit, max_iters=20000):

    best_h = h.copy()
    best_c5 = c5
    stuck = 0

            break
        if stuck > 3000:
            break

        k_active = int(np.argmax(corr))

        improved = False

        for _ in range(n_tries):

                break
