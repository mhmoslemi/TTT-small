def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h


def make_golden(n):
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return h


def make_m_seq(n):
    # Generate a maximal-length sequence of length n (n must be 2^m - 1)
    m = int(round(np.log2(n + 1)))
    if (1 << m) - 1 != n:
        return make_thue_morse(n)
    # Simple LFSR for m-sequence
    poly = {7: 0b10011, 15: 0b100011, 31: 0b100000001, 63: 0b100000000001,
            127: 0b10000000000000011, 255: 0b100010000000000000001,
            511: 0b1000000000000000000000000000000000001}
    if m > 2 and m not in {3, 4, 5, 6, 7, 8, 9}:
        return make_thue_morse(n)
    # Use a simple LFSR
    reg = 1
    taps = {3: [3, 1], 4: [4, 3], 5: [5, 3], 6: [6, 5], 7: [7, 6],
            8: [8, 6], 9: [9, 5]}
    if m not in taps:
        return make_thue_morse(n)
    tap_positions = [t - 1 for t in taps[m]]
    seq = []
    for _ in range(n):
        seq.append(reg & 1)
        bit = 0
        for t in tap_positions:
            bit ^= (reg >> t) & 1
        reg = (reg >> 1) | (bit << (m - 1))
    h = np.array(seq, dtype=float)
    if np.sum(h) > n / 2:
        h = 1.0 - h
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 8.0)


def obj_pows_sum(x, p, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_c = np.max(corr)
    if max_c < 1e-30:
        return 0.0
    ratios = np.maximum(corr, 0.0) / max_c
    log_sum = np.log(np.sum(np.power(ratios, p)))
    val = max_c * np.exp(log_sum / p)
    return val * dx


def obj_pws_grad(x, p, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_c = np.max(corr)
    if max_c < 1e-30:
        return np.zeros(n)
    safe_corr = np.maximum(corr, 1e-30)
    ratios = safe_corr / max_c
    log_sum = np.log(np.sum(np.power(ratios, p)))
    S = np.exp(log_sum)
    if S < 1e-30:
        return np.zeros(n)
    weights = np.power(ratios, p - 1) / S
    grad = np.zeros(n)
    nz = np.where(weights > 1e-15)[0]
    for k in nz:
        w = weights[k] * max_c
        s = k - (n - 1)
        if s >= 0:
            npair = n - s
            if npair <= 0:
                continue
            grad[:npair] += w * (1.0 - x[s:s + npair])
            grad[s:] -= w * x[:npair]
        else:
            abs_s = -s
            npair = n - abs_s
            if npair <= 0:
                continue
            grad[:npair] -= w * x[abs_s:abs_s + npair]
            grad[abs_s:] += w * (1.0 - x[:npair])
    return grad * dx


def active_set_grad(h, n, dx, top_k=5):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
    weights = np.zeros(len(corr))
    max_c = corr[top_idx[0]]
    for k in top_idx:
        weights[k] = 1.0 / len(top_idx)
    grad = np.zeros(n)
    for k in top_idx:
        w = weights[k]
        s = k - (n - 1)
        if s >= 0:
            npair = n - s
            if npair <= 0:
                continue
            grad[:npair] += w * (1.0 - h[s:s + npair])
            grad[s:] -= w * h[:npair]
        else:
            abs_s = -s
            npair = n - abs_s
            if npair <= 0:
                continue
            grad[:npair] -= w * h[abs_s:abs_s + npair]
            grad[abs_s:] += w * (1.0 - h[:npair])
    return grad * dx, top_idx


def do_lbfgs(h_start, n, p_list, budget_fn, maxiter_per_p=200):
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    for p in p_list:
        if budget_fn() < 10.0:
            break
        p_val = float(p)

        def func(x):
            h_f = np.clip(x, 0.0, 1.0)
            return obj_pows_sum(h_f, p_val, dx)

        def jac(x):
            h_f = np.clip(x, 0.0, 1.0)
            g = obj_pws_grad(h_f, p_val, dx)
            # Penalty for sum deviation
            diff = np.sum(h_f) - n / 2.0
            g = g + 10.0 * diff / n
            return g

        try:
            maxit = maxiter_per_p
            est_time = budget_fn() * 0.3
            maxit = min(maxit, max(20, int(est_time / 0.05)))
            res = minimize(func, h_cur, jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-8,
                                    'maxls': 50})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-14:
                c5_cur = c5_new
                h_cur = h_new
            else:
                # Try projected version of res.x
                h_proj = project_feasible(res.x, n)
                c5_proj = compute_c5(h_proj)
                if c5_proj < c5_cur - 1e-14:
                    c5_cur = c5_proj
                    h_cur = h_proj
        except Exception:
            pass
    return h_cur, c5_cur


def do_slsqp(h_start, n, p_list, budget_fn, maxiter_per_p=150):
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    for p in p_list:
        if budget_fn() < 15.0:
            break
        p_val = float(p)

        def func(x):
            h_f = np.clip(x, 0.0, 1.0)
            return obj_pows_sum(h_f, p_val, dx)

        def jac(x):
            h_f = np.clip(x, 0.0, 1.0)
            return obj_pws_grad(h_f, p_val, dx)

        def con_sum(x):
            return np.sum(x) - n / 2.0

        def con_jac(x):
            return np.ones(n)

        cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_jac}]
        try:
            maxit = maxiter_per_p
            est_time = budget_fn() * 0.35
            maxit = min(maxit, max(30, int(est_time / 0.15)))
            res = minimize(func, h_cur, jac=jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxit, 'ftol': 1e-14, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-14:
                c5_cur = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, c5_cur


def do_subgrad(h_best, n, budget_fn, rng, max_iters=200):
    h = h_best.copy()
    c5_best = compute_c5(h)
    dx = 2.0 / n
    no_improve = 0
    for it in range(max_iters):
        if budget_fn() < 3.0:
            break
        grad, _ = active_set_grad(h, n, dx, top_k=5)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        improved = False
        steps = [0.003, 0.006, 0.01, 0.02, 0.04, 0.08, 0.015, 0.03, 0.05]
        for step in steps:
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-13:
                c5_best = c5_trial
                h_best = h_trial
                h = h_trial.copy()
                improved = True
                no_improve = 0
                break
        if not improved:
            no_improve += 1
            if no_improve > 40:
                for attempt in range(8):
                    scale = 0.01 + 0.01 * attempt
                    noise = rng.randn(n) * scale
                    h_trial = project_feasible(h + noise, n)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < c5_best - 1e-13:
                        c5_best = c5_trial
                        h_best = h_trial.copy()
                        h = h_trial.copy()
                        no_improve = 0
                        break
                if no_improve > 50:
                    break
    return h_best, c5_best


def do_coord_descent(h_best, n, budget_fn, rng, max_sweeps=20):
    h = h_best.copy()
    c5_best = compute_c5(h)
    for sweep in range(max_sweeps):
        if budget_fn() < 2.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if budget_fn() < 1.0:
                break
            old_val = h[i]
            candidates = [0.0, 1.0]
            # Also try intermediate values if not already binary
            if 0.01 < old_val < 0.99:
                candidates.extend([0.1, 0.3, 0.5, 0.7, 0.9])
            for new_val in candidates:
                if abs(new_val - old_val) < 1e-7:
                    continue
                h[i] = new_val
                h = project_feasible(h, n)
                c5_test = compute_c5(h)
                if c5_test < c5_best - 1e-13:
                    c5_best = c5_test
                    h_best = h.copy()
                    any_imp = True
                    break
                else:
                    h = h_best.copy()
        if not any_imp and sweep > 5:
            break
    return h_best, c5_best


def do_multistart_slsqp(h_best, n, budget_fn, rng, n_starts=4):
    c5_best = compute_c5(h_best)
    for r in range(n_starts):
        if budget_fn() < 20.0:
            break
        noise_scale = 0.02 * (1.0 + 0.7 * r)
        h_pert = project_feasible(h_best + rng.randn(n) * noise_scale, n)
        h_cur, c5_cur = do_slsqp(h_pert, n, budget_fn, [50, 200, 800], maxiter_per_p=100)
        if c5_cur < c5_best - 1e-13:
            c5_best = c5_cur
            h_best = h_cur
    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    def budget_fn():
        return max(0.0, _G_BUDGET - (time.time() - _G_START) - 8.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial solution
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        candidates = [make_thue_morse(parent_n), make_golden(parent_n)]
        best_init = None
        best_init_c5 = float('inf')
        for c in candidates:
            c = project_feasible(c, parent_n)
            c5c = compute_c5(c)
            if c5c < best_init_c5:
                best_init_c5 = c5c
                best_init = c
        best_h = best_init.copy()
        best_c5 = best_init_c5
        best_n = parent_n
        print("No parent. Best init: n={}, C5={:.10f}".format(parent_n, best_c5))

    # Phase 1: L-BFGS-B with increasing power
    print("\n--- Phase 1: L-BFGS-B (n={}) ---".format(best_n))
    best_h, best_c5 = do_lbfgs(best_h, best_n, [4, 16, 64, 256, 1024, 4096], budget_fn, maxiter_per_p=150)
    print("  After L-BFGS-B: C5={:.10f}".format(best_c5))

    # Phase 2: SLSQP with moderate power
    if budget_fn() > 30:
        print("\n--- Phase 2: SLSQP moderate (n={}) ---".format(best_n))
        best_h, best_c5 = do_slsqp(best_h, best_n, [30, 100, 300], budget_fn, maxiter_per_p=120)
        print("  After SLSQP mod: C5={:.10f}".format(best_c5))

    # Phase 3: Subgradient refinement
    if budget_fn() > 15:
        print("\n--- Phase 3: Subgradient (n={}) ---".format(best_n))
        best_h, best_c5 = do_subgrad(best_h, best_n, budget_fn, rng, max_iters=150)
        print("  After subgrad: C5={:.10f}".format(best_c5))

    # Phase 4: SLSQP with high power
    if budget_fn() > 30:
        print("\n--- Phase 4: SLSQP high-p (n={}) ---".format(best_n))
        best_h, best_c5 = do_slsqp(best_h, best_n, [1000, 3000, 10000], budget_fn, maxiter_per_p=100)
        print("  After SLSQP high: C5={:.10f}".format(best_c5))

    # Phase 5: Try other resolutions
    print("\n--- Phase 5: Multi-resolution ---")
    resolutions = [240, 264, 272, 280, 288, 300, 320, 352, 384, 448, 512]
    for n_try in resolutions:
        if budget_fn() < 30.0:
            break
        if n_try == best_n:
            continue
        h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        c5_interp = compute_c5(h_interp)
        h_new, c5_new = do_lbfgs(h_interp, n_try, [16, 64, 256, 1024], budget_fn, maxiter_per_p=100)
        # Quick SLSQP polish
        if budget_fn() > 20:
            h_new, c5_new = do_slsqp(h_new, n_try, [200, 800], budget_fn, maxiter_per_p=60)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n_try
            print("  n={} IMPROVED: C5={:.10f}".format(n_try, best_c5))
        else:
            print("  n={} no improve: {:.10f}".format(n_try, c5_new))

    # Phase 6: Back to best_n, more refined optimization
    if budget_fn() > 30:
        print("\n--- Phase 6: Refined SLSQP (n={}) ---".format(best_n))
        best_h, best_c5 = do_slsqp(best_h, best_n, [50, 200, 800, 2000], budget_fn, maxiter_per_p=80)
        print("  After refined SLSQP: C5={:.10f}".format(best_c5))

    # Phase 7: Multistart
    if budget_fn() > 40:
        print("\n--- Phase 7: Multistart (n={}) ---".format(best_n))
        best_h, best_c5 = do_multistart_slsqp(best_h, best_n, budget_fn, rng, n_starts=4)
        print("  After multistart: C5={:.10f}".format(best_c5))

    # Phase 8: Subgradient again
    if budget_fn() > 15:
        print("\n--- Phase 8: Subgradient 2 (n={}) ---".format(best_n))
        best_h, best_c5 = do_subgrad(best_h, best_n, budget_fn, rng, max_iters=150)
        print("  After subgrad2: C5={:.10f}".format(best_c5))

    # Phase 9: Coordinate descent
    if budget_fn() > 10:
        print("\n--- Phase 9: Coord descent (n={}) ---".format(best_n))
        best_h, best_c5 = do_coord_descent(best_h, best_n, budget_fn, rng, max_sweeps=15)
        print("  After coord: C5={:.10f}".format(best_c5))

    # Phase 10: Final L-BFGS-B with very high power
    if budget_fn() > 15:
        print("\n--- Phase 10: Final L-BFGS-B (n={}) ---".format(best_n))
        best_h, best_c5 = do_lbfgs(best_h, best_n, [2048, 8192, 32768], budget_fn, maxiter_per_p=100)
        print("  After final LBFGS: C5={:.10f}".format(best_c5))

    # Phase 11: Final SLSQP polish
    if budget_fn() > 10:
        print("\n--- Phase 11: Final SLSQP polish ---")
        best_h, best_c5 = do_slsqp(best_h, best_n, [5000, 20000], budget_fn, maxiter_per_p=60)
        print("  After final SLSQP: C5={:.10f}".format(best_c5))

    # Phase 12: Final subgradient steps
    if budget_fn() > 5:
        best_h, best_c5 = do_subgrad(best_h, best_n, budget_fn, rng, max_iters=50)

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
