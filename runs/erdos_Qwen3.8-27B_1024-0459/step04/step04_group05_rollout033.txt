import numpy as np
from scipy.optimize import minimize
import time

_G_START = 0.0
_G_BUDGET = 1300.0
_G_N = 0
_G_TGT = 0.0
_G_BETA = 1.0
_G_DX = 1.0


def remaining():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 4.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    v = np.zeros(N)
    v[:n] = 1.0 - h
    fa = np.fft.fft(a)
    fv = np.fft.fft(v)
    corr = np.real(np.fft.ifft(np.conj(fa) * fv))
    return float(max(np.max(corr[:n]), np.max(corr[N - n + 1:]))) * dx


def get_full_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        idx = int(np.argmax(room))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def compute_subgrad(h, n, top_k=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
    grad = np.zeros(n)
    weights = np.zeros(len(top_idx))
    c_max = corr[top_idx[0]]
    for i, k_idx in enumerate(top_idx):
        w = np.exp(-(corr[k_idx] - c_max) * 10.0)
        weights[i] = w
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights = np.ones(len(top_idx))
        wsum = float(len(top_idx))
    for i, k_idx in enumerate(top_idx):
        offset = int(k_idx) - (n - 1)
        wg = weights[i] / wsum
        if offset >= 0:
            n_pairs = n - offset
            if n_pairs <= 0:
                continue
            grad[:n_pairs] += wg * (1.0 - h[offset:offset + n_pairs])
            grad[offset:] -= wg * h[:n_pairs]
        else:
            abs_off = -offset
            n_pairs = n - abs_off
            if n_pairs <= 0:
                continue
            grad[:n_pairs] -= wg * h[abs_off:abs_off + n_pairs]
            grad[abs_off:] += wg * (1.0 - h[:n_pairs])
    return grad * dx, top_idx, corr


def lse_objective(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _G_BETA
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    return (m + np.log(np.sum(exps)) / beta) * dx


def sum_constraint(x):
    return np.sum(x) - _G_TGT


def sum_jac(x):
    return np.ones(_G_N)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h


def make_golden_ratio(n):
    h = np.zeros(n)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return h


def make_random_bin(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h


def do_gradient_descent(h, n, max_iters, top_k=5, momentum=0.85, init_lr=0.005):
    global _G_N, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    c5_best = compute_c5(h)
    h_best = h.copy()
    h_cur = h.copy()
    v = np.zeros(n)
    lr = init_lr
    stall = 0
    for it in range(max_iters):
        if remaining() < 3.0:
            break
        grad, top_idx, corr = compute_subgrad(h_cur, n, top_k=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        v = momentum * v - lr * grad / np.sqrt(gnorm + 1e-10)
        h_new = project_feasible(h_cur + v, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-14:
            c5_best = c5_new
            h_best = h_new.copy()
            h_cur = h_new
            stall = 0
        else:
            stall += 1
            if stall % 20 == 0:
                lr *= 0.7
                v *= 0.5
            if stall > 80:
                h_cur = h_best.copy()
                v = np.zeros(n)
                lr = init_lr
                noise = np.random.randn(n) * 0.01
                h_cur = project_feasible(h_cur + noise, n)
                stall = 40
        if lr < 1e-7:
            lr = init_lr
            v = np.zeros(n)
    return h_best, c5_best


def do_slsqp(h, n, beta_list, maxiter_list=None):
    global _G_N, _G_BETA, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac}]
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for i, beta in enumerate(beta_list):
        if remaining() < 8.0:
            break
        _G_BETA = float(beta)
        if maxiter_list is not None:
            mi = maxiter_list[i]
        else:
            if beta < 200:
                mi = 300
            elif beta < 2000:
                mi = 200
            else:
                mi = 100
        try:
            res = minimize(lse_objective, h_cur, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': mi, 'ftol': 1e-18, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, c5_cur


def do_adaptive_subgrad(h, n, max_iters, top_k=5):
    global _G_N, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    c5_best = compute_c5(h)
    h_best = h.copy()
    h_cur = h.copy()
    no_improve = 0
    lr = 0.02
    for it in range(max_iters):
        if remaining() < 3.0:
            break
        grad, top_idx, corr = compute_subgrad(h_cur, n, top_k=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        improved = False
        for mult in [1.0, 0.7, 0.4, 0.2, 0.1, 0.05]:
            step = lr * mult
            h_trial = project_feasible(h_cur - step * grad, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-14:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h_cur = h_trial.copy()
                improved = True
                no_improve = 0
                lr = min(lr * 1.1, 0.1)
                break
        if not improved:
            no_improve += 1
            lr *= 0.7
            if no_improve > 20:
                noise_scale = 0.01 * (1 + no_improve / 20.0)
                h_trial = project_feasible(h_best + np.random.randn(n) * noise_scale, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < c5_best - 1e-14:
                    c5_best = c5_trial
                    h_best = h_trial.copy()
                    h_cur = h_trial.copy()
                    no_improve = 0
                else:
                    h_cur = h_best.copy()
                    if no_improve > 60:
                        break
    return h_best, c5_best


def do_coord_flip(h, n, max_passes=10):
    c5_best = compute_c5(h)
    h_best = h.copy()
    for p in range(max_passes):
        if remaining() < 5.0:
            break
        any_imp = False
        order = np.random.permutation(n)
        for i in order:
            if remaining() < 2.0:
                break
            old = h[i]
            for new_val in [0.0, 1.0]:
                if abs(new_val - old) < 1e-6:
                    continue
                h[i] = new_val
                h = project_feasible(h, n)
                c5_try = compute_c5(h)
                if c5_try < c5_best - 1e-14:
                    c5_best = c5_try
                    h_best = h.copy()
                    any_imp = True
                    break
                else:
                    h = h_best.copy()
        if not any_imp and p > 2:
            break
    return h_best, c5_best


def optimize_at_resolution(h_start, n, rng):
    if remaining() < 20.0:
        return h_start, compute_c5(h_start)
    h_cur = project_feasible(h_start, n)
    c5_cur = compute_c5(h_cur)

    h_cur, c5_cur = do_adaptive_subgrad(h_cur, n, max_iters=200)
    if remaining() < 10.0:
        return h_cur, c5_cur

    h_cur, c5_cur = do_slsqp(h_cur, n, [50, 200, 800, 3000], [200, 150, 100, 80])
    if remaining() < 10.0:
        return h_cur, c5_cur

    h_cur, c5_cur = do_gradient_descent(h_cur, n, max_iters=300, top_k=7, momentum=0.8, init_lr=0.003)
    if remaining() < 10.0:
        return h_cur, c5_cur

    h_cur, c5_cur = do_slsqp(h_cur, n, [5000, 20000], [80, 50])
    if remaining() < 8.0:
        return h_cur, c5_cur

    h_cur, c5_cur = do_adaptive_subgrad(h_cur, n, max_iters=100, top_k=7)

    return h_cur, c5_cur
