```python
def compute_swap_deltas(h, a, b, n):
    # a: position being changed 1->0
    # b: position being changed 0->1
    # Returns array of deltas for all 2n-1 shifts
    deltas = np.zeros(2*n-1)
    
    # For shift d > 0 (index d + n - 1 in full correlation):
    # delta_C(d) = (h[a-d]-1)*[0<=a-d and a<n] + (1-h[b-d])*[0<=b-d and b<n] + h[a+d]*[a<n-d] - h[b+d]*[b<n-d]
    
    for d in range(1, n):
        idx_pos = d + n - 1  # index in full correlation
        idx_neg = -(d) + n - 1  # index for shift -d
        
        # C(d) delta
        cd = 0
        if a - d >= 0 and a < n:
            cd += h[a-d] - 1
        if b - d >= 0 and b < n:
            cd += 1 - h[b-d]
        if a < n - d:
            cd += h[a+d]
        if b < n - d:
            cd -= h[b+d]
        deltas[idx_pos] = cd
        
        # C(-d) delta
        cd_neg = 0
        if a < n - d:
            cd_neg -= (1 - h[a+d])
        if b < n - d:
            cd_neg += (1 - h[b+d])
        if a - d >= 0:
            cd_neg += h[a-d]
        if b - d >= 0:
            cd_neg -= h[b-d]
        deltas[idx_neg] = cd_neg
    
    return deltas
```

```python
def compute_swap_deltas_vec(h, a, b, n):
    d = np.arange(1, n)
    deltas = np.zeros(2*n-1)
    
    # C(d) for d = 1..n-1
    # h[a-d] valid where a-d >= 0, i.e., d <= a
    mask_ad = d <= a  # a-d >= 0
    mask_bd = d <= b  # b-d >= 0
    mask_ad2 = d < n - a  # a < n-d
    mask_bd2 = d < n - b  # b < n-d
    
    cd = np.zeros(n-1)
    cd[mask_ad] += h[a - d[mask_ad]] - 1
    cd[mask_bd] += 1 - h[b - d[mask_bd]]
    cd[mask_ad2] += h[a + d[mask_ad2]]
    cd[mask_bd2] -= h[b + d[mask_bd2]]
    
    deltas[n:n-1+n-1] = cd  # indices n..2n-2 for positive shifts
    
    # C(-d) for d = 1..n-1
    cd_neg = np.zeros(n-1)
    cd_neg[mask_ad2] -= (1 - h[a + d[mask_ad2]])
    cd_neg[mask_bd2] += (1 - h[b + d[mask_bd2]])
    cd_neg[mask_ad] += h[a - d[mask_ad]]
    cd_neg[mask_bd] -= h[b - d[mask_bd]]
    
    deltas[0:n-1] = cd_neg  # indices 0..n-2 for negative shifts
    
    return deltas
```

For the peak at shift d, corr[d] = sum_i h[i+d]*(1-h[i]).

- Increase h[i] for positions where h[i+d]=1 (reduces the "1*1" type terms since 1-h[i] decreases)

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h
```
