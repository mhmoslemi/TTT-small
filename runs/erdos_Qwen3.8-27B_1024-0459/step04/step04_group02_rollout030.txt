For k=0: ∫ h(x)(1-h(x)) dx = ∫ h(x) dx - ∫ h(x)² dx = 1 - ∫ h(x)² dx

corr[k] = #{i : h[i]=1, h[i+offset]=0} (for the appropriate range)

When I flip h[i] from 0 to 1:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = int(np.sum(free))
        if nf > 0:
            h[free] += diff / nf
        else:
            if diff > 0:
                i = int(np.argmin(h))
            else:
                i = int(np.argmax(h))
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    return h

def _lbfgs_smooth_obj(h, beta, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    val = (m + np.log(max(wsum, 1e-300)) / beta) * dx
    val += lam * (np.sum(h) - n / 2.0) ** 2
    return val

def _lbfgs_smooth_grad(h, beta, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_vals / wsum

    grad = np.zeros(n)
    ncorr = len(corr)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset+npairs] += w * (1.0 - h[:npairs])
                grad[:npairs] -= w * h[offset:offset+npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - h[aoff:aoff+npairs])
                grad[aoff:aoff+npairs] -= w * h[:npairs]
    grad *= dx
    if lam > 0:
        grad += 2.0 * lam * (np.sum(h) - n / 2.0)
    return grad

def _make_lbfgs_callbacks(beta, lam, n):
    obj_fn = lambda x: _lbfgs_smooth_obj(x, beta, lam)
    grad_fn = lambda x: _lbfgs_smooth_grad(x, beta, lam)
    return obj_fn, grad_fn

def lbfgs_optimize(h, n, beta, lam, maxiter=500):
    obj_fn, grad_fn = _make_lbfgs_callbacks(beta, lam, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxls': 20, 'maxcor': 30})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def coord_descent(h, n, n_grid=60, max_sweeps=200, time_limit_fn=None):
    best = h.copy()
    best_c5 = compute_c5(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    t_grid = np.linspace(0.0, 1.0, n_grid)

    for sweep in range(max_sweeps):
        if time_limit_fn is not None and time_limit_fn() < 5.0:
            break
        improved = False
        best_corr = compute_corr(best)

        for i in range(n):
            t0 = best[i]

            # Compute a coefficients for each lag
            a = np.zeros(ncorr)
            b = np.zeros(ncorr)
            for k in range(ncorr):
                offset = k - (n - 1)
                if offset == 0:
                    b[k] = -1.0
                    a[k] = 1.0
                elif offset > 0:
                    if i < n - offset:
                        a[k] += (1.0 - best[i + offset])
                    if i >= offset:
                        a[k] -= best[i - offset]
                else:
                    aoff = -offset
                    if i >= aoff:
                        a[k] += (1.0 - best[i - aoff])
                    if i < n - aoff:
                        a[k] -= best[i + aoff]

            base = best_corr - a * t0 - b * t0 * t0

            # Evaluate at grid points
            vals = base[:, None] + a[:, None] * t_grid[None, :] + b[:, None] * t_grid[None, :] ** 2
            f_vals = np.max(vals, axis=0) * dx

            best_idx = int(np.argmin(f_vals))
            t_best = t_grid[best_idx]
            c5_best_local = f_vals[best_idx]

            if c5_best_local < best_c5 - 1e-14 and abs(t_best - t0) > 1e-8:
                best[i] = t_best
                best_corr = compute_corr(best)
                best_c5 = compute_c5(best)
                improved = True

        if not improved:
            break

    return best, best_c5

def binarize_decode(h, n, n_grid=20, max_sweeps=50, time_limit_fn=None):
    """Try pushing values towards 0 or 1."""
    best = h.copy()
    best_c5 = compute_c5(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1

    for sweep in range(max_sweeps):
        if time_limit_fn is not None and time_limit_fn() < 3.0:
            break
        improved = False
        best_corr = compute_corr(best)

        for i in range(n):
            t0 = best[i]
            # Try pushing towards 0 and 1
            for t_new in [0.0, 1.0]:
                if abs(t_new - t0) < 1e-8:
                    continue
                # Compute effect
                delta = t_new - t0
                a = np.zeros(ncorr)
                b = np.zeros(ncorr)
                for k in range(ncorr):
                    offset = k - (n - 1)
                    if offset == 0:
                        b[k] = -1.0
                        a[k] = 1.0
                    elif offset > 0:
                        if i < n - offset:
                            a[k] += (1.0 - best[i + offset])
                        if i >= offset:
                            a[k] -= best[i - offset]
                    else:
                        aoff = -offset
                        if i >= aoff:
                            a[k] += (1.0 - best[i - aoff])
                        if i < n - aoff:
                            a[k] -= best[i + aoff]
                base = best_corr - a * t0 - b * t0 * t0
                new_corr = base + a * t_new + b * t_new * t_new
                c5_trial = np.max(new_corr) * dx
                if c5_trial < best_c5 - 1e-12:
                    best[i] = t_new
                    best = project_feasible(best, n)
                    best_corr = compute_corr(best)
                    best_c5 = compute_c5(best)
                    improved = True
                    break

        if not improved:
            break

    return best, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
