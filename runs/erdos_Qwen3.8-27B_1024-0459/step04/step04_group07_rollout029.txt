        c5_512 = compute_c5(h_512)
        print("  Interpolated: C5={:.8f}".format(c5_512))
        
        # Multi-start at n=512
        inits = make_diverse_inits(best_h, best_n, n2, rng, count=6)
        scores = [(compute_c5(h), i) for i, h in enumerate(inits)]
        scores.sort()
        
        # Optimize the best 3 starts
        for score, idx in scores[:3]:
            if tlimit() < 40:
                break
            h_try = inits[idx].copy()
            c5_try = score
            for bv in [30, 100, 400, 1500, 5000]:
                if tlimit() < 15:
                    break
                if tlimit() > 60:
                    h_try = lbfgsb_run(h_try, n2, bv, maxiter=300, pen=80.0)
                h_try = slsqp_run(h_try, n2, bv, maxiter=200)
                c_new = compute_c5(h_try)
                if c_new < c5_try - 1e-13:
                    c5_try = c_new
                    h_try = h_try
            if c5_try < best_c5 - 1e-10:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n2
                print("  n=512 start{} improved: C5={:.8f}".format(idx, best_c5))
            else:
                print("  n=512 start{}: C5={:.8f}".format(idx, c5_try))
    
    # === Phase 3: Try n=768 ===
    if tlimit() > 150:
        print("\n=== Phase 3: n=768 ===")
        n3 = 768
        h_768 = project_feasible(interp_h(best_h, best_n, n3), n3)
        c5_768 = compute_c5(h_768)
        print("  Interpolated: C5={:.8f}".format(c5_768))
        
        h_try = h_768.copy()
        c5_try = c5_768
        for bv in [20, 80, 300, 1000]:
            if tlimit() < 20:
                break
            h_try = lbfgsb_run(h_try, n3, bv, maxiter=200, pen=60.0)
            if tlimit() > 25:
                h_try = slsqp_run(h_try, n3, bv, maxiter=150)
            c_new = compute_c5(h_try)
            if c_new < c5_try - 1e-13:
                c5_try = c_new
        
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n3
            print("  n=768 improved: C5={:.8f}".format(best_c5))
        else:
            print("  n=768: C5={:.8f} (no improvement)".format(c5_try))
    
    # === Phase 4: Come back to best resolution, aggressive polish ===
    if tlimit() > 50:
        print("\n=== Phase 4: polish n={} ===".format(best_n))
        n = best_n
        h_cur = best_h.copy()
        c5_cur = best_c5
        
        # Very high beta SLSQP
        for bv in [10000, 25000, 50000, 100000]:
            if tlimit() < 8:
                break
            h_new = slsqp_run(h_cur, n, bv, maxiter=500)
            c_new = compute_c5(h_new)
            if c_new < c5_cur - 1e-14:
                c5_cur = c_new
                h_cur = h_new
        
        # Active-set
        for it in range(8):
            if tlimit() < 6:
                break
            h_new = active_set_optimize(h_cur, n, 5000, maxiter=300)
            c_new = compute_c5(h_new)
            if c_new < c5_cur - 1e-14:
                c5_cur = c_new
                h_cur = h_new
        
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        print("  Phase 4 done: C5={:.8f}".format(best_c5))
    
    # === Phase 5: Diverse restarts at best resolution ===
    if tlimit() > 60:
        print("\n=== Phase 5: diverse restarts n={} ===".format(best_n))
        n = best_n
        
        for attempt in range(12):
            if tlimit() < 12:
                break
            # Various perturbation strategies
            strategy = attempt % 3
            h_pert = best_h.copy()
            
            if strategy == 0:
                # Random Gaussian perturbation
                n_pert = rng.randint(8, 30)
                idxs = rng.choice(n, size=n_pert, replace=False)
                for idx in idxs:
                    h_pert[idx] = np.clip(h_pert[idx] + rng.randn() * 0.4, 0.0, 1.0)
            elif strategy == 1:
                # Block perturbation
                block_size = rng.randint(4, 12)
                start = rng.randint(0, n - block_size)
                for idx in range(start, start + block_size):
                    h_pert[idx] = np.clip(h_pert[idx] + rng.randn() * 0.5, 0.0, 1.0)
            else:
                # Global small perturbation
                h_pert = np.clip(h_pert + rng.randn(n) * 0.08, 0.0, 1.0)
            
            h_pert = project_feasible(h_pert, n)
            c5_p = compute_c5(h_pert)
            
            if c5_p < best_c5 * 1.01:
                for bv in [200, 1000, 5000]:
                    if tlimit() < 5:
                        break
                    h_pert = slsqp_run(h_pert, n, bv, maxiter=200)
                    c5_p = compute_c5(h_pert)
                if c5_p < best_c5 - 1e-11:
                    best_c5 = c5_p
                    best_h = h_pert.copy()
                    best_n = n
                    print("  Restart {} (strat {}) improved: C5={:.8f}".format(attempt, strategy, best_c5))
    
    # === Phase 6: Final ultra-high-beta polish ===
    if tlimit() > 15:
        print("\n=== Phase 6: final polish ===")
        n = best_n
        for bv in [50000, 100000, 200000]:
            if tlimit() < 5:
                break
            h_new = slsqp_run(best_h, n, bv, maxiter=500)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new
        
        # One more active-set pass
        if tlimit() > 8:
            h_new = active_set_optimize(best_h, n, 10000, maxiter=400)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new
        
        print("  Phase 6 done: C5={:.8f}".format(best_c5))
    
    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass
    
    _G_PEN = 0.0
    _G_FOCUS = -1
    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
