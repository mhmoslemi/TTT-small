import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    """Project h to feasible set: 0<=h<=1, sum(h) = n/2"""
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            adj = diff / cnt
            # Check if adjustment would go out of bounds
            new_vals = h[mask] + adj
            if np.all(new_vals >= 0) and np.all(new_vals <= 1):
                h[mask] = new_vals
            else:
                # Only adjust those that can accommodate
                if adj > 0:
                    m2 = mask & (h < 1.0 - abs(adj))
                else:
                    m2 = mask & (h > abs(adj))
                cnt2 = np.sum(m2)
                if cnt2 > 0:
                    h[m2] += diff / cnt2
                else:
                    break
        else:
            break
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute among interior points
        interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(interior) > 0:
            h[interior[0]] = np.clip(h[interior[0]] + diff, 0, 1)
            diff2 = target - np.sum(h)
            if abs(diff2) > 1e-10 and len(interior) > 1:
                h[interior[1]] = np.clip(h[interior[1]] + diff2, 0, 1)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def smooth_max_obj_grad(h, n, beta):
    """Compute smooth approximation of max(corr)*dx and its gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    
    # Gradient
    nc = len(corr)
    grad = np.zeros(n)
    for k in range(nc):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset+npairs] += w * (1.0 - h[:npairs])
                grad[:npairs] -= w * h[offset:offset+npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - h[aoff:aoff+npairs])
                grad[aoff:aoff+npairs] -= w * h[:npairs]
    grad *= dx
    return val, grad

def slsqp_optimize(h, n, beta, maxiter):
    """SLSQP optimization with smooth max objective."""
    dx = 2.0 / n
    
    def obj(x):
        return smooth_max_obj_grad(x, n, beta)[0]
    
    def obj_grad(x):
        return smooth_max_obj_grad(x, n, beta)[1]
    
    def constraint_fun(x):
        return np.sum(x) - n / 2.0
    
    def constraint_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            if compute_c5(trial) <= compute_c5(h) + 1e-14:
                return trial
    except Exception:
        pass
    return h.copy()

def coord_descent(h, n, max_sweeps=50, time_limit=None):
    """Coordinate descent with exact 1D optimization per coordinate."""
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    t_start = time.time()
    
    for sweep in range(max_sweeps):
        if time_limit is not None and (time.time() - t_start) > time_limit:
            break
        if remaining_time() < 5:
            break
        improved = False
        # Process in order (or random)
        for i in range(n):
            if remaining_time() < 2:
                    break
            t0 = best[i]
            # Compute coefficients for each lag
            # corr[k] as function of t = h[i]:
            # For lag with offset off (= k - (n-1)):
            #   term involving h[i]: 
            #     if i is the "first" index (i >= 0 and i+off < n): h[i]*(1-h[i+off])
            #     if i is the "second" index (i-off >= 0 and i < n): h[i-off]*(1-h[i])
            #   So: a_k * t + b_k * t^2 where:
            #     if i+off valid: a_k += (1-h[i+off]), b_k += 0 (unless i+off==i, i.e., off=0)
            #     if i-off valid: a_k += (-h[i-off]), b_k += 0
            #     if off==0 and i valid: the term is t*(1-t) = t - t^2, so a_k += 1, b_k += -1
            
            a = np.zeros(ncorr)
            b = np.zeros(ncorr)
            
            for off in range(-(n-1), n):
                k = n - 1 + off
                if off == 0:
                    # Only contributes if 0 <= i < n (always true)
                    a[k] += 1.0
                    b[k] += -1.0
                else:
                    # Check if i is "first" in a pair: position i, partner at i+off
                    if 0 <= i + off < n:
                        a[k] += (1.0 - best[i + off])
                    # Check if i is "second" in a pair: position i-off, partner at i
                    if 0 <= i - off < n:
                        a[k] += (-best[i - off])
            
            t0 = best[i]
            base = best_corr - a * t0 - b * t0 * t0
            
            def obj_at_t(t):
                vals = base + a * t + b * t * t
                return np.max(vals) * dx
            
            # Golden section search on [0, 1]
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo, hi = 0.0, 1.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = obj_at_t(c1)
            f2 = obj_at_t(c2)
            
            for _ in range(50):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = obj_at_t(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = obj_at_t(c2)
            
            t_opt = (lo + hi) / 2.0
            f_opt = obj_at_t(t_opt)
            
            if f_opt < best_c5 - 1e-14:
                best[i] = t_opt
                best_corr = compute_corr(best)
                best_c5 = compute_c5(best)
                improved = True
        
        if not improved:
            break
    
    return best, best_c5

def balance_step(h, n, tol=0.3, max_iters=100):
    """Try to balance the top correlation values by adjusting h."""
    best = h.copy()
    best_c5 = compute_c5(best)
    dx = 2.0 / n
    
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        corr = compute_corr(best)
        max_corr = np.max(corr)
        
        # Find active lags (near maximum)
        active = np.where(corr >= max_corr - max_corr * tol)[0]
        if len(active) == 0:
            break
        
        # For each active lag, compute the "pressure" gradient
        # We want to decrease the active correlations
        # Gradient of corr[k] w.r.t. h[i]:
        #   If off = k-(n-1) >= 0: d(corr[k])/d(h[i]) = (1-h[i+off]) if i+off<n, minus h[i-off] if i-off>=0
        #   If off < 0: similar
        
        grad = np.zeros(n)
        for k_idx in active:
            offset = int(k_idx) - (n - 1)
            if offset >= 0:
                npairs = n - offset
                for i in range(npairs):
                    # h[i] appears as first: contributes (1-h[i+offset])
                    grad[i] += (1.0 - best[i + offset])
                    # h[i+offset] appears as second: contributes (-h[i])
                    grad[i + offset] += (-best[i])
            else:
                aoff = -offset
                npairs = n - aoff
                for i in range(npairs):
                    # h[i+aoff] appears as first: contributes (1-h[i])
                    grad[i + aoff] += (1.0 - best[i])
                    # h[i] appears as second: contributes (-h[i+aoff])
                    grad[i] += (-best[i + aoff])
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        grad = grad / gnorm
        
        # Try different step sizes
        improved = False
        for step in [0.05, 0.02, 0.01, 0.08, 0.005, 0.03]:
            h_trial = best - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-15:
                best = h_trial
                best_c5 = c5_trial
                improved = True
                break
        
        if not improved:
            break
    
    return best, best_c5

import numpy as np
from scipy.optimize import minimize

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            break
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(interior) > 0:
            h[interior[0]] = np.clip(h[interior[0]] + diff, 0, 1)
            diff2 = target - np.sum(h)
            if abs(diff2) > 1e-13 and len(interior) > 1:
                h[interior[1]] = np.clip(h[interior[1]] + diff2, 0, 1)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def smooth_max_val_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    
    nc = len(corr)
    grad = np.zeros(n)
    for k in range(nc):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset+npairs] += w * (1.0 - h[:npairs])
                grad[:npairs] -= w * h[offset:offset+npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - h[aoff:aoff+npairs])
                grad[aoff:aoff+npairs] -= w * h[:npairs]
    grad *= dx
    return val, grad

def slsqp_optimize(h, n, beta, maxiter):
    dx = 2.0 / n
    
    def obj(x):
        return smooth_max_val_grad(x, n, beta)[0]
    
    def obj_grad(x):
        return smooth_max_val_grad(x, n, beta)[1]
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            if compute_c5(trial) <= compute_c5(h) + 1e-14:
                return trial
    except Exception:
        pass
    return h.copy()

def compute_c5(h):

    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            break
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        interior = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(interior) > 0:
            h[interior[0]] = np.clip(h[interior[0]] + diff, 0, 1)
            diff2 = target - np.sum(h)
            if abs(diff2) > 1e-13 and len(interior) > 1:
                h[interior[1]] = np.clip(h[interior[1]] + diff2, 0, 1)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def smooth_max_val(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    return (m + np.log(wsum) / beta) * dx

def smooth_max_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    nc = len(corr)
    grad = np.zeros(n)
    for k in range(nc):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset+npairs] += w * (1.0 - h[:npairs])
                grad[:npairs] -= w * h[offset:offset+npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - h[aoff:aoff+npairs])
                grad[aoff:aoff+npairs] -= w * h[:npairs]
    grad *= dx
    return grad

def _slsqp_obj_wrapper(x, n, beta):
    return smooth_max_val(x, n, beta)

def _slsqp_grad_wrapper(x, n, beta):
    return smooth_max_grad(x, n, beta)

def slsqp_optimize(h, n, beta, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_slsqp_obj_wrapper, h, args=(n, beta),
                      jac=_slsqp_grad_wrapper, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            if compute_c5(trial) <= compute_c5(h) + 1e-14:

        h_new = slsqp_optimize(h, n, beta, 600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Extended coordinate descent at parent resolution
    if remaining_time() > 60:
        cd_time = min(300, remaining_time() - 200)
        h_new, c5_new = coord_descent_fast(h, n, max_sweeps=200, time_budget=cd_time)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: Balance step
    if remaining_time() > 30:
        h_new, c5_new = balance_step(best_h, best_n, tol=0.2, max_iters=300)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 4: More SLSQP + coordinate descent
    if remaining_time() > 60:
        h = best_h.copy()
        n = best_n
        for beta in [50000, 200000]:
            if remaining_time() < 25:
                break
            h_new = slsqp_optimize(h, n, beta, 500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

        if remaining_time() > 40:
            cd_time = min(150, remaining_time() - 60)
            h_new, c5_new = coord_descent_fast(h, n, max_sweeps=100, time_budget=cd_time)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 5: Try higher resolution
    for rn in [best_n * 2, best_n + 256, best_n * 3 // 2]:
        if rn % 2 != 0:
            rn += 1
        if rn > 1600 or rn < 128:
            continue
        if remaining_time() < 80:
            break
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)

        for beta in [1000, 5000, 20000]:
            if remaining_time() < 35:
                break
            h_new = slsqp_optimize(h_r, rn, beta, 400)
            c5_new = compute_c5(h_new)
            if c5_new < c5_r - 1e-14:
                h_r = h_new
                c5_r = c5_new

        if remaining_time() > 40:
            cd_time = min(120, remaining_time() - 50)
            h_r, c5_r = coord_descent_fast(h_r, rn, max_sweeps=80, time_budget=cd_time)

        if remaining_time() > 20:
            h_r, c5_r2 = balance_step(h_r, rn, tol=0.15, max_iters=200)
            if c5_r2 < c5_r:
                c5_r = c5_r2

        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn

    # Phase 6: Final polish at best resolution
    if remaining_time() > 20:
        h = best_h.copy()
        n = best_n
        for beta in [100000, 300000]:
            if remaining_time() < 10:
                break
            h_new = slsqp_optimize(h, n, beta, 400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 7: Final coordinate descent
    if remaining_time() > 15:
        cd_time = min(80, remaining_time() - 5)
        h_new, c5_new = coord_descent_fast(best_h, best_n, max_sweeps=60, time_budget=cd_time)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 8: Final balance
    if remaining_time() > 10:
        h_new, c5_new = balance_step(best_h, best_n, tol=0.1, max_iters=150)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 9: One more round of coordinate descent
    if remaining_time() > 10:
        cd_time = min(40, remaining_time() - 3)
        h_new, c5_new = coord_descent_fast(best_h, best_n, max_sweeps=40, time_budget=cd_time)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
