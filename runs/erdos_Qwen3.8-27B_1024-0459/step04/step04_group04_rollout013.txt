def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def full_smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -50, 50)
    ex = np.exp(cl)
    s = np.sum(ex)
    return (m + np.log(s) / beta) * dx_val

def full_smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -50, 50)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    
    g = np.zeros(n)
    offset = n - 1
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-16:
            continue
        off = k - offset
        if off >= 0:
            npairs = n - off
            g[off:off+npairs] += wk * (1.0 - x[:npairs])
            g[:npairs] -= wk * x[off:off+npairs]
        else:
            ao = -off
            npairs = n - ao
            g[:npairs] += wk * (1.0 - x[ao:ao+npairs])
            g[ao:ao+npairs] -= wk * x[:npairs]
