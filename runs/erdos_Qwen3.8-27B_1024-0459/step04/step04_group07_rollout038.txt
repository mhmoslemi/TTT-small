def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_BETA = 1.0
_G_DX = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = corr - m
    e = np.exp(np.clip(beta * s, -400.0, 400.0))
    val = (m + np.log(np.sum(e)) / beta) * dx
    return float(val)

def smooth_obj_pen(x):
    beta = _G_BETA
    dx = _G_DX
    n = _G_N
    corr = np.correlate(x, 1.0 - x, mode='full')

def con_jac(x):

    _G_BETA = float(beta)
    _G_DX = 2.0 / n

    try:
        res = minimize(smooth_obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        h_new = project_feasible(res.x, n)
    except Exception:
        h_new = project_feasible(h.copy(), n)
    return h_new

        w = np.exp(np.clip(shifted / tau, -500.0, 500.0))
        w_sum = np.sum(w)

        else:

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, best_c5))
    return (best_h, best_c5, best_n)
