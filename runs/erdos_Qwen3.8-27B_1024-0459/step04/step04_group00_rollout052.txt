from scipy.optimize import minimize, minimize_scalar
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        # Distribute to non-bound elements
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(free):
            h[free] += diff / np.count_nonzero(free)
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)

def lfbfgs_smooth(h, n, beta, maxiter=1000):
    """L-BFGS-B with smooth max objective + penalty for constraint."""
    dx_v = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        h_c = np.clip(x, 0, 1)
        corr = compute_corr(h_c)
        cd = corr * dx_v
        m = np.max(cd)
        sh = cd - m
        cl = np.clip(beta * sh, -50, 50)
        sm = m + np.log(np.sum(np.exp(cl))) / beta
        penalty = 0.01 * (np.sum(x) - target) ** 2
        return sm + penalty
    
    def grad(x):
        h_c = np.clip(x, 0, 1)
        corr = compute_corr(h_c)
        cd = corr * dx_v
        m = np.max(cd)
        sh = cd - m
        cl = np.clip(beta * sh, -50, 50)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        n_full = len(corr)  # 2n-1
        for k in range(n_full):
            wk = w[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:

                g[off:off + np_] += wk * (1.0 - h_c[:np_])
                g[:np_] -= wk * h_c[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - h_c[ao:ao + np_])
                g[ao:ao + np_] -= wk * h_c[:np_]
        g *= dx_v
        g += 0.02 * (np.sum(x) - target) / n
        return g
    
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    try:

                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        result = project_feasible(res.x, n)
        return result
    except:
        return h0

def subgradient_method(h, n, iterations=5000, alpha0=0.1):
    """Subgradient method for the true max objective."""
    dx_v = 2.0 / n
    target = n / 2.0
    h_cur = project_feasible(h, n)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    alpha = alpha0
    
    for it in range(iterations):
        if remaining() < 2:
            break
        corr = compute_corr(h_cur)
        cd = corr * dx_v
        k_max = np.argmax(cd)
        c_max = cd[k_max]

        off = k_max - (n - 1)
        g = np.zeros(n)
        if off >= 0:

            g[off:off + np_] += (1.0 - h_cur[:np_])
            g[:np_] -= h_cur[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += (1.0 - h_cur[ao:ao + np_])
            g[ao:ao + np_] -= h_cur[:np_]

            best_h = h_next.copy()
            # Decay alpha more slowly when improving
            alpha *= 0.999
        else:
            alpha *= 0.99
            
        h_cur = h_next
    
    return best_h, best_c5

                if c5_new < best_c5 - 1e-14:

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

def grad_corr_n(x, n):
    """Gradient of all 2n-1 correlations wrt x."""
    n_full = 2 * n - 1
    G = np.zeros((n_full, n))
    for k in range(n_full):
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            G[k, off:off + np_] += (1.0 - x[:np_])
            G[k, :np_] -= x[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            G[k, :np_] += (1.0 - x[ao:ao + np_])
            G[k, ao:ao + np_] -= x[:np_]
    return G


def lbfgs_smooth(h, n, beta, maxiter=300, penalty_coeff=100.0):

        corr = compute_corr(h_c)
        cd = corr * dx_v
        m = np.max(cd)
        sh = cd - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        sm = m + np.log(np.sum(np.exp(cl))) / beta
        pen = penalty_coeff * (np.sum(x) - target) ** 2

        corr = compute_corr(h_c)
        cd = corr * dx_v
        m = np.max(cd)
        sh = cd - m

        ex = np.exp(cl)

        G = grad_corr_n(h_c, n)
        g = (w * dx_v) @ G
        g += 2.0 * penalty_coeff * (np.sum(x) - target) / n
        return g

            best_c5 = c5_t
            best_h = h_feas.copy()
            best_n = n_t
            n = n_t
            h = h_feas.copy()

    # Phase 8: Final intensive refinement at best resolution
    n = best_n
    h = best_h.copy()

    # High-beta L-BFGS-B
    for beta in [50000, 200000, 800000]:
        if remaining() < 6:
            break
        h_new = lbfgs_smooth(h, n, beta, maxiter=500, penalty_coeff=5000.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Multi-lag equalize final
    if remaining() > 10:
        h_eq, c5_eq = multi_lag_equalize(h, n, top_k=6, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Coordinate descent final
    if remaining() > 12:
        h_cd, c5_cd = coordinate_descent(h, n, n_passes=15, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Subgradient final
    if remaining() > 15:
        h_sg, c5_sg = subgradient_method(h, n, iterations=3000, alpha0=0.01)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Final L-BFGS-B polish
    if remaining() > 6:
        h_new = lbfgs_smooth(best_h, n, 500000, maxiter=400, penalty_coeff=10000.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
