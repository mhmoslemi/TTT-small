def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:

        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:

    diff = target - np.sum(h)
    if abs(diff) > 1e-13:

            rooms = np.minimum(h, 1.0 - h)

def obj_lse(x):
    n = _G_N
    dx = _G_DX
    if n <= 384:
        corr = np.correlate(x, 1.0 - x, mode='full')
    else:
        corr = compute_corr_fft(x)
    m = np.max(corr)
    beta = _G_BETA
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    return (m + np.log(np.sum(exps)) / beta) * dx

def obj_lse_grad(x):
    """Gradient of the LSE objective."""
    n = _G_N
    dx = _G_DX
    if n <= 384:
        corr = np.correlate(x, 1.0 - x, mode='full')
    else:
        corr = compute_corr_fft(x)
    m = np.max(corr)
    beta = _G_BETA
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    weights = exps / np.sum(exps)
    # Gradient of sum of weighted correlations
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        offset = k_idx - (n - 1)
        if offset >= 0:
            n_pairs = n - offset
            if n_pairs <= 0:
                continue
            grad[:n_pairs] += w * (1.0 - x[offset:offset + n_pairs])
            grad[offset:] -= w * x[:n_pairs]
        else:
            abs_off = -offset
            n_pairs = n - abs_off
            if n_pairs <= 0:
                continue
            grad[:n_pairs] -= w * x[abs_off:abs_off + n_pairs]
            grad[abs_off:] += w * (1.0 - x[:n_pairs])
    return grad * dx

def compute_corr_direct(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_corr_fft(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    a = np.fft.fft(h, N)
    b = np.fft.fft(1.0 - h, N)
    circ = np.real(np.fft.ifft(a * np.conj(b)))
    return np.concatenate([circ[N - (n - 1):N], circ[0:n]])


def compute_c5(h):

def subgrad_topk(h, top_k):
    n = len(h)
    dx = 2.0 / n
    if n <= 384:
        corr = np.correlate(h, 1.0 - h, mode='full')
    else:

            c5_init = compute_c5(init_h)
            if c5_init >= best_c5:
                continue
            h_opt, c5_opt = do_slsqp(init_h, 512, budget_fn, [100, 500, 2000])
            if budget_fn() > 10:
                h_opt, c5_opt = do_subgrad(h_opt, 512, budget_fn, rng, top_k=5, max_iters=80)
            if c5_opt < best_c5 - 1e-14:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = 512
                print("  Alt init improved: C5={:.10f}".format(c5_opt))

    # Phase 12: Final tiny adjustments
    if budget_fn() > 8:
        best_h, best_c5 = do_subgrad(best_h, best_n, budget_fn, rng, top_k=5, max_iters=50)
    if budget_fn() > 8:
        best_h, best_c5 = do_slsqp(best_h, best_n, budget_fn, [50000, 200000])
    if budget_fn() > 5:
        best_h, best_c5 = do_coord_descent(best_h, best_n, budget_fn, rng, max_sweeps=5)

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
