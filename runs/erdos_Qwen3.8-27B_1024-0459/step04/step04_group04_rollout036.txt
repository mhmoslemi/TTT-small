import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1
    return h_bin


def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[off:off + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao + npairs])
        g[ao:ao + npairs] -= x[:npairs]
    return g


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    beta = _BETA
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    s = np.sum(ex)
    w = ex / s
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-14:
            g += w[k] * corr_grad_per_shift(x, n, k)
    return g * dx_val


def sum_con(x):
    return np.sum(x) - _TGT


def sum_con_jac(x):
    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=400):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def sa_binary(h, n, time_limit=60, seed=42):

    dx_val = 2.0 / n
    cur_max = float(np.max(compute_corr(h)))
    best_max = cur_max
    best_h = h.copy()

    T = 6.0
    alpha = 0.998
    start = time.time()

def active_set_descent(h, n, iters=5000, lr_init=0.08, tol_gap=3.0):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    lr = lr_init
    for it in range(iters):
        if remaining() < 5:
            break
        if it % 200 == 0 and it > 0:
            lr = max(0.001, lr_init * (0.95 ** (it // 200)))

        corr = compute_corr(h)
        m = float(np.max(corr))
        c5_cur = m * (2.0 / n)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()

        active = np.where(corr > m - tol_gap)[0]
        g = np.zeros(n)
        for k in active:
            g += corr_grad_per_shift(h, n, int(k))
        g /= float(len(active))

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        h_new = h - lr * g / gn
        h_new = project_feasible(h_new, n)

        c5_new = compute_c5(h_new)
        if c5_new <= best_c5 + 1e-10:
            h = h_new
        else:
            lr *= 0.7
            if lr < 1e-5:
                break

    return best_h, best_c5


def topk_gradient_descent(h, n, max_iters=500, top_k=8):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        m = float(np.max(corr))
        if m * (2.0 / n) >= best_c5 + 1e-8:
            break

        top_idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k in top_idx:
            g += corr_grad_per_shift(h, n, int(k))
        g /= float(len(top_idx))

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn

        improved = False
        for step in [0.05, 0.02, 0.01, 0.1, 0.005, 0.03, 0.008, 0.04]:
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-13:
                h = h_trial
                best_c5 = c_new
                best_h = h.copy()
                improved = True
                break

        if not improved:
            break

    return best_h, best_c5
