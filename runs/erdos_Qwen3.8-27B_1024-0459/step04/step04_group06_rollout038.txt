                h_best = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return h_best, c5_best


def iterated_local_search(h_bin, n, num_restarts=5, max_iter_per=20000, seed=42):
    """Iterated local search: hill climb, perturb, repeat"""
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    
    for restart in range(num_restarts):
        if remaining_time() < 10.0:
            break
        
        h_in = h_best.copy()
        # Perturb: do some random swaps
        num_pert = n // 8
        one_pos = np.where(h_in == 1.0)[0].tolist()
        zero_pos = np.where(h_in == 0.0)[0].tolist()
        for _ in range(num_pert):
            if not one_pos or not zero_pos:
                break
            oi = rng.randint(len(one_pos))
            zi = rng.randint(len(zero_pos))
            h_in[one_pos[oi]] = 0.0
            h_in[zero_pos[zi]] = 1.0
            one_pos[oi] = zero_pos[zi]
            zero_pos[zi] = one_pos[0]
            one_pos[0] = one_pos[1]
            one_pos.pop(1)
            zero_pos.pop(zi) if zi < len(zero_pos) else zero_pos.pop(0)
            # Simpler: just redo
            one_pos = np.where(h_in == 1.0)[0].tolist()
            zero_pos = np.where(h_in == 0.0)[0].tolist()
        
        h_new, c5_new = hill_climb_binary(h_in, n, max_iter=max_iter_per, seed=seed + restart * 1000)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new
    
    return h_best, c5_best


def con_sum(x):
    return np.sum(x) - _G_TGT


def obj_smooth(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(_G_BETA * shifted, -600, 600))
    return (m + np.log(np.sum(exps)) / _G_BETA) * dx


def obj_raw(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(corr)) * dx


def slsqp_polish(h, n, betas=[200, 500, 1000, 2000, 5000], max_iter_each=300):
    """SLSQP polish with increasing beta"""
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum}]
    
    c5_best = compute_c5(h)
    h_best = h.copy()
    
    for beta_val in betas:
        if remaining_time() < 5.0:
            break
        _G_BETA = float(beta_val)
        try:
            res = minimize(obj_smooth, h_best, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter_each, 'ftol': 1e-14, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-14:
                c5_best = c5_new
                h_best = h_new
        except Exception:
            pass
    
    return h_best, c5_best


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def subgrad_step(h, n, top_k=5, step_size=0.05):
    """One subgradient step focusing on top-k correlations"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    
    # Find top k correlation indices
    top_indices = np.argsort(corr)[-top_k:]
    top_indices = top_indices[np.argsort(corr[top_indices])[::-1]]
    
    grad = np.zeros(n)
    for k_idx in top_indices:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            n_valid = n - offset
            if n_valid <= 0:
                continue
            grad[offset:offset + n_valid] += (1.0 - h[:n_valid])
            grad[:n_valid] -= h[offset:offset + n_valid]
        else:
            m = -offset
            n_valid = n - m
            if n_valid <= 0:
                continue
            grad[:n_valid] += (1.0 - h[m:m + n_valid])
            grad[m:m + n_valid] -= h[:n_valid]
    
    gn = np.linalg.norm(grad)
    if gn < 1e-14:
        return h, compute_c5(h)
    grad = grad / gn
    
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new, compute_c5(h_new)


def subgrad_momentum(h, n, max_iter=500, step0=0.08, momentum=0.85, top_k=5):
    """Subgradient method with momentum"""
    c5_best = compute_c5(h)
    h_best = h.copy()
    v = np.zeros(n)
    step = step0
    patience = 0
    
    for it in range(max_iter):
        if remaining_time() < 5.0:
            break
        
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        n_corr = len(corr)
        top_indices = np.argsort(corr)[-top_k:]
        top_indices = top_indices[np.argsort(corr[top_indices])[::-1]]
        
        grad = np.zeros(n)
        for k_idx in top_indices:
            offset = int(k_idx) - (n - 1)
            if offset >= 0:
                n_valid = n - offset
                if n_valid <= 0:
                    continue
                grad[offset:offset + n_valid] += (1.0 - h[:n_valid])
                grad[:n_valid] -= h[offset:offset + n_valid]
            else:
                m = -offset
                n_valid = n - m
                if n_valid <= 0:
                    continue
                grad[:n_valid] += (1.0 - h[m:m + n_valid])
                grad[m:m + n_valid] -= h[:n_valid]
        
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad = grad / gn
        
        alpha = step
        improved = False
        for _ in range(10):
            v = momentum * v - alpha * grad
            h_trial = project_feasible(h + v, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-13:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
            alpha *= 0.5
        
        if improved:
            patience = 0
            step = min(alpha * 1.3, 0.3)
        else:
            patience += 1
            step *= 0.6
            v *= 0.5
            if patience > 15:
                break
    
    return h_best, c5_best
