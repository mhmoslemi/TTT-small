_G_DX = 1.0

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Adjust non-bound variables
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:

        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)

    h_new = project_feasible(h_new, n)

                    if c5_t < c5_best:
                        c5_best = c5_t
                        h_best = h_bin.copy()
                    found = True
                    break
                else:
                    h_bin[i1] = 1.0
                    h_bin[i2] = 1.0
                    h_bin[j1] = 0.0
                    h_bin[j2] = 0.0
            if not found:
                # perturb
                for _ in range(3):
                    i = idx1[rng.randint(len(idx1))]
                    j = idx0[rng.randint(len(idx0))]
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
                    idx1 = np.where(h_bin == 1.0)[0]
                    idx0 = np.where(h_bin == 0.0)[0]
    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET, _G_N, _G_BETA, _G_TGT, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent solution
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        best_h = project_feasible(make_thue_morse(parent_n), parent_n)
        best_c5 = compute_c5(best_h)
        best_n = parent_n
        print("TM: n={}, C5={:.10f}".format(parent_n, best_c5))

    n = best_n

    # Phase 1: Quick SLSQP at parent
    print("\n--- Phase 1: SLSQP warmup ---")
    for beta in [50, 200, 800]:
        if time_left() < 30:
            break
        iters = min(400, max(50, int(time_left() * 0.06 / 0.04)))
        best_h, c5 = do_slsqp(best_h, n, beta, iters)
        if c5 < best_c5:
            best_c5 = c5
            best_n = n
        print("  beta={}: C5={:.10f}".format(beta, best_c5))

    # Phase 2: Try alternative starts
    print("\n--- Phase 2: Alt starts ---")
    alts = [make_thue_morse(n), make_golden(n), make_random_bin(rng, n)]
    for alt in alts:
        if time_left() < 40:
            break
        alt = project_feasible(alt, n)
        alt_c5 = compute_c5(alt)
        if alt_c5 < best_c5:
            best_h = alt.copy()
            best_c5 = alt_c5
            best_n = n
            print("  Alt improved: C5={:.10f}".format(alt_c5))
        else:
            # Quick SLSQP on alt
            for beta in [200, 1000]:
                if time_left() < 20:
                    break
                alt, alt_c5 = do_slsqp(alt, n, beta, 150)
            if alt_c5 < best_c5:
                best_h = alt.copy()
                best_c5 = alt_c5
                best_n = n
                print("  Alt after SLSQP: C5={:.10f}".format(alt_c5))

    # Phase 3: Subgradient
    print("\n--- Phase 3: Subgradient ---")
    if time_left() > 30:
        max_sg = min(600, int(time_left() * 0.2 / 0.03))
        best_h, c5 = do_subgrad(best_h, best_n, rng, max_sg)
        if c5 < best_c5:
            best_c5 = c5
            best_n = best_n
        print("  After subgrad: C5={:.10f}".format(best_c5))

    # Phase 4: High-beta SLSQP
    print("\n--- Phase 4: High-beta SLSQP ---")
    for beta in [2000, 8000, 30000]:
        if time_left() < 25:
            break
        iters = min(400, max(40, int(time_left() * 0.08 / 0.04)))
        best_h, c5 = do_slsqp(best_h, best_n, beta, iters)
        if c5 < best_c5:
            best_c5 = c5
            best_n = best_n
        print("  beta={}: C5={:.10f}".format(beta, best_c5))

    # Phase 5: More subgradient
    print("\n--- Phase 5: Subgradient 2 ---")
    if time_left() > 20:
        max_sg2 = min(400, int(time_left() * 0.1 / 0.03))
        best_h, c5 = do_subgrad(best_h, best_n, rng, max_sg2)
        if c5 < best_c5:
            best_c5 = c5
        print("  After subgrad2: C5={:.10f}".format(best_c5))

    # Phase 6: Multi-resolution
    print("\n--- Phase 6: Multi-resolution ---")
    for n_try in [320, 384, 512, 640]:
        if time_left() < 50:
            break
        if n_try == best_n:
            continue
        h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        c5_interp = compute_c5(h_interp)
        for beta in [500, 3000, 10000]:
            if time_left() < 25:
                break
            iters = min(300, max(40, int(time_left() * 0.06 / 0.04)))
            h_interp, c5_interp = do_slsqp(h_interp, n_try, beta, iters)
        if c5_interp < best_c5 - 1e-15:
            best_c5 = c5_interp
            best_h = h_interp
            best_n = n_try
            print("  n={}: C5={:.10f} IMPROVED".format(n_try, best_c5))
        else:
            print("  n={}: C5={:.10f}".format(n_try, c5_interp))

    # Phase 7: Binary local search
    print("\n--- Phase 7: Binary search ---")
    if time_left() > 20:
        h_b, c5_b = do_binary_search(best_h, best_n, rng, 30)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b
            print("  Binary: C5={:.10f}".format(best_c5))
        else:
            print("  Binary best: C5={:.10f} (no improve)".format(c5_b))

    # Phase 8: Coordinate descent
    print("\n--- Phase 8: Coord descent ---")
    if time_left() > 10:
        best_h, c5 = do_coord_descent(best_h, best_n, rng, 8)
        if c5 < best_c5:
            best_c5 = c5
        print("  After coord: C5={:.10f}".format(best_c5))

    # Phase 9: Final high-beta polish
    print("\n--- Phase 9: Final polish ---")
    for beta in [50000, 200000, 500000]:
        if time_left() < 12:
            break
        best_h, c5 = do_slsqp(best_h, best_n, beta, 80)
        if c5 < best_c5:
            best_c5 = c5
            best_n = best_n
    print("  After polish: C5={:.10f}".format(best_c5))

    # Phase 10: Tiny subgradient steps
    if time_left() > 8:
        best_h, c5 = do_subgrad(best_h, best_n, rng, 100)
        if c5 < best_c5:
            best_c5 = c5

    # Phase 11: Final coord
    if time_left() > 5:
        best_h, c5 = do_coord_descent(best_h, best_n, rng, 5)
        if c5 < best_c5:
            best_c5 = c5

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
