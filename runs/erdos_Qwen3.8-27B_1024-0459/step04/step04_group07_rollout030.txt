    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c else 0.0
    return h

def obj_lob(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    b = _G_BETA
    s = corr - m
    e = np.exp(np.clip(b * s, -400, 400))
    val = (m + np.log(np.sum(e)) / b) * _G_DX
    if _G_FOCUS_LAGS is not None and _G_FOCUS_W > 0:
        for k in _G_FOCUS_LAGS:
            if 0 <= k < len(corr):
                val += _G_FOCUS_W * corr[k] * _G_DX
    if _G_PEN > 0:
        val += _G_PEN * (np.sum(x) - _G_TGT) ** 2
    return float(val)

def compute_smoothed_grad(h, n, beta):
    """Compute exact gradient of smoothed log-sum-exp objective."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    log_w = np.clip(beta * (corr - m), -400, 400)
    p = np.exp(log_w)
    psum = np.sum(p)
    if psum < 1e-300:
        return np.zeros(n)
    p /= psum
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    for L in range(-(n - 1), n):
        pm = p[center + L]
        if pm < 1e-18:
            continue
        if L == 0:
            grad += pm * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += pm * (1.0 - h[:n - L])
            grad[:n - L] -= pm * h[L:]
        else:
            M = -L
            grad[:n - M] += pm * (1.0 - h[M:])
            grad[M:] -= pm * h[:n - M]
    return dx * grad

def gradient_descent(h, n, beta, lr, momentum, max_iters, tlimit, beta_anneal=False):
    """Momentum gradient descent on the smoothed objective with sum constraint."""
    h_cur = h.copy()
    v = np.zeros(n)
    c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5
    current_beta = float(beta)
    
    for it in range(max_iters):
        if tlimit() < 3.0:
            break
        if beta_anneal and it % 500 == 0 and it > 0:
            current_beta *= 1.5
        
        g = compute_smoothed_grad(h_cur, n, current_beta)
        # Project gradient: remove component in direction of all-ones (sum constraint)
        g = g - np.mean(g)
        
        v = momentum * v - lr * g
        h_cur = h_cur + v
        h_cur = project_feasible(h_cur, n)
        
        c5 = compute_c5(h_cur)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cur.copy()
    
    return best_h, best_c5

def slsqp_run(h0, n, beta, maxiter, focus_lags=None, focus_w=0.0):
    global _G_BETA, _G_TGT, _G_DX, _G_PEN, _G_FOCUS_LAGS, _G_FOCUS_W
    _G_BETA = float(beta)
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    _G_PEN = 0.0
    _G_FOCUS_LAGS = focus_lags
    _G_FOCUS_W = focus_w
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    try:
        res = minimize(obj_lob, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def lbfgsb_run(h0, n, beta, maxiter, pen=100.0):
    global _G_BETA, _G_TGT, _G_DX, _G_PEN, _G_FOCUS_LAGS, _G_FOCUS_W
    _G_BETA = float(beta)
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    _G_PEN = float(pen)
    _G_FOCUS_LAGS = None
    _G_FOCUS_W = 0.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_lob, h0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-6, 'iprint': 0, 'maxfun': maxiter * 3})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def find_active_lags(h, n_top=5):
    corr = get_corr(h)
    top_indices = np.argsort(corr)[-n_top:][::-1]
    return [int(k) for k in top_indices]

def multi_lag_optimize(h, n, tlimit, max_rounds=20):
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    
    for rnd in range(max_rounds):
        if tlimit() < 10:
            break
        active_lags = find_active_lags(h_best, n_top=5)
        # Exclude self-correlation (center lag)
        center = n - 1
        focus = [l for l in active_lags if l != center]
        
        for focus_w in [0.5, 1.0]:
            if tlimit() < 8:
                break
            h_new = slsqp_run(h_best, n, 2000, maxiter=300, focus_lags=focus, focus_w=focus_w)
            c_new = compute_c5(h_new)
            if c_new < c5_best - 1e-12:
                c5_best = c_new
                h_best = h_new
    
    return h_best, c5_best

def coord_descent(h, n, max_sweeps, rng, tlimit):
    c5 = compute_c5(h)
    h_best = h.copy()
    
    for sweep in range(max_sweeps):
        if tlimit() < 5:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if tlimit() < 2:
                break
            v0 = h_best[i]
            for v in [0.0, 1.0]:
                if abs(v - v0) < 1e-9:
                    continue
                trial = h_best.copy()
                trial[i] = v
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 1e-10:
                    c5 = c
                    h_best = trial.copy()
                    improved = True
            for d in [0.1, -0.1, 0.05, -0.05, 0.2, -0.2, 0.02, -0.02]:
                if tlimit() < 2:
                    break
                trial = h_best.copy()
                trial[i] = np.clip(trial[i] + d, 0.0, 1.0)
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 1e-10:
                    c5 = c
                    h_best = trial.copy()
                    improved = True
        if not improved and sweep >= 3:
            break
    return h_best, c5
