        print("\n=== Phase 3: Focused lags ===")
        for rnd in range(30):
            if tleft() < 15:
                break
            c5v, _, k_max, corr = compute_c5_full(h_cur)
            top3 = np.argsort(corr)[-3:]
            top5 = np.argsort(corr)[-5:]
            if rnd % 2 == 0:
                h_new = focused_slsqp(h_cur, n, 5000, top3, maxiter=400)
            else:
                h_new = focused_slsqp(h_cur, n, 3000, top5, maxiter=400)
            c = compute_c5(h_new)
            if c < best_c5 - 1e-13:
                best_c5 = c
                best_h = h_new.copy()
                h_cur = h_new.copy()
                print("  Focus {}: C5={:.8f}".format(rnd, c))
            elif c < c5v - 1e-13:
                h_cur = h_new.copy()

    # Phase 4: Higher resolution
    if tleft() > 180:
        print("\n=== Phase 4: Higher res ===")
        for n_try in [512, 1024]:
            if tleft() < 80:
                break
            h_up = project_feasible(interp_h(best_h, best_n, n_try), n_try)
            c_up = compute_c5(h_up)
            print("  n={} interp: C5={:.8f}".format(n_try, c_up))
            h_cur2 = h_up.copy()
            for beta in [100, 500, 2000, 8000]:
                if tleft() < 30:
                    break
                if n_try <= 512:
                    h_new = slsqp_opt(h_cur2, n_try, beta, maxiter=400)
                else:
                    h_new = lbfgsb_opt(h_cur2, n_try, beta, maxiter=1500, pen=100.0)
                    h_new = slsqp_opt(h_new, n_try, beta, maxiter=200)
                c = compute_c5(h_new)
                if c < c_up - 1e-12:
                    c_up = c
                    h_cur2 = h_new.copy()
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    best_h = h_new.copy()
                    best_n = n_try
                    print("    n={} beta={}: C5={:.8f} BEST!".format(n_try, beta, c))
            if c_up < best_c5:
                best_c5 = c_up
                best_h = h_cur2.copy()
                best_n = n_try

    # Phase 5: Coordinate descent at best resolution
    if tleft() > 50:
        print("\n=== Phase 5: Coord descent n={} ===".format(best_n))
        best_h, best_c5 = coord_descent(best_h, best_n, max_sweeps=50, tcheck=tleft)
        print("  CD: C5={:.8f}".format(best_c5))

    # Phase 6: Multi-start perturbation
    if tleft() > 100:
        print("\n=== Phase 6: Multi-start ===")
        n = best_n
        for attempt in range(8):
            if tleft() < 25:
                break
            h_p = best_h.copy()
            n_pert = np.random.randint(3, 10)
            idxs = np.random.choice(n, size=n_pert, replace=False)
            for idx in idxs:
                h_p[idx] = np.clip(h_p[idx] + np.random.randn() * 0.25, 0.0, 1.0)
            h_p = project_feasible(h_p, n)
            c_p = compute_c5(h_p)
            if c_p < best_c5 * 1.05:
                h_cur_p = h_p.copy()
                for beta in [500, 3000, 10000]:
                    if tleft() < 10:
                        break
                    h_new = slsqp_opt(h_cur_p, n, beta, maxiter=300)
                    c = compute_c5(h_new)
                    if c < best_c5 - 1e-13:
                        best_c5 = c
                        best_h = h_new.copy()
                        h_cur_p = h_new.copy()
                        print("  Perturb {}: C5={:.8f}".format(attempt, c))
                    else:
                        h_cur_p = h_new

    # Phase 7: Final ultra-fine convergence
    if tleft() > 30:
        print("\n=== Phase 7: Final convergence ===")
        n = best_n
        h_cur = best_h.copy()
        for beta in [5000, 20000, 60000, 150000]:
            if tleft() < 12:
                break
            h_new, _ = pg_optimize(h_cur, n, beta, max_iter=15000)
            c = compute_c5(h_new)
            if c < best_c5 - 1e-14:
                best_c5 = c
                best_h = h_new.copy()
                h_cur = h_new.copy()
                print("  Fin PG beta={}: C5={:.8f}".format(beta, c))
            else:
                h_cur = h_new

    # Phase 8: Final focused + CD
    if tleft() > 40:
        print("\n=== Phase 8: Final focus ===")
        n = best_n
        for _ in range(10):
            if tleft() < 8:
                break
            c5v, _, _, corr = compute_c5_full(best_h)
            top2 = np.argsort(corr)[-2:]
            h_new = focused_slsqp(best_h, n, 50000, top2, maxiter=400)
            c = compute_c5(h_new)
            if c < best_c5 - 1e-14:
                best_c5 = c
                best_h = h_new
        if tleft() > 15:
            best_h, best_c5 = coord_descent(best_h, n, max_sweeps=20, tcheck=tleft)

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
