    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final fixup
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

def sum_constraint(h):
    return np.sum(h) - len(h) / 2.0

def make_smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    # log-sum-exp for stability
    s = corr - m
    max_s = np.max(s)
    lse = m + max_s + np.log(np.sum(np.exp(s - max_s))) / beta
    return lse * dx

def slsqp_optimize(h, n, beta, maxiter=200, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    
    def obj(x):
        return make_smooth_obj(x, beta)
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter,
                                                   'ftol': ftol, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def binary_anneal(h, n, rng, max_iters=5000):
    """Simulated annealing on binary flips while maintaining sum constraint."""
    c5 = compute_c5(h)
    h_best = h.copy()
    h_cur = h.copy()
    T = 0.05
    T_min = 0.001
    cooling = 0.999
    
    for it in range(max_iters):
        if remaining_time() < 5.0:
            break
        # Flip a random 1 to 0 and a random 0 to 1 (maintain sum)
        ones = np.where(h_cur > 0.5)[0]
        zeros = np.where(h_cur < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        if i == j:
            continue
        
        h_cur[i] = 0.0
        h_cur[j] = 1.0
        c5_new = compute_c5(h_cur)
        delta = c5_new - c5
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            c5 = c5_new
            if c5_new < compute_c5(h_best) - 1e-12:
                h_best = h_cur.copy()
        else:
            h_cur[i] = 1.0
            h_cur[j] = 0.0
        
        T = max(T * cooling, T_min)
    
    return h_best, compute_c5(h_best)

def block_optimize(h, n, block_size=16, n_blocks=20):
    """Optimize blocks of variables sequentially."""
    c5_best = compute_c5(h)
    h_best = h.copy()
    rng = np.random.RandomState(42)
    
    for _ in range(n_blocks):
        if remaining_time() < 5.0:
            break
        start = rng.randint(0, max(1, n - block_size))
        block = slice(start, start + block_size)
        
        # Optimize this block using SLSQP
        h_block = h_best[block].copy()
        
        def block_obj(x):
            h_trial = h_best.copy()
            # Maintain sum constraint
            fixed_sum = np.sum(h_best) - np.sum(h_block)
            h_block_adjusted = x + (block_size/2.0 - np.sum(x)) / block_size  # rough adjustment
            h_block_adjusted = np.clip(h_block_adjusted, 0, 1)
            h_trial[block] = h_block_adjusted
            h_trial = project_feasible(h_trial, n)
            return make_smooth_obj(h_trial, 500.0)
        
        bounds = [(0.0, 1.0)] * block_size
        try:
            res = minimize(block_obj, h_block, method='SLSQP', bounds=bounds,
                          options={'maxiter': 50, 'ftol': 1e-12, 'iprint': 0})
            h_trial = h_best.copy()
            h_trial[block] = res.x
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-13:
                c5_best = c5_trial
                h_best = h_trial.copy()
        except Exception:
            pass
    
    return h_best, c5_best

        try:
            res = minimize(block_obj, h_blk, method='SLSQP', bounds=bounds,
                          options={'maxiter': 40, 'ftol': 1e-12, 'iprint': 0})
            h_trial = h_best.copy()
            h_trial[block] = res.x
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-14:
                c5_best = c5_trial
                h_best = h_trial.copy()
        except Exception:
            pass

    return h_best, c5_best


def coordinate_descent(h, n, max_sweeps=8, rng=None):
    c5_best = compute_c5(h)
    h_best = h.copy()
    if rng is None:
        rng = np.random.RandomState(99)

    for sweep in range(max_sweeps):
        if remaining_time() < 5.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if remaining_time() < 3.0:
                break
            for dv in [0.0, 1.0]:
                if abs(h_best[i] - dv) < 1e-8:
                    continue
                h_trial = h_best.copy()
                h_trial[i] = dv
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < c5_best - 1e-13:
                    c5_best = c5_t
                    h_best = h_trial.copy()
                    any_imp = True
            for dv in [0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.2, -0.2]:
                h_trial = h_best.copy()
                h_trial[i] = np.clip(h_trial[i] + dv, 0.0, 1.0)
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < c5_best - 1e-13:
                    c5_best = c5_t
                    h_best = h_trial.copy()
                    any_imp = True
        if not any_imp and sweep >= 2:
            break

    return h_best, c5_best


def perturb_and_optimize(h, n, rng, num_pert=10):
    c5_best = compute_c5(h)
    h_best = h.copy()

    for p in range(num_pert):
        if remaining_time() < 10.0:
            break
        h_trial = h_best.copy()
        pt = p % 5
        if pt == 0:
            # Flip a segment
            start = rng.randint(0, max(1, n - 10))
            L = rng.randint(4, min(20, n // 4))
            h_trial[start:start + L] = 1.0 - h_trial[start:start + L]
        elif pt == 1:
            # Swap two segments
            s1 = rng.randint(0, n // 3)
            s2 = s1 + rng.randint(n // 4, max(n // 4 + 1, n - n // 3))
            L = rng.randint(3, min(10, n // 10))
            a = h_trial[s1:s1 + L].copy()
            h_trial[s1:s1 + L] = h_trial[s2:s2 + L]
            h_trial[s2:s2 + L] = a
        elif pt == 2:
            # Gaussian noise
            h_trial += 0.05 * rng.randn(n)
        elif pt == 3:
            # Random subset reassignment
            subset = rng.choice(n, size=n // 5, replace=False)
            h_trial[subset] = rng.rand(len(subset))
        else:
            # Sinusoidal perturbation
            freq = rng.randint(2, max(3, n // 6))
            phase = rng.rand() * 2 * np.pi
            t = np.arange(n) / float(n)
            h_trial += 0.06 * np.sin(2.0 * np.pi * freq * t + phase)

        h_trial = project_feasible(h_trial, n)
        c5_t = compute_c5(h_trial)

        # Quick SLSQP from perturbed point
        for beta in [200, 1000]:
            if remaining_time() < 5.0:
                break
            h_opt, c5_opt = slsqp_polish(h_trial, n, beta, maxiter=80, ftol=1e-13)
            if c5_opt < c5_t:
                h_trial = h_opt
                c5_t = c5_opt

        if c5_t < c5_best - 1e-12:
            c5_best = c5_t
            h_best = h_trial.copy()

    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n_init = 256
        best_h = np.zeros(n_init)
        for i in range(n_init):
            cnt = bin(i).count('1')
            best_h[i] = 1.0 if cnt % 2 == 1 else 0.0
        best_h = project_feasible(best_h, n_init)

    best_n = len(best_h)
    best_c5 = compute_c5(best_h)
    print("Initial: n={}, C5={:.10f}".format(best_n, best_c5))

    # Phase 1: SLSQP polish at original resolution
    print("\n--- Phase 1: SLSQP polish (n={}) ---".format(best_n))
    for beta in [50, 200, 1000, 5000]:
        if remaining_time() < 15.0:
            break
        h_new, c5_new = slsqp_polish(best_h, best_n, beta, maxiter=300, ftol=1e-14)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new
            print("  beta={}: C5={:.10f}".format(beta, best_c5))
    print("  Phase 1 done: C5={:.10f}".format(best_c5))

    # Phase 2: Subgradient optimization
    print("\n--- Phase 2: Subgradient ---")
    h_sub, c5_sub = subgrad_optimize(best_h, best_n, max_iter=300, step0=0.08)
    if c5_sub < best_c5 - 1e-13:
        best_c5 = c5_sub
        best_h = h_sub
    print("  Phase 2 done: C5={:.10f}".format(best_c5))

    # Phase 3: Binary annealing
    print("\n--- Phase 3: Binary annealing ---")
    h_bin, c5_bin = binary_anneal(best_h, best_n, rng, max_iters=4000)
    if c5_bin < best_c5 - 1e-13:
        best_c5 = c5_bin
        best_h = h_bin
        print("  Binary improved: C5={:.10f}".format(best_c5))
    else:
        print("  Binary: C5={:.10f} (no improvement)".format(c5_bin))

    # Phase 4: Perturbation exploration
    print("\n--- Phase 4: Perturbation exploration ---")
    best_h, best_c5 = perturb_and_optimize(best_h, best_n, rng, num_pert=12)
    print("  Phase 4 done: C5={:.10f}".format(best_c5))

    # Phase 5: Upsample to 512 and optimize
    print("\n--- Phase 5: Upsample to 512 ---")
    if remaining_time() > 60.0:
        n512 = 512
        h512 = interp_to(best_h, best_n, n512)
        h512 = project_feasible(h512, n512)
        c5512 = compute_c5(h512)
        print("  Upsampled: C5={:.10f}".format(c5512))

        for beta in [50, 200, 1000, 5000]:
            if remaining_time() < 20.0:
                break
            h512, c5512 = slsqp_polish(h512, n512, beta, maxiter=200, ftol=1e-14)
            print("  beta={}: C5={:.10f}".format(beta, c5512))

        # Subgradient at 512
        if remaining_time() > 20.0:
            h512, c5512 = subgrad_optimize(h512, n512, max_iter=200, step0=0.05)
            print("  Subgrad: C5={:.10f}".format(c5512))

        if c5512 < best_c5 - 1e-10:
            best_c5 = c5512
            best_h = h512.copy()
            best_n = n512
            print("  512 improved! C5={:.10f}".format(best_c5))
        else:
            print("  512: no improvement (C5={:.10f})".format(c5512))

    # Phase 6: Upsample to 1024
    print("\n--- Phase 6: Upsample to 1024 ---")
    if remaining_time() > 80.0:
        n1024 = 1024
        h1024 = interp_to(best_h, best_n, n1024)
        h1024 = project_feasible(h1024, n1024)
        c51024 = compute_c5(h1024)
        print("  Upsampled: C5={:.10f}".format(c51024))

        for beta in [50, 200, 1000]:
            if remaining_time() < 25.0:
                break
            h1024, c51024 = slsqp_polish(h1024, n1024, beta, maxiter=150, ftol=1e-14)
            print("  beta={}: C5={:.10f}".format(beta, c51024))

        if remaining_time() > 20.0:
            h1024, c51024 = subgrad_optimize(h1024, n1024, max_iter=150, step0=0.04)
            print("  Subgrad: C5={:.10f}".format(c51024))

        if c51024 < best_c5 - 1e-10:
            best_c5 = c51024
            best_h = h1024.copy()
            best_n = n1024
            print("  1024 improved! C5={:.10f}".format(best_c5))
        else:
            print("  1024: no improvement (C5={:.10f})".format(c51024))

    # Phase 7: Block optimization at best resolution
    print("\n--- Phase 7: Block optimization (n={}) ---".format(best_n))
    if remaining_time() > 30.0:
        best_h, best_c5 = block_optimize(best_h, best_n, block_size=20, n_rounds=40)
        print("  Block opt: C5={:.10f}".format(best_c5))

    # Phase 8: Another SLSQP round with high beta
    print("\n--- Phase 8: High-beta SLSQP (n={}) ---".format(best_n))
    if remaining_time() > 20.0:
        for beta in [5000, 20000, 100000]:
            if remaining_time() < 10.0:
                break
            h_new, c5_new = slsqp_polish(best_h, best_n, beta, maxiter=300, ftol=1e-15)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new
                print("  beta={}: C5={:.10f}".format(beta, best_c5))

    # Phase 9: Coordinate descent
    print("\n--- Phase 9: Coordinate descent ---")
    if remaining_time() > 10.0:
        best_h, best_c5 = coordinate_descent(best_h, best_n, max_sweeps=6, rng=rng)
        print("  Coord descent: C5={:.10f}".format(best_c5))

    # Phase 10: Final subgradient + SLSQP
    print("\n--- Phase 10: Final polish ---")
    if remaining_time() > 15.0:
        h_new, c5_new = subgrad_optimize(best_h, best_n, max_iter=150, step0=0.03)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new
        if remaining_time() > 8.0:
            for beta in [10000, 50000]:
                if remaining_time() < 5.0:
                    break
                h_new, c5_new = slsqp_polish(best_h, best_n, beta, maxiter=200, ftol=1e-15)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    best_h = h_new
        print("  Final polish: C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
