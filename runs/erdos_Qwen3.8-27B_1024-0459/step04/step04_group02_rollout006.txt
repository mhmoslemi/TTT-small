  Wait, the second term: h[i-d]*(1-t) - h[i-d]*(1-h[i]) = h[i-d]*(h[i]-t) = -h[i-d]*(t-h[i])
  
  So corr_d changes by: (t-h[i]) * [(1-h[i+d])·I(0≤i≤n-1-d) - h[i-d]·I(d≤i≤n-1)]

- corr_{-d} (d>0):
  - From j=i (if d≤i≤n-1): (t-h[i])(1-h[i-d])
  - From j=i+d (if d≤i+d≤n-1, i.e., 0≤i≤n-1-d): h[i+d]*(h[i]-t) = -h[i+d]*(t-h[i])
  - Total: (t-h[i]) * [(1-h[i-d])·I(d≤i≤n-1) - h[i+d]·I(0≤i≤n-1-d)]

So for each offset d (positive or negative), the change in corr is linear in t (for d≠0) with coefficient:
- For offset d>0: coeff_d = (1-h[i+d])·I(i≤n-1-d) - h[i-d]·I(i≥d)
- For offset -d (d>0): coeff_{-d} = (1-h[i-d])·I(i≥d) - h[i+d]·I(i≤n-1-d)
- For offset 0: the change is (t-h[i])(1-t-h[i]) which is quadratic in t

So the objective as a function of t:
f(t) = max_k (corr[k] + coeff_k * (t - h[i]) + quad_k * (t-h[i])(1-t-h[i]))

where quad_k is non-zero only for k=0 (offset 0).

For k≠0 (all offsets except 0): linear in t
For k=0: quadratic (concave, since (t-h[i])(1-t-h[i]) = (t-h[i]) - (t-h[i])(t+h[i])... let me just write it as t(1-t) - h[i](1-h[i]) which is a concave quadratic in t)

So f(t) = max(g_0(t), g_1(t), g_2(t), ..., g_{2n-2}(t))
where g_0 is concave quadratic and all others are linear.

The minimum of max of these over t ∈ [0,1] can be found by:
1. If the minimum is determined by the linear terms: it's the solution of a 1D LP (find t minimizing max of lines)
2. If the quadratic term is active: the minimum is where the parabola is tangent to the envelope of lines

For practical purposes, I can use bisection:
- The function f(t) is the max of concave and linear functions → it's convex
- So I can use ternary search or bisection on the derivative

Actually, f(t) is the max of a concave function and several linear functions. The max of any collection of functions... hmm, the max of concave functions is not necessarily convex. But the max of linear functions is convex, and adding a concave function on top...

Let me just use ternary search. It's guaranteed to find the minimum of a unimodal function. Is f(t) unimodal? 

f(t) = max(a*t² + b*t + c, line_1(t), line_2(t), ...)
where a = -1 (concave quadratic) and the lines are linear.

The maximum of a concave parabola and several lines... this should be unimodal (or at least the minimum is unique) because:
- The lines form a convex piecewise-linear function (their max)
- We're taking the max of that with a concave parabola

import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(2000):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = int(np.argmin(h))
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = int(np.argmax(h))
                h[idx] = max(0.0, h[idx] + diff)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-14:
            for j in range(n):
                if h[j] > 1e-14 and h[j] < 1.0 - 1e-14:
                    h[j] = np.clip(h[j] + diff2, 0.0, 1.0)
                    break
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def smooth_max_over_active(corr, active_idx, beta):
    """Smooth max over a subset of correlation lags."""
    subset = corr[active_idx]
    m = np.max(subset)
    shifted = np.minimum(beta * (subset - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    if wsum < 1e-300:
        wsum = 1.0
        weights = np.ones_like(exp_vals) / len(exp_vals)
    else:
        weights = exp_vals / wsum
    val = m + np.log(wsum) / beta
    return val, weights

def obj_active(x, n, active_idx, beta):
    """Objective: smooth max of active lags * dx."""
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    val, _ = smooth_max_over_active(corr, active_idx, beta)
    return val * dx

def obj_active_grad(x, n, active_idx, beta):
    """Gradient of smooth max over active lags."""
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    val, weights = smooth_max_over_active(corr, active_idx, beta)
    grad = np.zeros(n)
    for j, k in enumerate(active_idx):
        w = weights[j]
        if w < 1e-14:
            continue
        offset = int(k) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return grad

def slsqp_full(h, n, beta, maxiter=300):
    """SLSQP with smooth max over ALL lags."""
    dx = 2.0 / n

    def _obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        ev = np.exp(shifted)
        ws = np.sum(ev)
        if ws < 1e-300:
            ws = 1.0
            wt = np.ones_like(ev) / len(ev)
        else:
            wt = ev / ws
        return (m + np.log(ws) / beta) * dx

    def _obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        ev = np.exp(shifted)
        ws = np.sum(ev)
        if ws < 1e-300:
            ws = 1.0
            wt = np.ones_like(ev) / len(ev)
        else:
            wt = ev / ws
        grad = np.zeros(n)
        for k in range(len(corr)):
            w = wt[k]
            if w < 1e-14:
                continue
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                if npairs > 0:
                    grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                    grad[:npairs] -= w * x[offset:offset + npairs]
            else:
                aoff = -offset
                npairs = n - aoff
                if npairs > 0:
                    grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                    grad[aoff:aoff + npairs] -= w * x[:npairs]
        grad *= dx
        return grad

    def _cons(x):
        return np.sum(x) - n / 2.0

    def _cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h, jac=_obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            if compute_c5(trial) <= compute_c5(h) + 1e-15:
                return trial
    except Exception:
        pass
    return h.copy()

def slsqp_active(h, n, active_idx, beta, maxiter=200):
    """SLSQP with smooth max over active lags only."""
    dx = 2.0 / n

    def _obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        val, _ = smooth_max_over_active(corr, active_idx, beta)
        return val * dx

    def _obj_grad(x):
        return obj_active_grad(x, n, active_idx, beta)

    def _cons(x):
        return np.sum(x) - n / 2.0

    def _cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h, jac=_obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            if compute_c5(trial) <= compute_c5(h) + 1e-15:
                return trial
    except Exception:
        pass
    return h.copy()

def active_set_optimize(h, n, max_rounds=10, n_active=30, beta_start=500, beta_end=5000):
    """Iterative active-set optimization."""
    best = h.copy()
    best_c5 = compute_c5(best)
    
    for rnd in range(max_rounds):
        if remaining_time() < 15:
            break
        corr = compute_corr(best)
        max_corr = np.max(corr)
        # Find active lags
        threshold = max_corr - 0.5
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        if len(active) > n_active:
            active = np.argsort(corr)[::-1][:n_active]
        
        beta = int(beta_start * (beta_end / beta_start) ** (rnd / max(1, max_rounds - 1)))
        beta = max(100, min(beta, 20000))
        
        h_new = slsqp_active(best, n, active, beta, 200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-16:
            best = h_new
            best_c5 = c5_new
    
    return best, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

            # For offset d != 0:

def pairwise_swap_local(h, n, max_passes=5):
    """Try swapping pairs of values (toward binary) to reduce C5."""
    best = h.copy()
    best_c5 = compute_c5(best)
    rng = np.random.RandomState(123)
    
    for p in range(max_passes):
        if remaining_time() < 5:
            break
        # Find pairs that could benefit from swap
        # Points near 0 and 1 are "almost binary"
        lows = np.where(best < 0.3)[0]
        highs = np.where(best > 0.7)[0]
        if len(lows) < 2 or len(highs) < 2:
            break
        improved = False
        for _ in range(50):
            if remaining_time() < 2:
                break
            i = rng.choice(lows)
            j = rng.choice(highs)
            # Swap: h[i] -> 1-h[i], h[j] -> 1-h[j] (then reproject)
            trial = best.copy()
            trial[i] = 1.0 - best[i]
            trial[j] = 1.0 - best[j]
            trial = project_feasible(trial, n)
            c5_trial = compute_c5(trial)
            if c5_trial < best_c5 - 1e-15:
                best = trial
                best_c5 = c5_trial
                improved = True
        if not improved:
            break
    return best, best_c5

def binary_init(n, rng):
    """Create a random binary-like initialization."""
    h = np.zeros(n)
    positions = rng.choice(n, n // 2, replace=False)
    h[positions] = 1.0
    return project_feasible(h, n)

def structured_init(n, rng):
    """Create a structured (Thue-Morse-like) initialization."""
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1') & 1
        h[i] = 1.0 if cnt == 1 else 0.0
    # Perturb slightly
    h += rng.uniform(-0.1, 0.1, n)
    return project_feasible(h, n)

def block_init(n, rng):
    """Create a block-structured initialization."""
    n_blocks = rng.randint(4, 20)
    h = np.zeros(n)
    block_size = n // n_blocks
    for b in range(n_blocks):
        if b % 2 == 0:
            h[b*block_size:(b+1)*block_size] = 1.0
    # Perturb
    h += rng.uniform(-0.15, 0.15, n)
    return project_feasible(h, n)

            h_init = structured_init(tn, rng)
        else:
            h_init = block_init(tn, rng)
        
        c5_init = compute_c5(h_init)
        if c5_init > 0.45:
            continue  # Too bad a starting point
        
        # Optimize this init
        for beta in [300, 1500, 5000]:
            if remaining_time() < 15:
                break
            h_new = slsqp_full(h_init, tn, beta, 200)
            c5_new = compute_c5(h_new)
            if c5_new < c5_init:
                h_init = h_new
                c5_init = c5_new
            if c5_init < best_c5:
                best_c5 = c5_init
                best_h = h_init.copy()
                best_n = tn
        
        if remaining_time() > 15:
            h_new, c5_new = active_set_optimize(h_init, tn, max_rounds=5, n_active=25)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = tn

    # Phase 6: Final coordinate descent
    if remaining_time() > 20:
        h_new, c5_new = coord_descent_exact(best_h, best_n, max_sweeps=25)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 7: High-precision SLSQP polish
    if remaining_time() > 15:
        for beta in [10000, 30000]:
            if remaining_time() < 8:
                break
            h_new = slsqp_full(best_h, best_n, beta, 150)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 8: Final coordinate descent with more sweeps
    if remaining_time() > 15:
        h_new, c5_new = coord_descent_exact(best_h, best_n, max_sweeps=30)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 9: Pairwise swaps (try to push toward binary)
    if remaining_time() > 10:
        h_new, c5_new = pairwise_swap_local(best_h, best_n, max_passes=3)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 10: Final active-set polish
    if remaining_time() > 10:
        h_new, c5_new = active_set_optimize(best_h, best_n, max_rounds=5, n_active=20,
                                            beta_start=5000, beta_end=50000)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final coordinate descent
    if remaining_time() > 5:
        h_new, c5_new = coord_descent_exact(best_h, best_n, max_sweeps=15)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    # Evaluate with the harness
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
