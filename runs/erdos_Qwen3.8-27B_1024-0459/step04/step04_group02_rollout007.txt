            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
    
    # Phase 5: Final coordinate descent
    if remaining_time() > 20:
        h_cd2, c5_cd2 = exact_coord_descent(best_h, best_n, max_sweeps=40, top_m=30)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
    
    # Phase 6: Small perturbation + re-optimize (escape local optima)
    if remaining_time() > 60:
        for pert_trial in range(5):
            if remaining_time() < 20:
                break
            h_pert = best_h + rng.normal(0, 0.01, size=best_n)
            h_pert = project_feasible(h_pert, best_n)
            c5_pert = compute_c5(h_pert)
            
            if c5_pert < best_c5 + 0.01:  # Not too far off
                for beta in [500, 5000, 50000]:
                    if remaining_time() < 10:
                        break
                    h_pert = lbfgs_optimize(h_pert, best_n, beta, 200.0, maxiter=150)
                    c5_pert = compute_c5(h_pert)
                    if c5_pert < best_c5:
                        best_c5 = c5_pert
                        best_h = h_pert.copy()
                
                if remaining_time() > 15:
                    h_pert_cd, c5_pert_cd = exact_coord_descent(h_pert, best_n, max_sweeps=15, top_m=20)
                    if c5_pert_cd < best_c5:
                        best_c5 = c5_pert_cd
                        best_h = h_pert_cd.copy()
    
    # Phase 7: Try a binary-structured start at a different resolution
    if remaining_time() > 80:
        try_n = 1280
        if try_n % 2 != 0:
            try_n += 1
        # Thue-Morse at try_n
        h_tm = np.zeros(try_n)
        for i in range(try_n):
            v = i
            cnt = 0
            while v:
                cnt ^= (v & 1)
                v >>= 1
            h_tm[i] = 1.0 if cnt == 1 else 0.0
        h_tm = project_feasible(h_tm, try_n)
        c5_tm = compute_c5(h_tm)
        
        if c5_tm < 0.45:
            for beta in [200, 2000, 20000]:
                if remaining_time() < 15:
                    break
                h_tm = lbfgs_optimize(h_tm, try_n, beta, 100.0, maxiter=200)
                c5_tm = compute_c5(h_tm)
                if c5_tm < best_c5:
                    best_c5 = c5_tm
                    best_h = h_tm.copy()
                    best_n = try_n
            
            if remaining_time() > 20:
                h_tm_cd, c5_tm_cd = exact_coord_descent(h_tm, try_n, max_sweeps=20, top_m=15)
                if c5_tm_cd < best_c5:
                    best_c5 = c5_tm_cd
                    best_h = h_tm_cd.copy()
                    best_n = try_n
    
    # Phase 8: Final micro-polish
    if remaining_time() > 8:
        h_final = best_h.copy()
        h_final = lbfgs_optimize(h_final, best_n, 100000, 1000.0, maxiter=100)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
        
        if remaining_time() > 5:
            h_final_cd, c5_final_cd = exact_coord_descent(best_h, best_n, max_sweeps=15, top_m=30)
            if c5_final_cd < best_c5:
                best_c5 = c5_final_cd
                best_h = h_final_cd.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
