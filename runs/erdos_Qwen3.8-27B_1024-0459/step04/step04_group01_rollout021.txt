Actually, let me reconsider the local search algorithm. The key insight is:

For binary h, a swap (i→0, j→1) changes count(d) by:
Δ(d) = h[i+d] - h[j+d] + h[i-d] - h[j-d]

The new max = max_d (curr_corr[n-1+d] + Δ(d))

A swap is "good" if it reduces the max. The simplest check:
- For the current max shift d*, we need Δ(d*) < 0 (or at least small)
- For all other shifts, curr_corr[n-1+d] + Δ(d) should not exceed the old max

The most efficient way to check this:
1. Compute Δ for the top-K shifts
2. If any top shift worsens significantly, reject
3. For the rest, the worst case is curr_corr[d] + |Δ(d)|, and |Δ(d)| ≤ 4 (max for binary)
4. If curr_max - curr_corr[d] > 4 for most d, then only the top few need checking

This allows very fast rejection of bad swaps.

        if not found_improvement:
            # Try random kick + recompute
            if rng.rand() < 0.3:
                ones = np.where(h_b == 1)[0]
                zeros = np.where(h_b == 0)[0]
                for _ in range(rng.randint(1, 3)):
                    i = int(ones[rng.randint(len(ones))])
                    j = int(zeros[rng.randint(len(zeros))])
                    if i != j:
                        h_b[i], h_b[j] = h_b[j], h_b[i]
                corr = compute_corr(h_b)
                cur_max = float(np.max(corr))
            else:
                # No improvement found, slightly reduce time per iteration
                pass
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
    except Exception:
        parent_n = 312
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
    
    # Phase 1: Quick SLSQP polish at parent (20s)
    if remaining() > 25:
        h_new = slsqp_smooth(best_h, best_n, 2000, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 2: Binary local search at parent resolution (150s)
    if remaining() > 160:
        t_alloc = min(150.0, remaining() - 50.0)
        h_bin, c5_bin = binary_local_search_structured(
            best_h, best_n, time_limit=t_alloc, rng=rng, top_k=12, T0=1.0, T1=0.1)
        if c5_bin < best_c5:
            best_h = h_bin
            best_c5 = c5_bin
            best_n = best_n
    
    # Phase 3: SLSQP on best binary (15s)
    if remaining() > 20:
        h_new = slsqp_smooth(best_h, best_n, 5000, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 4: Try higher resolutions
    n_candidates = []
    for nn in [best_n + 60, best_n + 120, best_n + 200, 500, 600, 700, 800]:
        if nn >= 100 and nn <= 900 and nn % 2 == 0 and nn != best_n:
            n_candidates.append(nn)
    n_candidates = sorted(set(n_candidates))
    
    for n_t in n_candidates:
        if remaining() < 60:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = make_binary(h_interp, n_t, rng)
        
        # Quick binary search
        t_alloc = min(100.0, remaining() - 30.0)
        if t_alloc < 20:
            continue
        h_bin_t, c5_bin_t = binary_local_search_structured(
            h_bin_t, n_t, time_limit=t_alloc, rng=rng, top_k=12, T0=0.8, T1=0.05)
        
        # SLSQP polish
        if remaining() > 20:
            h_pol = slsqp_smooth(h_bin_t, n_t, 3000, maxiter=300)
            c5_pol = compute_c5(h_pol)
            if c5_pol < c5_bin_t:
                h_bin_t = h_pol
                c5_bin_t = c5_pol
        
        if c5_bin_t < best_c5:
            best_h = h_bin_t
            best_c5 = c5_bin_t
            best_n = n_t
    
    # Phase 5: More binary search at best resolution
    if remaining() > 40:
        t_alloc = min(120.0, remaining() - 20.0)
        h_bin, c5_bin = binary_local_search_structured(
            best_h, best_n, time_limit=t_alloc, rng=rng, top_k=15, T0=0.5, T1=0.05)
        if c5_bin < best_c5:
            best_h = h_bin
            best_c5 = c5_bin
    
    # Phase 6: Focused search
    if remaining() > 30:
        t_alloc = min(60.0, remaining() - 15.0)
        h_bin, c5_bin = binary_focused_search(
            best_h, best_n, time_limit=t_alloc, rng=rng)
        if c5_bin < best_c5:
            best_h = h_bin
            best_c5 = c5_bin
    
    # Phase 7: Final SLSQP polish
    if remaining() > 10:
        for beta in [5000, 15000, 30000]:
            if remaining() < 5:
                break
            h_new = slsqp_smooth(best_h, best_n, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    
    # Phase 8: One more binary pass
    if remaining() > 15:
        h_bin, c5_bin = binary_local_search_structured(
            best_h, best_n, time_limit=min(30.0, remaining() - 8.0), rng=rng, top_k=10, T0=0.3, T1=0.02)
        if c5_bin < best_c5:
            best_h = h_bin
            best_c5 = c5_bin
    
    # Final SLSQP
    if remaining() > 6:
        h_new = slsqp_smooth(best_h, best_n, 20000, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Validate and finalize
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
