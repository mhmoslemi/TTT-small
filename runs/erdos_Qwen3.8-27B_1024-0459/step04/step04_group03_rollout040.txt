            idx = np.arange(i_first_start, i_first_end + 1)
            grad[idx] += (1 - h[idx - l])
        
        # Second term: -h[i+l] for i in [i_second_start, i_second_end]
        if i_second_start >= 0 and i_second_end < n:
            idx = np.arange(i_second_start, i_second_end + 1)
            grad[idx] += -h[idx + l]
    
    grad *= dx / len(active_lags)
    return grad

        grad = compute_subgradient(h, active, dx)
        
        # Momentum update
        momentum = beta * momentum - alpha * grad
        h_new = h + momentum
        h_new = project_feasible(h_new, n)
        
        # Check improvement
        c5_new = compute_c5(h_new)
        c5_old = compute_c5(h)
        if c5_new < c5_old:
            h = h_new
        else:
            alpha *= 0.9
        
        if it % 100 == 0 and alpha < 0.001:
            break
    
    return h

**Alternative: Direct optimization of the max using bisection on t**

The problem is: min_h max_k c[k]*dx subject to constraints.
This is equivalent to: find the smallest t such that there exists h with all c[k]*dx ≤ t.

For a fixed t, the feasibility problem is:
Find h in [0,1]^n with sum(h) = n/2 and c[k]*dx ≤ t for all k.

This is a system of quadratic inequalities (since c[k] is quadratic in h). It's non-convex, so bisection on t doesn't directly give a convex feasibility problem.

However, if we linearize: for a given h, c[k] is linear in each h[i] individually. So we can use an iterative linearization approach (similar to the method of conditional gradients / Frank-Wolfe).

**Frank-Wolfe approach:**
1. Start with h0
2. At each iteration, compute the gradient of the objective (or subgradient)
3. Solve the linear minimization: min_{h in feasible} <grad, h>
   The feasible set is {h: 0 ≤ h[i] ≤ 1, sum(h) = n/2}
   The linear minimization over this set: put as much as possible in the coordinates with the most negative gradient, and as little as possible in those with the most positive gradient, while maintaining the sum constraint.
4. Take a step towards the linear minimizer
5. Repeat

This is attractive because step 3 is easy (sort by gradient, fill greedily).

        _LAGS_G = active
        x0 = np.concatenate([h, [t_init]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': _mm_ineq},
            {'type': 'eq', 'fun': _mm_eq}
        ]
        try:
            res = minimize(_mm_obj, x0, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_slsqp_iter, 'ftol': 1e-14, 'iprint': 0})
            if res.x is not None:
                h_new = project_feasible(res.x[:n], n)
                c5_new = compute_c5(h_new)
                c5_old = compute_c5(h)
                if c5_new < c5_old - 1e-16:
                    h = h_new
                else:
                    if rnd > 3:
                        break
        except Exception:
            break
    return h

def simulated_annealing_binary(h_init, n, max_iters=5000, rng=None):
    if rng is None:
        rng = np.random.RandomState(77)
    h = h_init.copy()
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    T = 0.05
    T_min = 0.001
    cooling = 0.997
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    h[ones] = 1.0
    h[zeros] = 0.0
    h = project_feasible(h, n)
    cur_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        if T < T_min:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cooling
    h_final = best_h.copy()
    h_final = project_feasible(h_final, n)
    return h_final, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S, _N_G, _BETA_G, _LAGS_G, _DX_G
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 256
        best_h = np.full(best_n, 0.5)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: Subgradient descent with momentum
    if remaining_time() > 60:
        h_sg, c5_sg = subgradient_descent(h, n, max_iters=2000, alpha0=0.04, momentum_coef=0.88)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 2: Frank-Wolfe
    if remaining_time() > 80:
        h_fw, c5_fw = frank_wolfe_method(h, n, max_iters=1500)
        if c5_fw < best_c5:
            best_c5 = c5_fw
            best_h = h_fw.copy()
            h = h_fw.copy()

    # Phase 3: Pair coordinate descent (continuous)
    if remaining_time() > 60:
        h_cd, c5_cd = pair_cd_continuous(h, n, max_sweeps=15, window=35, n_trials=13)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Phase 4: Simulated annealing on binary assignment
    if remaining_time() > 120:
        h_sa, c5_sa = simulated_annealing_binary(h, n, max_iters=4000, rng=rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            h = h_sa.copy()
        # After SA, do continuous refinement
        if remaining_time() > 30:
            h_cd2, c5_cd2 = pair_cd_continuous(h, n, max_sweeps=10, window=30, n_trials=11)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()
                h = h_cd2.copy()

    # Phase 5: SLSQP smooth optimization
    if remaining_time() > 40:
        for beta in [200, 800, 3000]:
            if remaining_time() < 15:
                break
            h_new = slsqp_smooth(h, n, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 6: Minimax SLSQP
    if remaining_time() > 30:
        h_mm = slsqp_minimax(h, n, max_rounds=12, max_slsqp_iter=200)
        c5_mm = compute_c5(h_mm)
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm.copy()
            h = h_mm.copy()

    # Phase 7: Multi-resolution exploration
    trial_ns = [192, 200, 224, 240, 256, 288, 320, 384, 448, 512]
    for trial_n in trial_ns:
        if remaining_time() < 25:
            break
        if trial_n == best_n:
            continue
        h_interp = interp_to(best_h, best_n, trial_n)
        h_interp = project_feasible(h_interp, trial_n)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5 + 0.005:
            # Try subgradient at new resolution
            if remaining_time() > 20:
                h_sg2, c5_sg2 = subgradient_descent(h_interp, trial_n, max_iters=800, alpha0=0.03, momentum_coef=0.87)
                if c5_sg2 < best_c5:
                    best_c5 = c5_sg2
                    best_h = h_sg2.copy()
                    best_n = trial_n
                    h_interp = h_sg2.copy()
            if remaining_time() > 15:
                h_cd3, c5_cd3 = pair_cd_continuous(h_interp, trial_n, max_sweeps=8, window=30, n_trials=11)
                if c5_cd3 < best_c5:
                    best_c5 = c5_cd3
                    best_h = h_cd3.copy()
                    best_n = trial_n
                    h_interp = h_cd3.copy()
            if remaining_time() > 12 and trial_n <= 384:
                h_new = slsqp_smooth(h_interp, trial_n, 1000, maxiter=200)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = trial_n

    # Phase 8: Second round of SA + refinement at best resolution
    if remaining_time() > 80:
        h_sa2, c5_sa2 = simulated_annealing_binary(best_h, best_n, max_iters=3000, rng=rng)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
        if remaining_time() > 30:
            h_cd4, c5_cd4 = pair_cd_continuous(best_h, best_n, max_sweeps=15, window=40, n_trials=15)
            if c5_cd4 < best_c5:
                best_c5 = c5_cd4
                best_h = h_cd4.copy()

    # Phase 9: Frank-Wolfe again on best solution
    if remaining_time() > 40:
        h_fw2, c5_fw2 = frank_wolfe_method(best_h, best_n, max_iters=1000)
        if c5_fw2 < best_c5:
            best_c5 = c5_fw2
            best_h = h_fw2.copy()

    # Phase 10: Final SLSQP polish
    if remaining_time() > 20:
        for beta in [5000, 15000]:
            if remaining_time() < 8:
                break
            h_new = slsqp_smooth(best_h, best_n, beta, maxiter=250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 11: Final pair CD
    if remaining_time() > 15:
        h_cd5, c5_cd5 = pair_cd_continuous(best_h, best_n, max_sweeps=10, window=35, n_trials=15)
        if c5_cd5 < best_c5:
            best_c5 = c5_cd5
            best_h = h_cd5.copy()

    # Phase 12: Final minimax
    if remaining_time() > 15:
        h_mm2 = slsqp_minimax(best_h, best_n, max_rounds=8, max_slsqp_iter=150)
        c5_mm2 = compute_c5(h_mm2)
        if c5_mm2 < best_c5:
            best_c5 = c5_mm2
            best_h = h_mm2.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
