def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_binary(h, n):
    """Project to binary with exactly n//2 ones."""
    h = (h > 0.5).astype(float)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        # Remove the "least important" ones (those contributing most to max corr)
        rng = np.random.RandomState(0)
        remove = rng.choice(ones, n_ones - target, replace=False)
        h[remove] = 0.0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        rng = np.random.RandomState(0)
        add = rng.choice(zeros, target - n_ones, replace=False)
        h[add] = 1.0
    return h

def simulated_annealing(h, n, time_limit, rng, t_start=1.0, t_end=0.01, alpha=0.999):
    """Simulated annealing on binary sequences."""
    best_h = h.copy()
    best_energy = compute_c5(h)
    cur_h = h.copy()
    cur_energy = best_energy
    corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
    cur_max_corr = float(np.max(corr))
    dx = 2.0 / n
    
    start = time.time()
    t = t_start
    n_ones = n // 2
    ones = np.where(cur_h == 1)[0]
    zeros = np.where(cur_h == 0)[0]
    
    while time.time() - start < time_limit and t > t_end:
        # Select a random 1-to-0 swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Apply swap
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        new_max = float(np.max(new_corr))
        new_energy = new_max * dx
        
        delta = new_energy - cur_energy
        
        if delta < 0 or rng.rand() < np.exp(-delta / t):
            # Accept
            cur_h = cur_h  # already applied
            cur_energy = new_energy
            corr = new_corr
            cur_max_corr = new_max
            # Update ones and zeros
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]

                best_h = cur_h.copy()
        else:
            # Reject
            cur_h[i], cur_h[j] = 1.0, 0.0
        
        t *= alpha

def slsqp_refine(h, n, beta, maxiter=300):
    dx_val = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -50, 50)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_val
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -50, 50)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        n_corr = len(corr)
        for k in range(n_corr):
            if w[k] > 1e-13:
                d = k - (n - 1)
                if d >= 0:
                    npairs = n - d
                    if npairs > 0:
                        g[d:d+npairs] += (1.0 - x[:npairs])
                        g[:npairs] -= x[d:d+npairs]
                else:
                    ao = -d
                    npairs = n - ao
                    if npairs > 0:
                        g[:npairs] += (1.0 - x[ao:ao+npairs])
                        g[ao:ao+npairs] -= x[:npairs]
        return g * dx_val
    
    def sum_fun(x):
        return np.sum(x) - target
    
    def sum_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        h_out = np.clip(res.x, 0.0, 1.0)
        # Project to feasible
        h_out = _project_feasible(h_out, n)
        return h_out
    except Exception:
        return h

def _project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        h_trial = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_trial)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _START) - 5.0)

def optimize_binary_at_n(h_init, n, time_budget, rng):
    """Run a sequence of binary optimization methods on a given n."""
    h = to_binary(h_init, n, rng)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    spent = 0.0
    
    # Phase 1: Greedy worst-shift descent
    t1 = min(60, time_budget * 0.3)
    h1, c51 = greedy_worst_shift(h, n, t1, rng)
    spent += time.time() - (time.time() - t1)
    if c51 < best_c5:
        best_c5 = c51
        best_h = h1.copy()
    
    # Phase 2: Simulated annealing from best
    t2 = min(120, time_budget * 0.4)
    if remaining_time() > t2 + 30:
        h2, c52 = sa_binary(best_h, n, t2, rng, t_start=0.3)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2.copy()
    
    # Phase 3: Full steepest on the worst shift
    t3 = min(90, time_budget * 0.3)
    if remaining_time() > t3 + 10:
        h3, c53 = full_steepest(best_h, n, t3, rng)
        if c53 < best_c5:
            best_c5 = c53
            best_h = h3.copy()
    
    # Phase 4: More SA with lower temperature
    t4 = min(60, time_budget * 0.3)
    if remaining_time() > t4 + 5:
        h4, c54 = sa_binary(best_h, n, t4, rng, t_start=0.1, t_end=0.01)
        if c54 < best_c5:
            best_c5 = c54
            best_h = h4.copy()
    
    # Phase 5: Final greedy
    if remaining_time() > 20:
        h5, c55 = greedy_worst_shift(best_h, n, min(40, remaining_time() - 10), rng)
        if c55 < best_c5:
            best_c5 = c55
            best_h = h5.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 312
        parent_h = random_binary(parent_n, rng)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    # First: quick SLSQP on parent to get a good continuous solution
    parent_n = len(best_h)
    if remaining_time() > 30:
        h_cont = slsqp_refine(best_h, parent_n, 500, maxiter=200)
        c5_cont = compute_c5(h_cont)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()
            best_n = parent_n
    
    # Now focus on binary optimization at various n values
    n_candidates = sorted(set([
        parent_n,
        parent_n + 8, parent_n + 16, parent_n + 32,
        parent_n - 8, parent_n - 16,
        256, 288, 320, 336, 352, 384, 400, 448, 512, 576, 640
    ]))
    n_candidates = [n for n in n_candidates if n >= 128 and n <= 700 and n % 2 == 0]
    
    # Sort by potential: prefer n values where 2/n * expected_min_corr is smallest
    # For now, just try them in order of proximity to parent_n, then larger n
    n_candidates.sort(key=lambda n: (abs(n - parent_n), n))
    
    # Allocate time: give more time to n values near the parent
    total_time = remaining_time()
    n_per_try = max(60, total_time // len(n_candidates))
    
    tried_n = set()
    for n_try in n_candidates:
        if n_try in tried_n:
            continue
        if remaining_time() < 40:
            break
        tried_n.add(n_try)
        
        # Interpolate parent to this n
        h_interp = interp_to(best_h, best_n, n_try)
        h_feas = _project_feasible(h_interp, n_try)
        
        # Quick binary optimization
        t_alloc = min(n_per_try, remaining_time() - 20)
        if t_alloc < 20:
            break
        
        h_opt, c5_opt = optimize_binary_at_n(h_feas, n_try, t_alloc, rng)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_try
    
    # Also try random starts at a few n values (to escape local optima)
    for n_rand in [parent_n, parent_n + 32, 384, 512]:
        if remaining_time() < 50:
            break
        if n_rand in tried_n:
            continue
        tried_n.add(n_rand)
        
        h_rand = random_binary(n_rand, rng)
        # Quick greedy to get a reasonable starting point
        h_g, c5_g = greedy_worst_shift(h_rand, n_rand, min(30, remaining_time() - 15), rng)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
            best_n = n_rand
        
        # SA from random
        if remaining_time() > 40:
            h_sa, c5_sa = sa_binary(h_g, n_rand, min(60, remaining_time() - 20), rng, t_start=0.5)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_rand
    
    # Final: intensive refinement on the best solution found
    if remaining_time() > 60:
        h_final = to_binary(best_h, best_n, rng)
        c5_final = compute_c5(h_final)
        if c5_final > best_c5:
            # The continuous solution is better, keep it
            pass
        else:
            best_c5 = c5_final
            best_h = h_final.copy()
        
        # More greedy + SA on the best
        t_fin = min(120, remaining_time() - 20)
        if t_fin > 30:
            h_f1, c5_f1 = full_steepest(best_h, best_n, t_fin // 2, rng)
            if c5_f1 < best_c5:
                best_c5 = c5_f1
                best_h = h_f1.copy()
        
        if remaining_time() > 30:
            h_f2, c5_f2 = sa_binary(best_h, best_n, min(60, remaining_time() - 10), rng, t_start=0.05, t_end=0.005)
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_f2.copy()
        
        if remaining_time() > 20:
            h_f3, c5_f3 = greedy_worst_shift(best_h, best_n, min(30, remaining_time() - 5), rng)
            if c5_f3 < best_c5:
                best_c5 = c5_f3
                best_h = h_f3.copy()
    
    # Validate
    best_n = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, len(best_h))
