def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

    global _IDX1_G, _IDX2_G, _MASK1_G, _MASK2_G
    shifts = np.arange(2 * n - 1) - (n - 1)
    js = np.arange(n)[:, None].astype(np.int32)
    sh = shifts[None, :].astype(np.int32)
    _IDX1_G = js - sh
    _IDX2_G = js + sh
    _MASK1_G = (_IDX1_G >= 0) & (_IDX1_G < n)

def smooth_max_eval(h, n, beta):
    dx = 2.0 / n
    h1 = 1.0 - h
    corr = np.correlate(h, h1, mode='full')
    max_c = np.max(corr)
    exp_b = np.exp(beta * (corr - max_c))
    sum_exp = np.sum(exp_b)
    obj = (max_c + np.log(sum_exp) / beta) * dx
    w = exp_b / sum_exp
    return obj, w

def compute_grad(h, n, w):

    dx = 2.0 / n
    h1 = 1.0 - h
    term1 = np.where(_MASK1_G, h1[_IDX1_G], 0.0) @ w
    term2 = np.where(_MASK2_G, h[_IDX2_G], 0.0) @ w
    return (term1 - term2) * dx

    obj, _ = smooth_max_eval(h, n, beta)

    best_h = h.copy()
    alpha = alpha_init

    max_c = np.max(corr)
    exp_b = np.exp(beta * (corr - max_c))
    return (max_c + np.log(np.sum(exp_b)) / beta) * dx

def _slsqp_eq(x):
    return np.sum(x) - _N_G / 2.0

def slsqp_refine(h, n, beta, maxiter=200):
    global _N_G, _BETA_G, _DX_G
    _N_G = n
    _BETA_G = beta
    _DX_G = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq}]
    try:
        res = minimize(_slsqp_obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def comb_search(h, n, rng, max_sweeps=15):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_imp = 0

    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        order = rng.permutation(n)
        for ii in order:
            if remaining_time() < 2:
                break
            i = int(ii)
            hi = best_h[i]
            if hi < 0.03 or hi > 0.97:
                continue
            for dj in [-5,-3,-2,-1,1,2,3,5,8,12,18,25,40]:
                j = i + dj
                if j < 0 or j >= n:
                    continue
                hj = best_h[j]
                if abs(hi - hj) < 0.08:
                    continue
                for frac in [0.4, 0.6]:
                    amt = (hi - hj) * frac
                    if abs(amt) < 0.02:
                        continue
                    h_try = best_h.copy()
                    h_try[i] = hi - amt
                    h_try[j] = hj + amt
                    h_try = project_feasible(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        improved = True
                        break
        if not improved:
            no_imp += 1
            if no_imp >= 3:
                break
        else:
            no_imp = 0

    return best_h

def perturb_restart(h, n, rng, sigma=0.03):
    h_p = h.copy()
    mask = (h_p > 0.05) & (h_p < 0.95)
    cnt = np.sum(mask)
    if cnt > 0:
        h_p[mask] += rng.normal(0, sigma, cnt)
    idx = rng.choice(n, size=min(4, n), replace=False)
    for i in idx:
        h_p[i] = 1.0 - h_p[i]
    h_p = project_feasible(h_p, n)
    return h_p
