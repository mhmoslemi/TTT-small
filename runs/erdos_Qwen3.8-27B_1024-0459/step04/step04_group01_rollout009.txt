    """Compute change in all correlations when h[i]:0->1, h[j]:1->0."""

        corr[k + n - 1] = d
    return corr

def coord_descent_h(h0, n, time_limit, rng):
    """Pair coordinate descent for continuous h."""
    h = h0.copy()
    start = time.time()
    best_c5 = compute_c5(h)
    half = n // 2

    for _ in range(10000):

            j = (i + 1) % n

        if hi - lo < 1e-14:
            continue

        # corr_k(t) = corr_k - h[i]*sl_i[k] + t*sl_i[k]
        # where sl_i[k] is slope of c_k w.r.t. h[i]
        # For the pair (i,j): corr_k(t) = corr_k + (t - h[i])*(sl_i[k] - sl_j[k])
        # We need sl_i and sl_j for all k

        d_sl = np.zeros(nc)

            # c_k = sum_n h[n+k]*(1-h[n])

            # More carefully: d/dh[p] c_k = sum over roles p plays
            # p appears as h[n+k] when n = p-k: contributes (1-h[p-k])... no
            # c_k = sum_{valid n} h[n+k]*(1-h[n])
            # d/dh[p]: 
            #   if p = n+k for some valid n (i.e., n=p-k valid): term is (1-h[p-k])
            #     but wait, h[n+k] = h[p], so d/dh[p] of h[p]*(1-h[p-k]) = (1-h[p-k])
            #   if p = n for some valid n: d/dh[p] of h[p+k]*(1-h[p]) = -h[p+k]
            # So sl[p, k] = (1-h[p-k]) * I[0<=p-k<n] - h[p+k] * I[0<=p+k<n]
            # Wait, need to be careful: the valid range of n depends on k.
            # For k >= 0: n in [0, n-1-k], so n+k in [k, n-1]
            #   p = n+k valid iff k <= p <= n-1, and n = p-k in [0, n-1-k]
            #   p = n valid iff 0 <= p <= n-1-k
            # For k < 0: let m = -k > 0, n in [m, n-1], n+k in [0, n-1-m]
            #   p = n+k valid iff 0 <= p <= n-1-m, and n = p+k = p-m >= m, so p >= 2m
            #   p = n valid iff m <= p <= n-1

            # where "valid as n+k" means p-k is a valid n value
            # and "valid as n" means p is a valid n value

                # p = n+k: valid n = p-k in [0, n-1-k] means p in [k, n-1]
                # p = n: valid if p in [0, n-1-k]

                if k <= p <= n - 1:

                if 0 <= p <= n - 1 - k:
                    si -= h[p + k]

    results = []
    for n_t in n_candidates:
        if n_t <= 0 or n_t % 2 != 0 or n_t == base_n:
            continue
        h_t = interp_to(base_h, base_n, n_t)
        h_t = project_feasible(h_t, n_t)
        c5_t = compute_c5(h_t)
        results.append((c5_t, n_t, h_t.copy()))
    results.sort(key=lambda x: x[0])
    return results


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 312
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Quick n-sweep
    n_cands = [best_n + d for d in [-6, -4, -2, 2, 4, 6, 8, 10, 12, 16, 20, 24, 32]]
    n_cands = [nn for nn in n_cands if 120 <= nn <= 600 and nn % 2 == 0]
    sweep = quick_n_sweep(best_h, best_n, n_cands, rng)
    for c5_s, n_s, h_s in sweep:
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
            best_n = n_s

    # Phase 2: SLSQP on best so far
    h_work = best_h.copy()
    n_work = best_n
    for beta in [300, 1000, 3000, 8000, 15000]:
        if remaining() < 8:
            break
        h_new = slsqp_refine(h_work, n_work, beta, maxiter=350)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_work = h_new.copy()
        else:
            h_work = h_new.copy()  # still use as starting point for next beta

    # Phase 3: Targeted binary search
    if remaining() > 20:
        time_limit = min(90, remaining() * 0.35)
        h_bin, c5_bin = targeted_binary_search(h_work, n_work, time_limit, rng)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            h_work = h_bin.copy()
            n_work = best_n

    # Phase 4: Coordinate descent on continuous
    if remaining() > 20:
        time_limit = min(60, remaining() * 0.3)
        h_cd, c5_cd = coord_descent_h(h_work, n_work, time_limit, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h_work = h_cd.copy()

    # Phase 5: SLSQP again after binary/cd
    for beta in [5000, 15000, 30000, 50000]:
        if remaining() < 6:
            break
        h_new = slsqp_refine(h_work, n_work, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_work = h_new.copy()
        else:
            h_work = h_new.copy()

    # Phase 6: Another binary search round (different random seed)
    rng2 = np.random.RandomState(seed + 12345)
    if remaining() > 15:
        time_limit = min(50, remaining() * 0.3)
        h_bin2, c5_bin2 = targeted_binary_search(h_work, n_work, time_limit, rng2)
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()
            h_work = h_bin2.copy()

    # Phase 7: Try a few other n values with full refinement
    n_other = [best_n + 2, best_n - 2, best_n + 4, best_n + 8]
    for n_t in n_other:
        if remaining() < 25 or n_t < 100 or n_t > 700 or n_t % 2 != 0 or n_t == best_n:
            continue
        h_i = interp_to(best_h, best_n, n_t)
        h_i = project_feasible(h_i, n_t)
        # Quick SLSQP
        if remaining() > 15:
            h_i = slsqp_refine(h_i, n_t, 3000, maxiter=250)
        if remaining() > 10:
            h_i, _ = coord_descent_h(h_i, n_t, min(20, remaining() - 10), rng)
        c5_t = compute_c5(h_i)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_i.copy()
            best_n = n_t

    # Phase 8: Final intensive refinement
    n_work = best_n
    h_work = best_h.copy()

    # Final SLSQP with very high beta
    for beta in [20000, 50000, 100000]:
        if remaining() < 5:
            break
        h_new = slsqp_refine(h_work, n_work, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_work = h_new.copy()

    # Final coordinate descent
    if remaining() > 15:
        time_limit = min(40, remaining() - 10)
        h_cd, c5_cd = coord_descent_h(h_work, n_work, time_limit, rng2)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Final binary micro-search
    if remaining() > 12:
        time_limit = min(25, remaining() - 6)
        h_b, c5_b = targeted_binary_search(best_h, best_n, time_limit, rng2)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Final SLSQP polish
    if remaining() > 8:
        h_new = slsqp_refine(best_h, best_n, 50000, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Validate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
