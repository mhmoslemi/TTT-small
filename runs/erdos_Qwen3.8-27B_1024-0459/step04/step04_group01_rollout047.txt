def solve_slp_iteration(h, n):
    """One SLP iteration: solve min t s.t. linearized constraints."""
    from scipy.optimize import linprog
    
    dx = 2.0 / n
    # For speed, work with corr/dx (i.e., the raw correlation sums)
    
    # Variables: [h_0, ..., h_{n-1}, t]
    # Objective: minimize t → c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x ≤ b_ub
    # For each lag k: sum_i h[i] * (1-h[i+k]) ≤ t
    # i.e., sum_i (1-h[i+k])*h[i] - t ≤ 0
    # For k from -(n-1) to (n-1):
    
    corr_raw = compute_corr(h)  # shape (2n-1,)
    n_corr = 2*n - 1
    
    # Build constraint matrix
    # For lag d = k - (n-1) (k=0..2n-2):
    # corr_raw[k] = sum_i h[i]*(1-h[i+d]) (valid i only)
    # Linearized: sum_i (1-h[i+d]) * h_new[i] ≤ t
    
    A_ub = np.zeros((n_corr, n + 1))
    b_ub = np.zeros(n_corr)
    
    for k in range(n_corr):
        d = k - (n-1)
        if d >= 0:
            # i ranges 0 to n-1-d
            for i in range(n-d):
                if 0 <= i+d < n:
                    A_ub[k, i] = (1.0 - h[i+d])
        else:
            # i ranges -d to n-1
            m = -d
            for i in range(m, n):
                if 0 <= i-m < n:
                    A_ub[k, i] = (1.0 - h[i-m])
        A_ub[k, n] = -1.0  # -t
    
    # Equality: sum h = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return h
