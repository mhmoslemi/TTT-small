def compute_c5(h):

def project_feasible(h, n):

def active_subgrad(h, n, top_k=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for k_idx in top_idx:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            n_pairs = n - offset
            if n_pairs <= 0:
                continue
            grad[:n_pairs] += (1.0 - h[offset:offset + n_pairs])
            grad[offset:] -= h[:n_pairs]
        else:
            abs_off = -offset
            n_pairs = n - abs_off
            if n_pairs <= 0:
                continue
            grad[:n_pairs] -= h[abs_off:abs_off + n_pairs]
            grad[abs_off:] += (1.0 - h[:n_pairs])
    return grad * dx

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_sum_bisect(h, n, target=None):
    """Project h onto {0<=h<=1, sum(h)=n/2} using bisection on lambda."""
    if target is None:
        target = n / 2.0
    h = np.array(h, dtype=float).copy()
    
    def sum_clipped(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0))
    
    lo = h.min() - target - 1.0
    hi = h.max() + target + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = sum_clipped(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(h - lam, 0.0, 1.0)
    
    # Fix residual
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        for _ in range(50):
            if diff > 0:
                idxs = np.where(h < 1.0 - 1e-12)[0]
            else:
                idxs = np.where(h > 1e-12)[0]
            if len(idxs) == 0:
                break
            idx = idxs[np.random.choice(len(idxs))]
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, -h[idx])
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-14:
                break
    return h

def obj_lse(x):

    m = np.max(corr)
    beta = _G_BETA
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -600, 600))
    return (m + np.log(np.sum(exps)) / beta) * dx


def con_sum(x):
    return np.sum(x) - _G_TGT


def active_subgrad(h, n, top_k=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for k_idx in top_idx:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            n_pairs = n - offset
            if n_pairs <= 0:
                continue
            grad[:n_pairs] += (1.0 - h[offset:offset + n_pairs])
            grad[offset:] -= h[:n_pairs]
        else:
            abs_off = -offset
            n_pairs = n - abs_off
            if n_pairs <= 0:
                continue
            grad[:n_pairs] -= h[abs_off:abs_off + n_pairs]
            grad[abs_off:] += (1.0 - h[:n_pairs])
    return grad * dx

def do_nesterov(h_cur, n, budget_fn, beta, max_iters=500):
    """Nesterov-accelerated projected gradient for LSE objective."""
    global _G_N, _G_BETA, _G_TGT, _G_DX
    _G_N = n
    _G_BETA = float(beta)
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    
    h = h_cur.copy()
    y = h_cur.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    
    step0 = 0.02
    for t in range(1, max_iters + 1):
        if budget_fn() < 3.0:
            break
        # Compute gradient of LSE at y
        corr = np.correlate(y, 1.0 - y, mode='full')
        m = np.max(corr)
        beta_v = beta
        shifted = corr - m
        exps = np.exp(np.clip(beta_v * shifted, -600, 600))
        weights = exps / np.sum(exps)
        
        # Gradient of sum_k weights_k * corr_k * dx
        grad = np.zeros(n)
        for k_idx in range(len(corr)):
            w = weights[k_idx]
            if w < 1e-10:
                continue
            offset = k_idx - (n - 1)
            if offset >= 0:
                n_pairs = n - offset
                if n_pairs <= 0:
                    continue
                grad[:n_pairs] += w * (1.0 - y[offset:offset + n_pairs])
                grad[offset:] -= w * y[:n_pairs]
            else:
                abs_off = -offset
                n_pairs = n - abs_off
                if n_pairs <= 0:
                    continue
                grad[:n_pairs] -= w * y[abs_off:abs_off + n_pairs]
                grad[abs_off:] += w * (1.0 - y[:n_pairs])
        grad *= _G_DX
        
        # Nesterov update
        alpha = 2.0 / (t + 1.0 + np.sqrt(1.0 + 4.0 * t * t))
        step = 2.0 / (1.0 + t)
        y_new = project_sum_bisect(h - step * grad, n)
        h = (1.0 - alpha) * y_new + alpha * (y_new - y)
        h = project_sum_bisect(h, n)
        y = y_new
        
        c5_cur = compute_c5(h)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h.copy()
    
    return h_best, c5_best

            h_new, c5_new = do_subgrad(h_new, n_try, budget_fn, rng, max_iters=80)
        
        if c5_new < best_c5 - 1e-15:
            best_h = h_new.copy()
            best_c5 = c5_new
            best_n = n_try
            print("  n={} improved: C5={:.10f}".format(n_try, best_c5))
        else:
            print("  n={} C5={:.10f} (best={:.10f})".format(n_try, c5_new, best_c5))

    # Phase 5: Subgradient at best resolution
    if budget_fn() > 15:
        print("\n--- Phase 5: Subgradient polish n={} ---".format(best_n))
        best_h, best_c5 = do_subgrad(best_h, best_n, budget_fn, rng, max_iters=250)
        print("  After subgrad: C5={:.10f}".format(best_c5))

    # Phase 6: Nesterov + SLSQP final round
    if budget_fn() > 15:
        print("\n--- Phase 6: Final Nesterov+SLSQP ---")
        h_nes, c5_nes = do_nesterov(best_h, best_n, budget_fn, 50000, max_iters=200)
        if c5_nes < best_c5:
            best_h = h_nes
            best_c5 = c5_nes
        h_new, c5_new = do_slsqp_round(best_h, best_n, 100000, 150, budget_fn)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        print("  After final: C5={:.10f}".format(best_c5))

    # Phase 7: Coordinate descent
    if budget_fn() > 10:
        print("\n--- Phase 7: Coordinate descent ---")
        best_h, best_c5 = do_coord_descent(best_h, best_n, budget_fn, max_sweeps=20)
        print("  After coord: C5={:.10f}".format(best_c5))

    # Phase 8: Try alternative starts at best resolution
    if budget_fn() > 30:
        print("\n--- Phase 8: Alternative starts ---")
        alts = []
        # Golden ratio
        gr_h = make_golden_ratio(best_n)
        alts.append(("golden", gr_h))
        # Thue-Morse
        tm_h = make_thue_morse(best_n)
        alts.append(("thue_morse", tm_h))
        # Random
        rng2 = np.random.RandomState(seed + 123)
        ridx = rng2.permutation(best_n)
        r_h = np.zeros(best_n)
        r_h[ridx[:best_n // 2]] = 1.0
        r_h = project_sum_bisect(r_h, best_n)
        alts.append(("random", r_h))
        # Perturbed best
        p_h = project_sum_bisect(best_h + rng.randn(best_n) * 0.02, best_n)
        alts.append(("perturbed_best", p_h))
        
        for name, h_alt in alts:
            if budget_fn() < 20:
                break
            c5_alt = compute_c5(h_alt)
            if c5_alt >= best_c5:
                continue
            # Quick optimization from this start
            h_opt, c5_opt = optimize_at_resolution(h_alt, best_n, budget_fn, [500, 5000], [200, 100])
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                print("  {} improved: C5={:.10f}".format(name, best_c5))
            else:
                print("  {} no improve: C5={:.10f}".format(name, c5_opt))

    # Phase 9: Final polish
    if budget_fn() > 8:
        best_h, best_c5 = do_subgrad(best_h, best_n, budget_fn, rng, max_iters=100)
    if budget_fn() > 8:
        h_new, c5_new = do_slsqp_round(best_h, best_n, 50000, 100, budget_fn)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    if budget_fn() > 5:
        best_h, best_c5 = do_coord_descent(best_h, best_n, budget_fn, max_sweeps=10)

    # Final validation
    best_h = project_sum_bisect(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
