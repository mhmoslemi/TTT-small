                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h.copy()
            else:
                h[i]=1; h[j]=0
    
    return h_best, c5_best

def sa_binary(h, n, budget_fn, rng, iters=20000, T0=0.003, T1=0.0001):
    """Simulated annealing on binary vectors."""
    h = h.astype(float).copy()
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    
    for it in range(iters):
        if budget_fn() < 1.0:
            break
        T = T0 * (T1/T0) ** (it / max(iters, 1))
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i]=0; h[j]=1
        c5_new = compute_c5(h)
        delta = c5_new - c5_cur
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
        else:
            h[i]=1; h[j]=0
    
    return h_best, c5_best

def slsqp_smoothed(h_start, n, budget_fn, beta_list):
    """SLSQP with log-sum-exp smoothing of the max."""
    N = n
    DX = 2.0 / n
    TGT = n / 2.0
    BOUNDS = [(0.0, 1.0)] * n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if beta < 1:
            return m * DX
        shifted = corr - m
        vals = np.exp(np.clip(shifted * beta, -500, 500))
        return (m + np.log(np.sum(vals)) / beta) * DX
    
    def obj_grad(x):
        # Not implementing analytical gradient; let SLSQP do finite differences
        return None
    
    def con_eq(x):
        return np.sum(x) - TGT
    
    def con_eq_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac}]
    
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    
    for beta in beta_list:
        if budget_fn() < 5.0:
            break
        try:
            maxiter = 300 if beta < 1000 else 200 if beta < 5000 else 100
            res = minimize(obj, h_cur, method='SLSQP',
                         bounds=BOUNDS, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    global _start_time, _budget
    _start_time = time.time()
    _budget = float(budget_s)
    rng = np.random.RandomState(seed)
    
    def remaining():
        return max(0.0, _budget - (time.time() - _start_time) - 5.0)
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 256
        h_init = make_thue_morse(n_init)
    
    # Find good binary starting points
    n = n_init
    starters = []
    
    # Binarized parent
    h_bin_parent = np.zeros(n)
    order = np.argsort(h_init)[::-1]
    h_bin_parent[order[:n//2]] = 1.0
    starters.append(('parent_bin', h_bin_parent))
    
    # Thue-Morse
    h_tm = make_thue_morse(n)
    starters.append(('thue_morse', h_tm))
    
    # Golden ratio
    h_gr = make_golden(n)
    starters.append(('golden', h_gr))
    
    # Random
    h_rand = make_random(rng, n)
    starters.append(('random', h_rand))
    
    # Evaluate all starters
    best_c5 = float('inf')
    best_h = None
    for name, h_s in starters:
        c5_s = compute_c5(h_s)
        print("  {0}: C5 = {1:.10f}".format(name, c5_s))
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
    
    # Also evaluate the continuous parent
    c5_parent = compute_c5(h_init)
    print("  parent_cont: C5 = {0:.10f}".format(c5_parent))
    if c5_parent < best_c5:
        best_c5 = c5_parent
        best_h = h_init.copy()
    
    print("Phase 0 best: C5 = {0:.10f}".format(best_c5))
    
    # Phase 1: Binary local search + SA at n
    print("\nPhase 1: Binary optimization at n={0}".format(n))
    t1_start = time.time()
    h_cur, c5_cur = binary_local_search(best_h, n, remaining, rng, max_iters=200000)
    print("  After local search: C5 = {0:.10f}".format(c5_cur))
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h_cur.copy()
    
    if remaining() > 30:
        h_sa, c5_sa = sa_binary(h_cur, n, remaining, rng, iters=50000)
        print("  After SA: C5 = {0:.10f}".format(c5_sa))
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
    
    # Phase 2: SLSQP continuous refinement
    if remaining() > 20:
        print("\nPhase 2: SLSQP continuous refinement at n={0}".format(n))
        h_cur, c5_cur = slsqp_smoothed(best_h, n, remaining, [50, 200, 1000, 5000])
        print("  After SLSQP: C5 = {0:.10f}".format(c5_cur))
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur
    
    # Phase 3: Try higher resolutions
    for n_new in [384, 512, 768, 1024]:
        if remaining() < 30:
            break

        # Interpolate best solution to new resolution

        x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
        h_interp = np.interp(x_new, xp, best_h)
        # Binarize
        h_bin_new = np.zeros(n_new)
        order = np.argsort(h_interp)[::-1]
