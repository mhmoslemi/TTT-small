def compute_delta(h, a, b, n):
    """Compute change in correlation vector when flipping h[a] 1→0 and h[b] 0→1."""
    delta = np.zeros(2*n-1)
    s = np.arange(-(n-1), n)  # all shifts
    k = s + (n-1)  # corresponding index in result array
    
    # Term 1: -(1-h[a-s]) for valid s (0 ≤ a-s < n)
    valid1 = (0 <= a - s) & (a - s < n)
    if np.any(valid1):
        delta[k[valid1]] += -(1 - h[a - s[valid1]])
    
    # Term 2: +h[a+s] for valid s (0 ≤ a+s < n)
    valid2 = (0 <= a + s) & (a + s < n)
    if np.any(valid2):
        delta[k[valid2]] += h[a + s[valid2]]
    
    # Term 3: +(1-h[b-s]) for valid s (0 ≤ b-s < n)
    valid3 = (0 <= b - s) & (b - s < n)
    if np.any(valid3):
        delta[k[valid3]] += (1 - h[b - s[valid3]])
    
    # Term 4: -h[b+s] for valid s (0 ≤ b+s < n)
    valid4 = (0 <= b + s) & (b + s < n)
    if np.any(valid4):
        delta[k[valid4]] -= h[b + s[valid4]]
    
    return delta

```python
def best_flip_step(h, n):
    """Find and apply the best single-pair flip to reduce max correlation."""
    corr = np.correlate(h, 1-h, mode='full')
    max_corr = np.max(corr)
    worst_k = np.argmax(corr)
    worst_s = worst_k - (n-1)
    
    s = worst_s
    # Compute A[a] for all a
    A = np.zeros(n)
    for a in range(n):
        # Term: -(1-h[a-s]) if 0≤a-s<n
        if 0 <= a-s < n:
            A[a] += -(1-h[a-s])
        # Term: +h[a+s] if 0≤a+s<n
        if 0 <= a+s < n:
            A[a] += h[a+s]
    
    # Compute B[b] for all b
    B = np.zeros(n)
    for b in range(n):
        # Term: +(1-h[b-s]) if 0≤b-s<n
        if 0 <= b-s < n:
            B[b] += (1-h[b-s])
        # Term: -h[b+s] if 0≤b+s<n
        if 0 <= b+s < n:
            B[b] += -h[b+s]
    
    # Find best a (h[a]=1) and best b (h[b]=0)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    best_a = ones[np.argmin(A[ones])]
    best_b = zeros[np.argmin(B[zeros])]
    
    # Verify
    h_trial = h.copy()
    h_trial[best_a] = 0
    h_trial[best_b] = 1
    corr_trial = np.correlate(h_trial, 1-h_trial, mode='full')
    max_trial = np.max(corr_trial)
    
    if max_trial < max_corr:
        h = h_trial
        return h, True
    else:
        return h, False
```

```python
def binary_anneal(h, n, max_iters, rng, T0=1.0, T1=0.01):
    """Simulated annealing for binary h."""
    corr = np.correlate(h, 1-h, mode='full')
    max_corr = np.max(corr)
    best_h = h.copy()
    best_max = max_corr
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for it in range(max_iters):
        # Choose flip based on current state
        # Strategy: with probability 0.7, use worst-lag guided; with 0.3, random
        if rng.random() < 0.7:
            # Worst-lag guided
            worst_k = np.argmax(corr)
            worst_s = worst_k - (n-1)
            s = worst_s
            # Compute A and B
            A = np.zeros(n)
            for a in range(n):
                if 0 <= a-s < n: A[a] += -(1-h[a-s])
                if 0 <= a+s < n: A[a] += h[a+s]
            B = np.zeros(n)
            for b in range(n):
                if 0 <= b-s < n: B[b] += (1-h[b-s])
                if 0 <= b+s < n: B[b] += -h[b+s]
            a_candidates = [a for a in range(n) if h[a] == 1]
            b_candidates = [b for range(n) if h[b] == 0]
            # Pick from top 5 candidates
            a_sorted = sorted(a_candidates, key=lambda x: A[x])[:5]
            b_sorted = sorted(b_candidates, key=lambda x: B[x])[:5]
            a = a_sorted[rng.randint(len(a_sorted))]
            b = b_sorted[rng.randint(len(b_sorted))]
        else:
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
        
        # Apply flip
        h[a] = 0
        h[b] = 1
        corr_new = np.correlate(h, 1-h, mode='full')
        max_new = np.max(corr_new)
        
        # Accept/reject
        delta = max_new - max_corr
        T = T0 * (T1/T0) ** (it / max_iters)
        if delta < 0 or rng.random() < np.exp(-delta / (T * 2.0/n)):
            # Accept
            max_corr = max_new
            corr = corr_new
            ones[ones.index(a)] = b  # update lists
            zeros[zeros.index(b)] = a
            if max_corr < best_max:
                best_max = max_corr
                best_h = h.copy()
        else:
            # Reject
            h[a] = 1
            h[b] = 0
    
    return best_h, best_max
```

The temperature scaling needs to be right. The "energy" is max_corr * dx = max_corr * 2/n. A change of 1 in max_corr corresponds to a change of 2/n in C₅. For n=256, that's 2/256 = 1/128 ≈ 0.0078.

So the energy difference is delta_energy = delta * 2/n. With T0 = 1.0 and T1 = 0.01, the acceptance probability for a worsening of delta_energy is exp(-delta_energy/T).

At the start (T=1), we accept almost anything. At the end (T=0.01), we only accept improvements or very small worsenings.

This should work well. Let me also add a "restart" mechanism: if we get stuck (no improvement for many iterations), restart from the best solution with a new random seed.

g_k is not convex or concave in general (it has quadratic terms with both + and - signs).

However, the maximum of a set of functions is convex if each function is convex. Since g_k is not convex, f(h) is not necessarily convex.

But empirically, for these types of problems, the optimum is often at a vertex of the feasible region, i.e., at a binary (0-1) point. This is because the feasible region is a hypercube (with one linear constraint), and the objective, while not convex, tends to have its minimum at extreme points.

For a fixed worst lag s*, we want to find (a,b) with h[a]=1, h[b]=0 that minimizes:
max_k (corr[k] + A_{s*}(a) + B_{s*}(b) + other changes at lag k)

Wait, I made an error earlier. The delta for lag k depends on s (the shift), not just s*. Let me redo this.

For a flip (a,b), the change at lag k (shift s = k-(n-1)) is:
delta(k) = A_s(a) + B_s(b)

where:
A_s(a) = [0≤a-s<n] * (-(1-h[a-s])) + [0≤a+s<n] * h[a+s]
B_s(b) = [0≤b-s<n] * (1-h[b-s]) + [0≤b+s<n] * (-h[b+s])

So the new correlation at lag k is:
new_corr(k) = corr(k) + A_s(a) + B_s(b) where s = k-(n-1)

We want to minimize max_k new_corr(k).

This is a minimax problem over (a,b). For a fixed a, the optimal b minimizes max_k (corr(k) + A_s(a) + B_s(b)).

This is still complex, but we can use the greedy heuristic: focus on the worst lag and find the flip that most reduces it.

For the worst lag k* (shift s*):
delta(k*) = A_{s*}(a) + B_{s*}(b)

To minimize this, we minimize A_{s*}(a) over a with h[a]=1, and B_{s*}(b) over b with h[b]=0. These are independent!

So: a* = argmin_{a: h[a]=1} A_{s*}(a), b* = argmin_{b: h[b]=0} B_{s*}(b).

Then verify: compute the full new correlation and check if max decreased.

If it didn't decrease (because another lag became the new maximum), we try the next-best candidates.

This is the key algorithm. Let me implement it efficiently.

For efficiency, I'll precompute A_s(a) and B_s(b) for the current worst lag, then do a greedy search with backtracking if needed.

def project_feasible(h, n):

def compute_delta_at_shift(h, a, b, s, n):
    """Compute delta at a single shift s for flipping h[a] 1→0, h[b] 0→1."""
    delta = 0.0
    # Term 1: -(1-h[a-s]) if 0 ≤ a-s < n
    if 0 <= a - s < n:
        delta += -(1.0 - h[a - s])
    # Term 2: +h[a+s] if 0 ≤ a+s < n
    if 0 <= a + s < n:
        delta += h[a + s]
    # Term 3: +(1-h[b-s]) if 0 ≤ b-s < n
    if 0 <= b - s < n:
        delta += (1.0 - h[b - s])
    # Term 4: -h[b+s] if 0 ≤ b+s < n
    if 0 <= b + s < n:
        delta += -h[b + s]
    return delta

def compute_delta_vector(h, a, b, n):
    """Compute the full delta vector for flipping h[a] 1→0, h[b] 0→1."""
    delta = np.zeros(2*n - 1)
    s_vals = np.arange(-(n-1), n)
    k_vals = s_vals + (n - 1)
    
    # Vectorized computation
    # A_s(a): contribution from h[a] flip
    # For each s: A_s(a) = [0≤a-s<n]*(-(1-h[a-s])) + [0≤a+s<n]*h[a+s]
    valid_a_minus = (a - s_vals >= 0) & (a - s_vals < n)
    valid_a_plus = (a + s_vals >= 0) & (a + s_vals < n)
    A = np.zeros(len(s_vals))
    A[valid_a_minus] = -(1.0 - h[a - s_vals[valid_a_minus]])
    A[valid_a_plus] += h[a + s_vals[valid_a_plus]]
    
    # B_s(b): contribution from h[b] flip
    valid_b_minus = (b - s_vals >= 0) & (b - s_vals < n)
    valid_b_plus = (b + s_vals >= 0) & (b + s_vals < n)
    B = np.zeros(len(s_vals))
    B[valid_b_minus] = (1.0 - h[b - s_vals[valid_b_minus]])
    B[valid_b_plus] += -h[b + s_vals[valid_b_plus]]
    
    delta = A + B
    return delta

def greedy_flip_step(h, n, top_k=5):
    """One step of greedy flip: find the best pair to flip at the worst lag."""
    corr = np.correlate(h, 1-h, mode='full')
    max_corr = np.max(corr)
    
    # Find top k worst lags
    top_lags = np.argsort(corr)[::-1][:top_k]
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    best_flip = None
    best_new_max = max_corr
    
    for k in top_lags[:3]:  # Try top 3 worst lags
        s = k - (n - 1)
        
        # Compute A[a] for all a with h[a]=1
        A_vals = np.zeros(len(ones))
        for i, a in enumerate(ones):
            delta = 0.0
            if 0 <= a - s < n:
                delta += -(1.0 - h[a - s])
            if 0 <= a + s < n:
                delta += h[a + s]
            A_vals[i] = delta
        
        # Compute B[b] for all b with h[b]=0
        B_vals = np.zeros(len(zeros))
        for i, b in enumerate(zeros):
            delta = 0.0
            if 0 <= b - s < n:
                delta += (1.0 - h[b - s])
            if 0 <= b + s < n:
                delta += -h[b + s]
            B_vals[i] = delta
        
        # Try top candidates
        a_order = np.argsort(A_vals)[:min(10, len(ones))]
        b_order = np.argsort(B_vals)[:min(10, len(zeros))]
        
        for ai in a_order[:5]:
            a = ones[ai]
            for bi in b_order[:5]:
                b = zeros[bi]
                if a == b:

                # Compute full delta and check

                h_trial[a] = 0
                h_trial[b] = 1

                new_max = np.max(corr_trial)
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_flip = (a, b)
    
    if best_flip is not None:
        h[best_flip[0]] = 0
        h[best_flip[1]] = 1
        return h, True
    return h, False

def binary_anneal(h, n, max_iters, rng, verbose=False):
    """Simulated annealing for binary h."""
    corr = np.correlate(h, 1-h, mode='full')
    max_corr = np.max(corr)
    best_h = h.copy()
    best_max = max_corr
    dx = 2.0 / n
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])

    T1 = 0.001
    
    no_improve_count = 0
    stagnation_limit = 500
    
    for it in range(max_iters):
        if (it % 1000 == 0) and remaining_time() < 5:
            break
        
        # Temperature schedule
        T = T0 * (T1 / T0) ** (it / max_iters)
        
        # Choose flip
        if rng.random() < 0.6:
            # Worst-lag guided
            worst_k = np.argmax(corr)
            s = worst_k - (n - 1)

            A_vals = np.zeros(n)
            for a in range(n):
                if h[a] != 1:
                    continue
                if 0 <= a - s < n:
                    A_vals[a] += -(1.0 - h[a - s])
                if 0 <= a + s < n:
                    A_vals[a] += h[a + s]
            
            # Compute B for all zeros
            B_vals = np.zeros(n)
            for b in range(n):
                if h[b] != 0:
                    continue
                if 0 <= b - s < n:
                    B_vals[b] += (1.0 - h[b - s])
                if 0 <= b + s < n:
                    B_vals[b] += -h[b + s]
            
            # Pick from top candidates
            ones_mask = (h == 1)
            zeros_mask = (h == 0)
            
            # Get top 10 a candidates
            A_sorted = np.argsort(A_vals)
            a_cands = [A_sorted[i] for i in range(min(10, n)) if ones_mask[A_sorted[i]]]
            if not a_cands:
                a_cands = list(ones_list)
            
            B_sorted = np.argsort(B_vals)
            b_cands = [B_sorted[i] for i in range(min(10, n)) if zeros_mask[B_sorted[i]]]
            if not b_cands:
                b_cands = list(zeros_list)
            
            a = a_cands[rng.randint(len(a_cands))]
            b = b_cands[rng.randint(len(b_cands))]
        else:
            # Random
            a = ones_list[rng.randint(len(ones_list))]
            b = zeros_list[rng.randint(len(zeros_list))]
        
        # Apply flip
        h[a] = 0
        h[b] = 1
        corr_new = np.correlate(h, 1-h, mode='full')
        max_new = np.max(corr_new)
        
        # Accept/reject
        energy_diff = (max_new - max_corr) * dx
        if energy_diff <= 0 or (T > 0 and rng.random() < np.exp(-energy_diff / T)):
            # Accept
            max_corr = max_new
            corr = corr_new
            # Update lists
            ones_list[ones_list.index(a)] = b
            zeros_list[zeros_list.index(b)] = a
            if max_corr < best_max:
                best_max = max_corr
                best_h = h.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            # Reject
            h[a] = 1
            h[b] = 0
            no_improve_count += 1
        
        # Restart if stuck
        if no_improve_count > stagnation_limit:
            h = best_h.copy()
            corr = np.correlate(h, 1-h, mode='full')
            max_corr = np.max(corr)
            ones_list = list(np.where(h == 1)[0])
            zeros_list = list(np.where(h == 0)[0])
            no_improve_count = 0
            # Random kick
            for _ in range(3):
                a = ones_list[rng.randint(len(ones_list))]
                b = zeros_list[rng.randint(len(zeros_list))]
                h[a] = 0
                h[b] = 1
                ones_list[ones_list.index(a)] = b
                zeros_list[zeros_list.index(b)] = a
            corr = np.correlate(h, 1-h, mode='full')
            max_corr = np.max(corr)
    
    return best_h, best_max * dx
