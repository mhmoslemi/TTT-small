    above = np.where(corr >= threshold)[0]
    all_lags = np.unique(np.concatenate([top_k, above]))
    return all_lags

def slsqp_optimize(h, n, beta, maxiter=500, top_lags=None):
    """Run SLSQP with smooth objective"""
    global _G_N, _G_BETA, _G_TGT, _G_DX, _G_TOP_LAGS
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    _G_BETA = float(beta)
    _G_TOP_LAGS = top_lags
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum}]
    
    if top_lags is not None and len(top_lags) > 0:
        func = obj_smooth_tight
    else:
        func = obj_smooth_full
    
    try:
        res = minimize(func, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        return h_new, c5_new
    except Exception as e:
        return h, compute_c5(h)

def coord_descent(h, n, max_sweeps=20, step_sizes=None):
    """Fine coordinate descent"""
    c5_best = compute_c5(h)
    h_best = h.copy()
    if step_sizes is None:
        step_sizes = [0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001, 0.0005, 0.0001]
    
    target = n / 2.0
    for sweep in range(max_sweeps):
        if remaining_time() < 3.0:
            break
        any_imp = False
        for i in range(n):
            if remaining_time() < 2.0:
                break
            old = h_best[i]
            for s in step_sizes:
                for direction in [1.0, -1.0]:
                    new_v = old + direction * s
                    if new_v < 0.0 or new_v > 1.0:
                        continue
                    h_trial = h_best.copy()
                    h_trial[i] = new_v
                    # Adjust another point to maintain sum
                    diff = new_v - old
                    # Find a point to adjust
                    if abs(diff) > 1e-15:
                        if diff > 0:
                            j = int(np.argmax(h_best))
                            if j == i:
                                j = int(np.argmax(np.where(np.arange(n) != i, h_best, -1)))
                        else:
                            j = int(np.argmin(h_best))
                            if j == i:
                                j = int(np.argmin(np.where(np.arange(n) != i, h_best, 2)))
                        if j != i:
                            h_trial[j] = h_best[j] - diff
                            h_trial[j] = np.clip(h_trial[j], 0.0, 1.0)
                    h_trial = project_feasible(h_trial, n)
                    c5_t = compute_c5(h_trial)
                    if c5_t < c5_best - 1e-14:
                        c5_best = c5_t
                        h_best = h_trial.copy()
                        any_imp = True
        if not any_imp:
            break
    return h_best, c5_best

def subgrad_step(h, n, top_k=10, step=0.05):
    """One subgradient step targeting top lags"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for k_idx in top_idx:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            nv = n - offset
            if nv <= 0:
                continue
            grad[offset:offset+nv] += (1.0 - h[:nv])
            grad[:nv] -= h[offset:offset+nv]
        else:
            m = -offset
            nv = n - m
            if nv <= 0:
                continue
            grad[:nv] += (1.0 - h[m:m+nv])
            grad[m:m+nv] -= h[:nv]
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h
    grad = grad / gn
    h_new = h - step * grad
    return project_feasible(h_new, n)
