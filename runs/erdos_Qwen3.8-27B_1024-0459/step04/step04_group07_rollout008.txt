def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:

        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_descent(h0, n, M, n_iter, lr_init, rng, tlimit):

    global
    h = h0.copy()
    c5 = compute_c5(h)
    h_best = h.copy()
    v = np.zeros(n)
    beta_mom = 0.9
    lr = lr_init
    
    for it in range(n_iter):
        if tlimit() < 2.0:
            break
        corr, grad, _ = corr_and_topM_grad(h, M)
        c_max = corr.max()
        lr_cur = lr_init / (1.0 + 0.05 * it)
        
        # Momentum step

                c5 = compute_c5(h_best)

        val += 0.02 * corr[_G_FOCUS] * _G_DX
    return float(val)

def slsqp_run(h0, n, beta, maxiter, use_focus=False):
    global _G_N, _G_TGT, _G_DX, _G_BETA, _G_PEN, _G_FOCUS
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    _G_BETA = float(beta)
    _G_PEN = 0.0
    _G_FOCUS = -1
    if use_focus:
        corr = np.correlate(h0, 1.0 - h0, mode='full')
        _G_FOCUS = int(np.argmax(corr))
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    try:
        res = minimize(obj_lob, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def lbfgsb_run(h0, n, beta, maxiter, pen=50.0):
    global _G_N, _G_TGT, _G_DX, _G_BETA, _G_PEN, _G_FOCUS
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    _G_BETA = float(beta)
    _G_PEN = float(pen)
    _G_FOCUS = -1
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_lob, h0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def coord_descent_transfer(h, n, max_sweeps, rng, tlimit):
    """Coordinate descent with transfer moves (move mass from i to j)."""
    c5 = compute_c5(h)
    h_best = h.copy()
    
    for sweep in range(max_sweeps):
        if tlimit() < 2.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if tlimit() < 1.0:
                break
            v0 = h_best[i]
            # Try setting to 0 or 1
            for v in [0.0, 1.0]:
                if abs(v - v0) < 1e-9:
                    continue
                trial = h_best.copy()
                trial[i] = v
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 5e-10:
                    c5 = c
                    h_best = trial.copy()
                    improved = True
                    break
            # Try transfers to random other indices
            n_transfers = 5
            transfer_idxs = rng.choice(n, size=n_transfers, replace=False)
            for j in transfer_idxs:
                if j == i or tlimit() < 0.5:
                    continue
                for delta in [0.1, -0.1, 0.05, -0.05, 0.2, -0.2]:
                    if tlimit() < 0.5:
                        break
                    if h_best[i] + delta < 0 or h_best[i] + delta > 1:
                        continue
                    if h_best[j] - delta < 0 or h_best[j] - delta > 1:
                        continue
                    trial = h_best.copy()
                    trial[i] += delta
                    trial[j] -= delta
                    trial = project_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < c5 - 5e-10:
                        c5 = c
                        h_best = trial.copy()
                        improved = True
                        break
        if not improved and sweep >= 3:
            break
    return h_best, c5

def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c else 0.0
    return h

def make_rand_bin(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def annealed_subgradient(h0, n, rng, tlimit, phases):
    """Multi-phase subgradient with annealing of M and learning rate."""
    h = h0.copy()
    c5 = compute_c5(h)
    h_best = h.copy()
    
    for phase_idx, (M, lr, n_iters) in enumerate(phases):
        if tlimit() < 3.0:
            break
        v = np.zeros(n)
        beta_mom = 0.7 + 0.3 * phase_idx / max(1, len(phases) - 1)
        for it in range(n_iters):
            if tlimit() < 1.5:
                break
            corr, grad, _ = corr_and_topM_grad(h, M)
            c_cur = corr.max()
            
            lr_cur = lr * (1.0 / (1.0 + 0.02 * it))
            v = beta_mom * v - lr_cur * grad
            h_trial = project_feasible(h + v, n)
            c_new = compute_c5(h_trial)
            
            if c_new < c5 - 1e-13:
                c5 = c_new
                h = h_trial
                h_best = h.copy()
            else:
                # Try half step
                h_trial2 = project_feasible(h + 0.5 * v, n)
                c2 = compute_c5(h_trial2)
                if c2 < c5 - 1e-13:
                    c5 = c2
                    h = h_trial2
                    h_best = h.copy()
                else:
                    v *= 0.5
                    h = h_best.copy()
                    c5 = compute_c5(h)
        # After each phase, do a quick SLSQP polish
        if tlimit() > 10 and phase_idx < len(phases) - 1:
            h_pol = slsqp_run(h_best, n, beta=200 + 300 * phase_idx, maxiter=100)
            c_pol = compute_c5(h_pol)
            if c_pol < c5 - 1e-12:
                c5 = c_pol
                h_best = h_pol
                h = h_pol
    
    return h_best, c5

def top_lag_balancing(h, n, rng, tlimit, n_rounds=20):
    """Identify top 2-3 lags and try to balance them via targeted SLSQP."""
    global _G_N, _G_TGT, _G_DX, _G_BETA, _G_PEN, _G_FOCUS
    h_best = h.copy()
    c5 = compute_c5(h)
    
    for rnd in range(n_rounds):
        if tlimit() < 3.0:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode='full')
        top3 = np.argsort(corr)[-3:]
        
        for focus_idx in top3:
            if tlimit() < 2.0:
                break
            _G_N = n
            _G_TGT = n / 2.0
            _G_DX = 2.0 / n
            _G_BETA = 800.0
            _G_PEN = 0.0
            _G_FOCUS = int(focus_idx)
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
            try:
                res = minimize(obj_lob, h_best, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 150, 'ftol': 1e-15, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new)
                if c_new < c5 - 1e-12:
                    c5 = c_new
                    h_best = h_new
            except Exception:
                pass
        _G_FOCUS = -1
    
    return h_best, c5

def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX, _G_PEN, _G_FOCUS
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    _G_PEN = 0.0
    _G_FOCUS = -1
    rng = np.random.RandomState(seed)

    def tlimit():
        return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = make_tm(parent_n)
        parent_h = project_feasible(parent_h, parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
        print("No parent. TM: n={}, C5={:.8f}".format(parent_n, best_c5))

    # === Phase 1: Annealed subgradient at parent resolution ===
    if tlimit() > 120:
        print("\n=== Phase 1: annealed subgradient n={} ===".format(best_n))
        n = best_n
        phases = [
            (3, 0.08, 200),   # M=3, moderate lr
            (5, 0.05, 200),   # M=5, smaller lr
            (7, 0.03, 200),   # M=7, even smaller
            (10, 0.02, 200),  # M=10, fine tuning
            (5, 0.01, 300),   # M=5, very fine
            (3, 0.005, 300),  # M=3, ultra fine
        ]
        h_cur, c5_cur = annealed_subgradient(best_h, n, rng, tlimit, phases)
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n
        print("  After subgradient: C5={:.8f}".format(best_c5))

    # === Phase 2: SLSQP multi-beta at parent resolution ===
    if tlimit() > 100:
        print("\n=== Phase 2: SLSQP polish n={} ===".format(best_n))
        n = best_n
        h_cur = best_h.copy()
        c5_cur = best_c5
        for bv in [100, 300, 800, 2000, 5000, 10000]:
            if tlimit() < 10:
                break
            h_new = slsqp_run(h_cur, n, bv, maxiter=300)
            c_new = compute_c5(h_new)
            if c_new < c5_cur - 1e-13:
                c5_cur = c_new
                h_cur = h_new
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n
        print("  After SLSQP: C5={:.8f}".format(best_c5))

    # === Phase 3: Top-lag balancing ===
    if tlimit() > 60:
        print("\n=== Phase 3: top-lag balancing n={} ===".format(best_n))
        n = best_n
        h_cur, c5_cur = top_lag_balancing(best_h, n, rng, tlimit, n_rounds=15)
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n
        print("  After balancing: C5={:.8f}".format(best_c5))

    # === Phase 4: Try n=512 ===
    if tlimit() > 200:
        print("\n=== Phase 4: n=512 exploration ===")
        n512 = 512
        h512 = project_feasible(interp_h(best_h, best_n, n512), n512)
        c5512 = compute_c5(h512)
        print("  Interpolated to 512: C5={:.8f}".format(c5512))
        
        # L-BFGS-B with penalty
        phases512 = [
            (5, 0.06, 150),
            (7, 0.04, 150),
            (5, 0.02, 200),
        ]
        h512, c5512 = annealed_subgradient(h512, n512, rng, tlimit, phases512)
        if c5512 < best_c5 - 1e-12:
            best_c5 = c5512
            best_h = h512.copy()
            best_n = n512
            print("  n=512 improved: C5={:.8f}".format(best_c5))
        
        # L-BFGS-B polish at 512
        if tlimit() > 30:
            for bv in [200, 800, 2000]:
                if tlimit() < 10:
                    break
                h512_new = lbfgsb_run(best_h, n512, bv, maxiter=500, pen=80.0)
                c_512 = compute_c5(h512_new)
                if c_512 < best_c5 - 1e-12:
                    best_c5 = c_512
                    best_n = n512
                    print("  n=512 LBFGS beta={}: C5={:.8f}".format(bv, best_c5))
        
        # SLSQP at 512 (slower but more precise)
        if tlimit() > 30 and best_n == n512:
            h512_new = slsqp_run(best_h, n512, 3000, maxiter=200)
            c_512 = compute_c5(h512_new)
            if c_512 < best_c5 - 1e-12:
                best_c5 = c_512
                print("  n=512 SLSQP: C5={:.8f}".format(best_c5))

        # Also try interpolating back to 256 if 512 didn't help
        if best_n == n512 and tlimit() > 30:
            h256_from512 = project_feasible(interp_h(best_h, n512, 256), 256)
            c_256 = compute_c5(h256_from512)
            if c_256 < best_c5:
                best_c5 = c_256
                best_h = h256_from512.copy()
                best_n = 256
                print("  Interp back to 256: C5={:.8f}".format(best_c5))

    # === Phase 5: Try n=384 ===
    if tlimit() > 80 and best_n != 384:
        print("\n=== Phase 5: n=384 exploration ===")
        n384 = 384
        h384 = project_feasible(interp_h(best_h, best_n, n384), n384)
        c5384 = compute_c5(h384)
        
        phases384 = [
            (5, 0.05, 150),
            (7, 0.03, 150),
            (3, 0.015, 200),
        ]
        h384, c5384 = annealed_subgradient(h384, n384, rng, tlimit, phases384)
        
        if tlimit() > 15:
            h384_new = slsqp_run(h384, n384, 2000, maxiter=200)
            c384 = compute_c5(h384_new)
            if c384 < c5384:
                c5384 = c384
                h384 = h384_new
        
        if c5384 < best_c5 - 1e-12:
            best_c5 = c5384
            best_h = h384.copy()
            best_n = n384
            print("  n=384 improved: C5={:.8f}".format(best_c5))
        else:
            print("  n=384 best: C5={:.8f} (no improvement)".format(c5384))

    # === Phase 6: Coordinate descent at best resolution ===
    if tlimit() > 40:
        print("\n=== Phase 6: coordinate descent n={} ===".format(best_n))
        best_h, best_c5 = coord_descent_transfer(best_h, best_n, max_sweeps=30, rng=rng, tlimit=tlimit)
        print("  After coord descent: C5={:.8f}".format(best_c5))

    # === Phase 7: Final subgradient + SLSQP polish ===
    if tlimit() > 30:
        print("\n=== Phase 7: final polish ===")
        n = best_n
        # Quick subgradient
        phases_fin = [
            (3, 0.02, 150),
            (5, 0.01, 200),
        ]
        h_fin, c5_fin = annealed_subgradient(best_h, n, rng, tlimit, phases_fin)
        if c5_fin < best_c5 - 1e-13:
            best_c5 = c5_fin
            best_h = h_fin
        
        # SLSQP ultra-fine
        for bv in [5000, 15000, 30000]:
            if tlimit() < 5:
                break
            h_fin = slsqp_run(best_h, n, bv, maxiter=400)
            c_fin = compute_c5(h_fin)
            if c_fin < best_c5 - 1e-13:
                best_c5 = c_fin
                best_h = h_fin
        
        # Top-lag balancing final
        if tlimit() > 10:
            best_h, best_c5 = top_lag_balancing(best_h, n, rng, tlimit, n_rounds=10)
        
        print("  After final polish: C5={:.8f}".format(best_c5))

    # === Phase 8: Random restart exploration ===
    if tlimit() > 40:
        print("\n=== Phase 8: restarts ===")
        n = best_n
        for attempt in range(6):
            if tlimit() < 15:
                break
            # Perturb best
            h_pert = best_h.copy()
            n_pert = rng.randint(4, 20)
            idxs = rng.choice(n, size=n_pert, replace=False)
            for idx in idxs:
                h_pert[idx] = np.clip(h_pert[idx] + rng.randn() * 0.25, 0.0, 1.0)
            h_pert = project_feasible(h_pert, n)
            c5_p = compute_c5(h_pert)
            
            if c5_p < best_c5 * 1.04:
                # Quick optimization from perturbation
                phases_p = [(3, 0.03, 100), (5, 0.01, 100)]
                h_opt, c5_opt = annealed_subgradient(h_pert, n, rng, tlimit, phases_p)
                if c5_opt < best_c5 - 1e-12:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    print("  Restart {} improved: C5={:.8f}".format(attempt, best_c5))

    # === Final: Coordinate descent + SLSQP ===
    if tlimit() > 10:
        n = best_n
        best_h, best_c5 = coord_descent_transfer(best_h, n, max_sweeps=20, rng=rng, tlimit=tlimit)
        if tlimit() > 8:
            h_fin = slsqp_run(best_h, n, 20000, maxiter=300)
            c_fin = compute_c5(h_fin)
            if c_fin < best_c5 - 1e-13:
                best_c5 = c_fin
                best_h = h_fin
        print("  Final: C5={:.8f}".format(best_c5))

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    _G_PEN = 0.0
    _G_FOCUS = -1
    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
