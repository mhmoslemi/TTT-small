    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def lbbgls_optimize(h, n, beta, penalty, maxiter=300):
    """L-BFGS-B optimization with smooth max objective + penalty for sum constraint."""
    dx = 2.0 / n
    target = n / 2.0
    
    ncorr = 2 * n - 1
    
    def obj_and_grad(x):
        x = np.clip(x, 0, 1)  # clip for numerical safety in objective
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        exp_vals = np.exp(shifted)
        wsum = np.sum(exp_vals)
        smooth_max = m + np.log(wsum) / beta
        val = smooth_max * dx
        s = np.sum(x) - target
        val += penalty * s * s
        
        # Gradient
        weights = exp_vals / wsum
        grad = np.zeros(n)
        for k in range(ncorr):
            wk = weights[k]
            if wk < 1e-15:
                continue
            offset = k - (n - 1)
            if offset >= 0:
                grad[:n-offset] += wk * (1.0 - x[offset:])
                grad[offset:] += -wk * x[:n-offset]
            else:
                aoff = -offset
                grad[aoff:] += wk * (1.0 - x[:-aoff])
                grad[:n-aoff] += -wk * x[aoff:]
        grad *= dx
        grad += 2.0 * penalty * s * np.ones(n)
        
        return val, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x: obj_and_grad(x)[0], h, 
                      jac=lambda x: obj_and_grad(x)[1],
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxfun': maxiter * 5})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            return trial
    except Exception:
        pass
    return h.copy()

def coord_descent(h, n, max_sweeps=50, time_limit=60):
    """Coordinate descent to minimize C5."""
    best = h.copy()
    best_c5 = compute_c5(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    t_start = time.time()
    
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        improved = False
        interior = np.where((best > 1e-8) & (best < 1.0 - 1e-8))[0]
        if len(interior) == 0:
            break
        order = np.random.permutation(interior)
        
        for i in order:
            if time.time() - t_start > time_limit - 1:
                break
            # Compute objective as function of t = h[i]
            t0 = best[i]
            # corr[k] = base_k + a_k * t (for k != n-1) or base_k + a_k*t - t² (for k = n-1)
            # Actually for k = n-1 (offset 0): h[i](1-h[i]) = t - t²
            # For other k: linear in t
            
            n_off = 2 * n - 1
            a = np.zeros(n_off)  # linear coefficient
            b = np.zeros(n_off)  # quadratic coefficient
            base = np.zeros(n_off)
            
            corr_now = np.correlate(best, 1.0 - best, mode='full')
            
            for off in range(-(n-1), n):
                k = n - 1 + off
                if off == 0:
                    # h[i](1-h[i]) = t(1-t) = t - t²
                    a[k] = 1.0
                    b[k] = -1.0
                else:
                    if 0 <= i + off < n:
                        a[k] += (1.0 - best[i + off])
                    if 0 <= i - off < n:
                        a[k] += (-best[i - off])
                
                base[k] = corr_now[k] - a[k] * t0 - b[k] * t0 * t0
            
            # Maximize: min over t of max_k (base[k] + a[k]*t + b[k]*t²) * dx
            # This is a 1D minimax problem
            # Try a grid + golden section
            candidates = np.linspace(0, 1, 50)
            best_t = t0
            best_val = best_c5
            for t in candidates:
                vals = base + a * t + b * t * t
                v = np.max(vals) * dx
                if v < best_val:
                    best_val = v
                    best_t = t
            
            # Refine with golden section around best_t
            if best_t != t0 and remaining_time() > 2:
                lo = max(0.0, best_t - 0.05)
                hi = min(1.0, best_t + 0.05)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = np.max(base + a*c1 + b*c1**2) * dx
                f2 = np.max(base + a*c2 + b*c2**2) * dx
                for _ in range(30):
                    if f1 < f2:
                        hi, c2, f2 = c2, c1, f1
                        c1 = hi - gr * (hi - lo)
                        f1 = np.max(base + a*c1 + b*c1**2) * dx
                    else:
                        lo, c1, f1 = c1, c2, f1
                        c2 = lo + gr * (hi - lo)
                        f2 = np.max(base + a*c2 + b*c2**2) * dx
                best_t = (lo + hi) / 2.0
                best_val = np.max(base + a*best_t + b*best_t**2) * dx
            
            if best_val < best_c5 - 1e-14:
                best[i] = best_t
                best_c5 = best_val
                improved = True
    
    return best, best_c5
