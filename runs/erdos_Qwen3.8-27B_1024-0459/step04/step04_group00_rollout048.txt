        if off >= 0:
            npairs = n - off
            g[off:off + npairs] += wk * (1.0 - x[:npairs])
            g[:npairs] -= wk * x[off:off + npairs]
        else:
            ao = -off
            npairs = n - ao
            g[:npairs] += wk * (1.0 - x[ao:ao + npairs])
            g[ao:ao + npairs] -= wk * x[:npairs]
    return g

def _sc(x):
    return np.sum(x) - _TGT

def _scj(x):
    return np.ones(_N)

def slsqp_smooth(h, n, beta, maxiter=500):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        obj_new = compute_c5(res.x)
        obj_old = compute_c5(h)
        if obj_new < obj_old:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def top_shifts_focus(h, n, top_k=5, max_iters=200):
    h = h.copy()
    dx_v = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min_top = corr[idx[-1]]
        if c_max - c_min_top < 1e-10:
            break
        g_t = np.zeros(n)
        g_b = np.zeros(n)
        for k, g in [(idx[0], g_t), (idx[-1], g_b)]:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += (1.0 - h[:np_])
                g[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= h[:np_]
        d = g_b - g_t
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        improved = False
        for step in [0.01, 0.03, 0.005, 0.05, 0.02, 0.08, 0.015, 0.04]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5

def coord_descent_true(h, n, max_rounds=5):
    """Coordinate descent on the true (non-smooth) C5 objective."""
    h = h.copy()
    best_c5 = compute_c5(h)
    dx_v = 2.0 / n
    
    for rd in range(max_rounds):
        if remaining() < 5:
            break
        improved = False
        for j in range(n):
            if remaining() < 2:
                break
            # Try moving h[j] to 0 and 1 (and current value)
            best_val = h[j]
            best_obj = best_c5
            
            # We need to maintain sum = n/2, so we adjust a partner
            # Simple approach: try h[j] in {0, 1, current} and project
            # But projection changes other values too.
            # Alternative: fix sum by adjusting the nearest "free" variable
            
            # Actually, let's just try a few values and project
            candidates = [0.0, 1.0, h[j] - 0.1, h[j] + 0.1, h[j] - 0.3, h[j] + 0.3]
            candidates = [max(0.0, min(1.0, c)) for c in candidates]
            candidates.append(h[j])  # keep current
            seen = set()
            for cv in candidates:
                if abs(cv - best_val) < 1e-10:
                    continue
                key = round(cv, 6)
                if key in seen:
                    continue
                seen.add(key)
                h[j] = cv
                h_p = project_feasible(h, n)
                obj_p = compute_c5(h_p)
                if obj_p < best_obj - 1e-12:
                    best_obj = obj_p
                    best_val = cv
                    improved = True
            
            h[j] = best_val
            if best_val != h[j]:  # this shouldn't happen
                h[j] = best_val
            h = project_feasible(h, n)
            if compute_c5(h) < best_c5:
                best_c5 = compute_c5(h)
        
        if not improved:
            break
    
    return h, best_c5

def binary_local_search(h_init, n, max_passes=50):
    """Systematic pair-flip local search on binary vectors."""
    h = (h_init > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            rem = np.random.RandomState(0).choice(ones_idx, n_ones - n_half, replace=False)
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            add = np.random.RandomState(0).choice(zeros_idx, n_half - n_ones, replace=False)
            h[add] = 1
    
    dx_v = 2.0 / n
    corr = compute_corr(h)
    cur_c5 = float(np.max(corr)) * dx_v
    best_h = h.copy()
    best_c5 = cur_c5
    
    for p in range(max_passes):
        if remaining() < 5:
            break
        ones_list = np.where(h == 1)[0].tolist()
        zeros_list = np.where(h == 0)[0].tolist()
        best_delta = 1e-10  # need improvement of at least this
        best_pair = None
        
        # Try all pairs (or a random subset if too many)
        total_pairs = len(ones_list) * len(zeros_list)
        if total_pairs > 50000:
            # Subsample
            rng = np.random.RandomState(p)
            indices = rng.choice(total_pairs, 50000, replace=False)
            pairs = [(ones_list[i // len(zeros_list)], zeros_list[i % len(zeros_list)]) for i in indices]
        else:
            pairs = [(oi, zi) for oi in ones_list for zi in zeros_list]
        
        for oi, zi in pairs:
            # Compute delta in C5
            h[oi] = 0
            h[zi] = 1
            nc = compute_corr(h)
            nc5 = float(np.max(nc)) * dx_v
            delta = nc5 - cur_c5
            h[oi] = 1
            h[zi] = 0
            
            if delta < best_delta:
                best_delta = delta
                best_pair = (oi, zi)
        
        if best_pair is not None:
            oi, zi = best_pair
            h[oi] = 0
            h[zi] = 1
            cur_c5 += best_delta
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            break  # Local minimum
    
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
