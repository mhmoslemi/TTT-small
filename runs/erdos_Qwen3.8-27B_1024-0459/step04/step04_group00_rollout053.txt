def lbfgsb_optimize(h_init, n, beta, penalty=1e4, maxiter=5000):
    h = h_init.copy()
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        dx = 2.0/n
        corr = np.correlate(x, 1.0-x, mode='full')
        m = np.max(corr)
        shifted = np.clip(beta*(corr-m), -50, 50)
        s = np.sum(np.exp(shifted))
        val = (m + np.log(s)/beta) * dx
        sd = np.sum(x) - n/2.0
        val += penalty * sd * sd
        return val
    
    def grad(x):
        dx = 2.0/n
        corr = np.correlate(x, 1.0-x, mode='full')
        m = np.max(corr)
        shifted = np.clip(beta*(corr-m), -50, 50)
        ex = np.exp(shifted)
        s = np.sum(ex)
        w = ex/s * dx
        
        g = np.zeros(n)
        for k in range(2*n-1):
            wk = w[k]
            if wk < 1e-15:
                continue
            off = k - (n-1)
            if off >= 0:
                np_ = n - off
                g[off:off+np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off+np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao+np_])
                g[ao:ao+np_] -= wk * x[:np_]
        
        sd = np.sum(x) - n/2.0
        g += 2.0 * penalty * sd
        return g
    
    res = minimize(obj, h, jac=grad, method='L-BFGS-B',

                best_c5 = c5_new
                best_h = h.copy()
                h = h.copy()

    # Phase 8: Final coordinate descent polish
    if remaining() > 15:
        h_cd, c5_cd = coordinate_descent(best_h, n, n_passes=30, rng_seed=seed + 42)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Final: ensure feasibility and validate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        best_c5 = final_c5
    else:
        # Use previous best
        pass

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
