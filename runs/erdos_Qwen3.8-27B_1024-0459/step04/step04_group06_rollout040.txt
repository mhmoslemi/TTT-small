    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)

def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 5.0)

def obj_smooth(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(_G_BETA * shifted, -600, 600))
    s = np.sum(exps)
    return (m + np.log(s) / _G_BETA) * dx

def obj_smooth_grad(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(_G_BETA * shifted, -600, 600))
    s = np.sum(exps)
    weights = exps / s
    
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    
    # Vectorized computation
    # For each offset, the gradient contribution is:
    # w * [(1-x[m_abs:]) for positions m_abs..n-1] - w * [x[m_abs:n] for positions 0..n-1-m_abs]
    
    # Let's do it with a loop over offsets (n-1 distinct absolute offsets, each appearing twice except m=0)
    # Actually offsets go from -(n-1) to (n-1), that's 2n-1 values
    # m_abs = 0 appears once, m_abs = 1..n-1 each appear twice (offset +m and -m)
    
    # For efficiency, let's group by m_abs:
    # For m_abs = 0: weight = weights[n-1] (offset 0)
    # For m_abs = k > 0: weight = weights[n-1+k] + weights[n-1-k] (offsets +k and -k)
    
    # Actually wait, offset = k_idx - (n-1), so:
    # k_idx = n-1 corresponds to offset 0
    # k_idx = n-1+m corresponds to offset +m
    # k_idx = n-1-m corresponds to offset -m
    
    # So weight for m_abs = 0 is weights[n-1]
    # weight for m_abs = m > 0 is weights[n-1+m] + weights[n-1-m]
    
    # And the gradient formula is the same for +m and -m!
    
    # So:
    # grad[m:] += (w_+m + w_-m) * (1 - x[:n-m])
    # grad[:n-m] -= (w_+m + w_-m) * x[m:n]
    
    # For m = 0:
    w0 = weights[n-1] * dx
    # grad[0:] += w0 * (1-x[:n])
    # grad[:n] -= w0 * x[0:n]
    # So grad += w0 * ((1-x) - x) = w0 * (1 - 2*x)
    grad += w0 * (1.0 - 2.0 * x)
    
    for m in range(1, n):
        w = (weights[n-1+m] + weights[n-1-m]) * dx
        if w < 1e-15:
            continue
        n_valid = n - m
        grad[m:] += w * (1.0 - x[:n_valid])
        grad[:n_valid] -= w * x[m:n]
    
    return grad

def con_sum(x):
    return np.sum(x) - _G_TGT

def slsqp_optimize(h, n, beta, maxiter=300):
    global _G_N, _G_BETA, _G_TGT, _G_DX
    _G_N = n
    _G_BETA = float(beta)
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_smooth, h, jac=obj_smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new)
    except Exception as e:
        return h, compute_c5(h)

def subgrad_step(h, n, top_k=5, step=0.1):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    # Get top_k correlations
    top_idx = np.argsort(corr)[-top_k:]
    top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
    
    grad = np.zeros(n)
    for k_idx in top_idx:
        offset = k_idx - (n - 1)
        m_abs = abs(offset)
        n_valid = n - m_abs
        if n_valid <= 0:
            continue
        grad[m_abs:] += (1.0 - h[:n_valid])
        grad[:n_valid] -= h[m_abs:m_abs + n_valid]
    
    gn = np.linalg.norm(grad)
    if gn < 1e-14:
        return h
    grad = grad / gn
    
    # Line search
    h_best = h.copy()
    c5_best = compute_c5(h)
    alpha = step
    for _ in range(15):
        h_trial = project_feasible(h - alpha * grad, n)
        c5_trial = compute_c5(h_trial)
        if c5_trial < c5_best - 1e-13:
            h_best = h_trial
            c5_best = c5_trial
            break
        alpha *= 0.5
    
    return h_best

def subgrad_momentum(h, n, max_iter=300, top_k_schedule=None):
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    v = np.zeros(n)
    step = 0.1
    patience = 0
    
    if top_k_schedule is None:
        top_k_schedule = [3, 5, 5, 8, 10, 15]
    
    for it in range(max_iter):
        if remaining_time() < 5.0:
            break
        tk = top_k_schedule[min(it // max(1, max_iter // len(top_k_schedule)), len(top_k_schedule) - 1)]
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-tk:]
        top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
        
        grad = np.zeros(n)
        for k_idx in top_idx:
            offset = k_idx - (n - 1)
            m_abs = abs(offset)
            n_valid = n - m_abs
            if n_valid <= 0:
                continue
            grad[m_abs:] += (1.0 - h[:n_valid])
            grad[:n_valid] -= h[m_abs:m_abs + n_valid]
        
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad = grad / gn
        
        alpha = step
        improved = False
        for _ in range(12):
            v = 0.85 * v - alpha * grad
            h_trial = project_feasible(h + v, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-13:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
            alpha *= 0.5

            patience = 0
            step = min(alpha * 1.3, 0.3)
        else:
            patience += 1
            step *= 0.5
            v *= 0.4

def swap_local_search(h, n, max_rounds=20, rng=None):
    """Try swapping pairs of h values to improve C5."""
    if rng is None:
        rng = np.random.RandomState(42)
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    
    for r in range(max_rounds):
        if remaining_time() < 5.0:
            break
        improved = False
        # Try random swaps
        for _ in range(n // 2):
            i, j = sorted(rng.choice(n, 2, replace=False))
            if abs(h_best[i] - h_best[j]) < 1e-6:
                continue
            h_trial = h_best.copy()
            h_trial[i], h_trial[j] = h_best[j], h_best[i]
            h_trial = project_feasible(h_trial, n)
            c5_t = compute_c5(h_trial)
            if c5_t < c5_best - 1e-12:
                c5_best = c5_t
                h_best = h_trial
                improved = True
        # Try bit flips (for near-binary values)
        for _ in range(n // 4):
            i = rng.randint(n)
            h_trial = h_best.copy()
            h_trial[i] = 1.0 - h_best[i]
            h_trial = project_feasible(h_trial, n)
            c5_t = compute_c5(h_trial)
            if c5_t < c5_best - 1e-12:
                c5_best = c5_t
                h_best = h_trial
                improved = True
        if not improved:
            break
    
    return h_best, c5_best

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

```python
def obj_smooth_grad(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(_G_BETA * shifted, -600, 600))
    s = np.sum(exps)
    weights = exps / s
    
    grad = np.zeros(n)
    
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        
        if offset >= 0:
            # corr = sum_{i=0}^{n-1-offset} x[i] * (1-x[i+offset])
            # d/dx[j] = (1-x[j+offset]) for j in [0, n-offset), -x[j-offset] for j in [offset, n)
            n_valid = n - offset
            grad[:n_valid] += w * (1.0 - x[offset:])
            grad[offset:] -= w * x[:n_valid]
        else:
            m_abs = -offset
            # corr = sum_{i=m_abs}^{n-1} x[i] * (1-x[i-m_abs])
            # = sum_{i=0}^{n-1-m_abs} x[i+m_abs] * (1-x[i])
            # d/dx[j] = (1-x[j-m_abs]) for j in [m_abs, n), -x[j+m_abs] for j in [0, n-m_abs)
            n_valid = n - m_abs
            grad[m_abs:] += w * (1.0 - x[:n_valid])
            grad[:n_valid] -= w * x[m_abs:]
    
    return grad
```
