def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 3.0)


def obj_smooth(x):

    m = np.max(corr)
    shifted = corr - m

    return (m + np.log(np.sum(exps)) / _G_BETA) * dx

def subgrad_level(h, n, max_iter=200, step0=0.04, top_k=3):

            else:

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
