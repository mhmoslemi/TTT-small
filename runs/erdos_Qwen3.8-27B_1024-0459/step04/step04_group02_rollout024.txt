            h_left[valid_left] = best[i_minus_d[valid_left]]
            h_right[valid_right] = best[i_plus_d[valid_right]]

            a = (1.0 - h_left) - h_right
            # Self term override
            a[self_idx] = 1.0

            base = best_corr - t0 * a
            base[self_idx] = best_corr[self_idx] - t0 + t0 * t0

            # 1D minimization: min_t max_k (base[k] + a[k]*t) - t^2 at self_idx
            def obj_1d(t):
                vals = base + a * t
                vals[self_idx] -= t * t
                return np.max(vals)

            # Grid search
            grid = np.linspace(0.0, 1.0, grid_size)
            grid_f = np.array([obj_1d(t) for t in grid])
            best_grid_idx = np.argmin(grid_f)
            t_best = grid[best_grid_idx]
            f_best = grid_f[best_grid_idx]

            # Golden section refinement
            lo = max(0.0, t_best - 1.0 / (grid_size - 1) * 2)
            hi = min(1.0, t_best + 1.0 / (grid_size - 1) * 2)
            if hi - lo > 1e-12:
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = obj_1d(c1)
                f2 = obj_1d(c2)
                for _ in range(40):
                    if hi - lo < 1e-13:
                        break
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = obj_1d(c1)
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = obj_1d(c2)
                t_gs = (lo + hi) / 2.0
                f_gs = obj_1d(t_gs)
                if f_gs < f_best:
                    t_best = t_gs
                    f_best = f_gs

            # Also check boundaries
            for t_check in [0.0, 1.0]:
                fc = obj_1d(t_check)
                if fc < f_best:
                    f_best = fc
                    t_best = t_check

            new_c5 = f_best * dx
            if new_c5 < best_c5 - 1e-15:
                best[i] = t_best
                best_corr = compute_corr(best)
                best_c5 = new_c5
                improved = True
                total_improvement += best_c5

        if not improved:
            break

    return best, best_c5

def slsqp_smooth(h, n, beta, maxiter=300):
    dx = 2.0 / n

    def obj_and_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        exp_vals = np.exp(shifted)
        wsum = np.sum(exp_vals)
        weights = exp_vals / wsum
        val = (m + np.log(wsum) / beta) * dx
        grad = np.zeros(n)
        for k in range(len(corr)):
            w = weights[k]
            if w < 1e-14:
                continue
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                if npairs > 0:
                    grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                    grad[:npairs] -= w * x[offset:offset + npairs]
            else:
                aoff = -offset
                npairs = n - aoff
                if npairs > 0:
                    grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                    grad[aoff:aoff + npairs] -= w * x[:npairs]
        grad *= dx
        return val, grad

    def obj_only(x):
        return obj_and_grad(x)[0]

    def grad_only(x):
        return obj_and_grad(x)[1]

    def cons_sum(x):
        return np.sum(x) - n / 2.0

    def cons_sum_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_sum, 'jac': cons_sum_jac}]
    try:
        res = minimize(obj_only, h, jac=grad_only, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            if compute_c5(trial) <= compute_c5(h) + 1e-15:
                return trial
    except Exception:
        pass
    return h.copy()

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial solution
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        best_h = np.random.RandomState(seed).uniform(0, 1, best_n)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP smoothing at parent resolution
    n = best_n
    h = best_h.copy()
    for beta in [50, 200, 500, 2000, 5000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_smooth(h, n, beta, 300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Exact coordinate descent at parent resolution
    if remaining_time() > 30:
        h_new, c5_new = coord_descent_exact(h, n, max_sweeps=150, grid_size=100)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: Try higher resolutions
    resolutions_to_try = []
    for rn in [960, 1280, 1024, 1600]:
        if rn % 2 == 0 and rn > best_n and rn <= 2560:
            resolutions_to_try.append(rn)
    resolutions_to_try.sort()

    for rn in resolutions_to_try:
        if remaining_time() < 60:
            break
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)

        # SLSQP at new resolution
        for beta in [200, 1000, 3000]:
            if remaining_time() < 25:
                break
            h_new = slsqp_smooth(h_r, rn, beta, 250)
            c5_new = compute_c5(h_new)
            if c5_new < c5_r:
                h_r = h_new
                c5_r = c5_new

        # Coordinate descent at new resolution
        if remaining_time() > 30:
            h_new, c5_new = coord_descent_exact(h_r, rn, max_sweeps=80, grid_size=80)
            if c5_new < c5_r:
                h_r = h_new
                c5_r = c5_new

        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn

    # Phase 4: Final coordinate descent on best solution
    if remaining_time() > 20:
        h_new, c5_new = coord_descent_exact(best_h, best_n, max_sweeps=100, grid_size=120)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 5: Try random perturbation + re-optimize
    if remaining_time() > 60:
        for attempt in range(3):
            if remaining_time() < 30:
                break
            h_pert = best_h + rng.randn(best_n) * 0.05
            h_pert = project_feasible(h_pert, best_n)
            c5_pert = compute_c5(h_pert)

            if c5_pert < best_c5 + 0.01:
                h_new, c5_new = coord_descent_exact(h_pert, best_n, max_sweeps=60, grid_size=80)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()

    # Phase 6: SLSQP polish with high beta
    if remaining_time() > 15:
        for beta in [10000, 50000]:
            if remaining_time() < 10:
                break
            h_new = slsqp_smooth(best_h, best_n, beta, 200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 7: Final coordinate descent polish
    if remaining_time() > 10:
        h_new, c5_new = coord_descent_exact(best_h, best_n, max_sweeps=50, grid_size=150)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 8: Binary projection attempt
    if remaining_time() > 30:
        h_bin = (best_h > 0.5).astype(float)
        target_sum = best_n / 2.0
        # Adjust to get exactly n/2 ones
        n_ones = int(np.sum(h_bin))
        if n_ones > int(target_sum):
            # Remove some ones (those with lowest "importance")
            ones = np.where(h_bin == 1)[0]
            # Remove from the ones with highest h value (least "committed")
            order = np.argsort(-best_h[ones])
            for idx in order[:n_ones - int(target_sum)]:
                h_bin[ones[idx]] = 0.0
        elif n_ones < int(target_sum):
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-best_h[zeros])
            for idx in order[:int(target_sum) - n_ones]:
                h_bin[zeros[idx]] = 1.0

        c5_bin = compute_c5(h_bin)
        # Try to improve binary with coordinate descent (will push toward fractional)
        if c5_bin < best_c5 + 0.005:
            h_new, c5_new = coord_descent_exact(h_bin, best_n, max_sweeps=80, grid_size=100)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Final verification
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
