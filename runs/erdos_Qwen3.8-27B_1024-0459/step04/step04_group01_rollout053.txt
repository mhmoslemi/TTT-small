    - Its own contribution (1-h[i+d]) is removed (decrease by (1-h[i+d]))
    - Its partner's contribution h[i-d]*(1-h[i]) increases by h[i-d] (since 1-h[i] goes from 0 to 1)
  - Net change at shift d for h[i]: -(1-h[i+d]) + h[i-d] (with bounds checking)
  
- Similarly for h[j] going from 0 to 1:
  - Its own contribution (1-h[j+d]) is added (increase by (1-h[j+d]))
  - Its partner's contribution h[j-d]*(1-h[j]) decreases by h[j-d] (since 1-h[j] goes from 1 to 0)
  - Net change at shift d for h[j]: (1-h[j+d]) - h[j-d]

Total delta at shift d: [-(1-h[i+d]) + h[i-d]] + [(1-h[j+d]) - h[j-d]] (with bounds)

For binary h, 1-h[x] is just the complement.

This can be computed for all d in O(n) time per swap. With n=312, that's 623 operations per swap evaluation. If we try 1000 swaps per iteration, that's 623K operations, which is fast.

But we need to be smarter about candidate selection. Instead of trying all O(n²) possible swaps, we'll focus on:
1. Swaps where i is a 1 that contributes to the worst shift
2. Swaps where j is a 0 that would "absorb" the contribution

def swap_delta(h, i, j, n):
    """Compute delta in correlation for swapping h[i]:1->0 and h[j]:0->1.
    Returns array of length 2n-1 with the change at each shift index."""
    # Shift index m in [0, 2n-2] corresponds to shift d = m - (n-1)
    # For binary h, the change at shift d:
    #   h[i] 1->0: -(1-h[partner_i]) + h[partner_i2]  (bounds checked)
    #   h[j] 0->1: (1-h[partner_j]) - h[partner_j2]   (bounds checked)
    # where for shift d:
    #   partner_i (as first element) = i+d if in [0,n-1]
    #   partner_i2 (as second element) = i-d if in [0,n-1]
    delta = np.zeros(2 * n - 1)
    # For i going 1->0:
    #   As h[i] in sum h[k]*(1-h[k+d]): contributes at shift d if i>=0, i+d<n
    #     -> d in [-i, n-1-i], partner at i+d, contribution = (1-h[i+d])
    #   As h[i-d] in sum h[k-d]*(1-h[k]): contributes at shift d if i-d in [0,n-1-d]... 
    #     Actually: h[i] appears as the (1-h[...]) factor when k = i-d, so k >= 0 means d <= i,
    #     and k+d = i < n always. Also k <= n-1-d means i-d <= n-1-d means i <= n-1 (always).
    #     So for shift d: if i-d >= 0 (i.e., d <= i), h[i-d]*(1-h[i]) is a term.
    #     When h[i]: 1->0, (1-h[i]): 0->1, so term increases by h[i-d] = h[max(0,i-d)]
    #     Contribution: +h[i-d] for d <= i and i-d >= 0
    #   As h[i] factor: if i in [0, n-1-d] (i.e., i+d <= n-1, d <= n-1-i) and i >= 0
    #     h[i]*(1-h[i+d]) is a term. When h[i]: 1->0, term goes from (1-h[i+d]) to 0.
    #     Contribution: -(1-h[i+d]) for d >= -(n-1) and i+d in [0,n-1] and d >= -(something)
    #     More precisely: d such that 0 <= i+d <= n-1, i.e., -i <= d <= n-1-i
    
    # i: 1 -> 0
    # Term as first factor h[i]*(1-h[i+d]): valid when 0<=i<n and 0<=i+d<n -> -i <= d <= n-1-i
    d_lo = max(-(n - 1), -i)
    d_hi = min(n - 1, n - 1 - i)
    if d_lo <= d_hi:
        partners = np.arange(d_lo, d_hi + 1) + i  # i+d
        delta[partners + (n - 1)] -= (1.0 - h[partners])
    # Term as (1-h[i]) factor: h[i-d]*(1-h[i]), valid when 0<=i-d<n and 0<=i<n -> 0<=i-d<=n-1 -> 0<=d<=i
    # Also need i-d+d=i < n (always) and i-d >= 0, i-d <= n-1
    d_lo2 = max(-(n - 1), 0)  # d >= 0 since i-d >= 0
    d_hi2 = min(n - 1, i)
    if d_lo2 <= d_hi2:
        partners2 = np.arange(d_lo2, d_hi2 + 1)  # d values
        src = i - partners2  # i-d
        delta[partners2 + (n - 1)] += h[src]
    
    # j: 0 -> 1
    # Term as first factor h[j]*(1-h[j+d]): was 0, now (1-h[j+d])
    d_lo = max(-(n - 1), -j)
    d_hi = min(n - 1, n - 1 - j)
    if d_lo <= d_hi:
        partners = np.arange(d_lo, d_hi + 1) + j
        delta[partners + (n - 1)] += (1.0 - h[partners])
    # Term as (1-h[j]) factor: h[j-d]*(1-h[j]), was h[j-d]*1, now h[j-d]*0
    d_lo2 = max(-(n - 1), 0)
    d_hi2 = min(n - 1, j)
    if d_lo2 <= d_hi2:
        partners2 = np.arange(d_lo2, d_hi2 + 1)
        src = j - partners2
        delta[partners2 + (n - 1)] -= h[src]
    
    return delta

def continuous_refine(h, n, time_limit):
    """Quick projected gradient refinement from a binary starting point."""
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    dx_val = 2.0 / n
    step = 0.01
    start = time.time()
    
    for it in range(2000):
        if time.time() - start > time_limit or remaining() < 5:
            break
        corr = compute_corr(h)
        m = np.max(corr)
        # Subgradient from top shifts
        top3 = np.argsort(-corr)[:3]
        wts = np.exp(-(corr[top3] - corr[top3[0]]) * 50)
        wts /= wts.sum()
        
        grad = np.zeros(n)
        for w, k in zip(wts, top3):
            k = int(k)
            d = k - (n - 1)
            if d >= 0:
                np_ = n - d
                grad[:np_] += w * (1.0 - h[d:d + np_])
                grad[d:d + np_] -= w * h[:np_]
            else:
                ao = -d
                np_ = n - ao
                grad[:np_] += w * (1.0 - h[ao:ao + np_])
                grad[ao:ao + np_] -= w * h[:np_]
        
        h_trial = h - step * grad
        h_trial = np.clip(h_trial, 0, 1)
        # Project to sum constraint
        diff = n / 2.0 - np.sum(h_trial)
        h_trial += diff / n
        h_trial = np.clip(h_trial, 0, 1)
        
        c_new = compute_c5(h_trial)
        if c_new < best_c5:
            best_c5 = c_new
            h = h_trial.copy()
            step = min(step * 1.02, 0.05)
        else:
            step *= 0.92
    
    return h, best_c5

                if d >= 0:
                    np_ = n - d
                    g[:np_] += w[k] * (1.0 - x[d:d + np_])
                    g[d:d + np_] -= w[k] * x[:np_]
                else:
                    ao = -d
                    np_ = n - ao
                    g[:np_] += w[k] * (1.0 - x[ao:ao + np_])
                    g[ao:ao + np_] -= w[k] * x[:np_]
        return g * dx_val
    
    def sum_fun(x):
        return np.sum(x) - n / 2.0
    
    def sum_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
    except Exception:
        parent_n = 312
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
    
    best_h = parent_h.copy()
    best_c5 = compute_c5(best_h)
    best_n = parent_n
    
    # Candidate n values (must be even)
    n_candidates = []
    for dn in [0, -8, -4, 4, 8, 16, -16, 24, 32, 48, 64, -32, 72, 80, 96, -24]:
        n_t = parent_n + dn
        if 120 <= n_t <= 600 and n_t % 2 == 0:
            n_candidates.append(n_t)
    # Add some standard sizes
    for ns in [312, 320, 336, 352, 384, 400, 256, 288]:
        if ns % 2 == 0 and ns not in n_candidates:
            n_candidates.append(ns)
    n_candidates = sorted(set(n_candidates))
    
    # Sort by potential (smaller dx * expected_max correlates with better C5)
    # Prioritize n where 2/n is "nice"
    n_candidates.sort(key=lambda x: -min(abs(x - 312), abs(x - 320), abs(x - 384)))
    n_candidates = n_candidates[:12]  # Limit to 12 values
    
    for n in n_candidates:
        if remaining() < 25:
            break
        
        time_for_n = min(remaining() - 10, max(10, (remaining() - 20) / (len(n_candidates) - n_candidates.index(n))))
        
        # Initialize for this n
        if n == parent_n:
            h_init = parent_h.copy()
        else:
            h_init = interp_to(best_h, best_n, n)
            h_init = project_feasible(h_init, n)
        
        h_bin = make_binary(h_init, n)
        
        # Phase A: Binary local search v3 (main workhorse)
        t_search = max(5, min(time_for_n * 0.6, remaining() - 15))
        h_bin_opt, c5_bin = binary_local_search_v3(h_bin, n, t_search, rng)
        
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin_opt.copy()
            best_n = n
        
        # Phase B: Try from sorted initial
        if remaining() > 20 and n != parent_n:
            h_sorted = np.zeros(n)
            h_sorted[:n // 2] = 1.0
            t_s2 = max(5, min(20, remaining() - 15))
            h_s_opt, c5_s = binary_local_search_v3(h_sorted, n, t_s2, rng)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s_opt.copy()
                best_n = n
        
        # Phase C: Quick continuous refinement on best binary
        if remaining() > 15:
            t_cont = max(3, min(15, remaining() - 10))
            h_cont, c5_cont = continuous_refine(h_bin_opt, n, t_cont)
            if c5_cont < best_c5:
                best_c5 = c5_cont
                best_h = h_cont.copy()
                best_n = n
        
        # Phase D: SLSQP polish
        if remaining() > 12:
            h_slq = slsqp_quick(h_bin_opt, n, 500, maxiter=150)
            c5_slq = compute_c5(h_slq)
            if c5_slq < best_c5:
                best_c5 = c5_slq
                best_h = h_slq.copy()
                best_n = n
    
    # Final intensive phase on best solution
    n_final = best_n
    h_final = best_h.copy()
    
    if remaining() > 20:
        # More local search on best
        h_bin_final = make_binary(h_final, n_final)
        t_final_search = max(10, min(remaining() - 15, 80))
        h_fs, c5_fs = binary_local_search_v3(h_bin_final, n_final, t_final_search, rng)
        if c5_fs < best_c5:
            best_c5 = c5_fs
            best_h = h_fs.copy()
            best_n = n_final
    
    if remaining() > 15:
        # Continuous refine
        t_cr = max(5, min(remaining() - 10, 40))
        h_cr, c5_cr = continuous_refine(best_h, best_n, t_cr)
        if c5_cr < best_c5:
            best_c5 = c5_cr
            best_h = h_cr.copy()
    
    if remaining() > 10:
        # SLSQP
        h_slq = slsqp_quick(best_h, best_n, 2000, maxiter=300)
        c5_slq = compute_c5(h_slq)
        if c5_slq < best_c5:
            best_c5 = c5_slq
            best_h = h_slq.copy()
    
    if remaining() > 8:
        # One more binary search round
        h_b2 = make_binary(best_h, best_n)
        h_b2s, c5_b2 = binary_local_search_v3(h_b2, best_n, max(3, remaining() - 6), rng)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2s.copy()
    
    # Validate
    best_h = np.clip(best_h, 0, 1)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5
    best_n = len(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
