_START_TIME = 0.0
_BUDGET_S = 1300.0
_N_WORK = 0
_BETA_WORK = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    # Project onto {0≤h≤1, sum(h)=n/2}
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        # Distribute uniformly among non-bound variables
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            # All at boundary - need to move one
            if diff > 0:
                idx = np.argmin(h)
            else:
                idx = np.argmax(h)
            h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Find the variable with most room
        if diff > 0:
            headroom = 1.0 - h
            idx = np.argmax(headroom)
        else:
            headroom = h
            idx = np.argmin(headroom)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def _smooth_obj(x):
    n = _N_WORK
    beta = _BETA_WORK
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    return (m + np.log(wsum) / beta) * dx

def _smooth_grad(x):
    n = _N_WORK
    beta = _BETA_WORK
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset+npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset+npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff+npairs])
                grad[aoff:aoff+npairs] -= w * x[:npairs]
    grad *= dx
    return grad

def _cons_sum(x):
    return np.sum(x) - _N_WORK / 2.0

def _cons_sum_jac(x):
    return np.ones(_N_WORK)

def slsqp_refine(h, n, beta, maxiter):
    global _N_WORK, _BETA_WORK
    _N_WORK = n
    _BETA_WORK = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_sum, 'jac': _cons_sum_jac}]
    try:
        res = minimize(_smooth_obj, h.copy(), jac=_smooth_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            if compute_c5(trial) <= compute_c5(h) + 1e-14:
                return trial
    except Exception:
        pass
    return h.copy()

def vectorized_coord_descent(h, n, max_sweeps=100, num_grid=300):
    """Fast vectorized coordinate descent for the exact (non-smooth) objective."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    corr = compute_corr(best_h)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        n_improved = 0
        
        for i in range(n):
            t_old = best_h[i]
            if t_old < 1e-8 or t_old > 1.0 - 1e-8:
                continue  # Skip boundary entries
            
            # Compute coefficients for each lag
            # coeff[k] for offset = k - (n-1)
            coeff = np.zeros(ncorr)
            for off in range(-(n-1), n):
                k = n - 1 + off
                if off == 0:
                    continue  # Handle separately
                c = 0.0
                if 0 <= i - off < n:
                    c += (1.0 - best_h[i - off])
                if 0 <= i + off < n:
                    c -= best_h[i + off]
                coeff[k] = c
            
            # For off=0: the change is t*(1-t) - t_old*(1-t_old)
            # = t - t² - t_old + t_old²
            
            # Evaluate on grid
            t_grid = np.linspace(0.0, 1.0, num_grid)
            # For off≠0: new_corr[k] = corr[k] + coeff[k] * (t - t_old)
            delta = (t_grid[:, None] - t_old) * coeff[None, :]
            new_corr = corr[None, :] + delta
            # For off=0: new_corr[n-1] = corr[n-1] + t*(1-t) - t_old*(1-t_old)
            off0_vals = corr[n-1] + t_grid * (1.0 - t_grid) - t_old * (1.0 - t_old)
            new_corr[:, n-1] = off0_vals
            
            c5_vals = np.max(new_corr, axis=1) * dx
            idx_best = np.argmin(c5_vals)
            
            if c5_vals[idx_best] < best_c5 - 1e-14:
                best_h[i] = t_grid[idx_best]
                corr = compute_corr(best_h)
                best_c5 = c5_vals[idx_best]
                improved = True
                n_improved += 1
        
        if not improved:
            break
    
    return best_h, best_c5
