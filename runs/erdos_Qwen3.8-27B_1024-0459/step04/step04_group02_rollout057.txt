```python
def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):

def project_feasible(h, n):

def smooth_slsqp(h, n, beta, maxiter):
    """Smooth max surrogate optimization via SLSQP."""
    dx = 2.0 / n
    ncorr = 2 * n - 1

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        wsum = np.sum(np.exp(shifted))
        return (m + np.log(wsum) / beta) * dx

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        exp_vals = np.exp(shifted)
        wsum = np.sum(exp_vals)
        weights = exp_vals / wsum
        grad = np.zeros(n)
        for k in range(ncorr):
            w = weights[k]
            if w < 1e-10:
                continue
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                if npairs > 0:
                    grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                    grad[:npairs] -= w * x[offset:offset + npairs]
            else:
                aoff = -offset
                npairs = n - aoff
                if npairs > 0:
                    grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                    grad[aoff:aoff + npairs] -= w * x[:npairs]
        grad *= dx
        return (obj(x), grad)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]

    try:
        res = minimize(obj_grad, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter,
                      'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            if compute_c5(trial) <= compute_c5(h) + 1e-14:
                return trial
    except Exception:
        pass
    return h.copy()
```

```python
    a[n : 2*n - 1 - j] += (1.0 - h[j+1 : n])

# j as second index: a[n-1-j : n-1] += -h[0 : j]
if j > 0:
    a[n-1-j : n-1] += -h[0 : j]
```

This is O(n) per coordinate, vectorized.

```python
import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            idx = np.argmin(h) if diff > 0 else np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        for j in range(n):
            if h[j] > 1e-14 and h[j] < 1.0 - 1e-14:
                h[j] = np.clip(h[j] + diff, 0.0, 1.0)
                break
    h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def coord_descent(h, n, max_sweeps=10, rng=None):
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    if rng is None:
        rng = np.random.RandomState(0)

    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        order = rng.permutation(n)

        for i in order:
            if remaining_time() < 1.5:
                break
            t0 = best[i]
            a = np.zeros(ncorr)
            b = np.zeros(ncorr)
            a[n - 1] = 1.0
            b[n - 1] = -1.0
            if i + 1 < n:
                a[n : 2 * n - 1 - i] += (1.0 - best[i + 1 : n])
            if i > 0:
                a[n - 1 - i : n - 1] += -best[0 : i]

            base = best_corr - a * t0 - b * t0 * t0

            # Coarse grid + golden section
            grid = np.concatenate([
                np.linspace(0, 1, 21),
                np.array([t0 - 0.3, t0 - 0.1, t0 - 0.03, t0, t0 + 0.03, t0 + 0.1, t0 + 0.3])
            ])
            grid = np.clip(grid, 0.0, 1.0)
            grid = np.unique(grid)

            best_t = t0
            best_val_local = np.max(base) * dx

            vals_grid = (base[None, :] + a[None, :] * grid[:, None] + b[None, :] * grid[:, None] ** 2)
            obj_grid = np.max(vals_grid, axis=1) * dx
            idx_best = np.argmin(obj_grid)
            if obj_grid[idx_best] < best_val_local - 1e-14:
                best_val_local = obj_grid[idx_best]
                best_t = grid[idx_best]

            # Golden section refinement
            if abs(best_t - t0) > 1e-9:
                lo = max(0.0, best_t - 0.08)
                hi = min(1.0, best_t + 0.08)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                f2 = np.max(base + a * c2 + b * c2 * c2) * dx
                for _ in range(40):
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = np.max(base + a * c2 + b * c2 * c2) * dx
                t_gs = (lo + hi) / 2.0
                f_gs = np.max(base + a * t_gs + b * t_gs * t_gs) * dx
                if f_gs < best_val_local - 1e-15:
                    best_val_local = f_gs
                    best_t = t_gs

            if best_val_local < best_c5 - 1e-14:
                best[i] = best_t
                best_corr = compute_corr(best)
                best_c5 = compute_c5(best)
                improved = True

        if not improved and sweep > 0:
            break

    return best, best_c5

def smooth_slsqp(h, n, beta, maxiter):
    dx = 2.0 / n
    ncorr = 2 * n - 1

    def _obj_val(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        wsum = np.sum(np.exp(shifted))
        return (m + np.log(wsum) / beta) * dx

    def _obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        exp_vals = np.exp(shifted)
        wsum = np.sum(exp_vals)
        weights = exp_vals / wsum
        val = (m + np.log(wsum) / beta) * dx
        grad = np.zeros(n)
        for k in range(ncorr):
            w = weights[k]
            if w < 1e-12:
                continue
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                if npairs > 0:
                    grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                    grad[:npairs] -= w * x[offset:offset + npairs]
            else:
                aoff = -offset
                npairs = n - aoff
                if npairs > 0:
                    grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                    grad[aoff:aoff + npairs] -= w * x[:npairs]
        grad *= dx
        return val, grad

    def _csum(x):
        return np.sum(x) - n / 2.0

    def _csum_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _csum, 'jac': _csum_jac}]

    try:
        res = minimize(_obj_grad, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter,
                      'ftol': 1e-20, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            if compute_c5(trial) <= compute_c5(h) + 1e-14:
                return trial
    except Exception:
        pass
    return h.copy()

def binary_start(n, rng):
    h = np.zeros(n)
    indices = rng.permutation(n)
    h[indices[:n // 2]] = 1.0
    return project_feasible(h, n)
```
