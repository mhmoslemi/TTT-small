def compute_corr(h):

def project_feasible(h, n):

def pg_descent(h, n, beta, max_iter=5000, lr_init=0.01, backtracking_cap=50):
    """Projected gradient descent with backtracking line search."""
    h = project_feasible(h.copy(), n)
    f = smooth_obj(h, n, beta)
    best_h = h.copy()
    best_f = f
    lr = lr_init
    
    for it in range(max_iter):
        if remaining() < 5:
            break
        g = smooth_grad(h, n, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        
        # Backtracking line search
        accepted = False
        for bt in range(backtracking_cap):
            h_new = project_feasible(h - lr * g, n)
            f_new = smooth_obj(h_new, n, beta)
            # Armijo condition
            if f_new < f - 1e-4 * lr * gn * gn:
                h = h_new
                f = f_new
                accepted = True
                break
            lr *= 0.5
            if lr < 1e-10:
                break
        
        if not accepted:
            # Try smaller step
            lr = max(lr * 0.1, 1e-12)
            h_new = project_feasible(h - lr * g, n)
            f_new = smooth_obj(h_new, n, beta)
            if f_new < f:
                h = h_new
                f = f_new
            else:
                # Try random perturbation

                    if f_pert < f:

                        f = f_pert
                        lr = lr_init * 0.1

                    break

            best_h = h.copy()
        
        # Adaptive lr
        if accepted and np.random.rand() < 0.1:
            lr = min(lr * 1.05, lr_init * 5)

    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    s = np.sum(ex)
    w = ex / s
    g = np.zeros(n)
    off_base = n - 1
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        off = k - off_base
        if off >= 0:
            npairs = n - off
            g[off:n] += wk * (1.0 - h[:npairs])
            g[:npairs] -= wk * h[off:off + npairs]
        else:
            ao = -off
            npairs = n - ao
            g[:npairs] += wk * (1.0 - h[ao:ao + npairs])
            g[ao:n] -= wk * h[:npairs]
    return g


def pg_descent(h, n, beta, max_iter=10000, lr_init=0.01):
    h = project_feasible(h.copy(), n)
    f = smooth_val(h, beta)
    best_h = h.copy()
    best_f = f
    lr = lr_init
    stall = 0

    for it in range(max_iter):
        if remaining() < 4:
            break
        g = smooth_grad(h, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        accepted = False
        for _bt in range(30):
            h_new = project_feasible(h - lr * g, n)
            f_new = smooth_val(h_new, beta)
            if f_new < f - 1e-10 * lr * gn:
                h = h_new
                f = f_new
                accepted = True
                break
            lr *= 0.5
            if lr < 1e-14:
                break

        if not accepted:
            lr = max(lr_init * 0.01, 1e-14)
            h_new = project_feasible(h - lr * g, n)
            f_new = smooth_val(h_new, beta)
            if f_new < f:
                h = h_new
                f = f_new
                lr = lr_init * 0.01
                accepted = True

        if f < best_f - 1e-14:
            best_f = f
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if accepted:
            lr = min(lr * 1.02, lr_init * 5.0)

        if stall > 500 and it < max_iter - 100:
            scale = 0.03 * (1.0 - stall / 1000.0)
            if scale > 0:
                h_pert = project_feasible(h + scale * (np.random.randn(n) - 0.5), n)
                f_pert = smooth_val(h_pert, beta)
                if f_pert < f:
                    h = h_pert
                    f = f_pert
                    lr = lr_init * 0.5
                    stall = 0

    return best_h


def slsqp_refine(h, n, beta, maxiter=300):
    h = project_feasible(h, n)
    target = n / 2.0

    def obj(x):
        return smooth_val(x, beta)

    def grad(x):
        return smooth_grad(x, beta)

    def cfun(x):
        return np.sum(x) - target

    def cjac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-17, 'iprint': 0})
        if res.fun < smooth_val(h, beta) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def binary_sa(h_init, n, time_limit=90, seed=42):
    rng = np.random.RandomState(seed)
    h = (h_init > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            rem = rng.choice(ones_idx, n_ones - n_ones + n_half, replace=False) if n_ones > n_half else np.array([])
            rem = rng.choice(ones_idx, n_ones - n_half, replace=False)
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            add = rng.choice(zeros_idx, n_half - n_ones, replace=False)
            h[add] = 1

    dx_v = 2.0 / n
    corr = compute_corr(h)
    cur_c5 = float(np.max(corr)) * dx_v
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    n_gen = 40000
    T0 = 0.8
    T1 = 0.0005

    for gen in range(n_gen):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break
        T = T0 * (T1 / T0) ** (gen / n_gen)
        if T < 0.0002:
            break

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_delta = 0.0
        best_move = None
        for _ in range(5):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0
            h[j] = 1
            nc = compute_corr(h)
            nc5 = float(np.max(nc)) * dx_v
            delta = nc5 - cur_c5
            h[i] = 1
            h[j] = 0
            if delta < best_delta:
                best_delta = delta
                best_move = (i, j)

        if best_move is None:
            continue
        i, j = best_move
        if best_delta <= 0 or rng.rand() < np.exp(-best_delta / max(T, 1e-10)):
            h[i] = 0
            h[j] = 1
            cur_c5 += best_delta
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def block_perturb(h, n, rng, block_sizes=(10, 20, 30, 50)):
    h_p = h.copy()
    bs = block_sizes[rng.randint(len(block_sizes))]
    start = rng.randint(0, max(1, n - bs))
    seg = h_p[start:start + bs].copy()
    h_p[start:start + bs] = 1.0 - seg
    h_p = project_feasible(h_p, n)
    return h_p


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Smooth gradient descent with increasing beta
    schedule = [
        (500, 4000, 0.01),
        (1000, 5000, 0.008),
        (3000, 5000, 0.006),
        (10000, 6000, 0.005),
        (30000, 8000, 0.004),
    ]
    for beta, iters, lr in schedule:
        if remaining() < 12:
            break
        h_new = pg_descent(h, n, beta, max_iter=iters, lr_init=lr)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: SLSQP refinement
    for beta, mi in [(5000, 400), (15000, 350), (40000, 250)]:
        if remaining() < 12:
            break
        h_new = slsqp_refine(h, n, beta, maxiter=mi)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: Block perturbations + re-optimize (escape local optima)
    n_perturb = 6
    for p_i in range(n_perturb):
        if remaining() < 30:
            break
        h_pert = block_perturb(best_h, best_n, rng)
        h_opt = pg_descent(h_pert, best_n, 10000, max_iter=4000, lr_init=0.005)
        h_opt = slsqp_refine(h_opt, best_n, 20000, maxiter=200)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5 - 1e-13:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h = h_opt.copy()

    # Phase 4: Resolution exploration
    n_candidates = []
    for dn in [-6, -4, -2, 2, 4, 6, 10, 14, 18, 24]:
        nc = best_n + dn
        if 256 <= nc <= 600 and nc % 2 == 0 and nc != best_n:
            n_candidates.append(nc)
    n_candidates = sorted(set(n_candidates))

    for n_t in n_candidates[:5]:
        if remaining() < 30:
            break
        h_t = interp_to(best_h, best_n, n_t)
        h_t = project_feasible(h_t, n_t)
        h_t = pg_descent(h_t, n_t, 3000, max_iter=4000, lr_init=0.006)
        h_t = pg_descent(h_t, n_t, 15000, max_iter=5000, lr_init=0.004)
        h_t = slsqp_refine(h_t, n_t, 20000, maxiter=200)
        c5_t = compute_c5(h_t)
        if c5_t < best_c5 - 1e-13:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t

    # Phase 5: Binary SA + continuous refinement
    if remaining() > 50:
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        if n_ones != best_n // 2:
            sort_idx = np.argsort(-best_h)
            h_bin = np.zeros(best_n)
            h_bin[sort_idx[:best_n // 2]] = 1.0

        sa_time = min(200, max(20, int(remaining() * 0.3)))
        h_b, c5_b = binary_sa(h_bin, best_n, time_limit=sa_time, seed=seed)

        # Continuous refinement of best binary
        h_br = h_b.copy()
        if remaining() > 20:
            h_br = pg_descent(h_br, best_n, 8000, max_iter=5000, lr_init=0.004)
            h_br = slsqp_refine(h_br, best_n, 20000, maxiter=200)
        c5_br = compute_c5(h_br)

        if c5_br < best_c5:
            best_c5 = c5_br
            best_h = h_br.copy()
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

        # Second SA from Thue-Morse
        if remaining() > 30:
            h_tm = np.zeros(best_n)
            for i in range(best_n):
                h_tm[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
            n_ones_tm = int(np.sum(h_tm))
            if n_ones_tm != best_n // 2:
                if n_ones_tm > best_n // 2:
                    oi = np.where(h_tm == 1)[0]
                    h_tm[oi[rng.choice(len(oi), n_ones_tm - best_n // 2, replace=False)]] = 0
                else:
                    zi = np.where(h_tm == 0)[0]
                    h_tm[zi[rng.choice(len(zi), best_n // 2 - n_ones_tm, replace=False)]] = 1
            sa_time2 = min(60, max(10, int(remaining() * 0.15)))
            if sa_time2 > 5:
                h_tm_b, c5_tm = binary_sa(h_tm, best_n, time_limit=sa_time2, seed=seed + 7)
                if c5_tm < best_c5:
                    best_c5 = c5_tm
                    best_h = h_tm_b.copy()

    # Phase 6: Final intensive SLSQP + gradient polish
    h = best_h.copy()
    n = best_n
    for beta, mi in [(10000, 400), (30000, 350), (60000, 250), (100000, 200)]:
        if remaining() < 6:
            break
        h_new = slsqp_refine(h, n, beta, maxiter=mi)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining() > 8:
        h_new = pg_descent(best_h, best_n, 80000, max_iter=6000, lr_init=0.002)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-12:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
