def binary_local_search(h, n, max_iters=10000):
    # h is binary (0 or 1) with sum = n/2
    # Try swaps: pick i where h[i]=1, j where h[j]=0, swap them
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    curr_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        best_delta = 1e-15
        best_i, best_j = -1, -1
        
        # Random subset of candidates
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        
        for i in ones[:50]:
            for j in zeros[:50]:
                h[i], h[j] = h[j], h[i]  # swap
                new_c5 = compute_c5(h)
                h[i], h[j] = h[j], h[i]  # unswap
                if new_c5 < curr_c5 - best_delta:
                    best_delta = curr_c5 - new_c5
                    best_i, best_j = i, j
        
        if best_i >= 0:
            h[best_i], h[best_j] = h[best_j], h[best_i]
            curr_c5 -= best_delta
        else:
            break
    
    return h, curr_c5

from scipy.optimize import minimize, fmin_l_bfgs_b
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt == 0:
            h[:] += diff / n
            h = np.clip(h, 0.0, 1.0)
            continue
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        # Sort by proximity to 0.5 and adjust
        dist = np.abs(h - 0.5)
        order = np.argsort(dist)
        for idx in order:
            new_val = h[idx] + diff
            if 0.0 <= new_val <= 1.0:
                h[idx] = new_val
                break
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def smooth_max_and_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    weights = exp_s / total
    lse_val = m + np.log(total) / beta
    
    # Gradient of lse_val w.r.t. h
    hw_full = np.correlate(h, weights, mode='full')
    idx = np.arange(n)
    grad = dx * (1.0 - hw_full[n-1-idx] - hw_full[n-1+idx])
    
    return lse_val * dx, grad

def obj_lbgfs(h, n, beta, penalty):
    val, grad_sm = smooth_max_and_grad(h, n, beta)
    sum_diff = np.sum(h) - n/2
    obj = val + penalty * sum_diff**2
    grad = grad_sm + 2*penalty*sum_diff
    return obj, grad

def lbfgs_optimize(h, n, beta, penalty, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    res = fmin_l_bfgs_b(
        lambda x: obj_lbgfs(x, n, beta, penalty)[0],
        h.copy(),
        fprime=lambda x: obj_lbgfs(x, n, beta, penalty)[1],
        pgtol=1e-12,
        maxiter=maxiter,
        iprint=0,
        bounds=bounds
    )
    return project_feasible(res, n)

def binary_round(h, n):
    """Round to binary, keeping exactly n/2 ones."""
    h_bin = np.zeros(n)
    # Sort by value descending, take top n/2
    order = np.argsort(h)[::-1]
    h_bin[order[:n//2]] = 1.0
    return h_bin

def binary_local_search(h_bin, n, max_sweeps=2000, time_limit=30):
    """Local search on binary solutions via swaps."""
    h = h_bin.copy()
    curr_c5 = compute_c5(h)
    corrections = np.correlate(h, 1.0-h, mode='full')
    t0 = time.time()
    
    for sweep in range(max_sweeps):
        if remaining_time() < 5 or (time.time() - t0) > time_limit:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_improvement = 1e-15
        best_pair = None
        
        # Evaluate all swaps (or a random subset for very large n)
        for i in ones:
            for j in zeros:
                # Try swap
                h[i] = 0
                h[j] = 1
                new_corr = np.correlate(h, 1.0-h, mode='full')
                new_max = np.max(new_corr)
                if new_max * (2.0/n) < curr_c5 - best_improvement:
                    best_improvement = curr_c5 - new_max * (2.0/n)
                    best_pair = (i, j)
                h[i] = 1
                h[j] = 0
        
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0
            h[j] = 1
            curr_c5 -= best_improvement
        else:
            break
    
    return h, curr_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt == 0:
            h[:] += diff / n
            h = np.clip(h, 0.0, 1.0)
            continue
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        dist = np.abs(h - 0.5)
        order = np.argsort(dist)
        for idx in order:
            nv = h[idx] + diff
            if 0.0 <= nv <= 1.0:
                h[idx] = nv
                break
    return h

def smooth_max_obj_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    weights = exp_s / total
    lse_val = m + np.log(total) / beta
    hw_full = np.correlate(h, weights, mode='full')
    idx = np.arange(n)
    grad = dx * (1.0 - hw_full[n - 1 - idx] - hw_full[n - 1 + idx])
    return lse_val * dx, grad

def obj_fn(h, n, beta, penalty):
    val, grad_sm = smooth_max_obj_grad(h, n, beta)
    sd = np.sum(h) - n / 2.0
    return val + penalty * sd * sd

def grad_fn(h, n, beta, penalty):
    val, grad_sm = smooth_max_obj_grad(h, n, beta)
    sd = np.sum(h) - n / 2.0
    return grad_sm + 2.0 * penalty * sd

    bounds = [(0.0, 1.0)] * n

        return obj_fn(x, n, beta, penalty)
    def g(x):
        return grad_fn(x, n, beta, penalty)

        res = fmin_l_bfgs_b(f, h0, fprime=g, pgtol=1e-14, maxiter=maxiter,
                            iprint=0, bounds=bounds, factr=1e12)

        return project_feasible(h0, n)

            h_bl3, c5_bl3 = binary_local_search_subsample(h_bin3, best_n, max_sweeps=800, time_budget_s=45, subsample=25)
        if c5_bl3 < best_c5:
            best_c5 = c5_bl3
            best_h = h_bl3.copy()
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
