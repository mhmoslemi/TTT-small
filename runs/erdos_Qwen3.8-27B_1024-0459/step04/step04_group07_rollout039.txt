def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))

    h = np.clip(h, 0.0, 1.0)

def slsqp_optimize(h0, n, beta, maxiter=300, focus_lags=None):
    """SLSQP with soft-max objective and optional lag focusing."""
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def objective(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = corr - m
        e = np.exp(np.clip(beta * s, -400, 400))
        val = (m + np.log(np.sum(e)) / beta) * dx
        if focus_lags is not None:
            for k in focus_lags:
                if 0 <= k < len(corr):
                    val += 0.05 * corr[k] * dx
        return float(val)
    
    def constraint_fun(x):
        return np.sum(x) - target
    
    cons = [{'type': 'eq', 'fun': constraint_fun}]
    
    try:
        res = minimize(objective, h0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:

        s = corr - m
        e = np.exp(np.clip(beta * s, -400, 400))
        val = (m + np.log(np.sum(e)) / beta) * dx
        val += pen * (np.sum(x) - target) ** 2
        return float(val)
    
    try:
        res = minimize(objective, h0, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

    """Simulated annealing on binary sequences."""
    h = h_binary.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    T = 0.05
    T_min = 0.005
    cooling = 0.99995
    
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    
    for it in range(max_iter):
        if t_limit() < 5:
            break
        if it % 10000 == 0 and it > 0:
            # Check time more frequently
            pass
        
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        # Random move: flip one 0 to 1 and one 1 to 0
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        h[i] = 1.0
        h[j] = 0.0
        new_c5 = compute_c5(h)
        delta = new_c5 - c5
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            c5 = new_c5
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = 0.0
            h[j] = 1.0
        
        T = max(T_min, T * cooling)
    
    return best_h, best_c5
