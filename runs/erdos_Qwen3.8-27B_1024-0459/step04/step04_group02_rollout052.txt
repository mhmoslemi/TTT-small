def make_thue_morse(n):
    """Create a Thue-Morse-like binary pattern with n/2 ones"""
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    # Ensure exactly n/2 ones
    ones = np.sum(h == 1)
    target = n // 2
    if ones > target:
        # Flip some 1s to 0s
        excess = ones - target
        idx = np.where(h == 1)[0]
        to_flip = idx[:excess]
        h[to_flip] = 0
    elif ones < target:
        deficit = target - ones
        idx = np.where(h == 0)[0]
        to_flip = idx[:deficit]
        h[to_flip] = 1
    return h

def make_block_pattern(n, nblocks=4):
    """Create a block pattern: alternating blocks of 1s and 0s"""
    h = np.zeros(n)
    block_size = n // nblocks
    for b in range(nblocks):
        start = b * block_size
        end = start + block_size if b < nblocks - 1 else n
        if b % 2 == 0:
            h[start:end] = 1.0
    # Fix sum
    h = project_feasible(h, n)
    return h

- h[j] appears in (1-h[m+a]) when m+a=j, i.e., m=j-a: contribution -h[j-a] if 0 ≤ j-a ≤ n-a-1, i.e., a ≤ j ≤ n-1

So:
- j ∈ [0, n-a-1] = [0, npairs-1] where npairs = n-a: add (1-h[j+a])
- j ∈ [a, n-1]: subtract h[j-a]

In code:
- grad[0:npairs] += (1-x[a:a+npairs]) → j=0..npairs-1: (1-h[j+a])
- grad[a:a+npairs] -= x[0:npairs] → j=a..n-1: -h[j-a]

grad[aoff:aoff + npairs] -= w * x[:npairs]

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        cnt = int(np.sum(mask))
        if cnt > 0:
            adj = diff / cnt
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
        else:
            if diff > 0:
                idx = np.argmin(h)
            else:
                idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13 and n > 0:
        for j in range(n):
            if h[j] > 1e-14 and h[j] < 1.0 - 1e-14:
                h[j] = np.clip(h[j] + diff, 0.0, 1.0)
                break

        if abs(diff2) > 1e-13:

def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    if wsum < 1e-300:
        wsum = 1.0
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    ncorr = len(corr)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]

            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def subgrad_direction(h, n, frac_thresh=0.35, n_active=8):
    corr = compute_corr(h)
    max_corr = np.max(corr)
    if max_corr < 1e-15:
        return None
    thresh = max_corr - max_corr * frac_thresh
    active = np.where(corr >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        active = np.argsort(corr)[::-1][:n_active]
    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return None
    return grad / gnorm

def vectorized_coord_descent(h, n, max_sweeps=80, rng_seed=42):
    """Efficient coordinate descent with precomputed coefficient structure."""
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    rng = np.random.RandomState(rng_seed)

    for sweep in range(max_sweeps):
        if remaining_time() < 2:
            break
        best_corr = compute_corr(best)
        best_c5 = np.max(best_corr) * dx
        max_corr = np.max(best_corr)
        # Identify variables that are in a "good" range
        interior = np.where((best > 0.005) & (best < 0.995))[0]
        if len(interior) == 0:
            break
        order = rng.permutation(interior)
        improved = False
        ncorr_axis = ncorr

        for i in order:
            if remaining_time() < 0.5:
                break
            # Precompute linear coefficients for variable i
            a = np.zeros(ncorr_axis)
            b = np.zeros(ncorr_axis)
            for off in range(-(n-1), n):
                k = n - 1 + off
                if off == 0:
                    a[k] += 1.0
                    b[k] += -1.0
                else:
                    if 0 <= i + off < n:
                        a[k] += (1.0 - best[i + off])
                    if 0 <= i - off < n:
                        a[k] += (-best[i - off])

            t0 = best[i]
            base = best_corr - a * t0 - b * t0 * t0

            # Evaluate at several candidate points
            candidates = np.array([0.0, 1.0, 0.1, 0.5, 0.9, t0*0.25, t0*0.75, t0*1.25, t0*1.75, 1.0-t0])
            candidates = np.clip(candidates, 0.0, 1.0)
            candidates = np.unique(candidates)

            vals = np.max(base + a * candidates + b * candidates * candidates) * dx
            idx_best = np.argmin(vals)
            c5_cand = vals[idx_best]
            t_cand = candidates[idx_best]

            if c5_cand < best_c5 - 1e-14 and abs(t_cand - t0) > 1e-9:
                # Golden section refinement
                lo = max(0.0, min(t0, t_cand) - 0.15)
                hi = min(1.0, max(t0, t_cand) + 0.15)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = np.max(base + a*c1 + b*c1*c1) * dx
                f2 = np.max(base + a*c2 + b*c2*c2) * dx
                for _ in range(35):
                    if f1 < f2:
                        hi = c2; c2 = c1; f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = np.max(base + a*c1 + b*c1*c1) * dx
                    else:
                        lo = c1; c1 = c2; f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = np.max(base + a*c2 + b*c2*c2) * dx
                t_gs = (lo + hi) / 2.0
                c5_gs = np.max(base + a*t_gs + b*t_gs*t_gs) * dx
                if c5_gs < c5_cand:
                    t_cand = t_gs
                    c5_cand = c5_gs
                if c5_cand < best_c5 - 1e-14:
                    best[i] = t_cand
                    best_c5 = c5_cand
                    improved = True
        if not improved:
            break
    return best, best_c5

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    ones = int(np.sum(h))
    target = n // 2
    if ones > target:
        idx = np.where(h == 1)[0]
        h[idx[:ones - target]] = 0
    elif ones < target:
        idx = np.where(h == 0)[0]
        h[idx[:target - ones]] = 1
    return h

        c5_r = compute_c5(h_r)
        
        h_r, c5_r = multi_start_slsqp(h_r, rn, [300, 1000, 4000, 12000, 30000], 400)
        
        if remaining_time() > 30:
            h_r, c5_r2 = vectorized_coord_descent(h_r, rn, 40, seed+3)
            if c5_r2 < c5_r:
                c5_r = c5_r2
        
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn
            n = rn

    # Phase 7: Try lower resolution (sometimes coarser grid gives better C5)
    if remaining_time() > 60:
        for rn in sorted(set([n//2, n//3, 320, 480])):
            if remaining_time() < 50:
                break
            if rn < 64 or rn % 2 != 0:
                continue
            h_low = interp_to(best_h, best_n, rn)
            h_low = project_feasible(h_low, rn)
            h_low, c5_low = multi_start_slsqp(h_low, rn, [200, 800, 3000, 10000], 400)
            if c5_low < best_c5:
                best_c5 = c5_low
                best_h = h_low.copy()
                best_n = rn
                n = rn

    # Phase 8: Different initializations at best resolution
    if remaining_time() > 80:
        inits = []
        # Thue-Morse
        h_tm = make_thue_morse(n)
        h_tm = project_feasible(h_tm, n)
        if compute_c5(h_tm) < 0.45:
            inits.append(h_tm)
        # Periodic
        h_per = make_periodic(n, 6)
        if compute_c5(h_per) < 0.45:
            inits.append(h_per)
        # Random binary
        h_rb = np.zeros(n)
        h_rb[rng.choice(n, n//2, replace=False)] = 1.0
        h_rb = project_feasible(h_rb, n)
        if compute_c5(h_rb) < 0.5:
            inits.append(h_rb)
        # Shifted parent
        h_sh = np.roll(best_h, n//7)
        h_sh = project_feasible(h_sh, n)
        inits.append(h_sh)
        
        for h_init in inits:
            if remaining_time() < 40:
                break
            h_t, c5_t = multi_start_slsqp(h_init, n, [200, 800, 3000, 10000, 30000], 350)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()

    # Phase 9: Final coordinate descent + SLSQP
    if remaining_time() > 20:
        h_cd, c5_cd = vectorized_coord_descent(best_h.copy(), best_n, 60, seed+10)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if remaining_time() > 15:
        h, c5_h = multi_start_slsqp(best_h.copy(), best_n, [20000, 50000, 100000], 400)
        if c5_h < best_c5:
            best_c5 = c5_h
            best_h = h.copy()

    # Phase 10: Micro subgradient refinement
    if remaining_time() > 10:
        h = best_h.copy()
        nn = best_n
        for it in range(500):
            if remaining_time() < 3:
                break
            grad = subgrad_direction(h, nn, 0.4, 6)
            if grad is None:
                break
            improved = False
            for step in [0.001, 0.003, 0.0005, 0.002, 0.005, 0.0015, 0.0008, 0.01]:
                h_trial = h - step * grad
                h_trial = project_feasible(h_trial, nn)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-16:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h = h_trial.copy()
                    improved = True
                    break
            if not improved:
                if it > 100:
                    break

    # Phase 11: Final coordinate descent
    if remaining_time() > 10:
        h_cd, c5_cd = vectorized_coord_descent(best_h.copy(), best_n, 40, seed+20)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 12: One more SLSQP with very high beta
    if remaining_time() > 8:
        h_new = slsqp_refine(best_h.copy(), best_n, 200000, 300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
