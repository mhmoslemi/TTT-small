def project_feasible(h, n):

def softmax_subgrad(h, n, top_k=6, temp=0.02):
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-top_k:][::-1]
    vals = corr[top_idx]
    max_val = vals[0]
    gaps = max_val - vals
    weights = np.exp(-gaps / temp)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for w, k_idx in zip(weights, top_idx):
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            nv = n - offset
            if nv <= 0:

            grad[offset:offset + nv] += w * (1.0 - h[:nv])
            grad[:nv] -= w * h[offset:offset + nv]

            if nv <= 0:

            grad[:nv] += w * (1.0 - h[m:m + nv])

        h_bal, c5_bal = balance_top_corr(h_try, n_try, num_rounds=20)
        if c5_bal < c5_try:
            c5_try = c5_bal
            h_try = h_bal
        if c5_try < best_c5 - 1e-12:
            best_h, best_c5, best_n = h_try, c5_try, n_try
            print("  n=320 improved: C5={:.10f}".format(best_c5))
        else:
            print("  n=320: C5={:.10f} (no improvement)".format(c5_try))

    # === Phase 4: Try n=512 ===
    if remaining_time() > 120.0:
        print("\n=== Phase 4: n=512 ===")
        n_try = 512
        h_try = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        _G_N = n_try
        _G_TGT = n_try / 2.0
        _G_DX = 2.0 / n_try
        bounds_h = [(0.0, 1.0)] * n_try
        cons_h = [{'type': 'eq', 'fun': con_sum}]
        h_try, c5_try = subgrad_momentum(h_try, n_try, max_iter=200, step0=0.08, momentum=0.9)
        for beta in [100, 500, 2000, 8000]:
            if remaining_time() < 12.0:
                break
            _G_BETA = float(beta)
            try:
                res = minimize(obj_logsumexp, h_try, method='SLSQP',
                              bounds=bounds_h, constraints=cons_h,
                              options={'maxiter': 250, 'ftol': 1e-15, 'iprint': 0})
                h_new = project_feasible(res.x, n_try)
                c5_new = compute_c5(h_new)
                if c5_new < c5_try:
                    c5_try = c5_new
                    h_try = h_new
            except Exception:
                pass
        if c5_try < best_c5 - 1e-12:
            best_h, best_c5, best_n = h_try, c5_try, n_try
            print("  n=512 improved: C5={:.10f}".format(best_c5))
        else:
            print("  n=512: C5={:.10f} (no improvement)".format(c5_try))

    # === Phase 5: Binary search at best resolution ===
    if remaining_time() > 60.0:
        print("\n=== Phase 5: Binary pair search at n={} ===".format(best_n))
        h_bin, c5_bin = binary_pair_search(best_h, best_n, max_iter=25000, rng=rng)
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin, c5_bin
            print("  Binary improved: C5={:.10f}".format(best_c5))
        else:
            print("  Binary: C5={:.10f}".format(c5_bin))
        # Refine the binary solution with SLSQP
        if c5_bin < best_c5 + 0.001:
            _G_N = best_n
            _G_TGT = best_n / 2.0
            _G_DX = 2.0 / best_n
            bounds_h = [(0.0, 1.0)] * best_n
            cons_h = [{'type': 'eq', 'fun': con_sum}]
            _G_BETA = 5000.0
            try:
                res = minimize(obj_logsumexp, h_bin, method='SLSQP',
                              bounds=bounds_h, constraints=cons_h,
                              options={'maxiter': 300, 'ftol': 1e-15, 'iprint': 0})
                h_new = project_feasible(res.x, best_n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new, c5_new
                    print("  Refined: C5={:.10f}".format(best_c5))
            except Exception:
                pass

    # === Phase 6: Block flip + finalize ===
    if remaining_time() > 30.0:
        print("\n=== Phase 6: Block flips ===")
        h_blk, c5_blk = block_flip_search(best_h, best_n, rng=rng, max_moves=8000)
        if c5_blk < best_c5:
            best_h, best_c5 = h_blk, c5_blk
        print("  Block: C5={:.10f}".format(best_c5))

    # === Phase 7: Final aggressive SLSQP ===
    if remaining_time() > 15.0:
        print("\n=== Phase 7: Final SLSQP ===")
        _G_N = best_n
        _G_TGT = best_n / 2.0
        _G_DX = 2.0 / best_n
        bounds_h = [(0.0, 1.0)] * best_n
        cons_h = [{'type': 'eq', 'fun': con_sum}]
        for beta in [5000, 15000, 50000]:
            if remaining_time() < 5.0:
                break
            _G_BETA = float(beta)
            try:
                res = minimize(obj_logsumexp, best_h, method='SLSQP',
                              bounds=bounds_h, constraints=cons_h,
                              options={'maxiter': 400, 'ftol': 1e-16, 'iprint': 0})
                h_new = project_feasible(res.x, best_n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:

                    best_h = h_new
                    print("  Final beta={}: C5={:.10f}".format(beta, best_c5))
            except Exception:
                pass
        # Balance one more time
        h_bal, c5_bal = balance_top_corr(best_h, best_n, num_rounds=15)
        if c5_bal < best_c5:
            best_h, best_c5 = h_bal, c5_bal

    # === Final validation ===
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
