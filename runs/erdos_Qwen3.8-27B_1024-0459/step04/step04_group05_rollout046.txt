import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - target) < 1e-14:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(80):
        tau = (lo + hi) / 2.0
        s = np.sum(np.clip(h - tau, 0.0, 1.0))
        if s > target:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2.0
    h = np.clip(h - tau, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_and_grad(h, n, lam):
    dx = 2.0 / n
    corr = compute_corr(h)
    M = float(np.max(corr))
    shifted = (corr - M) / lam
    shifted = np.clip(shifted, -200, 200)
    exp_s = np.exp(shifted)
    S = float(np.sum(exp_s))
    obj = (M + lam * np.log(S)) * dx
    w = exp_s / S
    grad = np.zeros(n)
    thresh = 1e-10
    for k_idx in range(len(corr)):
        wk = w[k_idx]
        if wk < thresh:
            continue
        wk *= dx
        offset = k_idx - (n - 1)
        if offset >= 0:
            np_ = n - offset
            if np_ <= 0:
                continue
            grad[:np_] += wk * (1.0 - h[offset:offset + np_])
            grad[offset:] -= wk * h[:np_]
        else:
            ao = -offset
            np_ = n - ao
            if np_ <= 0:
                continue
            grad[:np_] -= wk * h[ao:ao + np_]
            grad[ao:] += wk * (1.0 - h[:np_])
    return obj, grad


def project_grad_opt(h0, n, budget_fn, max_iters=20000, lam0=0.02, lam_min=0.0005):
    dx = 2.0 / n
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur
    lam = lam0
    obj_cur, grad = smooth_obj_and_grad(h_cur, n, lam)
    alpha = 1.0 / (2.0 * n)
    no_improve = 0
    stalled = 0

    for it in range(max_iters):
        if budget_fn() < 1.5:
            break

        a = alpha
        improved = False
        for _ in range(50):
            h_trial = project_feasible(h_cur - a * grad, n)
            obj_trial, _ = smooth_obj_and_grad(h_trial, n, lam)
            if obj_trial < obj_cur - 1e-18:
                h_cur = h_trial
                obj_cur = obj_trial
                c5_trial = compute_c5(h_cur)
                if c5_trial < best_c5 - 1e-15:
                    best_c5 = c5_trial
                    best_h = h_cur.copy()
                    no_improve = 0
                    stalled = 0
                else:
                    no_improve += 1
                alpha = min(a * 1.8, 5.0 / n)
                improved = True
                break
            a *= 0.4
        if not improved:
            alpha *= 0.3
            stalled += 1
            no_improve += 1
            if stalled > 500 or no_improve > 5000:
                break

        obj_cur, grad = smooth_obj_and_grad(h_cur, n, lam)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        if it % 1000 == 999 and lam > lam_min:
            lam = max(lam_min, lam * 0.85)
            obj_cur, grad = smooth_obj_and_grad(h_cur, n, lam)

    return best_h, best_c5


def binary_local_search(h0, n, budget_fn, max_sweeps=30):
    h = h0.copy()
    c5_best = compute_c5(h)
    best_h = h.copy()
    for sweep in range(max_sweeps):
        if budget_fn() < 2.0:
            break
        any_imp = False
        order = np.random.permutation(n)
        for i in order:
            if budget_fn() < 1.0:
                break
            old_val = h[i]
            for new_val in [0.0, 1.0]:
                if abs(new_val - old_val) < 1e-6:
                    continue
                h[i] = new_val
                h = project_feasible(h, n)
                c5_test = compute_c5(h)
                if c5_test < c5_best - 1e-14:
                    c5_best = c5_test
                    best_h = h.copy()
                    any_imp = True
                else:
                    h = best_h.copy()
        if not any_imp and sweep > 5:
            break
    return best_h, c5_best


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h


def make_random_binary(rng, n):

    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h


def do_slsqp_round(h0, n, budget_fn, multipliers):
    dx = 2.0 / n
    target = n / 2.0
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    def obj_fn(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        for mu in multipliers:
            shifted = corr - m
            exps = np.exp(np.clip(mu * shifted, -300, 300))
            m = m + np.log(np.sum(exps)) / mu
        return m * dx

    def con_fun(x):
        return np.sum(x) - target

    def con_jac(x):
        return np.ones(n)

    for mu in multipliers:
        if budget_fn() < 8.0:
            break
        multipliers_single = [mu]

        def obj_single(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(corr)
            shifted = corr - m
            exps = np.exp(np.clip(mu * shifted, -300, 300))
            return (m + np.log(np.sum(exps)) / mu) * dx

        try:
            cons = [({'type': 'eq', 'fun': con_fun, 'jac': con_jac})]
            res = minimize(obj_single, h_cur, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-15, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, c5_cur
