from scipy.optimize import minimize
import time

_G_START = 0.0
_G_BUDGET = 1300.0
_G_DX = 1.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            if diff > 0:
                idx = np.where(h < 1.0)[0]
            else:
                idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] += diff
                h = np.clip(h, 0.0, 1.0)
            break
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-15)[0]
        else:
            idx = np.where(h > 1e-15)[0]
        if len(idx) > 0:
            h[idx[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h


def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 3.0)


def grad_softmax(h, n, beta, top_k=None):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if top_k is not None and top_k < n_corr:
        top_idx = np.argsort(corr)[-top_k:]
        weights = np.zeros(n_corr)
        corr_top = corr[top_idx]
        cm = np.max(corr_top)
        w_top = np.exp(beta * (corr_top - cm))
        w_top /= np.sum(w_top)
        weights[top_idx] = w_top
    else:
        cm = np.max(corr)
        weights = np.exp(beta * (corr - cm))
        weights /= np.sum(weights)
    grad = np.zeros(n)
    for j in range(n_corr):
        wj = weights[j]
        if wj < 1e-16:
            continue
        offset = j - (n - 1)
        if offset >= 0:
            nv = n - offset
            if nv <= 0:
                continue
            grad[offset:offset + nv] += wj * (1.0 - h[:nv])
            grad[:nv] -= wj * h[offset:offset + nv]
        else:
            m = -offset
            nv = n - m
            if nv <= 0:
                continue
            grad[:nv] += wj * (1.0 - h[m:m + nv])
            grad[m:m + nv] -= wj * h[:nv]
    return grad, corr, weights


def nesterov_opt(h, n, max_iter=400, beta=50.0, step0=0.04, momentum=0.9, top_k=None):
    c5_best = compute_c5(h)
    h_best = h.copy()
    v = np.zeros(n)
    step = step0
    patience = 0
    for it in range(max_iter):
        if remaining_time() < 5.0:
            break
        h_look = project_feasible(h + v, n)
        grad, corr, weights = grad_softmax(h_look, n, beta, top_k=top_k)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn
        alpha = step
        improved = False
        for _ in range(18):
            v = momentum * v - alpha * grad
            h_trial = project_feasible(h + v, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-14:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
            alpha *= 0.5
        if improved:
            patience = 0
            step = min(alpha * 1.3, 0.25)
        else:
            patience += 1
            step *= 0.55
            v *= 0.4
            if patience > 25:
                break
    return h_best, c5_best

def slsqp_solve(h, n, beta, maxiter=300):
    global _G_N, _G_BETA, _G_TGT, _G_DX
    _G_N = n
    _G_BETA = float(beta)
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        return h_new, c5_new

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

        print("Thue-Morse: n={}, C5={:.10f}".format(best_n, best_c5))

    n = best_n
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n

    # Phase 1: Nesterov with increasing beta (full softmax)
    print("\n--- Phase 1: Nesterov full softmax ---")
    for beta_val in [15, 40, 100, 300]:
        if remaining_time() < 25.0:
            break
        best_h, best_c5 = nesterov_opt(best_h, n, max_iter=300, beta=beta_val,
                                       step0=0.035, momentum=0.9)
        print("  beta={}: C5={:.10f}".format(beta_val, best_c5))

    # Phase 2: Nesterov with top-k focus
    print("\n--- Phase 2: Nesterov top-k focus ---")
    for top_k in [5, 10, 20, n, None]:
        if remaining_time() < 15.0:
            break
        best_h, best_c5 = nesterov_opt(best_h, n, max_iter=250, beta=200.0,
                                       step0=0.025, momentum=0.88, top_k=top_k)
        label = str(top_k) if top_k is not None else "all"
        print("  top_k={}: C5={:.10f}".format(label, best_c5))

    # Phase 3: SLSQP multi-beta
    print("\n--- Phase 3: SLSQP multi-beta ---")
    for beta_val in [100, 300, 1000, 3000, 10000]:
        if remaining_time() < 12.0:
            break
        h_new, c5_new = slsqp_solve(best_h, n, beta_val, maxiter=350)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new
            print("  beta={}: C5={:.10f}".format(beta_val, best_c5))

    # Phase 4: Binary local search
    print("\n--- Phase 4: Binary local search ---")
    h_bin = (best_h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, n)
    c5_bin = compute_c5(h_bin)
    print("  Initial binary: C5={:.10f}".format(c5_bin))
    h_bin, c5_bin = binary_local_search(h_bin, n, max_iter=8000, rng=rng)
    print("  After binary search: C5={:.10f}".format(c5_bin))
    if c5_bin < best_c5 - 1e-14:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    # Continuous refinement of binary
    if remaining_time() > 40.0:
        h_ref = h_bin.copy()
        for beta_val in [50, 200, 800]:
            if remaining_time() < 10.0:
                break
            h_ref, c5_ref = nesterov_opt(h_ref, n, max_iter=200, beta=beta_val, step0=0.03)
            if c5_ref < best_c5 - 1e-14:
                best_c5 = c5_ref
                best_h = h_ref.copy()

    # Phase 5: Multi-start exploration
    print("\n--- Phase 5: Multi-start ---")
    best_h, best_c5 = multi_start_explore(best_h, n, best_c5, rng, num_starts=5)
    print("  C5={:.10f}".format(best_c5))

    # Phase 6: SLSQP round 2
    print("\n--- Phase 6: SLSQP round 2 ---")
    for beta_val in [500, 2000, 8000]:
        if remaining_time() < 10.0:
            break
        h_new, c5_new = slsqp_solve(best_h, n, beta_val, maxiter=400)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new
            print("  beta={}: C5={:.10f}".format(beta_val, best_c5))

    # Phase 7: Multi-resolution
    print("\n--- Phase 7: Multi-resolution ---")
    for n_try in [192, 320, 384, 512]:
        if remaining_time() < 35.0:
            break
        h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        c5_interp = compute_c5(h_interp)
        g_n = n_try
        _G_N = g_n
        _G_TGT = g_n / 2.0
        _G_DX = 2.0 / g_n
        for beta_val in [30, 100, 400]:
            if remaining_time() < 12.0:
                break
            h_interp, c5_interp = nesterov_opt(h_interp, n_try, max_iter=150,
                                               beta=beta_val, step0=0.03)
        if remaining_time() > 15.0:
            h_interp, c5_interp = slsqp_solve(h_interp, n_try, 500, maxiter=200)
        print("  n={}: C5={:.10f}".format(n_try, c5_interp))
        if c5_interp < best_c5 - 1e-12:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n_try
        _G_N = n
        _G_TGT = n / 2.0
        _G_DX = 2.0 / n

    # Phase 8: Coordinate descent
    print("\n--- Phase 8: Coordinate descent ---")
    n = best_n
    best_h, best_c5 = coordinate_refine(best_h, n, max_sweeps=15, rng=rng)
    print("  C5={:.10f}".format(best_c5))

    # Phase 9: Final high-beta SLSQP
    print("\n--- Phase 9: Final polish ---")
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    for beta_val in [5000, 15000, 30000]:
        if remaining_time() < 5.0:
            break
        h_new, c5_new = slsqp_solve(best_h, n, beta_val, maxiter=500)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new
            print("  beta={}: C5={:.10f}".format(beta_val, best_c5))

    # Phase 10: Tiny coordinate refinement
    if remaining_time() > 6.0:
        best_h, best_c5 = coordinate_refine(best_h, n, max_sweeps=5, rng=rng)

    # Final project and validate
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(n, final_c5))
    return (best_h, final_c5, n)
