def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def binarize(h, n):
    """Convert h to binary with exactly n/2 ones."""
    h = np.array(h, dtype=float).copy()
    target = n // 2
    # Threshold
    h_bin = (h >= 0.5).astype(float)
    count = int(np.sum(h_bin))
    if count > target:
        # Need to flip (count - target) ones to zeros
        # Flip the ones with smallest h values
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones_idx])
        to_flip = ones_idx[order[:count - target]]
        h_bin[to_flip] = 0
    elif count < target:
        # Need to flip (target - count) zeros to ones
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])
        to_flip = zeros_idx[order[:target - count]]
        h_bin[to_flip] = 1
    return h_bin

**New approach: Use the fact that for n=256, we can afford to recompute the correlation many times.**

Let me time it more carefully. For two arrays of length 256:
- np.correlate: This does a sum of products. The output has 511 elements. The total number of multiply-adds is sum_{k=1}^{256} k + sum_{k=1}^{255} k = 256*257/2 + 255*256/2 = 256*256 = 65536.
- In numpy, this is implemented in C and should take about 10-30μs.

Over 1000 seconds: 5M evaluations.

But each "iteration" only makes one improvement (or none). So we can make at most 5M improvements in total, which is way more than enough.

The real bottleneck is the Python loop overhead. Let me minimize it.

Actually, let me try a completely different approach. Instead of evaluating each swap separately, let me compute the effect of all possible swaps on the active shifts simultaneously.

If I know which shifts are "active" (achieving or near the maximum), I only need to check those shifts. For the max to decrease, ALL active shifts must decrease (or at least the maximum must decrease).

Here's a refined algorithm:
1. Compute corr, find max value M and the set S of shifts achieving M (or within ε of M)
2. For a swap to improve the solution, it must reduce ALL shifts in S (or at least reduce the max)
3. Compute Δcorr for each shift in S for all candidate swaps
4. Find swaps that reduce all shifts in S

This is much more targeted. Let me implement it.

Actually, the simplest effective approach might be:

**Phase 1**: Use the parent, binarize it, then do a quick greedy search to reach a local optimum.
**Phase 2**: Try many different resolutions (n=192 to n=512, step 16 or 32), for each:
  - Interpolate the best solution to that resolution
  - Binarize
  - Do a few hundred greedy iterations
  - Keep the best
**Phase 3**: For the best resolution found, do more intensive optimization (SA, more greedy).

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def binarize(h, n):
    h = np.array(h, dtype=float).copy()
    target = n // 2
    h_bin = (h >= 0.5).astype(float)
    count = int(np.sum(h_bin))
    if count > target:
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones_idx])
        h_bin[ones_idx[order[:count - target]]] = 0
    elif count < target:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])
        h_bin[zeros_idx[order[:target - count]]] = 1
    return h_bin

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:

def interp_binarize(src_h, src_n, dst_n):
    """Interpolate src_h to dst_n resolution and binarize."""
    if src_n == dst_n:
        return binarize(src_h, dst_n)
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    h_interp = np.interp(x_new, xp, src_h)
    return binarize(h_interp, dst_n)

def greedy_local_search(h, n, budget_fn, max_iters=10000, n_samples=5000):
    """Local search with random swap moves, best-improvement."""
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    rng = np.random.RandomState(999)
    
    ones_list = np.where(h_best == 1)[0].tolist()
    zeros_list = np.where(h_best == 0)[0].tolist()

        if budget_fn() < 5:
            break

        n_zeros = len(zeros_list)
        total = n_ones * n_zeros
        
        if total <= n_samples:
            candidates = [(i, j) for i in ones_list for j in zeros_list]
        else:
            idx_i = rng.choice(n_ones, size=n_samples, replace=False)
            idx_j = rng.choice(n_zeros, size=n_samples, replace=False)
            candidates = [(ones_list[idx_i[k]], zeros_list[idx_j[k]]) for k in range(n_samples)]
