        print("  After HC: C5={:.10f}".format(best_c5))

    # Phase 4: Interpolate to n=512 and optimize
    if _budget_remaining() > 120:
        print("\n--- Phase 4: n=512 ---")
        n512 = 512
        h512 = project_feasible(interp_to(best_h, best_n, n512), n512)
        c5_512 = compute_c5(h512)
        print("  Interpolated: C5={:.10f}".format(c5_512))
        h512, c5_512 = do_slsqp_lse(h512, n512, [30, 100, 300, 1000, 3000], maxiter_per_beta=250)
        print("  After SLSQP: C5={:.10f}".format(c5_512))
        if _budget_remaining() > 60:
            h512, c5_512 = do_slsqp_topk(h512, n512, top_k=7, n_rounds=2)
            print("  After topk: C5={:.10f}".format(c5_512))
        if _budget_remaining() > 40:
            h512, c5_512 = do_continuous_hc(h512, n512, rng, max_iters=4000)
            print("  After HC: C5={:.10f}".format(c5_512))
        if c5_512 < best_c5:
            best_c5 = c5_512
            best_h = h512.copy()
            best_n = n512
            print("  IMPROVED to n={}".format(n512))
        else:
            print("  No improvement at n=512 (best remains {:.10f})".format(best_c5))

    # Phase 5: Try n=1024 if time permits
    if _budget_remaining() > 150:
        print("\n--- Phase 5: n=1024 ---")
        n1024 = 1024
        h1024 = project_feasible(interp_to(best_h, best_n, n1024), n1024)
        c5_1024 = compute_c5(h1024)
        print("  Interpolated: C5={:.10f}".format(c5_1024))
        h1024, c5_1024 = do_slsqp_lse(h1024, n1024, [50, 200, 800], maxiter_per_beta=200)
        print("  After SLSQP: C5={:.10f}".format(c5_1024))
        if c5_1024 < best_c5:
            best_c5 = c5_1024
            best_h = h1024.copy()
            best_n = n1024
            print("  IMPROVED to n={}".format(n1024))

    # Phase 6: Binary local search at best resolution
    if _budget_remaining() > 40:
        print("\n--- Phase 6: Binary HC n={} ---".format(best_n))
        h_bin, c5_bin = do_binary_hc(best_h, best_n, rng, max_iters=25000)
        print("  Binary: C5={:.10f}".format(c5_bin))
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            print("  IMPROVED!")
        # SLSQP from binary solution
        if _budget_remaining() > 30:
            h_from_bin, c5_fb = do_slsqp_lse(h_bin, best_n, [200, 800, 2000], maxiter_per_beta=150)
            if c5_fb < best_c5:
                best_c5 = c5_fb
                best_h = h_from_bin.copy()
                print("  SLSQP from binary: C5={:.10f}".format(best_c5))

    # Phase 7: Perturbed restarts
    if _budget_remaining() > 40:
        print("\n--- Phase 7: Perturbed restarts ---")
        best_h, best_c5 = do_perturb_slsqp(best_h, best_n, rng, n_restarts=3,
                                           beta_list=[300, 1000, 3000])
        print("  After restarts: C5={:.10f}".format(best_c5))

    # Phase 8: Final high-beta SLSQP
    if _budget_remaining() > 20:
        print("\n--- Phase 8: High-beta polish ---")
        best_h, best_c5 = do_slsqp_lse(best_h, best_n, [5000, 15000, 40000], maxiter_per_beta=150)
        print("  After polish: C5={:.10f}".format(best_c5))

    # Phase 9: Final top-k + HC
    if _budget_remaining() > 20:
        best_h, best_c5 = do_slsqp_topk(best_h, best_n, top_k=5, n_rounds=2)
        if _budget_remaining() > 15:
            best_h, best_c5 = do_continuous_hc(best_h, best_n, rng, max_iters=2000)

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
