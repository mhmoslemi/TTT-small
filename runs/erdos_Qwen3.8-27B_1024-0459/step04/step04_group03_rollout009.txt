def corr_change_from_swap(h, p, q, n):
    """Compute change in full correlation when h[p]:1→0, h[q]:0→1."""
    delta = np.zeros(2*n-1)
    # p going 1→0
    # Effect on corr at lag l (full index = l + n - 1):
    # Term i+l=p: contributes -(1-h[p-l]) if valid i
    # Term i=p: contributes +h[p+l] if valid p+l
    for l in range(-(n-1), n):
        # i = p - l must be in [0, n-1]
        i1 = p - l
        if 0 <= i1 < n:
            delta[l + n - 1] += h[i1] - 1  # = -(1-h[i1])
        # i = p, need p+l in [0, n-1]
        j1 = p + l
        if 0 <= j1 < n:
            delta[l + n - 1] += h[j1]
    # q going 0→1
    for l in range(-(n-1), n):

            h[p], h[q] = h[q], h[p]
            corr = compute_corr(h)
            if float(np.max(corr)) < best_max - 1e-10:
                best_max = float(np.max(corr))
                best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count >= max_stall:
                # Kick: random multi-swap
                for _ in range(5):
                    p = int(rng.choice(ones))
                    q = int(rng.choice(zeros))
                    h[p], h[q] = h[q], h[p]
                corr = compute_corr(h)
                stall_count = 0
                max_stall = int(max_stall * 1.3)
    
    return best_h, best_max

def smooth_obj(x):
    n = _N_G
    beta = _BETA_G
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def smooth_eq(x):
    return np.sum(x) - _N_G / 2.0

def mm_obj(x):
    return x[_N_G]

def mm_ineq(x):
    n = _N_G
    dx = _DX_G
    h = x[:n]
    t = x[n]
    corr = compute_corr(h)
    return t - dx * corr[_LAGS_G]

def mm_eq(x):
    return np.sum(x[:_N_G]) - _N_G / 2.0

def slsqp_smooth(h_init, n, beta, maxiter=300):
    global _N_G, _BETA_G
    _N_G = n
    _BETA_G = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smooth_eq}]
    h = h_init.copy()
    try:
        res = minimize(smooth_obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def slsqp_minimax(h_init, n, max_rounds=10, max_slsqp_iter=200):
    global _N_G, _DX_G, _LAGS_G
    _N_G = n
    _DX_G = 2.0 / n
    h = h_init.copy()
    dx = 2.0 / n

    for rnd in range(max_rounds):
        if remaining_time() < 8:
            break
        corr = compute_corr(h)
        max_corr = float(np.max(corr))
        t_init = max_corr * dx + 1e-8

        # Active set: lags close to the maximum
        threshold = max_corr - 0.3
        active = np.where(corr >= threshold)[0]
        top5 = np.argsort(corr)[::-1][:5]
        active = np.unique(np.concatenate([active, top5]))
        active = active[np.argsort(active)]
        
        # Limit active set size
        if len(active) > 30:
            top30 = np.argsort(corr)[::-1][:30]
            active = top30[np.argsort(top30)]

        _LAGS_G = active
        x0 = np.concatenate([h, [t_init]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [
            {'type': 'ineq', 'fun': mm_ineq},
            {'type': 'eq', 'fun': mm_eq}
        ]
        try:
            res = minimize(mm_obj, x0, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_slsqp_iter, 'ftol': 1e-14, 'iprint': 0})
            if res.x is not None:
                h_new = project_feasible(res.x[:n], n)
                c5_new = compute_c5(h_new)
                c5_old = compute_c5(h)
                if c5_new < c5_old - 1e-16:
                    h = h_new
                else:
                    if rnd > 3:
                        break
        except Exception:
            break
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S, _N_G, _BETA_G, _LAGS_G, _DX_G
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent solution
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 256
        best_h = np.full(best_n, 0.5)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)
        parent_n = best_n
        parent_h = best_h.copy()

    # === Phase 1: Binary local search at parent resolution ===
    if remaining_time() > 200:
        h_bin = threshold_to_binary(parent_h, parent_n)
        time_alloc = min(200, remaining_time() * 0.35)
        h_bin_best, c5_bin = binary_local_search(h_bin, parent_n, time_alloc, rng)
        c5_bin_real = compute_c5(h_bin_best)
        if c5_bin_real < best_c5:
            best_c5 = c5_bin_real
            best_h = h_bin_best.copy()
            best_n = parent_n

        # Also try a few random binary starts
        for rs in range(3):
            if remaining_time() < 50:
                break
            h_rand = (rng.rand(parent_n) > 0.5).astype(float)
            ones = np.sum(h_rand == 1)
            target = parent_n // 2
            if ones > target:
                excess = int(ones - target)
                one_pos = np.where(h_rand == 1)[0]
                h_rand[int(one_pos[rng.randint(0, len(one_pos), excess)])] = 0
            elif ones < target:
                deficit = int(target - ones)
                zero_pos = np.where(h_rand == 0)[0]
                h_rand[int(zero_pos[rng.randint(0, len(zero_pos), deficit)])] = 1
            h_r, c5_r = binary_local_search(h_rand, parent_n, min(40, remaining_time()*0.5), rng)
            c5_r_real = compute_c5(h_r)
            if c5_r_real < best_c5:
                best_c5 = c5_r_real
                best_h = h_r.copy()
                best_n = parent_n

    # === Phase 2: Binary local search at higher resolution ===
    if remaining_time() > 150:
        for trial_n in [384, 512]:
            if remaining_time() < 60:
                break
            h_interp = interp_to(best_h, best_n, trial_n)
            h_bin_h = threshold_to_binary(h_interp, trial_n)
            time_alloc = min(80, remaining_time() * 0.3)
            h_b, c5_b = binary_local_search(h_bin_h, trial_n, time_alloc, rng)
            c5_b_real = compute_c5(h_b)
            if c5_b_real < best_c5:
                best_c5 = c5_b_real
                best_h = h_b.copy()
                best_n = trial_n

    # === Phase 3: Non-binary smooth optimization ===
    if remaining_time() > 100:
        h_opt = best_h.copy()
        n_opt = best_n
        for beta in [50, 200, 500, 2000, 8000]:
            if remaining_time() < 15:
                break
            h_new = slsqp_smooth(h_opt, n_opt, beta, maxiter=250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-16:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_opt = h_new.copy()

    # === Phase 4: Minimax SLSQP ===
    if remaining_time() > 60:
        h_mm = slsqp_minimax(best_h, best_n, max_rounds=12, max_slsqp_iter=200)
        c5_mm = compute_c5(h_mm)
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm.copy()

    # === Phase 5: Binarize + re-optimize (exploit structure) ===
    if remaining_time() > 80:
        h_bin2 = threshold_to_binary(best_h, best_n)
        c5_bin2 = compute_c5(h_bin2)
        # Short binary polish
        if c5_bin2 < best_c5 + 0.005:
            time_alloc = min(40, remaining_time() * 0.4)
            h_bp, c5_bp = binary_local_search(h_bin2, best_n, time_alloc, rng)
            c5_bp_real = compute_c5(h_bp)
            if c5_bp_real < best_c5:
                best_c5 = c5_bp_real
                best_h = h_bp.copy()
            # Non-binary polish from binary
            if remaining_time() > 20:
                h_np = slsqp_smooth(h_bp, best_n, 1000, maxiter=200)
                c5_np = compute_c5(h_np)
                if c5_np < best_c5:
                    best_c5 = c5_np
                    best_h = h_np.copy()

    # === Phase 6: Multi-resolution exploration ===
    if remaining_time() > 60:
        for trial_n in [192, 256, 320, 384, 448]:
            if remaining_time() < 20:
                break
            if trial_n == best_n:
                continue
            h_interp = interp_to(best_h, best_n, trial_n)
            h_interp = project_feasible(h_interp, trial_n)
            c5_i = compute_c5(h_interp)
            if c5_i < best_c5 + 0.01:
                maxit = 200 if trial_n <= 320 else 150
                for beta in [200, 1000]:
                    if remaining_time() < 8:
                        break
                    h_new = slsqp_smooth(h_interp, trial_n, beta, maxiter=maxit)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        best_n = trial_n
                        h_interp = h_new.copy()

    # === Phase 7: Targeted lag reduction ===
    if remaining_time() > 40:
        # Find worst lags and do focused optimization
        corr = compute_corr(best_h)
        worst_lags = np.argsort(corr)[::-1][:8]
        n_f = best_n
        dx_f = 2.0 / n_f
        
        # Custom objective: weighted max of worst lags
        weights = np.zeros(2*n_f - 1)
        for i, lag in enumerate(worst_lags):
            weights[lag] = 1.0 / (i + 1)
        
        # Use SLSQP to minimize weighted sum
        def weighted_obj(x):
            h = np.clip(x, 0, 1)
            c = compute_corr(h)
            return np.sum(weights * c) * dx_f
        
        def weighted_eq(x):
            return np.sum(x) - n_f / 2.0
        
        bounds_f = [(0.0, 1.0)] * n_f
        cons_f = [{'type': 'eq', 'fun': weighted_eq}]
        try:
            res = minimize(weighted_obj, best_h, method='SLSQP',
                          bounds=bounds_f, constraints=cons_f,
                          options={'maxiter': 200, 'ftol': 1e-14, 'iprint': 0})
            if res.x is not None:
                h_w = project_feasible(res.x, n_f)
                c5_w = compute_c5(h_w)
                if c5_w < best_c5:
                    best_c5 = c5_w
                    best_h = h_w.copy()
        except Exception:
            pass

    # === Phase 8: Final polish ===
    if remaining_time() > 20:
        for beta in [5000, 20000, 50000]:
            if remaining_time() < 8:
                break
            h_new = slsqp_smooth(best_h, best_n, beta, maxiter=150)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # === Phase 9: Final minimax ===
    if remaining_time() > 15:
        h_mm2 = slsqp_minimax(best_h, best_n, max_rounds=8, max_slsqp_iter=150)
        c5_mm2 = compute_c5(h_mm2)
        if c5_mm2 < best_c5:
            best_c5 = c5_mm2
            best_h = h_mm2.copy()

    # === Phase 10: Very fine binarization + micro-search ===
    if remaining_time() > 15:
        # Try thresholding at different levels
        for thresh in [0.3, 0.4, 0.5, 0.6, 0.7]:
            if remaining_time() < 5:
                break
            h_t = (best_h > thresh).astype(float)
            ones = np.sum(h_t == 1)
            target = best_n // 2
            if ones > target:
                order = np.argsort(best_h[np.where(h_t == 1)[0]])
                pos = np.where(h_t == 1)[0]
                for i in range(int(ones - target)):
                    h_t[int(pos[order[i]])] = 0
            elif ones < target:
                order = np.argsort(best_h[np.where(h_t == 0)[0]])[::-1]
                pos = np.where(h_t == 0)[0]
                for i in range(int(target - ones)):
                    h_t[int(pos[order[i]])] = 1
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
