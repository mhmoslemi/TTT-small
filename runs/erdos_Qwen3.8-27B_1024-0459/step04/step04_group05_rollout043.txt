def remaining_time():

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmax(np.minimum(h, 1.0 - h)))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def make_thue_morse(n):

def simulated_annealing_binary(h, n, rng, max_iters=3000, T_start=0.003, T_end=0.0001):

                h_best = h.copy()

def subgradient_refine(h, n, budget_fn, rng, max_iters=100):
    """Subgradient method for the min-max problem."""
    h = h.copy()
    c5_best = compute_c5(h)
    dx = 2.0 / n
    no_improve = 0

    for it in range(max_iters):
        if budget_fn() < 4:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        # Find active set (top k shifts)
        active = np.where(np.abs(corr - max_val) < 0.01 * max_val + 1e-6)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        # Compute subgradient
        grad = np.zeros(n)
        for k_idx in active[:5]:
            offset = int(k_idx) - (n - 1)
            if offset >= 0:
                n_pairs = n - offset
                if n_pairs > 0:
                    grad[:n_pairs] += (1.0 - h[offset:offset + n_pairs])
                    grad[offset:] -= h[:n_pairs]
            else:
                abs_off = -offset
                n_pairs = n - abs_off
                if n_pairs > 0:
                    grad[:n_pairs] -= h[abs_off:abs_off + n_pairs]
                    grad[abs_off:] += (1.0 - h[:n_pairs])
        grad *= dx

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        grad /= gnorm

        improved = False
        for step in [0.01, 0.02, 0.05, 0.1, 0.2, 0.03, 0.005]:
            h_trial = project_feasible(h - step * grad, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-14:
                c5_best = c5_trial
                h = h_trial.copy()
                improved = True
                no_improve = 0
                break
        if not improved:
            no_improve += 1
            if no_improve > 20:
                noise = rng.randn(n) * 0.01
                h_trial = project_feasible(h + noise, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < c5_best - 1e-14:
                    c5_best = c5_trial
                    h = h_trial.copy()
                    no_improve = 0

    return h, c5_best

        best_h, best_c5 = do_slsqp_multi(best_h, best_n, [500, 2000, 10000, 50000], budget_fn)
        print("  C5={:.10f}".format(best_c5))

    # Phase 4: Subgradient refinement
    if budget_fn() > 20:
        print("\n--- Phase 4: Subgradient ---")
        best_h, best_c5 = subgradient_refine(best_h, best_n, budget_fn, rng, max_iters=80)
        print("  C5={:.10f}".format(best_c5))

    # Phase 5: Multi-resolution
    print("\n--- Phase 5: Multi-resolution ---")
    for n_try in [384, 512, 640, 768, 1024]:
        if budget_fn() < 40:
            break
        if n_try % 2 != 0:
            continue
        h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        c5_interp = compute_c5(h_interp)

        h_new, c5_new = do_slsqp_multi(h_interp, n_try, [200, 1000, 5000], budget_fn)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n_try
            print("  n={} SLSQP: C5={:.10f}".format(n_try, c5_new))
        else:
            print("  n={} SLSQP: C5={:.10f} (best={:.10f})".format(n_try, c5_new, best_c5))

        # Binary at this resolution
        if budget_fn() > 20:
            h_b = binarize(best_h, best_n, rng)
            h_b, c5_b = greedy_binary_search(h_b, best_n, rng, max_iters=40)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                print("  n={} binary: C5={:.10f}".format(n_try, c5_b))

    # Phase 6: Simulated annealing at best resolution
    if budget_fn() > 30:
        print("\n--- Phase 6: Simulated Annealing ---")
        h_sa = binarize(best_h, best_n, rng)
        h_sa, c5_sa = simulated_annealing_binary(h_sa, best_n, rng, max_iters=5000)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        print("  C5={:.10f}".format(best_c5))

    # Phase 7: Multiple restarts
    if budget_fn() > 40:
        print("\n--- Phase 7: Restarts ---")
        for r in range(3):
            if budget_fn() < 20:
                break
            # Perturb and re-optimize
            noise_scale = 0.02 + 0.01 * r
            h_pert = project_feasible(best_h + rng.randn(best_n) * noise_scale, best_n)
            h_new, c5_new = do_slsqp_multi(h_pert, best_n, [500, 2000, 10000], budget_fn)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  Restart {}: C5={:.10f}".format(r, c5_new))

    # Phase 8: Final binary polish
    if budget_fn() > 15:
        print("\n--- Phase 8: Final binary polish ---")
        h_bin = binarize(best_h, best_n, rng)
        h_bin, c5_bin = greedy_binary_search(h_bin, best_n, rng, max_iters=50)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        print("  C5={:.10f}".format(best_c5))

    # Phase 9: Final SLSQP
    if budget_fn() > 10:
        print("\n--- Phase 9: Final SLSQP ---")
        best_h, best_c5 = do_slsqp_multi(best_h, best_n, [10000, 50000, 100000], budget_fn)
        print("  C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    # Ensure exact constraint
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-14:
        best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
