def min_one_shift(h, n, k_star, max_iter=200):
    """Minimize the correlation at a single shift k_star."""
    dx = 2.0 / n
    target = n / 2.0
    shift = int(k_star) - (n - 1)
    
    def obj(x):
        if shift >= 0:
            np_ = n - shift
            return np.sum(x[:np_] * (1.0 - x[shift:shift+np_])) * dx
        else:
            aoff = -shift
            np_ = n - aoff
            return np.sum(x[aoff:aoff+np_] * (1.0 - x[:np_])) * dx
    
    def obj_grad(x):
        g = np.zeros(n)
        if shift >= 0:
            np_ = n - shift
            g[:np_] += (1.0 - x[shift:shift+np_])
            g[shift:] -= x[:np_]
        else:
            aoff = -shift
            np_ = n - aoff
            g[:np_] -= x[aoff:aoff+np_]
            g[aoff:] += (1.0 - x[:np_])
        return g * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    
    res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                  bounds=bounds, constraints=cons,
                  options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
    return project_feasible(res.x, n)

    if time_left() > 15:
        print("\n--- Phase 6: Binary + continuous ---")
        h_bin, c5_bin = make_two_opt_binary(best_h, best_n, max_iters=300)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            print("  Binary improved: C5={:.10f}".format(best_c5))
        else:
            print("  Binary C5={:.10f} (no improve, best={:.10f})".format(c5_bin, best_c5))
        # Refine from binary with continuous opt
        h_work = project_feasible(best_h, best_n)
        h_work, c5_work = do_iterative_shift_opt(h_work, best_n, max_rounds=15)
        if c5_work < best_c5 - 1e-14:
            best_c5 = c5_work
            best_h = h_work.copy()
            print("  After cont refine: C5={:.10f}".format(best_c5))

    # Phase 7: Multi-resolution
    print("\n--- Phase 7: Multi-resolution ---")
    best_h, best_c5, best_n = do_multi_resolution_polish(best_h, best_n, best_c5)
    print("  Best: n={}, C5={:.10f}".format(best_n, best_c5))

    # Phase 8: Perturbed restarts at best resolution
    if time_left() > 30:
        print("\n--- Phase 8: Perturbed restarts ---")
        best_h, best_c5 = do_perturb_restart(best_h, best_n, n_restarts=5)
        print("  C5={:.10f}".format(best_c5))

    # Phase 9: Final polish
    if time_left() > 15:
        print("\n--- Phase 9: Final LSE + shift opt ---")
        best_h, best_c5 = do_lse_slsqp(best_h, best_n, [5000, 20000, 50000])
        print("  After LSE: C5={:.10f}".format(best_c5))
    if time_left() > 10:
        best_h, best_c5 = do_iterative_shift_opt(best_h, best_n, max_rounds=15)
        print("  After shift: C5={:.10f}".format(best_c5))

    # Phase 10: Tiny subgradient polish
    if time_left() > 8:
        print("\n--- Phase 10: Tiny subgrad ---")
        best_h, best_c5 = do_subgradient(best_h, best_n, max_iters=50, noise_scale=0.002)
        print("  C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
