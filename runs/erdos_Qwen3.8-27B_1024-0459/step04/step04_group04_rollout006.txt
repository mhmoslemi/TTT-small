Intuitively, we want the 1s and 0s to be as "interleaved" as possible at ALL scales, not just adjacent. This is related to the concept of a "well-mixed" sequence.

A good construction might be based on a van der Corput sequence or a similar low-discrepancy sequence. Or perhaps a婉 sequence based on Gray codes.

For the SA:
- State: binary array h of length n, with exactly n/2 ones
- Move: pick random i with h[i]=1 and random j with h[j]=0, swap them
- Energy: max(corr) where corr = np.correlate(h, 1-h, 'full')
- Cost: C₅ = max(corr) * 2/n

I'll use a multi-start SA with:

import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def max_corr_val(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def make_binary_from_continuous(h, n, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        perm = rng.permutation(len(idx))[:n_ones - target]
        h_bin[idx[perm]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        perm = rng.permutation(len(idx))[:target - n_ones]
        h_bin[idx[perm]] = 1
    return h_bin

def sa_binary_fast(h, n, time_limit, seed, T_init=None):

    h = h.copy()
    target = n // 2
    
    # Ensure exactly target ones
    n_ones = int(h.sum())
    if n_ones > target:

        perm = rng.permutation(len(idx))[:n_ones - target]

    elif n_ones < target:

        perm = rng.permutation(len(idx))[:target - n_ones]

    best_energy = cur_energy
    best_h = h.copy()

def targeted_sa(h, n, time_limit, seed):
    """SA with targeted moves: preferentially swap positions that affect high-correlation shifts."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    target = n // 2
    
    n_ones = int(h.sum())
    if n_ones > target:
        idx = np.where(h == 1)[0]
        perm = rng.permutation(len(idx))[:n_ones - target]
        h[idx[perm]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        perm = rng.permutation(len(idx))[:target - n_ones]
        h[idx[perm]] = 1
    
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_energy = float(corr.max())
    best_energy = cur_energy
    best_h = h.copy()
    
    T = max(2.0, cur_energy * 0.05)
    alpha = 0.9998
    start = time.time()
    stall = 0
    max_stall = 8000
    
    while time.time() - start < time_limit:
        if remaining_time() < 5:
            break
        
        # With probability 0.3, do a targeted move
        if rng.rand() < 0.3:
            # Find the shift with max correlation
            max_k = int(np.argmax(corr))
            off = max_k - (n - 1)
            # Try to swap a position that's involved in this correlation
            # For shift off > 0: c involves pairs (i, i+off) for i=0..n-1-off
            # We want to change a pair where h[i+off]=1, h[i]=0
            if off > 0:

                # Find pairs where h[i+off]=1, h[i]=0

                candidates = np.where(mask)[0]
                if len(candidates) > 0:

                    j = i + off
                    h[i], h[j] = 1.0, 0.0
                    new_corr = np.correlate(h, 1.0 - h, mode='full')
                    new_energy = float(new_corr.max())
                    delta = new_energy - cur_energy

                        corr = new_corr
                        if new_energy < best_energy:

                            stall = 0
                        else:
                            stall += 1
                    else:
                        h[i], h[j] = 0.0, 1.0
                        stall += 1
                    T *= alpha
                    if stall > max_stall:
                        T = max(T, cur_energy * 0.03)
                        stall = 0
                    continue
        
        # Standard random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_energy = float(new_corr.max())
        delta = new_energy - cur_energy
        
        accept = False
        if delta < 0:
            accept = True
        elif T > 0.5 and -delta / T < 700:
            accept = rng.rand() < np.exp(-delta / T)
        
        if accept:
            cur_energy = new_energy
            corr = new_corr
            if new_energy < best_energy:
                best_energy = new_energy
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
        
        T *= alpha
        if stall > max_stall:
            T = max(T, cur_energy * 0.03)
            stall = 0
    
    return best_h, best_energy * dx, best_energy

def smooth_obj_and_grad(x, n, beta):
    """Compute smooth max approximation and its gradient."""
    dx_val = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    val = (m + np.log(ex.sum()) / beta) * dx_val
    
    # Gradient
    w = ex / ex.sum()
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-14:
            off = k - (n - 1)
            if off >= 0:
                npairs = n - off
                g[off:off+npairs] += w[k] * (1.0 - x[:npairs])
                g[:npairs] -= w[k] * x[off:off+npairs]
            else:
                ao = -off
                npairs = n - ao
                g[:npairs] += w[k] * (1.0 - x[ao:ao+npairs])
                g[ao:ao+npairs] -= w[k] * x[:npairs]
    
    return val, g * dx_val

def slsqp_refine(h, n, beta, maxiter=300):
    def obj(x):
        v, _ = smooth_obj_and_grad(x, n, beta)
        return v
    
    def grad(x):
        _, g = smooth_obj_and_grad(x, n, beta)
        return g
    
    def cons_fun(x):
        return np.sum(x) - n/2.0
    
    def cons_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except:
        return h
