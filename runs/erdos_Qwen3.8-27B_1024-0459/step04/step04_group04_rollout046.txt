Actually, let me think about a smarter approach for the continuous case. 

**Key insight**: The objective max_k c_k is the upper envelope of linear functions. At the optimum, the subgradient (convex combination of active normals) must be in the normal cone of the feasible set. For box + equality constraints, this means:
- For the active shifts k_1,...,k_m, the weighted sum of their gradients should be 0 in the directions where 0<h[i]<1, and the right sign at the boundaries.

A practical algorithm:
1. Find the active set (shifts within epsilon of max)
2. Solve the KKT conditions: find weights λ_k ≥ 0, sum λ_k = 1, and a vector ν such that:
   - For 0 < h[i] < 1: sum_k λ_k * g_k[i] = ν
   - For h[i] = 0: sum_k λ_k * g_k[i] ≥ ν
   - For h[i] = 1: sum_k λ_k * g_k[i] ≤ ν
   
   where g_k is the gradient of c_k.

This is harder to solve directly, but we can approximate with iterative methods.

import numpy as np
from scipy.optimize import minimize

_START = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_grad_one(h, n, k):
    """Gradient of the k-th correlate value w.r.t. h"""
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[:npairs] += (1.0 - h[off:off+npairs])
        g[off:off+npairs] -= h[:npairs]
    else:
        ao = -off
        npairs = n - ao
        g[ao:ao+npairs] += (1.0 - h[:npairs])
        g[:npairs] -= h[ao:ao+npairs]
    return g

def corr_grad_avg(h, n, active_indices):
    """Average gradient over active indices"""
    g = np.zeros(n)
    for k in active_indices:
        g += corr_grad_one(h, n, int(k))
    g /= max(1, len(active_indices))
    return g

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def make_binary(h, rng):
    h_bin = (h > 0.5).astype(float).copy()
    n = len(h_bin)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        if len(idx) < n_ones - target:
            idx = np.where(h_bin == 1)[0]
        h_bin[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1
    return h_bin

def sa_binary_biased(h_bin, n, time_limit=60, seed=42):
    """SA with move biasing towards positions in worst shifts"""
    rng = np.random.RandomState(seed)
    h = h_bin.astype(float).copy()
    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()
    
    T = 4.0
    alpha = 0.9992
    start = time.time()
    stall = 0
    max_stall = 15000
    
    while time.time() - start < time_limit and T > 0.02:
        if remaining() < 8:
            break
        if stall > max_stall:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Bias towards worst shift
        worst_k = int(np.argmax(corr))
        off = worst_k - (n - 1)
        if off >= 0:
            npairs = n - off
            contrib_one = h[:npairs].copy()  # positions i where h[i]=1
            contrib_zero = (1.0 - h[off:off+npairs]).copy()  # positions i+off where h=0
            # Mark contributing positions
            bad = np.zeros(n)
            m = contrib_one * contrib_zero  # 1 where h[i]=1 and h[i+off]=0
            bad[:npairs] += m

def proj_subgrad(h, n, time_limit=60, seed=42):

    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx_val = 2.0 / n
    
    lr = 0.003
    momentum = 0.9
    v = np.zeros(n)
    
    start = time.time()
    while time.time() - start < time_limit and remaining() > 8:
        corr = compute_corr(h)
        m = float(np.max(corr))
        c5_cur = m * dx_val
        
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        
        # Active set: within 0.5 of max
        active = np.where(corr >= m - 0.5)[0]
        g = corr_grad_avg(h, n, active)
        
        # Nesterov-like step
        v = momentum * v - lr * g
        h_trial = h + v
        h_trial = project_feasible(h_trial, n)
        
        c_new = compute_c5(h_trial)
        if c_new < c5_cur:
            h = h_trial
            lr = min(lr * 1.02, 0.01)
        else:
            # Reject, reduce step
            lr *= 0.7
            v = 0.0  # Reset momentum
            if lr < 1e-7:
                break
    
    return best_h, best_c5

def smooth_slsqp(h, n, beta, maxiter=300):
    """SLSQP with log-sum-exp smoothing"""
    dx_val = 2.0 / n
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -50, 50)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_val
    
    def gradient(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        cl = np.clip(beta * shifted, -50, 50)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k_idx in range(len(corr)):
            if w[k_idx] > 1e-10:
                g += w[k_idx] * corr_grad_one(x, n, k_idx)
        return g * dx_val
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)
