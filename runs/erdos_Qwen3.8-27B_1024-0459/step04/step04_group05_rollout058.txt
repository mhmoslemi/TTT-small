            grad[d:d + pairs] += w * (1.0 - x[:pairs])
            grad[:pairs] -= w * x[d:d + pairs]
    return grad * _G_DX


def _con_eq(x):
    return np.sum(x) - _G_TGT


def _con_eq_jac(x):
    return np.ones(_G_N)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h


def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 8.0)


def do_slsqp(h_start, n, budget_fn, beta_list, verbose=True):
    global _G_N, _G_BETA, _G_TGT, _G_DX
    _G_N = n
    _G_TGT = n * 0.5
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_eq_jac}]
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    for beta_val in beta_list:
        if remaining_time() < 8.0:
            break
        _G_BETA = float(beta_val)
        if beta_val < 200:
            maxiter = 500
        elif beta_val < 2000:
            maxiter = 400
        elif beta_val < 20000:
            maxiter = 300
        else:
            maxiter = 200
        est_sec = remaining_time() * 0.35
        maxiter = min(maxiter, max(50, int(est_sec / 0.4)))
        try:
            res = minimize(_obj_func, h_cur, method='SLSQP', bounds=bounds,
                           constraints=constraints, jac=_jac_func,
                           options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                h_cur = h_new
                if verbose:
                    print("    beta={}: C5={:.10f}".format(beta_val, c5_cur))
        except Exception:
            pass
    return h_cur, c5_cur


def do_smoothed_pg(h_start, n, budget_fn, max_iters=300, beta_pg=100.0):
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    stall = 0
    for it in range(max_iters):
        if remaining_time() < 3.0:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        m = np.max(corr)
        shifted = np.clip(corr - m, -200.0 / beta_pg, 0.0)
        expo = np.exp(beta_pg * shifted)
        weights = expo / np.sum(expo)
        grad = np.zeros(n)
        for k in range(n_lags):
            w = weights[k]
            if w < 1e-7:
                continue
            offset = k - (n - 1)
            d = abs(offset)
            pairs = n - d
            if pairs <= 0:
                continue
            if offset >= 0:
                grad[:pairs] += w * (1.0 - h_cur[d:d + pairs])
                grad[d:d + pairs] -= w * h_cur[:pairs]
            else:
                grad[d:d + pairs] += w * (1.0 - h_cur[:pairs])
                grad[:pairs] -= w * h_cur[d:d + pairs]
        grad *= dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-17:
            break
        grad /= gnorm
        improved = False
        for step in [0.01, 0.02, 0.05, 0.1, 0.005, 0.2, 0.002, 0.5]:
            h_trial = project_feasible(h_cur - step * grad, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_cur - 1e-14:
                c5_cur = c5_trial
                h_cur = h_trial
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 40:
                noise = np.random.RandomState(it).randn(n) * 0.01
                h_trial = project_feasible(h_cur + noise, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < c5_cur:
                    c5_cur = c5_trial
                    h_cur = h_trial
                if stall > 80:
                    break
    return h_cur, c5_cur


def do_binary_local_search(h_start, n, budget_fn, rng, max_sweeps=10):
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur)
    for sweep in range(max_sweeps):
        if remaining_time() < 3.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if remaining_time() < 1.5:
                break
            old = h_cur[i]
            for nv in [0.0, 1.0]:
                if abs(nv - old) < 1e-7:
                    continue
                h_cur[i] = nv
                h_cur = project_feasible(h_cur, n)
                c5_t = compute_c5(h_cur)
                if c5_t < c5_cur - 1e-14:
                    c5_cur = c5_t
                    h_cur = h_cur.copy()
                    any_imp = True
                else:
                    h_cur[i] = old
                    h_cur = project_feasible(h_cur, n)
        if not any_imp and sweep > 2:
            break
    return h_cur, c5_cur


def do_perturbed_restart(h_best, n, budget_fn, rng, n_restarts=4):
    c5_best = compute_c5(h_best)
    for r in range(n_restarts):
        if remaining_time() < 20.0:
            break
        noise_scale = 0.02 * (1.0 + 0.5 * r)
        h_pert = project_feasible(h_best + rng.randn(n) * noise_scale, n)
        c5_pert = compute_c5(h_pert)
        h_cur = h_pert.copy() if c5_pert < c5_best else h_best.copy()
        c5_cur = min(c5_pert, c5_best)
        h_cur, c5_cur = do_slsqp(h_cur, n, budget_fn, [200, 1000, 5000], verbose=False)
        if c5_cur < c5_best - 1e-14:
            c5_best = c5_cur
            h_best = h_cur.copy()
    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    def budget_fn():
        return max(0.0, _G_BUDGET - (time.time() - _G_START) - 8.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent or create initial
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.10f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        parent_h = project_feasible(make_thue_morse(parent_n), parent_n)
        best_h = parent_h.copy()
        best_c5 = compute_c5(best_h)
        best_n = parent_n
        print("No parent. TM n={}, C5={:.10f}".format(parent_n, best_c5))

    # Phase 1: SLSQP with analytical gradient at parent resolution
    print("\n--- Phase 1: SLSQP n={} ---".format(best_n))
    beta1 = [20, 50, 100, 200, 500, 1000, 2000, 5000]
    best_h, best_c5 = do_slsqp(best_h, best_n, budget_fn, beta1)
    print("  After SLSQP: C5={:.10f}".format(best_c5))

    # Phase 2: Smoothed projected gradient
    if remaining_time() > 20:
        print("\n--- Phase 2: Smoothed PG ---")
        best_h, best_c5 = do_smoothed_pg(best_h, best_n, budget_fn, max_iters=300, beta_pg=80.0)
        print("  After PG(b=80): C5={:.10f}".format(best_c5))
        if remaining_time() > 20:
            best_h, best_c5 = do_smoothed_pg(best_h, best_n, budget_fn, max_iters=200, beta_pg=200.0)
            print("  After PG(b=200): C5={:.10f}".format(best_c5))

    # Phase 3: High-beta SLSQP
    if remaining_time() > 20:
        print("\n--- Phase 3: High-beta SLSQP ---")
        beta3 = [5000, 10000, 20000, 50000, 100000]
        best_h, best_c5 = do_slsqp(best_h, best_n, budget_fn, beta3)
        print("  After high-beta: C5={:.10f}".format(best_c5))

    # Phase 4: Try higher resolution
    print("\n--- Phase 4: Higher resolution ---")
    for n_try in [320, 384, 512]:
        if remaining_time() < 40.0:
            break
        h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        c5_interp = compute_c5(h_interp)
        h_new, c5_new = do_slsqp(h_interp, n_try, budget_fn, [100, 500, 2000, 10000], verbose=False)
        if remaining_time() > 15:
            h_new, c5_new = do_smoothed_pg(h_new, n_try, budget_fn, max_iters=100, beta_pg=150.0)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n_try
            print("  n={} improved: C5={:.10f}".format(n_try, best_c5))
        else:
            print("  n={} no improve: C5={:.10f} (best={:.10f})".format(n_try, c5_new, best_c5))

    # Phase 5: Perturbed restarts
    if remaining_time() > 30:
        print("\n--- Phase 5: Perturbed restarts ---")
        best_h, best_c5 = do_perturbed_restart(best_h, best_n, budget_fn, rng, n_restarts=4)
        print("  After restarts: C5={:.10f}".format(best_c5))

    # Phase 6: Binary local search
    if remaining_time() > 10:
        print("\n--- Phase 6: Binary local search ---")
        best_h, best_c5 = do_binary_local_search(best_h, best_n, budget_fn, rng, max_sweeps=5)
        print("  After binary: C5={:.10f}".format(best_c5))

    # Phase 7: Final SLSQP polish
    if remaining_time() > 15:
        print("\n--- Phase 7: Final polish ---")
        best_h, best_c5 = do_slsqp(best_h, best_n, budget_fn, [20000, 50000, 100000, 500000], verbose=False)
        print("  After final SLSQP: C5={:.10f}".format(best_c5))

    # Phase 8: Final smoothed PG
    if remaining_time() > 10:
        best_h, best_c5 = do_smoothed_pg(best_h, best_n, budget_fn, max_iters=100, beta_pg=500.0)
        print("  After final PG: C5={:.10f}".format(best_c5))

    # Phase 9: One more SLSQP with very high beta
    if remaining_time() > 10:
        best_h, best_c5 = do_slsqp(best_h, best_n, budget_fn, [200000, 1000000], verbose=False)
        print("  After ultra SLSQP: C5={:.10f}".format(best_c5))

    # Final validation and projection
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
