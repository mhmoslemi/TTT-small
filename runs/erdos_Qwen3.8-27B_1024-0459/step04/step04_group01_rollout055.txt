def project_feasible(h, n):

    target = n / 2.0
    # Binary search for the right shift
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0

    """Gradient of corr[k] w.r.t. h, where k is the full-correlation index."""
    g = np.zeros(n)
    off = k - (n - 1)

        if npairs > 0:
            g[:npairs] += (1.0 - h[ao:ao+npairs])
            g[ao:ao+npairs] -= h[:npairs]
    return g

    c_obj[-1] = 1.0
    A_ub = np.hstack([A_grads, -np.ones((m, 1))])
    b_ub = -curr_vals
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)]
    bounds.append((0.0, None))
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                           'primal_feasibility_tolerance': 1e-9,
                           'time_limit': 5.0})
    if res.status != 0:
        return h
    delta = res.x[:n].copy()
    delta_norm = np.linalg.norm(delta)
    if delta_norm > trust_r:
        delta *= trust_r / delta_norm
    h_new = np.clip(h + delta, 0.0, 1.0)
    h_new = project_feasible(h_new, n)
    return h_new

def lp_optimize(h, n, time_limit=300):
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    trust_r = 0.5
    n_top = None
    stagnation = 0
    for it in range(200000):
        if time.time() - start > time_limit:
            break
        if remaining() < 10:
            break
        if it < 30:
            cur_top = None
        elif it < 200:
            cur_top = 20
        elif it < 500:
            cur_top = 10
        else:
            cur_top = 5
        h_new = lp_step(h, n, trust_r, n_top=cur_top)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stagnation = 0
            trust_r = min(trust_r * 1.02, 1.5)
        else:
            stagnation += 1
            trust_r = max(trust_r * 0.92, 0.005)
            if stagnation > 80:
                break
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def slsqp_smooth(h, n, beta, maxiter=300):
    global _N_global, _BETA_global, _TGT_global
    _N_global = n
    _BETA_global = beta
    _TGT_global = n / 2.0
    dx_val = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        corr = compute_corr(x)
        mx = np.max(corr)
        shifted = corr - mx
        cl = np.clip(beta * shifted, -50, 50)
        ex = np.exp(cl)
        return (mx + np.log(np.sum(ex)) / beta) * dx_val

    def grad(x):
        corr = compute_corr(x)
        mx = np.max(corr)
        shifted = corr - mx
        cl = np.clip(beta * shifted, -50, 50)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k_idx in range(len(corr)):
            if w[k_idx] > 1e-14:
                g += w[k_idx] * corr_grad_shift(x, n, k_idx)
        return g * dx_val

    def con_fun(x):
        return np.sum(x) - _TGT_global

    def con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

_N_global = 0
_BETA_global = 1.0
_TGT_global = 0.0

def coord_descent_focused(h, n, time_limit=60, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = compute_corr(h)
    nc = 2 * n - 1
    n_half = n // 2
    for pass_num in range(5000):
        if time.time() - start > time_limit:
            break
        if remaining() < 10:
            break
        for i in range(n):
            j = (i + n_half) % n
            if i == j:
                j = (i + 1) % n
            s = h[i] + h[j]
            lo_t = max(0.0, s - 1.0)
            hi_t = min(1.0, s)
            if lo_t >= hi_t - 1e-14:
                continue
            sl_i = corr_grad_shift(h, n, i + 0)  # placeholder, need proper shift
            # Actually we need the slope of each corr entry w.r.t. h[i] and h[j]
            # Slope of corr[k] w.r.t. h[i]:
            # Let's compute using the gradient formula
            sl_i_vec = np.zeros(nc)
            sl_j_vec = np.zeros(nc)
            for k_idx in range(nc):
                g_k = corr_grad_shift(h, n, k_idx)
                sl_i_vec[k_idx] = g_k[i]
                sl_j_vec[k_idx] = g_k[j]
            diff_sl = sl_i_vec - sl_j_vec
            intercepts = corr - h[i] * diff_sl
            # Minimize max(intercepts + t * diff_sl) over t in [lo_t, hi_t]
            lo, hi = lo_t, hi_t
            for _ in range(40):
                mid = (lo + hi) / 2.0
                vals = diff_sl * mid + intercepts
                max_val = vals.max()
                active = vals >= max_val - 1e-13
                if active.sum() == 0:
                    break
                s_max = diff_sl[active].max()
                s_min = diff_sl[active].min()
                if s_max < -1e-14:
                    lo = mid
                elif s_min > 1e-14:
                    hi = mid
                else:
                    break
            t_opt = (lo + hi) / 2.0
            if abs(t_opt - h[i]) < 1e-13:
                continue
            delta = t_opt - h[i]
            corr += delta * diff_sl
            h[i] = t_opt
            h[j] = s - t_opt
        h = project_feasible(h, n)
        corr = compute_corr(h)
        c5_new = float(np.max(corr) * 2.0 / n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 312
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        best_h = project_feasible(parent_h, parent_n)
        best_c5 = compute_c5(best_h)
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: LP optimization on parent resolution (main workhorse)
    if remaining() > 20:
        h_lp, c5_lp = lp_optimize(h, n, time_limit=min(350, remaining() - 20))
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()
            h = h_lp.copy()

    # Phase 2: SLSQP polish
    for beta in [2000, 8000, 20000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: More LP after SLSQP
    if remaining() > 30:
        h_lp, c5_lp = lp_optimize(h, n, time_limit=min(200, remaining() - 30))
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()
            h = h_lp.copy()

    # Phase 4: Try other n values
    n_candidates = sorted(set([
        best_n - 4, best_n - 2, best_n + 2, best_n + 4,
        best_n + 8, best_n - 8, best_n + 16, best_n - 16,
    ]))
    n_candidates = [nn for nn in n_candidates if 64 <= nn <= 1024 and nn % 2 == 0 and nn != best_n]

    for n_t in n_candidates:
        if remaining() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        # LP on new n
        h_opt, c5_opt = lp_optimize(h_feas, n_t, time_limit=min(80, remaining() - 30))
        if remaining() < 10:
            break
        # SLSQP on new n
        h_opt = slsqp_smooth(h_opt, n_t, 5000, maxiter=200)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 5: Final intensive LP on best
    n = best_n
    h = best_h.copy()
    if remaining() > 20:
        h_lp, c5_lp = lp_optimize(h, n, time_limit=min(150, remaining() - 20))
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()
            h = h_lp.copy()

    # Phase 6: Coordinate descent polish
    if remaining() > 15:
        h_cd, c5_cd = coord_descent_focused(h, n, time_limit=min(60, remaining() - 15), rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Phase 7: Final LP + SLSQP
    if remaining() > 10:
        h_lp, c5_lp = lp_optimize(h, n, time_limit=min(40, remaining() - 10))
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()
            h = h_lp.copy()
    if remaining() > 8:
        h_new = slsqp_smooth(h, n, 30000, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 8: Final tiny LP
    if remaining() > 5:
        h_lp, c5_lp = lp_optimize(best_h, best_n, time_limit=min(20, remaining() - 5))
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()

    # Validate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
