def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

        best_h = h_cur.copy()
    print("  C5={:.10f}".format(best_c5))

    # Phase 2: Subgradient with momentum
    if remaining() > 25:
        print("\n--- Phase 2: Subgradient (n={}) ---".format(best_n))
        h_cur, c5_cur = subgrad_optimize(best_h, best_n, max_iter=350, step0=0.06,
                                         momentum=0.9, top_k=5, time_limit_fn=remaining)
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        # SLSQP follow-up
        h_cur, c5_cur = slsqp_polish(best_h, best_n, [5000, 10000, 30000],
                                      max_iter_each=200, time_limit_fn=remaining)
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        print("  C5={:.10f}".format(best_c5))

    # Phase 3: Targeted lag reduction
    if remaining() > 20:
        print("\n--- Phase 3: Targeted lag reduction (n={}) ---".format(best_n))
        best_h, best_c5 = targeted_lag_reduce(best_h, best_n, num_rounds=60, time_limit_fn=remaining)
        h_cur, c5_cur = slsqp_polish(best_h, best_n, [10000, 30000],
                                      max_iter_each=150, time_limit_fn=remaining)
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        print("  C5={:.10f}".format(best_c5))

    # Phase 4: Fine coordinate descent
    if remaining() > 15:
        print("\n--- Phase 4: Coordinate descent (n={}) ---".format(best_n))
        best_h, best_c5 = coordinate_descent_fine(best_h, best_n, max_sweeps=12,
                                                   rng=rng, time_limit_fn=remaining)
        print("  C5={:.10f}".format(best_c5))

    # Phase 5: Perturbation + SLSQP
    if remaining() > 20:
        print("\n--- Phase 5: Perturbation exploration (n={}) ---".format(best_n))
        best_h, best_c5 = alternating_block_perturb(best_h, best_n, num_tries=25,
                                                     rng=rng, time_limit_fn=remaining)
        h_cur, c5_cur = slsqp_polish(best_h, best_n, [5000, 15000, 40000],
                                      max_iter_each=200, time_limit_fn=remaining)
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        print("  C5={:.10f}".format(best_c5))

    # Phase 6: Resolution sweep
    if remaining() > 60:
        print("\n--- Phase 6: Resolution sweep ---")
        for n_try in [128, 160, 192, 288, 320, 384, 512]:
            if remaining() < 15:
                break
            if n_try == best_n:
                continue
            h_interp = project_feasible(interp_to(best_h, best_n, n_try), n_try)
            c5_try = compute_c5(h_interp)
            h_try, c5_try = subgrad_optimize(h_interp, n_try, max_iter=80, step0=0.05,
                                             top_k=4, time_limit_fn=remaining)
            h_try, c5_try = slsqp_polish(h_try, n_try, [1000, 5000, 15000],
                                         max_iter_each=100, time_limit_fn=remaining)
            if c5_try < best_c5 - 1e-11:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
                print("  n={}: C5={:.10f} IMPROVED".format(n_try, best_c5))
            else:
                print("  n={}: C5={:.10f}".format(n_try, c5_try))

    # Phase 7: Final SLSQP with very high beta
    if remaining() > 10:
        print("\n--- Phase 7: Final high-beta SLSQP (n={}) ---".format(best_n))
        _G_N = best_n
        _G_TGT = best_n / 2.0
        _G_DX = 2.0 / best_n
        h_cur, c5_cur = slsqp_polish(best_h, best_n, [20000, 50000, 100000],
                                      max_iter_each=300, time_limit_fn=remaining)
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        print("  C5={:.10f}".format(best_c5))

    # Phase 8: Another round of targeted + CD
    if remaining() > 12:
        print("\n--- Phase 8: Targeted + CD round 2 (n={}) ---".format(best_n))
        best_h, best_c5 = targeted_lag_reduce(best_h, best_n, num_rounds=40, time_limit_fn=remaining)
        h_cur, c5_cur = slsqp_polish(best_h, best_n, [50000, 100000],
                                      max_iter_each=200, time_limit_fn=remaining)
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        best_h, best_c5 = coordinate_descent_fine(best_h, best_n, max_sweeps=8,
                                                   rng=rng, time_limit_fn=remaining)
        print("  C5={:.10f}".format(best_c5))

    # Phase 9: Final micro-optimization
    if remaining() > 8:
        print("\n--- Phase 9: Micro-opt (n={}) ---".format(best_n))
        _G_N = best_n
        _G_TGT = best_n / 2.0
        _G_DX = 2.0 / best_n
        bounds = [(0.0, 1.0)] * best_n
        cons = [{'type': 'eq', 'fun': sum_constraint_wrapper}]
        for beta_val in [100000, 200000]:
            if remaining() < 4.0:
                break
            _G_BETA = float(beta_val)
            try:
                res = minimize(smooth_obj_wrapper, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-16, 'iprint': 0})
                h_new = project_feasible(res.x, best_n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass
        print("  C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
