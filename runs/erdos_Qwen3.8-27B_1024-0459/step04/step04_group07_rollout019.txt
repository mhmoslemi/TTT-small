        grad[s:] += ws * (1.0 - h[:n-s])

        grad[:n-s] -= ws * h[s:]

        ws = w[n-1-s]
        if ws < 1e-15:

        grad[:n-s] += ws * (1.0 - h[s:])
        # -h[t-s] for t = s..n-1
        grad[s:] -= ws * h[:n-s]
    
    grad *= dx
    return grad

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

    target = n / 2.0
    for _ in range(200):

        else:

        h = np.clip(h, 0.0, 1.0)

            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))

    m = np.max(corr)

def subgradient_step(h, n, lr, top_k=5):
    """Take a subgradient step on the exact max, projected."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top_k active lags
    idx = np.argsort(corr)[::-1][:top_k]
    # Average subgradient over top lags
    grad = np.zeros(n)
    for k in idx:
        s = k - (n - 1)
        a = abs(s)
        g = np.zeros(n)
        if a == 0:
            g = 1.0 - 2.0 * h
        else:
            if s >= 0:
                g[a:] += (1.0 - h[:n-a])
                g[:n-a] -= h[a:]
            else:
                g[:n-a] += (1.0 - h[a:])
                g[a:] -= h[:n-a]
        grad += g
    grad /= top_k
    grad *= dx
    
    h_new = h - lr * grad
    return project_feasible(h_new, n)

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 256
_G_BETA = 100.0
_G_START = 0.0
_G_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def _lob_func(x):
    n = _G_N
    beta = _G_BETA
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = corr - m
    e = np.exp(np.clip(beta * s, -500.0, 500.0))
    se = np.sum(e)
    return (m + np.log(se) / beta) * dx

def _lob_grad(x):
    n = _G_N
    beta = _G_BETA
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = corr - m
    e = np.exp(np.clip(beta * s, -500.0, 500.0))
    se = np.sum(e)
    w = e / se
    grad = np.zeros(n)
    w0 = w[n - 1]
    if w0 > 1e-15:
        grad += w0 * (1.0 - 2.0 * x)
    for shift in range(1, n):
        wp = w[n - 1 + shift]
        wn = w[n - 1 - shift]
        a = shift
        if wp > 1e-14:
            grad[a:] += wp * (1.0 - x[:n - a])
            grad[:n - a] -= wp * x[a:]
        if wn > 1e-14:
            grad[:n - a] += wn * (1.0 - x[a:])
            grad[a:] -= wn * x[:n - a]
    grad *= dx
    return grad

def _eq_fun(x):
    return np.sum(x) - _G_N / 2.0

def _eq_jac(x):
    return np.ones(_G_N)

def run_lbfgs(h0, maxiter, ftol):
    n = _G_N
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lob_func, h0, jac=_lob_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol,
                                               'gtol': 1e-12, 'maxfun': maxiter * 12, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)

def run_slsqp(h0, maxiter):
    n = _G_N
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_lob_func, h0, jac=_lob_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)

def subgradient_optimize(h0, n, n_steps, lr_init, decay, top_k):
    h = h0.copy()
    dx = 2.0 / n
    c5 = compute_c5(h)
    lr = lr_init
    momentum = np.zeros(n)
    mu = 0.85
    for step in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[::-1][:top_k]
        weights = np.zeros(len(corr))
        weights[top_idx] = 1.0 / top_k
        grad = np.zeros(n)
        for k in top_idx:
            s = k - (n - 1)
            a = abs(s)
            g = np.zeros(n)
            if a == 0:
                g = 1.0 - 2.0 * h
            elif s >= 0:
                g[a:] += (1.0 - h[:n - a])
                g[:n - a] -= h[a:]
            else:
                g[:n - a] += (1.0 - h[a:])
                g[a:] -= h[:n - a]
            grad += g
        grad *= dx
        momentum = mu * momentum + (1 - mu) * grad
        h_new = h - lr * momentum
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < c5:
            c5 = c_new
            h = h_new
        else:
            lr *= decay
    return h, c5

def coord_descent(h_init, n, rng, tlimit, max_sweeps):

                trial = project_feasible(trial, n)

                    h = trial.copy()
                    improved = True

                if tlimit() < 0.5:
                    break

                trial = project_feasible(trial, n)

                    c5 = c
                    h = trial.copy()
                    improved = True

            break
    return h, c5

def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c else 0.0
    return h

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()

    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)
