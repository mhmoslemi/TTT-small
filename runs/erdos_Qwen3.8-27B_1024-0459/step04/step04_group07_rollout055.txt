_G_DX = 1.0
_G_PEN = 0.0
_G_TGT = 0.0
_G_FOCUS = -1
_G_TOPK = 1

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_lse(x):

    s = corr - m

def obj_topk(x):

    _G_PEN = 0.0
    _G_TGT = n / 2.0
    _G_FOCUS = -1
    _G_TOPK = topk

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

        else:
            mask = h > 1e-10

        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
