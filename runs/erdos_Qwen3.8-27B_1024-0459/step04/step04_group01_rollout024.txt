            # For coordinate i, the objective is max_k corr[k]
            # As a function of h[i], corr[k] is linear: corr[k] = a_k * h[i] + b_k
            # We want to minimize max_k (a_k * h[i] + b_k) over h[i] in [0,1]
            # subject to sum constraint (which we handle by projection)
            
            # Compute slopes a_k for all k
            # corr[k] = sum_j h[j]*(1-h[j+k-(n-1)])
            # The terms involving h[i] are:
            #   j = i: h[i]*(1-h[i+k-(n-1)]) -> contributes h[i] - h[i]*h[i+k-(n-1)]
            #   j+k-(n-1) = i, i.e., j = i - k + (n-1): h[j]*(1-h[i]) -> contributes -h[j]*h[i]
            
            # This is complex for non-binary h. Let's use a simpler approach:
            # Try h[i] = 0, h[i] = 1, and the current value, pick the best
            old_val = h[i]
            candidates = [0.0, 1.0, old_val]
            # Also try values that equalize top correlations
            best_candidate = old_val
            best_max = float('inf')

                h[i] = cv
                c = compute_corr(h)
                mx = np.max(c)
                if mx < best_max:
                    best_max = mx
                    best_candidate = cv
            
            if best_candidate != old_val:
                h[i] = best_candidate
                improved = True
        
        h = project_feasible(h, n)
        corr = compute_corr(h)
        c5_new = float(np.max(corr)) * 2.0 / n
        if c5_new < best_c5:
            best_c5 = c5_new
        
        if not improved and pass_num > 5:
            break
