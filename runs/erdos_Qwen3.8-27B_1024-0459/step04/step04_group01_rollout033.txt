        T = max(T * cool_rate, T_min)
    
    return best_h, best_max * dx_val

# Improved pair coordinate descent with exact 1D convex optimization
def corr_slope_at_i(h, i, n):
    """Compute slope of corr[d] w.r.t. h[i] for all d."""
    nc = 2 * n - 1
    slopes = np.zeros(nc)
    for idx in range(nc):
        d = idx - (n - 1)
        s = 0.0
        if d >= 0:
            if d <= i < n:
                s += (1.0 - h[i - d])
            if 0 <= i < n - d:
                s -= h[i + d]
        else:
            m = -d
            if 0 <= i < n - m:
                s += (1.0 - h[i + m])
            if i >= m and i < n:
                s -= h[i - m]
        slopes[idx] = s
    return slopes

def solve_1d_convex(slopes, intercepts, lo, hi):
    """Minimize max(slopes*t + intercepts) over t in [lo, hi] exactly."""
    if lo >= hi:
        return lo
    
    # The function f(t) = max_d (slopes[d]*t + intercepts[d]) is convex
    # The minimum is at a point where two (or more) lines are tied for the max
    # We can find it by binary search on the derivative (subgradient)
    
    # Subgradient approach: at optimal t, 0 is in the subgradient
    # Subgradient = {slopes[d] : d is argmax}
    
    def subgradient_contains_zero(t):
        vals = slopes * t + intercepts
        max_val = vals.max()
        active = vals >= max_val - 1e-12
        return slopes[active].min() <= 0 <= slopes[active].max()
    
    lo_t, hi_t = lo, hi
    for _ in range(60):
        mid = (lo_t + hi_t) / 2.0
        vals = slopes * mid + intercepts
        max_val = vals.max()
        active = vals >= max_val - 1e-12
        min_slope = slopes[active].min()
        max_slope = slopes[active].max()
        
        if min_slope > 0:
            # Function is increasing at mid, so optimum is to the left
            hi_t = mid
        elif max_slope < 0:
            # Function is decreasing at mid, so optimum is to the right
            lo_t = mid
        else:
            break
    
    return (lo_t + hi_t) / 2.0

def pair_cd_exact(h, n, max_passes=200, time_limit=60, rng=None):
    """Pair coordinate descent with exact 1D convex solver."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    start = time.time()
    best_c5 = compute_c5(h)
    corr = compute_corr(h)
    
    for pass_num in range(max_passes):
        if time.time() - start > time_limit:
            break
        if remaining() < 10:
            break
        
        # Try different pairing strategies
        if pass_num % 3 == 0:
            order = np.arange(n)
        elif pass_num % 3 == 1:
            order = rng.permutation(n)
        else:
            order = np.arange(n)  # forward
        
        for idx in range(n):
            i = int(order[idx])
            j = (i + n // 2) % n
            if i == j:
                j = (i + 1) % n
            
            s = h[i] + h[j]
            lo_t = max(0.0, s - 1.0)
            hi_t = min(1.0, s)
            if lo_t >= hi_t - 1e-14:
                continue
            
            sl_i = corr_slope_at_i(h, i, n)
            sl_j = corr_slope_at_i(h, j, n)
            diff_sl = sl_i - sl_j
            
            intercepts = corr - h[i] * diff_sl
            t_opt = solve_1d_convex(diff_sl, intercepts, lo_t, hi_t)
            
            if abs(t_opt - h[i]) > 1e-12:
                delta = t_opt - h[i]
                corr += delta * diff_sl
                h[i] = t_opt
                h[j] = s - t_opt
        
        h = project_feasible(h, n)
        corr = compute_corr(h)
        
        c5_new = float(np.max(corr) * 2.0 / n)
        if c5_new < best_c5:
            best_c5 = c5_new
    
    return h, best_c5

def block_binary_search(h, n, time_limit=60, rng=None):
    """Binary local search with block moves."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h))
    target = n // 2
    
    if n_ones > target:
        ones_idx = np.where(h == 1)[0]
        to_remove = rng.choice(ones_idx, n_ones - target, replace=False)
        h[to_remove] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h == 0)[0]
        to_add = rng.choice(zeros_idx, target - n_ones, replace=False)
        h[to_add] = 1.0
    
    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx_val
    best_h = h.copy()
    
    start = time.time()
    stall = 0
    T = 1.0
    
    for it in range(100000):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        found = False
        n_try = 300
        
        for _ in range(n_try):
            # Choose move type
            r = rng.rand()
            
            if r < 0.5:
                # Single swap biased toward worst shift contributors
                k_max = int(np.argmax(corr))
                offset = k_max - (n - 1)
                
                if offset >= 0 and offset < n:
                    # corr at this shift involves pairs (i, i+offset) for i in [0, n-1-offset]
                    # We want to change h[i]=1, h[i+offset]=0 pairs
                    candidates_i = [i for i in ones if i + offset < n and h[i + offset] == 0]
                    if len(candidates_i) > 0:
                        i = candidates_i[rng.randint(len(candidates_i))]
                    else:
                        i = ones[rng.randint(len(ones))]
                elif offset < 0:
                    m = -offset
                    candidates_i = [i for i in ones if i - m >= 0 and h[i - m] == 0]
                    if len(candidates_i) > 0:
                        i = candidates_i[rng.randint(len(candidates_i))]
                    else:
                        i = ones[rng.randint(len(ones))]
                else:
                    i = ones[rng.randint(len(ones))]
                
                j = zeros[rng.randint(len(zeros))]
                
                h[i] = 0.0
                h[j] = 1.0
                new_corr = compute_corr(h)
                new_max = float(np.max(new_corr))
                
                if new_max < cur_max:
                    corr = new_corr
                    cur_max = new_max
                    best_c5 = cur_max * dx_val
                    best_h = h.copy()
                    found = True
                    break
                elif new_max < cur_max + T:
                    # Accept with some probability (SA-like)
                    if rng.rand() < np.exp(-(new_max - cur_max) / max(T, 0.01)):
                        corr = new_corr
                        cur_max = new_max
                        found = True
                        break
                    else:
                        h[i] = 1.0
                        h[j] = 0.0
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            
            elif r < 0.8:
                # Double swap
                if len(ones) < 2 or len(zeros) < 2:
                    continue
                i1, i2 = rng.choice(ones, 2, replace=False)
                j1, j2 = rng.choice(zeros, 2, replace=False)
                
                h[i1] = 0.0
                h[i2] = 0.0
                h[j1] = 1.0
                h[j2] = 1.0
                new_corr = compute_corr(h)
                new_max = float(np.max(new_corr))
                
                if new_max < cur_max:
                    corr = new_corr
                    cur_max = new_max
                    best_c5 = cur_max * dx_val
                    best_h = h.copy()
                    found = True
                    break
                else:
                    h[i1] = 1.0
                    h[i2] = 1.0
                    h[j1] = 0.0
                    h[j2] = 0.0
            
            else:
                # Segment move: flip a small segment and compensate
                seg_len = rng.randint(2, 5)
                start_idx = rng.randint(0, n - seg_len)
                seg = h[start_idx:start_idx + seg_len].copy()
                n_ones_seg = int(np.sum(seg))
                
                # Find a location to put the complement with same number of ones
                # For simplicity, just flip and accept if improves
                h[start_idx:start_idx + seg_len] = 1.0 - seg
                new_corr = compute_corr(h)
                new_max = float(np.max(new_corr))
                
                if new_max < cur_max:
                    corr = new_corr
                    cur_max = new_max
                    best_c5 = cur_max * dx_val
                    best_h = h.copy()
                    found = True
                    break
                else:
                    h[start_idx:start_idx + seg_len] = seg
        
        if found:
            stall = 0
            T = max(0.01, T * 0.999)
        else:
            stall += 1
            T = max(0.01, T * 0.99)
            if stall > 200:
                # Restart temperature
                T = 0.5
                stall = 0
    
    return best_h, best_c5

Let me think of a faster approach. The key insight is that for a single swap (move 1 from position i to position j), the change in correlation can be computed in O(n) time instead of O(n²).

For a swap of h[i] -= 1 and h[j] += 1:
- For each shift d, corr[d] changes by:
  - From h[i]: the term involving h[i] changes
  - From h[j]: the term involving h[j] changes

Specifically, corr[d] = Σ_k h[k](1-h[k-d]) (for appropriate ranges)

When h[i] changes by -1:
- Terms where h[i] is the "1" part: (1-h[i-d]) → decreases by 1 if i-d is valid and h[i-d]=0
- Terms where h[i] is the "0" part: h[k](1-h[i]) → increases by h[k] if k+i... 

This is getting complicated. Let me just use the slope computation.

The change in corr when we change h[i] by δ_i and h[j] by δ_j is:
Δcorr[d] = δ_i * slope_i[d] + δ_j * slope_j[d]

where slope_i[d] is the partial derivative of corr[d] w.r.t. h[i].

For a swap (h[i] goes from 1 to 0, h[j] goes from 0 to 1):
δ_i = -1, δ_j = +1
Δcorr[d] = -slope_i[d] + slope_j[d] = slope_j[d] - slope_i[d]

So we can compute the new corr as: corr_new = corr + (slope_j - slope_i)

This is O(n) to compute the slopes (well, O(n * n_corr) = O(n²) total), but we only need to do it once per candidate, not for the full recompute.

Actually, computing slope_i and slope_j each take O(n) time (since there are 2n-1 correlation values and each slope computation is O(1) per value). So the total per swap is O(n), which is much better than O(n²) for the full recompute.

def corr_slope_at_i_vec(h, i, n):
    """Vectorized computation of slope of corr w.r.t. h[i]."""
    nc = 2 * n - 1
    d = np.arange(nc) - (n - 1)
    idx1 = i - d
    valid1 = (idx1 >= 0) & (idx1 < n)
    safe1 = np.clip(idx1, 0, n - 1)
    term1 = np.where(valid1, 1.0 - h[safe1], 0.0)
    idx2 = i + d
    valid2 = (idx2 >= 0) & (idx2 < n)
    safe2 = np.clip(idx2, 0, n - 1)
    term2 = np.where(valid2, -h[safe2], 0.0)
    return term1 + term2


def solve_1d_convex(slopes, intercepts, lo, hi):
    """Minimize max(slopes*t + intercepts) over t in [lo, hi]."""
    if lo >= hi:
        return lo
    lo_t, hi_t = lo, hi
    for _ in range(50):
        mid = (lo_t + hi_t) / 2.0
        vals = slopes * mid + intercepts
        max_val = vals.max()
        active = vals >= max_val - 1e-12
        min_slope = slopes[active].min()
        max_slope = slopes[active].max()
        if min_slope > 1e-14:
            hi_t = mid
        elif max_slope < -1e-14:
            lo_t = mid
        else:
            break
    return (lo_t + hi_t) / 2.0


def pair_cd(h, n, max_passes=300, time_limit=60, rng=None):
    """Pair coordinate descent with vectorized slope computation."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    start = time.time()
    best_c5 = compute_c5(h)
    corr = compute_corr(h)

    for pass_num in range(max_passes):
        if time.time() - start > time_limit:
            break
        if remaining() < 10:
            break

        if pass_num % 4 == 0:
            # Standard pairing: i with i+n/2
            pairs = [(i, (i + n // 2) % n) for i in range(n // 2)]
        elif pass_num % 4 == 1:
            # Adjacent pairing
            pairs = [(i, (i + 1) % n) for i in range(0, n - 1, 2)]
            if n % 2 == 0:
                pairs[-1] = (n - 2, n - 1)
        elif pass_num % 4 == 2:
            # Random pairing
            perm = rng.permutation(n)
            pairs = [(int(perm[2 * i]), int(perm[2 * i + 1])) for i in range(n // 2)]
        else:
            # Shifted pairing
            shift = 1 + (pass_num // 4) % max(1, n // 4)
            pairs = [(i, (i + shift) % n) for i in range(n // 2)]
            # Remove duplicate pairs
            seen = set()
            unique_pairs = []
            for a, b in pairs:
                key = (min(a, b), max(a, b))
                if key not in seen:
                    seen.add(key)
                    unique_pairs.append((a, b))
            pairs = unique_pairs

        for (i, j) in pairs:
            s = h[i] + h[j]
            lo_t = max(0.0, s - 1.0)
            hi_t = min(1.0, s)
            if lo_t >= hi_t - 1e-14:
                continue

            sl_i = corr_slope_at_i_vec(h, i, n)
            sl_j = corr_slope_at_i_vec(h, j, n)
            diff_sl = sl_i - sl_j

            intercepts = corr - h[i] * diff_sl
            t_opt = solve_1d_convex(diff_sl, intercepts, lo_t, hi_t)

            if abs(t_opt - h[i]) > 1e-12:
                delta = t_opt - h[i]
                corr += delta * diff_sl
                h[i] = t_opt
                h[j] = s - t_opt

        h = project_feasible(h, n)
        corr = compute_corr(h)
        c5_new = float(np.max(corr) * 2.0 / n)
        if c5_new < best_c5:
            best_c5 = c5_new

    return h, best_c5

    best_T = T

    while time.time() - start < time_limit and remaining() > 5:
        ones_list = np.where(h == 1)[0]
        zeros_list = np.where(h == 0)[0]
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break

        # Find worst shifts
        k_max = int(np.argmax(corr))
        offset = k_max - (n - 1)

        # Precompute good candidates for this shift
        if offset >= 0 and offset < n:
            valid = ones_list[ones_list + offset < n]
            good_i = valid[h[valid + offset] == 0]
        elif offset < 0 and -offset < n:
            m = -offset
            valid = ones_list[ones_list >= m]
            good_i = valid[h[valid - m] == 0]
        else:
            good_i = ones_list

        found = False
        n_try = 400

        for _ in range(n_try):
            if len(good_i) > 5 and rng.rand() < 0.6:
                i = int(good_i[rng.randint(len(good_i))])
            else:
                i = int(ones_list[rng.randint(len(ones_list))])
            j = int(zeros_list[rng.randint(len(zeros_list))])
            if i == j:
                continue

            sl_i = corr_slope_at_i_vec(h, i, n)
            sl_j = corr_slope_at_i_vec(h, j, n)
            delta = sl_j - sl_i
            new_max = float(np.max(corr + delta))

            if new_max <= cur_max + 1e-14:
                h[i] = 0.0
                h[j] = 1.0
                corr += delta
                cur_max = new_max
                if cur_max * dx_val < best_c5:
                    best_c5 = cur_max * dx_val
                    best_h = h.copy()
                found = True
                break
            else:
                if rng.rand() < np.exp(-(new_max - cur_max) / max(T, 0.01)):
                    h[i] = 0.0
                    h[j] = 1.0
                    corr += delta
                    cur_max = new_max
                    found = True
                    break

        if found:
            best_T = T
            T = max(0.005, T * 0.9997)
        else:
            T = max(0.005, T * 0.995)
            if T <= 0.006:
                T = best_T * 0.5

    return best_h, best_c5


def binary_double_swap(h, n, time_limit=60, rng=None):
    """Binary local search with double swaps for escaping local minima."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h))
    target = n // 2

    if n_ones > target:
        ones_idx = np.where(h == 1)[0]
        to_remove = rng.choice(ones_idx, n_ones - target, replace=False)
        h[to_remove] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h == 0)[0]
        to_add = rng.choice(zeros_idx, target - n_ones, replace=False)
        h[to_add] = 1.0

    dx_val = 2.0 / n
    corr = compute_corr(h).copy()
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx_val
    best_h = h.copy()
    start = time.time()

    while time.time() - start < time_limit and remaining() > 5:
        ones_list = np.where(h == 1)[0]
        zeros_list = np.where(h == 0)[0]
        if len(ones_list) < 2 or len(zeros_list) < 2:
            break

        k_max = int(np.argmax(corr))
        offset = k_max - (n - 1)

        # Try double swap: move two 1s to two 0 positions
        # Bias toward worst shift
        if offset >= 0 and offset < n:
            candidates1 = [i for i in ones_list if i + offset < n and h[i + offset] == 0]
        elif offset < 0:
            m = -offset
            candidates1 = [i for i in ones_list if i - m >= 0 and h[i - m] == 0]
        else:
            candidates1 = list(ones_list)

        found = False
        for _ in range(200):
            if len(candidates1) > 3 and rng.rand() < 0.5:
                i1 = candidates1[rng.randint(len(candidates1))]
            else:
                i1 = int(ones_list[rng.randint(len(ones_list))])
            i2 = int(ones_list[rng.randint(len(ones_list))])
            if i1 == i2:
                continue
            j1 = int(zeros_list[rng.randint(len(zeros_list))])
            j2 = int(zeros_list[rng.randint(len(zeros_list))])
            if j1 == j2 or j1 == i1 or j2 == i1 or j1 == i2 or j2 == i2:
                continue

            sl_i1 = corr_slope_at_i_vec(h, i1, n)
            sl_i2 = corr_slope_at_i_vec(h, i2, n)
            sl_j1 = corr_slope_at_i_vec(h, j1, n)
            sl_j2 = corr_slope_at_i_vec(h, j2, n)
            delta = -sl_i1 - sl_i2 + sl_j1 + sl_j2
            new_max = float(np.max(corr + delta))

            if new_max <= cur_max + 1e-14:
                h[i1] = 0.0
                h[i2] = 0.0
                h[j1] = 1.0
                h[j2] = 1.0
                corr += delta
                cur_max = new_max
                if cur_max * dx_val < best_c5:
                    best_c5 = cur_max * dx_val
                    best_h = h.copy()
                found = True
                break

        if not found:
            # Try single swap as fallback
            for _ in range(100):
                i = int(ones_list[rng.randint(len(ones_list))])
                j = int(zeros_list[rng.randint(len(zeros_list))])
                if i == j:
                    continue
                sl_i = corr_slope_at_i_vec(h, i, n)
                sl_j = corr_slope_at_i_vec(h, j, n)
                delta = sl_j - sl_i
                new_max = float(np.max(corr + delta))
                if new_max <= cur_max + 1e-14:
                    h[i] = 0.0
                    h[j] = 1.0
                    corr += delta
                    cur_max = new_max
                    if cur_max * dx_val < best_c5:
                        best_c5 = cur_max * dx_val
                        best_h = h.copy()
                    found = True
                    break

    return best_h, best_c5


def greedy_binary_init(n, rng=None):
    """Create a greedy binary initial solution."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = np.zeros(n)
    target = n // 2
    # Start with random placement
    ones = rng.choice(n, target, replace=False)
    h[ones] = 1.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 312
        parent_h = np.zeros(parent_n)
        parent_h[::2] = 1.0
        if int(np.sum(parent_h)) != parent_n // 2:
            parent_h = np.zeros(parent_n)
            parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Quick SLSQP refinement
    for beta in [500, 2000, 8000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Pair coordinate descent
    if remaining() > 20:
        h_pc, c5_pc = pair_cd(h, n, max_passes=200, time_limit=70, rng=rng)
        if c5_pc < best_c5:
            best_c5 = c5_pc
            best_h = h_pc.copy()
            h = h_pc.copy()

    # Phase 3: SLSQP after CD
    for beta in [5000, 20000]:
        if remaining() < 6:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=350)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Main binary search with SA (most important)
    if remaining() > 30:
        h_bin = (best_h > 0.5).astype(float)
        h_b, c5_b = fast_binary_search(h_bin, n,
                                       time_limit=min(180, max(30, remaining() // 3)), rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h = h_b.copy()

    # Phase 5: Double swap search
    if remaining() > 30:
        h_bin = (best_h > 0.5).astype(float)
        h_b, c5_b = binary_double_swap(h_bin, n,
                                        time_limit=min(80, max(20, remaining() // 4)), rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h = h_b.copy()

    # Phase 6: Try different n values
    n_candidates = sorted(set([
        n + 8, n + 16, n + 32, n + 64,
        n - 8, n - 16,
        256, 320, 384, 448, 512, 576, 640
    ]))
    n_candidates = [nc for nc in n_candidates if 64 <= nc <= 800 and nc % 2 == 0 and nc != n]

    for n_t in n_candidates[:6]:
        if remaining() < 30:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)

        # Quick SLSQP
        h_opt = slsqp_smooth(h_feas, n_t, 3000, maxiter=200)

        # Pair CD
        if remaining() > 15:
            h_opt, _ = pair_cd(h_opt, n_t, max_passes=100, time_limit=20, rng=rng)

        # Binary search
        if remaining() > 15:
            h_bin = (h_opt > 0.5).astype(float)
            h_opt, _ = fast_binary_search(h_bin, n_t, time_limit=min(25, max(10, remaining() // 3)), rng=rng)

        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 7: More binary search on best
    if remaining() > 30:
        n = best_n
        h_bin = (best_h > 0.5).astype(float)
        h_b, c5_b = fast_binary_search(h_bin, n,
                                       time_limit=min(120, max(20, remaining() // 3)), rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h = h_b.copy()

    # Phase 8: Final continuous polish from best binary
    n = best_n
    h = best_h.copy()
    for beta in [10000, 30000, 80000]:
        if remaining() < 5:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 9: Final pair CD
    if remaining() > 10:
        h_pc, c5_pc = pair_cd(h, n, max_passes=150,
                               time_limit=min(30, max(5, remaining() - 5)), rng=rng)
        if c5_pc < best_c5:
            best_c5 = c5_pc
            best_h = h_pc.copy()

    # Phase 10: One more binary pass if time
    if remaining() > 15:
        h_bin = (best_h > 0.5).astype(float)
        h_b, c5_b = fast_binary_search(h_bin, n,
                                       time_limit=min(40, max(10, remaining() // 2)), rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Final SLSQP polish
    if remaining() > 8:
        h_new = slsqp_smooth(best_h, best_n, 50000, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Validate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
