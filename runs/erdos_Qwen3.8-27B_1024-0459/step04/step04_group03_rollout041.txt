def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):

            break

import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        for j in range(n):
            nv = h[j] + diff
            if 0.0 <= nv <= 1.0:
                h[j] = nv
                break
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def compute_smoothed_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    f = corr * dx
    f_max = f.max()
    ex = np.exp(beta * (f - f_max))
    w_sum = ex.sum()
    if w_sum < 1e-300:
        ex = np.ones_like(f)
        w_sum = len(f)
    w = ex / w_sum
    grad = np.zeros(n)
    sig = np.where(w > 1e-12)[0]
    for idx in sig:
        wi = w[idx]
        k = idx - (n - 1)
        if k == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += wi * (1.0 - h[k:])
            grad[k:] += wi * (-h[:n-k])
        else:
            ka = -k
            grad[:n-ka] += wi * (-h[ka:])
            grad[ka:] += wi * (1.0 - h[:n-ka])
    return grad

def projected_grad_descent(h, n, beta, max_iter=200):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.5 / max(1.0, beta ** 0.3)
    momentum = 0.7
    v = np.zeros(n)
    
    for it in range(max_iter):
        if remaining_time() < 2:
            break
        grad = compute_smoothed_grad(h, n, beta)
        v = momentum * v - step * grad
        h_trial = project_feasible(h + v, n)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5 - 1e-15:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            h = h_trial
        else:
            # Backtrack
            step *= 0.5
            momentum = 0.5 * momentum
            if step < 1e-7:
                step = 0.5 / max(1.0, beta ** 0.3)
                momentum = 0.7
            h_trial = project_feasible(h - step * grad, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-15:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial
    return best_h, best_c5

def _mm_ineq(x):
    n_grp = x[-1]
    n = int(n_grp)
    dx = 2.0 / n
    h = x[:n]
    t = x[n]
    corr = np.correlate(h, 1.0 - h, mode='full')
    active = np.argsort(corr)[::-1][:5]
    return np.min(t - dx * corr[active])

def _mm_eq(x):
    n = int(x[-1])
    return np.sum(x[:n]) - n / 2.0

def slsqp_minimax(h_init, n, max_rounds=20, max_slsqp_iter=300):
    dx = 2.0 / n
    h = h_init.copy()
    
    for rnd in range(max_rounds):

        def mm_obj(x):
            return x[n]
        
        def mm_ineq_local(x):
            hh = x[:n]
            tt = x[n]
            c = np.correlate(hh, 1.0 - hh, mode='full')
            return np.min(tt - dx * c[top_k])
        
        def mm_eq_local(x):
            return np.sum(x[:n]) - n / 2.0

            {'type': 'ineq', 'fun': mm_ineq_local},
            {'type': 'eq', 'fun': mm_eq_local}
        ]

def _slsqp_smooth_obj(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def _slsqp_eq(x, n):
    return np.sum(x) - n / 2.0

    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(lambda x: _slsqp_smooth_obj(x, n, beta), h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()
