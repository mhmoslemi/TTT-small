def project_feasible(h, n):

            tm[i] = tm[i >> 1]
    return tm

def structured_init(n, rng, pattern_type=0):
    """Generate structured initial binary sequence."""
    h = np.zeros(n)
    target = n // 2
    if pattern_type == 0:
        # Thue-Morse
        h = thue_morse(n)
    elif pattern_type == 1:
        # Alternating blocks of varying size
        pos = 0
        block = 1
        val = 1
        while pos < n:
            end = min(pos + block, n)
            h[pos:end] = val
            pos = end
            block += rng.choice([1, 1, 2])
            val = 1 - val
    elif pattern_type == 2:
        # Grayscale gradient then threshold
        h = (np.arange(n) / n > 0.5).astype(float)
        # Shuffle blocks
        block_size = n // 8
        for _ in range(5):
            i = rng.randint(0, 7)
            j = rng.randint(0, 7)
            block_i = h[i*block_size:(i+1)*block_size]
            block_j = h[j*block_size:(j+1)*block_size]
            h[i*block_size:(i+1)*block_size] = block_j
            h[j*block_size:(j+1)*block_size] = block_i
    elif pattern_type == 3:
        # Random with long-range correlation
        h = (rng.rand(n) > 0.5).astype(float)
    elif pattern_type == 4:
        # Periodic with random period
        p = rng.randint(3, 12)
        base = (rng.rand(p) > 0.5).astype(float)
        while int(np.sum(base)) * (n // p) != target:
            # Adjust base to have right density
            base = (rng.rand(p) > 0.5).astype(float)
        h = np.tile(base, n // p + 1)[:n]
    # Fix to exact count
    h = make_binary(h, n, rng)
    return h

def sa_binary(h, n, time_limit=60, seed=42, use_double_swap=True):
    """Improved binary SA with single and double swap moves."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    
    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()
    
    T = 10.0
    alpha = 0.9990
    start = time.time()
    stall = 0
    max_stall = 8000
    
    while time.time() - start < time_limit and T > 0.05:
        if remaining() < 10:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
            
        if use_double_swap and rng.rand() < 0.3 and len(ones) >= 2 and len(zeros) >= 2:
            # Double swap: pick 2 ones and 2 zeros
            i1, i2 = rng.choice(len(ones), 2, replace=False)
            j1, j2 = rng.choice(len(zeros), 2, replace=False)
            p1, p2 = ones[i1], ones[i2]
            q1, q2 = zeros[j1], zeros[j2]
            h[p1] = 0; h[p2] = 0; h[q1] = 1; h[q2] = 1
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            delta = new_max - cur_max
            if delta < 0 or (T > 0.05 and rng.rand() < np.exp(-max(0, delta) / T)):
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                h[p1] = 1; h[p2] = 1; h[q1] = 0; h[q2] = 0
                stall += 1
        else:
            # Single swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0; h[j] = 1
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            delta = new_max - cur_max
            if delta < 0 or (T > 0.05 and rng.rand() < np.exp(-max(0, delta) / T)):
                cur_max = new_max
                if new_max < best_max:
                    best_max = new_max
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                h[i] = 1; h[j] = 0
                stall += 1
        T *= alpha
        if stall > max_stall:
            T = max(T, 3.0)
            stall = 0
    
    return best_h, best_max * dx_val

def local_search_single_swap(h, n, max_rounds=50):
    """Exhaustive single-swap local search."""
    h = h.copy()
    dx_val = 2.0 / n
    cur_c5 = compute_c5(h)
    
    for rnd in range(max_rounds):
        if remaining() < 5:
            break

        best_j = -1
        
        ones = np.where(h == 1)[0]

        # For each one and zero pair, compute the change in max corr
        # This is O(n^3) but for n=360-720 it's feasible with vectorization
        for i in ones:
            # Flipping h[i] from 1 to 0 changes corr
            # corr_new[k] = corr[k] + delta_i[k] where delta_i depends on position
            # For position i going 1→0:
            #   In corr, position i was contributing h[i]=1 to the "h" array
            #   After flip, it contributes 0. Also (1-h[i]) goes from 0 to 1.
            # corr[k] = sum_n h[n+k]*(1-h[n])
            # Remove contribution of h[i]: subtract (1-h[n]) for all n where n+k=i, i.e., n=i-k
            # Add contribution of (1-h[i])=1: add h[n] for all n where n=i (in the 1-h[n] position)
            
            # This is complex. Let me just compute directly for small n.
            pass
        
        # Direct approach: try all swaps, keep best
        # For n=720, n/2 * n/2 = 129600 swaps, each requiring O(n) correlate
        # That's 129600 * 720 * 720 ≈ 67 billion ops. Too slow.
        
        # Better: use the structure of the correlation update
        # When we swap h[i]=1→0 and h[j]=0→1:
        # The change in h is: h[i]-=1, h[j]+=1
        # Change in (1-h) is: (1-h)[i]+=1, (1-h)[j]-=1
        # corr[k] = sum_n h[n+k]*(1-h[n])
        # Δcorr[k] = sum_n Δh[n+k]*(1-h[n]) + sum_n h[n+k]*Δ(1-h)[n]
        
        # This is O(n) to compute for each (i,j) pair. Total: O(n^3)
        # For n=720: 720^3 ≈ 3.7e8. Feasible in ~5-10 seconds with numpy.
        
        # Let me use a vectorized approach
        # Δh has -1 at i, +1 at j
        # Δ(1-h) has +1 at i, -1 at j
        # Δcorr[k] = sum_n Δh[n+k]*(1-h[n]) + sum_n h[n+k]*Δ(1-h)[n]
        # First term: (Δh * (1-h)) correlated... it's like a convolution
        # Actually Δcorr[k] = corr(Δh, 1-h)[k] + corr(h, Δ(1-h))[k]
        
        # corr(Δh, 1-h): Δh is sparse (only 2 non-zero entries)
        # = (1-h)[i-k] - (1-h)[j-k] for valid indices... 
        # This is getting complicated. Let me just do it naively for n ≤ 400
        # and use a smarter approach for larger n.
        
        if n <= 400:
            # Naive: evaluate all swaps
            best_c5_local = cur_c5
            for i in ones:
                h[i] = 0
                for j in zeros:
                    h[j] = 1
                    c = float(np.max(compute_corr(h)) * dx_val)
                    if c < best_c5_local:
                        best_c5_local = c
                        best_i_try = i
                        best_j_try = j
                    h[j] = 0
                h[i] = 1
            
            if best_c5_local < cur_c5 - 1e-15:
                h[best_i_try] = 0
                h[best_j_try] = 1
                cur_c5 = best_c5_local
            else:
                break
        else:
            # For larger n, use random sampling of swaps
            tried = 0
            max_tries = 5000
            improved = False
            while tried < max_tries and remaining() > 2:
                i = ones[rng.randint(len(ones))] if 'rng' in dir() else ones[np.random.randint(len(ones))]
                j = zeros[np.random.randint(len(zeros))]
                h[i] = 0; h[j] = 1
                c = float(np.max(compute_corr(h)) * dx_val)
                if c < cur_c5 - 1e-15:
                    cur_c5 = c
                    improved = True
                else:
                    h[i] = 1; h[j] = 0
                tried += 1
            if not improved:
                break
    
    return h, cur_c5

def focused_local_search(h, n, max_iters=200):
    """Iterative improvement: try random swaps, keep if better."""

def smooth_obj(x, n, beta):
    dx_val = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val

def corr_grad_for_lag(x, n, lag):
    """Gradient of corr[lag] w.r.t. x, where lag is the index in full correlation."""
    g = np.zeros(n)
    off = lag - (n - 1)
    if off >= 0:
        npairs = n - off
        # corr[lag] = sum_{i=0}^{npairs-1} x[i+off] * (1-x[i])
        # dx[i+off]: contributes (1-x[i]) for i=0..npairs-1
        # dx[i]: contributes -x[i+off] for i=0..npairs-1
        g[off:off+npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[off:off+npairs]
    else:
        ao = -off
        npairs = n - ao
        g[:npairs] += (1.0 - x[ao:ao+npairs])
        g[ao:ao+npairs] -= x[:npairs]
    return g

def smooth_grad(x, n, beta):
    dx_val = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-12:
            g += w[k] * corr_grad_for_lag(x, n, k)
    return g * dx_val
