The Jacobian of the system (c_k(h) - t = 0 for k in A, sum(h) = n/2) with respect to (h, t) is:
[dc_1/dh  -1]
[dc_2/dh  -1]
...
[dc_m/dh  -1]
[1  ...  1   0]

where m = |A|.

If m + 1 <= n (which is typically true), this is an underdetermined system, and we can minimize t by choosing the right "free" variables.

This is getting complicated. Let me just focus on a practical approach that's likely to work:

1. **Improved SLSQP**: Use a better smooth objective and more iterations
2. **Better gradient**: Use the vectorized approach
3. **More aggressive multi-resolution**: Try n values that are common divisors/multiples
4. **Custom local search**: Instead of generic SA, use a "correlation-focused" local search that targets the specific over-correlated lags

Let me write a cleaner, more focused implementation:

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    # Bisection for exact projection
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine correction
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        free = (result > 1e-12) & (result < 1.0 - 1e-12)
        if np.sum(free) > 0:
            result[free] += diff / np.sum(free)
            result = np.clip(result, 0.0, 1.0)
    return result

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)

def smooth_objective(x, n, beta):
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(shifted)
    return (m + np.log(np.sum(ex)) / beta) * dx

def smooth_gradient(x, n, beta):
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(shifted)
    w = ex / np.sum(ex)
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - x
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-15:
            continue
        grad[k:] += wk * one_minus_h[:n - k]
        grad[:n - k] -= wk * x[k:]
    for l in range(1, n):
        wl = w[n - 1 - l]
        if wl < 1e-15:
            continue
        grad[l:] += wl * one_minus_h[:n - l]
        grad[:n - l] -= wl * x[l:]
    
    grad *= dx
    return grad

def slsqp_solve(h, n, beta, maxiter=800):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_objective(x, n, beta)
    
    def grad(x):
        return smooth_gradient(x, n, beta)
    
    def cons_fun(x):
        return np.sum(x) - n / 2.0
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def projected_gradient_descent(h, n, beta=1000, max_iter=2000, momentum=0.9):
    """Momentum-based projected gradient on smoothed objective."""
    h = project_feasible(h, n).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Initial gradient
    grad = smooth_gradient(h, n, beta)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        return best_h, best_c5
    
    # LR: use 1/L where L is a rough Lipschitz estimate
    # For the smoothed objective with beta, L ~ beta * (2/n) * n = 2*beta... rough
    lr = 2.0 / (beta * 4.0)  # conservative
    vel = np.zeros(n)
    
    for it in range(max_iter):
        if remaining() < 3:
            break
        
        vel = momentum * vel - lr * grad
        h_new = project_feasible(h + vel, n)
        
        obj_new = smooth_objective(h_new, n, beta)
        grad_new = smooth_gradient(h_new, n, beta)
        
        # Check improvement on smoothed objective
        # (We track exact C5 separately)
        
        h = h_new
        grad = grad_new
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Adaptive lr
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
    
    return best_h, best_c5

def correlation_equalize(h, n, top_k=3, max_iters=500):
    """Equalize top-k correlations using gradient steps."""
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:top_k]
        c_top = corr[top_idx[0]]
        c_min_top = corr[top_idx[-1]]
        
        if c_top - c_min_top < 1e-10:
            break
        
        # Direction: decrease max correlation, increase min of top-k
        g_max = np.zeros(n)
        g_min = np.zeros(n)
        
        # Gradient of max correlation
        km = top_idx[0]
        off = km - (n - 1)
        if off >= 0:
            np_ = n - off
            g_max[off:off + np_] += (1.0 - h[:np_])
            g_max[:np_] -= h[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g_max[:np_] += (1.0 - h[ao:ao + np_])
            g_max[ao:ao + np_] -= h[:np_]
        
        # Gradient of min top correlation
        kb = top_idx[-1]
        off = kb - (n - 1)
        if off >= 0:
            np_ = n - off
            g_min[off:off + np_] += (1.0 - h[:np_])
            g_min[:np_] -= h[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g_min[:np_] += (1.0 - h[ao:ao + np_])
            g_min[ao:ao + np_] -= h[:np_]
        
        d = g_min - g_max  # direction that increases min and decreases max
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        
        improved = False
        for step in [0.05, 0.02, 0.01, 0.08, 0.005, 0.03, 0.1, 0.015, 0.04]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        
        if not improved:
            # Try smaller steps
            for step in [0.001, 0.002, 0.003]:
                ht = project_feasible(h + step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
            if not improved:
                break
    
    return h, best_c5
