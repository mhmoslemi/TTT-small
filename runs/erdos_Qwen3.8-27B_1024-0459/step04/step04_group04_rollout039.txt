    # add_impact[p, k] = change in corr[k] when h[p] goes 0->1

    # For each p and offset:
    # valid_first: max(0,-offset) <= p <= min(n-1, n-1-offset)
    # valid_second: max(0,offset) <= p <= min(n-1,n-1+offset)

    P = np.arange(n)[:, None]  # (n, 1)
    O = offsets[None, :]  # (1, 2n-1)
    
    valid_first = (P >= np.maximum(0, -O)) & (P <= np.minimum(n-1, n-1-O))
    valid_second = (P >= np.maximum(0, O)) & (P <= np.minimum(n-1, n-1+O))
    
    # For first factor: 1-h[p+offset]
    # p+offset: need to index h at position p+offset (if valid)
    # For second factor: h[p-offset]
    
    # Create index arrays
    p_plus_o = P + O  # (n, 2n-1)
    p_minus_o = P - O  # (n, 2n-1)
    
    # Clip for safe indexing
    p_plus_o_clip = np.clip(p_plus_o, 0, n-1)
    p_minus_o_clip = np.clip(p_minus_o, 0, n-1)
    
    one_minus_h_po = (1 - h[p_plus_o_clip])  # (n, 2n-1)
    h_pmo = h[p_minus_o_clip]  # (n, 2n-1)
    
    add_impact = valid_first * one_minus_h_po - valid_second * h_pmo  # (n, 2n-1)
    
    # Now greedy search
    h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for round in range(max_rounds):
        # Recompute corr and add_impact (since h changed)
        corr = np.correlate(h, 1-h, 'full')
        current_max = np.max(corr)
        
        # Recompute add_impact
        one_minus_h_po = (1 - h[p_plus_o_clip])
        h_pmo = h[p_minus_o_clip]
        add_impact = valid_first * one_minus_h_po - valid_second * h_pmo
        
        # For each j in zeros, find best i in ones
        best_delta = 0  # We want to minimize the new max
        best_swap = None
        best_new_max = current_max + 1  # current_max in units of dx
        
        for j in zeros:
            B_j = corr + add_impact[j]  # (2n-1,)
            # For each i in ones: new_max = max(B_j - add_impact[i])
            D = B_j[None, :] - add_impact[ones, :]  # (n_ones, 2n-1)
            new_maxes = D.max(axis=1)  # (n_ones,)
            idx = np.argmin(new_maxes)

                best_swap = (ones[idx], j)
        
        if best_swap is None or best_new_max >= current_max:

        i, j = best_swap
        h[i], h[j] = 0, 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    
    return h, np.max(np.correlate(h, 1-h, 'full'))

import numpy as np

_START = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _START) - 5.0)


def make_binary_from_continuous(h, n, rng):
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        to_flip = n_ones - target
        h_bin[idx[rng.choice(len(idx), to_flip, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        to_flip = target - n_ones
        h_bin[idx[rng.choice(len(idx), to_flip, replace=False)]] = 1
    return h_bin


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def precompute_masks(n):
    P = np.arange(n)[:, None].astype(float)
    offsets = np.arange(-(n - 1), n, dtype=float)[None, :]
    valid_first = (P >= np.maximum(0.0, -offsets)) & (P <= np.minimum(n - 1.0, n - 1.0 - offsets))
    valid_second = (P >= np.maximum(0.0, offsets)) & (P <= np.minimum(n - 1.0, n - 1.0 + offsets))
    p_plus_o = P + offsets
    p_minus_o = P - offsets
    p_plus_o_clip = np.clip(p_plus_o, 0, n - 1).astype(np.intp)
    p_minus_o_clip = np.clip(p_minus_o, 0, n - 1).astype(np.intp)
    return valid_first, valid_second, p_plus_o_clip, p_minus_o_clip


def compute_add_impact(h, valid_first, valid_second, p_plus_o_clip, p_minus_o_clip):
    one_minus_h_po = 1.0 - h[p_plus_o_clip]
    h_pmo = h[p_minus_o_clip]
    return valid_first * one_minus_h_po - valid_second * h_pmo

            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
                best_n = n_t

    # Final: one more greedy pass on best
    if remaining_time() > 10:
        h_bin = make_binary_from_continuous(best_h, best_n, rng)
        h_res, c5_res = greedy_local_search(h_bin, best_n, time_limit=min(60, remaining_time() * 0.6), seed=seed + 4242)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

    # Final validation
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
