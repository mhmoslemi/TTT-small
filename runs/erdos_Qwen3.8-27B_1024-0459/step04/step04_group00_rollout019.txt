    m = np.max(corr)

    g = np.zeros(n)

        best_d = 0.0

def binary_sa(h_init, n, time_limit=30, rng=None):
    """Simulated annealing on binary vectors."""
    if rng is None:
        rng = np.random.RandomState(99)
    h = (h_init > 0.5).astype(float).copy()

    n_ones = int(np.sum(h))

            ones_idx = np.where(h == 1)[0]
            rem = rng.choice(ones_idx, n_ones - n_half, replace=False)
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            add = rng.choice(zeros_idx, n_half - n_ones, replace=False)
            h[add] = 1
    dx_v = 2.0 / n
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    T0 = 1.0
    T1 = 0.005
    n_est = int(time_limit * 800)
    for gen in range(n_est):
        if time.time() - start > time_limit or remaining() < 5:
            break
        T = T0 * (T1 / T0) ** (gen / max(n_est, 1))
        if T < 0.001:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0
        h[j] = 1
        nc5 = compute_c5(h)
        delta = nc5 - cur_c5
        if delta <= 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = nc5
