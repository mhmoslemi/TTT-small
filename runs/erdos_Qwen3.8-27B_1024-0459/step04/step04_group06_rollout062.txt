            grad[offset:offset+nv] += w * (1.0 - h[:nv])
            grad[:nv] -= w * h[offset:offset+nv]
        else:

            nv = n - m
            if nv <= 0:
                continue
            grad[:nv] += w * (1.0 - h[m:m+nv])
            grad[m:m+nv] -= w * h[:nv]
    return grad, float(np.max(corr)) * (2.0/n)

def obj_lse(x):

    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(_G_BETA * shifted, -600, 600))
    return (m + np.log(np.sum(exps)) / _G_BETA) * dx

def obj_lse_grad(x):

    dx = _G_DX

    shifted = corr - m

        offset = int(j) - (n - 1)
        if offset >= 0:
            nv = n - offset
            if nv <= 0:

            if nv <= 0:
                continue
            grad[:nv] += w * (1.0 - x[m_off:m_off+nv])
            grad[m_off:m_off+nv] -= w * x[:nv]

def nesterov_opt(h, n, max_iter=300):
    """Nesterov-accelerated projected subgradient with full active set."""
    c5_best = compute_c5(h)
    h_best = h.copy()
    x = h.copy()
    x_prev = h.copy()
    
    for t in range(max_iter):
        if remaining_time() < 5.0:
            break
        
        # Nesterov extrapolation
        beta_t = t / (t + 2.0 + np.sqrt(t*t + 4.0*t + 4.0))  # approximately (t-1)/(t+2) for t>=1
        beta_t = min(t / (t + 2.0), 1.0) if t > 0 else 0.0
        y = x + beta_t * (x - x_prev)
        y = project_feasible(y, n)
        
        # Compute subgradient at y
        corr = np.correlate(y, 1.0 - y, mode='full')
        c5_y = np.max(corr) * (2.0/n)
        
        # Active set: all shifts within 90% of max
        threshold = 0.90 * np.max(corr)
        active = np.where(corr >= threshold)[0]
        
        grad = subgrad_full(y, n, active)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        
        # Polyak step size
        alpha = (c5_y - c5_best) / (2.0 * gn * gn + 1e-10)
        alpha = np.clip(alpha, 0.001, 0.3)

        x_new = project_feasible(x_new, n)

        x_prev = x
        x = x_new
        
        if c5_new < c5_best - 1e-14:
            c5_best = c5_new
            h_best = x_new.copy()
    
    return h_best, c5_best

                contrib = h_best[:nv] * (1.0 - h_best[k:k + nv])
                top_i = np.argsort(contrib)[-5:]
                for i in top_i:
                    for delta in [0.08, -0.08, 0.04, -0.04, 0.02, -0.02]:
                        h_t = h_best.copy()
                        h_t[i] = np.clip(h_t[i] + delta, 0.0, 1.0)
                        h_t = project_feasible(h_t, n)
                        c5_t = compute_c5(h_t)
                        if c5_t < c5_best + best_improve:
                            best_improve = c5_t - c5_best
                            best_trial = h_t
            else:
                m = -k
                nv = n - m
                if nv <= 1:
                    continue
                contrib = h_best[m:m + nv] * (1.0 - h_best[:nv])
                top_i = np.argsort(contrib)[-5:]
                for i in top_i:
                    for delta in [0.08, -0.08, 0.04, -0.04, 0.02, -0.02]:
                        h_t = h_best.copy()
                        h_t[m + i] = np.clip(h_t[m + i] + delta, 0.0, 1.0)
                        h_t = project_feasible(h_t, n)
                        c5_t = compute_c5(h_t)
                        if c5_t < c5_best + best_improve:
                            best_improve = c5_t - c5_best
                            best_trial = h_t
        if best_trial is not None and best_improve < -1e-14:
            h_best = best_trial
            c5_best = compute_c5(h_best)
        else:
            # Try random perturbation to escape
            h_t = h_best + 0.01 * rng_local.randn(n)
            h_t = project_feasible(h_t, n)
            c5_t = compute_c5(h_t)
            if c5_t < c5_best - 1e-14:
                h_best = h_t
                c5_best = c5_t
            else:
                break
    return h_best, c5_best


def momentum_subgrad(h, n, max_iter=300, frac=0.90):
    c5_best = compute_c5(h)
    h_best = h.copy()
    x = h.copy()
    v = np.zeros(n)
    step = 0.15
    patience = 0
    for t in range(max_iter):
        if remaining_time() < 4.0:
            break
        corr = np.correlate(x, 1.0 - x, mode='full')
        active = get_active_idx(corr, frac)
        grad = subgrad_full(x, n, active)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        gnorm = grad
        alpha = step
        improved = False
        for _ in range(12):
            v = 0.85 * v - alpha * gnorm
            x_trial = x + v
            x_trial = project_feasible(x_trial, n)
            c5_t = compute_c5(x_trial)
            if c5_t < c5_best - 1e-14:
                c5_best = c5_t
                h_best = x_trial.copy()
                x = x_trial.copy()
                improved = True
                break
            alpha *= 0.5
        if improved:
            patience = 0
            step = min(alpha * 1.3, 0.3)
        else:
            patience += 1
            step *= 0.65
            v *= 0.4
            if patience > 20:
                break
    return h_best, c5_best


def perturb_and_optimize(h_best, n_best, rng, num_pert=10):
    c5_best = compute_c5(h_best)
    global _G_N, _G_TGT, _G_DX
    _G_N = n_best
    _G_TGT = n_best / 2.0
    _G_DX = 2.0 / n_best
    bounds = [(0.0, 1.0)] * n_best
    cons = [{'type': 'eq', 'fun': con_sum}]
    for p in range(num_pert):
        if remaining_time() < 15.0:
            break
        h_t = h_best.copy()
        mode = p % 5
        if mode == 0:
            start = rng.randint(0, max(1, n_best - 20))
            L = rng.randint(6, min(25, n_best // 5))
            h_t[start:start + L] = 1.0 - h_t[start:start + L]
        elif mode == 1:
            idx = rng.choice(n_best, size=max(5, n_best // 8), replace=False)
            h_t[idx] = 1.0 - h_t[idx]
        elif mode == 2:
            h_t += 0.04 * rng.randn(n_best)
        elif mode == 3:
            f = rng.randint(1, max(2, n_best // 8))
            ph = rng.rand() * 2 * np.pi
            t_arr = np.arange(n_best) / float(n_best)
            h_t += 0.03 * np.sin(2.0 * np.pi * f * t_arr + ph)
        else:
            s1 = rng.randint(0, n_best // 3)
            s2 = s1 + rng.randint(5, n_best - n_best // 3 - 5)
            L = min(rng.randint(4, 12), n_best - s2)
            tmp = h_t[s1:s1 + L].copy()
            h_t[s1:s1 + L] = h_t[s2:s2 + L]
            h_t[s2:s2 + L] = tmp
        h_t = project_feasible(h_t, n_best)
        # Quick momentum
        h_t, c5_t = momentum_subgrad(h_t, n_best, max_iter=60, frac=0.93)
        # SLSQP from perturbed
        _G_BETA = 1000.0
        try:
            res = minimize(obj_lse, h_t, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-14, 'iprint': 0})
            h_new = project_feasible(res.x, n_best)
            c5_new = compute_c5(h_new)
            if c5_new < c5_t:
                c5_t = c5_new
                h_t = h_new
        except Exception:
            pass
        if c5_t < c5_best - 1e-12:
            c5_best = c5_t
            h_best = h_t.copy()
    return h_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_BETA, _G_TGT, _G_START, _G_BUDGET, _G_DX
    _G_START = time.time()
    _G_BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Phase 0: Initialize from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 256
        h0 = np.zeros(parent_n)
        for i in range(parent_n):
            h0[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
        best_h = project_feasible(h0, parent_n)
        best_c5 = compute_c5(best_h)
        best_n = parent_n

    _G_N = best_n
    _G_TGT = best_n / 2.0
    _G_DX = 2.0 / best_n
    print("Initial: n={}, C5={:.10f}".format(best_n, best_c5))

    # Phase 1: Nesterov subgradient (full active set)
    print("Phase 1: Nesterov subgradient")
    best_h, best_c5 = nesterov_subgrad(best_h, best_n, max_iter=400, frac=0.92)
    print("  C5={:.10f}".format(best_c5))

    # Phase 2: SLSQP with increasing beta
    print("Phase 2: SLSQP")
    best_h, best_c5 = slsqp_polish(best_h, best_n, [200, 500, 1500, 4000, 10000])
    print("  C5={:.10f}".format(best_c5))

    # Phase 3: Local search targeting active shifts
    print("Phase 3: Local search")
    best_h, best_c5 = local_search(best_h, best_n, max_rounds=60)
    print("  C5={:.10f}".format(best_c5))

    # Phase 4: Momentum subgradient + SLSQP
    if remaining_time() > 60.0:
        print("Phase 4: Momentum + SLSQP round 2")
        best_h, best_c5 = momentum_subgrad(best_h, best_n, max_iter=250, frac=0.90)
        best_h, best_c5 = slsqp_polish(best_h, best_n, [3000, 8000, 20000, 50000])
        print("  C5={:.10f}".format(best_c5))

    # Phase 5: Perturbation exploration
    if remaining_time() > 80.0:
        print("Phase 5: Perturbation")
        best_h, best_c5 = perturb_and_optimize(best_h, best_n, rng, num_pert=8)
        print("  C5={:.10f}".format(best_c5))

    # Phase 6: Higher resolution
    if remaining_time() > 150.0:
        print("Phase 6: Higher resolution")
        for n_try in [320, 384, 512]:
            if remaining_time() < 50.0:
                break
            h_int = project_feasible(interp_to(best_h, best_n, n_try), n_try)
            _G_N = n_try
            _G_TGT = n_try / 2.0
            _G_DX = 2.0 / n_try
            c5_h = compute_c5(h_int)
            h_int, c5_h = nesterov_subgrad(h_int, n_try, max_iter=120, frac=0.93)
            h_int, c5_h = slsqp_polish(h_int, n_try, [500, 2000, 8000])
            h_int, c5_h = local_search(h_int, n_try, max_rounds=30)
            if c5_h < best_c5 - 1e-12:
                best_c5 = c5_h
                best_h = h_int.copy()
                best_n = n_try
                print("  n={}: C5={:.10f} (NEW BEST)".format(n_try, best_c5))
            else:
                print("  n={}: C5={:.10f}".format(n_try, c5_h))

    # Phase 7: Final polish at best resolution
    if remaining_time() > 20.0:
        print("Phase 7: Final polish")
        _G_N = best_n
        _G_TGT = best_n / 2.0
        _G_DX = 2.0 / best_n
        best_h, best_c5 = nesterov_subgrad(best_h, best_n, max_iter=200, frac=0.95)
        best_h, best_c5 = slsqp_polish(best_h, best_n, [10000, 30000, 80000])
        best_h, best_c5 = local_search(best_h, best_n, max_rounds=40)
        print("  C5={:.10f}".format(best_c5))

    # Phase 8: One more Nesterov + tiny SLSQP
    if remaining_time() > 10.0:
        best_h, best_c5 = nesterov_subgrad(best_h, best_n, max_iter=100, frac=0.96)
        _G_N = best_n
        _G_TGT = best_n / 2.0
        _G_DX = 2.0 / best_n
        _G_BETA = 50000.0
        bounds = [(0.0, 1.0)] * best_n
        cons = [{'type': 'eq', 'fun': con_sum}]
        try:
            res = minimize(obj_lse, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-15, 'iprint': 0})
            h_new = project_feasible(res.x, best_n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            final_c5 = min(final_c5, eval_c5)
    except Exception:
        pass

    print("FINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
