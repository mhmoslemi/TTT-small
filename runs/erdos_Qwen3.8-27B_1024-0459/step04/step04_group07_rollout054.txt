dc[m]/dh[j] = (1-h[j-m+n-1]) - h[j+m-n+1]  (with 0 for out-of-bounds)

Now for the log-sum-exp objective:
L(h) = (1/β) * log(sum_{m=0}^{2n-2} exp(β * c[m]))

dL/dh[j] = (1/sum_m exp(β*c[m])) * sum_m exp(β*c[m]) * dc[m]/dh[j]
= sum_m w[m] * dc[m]/dh[j]

where w[m] = exp(β*c[m]) / sum_m exp(β*c[m])

This is a weighted average of the partial gradients at each lag.

For Frank-Wolfe, I need to minimize dL/dh[j] * h[j] over the constraint set. The linear oracle gives:
h* = 1 where dL/dh is smallest (n/2 of them), 0 elsewhere.

import numpy as np
from scipy.optimize import minimize
import time

_G_BETA = 1.0
_G_TGT = 0.0
_G_DX = 1.0
_G_PEN = 0.0
_G_FOCUS = -1
_G_START = 0.0
_G_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c else 0.0
    return h

def make_rand_bin(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def make_sine(n, f1, f2, phase):
    t = np.arange(n) / n
    h = 0.5 + 0.35 * np.sin(2.0 * np.pi * f1 * t + phase) + 0.2 * np.sin(2.0 * np.pi * f2 * t + 1.3)
    return np.clip(h, 0.0, 1.0)

def obj_lob(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    b = _G_BETA
    s = corr - m
    e = np.exp(np.clip(b * s, -400, 400))
    val = (m + np.log(np.sum(e)) / b) * _G_DX
    if _G_PEN > 0:
        val += _G_PEN * (np.sum(x) - _G_TGT) ** 2
    if _G_FOCUS >= 0 and _G_FOCUS < len(corr):
        val += 0.01 * corr[_G_FOCUS] * _G_DX
    return float(val)

def grad_fft(x, beta):
    n = len(x)
    dx = 2.0 / n
    N = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    m_max = np.max(corr)
    s = corr - m_max
    e = np.exp(np.clip(beta * s, -400, 400))
    w = e / np.sum(e)
    size = 1 << (3 * n - 2).bit_length()
    w_fft = np.fft.rfft(w, size)
    x_fft = np.fft.rfft(x, size)
    conv1 = np.fft.irfft(w_fft * x_fft, size)
    w_rev = w[::-1]
    wr_fft = np.fft.rfft(w_rev, size)
    conv2 = np.fft.irfft(wr_fft * x_fft, size)
    T1 = conv1[np.arange(n) + n - 1]
    T2 = conv2[np.arange(n) + n - 1]
    grad = (1.0 - T1 - T2) * dx
    return grad, w

def slsqp_run(h0, n, beta, maxiter):
    global _G_BETA, _G_TGT, _G_DX, _G_PEN, _G_FOCUS
    _G_BETA = float(beta)
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    _G_PEN = 0.0
    _G_FOCUS = -1
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    try:
        res = minimize(obj_lob, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def lbfgsb_run(h0, n, beta, maxiter, pen=100.0):

def frank_wolfe(h0, n, beta, max_iter, tlimit):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iter):
        if tlimit() < 3.0:
            break
        grad, w = grad_fft(h, beta)
        # Linear oracle: minimize grad^T x over {0<=x<=1, sum(x)=n/2}
        # Optimal: x* = 1 where grad is smallest (n/2 of them)
        order = np.argsort(grad)
        x_star = np.zeros(n)
        x_star[order[:n // 2]] = 1.0
        # Frank-Wolfe step
        t_fw = 2.0 / (it + 2)
        h_new = (1.0 - t_fw) * h + t_fw * x_star
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        # Occasionally recompute with SLSQP polish
        if it % 50 == 49 and tlimit() > 10:
            local_n = n
            local_h = slsqp_run(h, local_n, beta, 100)
            local_c5 = compute_c5(local_h)
            if local_c5 < best_c5:
                best_c5 = local_c5
                best_h = local_h.copy()
                h = local_h
    return best_h, best_c5

def coord_descent(h, n, max_sweeps, rng, tlimit):

    h_best = h.copy()

        improved = False
        order = rng.permutation(n)
        for i in order:
            if tlimit() < 1.0:
                break
            v0 = h_best[i]
            for d in [0.15, -0.15, 0.08, -0.08, 0.25, -0.25, 0.04, -0.04, 0.02, -0.02]:
                trial = h_best.copy()
                trial[i] = np.clip(trial[i] + d, 0.0, 1.0)
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 1e-10:
                    c5 = c
                    h_best = trial.copy()
                    improved = True
            for v in [0.0, 1.0]:
                if abs(v - v0) < 1e-9:
                    continue
                trial = h_best.copy()
                trial[i] = v
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 1e-10:
                    c5 = c
                    h_best = trial.copy()
                    improved = True
        if not improved and sweep >= 2:
            break
    return h_best, c5

        inits_512 = [(h_512, c5_512), (h_tm512, c5_tm512), (h_r512, c5_r512), (h_s512, c5_s512)]
        inits_512.sort(key=lambda x: x[1])
        print("  n={}: best init C5={:.8f}".format(N1, inits_512[0][1]))

        # Quick SLSQP on top 2 inits
        for ii in range(min(2, len(inits_512))):
            if tlimit() < 20:
                break
            h_try = inits_512[ii][0].copy()
            c_try = inits_512[ii][1]
            h_new = slsqp_run(h_try, N1, 200, 200)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-12:
                best_c5 = c_new
                best_h = h_new.copy()
                best_n = N1
                print("  SLSQP init {}: C5={:.8f}".format(ii, best_c5))

    # Phase 3: Frank-Wolfe at n=512
    n_curr = best_n if best_n >= 512 else 512
    if best_n < 512:
        h_curr = project_feasible(interp_h(best_h, best_n, n_curr), n_curr)
    else:
        h_curr = best_h.copy()

    print("\n=== Phase 3: Frank-Wolfe n={} ===".format(n_curr))
    if tlimit() > 60:
        h_fw, c5_fw = frank_wolfe(h_curr, n_curr, beta=100.0, max_iter=500, tlimit=tlimit)
        if c5_fw < best_c5 - 1e-12:
            best_c5 = c5_fw
            best_h = h_fw.copy()
            best_n = n_curr
            print("  FW improved: C5={:.8f}".format(best_c5))
        else:
            print("  FW best: C5={:.8f} (best so far: {:.8f})".format(c5_fw, best_c5))

        # Second FW pass with higher beta
        if tlimit() > 30:
            h_curr2 = best_h if best_n == n_curr else h_fw
            h_fw2, c5_fw2 = frank_wolfe(h_curr2, n_curr, beta=300.0, max_iter=300, tlimit=tlimit)
            if c5_fw2 < best_c5 - 1e-12:
                best_c5 = c5_fw2
                best_h = h_fw2.copy()
                best_n = n_curr
                print("  FW2 improved: C5={:.8f}".format(best_c5))

    # Phase 4: SLSQP multi-beta refinement at n=512
    print("\n=== Phase 4: SLSQP refinement n={} ===".format(best_n))
    n_curr = best_n
    if tlimit() > 100:
        beta_schedule = [50, 150, 400, 800, 1500, 3000, 6000, 12000]
        h_cur = best_h.copy()
        c5_cur = best_c5
        for bv in beta_schedule:
            if tlimit() < 10:
                break
            h_new = slsqp_run(h_cur, n_curr, bv, 300)
            c_new = compute_c5(h_new)
            if c_new < c5_cur - 1e-13:
                c5_cur = c_new
                h_cur = h_new
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n_curr
        print("  After multi-beta: C5={:.8f}".format(best_c5))

    # Phase 5: Focused iteration
    if tlimit() > 50:
        print("\n=== Phase 5: Focused SLSQP ===")
        h_f, c5_f = focused_slsqp(best_h, best_n, rng, tlimit, max_rounds=25)
        if c5_f < best_c5 - 1e-12:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = best_n
        print("  After focused: C5={:.8f}".format(best_c5))

    # Phase 6: Multi-start perturbation
    print("\n=== Phase 6: Perturbation escapes ===")
    n_curr = best_n
    if tlimit() > 80:
        for attempt in range(10):
            if tlimit() < 12:
                break
            h_pert = best_h.copy()
            n_pert = rng.randint(5, 25)
            idxs = rng.choice(n_curr, size=n_pert, replace=False)
            for idx in idxs:
                h_pert[idx] = np.clip(h_pert[idx] + rng.randn() * 0.4, 0.0, 1.0)
            h_pert = project_feasible(h_pert, n_curr)
            c5_p = compute_c5(h_pert)

            if c5_p < best_c5 * 1.08:
                h_attempt = h_pert.copy()
                for bv in [100, 500, 2000]:
                    if tlimit() < 6:
                        break
                    h_at = slsqp_run(h_attempt, n_curr, bv, 200)
                    c_at = compute_c5(h_at)
                    if c_at < best_c5 - 1e-12:
                        best_c5 = c_at
                        best_h = h_at.copy()
                        best_n = n_curr
                        h_attempt = h_at
                        print("  Perturb {} improved: C5={:.8f}".format(attempt, best_c5))
                        break

    # Phase 7: Try n=1024
    N2 = 1024
    if tlimit() > 120:
        print("\n=== Phase 7: Try n={} ===".format(N2))
        h_1024 = project_feasible(interp_h(best_h, best_n, N2), N2)
        c5_1024 = compute_c5(h_1024)
        print("  Interpolated: C5={:.8f}".format(c5_1024))

        h_1024 = lbfgsb_run(h_1024, N2, 200, 300, pen=50.0)
        c5_1024 = compute_c5(h_1024)
        h_1024 = slsqp_run(h_1024, N2, 500, 200)
        c5_1024 = compute_c5(h_1024)

        h_1024, c5_1024 = frank_wolfe(h_1024, N2, 200, 200, tlimit)
        c5_1024 = compute_c5(h_1024)

        if c5_1024 < best_c5 - 1e-9:
            best_c5 = c5_1024
            best_h = h_1024.copy()
            best_n = N2
            print("  n={} improved: C5={:.8f}".format(N2, best_c5))
        else:
            print("  n={} best: C5={:.8f} (no improvement over {:.8f})".format(N2, c5_1024, best_c5))

    # Phase 8: Coordinate descent
    if tlimit() > 30:
        print("\n=== Phase 8: Coordinate descent n={} ===".format(best_n))
        best_h, best_c5 = coord_descent(best_h, best_n, max_sweeps=30, rng=rng, tlimit=tlimit)
        print("  After coord: C5={:.8f}".format(best_c5))

    # Phase 9: Final SLSQP polish
    if tlimit() > 15:
        print("\n=== Phase 9: Final polish ===")
        for bv in [3000, 8000, 20000]:
            if tlimit() < 5:
                break
            h_new = slsqp_run(best_h, best_n, bv, 400)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-13:
                best_c5 = c_new
                best_h = h_new
        print("  After polish: C5={:.8f}".format(best_c5))

    # Phase 10: Final focused + coord
    if tlimit() > 15:
        best_h, best_c5 = focused_slsqp(best_h, best_n, rng, tlimit, max_rounds=15)
        print("  Final focused: C5={:.8f}".format(best_c5))

    if tlimit() > 10:
        best_h, best_c5 = coord_descent(best_h, best_n, max_sweeps=20, rng=rng, tlimit=tlimit)
        print("  Final coord: C5={:.8f}".format(best_c5))

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    _G_PEN = 0.0
    _G_FOCUS = -1
    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
