import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 5.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def _obj_lse(x, n, beta, dx_val):
    corr = compute_corr(x)
    m = float(np.max(corr))
    shifted = corr - m
    cl = np.clip(beta * shifted, -50, 50)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val


def _obj_lse_grad(x, n, beta, dx_val):
    corr = compute_corr(x)
    m = float(np.max(corr))
    shifted = corr - m
    cl = np.clip(beta * shifted, -50, 50)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-13:
            off = k - (n - 1)
            if off >= 0:
                npairs = n - off
                g[:npairs] += w[k] * (1.0 - x[off:off + npairs])
                g[off:off + npairs] -= w[k] * x[:npairs]
            else:
                ao = -off
                npairs = n - ao
                g[ao:ao + npairs] += w[k] * (1.0 - x[:npairs])
                g[:npairs] -= w[k] * x[ao:ao + npairs]
    return g * dx_val


def _sum_con(x, target):
    return np.sum(x) - target


def _sum_con_jac(x):
    return np.ones(len(x))


def slsqp_run(h, n, beta, maxiter=400, ftol=1e-15):
    dx_val = 2.0 / n
    target = n / 2.0

    def obj(x):
        return _obj_lse(x, n, beta, dx_val)

    def obj_grad(x):
        return _obj_lse_grad(x, n, beta, dx_val)

    def sfun(x):
        return _sum_con(x, target)

    def sjac(x):
        return _sum_con_jac(x)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h

    best_h = h.copy()
    start = time.time()
    stall = 0

    while time.time() - start < time_limit and remaining() > 8:
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_swap = None
        best_new_max = cur_max

        # Try a random subset of swaps if too many
        if len(ones) * len(zeros) > 20000:
            swaps_to_try = []
            for _ in range(200):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                swaps_to_try.append((i, j))
        else:
            swaps_to_try = [(i, j) for i in ones for j in zeros]

        for i, j in swaps_to_try:
            h[i], h[j] = 0.0, 1.0
            new_corr = compute_corr(h)
            new_max = int(np.max(new_corr))
            h[i], h[j] = 1.0, 0.0
            if new_max < best_new_max:
                best_new_max = new_max
                best_swap = (i, j)

        if best_swap is not None:
            i, j = best_swap
            h[i], h[j] = 0.0, 1.0
            cur_max = best_new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
            if stall > 50:
                # Perturb: random 5 swaps
                for _ in range(5):
                    ones = list(np.where(h == 1)[0])
                    zeros = list(np.where(h == 0)[0])
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i], h[j] = 0.0, 1.0
                corr = compute_corr(h)
                cur_max = int(np.max(corr))
                stall = 0

    return best_h, best_max * dx_val


def multi_restart_binary(h, n, time_limit=120, seed=42):
    rng = np.random.RandomState(seed)
    best_h = (h > 0.5).astype(float).copy()
    target = n // 2
    best_c5 = compute_c5(best_h)

    n_restarts = max(1, min(5, int(time_limit / 30)))
    t_each = time_limit / n_restarts

    for r in range(n_restarts):
        if remaining() < 10:
            break
        if r == 0:
            h_init = (h > 0.5).astype(float)
        else:
            # Perturb best
            h_init = best_h.copy()
            n_perturb = max(3, n // 30)
            idx_ones = np.where(h_init == 1)[0]
            idx_zeros = np.where(h_init == 0)[0]
            for _ in range(n_perturb):
                i = idx_ones[rng.randint(len(idx_ones))]
                j = idx_zeros[rng.randint(len(idx_zeros))]
                h_init[i], h_init[j] = 0.0, 1.0
            corr = compute_corr(h_init)
            # Fix count
            n_1 = int(np.sum(h_init))
            if n_1 > target:
                idx = np.where(h_init == 1)[0]
                h_init[idx[rng.choice(len(idx), n_1 - target, replace=False)]] = 0
            elif n_1 < target:
                idx = np.where(h_init == 0)[0]
                h_init[idx[:target - n_1]] = 1

        h_res, c5_res = greedy_binary_search(h_init, n, time_limit=t_each, seed=seed + r * 7919)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

    return best_h, best_c5


def targeted_lag_descent(h, n, time_limit=60):
    """Aggressively target the specific lags at the maximum."""
    dx_val = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    stagnant = 0

    while time.time() - start < time_limit and remaining() > 5:
        corr = compute_corr(h)
        m = float(np.max(corr))

        # Find all lags within 0.1% of max
        threshold = m - 0.001 * m
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.where(corr >= m * 0.999)[0]

        # Compute average gradient over active lags
        g = np.zeros(n)
        for k in active:
            kk = int(k)
            off = kk - (n - 1)
            if off >= 0:
                npairs = n - off
                g[:npairs] += (1.0 - h[off:off + npairs])
                g[off:off + npairs] -= h[:npairs]
            else:
                ao = -off
                npairs = n - ao
                g[ao:ao + npairs] += (1.0 - h[:npairs])
                g[:npairs] -= h[ao:ao + npairs]
        g /= len(active)
        g *= dx_val

        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break

        improved = False
        for step in [0.005, 0.002, 0.001, 0.01, 0.0005, 0.003, 0.02]:
            h_trial = h - step * g
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-14:
                h = h_trial
                best_c5 = c_new
                improved = True
                stagnant = 0
                break

        if not improved:
            stagnant += 1
            if stagnant > 30:
                break

    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 360
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: SLSQP refinement at parent resolution
    h_work = best_h.copy()
    for beta in [2000, 8000, 30000]:
        if remaining() < 30:
            break
        h_new = slsqp_run(h_work, best_n, beta, maxiter=500, ftol=1e-15)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_work = h_new.copy()

    # Targeted descent at parent
    if remaining() > 20:
        h_td, c5_td = targeted_lag_descent(h_work, best_n, time_limit=min(30, remaining() * 0.15))
        if c5_td < best_c5:
            best_c5 = c5_td
            best_h = h_td.copy()
            h_work = h_td.copy()

    # Phase 2: Binary local search at parent resolution
    if remaining() > 60:
        h_bin = (best_h > 0.5).astype(float)
        t_bin = min(180, remaining() * 0.25)
        h_bl, c5_bl = multi_restart_binary(h_bin, best_n, time_limit=t_bin, seed=seed)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()

    # Phase 3: Try higher resolution n=720
    if remaining() > 120:
        n_up = 720
        h_up = interp_to(best_h, best_n, n_up)
        h_up = project_feasible(h_up, n_up)

        # SLSQP at n=720
        for beta in [2000, 10000]:
            if remaining() < 40:
                break
            h_opt = slsqp_run(h_up, n_up, beta, maxiter=300, ftol=1e-14)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_up = h_opt.copy()

        # Targeted descent
        if remaining() > 30:
            h_td, c5_td = targeted_lag_descent(h_up, n_up, time_limit=min(40, remaining() * 0.2))
            if c5_td < best_c5:
                best_c5 = c5_td
                best_h = h_td.copy()
                best_n = n_up
            elif c5_td < best_c5:
                best_n = n_up

        # Binary at 720
        if remaining() > 60 and best_n == n_up:
            h_bin720 = (best_h > 0.5).astype(float)
            h_bl720, c5_bl720 = multi_restart_binary(h_bin720, n_up, time_limit=min(80, remaining() * 0.2), seed=seed + 500)
            if c5_bl720 < best_c5:
                best_c5 = c5_bl720
                best_h = h_bl720.copy()

    # Phase 4: Try n=540 and n=480 (alternative resolutions)
    for n_alt in [540, 480, 400]:
        if remaining() < 60:
            break
        h_alt = interp_to(best_h, best_n, n_alt)
        h_alt = project_feasible(h_alt, n_alt)

        h_opt = slsqp_run(h_alt, n_alt, 5000, maxiter=300, ftol=1e-14)
        c5_opt = compute_c5(h_opt)

        if remaining() > 20:
            h_td, c5_td = targeted_lag_descent(h_opt, n_alt, time_limit=min(20, remaining() * 0.15))
            if c5_td < c5_opt:
                c5_opt = c5_td
                h_opt = h_td

        h_bin_alt = (h_opt > 0.5).astype(float)
        h_bl, c5_bl = multi_restart_binary(h_bin_alt, n_alt, time_limit=min(30, remaining() * 0.15), seed=seed + n_alt)
        if c5_bl < c5_opt:
            c5_opt = c5_bl
            h_opt = h_bl

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_alt

    # Phase 5: Final intensive polish on best solution
    if remaining() > 30:
        h_work = best_h.copy()
        for beta in [10000, 50000, 100000]:
            if remaining() < 10:
                break
            h_new = slsqp_run(h_work, best_n, beta, maxiter=400, ftol=1e-15)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_work = h_new.copy()

    if remaining() > 15:
        h_td, c5_td = targeted_lag_descent(best_h, best_n, time_limit=min(20, remaining() * 0.4))
        if c5_td < best_c5:
            best_c5 = c5_td
            best_h = h_td.copy()

    # Phase 6: Final binary push
    if remaining() > 30:
        h_bin_final = (best_h > 0.5).astype(float)
        h_bl, c5_bl = multi_restart_binary(h_bin_final, best_n, time_limit=min(40, remaining() * 0.5), seed=seed + 9999)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()

    # Final continuous micro-polish
    if remaining() > 10:
        h_final = best_h.copy()
        h_new = slsqp_run(h_final, best_n, 200000, maxiter=300, ftol=1e-15)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Validate and return
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-12:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
