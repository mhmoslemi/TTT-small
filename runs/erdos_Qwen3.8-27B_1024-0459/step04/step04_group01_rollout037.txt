def single_coord_descent(h, n, n_passes, lam=0.0, rng=None, time_limit=float('inf')):
    """Single coordinate descent with optional Lagrangian for sum constraint."""
    h = h.copy()
    start = time.time()
    best_c5 = compute_c5(h)
    corr = compute_corr(h)
    nc = 2 * n - 1
    center = n - 1
    dx = 2.0 / n
    
    for p in range(n_passes):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break
        if rng is not None and p > 0 and p % 10 == 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        
        for i in order:
            # Compute coefficients for h[i]
            b = np.zeros(nc)
            # d = 0
            b[center] = 1.0 - 2.0 * h[i]
            # d > 0
            for d in range(1, n):
                k = center + d
                if i + d < n:
                    b[k] += (1.0 - h[i + d])
                if i >= d:
                    b[k] -= h[i - d]
            # d < 0 (m = -d > 0)
            for m in range(1, n):
                k = center - m
                if i >= m:
                    b[k] += (1.0 - h[i - m])
                if i + m < n:
                    b[k] -= h[i + m]
            
            # Solve: min_t max_k (corr[k] + (t-h[i])*b[k]) + lam*t over t in [0,1]
            # = min_t max_k (corr[k] - h[i]*b[k] + t*b[k]) + lam*t
            v = corr - h[i] * b  # intercepts
            # f(t) = max_k(v[k] + t*b[k]) + lam*t
            # Subgradient at t: {b[k] + lam : k is argmax of (v[k] + t*b[k])}
            
            # Bisection
            lo_t, hi_t = 0.0, 1.0
            for _ in range(50):
                mid = (lo_t + hi_t) / 2.0
                vals = v + mid * b
                max_val = vals.max()
                active = vals >= max_val - 1e-12
                slopes = b[active] + lam
                if slopes.min() > 1e-14:
                    hi_t = mid  # function increasing, go left
                elif slopes.max() < -1e-14:
                    lo_t = mid  # function decreasing, go right
                else:
                    break
            
            t_opt = (lo_t + hi_t) / 2.0
            
            if abs(t_opt - h[i]) > 1e-14:
                delta = t_opt - h[i]
                corr += delta * b
                h[i] = t_opt
        
        # Project to feasible set
        h = project_feasible(h, n)
        corr = compute_corr(h)
        
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
    
    return h, best_c5
