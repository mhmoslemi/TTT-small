    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h


def binary_local_search_v2(h_start, n, time_limit=60, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = h_start.astype(float).copy()
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        ones_idx = np.where(h == 1.0)[0]
        remove = rng.choice(ones_idx, n_ones - n // 2, replace=False)
        h[remove] = 0.0
    elif n_ones < n // 2:
        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(zeros_idx, n // 2 - n_ones, replace=False)
        h[add] = 1.0

    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * 2.0 / n
    best_h = h.copy()
    start = time.time()
    stall = 0

    for it in range(500000):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        k_max = int(np.argmax(corr))
        offset = k_max - (n - 1)

        if offset >= 0 and offset < n:
            candidates = np.array([i for i in ones if i + offset < n and h[i + offset] == 0.0])
        elif offset < 0 and offset > -(n-1):
            candidates = np.array([i for i in ones if i - offset < n and h[i - offset] == 0.0])
        else:
            candidates = ones

        found = False
        n_try = min(300, 50 + it // 100)
        for _ in range(n_try):
            if len(candidates) > 0 and rng.rand() < 0.75:
                i = int(candidates[rng.randint(len(candidates))])
            else:
                i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])

            h[i] = 0.0
            h[j] = 1.0
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))

            if new_max < cur_max - 1e-12:
                corr = new_corr
                cur_max = new_max
                best_c5 = cur_max * 2.0 / n
                best_h = h.copy()
                found = True
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                break
            else:
                h[i] = 1.0
                h[j] = 0.0

        if found:
            stall = 0
        else:
            stall += 1
            if stall > 200:
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                h[i] = 0.0
                h[j] = 1.0
                corr = compute_corr(h)
                cur_max = float(np.max(corr))
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                stall = 0

    return best_h, best_c5


def multi_restart_binary(h_start, n, time_limit=60, rng=None, n_restarts=5):
    if rng is None:
        rng = np.random.RandomState(42)
    best_h = h_start.copy()
    best_c5 = compute_c5(h_start)
    time_per = time_limit / n_restarts

    for r in range(n_restarts):
        if remaining() < time_per + 5:
            break
        h_cur = h_start.astype(float).copy()
        n_ones = int(np.sum(h_cur))
        if n_ones != n // 2:
            if n_ones > n // 2:
                idx = np.where(h_cur == 1.0)[0]
                rem = rng.choice(idx, n_ones - n // 2, replace=False)
                h_cur[rem] = 0.0
            else:
                idx = np.where(h_cur == 0.0)[0]
                add = rng.choice(idx, n // 2 - n_ones, replace=False)
                h_cur[add] = 1.0

        n_flips = rng.randint(n // 20, n // 5)
        ones = np.where(h_cur == 1.0)[0]
        zeros = np.where(h_cur == 0.0)[0]
        flip_ones = rng.choice(ones, min(n_flips, len(ones)), replace=False)
        flip_zeros = rng.choice(zeros, min(n_flips, len(zeros)), replace=False)
        h_cur[flip_ones] = 0.0
        h_cur[flip_zeros] = 1.0

        h_res, c5_res = binary_local_search_v2(h_cur, n, time_limit=min(time_per, remaining() - 5), rng=rng)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

    return best_h, best_c5


def single_coord_descent(h, n, max_passes=100, time_limit=60):
    h = h.copy()
    start = time.time()
    best_c5 = compute_c5(h)
    n_corr = 2 * n - 1

    for pass_num in range(max_passes):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break

        corr = compute_corr(h)
        cur_max = float(np.max(corr)) * 2.0 / n
        improved = False

        for i in range(n):
            if time.time() - start > time_limit:
                break
            if remaining() < 5:
                break

            slopes = np.zeros(n_corr)
            for d in range(-(n-1), n):
                idx = d + (n - 1)
                s = 0.0
                if d >= 0:
                    if d <= i <= n - 1:
                        s += (1.0 - h[i - d])
                    if 0 <= i <= n - 1 - d:
                        s -= h[i + d]
                else:
                    m = -d
                    if 0 <= i <= n - 1 - m:
                        s += (1.0 - h[i + m])
                    if i >= m:
                        s -= h[i - m]
                slopes[idx] = s

            intercepts = corr - h[i] * slopes
            vals_at_0 = intercepts
            vals_at_1 = slopes + intercepts

            max_at_0 = float(np.max(vals_at_0))
            max_at_1 = float(np.max(vals_at_1))

            if max_at_0 >= max_at_1:
                t_opt = 0.0
            else:
                t_opt = 1.0
            # Check if interior is better
            # Binary search for where max is minimized
            # The function f(t) = max_k (slopes[k]*t + intercepts[k]) is convex in t
            lo, hi = 0.0, 1.0
            for _ in range(30):
                mid = (lo + hi) / 2.0
                vals = slopes * mid + intercepts
                max_val = float(np.max(vals))
                active = vals >= max_val - 1e-13
                if slopes[active].max() < -1e-14:
                    lo = mid
                elif slopes[active].min() > 1e-14:
                    hi = mid
                else:
                    break
            t_opt = (lo + hi) / 2.0

            if abs(t_opt - h[i]) > 1e-10:
                h[i] = t_opt
                improved = True

        if not improved:
            break

        h = project_feasible(h, n)
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new

    return h, best_c5


def subgradient_momentum(h, n, max_iters=5000, time_limit=60, top_k=5):
    h = h.copy()
    start = time.time()
    best_c5 = compute_c5(h)
    momentum_vec = np.zeros(n)
    beta_mom = 0.85
    step_size = 0.02

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining() < 10:
            break

        corr = compute_corr(h)
        sorted_idx = np.argsort(-corr)
        top = sorted_idx[:top_k]
        weights = np.exp((corr[top] - corr[top[0]]) * 200.0)
        weights /= weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top):
            grad += w * corr_grad_per_shift(h, n, int(k))

        momentum_vec = beta_mom * momentum_vec + grad
        h_trial = h - step_size * momentum_vec
        h = project_feasible(h_trial, n)

        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            step_size = min(step_size * 1.005, 0.08)
        else:
            if c5_new > best_c5 * 1.005:
                step_size *= 0.95
            else:
                step_size = min(step_size * 1.002, 0.08)

    return h, best_c5


def optimize_at_n(h_init, n, time_alloc, rng):
    """Run full optimization pipeline at resolution n."""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: SLSQP
    t_remaining = time_alloc
    n_beta = min(4, max(1, int(t_remaining / 15)))
    betas = [500, 2000, 8000, 30000][:n_beta]
    t_per_slsqp = t_remaining / (n_beta + 2)

    for beta in betas:
        if remaining() < 10:
            break
        time_lim = min(t_per_slsqp, remaining() - 10)
        if time_lim < 3:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Subgradient
    if remaining() > 15 and time_alloc > 30:
        t_sg = min(30, time_alloc * 0.25, remaining() - 10)
        if t_sg > 5:
            h_sg, c5_sg = subgradient_momentum(h, n, max_iters=3000, time_limit=t_sg, top_k=5)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()
                h = h_sg.copy()

    # Phase 3: SLSQP polish
    if remaining() > 10:
        h_new = slsqp_smooth(h, n, 15000, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Binary local search with restarts
    if remaining() > 20 and time_alloc > 40:
        h_bin = (best_h > 0.5).astype(float)
        t_bin = min(remaining() - 10, time_alloc * 0.35)
        if t_bin > 10:
            n_r = max(2, int(t_bin / 20))
            h_b, c5_b = multi_restart_binary(h_bin, n, time_limit=t_bin, rng=rng, n_restarts=n_r)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()

    # Phase 5: Final SLSQP on best
    if remaining() > 8:
        h_new = slsqp_smooth(best_h, n, 30000, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 6: Final binary refinement
    if remaining() > 10:
        h_bin = (best_h > 0.5).astype(float)
        t_bin2 = min(remaining() - 5, 20)
        if t_bin2 > 5:
            h_b, c5_b = binary_local_search_v2(h_bin, n, time_limit=t_bin2, rng=rng)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()

    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
    except Exception:
        parent_n = 312
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)

    best_h = parent_h.copy()
    best_c5 = parent_c5
    best_n = parent_n

    # Step 1: Quick optimization at parent resolution
    t1 = min(80, budget_s * 0.1)
    h1, c5_1 = optimize_at_n(parent_h, parent_n, t1, rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = parent_n

    # Step 2: Explore multiple n values
    # Choose n values based on where the target C5 lands on nice grid
    # For target C5 = 0.38085: need max_corr * 2/n < 0.38085
    # i.e., max_corr < 0.19042 * n
    candidate_ns = []
    for n_try in range(300, 600, 4):
        if n_try % 2 != 0:
            continue
        if n_try == parent_n:
            continue
        # Check if binary solutions at this n could potentially beat target
        # (heuristic: max_corr target = 0.19042*n, should be >= n/8 + some margin)
        target_max_corr = 0.19042 * n_try
        # For feasibility, target_max_corr should be > n/8 (the average)
        if target_max_corr > n_try * 0.125:
            candidate_ns.append(n_try)

    # Prioritize n values near known-good range and with "nice" dx
    # dx = 2/n. We want 0.38085/dx to be close to an integer (for binary solutions)
    scored_ns = []
    for n_try in candidate_ns:
        dx_try = 2.0 / n_try
        target_units = 0.38085 / dx_try
        frac = target_units - int(target_units)
        # Score: closer to integer is better (easier to achieve with binary)
        score = min(frac, 1 - frac)
        # Also prefer n near parent_n (less interpolation distortion)
        proximity = 1.0 / (1.0 + abs(n_try - parent_n) / 50.0)
        total_score = -score + 0.3 * proximity
        scored_ns.append((total_score, n_try))

    scored_ns.sort(key=lambda x: -x[0])
    top_ns = [n_try for _, n_try in scored_ns[:12]]

    # Ensure we include some specific values
    for extra in [320, 336, 384, 400, 448, 480]:
        if extra not in top_ns and extra != parent_n:
            top_ns.append(extra)
    top_ns = top_ns[:10]

    total_remaining = budget_s - (time.time() - _START) - 100  # reserve 100s for final
    t_per_n = max(20, total_remaining / len(top_ns))

    for n_try in top_ns:
        if remaining() < 30:
            break
        h_interp = interp_to(best_h, best_n, n_try)
        h_feas = project_feasible(h_interp, n_try)

        t_alloc = min(t_per_n, remaining() - 20)
        h_opt, c5_opt = optimize_at_n(h_feas, n_try, t_alloc, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_try

    # Step 3: Intensive optimization on best solution found
    n_best = best_n
    h_best = best_h.copy()
    t_final = remaining() - 20

    if t_final > 60:
        # More SLSQP
        for beta in [5000, 20000, 50000]:
            if remaining() < 15:
                break
            h_new = slsqp_smooth(h_best, n_best, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_best = h_new.copy()

        # Binary search with many restarts
        if remaining() > 40:
            h_bin = (best_h > 0.5).astype(float)
            t_bin = min(remaining() - 20, 60)
            h_b, c5_b = multi_restart_binary(h_bin, n_best, time_limit=t_bin, rng=rng, n_restarts=8)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                h_best = h_b.copy()

        # Single coordinate descent for continuous refinement
        if remaining() > 30:
            h_cd, c5_cd = single_coord_descent(best_h, n_best, max_passes=50,
                                               time_limit=min(30, remaining() - 15))
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                h_best = h_cd.copy()

        # Final SLSQP polish
        if remaining() > 15:
            h_new = slsqp_smooth(best_h, n_best, 50000, maxiter=500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

        # Final binary
        if remaining() > 15:
            h_bin = (best_h > 0.5).astype(float)
            h_b, c5_b = binary_local_search_v2(h_bin, n_best,
                                                time_limit=min(15, remaining() - 8), rng=rng)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
