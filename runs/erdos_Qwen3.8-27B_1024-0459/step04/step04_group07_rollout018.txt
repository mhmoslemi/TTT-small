import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    # Project h to [0,1]^n with sum = n/2
    h = np.clip(h, 0, 1)
    for _ in range(200):
        diff = n/2.0 - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0, 1)
    # Final adjustment
    diff = n/2.0 - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] += diff
        h = np.clip(h, 0, 1)
    return h

def smooth_obj_and_grad(h, beta, pen_weight):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    # Softmax weights
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    # Smooth objective value
    log_sum = np.log(np.sum(np.exp(shifted)))  # = log(Σ exp(-β(m-corr_k)))
    obj = (m + log_sum / beta) * dx
    
    # Add penalty for sum constraint (if using without constraint)
    if pen_weight > 0:
        s = np.sum(h) - n/2.0
        obj += pen_weight * s * s
    
    # Gradient
    # d(corr[k])/dh[i] = -h[i+s_k]*[0≤i+s_k<n] + (1-h[i-s_k])*[0≤i-s_k<n]
    # where s_k = k - (n-1)
    grad = np.zeros(n)
    # Only need to compute for significant weights
    # For efficiency, only compute for weights > threshold
    thresh = 1.0 / (n * 10)  # threshold
    active = np.where(weights > thresh)[0]
    
    for k in active:
        s = k - (n - 1)
        w = weights[k]
        if w < 1e-15:
            continue
        # Contribution to gradient from this lag
        # d(corr[k])/dh[i] = -h[i+s]*[0≤i+s<n] + (1-h[i-s])*[0≤i-s<n]

        #   and 0 ≤ i ≤ n-1
        lo2 = max(0, s)
        hi2 = min(n-1, n-1+s)
        if lo2 <= hi2:
            grad[lo2:hi2+1] += w * ((1.0 - h[lo2-s:hi2-s+1]))
    
    grad *= dx
    
    if pen_weight > 0:
        s = np.sum(h) - n/2.0
        grad += 2 * pen_weight * s
    
    return obj, grad

def lbfgsb_smooth(h0, n, beta, maxiter, pen_weight=0.0):
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        lambda x: smooth_obj_and_grad(x, beta, pen_weight)[0],
        lambda x: smooth_obj_and_grad(x, beta, pen_weight)[1],
        h0, method='L-BFGS-B', bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'iprint': 0}
    )
    return project_feasible(res.x, n)

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def smooth_objective(h, beta, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    val = (m + log_sum / beta) * dx
    if pen > 0:
        s = np.sum(h) - n / 2.0
        val += pen * s * s
    return val

def smooth_gradient(h, beta, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    weights = np.exp(shifted)
    total = np.sum(weights)
    if total < 1e-300:
        weights = np.zeros_like(corr)
        weights[int(np.argmax(corr))] = 1.0
    else:
        weights /= total
    
    # Gradient computation
    grad = np.zeros(n)
    # For efficiency, only process lags with significant weight
    threshold = 1.0 / (len(corr) * 100)
    active = np.where(weights > threshold)[0]
    
    n1 = n - 1
    for k in active:
        s = k - n1
        w = weights[k]
        # d(corr[k])/dh[i] = -h[i+s]*[0<=i+s<n] + (1-h[i-s])*[0<=i-s<n]
        # Term 1: -w * h[i+s] for i in [max(0,-s), min(n-1, n-1-s)]
        lo1 = max(0, -s)
        hi1 = min(n-1, n1-s)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += -w * h[lo1+s:hi1+s+1]
        # Term 2: w * (1-h[i-s]) for i in [max(0,s), min(n-1, n-1+s)]
        lo2 = max(0, s)
        hi2 = min(n-1, n1+s)
        if lo2 <= hi2:
            grad[lo2:hi2+1] += w * (1.0 - h[lo2-s:hi2-s+1])
    
    grad *= dx
    if pen > 0:
        s_val = np.sum(h) - n / 2.0
        grad += 2.0 * pen * s_val
    return grad

def optimize_lbgfsb(h0, n, beta, maxiter, pen=0.0):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            lambda x: smooth_objective(x, beta, pen),
            lambda x: smooth_gradient(x, beta, pen),
            h0, method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 50, 'iprint': 0}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h0

def optimize_slsqp(h0, n, beta, maxiter, pen=0.0):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                   'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(
            lambda x: smooth_objective(x, beta, pen),
            lambda x: smooth_gradient(x, beta, pen),
            h0, method='SLSQP', bounds=bounds,
            constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h0

def _obj_fn(x):

    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')

    s = np.sum(np.exp(shifted))
    if s < 1e-300:
        return m * dx
    log_sum = np.log(s)
    val = (m + log_sum / _G_BETA) * dx
    if _G_PEN > 0:
        ss = float(np.sum(x)) - n / 2.0
        val += _G_PEN * ss * ss
    return float(val)

def _grad_fn(x):
    n = _G_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    shifted = np.clip(_G_BETA * (corr - m), -500.0, 0.0)
    weights = np.exp(shifted)
    total = float(np.sum(weights))
    if total < 1e-300:
        am = int(np.argmax(corr))
        weights = np.zeros(len(corr))
        weights[am] = 1.0
    else:
        weights /= total
    grad = np.zeros(n)
    n1 = n - 1
    thresh = 1.0 / (len(corr) * 200.0)
    active = np.where(weights > thresh)[0]
    for k in active:
        s = k - n1
        w = weights[k]
        lo1 = max(0, -s)
        hi1 = min(n1, n1 - s)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += -w * x[lo1+s:hi1+s+1]
        lo2 = max(0, s)
        hi2 = min(n1, n1 + s)
        if lo2 <= hi2:
            grad[lo2:hi2+1] += w * (1.0 - x[lo2-s:hi2-s+1])
    grad *= dx
    if _G_PEN > 0:
        ss = float(np.sum(x)) - n / 2.0
        grad += 2.0 * _G_PEN * ss

def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c else 0.0
    return h

def make_rand_bin(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0

def make_spectral_init(n, rng):
    """Initialize using flat-ish low-frequency spectrum."""
    freqs = np.fft.rfftfreq(n)
    mask = np.arange(n // 2 + 1)
    H = np.zeros(n // 2 + 1, dtype=complex)

    phase = rng.uniform(0, 2 * np.pi, size=band)
    H[:band] = np.exp(1j * phase)
    H[band:] = 0.0
    h = np.fft.irfft(H * n, n=n)
    h = np.clip(h, 0.0, 1.0)
    h = project_feasible(h, n)
    if compute_c5(h) > 0.45:
        h = project_feasible(make_tm(n), n)
    return h

def make_zadoff(rng, n):

def subgradient_step(h, n, step_size):
    """One subgradient step on the exact max."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    n1 = n - 1
    s = k_star - n1
    grad = np.zeros(n)
    lo1 = max(0, -s)
    hi1 = min(n1, n1 - s)
    if lo1 <= hi1:
        grad[lo1:hi1+1] = -h[lo1+s:hi1+s+1]
    lo2 = max(0, s)
    hi2 = min(n1, n1 + s)
    if lo2 <= hi2:
        grad[lo2:hi2+1] += (1.0 - h[lo2-s:hi2-s+1])
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new
