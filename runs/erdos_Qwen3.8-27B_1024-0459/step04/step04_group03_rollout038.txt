    """Update D after changing h[old_i] and h[old_j]."""
    for k in range(n):
        # h[old_i] changed from 1 to new_val_i (which is 0)
        delta_i = new_val_i - 1  # = -1
        if delta_i != 0:
            # k - lag = old_i → lag = k - old_i
            lag1 = k - old_i
            if -(n-1) <= lag1 <= n-1:
                D[k, lag1 + n - 1] += delta_i
            # k + lag = old_i → lag = old_i - k
            lag2 = old_i - k
            if -(n-1) <= lag2 <= n-1:

        delta_j = new_val_j - 0  # = +1

    best_h = h.copy()

            break

        max_corr = np.max(corr)
        if max_corr * dx < best_c5:
            best_c5 = max_corr * dx
            best_h = h.copy()
        
        # Find active lags (within epsilon of max)
        threshold = max_corr - 0.1
        active = np.where(corr >= threshold)[0]
        
        # Compute subgradient: weighted average of gradients of active corr
        g = np.zeros(n)
        for m in active:
            lag = m - (n - 1)
            weight = corr[m] - threshold  # proportional weight
            for i in range(n):
                val = 0.0
                if 0 <= i + lag < n:
                    val += (1 - h[i + lag])
                if 0 <= i - lag < n:
                    val -= h[i - lag]
                g[i] += weight * val
        
        if np.sum(np.abs(g)) < 1e-10:
            break
        
        # Normalized subgradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-10:
            break
        g = g / g_norm
        
        # Learning rate
        lr = lr0 / (1 + 0.01 * it)
        
        # Projected gradient step (maintain sum constraint)
        g_centered = g - np.mean(g)
        h = h - lr * g_centered
        h = project_feasible(h, n)
    
    return best_h, best_c5
