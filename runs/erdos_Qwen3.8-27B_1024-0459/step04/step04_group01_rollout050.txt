    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n

def corr_grad_per_shift(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        if npairs > 0:
            g[off:off + npairs] += (1.0 - x[:npairs])
            g[:npairs] -= x[off:off + npairs]
    else:
        ao = -off
        npairs = n - ao
        if npairs > 0:
            g[:npairs] += (1.0 - x[ao:ao + npairs])
            g[ao:ao + npairs] -= x[:npairs]
    return g

    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -60, 60)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_val

    n = _N

    beta = _BETA

    m = np.max(corr)

    g = np.zeros(n)

def swap_delta(h, i, j, n):
    """Compute correlation delta for h[i]:1->0, h[j]:0->1. Returns delta array of length 2n-1."""
    delta = np.zeros(2 * n - 1)
    # h[i] 1->0: positive shifts d where i+d valid
    if i + 1 < n:
        delta[n:n + (n - 1 - i)] -= (1.0 - h[i + 1:n])
    # h[i] 1->0: negative shifts, i-m valid
    if i > 0:
        delta[n - i:n - 1] += h[0:i][::-1]
    # h[j] 0->1: positive shifts d where j+d valid
    if j + 1 < n:
        delta[n:n + (n - 1 - j)] += (1.0 - h[j + 1:n])
    # h[j] 0->1: negative shifts, j-m valid
    if j > 0:
        delta[n - j:n - 1] -= h[0:j][::-1]
    return delta

def sturmian_init(n, alpha, rng):
    """Generate a balanced binary sequence with given density."""
    h = np.zeros(n)
    # Use three-distance theorem / mechanical word
    target = n // 2
    # Simple approach: distribute ones as evenly as possible with a golden-ratio-like offset
    irat = np.floor(np.arange(n) * alpha + rng.rand())
    h[irat % n] = 1.0
    # Fix count
    n_ones = int(np.sum(h))
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0.0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1.0
    return h


def thue_morse_init(n, rng):
    """Generate Thue-Morse based initialization."""
    # Thue-Morse sequence: t(i) = popcount(i) % 2
    h = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    # Ensure density 1/2
    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0.0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1.0
    return h

    t1 = min(350, max(50, remaining() * 0.3))
    h1, c5_1 = optimize_at_n(best_h.copy(), best_n, t1, rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = best_n

    # Phase 2: Try multiple n values
    # Choose n values around parent_n
    n_candidates = sorted(set([
        best_n, best_n + 2, best_n + 4, best_n + 6, best_n + 8,
        best_n + 10, best_n + 12, best_n + 16, best_n + 20, best_n + 24,
        best_n + 32, best_n + 40,
        best_n - 2, best_n - 4, best_n - 6, best_n - 8,
        max(100, best_n - 16),
        256, 288, 320, 336, 360, 400, 448, 512
    ]))
    n_candidates = [nn for nn in n_candidates if 80 <= nn <= 600 and nn % 2 == 0]
    n_candidates = [nn for nn in n_candidates if nn != best_n]
    # Sort by distance from best_n
    n_candidates.sort(key=lambda x: abs(x - best_n))

    # Also try some structured initializations at a few n values
    structured_ns = []
    for nn in [320, 336, 352, 368, 384, 400]:
        if nn % 2 == 0 and nn != best_n and nn not in n_candidates[:4]:
            structured_ns.append(nn)

    phase2_per_n = min(80, max(15, (remaining() * 0.5) / max(1, len(n_candidates[:8]))))

    for n_t in n_candidates[:12]:
        if remaining() < 30:
            break
        t_per_n = min(phase2_per_n, max(10, remaining() * 0.08))
        # Interpolate from best
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)

        h_t, c5_t = optimize_at_n(h_feas, n_t, t_per_n, rng)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t

    # Phase 3: Structured initializations
    for n_s in structured_ns[:3]:
        if remaining() < 40:
            break
        t_s = min(60, max(15, remaining() * 0.06))
        # Thue-Morse init
        h_tm = thue_morse_init(n_s, rng)
        h_t, c5_t = optimize_at_n(h_tm, n_s, t_s, rng)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_s

        # Golden ratio init
        if remaining() > 20:
            alpha = 0.5 + 0.01 * rng.randn()
            h_gr = sturmian_init(n_s, max(0.3, min(0.7, alpha)), rng)
            h_t2, c5_t2 = optimize_at_n(h_gr, n_s, t_s * 0.7, rng)
            if c5_t2 < best_c5:
                best_c5 = c5_t2
                best_h = h_t2.copy()
                best_n = n_s

    # Phase 4: Intensive polish on best solution
    n = best_n
    h = best_h.copy()
    t_polish = min(200, max(30, remaining() * 0.2))

    # More SLSQP
    for beta in [5000, 15000, 40000, 80000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new

    # Re-round and binary polish
    h_bin = to_binary(best_h, n, rng)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    h = best_h.copy()
    if remaining() > 20:
        t_g = min(60, max(10, remaining() * 0.15))
        h, c5 = binary_greedy_1swap(h, n, t_g, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        h = h.copy()

    if remaining() > 15:
        t_sa = min(40, max(5, remaining() * 0.1))
        h, c5 = binary_simulated_annealing(h, n, t_sa, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        h = h.copy()

    if remaining() > 15:
        t_g2 = min(40, max(5, remaining() * 0.15))
        h, c5 = binary_greedy_1swap(h, n, t_g2, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 5: Try 2-swap on best
    if remaining() > 20:
        h, c5 = binary_2swap_search(best_h.copy(), best_n, min(30, remaining() * 0.1), rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    s = np.sum(best_h)
    target = best_n / 2.0
    if abs(s - target) > 1e-10:
        best_h = project_feasible(best_h, best_n)

    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    # Ensure exact feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5 + 1e-12:
        best_c5 = final_c5

    return (best_h, best_c5, best_n)
