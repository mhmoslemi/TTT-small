def swap_delta_corr(h, i, j, n):
    """Compute the change in correlation when flipping h[i]:1->0, h[j]:0->1.
    Returns delta array of length 2n-1 (full correlation indices)."""
    delta = np.zeros(2 * n - 1)
    # h[i] goes 1->0
    # For shift d>=0: corr[n-1+d] = sum_{a=0}^{n-1-d} h[a+d]*(1-h[a])
    # h[i] as unshifted (a=i): valid if i <= n-1-d, i.e., d <= n-1-i
    #   (1-h[i]) goes 0->1: corr increases by h[i+d] for d=1..n-1-i (d=0: h[i]*(1-h[i]) unchanged for binary)
    for d in range(1, n - i):
        delta[n - 1 + d] += h[i + d]
    # h[i] as shifted (a+d = i, a = i-d): valid if i-d >= 0, i.e., d <= i
    #   h[i] goes 1->0: corr decreases by (1-h[i-d])
    for d in range(1, i + 1):
        delta[n - 1 + d] -= (1.0 - h[i - d])
    # h[j] goes 0->1
    # h[j] as unshifted (a=j): (1-h[j]) goes 1->0: corr decreases by h[j+d]
    for d in range(1, n - j):
        delta[n - 1 + d] -= h[j + d]
    # h[j] as shifted (a+d=j, a=j-d): h[j] goes 0->1: corr increases by (1-h[j-d])
    for d in range(1, j + 1):
        delta[n - 1 + d] += (1.0 - h[j - d])
    # Handle negative shifts by symmetry: corr[n-1-d] = corr[n-1+d]
    # But wait - for the full correlation of same-length sequences, is it symmetric?
    # corr[n-1+d] = sum_{a=0}^{n-1-d} h[a+d]*(1-h[a])
    # corr[n-1-d] = sum_{a=d}^{n-1} h[a]*(1-h[a-d]) = sum_{b=0}^{n-1-d} h[b+d]*(1-h[b])
    # Yes, symmetric!
    for d in range(1, n):
        delta[n - 1 - d] = delta[n - 1 + d]
    return delta

def coord_descent_continuous(h, n, max_passes=100, time_limit=30, rng=None):

            if i == j:

            hi_t = min(1.0, s)

                    if m < n and i - m >= 0:
                        s_val -= h[i - m]

def top_k_equalize(h, n, top_k=5, max_iters=500, time_limit=30):
    """Reduce the top-K correlations toward each other."""
    h = h.copy()
    dx_val = 2.0 / n
    best_c5 = compute_c5(h)
    start = time.time()
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break
        
        corr = compute_corr(h)
        top = np.argsort(-corr)[:top_k]
        worst_shift = int(top[0])
        c_worst = corr[worst_shift]
        c_min_top = corr[top[-1]]
        
        if c_worst - c_min_top < 1e-10:
            break
        
        # Direction: negative gradient of worst shift, positive gradient of min top shift
        g_worst = corr_grad_per_shift(h, n, worst_shift)
        g_min = corr_grad_per_shift(h, n, int(top[-1]))
        d = g_min - g_worst
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-14:
            break
        d /= dnorm
        
        improved = False
        for step in [0.002, 0.005, 0.01, 0.02, 0.003, 0.015, 0.03, 0.008]:
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-12:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            # Try opposite direction
            for step in [0.003, 0.008, 0.015, 0.005]:
                h_trial = h - step * d
                h_trial = project_feasible(h_trial, n)
                c_new = compute_c5(h_trial)
                if c_new < best_c5 - 1e-12:
                    h = h_trial
                    best_c5 = c_new
                    improved = True
                    break
        if not improved:
            break
    
    return h, best_c5

    n = best_n
    h = best_h.copy()

    # Phase 1: Quick SLSQP refinement
    if remaining() > 20:
        for beta in [500, 2000]:
            if remaining() < 8:
                break
            h_new = slsqp_smooth(h, n, beta, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 2: Binary local search (main workhorse - improved)
    if remaining() > 30:
        h_bin = best_h.copy()
        h_b, c5_b = binary_local_search_v2(h_bin, n, time_limit=min(80, remaining() - 20), rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h = h_b.copy()
            best_n = n

    # Phase 3: Directed swap search (focus on worst shift)
    if remaining() > 30:
        h_d, c5_d = directed_swap_search(best_h, best_n, time_limit=min(60, remaining() - 15), rng=rng)
        if c5_d < best_c5:
            best_c5 = c5_d
            best_h = h_d.copy()
            h = h_d.copy()

    # Phase 4: SLSQP polish after binary optimization
    if remaining() > 15:
        for beta in [3000, 10000, 30000]:
            if remaining() < 8:
                break
            h_new = slsqp_smooth(best_h, best_n, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 5: Top-K equalization
    if remaining() > 15:
        h_eq, c5_eq = top_k_equalize(best_h, best_n, top_k=5, max_iters=300,
                                     time_limit=min(30, remaining() - 10))
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 6: Try higher resolution
    n_tries = sorted(set([
        best_n + 4, best_n + 8, best_n + 16,
        384, 400, 448, 512
    ]))
    n_tries = [nn for nn in n_tries if 100 <= nn <= 600 and nn % 4 == 0 and nn != best_n]
    
    for n_t in n_tries:
        if remaining() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        
        # SLSQP on new resolution
        h_opt = h_feas
        for beta in [1000, 5000]:
            if remaining() < 10:
                break
            h_opt = slsqp_smooth(h_opt, n_t, beta, maxiter=200)
        
        # Binary refinement
        if remaining() > 20:
            h_opt_b, c5_opt_b = binary_local_search_v2(h_opt, n_t,
                                                        time_limit=min(30, remaining() - 15),
                                                        rng=rng)
            if c5_opt_b < compute_c5(h_opt):
                h_opt = h_opt_b
        
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
            h = h_opt.copy()

    # Phase 7: Multi-start at best resolution
    if remaining() > 40:
        h_ms, c5_ms = multi_start_binary(best_n, num_starts=3,
                                         time_per_start=min(15, (remaining() - 10) // 3),
                                         seed=seed + 100)
        if c5_ms < best_c5:
            best_c5 = c5_ms
            best_h = h_ms.copy()
            h = h_ms.copy()

    # Phase 8: Final intensive refinement
    n = best_n
    h = best_h.copy()
    
    # Final binary search
    if remaining() > 20:
        h_f, c5_f = binary_local_search_v2(h, n, time_limit=min(30, remaining() - 10), rng=rng)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            h = h_f.copy()

    # Final SLSQP
    if remaining() > 10:
        for beta in [10000, 50000]:
            if remaining() < 6:
                break
            h_new = slsqp_smooth(best_h, best_n, beta, maxiter=250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Final coordinate descent
    if remaining() > 12:
        h_cd, c5_cd = coord_descent_continuous(best_h, best_n, max_passes=80,
                                               time_limit=min(20, remaining() - 8), rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Final polish
    if remaining() > 6:
        h_new = slsqp_smooth(best_h, best_n, 20000, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Validate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
