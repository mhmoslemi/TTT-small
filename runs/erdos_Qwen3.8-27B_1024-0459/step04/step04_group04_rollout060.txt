_BETA = 1.0
_TGT = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = float(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)

    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def corr_grad_single(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[:npairs] += (1.0 - x[off:off+npairs])
        g[off:off+npairs] -= x[:npairs]
    else:
        ao = -off
        npairs = n - ao
        g[ao:ao+npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[ao:ao+npairs]
    return g

    m = np.max(corr)
    beta = _BETA

    ex = np.exp(cl)
    s = np.sum(ex)
    if s < 1e-300:
        s = 1e-300
    return (m + np.log(s) / beta) * dx_val

def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -50, 50)

    s = np.sum(ex)

        s = 1e-300

    for k in range(len(corr)):
        wk = w[k]
        if wk > 1e-10:
            g += wk * corr_grad_single(x, n, k)
    return g * dx_val

    return np.sum(x) - _TGT

    return np.ones(_N)

def slsqp_optimize(h, n, beta, maxiter=300):
    global _N, _BETA, _TGT
    _N = n
    _TGT = float(n) / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:

                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})

def topk_grad_descent(h, n, max_iters=300, top_k=5):

        m = float(np.max(corr))

            g += corr_grad_single(h, n, int(k))
        g /= len(top_idx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        improved = False
        for step in [0.03, 0.01, 0.005, 0.05, 0.002, 0.02, 0.08]:
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)

import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = float(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _START) - 3.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def corr_grad_single(x, n, k):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        npairs = n - off
        g[:npairs] += (1.0 - x[off:off + npairs])
        g[off:off + npairs] -= x[:npairs]
    else:
        ao = -off
        npairs = n - ao
        g[ao:ao + npairs] += (1.0 - x[:npairs])
        g[:npairs] -= x[ao:ao + npairs]
    return g


def smooth_obj(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -50, 50)
    ex = np.exp(cl)
    s = np.sum(ex)
    if s < 1e-300:
        s = 1e-300
    return (m + np.log(s) / beta) * dx_val


def smooth_grad(x):
    n = _N
    dx_val = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _BETA
    shifted = corr - m
    cl = np.clip(beta * shifted, -50, 50)
    ex = np.exp(cl)
    s = np.sum(ex)
    if s < 1e-300:
        s = 1e-300
    w = ex / s
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk > 1e-10:
            g += wk * corr_grad_single(x, n, k)
    return g * dx_val


def sum_con(x):
    return np.sum(x) - _TGT


def sum_con_jac(x):
    return np.ones(_N)


def slsqp_optimize(h, n, beta, maxiter=250):
    global _N, _BETA, _TGT
    _N = n
    _TGT = float(n) / 2.0
    _BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)


def topk_grad_descent(h, n, max_iters=300, top_k=5):
    h = h.copy()
    best_c5 = compute_c5(h)
    dx_val = 2.0 / n
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        m = float(np.max(corr))
        if m * dx_val >= best_c5 + 1e-8:
            break
        top_idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k in top_idx:
            g += corr_grad_single(h, n, int(k))
        g /= len(top_idx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        improved = False
        steps = [0.04, 0.02, 0.01, 0.06, 0.005, 0.03, 0.08, 0.002]
        for step in steps:
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-14:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            for step in [0.001, 0.0005, 0.002, 0.0003]:
                h_trial = h + step * d
                h_trial = project_feasible(h_trial, n)
                c_new = compute_c5(h_trial)
                if c_new < best_c5 - 1e-14:
                    h = h_trial
                    best_c5 = c_new
                    improved = True
                    break
            if not improved:
                break
    return h, best_c5


def subgradient_polish(h, n, max_iters=500):
    """Diminishing-step subgradient on the max correlation."""
    h = h.copy()
    best_c5 = compute_c5(h)
    dx_val = 2.0 / n
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        m = float(np.max(corr))
        if m * dx_val >= best_c5 + 1e-8:
            break
        # Active set: all k where corr[k] > m - 1.0
        active = np.where(corr > m - 1.5)[0]
        if len(active) == 0:
            active = np.argsort(-corr)[:3]
        g = np.zeros(n)
        for k in active:
            g += corr_grad_single(h, n, int(k))
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        # Diminishing step
        step = 1.0 / (np.sqrt(it + 1) * 20.0)
        h_trial = h + step * (-g / gn)
        h_trial = project_feasible(h_trial, n)
        c_new = compute_c5(h_trial)
        if c_new < best_c5:
            best_c5 = c_new
            h = h_trial.copy()
    return h, best_c5


def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        rem = rng.choice(len(idx), n_ones - target, replace=False)
        h_bin[idx[rem]] = 0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        add = rng.choice(len(idx), target - n_ones, replace=False)
        h_bin[idx[add]] = 1
    return h_bin


def sa_binary(h, n, time_limit=25, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[rng.choice(len(idx), n_ones - target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[rng.choice(len(idx), target - n_ones, replace=False)]] = 1

    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()

    T = 6.0
    alpha = 0.9993
    start = time.time()
    stall = 0
    max_stall = 12000

    while time.time() - start < time_limit and T > 0.02:
        if remaining() < 8:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        delta = new_max - cur_max
        if delta < 0 or (rng.rand() < np.exp(-delta / max(T, 1e-10))):
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
        T *= alpha
        if stall > max_stall:
            T = max(T, 3.0)
            stall = 0
    return best_h, best_max * dx_val

            best_h = h_new.copy()

        h_td, c5_td = topk_grad_descent(h, n, max_iters=200, top_k=5)
        if c5_td < best_c5:
            best_c5 = c5_td
            best_h = h_td.copy()
            h = h_td.copy()

    # Stage C: Subgradient polish
    if time.time() - start < time_limit * 0.8 and remaining() > 8:
        h_sg, c5_sg = subgradient_polish(h, n, max_iters=300)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Stage D: Final SLSQP at very high beta
