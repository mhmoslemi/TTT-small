def compute_c5(h):
    dx = 2.0/len(h)
    corr = np.correlate(h, 1.0-h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n/2.0
    # Simple adjustment by adding a delta to all non-boundary points
    diff = target - np.sum(h)
    if abs(diff) < 1e-10: return h
    # Distribute adjustment to minimize max change
    mask = (h > 1e-8) & (h < 1.0-1e-8)
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    # Clip again to maintain boundaries
    h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-8:
        for i in range(n):
            if h[i] > 1e-8 and h[i] < 1.0-1e-8:
                h[i] = np.clip(h[i] + diff2, 0.0, 1.0)
                break
    return h
