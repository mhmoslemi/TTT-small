    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def fast_coordinate_descent(h_init, n, max_sweeps=60, seed=0):
    rng = np.random.RandomState(seed)
    best = h_init.copy()
    best_corr = compute_corr(best)
    best_c5 = compute_c5(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    k0 = n - 1  # index of lag 0 in the full correlation

    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        order = np.arange(n)
        rng.shuffle(order)

        for i in order:
            if remaining_time() < 2:
                break
            t0 = best[i]
            if t0 < 1e-10 or t0 > 1.0 - 1e-10:
                # Try moving to 0.5 or the other extreme
                candidates = [1.0 - t0, 0.5]
            else:
                candidates = [0.0, 1.0, t0 * 0.5, min(1.0, t0 + 0.15), max(0.0, t0 - 0.15), 1.0 - t0]

            # Compute base and coefficients
            a = np.zeros(ncorr)
            b = np.zeros(ncorr)
            # For lag 0 (k = k0): quadratic term t - t^2
            b[k0] = -1.0
            a[k0] = 1.0  # coefficient of t from the t*(1-t) term, will be overridden below

            # Recompute a properly
            a[:] = 0.0
            b[:] = 0.0
            b[k0] = -1.0

            for off in range(-(n-1), n):
                k = n - 1 + off
                if off == 0:
                    # t*(1-t) = t - t^2, so a[k] = 1, b[k] = -1
                    a[k] = 1.0
                else:
                    coeff = 0.0
                    if 0 <= i + off < n:
                        coeff += (1.0 - best[i + off])
                    if 0 <= i - off < n:
                        coeff += (-best[i - off])
                    a[k] = coeff

            base = best_corr - a * t0 - b * t0 * t0

            best_t = t0
            best_val = best_c5

            for t in candidates:
                t = max(0.0, min(1.0, t))
                if abs(t - t0) < 1e-12:
                    continue
                vals = base + a * t + b * t * t
                c = np.max(vals) * dx
                if c < best_val - 1e-14:
                    best_val = c
                    best_t = t

            # Also try the vertex of the quadratic at k0
            vertex = a[k0] / 2.0
            if 0.01 < vertex < 0.99 and abs(vertex - t0) > 1e-6:
                vals = base + a * vertex + b * vertex * vertex
                c = np.max(vals) * dx
                if c < best_val - 1e-14:
                    best_val = c
                    best_t = vertex

            # Golden section refinement if we found improvement
            if best_t != t0:
                center = (best_t + t0) / 2.0
                spread = max(0.05, abs(best_t - t0) * 0.6)
                lo = max(0.0, center - spread)
                hi = min(1.0, center + spread)
                # Ensure best_t and t0 are in [lo, hi]
                lo = min(lo, min(best_t, t0) - 0.01)
                hi = max(hi, max(best_t, t0) + 0.01)
                lo = max(0.0, lo)
                hi = min(1.0, hi)
                if hi - lo > 1e-8:
                    gr = (np.sqrt(5.0) - 1.0) / 2.0
                    c1 = hi - gr * (hi - lo)
                    c2 = lo + gr * (hi - lo)
                    f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                    f2 = np.max(base + a * c2 + b * c2 * c2) * dx
                    for _ in range(35):
                        if f1 < f2:
                            hi = c2
                            c2 = c1
                            f2 = f1
                            c1 = hi - gr * (hi - lo)
                            f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                        else:
                            lo = c1
                            c1 = c2
                            f1 = f2
                            c2 = lo + gr * (hi - lo)
                            f2 = np.max(base + a * c2 + b * c2 * c2) * dx
                    gs_t = (lo + hi) / 2.0
                    gs_c = np.max(base + a * gs_t + b * gs_t * gs_t) * dx
                    if gs_c < best_val:
                        best_val = gs_c
                        best_t = gs_t

            if best_val < best_c5 - 1e-13:
                # Update correlation incrementally
                dt = best_t - t0
                dt2 = best_t * best_t - t0 * t0
                for off in range(-(n-1), n):
                    k = n - 1 + off
                    if a[k] != 0.0 or b[k] != 0.0:
                        best_corr[k] += a[k] * dt + b[k] * dt2
                best[i] = best_t
                best_c5 = best_val
                improved = True

        if not improved:
            if sweep > 3:
                break

    # Clip and project
    best = np.clip(best, 0.0, 1.0)
    best = project_feasible(best, n)
    best_c5 = compute_c5(best)
    return best, best_c5

def pair_swap_descent(h_init, n, max_iters=2000, seed=0):
    """Try swapping small amounts between pairs of coordinates."""
    rng = np.random.RandomState(seed + 1000)
    best = h_init.copy()
    best_corr = compute_corr(best)
    best_c5 = compute_c5(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1

    for it in range(max_iters):
        if remaining_time() < 3:
            break
        # Pick two random coordinates
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        ti = best[i]
        tj = best[j]
        if ti <= 1e-8 or ti >= 1.0 - 1e-8:
            continue
        if tj <= 1e-8 or tj >= 1.0 - 1e-8:
            continue
        # Try transferring a small amount from i to j (or vice versa)
        for delta in [0.05, -0.05, 0.1, -0.1, 0.02, -0.02]:
            ni = ti + delta
            nj = tj - delta
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            # Evaluate effect on correlation
            # Change in corr for coordinate i: a_i * delta + b_i * delta^2
            # Change in corr for coordinate j: a_j * (-delta) + b_j * delta^2
            # Total change: (a_i - a_j) * delta + (b_i + b_j) * delta^2

            # Quick check: just evaluate the max change
            worst = 0.0
            for off_i in range(-(n-1), n):
                ki = n - 1 + off_i
                ai = 0.0
                bi = 0.0
                if off_i == 0:
                    ai = 1.0
                    bi = -1.0
                else:
                    if 0 <= i + off_i < n:
                        ai += (1.0 - best[i + off_i])
                    if 0 <= i - off_i < n:
                        ai += (-best[i - off_i])
                # Only need to check if this lag is near the max
                if ki < len(best_corr) and best_corr[ki] > best_c5 / dx - 0.3:
                    new_val = best_corr[ki] + ai * delta + bi * delta * delta
                    change = new_val - best_corr[ki]
                    if change > worst:
                        worst = change

            if worst < -0.001 * dx:
                # Actually evaluate
                best[i] = ni
                best[j] = nj
                new_corr = compute_corr(best)
                new_c5 = np.max(new_corr) * dx
                if new_c5 < best_c5 - 1e-14:
                    best_corr = new_corr
                    best_c5 = new_c5
                else:
                    best[i] = ti
                    best[j] = tj

    best = project_feasible(best, n)
    best_c5 = compute_c5(best)
    return best, best_c5

def smooth_slsqp(h, n, beta, maxiter, tol=1e-18):
    """Smoothed SLSQP optimization."""
    dx = 2.0 / n

    def obj_func(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        exp_vals = np.exp(shifted)
        wsum = np.sum(exp_vals)
        return (m + np.log(wsum) / beta) * dx

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        exp_vals = np.exp(shifted)
        wsum = np.sum(exp_vals)
        weights = exp_vals / wsum
        grad = np.zeros(n)
        for k in range(len(corr)):
            w = weights[k]
            if w < 1e-14:
                continue
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                if npairs > 0:
                    grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                    grad[:npairs] -= w * x[offset:offset + npairs]
            else:
                aoff = -offset
                npairs = n - aoff
                if npairs > 0:
                    grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                    grad[aoff:aoff + npairs] -= w * x[:npairs]
        grad *= dx
        return grad

    def constraint_fun(x):
        return np.sum(x) - n / 2.0

    def constraint_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]

    try:
        res = minimize(obj_func, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            return trial
    except Exception:
        pass
    return h.copy()

def structured_perturbation(h, n, amount, rng):
    """Apply a structured perturbation: shift, flip, or blend."""
    h_new = h.copy()
    strategy = rng.randint(4)
    if strategy == 0:
        # Circular shift by 1-3 positions
        shift = rng.randint(1, 4)
        h_new = np.roll(h, shift)
    elif strategy == 1:
        # Flip a random block
        start = rng.randint(n - 40)
        end = min(n, start + rng.randint(10, 40))
        h_new[start:end] = 1.0 - h_new[start:end]
        h_new = project_feasible(h_new, n)
    elif strategy == 2:
        # Blend with a different phase
        phase = rng.uniform(0, 2)
        for i in range(n):
            h_new[i] += amount * np.sin(2 * np.pi * i / n + phase) * 0.1
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
    else:
        # Random small perturbation
        h_new += rng.uniform(-amount, amount, n)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Initialize from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 256
        h_tmp = np.zeros(best_n)
        # Thue-Morse-like pattern
        for i in range(best_n):
            cnt = bin(i).count('1')
            h_tmp[i] = 1.0 if cnt % 2 == 1 else 0.0
        best_h = project_feasible(h_tmp, best_n)
        best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent at parent resolution
    n = best_n
    h_work = best_h.copy()
    h_work, c5_work = fast_coordinate_descent(h_work, n, max_sweeps=80, seed=seed)
    if c5_work < best_c5:
        best_c5 = c5_work
        best_h = h_work.copy()

    # Phase 2: SLSQP at moderate resolution
    if remaining_time() > 120:
        slsq_n = min(n, 256)
        h_s = interp_to(best_h, best_n, slsq_n)
        h_s = project_feasible(h_s, slsq_n)
        for beta in [200, 1000, 5000]:
            if remaining_time() < 30:
                break
            h_s = smooth_slsqp(h_s, slsq_n, beta, 200)
            c5_s = compute_c5(h_s)
            # Transfer back to full resolution
            h_full = interp_to(h_s, slsq_n, n)
            h_full = project_feasible(h_full, n)
            c5_full = compute_c5(h_full)
            if c5_full < best_c5 - 1e-14:
                best_c5 = c5_full
                best_h = h_full.copy()

    # Phase 3: More coordinate descent
    if remaining_time() > 60:
        h_work = best_h.copy()
        h_work, c5_work = fast_coordinate_descent(h_work, best_n, max_sweeps=100, seed=seed + 100)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()

    # Phase 4: Try higher resolution
    if remaining_time() > 100:
        for hi_n in [int(best_n * 1.5), best_n * 2]:
            if remaining_time() < 50:
                break
            if hi_n % 2 != 0:
                hi_n += 1
            if hi_n > 1600:
                continue
            h_hi = interp_to(best_h, best_n, hi_n)
            h_hi = project_feasible(h_hi, hi_n)
            h_hi, c5_hi = fast_coordinate_descent(h_hi, hi_n, max_sweeps=40, seed=seed + 500)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi.copy()
                best_n = hi_n

    # Phase 5: Pair swap
    if remaining_time() > 40:
        h_work = best_h.copy()
        h_work, c5_work = pair_swap_descent(h_work, best_n, max_iters=3000, seed=seed)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()

    # Phase 6: Structured perturbations + re-optimize
    if remaining_time() > 80:
        for pert_iter in range(6):
            if remaining_time() < 20:
                break
            h_pert = structured_perturbation(best_h, best_n, 0.05, rng)
            h_pert, c5_pert = fast_coordinate_descent(h_pert, best_n, max_sweeps=50, seed=seed + 2000 + pert_iter)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()

    # Phase 7: Final coordinate descent polish
    if remaining_time() > 20:
        h_work = best_h.copy()
        h_work, c5_work = fast_coordinate_descent(h_work, best_n, max_sweeps=100, seed=seed + 9999)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()

    # Phase 8: Try a completely different initialization at current resolution
    if remaining_time() > 60:
        # Try a "block" pattern
        h_block = np.zeros(best_n)
        block_len = best_n // 4
        h_block[0:block_len] = 1.0
        h_block[best_n // 2:best_n // 2 + block_len] = 1.0
        h_block = project_feasible(h_block, best_n)
        h_block, c5_block = fast_coordinate_descent(h_block, best_n, max_sweeps=60, seed=seed + 5000)
        if c5_block < best_c5:
            best_c5 = c5_block
            best_h = h_block.copy()

        # Try random initialization
        h_rand = np.zeros(best_n)
        chosen = rng.choice(best_n, best_n // 2, replace=False)
        h_rand[chosen] = 1.0
        h_rand = project_feasible(h_rand, best_n)
        h_rand, c5_rand = fast_coordinate_descent(h_rand, best_n, max_sweeps=60, seed=seed + 6000)
        if c5_rand < best_c5:
            best_c5 = c5_rand
            best_h = h_rand.copy()

    # Phase 9: Final micro-optimization
    if remaining_time() > 10:
        h_work = best_h.copy()
        h_work, c5_work = fast_coordinate_descent(h_work, best_n, max_sweeps=50, seed=seed + 7777)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
