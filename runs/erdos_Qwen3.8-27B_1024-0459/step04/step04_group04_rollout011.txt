def continuous_refine(h, n, time_limit, seed):
    """Simple gradient-based refinement for non-binary solutions."""
    from scipy.optimize import minimize
    rng = np.random.RandomState(seed)

    h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def objective(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        # Soft max approximation
        beta = 500
        shifted = c - m
        val = (m + np.log(np.sum(np.exp(np.clip(beta * shifted, -50, 50)))) / beta)
        return val * dx

    def constraint(x):
        return np.sum(x) - target

    start = time.time()
    for beta_scale in [200, 500, 1000, 2000]:
        if time.time() - start > time_limit * 0.8:
            break

        def obj_scaled(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(c)
            shifted = c - m
            cl = np.clip(beta_scale * shifted, -50, 50)
            return (m + np.log(np.sum(np.exp(cl))) / beta_scale) * dx

        try:
            res = minimize(obj_scaled, h, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': constraint}],
                          options={'maxiter': 300, 'ftol': 1e-14})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h = h_new.copy()
        except Exception:
            pass

    return h, best_c5

                h_init = make_binary(parent_h, parent_n, n_t, seed + n_t * 10 + si)
            elif si == 1:
                h_init = random_binary(n_t, seed + 100000 + n_t * 7 + si)
            else:
                # Thue-Morse inspired
                h_init = np.zeros(n_t)
                tm = 0
                for i in range(n_t):
                    tm = i ^ (i >> 1)
                    h_init[i] = 1.0 if (bin(tm).count('1') % 2 == 0) else 0.0
                # Fix count
                target = n_t // 2
                n_ones = int(np.sum(h_init))
                rng_fix = np.random.RandomState(seed + n_t + si)
                if n_ones > target:
                    ones = np.where(h_init == 1)[0]
                    h_init[ones[rng_fix.choice(len(ones), n_ones - target, replace=False)]] = 0
                elif n_ones < target:
                    zeros = np.where(h_init == 0)[0]
                    h_init[zeros[rng_fix.choice(len(zeros), target - n_ones, replace=False)]] = 1

            h_sa, c5_sa = sa_binary_fast(h_init, n_t, time_limit=t_per_start,
                                         seed=seed + 50000 + n_t * 13 + si * 1000,
                                         T_initial=6.0, alpha=0.9995)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_t

    # Phase 2: Targeted local search on best and nearby n values
    n_phase3 = sorted(set([best_n - 4, best_n - 2, best_n, best_n + 2, best_n + 4,
                            360, 364, 368, 372, 376]))
    n_phase3 = [n for n in n_phase3 if n % 2 == 0 and 200 <= n <= 500]

    for n_t in n_phase3:
        if rem() < 30:
            break
        if n_t == best_n:
            h_init = best_h.copy()
        else:
            h_init = make_binary(best_h, best_n, n_t, seed + n_t * 31)

        t_ts = min(60, max(10, rem() / (len(n_phase3) + 1)))
        h_ts, c5_ts = targeted_local_search(h_init, n_t, time_limit=t_ts, seed=seed + n_t * 17)
        if c5_ts < best_c5:
            best_c5 = c5_ts
            best_h = h_ts.copy()
            best_n = n_t

    # Phase 3: More SA restarts on the best n
    if rem() > 60:
        n_best = best_n
        for restart in range(5):
            if rem() < 20:
                break
            if restart == 0:
                h_init = best_h.copy()
            else:
                h_init = make_binary(best_h, best_n, best_n, seed + 90000 + restart * 777)
                # Add some perturbation
                rng_p = np.random.RandomState(seed + 99999 + restart)
                n_ones = int(np.sum(h_init))
                n_zeros = best_n - n_ones
                n_flip = best_n // 10
                for _ in range(n_flip):
                    a = int(rng_p.choice(np.where(h_init == 1)[0]))
                    b = int(rng_p.choice(np.where(h_init == 0)[0]))
                    h_init[a], h_init[b] = 0.0, 1.0

            t_sa = min(80, max(20, rem() / 2))
            h_sa, c5_sa = sa_binary_fast(h_init, best_n, time_limit=t_sa,
                                         seed=seed + 200000 + restart * 333,
                                         T_initial=4.0, alpha=0.9998)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = best_n

    # Phase 4: Targeted local search again
    if rem() > 30:
        t_ts2 = min(60, max(10, rem() * 0.5))
        h_ts, c5_ts = targeted_local_search(best_h, best_n, time_limit=t_ts2,
                                            seed=seed + 77777)
        if c5_ts < best_c5:
            best_c5 = c5_ts
            best_h = h_ts.copy()

    # Phase 5: Try continuous refinement if binary not optimal
    if rem() > 30:
        h_cont, c5_cont = continuous_refine(best_h, best_n, time_limit=min(60, rem() * 0.6),
                                            seed=seed + 55555)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()

    # Phase 6: Final polish
    if rem() > 15:
        h_final, c5_final = targeted_local_search(best_h, best_n, time_limit=min(30, rem() * 0.7),
                                                  seed=seed + 88888)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    else:
        best_c5 = final_c5

    # Try evaluate_erdos_solution
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and np.isfinite(eval_c5) and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    # One more check: validate constraints
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10), "h out of bounds"
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-6, "sum constraint violated"
    best_h = np.clip(best_h, 0.0, 1.0)

    return (best_h, best_c5, best_n)
