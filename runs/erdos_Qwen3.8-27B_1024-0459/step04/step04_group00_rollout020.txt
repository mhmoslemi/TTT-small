Let me also add one more trick: **antagonistic pair search**. 

The idea: if the top correlation is at shift d, then the "bad" pairs are (i, i+d) where h[i+d] > 0.5 and h[i] < 0.5. For each such pair, I want to either decrease h[i+d] or increase h[i]. But I also need to maintain the sum constraint, so I should do both: decrease h[i+d] and increase h[i] (or decrease some h[a] and increase some h[b] for different a, b).

The most effective move is to find a pair (i, i+d) that is "very bad" (large h[i+d] and small h[i]) and a pair (j, j+d') that is "very good" (small h[j+d'] and large h[j]), and swap the "1-ness" between them.

This is essentially what the 2D coordinate search does, but guided by the correlation structure.

_GB = 1.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / n)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = int(free.sum())
        if nf > 0:
            h[free] += diff / nf
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def rtime():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)

def _obj(x):
    n = _GN
    h = np.clip(x, 0.0, 1.0)
    c = compute_corr(h)
    m = c.max()
    s = np.clip(_GB * (c - m), -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(s))) / _GB) * (2.0 / n)

def _grad(x):
    n = _GN
    h = np.clip(x, 0.0, 1.0)
    c = compute_corr(h)
    m = c.max()

    ev = np.exp(s)
    w = ev / ev.sum() * (2.0 / n)

    return np.ones(_GN)

    try:

        hn = project_feasible(r.x, n)
        if compute_c5(hn) < compute_c5(h0):

    return h0

                    if abs(h2i - h2j) < 1e-8:
                        continue
                    for f in [0.3, 0.5, 0.7]:
                        d = f * (h2i - h2j)
                        ht = h.copy()
                        ht[i] -= d
                        ht[j] += d
                        ht = np.clip(ht, 0.0, 1.0)
                        ht = project_feasible(ht, n)
                        ct = compute_c5(ht)
                        if ct < bc - 1e-15:
                            h = ht
                            bc = ct
                            imp2 = True
                            break
                    if imp2:
                        break
                if imp2:
                    break
            if not imp2:
                break
    return h, bc

def interp(a, sn, dn):
    if sn == dn:
        return a.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, a)
