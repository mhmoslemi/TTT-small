            if m < n:
                slopes[:n - m, idx] += (1.0 - h[m:m + n - m])
                slopes[m:m + n - m, idx] -= h[:n - m]
    return slopes

            # Find best improvement swap
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            
            best_delta = -1e-10
            best_i, best_j = -1, -1
            
            # Try swaps biased toward the worst shift
            n_try = 300
            for _ in range(n_try):
                if offset >= 0 and offset < n:
                    # Bad: h[i]=1, h[i+offset]=0
                    bad_i = [x for x in ones if x + offset < n and h_bin[x + offset] == 0]
                    if len(bad_i) > 0 and rng.rand() < 0.6:
                        i = bad_i[rng.randint(len(bad_i))]
                    else:
                        i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                elif offset < 0 and abs(offset) < n:
                    m = -offset
                    bad_j = [x for x in zeros if x + m < n and h_bin[x + m] == 1]
                    if len(bad_j) > 0 and rng.rand() < 0.6:
                        j = bad_j[rng.randint(len(bad_j))]
                    else:
                        j = zeros[rng.randint(len(zeros))]
                    i = ones[rng.randint(len(ones))]
                else:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                
                if i == j:
                    continue
                h_bin[i] = 0
                h_bin[j] = 1
                new_corr = compute_corr(h_bin)
                new_max = float(np.max(new_corr))
                delta = cur_max - new_max
                if delta > best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
                    best_corr = new_corr.copy()
                    best_max = new_max
                    # Don't revert, keep the best
                else:
                    h_bin[i] = 1
                    h_bin[j] = 0
            
            if best_i >= 0 and best_delta > 0:
                h_bin[best_i] = 0
                h_bin[best_j] = 1
                corr = best_corr
                cur_max = best_max
                stall = 0
            else:
                stall += 1
                if stall > 200:
                    # Random kick
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h_bin[i] = 0
                    h_bin[j] = 1
                    corr = compute_corr(h_bin)
                    cur_max = float(np.max(corr))
                    stall = 0
        
        c5_bin = float(np.max(compute_corr(h_bin))) * dx
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
    
    return best_h, best_c5


def lcos_optimize(h, n, max_iters=200, time_limit=30):
    """Level-setting / equalization optimization for top correlations."""
    h = h.copy()
    start = time.time()
    dx_val = 2.0 / n
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break
        
        corr = compute_corr(h)
        c_max = np.max(corr)
        # Find all shifts within 1% of max (active set)
        threshold = c_max - 0.01 * c_max
        active = np.where(corr >= threshold)[0]
        
        if len(active) == 0:
            break
        if len(active) == 1:
            break
        
        # Compute gradient: direction that reduces max and raises min of active set
        g_max = np.zeros(n)
        g_min = np.zeros(n)
        w_max = np.zeros(len(active))
        w_min = np.zeros(len(active))
        
        # Weight by proximity to max/min
        for ai in active:
            w_max[ai] = corr[ai] / np.sum(corr[active])
            w_min[ai] = 1.0 / len(active)
        
        for ai, ai_idx in zip(active, range(len(active))):
            g = corr_grad_per_shift(h, n, int(ai))
            g_max += w_max[ai_idx] * g
            g_min += w_min[ai_idx] * g
        
        # Direction: reduce weighted max, increase weighted min
        d = g_min - g_max
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-14:
            break
        d /= dnorm
        
        # Line search
        improved = False
        for step in [0.003, 0.005, 0.01, 0.02, 0.002, 0.007, 0.015, 0.03, 0.05]:
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-12:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        
        if not improved:
            # Try opposite direction
            for step in [0.003, 0.005, 0.01, 0.02]:
                h_trial = h - step * d
                h_trial = project_feasible(h_trial, n)
                c_new = compute_c5(h_trial)
                if c_new < best_c5 - 1e-12:
                    h = h_trial
                    best_c5 = c_new
                    improved = True
                    break
        
        if not improved:
            break
    
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 312
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP with increasing beta (main continuous optimizer)
    for beta in [200, 500, 1500, 4000, 10000, 25000, 60000]:
        if remaining() < 12:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Pair coordinate descent (intensive)
    if remaining() > 60:
        h_pc, c5_pc = pair_coord_descent_v2(h, n, max_passes=600,
                                            time_limit=min(150, remaining() - 20), rng=rng)
        c5_pc = compute_c5(h_pc)
        if c5_pc < best_c5:
            best_c5 = c5_pc
            best_h = h_pc.copy()
            h = h_pc.copy()

    # Phase 3: SLSQP again after coord descent
    for beta in [10000, 30000, 80000, 200000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Active shift local search
    if remaining() > 40:
        h_as, c5_as = active_shift_local_search(h, n,
                                                time_limit=min(120, remaining() - 15),
                                                top_k=12, rng=rng)
        c5_as = compute_c5(h_as)
        if c5_as < best_c5:
            best_c5 = c5_as
            best_h = h_as.copy()
            h = h_as.copy()

    # Phase 5: Level-setting equalization
    if remaining() > 20:
        h_ls, c5_ls = lcos_optimize(h, n, max_iters=300,
                                    time_limit=min(40, remaining() - 8))
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            h = h_ls.copy()

    # Phase 6: Simulated annealing on continuous
    if remaining() > 50:
        h_sa, c5_sa = simulated_annealing_continuous(h, n,
                                                     time_limit=min(100, remaining() - 20),
                                                     rng=rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            h = h_sa.copy()
        # Quick SLSQP after SA
        if remaining() > 15:
            h_new = slsqp_smooth(h, n, 50000, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 7: Try different n values
    n_candidates = sorted(set([
        n + 4, n + 8, n + 16, n + 24, n + 32, n + 48,
        n - 4, n - 8, n - 16,
        320, 336, 352, 368, 384,
        288, 296, 304
    ]))
    n_candidates = [nn for nn in n_candidates if 200 <= nn <= 600 and nn != n]
    # Prioritize nearby values
    n_candidates.sort(key=lambda x: abs(x - n))
    n_candidates = n_candidates[:8]

    for n_t in n_candidates:
        if remaining() < 45:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        
        # SLSQP on new n
        h_opt = h_feas
        for beta in [1000, 5000, 20000]:
            if remaining() < 10:
                break
            h_opt = slsqp_smooth(h_opt, n_t, beta, maxiter=350)
        
        # Quick coord descent
        if remaining() > 25:
            h_opt, _ = pair_coord_descent_v2(h_opt, n_t, max_passes=200,
                                              time_limit=min(30, remaining() - 10), rng=rng)
        
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 8: Final intensive refinement on best
    n = best_n
    h = best_h.copy()

    # Final SLSQP with very high beta
    for beta in [50000, 150000, 500000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final coord descent
    if remaining() > 20:
        h_pc, c5_pc = pair_coord_descent_v2(h, n, max_passes=400,
                                            time_limit=min(60, remaining() - 15), rng=rng)
        c5_pc = compute_c5(h_pc)
        if c5_pc < best_c5:
            best_c5 = c5_pc
            best_h = h_pc.copy()

    # Final active shift search
    if remaining() > 15:
        h_as, c5_as = active_shift_local_search(best_h, best_n,
                                                time_limit=min(30, remaining() - 8),
                                                top_k=8, rng=rng)
        c5_as = compute_c5(h_as)
        if c5_as < best_c5:
            best_c5 = c5_as
            best_h = h_as.copy()

    # Final lcos
    if remaining() > 10:
        h_ls, c5_ls = lcos_optimize(best_h, best_n, max_iters=200,
                                    time_limit=min(15, remaining() - 5))
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Final polish
    if remaining() > 8:
        h_new = slsqp_smooth(best_h, best_n, 100000, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Validate and finalize
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
