        h = project_feasible(h0, n)
    _G['pen'] = 0.0
    return h

def active_set_refine(h, n, tlimit, max_rounds=40):
    """Iteratively refine using identified active lags."""
    c5 = compute_c5(h)
    h_best = h.copy()
    for rnd in range(max_rounds):
        if tlimit() < 5.0:
            break
        corr = np.correlate(h_best, 1.0 - h_best, mode='full')
        c_max = np.max(corr)
        active_tol = 1e-4
        active_mask = corr >= c_max - active_tol
        n_active = int(np.sum(active_mask))
        
        # Try topk with k = min(n_active, 3)
        k = min(n_active, 3)
        h_new = slsqp_opt(h_best, n, beta=5000, maxiter=300, mode='topk', topk_val=k)
        c_new = compute_c5(h_new)
        if c_new < c5 - 1e-13:
            c5 = c_new
            h_best = h_new
        
        # Try weighted objective on active lags
        weights = np.zeros(2 * n - 1)
        weights[active_mask] = 1.0 / max(n_active, 1)
        h_new2 = slsqp_opt(h_best, n, beta=1.0, maxiter=200, mode='weighted', weights=weights)
        c_new2 = compute_c5(h_new2)
        if c_new2 < c5 - 1e-13:
            c5 = c_new2
            h_best = h_new2
        
        # Try L-BFGS with topk
        if tlimit() > 10:
            h_new3 = lbfgs_opt(h_best, n, beta=5000, maxiter=500, pen=10000, mode='topk', topk_val=k)
            c_new3 = compute_c5(h_new3)
            if c_new3 < c5 - 1e-13:
                c5 = c_new3
                h_best = h_new3
    
    return h_best, c5

def fine_coord_descent(h, n, rng, tlimit, max_sweeps=60):
    """Fine coordinate descent with many trial values."""
    c5 = compute_c5(h)
    h_best = h.copy()
    # Precompute trial values
    trial_vals = np.concatenate([[0.0, 1.0], np.linspace(0.0, 1.0, 33)[1:-1]])
    for sweep in range(max_sweeps):
        if tlimit() < 2.0:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if tlimit() < 0.5:
                break
            v0 = h_best[i]
            for v in trial_vals:
                if tlimit() < 0.3:
                    break
                if abs(v - v0) < 1e-12:
                    continue
                trial = h_best.copy()
                trial[i] = v
                trial = project_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5 - 1e-12:
                    c5 = c
                    h_best = trial.copy()
                    improved = True
        if not improved:
            break
    return h_best, c5

def pairwise_swap_descent(h, n, rng, tlimit, max_sweeps=20):
    """Try pairwise modifications: add d to one, subtract d from another."""
    c5 = compute_c5(h)
    h_best = h.copy()
    for sweep in range(max_sweeps):
        if tlimit() < 2.0:
            break
        improved = False
        for _ in range(200):
            if tlimit() < 1.0:
                break
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            d = float(rng.choice([0.05, 0.1, 0.15, -0.05, -0.1, -0.15]))
            trial = h_best.copy()
            trial[i] = np.clip(trial[i] + d, 0.0, 1.0)
            trial[j] = np.clip(trial[j] - d, 0.0, 1.0)
            trial = project_feasible(trial, n)
            c = compute_c5(trial)
            if c < c5 - 1e-12:
                c5 = c
                h_best = trial.copy()
                improved = True
        if not improved:
            break
    return h_best, c5

def run(seed=42, budget_s=1300, **kwargs):
    _G['beta'] = 1.0
    _G['tgt'] = 0.0
    _G['dx'] = 1.0
    _G['pen'] = 0.0
    _G['topk'] = 2
    _G['weights'] = None
    t_start = time.time()
    budget = float(budget_s)
    rng = np.random.RandomState(seed)

    def tlimit():
        return max(0.0, budget - (time.time() - t_start) - 5.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        parent_n = 256
        best_h = make_tm(parent_n)
        best_h = project_feasible(best_h, parent_n)
        best_c5 = compute_c5(best_h)
        best_n = parent_n
        print("No parent. TM: n={}, C5={:.8f}".format(parent_n, best_c5))

    # Phase 1: L-BFGS-B with penalty (fast)
    if tlimit() > 120:
        print("\n=== Phase 1: L-BFGS-B, n={} ===".format(best_n))
        n = best_n
        h_cur = best_h.copy()
        c5_cur = best_c5

        # LSE mode with increasing penalty
        for beta, pen in [(100, 50), (300, 200), (500, 500), (1000, 1000), (2000, 5000), (5000, 10000)]:
            if tlimit() < 10:
                break
            h_new = lbfgs_opt(h_cur, n, beta, maxiter=800, pen=pen, mode='lse')
            c_new = compute_c5(h_new)
            if c_new < c5_cur - 1e-14:
                c5_cur = c_new
                h_cur = h_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    print("  LSE beta={}, pen={}: C5={:.8f}".format(beta, pen, best_c5))

        # Topk mode
        for k in [2, 3]:
            for beta, pen in [(500, 500), (2000, 5000), (5000, 10000)]:
                if tlimit() < 5:
                    break
                h_new = lbfgs_opt(h_cur, n, beta, maxiter=500, pen=pen, mode='topk', topk_val=k)
                c_new = compute_c5(h_new)
                if c_new < c5_cur - 1e-14:
                    c5_cur = c_new
                    h_cur = h_new
                    if c5_cur < best_c5:
                        best_c5 = c5_cur
                        best_h = h_cur.copy()
                        print("  TOPK k={} beta={} pen={}: C5={:.8f}".format(k, beta, pen, best_c5))

        print("  Phase 1 done: C5={:.8f}".format(best_c5))

    # Phase 2: SLSQP with topk (handles constraint exactly, smoother objective)
    if tlimit() > 100:
        print("\n=== Phase 2: SLSQP topk, n={} ===".format(best_n))
        n = best_n
        h_cur = best_h.copy()
        c5_cur = best_c5

        for k in [1, 2, 3]:
            for beta in [1000, 5000, 10000]:
                if tlimit() < 10:
                    break
                h_new = slsqp_opt(h_cur, n, beta=beta, maxiter=400, mode='topk', topk_val=k)
                c_new = compute_c5(h_new)
                if c_new < c5_cur - 1e-14:
                    c5_cur = c_new
                    h_cur = h_new
                    if c5_cur < best_c5:
                        best_c5 = c5_cur
                        best_h = h_cur.copy()
                        print("  SLSQP topk={} beta={}: C5={:.8f}".format(k, beta, best_c5))

        # LSE mode
        for beta in [2000, 5000, 10000, 20000]:
            if tlimit() < 10:
                break
            h_new = slsqp_opt(h_cur, n, beta=beta, maxiter=400, mode='lse')
            c_new = compute_c5(h_new)
            if c_new < c5_cur - 1e-14:
                c5_cur = c_new
                h_cur = h_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    print("  SLSQP lse beta={}: C5={:.8f}".format(beta, best_c5))

        print("  Phase 2 done: C5={:.8f}".format(best_c5))

    # Phase 3: Active set refinement
    if tlimit() > 60:
        print("\n=== Phase 3: Active set, n={} ===".format(best_n))
        n = best_n
        h_cur, c5_cur = active_set_refine(best_h, n, tlimit, max_rounds=30)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            print("  After active set: C5={:.8f}".format(best_c5))

    # Phase 4: Fine coordinate descent
    if tlimit() > 40:
        print("\n=== Phase 4: Fine coord descent, n={} ===".format(best_n))
        n = best_n
        best_h, best_c5 = fine_coord_descent(best_h, n, rng, tlimit, max_sweeps=60)
        print("  After fine coord: C5={:.8f}".format(best_c5))

    # Phase 5: Pairwise swap descent
    if tlimit() > 30:
        print("\n=== Phase 5: Pairwise swap, n={} ===".format(best_n))
        n = best_n
        best_h, best_c5 = pairwise_swap_descent(best_h, n, rng, tlimit, max_sweeps=20)
        print("  After pairwise: C5={:.8f}".format(best_c5))

    # Phase 6: Try n=512
    if tlimit() > 100:
        print("\n=== Phase 6: n=512 ===")
        n_hi = 512
        h_hi = project_feasible(interp_h(best_h, best_n, n_hi), n_hi)
        c5_hi = compute_c5(h_hi)
        print("  Interpolated n=512: C5={:.8f}".format(c5_hi))

        if c5_hi < best_c5 + 0.002:
            for beta, pen in [(200, 100), (500, 500), (1000, 2000), (3000, 10000)]:
                if tlimit() < 15:
                    break
                h_new = lbfgs_opt(h_hi, n_hi, beta, maxiter=300, pen=pen, mode='lse')
                c_new = compute_c5(h_new)
                if c_new < c5_hi - 1e-14:
                    c5_hi = c_new
                    h_hi = h_new
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h_new.copy()
                    best_n = n_hi
                    print("  n=512 LSE beta={}: C5={:.8f}".format(beta, best_c5))

            # Topk at n=512
            if tlimit() > 30 and best_n == n_hi:
                for k in [2, 3]:
                    if tlimit() < 15:
                        break
                    h_new = lbfgs_opt(h_hi, n_hi, 3000, maxiter=300, pen=10000, mode='topk', topk_val=k)
                    c_new = compute_c5(h_new)
                    if c_new < c5_hi - 1e-14:
                        c5_hi = c_new
                        h_hi = h_new
                    if c_new < best_c5:
                        best_c5 = c_new
                        best_h = h_new.copy()
                        best_n = n_hi
                        print("  n=512 TOPK k={}: C5={:.8f}".format(k, best_c5))

            # Coord descent at n=512
            if tlimit() > 20 and best_n == n_hi:
                best_h, best_c5 = fine_coord_descent(best_h, best_n, rng, tlimit, max_sweeps=20)
                print("  n=512 after coord: C5={:.8f}".format(best_c5))

        print("  Phase 6 done: best C5={:.8f}".format(best_c5))

    # Phase 7: Final SLSQP polish at best resolution
    if tlimit() > 40:
        print("\n=== Phase 7: Final polish, n={} ===".format(best_n))
        n = best_n
        h_cur = best_h.copy()
        c5_cur = best_c5

        for beta in [10000, 20000, 50000]:
            if tlimit() < 10:
                break
            h_new = slsqp_opt(h_cur, n, beta=beta, maxiter=600, mode='lse')
            c_new = compute_c5(h_new)
            if c_new < c5_cur - 1e-14:
                c5_cur = c_new
                h_cur = h_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    print("  Polish beta={}: C5={:.8f}".format(beta, best_c5))

        # Topk polish
        for k in [2]:
            if tlimit() < 10:
                break
            h_new = slsqp_opt(h_cur, n, beta=10000, maxiter=400, mode='topk', topk_val=k)
            c_new = compute_c5(h_new)
            if c_new < c5_cur - 1e-14:
                c5_cur = c_new
                h_cur = h_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    print("  Polish topk={}: C5={:.8f}".format(k, best_c5))

        # Final fine coord
        if tlimit() > 15:
            best_h, best_c5 = fine_coord_descent(best_h, best_n, rng, tlimit, max_sweeps=40)
            print("  After final coord: C5={:.8f}".format(best_c5))

    # Phase 8: Perturbation escape
    if tlimit() > 50:
        print("\n=== Phase 8: Perturbation ===")
        n = best_n
        for attempt in range(8):
            if tlimit() < 12:
                break
            h_pert = best_h.copy()
            n_pert = rng.randint(2, 8)
            idxs = rng.choice(n, size=n_pert, replace=False)
            for idx in idxs:
                h_pert[idx] = np.clip(h_pert[idx] + rng.randn() * 0.15, 0.0, 1.0)
            h_pert = project_feasible(h_pert, n)
            c5_p = compute_c5(h_pert)

            if c5_p < best_c5 * 1.02:
                h_cur = h_pert.copy()
                c5_cur = c5_p
                # Quick re-optimization
                for bv in [2000, 8000]:
                    if tlimit() < 5:
                        break
                    h_new = slsqp_opt(h_cur, n, beta=bv, maxiter=200, mode='topk', topk_val=2)
                    c_new = compute_c5(h_new)
                    if c_new < c5_cur - 1e-13:
                        c5_cur = c_new
                        h_cur = h_new
                if c5_cur < best_c5 - 1e-12:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n
                    print("  Perturb {}: C5={:.8f}".format(attempt, best_c5))

    # Phase 9: Ultra-fine final coord descent
    if tlimit() > 10:
        best_n = len(best_h)
        best_h, best_c5 = fine_coord_descent(best_h, best_n, rng, tlimit, max_sweeps=30)
        print("  After ultra-fine coord: C5={:.8f}".format(best_c5))

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            ec = float(eval_c5)
            if ec < final_c5:
                final_c5 = ec
    except Exception:
        pass

    _G['pen'] = 0.0
    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
