                grad[aoff:aoff+npair] -= w * x[:npair]
    grad *= dx
    return val, grad

def slsqp_step(h, n, beta, maxiter=500):

    _N_GLOBAL = n
    _BETA_GLOBAL = beta
    
    def obj(x):
        return smooth_obj_and_grad(x, n, beta)[0]
    
    def obj_grad(x):
        return smooth_obj_and_grad(x, n, beta)[1]

        return np.sum(x) - n / 2.0
    
    def cons_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

            if compute_c5(trial) <= compute_c5(h) + 1e-14:

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src)

def coord_descent(h, n, max_sweeps=50, sweep_range=0.4):

    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best).astype(float)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        order = np.random.permutation(n)
        
        for i in order:
            if remaining_time() < 1:
                break

            a = np.zeros(ncorr)
            b = np.zeros(ncorr)  # quadratic coeff
            t0 = best[i]

                k = n - 1 + off
                if off == 0:
                    a[k] += 1.0
                    b[k] += -1.0
                else:
                    if 0 <= i + off < n:
                        a[k] += (1.0 - best[i + off])
                    if 0 <= i - off < n:
                        a[k] += (-best[i - off])

            def obj_at_t(t):
                vals = base + a * t + b * t * t
                return np.max(vals) * dx

            t_best = t0
            c5_local = best_c5
            
            # Wider candidate set
            candidates = [0.0, 1.0, 0.5, t0 + 0.25, t0 - 0.25, 
                         t0 + 0.5, t0 - 0.5, 1.0 - t0, t0 * 0.5, t0 * 2.0]
            for c in candidates:
                c = max(0.0, min(1.0, c))
                if abs(c - t0) < 1e-12:
                    continue
                c_val = obj_at_t(c)
                if c_val < c5_local - 1e-15:
                    c5_local = c_val
                    t_best = c
            
            # If improvement, refine with golden section in a wider window
            if abs(t_best - t0) > 1e-10:
                lo = max(0.0, min(t0, t_best) - sweep_range)
                hi = min(1.0, max(t0, t_best) + sweep_range)
                if hi - lo > 0.01:
                    gr = (np.sqrt(5.0) - 1.0) / 2.0
                    c1 = hi - gr * (hi - lo)
                    c2 = lo + gr * (hi - lo)
                    f1 = obj_at_t(c1)
                    f2 = obj_at_t(c2)

                            c2 = c1
                            f2 = f1
                            c1 = hi - gr * (hi - lo)
                            f1 = obj_at_t(c1)
                        else:
                            lo = c1
                            c1 = c2
                            f1 = f2
                            c2 = lo + gr * (hi - lo)
                            f2 = obj_at_t(c2)
                    t_gs = (lo + hi) / 2.0
                    c_gs = obj_at_t(t_gs)
                    if c_gs < c5_local:
                        c5_local = c_gs
                        t_best = t_gs
            
            if c5_local < best_c5 - 1e-14:
                best[i] = t_best
                best_corr = compute_corr(best).astype(float)
                best_c5 = c5_local
                improved = True
        
        if not improved:
            break
    
    return best, best_c5

def binary_rounding(h, n, threshold=0.03):

            best_ti = ti

                    f2 = obj_2d(c2, best_tj)
                    if f1 < f2:
                        hi = c2
                    else:
                        lo = c1
                best_ti = (lo + hi) / 2.0

                    f2 = obj_2d(best_ti, c2)
                    if f1 < f2:
                        hi = c2
                    else:
                        lo = c1

                    best_local = c_final
                
                if best_local < best_c5 - 1e-14:
                    best[i] = best_ti
                    best[i+1] = best_tj
                    best_c5 = best_local
                    improved = True

    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(interior):
            cnt = float(np.sum(interior))
            step = diff / cnt
            if step > 0:
                max_s = (1.0 - h[interior]).min()
                step = min(step, max_s * 0.999)
            else:
                max_s = h[interior].min()
                step = max(step, -max_s * 0.999)
            h[interior] += step
        else:
            idx = int(np.argmin(np.abs(h - 0.5)))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

        h += diff / n

        pass
    return h.copy()

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src)

def coord_descent(h, n, max_sweeps=50, rng=None):
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best).astype(float)
    dx = 2.0 / n
    ncorr = 2 * n - 1

    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for i in order:
            if remaining_time() < 1:
                break
            t0 = best[i]
            a = np.zeros(ncorr)
            b = np.zeros(ncorr)
            for off in range(-(n-1), n):
                k = n - 1 + off
                if off == 0:
                    a[k] += 1.0
                    b[k] += -1.0
                else:
                    if 0 <= i + off < n:
                        a[k] += (1.0 - best[i + off])
                    if 0 <= i - off < n:
                        a[k] += (-best[i - off])

            base = best_corr - a * t0 - b * t0 * t0

            def obj_at_t(t):
                vals = base + a * t + b * t * t
                return float(np.max(vals) * dx)

            t_best = t0
            c5_local = best_c5
            candidates = [0.0, 1.0, 0.5, t0+0.3, t0-0.3, t0+0.6, t0-0.6,
                         1.0-t0, t0*0.5, t0*2.0, t0+0.15, t0-0.15]
            for c in candidates:
                c = max(0.0, min(1.0, c))
                if abs(c - t0) < 1e-12:
                    continue
                cv = obj_at_t(c)
                if cv < c5_local - 1e-15:
                    c5_local = cv
                    t_best = c

            if abs(t_best - t0) > 1e-10:
                lo = max(0.0, min(t0, t_best) - 0.4)
                hi = min(1.0, max(t0, t_best) + 0.4)
                if hi - lo > 0.01:
                    gr = (np.sqrt(5.0) - 1.0) / 2.0
                    c1 = hi - gr * (hi - lo)
                    c2 = lo + gr * (hi - lo)
                    f1 = obj_at_t(c1)
                    f2 = obj_at_t(c2)
                    for _ in range(50):
                        if f1 < f2:
                            hi = c2; c2 = c1; f2 = f1
                            c1 = hi - gr * (hi - lo); f1 = obj_at_t(c1)
                        else:
                            lo = c1; c1 = c2; f1 = f2
                            c2 = lo + gr * (hi - lo); f2 = obj_at_t(c2)
                    t_gs = (lo + hi) / 2.0
                    c_gs = obj_at_t(t_gs)
                    if c_gs < c5_local:
                        c5_local = c_gs
                        t_best = t_gs

            if c5_local < best_c5 - 1e-14:
                best[i] = t_best
                best_corr = compute_corr(best).astype(float)
                best_c5 = c5_local
                improved = True

        if not improved:
            break

    return best, best_c5

def binary_rounding(h, n, threshold=0.04):
    best = h.copy()
    best_c5 = compute_c5(best)
    changed = True
    while changed:
        changed = False
        for i in range(n):
            if abs(best[i]) < threshold:
                trial = best.copy()
                trial[i] = 0.0
                trial = project_feasible(trial, n)
                cv = compute_c5(trial)
                if cv < best_c5 - 1e-15:
                    best = trial; best_c5 = cv; changed = True
            elif abs(best[i] - 1.0) < threshold:
                trial = best.copy()
                trial[i] = 1.0
                trial = project_feasible(trial, n)
                cv = compute_c5(trial)
                if cv < best_c5 - 1e-15:
                    best = trial; best_c5 = cv; changed = True
    return best, best_c5

def perturbation_search(h, n, rng, max_iters=50):
    best = h.copy()
    best_c5 = compute_c5(best)
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        # Pick 3-8 random coordinates to perturb
        n_pert = rng.randint(3, 9)
        idx = rng.choice(n, n_pert, replace=False)
        trial = best.copy()
        for i in idx:
            trial[i] = rng.uniform(0.0, 1.0)
        trial = project_feasible(trial, n)
        cv = compute_c5(trial)
        if cv < best_c5:
            # Local refine
            h_r, c5_r = coord_descent(trial, n, max_sweeps=15, rng=rng)
            if c5_r < best_c5:
                best = h_r
                best_c5 = c5_r
    return best, best_c5

def multi_beta_slsqp(h, n, betas, maxiter=500):
    current = h.copy()
    for beta in betas:
        if remaining_time() < 15:
            break
        new = slsqp_step(current, n, beta, maxiter)
        cv = compute_c5(new)
        cv_old = compute_c5(current)
        if cv < cv_old - 1e-16:
            current = new
    return current

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        h_tmp = np.zeros(best_n)
        for i in range(best_n):
            v = i
            cnt = 0
            while v:
                cnt ^= (v & 1)
                v >>= 1
            h_tmp[i] = 1.0 if cnt == 1 else 0.0
        best_h = project_feasible(h_tmp, best_n)
        best_c5 = compute_c5(best_h)

    # Phase 1: Multi-beta SLSQP at parent resolution
    n = best_n
    h = best_h.copy()
    h = multi_beta_slsqp(h, n, [200, 500, 1000, 3000, 8000, 20000, 50000, 100000], 600)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        best_n = n

    # Phase 2: Coordinate descent
    if remaining_time() > 30:
        hc, cc = coord_descent(best_h, best_n, max_sweeps=80, rng=rng)
        if cc < best_c5:
            best_c5 = cc
            best_h = hc.copy()

    # Phase 3: Binary rounding + coord descent
    if remaining_time() > 25:
        hb, cb = binary_rounding(best_h, best_n, threshold=0.05)
        if cb < best_c5:
            best_c5 = cb
            best_h = hb.copy()
        if remaining_time() > 25:
            hc, cc = coord_descent(best_h, best_n, max_sweeps=50, rng=rng)
            if cc < best_c5:
                best_c5 = cc
                best_h = hc.copy()

    # Phase 4: SLSQP again after coord descent
    if remaining_time() > 30:
        h2 = multi_beta_slsqp(best_h, best_n, [5000, 20000, 80000], 500)
        c52 = compute_c5(h2)
        if c52 < best_c5:
            best_c5 = c52
            best_h = h2.copy()

    # Phase 5: Try different resolutions
    for rn in [320, 960, 1280]:
        if remaining_time() < 50:
            break
        if rn == best_n or rn % 2 != 0:
            continue
        hr = interp_to(best_h, best_n, rn)
        hr = project_feasible(hr, rn)
        cr = compute_c5(hr)
        hr = multi_beta_slsqp(hr, rn, [500, 2000, 8000, 30000], 400)
        cr = compute_c5(hr)
        if cr < best_c5:
            best_c5 = cr
            best_h = hr.copy()
            best_n = rn
        elif cr < best_c5 + 0.01:
            hc, cc = coord_descent(hr, rn, max_sweeps=25, rng=rng)
            if cc < best_c5:
                best_c5 = cc
                best_h = hc.copy()
                best_n = rn

    # Phase 6: Back to best_n, final SLSQP + CD
    if remaining_time() > 40:
        h3 = multi_beta_slsqp(best_h, best_n, [10000, 40000, 150000], 400)
        c53 = compute_c5(h3)
        if c53 < best_c5:
            best_c5 = c53
            best_h = h3.copy()

    if remaining_time() > 20:
        hc, cc = coord_descent(best_h, best_n, max_sweeps=60, rng=rng)
        if cc < best_c5:
            best_c5 = cc
            best_h = hc.copy()

    # Phase 7: Perturbation search
    if remaining_time() > 40:
        hp, cp = perturbation_search(best_h, best_n, rng, max_iters=40)
        if cp < best_c5:
            best_c5 = cp
            best_h = hp.copy()

    # Phase 8: Binary rounding + final CD
    if remaining_time() > 20:
        hb, cb = binary_rounding(best_h, best_n, threshold=0.02)
        if cb < best_c5:
            best_c5 = cb
            best_h = hb.copy()
        hc, cc = coord_descent(best_h, best_n, max_sweeps=40, rng=rng)
        if cc < best_c5:
            best_c5 = cc
            best_h = hc.copy()

    # Phase 9: Final SLSQP polish
    if remaining_time() > 10:
        hf = slsqp_step(best_h, best_n, 100000, 300)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf.copy()

    # Final: one more micro coord descent
    if remaining_time() > 5:
        hc, cc = coord_descent(best_h, best_n, max_sweeps=20, rng=rng)
        if cc < best_c5:
            best_c5 = cc
            best_h = hc.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if isinstance(eval_c5, float) and eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
