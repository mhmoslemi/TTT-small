    """Compute gradient of weighted sum of correlations.
    weights[k] corresponds to shift d = k - (n-1)
    """
    n = len(h)
    g = np.zeros(n)
    for k in range(2*n - 1):
        w = weights[k]
        if abs(w) < 1e-15:
            continue
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            if m <= 0:
                continue
            g[d:d+m] += w * (1.0 - h[:m])
            g[:m] -= w * h[d:d+m]
        else:
            a = -d
            m = n - a
            if m <= 0:
                continue
            g[a:a+m] += w * (1.0 - h[:m])
            g[:m] -= w * h[a:a+m]
    return g

Actually, the water-filling projection is:
- We want to find $p$ that minimizes $\|h - p\|^2$ subject to $0 \leq p[i] \leq 1, \sum p = target$.
- The solution is: $p[i] = \text{clip}(h[i] - \lambda, 0, 1)$ for some $\lambda$.
- Find $\lambda$ by bisection: $\sum \text{clip}(h[i] - \lambda, 0, 1) = target$.

    target = n / 2.0
    # Bisection on lambda
    lo, hi = h.min() - 1.0, h.max() + 1.0

        p = np.clip(h - mid, 0.0, 1.0)

```python
def smooth_max_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    # Only keep top values for efficiency
    sh = corr - m
    cl = np.clip(beta * sh, -100.0, 0.0)  # top values get 0 (exp=1), others get very negative
    ex = np.exp(cl)
    Z = np.sum(ex)
    w = ex / Z
    # Weighted gradient
    g = weighted_corr_grad(h, w)
    obj = (m + np.log(Z) / beta) * dx
    grad = g * dx
    return obj, grad

def weighted_corr_grad(h, w):
    """Gradient of sum_k w[k] * corr[k] with respect to h."""
    n = len(h)
    g = np.zeros(n)
    # Only process non-negligible weights
    thresh = 1e-10
    active = np.where(w > thresh)[0]
    for k in active:
        wk = w[k]
        d = k - (n - 1)
        if d >= 0:
            m = n - d
            if m <= 0: continue
            g[d:d+m] += wk * (1.0 - h[:m])
            g[:m] -= wk * h[d:d+m]
        else:
            a = -d
            m = n - a
            if m <= 0: continue
            g[a:a+m] += wk * (1.0 - h[:m])
            g[:m] -= wk * h[a:a+m]
    return g
```
