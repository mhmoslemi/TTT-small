    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    # Sort correlations descending
    sorted_corr = np.sort(corr)[::-1]
    top = sorted_corr[:top_k]
    t = top[0]
    # Log-sum-exp for the top
    m = top[0]
    lse = m + np.log(np.sum(np.exp(top - m))) / beta
    # Also add variance penalty
    var_pen = np.sum((top - t)**2) * 0.01
    return (lse + var_pen) * dx

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

        else:

        h = np.clip(h, 0.0, 1.0)

    exps = np.exp(np.clip(_G_BETA * shifted, -600, 600))

                nv = n - offset
                if nv <= 0:

    if remaining_time() > 50.0:
        print("\n--- Phase 9: Try n=192 ---")
        n_mid = 192
        h_mid = project_feasible(interp_to(best_h, best_n, n_mid), n_mid)
        h_bin_m, c5_bin_m = project_binary(h_mid, n_mid)
        if c5_bin_m < compute_c5(h_mid):
            h_mid = h_bin_m
        _G_N = n_mid
        _G_TGT = n_mid / 2.0
        _G_DX = 2.0 / n_mid
        h_mid, c5_mid = slsqp_refine(
            h_mid, n_mid,
            betas=[100, 500, 2000, 8000],
            maxiter_list=[200, 300, 400, 500]
        )
        h_bin_m2, c5_bin_m2 = project_binary(h_mid, n_mid)
        if c5_bin_m2 < c5_mid:
            c5_mid = c5_bin_m2
            h_mid = h_bin_m2
        if c5_mid < best_c5 - 1e-12:
            best_c5 = c5_mid
            best_h = h_mid.copy()
            best_n = n_mid
            print("  n=192: C5={:.10f} (improved!)".format(best_c5))
        else:
            print("  n=192: C5={:.10f} (no improvement)".format(c5_mid))

    # Phase 10: Try n=320
    if remaining_time() > 50.0:
        print("\n--- Phase 10: Try n=320 ---")
        n_fine = 320
        h_fine = project_feasible(interp_to(best_h, best_n, n_fine), n_fine)
        h_bin_f, c5_bin_f = project_binary(h_fine, n_fine)
        if c5_bin_f < compute_c5(h_fine):
            h_fine = h_bin_f
        _G_N = n_fine
        _G_TGT = n_fine / 2.0
        _G_DX = 2.0 / n_fine
        h_fine, c5_fine = slsqp_refine(
            h_fine, n_fine,
            betas=[100, 500, 2000],
            maxiter_list=[200, 300, 400]
        )
        if c5_fine < best_c5 - 1e-12:
            best_c5 = c5_fine
            best_h = h_fine.copy()
            best_n = n_fine
            print("  n=320: C5={:.10f} (improved!)".format(best_c5))
        else:
            print("  n=320: C5={:.10f} (no improvement)".format(c5_fine))

    # Phase 11: Final polish on best solution
    print("\n--- Phase 11: Final polish (n={}) ---".format(best_n))
    n = best_n
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n

    # Very high beta SLSQP
    best_h, best_c5 = slsqp_refine(
        best_h, n,
        betas=[10000, 30000, 60000, 100000],
        maxiter_list=[500, 600, 700, 800]
    )

    # Another binary projection + local search
    h_bin_final, c5_bin_final = project_binary(best_h, n)
    if c5_bin_final < best_c5:
        best_c5 = c5_bin_final
        best_h = h_bin_final
    best_h, best_c5 = binary_swap_search(best_h, n, max_rounds=2000, rng=rng)

    # Coordinate descent
    best_h, best_c5 = coordinate_descent(best_h, n, max_sweeps=6, rng=rng)

    # One more SLSQP
    if remaining_time() > 5.0:
        best_h, best_c5 = slsqp_refine(
            best_h, n,
            betas=[5000, 20000],
            maxiter_list=[400, 600]
        )

    print("  Final C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
