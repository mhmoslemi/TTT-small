def proj(h, n):

    target = n / 2.0
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-15:

                   'jac': lambda x: np.ones(best_n)}]
        if tl() > 15:
            h_cur = slsqp_opt(best_h, best_n, 2000, 200, bounds_b, cons_b)
            c5_cur = c5_val(h_cur)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
        print("  After Phase 6: C5={:.8f}".format(best_c5))

    # Phase 7: Coordinate descent
    if tl() > 15:
        print("\n=== Phase 7: Coord descent ===")
        best_h, best_c5 = coord_descent(best_h, best_n, tl, rng, sweeps=10)
        print("  After coord: C5={:.8f}".format(best_c5))

    # Phase 8: Final active-set + SLSQP polish
    if tl() > 15:
        print("\n=== Phase 8: Final polish ===")
        bounds_f = [(0.0, 1.0)] * best_n
        cons_f = [{'type': 'eq', 'fun': lambda x: np.sum(x) - best_n / 2.0,
                   'jac': lambda x: np.ones(best_n)}]
        for beta in [5000, 10000, 30000]:
            if tl() < 10:
                break
            h_cur = slsqp_opt(best_h, best_n, beta, 150, bounds_f, cons_f)
            c5_cur = c5_val(h_cur)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
        # Active set
        h_a, c5_a = active_set_opt(best_h, best_n, bounds_f, cons_f, 3, 100, tl)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a.copy()
        print("  After polish: C5={:.8f}".format(best_c5))

    # Phase 9: Tiny SA + final coord
    if tl() > 10:
        h_sa, c5_sa = sa_search(best_h, best_n, tl, rng, n_steps=1000)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
    if tl() > 8:
        best_h, best_c5 = coord_descent(best_h, best_n, tl, rng, sweeps=5)

    # Final: one more SLSQP with very high beta
    if tl() > 8:
        bounds_f = [(0.0, 1.0)] * best_n
        cons_f = [{'type': 'eq', 'fun': lambda x: np.sum(x) - best_n / 2.0,
                   'jac': lambda x: np.ones(best_n)}]
        h_cur = slsqp_opt(best_h, best_n, 50000, 100, bounds_f, cons_f)
        c5_cur = c5_val(h_cur)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()

    # Final projection and validation
    best_h = proj(best_h, best_n)
    final_c5 = c5_val(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
