        print("P5 PG2: C5={:.10f}".format(best_c5))

    # Phase 6: Very high beta SLSQP
    if remaining_time() > 25:
        best_h, best_c5 = do_slsqp(best_h, best_n, [30000, 50000, 100000])
        print("P6 SLSQP high: C5={:.10f}".format(best_c5))

    # Phase 7: Try nearby resolutions
    for dn in [2, -2, 4, -4, 6, -6]:
        n_try = best_n + dn
        if n_try < 128 or n_try % 2 != 0:
            continue
        if remaining_time() < 25:
            break
        h_t = project_feasible(interp_to(best_h, best_n, n_try), n_try)
        c5_t = compute_c5(h_t)
        h_t, c5_t = do_slsqp(h_t, n_try, [500, 3000])
        if remaining_time() > 10:
            h_t, c5_t = do_pg(h_t, n_try, 80, 3)
        if c5_t < best_c5 - 1e-13:
            best_h, best_c5, best_n = h_t, c5_t, n_try
            print("  P7 n={} improved: C5={:.10f}".format(n_try, c5_t))
        else:
            print("  P7 n={} no improve: {:.10f}".format(n_try, c5_t))

    # Phase 8: Final intensive polish at best resolution
    if remaining_time() > 20:
        best_h, best_c5 = do_pg(best_h, best_n, 150, 3)
        print("P8 PG final: C5={:.10f}".format(best_c5))

    if remaining_time() > 15:
        best_h, best_c5 = do_slsqp(best_h, best_n, [50000, 100000, 200000])
        print("P9 SLSQP final: C5={:.10f}".format(best_c5))

    if remaining_time() > 10:
        best_h, best_c5 = do_eq_grad(best_h, best_n)
        print("P10 EqGrad final: C5={:.10f}".format(best_c5))

    if remaining_time() > 8:
        best_h, best_c5 = do_pg(best_h, best_n, 80, 2)
        print("P11 PG top2: C5={:.10f}".format(best_c5))

    if remaining_time() > 8:
        best_h, best_c5 = do_slsqp(best_h, best_n, [200000, 500000])
        print("P12 SLSQP ultra: C5={:.10f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("FINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
