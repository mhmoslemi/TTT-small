from scipy.optimize import minimize
import time

_G_N = 0
_G_BETA = 1.0
_G_TGT = 0.0
_G_START = 0.0
_G_BUDGET = 1300.0
_G_DX = 1.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) > 0:
            h[idx[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h


def remaining_time():
    return max(0.0, _G_BUDGET - (time.time() - _G_START) - 3.0)


def obj_smooth(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(_G_BETA * shifted, -600, 600))
    return (m + np.log(np.sum(exps)) / _G_BETA) * dx


def con_sum(x):
    return np.sum(x) - _G_TGT


def subgrad_topk(h, n, top_k):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    if top_k >= n_corr:
        top_indices = np.arange(n_corr)
    else:
        top_indices = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for k_idx in top_indices:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            n_valid = n - offset
            if n_valid <= 0:
                continue
            grad[offset:offset + n_valid] += (1.0 - h[:n_valid])
            grad[:n_valid] -= h[offset:offset + n_valid]
        else:
            m = -offset
            n_valid = n - m
            if n_valid <= 0:
                continue
            grad[:n_valid] += (1.0 - h[m:m + n_valid])
            grad[m:m + n_valid] -= h[:n_valid]
    return grad, top_indices, corr


def lse_slsqp(h_init, n, beta_schedule, max_iter=500, ftol=1e-16):
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum}]
    h_curr = h_init.copy()
    c5_curr = compute_c5(h_curr)
    for beta in beta_schedule:
        if remaining_time() < 5.0:
            break
        _G_BETA = float(beta)
        try:
            res = minimize(obj_smooth, h_curr, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_curr - 1e-15:
                c5_curr = c5_new
                h_curr = h_new.copy()
        except Exception:
            pass
    return h_curr, c5_curr


def subgrad_linesearch(h_init, n, max_iter=500, step0=0.15, momentum=0.85,
                       top_k_schedule=None):
    h_best = h_init.copy()
    c5_best = compute_c5(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    step = step0
    patience = 0
    if top_k_schedule is None:
        top_k_schedule = [3, 5, 5, 8, 10, 15, 20, 30]
    for it in range(max_iter):
        if remaining_time() < 3.0:
            break
        tk = top_k_schedule[min(it * len(top_k_schedule) // max_iter, len(top_k_schedule) - 1)]
        grad, _, _ = subgrad_topk(h, n, tk)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad = grad / gn
        alpha = step
        improved = False
        for _ in range(15):
            v = momentum * v - alpha * grad
            h_trial = h + v
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_best - 1e-13:
                c5_best = c5_trial
                h_best = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
            alpha *= 0.5
        if improved:
            patience = 0
            step = min(alpha * 1.4, 0.5)
        else:
            patience += 1
            step *= 0.6
            v *= 0.4
            if patience > 15:
                break
    return h_best, c5_best


def simulated_anneal(h_init, n, rng, max_iters=3000, T0=0.04, T_end=0.0003):
    h_best = h_init.copy()
    c5_best = compute_c5(h_init)
    h_curr = h_init.copy()
    c5_curr = c5_best
    stall_count = 0
    for it in range(max_iters):
        if remaining_time() < 2.0:
            break
        T = T0 * (T_end / T0) ** (float(it) / float(max_iters))
        top_k = min(3 + it // 300, 25)
        grad, _, _ = subgrad_topk(h_curr, n, top_k)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad = grad / gn
        step_size = T * 1.5
        noise = rng.randn(n) * T * 0.4
        h_prop = h_curr - step_size * grad + noise
        h_prop = project_feasible(h_prop, n)
        c5_prop = compute_c5(h_prop)
        delta = c5_prop - c5_curr
        accept = delta < 0 or (T > 1e-8 and rng.rand() < np.exp(-delta / T))
        if accept:
            h_curr = h_prop
            c5_curr = c5_prop
            if c5_curr < c5_best - 1e-14:
                c5_best = c5_curr
                h_best = h_curr.copy()
                stall_count = 0
            else:
                stall_count += 1
        else:
            stall_count += 1
        if stall_count > 500 and T < 0.01:
            break
    return h_best, c5_best


def targeted_perturb(h_best, n, rng, num_rounds=20):
    c5_best = float(compute_c5(h_best))
    h_best = h_best.copy()
    _G_N = n
    _G_TGT = n / 2.0
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    for r in range(num_rounds):
        if remaining_time() < 8.0:
            break
        h_trial = h_best.copy()
        pt = r % 8
        if pt == 0:
            # Block flip
            L = rng.randint(3, min(15, n // 5))
            start = rng.randint(0, n - L)

        elif pt == 1:
            # Block swap
            L = rng.randint(3, min(10, n // 10))
            s1 = rng.randint(0, n // 3)
            s2 = s1 + rng.randint(n // 4, n - n // 4 - L)
            tmp = h_trial[s1:s1 + L].copy()
            h_trial[s1:s1 + L] = h_trial[s2:s2 + L]
            h_trial[s2:s2 + L] = tmp
        elif pt == 2:
            # Gaussian noise
            h_trial += rng.randn(n) * 0.02
        elif pt == 3:
            # Random subset randomize
            subset = rng.choice(n, size=n // 8, replace=False)
            h_trial[subset] = rng.rand(len(subset))
        elif pt == 4:
            # Frequency perturbation
            freq = rng.randint(1, n // 8)
            phase = rng.rand() * 2 * np.pi
            t = np.arange(n) / float(n)
            h_trial += 0.03 * np.sin(2.0 * np.pi * freq * t + phase)
        elif pt == 5:

        elif pt == 6:

            scale = rng.uniform(0.8, 1.2)
            h_trial = h_best * scale
        else:

            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best - 1e-12:
                c5_best = c5_new
                h_best = h_new.copy()
        except Exception:
            pass

    return h_best, c5_best


def coord_descent_focused(h_init, n, rng, max_sweeps=20):
    h_best = h_init.copy()
    c5_best = compute_c5(h_best)
    for sweep in range(max_sweeps):
        if remaining_time() < 2.0:
            break
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            if remaining_time() < 1.0:
                break
            old = h_best[i]
            candidates = [0.0, 1.0]
            for d in [0.05, -0.05, 0.1, -0.1, 0.15, -0.15]:
                c = old + d
                if 0.0 <= c <= 1.0 and abs(c - old) > 1e-8:
                    candidates.append(c)
            for new_v in candidates:
                h_trial = h_best.copy()
                h_trial[i] = new_v
                h_trial = project_feasible(h_trial, n)
                c5_t = compute_c5(h_trial)
                if c5_t < c5_best - 1e-14:
                    c5_best = c5_t
                    h_best = h_trial.copy()
                    any_imp = True
                    break
        if not any_imp and sweep > 4:
            break
    return h_best, c5_best
