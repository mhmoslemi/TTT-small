import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.sum(mask)
        if cnt > 0:
            step = diff / cnt
            h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
        else:
            if diff > 0:
                idx = int(np.argmin(h))
            else:
                idx = int(np.argmax(h))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)

def fast_gradient(h, n, beta, corr=None):
    """Compute gradient of smooth max objective."""
    dx = 2.0 / n
    if corr is None:
        corr = np.correlate(h, 1.0 - h, mode='full')
    ncorr = 2 * n - 1
    # Log-sum-exp weights
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_v = np.exp(shifted)
    wsum = np.sum(exp_v)
    w = exp_v / wsum  # weights, sum to 1
    
    grad = np.zeros(n)
    # Precompute for speed
    for k_idx in range(ncorr):
        wk = w[k_idx]
        if wk < 1e-15:
            continue
        offset = k_idx - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[:npairs] += wk * (1.0 - h[offset:offset+npairs])
                grad[offset:offset+npairs] -= wk * h[:npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[aoff:aoff+npairs] += wk * (1.0 - h[:npairs])
                grad[:npairs] -= wk * h[aoff:aoff+npairs]
    grad *= dx
    return grad

def adam_smooth_optimize(h, n, betas, iters, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimization with schedule of betas."""
    h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for beta in betas:
        m = np.zeros(n)
        v = np.zeros(n)
        for it in range(iters):
            if remaining_time() < 3:
                return best_h, best_c5
            t += 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_c = np.max(corr)
            shifted = beta * (corr - max_c)
            shifted = np.clip(shifted, -50, 50)
            exp_v = np.exp(shifted)
            wsum = np.sum(exp_v)
            w = exp_v / wsum
            val = (max_c + np.log(wsum) / beta) * (2.0 / n)
            
            grad = np.zeros(n)
            for k_idx in range(2*n-1):
                wk = w[k_idx]
                if wk < 1e-15:
                    continue
                offset = k_idx - (n - 1)
                if offset >= 0:
                    npairs = n - offset
                    if npairs > 0:
                        grad[:npairs] += wk * (1.0 - h[offset:offset+npairs])
                        grad[offset:offset+npairs] -= wk * h[:npairs]
                else:
                    aoff = -offset
                    npairs = n - aoff
                    if npairs > 0:
                        grad[aoff:aoff+npairs] += wk * (1.0 - h[:npairs])
                        grad[:npairs] -= wk * h[aoff:aoff+npairs]
            grad *= (2.0 / n)
            
            # Add constraint gradient (penalty for sum != n/2)
            sum_diff = np.sum(h) - n / 2.0
            grad += 50.0 * sum_diff
            
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            lr_t = lr / (1.0 + 0.001 * t)
            h = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            h = project_feasible(h, n)
            
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5

def coord_descent(h, n, max_sweeps=20, verbose=False):
    """Coordinate descent with exact line search per coordinate."""
    h = h.copy()
    best_c5 = compute_c5(h)
    if best_c5 < 0.38:
        return h, best_c5
    
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        # Focus on interior points
        interior = np.where((h > 0.01) & (h < 0.99))[0]
        if len(interior) == 0:
            break
        
        for j in interior:
            if remaining_time() < 1:
                break
            t0 = h[j]
            # Compute coefficients for each lag
            # c_k(t) = base_k + a_k * t + b_k * t^2
            ncorr = 2 * n - 1
            a = np.zeros(ncorr)
            b = np.zeros(ncorr)
            
            for off in range(-(n-1), n):
                k = n - 1 + off
                if off == 0:
                    # c_0 contribution from j: t*(1-t) = t - t^2
                    a[k] += 1.0
                    b[k] += -1.0
                else:
                    if off > 0:
                        if j < n - off:
                            a[k] += (1.0 - h[j + off])  # h[j] as first factor
                        if j >= off:
                            a[k] += (-h[j - off])  # h[j] as second factor (1-h[j])
                    else:
                        m = -off
                        if j >= m:
                            a[k] += (1.0 - h[j - m])  # h[j] as first factor
                        if j + m < n:
                            a[k] += (-h[j + m])  # h[j] as second factor
            
            # base_k = c_k(t0) - a_k*t0 - b_k*t0^2
            t0_val = t0
            base = corr - a * t0_val - b * t0_val * t0_val
            
            # Evaluate f(t) at candidate points
            # Candidates: 0, 1, t0, and where the max changes
            candidates = [0.0, 1.0, t0]
            # Add some heuristic candidates
            candidates.append(0.5)
            candidates.append(max(0.0, t0 - 0.1))
            candidates.append(min(1.0, t0 + 0.1))
            candidates.append(max(0.0, t0 - 0.05))
            candidates.append(min(1.0, t0 + 0.05))
            
            # Find minimum by evaluating at candidates and doing local search
            best_t = t0
            best_val = np.inf
            for t in candidates:
                t = max(0.0, min(1.0, t))
                vals = base + a * t + b * t * t
                f = np.max(vals) * dx
                if f < best_val:
                    best_val = f
                    best_t = t
            
            # Local refinement with golden section around best_t
            if best_t != t0:
                lo = max(0.0, best_t - 0.05)
                hi = min(1.0, best_t + 0.05)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = np.max(base + a*c1 + b*c1*c1) * dx
                f2 = np.max(base + a*c2 + b*c2*c2) * dx
                for _ in range(30):
                    if f1 < f2:
                        hi = c2; c2 = c1; f2 = f1
                        c1 = hi - gr*(hi-lo)
                        f1 = np.max(base + a*c1 + b*c1*c1) * dx
                    else:
                        lo = c1; c1 = c2; f1 = f2
                        c2 = lo + gr*(hi-lo)
                        f2 = np.max(base + a*c2 + b*c2*c2) * dx
                t_star = (lo + hi) / 2.0
                f_star = np.max(base + a*t_star + b*t_star*t_star) * dx
                if f_star < best_val:
                    best_val = f_star
                    best_t = t_star
            
            if best_val < best_c5 - 1e-10:
                h[j] = best_t
                corr = np.correlate(h, 1.0 - h, mode='full')
                best_c5 = compute_c5(h)
                improved = True
        
        if not improved:
            break
    
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
