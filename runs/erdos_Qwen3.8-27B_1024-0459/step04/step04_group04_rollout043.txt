    rng = np.random.RandomState(seed)
    h = h.copy()
    target = n // 2
    assert int(np.sum(h)) == target
    
    corr = compute_corr(h)

def to_binary(h, n, rng=None):
    target = n // 2
    if rng is not None:
        h = h + rng.rand(n) * 1e-10
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin


def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = bin(i).count('1') % 2
    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones > target:
        idx = np.where(h == 1)[0]
        for k in range(n_ones - target):
            h[idx[k]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        for k in range(target - n_ones):
            h[idx[k]] = 1
    return h


def random_binary(n, rng):
    h = np.zeros(n)
    target = n // 2
    idx = rng.choice(n, target, replace=False)
    h[idx] = 1.0
    return h


def block_sequence(n, block_size=8):
    h = np.zeros(n)
    target = n // 2
    for i in range(n):
        if (i // block_size) % 2 == 0:
            h[i] = 1.0
    n_ones = int(np.sum(h))
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[:n_ones - target]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[:target - n_ones]] = 1
    return h


def sa_binary(h, n, time_limit, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    target = n // 2
    if int(np.sum(h)) != target:
        h = to_binary(h, n, rng)

    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()

    dx = 2.0 / n
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    start = time.time()
    T = 3.0
    T_min = 0.05
    alpha = 0.99995
    stall = 0

    while time.time() - start < time_limit and remaining() > 5:
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        h[i] = 0.0
        h[j] = 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))

        delta = new_max - cur_max
        if delta < -1e-12 or (T > T_min and rng.rand() < np.exp(-max(delta, 0) / T)):
            cur_max = new_max
            corr = new_corr
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1

        T *= alpha
        if stall > 2000:
            T = max(T, 1.0)
            stall = 0

    return best_h, best_max * dx


def greedy_local_search(h, n, time_limit, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    target = n // 2

    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()

    dx = 2.0 / n
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    start = time.time()
    no_improve = 0

    while time.time() - start < time_limit and remaining() > 5:
        n_trials = min(2000, len(ones) * len(zeros))
        improved_this_round = False

        for _ in range(n_trials):
            if time.time() - start > time_limit or remaining() < 5:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

            h[i] = 0.0
            h[j] = 1.0
            new_max = float(np.max(compute_corr(h)))

            if new_max < cur_max - 1e-12:
                cur_max = new_max
                corr = compute_corr(h)
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if new_max < best_max - 1e-12:
                    best_max = new_max
                    best_h = h.copy()
                    no_improve = 0
                improved_this_round = True
            else:
                h[i] = 1.0
                h[j] = 0.0

        if not improved_this_round:
            no_improve += 1
            if no_improve > 20:
                for _ in range(3):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i], h[j] = 0.0, 1.0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                cur_max = float(np.max(compute_corr(h)))
                corr = compute_corr(h)
                no_improve = 0

    return best_h, best_max * dx


def sa_then_greedy(h, n, sa_time, greedy_time, seed=42):
    h_sa, c5_sa = sa_binary(h, n, sa_time, seed)
    h_final, c5_final = greedy_local_search(h_sa, n, greedy_time, seed + 999)
    return h_final, c5_final


def subgradient_descent(h, n, time_limit, lr=0.01, momentum=0.9):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    corr = compute_corr(h)
    cur_c5 = float(np.max(corr))
    best_h = h.copy()
    best_c5 = cur_c5

    dx = 2.0 / n
    start = time.time()
    v = np.zeros(n)

    while time.time() - start < time_limit and remaining() > 5:
        corr = compute_corr(h)
        m = float(np.max(corr))
        if m < best_c5:
            best_c5 = m
            best_h = h.copy()

        # Find top lags
        top_k = np.argsort(-corr)[:5]
        g = np.zeros(n)
        for k in top_k:
            ki = int(k)
            off = ki - (n - 1)
            if off >= 0:
                npairs = n - off
                g[off:off + npairs] += (1.0 - h[:npairs])
                g[:npairs] -= h[off:off + npairs]
            else:
                ao = -off
                npairs = n - ao
                g[:npairs] += (1.0 - h[ao:ao + npairs])
                g[ao:ao + npairs] -= h[:npairs]
        g /= len(top_k)

        v = momentum * v - lr * g
        h_new = h + v
        h_new = project_feasible(h_new, n)

        new_corr = compute_corr(h_new)
        new_m = float(np.max(new_corr))
        if new_m < m * (1 + 0.1):
            h = h_new
        else:
            lr *= 0.9
            v *= 0.5

    final_c5 = compute_c5(h)
    if compute_c5(best_h) < final_c5:
        return best_h, compute_c5(best_h)
    return h, final_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 360
        parent_h = random_binary(parent_n, rng)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    def check_and_update(h_new, c5_new, n_new):
        nonlocal best_h, best_c5, best_n
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            best_n = n_new
            return True
        return False

    # Phase 1: Binary optimization at parent n
    n_parent = best_n
    h_bin_parent = to_binary(best_h, n_parent, rng)
    c5_bin = compute_c5(h_bin_parent)
    if c5_bin < best_c5:
        best_h = h_bin_parent.copy()
        best_c5 = c5_bin
        best_n = n_parent

    if best_c5 > _TARGET_C5 and remaining() > 60:
        # Run SA + greedy at parent n
        t_sa = min(120, remaining() * 0.3)
        t_greedy = min(60, remaining() * 0.15)
        h_opt, c5_opt = sa_then_greedy(h_bin_parent, n_parent, t_sa, t_greedy, seed)
        check_and_update(h_opt, c5_opt, n_parent)

    # Phase 2: Try multiple n values with diverse initializations
    n_list = [360, 400, 440, 380, 420, 460, 340, 320]
    n_list = [n for n in n_list if n % 2 == 0 and 100 <= n <= 600]
    # Remove parent n (already done)
    n_list = [n for n in n_list if n != n_parent]

    for n_t in n_list:
        if remaining() < 40:
            break
        if best_c5 <= _TARGET_C5:
            break

        # Try multiple initializations
        inits = []
        # From parent (interpolated)
        xp = (np.arange(best_n) + 0.5) * 2.0 / best_n
        x_new = (np.arange(n_t) + 0.5) * 2.0 / n_t
        h_interp = np.interp(x_new, xp, best_h)
        inits.append(to_binary(h_interp, n_t, rng))
        # Thue-Morse
        inits.append(thue_morse(n_t))
        # Random
        inits.append(random_binary(n_t, rng))

        for init_h in inits:
            if remaining() < 20:
                break
            t_sa = min(40, max(10, remaining() * 0.2))
            t_greedy = min(20, max(5, remaining() * 0.1))
            h_opt, c5_opt = sa_then_greedy(init_h, n_t, t_sa, t_greedy, seed + n_t + len(inits))
            check_and_update(h_opt, c5_opt, n_t)

    # Phase 3: Intensive optimization on best solution found
    if remaining() > 60 and best_c5 > _TARGET_C5:
        n_best = best_n
        h_work = to_binary(best_h, n_best, rng)

        # Long SA
        t_sa = min(180, remaining() * 0.4)
        h_opt, c5_opt = sa_binary(h_work, n_best, t_sa, seed + 7777)
        check_and_update(h_opt, c5_opt, n_best)

        # Greedy polish
        if remaining() > 30:
            t_greedy = min(60, remaining() * 0.3)
            h_opt2, c5_opt2 = greedy_local_search(h_work, n_best, t_greedy, seed + 8888)
            check_and_update(h_opt2, c5_opt2, n_best)

    # Phase 4: Continuous optimization (backup)
    if remaining() > 60 and best_c5 > _TARGET_C5:
        n_cont = best_n
        h_cont = project_feasible(best_h, n_cont)
        t_sg = min(80, remaining() * 0.4)
        h_sg, c5_sg = subgradient_descent(h_cont, n_cont, t_sg)
        check_and_update(h_sg, c5_sg, n_cont)

    # Phase 5: Final binary push with multiple seeds
    if remaining() > 30 and best_c5 > _TARGET_C5:
        n_best = best_n
        h_work = to_binary(best_h, n_best, rng)
        for s in [seed + 11111, seed + 22222, seed + 33333]:
            if remaining() < 15:
                break
            t_sa = min(20, remaining() * 0.4)
            h_opt, c5_opt = sa_binary(h_work, n_best, t_sa, s)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n = n_best

    # Final: also try converting best continuous to binary
    h_bin_final = to_binary(best_h, best_n, rng)
    c5_bin_final = compute_c5(h_bin_final)
    if c5_bin_final < best_c5:
        best_h = h_bin_final.copy()
        best_c5 = c5_bin_final

    # Validate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
