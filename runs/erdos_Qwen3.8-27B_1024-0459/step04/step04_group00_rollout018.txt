import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            adj = diff / cnt
            for it in range(50):
                h[mask] += adj
                h = np.clip(h, 0.0, 1.0)
                diff = target - np.sum(h)
                if abs(diff) < 1e-15:
                    break
                mask = (h > 1e-12) & (h < 1.0 - 1e-12)
                cnt = np.count_nonzero(mask)
                if cnt == 0:
                    break
                adj = diff / cnt
        else:
            h = h + diff / n
            h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)


def compute_A_for_j(j, h, n):
    """Compute A[k] = d(corr[k])/d(h[j]) for all k, vectorized."""
    A = np.zeros(2 * n - 1)
    A[n - 1] = 1.0 - 2.0 * h[j]
    a = np.arange(1, n)
    # o = +a >= 0
    m1 = a <= j
    m2 = a <= (n - 1 - j)
    if m1.any():
        A[n - 1 + a[m1]] += (1.0 - h[j - a[m1]])
    if m2.any():
        A[n - 1 + a[m2]] += -h[j + a[m2]]
    # o = -a < 0
    m3 = a <= (n - 1 - j)
    m4 = a <= j
    if m3.any():
        A[n - 1 - a[m3]] += (1.0 - h[j + a[m3]])
    if m4.any():
        A[n - 1 - a[m4]] += -h[j - a[m4]]
    return A


def find_best_t(A, B):
    """Find t in [0,1] minimizing max(A*t + B) via bisection on subgradient."""
    f0 = np.max(B)
    f1 = np.max(A + B)
    # Subgradient at 0
    tol = 1e-12 * max(1.0, f0)
    amax0 = np.max(B)
    argmax0 = np.where(B >= amax0 - tol)[0]
    if len(argmax0) > 0 and np.min(A[argmax0]) >= -1e-14:
        return 0.0
    # Subgradient at 1
    amax1 = np.max(A + B)
    argmax1 = np.where(A + B >= amax1 - tol)[0]
    if len(argmax1) > 0 and np.max(A[argmax1]) <= 1e-14:
        return 1.0
    # Bisection
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        vals = A * mid + B
        mx = np.max(vals)
        t2 = 1e-12 * max(1.0, abs(mx))
        argmax = np.where(vals >= mx - t2)[0]
        if len(argmax) == 0:
            break
        amin = np.min(A[argmax])
        amax = np.max(A[argmax])
        if amin > 1e-14:
            hi = mid
        elif amax < -1e-14:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5


def coordinate_descent(h, n, max_sweeps=100, rng=None):
    """Exact coordinate descent with incremental correlation updates."""
    h = project_feasible(h, n)
    corr = compute_corr(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    prev_c5 = best_c5 + 0.01

    for sweep in range(max_sweeps):
        if remaining() < 5:
            break
        if rng is not None and sweep % 3 == 1:
            idx = rng.permutation(n)
        elif rng is not None and sweep % 3 == 2:
            idx = np.arange(n - 1, -1, -1)
        else:
            idx = np.arange(n)

        for j in idx:
            old = h[j]
            A = compute_A_for_j(int(j), h, n)
            B = corr - A * old
            t_star = find_best_t(A, B)

            if abs(t_star - old) > 1e-12:
                delta = t_star - old
                h[j] = t_star
                corr += A * delta

        h = project_feasible(h, n)
        corr = compute_corr(h)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
        if abs(c5 - prev_c5) < 1e-12:
            break
        prev_c5 = c5

    return best_h, best_c5


def _sm_obj(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / _BETA) * dx_v


def _sm_grad_sum(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    wc = w * dx_v
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            npairs = n - off
            g[off:off + npairs] += wk * (1.0 - x[:npairs])
            g[:npairs] -= wk * x[off:off + npairs]
        else:
            ao = -off
            npairs = n - ao
            g[:npairs] += wk * (1.0 - x[ao:ao + npairs])
            g[ao:ao + npairs] -= wk * x[:npairs]
    return g


def _sc(x):
    return np.sum(x) - _TGT


def _scj(x):
    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=500):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad_sum, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.success or (hasattr(res, 'fun') and res.fun < _sm_obj(h)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def top_shifts_focus(h, n, top_k=5, max_iters=200):

            break
        k_top = idx[0]
        k_bot = idx[-1]
        g_t = np.zeros(n)
        g_b = np.zeros(n)
        for k, g in [(k_top, g_t), (k_bot, g_b)]:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += (1.0 - h[:np_])
                g[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= h[:np_]
        d = g_b - g_t
        dn = np.linalg.norm(d)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1')
        h[i] = 1.0 if c % 2 == 1 else 0.0
    n_ones = int(np.sum(h))
    if n_ones != n // 2:
        sort_idx = np.argsort(-h)
        h2 = np.zeros(n)
        h2[sort_idx[:n // 2]] = 1.0
        h = h2
    return h


def random_balanced(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def perturb_h(h, n, rng, eps=0.1):

    return project_feasible(h_new, n)
