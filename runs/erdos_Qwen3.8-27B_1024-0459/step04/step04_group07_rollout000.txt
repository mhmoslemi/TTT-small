import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def interp_h(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def make_tm(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') & 1
        h[i] = 1.0 if c else 0.0
    return h

def make_rand_bin(rng, n):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def spectral_init(rng, n):
    H = np.zeros(n, dtype=complex)
    for k in range(1, n):
        if n % 2 == 0 and k == n // 2:
            H[k] = rng.uniform(-0.5, 0.5)
            continue
        r = rng.uniform(0.3, 1.0)
        theta = rng.uniform(0, 2 * np.pi)
        H[k] = r * (np.cos(theta) + 1j * np.sin(theta))
    for k in range(1, (n // 2) + 1):
        nk = (n - k) % n
        if nk != k and nk != 0 and nk < n:
            H[nk] = np.conj(H[k])
    h = np.real(np.fft.ifft(H))
    h = h - np.mean(h)
    hmin, hmax = h.min(), h.max()
    if hmax > hmin:
        h = (h - hmin) / (hmax - hmin)
    else:
        h = np.full(n, 0.5)
    return h

def obj_mob(x, n, beta, pen, focus):
    corr = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    s = corr - m
    e = np.exp(np.clip(beta * s, -400, 400))
    val = (m + np.log(np.sum(e)) / beta) * dx
    if pen > 0:
        val += pen * (np.sum(x) - n / 2.0) ** 2
    if focus >= 0 and focus < len(corr):
        val += 0.005 * corr[focus] * dx
    return float(val)

def slsqp_opt(h0, n, beta, maxiter, focus=-1):
    def obj(x):
        return obj_mob(x, n, beta, 0.0, focus)
    def eq_fun(x):
        return np.sum(x) - n / 2.0
    def eq_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)

def lbfgsb_opt(h0, n, beta, maxiter, pen=100.0):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return obj_mob(x, n, beta, pen, -1)
    try:
        res = minimize(obj, h0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)

        if tlimit() > 10:
            h_bin = (best_h > 0.5).astype(float)
            h_bin = project_feasible(h_bin, n)
            h_bin, c5_bin = swap_search(h_bin, n, rng, tlimit, max_iter=10000)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
        print("  C5={:.8f}".format(best_c5))

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < final_c5:
                final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
