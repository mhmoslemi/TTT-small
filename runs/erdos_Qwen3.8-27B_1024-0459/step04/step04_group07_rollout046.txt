            h_low = project_feasible(interp_h(best_h, best_n, n_low), n_low)
            c5_l = compute_c5(h_low)
            for bv in [1000, 5000, 15000]:
                if tlimit() < 10:
                    break
                h_new = slsqp_opt(h_low, n_low, bv, maxiter=400, topk=3)
                c_new = compute_c5(h_new)
                if c_new < c5_l:
                    c5_l = c_new
                    h_low = h_new
            if c5_l < best_c5 - 1e-10:
                best_c5 = c5_l
                best_h = h_low.copy()
                best_n = n_low
                print("  IMPROVED at n={}: C5={:.8f}".format(n_low, best_c5))
            else:
                print("  n={}: C5={:.8f} (no improvement)".format(n_low, c5_l))

    # Phase 5: Multi-start perturbation
    if tlimit() > 80:
        print("\n=== Phase 5: Multi-start perturbations ===")
        n = best_n
        for attempt in range(6):
            if tlimit() < 18:
                break
            if attempt == 0:
                h_pert = best_h + rng.randn(n) * 0.08
            elif attempt == 1:
                h_pert = best_h.copy()
                idxs = rng.choice(n, size=min(30, n), replace=False)
                for idx in idxs:
                    h_pert[idx] = np.clip(h_pert[idx] + rng.randn() * 0.15, 0.0, 1.0)
            elif attempt == 2:
                h_pert = best_h.copy()
                idxs = rng.choice(n, size=min(20, n), replace=False)
                for i in range(0, min(18, n), 2):
                    a, b = idxs[i], idxs[i + 1]
                    h_pert[a], h_pert[b] = h_pert[b], h_pert[a]
            elif attempt == 3:
                h_pert = make_sine(rng, n)
            elif attempt == 4:
                h_pert = make_tm(n)
            else:
                h_pert = best_h.copy()
                h_f = np.fft.rfft(h_pert)
                for _ in range(3):
                    freq = rng.randint(1, len(h_f))
                    h_f[freq] += rng.randn() * 0.5
                    h_f[len(h_f) - 1 - freq] += rng.randn() * 0.5
                h_pert = np.real(np.fft.irfft(h_f, n))

            h_pert = project_feasible(h_pert, n)
            c5_p = compute_c5(h_pert)

            if c5_p < best_c5 * 1.05:
                h_cur = h_pert.copy()
                c5_cur = c5_p
                for bv in [500, 2000, 8000, 20000]:
                    if tlimit() < 6:
                        break
                    h_new = slsqp_opt(h_cur, n, bv, maxiter=300, topk=3)
                    c_new = compute_c5(h_new)
                    if c_new < c5_cur - 1e-14:
                        c5_cur = c_new
                        h_cur = h_new
                if c5_cur < best_c5 - 1e-11:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n
                    print("  Start {}: C5={:.8f}".format(attempt, c5_cur))
            else:
                print("  Start {}: C5={:.6f} (too far, skip)".format(attempt, c5_p))

    # Phase 6: Block coordinate descent
    if tlimit() > 40:
        print("\n=== Phase 6: Block coordinate descent ===")
        best_h, best_c5 = block_descent(best_h, best_n, [4, 8, 16],
                                        max_sweeps=25, rng=rng, tlimit=tlimit)
        best_n = len(best_h)
        print("  After block CD: C5={:.8f}".format(best_c5))

    # Phase 7: Final aggressive polish
    if tlimit() > 20:
        print("\n=== Phase 7: Final polish ===")
        n = best_n
        for bv in [10000, 25000, 50000, 100000]:
            if tlimit() < 5:
                break
            h_new = slsqp_opt(best_h, n, bv, maxiter=600, topk=0)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new
        # Top-k variants
        for tk in [2, 3, 5]:
            if tlimit() < 5:
                break
            h_new = slsqp_opt(best_h, n, 50000, maxiter=400, topk=tk)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new

    # Final micro coordinate descent
    if tlimit() > 10:
        c5_cur = compute_c5(best_h)
        n = best_n
        h_best = best_h.copy()
        for sweep in range(5):
            if tlimit() < 2:
                break
            for i in range(n):
                if tlimit() < 1:
                    break
                for delta in [0.02, -0.02, 0.01, -0.01, 0.05, -0.05]:
                    trial = h_best.copy()
                    trial[i] = np.clip(trial[i] + delta, 0.0, 1.0)
                    trial = project_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < c5_cur - 1e-13:
                        c5_cur = c
                        h_best = trial.copy()
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_best

    # Final validation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    _G_PEN = 0.0
    _G_TOPK = 0
    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
