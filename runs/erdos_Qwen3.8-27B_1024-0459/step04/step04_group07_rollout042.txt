def slsqp_run(h0, n, beta, maxiter):
    """SLSQP with smooth max objective"""
    global _G_N, _G_BETA, _G_TGT, _G_DX, _G_PEN, _G_FOCUS
    
    def _obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = corr - m
        e = np.exp(np.clip(_G_BETA * s, -400, 400))
        return (m + np.log(np.sum(e)) / _G_BETA) * _G_DX
    
    _G_BETA = float(beta)
    _G_DX = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    try:
        res = minimize(_obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h = project_feasible(res.x, n)
    except Exception:
        h = project_feasible(h0, n)
    return h

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

        if abs(diff) < 1e-15:

        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)

        else:

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    s = corr - m
    exp_s = np.exp(np.clip(beta * s, -400, 400))
    total = np.sum(exp_s)
    weights = exp_s / total
    obj = (m + np.log(total) / beta) * dx
    grad = np.zeros(n)
    nc = 2 * n - 1
    for k in range(nc):
        w = weights[k]
        if w < 1e-14:
            continue
        mlag = k - (n - 1)
        i_lo = max(0, -mlag)
        i_hi = min(n - 1, n - 1 - mlag)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += w * (1.0 - h[idx + mlag])
        j_lo = max(0, mlag)
        j_hi = min(n - 1, n - 1 + mlag)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            grad[idx] += -w * h[idx - mlag]
    grad *= dx
    return obj, grad
