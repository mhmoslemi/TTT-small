def slsqp_grad(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    ex = np.exp(np.clip(beta * shifted, -50, 50))
    w = ex / np.sum(ex)
    dx_val = 2.0 / n
    g = np.zeros(n)

    # corr[k] = sum_i x[i] * (1 - x[i + k - (n-1)]) for valid i
    # Let s = k - (n-1)
    # If s >= 0: corr = sum_{i=0}^{n-s-1} x[i] * (1-x[i+s])
    #   d/dx[i] = (1-x[i+s]) for i <= n-s-1
