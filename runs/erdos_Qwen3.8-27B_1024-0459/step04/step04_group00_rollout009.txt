        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = float(c)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            remove = np.random.choice(ones, n_ones - target, replace=False)
            h[remove] = 0
        else:
            zeros = np.where(h == 0)[0]
            add = np.random.choice(zeros, target - n_ones, replace=False)
            h[add] = 1
    return h


def random_balanced(n, rng):
    idx = np.arange(n)
    rng.shuffle(idx)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h


def greedy_construct(n, rng, w_seed=0):
    """Greedy: add ones one at a time minimizing max correlation increase."""
    h = np.zeros(n)
    target = n // 2
    dx = 2.0 / n
    for step in range(target):
        best_pos = -1
        best_c5 = float('inf')
        candidates = np.where(h == 0)[0]
        # Sample a subset if too many candidates
        if len(candidates) > 60:
            candidates = candidates[rng.choice(len(candidates), 60, replace=False)]
        for pos in candidates:
            h[pos] = 1.0
            corr = compute_corr(h)
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_pos = pos
            h[pos] = 0.0
        if best_pos >= 0:
            h[best_pos] = 1.0
    return h


def binary_sa_multi(h_init, n, time_limit, rng, T0=3.0, T1=0.005,
                    multi_rate=0.3, tabu_len=300):
    """Simulated annealing with single and multi-swap moves."""
    h = (h_init > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            rem = rng.choice(ones_idx, n_ones - n_half, replace=False)
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            add = rng.choice(zeros_idx, n_half - n_ones, replace=False)
            h[add] = 1
    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr)) * dx
    best_h = h.copy()
    best_c5 = cur_max
    best_corr_max = float(np.max(corr))
    start = time.time()
    tabu = {}
    step_count = 0
    n_iters = 0
    no_improve = 0
    stall_limit = 5000

    while time.time() - start < time_limit and remaining() > 5:
        n_iters += 1
        progress = min(1.0, (time.time() - start) / time_limit)
        T = T0 * (T1 / T0) ** progress

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        best_delta = 0
        best_move_single = None
        best_move_multi = None

        # Try single swaps
        n_tries = 12
        for _ in range(n_tries):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            key = (min(i, j), max(i, j))
            if key in tabu and step_count - tabu[key] < tabu_len:
                continue
            h[i] = 0
            h[j] = 1
            nc = compute_corr(h)
            nm = float(np.max(nc))
            h[i] = 1
            h[j] = 0
            delta = nm - best_corr_max
            if best_move_single is None or delta < best_delta:
                best_delta = delta
                best_move_single = (i, j, None)

        # Try multi-swaps (2 pairs) with some probability
        if rng.rand() < multi_rate and len(ones) >= 4 and len(zeros) >= 4:
            n_mtries = 6
            for _ in range(n_mtries):
                i1, i2 = rng.choice(ones, 2, replace=False)
                j1, j2 = rng.choice(zeros, 2, replace=False)
                key1 = (min(i1, j1), max(i1, j1))
                key2 = (min(i2, j2), max(i2, j2))
                if (key1 in tabu and step_count - tabu[key1] < tabu_len) or \
                   (key2 in tabu and step_count - tabu[key2] < tabu_len):
                    continue
                h[i1] = 0
                h[i2] = 0
                h[j1] = 1
                h[j2] = 1
                nc = compute_corr(h)
                nm = float(np.max(nc))
                h[i1] = 1
                h[i2] = 1
                h[j1] = 0
                h[j2] = 0
                delta = nm - best_corr_max
                if best_move_multi is None or delta < best_delta:
                    best_delta = delta
                    best_move_multi = (i1, i2, (j1, j2))

        # Choose best move from single and multi
        do_multi = (best_move_multi is not None and
                    (best_move_single is None or best_move_multi is not best_move_single))
        if best_move_multi is not None and not do_multi:
            # Compare
            # best_delta already tracks the global best
            pass

        if best_move_single is None and best_move_multi is None:
            continue

        # Determine which move has smaller delta
        if best_move_single is not None and best_move_multi is not None:
            # Re-evaluate to compare
            # We used same best_delta tracking, so the last update won
            # Actually, let's just pick based on which was updated last with best_delta
            # Since we compared both against best_delta, the one that set it best is the winner
            # Simpler: just use the one with smaller delta (they share best_delta)
            use_multi = rng.rand() < 0.4
        elif best_move_single is not None:
            use_multi = False
        else:
            use_multi = True

        if use_multi and best_move_multi is not None:
            i1, i2, (j1, j2) = best_move_multi
            h[i1] = 0
            h[i2] = 0
            h[j1] = 1
            h[j2] = 1
            tabu[(min(i1, j1), max(i1, j1))] = step_count
            tabu[(min(i2, j2), max(i2, j2))] = step_count
        else:
            if best_move_single is None or best_move_multi is not None:
                if best_move_multi is not None:
                    i1, i2, (j1, j2) = best_move_multi
                    h[i1] = 0
                    h[i2] = 0
                    h[j1] = 1
                    h[j2] = 1
                    tabu[(min(i1, j1), max(i1, j1))] = step_count
                    tabu[(min(i2, j2), max(i2, j2))] = step_count
                    step_count += 1
                    corr = compute_corr(h)
                    best_corr_max = float(np.max(corr))
                    cur_max = best_corr_max * dx
                    if cur_max < best_c5:
                        best_c5 = cur_max
                        best_h = h.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                    continue
            i, j, _ = best_move_single
            h[i] = 0
            h[j] = 1
            tabu[(min(i, j), max(i, j))] = step_count

        step_count += 1
        corr = compute_corr(h)
        best_corr_max = float(np.max(corr))
        cur_max = best_corr_max * dx
        if cur_max < best_c5:
            best_c5 = cur_max
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > stall_limit and T < 0.01:
            break

    return best_h, best_c5


def continuous_refine(h_init, n, time_limit, rng):
    """Optimize continuous values using active-set projected subgradient."""
    h = project_feasible(h_init.copy(), n)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    lr = 0.05
    decay = 0.995

    for it in range(200000):
        if time.time() - start > time_limit or remaining() < 5:
            break
        corr = compute_corr(h)
        cmax = float(np.max(corr))
        active = np.where(corr > cmax - 0.5)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])
        elif len(active) > 20:
            top_idx = np.argsort(-corr)[:20]
            active = top_idx

        # Compute gradient of average of active correlations
        g = np.zeros(n)
        for k in active:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += (1.0 - h[:np_])
                g[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= h[:np_]
        g /= len(active)

        # Projected gradient step
        h_new = h - lr * g
        h_new = np.clip(h_new, 0.0, 1.0)
        # Fix sum
        diff = n / 2.0 - np.sum(h_new)
        mask = (h_new > 1e-8) & (h_new < 1.0 - 1e-8)
        if np.count_nonzero(mask) > 0:
            h_new[mask] += diff / np.count_nonzero(mask)
        else:
            h_new[:] += diff / n
        h_new = np.clip(h_new, 0.0, 1.0)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new.copy()
        elif c5_new < best_c5 + 1e-8:
            h = h_new.copy()
        else:
            lr *= 0.9

        lr = max(lr * decay, 0.001)

    # Final SLSQP on a reduced set of free variables
    if remaining() > 10:
        free = (h > 0.01) & (h < 0.99)
        n_free = np.count_nonzero(free)
        if n_free >= 2 and n_free <= 40:
            free_idx = np.where(free)[0]
            fixed_vals = h.copy()

            def obj_fun(x):
                h_temp = fixed_vals.copy()
                h_temp[free_idx] = x
                return compute_c5(h_temp)

            x0 = h[free_idx].copy()
            bounds = [(0.0, 1.0)] * n_free
            try:
                res = minimize(obj_fun, x0, method='Nelder-Mead',
                               options={'maxiter': 500, 'xatol': 1e-10, 'fatol': 1e-12})
                h_temp = fixed_vals.copy()
                h_temp[free_idx] = np.clip(res.x, 0, 1)
                h_temp = project_feasible(h_temp, n)
                c5_temp = compute_c5(h_temp)
                if c5_temp < best_c5:
                    best_h = h_temp.copy()
                    best_c5 = c5_temp
            except Exception:
                pass

    return best_h, best_c5


def coarse_to_fine(h_coarse, n_coarse, n_fine, time_per_level, rng):
    """Interpolate from coarse to fine and do local search."""
    xp = (np.arange(n_coarse) + 0.5) * 2.0 / n_coarse
    x_new = (np.arange(n_fine) + 0.5) * 2.0 / n_fine
    h_interp = np.interp(x_new, xp, h_coarse)
    h_feas = project_feasible(h_interp, n_fine)

    # Binary SA at fine resolution
    h_bin = (h_feas > 0.5).astype(float)
    n_half = n_fine // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h_bin == 1)[0]
            h_bin[ones_idx[rng.choice(len(ones_idx), n_ones - n_half, replace=False)]] = 0
        else:
            zeros_idx = np.where(h_bin == 0)[0]
            h_bin[zeros_idx[rng.choice(len(zeros_idx), n_half - n_ones, replace=False)]] = 1

    h_opt, c5_opt = binary_sa_multi(h_bin, n_fine, time_per_level, rng,
                                    T0=2.0, T1=0.01, tabu_len=200)
    return h_opt, c5_opt


def sequential_bisection_refine(h, n, time_limit):
    """Iteratively bisect the gap between max and second-max correlations."""
    h = h.copy()
    dx = 2.0 / n
    start = time.time()
    best_c5 = compute_c5(h)

    for outer in range(200):
        if time.time() - start > time_limit or remaining() < 3:
            break
        corr = compute_corr(h)
        cmax = float(np.max(corr))
        top_k = np.argsort(-corr)[:8]
        cmin_top = corr[top_k[-1]]

        if cmax - cmin_top < 0.01:
            break

        # For each top shift, compute gradient and find direction to reduce it
        # while not increasing others
        k_top = top_k[0]
        off = k_top - (n - 1)
        g_reduce = np.zeros(n)
        if off >= 0:
            np_ = n - off
            g_reduce[off:off + np_] -= (1.0 - h[:np_])
            g_reduce[:np_] += h[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g_reduce[:np_] -= (1.0 - h[ao:ao + np_])
            g_reduce[ao:ao + np_] += h[:np_]

        # Also gradient to reduce a second top shift
        k_top2 = top_k[1] if len(top_k) > 1 else k_top
        off2 = k_top2 - (n - 1)
        g_reduce2 = np.zeros(n)
        if off2 >= 0:
            np_ = n - off2
            g_reduce2[off2:off2 + np_] -= (1.0 - h[:np_])
            g_reduce2[:np_] += h[off2:off2 + np_]
        else:
            ao = -off2
            np_ = n - ao
            g_reduce2[:np_] -= (1.0 - h[ao:ao + np_])
            g_reduce2[ao:ao + np_] += h[:np_]

        d = g_reduce + 0.5 * g_reduce2
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn

        # Line search
        improved = False
        for step in [0.01, 0.02, 0.005, 0.03, 0.002, 0.05, 0.001]:
            h_new = project_feasible(h + step * d, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                h = h_new
                best_c5 = c5_new
                improved = True
                break

        if not improved:
            # Try random perturbation
            h_pert = h + np.random.randn(n) * 0.01
            h_pert = project_feasible(h_pert, n)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5:
                h = h_pert
                best_c5 = c5_pert
            else:
                break

    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        np.random.seed(seed)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    np.random.seed(seed + 100)

    n_default = best_n

    # Phase 1: Coarse-to-fine binary search
    if remaining() > 100:
        n_coarse = max(64, n_default // 4)
        # Use parent structure at coarse level
        xp = (np.arange(n_default) + 0.5) * 2.0 / n_default
        x_c = (np.arange(n_coarse) + 0.5) * 2.0 / n_coarse
        h_c_init = np.interp(x_c, xp, best_h)
        h_c_bin = (h_c_init > 0.5).astype(float)
        n_half_c = n_coarse // 2
        n_ones_c = int(np.sum(h_c_bin))
        if n_ones_c != n_half_c:
            if n_ones_c > n_half_c:
                oi = np.where(h_c_bin == 1)[0]
                h_c_bin[oi[rng.choice(len(oi), n_ones_c - n_half_c, replace=False)]] = 0
            else:
                zi = np.where(h_c_bin == 0)[0]
                h_c_bin[zi[rng.choice(len(zi), n_half_c - n_ones_c, replace=False)]] = 1

        t_coarse = min(60, max(10, int(remaining() * 0.08)))
        h_c_sa, c5_c = binary_sa_multi(h_c_bin, n_coarse, t_coarse, rng, T0=3.0, T1=0.05, tabu_len=100)

        # Level 2: n_coarse * 2
        if remaining() > 60:
            n_mid = max(96, n_default // 2)
            t_mid = min(80, max(10, int(remaining() * 0.1)))
            h_mid, c5_mid = coarse_to_fine(h_c_sa, n_coarse, n_mid, t_mid, rng)
            if c5_mid < best_c5:
                best_c5 = c5_mid
                best_h = h_mid.copy()
                best_n = n_mid

        # Level 3: full resolution
        if remaining() > 50:
            n_full = n_default
            t_full = min(200, max(20, int(remaining() * 0.2)))
            h_f, c5_f = coarse_to_fine(h_mid, n_mid, n_full, t_full, rng)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()
                best_n = n_full

    # Phase 2: Direct binary SA at full resolution from multiple starts
    if remaining() > 120:
        n = best_n
        starts = []
        # Start 1: Best so far rounded
        starts.append((best_h > 0.5).astype(float))
        # Start 2: Thue-Morse
        starts.append(thue_morse(n))
        # Start 3: Parent rounded
        if 'parent_h' in dir():
            starts.append((parent_h > 0.5).astype(float))
        # Start 4: Random balanced
        starts.append(random_balanced(n, rng))
        # Start 5: Another random
        starts.append(random_balanced(n, rng))
        # Start 6: Inverted Thue-Morse
        tm = thue_morse(n)
        starts.append(1.0 - tm)

        n_starts = len(starts)
        time_per_start = min(80, max(10, int((remaining() - 10) / n_starts)))

        for si, s in enumerate(starts):
            if remaining() < 15:
                break
            # Ensure correct number of ones
            n_half = n // 2
            n_ones = int(np.sum(s))
            if n_ones != n_half:
                if n_ones > n_half:
                    oi = np.where(s == 1)[0]
                    s[oi[rng.choice(len(oi), n_ones - n_half, replace=False)]] = 0
                else:
                    zi = np.where(s == 0)[0]
                    s[zi[rng.choice(len(zi), n_half - n_ones, replace=False)]] = 1

            c5_start = compute_c5(s)
            if c5_start < best_c5:
                best_c5 = c5_start
                best_h = s.copy()

            t_sa = time_per_start
            if si == 0:
                t_sa = min(120, max(20, int(remaining() * 0.3)))
            h_sa, c5_sa = binary_sa_multi(s, n, t_sa, rng,
                                          T0=2.5, T1=0.008, multi_rate=0.3, tabu_len=400)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n

    # Phase 3: Continuous refinement
    if remaining() > 30:
        n = best_n
        h_cont = best_h.copy()
        # Make it slightly continuous for smoother optimization
        h_cont = np.clip(h_cont + rng.randn(n) * 0.02, 0, 1)
        h_cont = project_feasible(h_cont, n)

        t_cont = min(250, max(10, int(remaining() * 0.35)))
        h_ref, c5_ref = continuous_refine(h_cont, n, t_cont, rng)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            best_n = n

    # Phase 4: Sequential bisection refinement
    if remaining() > 20:
        n = best_n
        t_bisect = min(100, max(5, int(remaining() * 0.25)))
        h_b, c5_b = sequential_bisection_refine(best_h, n, t_bisect)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 5: Try different resolutions
    if remaining() > 40:
        n_candidates = sorted(set([
            best_n, best_n + 16, best_n - 16,
            best_n + 32, best_n - 32,
            int(best_n * 1.33), int(best_n * 0.75),
        ]))
        n_candidates = [nn for nn in n_candidates if 96 <= nn <= 800 and nn % 2 == 0]
        n_candidates = [nn for nn in n_candidates if nn != best_n]
        n_candidates.sort(key=lambda x: abs(x - best_n))

        for n_t in n_candidates[:4]:
            if remaining() < 15:
                break
            xp = (np.arange(best_n) + 0.5) * 2.0 / best_n
            x_new = (np.arange(n_t) + 0.5) * 2.0 / n_t
            h_int = np.interp(x_new, xp, best_h)
            h_int = project_feasible(h_int, n_t)

            # Quick binary SA
            h_bin = (h_int > 0.5).astype(float)
            n_half_t = n_t // 2
            n_ones_t = int(np.sum(h_bin))
            if n_ones_t != n_half_t:
                if n_ones_t > n_half_t:
                    oi = np.where(h_bin == 1)[0]
                    h_bin[oi[rng.choice(len(oi), n_ones_t - n_half_t, replace=False)]] = 0
                else:
                    zi = np.where(h_bin == 0)[0]
                    h_bin[zi[rng.choice(len(zi), n_half_t - n_ones_t, replace=False)]] = 1

            tTrial = min(60, max(5, int(remaining() * 0.15)))
            h_t, c5_t = binary_sa_multi(h_bin, n_t, tTrial, rng, T0=2.0, T1=0.01, tabu_len=200)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t

            # Quick continuous refine
            if remaining() > 10:
                h_tc, c5_tc = continuous_refine(h_t, n_t, min(30, int(remaining() * 0.15)), rng)
                if c5_tc < best_c5:
                    best_c5 = c5_tc
                    best_h = h_tc.copy()
                    best_n = n_t

    # Phase 6: Final intensive local search + continuous polish
    if remaining() > 15:
        n = best_n
        # Binary local search (short)
        h_bin = (best_h > 0.5).astype(float)
        n_half = n // 2
        n_ones = int(np.sum(h_bin))
        if n_ones != n_half:
            if n_ones > n_half:
                oi = np.where(h_bin == 1)[0]
                h_bin[oi[rng.choice(len(oi), n_ones - n_half, replace=False)]] = 0
            else:
                zi = np.where(h_bin == 0)[0]
                h_bin[zi[rng.choice(len(zi), n_half - n_ones, replace=False)]] = 1
        tFinal = min(60, max(5, int(remaining() * 0.4)))
        h_fb, c5_fb = binary_sa_multi(h_bin, n, tFinal, rng, T0=1.0, T1=0.003,
                                      multi_rate=0.4, tabu_len=500)
        if c5_fb < best_c5:
            best_c5 = c5_fb
            best_h = h_fb.copy()

        # Continuous polish
        if remaining() > 8:
            h_cp = best_h + rng.randn(n) * 0.005
            h_cp = project_feasible(h_cp, n)
            h_pf, c5_pf = continuous_refine(h_cp, n, min(30, max(3, int(remaining() * 0.4))), rng)
            if c5_pf < best_c5:
                best_c5 = c5_pf
                best_h = h_pf.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    # Validation
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        best_c5 = min(best_c5, eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
