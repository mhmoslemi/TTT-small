def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_obj(x, n, dx, tau):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    # Softmax-based smooth max
    shifted = (corr - m) / tau
    shifted = np.clip(shifted, -50, 0)  # numerical stability
    return float((m + tau * np.log(np.sum(np.exp(shifted)))) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def subgradient_step(h, alpha, n):
    """Take a subgradient step for the max-k term."""
    corr = compute_corr(h)
    k_max = np.argmax(corr)
    # Gradient of corr[k_max] w.r.t. h
    # corr[k] = sum_i h[i]*(1-h[i+k-(n-1)]) for valid i
    # d/dh[j] corr[k] = (1-h[j+k-(n-1)]) for j where i=j is valid
    #                  + (-h[j-(n-1)-k]) for j where i+k-(n-1)=j is valid
    dx = 2.0 / n

            # term is h[i]*(1-h[j]), derivative w.r.t h[i] is (1-h[j]), w.r.t h[j] is -h[i]
            grad[i] += (1.0 - h[j])
            if j != i:
                grad[j] += (-h[i])
    h_new = h - alpha * grad
    return np.clip(h_new, 0.0, 1.0)

So corr_s depends on h[i] through: h[i]*(1-h[i+s]) + (1-h[i])*h[i-s] = h[i] - h[i]*h[i+s] + h[i-s] - h[i]*h[i-s]
= h[i](1 - h[i+s] - h[i-s]) + h[i-s]

This is linear in h[i]! (Since h[i+s] and h[i-s] are fixed when optimizing h[i].)

Wait, that's only if i+s ≠ i and i-s ≠ i, i.e., s ≠ 0. For s=0: h[i]*(1-h[i]) which is quadratic.

So for s≠0, the contribution to corr_s from h[i] is LINEAR in h[i]. And the max of linear functions is convex in h[i]! So for each coordinate, we're minimizing a convex piecewise-linear function (max of lines), subject to 0≤h[i]≤1.

This is great! We can do coordinate ascent (actually descent since we minimize) efficiently:
- For h[i], the objective max_k corr_k * dx is a max of linear functions in h[i] (for k where s_k ≠ 0) plus possibly one quadratic term (for k where s_k = 0, i.e., k = n-1).

The max of linear functions is achieved at a breakpoint or at a boundary. We can find the optimum by:
1. Computing all the slopes and intercepts
2. Finding the maximum of lines at each candidate point
3. The minimum of the max is at a specific h[i] value

Actually, this is a 1D convex optimization (minimize max of affine functions). The optimum is where the "upper envelope" of the lines is minimized, which occurs at an intersection of two lines (or at a boundary).

This gives us an efficient coordinate descent method! Let me implement this.

For coordinate i, the objective is:
f(h[i]) = max_k corr_k(h) * dx

where corr_k(h) = A_k * h[i] + B_k (for the shifts where i appears with non-zero coefficient)

The A_k and B_k can be precomputed. The maximum of affine functions in one variable is a convex piecewise-linear function. Its minimum is at:
- h[i] = 0, or
- h[i] = 1, or
- An intersection point of two of the lines

To find the intersection points, we'd need to sort all O(n) lines and find where the max switches. But since we only need the minimum of the max, we can use a ternary search or binary search on the slope.

Actually, a simpler approach: since it's a 1D convex problem, we can use scipy.optimize.minimize_scalar or just evaluate at a few candidate points.

Even simpler: The subgradient at the current point tells us which direction to move. If the subgradient (slope of the max) is positive, decrease h[i]; if negative, increase. The optimal point is where the subgradient changes sign, which is at an intersection.

The minimum of a convex piecewise-linear function is where the subgradient contains 0, i.e., where the slope changes from negative to positive. This happens at a "kink" point (intersection of two lines).

Algorithm:
1. Compute all D_k values
2. The subgradient at t is D_k* for the k that achieves the max at t (or convex hull if multiple achieve it)
3. Binary search: at t=0, compute which k is max, get slope. At t=1, similarly.
4. If slope at 0 > 0, optimum is t=0. If slope at 1 < 0, optimum is t=1.
5. Otherwise, binary search for where slope changes sign.

But "slope" for a max function is the max of the active D_k values (for the non-negative side) and min (for non-positive side). Actually the subgradient of max_k (a_k*t + b_k) is conv{a_k : k ∈ argmax}. So the subgradient contains 0 iff there exist active k's with some having a_k < 0 and some having a_k > 0 (or one with a_k = 0).

import numpy as np
from scipy.optimize import minimize, minimize_scalar

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_obj(x, n, dx, tau):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / tau
    shifted = np.clip(shifted, -60, 0)
    logsum = np.log(np.sum(np.exp(shifted)))
    return float((m + tau * logsum) * dx)

def fix_sum(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h

def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def slimdown(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    if n_new > n_old:
        return upsample(h_old, n_new)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def coord_descent(h, n_passes=10, tol=1e-10):
    """Coordinate descent for max correlation minimization."""
    n = len(h)
    dx = 2.0 / n
    n_full = 2 * n - 1
    h = h.copy()
    corr = compute_corr(h)
    
    for sweep in range(n_passes):
        improved = False
        for i in range(n):
            old = h[i]
            
            # Compute D_k = partial corr_k / partial h[i]
            D = np.zeros(n_full)
            for k in range(n_full):
                s = k - (n - 1)
                if s == 0:
                    continue
                d = 0.0
                if 0 <= i + s < n:
                    d += (1.0 - h[i + s])
                if 0 <= i - s < n:
                    d += (-h[i - s])
                D[k] = d
            
            # Golden section search for optimal h[i]
            def g(t):
                c_new = corr + (t - old) * D
                c_new[n-1] = corr[n-1] - old * (1.0 - old) + t * (1.0 - t)
                return np.max(c_new) * dx
            
            a, b = 0.0, 1.0
            gr = 0.6180339887498949
            c = b - gr * (b - a)
            d = a + (1.0 - gr) * (b - a)
            fc = g(c)
            fd = g(d)
            for _ in range(25):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    fc = g(c)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + (1.0 - gr) * (b - a)
                    fd = g(d)
            
            new_val = (a + b) / 2.0
            delta = new_val - old
            
            if abs(delta) > 1e-12:
                corr = corr + delta * D
                corr[n-1] = corr[n-1] - old * (1.0 - old) + new_val * (1.0 - new_val)
                h[i] = new_val
                improved = True
        
        # Fix sum constraint
        h = fix_sum(h, n)
        corr = compute_corr(h)
        
        if not improved:
            break
    
    return h

def compute_c5(h):
    dx = 2.0 / len(h)

    return np.correlate(h, 1.0 - h, mode="full")


def smooth_max_obj(x, n, dx, tau):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / tau
    shifted = np.clip(shifted, -60.0, 0.0)
    logsum = np.log(np.sum(np.exp(shifted)))
    return float((m + tau * logsum) * dx)


def fix_sum(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0.0, 1.0)

def upsample(h_old, n_new):

        return h_old.copy()

    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

    for sweep in range(n_passes):
        for i in range(n):

                    continue
                d = 0.0

                    d += (-h[i - s])

                b = 0.3

            c_new_c[n - 1] = corr[n - 1] - old * (1.0 - old) + c * (1.0 - c)
            fc = np.max(c_new_c) * dx

            c_new_d = corr + (d - old) * D
            c_new_d[n - 1] = corr[n - 1] - old * (1.0 - old) + d * (1.0 - d)

                    d = c
                    fd = fc

def subgradient_step(h, alpha, n):
    corr = compute_corr(h)
    k_max = np.argmax(corr)

            if 0 <= i - s < n:
                d += (-h[i - s])
