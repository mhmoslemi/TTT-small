_g_beta = 30.0
_g_top_lags = None
_g_lambda_focus = 0.0

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr * dx

def _project_feasible(h, n):
    h = h.astype(float).copy()
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nfree = np.sum(free)
        if nfree == 0:
            h += diff / n
        else:
            h[free] += diff / nfree
    h = np.clip(h, 0.0, 1.0)
    final_diff = target - np.sum(h)
    if abs(final_diff) > 1e-10:
        free = (h > 0) & (h < 1.0)
        if np.sum(free) > 0:
            h[free] += final_diff / np.sum(free)
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj_lse(x):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx

    h_new = np.interp(x_new, xp, h)
    return _project_feasible(h_new, n_new)

def _subgrad_step(h, n, alpha, n_active):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    active = np.where(corr > m - alpha * 0.01)[0]
    if len(active) == 0:
        active = np.argmax(corr)
        active = np.array([active])
    elif len(active) > n_active:
        idx = np.argsort(-corr[:len(active)])[:n_active]
        active = active[idx]
    grad = np.zeros_like(h)
    for k in active:
        dk = int(k)
        if dk == 0:
            g = 1.0 - 2.0 * h
        elif dk < n:
            # corr[dk] = sum_{i=0}^{n-1-dk} h[i]*(1-h[i+dk])
            # d/dh[i] = (1-h[i+dk]) for i < n-dk
            # d/dh[i+dk] = -h[i] for i+dk < n i.e. i < n-dk
            for i in range(n - dk):
                grad[i] += (1.0 - h[i + dk])
                grad[i + dk] += (-h[i])
        else:
            dk2 = dk - (n - 1)
            # corr[n-1+dk2] = sum_{i=dk2}^{n-1} h[i]*(1-h[i-dk2])
            for i in range(dk2, n):
                grad[i] += (1.0 - h[i - dk2])

    grad = grad / len(active) * dx

    h_new = _project_feasible(h_new, n)
    return h_new
