            best_c5 = c5_val
            best_n = n_cand
            return True
        return False
    
    # Phase 1: Quick optimization at parent resolution
    # Beta 10 for initial
    if remaining() > 150:
        h_opt = run_slsqp(best_h, best_n, best_n / 2.0, beta=8.0, max_iter=100, ftol=1e-12)
        try_update(h_opt, best_n)
    
    # Beta 30
    if remaining() > 120:
        h_opt = run_slsqp(best_h, best_n, best_n / 2.0, beta=30.0, max_iter=150, ftol=1e-13)
        try_update(h_opt, best_n)
    
    # Beta 80
    if remaining() > 80:
        h_opt = run_slsqp(best_h, best_n, best_n / 2.0, beta=80.0, max_iter=100, ftol=1e-13)
        try_update(h_opt, best_n)
    
    # Phase 2: Try higher resolutions
    resolutions = [96, 100, 160, 190, 200, 250, 300]
    for n_try in resolutions:
        if remaining() < 120:
            break
        target_try = n_try / 2.0
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_to_constraint(h_interp, target_try)
        
        # Quick optimize at this resolution
        time_for_opt = min(remaining() * 0.4, 200)
        h_opt1 = run_slsqp(h_interp, n_try, target_try, beta=15.0, max_iter=int(50 + time_for_opt * 0.3), ftol=1e-12)
        try_update(h_opt1, n_try)
        
        if remaining() < 80:
            continue
        
        h_opt2 = run_slsqp(best_h if best_n == n_try else project_to_constraint(interpolate_h(best_h, best_n, n_try), target_try),
                          n_try, target_try, beta=50.0, max_iter=int(30 + remaining() * 0.1), ftol=1e-13)
        try_update(h_opt2, n_try)
    
    # Phase 3: Try random bang-bang starts at even resolutions
    for n_try in [100, 192, 200]:
        if remaining() < 100:
            break
        target_try = n_try / 2.0
        if target_try == int(target_try):
            for attempt in range(3):
                if remaining() < 60:
                    break
                h_rand = random_bangbang(n_try, seed + attempt * 100 + n_try)
                h_rand = project_to_constraint(h_rand, target_try)
                c5_rand = compute_c5(h_rand)
                if c5_rand < 0.5:  # Only worth optimizing if decent start
                    h_opt = run_slsqp(h_rand, n_try, target_try, beta=20.0,
                                     max_iter=80, ftol=1e-13)
                    try_update(h_opt, n_try)
    
    # Phase 4: Local search on best solution
    if remaining() > 60:
        n_best = best_n
        target_best = n_best / 2.0
        ls_time = min(remaining() * 0.35, 300)
        n_iters = int(ls_time * 500)  # ~500 iterations per second estimate
        h_ls, c5_ls = local_search(best_h, n_best, target_best, n_iters, seed + 777, ls_time)
        try_update(h_ls, n_best)
    
    # Phase 5: Coordinate descent on best
    if remaining() > 60:
        n_best = best_n
        target_best = n_best / 2.0
        cd_time = min(remaining() * 0.3, 200)
        h_cd, c5_cd = coordinate_descent(best_h, n_best, target_best, beta=40.0,
                                         sweeps=20, time_limit=cd_time)
        try_update(h_cd, n_best)
    
    # Phase 6: Final sharp SLSQP
    if remaining() > 40:
        n_best = best_n
        target_best = n_best / 2.0
        h_final = run_slsqp(best_h, n_best, target_best, beta=150.0,
                           max_iter=int(min(80, remaining() * 0.5)), ftol=1e-14)
        try_update(h_final, best_n)
    
    # Phase 7: One more local search pass
    if remaining() > 30:
        n_best = best_n
        target_best = n_best / 2.0
        h_ls2, c5_ls2 = local_search(best_h, n_best, target_best,
                                     int(remaining() * 0.7 * 400), seed + 999, remaining() * 0.6)
        try_update(h_ls2, best_n)
    
    # Final cleanup
    n_final = best_n
    tgt_final = n_final / 2.0
    best_h = project_to_constraint(best_h, tgt_final)
    final_c5 = compute_c5(best_h)
    
    # If somehow the constraint is off, fix it
    s = np.sum(best_h)
    if abs(s - tgt_final) > 1e-12:
        best_h = project_to_constraint(best_h, tgt_final)
        final_c5 = compute_c5(best_h)
    
    return (best_h, final_c5, best_n)
