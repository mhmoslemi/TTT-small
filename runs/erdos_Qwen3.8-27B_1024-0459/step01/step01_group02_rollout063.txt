def binary_local_search(h_init, n, max_iter=100000, time_limit=30):
    """Local search on binary solutions using swap moves"""
    h = (h_init > 0.5).astype(float).copy()
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h))
    n_want = n // 2
    if n_ones > n_want:
        ones = np.where(h == 1)[0]
        order = np.random.permutation(ones)[:n_ones - n_want]
        h[order] = 0

        zeros = np.where(h == 0)[0]
        order = np.random.permutation(zeros)[:n_want - n_ones]
        h[order] = 1
    
    best_c5 = compute_c5(h)

    start = time.time()
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])

For the binary case:
- Compute corr = np.correlate(h, 1-h, 'full')
- Find k* = argmax(corr)
- The shift is k* - (n-1)
- To reduce corr at k*, I want to find pairs (a in S, b in S^c) such that swapping reduces corr at k*

For shift k (where k = k*-(n-1)):
corr[k*] = sum_j h[j+k] * (1-h[j])

A swap (a: 1→0, b: 0→1) changes corr by:
Δcorr = -[a-k valid] * (1-h[a-k]) + [a+k valid] * h[a+k] + [b-k valid] * (1-h[b-k]) - [b+k valid] * h[b+k]

To minimize corr[k*], I want Δcorr < 0, i.e.:
h[a+k] + (1-h[b-k]) < (1-h[a-k]) + h[b+k]

For binary: h[a+k] + (1-h[b-k]) < (1-h[a-k]) + h[b+k]

This gives conditions on the relationship between a and b relative to k.

But checking all 48*48 pairs for the best one is 2304 evaluations, each O(1) (just checking a few array values). So I can do a full neighborhood search in ~2304 ops per iteration.

import numpy as np
from scipy.optimize import minimize
import time

_G_n = 0
_G_target = 0.0
_G_beta = 50.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(x):
    n = _G_n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _G_beta
    exps = np.exp(np.clip(beta * (corr - m), -60, 60))
    return (m + np.log(float(np.sum(exps))) / beta) * dx

def obj_raw(x):
    n = _G_n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(corr) * dx)

def con_eq(x):
    return np.sum(x) - _G_target

def _ensure_balance(h, n, rng):

        to_add = rng.choice(zeros, n_want - n_ones, replace=False)

    return h

def binary_ls(h_init, n, time_limit, rng, directed=True):

            shift = k_star - (n - 1)

            f_vals = np.zeros(len(one_indices))
            g_vals = np.zeros(len(zero_indices))

            for i in range(len(one_indices)):
                a = one_indices[i]
                fv = 0.0
                if 0 <= a - shift < n:

def simulated_annealing_bin(h_init, n, time_limit, rng):
    h = _ensure_balance(h_init, n, rng)
    c5 = compute_c5(h)
    best_h = h.copy()

def subgradient_step(h, n, steps, time_limit):
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    alpha = 0.05
    start = time.time()

    for it in range(steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        shift = k_star - (n - 1)

        s = np.zeros(n)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        if lo1 < hi1:
            s[lo1:hi1] += (1.0 - h[lo1 + shift:hi1 + shift])
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if lo2 < hi2:
            s[lo2:hi2] -= h[lo2 - shift:hi2 - shift]

        h = h - alpha * s
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        alpha *= 0.9995
        alpha = max(alpha, 0.0005)

    return best_h, best_c5
