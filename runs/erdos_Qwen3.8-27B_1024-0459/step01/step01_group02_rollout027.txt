            h_new[i] += delta
            h_new[j] -= delta
            if h_new[i] <= 1.0 and h_new[j] >= 0.0:
                h_new = project_feasible(h_new, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
                    break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        if parent_n > 0:
            parent_h = project_feasible(parent_h, parent_n)
            parent_c5 = compute_c5(parent_h)
            best_h = parent_h.copy()
            best_c5 = parent_c5
            best_n = parent_n
    except Exception:
        parent_h = None
        parent_n = 0
    
    if best_h is None:
        best_n = 96
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)
    
    print("Initial: n={}, C5={:.6f}".format(best_n, best_c5))
    
    def elapsed():
        return time.time() - start_time
    
    def remaining():
        return max(0.0, budget_s - elapsed() - 10.0)
    
    # Resolutions to try
    resolutions = sorted(set([parent_n if parent_n > 0 else 96, 96, 100, 128, 95]))
    resolutions = [r for r in resolutions if 64 <= r <= 200]
    
    for n in resolutions:
        if remaining() < 100:
            break
        
        print("Optimizing at n={}".format(n))
        
        # Multiple DE runs with different seeds
        for run_seed in [seed, seed + 1, seed + 2]:
            if remaining() < 50:
                break
            
            bounds = [(0.0, 1.0)] * n
            
            # Custom objective with penalty
            def obj(h):
                return de_penalized(h, n)
            
            # Initial population hint: include parent (interpolated)
            # scipy DE doesn't support custom init easily, so we just run it
            
            try:
                result = differential_evolution(
                    obj, bounds,
                    seed=run_seed,
                    maxiter=80,
                    popsize=30,
                    tol=1e-10,
                    mutation=(0.5, 1.0),
                    recombination=0.9,
                    polish=False,  # We'll do our own polish
                    init='latinhypercube',
                    update_interval=0,
                    polish=True  # Actually let scipy polish
                )
                h_de = project_feasible(result.x, n)
                c5_de = compute_c5(h_de)
                if c5_de < best_c5:
                    best_c5 = c5_de
                    best_h = h_de.copy()
                    best_n = n
                    print("  DE improved: n={}, C5={:.6f}".format(n, best_c5))
            except Exception as e:
                print("  DE failed: {}".format(e))
        
        # Local search on the best so far (interpolated to this n)
        if remaining() > 30:
            h_ls = best_h.copy()
            if len(h_ls) != n:
                x_old = (np.arange(len(h_ls)) + 0.5) * (2.0 / len(h_ls))
                x_new = (np.arange(n) + 0.5) * (2.0 / n)
                h_ls = np.interp(x_new, x_old, h_ls)
                h_ls = project_feasible(h_ls, n)
            
            ls_rng = np.random.RandomState(seed + 100 + n)
            h_ls, c5_ls = local_search(h_ls, n, ls_rng, n_trials=50000, time_limit=min(60, remaining()-10))
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
                best_n = n
                print("  Local search improved: n={}, C5={:.6f}".format(n, best_c5))
    
    # Final polish: coordinate descent on best
    if remaining() > 20:
        n_final = best_n
        h_final = best_h.copy()
        # Try small random perturbations
        for trial in range(int(min(remaining()-10, 30) * 50)):
            if elapsed() > budget_s - 5:
                break
            i = rng.randint(n_final)
            j = rng.randint(n_final)
            if i == j:
                continue
            delta = rng.uniform(-0.1, 0.1)
            h_new = h_final.copy()
            h_new[i] += delta
            h_new[j] -= delta
            if 0 <= h_new[i] <= 1 and 0 <= h_new[j] <= 1:
                h_new = project_feasible(h_new, n_final)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
    
    # Try binary rounding
    if best_n % 2 == 0:
        h_bin = binary_rounding(best_h, best_n)
        if h_bin is not None:
            c5_bin = compute_c5(h_bin)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                print("  Binary rounding: C5={:.6f}".format(best_c5))
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
