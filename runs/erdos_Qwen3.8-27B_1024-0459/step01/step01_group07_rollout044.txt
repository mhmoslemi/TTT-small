def compute_smooth_grad(h, n_points, temp=20.0):
    n = n_points
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(conv)
    weights = np.exp((conv - c_max) / temp)
    weights /= weights.sum()

        wk = weights[k] * dx
        if wk < 1e-12:
            continue
        m = k - (n - 1)
        if m >= 0:
            # First term: (1-h[j+m]) for j=0..n-1-m
            grad[:n-m] += wk * (1.0 - h[m:])
            # Second term: -h[j-m] for j=m..n-1
            grad[m:] += -wk * h[:n-m]

            grad[p:] += wk * (1.0 - h[:n-p])
            # Second term: -h[j+p] for j=0..n-1-p
            grad[:n-p] += -wk * h[p:]
    return grad

import numpy as np
import time


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(conv)


def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-15
            if not np.any(free):
                break
            step = diff / np.sum(free)
            h[free] = np.minimum(h[free] + step, 1.0)
        else:
            free = h > 1e-15
            if not np.any(free):
                break
            step = diff / np.sum(free)
            h[free] = np.maximum(h[free] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        free = (h > 0.005) & (h < 0.995)
        if np.any(free):
            h[free] += (target - s) / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_grad(h, n_points, temp):
    n = n_points
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(conv)
    weights = np.exp((conv - c_max) / temp)
    weights /= weights.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-12:
            continue
        m = k - (n - 1)
        if m >= 0:
            grad[:n - m] += wk * (1.0 - h[m:]) * dx
            grad[m:] += -wk * h[:n - m] * dx
        else:
            p = -m
            grad[p:] += wk * (1.0 - h[:n - p]) * dx
            grad[:n - p] += -wk * h[p:] * dx
    return grad


def adam_optimize(h0, n_points, time_limit, lr=0.05, beta1=0.9, beta2=0.999,
                  eps=1e-8, temp_start=40.0, temp_end=3.0, n_iter=50000):
    n = n_points
    t_start = time.time()
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_val = compute_c5(h, n)
    no_improve = 0
    t = 0
    while t < n_iter:
        elapsed = time.time() - t_start
        if elapsed > time_limit * 0.95:
            break
        t += 1
        progress = min(1.0, t / n_iter)
        temp = temp_start * (temp_end / temp_start) ** progress
        grad = smooth_max_grad(h, n, temp)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        actual_val = compute_c5(h_new, n)
        if actual_val < best_val - 1e-12:
            best_val = actual_val
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        if no_improve > 800 and t < n_iter * 0.7:
            h = h + 0.08 * np.random.randn(n)
            h = project_feasible(h, n)
            no_improve = 0
            m = np.zeros(n)
            v = np.zeros(n)
    return best_h, best_val


def structured_init(n, rng, seed_offset=0):

    if strategy == 0:
        h = rng.uniform(0, 1, n)

    elif strategy == 3:

    elif strategy == 4:

        if time_left() < 250:
            break
        x_coarse = np.linspace(0, 2, best_n, endpoint=False)
        x_fine_grid = np.linspace(0, 2, n_fine, endpoint=False)
        h_fine = np.interp(x_fine_grid, x_coarse, best_h)
        h_fine = project_feasible(h_fine, n_fine)
        tl = min(400, time_left() - 150)
        h_try, v_try = adam_optimize(h_fine, n_fine, tl, lr=0.03,
                                     temp_start=35.0, temp_end=3.0, n_iter=50000)
        if v_try < best_val:
            best_h, best_val, best_n = h_try, v_try, n_fine
        # Random init at this resolution
        if time_left() > 200:
            h_rand_fine = structured_init(n_fine, rng)
            tl2 = min(200, time_left() - 100)
            h_try, v_try = adam_optimize(h_rand_fine, n_fine, tl2, lr=0.04,
                                         temp_start=40.0, temp_end=4.0, n_iter=40000)
            if v_try < best_val:
                best_h, best_val, best_n = h_try, v_try, n_fine

    # Phase 4: Low-temp polish on best
    if time_left() > 80:
        tl = min(120, time_left() - 40)
        h_try, v_try = adam_optimize(best_h, best_n, tl, lr=0.015,
                                     temp_start=10.0, temp_end=1.0, n_iter=60000)
        if v_try < best_val:
            best_h, best_val, best_n = h_try, v_try, best_n

    # Phase 5: Nudge / coordinate descent
    if time_left() > 40:
        h_try, v_try = nudge_search(best_h, best_n, time_left() - 20, n_rounds=500)
        if v_try < best_val:
            best_h, best_val = h_try, v_try

    # Phase 6: Final tiny Adam
    if time_left() > 20:
        h_try, v_try = adam_optimize(best_h, best_n, time_left() - 10, lr=0.01,
                                     temp_start=5.0, temp_end=0.5, n_iter=30000)
        if v_try < best_val:
            best_h, best_val, best_n = h_try, v_try, best_n

    best_h = project_feasible(best_h, best_n)
    best_val = compute_c5(best_h, best_n)
    return (best_h, best_val, best_n)
