    n = h.size
    target = 0.5 * n
    best_c = evaluate_erdos_local(h)
    stall = 0
    for _ in range(500):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        center = n - 1
        idxs = np.argsort(corr)[-top_k:]
        # Compute gradient of average of top-k correlations
        g = np.zeros(n)
        for t in idxs:
            t = int(t)
            k = t - center
            if k == 0:
                g += (1.0 - 2.0 * h)
            else:
                lag = abs(k)
                if lag >= n:
                    continue
                if k > 0:
                    g[lag:] += (1.0 - h[:n - lag])
                    g[:n - lag] -= h[lag:]
                else:
                    g[:n - lag] += (1.0 - h[lag:])
                    g[lag:] -= h[:n - lag]
        g /= float(top_k)
        # Project gradient to tangent space (sum=0)
        g -= np.mean(g)
        norm = np.linalg.norm(g)
        if norm < 1e-14:
            break
        d = -g / norm
        improved = False
        for step in (0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4):
            cand = project_density(h + step * d, target)
            c = evaluate_erdos_local(cand)
            if c < best_c - 1e-13:
                h = cand
                best_c = c
                improved = True
                break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
    return h, best_c

def slsqp_optimize(h0, n, deadline, alpha_list=None, maxiter_per_alpha=120):
    n = int(n)
    target = 0.5 * n
    if alpha_list is None:
        if n <= 100:
            alpha_list = [5, 15, 50, 150, 400, 1000]
        elif n <= 250:
            alpha_list = [8, 25, 80, 200, 600]
        else:
            alpha_list = [10, 30, 100, 300]
    _G['n'] = n
    _G['target'] = target
    h = project_density(np.asarray(h0, dtype=float), target)
    best_h = h.copy()
    best_c = evaluate_erdos_local(best_h)
    bounds = [(0.0, 1.0)] * n
    constraints = ({"type": "eq", "fun": sum_con, "jac": sum_con_jac},)
    for alpha in alpha_list:
        if time.perf_counter() >= deadline - 5.0:
            break
        _G['alpha'] = float(alpha)
        try:
            res = minimize(
                smooth_obj, h, method="L-BFGS-B", jac=smooth_obj_jac,
                bounds=bounds,
            )
            # L-BFGS-B doesn't handle equality constraints, so we project
            xg = np.asarray(res.x, dtype=float)
            if np.all(np.isfinite(xg)):
                cand = project_density(xg, target)
                c = evaluate_erdos_local(cand)
                if c < best_c - 1e-12:
                    best_h = cand
                    best_c = c
                    h = cand
                elif c < best_c + 0.05:
                    h = cand
        except Exception:
            pass
        # Also try SLSQP for the equality constraint
        if time.perf_counter() < deadline - 10.0:
            try:
                res2 = minimize(
                    smooth_obj, h, method="SLSQP", jac=smooth_obj_jac,
                    bounds=bounds, constraints=constraints,
                    options={"maxiter": maxiter_per_alpha, "ftol": 1e-10, "disp": False}
                )
                xg2 = np.asarray(res2.x, dtype=float)
                if np.all(np.isfinite(xg2)):
                    cand2 = project_density(xg2, target)
                    c2 = evaluate_erdos_local(cand2)
                    if c2 < best_c - 1e-12:
                        best_h = cand2
                        best_c = c2
                        h = cand2
            except Exception:
                pass
    return best_h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 12.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float('inf')
    best_n = 0

    # Load parent
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.clip(np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        nparent = parent.size
        proj_parent = project_density(parent, 0.5 * nparent)
        cp = evaluate_erdos_local(proj_parent)
        if cp < best_c:
            best_h = proj_parent.copy()
            best_c = cp
            best_n = nparent

    if best_h is None:
        best_h = make_binary_pattern(90, 1, rng)
        best_c = evaluate_erdos_local(best_h)
        best_n = 90

    n_parent = int(best_h.size)

    # Phase 1: Optimize at parent resolution with multiple strategies
    # 1a: Binary local search
    if deadline - time.perf_counter() > 30:
        h_bin = project_density(np.where(best_h > 0.5, 1.0, 0.0), 0.5 * n_parent)
        h_bin, c_bin = targeted_local_search(h_bin, min(deadline - 20, start + 80), max_iter=5000, top_k=8)
        if c_bin < best_c:
            best_h, best_c, best_n = h_bin, c_bin, n_parent

    # 1b: Gradient refine from parent
    if deadline - time.perf_counter() > 20:
        h_gr, c_gr = gradient_refine(best_h, min(deadline - 10, start + 120), top_k=10)
        if c_gr < best_c:
            best_h, best_c, best_n = h_gr, c_gr, n_parent

    # 1c: SLSQP at parent res
    if deadline - time.perf_counter() > 30:
        h_s, c_s = slsqp_optimize(best_h, n_parent, min(deadline - 10, start + 250))
        if c_s < best_c:
            best_h, best_c, best_n = h_s, c_s, n_parent

    # 1d: More binary local search from best
    if deadline - time.perf_counter() > 20:
        h_bin2 = project_density(np.where(best_h > 0.5, 1.0, 0.0), 0.5 * n_parent)
        h_bin2, c_bin2 = targeted_local_search(h_bin2, min(deadline - 10, start + 300), max_iter=8000, top_k=10)
        if c_bin2 < best_c:
            best_h, best_c, best_n = h_bin2, c_bin2, n_parent

    # 1e: Multi-start with structured patterns at parent res
    for mode in [0, 2, 3, 4]:
        if deadline - time.perf_counter() < 40:
            break
        h_new = make_binary_pattern(n_parent, mode, rng)
        h_new, c_new = targeted_local_search(h_new, min(deadline - 15, start + 350 + 30 * mode), max_iter=3000, top_k=6)
        h_new, c_new2 = gradient_refine(h_new, min(deadline - 10, start + 370 + 30 * mode), top_k=8)
        if c_new2 < c_new:
            c_new = c_new2
        if c_new < best_c:
            best_h, best_c, best_n = h_new, c_new, n_parent

    # Phase 2: Upsample to 2x and optimize
    n2 = n_parent * 2
    if deadline - time.perf_counter() > 80 and n2 <= 360:
        h_up = upsample_h(best_h, n2)
        # Binary local search at higher res
        if deadline - time.perf_counter() > 40:
            h_b2 = project_density(np.where(h_up > 0.5, 1.0, 0.0), 0.5 * n2)
            h_b2, c_b2 = targeted_local_search(h_b2, min(deadline - 20, start + 550), max_iter=4000, top_k=8)
            if c_b2 < best_c:
                best_h, best_c, best_n = h_b2, c_b2, n2
            h_up = h_b2
        # SLSQP at higher res
        if deadline - time.perf_counter() > 30:
            h_s2, c_s2 = slsqp_optimize(h_up, n2, min(deadline - 10, start + 800))
            if c_s2 < best_c:
                best_h, best_c, best_n = h_s2, c_s2, n2

    # Phase 3: Upsample to 4x if time permits
    n4 = n_parent * 4
    if deadline - time.perf_counter() > 100 and n4 <= 360:
        h_up4 = upsample_h(best_h, n4)
        if deadline - time.perf_counter() > 50:
            h_s4, c_s4 = slsqp_optimize(h_up4, n4, min(deadline - 10, start + 1100))
            if c_s4 < best_c:
                best_h, best_c, best_n = h_s4, c_s4, n4

    # Phase 4: Final polish
    if best_h is not None and deadline - time.perf_counter() > 15:
        nb = int(best_h.size)
        # Gradient refine
        if deadline - time.perf_counter() > 10:
            h_pf, c_pf = gradient_refine(best_h, deadline - 5, top_k=12)
            if c_pf < best_c:
                best_h, best_c = h_pf, c_pf
        # More SLSQP with high alpha
        if deadline - time.perf_counter() > 15:
            _G['n'] = nb
            _G['target'] = 0.5 * nb
            _G['alpha'] = 2000.0
            try:
                bounds = [(0.0, 1.0)] * nb
                constraints = ({"type": "eq", "fun": sum_con, "jac": sum_con_jac},)
                res = minimize(smooth_obj, best_h, method="SLSQP", jac=smooth_obj_jac,
                              bounds=bounds, constraints=constraints,
                              options={"maxiter": 60, "ftol": 1e-12, "disp": False})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    cand = project_density(res.x, 0.5 * nb)
                    c = evaluate_erdos_local(cand)
                    if c < best_c:
                        best_h, best_c = cand, c
            except Exception:
                pass
        # Final binary local search
        if deadline - time.perf_counter() > 8:
            h_fin = project_density(np.where(best_h > 0.5, 1.0, 0.0), 0.5 * nb)
            h_fin, c_fin = targeted_local_search(h_fin, deadline - 3, max_iter=2000, top_k=10)
            if c_fin < best_c:
                best_h, best_c = h_fin, c_fin

    # Final projection and evaluation
    if best_h is None:
        best_h = project_density(np.full(90, 0.5), 45.0)
        best_n = 90
    best_h = project_density(best_h, 0.5 * int(best_h.size))
    best_c = evaluate_erdos_local(best_h)
    best_n = int(best_h.size)
    return best_h, float(best_c), best_n
