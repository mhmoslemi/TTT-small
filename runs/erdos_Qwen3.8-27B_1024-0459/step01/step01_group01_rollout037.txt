    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    weights = np.exp((corr - m) / tau)
    sw = weights.sum()
    if sw < 1e-300:
        return np.zeros(n)
    weights /= sw
    
    # k for each idx: k = idx - (n-1)
    # grad[j] = dx * tau * sum_idx weights[idx] * ((1-h[j-k])*[0<=j-k<n] - h[j+k]*[0<=j+k<n])
    
    grad = np.zeros(n)
    for idx in range(2*n-1):
        k = idx - (n-1)
        w = weights[idx]
        if w < 1e-12:
            continue
        # Part 1: (1-h[j-k]) for j where 0<=j-k<n, i.e., k<=j<=k+n-1, intersect [0,n-1]
        js = max(0, k)
        je = min(n-1, k+n-1)
        if js <= je:

            mg = range(js-k, je-k+1)  # indices into h for (1-h[...])

        js2 = max(0, -k)
        je2 = min(n-1, n-1-k)
        if js2 <= je2:
            # j+k ranges from js2+k to je2+k
            mg2 = range(js2+k, je2+k+1)
            grad[js2:je2+1] -= w * h[mg2]
    
    grad *= dx * tau
    return grad
