from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_and_shifts(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr * dx

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.array(h, dtype=float).copy()
    # Clip
    h = np.clip(h, 0, 1)
    # Adjust sum
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Distribute adjustment
    h = h + diff / n
    h = np.clip(h, 0, 1)
    # Fix remaining
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            # Need to increase sum; add to elements < 1
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if np.sum(mask) > 0:
            per = diff / np.sum(mask)
            h[mask] += per
            h = np.clip(h, 0, 1)
    # Final fix: force exact sum
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        # Find one element to adjust
        if diff > 0:
            idx = np.argmin(h)  # smallest has most room to grow
        else:
            idx = np.argmax(h)
        h[idx] += diff
        h = np.clip(h, 0, 1)
    return h

    m = np.max(corr)

    exps = np.exp(beta * (corr - m))
    return (m + np.log(np.sum(exps)) / beta) * dx

    dx = 2.0 / n

    s = k_star - (n - 1)  # effective shift

def coordinate_descent(h, n, max_inner_iters=500, tol=1e-10):
    """
    Coordinate descent: try adjusting pairs (i, j) to reduce the max correlation.
    """
    dx = 2.0 / n
    best_c5 = compute_c5(h)

        improved = False

            # If h[i]=0 and h[j]=1, this is a (0,1) pair -> doesn't contribute to this shift's overlap

Let me also think about a key mathematical insight: The optimal solution likely has h values that are very close to 0 or 1 (bimodal), with the constraint sum=n/2 forcing exactly half to be 1 and half to be 0. The "fractional" values (between 0 and 1) only appear at a few positions to fine-tune.

For binary solutions (n even), the problem is combinatorial: find a {0,1} sequence of length n with n/2 ones minimizing max_s #{i: h[i]=1, h[i+s]=0}.

This is equivalent to: minimize max_s (number of 1s in first n-s positions that are followed by a 0 after s positions).

For a binary sequence, let's denote A = set of positions with 1, B = set of positions with 0.
For shift s: count = |{i ∈ A : i-s ∈ B}| = |{i ∈ A : h[i-s] = 0}|

This is related to the "binary autocorrelation" problem.

Let me try a different approach: **Simulated Annealing** on the binary problem.
- State: binary sequence with n/2 ones
- Move: swap a 1 with a 0 (maintains the count)
- Energy: C₅ (max correlation)
- Accept with probability exp(-ΔE/T)

This is simple, effective, and naturally handles the discrete structure.

But wait - if we restrict to binary, we might not get the best answer. The continuous relaxation might be slightly better. However, the improvement from going binary → continuous is likely very small (on the order of 1/n).

Let me implement a hybrid:
1. Simulated annealing / local search on binary sequences (at even n)
2. Then relax to continuous and do SLSQP polish
3. Compare with direct SLSQP on continuous

For the binary local search, the key operation is:
- Pick a position i with h[i]=1 and a position j with h[j]=0
- Swap them
- Recompute the affected correlations
- Accept if improvement (or with probability for SA)

The affected correlations are those k where the swap changes the count. If we change h[i] from 1 to 0, then for each shift s:
- The pair (i, i+s) changes from (1,?) to (0,?) - removes a (1,0) if h[i+s]=0
- The pair (i-s, i) changes from (?,1) to (?,0) - adds a (1,0) if h[i-s]=1

Similarly for h[j] going from 0 to 1.

This can be done efficiently if we maintain the correlation array.

Let me implement this:

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    # Force exact sum
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    # For numerical stability
    max_shifted = np.max(shifted)
    if max_shifted > 50:
        shifted = shifted - max_shifted
        return (m + (m + max_shifted/beta + np.log(np.sum(np.exp(shifted))) / beta)) * dx
    exps = np.exp(shifted)
    log_sum = np.log(np.sum(exps))
    return (m + log_sum / beta) * dx

    """Sum of top-k correlations, as a smooth proxy for max"""

    Simulated annealing on binary sequence.
    h_bin must be a binary array with exactly n/2 ones.
    """

    h = h_bin.copy()

    zeros = np.where(h == 0.0)[0]

        j = zeros[rng_local.randint(len(zeros))]

        h[i], h[j] = 0.0, 1.0
        
        # Recompute correlation
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = np.max(new_corr) * dx

            if new_c5 < best_c5:
                best_c5 = new_c5
        else:
            # Reject
            h[i], h[j] = 1.0, 0.0
    
    return h, best_c5

A better approach is to use the **average subgradient** (average over the last K iterations' active shifts).

Actually, an even better approach for this specific problem is the **multiplicative weights / mirror descent** method:
- Maintain weights w_k for each shift k
- The objective becomes Σ_k w_k * f_k(h)
- Update w_k proportional to f_k(h)
- Optimize the weighted sum

This is the dual approach to the primal max problem.
