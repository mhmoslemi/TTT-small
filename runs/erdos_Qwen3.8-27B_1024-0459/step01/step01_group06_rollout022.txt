import time

def compute_c5_array(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_c5_array(h)) * (2.0 / n))

def softmax_obj(h, beta):
    n = len(h)
    c = compute_c5_array(h) * beta
    m = np.max(c)
    val = (m + np.log(np.sum(np.exp(c - m)))) / beta
    return float(val * (2.0 / n))

    n = len(h)
    dx = 2.0 / n
    c = compute_c5_array(h) * beta
    m = np.max(c)
    w = np.exp(c - m)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-13:
            continue
        k_j = j - (n - 1)
        if k_j >= 0:
            # d(c_j)/d(h[p]) = (1-h[p+k_j]) for p in [0, n-1-k_j)
            #                - h[p-k_j] for p in [k_j, n)

                grad[k_j:] -= w[j] * h[:n-k_j]
        else:
            d = -k_j
            if n - d > 0:
                grad[d:] += w[j] * (1.0 - h[:n-d])
            grad[:n-d] -= w[j] * h[d:]
    return grad * dx

        d = -k

            grad[d:] += (1.0 - h[:n-d])
        grad[:n-d] -= h[d:]

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

        if diff > 0:

        else:

    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old * 2.0
    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    return np.interp(x_new, x_old, h_old)

def coordinate_descent_step(h, n, max_rounds=5):
    """Try to improve by adjusting individual coordinates."""
    h = h.copy()
    dx = 2.0 / n
    target = n / 2.0
    best_c = compute_c5(h)
    
    for _round in range(max_rounds):
        improved = False
        for i in range(n):
            # Try adjusting h[i] slightly
            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                h_trial = h.copy()
                h_trial[i] = np.clip(h[i] + delta, 0.0, 1.0)
                # Fix sum constraint by adjusting another coordinate
                diff = target - h_trial.sum()
                if abs(diff) > 1e-12:
                    # Distribute correction to all non-bound coordinates
                    mask = (h_trial > 1e-6) & (h_trial < 1.0 - 1e-6)
                    if mask.sum() == 0:

                    h_trial[mask] += diff / mask.sum()
                    h_trial = np.clip(h_trial, 0.0, 1.0)
                if compute_c5(h_trial) < best_c - 1e-10:
                    h = h_trial
                    best_c = compute_c5(h)
                    improved = True
                    break
        if not improved:
