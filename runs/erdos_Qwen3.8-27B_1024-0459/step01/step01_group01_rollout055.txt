import numpy as np
from scipy.optimize import minimize
import functools
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h


def smooth_max_val(x, tau):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    if tau < 1e-12:
        return m * dx
    return (m + tau * np.log(np.sum(np.exp((corr - m) / tau)))) * dx


def smooth_max_grad(x, tau):
    h = np.clip(x, 0.0, 1.0)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)

    if tau < 1e-12:
        s_max = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            js = j - s_max
            if 0 <= js < n:
                grad[j] += (1.0 - h[js])
            jp = j + s_max
            if 0 <= jp < n:
                grad[j] -= h[jp]
        return grad * dx

    w = np.exp((corr - m) / tau)
    w_sum = np.sum(w)
    w = w / w_sum

    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        for s in range(-(n - 1), n):
            k = n - 1 + s
            if w[k] > 1e-15:
                js = j - s
                if 0 <= js < n:
                    val += w[k] * (1.0 - h[js])
                jp = j + s
                if 0 <= jp < n:
                    val -= w[k] * h[jp]
        grad[j] = val * dx
    return grad


def _con_fun(x, n):
    return np.array([x.sum() - n / 2.0])


def _con_jac(x, n):
    return np.array([np.ones(n)])


def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def downsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def binary_local_search(h, n, max_iters=5000, rng=None):
    """Flip 0->1 and 1->0 pairs to reduce max overlap."""
    h = fix_sum(h, n).copy()
    if rng is None:
        rng = np.random.default_rng(0)
    best = compute_c5(h)
    ones = np.where(h > 0.99)[0]
    zeros = np.where(h < 0.01)[0]
    for _ in range(max_iters):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h)
        if c < best - 1e-15:
            best = c
        else:
            h[i], h[j] = h[j], h[i]
    return h, best


def periodic_binary_init(n, period, duty, rng):
    """Create a binary periodic sequence with given period and duty cycle."""
    h = np.zeros(n)
    for i in range(n):
        phase = (i % period)
        if phase < duty:
            h[i] = 1.0
    h = fix_sum(h, n)
    return h
