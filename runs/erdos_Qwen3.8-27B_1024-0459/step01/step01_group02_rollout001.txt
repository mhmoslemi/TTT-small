def compute_subgrad(h, tol_frac=1e-6):
   n=len(h); dx=2/n
   corr=np.correlate(h,1-h,'full')
   m=corr.max(); c=m*dx
   tol=max(1e-10, tol_frac*max(1.0,m))
   idxs=np.where(corr >= m - tol)[0]
   if len(idxs)==0: idxs=np.array([int(np.argmax(corr))])
   slacks=corr[idxs] - (m - tol)
   if slacks.sum()<=0:
      weights=np.full_like(idxs,1.0/len(idxs),dtype=float)
   else:
      weights=slacks/slacks.sum()
   center=n-1; g=np.zeros(n)
   for idx,ww in zip(idxs,weights):
      if ww <=1e-12: continue
      k=idx-center
      # add ww * grad_lag
      p_start=max(0,k); p_end=min(n-1,n-1+k)
      if p_start <= p_end:
         p=np.arange(p_start,p_end+1)
         g[p] += ww*(1.0 - h[p-k])
      p_start=max(0,-k); p_end=min(n-1,n-1-k)
      if p_start <= p_end:
         p=np.arange(p_start,p_end+1)
         g[p] -= ww*h[p+k]
   return dx*g, c

def compute_subgrad(h, tol_frac=1e-6):

def eval_c5(h):
    n = len(h)
    if n == 0:
        return float("inf")
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / float(n)))


def corr_full(h):

def project_feasible(v, target=None):

def round_to_binary(h, target=None):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if target is None:
        target = 0.5 * n
    target = float(target)
    order = np.argsort(-h)
    x = np.zeros(n, dtype=float)
    k = int(np.floor(target + 1e-12))
    if k > 0:
        x[order[:k]] = 1.0
    fr = target - float(k)
    if fr > 1e-12 and k < n:
        x[order[k]] = 0.5
    return project_feasible(x, target)


def interpolate_to_n(h, n):
    m = len(h)
    if m == 0:
        return np.zeros(int(n), dtype=float)
    dx_old = 2.0 / float(m)
    dx_new = 2.0 / float(n)
    xp = (np.arange(m) + 0.5) * dx_old
    xq = (np.arange(n) + 0.5) * dx_new
    y = np.interp(xq, xp, h.astype(float))
    return project_feasible(y, 0.5 * n)


def make_sinusoid(n, periods, amp, phase, periods2=0, amp2=0.0, phase2=0.0):
    t = (np.arange(n) + 0.5) / float(n)
    h = 0.5 + float(amp) * np.cos(2.0 * np.pi * float(periods) * t + float(phase))
    if periods2:
        h = h + float(amp2) * np.cos(2.0 * np.pi * float(periods2) * t + float(phase2))
    return project_feasible(h, 0.5 * n)


def random_binary(n, rng):
    target = 0.5 * n
    k = int(np.floor(target + 1e-12))
    x = np.zeros(n, dtype=float)
    if k > 0:
        idx = rng.choice(n, k, replace=False)
        x[idx] = 1.0
    if target - float(k) > 1e-12:
        zeros = np.where(x == 0.0)[0]
        if zeros.size > 0:
            x[zeros[int(rng.randint(zeros.size))]] = 0.5
    return project_feasible(x, target)


def add_grad_for_lag(g, h, k, w):
    n = len(h)
    s = max(0, int(k))
    e = min(n - 1, n - 1 + int(k))
    if s <= e:
        g[s:e + 1] += w * (1.0 - h[s - int(k):e + 1 - int(k)])
    s = max(0, -int(k))
    e = min(n - 1, n - 1 - int(k))
    if s <= e:
        g[s:e + 1] -= w * h[s + int(k):e + 1 + int(k)]


def compute_subgrad(h, tol_frac=1e-6):
    n = len(h)
    dx = 2.0 / float(n)
    corr = corr_full(h)
    m = float(np.max(corr))
    c = m * dx
    tol = max(1e-10, float(tol_frac) * max(1.0, m))
    idxs = np.where(corr >= m - tol)[0]
    if idxs.size == 0:
        idxs = np.array([int(np.argmax(corr))])
    slacks = corr[idxs] - (m - tol)
    ss = float(slacks.sum())
    if ss <= 1e-300:
        weights = np.full(idxs.shape, 1.0 / float(idxs.size), dtype=float)
    else:
        weights = slacks / ss
    center = n - 1
    g = np.zeros(n, dtype=float)
    for ii, ww in zip(idxs, weights):
        ww = float(ww)
        if ww <= 1e-14:
            continue
        k = int(ii) - center
        add_grad_for_lag(g, h, k, ww)
    g = g - float(np.mean(g))
    return dx * g, c


def active_term_moves(h, stop_time, rng):
    n = len(h)
    target = 0.5 * n
    cur = np.array(h, dtype=float, copy=True)
    best_c = eval_c5(cur)
    best_h = cur.copy()
    stall = 0
    fracs = (0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5)

    while time.time() < stop_time - 2.0 and stall < 250:
        corr = corr_full(cur)
        m = float(np.max(corr))
        c = m * (2.0 / float(n))
        if c < best_c:
            best_c = c
            best_h = cur.copy()

        improved = False
        idx_sorted = np.argsort(corr)[::-1]
        for idx in idx_sorted[:5]:
            val = float(corr[idx])
            if val < m - max(1e-9, 1e-4 * m):
                break
            k = int(idx) - (n - 1)
            if k == 0:
                continue
            i_start = max(0, -k)
            i_end = min(n - 1, n - 1 - k)
            if i_start > i_end:
                continue
            ii = np.arange(i_start, i_end + 1)
            jj = ii + k
            terms = cur[jj] * (1.0 - cur[ii])
            order = np.argsort(terms)[::-1][:12]
            for t in order:
                p = int(ii[t])
                q = int(jj[t])
                if p == q:
                    continue
                maxd = min(cur[q], 1.0 - cur[p])
                if maxd <= 1e-9:
                    continue
                for f in fracs:
                    if time.time() >= stop_time - 1.0:
                        break
                    d = float(f) * maxd
                    if d <= 1e-9:
                        continue
                    cand = cur.copy()
                    cand[p] = min(1.0, cur[p] + d)
                    cand[q] = max(0.0, cur[q] - d)
                    c2 = eval_c5(cand)
                    if c2 < best_c - 1e-13:
                        best_c = c2
                        best_h = cand.copy()
                        cur = cand
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            g, cg = compute_subgrad(cur, 1e-5)
            if cg < best_c:
                best_c = cg
                best_h = cur.copy()
            gn = float(np.linalg.norm(g))
            if gn > 1e-14:
                gu = g / gn
                for a in (0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002):
                    if time.time() >= stop_time - 1.0:
                        break
                    cand = project_feasible(cur - float(a) * gu, target)
                    c2 = eval_c5(cand)
                    if c2 < best_c - 1e-13:
                        best_c = c2
                        best_h = cand.copy()
                        cur = cand
                        improved = True
                        break

            if not improved:
                scale = 0.002 + 0.001 * min(10, stall)
                noise = rng.normal(0.0, scale, n)
                noise -= float(noise.mean())
                cand = project_feasible(cur + noise, target)
                c2 = eval_c5(cand)
                if c2 < best_c - 1e-14:
                    best_c = c2
                    best_h = cand.copy()
                    cur = cand
                    improved = True

        if improved:
            stall = 0
        else:
            stall += 1

    best_h = project_feasible(best_h, target)
    best_c = eval_c5(best_h)
    return best_h, best_c


def score_pair_search(h, stop_time, rng):
    n = len(h)
    target = 0.5 * n
    cur = np.array(h, dtype=float, copy=True)
    best_c = eval_c5(cur)
    best_h = cur.copy()
    stall = 0
    top_pairs = 40 if n <= 160 else 25
    fracs = (0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0)

    while time.time() < stop_time - 2.0 and stall < 300:
        g, cg = compute_subgrad(cur, 1e-5)
        if cg < best_c:
            best_c = cg
            best_h = cur.copy()

        scores = g[:, None] - g[None, :]
        feasible = (cur[:, None] > 1e-9) & (cur[None, :] < 1.0 - 1e-9)
        diag = np.arange(n)
        feasible[diag, diag] = False
        scores = np.where(feasible, scores, np.inf)
        flat = scores.ravel()
        finite_count = int(np.count_nonzero(flat < np.inf))
        improved = False

        if finite_count > 0:
            K = min(top_pairs, finite_count)
            part = np.argpartition(flat, K - 1)[:K]
            part = part[np.argsort(flat[part])]
            for pidx in part:
                sc = float(flat[pidx])
                if sc >= -1e-12:
                    break
                i = int(pidx // n)
                j = int(pidx % n)
                if i == j:
                    continue
                maxd = min(cur[j], 1.0 - cur[i])
                if maxd <= 1e-9:
                    continue
                for f in fracs:
                    if time.time() >= stop_time - 1.0:
                        break
                    d = float(f) * maxd
                    if d <= 1e-9:
                        continue
                    cand = cur.copy()
                    cand[i] = min(1.0, cur[i] + d)
                    cand[j] = max(0.0, cur[j] - d)
                    c2 = eval_c5(cand)
                    if c2 < best_c - 1e-13:
                        best_c = c2
                        best_h = cand.copy()
                        cur = cand
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            gn = float(np.linalg.norm(g))
            if gn > 1e-14:
                gu = g / gn
                for a in (0.03, 0.015, 0.007, 0.003, 0.0015, 0.0007, 0.0003):
                    if time.time() >= stop_time - 1.0:
                        break
                    cand = project_feasible(cur - float(a) * gu, target)
                    c2 = eval_c5(cand)
                    if c2 < best_c - 1e-13:
                        best_c = c2
                        best_h = cand.copy()
                        cur = cand
                        improved = True
                        break

            if not improved:
                scale = 0.002 + 0.0005 * min(20, stall)
                noise = rng.normal(0.0, scale, n)
                noise -= float(noise.mean())
                cand = project_feasible(cur + noise, target)
                c2 = eval_c5(cand)
                if c2 < best_c - 1e-14:
                    best_c = c2
                    best_h = cand.copy()
                    cur = cand
                    improved = True

        if improved:
            stall = 0
        else:
            stall += 1

    best_h = project_feasible(best_h, target)
    best_c = eval_c5(best_h)
    return best_h, best_c


def slsq_obj(x):
    n = int(_SLSQP_N)
    if n <= 0:
        return 1.0
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(corr))
    dx = 2.0 / float(n)
    beta = float(_SLSQP_BETA)
    if beta <= 1e-12:
        return m * dx
    z = beta * (corr - m)
    z = z - float(np.max(z))
    s = float(np.sum(np.exp(z)))
    if s <= 1e-300:
        return m * dx
    return (m + float(np.log(s)) / beta) * dx


def slsq_con(x):
    return float(np.sum(x) - 0.5 * _SLSQP_N)


def slsqpolish(h, stop_time, rng):
    global _SLSQP_N, _SLSQP_BETA
    n = len(h)
    best = eval_c5(h)
    _SLSQP_N = n
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": slsq_con}]
    for b in (30.0, 100.0):
        if time.time() >= stop_time - 10.0:
            break
        _SLSQP_BETA = b
        try:
            res = minimize(
                slsq_obj,
                h,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 45, "ftol": 1e-12, "iprint": 0},
            )
            h2 = project_feasible(res.x, 0.5 * n)
            c2 = eval_c5(h2)
            if c2 < best:
                h = h2
                best = c2
        except Exception:
            pass
    return h, best


def polish_candidate(h, stop_time, rng):
    h = project_feasible(np.array(h, dtype=float, copy=True), 0.5 * len(h))
    best_c = eval_c5(h)
    now = time.time()
    if stop_time <= now + 10.0:
        return h, best_c

    dur = max(1.0, float(stop_time - now))
    stop1 = min(time.time() + 0.40 * dur, _DEADLINE - 60.0)
    stop2 = min(time.time() + 0.75 * dur, _DEADLINE - 40.0)
    stop3 = min(float(stop_time), _DEADLINE - 10.0)

    if stop1 > time.time() + 10.0:
        h, best_c = active_term_moves(h, stop1, rng)

    if stop2 > time.time() + 10.0:
        h2, c2 = score_pair_search(h, stop2, rng)
        if c2 < best_c:
            h, best_c = h2, c2

    if stop3 > time.time() + 10.0:
        h2, c2 = active_term_moves(h, stop3, rng)
        if c2 < best_c:
            h, best_c = h2, c2

    if len(h) <= 128 and time.time() < _DEADLINE - 120.0 and stop_time - time.time() > 60.0:
        sstop = min(time.time() + 60.0, _DEADLINE - 30.0)
        h2, c2 = slsqpolish(h, sstop, rng)
        if c2 < best_c:
            h, best_c = h2, c2

    h = project_feasible(h, 0.5 * len(h))
    c = eval_c5(h)
    if c < best_c:
        best_c = c
    return h, best_c


def fourier_params_to_h(params, n):
    t = (np.arange(n) + 0.5) / float(n)
    h = np.full(n, 0.5, dtype=float)
    for m in (1, 2, 3):
        amp = float(params[2 * m - 2])
        ph = float(params[2 * m - 1])
        h = h + amp * np.cos(2.0 * np.pi * float(m) * t + ph)
    return project_feasible(h, 0.5 * n)


def fourier_obj(params):
    n = int(_FOURIER_N)
    if n <= 0:
        return 1.0
    h = fourier_params_to_h(params, n)
    return eval_c5(h)


def add_candidate(cand_list, h):
    if h is None:
        return
    n = len(h)
    if n <= 0:
        return
    if not np.all(np.isfinite(h)):
        return
    h = project_feasible(h, 0.5 * n)
    c = eval_c5(h)
    if np.isfinite(c):
        cand_list.append((c, h))


def add_fourier_candidates(cand_list, rng, stop_time):
    global _FOURIER_N
    for n in (95, 96, 128):
        if time.time() >= stop_time:
            return
        _FOURIER_N = n
        for _ in range(3):
            if time.time() >= stop_time:
                return
            x0 = np.empty(6, dtype=float)
            x0[0] = rng.uniform(0.15, 0.45)
            x0[1] = rng.uniform(0.0, 2.0 * np.pi)
            x0[2] = rng.uniform(0.05, 0.25)
            x0[3] = rng.uniform(0.0, 2.0 * np.pi)
            x0[4] = rng.uniform(0.00, 0.15)
            x0[5] = rng.uniform(0.0, 2.0 * np.pi)
            try:
                res = minimize(
                    fourier_obj,
                    x0,
                    method="Nelder-Mead",
                    options={"maxiter": 130, "xatol": 1e-3, "fatol": 1e-6},
                )
                h = fourier_params_to_h(res.x, n)
                add_candidate(cand_list, h)
            except Exception:
                pass


def select_candidates(cand_list, parent_h, parent_c, max_sel=4):
    if not cand_list:
        return []
    cand_list.sort(key=itemgetter(0))
    selected = []
    best_for_n = {}
    if parent_h is not None:
        selected.append((parent_c, parent_h))
        best_for_n[len(parent_h)] = parent_c

    for c, h in cand_list:
        if len(selected) >= max_sel:
            break
        n = len(h)
        if n in best_for_n:
            if c >= best_for_n[n] - 1e-8:
                continue
            for i in range(len(selected)):
                sc, sh = selected[i]
                if len(sh) == n and c < sc - 1e-8:
                    selected[i] = (c, h)
                    best_for_n[n] = c
                    break
            continue
        selected.append((c, h))
        best_for_n[n] = c
    return selected


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _DEADLINE
    _START = time.time()
    _DEADLINE = _START + float(budget_s)
    rng = np.random.RandomState(seed)

    parent_h = None
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
    except Exception:
        parent_h = None

    best_h = None
    best_c = float("inf")
    parent_proj = None
    parent_c = None

    if parent_h is not None and parent_h.size > 0 and np.all(np.isfinite(parent_h)):
        parent_proj = project_feasible(parent_h.copy(), 0.5 * len(parent_h))
        parent_c = eval_c5(parent_proj)
        best_h = parent_proj.copy()
        best_c = parent_c

    if best_h is None:
        n0 = 96
        best_h = project_feasible(random_binary(n0, rng), 0.5 * n0)
        best_c = eval_c5(best_h)
        parent_proj = None

    cand_list = []
    if parent_proj is not None:
        cand_list.append((parent_c, parent_proj))

    local_reserve = max(200.0, 0.70 * float(budget_s))
    cand_stop = max(time.time() + 20.0, min(_DEADLINE - local_reserve, time.time() + 180.0))

    if parent_proj is not None and parent_proj.size > 0:
        pn = len(parent_proj)
        for n in (max(48, pn - 10), pn + 1 if (pn + 1) % 2 == 0 else pn + 2, 128):
            if 48 <= n <= 256 and n != pn and time.time() < cand_stop:
                add_candidate(cand_list, interpolate_to_n(parent_proj, int(n)))
        for scale in (0.01, 0.03, 0.08):
            if time.time() >= cand_stop:
                break
            noise = rng.normal(0.0, scale, pn)
            noise -= float(noise.mean())
            add_candidate(cand_list, parent_proj + noise)

    for n in (80, 95, 96, 112, 128):
        if time.time() >= cand_stop:
            break
        for per, amp in ((1, 0.5), (1, 0.35), (1, 0.65), (2, 0.5), (2, 0.35), (3, 0.40), (4, 0.30)):
            if time.time() >= cand_stop:
                break
            for ph in (0.0, np.pi / 4.0, np.pi / 2.0, 3.0 * np.pi / 4.0, np.pi):
                if time.time() >= cand_stop:
                    break
                add_candidate(cand_list, make_sinusoid(n, per, amp, ph))

    for n in (96, 128):
        if time.time() >= cand_stop:
            break
        for _ in range(4):
            if time.time() >= cand_stop:
                break
            add_candidate(cand_list, random_binary(n, rng))

    if float(budget_s) > 600 and time.time() < cand_stop:
        add_fourier_candidates(cand_list, rng, cand_stop)

    selected = select_candidates(cand_list, parent_proj, parent_c, max_sel=4)
    if not selected:
        selected = [(best_c, best_h)]

    local_deadline = max(time.time() + 1.0, _DEADLINE - 30.0)
    total_local = max(0.0, local_deadline - time.time())
    weights = (0.55, 0.25, 0.15, 0.05)

    for i, (_, hh) in enumerate(selected):
        if time.time() + 40.0 >= local_deadline:
            break
        if i >= len(weights):
            continue
        alloc = total_local * float(weights[i])
        alloc = min(alloc, local_deadline - time.time())
        if alloc < 40.0:
            break
        stop_time = min(time.time() + alloc, local_deadline)
        hh2, cc2 = polish_candidate(hh, stop_time, rng)
        if cc2 < best_c:
            best_h = hh2
            best_c = cc2

    if best_h is not None and time.time() + 120.0 < local_deadline:
        hh2, cc2 = polish_candidate(best_h, local_deadline, rng)
        if cc2 < best_c:
            best_h = hh2
            best_c = cc2

    if best_h is None:
        best_h = project_feasible(random_binary(96, rng), 48.0)
        best_c = eval_c5(best_h)

    best_h = project_feasible(best_h, 0.5 * len(best_h))
    best_c = eval_c5(best_h)

    if time.time() + 15.0 < _DEADLINE:
        rb = round_to_binary(best_h, 0.5 * len(best_h))
        c_rb = eval_c5(rb)
        if c_rb < best_c:
            best_h = rb
            best_c = c_rb

    if time.time() + 15.0 < _DEADLINE:
        hh2, cc2 = active_term_moves(best_h, min(_DEADLINE - 5.0, time.time() + 10.0), rng)
        if cc2 < best_c:
            best_h = hh2
            best_c = cc2

    best_h = project_feasible(best_h, 0.5 * len(best_h))
    final_c = eval_c5(best_h)

    try:
        ec = float(evaluate_erdos_solution(best_h))
        if np.isfinite(ec) and ec < final_c:
            final_c = ec
    except Exception:
        pass

    return best_h, float(final_c), int(len(best_h))
