def project_box_sum(h, n):
    """Project h onto {h in [0,1]^n : sum(h) = n/2}"""
    h = h.copy()
    target = n / 2.0
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Exact fix: distribute remaining
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-15:
        # Find coordinates not at bounds
        free = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
        if np.sum(free) > 0:
            h_proj[free] += diff / np.sum(free)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def subgradient_step(h, n, alpha, top_k=3):
    """One subgradient step with projection"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    # Find top k maximizing shifts
    top_indices = np.argsort(corr)[-top_k:]
    # Average gradient over top shifts
    grad = np.zeros(n)
    for k_idx in top_indices:

            grad[j] += g

    h_new = h - alpha * grad

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def periodic_obj(t):
    n_rep = _g_n // _g_period
    h = np.tile(np.array(t, dtype=float), n_rep)
    return compute_c5(h)

def periodic_con(t):
    return float(np.sum(t) - _g_period / 2.0)

def periodic_con_neg(t):
    # For inequality constraints: t[i] >= 0 (already handled by bounds)
    return 0.0

def subgrad_with_backtrack(h, n, max_iter, alpha_init=0.1, top_k=5):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha = alpha_init
    stall_count = 0
    
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = float(np.max(corr) * (2.0 / n))
        if cmax < best_c5 - 1e-15:
            best_c5 = cmax
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 200:
                alpha *= 0.5
                stall_count = 0
                if alpha < 1e-8:
                    break
        
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for ki in top_idx:
            k = int(ki) - (n - 1)
            if 0 <= k < n:
                if n - k > 0:
                    grad[k:] += (1.0 - h[:n-k])
                    grad[:n-k] += (-h[k:])
            elif -n < k < 0:
                ka = -k
                if n - ka > 0:
                    grad[:n-ka] += (1.0 - h[ka:])
                    grad[ka:] += (-h[:n-ka])
        grad /= float(len(top_idx))
        grad -= np.mean(grad)  # Project onto sum=0 subspace
        
        # Backtracking
        step = alpha
        improved = False
        for _ in range(15):
            h_try = h - step * grad
            h_try = project_box_sum(h_try, n)
            if np.min(h_try) >= -1e-10 and np.max(h_try) <= 1.0 + 1e-10:
                f_try = compute_c5(h_try)
                if f_try < cmax - 1e-15:
                    h = h_try
                    improved = True
                    break
            step *= 0.5
        
        if not improved and it % 100 == 0:
            alpha *= 0.7
    
    return best_h, best_c5

def pair_coord_descent(h, n, n_sweeps, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for j in order:
            k = (j + n // 2) % n
            if k == j:
                k = (j + 1) % n
            
            t_lo = max(-h[j], h[k] - 1.0)
            t_hi = min(1.0 - h[j], h[k])
            
            if t_hi - t_lo < 1e-12:
                continue
            
            best_t = 0.0
            best_f = best_c5
            cur_f = best_c5
            
            # Try several values
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9, -0.1, -0.3, -0.5, -0.7, -0.9]:
                t = frac * (t_hi - t_lo) * 0.5
                if t <= t_lo + 1e-12 or t >= t_hi - 1e-12:
                    continue
                h[j] += t
                h[k] -= t
                f = compute_c5(h)
                if f < best_f:
                    best_f = f
                    best_t = t
                else:
                    h[j] -= t
                    h[k] += t
            
            if best_t != 0.0:
                h[j] += best_t
                h[k] -= best_t
                best_c5 = best_f
    
    return h, best_c5

def interp_to(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    h_new = np.interp(x_new, x_old, h_src)
    return h_new

            best_h = h_cd

            if c5_sg2 < best_c5:
                best_c5 = c5_sg2
                best_h = h_sg2
                best_n = n_try

    if time_left() > 50:
        n = best_n
        n_sweeps = int((time_left() - 20) / 0.3)

        h_cd2, c5_cd2 = pair_coord_descent(best_h, n, n_sweeps, rng)

            best_c5 = c5_cd2
            best_h = h_cd2

            pass
    
    # Final check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
