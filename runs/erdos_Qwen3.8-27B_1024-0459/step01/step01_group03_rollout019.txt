        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.clip(np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        p = _proj_density(parent, 0.5 * float(parent.size))
        c = _eval(p)
        best_h, best_c = p.copy(), c

    if best_h is None:
        h0 = _make_random_start(90, rng)
        best_h, best_c = h0, _eval(h0)

    plan = []
    if parent is not None:
        pn = int(parent.size)
        plan.append(pn)
        plan.append(int(pn * 1.5))
        plan.append(int(pn * 2))
        plan.append(int(pn * 3))
    plan.extend([90, 120, 150, 180, 240])
    plan = sorted(set(p for p in plan if 4 <= p <= 300))

    for n_res in plan:
        if deadline - time.perf_counter() < 40.0:
            break
        if n_res > 150 and deadline - time.perf_counter() < 100.0:
            break

        starts = []
        if int(best_h.size) == n_res:
            starts.append(best_h.copy())
        else:
            starts.append(_upsample(best_h, n_res))
        if parent is not None and int(parent.size) != n_res:
            starts.append(_upsample(parent, n_res))
        starts.append(_make_random_start(n_res, rng))
        if rng.random() < 0.7:
            starts.append(_make_random_start(n_res, rng))

        uniq = []
        for s in starts:
            s = _proj_density(s, 0.5 * float(n_res))
            dup = any(np.allclose(s, u, atol=5e-5) for u in uniq)
            if not dup:
                uniq.append(s)

        for s in uniq:
            rem = deadline - time.perf_counter()
            if rem < 25.0:
                break
            left = len(uniq)
            if n_res <= 100:
                cap = 80.0
            elif n_res <= 180:
                cap = 70.0
            else:
                cap = 55.0
            per = min(rem / float(left), cap)
            per = max(20.0, per)
            sub_dl = time.perf_counter() + min(per, rem)

            h_out, c_out = _optimize_resolution(s, n_res, sub_dl, rng)
            if h_out is not None:
                h_out = _proj_density(h_out, 0.5 * float(n_res))
                c_out = _eval(h_out)
                if c_out < best_c:
                    best_h = h_out.copy()
                    best_c = c_out

    if best_h is not None and deadline - time.perf_counter() > 20.0:
        n_b = int(best_h.size)
        h2, c2 = _subgradient_polish(best_h, deadline, max_iter=800, top_k=4)
        if c2 < best_c:
            best_h, best_c = h2, c2
            n_b = int(best_h.size)

        if n_b <= 200 and deadline - time.perf_counter() > 15.0:
            h3, c3 = _pair_search(best_h, deadline, max_iter=500, top_k=5)
            if c3 < best_c:
                best_h, best_c = h3, c3
                n_b = int(best_h.size)

        if deadline - time.perf_counter() > 10.0:
            h4, c4 = _subgradient_polish(best_h, deadline, max_iter=500, top_k=3)
            if c4 < best_c:
                best_h, best_c = h4, c4

    if best_h is None:
        fallback = _proj_density(np.full(90, 0.5), 45.0)
        return fallback, _eval(fallback), 90

    final_h = _proj_density(best_h, 0.5 * float(best_h.size))
    final_c = _eval(final_h)
    return final_h, float(final_c), int(final_h.size)
