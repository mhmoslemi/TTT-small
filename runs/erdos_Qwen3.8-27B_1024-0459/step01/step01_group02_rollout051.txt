import numpy as np
from scipy.optimize import minimize
import time

_g_start = 0.0
_g_budget = 1300.0
_g_obj_n = 0
_g_obj_beta = 50.0
_g_con_n = 0
_g_active_shifts = None
_g_rng = None


def _remaining():
    return max(0.0, _g_budget - (time.time() - _g_start) - 5.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_c5_detail(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    return h


def resample(h, old_n, new_n):
    if old_n == new_n:
        return h.copy()
    xp = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    interp = np.interp(x_new, xp, h)
    return project_feasible(interp, new_n)


def _con_eq(x):
    return np.sum(x) - _g_con_n / 2.0


def _obj_smooth_global(x):
    n = _g_obj_n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _g_obj_beta
    exps = np.exp(beta * (corr - m))
    s = np.sum(exps)
    if s < 1e-300:
        return m * dx
    return (m + np.log(s) / beta) * dx


def _obj_focused_global(x):
    n = _g_obj_n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    active = _g_active_shifts
    a = corr[active]
    ma = np.max(a)
    beta = _g_obj_beta
    exps = np.exp(beta * (a - ma))
    s = np.sum(exps)
    if s < 1e-300:
        return ma * dx
    return (ma + np.log(s) / beta) * dx


def _obj_raw_global(x):
    n = _g_obj_n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return np.max(corr) * dx


def run_de(h_init, n, pop_size, n_gen, F, Cr, seed):
    rng = np.random.RandomState(seed)
    pop = np.zeros((pop_size, n))
    pop[0] = h_init.copy()
    for i in range(1, pop_size):
        r = rng.rand()
        if r < 0.35:
            h_bin = np.zeros(n)
            idx = rng.choice(n, size=n // 2, replace=False)
            h_bin[idx] = 1.0
            pop[i] = h_bin
        elif r < 0.65:
            pop[i] = h_init + rng.randn(n) * (0.08 + 0.08 * rng.rand())
        elif r < 0.85:
            t = np.arange(n) / n
            freq = rng.randint(1, 5)
            phase = rng.rand() * 2.0 * np.pi
            amp = 0.3 + 0.2 * rng.rand()
            pop[i] = 0.5 + amp * np.sin(2.0 * np.pi * freq * t + phase)
        else:
            pop[i] = rng.rand(n)
        pop[i] = project_feasible(pop[i], n)

    fitness = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_val = float(np.min(fitness))
    best_sol = pop[np.argmin(fitness)].copy()

    for gen in range(n_gen):
        if _remaining() < 25:
            break
        for i in range(pop_size):
            r1, r2, r3 = rng.choice(pop_size, size=3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            trial = pop[i].copy()
            j0 = rng.randint(n)
            cmask = rng.rand(n) < Cr
            cmask[j0] = True
            trial[cmask] = mutant[cmask]
            trial = project_feasible(trial, n)
            fit = compute_c5(trial)
            if fit <= fitness[i]:
                pop[i] = trial
                fitness[i] = fit
                if fit < best_val:
                    best_val = fit
                    best_sol = trial.copy()

    return best_sol, best_val


def run_sa(h_init, n, n_steps, T_init, T_final, seed):
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    c5_cur = best_c5
    for step in range(n_steps):
        if _remaining() < 10:
            break
        frac = step / max(1, n_steps - 1)
        T = T_init * (T_final / T_init) ** frac
        n_pert = max(2, int(n * 0.04))
        idx = rng.choice(n, size=n_pert, replace=False)
        h_new = h.copy()
        h_new[idx] += rng.randn(n_pert) * T * 3.0
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        accept = False
        if c5_new < c5_cur:
            accept = True
        else:
            delta = (c5_new - c5_cur) / max(T * 0.005, 1e-12)
            if delta < 50.0 and rng.rand() < np.exp(-delta):
                accept = True
        if accept:
            h = h_new
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
    return best_h, best_c5


def run_focused_slsqp(h_start, n, K, max_rounds, beta):
    global _g_obj_n, _g_obj_beta, _g_active_shifts, _g_con_n
    _g_obj_n = n
    _g_obj_beta = beta
    _g_con_n = n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq}]

    for rnd in range(max_rounds):
        if _remaining() < 18:
            break
        _, corr = compute_c5_detail(h)
        m = np.max(corr)
        if m < 1e-15:
            break
        thresh = m * 0.93
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr))])
        if len(active) > K:
            active = np.argsort(corr)[::-1][:K]
        _g_active_shifts = active

        try:
            r = minimize(_obj_focused_global, h, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 250, 'ftol': 1e-13, 'iprint': 0})
            h_new = project_feasible(r.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-10:
                best_c5 = c5_new
                h = h_new.copy()
            else:
                K = min(int(K * 1.8), 2 * n - 1)
        except Exception:
            break

    return h, best_c5


def run_smooth_slsqp(h_start, n, beta, maxiter):
    global _g_obj_n, _g_obj_beta, _g_con_n
    _g_obj_n = n
    _g_obj_beta = beta
    _g_con_n = n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq}]
    try:
        r = minimize(_obj_smooth_global, h, method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-13, 'iprint': 0})
        h_new = project_feasible(r.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            h = h_new.copy()
    except Exception:
        pass
    return h, best_c5


def run_coordinate_descent(h_start, n, max_rounds):
    h = h_start.copy()
    best_c5 = compute_c5(h)
    deltas = [-0.08, -0.04, -0.02, -0.01, 0.01, 0.02, 0.04, 0.08]
    for rnd in range(max_rounds):
        if _remaining() < 10:
            break
        improved = False
        order = np.random.RandomState(rnd * 31 + 7).permutation(n)
        for i in order:
            if _remaining() < 10:
                break
            for d in deltas:
                h_try = h.copy()
                h_try[i] += d
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def try_binary_round(h, n):
    if n % 2 != 0:
        return h.copy(), compute_c5(h)
    h_round = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_round))
    n_want = n // 2
    if n_ones > n_want:
        ones = np.where(h_round == 1.0)[0]
        vals = h[ones]
        order = np.argsort(vals)[:n_ones - n_want]
        h_round[ones[order]] = 0.0
    elif n_ones < n_want:
        zeros = np.where(h_round == 0.0)[0]
        vals = h[zeros]
        order = np.argsort(-vals)[:n_want - n_ones]
        h_round[zeros[order]] = 1.0
    c5 = compute_c5(h_round)
    return h_round, c5
