def eval_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    c = np.correlate(h, 1.0 - h, mode="full")
    if c.size == 0:
        return float("inf")
    return float(np.max(c) * (2.0 / float(n)))


def corr_of(h):
    return np.correlate(h, 1.0 - h, mode="full")


def fix_sum_small(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    res = float(target - np.sum(h))
    if not np.isfinite(res) or abs(res) <= 1e-13:
        return h

    if res > 0.0:
        free = np.where(h < 1.0 - 1e-15)[0]
        if free.size:
            w = 1.0 - h[free]
            ws = float(np.sum(w))
            if ws > 0.0:
                h[free] += res * (w / ws)
    else:
        free = np.where(h > 1e-15)[0]
        if free.size:
            w = h[free]
            ws = float(np.sum(w))
            if ws > 0.0:
                h[free] += res * (w / ws)

    h = np.clip(h, 0.0, 1.0)
    res = float(target - np.sum(h))
    if abs(res) > 1e-12:
        free = np.where((h > 1e-15) & (h < 1.0 - 1e-15))[0]
        if free.size:
            h[free[0]] += res
            h = np.clip(h, 0.0, 1.0)
    return h


def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    if n == 0:
        return x

    res = float(target - np.sum(x))
    if abs(res) <= 1e-12 * max(1, n):
        return x

    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < float(target):
            lo = mid
        else:
            hi = mid

    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    return fix_sum_small(h, target)


def is_feasible(h, target, tol=1e-6):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    if abs(float(np.sum(h)) - float(target)) > max(1e-8, tol * max(1, h.size)):
        return False
    return True


def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    n = int(n)
    if n0 == n:
        return project_density(p, 0.5 * n)
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * max(1, n))

    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)


def downsample_h(h, new_n):
    h = np.asarray(h, dtype=float)
    n = h.size
    new_n = int(new_n)
    if new_n <= 0:
        new_n = 1
    if n == new_n:
        return project_density(h, 0.5 * new_n)
    if n == 0:
        return project_density(np.full(new_n, 0.5), 0.5 * new_n)

    out = np.empty(new_n, dtype=float)
    for i in range(new_n):
        a = (i * n) // new_n
        b = ((i + 1) * n) // new_n
        if b > a:
            out[i] = float(np.mean(h[a:b]))
        else:
            out[i] = float(h[a])
    return project_density(out, 0.5 * new_n)


def transform_h(h, nn):
    h = np.asarray(h, dtype=float)
    nn = int(nn)
    if h.size == nn:
        return project_density(h, 0.5 * nn)
    if nn > h.size:
        return upsample_h(h, nn)
    return downsample_h(h, nn)


def make_fourier_candidate(n, rng, mode=0):
    n = int(n)
    if n <= 1:
        return project_density(np.array([0.5], dtype=float), 0.5 * max(1, n))

    if mode == 1:
        h = np.empty(n, dtype=float)
        pos = 0
        levels = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
        val = 0.5
        while pos < n:
            rem = n - pos
            length = int(rng.integers(1, max(1, rem)))
            val = float(rng.choice(levels))
            h[pos:pos + length] = val
            pos += length
        return project_density(h, 0.5 * n)

    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.full(n, 0.5, dtype=float)
    maxm = 8 if mode == 0 else 5
    for m in range(1, maxm + 1):
        amp = float(rng.uniform(0.04, 0.30)) * (0.65 ** (m - 1))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        if rng.random() < 0.5:
            h += amp * np.sin(0.5 * np.pi * m * x + phase)
        else:
            h += amp * np.cos(0.5 * np.pi * m * x + phase)
    return project_density(h, 0.5 * n)


def make_shape_candidates(n):
    n = int(n)
    out = []
    if n <= 1:

    h = 0.5 + 0.20 * np.sin(np.pi * x + 0.7)

    out.append(project_density(h, 0.5 * n))
    return out


def build_start_list(n, parent, parent_best, rng):
    starts = []
    if parent is not None:
        starts.append(upsample_h(parent, n))

def vector_close(a, b):
    if a.size != b.size:
        return False
    return bool(np.allclose(a, b, rtol=0.0, atol=1e-7))


def select_best_start(starts, current_best=None, current_c=None):

    best_c = float("inf")
    seen = []

def add_lag_gradient(g, h, t, weight):
    n = h.size
    if weight == 0.0:
        return
    k = t - (n - 1)
    if k == 0:
        g += weight * (1.0 - 2.0 * h)
    elif k > 0:
        lag = k
        g[lag:] += weight * (1.0 - h[:n - lag])
        g[:n - lag] -= weight * h[lag:]
    else:
        lag = -k
        g[:n - lag] += weight * (1.0 - h[lag:])
        g[lag:] -= weight * h[:n - lag]


def active_average_gradient(h, c, max_count=16):
    n = h.size
    c = np.asarray(c, dtype=float)
    if c.size == 0:
        return np.zeros(n, dtype=float)
    m = float(np.max(c))
    tol = max(1e-12, 1e-8 * max(1.0, m))
    idx = np.where(c >= m - tol)[0]
    if idx.size > max_count:
        idx = idx[np.argsort(c[idx])][-max_count:]
    if idx.size == 0:
        idx = np.array([int(np.argmax(c))], dtype=int)

    g = np.zeros(n, dtype=float)
    w = 1.0 / float(idx.size)
    for t in idx:
        add_lag_gradient(g, h, int(t), w)
    return g


def smooth_jac_only(h, alpha):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return np.zeros(0, dtype=float)
    c = corr_of(h)
    m = float(np.max(c))
    y = alpha * (c - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return np.zeros(n, dtype=float)
    w = e / z
    g = np.zeros(n, dtype=float)
    for t in range(c.size):
        p = float(w[t])
        if p > 1e-11:
            add_lag_gradient(g, h, int(t), p)
    return (2.0 / float(n)) * g


def set_smooth_context(n, alpha, target):
    global _N, _ALPHA, _TARGET
    _N = int(n)
    _ALPHA = float(alpha)
    _TARGET = float(target)


def context_obj(x):
    x = np.asarray(x, dtype=float)
    n = _N
    alpha = _ALPHA
    if x.size != n or not np.all(np.isfinite(x)):
        return 1e100
    c = corr_of(x)
    m = float(np.max(c))
    y = alpha * (c - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return 1e100
    return float((2.0 / float(n)) * (m + np.log(z) / alpha))


def context_jac(x):
    return smooth_jac_only(x, _ALPHA)


def context_eq(x):
    return float(np.sum(np.asarray(x, dtype=float)) - _TARGET)


def context_eq_jac(x):
    return np.ones_like(np.asarray(x, dtype=float), dtype=float)


def slsqp_polish(h, deadline):
    h = project_density(h, 0.5 * h.size)
    n = h.size
    target = 0.5 * float(n)
    best_h = h.copy()
    best_c = eval_c5(best_h)
    if n > 200 or deadline - time.perf_counter() < 20.0:
        return best_h, best_c

    if n <= 100:
        alphas = (20.0, 60.0, 150.0, 300.0)
        maxiter = 90
        sl_time = 120.0
    elif n <= 180:
        alphas = (20.0, 60.0, 150.0)
        maxiter = 60
        sl_time = 55.0
    else:
        return best_h, best_c

    sub_deadline = min(deadline, time.perf_counter() + sl_time)
    bounds = [(0.0, 1.0)] * n
    constraints = ({"type": "eq", "fun": context_eq, "jac": context_eq_jac},)
    h_work = best_h.copy()

    for alpha in alphas:
        if time.perf_counter() >= sub_deadline:
            break
        set_smooth_context(n, alpha, target)
        try:
            res = minimize(
                context_obj,
                h_work,
                method="SLSQP",
                jac=context_jac,
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": maxiter, "ftol": 1e-10, "disp": False},
            )
            xg = getattr(res, "x", None)
            if xg is not None:
                xg = np.asarray(xg, dtype=float)
                if xg.size == n and np.all(np.isfinite(xg)):
                    cand = project_density(xg, target)
                    c = eval_c5(cand)
                    if c < best_c:
                        best_h = cand
                        best_c = c
                        h_work = cand
                    elif c <= best_c + 0.01:
                        h_work = cand
                    else:
                        h_work = best_h
        except Exception:
            h_work = best_h

    return best_h, best_c


def pair_LQ(h, v, i, j, ar):
    n = h.size
    m = 2 * n - 1
    center = n - 1
    L = np.zeros(m, dtype=float)

    L[center + i - ar] += v
    L[center + j - ar] -= v
    L[center - i + ar] -= h
    L[center - j + ar] += h

    Q = np.zeros(m, dtype=float)
    Q[center] = 2.0
    d = abs(i - j)
    if d > 0:
        Q[center + d] = -1.0
        Q[center - d] = -1.0
    return L, Q


def pair_line_search(h, c, i, j, ar, G=64, refine=True, current_C=None):
    n = h.size
    dx = 2.0 / float(n)
    target = 0.5 * float(n)
    v = 1.0 - h

    lo = max(-float(h[i]), float(h[j]) - 1.0)
    hi = min(1.0 - float(h[i]), float(h[j]))
    if hi - lo <= 1e-13:
        return None

    L, Q = pair_LQ(h, v, int(i), int(j), ar)
    if current_C is None:
        current_C = float(np.max(c) * dx)

    if G < 8:
        G = 8
    gs = np.linspace(lo, hi, int(G))
    base = c[None, :] + L[None, :] * gs[:, None] - Q[None, :] * gs[:, None] * gs[:, None]
    obj = np.max(base, axis=1)
    if not np.all(np.isfinite(obj)):
        return None
    idx = int(np.argmin(obj))

    if refine:
        for _ in range(5):
            if idx <= 0:
                lo2 = float(gs[0])
                hi2 = float(gs[1])
            elif idx >= gs.size - 1:
                lo2 = float(gs[-2])
                hi2 = float(gs[-1])
            else:
                lo2 = float(gs[idx - 1])
                hi2 = float(gs[idx + 1])
            if hi2 - lo2 <= 1e-12:
                break
            gs = np.linspace(lo2, hi2, 48)
            base = c[None, :] + L[None, :] * gs[:, None] - Q[None, :] * gs[:, None] * gs[:, None]
            obj = np.max(base, axis=1)
            if not np.all(np.isfinite(obj)):
                break
            idx = int(np.argmin(obj))

    delta = float(gs[idx])
    new_c = c + L * delta - Q * delta * delta
    C2 = float(np.max(new_c) * dx)
    if C2 >= current_C - 1e-12:
        return None

    h2 = h.copy()
    h2[i] += delta
    h2[j] -= delta
    h2 = np.clip(h2, 0.0, 1.0)
    h2 = fix_sum_small(h2, target)
    c2 = corr_of(h2)
    C2b = float(np.max(c2) * dx)
    if C2b < C2:
        C2 = C2b
    return (h2, c2, float(C2))


def find_best_pair_move(h, c, pairs, deadline, ar, coarse_grid=48, refine_count=4):
    n = h.size
    current_C = float(np.max(c) * (2.0 / float(n)))
    top = []
    if not pairs:
        return None

    for pair in pairs:
        if time.perf_counter() >= deadline:
            break
        ii, jj = int(pair[0]), int(pair[1])
        if ii == jj:
            continue
        mv = pair_line_search(h, c, ii, jj, ar, G=coarse_grid, refine=False, current_C=current_C)
        if mv is not None and mv[2] < current_C - 1e-12:
            top.append((mv[2], ii, jj))
            if len(top) > 24:
                top.sort()
                top = top[:12]

    if not top:
        return None

    top.sort()
    best = None
    for Cv, ii, jj in top[:max(1, int(refine_count))]:
        mv = pair_line_search(h, c, ii, jj, ar, G=128, refine=True, current_C=Cv)
        if mv is not None and (best is None or mv[2] < best[2]):
            best = mv
    return best


def top_pairs_from_gradient(g, K, n, rand_count=12):
    g = np.asarray(g, dtype=float).ravel()
    n = int(n)
    K = int(K)
    if n < 2 or K < 1:
        return []
    scores = np.abs(g[:, None] - g[None, :])
    iu, ju = np.triu_indices(n, 1)
    sc = scores[iu, ju]
    if sc.size == 0:
        return []

    k = min(K, sc.size)
    if k < sc.size:
        part = np.argpartition(sc, -k)[-k:]
    else:
        part = np.arange(sc.size)
    order = np.argsort(sc[part])[::-1]
    part = part[order]

    pairs = []
    seen = set()
    for idx in part:
        p = (int(iu[idx]), int(ju[idx]))
        if p not in seen:
            seen.add(p)
            pairs.append(p)

    rng = np.random.default_rng(0)
    # A fixed local rng is okay here; run-level rng is passed in by callers when needed.
    for _ in range(max(0, int(rand_count))):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        if i > j:
            i, j = j, i
        p = (i, j)
        if p not in seen:
            seen.add(p)
            pairs.append(p)
    return pairs


def random_pairs(rng, n, K, rand_count=12):
    out = []
    seen = set()
    K = int(K)
    if n < 2:
        return out
    # Use main rng for genuinely different pairs.
    for _ in range(max(K * 2, 20)):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        if i > j:
            i, j = j, i
        p = (i, j)
        if p not in seen:
            seen.add(p)
            out.append(p)
            if len(out) >= K:
                break
    for _ in range(max(0, int(rand_count))):
        if not out:
            continue
        out.append(tuple(int(x) for x in rng.choice(out, size=2, replace=False)))
        # ensure distinct
        if out[-1][0] == out[-1][1]:
            out.pop()
    return out


def combined_pairs_from_gradient(g, K, n, rng, rand_count=12):
    pairs = top_pairs_from_gradient(g, K, n, 0)
    extra = random_pairs(rng, n, min(16, max(2, K // 8)), rand_count)
    seen = set()
    uniq = []
    for p in pairs + extra:
        i, j = int(p[0]), int(p[1])
        if i == j:
            continue
        if i > j:
            i, j = j, i
        p = (i, j)
        if p not in seen:
            seen.add(p)
            uniq.append(p)
    return uniq


def active_line_search(h, c, max_count=16):
    n = h.size
    target = 0.5 * float(n)
    dx = 2.0 / float(n)
    current_C = float(np.max(c) * dx)
    g = active_average_gradient(h, c, max_count)
    d = -g
    d -= np.mean(d)
    norm = float(np.linalg.norm(d))
    if norm < 1e-14:
        return None
    d /= norm

    best = None
    scales = (0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0)
    for s in scales:
        h2 = project_density(h + float(s) * d, target)
        c2 = corr_of(h2)
        C2 = float(np.max(c2) * dx)
        if C2 < (best[2] if best is not None else current_C) - 1e-12:
            best = (h2, c2, C2)
    return best

def pair_only_polish(h, deadline, rng, pair_count=40, coarse_grid=24):
    n = h.size
    if n < 2:
        return h, eval_c5(h)
    target = 0.5 * float(n)
    best = project_density(h, target)
    best_c = eval_c5(best)
    ar = np.arange(n, dtype=int)
    stall = 0

    while time.perf_counter() < deadline - 1.0:
        c = corr_of(best)
        g = active_average_gradient(best, c, 8)
        pairs = combined_pairs_from_gradient(g, pair_count, n, rng, rand_count=6)
        mv = find_best_pair_move(best, c, pairs, deadline, ar, coarse_grid=coarse_grid, refine_count=2)
        if mv is not None and mv[2] < best_c - 1e-12:
            best = mv[0]
            best_c = mv[2]
            stall = 0
        else:
            stall += 1
            if stall > 6:
                break
    best = project_density(best, target)
    best_c = eval_c5(best)
    return best, best_c


def polish_general(h, deadline, rng, do_slsqp=True):

    ar = np.arange(n, dtype=int)
    stall = 0
    pert = 0

    while time.perf_counter() < deadline - 1.0:
        c = corr_of(best)
        cur_C = float(np.max(c) * (2.0 / float(n)))
        if cur_C < best_c:
            best_c = cur_C

        g = active_average_gradient(best, c, 12)
        K = 110 if n <= 100 else 70
        pairs = combined_pairs_from_gradient(g, K, n, rng, rand_count=14)

        mv = find_best_pair_move(best, c, pairs, deadline, ar, coarse_grid=48, refine_count=4)
        improved = False
        if mv is not None and mv[2] < best_c - 1e-12:
            best = mv[0]
            best_c = mv[2]
            improved = True
            stall = 0
            pert = 0

        if not improved:
            mv = active_line_search(best, c, 16)
            if mv is not None and mv[2] < best_c - 1e-12:
                best = mv[0]
                best_c = mv[2]
                improved = True
                stall = 0
                pert = 0

        if not improved:
            mv = smooth_line_search(best, c, 80 if n <= 120 else 50)
            if mv is not None and mv[2] < best_c - 1e-12:
                best = mv[0]
                best_c = mv[2]
                improved = True
                stall = 0
                pert = 0

        if not improved:
            stall += 1
            if stall >= 10:
                remaining = max(0.0, deadline - time.perf_counter())
                if remaining > 80.0 and pert < 5:
                    scale = 0.02 + 0.015 * float(pert)
                    hp = project_density(best + scale * rng.normal(size=n), target)
                    cp = eval_c5(hp)
                    if cp < best_c + 0.002:
                        sub_dead = min(deadline, time.perf_counter() + 45.0)
                        hp2, cp2 = pair_only_polish(hp, sub_dead, rng, pair_count=34, coarse_grid=24)
                        if cp2 < best_c:
                            best = hp2
                            best_c = cp2
                            stall = 0
                            pert = 0
                            continue
                    pert += 1
                    stall = 0

            if stall > 35:
                break

    best = project_density(best, target)
    best_c = eval_c5(best)
    return best, best_c


def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    budget = max(5.0, budget)
    deadline = start + max(1.0, budget - 12.0)

    rng = np.random.default_rng(seed)

    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    best_h = None
    best_c = float("inf")

    if parent is not None:
        nparent = int(parent.size)
        pproj = project_density(parent, 0.5 * nparent)
        cp = eval_c5(pproj)
        if cp < best_c:
            best_h = pproj
            best_c = cp

    if best_h is None:
        base_n0 = 90
        starts = build_start_list(base_n0, parent, None, rng)
        best_h, best_c = select_best_start(starts)
        if best_h is None:
            best_h = project_density(np.full(90, 0.5), 45.0)
            best_c = eval_c5(best_h)

    base_n = int(best_h.size) if best_h.size > 1 else (nparent if parent is not None else 90)
    if base_n < 20:
        base_n = 90
        best_h = project_density(best_h, 0.5 * best_h.size)
        best_c = eval_c5(best_h)

    # cheap improvement from simple starts at the primary resolution
    starts = build_start_list(base_n, parent, best_h, rng)
    primary_h, primary_c = select_best_start(starts, best_h, best_c)
    if primary_h is not None and primary_c < best_c:
        best_h = primary_h
        best_c = primary_c

    span = max(1.0, deadline - start)

    # Primary high-resolution/parent polish.
    primary_deadline = start + min(0.58 * span, 900.0)
    primary_deadline = max(primary_deadline, start + 20.0)
    h1, c1 = polish_general(best_h, primary_deadline, rng, do_slsqp=True)
    if h1 is not None and c1 < best_c:
        best_h = h1
        best_c = c1

    # Extra resolutions.  Coarser resolutions are tested because the discrete
    # objective is exact for the returned resolution.
    extra_ns = []
    if base_n >= 60:
        extra_ns.append(max(30, base_n // 2))
    if base_n >= 40:
        extra_ns.append(max(24, base_n // 3))
    if base_n < 180:
        extra_ns.append(min(180, max(180, 2 * base_n)))
    if base_n < 240:
        extra_ns.append(min(240, 3 * base_n))

    # de-duplicate and sort by closeness to current size
    uniq_ns = []
    for nn in extra_ns:
        if nn < 20:
            continue
        if nn not in uniq_ns:
            uniq_ns.append(nn)
    uniq_ns.sort(key=lambda qq: abs(qq - int(best_h.size)))

    # Budget split: ~22% to first extra, ~12% to second, ~10% to third, final rest.
    allocs = [0.22, 0.12, 0.10, 0.06]
    for idx, nn in enumerate(uniq_ns[:4]):
        now = time.perf_counter()
        if now >= deadline - 30.0:
            break
        frac = allocs[idx] if idx < len(allocs) else 0.05
        sub_dead = min(deadline, now + max(20.0, frac * span))

        h_try = transform_h(best_h, nn)
        # add a few local starts at this resolution
        starts_y = build_start_list(nn, parent, h_try, rng)
        h_try, c_try = select_best_start(starts_y, h_try, eval_c5(h_try))
        if c_try >= best_c + 0.02:
            # Do not spend time polishing a clearly worse resolution.
            continue
        h2, c2 = polish_general(h_try, sub_dead, rng, do_slsqp=(nn <= 180))
        if h2 is not None and c2 < best_c:
            best_h = h2
            best_c = c2

    # Final long polish on the best found resolution.
    now = time.perf_counter()
    if now < deadline - 10.0:
        final_deadline = deadline
        hf, cf = polish_general(best_h, final_deadline, rng, do_slsqp=(int(best_h.size) <= 120))
        if hf is not None and cf < best_c:
            best_h = hf
            best_c = cf

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = eval_c5(final_h)
    if final_c == float("inf"):
        final_h = project_density(np.full(90, 0.5), 45.0)
        final_c = eval_c5(final_h)

    return final_h, float(final_c), int(final_h.size)
