def soft_max_obj(h, n_points, temp):

def project_feasible(h, n_points):

def adam_optimize(h0, n, max_iter, temp_start, temp_end, lr, time_limit):
    """Adam optimizer with soft-max objective and cooling."""
    n_pts = n
    h = h0.copy()
    m_v = np.zeros(n_pts)
    v_v = np.zeros(n_pts)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    best_h = h.copy()
    best_val = compute_c5(h, n_pts)
    
    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        # Cool temperature
        frac = it / max(max_iter - 1, 1)
        temp = temp_start * (temp_end / temp_start) ** frac
        
        # Compute objective and gradient
        if temp > 0.05:
            f = soft_max_obj(h, n_pts, temp)
            g = compute_grad_soft(h, n_pts, temp)
        else:
            f = compute_c5(h, n_pts)
            g = compute_grad_max(h, n_pts)
        
        t += 1
        m_v = beta1 * m_v + (1 - beta1) * g
        v_v = beta2 * v_v + (1 - beta2) * g**2
        m_hat = m_v / (1 - beta1**t)
        v_hat = v_v / (1 - beta2**t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n_pts)
        
        val = compute_c5(h_new, n_pts)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_val

def coordinate_descent(h0, n, n_sweeps=5, time_limit=60):
    """Simple coordinate descent: try flipping small amounts at each coordinate."""
    h = h0.copy()
    best_val = compute_c5(h, n)
    t0 = time.time()
    
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_limit:
            break
        for i in range(n):
            if time.time() - t0 > time_limit:
                break
            # Try moving h[i] by small amounts
            original = h[i]
            for delta in [0.02, -0.02, 0.05, -0.05, 0.1, -0.1]:
                new_val = original + delta
                if new_val < 0 or new_val > 1:
                    continue
                h[i] = new_val
                # Adjust another coordinate to maintain sum constraint
                # Find the coordinate with most "room" in the opposite direction
                needed = -delta  # need to compensate
                if needed > 0:
                    # Need to decrease another coordinate
                    candidates = np.where((h < 1.0) & (np.arange(n) != i))[0]
                    if len(candidates) == 0:
                        continue
                    # Pick the one with highest value
                    j = candidates[np.argmax(h[candidates])]
                    room = h[j]  # max we can decrease j
                    actual_remove = min(needed, room)
                    h[j] -= actual_remove
                else:
                    # Need to increase another coordinate
                    candidates = np.where((h > 0.0) & (np.arange(n) != i))[0]
                    if len(candidates) == 0:
                        continue
                    j = candidates[np.argmin(h[candidates])]
                    room = 1.0 - h[j]
                    actual_add = min(-needed, room)
                    h[j] += actual_add
                
                # Check if sum is correct
                s = np.sum(h)
                target = n / 2.0
                if abs(s - target) > 0.01:
                    h[i] = original
                    continue
                
                val = compute_c5(h, n)
                if val < best_val:
                    best_val = val
                    # Keep this change
                else:
                    # Revert
                    h[i] = original
                    if needed > 0:
                        h[j] += actual_remove
                    else:
                        h[j] -= actual_add
    
    return h, best_val

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)


def soft_max_obj(h, n_points, temp):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(conv)
    if temp < 1e-10:
        return m
    shifted = (conv - m) / temp
    max_shift = np.max(shifted)
    return temp * (np.log(np.sum(np.exp(shifted - max_shift)) + max_shift) + m / temp)


def project_feasible(h, n_points):

def compute_grad_max(h, n_points):

    dx = 2.0 / n

    k_max = int(np.argmax(conv))

    grad = np.zeros(n)
    if m_lag >= 0:
        grad[:n - m_lag] += (1.0 - h[m_lag:n]) * dx
        if m_lag > 0:
            grad[m_lag:] -= h[:n - m_lag] * dx
    else:
        p = -m_lag
        grad[:n - p] -= h[p:n] * dx
        grad[p:] += (1.0 - h[:n - p]) * dx
    return grad


def compute_grad_soft(h, n_points, temp):

    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(conv)

    max_s = np.max(shifted)

    t0 = time.time()
    deltas = [0.05, -0.05, 0.02, -0.02, 0.1, -0.1]

    for sweep in range(n_sweeps):
        if time.time() - t0 > time_limit:
            break
        for i in range(n):
            if time.time() - t0 > time_limit:
                break
            orig = h[i]
            for d in deltas:
                nv = orig + d
                if nv < 0.0 or nv > 1.0:
                    continue
                h[i] = nv
                needed = -d
                if needed > 0:
                    cands = np.where((h <= 1.0) & (np.arange(n) != i))[0]
                    if len(cands) == 0:
                        h[i] = orig
                        break
                    j = cands[np.argmax(h[cands])]
                    rm = min(needed, h[j])
                    h[j] -= rm
                else:
                    cands = np.where((h >= 0.0) & (np.arange(n) != i))[0]
                    if len(cands) == 0:
                        h[i] = orig
                        break
                    j = cands[np.argmin(h[cands])]
                    ad = min(-needed, 1.0 - h[j])
                    h[j] += ad

                s = np.sum(h)
                if abs(s - n / 2.0) > 0.05:
                    h[i] = orig
                    continue

                val = compute_c5(h, n)
                if val < best_val:
                    best_val = val
                else:
                    h[i] = orig
                    if needed > 0:
                        h[j] += rm
                    else:
                        h[j] -= ad
                    break
    return h, best_val


def slsqp_polish(h_init, n_points, maxiter):
    n = n_points
    target = n / 2.0

    def obj_wrapper(x):
        return compute_c5(x, n)

    def jac_wrapper(x):
        return compute_grad_max(x, n)

    def con_fun(x):
        return np.sum(x) - target

    def con_jac(x):
        return np.ones(n)

    res = minimize(obj_wrapper, h_init.copy(), jac=jac_wrapper,
                   method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x, res.fun


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    # Get initial solution from parent
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if h0 is None or len(h0) == 0:
        n0 = 100
        h0 = np.full(n0, 0.5)
    else:
        n0 = len(h0)

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_val = compute_c5(best_h, n0)
    best_n = n0

    # Phase 1: Adam soft-max at parent resolution
    if time_left() > 80:
        alloc = min(60, time_left() - 20)
        h_try, v_try = adam_loop(h0, n0, max_iter=3000,
                                  temp_start=0.15, temp_end=0.005,
                                  lr=0.02, time_limit=alloc)
        h_try = project_feasible(h_try, n0)
        v_try = compute_c5(h_try, n0)
        if v_try < best_val:
            best_h, best_val = h_try, v_try

    # Phase 2: Try multiple resolutions via interpolation
    for n_fine in [140, 200, 280, 350]:
        if time_left() < 200:
            break
        x_coarse = np.linspace(0, 2, best_n, endpoint=False)
        x_fine = np.linspace(0, 2, n_fine, endpoint=False)
        h_fine = np.interp(x_fine, x_coarse, best_h)
        h_fine = project_feasible(h_fine, n_fine)

        alloc = min(70, time_left() - 60)
        if alloc < 20:
            break
        h_try, v_try = adam_loop(h_fine, n_fine, max_iter=2000,
                                  temp_start=0.1, temp_end=0.005,
                                  lr=0.01, time_limit=alloc)
        h_try = project_feasible(h_try, n_fine)
        v_try = compute_c5(h_try, n_fine)
        if v_try < best_val:
            best_h, best_val, best_n = h_try, v_try, n_fine

    # Phase 3: SLSQP polish on best
    if time_left() > 80:
        try:
            h_try, _ = slsqp_polish(best_h, best_n, maxiter=800)
            h_try = project_feasible(h_try, best_n)
            v_try = compute_c5(h_try, best_n)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
        except Exception:
            pass

    # Phase 4: Multi-start with Adam at best_n and nearby resolutions
    attempt = 0
    while time_left() > 120 and attempt < 40:
        strat = attempt % 5
        n_try = best_n
        if strat == 0:
            h_rand = best_h + rng.normal(0, 0.15, best_n)
        elif strat == 1:
            h_rand = best_h.copy()
            n_pert = max(3, best_n // 5)
            idx = rng.choice(best_n, size=n_pert, replace=False)
            h_rand[idx] = np.clip(h_rand[idx] + rng.normal(0, 0.25, n_pert), 0, 1)
        elif strat == 2:
            h_rand = rng.uniform(0, 1, best_n)
        elif strat == 3:
            # Sine-like structure
            x = np.linspace(0, 2, best_n, endpoint=False)
            h_rand = 0.5 + 0.4 * np.sin(2 * np.pi * rng.randint(1, 5) * x)
            h_rand = np.clip(h_rand + rng.normal(0, 0.1, best_n), 0, 1)
        else:
            # Binary random then smooth
            h_rand = (rng.random(best_n) < 0.5).astype(float)
            # Smooth slightly
            kernel = np.array([0.1, 0.3, 0.3, 0.1, 0.1])
            h_rand = np.convolve(h_rand, kernel, mode='same')
            h_rand = np.clip(h_rand, 0, 1)

        h_rand = project_feasible(h_rand, n_try)
        alloc = min(35, time_left() - 40)
        if alloc < 10:
            break
        h_try, v_try = adam_loop(h_rand, n_try, max_iter=1000,
                                  temp_start=0.06, temp_end=0.005,
                                  lr=0.008, time_limit=alloc)
        h_try = project_feasible(h_try, n_try)
        v_try = compute_c5(h_try, n_try)
        if v_try < best_val:
            best_h, best_val, best_n = h_try, v_try, n_try
        attempt += 1

    # Phase 5: Try lower resolution (coarser can sometimes be better)
    if time_left() > 150:
        for n_coarse in [70, 80, 90]:
            if time_left() < 100:
                break
            if n_coarse >= best_n:
                continue
            x_coarse = np.linspace(0, 2, best_n, endpoint=False)
            x_target = np.linspace(0, 2, n_coarse, endpoint=False)
            h_coarse = np.interp(x_target, x_coarse, best_h)
            h_coarse = project_feasible(h_coarse, n_coarse)
            alloc = min(40, time_left() - 50)
            if alloc < 15:
                break
            h_try, v_try = adam_loop(h_coarse, n_coarse, max_iter=1500,
                                      temp_start=0.08, temp_end=0.005,
                                      lr=0.01, time_limit=alloc)
            h_try = project_feasible(h_try, n_coarse)
            v_try = compute_c5(h_try, n_coarse)
            if v_try < best_val:
                best_h, best_val, best_n = h_try, v_try, n_coarse

    # Phase 6: Coordinate descent polish
    if time_left() > 40:
        alloc = min(30, time_left() - 15)
        best_h, best_val = coord_descent_polish(best_h, best_n, n_sweeps=3, time_limit=alloc)

    # Phase 7: Final SLSQP
    if time_left() > 30:
        try:
            h_try, _ = slsqp_polish(best_h, best_n, maxiter=1000)
            h_try = project_feasible(h_try, best_n)
            v_try = compute_c5(h_try, best_n)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
        except Exception:
            pass

    # Phase 8: One more coordinate descent
    if time_left() > 15:
        best_h, best_val = coord_descent_polish(best_h, best_n, n_sweeps=2, time_limit=10)

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_val = compute_c5(best_h, best_n)

    # Ensure exact constraint
    target = best_n / 2.0
    s = np.sum(best_h)
    if abs(s - target) > 1e-12:
        best_h = project_feasible(best_h, best_n)
        best_val = compute_c5(best_h, best_n)

    return (best_h, best_val, best_n)
