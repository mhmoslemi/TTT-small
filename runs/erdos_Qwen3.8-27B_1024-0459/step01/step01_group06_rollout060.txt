```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time
import math

def compute_all_lags(h, n):
    """Compute all correlation values c[k] for k = -(n-1) to n-1"""
    return np.correlate(h, 1.0 - h, mode="full")

def c5_value(c, n):
    return float(np.max(c) * (2.0 / n))

def p_mean_obj(h, n, p):
    c = compute_all_lags(h, n)
    dx = 2.0 / n
    # Use log-sum-exp for numerical stability
    log_powers = p * np.log(np.maximum(c, 1e-30))
    m = np.max(log_powers)
    val = (m + np.log(np.sum(np.exp(log_powers - m)))) / p
    return float(np.exp(val) * dx)

def p_mean_grad(h, n, p):
    c = compute_all_lags(h, n)
    dx = 2.0 / n
    # Compute gradient of (1/K * sum c[k]^p)^(1/p)
    K = 2 * n - 1
    c_clipped = np.maximum(c, 1e-30)
    weights = c_clipped ** (p - 1)
    # f = (1/K * sum c[k]^p)^(1/p)
    log_powers = p * np.log(c_clipped)
    m = np.max(log_powers)
    log_f = (m + np.log(np.sum(np.exp(log_powers - m))) - np.log(K)) / p
    f = np.exp(log_f)
    
    # grad f / grad h[j] = (f^{p-1} / K) * sum_k c[k]^{p-1} * dc[k]/dh[j]
    factor = f ** (p - 1) / K
    grad = np.zeros(n)
    
    for k_idx in range(K):
        w = weights[k_idx]
        if w < 1e-20:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            # dc/dh[j] = (1-h[j+lag]) for j <= n-1-lag
            # dc/dh[j] -= h[j-lag] for j >= lag
            idx1 = np.arange(n - lag)
            grad[idx1] += w * (1.0 - h[idx1 + lag])
            idx2 = np.arange(lag, n)
            grad[idx2] -= w * h[idx2 - lag]
        else:
            l = -lag
            idx1 = np.arange(n - l)
            grad[idx1] += w * (1.0 - h[idx1 + l])
            idx2 = np.arange(l, n)
            grad[idx2] -= w * h[idx2 - l]
    
    return grad * factor * dx

def hard_grad(h, n):
    c = compute_all_lags(h, n)
    k_star = np.argmax(c)
    lag = k_star - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag >= 0:

        grad[idx1] += (1.0 - h[idx1 + lag])
        idx2 = np.arange(lag, n)

        grad[idx1] += (1.0 - h[idx1 + l])
        idx2 = np.arange(l, n)
        grad[idx2] -= h[idx2 - l]

def project_feasible(h, n, tol=1e-15):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = h.sum() - target
        if abs(diff) < tol:
            break
        if diff > 0:
            mask = h > tol
        else:
            mask = h < 1.0 - tol
        if mask.sum() == 0:
            h[:] = target / n
            break
        h[mask] -= diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def projected_gradient_step(h, grad, step_size, n):
    """Take a projected gradient step"""
    h_new = h - step_size * grad
    return project_feasible(h_new, n)

def backtracking_line_search(h, f_h, grad, n, alpha=0.5, c1=1e-4, max_iter=50):
    """Simple backtracking line search"""
    step = 1.0
    for _ in range(max_iter):
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        f_new = p_mean_obj(h_new, n, p=50)  # or whatever p we're using
        if f_new <= f_h + c1 * step * np.dot(grad, h_new - h):
            return h_new, f_new
        step *= alpha
    return h_new, f_new
```

```python
def pnorm_grad(h, n, p):
    c = np.correlate(h, 1.0-h, mode="full")
    K = 2*n - 1
    c_safe = np.maximum(c, 1e-30)
    w = c_safe ** (p-1)
    sp = np.sum(c_safe ** p)
    f = sp ** (1.0/p)
    factor = f ** (1-p) * (2.0/n)
    
    grad = np.zeros(n)
    # Positive and zero lags: k_idx from n-1 to 2n-2
    for k_idx in range(n-1, K):
        wk = w[k_idx]
        if wk < 1e-25:
            continue
        lag = k_idx - (n-1)
        if lag == 0:
            grad[:] += wk * (1.0 - h)  # (1-h[j-0]) for all j
            grad[:] -= wk * h  # -h[j+0] for all j
        else:
            grad[lag:] += wk * (1.0 - h[:n-lag])  # (1-h[j-lag]) for j=lag..n-1
            grad[:n-lag] -= wk * h[lag:]  # -h[j+lag] for j=0..n-1-lag
    # Negative lags: k_idx from 0 to n-2
    for k_idx in range(n-1):
        wk = w[k_idx]
        if wk < 1e-25:
            continue
        l = (n-1) - k_idx  # this is -lag = l
        grad[:n-l] += wk * (1.0 - h[l:])  # (1-h[j+l]) for j=0..n-1-l
        grad[l:] -= wk * h[:n-l]  # -h[j-l] for j=l..n-1
    
    return grad * factor, c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full")

def c5_of(h, n):
    return float(np.max(compute_c(h, n)) * (2.0 / n))

def pnorm_obj(h, n, p):
    c = compute_c(h, n)
    c_safe = np.maximum(c, 1e-30)
    dx = 2.0 / n
    log_cp = p * np.log(c_safe)
    m = np.max(log_cp)
    log_val = (m + np.log(np.sum(np.exp(log_cp - m)))) / p
    return float(np.exp(log_val) * dx)

def pnorm_grad(h, n, p):
    c = compute_c(h, n)
    K = 2 * n - 1
    c_safe = np.maximum(c, 1e-30)
    w = c_safe ** (p - 1)
    sp = np.sum(c_safe ** p)
    f = sp ** (1.0 / p)
    factor = f ** (1 - p) * (2.0 / n)
    
    grad = np.zeros(n)
    for k_idx in range(n - 1, K):
        wk = w[k_idx]
        if wk < 1e-25:
            continue
        lag = k_idx - (n - 1)
        if lag == 0:
            grad[:] += wk * (1.0 - h) - wk * h
        else:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] -= wk * h[lag:]
    for k_idx in range(n - 1):
        wk = w[k_idx]
        if wk < 1e-25:
            continue
        l = (n - 1) - k_idx
        grad[:n - l] += wk * (1.0 - h[l:])
        grad[l:] -= wk * h[:n - l]
    return grad * factor

def hard_grad(h, n):
    c = compute_c(h, n)
    k_star = np.argmax(c)
    lag = k_star - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - h) - h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag])
        grad[:n - lag] -= h[lag:]
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:])
        grad[l:] -= h[:n - l]
    return grad * dx

def sum_eq_fun(h, n):
    return float(np.sum(h) - n / 2.0)

def sum_eq_jac(h, n):
    return np.ones(n)

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(1000):
        diff = h.sum() - target
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            m = h > 1e-14
        else:
            m = h < 1.0 - 1e-14
        if not m.any():
            h[:] = 0.5
            break
        h[m] -= diff / m.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except:
        n = 96
        h_init = np.full(n, 0.5)
    
    h_init = project(h_init, n)
    best_h = h_init.copy()
    best_c5 = c5_of(best_h, n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n,)}]
    
    # Phase 1: p-norm optimization with L-BFGS-B (no equality constraint, handle via projection)
    # Actually, L-BFGS-B handles bounds but not equality constraints well.
    # Let's use SLSQP for the equality constraint version, or handle it differently.
    
    # Alternative: parameterize h as 0.5 + 0.5*x where sum(x) = 0 and |x| ≤ 1
    # Then x is in the hyperplane {x: sum(x)=0} ∩ [-1,1]^n
    # L-BFGS-B can handle bounds on x but not the equality.
    
    # Better: use SLSQP with the equality constraint, but with the p-norm objective.
    
    for p_val in [8, 20, 80, 300]:
        if time.time() - t0 > budget_s - 120:
            break
        try:
            res = minimize(
                fun=pnorm_obj,
                x0=best_h,
                jac=pnorm_grad,
                args=(n, p_val),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-14}

            h_new = project(np.clip(res.x, 0.0, 1.0), n)
            c5_new = c5_of(h_new, n)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    for _ in range(3):

                fun=lambda h: c5_of(h, n),
                x0=best_h,
                jac=lambda h: hard_grad(h, n),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 2000, 'ftol': 1e-15}
            )
            h_new = project(np.clip(res.x, 0.0, 1.0), n)
            c5_new = c5_of(h_new, n)
            if c5_new < best_c5:

                best_c5 = c5_new
```

```python
def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full")

    return float(np.max(compute_c(h, n)) * (2.0 / n))


def pnorm_obj(h, n, p):
    c = compute_c(h, n)
    c_safe = np.maximum(c, 1e-30)
    dx = 2.0 / n
    log_cp = p * np.log(c_safe)
    m = np.max(log_cp)
    log_val = (m + np.log(np.sum(np.exp(log_cp - m)))) / p
    return float(np.exp(log_val) * dx)

    c = compute_c(h, n)
    K = 2 * n - 1
    c_safe = np.maximum(c, 1e-30)
    w = c_safe ** (p - 1)
    sp = np.sum(c_safe ** p)
    f = sp ** (1.0 / p)
    factor = f ** (1 - p) * (2.0 / n)
    grad = np.zeros(n)
    for k_idx in range(n - 1, K):
        wk = w[k_idx]
        if wk < 1e-25:
            continue
        lag = k_idx - (n - 1)
        if lag == 0:
            grad[:] += wk * (1.0 - h) - wk * h
        else:
            grad[lag:] += wk * (1.0 - h[:n - lag])

            continue
        l = (n - 1) - k_idx
        grad[:n - l] += wk * (1.0 - h[l:])
        grad[l:] -= wk * h[:n - l]
    return grad * factor
```
