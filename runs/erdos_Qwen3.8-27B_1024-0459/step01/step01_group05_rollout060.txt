import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def corr_grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    i = np.arange(n)
    idx1 = i - lag
    mask1 = (idx1 >= 0) & (idx1 < n)
    g[mask1] += (1.0 - h[idx1[mask1]])
    idx2 = i + lag
    mask2 = (idx2 >= 0) & (idx2 < n)
    g[mask2] -= h[idx2[mask2]]
    return g

def project_constraint(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def gradient_descent(h, target, t0, time_budget, top_k_init, top_k_min, lr_init, momentum_init):
    n = len(h)
    h = project_constraint(h, target).copy()
    lr = lr_init
    mom = momentum_init
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stall = 0
    it = 0
    
    while time.time() - t0 < time_budget:
        corr = compute_corr(h)
        max_val = corr.max()
        c5_val = max_val * (2.0 / n)
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                break
        
        frac = min(1.0, it / 5000.0)
        top_k = max(top_k_min, int(top_k_init - frac * (top_k_init - top_k_min)))
        top_k = min(top_k, len(corr))
        
        top_idx = np.argsort(corr)[::-1][:top_k]
        
        g = np.zeros(n)

            mom = max(0.5, mom * 0.9999)
        else:
            lr *= 0.4
            v *= 0.5
        
        it += 1
    
    return best_h, best_c5

def slsqp_smooth_obj(h, n, target, beta):
    corr = compute_corr(h)
    m = corr.max()
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -80.0, 80.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def slsqp_smooth_grad(h, n, target, beta):
    corr = compute_corr(h)
    m = corr.max()
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -80.0, 80.0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    g = np.zeros(n)
    for ki in range(len(corr)):
        if weights[ki] > 1e-12:
            lag = ki - (n - 1)
            g += weights[ki] * corr_grad_lag(h, lag)
    return g

def slsqp_sum_fun(h, n, target):
    return np.sum(h) - target

def slsqp_sum_jac(h, n, target):
    return np.ones(n)

def slsqp_polish(h_init, n, t0, time_budget, beta):
    target = n / 2.0
    h0 = project_constraint(h_init, target).copy()
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq',
                    'fun': slsqp_sum_fun,
                    'jac': slsqp_sum_jac}]
    
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

                         constraints=constraints,
                         options={'maxiter': 150, 'ftol': 1e-12})
