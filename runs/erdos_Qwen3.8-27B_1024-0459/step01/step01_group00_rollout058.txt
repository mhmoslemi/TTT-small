def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    return project_feasible(h, n)

def smooth_max_and_grad(h, beta):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    c_shifted = c - c_max
    p = np.exp(beta * c_shifted)
    p_sum = p.sum()
    p /= p_sum
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        m = k - (n - 1)
        pk = p[k]
        if pk < 1e-14:
            continue
        if m >= 0:
            if m < n:
                idx = np.arange(m, n)
                if len(idx) > 0:
                    grad[idx] += pk * (1.0 - h[idx - m])
            if n - m > 0:
                idx2 = np.arange(0, n - m)
                grad[idx2] -= pk * h[idx2 + m]
        else:
            neg_m = -m
            if n - neg_m > 0:
                idx = np.arange(0, n - neg_m)
                grad[idx] += pk * (1.0 - h[idx + neg_m])
            if neg_m < n:
                idx2 = np.arange(neg_m, n)
                grad[idx2] -= pk * h[idx2 - neg_m]
    
    f_val = (c_max + np.log(p_sum)) / beta
    return f_val, grad, c_max

def smooth_max_optimize(h, n, beta_start=5.0, beta_end=400.0, n_beta_steps=10,
                        steps_per_beta=300, eta_init=0.5, momentum=0.85,
                        time_limit=80, seed=42):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    
    for bs in range(n_beta_steps):
        t = bs / max(n_beta_steps - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** t
        eta = eta_init / (1.0 + 0.5 * bs)
        v = np.zeros(n)
        
        for step in range(steps_per_beta):
            if time.time() - start > time_limit:
                return best_h, best_c5
            f_val, grad, c_max = smooth_max_and_grad(h, beta)
            v = momentum * v + (1.0 - momentum) * grad
            h_new = h - eta * v
            h_new = project_feasible(h_new, n)
            c5 = compute_c5(h_new)

                eta *= 0.9
                v *= 0.5
        
        h = best_h.copy()
    
    return best_h, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    vals = np.interp(x_new, x_old, h_src)
    return np.clip(vals, 0.0, 1.0)

def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return project_feasible(h, n)

def periodic_init(n, pattern):
    h = np.tile(pattern, n // len(pattern) + 1)[:n]
    return project_feasible(h, n)

def weyl_init(n, alpha):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i * alpha) % 1.0 < 0.5 else 0.0
    return project_feasible(h, n)

def slsqp_optimize(n, h_init, max_iter=2000):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    def constraint(x):
        return np.sum(x) - n / 2.0
    try:
        res = minimize(obj, h_init, method='SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': constraint}],
                      options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.fun < 1e10:
            h_out = project_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None
