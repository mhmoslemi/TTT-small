import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def symmetrize(h, n):
    half = n // 2
    v = (h[:half] + h[n-1:half-1:-1]) / 2.0
    h_sym = np.empty(n)
    h_sym[:half] = v
    h_sym[half:] = v[::-1]
    return h_sym, v

def half_to_full(v, n):
    half = n // 2
    h = np.empty(n)
    h[:half] = v
    h[half:] = v[::-1]
    return h

def thue_morse_half(n_half):
    tm = np.zeros(n_half)
    for i in range(n_half):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return tm

def smooth_max_val(corr, beta):
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted = shifted - np.max(shifted)
    return cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))

def smooth_max_obj(v, n, beta):
    h = half_to_full(v, n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_val(corr, beta)

def optimize_sym_slsqp(v, n, beta=50, max_iter=2000, ftol=1e-14):
    half = n // 2
    target_sum = n / 4.0

    def obj(x):
        return smooth_max_obj(x, n, beta)

    def con_eq(x):
        return np.sum(x) - target_sum

    bounds = [(0.0, 1.0)] * half
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': lambda x: np.ones(half)}]

    try:
        res = minimize(obj, v, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
        if res.success or res.fun < 1e10:
            v_opt = np.clip(res.x, 0.0, 1.0)
            diff = target_sum - np.sum(v_opt)

                v_opt = np.clip(v_opt, 0.0, 1.0)
            h_opt = half_to_full(v_opt, n)
            return v_opt, compute_c5(h_opt)
    except Exception:

    return None, None

        i_flip = ones[rng.randint(len(ones))]
        j_flip = zeros[rng.randint(len(zeros))]

        v[i_flip] = 0
        v[j_flip] = 1
        h_try = half_to_full(v, n)
        c5_try = compute_c5(h_try)

        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            no_improve = 0
        else:
            v[i_flip] = 1
            v[j_flip] = 0
            no_improve += 1

    return v, best_c5

def binary_local_search_multi(v_bin, n, time_limit=30, seed=42, n_restarts=3):
    rng = np.random.RandomState(seed)
    start = time.time()
    half = n // 2
    target_sum = n / 4.0
    best_v = v_bin.copy()
    best_c5 = compute_c5(half_to_full(v_bin, n))

    for restart in range(n_restarts):
        if time.time() - start > time_limit:
            break
        v_cur = best_v.copy()
        if restart > 0:
            for _ in range(3):
                ones = np.where(v_cur == 1)[0]
                zeros = np.where(v_cur == 0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    v_cur[ones[rng.randint(len(ones))]] = 0
                    v_cur[zeros[rng.randint(len(zeros))]] = 1

        v_res, c5_res = binary_local_search(v_cur, n, time_limit=6, seed=seed + restart)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_v = v_res.copy()

    return best_v, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def make_half_feasible(v, n):
    half = n // 2
    target_sum = n / 4.0
    v = np.clip(v, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(v)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = v < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            v[mask] += diff / cnt
        else:
            mask = v > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            v[mask] += diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v

def coordinate_descent_sym(v, n, n_iters=2000, time_limit=30):
    start = time.time()
    half = n // 2
    target_sum = n / 4.0
    v = make_half_feasible(v.copy(), n)
    h = half_to_full(v, n)
    best_c5 = compute_c5(h)
    step = 0.02

    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        improved = False
        idxs = np.random.permutation(half)
        for i in idxs:
            if time.time() - start > time_limit:
                break
            old = v[i]
            for delta in [step, -step, step/2, -step/2]:
                new = old + delta
                if new < 0 or new > 1:
                    continue
                v[i] = new
                diff = target_sum - np.sum(v)
                if abs(diff) > 1e-8:
                    others = np.concatenate([np.arange(0, i), np.arange(i+1, half)])
                    mask_o = (v[others] > 0.01) & (v[others] < 0.99)
                    if np.sum(mask_o) == 0:
                        v[i] = old
                        break
                    v[others[mask_o]] += diff / np.sum(mask_o)
                    v = np.clip(v, 0, 1)
                h_try = half_to_full(v, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-13:
                    best_c5 = c5_try
                    improved = True
                    break
                else:
                    v[i] = old
        if not improved and it > 100:
            step *= 0.7
            if step < 0.001:
                break

    return v, best_c5

def non_sym_slsqp(h, n, beta=50, max_iter=1000, ftol=1e-14):
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max_val(corr, beta)

    def con_eq(x):
        return np.sum(x) - target

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': lambda x: np.ones(n)}]

    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
        if res.success or res.fun < 1e10:
            h_opt = make_feasible(res.x, n)
            return h_opt, compute_c5(h_opt)
    except Exception:
        pass
    return None, None

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is None or len(h_parent) == 0:
            h_parent = None
    except (NameError, Exception):
        h_parent = None

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if h_parent is not None:
        best_h = h_parent.copy()
        best_n = len(best_h)
        best_c5 = compute_c5(best_h)
    else:
        n_init = 160
        best_n = n_init
        v_init = np.ones(n_init // 2) * 0.5
        best_h = half_to_full(v_init, n_init)
        best_c5 = compute_c5(best_h)

    # Determine target resolutions
    target_ns = set()
    if best_n % 2 == 0:
        target_ns.add(best_n)
    target_ns.update([256, 320, 400])

    # If best_n is small, add it
    if best_n < 256:
        target_ns.add(best_n)

    target_ns = sorted([n for n in target_ns if n >= 64 and n % 2 == 0])

    # Phase 1: Symmetric optimization at each target resolution
    for n in target_ns:
        if remaining() < 60:
            break

        half = n // 2
        target_sum = n / 4.0
        v_inits = []

        # Init 1: Thue-Morse (adjusted for correct sum)
        tm = thue_morse_half(half)
        diff = target_sum - np.sum(tm)
        if abs(diff) > 0.5:
            if diff > 0:
                zeros = np.where(tm == 0)[0]
                for i in range(min(int(diff), len(zeros))):
                    tm[zeros[i]] = 1.0
            else:
                ones = np.where(tm == 1)[0]
                for i in range(min(int(-diff), len(ones))):
                    tm[ones[i]] = 0.0
        v_inits.append(tm.copy())

        # Init 2: From parent (interpolated + symmetrized)
        if h_parent is not None:
            h_interp = interpolate_to_n(h_parent, n)
            h_sym, v_sym = symmetrize(h_interp, n)
            v_sym = make_half_feasible(v_sym, n)
            v_inits.append(v_sym)

        # Init 3: From current best (interpolated + symmetrized)
        if best_h is not None and len(best_h) != n:
            h_interp2 = interpolate_to_n(best_h, n)
            _, v_good = symmetrize(h_interp2, n)
            v_good = make_half_feasible(v_good, n)
            v_inits.append(v_good)

        # Init 4: Random binary (several)
        for ri in range(4):
            v_rand = np.zeros(half)
            ones_needed = int(target_sum)
            idx = np.random.choice(half, ones_needed, replace=False)
            v_rand[idx] = 1.0
            v_inits.append(v_rand)

        # Init 5: Structured patterns
        if half % 4 == 0:
            v_block = np.tile([1.0, 1.0, 0.0, 0.0], half // 4)
            if np.sum(v_block) == target_sum:
                v_inits.append(v_block.copy())
        if half % 2 == 0:
            v_alt = np.tile([1.0, 0.0], half // 2)
            if np.sum(v_alt) == target_sum:
                v_inits.append(v_alt.copy())

        # Deduplicate
        unique_inits = [v_inits[0]]
        for v in v_inits[1:]:
            is_dup = False
            for u in unique_inits:
                if np.max(np.abs(v - u)) < 0.05:
                    is_dup = True
                    break
            if not is_dup:
                unique_inits.append(v)
        v_inits = unique_inits[:8]

        # Optimize each initialization
        for vi, v0 in enumerate(v_inits):
            if remaining() < 20:
                break

            v0 = make_half_feasible(v0, n)
            h0 = half_to_full(v0, n)
            c5_0 = compute_c5(h0)

            # Binary local search (if close to binary)
            if np.sum(np.abs(v0 - np.round(v0))) < 0.5:
                v_bin = np.round(v0)
                diff_b = target_sum - np.sum(v_bin)
                if abs(diff_b) > 0.5:
                    if diff_b > 0:
                        zeros = np.where(v_bin == 0)[0]
                        for i in range(min(int(diff_b), len(zeros))):
                            v_bin[zeros[i]] = 1
                    else:
                        ones = np.where(v_bin == 1)[0]
                        for i in range(min(int(-diff_b), len(ones))):
                            v_bin[ones[i]] = 0
                if remaining() > 25:
                    v_bl, c5_bl = binary_local_search_multi(v_bin, n, time_limit=20, seed=seed+vi)
                    if c5_bl < best_c5:
                        best_c5 = c5_bl
                        best_h = half_to_full(v_bl, n).copy()
                        best_n = n

            # Smooth max SLSQP (symmetric)
            if remaining() > 25:
                v_work = v0.copy()
                for beta in ([20, 50, 100, 200] if remaining() > 80 else [50, 100]):
                    if remaining() < 15:
                        break
                    v_opt, c5_opt = optimize_sym_slsqp(v_work, n, beta=beta, max_iter=1500)
                    if v_opt is not None:
                        if c5_opt < best_c5:
                            best_c5 = c5_opt
                            best_h = half_to_full(v_opt, n).copy()
                            best_n = n
                        v_work = v_opt.copy()

            # Coordinate descent polish
            if remaining() > 15 and best_n == n:
                v_cd, c5_cd = coordinate_descent_sym(v_work, n, n_iters=1000, time_limit=15)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = half_to_full(v_cd, n).copy()
                    best_n = n

    # Phase 2: Non-symmetric polish on the best solution
    if remaining() > 40:
        n_cur = best_n
        h_work = best_h.copy()
        for beta in ([30, 80, 150] if remaining() > 100 else [80]):
            if remaining() < 20:
                break
            h_opt, c5_opt = non_sym_slsqp(h_work, n_cur, beta=beta, max_iter=1000)
            if h_opt is not None and c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h_work = h_opt.copy()

    # Phase 3: Final local perturbation
    if remaining() > 20:
        n_cur = best_n
        h_work = best_h.copy()
        n_trials = min(3000, int((remaining() - 5) / 0.5))
        best_local_c5 = compute_c5(h_work)

        for _ in range(n_trials):
            if remaining() < 5:
                break
            h_try = h_work.copy()
            n_mod = np.random.randint(1, 4)
            idxs = np.random.choice(n_cur, n_mod, replace=False)
            for idx in idxs:
                h_try[idx] += np.random.uniform(-0.03, 0.03)
            h_try = np.clip(h_try, 0.0, 1.0)
            diff = n_cur / 2.0 - np.sum(h_try)
            mask = (h_try > 0.01) & (h_try < 0.99)
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h_try[mask] += diff / cnt
            h_try = np.clip(h_try, 0.0, 1.0)
            c5_try = compute_c5(h_try)
            if c5_try < best_local_c5 - 1e-13:
                h_work = h_try
                best_local_c5 = c5_try

        if best_local_c5 < best_c5:
            best_c5 = best_local_c5
            best_h = h_work.copy()
            best_n = n_cur

    # Phase 4: Try upscaling to higher resolution if time permits
    if remaining() > 80 and best_n < 512:
        n_up = min(best_n * 2, 512)
        if n_up % 2 != 0:
            n_up += 1
        h_interp = interpolate_to_n(best_h, n_up)
        h_sym, v_sym = symmetrize(h_interp, n_up)
        v_sym = make_half_feasible(v_sym, n_up)
        h_work = half_to_full(v_sym, n_up)

        for beta in [50, 100]:
            if remaining() < 30:
                break
            v_opt, c5_opt = optimize_sym_slsqp(v_sym, n_up, beta=beta, max_iter=1500)
            if v_opt is not None:
                h_work = half_to_full(v_opt, n_up)
                v_sym = v_opt.copy()
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_work.copy()
                    best_n = n_up

    # Final feasibility check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
