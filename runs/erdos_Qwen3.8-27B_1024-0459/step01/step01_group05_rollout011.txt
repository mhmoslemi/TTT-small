```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -50, 50)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_grad(x, beta):
    """Return (value, gradient) of smooth max."""
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -50, 50)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    val = m + np.log(sum_exp) / beta
    grad = exp_vals / sum_exp  # This is the weight for each component
    return val, grad

def project_to_feasible(h, target):
    """Project h to {sum(h)=target, 0<=h<=1}."""
    h = np.clip(h.astype(float), 0, 1)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        # Find free variables
        if diff > 0:
            free = h < 1.0
        else:
            free = h > 0.0
        if not np.any(free):
            break
        per = diff / np.sum(free)
        h[free] += per
        h = np.clip(h, 0, 1)
    return h

def make_objective(h, beta):
    hc = np.clip(h, 0, 1)
    corr = np.correlate(hc, 1.0 - hc, mode='full')
    return smooth_max(corr, beta)

def make_objective_grad(h, beta):
    """Numerical gradient of smooth max of correlations."""
    n = len(h)
    hc = np.clip(h, 0, 1)
    f0 = make_objective(hc, beta)
    g = np.zeros(n)
    eps = 1e-7
    for i in range(n):
        hc2 = hc.copy()
        hc2[i] = min(1.0, max(0.0, hc2[i] + eps))
        g[i] = (make_objective(hc2, beta) - f0) / eps
    return f0, g
```

```python
    for i, (b, mi) in enumerate(zip(betas_parent, iters_parent)):
        if elapsed_time(t0) > budget_s * 0.18:

        h_cur = run_slsqp(best_h, best_n, best_n / 2.0, b, mi, 1e-13)

    if elapsed_time(t0) < budget_s * 0.35:

    s = np.sum(best_h)

    if abs(diff) > 1e-12:

        best_h = np.clip(best_h, 0.0, 1.0)
    
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n_final)
```
