    #   Δ = +h[a-m] for m where 0<=a-m<n, i.e., a-n<m<=a, i.e., m in (a-n, a]
    
    # For h[b] going 0→1:
    #   As h[i] (i=b): ΔO(m) += +(1-h[b+m]) for m where 0<=b+m<n
    #   As (1-h[i+m]) (i+m=b, i=b-m):
    #   Old: h[b-m]*(1-h[b]) = h[b-m]*1 = h[b-m]
    #   New: h[b-m]*(1-h[b]) = h[b-m]*0 = 0
    #   Δ = -h[b-m] for m where 0<=b-m<n

        delta = 0.0

        if 0 <= a + m < n:
            delta -= (1.0 - h[a + m])
        # h[a] 1→0 as 1-h[i+m]:

        # h[b] 0→1 as 1-h[i+m]:
        if 0 <= b - m < n:
            delta -= h[b - m]
        deltas[idx] = delta
    
    return deltas

def compute_swap_delta(h, a, b, n):
    """Compute delta in all 2n-1 overlap values when h[a] goes 1→0 and h[b] goes 0→1."""
    delta = np.zeros(2*n-1)
    
    # Four contributions:
    # 1. h[a] as first element: pair (a, a+m), Δ = -(1-h[a+m]) for valid a+m
    # 2. h[a] as second element: pair (a-m, a), Δ = +h[a-m] for valid a-m  [NOTE: this uses OLD h[a]=1, so 1-h[a]=0, new 1-h[a]=1, change in (1-h[a]) is +1, multiplied by h[a-m]]
    # 3. h[b] as first element: pair (b, b+m), Δ = +(1-h[b+m]) for valid b+m
    # 4. h[b] as second element: pair (b-m, b), Δ = -h[b-m] for valid b-m
    
    # Contribution 1: m in range where 0 ≤ a+m < n → m in [-a, n-1-a]
    m_lo = max(-(n-1), -a)
    m_hi = min(n-1, n-1-a)
    if m_lo <= m_hi:
        ms = np.arange(m_lo, m_hi+1)
        delta[ms + (n-1)] -= (1.0 - h[a + ms])
    
    # Contribution 2: m in range where 0 ≤ a-m < n → m in [a-n+1, a]
    m_lo = max(-(n-1), a-n+1)
    m_hi = min(n-1, a)
    if m_lo <= m_hi:
        ms = np.arange(m_lo, m_hi+1)
        delta[ms + (n-1)] += h[a - ms]
    
    # Contribution 3: m in range where 0 ≤ b+m < n → m in [-b, n-1-b]
    m_lo = max(-(n-1), -b)
    m_hi = min(n-1, n-1-b)
    if m_lo <= m_hi:
        ms = np.arange(m_lo, m_hi+1)
        delta[ms + (n-1)] += (1.0 - h[b + ms])
    
    # Contribution 4: m in range where 0 ≤ b-m < n → m in [b-n+1, b]
    m_lo = max(-(n-1), b-n+1)
    m_hi = min(n-1, b)

        delta[ms + (n-1)] -= h[b - ms]
