            #   If j >= lag and j <= n-1: contributes (1-h[j-lag]) from the term i=j
            #   If j >= 0 and j <= n-1-lag: contributes -h[j+lag] from the term i=j+lag (where h[i-lag]=h[j])
            contrib_plus = np.zeros(n)
            contrib_plus[lag:] = (1.0 - h[:n-lag])  # for i in [lag, n-1], (1-h[i-lag])
            contrib_minus = np.zeros(n)
            if n - lag > 0:
                contrib_minus[:n-lag] = h[lag:]  # for i in [lag, n-1], h[i] = h[j+lag] where j=i-lag
            grad += wk * (contrib_plus - contrib_minus)
        else:
            # R(-m) where m = -lag > 0
            # R(-m) = sum_{i=0}^{n-1-m} h[i](1-h[i+m])
            m = -lag
            contrib_plus = np.zeros(n)
            contrib_plus[:n-m] = (1.0 - h[m:])
            contrib_minus = np.zeros(n)
            if n - m > 0:
                contrib_minus[m:] = h[:n-m]
            grad += wk * (contrib_plus - contrib_minus)
    return grad

def hard_max_obj(h, n):
    c = get_c5_values(h, n)
    return float(np.max(c) * (2.0 / n))

def hard_max_grad(h, n):
    c = get_c5_values(h, n)
    k_max = int(np.argmax(c))
    dx = 2.0 / n
    grad = np.zeros(n)

    if lag >= 0:
        grad[lag:] += (1.0 - h[:n-lag])
        if n - lag > 0:

    else:
        m = -lag
        grad[:n-m] += (1.0 - h[m:])
        if n - m > 0:
            grad[m:] -= h[:n-m]
    return grad * dx

        else:
            pairs = [(i, j) for i in one_list for j in zero_list]

            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h, n)
            h[i] = 1.0
            h[j] = 0.0

                best_improvement = best_c - c_new
                best_pair = (i, j)
                best_direction = 1  # 1->0, 0->1
                best_c_new = c_new
            elif c_new < best_c:
                best_improvement = best_c - c_new
                best_pair = (i, j)

                ones.add(j)
                zeros.remove(j)
                zeros.add(i)

        else:
            # Try reverse: h[i] 0->1, h[j] 1->0
            # Actually we already tried both directions via all pairs
            break

Actually, I want to add one more technique: **iterative reweighting on lags**.

The idea:
1. Compute c[k] for all k
2. Identify the "active" lags (those with c[k] > threshold)
3. Solve: min_h max_{k in active} c[k] using SLSQP
4. Update active set based on new c[k] values
5. Repeat

This focuses the optimization on the lags that actually matter.

But this is essentially what the soft-max already does (it weights all lags by their exponential of the value). So I'll stick with the soft-max approach but with more aggressive annealing.

import numpy as np
from scipy.optimize import minimize
import time


def get_c5_values(h, n):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h, n):
    c = get_c5_values(h, n)
    return float(np.max(c) * (2.0 / n))


def soft_max_obj(h, n, beta):
    c = get_c5_values(h, n) * (2.0 / n)
    scaled = beta * c
    m = np.max(scaled)
    if m > 500.0 / max(beta, 1e-6):
        return float(m)
    se = scaled - m
    se[se < -50] = -50.0
    return float(m + np.log(np.sum(np.exp(se))))


def soft_max_grad(h, n, beta):
    c = get_c5_values(h, n)
    dx = 2.0 / n
    scaled = beta * c * dx
    m = np.max(scaled)
    se = scaled - m
    se[se < -50] = -50.0
    w = np.exp(se)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = beta * dx * w[k_idx]
        if wk < 1e-14:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            if n - lag > 0:
                grad[:n - lag] -= wk * h[lag:]
        else:
            m2 = -lag
            grad[:n - m2] += wk * (1.0 - h[m2:])
            if n - m2 > 0:
                grad[m2:] -= wk * h[:n - m2]
    return grad

        grad[:n - m2] += (1.0 - h[m2:])

def constraint_jac(x, n):

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-12
        else:
            mask = h < 1.0 - 1e-12
        if mask.sum() == 0:
            break
        h[mask] -= diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def subgradient_descent(h, n, max_iter, t0, budget_s, margin):
    h_cur = h.copy()
    best_c = compute_c5(h_cur, n)
    best_h = h_cur.copy()
    step = 0.03
    patience = 0
    for it in range(max_iter):

        c_new = compute_c5(h_new, n)

            best_c = c_new
            best_h = h_new.copy()
            h_cur = h_new
            patience = 0

            h_cur = h_new
            patience += 1
            if patience % 50 == 0:
                step *= 0.9
                step = max(step, 0.0005)
