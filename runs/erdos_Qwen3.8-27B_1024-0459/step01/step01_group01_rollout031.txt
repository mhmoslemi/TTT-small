import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_obj_grad(h, tau, lam):
    L = len(h)
    dx = 2.0 / L
    v = 1.0 - h
    corr = np.correlate(h, v, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / tau
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    sum_exp = np.sum(exp_shifted)
    val = (m + tau * (np.log(sum_exp) + max_shifted)) * dx
    weights = exp_shifted / sum_exp
    
    # Penalty term
    penalty = lam * (h.sum() / L - 0.5) ** 2
    val += penalty
    
    # Gradient of smooth max
    grad = np.zeros(L)
    for i in range(2 * L - 1):
        wi = weights[i]
        if wi < 1e-14:
            continue
        k = i - (L - 1)
        j_lo = max(0, -k)
        j_hi = min(L - 1, L - 1 - max(0, k))
        if j_lo > j_hi:
            continue
        js = np.arange(j_lo, j_hi + 1)
        grad[js] += wi * ((1.0 - h[js - k]) - h[js + k])
    grad *= dx
    
    # Gradient of penalty
    grad += 2.0 * lam * (h.sum() / L - 0.5) / L * np.ones(L)
    
    return val, grad

def smooth_max_obj(h, tau, lam):
    val, _ = smooth_max_obj_grad(h, tau, lam)
    return val

def smooth_max_jac(h, tau, lam):
    _, grad = smooth_max_obj_grad(h, tau, lam)
    return grad

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:

        if diff > 0:

        else:

        if len(idx) == 0:
            break
        per = diff / len(idx)
        # Vectorized update
        h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h

def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def compute_gradient(h, weights):
    L = len(h)
    grad = np.zeros(L)
    for i in range(2 * L - 1):
        wi = weights[i]
        if wi < 1e-14:
            continue
        k = i - (L - 1)  # actual lag
        
        # First term: (1-h[j-k]) for valid j
        # Valid j range: 
        #   if k >= 0: k <= j <= L-1
        #   if k < 0: 0 <= j <= L-1+k
        if k >= 0:
            j1_lo = k
            j1_hi = L - 1
        else:
            j1_lo = 0
            j1_hi = L - 1 + k
        
        if j1_lo <= j1_hi:
            js1 = np.arange(j1_lo, j1_hi + 1)
            grad[js1] += wi * (1.0 - h[js1 - k])
        
        # Second term: -h[j+k] for valid j
        # Valid j range:
        #   if k >= 0: 0 <= j <= L-1-k
        #   if k < 0: -k <= j <= L-1
        if k >= 0:
            j2_lo = 0
            j2_hi = L - 1 - k
        else:
            j2_lo = -k
            j2_hi = L - 1
        
        if j2_lo <= j2_hi:
            js2 = np.arange(j2_lo, j2_hi + 1)
            grad[js2] -= wi * h[js2 + k]
    
    return grad

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max_obj_grad(h, tau, lam):
    L = len(h)
    dx = 2.0 / L
    v = 1.0 - h
    corr = np.correlate(h, v, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / tau
    max_sh = np.max(shifted)
    exp_sh = np.exp(shifted - max_sh)
    sum_exp = np.sum(exp_sh)
    val = (m + tau * (np.log(sum_exp) + max_sh)) * dx
    weights = exp_sh / sum_exp

    penalty = lam * (h.sum() / L - 0.5) ** 2
    val += penalty

    grad = np.zeros(L)
    for i in range(2 * L - 1):
        wi = weights[i]
        if wi < 1e-14:
            continue
        k = i - (L - 1)
        if k >= 0:
            j1_lo = k
            j1_hi = L - 1
            j2_lo = 0
            j2_hi = L - 1 - k
        else:
            j1_lo = 0
            j1_hi = L - 1 + k
            j2_lo = -k
            j2_hi = L - 1
        if j1_lo <= j1_hi:
            js1 = np.arange(j1_lo, j1_hi + 1)
            grad[js1] += wi * (1.0 - h[js1 - k])
        if j2_lo <= j2_hi:
            js2 = np.arange(j2_lo, j2_hi + 1)
            grad[js2] -= wi * h[js2 + k]
    grad *= dx

    grad += 2.0 * lam * (h.sum() / L - 0.5) / L

    return val, grad


def smooth_max_obj(h, tau, lam):
    return smooth_max_obj_grad(h, tau, lam)[0]


def smooth_max_jac(h, tau, lam):
    return smooth_max_obj_grad(h, tau, lam)[1]


def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h


def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def do_lbgfsb(h0, n, tau, lam, maxiter, ftol, gtol):
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        smooth_max_obj, h0, args=(tau, lam),
        jac=smooth_max_jac,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol, 'disp': False}
    )
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = fix_sum(h_new, n)
    return h_new


def periodic_pattern(n, period_frac, duty_frac):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    period = 2.0 * period_frac
    h = np.zeros(n)
    for i in range(n):
        phase = (x[i] % period) / period
        h[i] = 1.0 if phase < duty_frac else 0.0
    return fix_sum(h, n)


def cosine_init(n, freq):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.45 * np.cos(2.0 * np.pi * freq * x)
    return fix_sum(np.clip(h, 0.0, 1.0), n)
