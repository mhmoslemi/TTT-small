import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)


def compute_all_shifts(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    return conv


def smooth_max_obj(h, n_points, temp):
    conv = compute_all_shifts(h, n_points)
    c = np.max(conv)
    if temp < 1e-12:
        return c
    shifted = (conv - c) / temp
    max_shifted = np.max(shifted)
    return c + temp * (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted))))


def smooth_max_grad(h, n_points, temp):
    n = n_points
    dx = 2.0 / n
    conv = compute_all_shifts(h, n)
    c = np.max(conv)
    if temp < 1e-12:
        k_max = int(np.argmax(conv))
        m = k_max - (n - 1)
        grad = np.zeros(n)
        if m >= 0:
            if m < n:
                grad[:n - m] += (1.0 - h[m:]) * dx
                grad[m:] += (-h[:n - m]) * dx
        else:
            p = -m
            if p < n:
                grad[p:] += (1.0 - h[:n - p]) * dx
                grad[:n - p] += (-h[p:]) * dx
        return grad
    shifted = (conv - c) / temp
    max_shifted = np.max(shifted)
    weights = np.exp(shifted - max_shifted)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for k_idx in active:
        w = weights[k_idx]
        m = k_idx - (n - 1)
        if m >= 0:
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:]) * dx
                grad[m:] += w * (-h[:n - m]) * dx
        else:
            p = -m
            if p < n:
                grad[p:] += w * (1.0 - h[:n - p]) * dx
                grad[:n - p] += w * (-h[p:]) * dx
    return grad


def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-12
            cnt = np.sum(free)
            if cnt == 0:
                break
            step = diff / cnt
            h[free] = np.minimum(h[free] + step, 1.0)
        else:
            free = h > 1e-12
            cnt = np.sum(free)
            if cnt == 0:
                break
            step = diff / cnt
            h[free] = np.maximum(h[free] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    residual = target - s
    if abs(residual) > 1e-13:
        if residual > 0:
            free = h < 1.0
            if np.any(free):
                h[free] = np.minimum(h[free] + residual / np.sum(free), 1.0)
        else:
            free = h > 0.0
            if np.any(free):
                h[free] = np.maximum(h[free] + residual / np.sum(free), 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h


def slsqp_smooth(h0, n_points, temp, maxiter=300, ftol=1e-12):
    n = n_points
    target = n / 2.0

    def obj(x):
        return smooth_max_obj(x, n, temp)

    def jac(x):
        return smooth_max_grad(x, n, temp)

    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    res = minimize(obj, h0.copy(), jac=jac, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x, res.fun


def lbfgsb_smooth(h0, n_points, temp, maxiter=500, ftol=1e-12):
    n = n_points
    target = n / 2.0
    penalty = 1000.0

    def obj(x):
        x_clip = np.clip(x, 0, 1)
        base = smooth_max_obj(x_clip, n, temp)
        s = np.sum(x_clip) - target
        return base + penalty * s * s

    def jac(x):
        x_clip = np.clip(x, 0, 1)
        g = smooth_max_grad(x_clip, n, temp)
        s = np.sum(x_clip) - target
        g += 2.0 * penalty * s * np.ones(n)
        return g

    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0.copy(), jac=jac, method='L-BFGS-B',
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
    return res.x, res.fun


def slsqp_hard(h0, n_points, maxiter=300, ftol=1e-12):

def structured_perturb(h, n_points, rng, scale=0.1):
    n = n_points
    h_p = h.copy()
    n_bumps = rng.randint(1, 4)
    for _ in range(n_bumps):
        center = rng.randint(0, n)
        width = rng.randint(max(3, n // 20), n // 4)
        amp = rng.uniform(-scale, scale)

                h_p[i] += amp * np.cos(np.pi * d / (2.0 * width)) ** 2
    h_p = np.clip(h_p, 0, 1)
    return project_feasible(h_p, n)


def segment_flip_perturb(h, n_points, rng):

            except Exception:
                pass

    # Phase 8: Final aggressive polish
    for temp_final in [0.005, 0.002, 0.001, 0.0005]:
        if time_left() < 15:
            break
        try:
            h_try, _ = slsqp_smooth(best_h, best_n, temp_final, maxiter=400)
            h_try = project_feasible(h_try, best_n)
            v_try = compute_c5(h_try, best_n)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
            elif v_try >= best_val:
                break
        except Exception:
            break

    # Phase 9: One final hard max SLSQP
    if time_left() > 10:
        try:
            h_try, _ = slsqp_hard(best_h, best_n, maxiter=500)
            h_try = project_feasible(h_try, best_n)
            v_try = compute_c5(h_try, best_n)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
        except Exception:
            pass

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_val = compute_c5(best_h, best_n)

    # Safety: ensure sum constraint is met exactly
    target = best_n / 2.0
    s = np.sum(best_h)
    if abs(s - target) > 1e-10:
        best_h += (target - s) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)
        best_val = compute_c5(best_h, best_n)

    return (best_h, best_val, best_n)
