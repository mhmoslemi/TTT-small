        h += (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_func(x):
    return compute_c5(x, _g_n)


def obj_grad(x):
    return compute_grad_max(x, _g_n)


def eq_constraint(x):
    return np.sum(x) - _g_target


def eq_constraint_jac(x):
    return np.ones(_g_n)


def is_prime(n):
    if n < 2:
        return False
    if n < 4:
        return True
    if n % 2 == 0:
        return False
    for i in range(3, int(n**0.5) + 1, 2):
        if n % i == 0:
            return False
    return True


def gen_qr_sequence(n):
    for p in [n + 1, n + 3, n + 5, n + 7, n + 9, n + 11, n + 15, n + 17, n + 19]:
        if p <= 3:
            continue
        if is_prime(p) and p % 4 == 3:
            h = np.zeros(n)
            cnt = 0
            for i in range(n):
                val = (i + 1) % p
                if val > 0 and pow(val, (p - 1) // 2, p) == 1:
                    h[i] = 1.0
                    cnt += 1
            if cnt == n // 2:
                return h
            break
    return None


def greedy_construction(n):
    h = np.zeros(n)
    remaining = list(range(n))
    for step in range(n // 2):
        best_pos = remaining[0]
        best_val = float('inf')
        for pos in remaining:
            h[pos] = 1.0
            val = compute_c5(h, n)
            h[pos] = 0.0
            if val < best_val:
                best_val = val
                best_pos = pos
        h[best_pos] = 1.0
        remaining.remove(best_pos)
    return h


def gen_candidates(n, seed, h_parent=None):
    rng = np.random.RandomState(seed)
    candidates = []

    if h_parent is not None and len(h_parent) == n:
        h_bin = (h_parent > 0.5).astype(float)
        if np.sum(h_bin) == n // 2:
            candidates.append(h_bin)

    h_greedy = greedy_construction(n)
    candidates.append(h_greedy)

    for _ in range(6):
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
        candidates.append(h)

    h_alt = np.zeros(n)
    h_alt[::2] = 1.0
    if np.sum(h_alt) == n // 2:
        candidates.append(h_alt)

    h_block = np.zeros(n)
    h_block[:n // 2] = 1.0
    candidates.append(h_block)

    h_qr = gen_qr_sequence(n)
    if h_qr is not None:
        candidates.append(h_qr)

    for freq in [3, 5, 7, 11, 13, 17, 21]:
        x = np.linspace(0, 2, n, endpoint=False)
        h_s = (np.sin(2 * np.pi * freq * x / 2.0) > 0).astype(float)
        if np.sum(h_s) == n // 2:
            candidates.append(h_s)

    return candidates


def binary_steepest_descent(h, n, max_rounds=300, t_start=0.0, time_limit=120.0):
    h = (h > 0.5).astype(float).copy()
    best_val = compute_c5(h, n)
    best_h = h.copy()

    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

        best_improve = 1e-12
        best_a, best_b = -1, -1

        for a in ones:
            for b in zeros:
                h[a], h[b] = h[b], h[a]
                val = compute_c5(h, n)
                h[a], h[b] = h[b], h[a]
                if best_val - val > best_improve:
                    best_improve = best_val - val
                    best_a, best_b = a, b

        if best_a >= 0:
            h[best_a], h[best_b] = h[best_b], h[best_a]
            best_val -= best_improve
            best_h = h.copy()
        else:
            break

    return best_h, best_val


def sa_jitter(h, n, best_val, seed, n_steps=100, T=0.05):
    rng = np.random.RandomState(seed)
    h = h.copy()
    for _ in range(n_steps):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h[a], h[b] = h[b], h[a]
        new_val = compute_c5(h, n)
        delta = new_val - best_val
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            if new_val < best_val:
                best_val = new_val
        else:
            h[a], h[b] = h[b], h[a]
    return h, best_val


def continuous_refine(h, n, maxiter=500, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = n / 2.0

    h0 = h.astype(float).copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_constraint, 'jac': eq_constraint_jac}]

    try:
        res = minimize(obj_func, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_res = project_feasible(res.x, n)
        val = compute_c5(h_res, n)
        return h_res, val
    except Exception:
        return h0, compute_c5(h0, n)


def multi_cycle_search(h_init, n, seed, t_start, time_limit, n_cycles=5):
    h_bin = (h_init > 0.5).astype(float)
    h_cur, v_cur = binary_steepest_descent(h_bin, n, max_rounds=200,
                                           t_start=t_start, time_limit=time_limit)
    h_best = h_cur.copy()
    v_best = v_cur

    for cyc in range(n_cycles):
        if time.time() - t_start > time_limit - 30:
            break
        T = 0.03 + 0.02 * cyc
        h_j, v_j = sa_jitter(h_best, n, v_best, seed + cyc * 100,
                             n_steps=60 + 20 * cyc, T=T)
        h_new, v_new = binary_steepest_descent(h_j, n, max_rounds=150,
                                               t_start=t_start, time_limit=time_limit)
        if v_new < v_best:
            h_best = h_new
            v_best = v_new

    return h_best, v_best


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    best_h = None
    best_val = float('inf')
    best_n = None

    if h_parent is not None and len(h_parent) > 0:
        n_p = len(h_parent)
        h_p = project_feasible(h_parent, n_p)
        v_p = compute_c5(h_p, n_p)
        best_h = h_p.copy()
        best_val = v_p
        best_n = n_p

    n_parent = best_n if best_n is not None else 70

    # Phase 1: Thorough search at parent resolution
    candidates = gen_candidates(n_parent, seed, h_parent)

    # Quick screen all candidates
    scored = []
    for c in candidates:
        c_bin = (c > 0.5).astype(float)
        v = compute_c5(c_bin, n_parent)
        scored.append((v, c_bin))
    scored.sort(key=lambda x: x[0])

    n_top = min(8, len(scored))
    for i in range(n_top):
        if budget_s - (time.time() - t0) < 90:
            break

        h_cand = scored[i][1]
        h_bin, v_bin = multi_cycle_search(h_cand, n_parent, seed + i * 7,
                                          t0, budget_s - 90, n_cycles=4)

        v_final = v_bin
        h_final = h_bin.copy()

        if budget_s - (time.time() - t0) > 60:
            h_cont, v_cont = continuous_refine(h_bin, n_parent, maxiter=400, ftol=1e-14)
            if v_cont < v_final:
                v_final = v_cont
                h_final = h_cont

        if v_final < best_val:
            best_val = v_final
            best_h = h_final.copy()
            best_n = n_parent

    # Phase 2: Try nearby n values
    n_nearby = [n_parent + d for d in [-2, 2, -4, 4, -6, 6, 8, -8, 10, -10, 12, 14, 16, 18, 20, 24, 28, 30, 32, 36, 40, 50, 60, 70]]
    n_nearby = [n for n in n_nearby if n >= 40 and n % 2 == 0 and n != n_parent]
    n_nearby = sorted(set(n_nearby), key=lambda x: abs(x - n_parent))

    for n_try in n_nearby:
        if budget_s - (time.time() - t0) < 50:
            break

        cands_n = gen_candidates(n_try, seed + n_try)
        scored_n = []
        for c in cands_n[:6]:
            c_bin = (c > 0.5).astype(float)
            v = compute_c5(c_bin, n_try)
            scored_n.append((v, c_bin))
        scored_n.sort(key=lambda x: x[0])

        for i in range(min(2, len(scored_n))):
            if budget_s - (time.time() - t0) < 35:
                break

            h_cand = scored_n[i][1]
            h_bin, v_bin = multi_cycle_search(h_cand, n_try, seed + n_try + i,
                                              t0, budget_s - 40, n_cycles=3)

            v_final = v_bin
            h_final = h_bin.copy()

            if budget_s - (time.time() - t0) > 40:
                h_cont, v_cont = continuous_refine(h_bin, n_try, maxiter=300, ftol=1e-13)
                if v_cont < v_final:
                    v_final = v_cont
                    h_final = h_cont

            if v_final < best_val:
                best_val = v_final
                best_h = h_final.copy()
                best_n = n_try

    # Phase 3: Final polish on best solution
    if best_h is not None and budget_s - (time.time() - t0) > 30:
        for attempt in range(6):
            if budget_s - (time.time() - t0) < 15:
                break
            if attempt < 3:
                h_ref, v_ref = continuous_refine(best_h, best_n, maxiter=600, ftol=1e-15)
            else:
                rng_p = np.random.RandomState(seed + 5000 + attempt)
                h_pert = best_h + rng_p.normal(0, 0.04 + 0.02 * attempt, best_n)
                h_pert = project_feasible(h_pert, best_n)
                h_ref, v_ref = continuous_refine(h_pert, best_n, maxiter=400, ftol=1e-14)

            if v_ref < best_val - 1e-13:
                best_val = v_ref
                best_h = h_ref.copy()

    # Final validation
    if best_h is not None:
        best_h = project_feasible(best_h, best_n)
        best_val = compute_c5(best_h, best_n)
    else:
        best_h = np.full(70, 0.5)
        best_n = 70
        best_val = compute_c5(best_h, best_n)

    return (best_h, best_val, best_n)
