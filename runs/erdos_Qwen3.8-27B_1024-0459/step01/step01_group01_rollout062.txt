    """Project y onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo, hi = -2.0, 2.0  # Range for lambda
    for _ in range(60):

        h = np.clip(y - lam, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    h = np.clip(y - (lo+hi)/2, 0.0, 1.0)
    # Fix any tiny numerical issues
    diff = target - h.sum()

        mask = (h > 0) & (h < 1)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h

import numpy as np
import time
import math


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def all_corr_values(h):
    return np.correlate(h, 1.0 - h, mode="full")


def grad_corr_at(h, n, m):
    grad = np.zeros(n)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m > 0:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    else:
        p = -m
        grad[p:] += (1.0 - h[:n - p])
        grad[:n - p] -= h[p:]
    return grad


def subgradient_topk(h, n, top_k=5):
    corr = all_corr_values(h)
    m_vals = np.arange(2 * n - 1) - (n - 1)
    top_idx = np.argsort(corr)[-top_k:]
    top_idx = top_idx[::-1]
    g = np.zeros(n)
    for k_full in top_idx:
        g += grad_corr_at(h, n, m_vals[k_full])
    g /= len(top_idx)
    return g, corr, m_vals


def fw_binary_oracle(g, n):
    n_half = n // 2
    idx = np.argsort(g)[:n_half]
    s = np.zeros(n)
    s[idx] = 1.0
    return s


def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(100):
        lam = (lo + hi) / 2.0
        h_tmp = np.clip(y - lam, 0.0, 1.0)
        s = h_tmp.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    h_tmp = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h_tmp.sum()
    if abs(diff) > 1e-12:
        mask = (h_tmp > 1e-10) & (h_tmp < 1.0 - 1e-10)
        if mask.sum() > 0:
            h_tmp[mask] += diff / mask.sum()
            h_tmp = np.clip(h_tmp, 0.0, 1.0)
    return h_tmp


def fix_sum(h, n):
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        for i in idx:
            h[i] = np.clip(h[i] + per, 0.0, 1.0)
    return h


def round_to_binary(h, n):
    n_half = n // 2
    sorted_idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[sorted_idx[:n_half]] = 1.0
    return b


def binary_sa(h_bin, n, seed, t_budget):
    rng = np.random.default_rng(seed)
    h = h_bin.copy()
    overlap = all_corr_values(h).astype(float)
    current_max = overlap.max()
    best_h = h.copy()
    best_max = current_max

    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])

    m_vals = np.arange(2 * n - 1) - (n - 1)

    T = max(current_max * 0.15, 0.01)
    T_min = max(current_max * 0.002, 0.001)
    n_iter_target = int(t_budget / 0.00003)
    cooling_rate = (T_min / T) ** (1.0 / max(1, n_iter_target))

    t_start = time.time()
    iter_count = 0
    while time.time() - t_start < t_budget:
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]

        delta = np.zeros(2 * n - 1)
        idx1 = (i + m_vals >= 0) & (i + m_vals < n)
        if idx1.any():
            delta[idx1] += h[i + m_vals[idx1]]
        idx2 = (i - m_vals >= 0) & (i - m_vals < n)
        if idx2.any():
            delta[idx2] += h[i - m_vals[idx2]]
        idx3 = (j + m_vals >= 0) & (j + m_vals < n)
        if idx3.any():
            delta[idx3] -= h[j + m_vals[idx3]]
        idx4 = (j - m_vals >= 0) & (j - m_vals < n)
        if idx4.any():
            delta[idx4] -= h[j - m_vals[idx4]]

        new_overlap = overlap + delta
        new_max = new_overlap.max()

        accept = new_max <= current_max
        if not accept and T > T_min:
            ratio = (current_max - new_max) / T
            if ratio > -50:
                accept = rng.random() < math.exp(ratio)

        if accept:
            h[i] = 0.0
            h[j] = 1.0
            ones_list.remove(i)
            ones_list.append(j)
            zeros_list.remove(j)
            zeros_list.append(i)
            overlap = new_overlap
            current_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()

        T = max(T * cooling_rate, T_min)
        iter_count += 1
        if iter_count % 50000 == 0:
            T = max(T * 0.995, T_min)

    return best_h, best_max


def frank_wolfe_opt(h0, n, t_budget, seed, top_k=5):

def projected_grad_opt(h0, n, t_budget, seed, tau=0.05):
    rng = np.random.default_rng(seed)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    m_vals = np.arange(2 * n - 1) - (n - 1)
    n_corr = 2 * n - 1

    t_start = time.time()
    t = 0
    alpha = 0.05
    momentum = np.zeros(n)
    beta_mom = 0.9

    while time.time() - t_start < t_budget:
        corr = all_corr_values(h)
        scaled = (corr - corr.max()) / tau
        weights = np.exp(scaled)
        weights /= weights.sum()

        g = np.zeros(n)
        for k_idx in range(n_corr):
            if weights[k_idx] > 1e-8:
                g += weights[k_idx] * grad_corr_at(h, n, m_vals[k_idx])

        momentum = beta_mom * momentum + (1 - beta_mom) * g
        h_raw = h - alpha * momentum
        h_new = project_feasible(h_raw, n)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

        h = h_new
        t += 1
        alpha = max(0.01, 0.05 / math.sqrt(1 + t * 0.01))

        if t % 200 == 0:
            s_bin = round_to_binary(h, n)
            c5_bin = compute_c5(s_bin)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = s_bin.copy()

    return best_h, best_c5

def multi_tau_fw(h0, n, t_budget, seed):

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    t_start = time.time()
    per_tau = t_budget / len(tau_list)

        if time.time() - t_start > t_budget * 0.95:

        t0 = time.time()

            m_val = corr - corr.max()
            weights = np.exp(m_val / tau)

            g = np.zeros(n)
            active = np.where(weights > 1e-6)[0]

                g += weights[k_idx] * grad_corr_at(h_cur, n, m_vals[k_idx])

            s = fw_binary_oracle(g, n)
            alpha = 2.0 / (t + 2.0)

            h_new = np.clip(h_new, 0.0, 1.0)

                best_h = h_new.copy()

def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_old)
    h_interp = fix_sum(h_interp, n_new)
    return h_interp


def periodic_binary(n, period, duty):
    n_half = n // 2
    h = np.zeros(n)
    cyc = list(period.keys())
    seq = []
    for p in cyc:
        for _ in range(p):
            seq.append(1)
    for p in cyc:
        for _ in range(p):
            seq.append(0)
    for i in range(n):
        h[i] = seq[i % len(seq)]
    sorted_idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[sorted_idx[:n_half]] = 1.0
    return b

                best_c5 = fw_hi[1]
                best_h = fw_hi[0].copy()

    # Phase 6: Final polish
    if elapsed() < budget_s * 0.96:
        n_fin = len(best_h)
        h_fin = best_h.copy()
        # Small projected gradient with very small step
        m_vals = np.arange(2 * n_fin - 1) - (n_fin - 1)
        alpha = 0.01
        momentum = np.zeros(n_fin)
        t0_pg = time.time()
        while time.time() - t0_pg < budget_s * 0.04:
            corr = all_corr_values(h_fin)
            m_scaled = (corr - corr.max()) / 0.01
            weights = np.exp(m_scaled)
            weights /= weights.sum()
            g = np.zeros(n_fin)
            active = np.where(weights > 1e-5)[0]
            for k_idx in active:
                g += weights[k_idx] * grad_corr_at(h_fin, n_fin, m_vals[k_idx])
            momentum = 0.95 * momentum + 0.05 * g
            h_raw = h_fin - alpha * momentum
            h_new = project_feasible(h_raw, n_fin)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_fin = h_new
            alpha *= 0.9999

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    if elapsed() > budget_s:
        pass

    return (best_h, best_c5, len(best_h))
