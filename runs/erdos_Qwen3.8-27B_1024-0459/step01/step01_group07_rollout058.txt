    target = n / 2.0
    h = np.array(h, dtype=float).copy()

        s = np.sum(h)

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0
            if not np.any(free):
                break
            h[free] = np.minimum(h[free] + diff / np.count_nonzero(free), 1.0)
        else:
            free = h > 0.0
            if not np.any(free):

            h[free] = np.maximum(h[free] + diff / np.count_nonzero(free), 0.0)

    """Compute smooth max of all lags plus constraint penalty, and its gradient."""
    lags = compute_lags(h, n)
    max_lag = lags.max()
    # Stable log-sum-exp
    shifted = (lags - max_lag) / alpha
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth_val = max_lag + alpha * np.log(sum_exp)
    
    # Weights for each lag
    weights = exp_shifted / sum_exp  # these sum to 1
    
    # Gradient of smooth max:
    # ∂smooth/∂h[j] = sum_m weights[m] * ∂(lag_m)/∂h[j]
    # ∂(lag_m)/∂h[j] = dx * [(1-h[j+m])*I(0<=j+m<n) - h[j-m]*I(0<=j-m<n)]
    # where m = k - (n-1) for k-th element of correlate output
    
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # Vectorized: for each significant lag, add weighted gradient
    # Only process lags with weight > threshold
    active = weights > 1e-10
    for k_idx in np.where(active)[0]:
        m = k_idx - (n - 1)  # shift
        w = weights[k_idx]
        # grad[j] += w * dx * [(1-h[j+m]) if 0<=j+m<n else 0 - h[j-m] if 0<=j-m<n else 0]
        if m >= 0:

            valid_first = np.arange(0, n - m)  # j where j+m < n
            grad[valid_first] += w * dx * (1.0 - h[valid_first + m])
            # j-m valid when j >= m
            valid_second = np.arange(m, n)  # j where j-m >= 0
            grad[valid_second] -= w * dx * h[valid_second - m]
        else:

            # j+m = j-p valid when j >= p
            valid_first = np.arange(p, n)
            grad[valid_first] += w * dx * (1.0 - h[valid_first - p])
            # j-m = j+p valid when j <= n-1-p
            valid_second = np.arange(0, n - p)
            grad[valid_second] -= w * dx * h[valid_second + p]
    
    # Penalty gradient
    s = h.sum()
    target = n / 2.0
    smooth_val += penalty * (s - target) ** 2
    grad += 2.0 * penalty * (s - target)

def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(1000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] = np.minimum(h[mask] + diff/cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] = np.maximum(h[mask] + diff/cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha, pen):
    lags = compute_lags(h, n)
    mx = lags.max()
    sh = (lags - mx) / alpha
    e = np.exp(sh)
    se = e.sum()
    val = mx + alpha * np.log(se)
    w = e / se
    dx = 2.0 / n
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for ki in active:
        m = int(ki) - (n - 1)
        wi = w[ki]
        if m >= 0:
            if n - m > 0:
                grad[:n-m] += wi * dx * (1.0 - h[m:])
            if m < n:
                grad[m:] -= wi * dx * h[:n-m]
        else:

            if p < n:
                grad[p:] += wi * dx * (1.0 - h[:n-p])
            if n - p > 0:
                grad[:n-p] -= wi * dx * h[p:]
    s = h.sum()
    tgt = n / 2.0
    val += pen * (s - tgt) ** 2
    grad += 2.0 * pen * (s - tgt)
    return val, grad

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)


def compute_lags(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv * dx


def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(2000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)


def smooth_obj_grad(h, n, alpha, pen):
    lags = compute_lags(h, n)
    mx = lags.max()
    sh = (lags - mx) / alpha
    e = np.exp(sh)
    se = e.sum()
    val = mx + alpha * np.log(se)
    w = e / se
    dx = 2.0 / n
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for ki in active:
        m = int(ki) - (n - 1)
        wi = w[ki]
        if m >= 0:
            if n - m > 0:
                grad[:n - m] += wi * dx * (1.0 - h[m:])
            if m < n:
                grad[m:] -= wi * dx * h[:n - m]
        else:
            p = -m
            if p < n:
                grad[p:] += wi * dx * (1.0 - h[:n - p])
            if n - p > 0:
                grad[:n - p] -= wi * dx * h[p:]
    s = h.sum()
    tgt = n / 2.0
    val += pen * (s - tgt) ** 2
    grad += 2.0 * pen * (s - tgt)
    return val, grad


def optimize_lbfgsb(h0, n, alpha, pen, maxiter=2000):
    def f(x):
        return smooth_obj_grad(x, n, alpha, pen)[0]

    def g(x):
        return smooth_obj_grad(x, n, alpha, pen)[1]
    res = minimize(f, h0, jac=g, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-9, 'maxfun': 8000})
    return project_feasible(res.x, n)


def hill_climb(h, n, n_rounds=3):
    h = h.copy()
    val = compute_c5(h, n)
    for _ in range(n_rounds):
        improved = False
        for j in range(n):
            for delta in [0.02, -0.02, 0.05, -0.05, 0.1, -0.1]:
                h_new = h.copy()
                h_new[j] = np.clip(h_new[j] + delta, 0.0, 1.0)
                h_new = project_feasible(h_new, n)
                v_new = compute_c5(h_new, n)
                if v_new < val - 1e-12:
                    h = h_new
                    val = v_new
                    improved = True
                    break
        if not improved:
            break
    return h, val
