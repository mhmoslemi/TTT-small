                grad[n-m_val:n] += wk * (-h[0:m_val])
    return grad * dx

def adam_optimize(h, n, beta, max_iter, lr=0.01, tol=1e-12):
    """Adam optimizer with projection for the soft-max objective."""
    h = h.copy()
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_c5 = compute_c5(h, n)
    h_best = h.copy()
    prev_val = soft_max_val(h, n, beta)
    
    for it in range(max_iter):
        g = soft_max_grad(h, n, beta)
        # Add gradient for sum constraint (soft)
        sum_diff = h.sum() - n / 2.0
        g += 0.1 * sum_diff
        
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * (g * g)
        m_hat = m / (1 - b1 ** (it + 1))
        v_hat = v / (1 - b2 ** (it + 1))
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        
        val = soft_max_val(h_new, n, beta)
        if val < prev_val - tol:
            h = h_new
            prev_val = val
            c5 = compute_c5(h, n)
            if c5 < best_c5:
                best_c5 = c5
                h_best = h.copy()
        else:
            # Try smaller step
            h_new2 = h - lr * 0.5 * m_hat / (np.sqrt(v_hat) + eps)
            h_new2 = project_feasible(h_new2, n)
            val2 = soft_max_val(h_new2, n, beta)
            if val2 < prev_val:
                h = h_new2
                lr *= 0.5
                val = val2
            else:
                lr *= 0.5
                if lr < 1e-8:
                    break
    
    return h_best, best_c5

def lbfgs_polish(h, n, beta):
    """L-BFGS-B polish with soft max objective."""
    def obj_func(hv):
        return soft_max_val(hv, n, beta)
    
    def obj_grad(hv):
        return soft_max_grad(hv, n, beta)
    
    bounds = [(0.0, 1.0)] * n

            jac=obj_grad,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 2000, 'ftol': 1e-14, 'gtol': 1e-8, 'maxfun': 5000}
        )
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h

def compute_correlation(h, n):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h, n):
    c = compute_correlation(h, n)
    return float(np.max(c) * (2.0 / n))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = h.sum() - n / 2.0
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h > 1e-15
        else:
            mask = h < 1.0 - 1e-15
        if mask.sum() == 0:
            break
        h[mask] -= diff / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - n / 2.0
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] - diff, 0.0, 1.0)
        diff2 = h.sum() - n / 2.0
        if abs(diff2) > 1e-15:
            mask = (h > 1e-15) if diff2 > 0 else (h < 1.0 - 1e-15)
            if mask.sum() > 0:
                h[mask] -= diff2 / mask.sum()
                h = np.clip(h, 0.0, 1.0)
    return h


def soft_max_val(h, n, beta):
    c = compute_correlation(h, n)
    scaled = beta * c
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / beta * (2.0 / n)


def soft_max_grad(h, n, beta):
    c = compute_correlation(h, n)
    scaled = beta * c
    m = np.max(scaled)
    w = np.exp(scaled - m)
    wsum = w.sum()
    w = w / wsum
    dx = 2.0 / n
    grad = np.zeros(n)
    threshold = 1e-12
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < threshold:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            if lag == 0:
                grad += wk * (1.0 - 2.0 * h)
            else:
                grad[0:lag] += wk * (-h[lag:n])
                if lag <= n - 1 - lag:
                    s = lag
                    e = n - lag
                    grad[s:e] += wk * (1.0 - h[0:e-s] - h[s+lag:e+lag])
                grad[n-lag:n] += wk * (1.0 - h[0:lag])
        else:
            m_val = -lag
            if m_val == 0:
                grad += wk * (1.0 - 2.0 * h)
            else:
                grad[0:m_val] += wk * (1.0 - h[m_val:m_val+m_val])
                if m_val <= n - 1 - m_val:
                    s = m_val
                    e = n - m_val
                    grad[s:e] += wk * (1.0 - h[s+m_val:e+m_val] - h[s-m_val:e-m_val])
                grad[n-m_val:n] += wk * (-h[0:m_val])
    return grad * dx

        g += constraint_weight * sum_diff

        if n_struct == best_n:
            x = np.linspace(0, 2, n_struct, endpoint=False)
            # Periodic 3-region pattern
            h_s1 = 0.5 + 0.5 * np.sign(np.sin(np.pi * x / (2.0/3.0)))
            h_s1 = np.clip(h_s1 + 0.01 * rng.randn(n_struct), 0, 1)
            structural_inits.append(('block3', h_s1, n_struct))

            # Better: use tanh for smooth boundaries
            h_s2 = 0.5 + 0.5 * np.tanh(3.0 * np.sin(2 * np.pi * x))
            structural_inits.append(('sin2', h_s2, n_struct))
        else:
            x = np.linspace(0, 2, n_struct, endpoint=False)
            h_s = 0.5 + 0.5 * np.tanh(4.0 * np.sin(2 * np.pi * x * 1.5))
            structural_inits.append(('sin15', h_s, n_struct))

    for name, h_struct, n_struct in structural_inits:
        if time.time() - t0 > budget_s - 80:
            break
        h_struct = project_feasible(h_struct, n_struct)
        c5_s = compute_c5(h_struct, n_struct)

        if c5_s < best_c5 * 1.1:  # Only pursue if reasonably good
            h_opt, c5_opt = adam_optimize(h_struct, n_struct, 20.0, max_iter=1000, lr=0.03)
            if c5_opt < best_c5:
                best_h = h_opt
                best_n = n_struct
                best_c5 = c5_opt
            if time.time() - t0 < budget_s - 40:
                h_pol = lbfgs_polish(best_h, best_n, beta=100.0, max_iter=2000)
                c5_p = compute_c5(h_pol, best_n)
                if c5_p < best_c5:
                    best_h = h_pol
                    best_c5 = c5_p

    # === Phase 4: Perturbation restarts ===
    remaining = budget_s - (time.time() - t0)
    n_restarts = int(min(200, remaining / 3.0))
    for i in range(n_restarts):
        if time.time() - t0 > budget_s - 20:
            break
        scale = 0.015 + 0.03 * (i / max(1, n_restarts))
        h_pert = best_h + rng.randn(best_n) * scale
        h_pert = project_feasible(h_pert, best_n)

        beta_r = [20.0, 100.0][i % 2]
        h_opt, c5_opt = adam_optimize(h_pert, best_n, beta_r, max_iter=600, lr=0.02)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # === Phase 5: Final aggressive refinement ===
    if time.time() - t0 < budget_s - 15:
        for beta in [200.0, 1000.0]:
            if time.time() - t0 > budget_s - 8:
                break
            h_opt, c5_opt = adam_optimize(best_h, best_n, beta, max_iter=2000, lr=0.01)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt

    # L-BFGS final polish
    if time.time() - t0 < budget_s - 5:
        h_pol = lbfgs_polish(best_h, best_n, beta=500.0, max_iter=2000)
        c5_p = compute_c5(h_pol, best_n)
        if c5_p < best_c5:
            best_h = h_pol
            best_c5 = c5_p

    # Coordinate descent final
    if time.time() - t0 < budget_s - 3:
        h_cd, c5_cd = coordinate_descent_polish(best_h, best_n, max_rounds=5)
        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    # Validate with evaluator if available
    try:
        val = evaluate_erdos_solution(best_h, best_n)
        if val is not None and 0.0 < val < 1.0:
            best_c5 = min(best_c5, float(val))
    except Exception:
        pass

    return (best_h, best_c5, best_n)
