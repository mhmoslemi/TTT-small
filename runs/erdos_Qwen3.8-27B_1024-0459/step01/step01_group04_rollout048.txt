def _compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0 or not np.all(np.isfinite(h)):
        return 1e6
    h = np.clip(h, 0.0, 1.0)
    conv = np.correlate(h, 1.0 - h, mode="full")
    c = float(np.max(conv) * (2.0 / n))
    return c if math.isfinite(c) else 1e6

def _as_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    if h.size != n:
        if h.size > n:
            h = h[:n]
        else:
            tmp = np.zeros(n)
            m = min(n, h.size)
            if m > 0:
                tmp[:m] = h[:m]
            h = tmp
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(h + mid, 0.0, 1.0))) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.any():
            per = diff / float(mask.sum())
            h[mask] = np.clip(h[mask] + per, 0.0, 1.0)
    return h

def _scores(h, n):
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, h, mode="full")
    ac = corr[n-1:n-1+n]
    prefix = np.zeros(n+1)
    np.cumsum(h, out=prefix[1:])
    total = prefix[n]
    scores = np.empty(2*n-1)
    scores[0] = total - ac[0]
    for d in range(1, n):
        scores[2*d-1] = prefix[n-d] - ac[d]
        scores[2*d] = total - prefix[d] - ac[d]
    return scores

def _obj(x, alpha):
    _check()
    n = x.size
    if n == 0:
        return 1e6
    h = np.clip(x, 0.0, 1.0)
    scores = _scores(h, n)
    m = float(scores.max())
    if alpha > 1e-14:
        shifted = scores - m
        w = np.exp(shifted / alpha)
        A = float(w.sum())
        if A > 0 and math.isfinite(A):
            val = m + alpha * math.log(A)
        else:
            val = m
    else:
        val = m
    return float(val) * (2.0 / n)

def _grad(x, alpha):
    _check()
    n = x.size
    if n == 0:
        return np.zeros(0)
    h = np.clip(x, 0.0, 1.0)
    scores = _scores(h, n)
    m = float(scores.max())
    if alpha > 1e-14:
        shifted = scores - m
        w = np.exp(shifted / alpha)
        A = float(w.sum())
        if A > 0 and math.isfinite(A):
            w = w / A
        else:
            idx = int(np.argmax(scores))
            w = np.zeros(2*n-1)
            w[idx] = 1.0
    else:
        tol = max(1e-10, 1e-9 * max(1.0, m))
        active = np.flatnonzero(scores >= m - tol)
        w = np.zeros(2*n-1)
        if active.size > 0:
            w[active] = 1.0 / active.size
        else:
            w[int(np.argmax(scores))] = 1.0

    grad = np.zeros(n)
    # shift 0
    grad += w[0] * (1.0 - 2.0 * h)
    # shifts d = 1..n-1
    for d in range(1, n):
        cp = w[2*d-1]
        cn = w[2*d]
        m2 = n - d
        if cp > 1e-14:
            grad[:m2] += cp * (1.0 - h[d:])
            grad[d:] += cp * (-h[:m2])
        if cn > 1e-14:
            grad[d:] += cn * (1.0 - h[:m2])
            grad[:m2] += cn * (-h[d:])
    grad *= 2.0 / float(n)
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n)
    return grad

def _con_eq(x):
    n = max(1, x.size)
    return float(np.sum(x)) - 0.5 * float(n)

def _con_jac(x):
    n = max(1, x.size)
    return np.ones(n)

def _upsample(parent, n):
    n0 = int(parent.size)
    if n == n0:
        return parent.copy()
    if n0 <= 0:
        return np.full(n, 0.5)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x, x0, parent.astype(float))
