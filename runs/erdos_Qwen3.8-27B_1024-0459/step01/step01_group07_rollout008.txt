import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(conv)

def compute_active_grad(h, n_points):
    """Compute gradient of the active (max) correlation term."""
    n = n_points
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(conv))
    m = k_max - (n - 1)  # lag
    grad = np.zeros(n)
    p = abs(m)
    if p == 0:
        # h[i] * (1-h[i]) = h[i] - h[i]^2
        grad[:] = (1.0 - 2.0 * h) * dx
    else:
        # For positive lag p: sum_{i=0}^{n-1-p} h[i+p]*(1-h[i])
        # d/dh[j] = (1-h[j-p])*[p<=j<=n-1] - h[j+p]*[0<=j<=n-1-p]
        grad[p:] += (1.0 - h[:n-p]) * dx
        grad[:n-p] += (-h[p:]) * dx
    return grad

def project_feasible(h, n_points):
    """Project to {0<=h<=1, sum(h)=n/2} using binary search on offset."""
    n = n_points
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    # Binary search for offset t such that sum(clip(h-t, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(h - t, 0.0, 1.0)
    # Fine-tune
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, n_points, temp=0.02):
    """Smooth approximation of max using log-sum-exp."""
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = np.max(conv)
    return temp * np.log(np.sum(np.exp((conv - mx) / temp))) + mx

def smooth_max_grad(h, n_points, temp=0.02):
    """Gradient of smooth max approximation."""
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = np.max(conv)
    weights = np.exp((conv - mx) / temp) / np.sum(np.exp((conv - mx) / temp))
    # Gradient is weighted sum of individual correlations' gradients
    n = n_points
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if weights[k_idx] < 1e-10:
            continue
        m = k_idx - (n - 1)
        p = abs(m)
        if p == 0:
            grad[:] += weights[k_idx] * (1.0 - 2.0 * h) * dx
        else:
            grad[p:] += weights[k_idx] * (1.0 - h[:n-p]) * dx
            grad[:n-p] += weights[k_idx] * (-h[p:]) * dx
    return grad

This is a problem in additive combinatorics / harmonic analysis. The "best" sets for this are related to Bohr sets or structured sets.

For the continuous case, if A = [0,1], then |A ∩ (A+k)| = max(0, 1-k) for k ∈ [0,1], and 0 for k > 1.
So max_k (1 - |A∩(A+k)|) = max_k (1 - max(0,1-k)) = max(max(0, k), 1) = 1 for k ≥ 1.
And for k ∈ [0,1]: 1 - (1-k) = k. Max at k=1 gives 1.
So C5 = 1 for A = [0,1]. That's terrible.

What about A = [0, 0.5] ∪ [1, 1.5]? |A| = 1.
For k = 0.5: A ∩ (A+0.5) = [0,0.5]∩[0.5,1] ∪ [1,1.5]∩[1.5,2] = ∅. So |A∩(A+0.5)| = 0, giving C5 contribution of 1.

def compute_c5(h, n):
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def compute_grad_active(h, n):
    """Exact gradient of the argmax correlation term."""
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(conv))
    p = abs(k_max - (n - 1))
    grad = np.zeros(n)
    if p == 0:
        grad[:] = (1.0 - 2.0*h) * dx
    else:
        grad[p:] += (1.0 - h[:n-p]) * dx
        grad[:n-p] += (-h[p:]) * dx
    return grad

def compute_grad_smooth(h, n, temp):
    """Gradient of log-sum-exp smoothed max."""
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = np.max(conv)
    w = np.exp((conv - mx) / temp)
    w /= np.sum(w)
    n_terms = 2*n - 1
    grad = np.zeros(n)
    for ki in range(n_terms):
        if w[ki] < 1e-12:
            continue
        p = abs(ki - (n-1))
        if p == 0:
            grad[:] += w[ki] * (1.0 - 2.0*h) * dx
        else:
            grad[p:] += w[ki] * (1.0 - h[:n-p]) * dx
            grad[:n-p] += w[ki] * (-h[p:]) * dx
    return grad

    v = np.zeros(n)

    best_val = compute_c5(h, n)
    temp = temp_init
    for t in range(1, iters+1):
        if time.time() - t_check[0] > t_check[1]:
            break
        temp_t = temp_init * (temp_end/temp_init) ** (t/iters)
        g = compute_grad_smooth(h, n, temp_t)
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*g*g
        mh = m / (1 - beta1**t)
        vh = v / (1 - beta2**t)
        h = project_feasible(h - lr * mh / (np.sqrt(vh) + 1e-8), n)
        val = compute_c5(h, n)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    return best_h, best_val

def binary_hill_climb(h, n, max_rounds=200, t_check=None):
    # Round to binary
    target = int(n // 2)
    h_b = np.zeros(n)
    idx = np.argsort(-h)
    h_b[idx[:target]] = 1.0
    best_val = compute_c5(h_b, n)
    best_h = h_b.copy()
    for r in range(max_rounds):
        if t_check is not None and time.time() - t_check[0] > t_check[1]:
            break
        improved = False
        for j in range(n):
            h_b[j] = 1.0 - h_b[j]
            val = compute_c5(h_b, n)
            if val < best_val:
                best_val = val
                best_h = h_b.copy()
                improved = True
            else:
                h_b[j] = 1.0 - h_b[j]
        if not improved:
            break
    return best_h, best_val

def slsqp_polish(h0, n, maxiter=500, t_check=None):
