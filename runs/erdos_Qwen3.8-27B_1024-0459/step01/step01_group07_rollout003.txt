```
def compute_values(h, n):
    h = np.asarray(h, dtype=float)
    if n == 1:
       dx=2.0
       return np.array([dx*(h[0] - h[0]*h[0])])
    corr = np.correlate(h, h, mode='full')
    r = corr[n-1:2*n-1]
    pref = np.empty(n+1, dtype=float)
    pref[0]=0.0
    np.cumsum(h, out=pref[1:])
    dx=2.0/float(n)
    vals = np.empty(2*n-1, dtype=float)
    vals[0] = dx*(pref[n] - r[0])
    if n > 1:
       m = np.arange(1, n, dtype=int)
       pos = dx*(pref[n - m] - r[1:])
       neg = dx*(pref[n] - pref[m] - r[1:])
       vals[1::2] = pos
       vals[2::2] = neg
    return vals
```

```python
import numpy as np
import time
from scipy.optimize import minimize

_N = 1
_TAU = 0.03
_ACTIVE_EPS = 1e-6


def project_feasible(y, n):
    n = int(n)
    if n <= 0:
        return np.array([], dtype=float)
    y = np.asarray(y, dtype=float)
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * float(n)

    mn = float(np.min(y))
    mx = float(np.max(y))
    lo = -mx - 1.0
    hi = 1.0 - mn + 1.0

    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(y + lam, 0.0, 1.0)
    h = exact_sum_fix(h, n)
    return h


def exact_sum_fix(h, n):
    n = int(n)
    target = 0.5 * float(n)
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

    for _ in range(25):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            w = 1.0 - h
            s = float(np.sum(w))
            if s <= 1e-15:
                break
            h += (diff / s) * w
        else:
            w = h
            s = float(np.sum(w))
            if s <= 1e-15:
                break
            h += (diff / s) * w
        h = np.clip(h, 0.0, 1.0)

    diff = target - float(np.sum(h))
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(-(1.0 - h))
        else:
            order = np.argsort(h)
        rem = diff
        for idx in order:
            if rem > 1e-15:
                add = min(rem, 1.0 - h[idx])
                if add <= 1e-18:
                    continue
                h[idx] += add
                rem -= add
            elif rem < -1e-15:
                add = max(rem, -h[idx])
                if add >= -1e-18:
                    continue
                h[idx] += add
                rem -= add
            if abs(rem) < 1e-15:
                break
    return h


def compute_values(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float)
    if n == 1:
        dx = 2.0
        return np.array([dx * (h[0] - h[0] * h[0])])

    corr = np.correlate(h, h, mode="full")
    r = corr[n - 1:2 * n - 1]

    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])

    dx = 2.0 / float(n)
    vals = np.empty(2 * n - 1, dtype=float)
    vals[0] = dx * (pref[n] - r[0])

    if n > 1:
        m = np.arange(1, n, dtype=int)
        pos = dx * (pref[n - m] - r[1:])
        neg = dx * (pref[n] - pref[m] - r[1:])
        vals[1::2] = pos
        vals[2::2] = neg
    return vals


def evaluate_exact(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def weighted_gradient(h, n, weights):
    n = int(n)
    dx = 2.0 / float(n)
    grad = np.zeros(n, dtype=float)
    w0 = weights[0]
    if w0 != 0.0:
        grad += w0 * dx * (1.0 - 2.0 * h)

    for m in range(1, n):
        ip = 1 + 2 * (m - 1)
        a = weights[ip]
        b = weights[ip + 1]
        if abs(a) < 1e-16 and abs(b) < 1e-16:
            continue
        c = a + b
        L = n - m
        if a != 0.0:
            grad[:L] += a * dx
        if b != 0.0:
            grad[m:] += b * dx
        if c != 0.0:
            grad[:L] -= c * dx * h[m:]
            grad[m:] -= c * dx * h[:L]
    return grad


def lse_val(h, n, tau):
    n = int(n)
    vals = compute_values(h, n)
    zmax = float(np.max(vals))
    x = (vals - zmax) / tau
    x = np.clip(x, -700.0, 700.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    F = zmax + tau * float(np.log(s))
    p = ex / s
    g = weighted_gradient(h, n, p)
    return F, g


def area_fun(x):
    return float(np.sum(x) - 0.5 * float(_N))


def area_jac(x):
    return np.ones(_N, dtype=float)


def lse_obj_wrapper(x):
    return lse_val(x, _N, _TAU)[0]


def lse_jac_wrapper(x):
    return lse_val(x, _N, _TAU)[1]


def exact_obj_wrapper(x):
    vals = compute_values(x, _N)
    return float(np.max(vals))


def active_jac_wrapper(x):
    vals = compute_values(x, _N)
    zm = float(np.max(vals))
    active = vals >= (zm - _ACTIVE_EPS)
    if not np.any(active):
        active = np.zeros(vals.shape, dtype=bool)
        active[int(np.argmax(vals))] = True
    w = np.zeros(vals.shape, dtype=float)
    w[active] = 1.0 / float(np.sum(active))
    return weighted_gradient(x, _N, w)


def slsqp_smooth(h0, n, tau, maxiter, ftol):

    _TAU = float(tau)
    h0 = project_feasible(h0, n)

    lr_max = 0.8

        it += 1
        gnorm = float(np.linalg.norm(g))

                F = F2
                v = v_next
                lr = min(lr_max, lr * 1.05)
                accepted = True
                if F < F_best - 1e-12:
                    F_best = F
                    no_improve = 0
                else:

            alpha *= 0.5

        if not accepted:
            lr *= 0.5
            v *= 0.5

            break

        if it % 40 == 0:
            c = evaluate_exact(h, n)
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()

    c = evaluate_exact(h, n)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c

def binary_round(h, n):

def downsample_h(h, n_from, n_to):
    h = np.asarray(h, dtype=float)
    n_from = int(n_from)
    n_to = int(n_to)
    if n_to <= 0:
        return h
    if n_to == n_from:
        return h.copy()

    if n_to < n_from:
        edges = np.linspace(0, n_from, n_to + 1).astype(int)
        out = np.empty(n_to, dtype=float)
        for i in range(n_to):
            a = int(edges[i])
            b = int(edges[i + 1])
            if b > a:
                out[i] = float(np.mean(h[a:b]))
            else:
                out[i] = float(h[a])
        return project_feasible(out, n_to)

    if n_from > 1 and n_to % n_from == 0:
        return np.repeat(h, n_to // n_from)
    idx = np.floor(np.linspace(0, 1, n_to, endpoint=False) * n_from).astype(int)
    idx = np.clip(idx, 0, n_from - 1)
    return project_feasible(h[idx], n_to)


def upsample_h(h, n_from, n_to):

def perturb_h(h, rng, scale):

    for res in candidates:
        if res[1] < best_c - 1e-12:
            best_h, best_c, best_n = res

    polished_ns = []
    polished_count = 0
    for res in candidates:
        if time.time() > hard_stop - 60:
            break
        if polished_count >= 7:
            break
        n = int(res[2])
        if n < 8 or n > 320 or n in polished_ns:
            continue
        polished_ns.append(n)
        polished_count += 1

        if polished_count == 1:
            tmax = max(120.0, 0.50 * budget_s)
        elif polished_count == 2:
            tmax = max(90.0, 0.30 * budget_s)
        elif polished_count == 3:
            tmax = max(70.0, 0.20 * budget_s)
        else:
            tmax = max(50.0, 0.10 * budget_s)

        stop = min(hard_stop, time.time() + tmax)
        h_try, c_try = polish_candidate(res[0], n, stop)
        if c_try < best_c - 1e-12:
            best_h, best_c, best_n = h_try, float(c_try), n

    if best_h is not None and time.time() < hard_stop - 120:
        for factor in [2, 3, 4]:
            if time.time() > hard_stop - 120:
                break
            nf = best_n * factor
            if nf % 2:
                nf += 1
            if nf < 8 or nf > 320:
                continue
            h_f = upsample_h(best_h, best_n, nf)
            res = make_candidate(h_f, nf)
            if res is None:
                continue
            if res[1] < best_c - 1e-12:
                best_h, best_c, best_n = res
            stop = min(hard_stop, time.time() + max(70.0, 0.12 * budget_s))
            h2, c2 = polish_candidate(best_h, best_n, stop)
            if c2 < best_c - 1e-12:
                best_h, best_c, best_n = h2, float(c2), best_n

    if best_h is not None and best_n > 40 and time.time() < hard_stop - 120:
        for den in [2, 3]:
            if time.time() > hard_stop - 90:
                break
            nd = best_n // den
            if nd % 2:
                nd -= 1
            if nd < 8 or nd > 320:
                continue
            h_d = downsample_h(best_h, best_n, nd)
            res = make_candidate(h_d, nd)
            if res is None:
                continue
            if res[1] < best_c - 1e-12:
                best_h, best_c, best_n = res
            stop = min(hard_stop, time.time() + max(50.0, 0.08 * budget_s))
            h2, c2 = polish_candidate(h_d, nd, stop)
            if c2 < best_c - 1e-12:
                best_h, best_c, best_n = h2, float(c2), nd

    if best_h is not None and time.time() < hard_stop - 120:
        bin_ns = []
        if best_n % 2 == 0 and 8 <= best_n <= 160:
            bin_ns.append(best_n)
        for bn in [70, 100, 140]:
            if bn % 2:
                bn -= 1
            if 8 <= bn <= 160 and bn not in bin_ns:
                bin_ns.append(bn)
        for bn in bin_ns:
            if time.time() > hard_stop - 90:
                break
            hb = binary_round(downsample_h(best_h, best_n, bn), bn)
            h2, c2 = binary_local_search(hb, bn, min(hard_stop, time.time() + 120))
            if c2 < best_c - 1e-12:
                best_h, best_c, best_n = h2, float(c2), int(bn)

    if best_h is not None and time.time() < hard_stop - 120:
        pn = best_n if 10 <= best_n <= 160 else 100
        if pn % 2:
            pn -= 1
        if pn < 10:
            pn = 70
        for scale in [0.08, 0.15, 0.25]:
            if time.time() > hard_stop - 80:
                break
            base_p = downsample_h(best_h, best_n, pn)
            h_p = perturb_h(base_p, rng, scale)
            stop = min(hard_stop, time.time() + 120)
            h2, c2 = polish_candidate(h_p, pn, stop)
            if c2 < best_c - 1e-12:
                best_h, best_c, best_n = h2, float(c2), int(pn)

    best_h = project_feasible(best_h, int(best_n))
    best_c = evaluate_exact(best_h, int(best_n))
    if not np.isfinite(best_c):
        best_h = project_feasible(np.full(70, 0.5), 70)
        best_c = evaluate_exact(best_h, 70)
        best_n = 70
    if best_c < 0.0 and best_c > -1e-12:
        best_c = 0.0

    return best_h, float(best_c), int(best_n)
```
