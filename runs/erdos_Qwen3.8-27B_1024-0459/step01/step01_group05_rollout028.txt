import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    """Compute exact C5 value."""
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    """Compute all correlation values."""
    n = len(h)
    comp = 1.0 - h
    return np.correlate(h, comp, mode='full')

def smooth_max_obj(h, beta):
    """Smooth max of correlation values."""
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_grad(h, beta):
    """Analytical gradient of smooth max of correlation values."""
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)

    # where lag = k - (n-1)

            continue
        w = weights[k]

            js = np.arange(j_min, j_max + 1)
            grad[js] += w * (1.0 - h[js - lag])
        
        # Term 2: -h[j+lag] for j+lag in S_lag
        # j+lag in S_lag means: max(0,lag) <= j+lag <= min(n-1, n-1+lag)
        # j >= max(0,lag) - lag and j <= min(n-1, n-1+lag) - lag
        # Also need 0 <= j < n (always true) and 0 <= j+lag < n
        
        if lag >= 0:
            # j >= 0 - lag... wait, max(0,lag)=lag, so j+lag >= lag → j >= 0
            # j+lag <= n-1+lag → j <= n-1

            # So j <= n-1-lag

            j2_max = n - 1
        
        if j2_min <= j2_max:
            js2 = np.arange(j2_min, j2_max + 1)

def project_sum(h, target):
    """Project h onto {sum(h)=target, 0<=h<=1}."""
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            free = np.where(h < 1.0)[0]
            if len(free) > 0:
                per = diff / len(free)
                h[free] = np.clip(h[free] + per, 0.0, 1.0)
        else:
            free = np.where(h > 0.0)[0]
            if len(free) > 0:
                per = diff / len(free)
                h[free] = np.clip(h[free] + per, 0.0, 1.0)
    return h

def sine_init(n):
    """Create sinusoidal initialization."""
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.3 * np.sin(np.pi * x)
    h = project_sum(h, n / 2.0)
    return h

def run_optimize_analytic(h_start, n, target, beta, max_iter, ftol, time_limit):
    """Run SLSQP with analytical gradient."""
    t_start = time.time()
    h0 = project_sum(h_start.copy(), target)
    
    def obj(h):
        return smooth_max_obj(np.clip(h, 0, 1), beta)
    
    def grad(h):
        return smooth_max_grad(np.clip(h, 0, 1), beta)
    
    def cons_fun(h):
        return np.sum(h) - target
    
    def cons_jac(h):
        return np.ones(n)
    
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    
    result = None
    try:
        result = minimize(obj, h0, jac=grad, method='SLSQP',
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': max_iter, 'ftol': ftol,
                                  'iprint': 0, 'eps': 1e-8})
    except Exception:
        pass
    
    elapsed = time.time() - t_start
    if result is not None and result.x is not None:
        return project_sum(result.x, target), elapsed
    return h0, elapsed
