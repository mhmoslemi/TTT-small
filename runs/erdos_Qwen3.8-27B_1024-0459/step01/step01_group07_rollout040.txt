def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

        diff = target - s
        if abs(diff) < 1e-12:

        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def binary_local_search(h, n, max_sweeps=100):
    """Greedy local search: flip pairs (1→0, 0→1) if it reduces C5."""
    h = np.array(h, dtype=float).copy()
    # Round to binary
    target = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    h = h_bin
    
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    best_val = np.max(conv) * dx
    
    for sweep in range(max_sweeps):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for a in ones:
            for b in zeros:
                # Compute delta for flipping a (1→0) and b (0→1)
                delta = compute_flip_delta(a, b, h, n)
                new_conv = conv + delta
                new_max = np.max(new_conv) * dx
                if new_max < best_val - 1e-12:
                    h[a] = 0.0
                    h[b] = 1.0
                    conv = new_conv
                    best_val = new_max
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_val

def compute_flip_delta(a, b, h, n):
    """Compute change in conv when flipping h[a] 1→0 and h[b] 0→1."""
    delta = np.zeros(2*n-1)
    offset_a = a
    offset_b = b
    # delta[k] = h[a+k-(n-1)] - h[b+k-(n-1)] + h[a-k+(n-1)] - h[b-k+(n-1)]

            d += h[idx3]
        idx4 = b - k + (n-1)
        if 0 <= idx4 < n:
            d -= h[idx4]
        delta[k] = d
    return delta

    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')

    weights = np.exp(shifted / T)
    p = weights / np.sum(weights)

    for k in range(n_conv):
        pk = p[k]
        if pk < 1e-15:

        offset = k - (n - 1)

        # j - offset in [0, n-1] => j in [offset, n-1+offset]
        j_start = max(0, offset)

        if j_start2 < j_end2:

import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)


def compute_conv(h, n):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_smooth_grad(h, n, T):
    dx = 2.0 / n
    conv = compute_conv(h, n)
    cmax = np.max(conv)
    shifted = (conv - cmax) / T
    weights = np.exp(shifted)
    p = weights / np.sum(weights)
    grad = np.zeros(n)
    n_conv = 2 * n - 1
    for k in range(n_conv):
        pk = p[k]
        if pk < 1e-15:
            continue
        offset = k - (n - 1)
        j_start = max(0, offset)
        j_end = min(n, n + offset)
        if j_start < j_end:
            grad[j_start:j_end] += pk * (1.0 - h[j_start - offset:j_end - offset])
        j_start2 = max(0, -offset)
        j_end2 = min(n, n - offset)
        if j_start2 < j_end2:
            grad[j_start2:j_end2] -= pk * h[j_start2 + offset:j_end2 + offset]
    grad *= dx
    return grad


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h


def to_binary(h, n):
    target = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin


def flip_delta(a, b, h, n):
    delta = np.zeros(2 * n - 1)
    k1s = max(0, n - 1 - a)
    k1e = min(2 * n - 1, 2 * n - 1 - a)
    if k1s < k1e:
        delta[k1s:k1e] += h[a + np.arange(k1s, k1e) - (n - 1)]
    k2s = max(0, n - 1 - b)
    k2e = min(2 * n - 1, 2 * n - 1 - b)
    if k2s < k2e:
        delta[k2s:k2e] -= h[b + np.arange(k2s, k2e) - (n - 1)]
    k3s = max(0, a)
    k3e = min(2 * n - 1, a + n)
    if k3s < k3e:
        delta[k3s:k3e] += h[a - np.arange(k3s, k3e) + (n - 1)]
    k4s = max(0, b)
    k4e = min(2 * n - 1, b + n)
    if k4s < k4e:
        delta[k4s:k4e] -= h[b - np.arange(k4s, k4e) + (n - 1)]
    return delta


def binary_local_search(h, n, max_sweeps=200, time_limit=None, t_start=None):
    h = to_binary(h, n).copy()
    dx = 2.0 / n
    conv = compute_conv(h, n)
    best_val = np.max(conv) * dx
    for sweep in range(max_sweeps):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_improve = 1e-15
        best_a, best_b = -1, -1
        best_new_conv = None
        for a in ones:
            for b in zeros:
                d = flip_delta(a, b, h, n)
                new_conv = conv + d
                new_max = np.max(new_conv) * dx
                if new_max < best_val - best_improve:
                    best_improve = best_val - new_max
                    best_a, best_b = a, b
                    best_new_conv = new_conv
        if best_a >= 0:
            h[best_a] = 0.0
            h[best_b] = 1.0
            conv = best_new_conv
            best_val = np.max(conv) * dx
            improved = True
        if not improved:
            break
    return h, best_val


def simulated_annealing(h, n, rng, t_start, time_limit, max_iters=500000,
                        T_init=0.02, T_min=0.001, cooling=0.9998):
    h = to_binary(h, n).copy()
    dx = 2.0 / n
    conv = compute_conv(h, n)
    best_val = np.max(conv) * dx
    best_h = h.copy()
    T = T_init
    current_val = best_val
    no_improve = 0
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        d = flip_delta(a, b, h, n)
        new_conv = conv + d
        new_val = np.max(new_conv) * dx
        delta_E = new_val - current_val
        if delta_E < 0 or rng.random() < np.exp(-delta_E / T):
            h[a] = 0.0
            h[b] = 1.0
            conv = new_conv
            current_val = new_val
            if current_val < best_val:
                best_val = current_val
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > 5000:
            T = max(T_min, T * 2.0)
            no_improve = 0
        T = max(T_min, T * cooling)
    return best_h, best_val, best_val


def smooth_gradient_optimize(h0, n, t_start, time_limit, T_vals=None,
                             max_iters=2000, lr_init=0.1):
    if T_vals is None:
        T_vals = [0.1, 0.05, 0.03, 0.015, 0.008, 0.004]
    h = project_feasible(h0, n)
    best_val = compute_c5(h, n)
    best_h = h.copy()
    for T in T_vals:
        if time.time() - t_start > time_limit:
            break
        lr = lr_init
        prev_val = compute_c5(h, n)
        for it in range(max_iters):
            if time.time() - t_start > time_limit:
                break
            grad = compute_smooth_grad(h, n, T)
            # Backtracking line search
            alpha = lr
            for _ in range(20):
                h_new = h - alpha * grad
                h_new = project_feasible(h_new, n)
                val_new = compute_c5(h_new, n)
                # Simple sufficient decrease
                if val_new < prev_val + 1e-10 or alpha < 1e-8:
                    break
                alpha *= 0.5
            if alpha < 1e-8:
                break
            h = h_new
            val = compute_c5(h, n)
            if val < best_val:
                best_val = val
                best_h = h.copy()
            prev_val = val
            lr = min(lr * 1.01, 0.5)
    return best_h, best_val

    t0 = time.time()

    def time_left():

    # Initialize from parent

        # Binary local search

                                                max_iters=300000, T_init=0.012,

                return compute_c5(x, best_n)
