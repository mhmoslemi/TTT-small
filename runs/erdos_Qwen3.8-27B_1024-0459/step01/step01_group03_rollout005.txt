def active_gradient(full,h,k,beta):
    n=h.size
    g=np.zeros(n,float)
    if n==0: return g
    if full.size==0: return g
    maxc=float(np.max(full))
    kk=min(k, full.size)
    idx=np.argsort(full)[-kk:]
    scale=2.0/n
    wraw=np.exp(beta*(full[idx]-maxc)*scale)
    s=wraw.sum()
    if s<=0: wraw=np.ones(kk)/kk
    else: wraw/=s

```
def project_density(x, target):
    x=np.asarray(x,dtype=float)
    n=x.size
    if n==0: return x
    x=np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    target=float(target)
    # quick if sum close and in bounds
    s=float(np.sum(x))
    if abs(s-target) <= 1e-13*max(1,n) and np.all(x>=0) and np.all(x<=1):
         return x
    lo=-float(np.max(x)) - 1.0
    hi=float(np.max(x)) + 1.0
    for _ in range(45):
       mid=0.5*(lo+hi)
       s=float(np.sum(np.clip(x+mid,0.0,1.0)))
       if s < target:
          lo=mid
       else:
          hi=mid
    h=np.clip(x+0.5*(lo+hi),0.0,1.0)
    # adjust
    for _ in range(5):
       res=target-float(np.sum(h))
       if abs(res) <= 1e-12: break
       free=np.where((h>1e-14)&(h<1.0-1e-14))[0]
       if free.size==0:
          free=np.where((h>0.0)&(h<1.0))[0]
       if free.size==0:
          break
       if res>0:
          idx=int(free[np.argmax(1.0-h[free])])
          add=min(res,1.0-h[idx])
          h[idx]+=add
       else:
          idx=int(free[np.argmax(h[free])])
          sub=min(-res,h[idx])
          h[idx]-=sub
    return h
```

```python
import time
import math
import numpy as np


def evaluate_exact_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * (2.0 / float(n)))


def is_feasible_h(h, target, tol=1e-8):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    if abs(float(np.sum(h)) - float(target)) > max(1e-8, tol * max(1, h.size)):
        return False
    return True


def project_density(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    if n == 0:
        return x
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    target = float(target)

    s = float(np.sum(x))
    if (abs(s - target) <= 1e-13 * max(1, n) and
            np.all(x >= 0.0) and np.all(x <= 1.0)):
        return x

    lo = -float(np.max(x)) - 1.0
    hi = float(np.max(x)) + 1.0

    for _ in range(45):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid

    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(6):
        res = target - float(np.sum(h))
        if abs(res) <= 1e-12:
            break
        free = np.where((h > 1e-14) & (h < 1.0 - 1e-14))[0]
        if free.size == 0:
            free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        if res > 0.0:
            idx = int(free[np.argmax(1.0 - h[free])])
            add = min(res, 1.0 - h[idx])
            h[idx] += add
        else:
            idx = int(free[np.argmax(h[free])])
            sub = min(-res, h[idx])
            h[idx] -= sub
    return h


def adjust_sum_light(h, target, tol=1e-10):
    h = np.asarray(h, dtype=float).copy()
    res = float(target) - float(np.sum(h))
    if abs(res) <= tol:
        return h
    free = np.where((h > 1e-14) & (h < 1.0 - 1e-14))[0]
    if free.size:
        if res > 0.0:
            idx = int(free[np.argmax(1.0 - h[free])])
            h[idx] += min(res, 1.0 - h[idx])
        else:
            idx = int(free[np.argmax(h[free])])
            h[idx] -= min(-res, h[idx])
    if abs(float(target) - float(np.sum(h))) > tol:
        h = project_density(h, target)
    return h


def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    n = int(n)
    if n0 == n:
        return p.copy()
    if n0 <= 1:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)


def random_composition(total, parts, rng):
    total = int(total)
    parts = int(parts)
    if total <= 0:
        return np.array([], dtype=int)
    parts = min(parts, total)
    if parts <= 0:
        return np.array([total], dtype=int)
    if parts == 1 or total <= 1:
        return np.array([total], dtype=int)
    cuts = np.sort(rng.choice(total - 1, size=parts - 1, replace=False))
    out = []
    prev = 0
    c = cuts
    for k in range(c.size):
        cur = int(c[k])
        out.append(cur - prev)
        prev = cur
    out.append(total - prev)
    return np.array(out, dtype=int)


def make_binary_blocks(n, rng, start_one=True):
    n = int(n)
    target = n // 2
    zeros = n - target
    if target <= 0 or zeros <= 0:
        return np.zeros(n, dtype=np.int8)

    maxR = min(target, zeros)
    minR = min(maxR, max(2, n // 20))
    if minR > maxR:
        minR = 1
    if maxR < 1:
        maxR = 1
    R = int(rng.integers(minR, maxR + 1))

    one_l = random_composition(target, R, rng)
    zero_l = random_composition(zeros, R, rng)

    b = np.empty(n, dtype=np.int8)
    pos = 0
    if start_one:
        for a, z in zip(one_l, zero_l):
            a = int(a)
            z = int(z)
            if a > 0:
                b[pos:pos + a] = 1
                pos += a
            if z > 0:
                b[pos:pos + z] = 0
                pos += z
            if pos >= n:
                break
    else:
        for z, a in zip(zero_l, one_l):
            a = int(a)
            z = int(z)
            if z > 0:
                b[pos:pos + z] = 0
                pos += z
            if a > 0:
                b[pos:pos + a] = 1
                pos += a
            if pos >= n:
                break

    if pos < n:
        b[pos:] = 0
    return b


def binarize_top(h, target_ones):
    h = np.asarray(h, dtype=float)
    n = h.size
    target = int(target_ones)
    if target <= 0:
        return np.zeros(n, dtype=np.int8)
    if target >= n:
        return np.ones(n, dtype=np.int8)
    idx = np.argsort(-h)
    b = np.zeros(n, dtype=np.int8)
    b[idx[:target]] = 1
    return b


def balance_binary(b, target_ones, rng=None):
    b = np.asarray(b, dtype=np.int8).copy()
    n = b.size
    target = int(target_ones)
    target = min(max(0, target), n)
    cur = int(np.sum(b))

    if cur > target:
        idx = np.where(b == 1)[0]
        need = cur - target
        if need > 0 and idx.size >= need:
            if rng is not None and need < idx.size:
                flip = idx[rng.choice(idx.size, size=need, replace=False)]
            else:
                flip = idx[:need]
            b[flip] = 0
    elif cur < target:
        idx = np.where(b == 0)[0]
        need = target - cur
        if need > 0 and idx.size >= need:
            if rng is not None and need < idx.size:
                flip = idx[rng.choice(idx.size, size=need, replace=False)]
            else:
                flip = idx[:need]
            b[flip] = 1
    return b


def make_binary_start(n, rng, mode, parent_h=None):
    n = int(n)
    target = n // 2
    b = None

    if mode == 0 and parent_h is not None and np.size(parent_h) > 0:
        hu = upsample_h(parent_h, n)
        b = binarize_top(hu, target)
    elif mode == 1:
        b = make_binary_blocks(n, rng, True)
    elif mode == 2:
        b = make_binary_blocks(n, rng, False)
    else:
        x = (np.arange(n) + 0.5) * (2.0 / float(n))
        f1 = int(rng.integers(1, 6))
        f2 = int(rng.integers(1, 6))
        p1 = float(rng.uniform(0.0, 2.0 * np.pi))
        p2 = float(rng.uniform(0.0, 2.0 * np.pi))
        a1 = float(rng.uniform(0.25, 0.55))
        a2 = float(rng.uniform(0.10, 0.35))
        vals = 0.5 + a1 * np.sin(np.pi * f1 * x + p1) + a2 * np.sin(np.pi * f2 * x + p2)
        b = binarize_top(vals, target)

    return balance_binary(b, target, rng)


def init_binary_C(b, n):
    C = np.zeros(2 * n - 1, dtype=np.int32)
    zeros = np.where(b == 0)[0]
    if zeros.size == 0:
        return C
    off = n - 1
    ones = np.where(b == 1)[0]
    for i in ones:
        C[off + zeros - int(i)] += 1
    return C


def top_k_sum_arr(arr, k):
    arr = np.asarray(arr)
    if arr.size <= 1:
        return int(np.sum(arr))
    k = max(1, min(int(k), int(arr.size)))
    if arr.size <= k:
        return int(np.sum(arr))
    part = np.partition(arr, -k)
    return int(part[-k:].sum())


def binary_edge_candidate(b, n, lag, rng):
    lag = int(lag)
    if lag >= 0:
        m = n - lag
        if m <= 0:
            return None
        src = b[:m]
        tgt = b[lag:lag + m]
        valid = np.where((src == 1) & (tgt == 0))[0]
        if valid.size == 0:
            return None
        s = int(valid[rng.integers(valid.size)])
        t = s + lag
    else:
        l = -lag
        m = n - l
        if m <= 0:
            return None
        src = b[l:]
        tgt = b[:m]
        valid = np.where((src == 1) & (tgt == 0))[0]
        if valid.size == 0:
            return None
        s = int(valid[rng.integers(valid.size)]) + l
        t = s - l

    if t < 0 or t >= n:
        return None
    if b[s] != 1 or b[t] != 0:
        return None
    return (t, s)


def binary_swap_eval(C, b, om, i, j, n, idx):
    C2 = C.copy()
    off = n - 1

    C2[off + idx - i] += om
    C2[off + i - idx] -= b
    C2[off + idx - j] -= om
    C2[off + j - idx] += b

    C2[off] -= 2
    C2[off + j - i] += 1
    C2[off + i - j] += 1

    return C2, int(C2.max())


def binary_local_search(b0, deadline, rng, target_count=-1):
    n = int(b0.size)
    if n < 2:
        return np.asarray(b0, dtype=np.int8).copy(), 0

    b = balance_binary(b0, n // 2, rng)
    C = init_binary_C(b, n)
    if C.size == 0:
        return b, 0

    om = (1 - b).astype(np.int8)
    idx = np.arange(n)
    cur_max = int(C.max())
    cur_tie = top_k_sum_arr(C, 5)
    best_max = cur_max
    best_b = b.copy()

    temp = 0.35
    no_best = 0
    it = 0
    max_iter = 1000000

    while it < max_iter and time.perf_counter() < deadline:
        it += 1

        top_count = min(8, 2 * n - 1)
        top_idx = np.argsort(C)[-top_count:]

        cand = []
        for tidx in top_idx:
            lag = int(tidx - (n - 1))
            if lag == 0:
                continue
            c = binary_edge_candidate(b, n, lag, rng)
            if c is not None:
                zero_idx, one_idx = c
                if b[zero_idx] == 0 and b[one_idx] == 1:
                    cand.append((zero_idx, one_idx))
            if len(cand) >= 4:
                break

        zeros = np.where(b == 0)[0]
        ones = np.where(b == 1)[0]
        if zeros.size > 0 and ones.size > 0:
            zi = int(zeros[rng.integers(zeros.size)])
            oi = int(ones[rng.integers(ones.size)])
            cand.append((zi, oi))

        accepted = False
        for zero_idx, one_idx in cand:
            if time.perf_counter() >= deadline:
                break
            if zero_idx == one_idx:
                continue
            if b[zero_idx] != 0 or b[one_idx] != 1:
                continue

            C2, nm = binary_swap_eval(C, b, om, zero_idx, one_idx, n, idx)
            accept = False

            if nm < cur_max:
                accept = True
            elif nm == cur_max:
                tie2 = top_k_sum_arr(C2, 5)
                if tie2 < cur_tie:
                    accept = True
            else:
                if nm <= cur_max + 2:
                    p = math.exp(-float(nm - cur_max) / max(temp, 1e-6))
                    if rng.random() < p:
                        accept = True

            if accept:
                b[zero_idx] = 1
                b[one_idx] = 0
                om[zero_idx] = 0
                om[one_idx] = 1

                C = C2
                cur_max = nm
                cur_tie = top_k_sum_arr(C, 5)

                if nm < best_max:
                    best_max = nm
                    best_b = b.copy()
                    no_best = 0
                else:
                    no_best += 1

                temp = max(0.04, temp * 0.999)
                accepted = True

                if target_count >= 0 and best_max <= target_count:
                    break
                break

        if not accepted:
            no_best += 1
            temp = max(0.04, temp * 0.999)
            if no_best > 800:
                break

        if target_count >= 0 and best_max <= target_count:
            break

    return best_b, best_max


def fft_len_for(n):
    m = max(1, 2 * int(n) - 1)
    N = 1
    while N < m:
        N <<= 1
    return N


def corr_full_fft(h):
    n = int(h.size)
    if n == 0:
        return np.array([], dtype=float)
    if n == 1:
        return np.array([float(h[0]) * (1.0 - float(h[0]))], dtype=float)

    N = fft_len_for(n)
    fh = np.fft.rfft(np.asarray(h, dtype=float), N)
    fv = np.fft.rfft(1.0 - np.asarray(h, dtype=float), N)

    c = np.fft.irfft(fv * np.conj(fh), N)
    c = np.asarray(c, dtype=float)

    out = np.empty(2 * n - 1, dtype=float)
    out[:n - 1] = c[N - n + 1:N]
    out[n - 1] = c[0]
    out[n:] = c[1:n]

    if not np.all(np.isfinite(out)):
        out = np.nan_to_num(out, nan=0.0, posinf=1e100, neginf=-1e100)
    return out


def fft_max_count(h):
    full = corr_full_fft(h)
    if full.size == 0:
        return float("inf")
    m = float(np.max(full))
    if not np.isfinite(m):
        return float("inf")
    return m


def active_gradient(full, h, k, beta):
    n = h.size
    g = np.zeros(n, dtype=float)
    if n == 0 or full.size == 0:
        return g
    if not np.all(np.isfinite(full)):
        full = np.nan_to_num(full, nan=0.0, posinf=1e100, neginf=-1e100)

    maxc = float(np.max(full))
    if not np.isfinite(maxc):
        return g

    k = max(1, min(int(k), full.size))
    idx = np.argsort(full)[-k:]
    scale = 2.0 / float(n)
    beta = float(beta)

    wraw = np.exp(beta * (full[idx] - maxc) * scale)
    wsum = float(np.sum(wraw))
    if wsum <= 0.0 or not np.isfinite(wsum):
        wraw = np.ones(k, dtype=float) / float(k)
    else:
        wraw = wraw / wsum

    off = n - 1
    for p in range(k):
        t = int(idx[p])
        w = float(wraw[p])
        lag = t - off
        if lag == 0:
            g += w * (1.0 - 2.0 * h)
        elif lag > 0:
            m = n - lag
            if m > 0:
                g[:m] += w * (1.0 - h[lag:])
                g[lag:] -= w * h[:m]
        else:
            l = -lag
            m = n - l
            if m > 0:
                g[l:] += w * (1.0 - h[:m])
                g[:m] -= w * h[l:]
    return g


def local_fractional(h0, n, deadline, rng, max_iter=20000):
    n = int(n)
    if n < 2:
        return np.asarray(h0, dtype=float).copy(), 0.0

    target = 0.5 * float(n)
    h = project_density(h0, target)
    S = target

    cur_count = fft_max_count(h)
    if not np.isfinite(cur_count):
        h = project_density(np.full(n, 0.5), S)
        cur_count = fft_max_count(h)

    best_h = h.copy()
    best_count = cur_count
    stall = 0
    active_k = 12 if n <= 220 else 8
    beta = 80.0

    for _ in range(int(max_iter)):
        if time.perf_counter() >= deadline - 2.0:
            break

        full = corr_full_fft(h)
        cur_count = float(np.max(full))
        if not np.isfinite(cur_count):
            cur_count = fft_max_count(h)

        if cur_count < best_count:
            best_count = cur_count
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 350:
                break

        g = active_gradient(full, h, active_k, beta)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n, dtype=float)

        improved = False

        d = -g
        d = d - float(np.mean(d))
        maxabs = float(np.max(np.abs(d)))
        if maxabs > 1e-12:
            d = d / maxabs
            for stp in (0.03, 0.08, 0.16, 0.012, 0.004):
                cand = project_density(h + stp * d, S)
                c = fft_max_count(cand)
                if c < cur_count - 1e-9:
                    h = cand
                    cur_count = c
                    improved = True
                    if c < best_count:
                        best_count = c
                        best_h = h.copy()
                        stall = 0
                    break

        if not improved:
            m = 8 if n <= 170 else 5
            lows = np.argsort(g)[:m]
            highs = np.argsort(-g)[:m]

            pairs = []
            for ii in lows:
                ii = int(ii)
                for jj in highs:
                    jj = int(jj)
                    if ii == jj:
                        continue
                    if g[ii] - g[jj] > -1e-8:
                        continue
                    dup = False
                    for pr in pairs:
                        if pr[0] == ii and pr[1] == jj:
                            dup = True
                            break
                    if not dup:
                        pairs.append((ii, jj))

            if len(pairs) < 10 and n > 4:
                for _ in range(6):
                    ii = int(rng.integers(n))
                    jj = int(rng.integers(n))
                    if ii == jj:
                        continue
                    dup = False
                    for pr in pairs:
                        if pr[0] == ii and pr[1] == jj:
                            dup = True
                            break
                    if not dup:
                        pairs.append((ii, jj))

            pairs = pairs[:24]
            best_cand = None
            best_cand_count = cur_count

            for ii, jj in pairs:
                cap = min(1.0 - float(h[ii]), float(h[jj]))
                if cap <= 1e-11:
                    continue

                deltas = []
                for frac in (1.0, 0.5, 0.25, 0.1, 0.05):
                    dv = frac * cap
                    if dv <= 1e-11:
                        continue
                    if dv >= cap:
                        dv = cap - 1e-12
                    dupd = False
                    for ud in deltas:
                        if abs(ud - dv) < 1e-12:
                            dupd = True
                            break
                    if not dupd:
                        deltas.append(dv)

                for dv in deltas:
                    cand = h.copy()
                    cand[ii] += dv
                    cand[jj] -= dv

                    if cand[ii] < 0.0:
                        cand[ii] = 0.0
                    elif cand[ii] > 1.0:
                        cand[ii] = 1.0
                    if cand[jj] < 0.0:
                        cand[jj] = 0.0
                    elif cand[jj] > 1.0:
                        cand[jj] = 1.0

                    c = fft_max_count(cand)
                    if c < best_cand_count - 1e-9:
                        best_cand = cand
                        best_cand_count = c

            if best_cand is not None and best_cand_count < cur_count - 1e-9:
                h = best_cand
                cur_count = best_cand_count
                improved = True
                if cur_count < best_count:
                    best_count = cur_count
                    best_h = h.copy()
                    stall = 0

        if stall > 0 and stall % 60 == 0:
            if stall > 220:
                break
            pert = 0.02 * rng.standard_normal(n)
            pert = pert - float(np.mean(pert))
            cand = project_density(h + pert, S)
            c = fft_max_count(cand)
            if c < cur_count - 1e-9:
                h = cand
                cur_count = c
                stall = 120

    best_h = project_density(best_h, target)
    best_count = fft_max_count(best_h)
    return best_h, best_count


def make_frac_candidate(n, rng, mode):
    n = int(n)
    if n <= 1:
        return project_density(np.array([0.5], dtype=float), 0.5 * n)

    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.full(n, 0.5, dtype=float)

    if mode == 0:
        for m in range(1, 7):
            amp = float(rng.uniform(0.05, 0.35)) * (0.65 ** (m - 1))
            freq = int(max(1, np.round(m * float(rng.uniform(0.5, 1.7)))))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            h += amp * np.sin(np.pi * freq * x + phase)
    elif mode == 1:
        b = make_binary_blocks(n, rng, True)
        h = b.astype(float)
    else:
        f1 = int(rng.integers(1, 5))
        f2 = int(rng.integers(1, 5))
        p1 = float(rng.uniform(0.0, 2.0 * np.pi))
        p2 = float(rng.uniform(0.0, 2.0 * np.pi))
        a1 = float(rng.uniform(0.30, 0.60))
        a2 = float(rng.uniform(0.10, 0.30))
        h = 0.5 + a1 * np.sin(np.pi * f1 * x + p1) + a2 * np.sin(np.pi * f2 * x + p2)

    return project_density(h, 0.5 * n)


def add_unique_array(arr, lst):
    if arr is None:
        return
    arr = np.asarray(arr)
    if arr.size == 0:
        return
    for u in lst:
        if np.asarray(u).size == arr.size and bool(np.array_equal(np.asarray(u), arr)):
            return
    lst.append(arr)


def add_int_unique(lst, val, max_val=1000):
    val = int(val)
    if val < 2 or val > int(max_val):
        return
    for u in lst:
        if int(u) == val:
            return
    lst.append(val)


def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    if budget < 5.0:
        n = 90
        h = project_density(make_frac_candidate(n, np.random.default_rng(123), 0), 0.5 * n)
        return h, evaluate_exact_c5(h), n

    total_deadline = start + max(1.0, budget - 30.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        p = project_density(parent, 0.5 * float(parent.size))
        cp = evaluate_exact_c5(p)
        if np.isfinite(cp) and cp < best_c:
            best_h = p
            best_c = cp
            best_n = int(p.size)

    if best_h is None or not np.isfinite(best_c):
        n0 = 90
        best_h = make_frac_candidate(n0, rng, 0)
        best_h = project_density(best_h, 0.5 * float(n0))
        best_c = evaluate_exact_c5(best_h)
        best_n = n0

    binary_alloc = min(300.0, max(50.0, 0.22 * budget))
    binary_deadline = min(total_deadline - 140.0, start + binary_alloc)
    if binary_deadline < start + 10.0:
        binary_deadline = min(total_deadline - 10.0, start + 10.0)

    binary_ns = []
    if parent is not None:
        bp = int(parent.size)
        if 2 <= bp <= 400 and bp % 2 == 0:
            binary_ns.append(bp)
        else:
            for cand_n in (bp + 1, bp - 1):
                if 2 <= cand_n <= 400 and cand_n % 2 == 0:
                    binary_ns.append(cand_n)
                    break

    for nn in (90, 96, 100, 108, 120, 144, 180, 220, 252, 300, 360):
        if nn not in binary_ns:
            binary_ns.append(nn)

    if len(binary_ns) > 7:
        binary_ns = binary_ns[:7]

    for bn in binary_ns:
        if time.perf_counter() > binary_deadline - 10.0:
            break

        if np.isfinite(best_c):
            target_count = int(math.ceil(float(bn) * best_c / 2.0 - 1e-12)) - 1
        else:
            target_count = int(math.ceil(0.4 * float(bn))) - 1
        if target_count < -1:
            target_count = -1

        starts = []
        if parent is not None:
            add_unique_array(make_binary_start(bn, rng, 0, parent), starts)
        if best_h is not None:
            add_unique_array(make_binary_start(bn, rng, 0, best_h), starts)
        add_unique_array(make_binary_start(bn, rng, 1), starts)
        add_unique_array(make_binary_start(bn, rng, 2), starts)
        add_unique_array(make_binary_start(bn, rng, 3), starts)

        starts = starts[:3]
        for b0 in starts:
            if time.perf_counter() > binary_deadline - 6.0:
                break
            sub_deadline = min(binary_deadline - 2.0, time.perf_counter() + 12.0)
            bb, _ = binary_local_search(b0, sub_deadline, rng, target_count)
            if bb is None or bb.size == 0:
                continue
            ht = bb.astype(float)
            ct = evaluate_exact_c5(ht)
            if np.isfinite(ct) and ct < best_c:
                best_h = ht
                best_c = ct
                best_n = int(bn)

    frac_alloc = min(900.0, max(220.0, 0.62 * budget))
    frac_deadline = min(total_deadline - 80.0, start + binary_alloc + frac_alloc)
    if frac_deadline < binary_deadline + 30.0:
        frac_deadline = min(total_deadline - 30.0, binary_deadline + 60.0)

    frac_ns = []
    if best_n > 0:
        add_int_unique(frac_ns, best_n, 1000)
        add_int_unique(frac_ns, 2 * best_n, 1000)
    if parent is not None:
        add_int_unique(frac_ns, int(parent.size), 1000)
        add_int_unique(frac_ns, 2 * int(parent.size), 1000)
    for nn in (90, 180, 360, 720, 900):
        add_int_unique(frac_ns, nn, 1000)

    if len(frac_ns) > 5:
        frac_ns = frac_ns[:5]

    for fn in frac_ns:
        if time.perf_counter() > frac_deadline - 20.0:
            break

        starts = []
        if best_h is not None:
            add_unique_array(upsample_h(best_h, fn), starts)
        if parent is not None:
            add_unique_array(upsample_h(parent, fn), starts)
        if fn <= 220:
            mode = int(rng.integers(0, 3))
            add_unique_array(make_frac_candidate(fn, rng, mode), starts)

        max_starts = 2 if fn <= 180 else 1
        starts = starts[:max_starts]

        for si, sh in enumerate(starts):
            rem = frac_deadline - time.perf_counter()
            if rem < 18.0:
                break
            left = max(1, len(starts) - si)
            per = rem / float(left)
            if fn <= 180:
                cap = 160.0
            elif fn <= 360:
                cap = 110.0
            else:
                cap = 70.0
            per = max(25.0, min(per, cap))
            sub_deadline = time.perf_counter() + per
            if sub_deadline > frac_deadline - 3.0:
                sub_deadline = frac_deadline - 3.0

            h_out, _ = local_fractional(sh, fn, sub_deadline, rng, max_iter=30000)
            if h_out is None or h_out.size == 0:
                continue

            hfe = adjust_sum_light(h_out, 0.5 * float(fn))
            cfe = evaluate_exact_c5(hfe)
            coff = evaluate_exact_c5(h_out)

            hbest = hfe
            cbest = cfe
            if is_feasible_h(h_out, 0.5 * float(fn), 1e-7) and coff < cbest:
                hbest = h_out
                cbest = coff

            if np.isfinite(cbest) and cbest < best_c:
                best_h = hbest
                best_c = cbest
                best_n = int(fn)

    if time.perf_counter() < total_deadline - 45.0 and best_h is not None:
        nfin = int(best_n)
        if nfin < 2:
            nfin = 90
        hfin, _ = local_fractional(best_h, nfin, total_deadline - 12.0, rng, max_iter=30000)
        if hfin is not None and hfin.size == nfin:
            hfe = adjust_sum_light(hfin, 0.5 * float(nfin))
            cfe = evaluate_exact_c5(hfe)
            coff = evaluate_exact_c5(hfin)
            if is_feasible_h(hfin, 0.5 * float(nfin), 1e-7) and coff < cfe:
                hfe = hfin
                cfe = coff
            if np.isfinite(cfe) and cfe < best_c:
                best_h = hfe
                best_c = cfe
                best_n = nfin

    if time.perf_counter() < total_deadline - 18.0 and best_h is not None:
        nbin = int(best_n)
        if nbin >= 4 and nbin % 2 == 0 and nbin <= 420:
            b0 = binarize_top(best_h, nbin // 2)
            if np.isfinite(best_c):
                tb = int(math.ceil(float(nbin) * best_c / 2.0 - 1e-12)) - 1
            else:
                tb = -1
            bb, _ = binary_local_search(b0, total_deadline - 6.0, rng, tb)
            ht = bb.astype(float)
            ct = evaluate_exact_c5(ht)
            if np.isfinite(ct) and ct < best_c:
                best_h = ht
                best_c = ct
                best_n = nbin

    if best_h is None:
        n = 90
        best_h = make_frac_candidate(n, rng, 0)
        best_c = evaluate_exact_c5(best_h)
        best_n = n

    target = 0.5 * float(int(best_h.size))
    h_alt = project_density(best_h, target)
    c_alt = evaluate_exact_c5(h_alt)
    c_cur = evaluate_exact_c5(best_h)

    if is_feasible_h(best_h, target, 1e-7):
        if c_alt < c_cur:
            best_h = h_alt
            best_c = c_alt
        else:
            best_c = min(c_cur, best_c)
    else:
        best_h = h_alt
        best_c = c_alt

    if not is_feasible_h(best_h, target, 1e-7):
        best_h = adjust_sum_light(best_h, target, tol=1e-9)
        best_c = evaluate_exact_c5(best_h)

    final_c = evaluate_exact_c5(best_h)
    if np.isfinite(final_c) and final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), int(best_h.size)
```
