def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-16)[0]
        else:
            idx = np.where(h > 1e-16)[0]

            break

    return h

    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

    return x.sum() - n / 2.0


def eq_jac(x, n):
    return np.ones(n)


def smooth_max_gradient(h, tau):

    m = np.max(corr)

def slsqp_optimize(x0, n, tau, maxiter, time_limit):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    t_start = time.time()

    def fun(x):
        return smooth_max_obj(x, n, dx, tau)

        return eq_con(x, n)

    def con_jac(x):
        return eq_jac(x, n)

    try:
        res = minimize(
            fun, x0, method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': con, 'jac': con_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-13, 'disp': False}
        )
        return np.clip(res.x, 0.0, 1.0)

                best_h, n_best, tau_ng, 8000, 1e-12, remain * 0.5
            )
            h_new = fix_sum(h_new, n_best)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 5: Diversification - try complement, shifts, structured starts
    if time.time() - t0 < budget_s * 0.92:
        n_div = len(best_h)
        candidates = []

        # Complement
        comp = fix_sum(1.0 - best_h, n_div)
        # Shift by 1
        shift1 = fix_sum(np.roll(best_h, n_div // 4), n_div)
        # Shift by half
        shift2 = fix_sum(np.roll(best_h, n_div // 2), n_div)

        candidates.append(comp)
        candidates.append(shift1)
        candidates.append(shift2)

        for tau in [0.01, 0.005]:
            for cand in candidates:
                if time.time() - t0 > budget_s * 0.92:
                    break
                maxiter = 1000
                h_new = slsqp_optimize(cand, n_div, tau, maxiter, 30)
                h_new = fix_sum(h_new, n_div)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new

        # Noise restarts
        for r in range(8):
            if time.time() - t0 > budget_s * 0.94:
                break
            noise = rng.normal(0, 0.04, n_div)
            x_pert = fix_sum(np.clip(best_h + noise, 0.0, 1.0), n_div)
            h_new = slsqp_optimize(x_pert, n_div, 0.003, 600, 20)
            h_new = fix_sum(h_new, n_div)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 6: Final ultra-fine polish
    if time.time() - t0 < budget_s * 0.97:
        n_fin = len(best_h)
        remain = budget_s * 0.97 - (time.time() - t0)
        h_new, c5_new = projected_gradient_opt(
            best_h, n_fin, 0.0005, 10000, 1e-14, remain
        )
        h_new = fix_sum(h_new, n_fin)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Final SLSQP polish with tiny tau
    if time.time() - t0 < budget_s * 0.985:
        n_fin = len(best_h)
        h_new = slsqp_optimize(fix_sum(best_h, n_fin), n_fin, 0.001, 2000, 10)
        h_new = fix_sum(h_new, n_fin)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
