def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)

def project_density(x, target):

    n = x.size
    if n == 0:
        return x
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)

        return x
    lo, hi = -max(s-target, 0) - 1.0, -max(target-s, 0) - 1.0

    # If s > target: need to decrease, so lo is negative

        else:

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)

def project_density(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    if n == 0:
        return x
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    target = float(target)
    s = float(np.sum(x))
    if abs(s - target) <= 1e-13:
        return x
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s_mid = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s_mid < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
