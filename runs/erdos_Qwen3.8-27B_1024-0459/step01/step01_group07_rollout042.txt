Actually, one more thought. I realize that the smooth max with temperature T gives F ≈ max(f) + T·log(number of near-max elements). As T → 0, F → max(f). But the gradient also becomes concentrated on the max element. This can cause numerical issues.

A better approach might be to use "soft alphabetic" or "huber" smoothing:
F_α = max(f) + (1/α) · log(Σ (f_m - max(f))^α) for α > 1

Or simply use the log-sum-exp with a floor on T (e.g., T ≥ 0.005) and then do a final "sharp" optimization where we only consider the top few lags.

Actually, the simplest approach that works well in practice:
1. Use log-sum-exp with T annealing from 0.1 down to 0.005
2. At the end, identify the argmax lag and do a local optimization that just minimizes f at that specific lag (using SLSQP or similar)
3. Iterate: re-identify argmax, optimize, repeat

Let me implement this "alternating" approach:
- Step A: Minimize f(k*) where k* is the current argmax (this is a smooth problem)
- Step B: Re-evaluate, find new argmax
- Repeat until convergence

This is essentially a coordinate descent on the max, and it often works well for min-max problems.

For step A, minimizing f(k*) = Σ_j h[j+k*]·(1-h[j])·dx subject to sum(h) = n/2, 0 ≤ h ≤ 1:
This is a quadratic optimization problem (the objective is quadratic in h). We can use:
- SLSQP with bounds and equality constraint
- Or L-BFGS-B with the v-parameterization (since the objective is smooth for fixed k*)

The gradient of f(k*) w.r.t. h[i] is straightforward:
∂f(k*)/∂h[i] = dx · [(1-h[i-k*])·1[valid] - h[i+k*]·1[valid]]

This is O(n) to compute, much faster than the smooth max gradient!

```python
def alt_minimize(h, n, k_star, maxiter=200):
    """Minimize f(k_star) = sum h[j+k]*(1-h[j])*dx subject to constraints."""
    dx = 2.0 / n
    lag = k_star - (n - 1)
    
    def obj_v(v_free):
        v = np.zeros(n)
        v[:n-1] = v_free
        v[n-1] = -np.sum(v_free)
        h = 0.5 + v
        pen = 500.0 * (np.sum(np.maximum(0, -h)**2) + np.sum(np.maximum(0, h-1)**2))
        # f(k_star) = dx * sum_j h[j+lag] * (1-h[j])
        a = max(0, -lag)
        b = min(n-1, n-1-lag)
        js = np.arange(a, b+1)
        f_val = dx * np.sum(h[js+lag] * (1-h[js]))
        # Gradient
        a = max(0, -lag)
        b = min(n-1, n-1-lag)
        grad_h = np.zeros(n)
        # First term
        i1s = max(a+lag, 0)
        i1e = min(b+lag, n-1)
        if i1s <= i1e:
            idx = np.arange(i1s, i1e+1)
            grad_h[idx] += dx * (1-h[idx-lag])
        # Second term
        i2s = max(a, 0)
        i2e = min(b, n-1)
        if i2s <= i2e:
            idx = np.arange(i2s, i2e+1)
            valid = (idx+lag >= 0) & (idx+lag < n)
            if np.any(valid):
                grad_h[idx[valid]] -= dx * h[idx[valid]+lag]
        # Penalty gradient
        grad_pen = np.zeros(n)
        below = h < 0
        above = h > 1
        grad_pen[below] = -1000.0 * h[below]
        grad_pen[above] = 1000.0 * (h[above] - 1)
        # Chain rule
        grad_v = grad_h[:n-1] - grad_h[n-1] + grad_pen[:n-1] - grad_pen[n-1]
        return f_val + pen, grad_v
    
    v0 = h - 0.5
    v0_free = v0[:n-1].copy()
    bounds = [(-0.5, 0.5)] * (n-1)
    
    res = minimize(obj_v, v0_free, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    
    v = np.zeros(n)
    v[:n-1] = res.x
    v[n-1] = -np.sum(res.x)
    h_new = np.clip(0.5 + v, 0, 1)
    h_new = project_feasible(h_new, n)
    return h_new

def alternating_optimize(h, n, max_rounds=20, maxiter_per_round=200):
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n
    for round in range(max_rounds):
        conv = np.correlate(h, 1-h, mode='full') * dx
        k_star = int(np.argmax(conv))
        h = alt_minimize(h, n, k_star, maxiter=maxiter_per_round)
        new_conv = np.correlate(h, 1-h, mode='full') * dx
        if np.max(new_conv) <= np.max(conv) * (1 - 1e-8) + 1e-12:
            # Check if k_star changed
            new_k = int(np.argmax(new_conv))
            if new_k == k_star:
                break
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(conv)


def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-15
            if not np.any(free):
                break
            h[free] = np.minimum(h[free] + diff / np.sum(free), 1.0)
        else:
            free = h > 1e-15
            if not np.any(free):
                break
            h[free] = np.maximum(h[free] + diff / np.sum(free), 0.0)
    return h


def smooth_max_and_grad(h, n, T):
    dx = 2.0 / n
    conv_raw = np.correlate(h, 1.0 - h, mode='full')
    f = conv_raw * dx
    max_f = np.max(f)
    exp_f = np.exp((f - max_f) / T)
    sum_exp = np.sum(exp_f)
    weights = exp_f / sum_exp
    F = max_f + T * np.log(sum_exp)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m] * dx
        if w < 1e-10:
            continue
        lag = m - (n - 1)
        a = max(0, -lag)
        b = min(n - 1, n - 1 - lag)
        if a > b:
            continue
        i1s = max(a + lag, 0)
        i1e = min(b + lag, n - 1)
        if i1s <= i1e:
            idx = np.arange(i1s, i1e + 1)
            grad[idx] += w * (1.0 - h[idx - lag])
        i2s = max(a, 0)
        i2e = min(b, n - 1)
        if i2s <= i2e:
            idx = np.arange(i2s, i2e + 1)
            valid = (idx + lag >= 0) & (idx + lag < n)
            if np.any(valid):
                grad[idx[valid]] -= w * h[idx[valid] + lag]
    return F, grad


def alt_minimize(h, n, k_star, maxiter=300):
    dx = 2.0 / n
    lag = k_star - (n - 1)
    a = max(0, -lag)
    b = min(n - 1, n - 1 - lag)

    def obj_v(v_free):
        v = np.zeros(n)
        v[:n - 1] = v_free
        v[n - 1] = -np.sum(v_free)
        h = 0.5 + v
        pen = 500.0 * (np.sum(np.maximum(0.0, -h) ** 2) + np.sum(np.maximum(0.0, h - 1.0) ** 2))
        if a <= b:
            js = np.arange(a, b + 1)
            f_val = dx * np.sum(h[js + lag] * (1.0 - h[js]))
        else:
            f_val = 0.0
        grad_h = np.zeros(n)
        if a <= b:
            i1s = max(a + lag, 0)
            i1e = min(b + lag, n - 1)
            if i1s <= i1e:
                idx = np.arange(i1s, i1e + 1)
                grad_h[idx] += dx * (1.0 - h[idx - lag])
            i2s = max(a, 0)
            i2e = min(b, n - 1)
            if i2s <= i2e:
                idx = np.arange(i2s, i2e + 1)
                valid = (idx + lag >= 0) & (idx + lag < n)
                if np.any(valid):
                    grad_h[idx[valid]] -= dx * h[idx[valid] + lag]
        grad_pen = np.zeros(n)
        below = h < 0
        above = h > 1
        grad_pen[below] = -1000.0 * h[below]
        grad_pen[above] = 1000.0 * (h[above] - 1.0)
        grad_v = grad_h[:n - 1] - grad_h[n - 1] + grad_pen[:n - 1] - grad_pen[n - 1]
        return f_val + pen, grad_v

    v0 = h - 0.5
    v0_free = v0[:n - 1].copy()
    bounds = [(-0.5, 0.5)] * (n - 1)
    res = minimize(obj_v, v0_free, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
    v = np.zeros(n)
    v[:n - 1] = res.x
    v[n - 1] = -np.sum(res.x)
    h_new = np.clip(0.5 + v, 0, 1)
    h_new = project_feasible(h_new, n)
    return h_new


def alternating_optimize(h, n, max_rounds=30, maxiter_per_round=300, time_check=None):
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n
    best_val = compute_c5(h, n)
    for rnd in range(max_rounds):
        if time_check is not None and time_check() < 20:
            break
        conv = np.correlate(h, 1.0 - h, mode='full') * dx
        top_lags = np.argsort(-conv)[:3]
        improved = False
        for k_star in top_lags:
            if time_check is not None and time_check() < 20:
                break
            h_new = alt_minimize(h, n, int(k_star), maxiter=maxiter_per_round)
            v_new = compute_c5(h_new, n)
            if v_new < best_val - 1e-10:
                h = h_new
                best_val = v_new
                improved = True
        if not improved:
            break
    return h, best_val


def smooth_max_optimize(h, n, T_init=0.08, T_min=0.003, n_phases=6, maxiter_per_phase=150, time_check=None):
    h = project_feasible(h.copy(), n)

    def obj_v(v_free, T):
        v = np.zeros(n)
        v[:n - 1] = v_free
        v[n - 1] = -np.sum(v_free)
        h_local = 0.5 + v
        pen = 500.0 * (np.sum(np.maximum(0.0, -h_local) ** 2) + np.sum(np.maximum(0.0, h_local - 1.0) ** 2))
        F, gh = smooth_max_and_grad(h_local, n, T)
        grad_pen = np.zeros(n)
        below = h_local < 0
        above = h_local > 1
        grad_pen[below] = -1000.0 * h_local[below]
        grad_pen[above] = 1000.0 * (h_local[above] - 1.0)
        grad_v = gh[:n - 1] - gh[n - 1] + grad_pen[:n - 1] - grad_pen[n - 1]
        return F + pen, grad_v

    v0 = h - 0.5
    v0_free = v0[:n - 1].copy()
    bounds = [(-0.5, 0.5)] * (n - 1)

    for phase in range(n_phases):
        if time_check is not None and time_check() < 30:
            break
        T = T_init * (T_min / T_init) ** (phase / max(n_phases - 1, 1))
        if T < T_min:
            T = T_min
        res = minimize(lambda vf: obj_v(vf, T), v0_free, method='L-BFGS-B', jac=True,
                       bounds=bounds, options={'maxiter': maxiter_per_phase, 'ftol': 1e-13})
        v0_free = res.x
        v = np.zeros(n)
        v[:n - 1] = v0_free
        v[n - 1] = -np.sum(v0_free)
        h = np.clip(0.5 + v, 0, 1)
        h = project_feasible(h, n)

    return h, compute_c5(h, n)
```
