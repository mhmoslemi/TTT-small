```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx


def compute_grad_single(h, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(conv))
    m = idx - (n - 1)
    grad = np.zeros(n)
    if m >= 0:
        if m == 0:
            grad[:] = (1.0 - 2.0 * h) * dx
        elif m < n:
            grad[m:] += (1.0 - h[:n - m]) * dx
            grad[:n - m] += (-h[m:]) * dx
    else:
        p = -m
        if p == 0:
            grad[:] = (1.0 - 2.0 * h) * dx
        elif p < n:
            grad[:n - p] += (1.0 - h[p:]) * dx
            grad[p:] += (-h[:n - p]) * dx
    return grad


def compute_grad_soft(h, n, n_top=8, temp=0.5):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(conv)
    scale = max(temp * max_val * 0.02, 0.5)
    shifted = conv - max_val
    weights = np.exp(shifted / scale)
    total_w = np.sum(weights)
    if total_w < 1e-30:
        top_idx = np.argsort(conv)[-n_top:]
    else:
        weights /= total_w
        top_idx = np.argsort(conv)[-n_top:]
    grad = np.zeros(n)
    for k_idx in top_idx:
        w = weights[k_idx] if total_w >= 1e-30 else 1.0 / n_top
        if w < 1e-8:
            continue
        m = k_idx - (n - 1)
        if m >= 0:
            if m == 0:
                grad += w * (1.0 - 2.0 * h) * dx
            elif m < n:
                grad[m:] += w * (1.0 - h[:n - m]) * dx
                grad[:n - m] += w * (-h[m:]) * dx
        else:
            p = -m
            if p == 0:
                grad += w * (1.0 - 2.0 * h) * dx
            elif p < n:
                grad[:n - p] += w * (1.0 - h[p:]) * dx
                grad[p:] += w * (-h[:n - p]) * dx
    return grad


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-15
            cnt = np.sum(free)
            if cnt == 0:
                break
            h[free] = np.minimum(h[free] + diff / cnt, 1.0)
        else:
            free = h > 1e-15
            cnt = np.sum(free)
            if cnt == 0:
                break
            h[free] = np.maximum(h[free] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h


def adam_optimize(h0, n, max_iter=5000, lr=0.02, beta1=0.9, beta2=0.999,
                  eps=1e-8, time_budget=100.0, t_start=None):
    h = h0.copy()
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    best_h = h.copy()
    best_val = compute_c5(h, n)
    for t in range(1, max_iter + 1):
        if t_start is not None and time.time() - t_start > time_budget:
            break
        progress = t / max_iter
        temp = 1.0 - 0.8 * progress
        n_top = max(2, int(8 * (1 - 0.5 * progress)))
        val = compute_c5(h, n)
        if val < best_val - 1e-15:
            best_val = val
            best_h = h.copy()
        grad = compute_grad_soft(h, n, n_top=n_top, temp=temp)
        m_hat = beta1 * m_hat + (1.0 - beta1) * grad
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad ** 2)
        m_unb = m_hat / (1.0 - beta1 ** t)
        v_unb = v_hat / (1.0 - beta2 ** t)
        h_new = h - lr * m_unb / (np.sqrt(v_unb) + eps)
        h_new = project_feasible(h_new, n)
        h = h_new
    val = compute_c5(h, n)
    if val < best_val:
        best_val = val
        best_h = h.copy()
    return best_h, best_val


def binary_local_search(h_start, n, n_rounds=20000, time_budget=60.0,
                        t_start=None, rng=None):

    h = np.zeros(n)
    order = np.argsort(-h_start)
    h[order[:target]] = 1.0
    best_h = h.copy()
    best_val = compute_c5(h, n)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    no_improve = 0
    for r in range(n_rounds):
        if t_start is not None and time.time() - t_start > time_budget:
            break
        improved = False
        n_ones = len(ones)
        n_zeros = len(zeros)
        if n_ones == 0 or n_zeros == 0:
            break
        n_tries = min(300, n_ones * n_zeros)
        for _ in range(n_tries):
            i = ones[rng.randint(n_ones)]
            j = zeros[rng.randint(n_zeros)]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            val = compute_c5(h, n)
            if val < best_val - 1e-14:
                best_val = val
                best_h = h.copy()
                improved = True
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
                n_ones = len(ones)
                n_zeros = len(zeros)
            else:
                h[i] = 1.0
                h[j] = 0.0
        if not improved:
            no_improve += 1
            if no_improve > 8:
                break
        else:
            no_improve = 0
    return best_h, best_val


def slsqp_polish(h0, n, maxiter=2000, ftol=1e-14):
    target = n / 2.0
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return compute_c5(x, n)

    def obj_grad(x):
        return compute_grad_single(x, n)

    res = minimize(obj, h0.copy(), jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x, res.fun


def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    # Get initial solution
    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if h_init is not None and len(h_init) > 0:
        n_init = len(h_init)
    else:
        n_init = 70
        h_init = np.full(n_init, 0.5)

    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_val = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Adam on parent resolution
    if time_left() > 80:
        for att in range(3):
            if time_left() < 30:
                break
            if att == 0:
                h0 = best_h.copy()
            else:
                h0 = best_h + rng.normal(0, 0.08 * (1 + att), n_init)
            h0 = project_feasible(h0, n_init)
            h_try, v_try = adam_optimize(h0, n_init, max_iter=4000, lr=0.015,
                                         time_budget=min(60, time_left() - 10),
                                         t_start=time.time())
            h_try = project_feasible(h_try, n_init)
            v_try = compute_c5(h_try, n_init)
            if v_try < best_val:
                best_h, best_val, best_n = h_try, v_try, n_init

    # Phase 2: Try higher resolutions with Adam
    resolutions = [100, 140, 200, 280]
    for n_fine in resolutions:
        if time_left() < 60:
            break
        h_fine = interpolate_h(best_h, best_n, n_fine)
        budget_this = min(150, time_left() - 30)
        if budget_this < 20:
            break
        h_try, v_try = adam_optimize(h_fine, n_fine, max_iter=8000, lr=0.012,
                                     time_budget=budget_this, t_start=time.time())
        h_try = project_feasible(h_try, n_fine)
        v_try = compute_c5(h_try, n_fine)
        if v_try < best_val:
            best_h, best_val, best_n = h_try, v_try, n_fine

    # Phase 3: Binary local search at multiple resolutions
    for n_bs in [best_n, 100, 140]:
        if time_left() < 40:
            break
        if n_bs == best_n:
            h_bs_start = best_h
        else:
            h_bs_start = interpolate_h(best_h, best_n, n_bs)
        h_bin, v_bin = binary_local_search(h_bs_start, n_bs, n_rounds=15000,
                                           time_budget=min(80, time_left() - 10),
                                           t_start=time.time(), rng=rng)
        if v_bin < best_val:
            best_h, best_val, best_n = h_bin, v_bin, n_bs
        # Continuous refine from binary
        if time_left() > 50:
            h_refine = h_bin + rng.normal(0, 0.05, n_bs)
            h_refine = project_feasible(h_refine, n_bs)
            h_try, v_try = adam_optimize(h_refine, n_bs, max_iter=3000, lr=0.01,
                                         time_budget=min(40, time_left() - 15),
                                         t_start=time.time())
            h_try = project_feasible(h_try, n_bs)
            v_try = compute_c5(h_try, n_bs)
            if v_try < best_val:
                best_h, best_val, best_n = h_try, v_try, n_bs

    # Phase 4: SLSQP polish
    if time_left() > 50:
        n = best_n
        try:
            h_try, v_try = slsqp_polish(best_h, n, maxiter=3000, ftol=1e-14)
            h_try = project_feasible(h_try, n)
            v_try = compute_c5(h_try, n)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
        except Exception:
            pass

    # Phase 5: More random restarts at best resolution
    n = best_n
    n_restarts = 0
    while time_left() > 40 and n_restarts < 40:
        n_restarts += 1
        strat = n_restarts % 4
        if strat == 0:
            h_r = rng.uniform(0, 1, n)
        elif strat == 1:
            h_r = best_h + rng.normal(0, 0.15, n)
        elif strat == 2:
            h_r = rng.randint(0, 2, n).astype(float)
            h_r += rng.normal(0, 0.03, n)
        else:
            idx = rng.choice(n, size=max(1, n // 4), replace=False)
            h_r = best_h.copy()
            h_r[idx] = rng.uniform(0, 1, len(idx))
        h_r = project_feasible(h_r, n)
        bud = min(25, time_left() - 10)
        if bud < 5:
            break
        h_try, v_try = adam_optimize(h_r, n, max_iter=2000, lr=0.012,
                                     time_budget=bud, t_start=time.time())
        h_try = project_feasible(h_try, n)
        v_try = compute_c5(h_try, n)
        if v_try < best_val:
            best_h, best_val = h_try, v_try

    # Phase 6: Final SLSQP on best
    if time_left() > 30:
        n = best_n
        try:
            h_try, v_try = slsqp_polish(best_h, n, maxiter=2000, ftol=1e-15)
            h_try = project_feasible(h_try, n)
            v_try = compute_c5(h_try, n)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
        except Exception:
            pass

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_val = compute_c5(best_h, best_n)
    # Verify valid
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_val = compute_c5(best_h, best_n)

    return (best_h, best_val, best_n)
```
