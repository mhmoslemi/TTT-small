    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-8
        else:
            mask = h > 1e-8
        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        # Don't overshoot
        if diff > 0:
            max_step = (1.0 - h[mask]).min()
            step = min(step, max_step)
        else:
            max_step = h[mask].min()
            step = max(step, -max_step)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    exps = np.exp(np.clip(beta * (corr - m), -50, 50))
    return (m + np.log(np.sum(exps)) / beta) * dx

        break
    
    # Grid search fallback (very fine)
    ts = np.linspace(0.0, 1.0, 50)
    for t in ts:
        c_t = c0 + t * delta
        mt = np.max(c_t)
        if mt < best_val:
            best_val = mt
            best_t = t
    
    # Refine around best_t
    lo = max(0.0, best_t - 2/50.0)
    hi = min(1.0, best_t + 2/50.0)
    ts2 = np.linspace(lo, hi, 20)
    for t in ts2:
        c_t = c0 + t * delta
        mt = np.max(c_t)
        if mt < best_val:
            best_val = mt
            best_t = t
    
    return best_t

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def remaining():
        return max(0.0, budget_s - (time.time() - start_time) - 5.0)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_h = None
        parent_n = 0
    
    if best_h is None:
        best_n = 96
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)
    
    print("Start: n={}, C5={:.6f}".format(best_n, best_c5))
    
    n = best_n
    
    # Phase 1: Coordinate descent (exploits 1D linear structure)
    if remaining() > 350:
        h_cd = best_h.copy()
        c5_cd = compute_c5(h_cd)
        no_improve_count = 0
        
        for sweep in range(150):
            if remaining() < 250:
                break
            sweep_improved = False
            order = rng.permutation(n)
            
            for j in order:
                if remaining() < 250:
                    break
                new_t = coordinate_descent_step(h_cd, j, n, rng)
                if abs(new_t - h_cd[j]) > 1e-10:
                    h_cd[j] = new_t
                    h_cd = project_feasible(h_cd, n)
                    c5_cd = compute_c5(h_cd)
                    if c5_cd < best_c5:
                        best_c5 = c5_cd
                        best_h = h_cd.copy()
                        best_n = n
                        sweep_improved = True
                    no_improve_count = 0
                else:
                    no_improve_count += 1
            
            if not sweep_improved and no_improve_count > n:
                break
        
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n
        print("After CD: C5={:.6f}".format(best_c5))
    
    # Phase 2: Simulated Annealing with structured moves
    if remaining() > 200:
        h_sa = best_h.copy()
        c5_sa = compute_c5(h_sa)
        T_start = 0.008
        T_end = 0.00002
        n_sa = 25000
        cool_rate = (T_end / T_start) ** (1.0 / n_sa)
        T = T_start
        
        for it in range(n_sa):
            if remaining() < 100:
                break
            
            h_new = h_sa.copy()
            move = rng.rand()
            
            if move < 0.30:
                # Perturb 1-3 random coordinates
                npert = rng.randint(1, 4)
                idx = rng.choice(n, size=npert, replace=False)
                h_new[idx] += rng.randn(npert) * 0.06
                h_new = project_feasible(h_new, n)
            elif move < 0.50:
                # Swap two values
                i1, i2 = rng.choice(n, size=2, replace=False)
                h_new[i1], h_new[i2] = h_new[i2], h_new[i1]
                h_new = project_feasible(h_new, n)
            elif move < 0.70:
                # Invert one or two values
                idx = rng.choice(n, size=rng.randint(1, 3), replace=False)
                h_new[idx] = 1.0 - h_new[idx]
                h_new = project_feasible(h_new, n)
            elif move < 0.85:
                # Block rotation
                blen = rng.randint(2, max(3, n // 6))
                bstart = rng.randint(0, n - blen)
                h_new[bstart:bstart+blen] = np.roll(h_new[bstart:bstart+blen], rng.randint(-3, 4))
                h_new = project_feasible(h_new, n)
            else:
                # Targeted: find active lag and perturb coordinates that contribute to it
                over = all_overlaps(h_sa)
                active_k = np.argmax(over)
                # Perturb random coordinates
                idx = rng.choice(n, size=3, replace=False)
                h_new[idx] += rng.randn(3) * 0.08
                h_new = project_feasible(h_new, n)
            
            c5_new = compute_c5(h_new)
            delta = c5_new - c5_sa
            if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-14)):
                h_sa = h_new
                c5_sa = c5_new
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
                    best_n = n
            
            T *= cool_rate
        
        print("After SA: C5={:.6f}".format(best_c5))
    
    # Phase 3: Differential Evolution (global exploration)
    if remaining() > 180:
        pop_size = 25
        pop = np.zeros((pop_size, n))
        pop[0] = best_h.copy()
        
        for i in range(1, pop_size):
            style = i % 4
            if style == 0:
                h_init = best_h + rng.randn(n) * 0.12
            elif style == 1:
                h_init = np.where(rng.rand(n) < 0.5, 1.0, 0.0)
            elif style == 2:
                freq = rng.randint(1, 6)
                phase = rng.rand() * 2 * np.pi
                h_init = 0.5 + 0.45 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            else:
                # Two-block structure
                h_init = np.zeros(n)
                split = rng.randint(n // 4, 3 * n // 4)
                h_init[split:] = 1.0
                # Add noise
                h_init += rng.randn(n) * 0.1
            pop[i] = project_feasible(np.clip(h_init, 0, 1), n)
        
        scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
        
        for gen in range(50):
            if remaining() < 100:
                break
            F_gen = max(0.4, 0.85 - 0.005 * gen)
            CR_gen = max(0.5, 0.8 - 0.002 * gen)
            
            for i in range(pop_size):
                if remaining() < 100:
                    break
                others = np.delete(np.arange(pop_size), i)
                r1, r2, r3 = rng.choice(others, size=3, replace=False)
                v = pop[r1] + F_gen * (pop[r2] - pop[r3])
                cross = rng.rand(n) < CR_gen
                cross[rng.randint(n)] = True
                trial = np.where(cross, v, pop[i])
                trial = project_feasible(trial, n)
                sc = compute_c5(trial)
                if sc <= scores[i]:
                    pop[i] = trial
                    scores[i] = sc
                if sc < best_c5:
                    best_c5 = sc
                    best_h = trial.copy()
                    best_n = n
        
        # Use best from population for next phase
        de_best = np.argmin(scores)
        if scores[de_best] < best_c5:
            best_c5 = scores[de_best]
            best_h = pop[de_best].copy()
            best_n = n
        print("After DE: C5={:.6f}".format(best_c5))
    
    # Phase 4: Second coordinate descent round
    if remaining() > 80:
        h_cd2 = best_h.copy()
        for sweep in range(60):
            if remaining() < 50:
                break
            sweep_improved = False
            order = rng.permutation(n)
            for j in order:
                if remaining() < 50:
                    break
                new_t = coordinate_descent_step(h_cd2, j, n, rng)
                if abs(new_t - h_cd2[j]) > 1e-10:
                    h_cd2[j] = new_t
                    h_cd2 = project_feasible(h_cd2, n)
                    sweep_improved = True
                    c5_now = compute_c5(h_cd2)
                    if c5_now < best_c5:
                        best_c5 = c5_now
                        best_h = h_cd2.copy()
                        best_n = n
            if not sweep_improved:
                break
        c5_cd2 = compute_c5(h_cd2)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
            best_n = n
        print("After CD2: C5={:.6f}".format(best_c5))
    
    # Phase 5: SLSQP polish with multiple beta values
    if remaining() > 50:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
        
        for beta in [25.0, 60.0, 150.0, 400.0]:
            if remaining() < 25:
                break
            try:
                r = minimize(smooth_obj, best_h.copy(), args=(beta,), method='SLSQP',
                            bounds=bounds, constraints=cons,
                            options={'maxiter': 500, 'ftol': 1e-15, 'iprint': 0})
                h_pol = project_feasible(r.x, n)
                c5_pol = compute_c5(h_pol)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()
                    best_n = n
                    print("  SLSQP beta={}: C5={:.6f}".format(beta, best_c5))
                else:
                    best_h = h_pol.copy()  # use projected even if not better (might help next)
                    best_h = project_feasible(best_h, n)
            except Exception:
                pass
        
        # Raw objective SLSQP
        if remaining() > 30:
            try:
                r = minimize(raw_obj, best_h.copy(), method='SLSQP',
                            bounds=bounds, constraints=cons,
                            options={'maxiter': 400, 'ftol': 1e-15, 'iprint': 0})
                h_pol = project_feasible(r.x, n)
                c5_pol = compute_c5(h_pol)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()
                    best_n = n
                    print("  SLSQP raw: C5={:.6f}".format(best_c5))
            except Exception:
                pass
    
    # Phase 6: Try higher resolution
    if remaining() > 80:
        for n_h in [128, 160, 192, 256]:
            if remaining() < 60:
                break
            
            x_o = (np.arange(best_n) + 0.5) * (2.0 / best_n)
            x_n = (np.arange(n_h) + 0.5) * (2.0 / n_h)
            h_h = np.interp(x_n, x_o, best_h)
            h_h = project_feasible(h_h, n_h)
            
            # SLSQP at higher res
            bounds_h = [(0.0, 1.0)] * n_h
            cons_h = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_h / 2.0}]
            try:
                r = minimize(smooth_obj, h_h.copy(), args=(50.0,), method='SLSQP',
                            bounds=bounds_h, constraints=cons_h,
                            options={'maxiter': 200, 'ftol': 1e-14, 'iprint': 0})
                h_pol = project_feasible(r.x, n_h)
                c5_pol = compute_c5(h_pol)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()
                    best_n = n_h
                    print("  High-res n={}: C5={:.6f}".format(n_h, best_c5))
            except Exception:
                pass
            
            # Quick CD at higher res
            if remaining() > 40 and best_n == n_h:
                h_cd_h = best_h.copy()
                for j in rng.permutation(n_h):
                    if remaining() < 30:
                        break
                    new_t = coordinate_descent_step(h_cd_h, j, n_h, rng)
                    if abs(new_t - h_cd_h[j]) > 1e-10:
                        h_cd_h[j] = new_t
                        h_cd_h = project_feasible(h_cd_h, n_h)
                c5_h = compute_c5(h_cd_h)
                if c5_h < best_c5:
                    best_c5 = c5_h
                    best_h = h_cd_h.copy()
                    best_n = n_h
                    print("  HiCD n={}: C5={:.6f}".format(n_h, best_c5))
    
    # Final: binary rounding attempt (only if n is even)
    if remaining() > 10 and best_n % 2 == 0:
        h_round = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_round))
        n_want = best_n // 2
        if n_ones > n_want:
            ones_idx = np.where(h_round == 1.0)[0]
            order = np.argsort(best_h[ones_idx])[:n_ones - n_want]
            h_round[ones_idx[order]] = 0.0
        elif n_ones < n_want:
            zero_idx = np.where(h_round == 0.0)[0]
            order = np.argsort(-best_h[zero_idx])[:n_want - n_ones]
            h_round[zero_idx[order]] = 1.0
        c5_round = compute_c5(h_round)
        if c5_round < best_c5:
            best_c5 = c5_round
            best_h = h_round.copy()
            best_n = best_n
            print("  Binary: C5={:.6f}".format(best_c5))
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
