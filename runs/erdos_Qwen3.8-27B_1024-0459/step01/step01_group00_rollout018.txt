            old_val = h[j]

    for _ in range(500):

        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h_init, n, max_sweeps=5000, tol=1e-14, check_every=50):
    h = make_feasible(h_init.copy(), n)
    target = n / 2.0
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    best_c5 = compute_c5(h)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if sweep > 0 and sweep % check_every == 0:
            c5_now = compute_c5(h)
            if c5_now < best_c5 - 1e-15:
                best_c5 = c5_now
                no_improve = 0
            else:
                no_improve += check_every
            if no_improve >= 2000:
                break
        
        improved = False
        max_change = 0.0
        
        for j in range(n):
            old_val = h[j]
            
            h[j] = 0.0
            corr_0 = np.correlate(h, 1.0 - h, mode='full')
            h[j] = 1.0
            corr_1 = np.correlate(h, 1.0 - h, mode='full')
            h[j] = old_val
            
            slope = corr_1 - corr_0
            
            a, b = 0.0, 1.0
            for _ in range(40):
                c_pt = b - (b - a) / gr
                d_pt = a + (b - a) / gr
                fc = np.max(corr_0 + c_pt * slope)
                fd = np.max(corr_0 + d_pt * slope)
                if fc < fd:
                    b = d_pt
                else:
                    a = c_pt
                if b - a < 1e-16:
                    break
            
            t_opt = (a + b) / 2.0
            val_opt = np.max(corr_0 + t_opt * slope)
            
            val_0 = np.max(corr_0)
            val_1 = np.max(corr_1)
            
            best_t = t_opt
            best_val = val_opt
            if val_0 < best_val:
                best_t = 0.0
                best_val = val_0
            if val_1 < best_val:
                best_t = 1.0
                best_val = val_1
            
            if abs(best_t - old_val) > 1e-15:
                h[j] = best_t
                max_change = max(max_change, abs(best_t - old_val))
                improved = True
        
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-12:
            if diff > 0:
                mask = h < 1.0 - 1e-12
            else:
                mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - np.sum(h)
            if abs(diff2) > 1e-12:
                mask = (h > 1e-12) & (h < 1.0 - 1e-12)
                cnt2 = np.count_nonzero(mask)
                if cnt2 > 0:
                    h[mask] += diff2 / cnt2
                h = np.clip(h, 0.0, 1.0)
        
        if not improved or max_change < tol:
            break
    
    h = make_feasible(h, n)
    return h

def pair_moves(h, n, n_pairs=200, max_rounds=50):
    h = make_feasible(h.copy(), n)
    target = n / 2.0
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    best_c5 = compute_c5(h)
    
    for rnd in range(max_rounds):
        improved = False
        for _ in range(n_pairs):
            j = np.random.randint(n)
            k = np.random.randint(n)
            if j == k:
                continue
            
            old_j, old_k = h[j], h[k]
            
            h[j] = 0.0
            h[k] = 0.0
            corr_base = np.correlate(h, 1.0 - h, mode='full')
            
            h[j] = 1.0
            h[k] = 0.0
            corr_j1 = np.correlate(h, 1.0 - h, mode='full')
            slope_j = corr_j1 - corr_base
            
            h[j] = 0.0
            h[k] = 1.0
            corr_k1 = np.correlate(h, 1.0 - h, mode='full')
            slope_k = corr_k1 - corr_base
            
            h[j] = old_j
            h[k] = old_k
            
            # h[j] = t + delta, h[k] = t - delta where t = (old_j+old_k)/2, delta = (old_j-old_k)/2
            # Actually let's parameterize: h[j] = old_j + s, h[k] = old_k - s (preserves sum)
            # corr(s) = corr_base + old_j*slope_j + old_k*slope_k + s*(slope_j - slope_k)
            
            base_corr = corr_base + old_j * slope_j + old_k * slope_k
            combined_slope = slope_j - slope_k
            
            # Minimize max(base_corr + s * combined_slope) over s in [max(-old_j, old_k-1), min(1-old_j, old_k)]
            s_lo = max(-old_j, old_k - 1.0)
            s_hi = min(1.0 - old_j, old_k)
            if s_lo >= s_hi:
                continue
            
            a, b = s_lo, s_hi
            for _ in range(35):
                c_pt = b - (b - a) / gr
                d_pt = a + (b - a) / gr
                fc = np.max(base_corr + c_pt * combined_slope)
                fd = np.max(base_corr + d_pt * combined_slope)
                if fc < fd:
                    b = d_pt
                else:
                    a = c_pt
                if b - a < 1e-15:
                    break
            
            s_opt = (a + b) / 2.0
            h_try_j = old_j + s_opt
            h_try_k = old_k - s_opt
            
            if 0.0 <= h_try_j <= 1.0 and 0.0 <= h_try_k <= 1.0:
                h[j] = h_try_j
                h[k] = h_try_k
                if abs(s_opt) > 1e-14:
                    improved = True
        
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-12:
            mask = (h > 1e-12) & (h < 1.0 - 1e-12)
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
        elif not improved:
            break
    
    return make_feasible(h, n)

def slsqp_polish(h, n, max_iter=3000):
    def obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out
    except Exception:
        pass
    return None

def thue_morse(n):
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return tm

def random_binary(n, rng):
    h = np.zeros(n)
    ones_pos = rng.choice(n, n // 2, replace=False)
    h[ones_pos] = 1.0
    return h

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def generate_sin_init(n, phase, freq):
    h = 0.5 + 0.3 * np.sin(2.0 * np.pi * freq * np.linspace(0, 2, n, endpoint=False) / 2.0 + phase)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    # Get parent if available
    h_parent = None
    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            h_parent = np.array(initial_h_values, dtype=float)
            if len(h_parent) == 0:
                h_parent = None
    except Exception:
        h_parent = None
    
    best_h = None
    best_c5 = 1e10
    best_n = 160
    
    # Generate starting points at multiple resolutions
    resolutions = [160, 200, 256, 320]
    if h_parent is not None:
        pn = len(h_parent)
        if pn not in resolutions:
            resolutions.insert(0, pn)
    
    resolutions = sorted(set(resolutions))
    
    # Build candidate pool: (h, n, c5)
    candidates = []
    
    for n in resolutions:
        if remaining() < 30:
            break
        
        starts = []
        
        # Parent interpolated
        if h_parent is not None and len(h_parent) != n:
            starts.append(make_feasible(interpolate_to_n(h_parent, n), n))
        elif h_parent is not None and len(h_parent) == n:
            starts.append(make_feasible(h_parent.copy(), n))
        
        # Thue-Morse
        tm = thue_morse(n)
        if abs(np.sum(tm) - n/2.0) < 0.01:
            starts.append(tm)
        else:
            starts.append(make_feasible(tm, n))
        
        # Random binary (2 variants)
        for _ in range(2):
            starts.append(random_binary(n, rng))
        
        # Sinusoidal
        starts.append(generate_sin_init(n, 0.0, 3))
        starts.append(generate_sin_init(n, np.pi/3, 5))
        
        # Gray-like: alternating blocks
        if n % 4 == 0:
            blk = n // 4
            hb = np.zeros(n)
            hb[0:blk] = 1.0
            hb[2*blk:3*blk] = 1.0
            starts.append(hb)
        
        # Interleaved
        if n % 2 == 0:
            hi = np.zeros(n)
            hi[0::4] = 1.0
            hi[2::4] = 1.0
            cnt = np.sum(hi)
            if cnt != n//2:
                starts.append(make_feasible(hi, n))
            else:
                starts.append(hi)
        
        for h0 in starts:
            if remaining() < 10:
                break
            h0 = make_feasible(h0, n)
            c5_0 = compute_c5(h0)
            candidates.append((h0, n, c5_0))
    
    # Sort by initial quality
    candidates.sort(key=lambda x: x[2])
    
    # Phase 1: Coordinate descent on top candidates
    n_cd = min(6, len(candidates))
    for idx in range(n_cd):
        if remaining() < 30:
            break
        h0, n, c5_0 = candidates[idx]
        
        # Coordinate descent
        h_cd = coord_descent(h0, n, max_sweeps=3000, check_every=30)
        c5_cd = compute_c5(h_cd)
        
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n
        
        # Pair moves refinement
        if remaining() > 30:
            h_pm = pair_moves(h_cd, n, n_pairs=100, max_rounds=30)
            c5_pm = compute_c5(h_pm)
            if c5_pm < best_c5:
                best_c5 = c5_pm
                best_h = h_pm.copy()
                best_n = n
        
        # Interpolate best to parent resolution and try there too
        if h_parent is not None and n != len(h_parent):
            if remaining() > 30:
                pn = len(h_parent)
                h_interp = make_feasible(interpolate_to_n(h_cd, pn), pn)
                h_cd2 = coord_descent(h_interp, pn, max_sweeps=2000, check_every=30)
                c5_cd2 = compute_c5(h_cd2)
                if c5_cd2 < best_c5:
                    best_c5 = c5_cd2
                    best_h = h_cd2.copy()
                    best_n = pn
    
    # Phase 2: Focused coordinate descent on the best solution with longer runs
    if remaining() > 60:
        h_long = coord_descent(best_h, best_n, max_sweeps=10000, check_every=50)
        c5_long = compute_c5(h_long)
        if c5_long < best_c5:
            best_c5 = c5_long
            best_h = h_long.copy()
    
    # Phase 3: SLSQP polish
    if remaining() > 40:
        h_sl = slsqp_polish(best_h, best_n, max_iter=3000)
        if h_sl is not None:
            c5_sl = compute_c5(h_sl)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()
    
    # Phase 4: More coordinate descent after SLSQP
    if remaining() > 40:
        h_cd3 = coord_descent(best_h, best_n, max_sweeps=8000, check_every=50)
        c5_cd3 = compute_c5(h_cd3)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3.copy()
    
    # Phase 5: Pair moves on best
    if remaining() > 30:
        h_pm2 = pair_moves(best_h, best_n, n_pairs=300, max_rounds=80)
        c5_pm2 = compute_c5(h_pm2)
        if c5_pm2 < best_c5:
            best_c5 = c5_pm2
            best_h = h_pm2.copy()
    
    # Phase 6: Final SLSQP
    if remaining() > 20:
        h_sl2 = slsqp_polish(best_h, best_n, max_iter=5000)
        if h_sl2 is not None:
            c5_sl2 = compute_c5(h_sl2)
            if c5_sl2 < best_c5:
                best_c5 = c5_sl2
                best_h = h_sl2.copy()
    
    # Phase 7: Final coordinate descent
    if remaining() > 20:
        h_fd = coord_descent(best_h, best_n, max_sweeps=5000, check_every=50)
        c5_fd = compute_c5(h_fd)
        if c5_fd < best_c5:
            best_c5 = c5_fd
            best_h = h_fd.copy()
    
    # Phase 8: Try different resolution for the best pattern
    if remaining() > 60:
        for n_alt in [best_n + 20, best_n - 20, best_n + 40]:
            if n_alt < 50 or remaining() < 20:
                continue
            h_alt = make_feasible(interpolate_to_n(best_h, n_alt), n_alt)
            h_alt_cd = coord_descent(h_alt, n_alt, max_sweeps=3000, check_every=30)
            c5_alt = compute_c5(h_alt_cd)
            if c5_alt < best_c5:
                best_c5 = c5_alt
                best_h = h_alt_cd.copy()
                best_n = n_alt
    
    # Final feasibility
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    # Verify
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10)
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-6
    
    return (best_h, best_c5, best_n)
