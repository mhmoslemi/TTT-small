import time
import numpy as np
from scipy.optimize import minimize


def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)


def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) <= 1e-14 * max(1, n):
        return x
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-12:
        free = np.where((h > 1e-14) & (h < 1.0 - 1e-14))[0]
        if free.size:
            if res > 0:
                idx = int(free[np.argmax(1.0 - h[free])])
            else:
                idx = int(free[np.argmax(h[free])])
            h[idx] += res
            h = np.clip(h, 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-14:
        free = np.where((h > 1e-14) & (h < 1.0 - 1e-14))[0]
        if free.size:
            h[free] += res / float(free.size)
            h = np.clip(h, 0.0, 1.0)

        return p.copy()
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)


def make_structured_start(n, mode, rng):
    n = int(n)
    if n <= 1:
        return project_density(np.array([0.5]), 0.5)
    x = np.arange(n, dtype=float) + 0.5

    if mode == 0:
        h = x / float(n)
    elif mode == 1:
        half = n // 2
        rem = n - half
        h1 = (np.arange(half, dtype=float) + 0.5) / half if half > 0 else np.array([0.5])
        h2 = (np.arange(rem, dtype=float) + 0.5) / rem if rem > 0 else np.array([0.5])
        h = np.concatenate([h1, h2])
    elif mode == 2:
        t = x / float(n)
        h = 1.0 - np.abs(2.0 * t - 1.0)
    elif mode == 3:
        freq = int(rng.integers(1, 4))
        phase = float(rng.uniform(0, 2 * np.pi))
        h = 0.5 + 0.45 * np.sin(2.0 * np.pi * freq * x / n + phase)
    elif mode == 4:
        h = 0.5
        for m in range(1, 5):
            amp = 0.3 / m
            freq = int(rng.integers(1, 3)) * m
            phase = float(rng.uniform(0, 2 * np.pi))
            h += amp * np.sin(2.0 * np.pi * freq * x / n + phase)
    elif mode == 5:
        nblocks = int(rng.integers(2, 6))
        block_len = n // nblocks
        h = np.zeros(n)
        for b in range(nblocks):
            start = b * block_len
            end = start + block_len if b < nblocks - 1 else n
            val = float(rng.random())
            h[start:end] = val
    else:
        u = rng.random(n)
        for _ in range(3):
            kernel = np.ones(int(max(2, n // 50)))
            u = np.convolve(u, kernel, mode='same') / kernel.sum()
        thresh = float(np.median(u))
        h = (u > thresh).astype(float)

    h = project_density(h, 0.5 * n)
    return h


def _nc_global():
    return _G['n']


def smooth_objective(x):
    n = _G['n']
    alpha = _G['alpha']
    if n <= 1:
        return 0.0
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return 1e100
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return 1e100
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return 1e100
    return float((2.0 / float(n)) * (m + np.log(z) / alpha))


def smooth_jacobian(x):
    n = _G['n']
    alpha = _G['alpha']
    if n <= 1:
        return np.zeros(max(1, n))
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return np.zeros(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return np.zeros(n)
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return np.zeros(n)
    w = e / z
    center = n - 1
    g = np.zeros(n)
    threshold = 1e-10
    active = np.where(w > threshold)[0]
    if active.size == 0:
        active = np.array([int(np.argmax(w))])
    for t in active:
        t = int(t)
        p = float(w[t])
        if p <= 0:
            continue
        k = t - center
        if k == 0:
            g += p * (1.0 - 2.0 * h)
        else:
            lag = abs(k)
            if lag >= n:
                continue
            if k > 0:
                g[lag:] += p
                g[:n - lag] -= p * h[lag:]
                g[lag:] -= p * h[:n - lag]
            else:
                g[:n - lag] += p
                g[:n - lag] -= p * h[lag:]
                g[lag:] -= p * h[:n - lag]
    return (2.0 / float(n)) * g


_G = {'n': 0, 'alpha': 1.0, 'target': 0.0}


def _sum_con(x):
    return float(np.sum(np.asarray(x, dtype=float)) - _G['target'])


def _sum_con_jac(x):
    return np.ones_like(np.asarray(x, dtype=float))


def active_set_polish(h, deadline, max_iter=500, active_k=6):
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    if n < 2:
        return best, best_c
    k = min(int(active_k), 2 * n - 1)
    stall = 0
    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        idxs = np.argsort(corr)[-k:]
        center = n - 1
        g = np.zeros(n)
        for t in idxs:
            t = int(t)
            kk = t - center
            if kk == 0:
                g += (1.0 - 2.0 * best)
            else:
                lag = abs(kk)
                if lag >= n:
                    continue
                if kk > 0:
                    g[lag:] += 1.0
                    g[:n - lag] -= best[lag:]
                    g[lag:] -= best[:n - lag]
                else:
                    g[:n - lag] += 1.0
                    g[:n - lag] -= best[lag:]
                    g[lag:] -= best[:n - lag]
        g /= float(k)
        d = -g
        d -= float(np.mean(d))
        norm = float(np.linalg.norm(d))
        if norm < 1e-14:
            break
        d /= norm
        improved = False
        for step in (0.02, 0.05, 0.1, 0.2, 0.4, 0.7, 1.0):
            cand = project_density(best + step * d, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-12:
                best = cand
                best_c = c
                improved = True
                break
        if not improved:
            for step in (0.01, 0.005, 0.002):
                cand = project_density(best + step * d, target)
                c = evaluate_erdos(cand)
                if c < best_c - 1e-12:
                    best = cand
                    best_c = c
                    improved = True
                    break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break
    best = project_density(best, target)
    best_c = evaluate_erdos(best)
    return best, best_c


def local_pair_search(h, deadline, max_iter=800, active_k=5):
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    if n < 2:
        return best, best_c
    k = min(int(active_k), 2 * n - 1)
    stall = 0
    center = n - 1
    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        idxs = np.argsort(corr)[-k:]
        lags = np.abs(idxs - center).astype(int)
        g = np.zeros(n)
        for t in idxs:
            t = int(t)
            kk = t - center
            if kk == 0:
                g += (1.0 - 2.0 * best)
            else:
                lag = abs(kk)
                if lag >= n:
                    continue
                if kk > 0:
                    g[lag:] += 1.0
                    g[:n - lag] -= best[lag:]
                    g[lag:] -= best[:n - lag]
                else:
                    g[:n - lag] += 1.0
                    g[:n - lag] -= best[lag:]
                    g[lag:] -= best[:n - lag]
        g /= float(k)
        npos = min(5, n // 3)
        nneg = min(5, n // 3)
        pos_idx = np.argsort(-g)[:npos]
        neg_idx = np.argsort(g)[:nneg]
        found = False
        for ii in neg_idx:
            ii = int(ii)
            for jj in pos_idx:
                jj = int(jj)
                if ii == jj:
                    continue
                cap_pos = 1.0 - best[ii]
                cap_neg = best[jj]
                cap = min(cap_pos, cap_neg)
                if cap <= 1e-13:
                    continue
                delta_opt = None
                a = g[ii] - g[jj]
                if a < -1e-12:
                    b = 0.0
                    for lag in lags:
                        lag = int(lag)
                        if lag == 0:
                            b += -2.0
                        elif abs(ii - jj) == lag:
                            b += 1.0
                    b /= float(k)
                    if b > 1e-12:
                        delta_opt = -a / (2.0 * b)
                        if delta_opt < 0:
                            delta_opt = None
                deltas = []
                if delta_opt is not None and 0 < delta_opt < cap:
                    deltas.append(delta_opt)
                deltas.extend([0.1 * cap, 0.3 * cap, 0.5 * cap, 0.7 * cap, cap * 0.999])
                for d in deltas:
                    if d <= 1e-13 or d >= cap:
                        if d >= cap:
                            d = cap * 0.999
                        if d <= 1e-13:
                            continue
                    cand = best.copy()
                    cand[ii] += d
                    cand[jj] -= d
                    cand = np.clip(cand, 0.0, 1.0)
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-12:
                        best = cand
                        best_c = c
                        found = True
                        break
                if found:
                    break
            if found:
                break
        if not found:
            d = -g
            d -= float(np.mean(d))
            norm = float(np.linalg.norm(d))
            if norm > 1e-13:
                d /= norm
                for step in (0.03, 0.1, 0.3):
                    cand = project_density(best + step * d, target)
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-12:
                        best = cand
                        best_c = c
                        found = True
                        break
        if found:
            stall = 0
        else:
            stall += 1
            if stall >= 25:
                break
    best = project_density(best, target)
    best_c = evaluate_erdos(best)
    return best, best_c


def optimize_slsqp(h0, n, sub_deadline):
    n = int(n)
    target = 0.5 * n
    h = project_density(h0, target)
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)
    now = time.perf_counter()
    span = max(1.0, sub_deadline - now)
    polish_reserve = min(30.0, max(5.0, 0.25 * span))
    if polish_reserve > 0.5 * span:
        polish_reserve = 0.15 * span
    solve_deadline = sub_deadline - polish_reserve

    if n <= 120:
        alphas = (10.0, 40.0, 100.0, 250.0)
        maxiter = 200
    elif n <= 300:
        alphas = (12.0, 50.0, 150.0)
        maxiter = 120
    elif n <= 600:
        alphas = (15.0, 50.0)
        maxiter = 80
    else:
        alphas = (15.0, 50.0)
        maxiter = 50

    bounds = [(0.0, 1.0)] * n
    constraints = ({"type": "eq", "fun": _sum_con, "jac": _sum_con_jac},)

    for alpha in alphas:
        if time.perf_counter() >= solve_deadline:
            break
        _G['n'] = n
        _G['alpha'] = alpha
        _G['target'] = target
        try:
            res = minimize(
                smooth_objective, h,
                method="SLSQP", jac=smooth_jacobian,
                bounds=bounds, constraints=constraints,
                options={"maxiter": maxiter, "ftol": 1e-10, "disp": False}
            )
            xg = getattr(res, "x", None)
            if xg is not None:
                xg = np.asarray(xg, dtype=float)
                if np.all(np.isfinite(xg)):
                    cand = project_density(xg, target)
                    c = evaluate_erdos(cand)
                    if c < best_c:
                        best_h = cand
                        best_c = c
                    if c <= best_c + 0.05:
                        h = cand
                    else:
                        h = best_h
        except Exception:
            h = best_h

    if time.perf_counter() < sub_deadline - 2.0:
        h2, c2 = active_set_polish(best_h, sub_deadline - 2.0, max_iter=200, active_k=6)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    return best_h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start_time + max(1.0, budget - 15.0)

    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 4:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        nparent = int(parent.size)
        cp = evaluate_erdos(parent)
        if cp < best_c:
            best_h = parent.copy()
            best_c = cp
            best_n = nparent
    else:
        h_init = make_structured_start(90, 2, rng)
        best_h = h_init
        best_c = evaluate_erdos(best_h)
        best_n = 90

    nlist = sorted(set([int(best_n), int(best_n * 2), int(best_n * 4),
                        int(best_n * 8), int(best_n * 16)]))
    nlist = [n for n in nlist if 2 <= n <= 2000]
    for extra in (180, 360, 720, 1440):
        if extra not in nlist:
            nlist.append(extra)
    nlist = sorted(set([n for n in nlist if 10 <= n <= 2000]))

    for ni in nlist:
        if deadline - time.perf_counter() < 30.0:
            break

        starts = []
        if best_h is not None:
            if int(best_h.size) == ni:
                starts.append(best_h.copy())
            else:
                starts.append(upsample_h(best_h, ni))

        if parent is not None:
            starts.append(upsample_h(parent, ni))

        n_struct = 3 if ni <= 360 else 2
        modes = [0, 1, 2, 3, 4, 5, 6]
        for m in range(n_struct):
            starts.append(make_structured_start(ni, modes[m % len(modes)], rng))

        uniq = []
        seen = []
        for s in starts:
            s = project_density(s, 0.5 * ni)
            if s.size != ni:
                continue
            dup = False
            for u in seen:
                if np.allclose(s, u, atol=1e-6):
                    dup = True
                    break
            if not dup:
                uniq.append(s)
                seen.append(s.copy())

        if not uniq:
            continue

        rem = deadline - time.perf_counter()
        left_starts = len(uniq)
        if ni <= 120:
            cap_per = 180.0
        elif ni <= 300:
            cap_per = 120.0
        elif ni <= 720:
            cap_per = 80.0
        else:
            cap_per = 50.0

        for idx_s, s in enumerate(uniq):
            remaining = deadline - time.perf_counter()
            left = max(1, left_starts - idx_s)
            per = min(remaining / float(left), cap_per)
            per = max(15.0, per)
            if remaining < 15.0:
                break
            sub_dl = time.perf_counter() + min(per, remaining - 2.0)

            h_out, c_out = optimize_slsqp(s, ni, sub_dl)
            if h_out is not None:
                h_out = project_density(h_out, 0.5 * ni)
                c_out = evaluate_erdos(h_out)
                if c_out < best_c:
                    best_h = h_out
                    best_c = c_out
                    best_n = ni

    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 20.0:
        nbest = int(best_h.size)
        if nbest <= 400:
            h2, c2 = local_pair_search(best_h, deadline - 5.0, max_iter=500, active_k=5)
            if c2 < best_c:
                best_h = h2
                best_c = c2
                best_n = nbest

        if deadline - time.perf_counter() > 10.0:
            h3, c3 = active_set_polish(best_h, deadline - 3.0, max_iter=300, active_k=8)
            if c3 < best_c:
                best_h = h3
                best_c = c3
                best_n = int(best_h.size)

        if deadline - time.perf_counter() > 10.0:
            n_up = min(int(best_h.size) * 2, 2000)
            if n_up > int(best_h.size):
                h_up = upsample_h(best_h, n_up)
                h4, c4 = active_set_polish(h_up, deadline - 3.0, max_iter=200, active_k=6)
                if c4 < best_c:
                    best_h = h4
                    best_c = c4
                    best_n = n_up
                elif deadline - time.perf_counter() > 5.0:
                    h5, c5 = local_pair_search(h_up, deadline - 2.0, max_iter=200, active_k=5)
                    if c5 < best_c:
                        best_h = h5
                        best_c = c5
                        best_n = n_up

    if best_h is None:
        fallback = project_density(np.full(45, 0.5), 22.5)
        return fallback, evaluate_erdos(fallback), 45

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
