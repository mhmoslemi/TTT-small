    # Zero lag (index n-1):
    # S_0 = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
    scores[n-1] = np.sum(h) - np.sum(h**2)
    grads[n-1] = 1 - 2*h

        scores[n-1+d] = np.dot(h[d:], 1-h[:m])

        g = np.zeros(n)

        g[d:] += (1 - h[:m])

    h = np.asarray(h, dtype=float)
    scores = np.zeros(2*n-1)
    # Zero lag: index n-1
    scores[n-1] = np.sum(h * (1-h))
    # Positive lags: index n-1+d
    for d in range(1, n):
        m = n - d
        scores[n-1+d] = np.dot(h[d:], 1-h[:m])
    # Negative lags: index n-1-d
    for d in range(1, n):
        m = n - d
        scores[n-1-d] = np.dot(h[:m], 1-h[d:])
    return scores

def score_grads(h, n):
    """Compute gradients of all 2n-1 scores w.r.t. h."""
    grads = np.zeros((2*n-1, n))
    # Zero lag
    grads[n-1] = 1 - 2*h
    # Positive lags
    for d in range(1, n):
        m = n - d
        g = np.zeros(n)
        g[d:] += (1 - h[:m])  # h[d+i] appears with coeff (1-h[i])
        g[:m] -= h[d:]         # h[i] appears with coeff -h[d+i]
        grads[n-1+d] = g
    # Negative lags
    for d in range(1, n):
        m = n - d
        g = np.zeros(n)
        g[:m] += (1 - h[d:])
        g[d:] -= h[:m]
        grads[n-1-d] = g
    return grads

def c5_value(h, n):
    scores = compute_scores(h, n)
    return float(np.max(scores)) * (2.0/n)

def lp_step(h, n, delta=0.3, tol=2.0):
    """One LP refinement step.
    
    Minimize t
    s.t. for active k: scores[k](h0) + grad_k(h0)·(h-h0) <= t
         sum(h) = n/2
         |h[i]-h0[i]| <= delta
         0 <= h[i] <= 1
    """
    h0 = np.asarray(h, dtype=float)
    scores = compute_scores(h0, n)
    max_s = np.max(scores)
    
    # Select active constraints (near the max)
    active_mask = scores >= max_s - tol
    if np.sum(active_mask) < 3:
        # Take top 10
        top_idx = np.argsort(scores)[-10:]
        active_mask = np.zeros(2*n-1, dtype=bool)
        active_mask[top_idx] = True
    
    active_idx = np.flatnonzero(active_mask)
    grads = score_grads(h0, n)
    active_grads = grads[active_idx]  # shape (len(active), n)
    active_scores = scores[active_idx]
    
    # LP variables: h[0..n-1], t
    # Objective: minimize t
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0  # t is the last variable
    
    # Inequality: for each active k:
    #   active_scores[k] + active_grads[k] · (h - h0) <= t
    #   => active_grads[k] · h - t <= active_grads[k] · h0 - active_scores[k]
    #   => -active_grads[k] · h + t >= active_scores[k] - active_grads[k] · h0
    # In linprog form (A_ub @ x <= b_ub):
    #   active_grads[k] · h - t <= active_grads[k] · h0 - active_scores[k]
    
    A_ub = np.zeros((len(active_idx), n+1))
    b_ub = np.zeros(len(active_idx))
    for i, idx in enumerate(active_idx):
        A_ub[i, :n] = active_grads[i]
        A_ub[i, n] = -1.0
        b_ub[i] = np.dot(active_grads[i], h0) - active_scores[i]
    
    # Trust region: |h[i] - h0[i]| <= delta
    # -delta <= h[i] - h0[i] <= delta
    # h[i] <= h0[i] + delta
    # -h[i] <= -(h0[i] - delta)
    # Add to bounds instead
    
    lb = np.maximum(0.0, h0 - delta)
    ub = np.minimum(1.0, h0 + delta)
    bounds = list(zip(lb, ub))
    bounds.append((0.0, max_s + n))  # t >= 0, upper bound is loose
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        h_new = res.x[:n]
        t_new = res.x[n]
        return h_new, t_new
    return h0, max_s

        pass
    return h0.copy(), max_s


def lp_refine(h, n, max_iter=40, delta_init=0.5, verbose=False):
    global _DEADLINE
    h = np.asarray(h, dtype=float).copy()
    n = int(n)
    best_h = h.copy()
    best_c = c5_value(h, n)
    delta = delta_init
    stall = 0
    for it in range(max_iter):
        check_deadline()
        h_new, t_new = lp_step(h, n, delta=delta)
        c_new = c5_value(h_new, n)
        if c_new < best_c - 1e-9:
            h = h_new
            best_h = h_new.copy()
            best_c = c_new
            stall = 0
            if verbose:
                print(f"  LP iter {it}: c={c_new:.6f} delta={delta:.3f}")
        elif c_new <= best_c + 1e-8:
            h = h_new
            stall += 1
        else:
            delta *= 0.7
            stall += 2
        if stall > 15:
            delta *= 0.5
            if delta < 0.005:
                break
            stall = 0
    return best_h, best_c


def binary_flip_update(scores, h, idx, new_val, n):
    i = int(idx)
    nv = int(new_val)
    for d in range(1, n):
        pidx = 2 * d - 1
        midx = 2 * d
        if i - d >= 0:
            hj = int(h[i - d])
            if nv == 1:
                scores[pidx] -= hj
                scores[midx] += 1.0 - hj
            else:
                scores[pidx] += hj
                scores[midx] -= 1.0 - hj
        if i + d < n:
            hj = int(h[i + d])
            if nv == 1:
                scores[pidx] += 1.0 - hj
                scores[midx] -= hj
            else:
                scores[pidx] -= 1.0 - hj
                scores[midx] += hj
    h[i] = nv


def binary_scores(h, n):
    hf = h.astype(float)
    scores = np.zeros(2 * n - 1)
    scores[n - 1] = 0.0
    for d in range(1, n):
        m = n - d
        scores[n - 1 - d] = float(np.dot(hf[:m], 1.0 - hf[d:]))
        scores[n - 1 + d] = float(np.dot(hf[d:], 1.0 - hf[:m]))
    return scores


def binary_local_search(h0, n, time_limit, rng):
    n = int(n)
    h = np.asarray(h0, dtype=float).copy()
    order = np.argsort(h)
    nb = np.zeros(n, dtype=np.int8)
    nb[order[n // 2:]] = 1
    scores = binary_scores(nb, n)
    cur_max = float(np.max(scores))
    best_max = cur_max
    best_h = nb.copy()
    start_t = time.time()
    deadline = start_t + float(time_limit)
    if _DEADLINE is not None:
        deadline = min(deadline, _DEADLINE)
    ops = 0
    stall = 0
    max_stall = max(2000, 30 * n)
    while ops < 2000000 and time.time() < deadline - 1.0:
        check_deadline()
        ops += 1
        if ops % 5000 == 0 and time.time() > deadline:
            break
        r = rng.random()
        if r < 0.75:
            active = np.flatnonzero(scores >= cur_max - 1e-9)
            if active.size == 0:
                k = int(np.argmax(scores))
            else:
                k = int(active[int(rng.integers(active.size))])
            if k == 0:
                k = int(np.argmax(scores))
            if k % 2 == 1:
                d = (k + 1) // 2
                i_start = 0
                i_end = n - d
                need_hi = 0
                need_lo = 1
            else:
                d = k // 2
                i_start = 0
                i_end = n - d
                need_hi = 1
                need_lo = 0
            cand_i = np.flatnonzero((nb[i_start:i_end] == need_hi) & (nb[i_start:d + i_start:i_start] == 0) if False else (nb[:i_end] == need_hi) & (nb[d:d + i_end] == need_lo))
            if cand_i.size == 0:
                zeros = np.flatnonzero(nb == 0)
                ones = np.flatnonzero(nb == 1)
                if zeros.size == 0 or ones.size == 0:
                    break
                i = int(zeros[int(rng.integers(zeros.size))])
                j = int(ones[int(rng.integers(ones.size))])
            else:
                i = int(cand_i[int(rng.integers(cand_i.size))])
                j = i + d
        else:
            zeros = np.flatnonzero(nb == 0)
            ones = np.flatnonzero(nb == 1)
            if zeros.size == 0 or ones.size == 0:
                break
            i = int(zeros[int(rng.integers(zeros.size))])
            j = int(ones[int(rng.integers(ones.size))])
        if i == j or i >= n or j >= n or i < 0 or j < 0:
            continue
        if nb[i] == nb[j]:
            continue
        binary_flip_update(scores, nb, i, 1 - int(nb[i]), n)
        binary_flip_update(scores, nb, j, 1 - int(nb[j]), n)
        new_max = float(np.max(scores))
        if new_max < cur_max - 1e-9:
            cur_max = new_max
            stall = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = nb.copy()
        elif new_max <= cur_max + 1e-9:
            if stall > max_stall * 0.5 and rng.random() < 0.1:
                pass
            else:
                binary_flip_update(scores, nb, i, int(nb[i]), n)
                binary_flip_update(scores, nb, j, int(nb[j]), n)
            stall += 1
        else:
            binary_flip_update(scores, nb, i, 1 - int(nb[i]), n)
            binary_flip_update(scores, nb, j, 1 - int(nb[j]), n)
            stall += 1
        if stall > max_stall:
            break
    final_c = cur_max * (2.0 / n)
    if final_c < best_max * (2.0 / n) + 1e-12:
        best_h = nb.copy()
        best_max = cur_max
    return best_h.astype(float), best_max * (2.0 / n)


def upsample_to(parent, n):
    n0 = int(parent.size)
    if n == n0:
        return parent.astype(float).copy()
    if n0 <= 0:
        return np.full(n, 0.5)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x, x0, parent.astype(float))


def get_parent():
    try:
        p = np.asarray(initial_h_values, dtype=float)
        if p.ndim != 1 or p.size < 2:
            return None
        return p.copy()
    except Exception:
        return None


def make_initial(n, rng, parent=None):
    if parent is not None:
        h = upsample_to(parent, n)
        return as_feasible(h, n)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    best_h = None
    best_c = 1e6
    for f in range(1, 5):
        for ph in (0.0, math.pi / 2, math.pi):
            h = 0.5 + 0.4 * np.cos(math.pi * f * x + ph)
            h = as_feasible(h, n)
            c = c5_value(h, n)
            if c < best_c:
                best_c = c
                best_h = h
    if best_h is None:
        best_h = as_feasible(np.full(n, 0.5), n)
    return best_h


def polish_slsqp(h, n, time_limit):
    from scipy.optimize import minimize
    h = as_feasible(h, n)
    start_t = time.time()
    deadline = start_t + max(5.0, float(time_limit))
    if _DEADLINE is not None:
        deadline = min(deadline, _DEADLINE)

    def objective(x):
        h_x = np.clip(x, 0.0, 1.0)
        scores = compute_scores(h_x, n)
        m = float(np.max(scores))
        alpha = 0.1
        shifted = scores - m
        shifted = np.clip(shifted / alpha, -50, 50)
        w = np.exp(shifted)
        A = float(np.sum(w))
        if A <= 0 or not math.isfinite(A):
            return m * (2.0 / n)
        return (m + alpha * math.log(A)) * (2.0 / n)

    def gradient(x):
        h_x = np.clip(x, 0.0, 1.0)
        scores = compute_scores(h_x, n)
        m = float(np.max(scores))
        grads = score_grads(h_x, n)
        alpha = 0.1
        shifted = (scores - m) / alpha
        shifted = np.clip(shifted, -50, 50)
        w = np.exp(shifted)
        A = float(np.sum(w))
        if A <= 0 or not math.isfinite(A):
            idx = int(np.argmax(scores))
            g = grads[idx].copy()
        else:
            g = np.zeros(n)
            for i in range(2 * n - 1):
                if w[i] / A > 1e-8:
                    g += (w[i] / A) * grads[i]
        g *= (2.0 / n)
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 0.5 * n,
             'jac': lambda x: np.ones(n)}]

    best_h = h.copy()
    best_c = c5_value(h, n)
    alpha_list = [0.5, 0.2, 0.05, 0.0]
    for alpha in alpha_list:
        if time.time() > deadline - 5.0:
            break
        maxit = max(20, int(80.0 * (60.0 / n)))
        maxit = min(maxit, 300)
        try:
            res = minimize(objective, h, method='SLSQP', jac=gradient,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxit, 'ftol': 1e-10, 'disp': 0})
            if res.success or res.status == 0:
                h_new = as_feasible(res.x, n)
                c_new = c5_value(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    h = h_new
        except Exception:
            pass
    return best_h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    global _DEADLINE
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except (ValueError, TypeError):
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0:
        budget_s = 30.0
    _DEADLINE = t0 + max(1.0, budget_s - _RESERVE)

    rng = np.random.default_rng(seed)
    parent = get_parent()

    best_h = None
    best_c = 1e6
    best_n = 0

    def update_state(h, n):
        nonlocal best_h, best_c, best_n
        if h is None:
            return
        h = np.asarray(h, dtype=float)
        n = int(n)
        h2 = as_feasible(h, n)
        c2 = c5_value(h2, n)
        if math.isfinite(c2) and c2 < best_c:
            best_h = h2.copy()
            best_c = c2
            best_n = n
            if _DEADLINE is not None:
                remaining = _DEADLINE - time.time()
                if remaining > 0:
                    pass

    if parent is not None:
        n0 = int(parent.size)
        h0 = as_feasible(parent, n0)
        c0 = c5_value(h0, n0)
        update_state(h0, n0)
    else:
        n0 = 70
        h0 = make_initial(n0, rng)
        c0 = c5_value(h0, n0)
        update_state(h0, n0)

    resolution_list = []
    for mult in [1.0, 1.5, 2.0, 2.5, 3.0, 4.0]:
        nn = int(round(mult * n0 / 2.0) * 2)
        if nn >= 4 and nn <= 6 * n0 and nn not in resolution_list:
            resolution_list.append(nn)
    if n0 not in resolution_list:
        resolution_list.insert(0, n0)
    resolution_list.sort()

    for n in resolution_list:
        if _DEADLINE is None or time.time() > _DEADLINE - 60.0:
            break
        remaining = float(_DEADLINE - time.time()) if _DEADLINE else 300.0
        if remaining < 80.0:
            break

        time_per_phase = remaining / 3.0

        # Phase 1: Binary local search
        bin_time = min(time_per_phase * 0.9, max(10.0, 60.0 * (n / 70.0)))
        if n == n0 and parent is not None:
            bin_start = parent
        else:
            bin_start = upsample_to(parent, n) if parent is not None else make_initial(n, rng)

        if n % 2 == 0:
            try:
                bh, bc = binary_local_search(bin_start, n, bin_time, rng)
                if bh is not None and math.isfinite(bc):
                    bhf = as_feasible(bh, n)
                    update_state(bhf, n)
            except BudgetError:
                pass
            except Exception:
                pass

        # Phase 2: LP refinement from best binary
        if _DEADLINE is not None and time.time() > _DEADLINE - 30.0:
            break
        lp_time = remaining - (time.time() - (t0 + budget_s - _RESERVE)) * 0  # just use remaining
        lp_start_h = None
        if best_n == n:
            lp_start_h = best_h
        else:
            lp_start_h = as_feasible(bin_start, n)
            update_state(lp_start_h, n)

        try:
            max_lp_iter = min(50, max(10, int(120.0 / max(1, n // 70))))
            h_lp, c_lp = lp_refine(lp_start_h, n, max_iter=max_lp_iter, delta_init=0.4)
            if h_lp is not None and math.isfinite(c_lp):
                update_state(h_lp, n)
        except BudgetError:
            pass
        except Exception:
            pass

        # Phase 3: Polish with SLSQP
        if _DEADLINE is not None and time.time() > _DEADLINE - 20.0:
            break
        if best_n == n:
            polish_time = min(30.0, float(_DEADLINE - time.time()) - 10.0) if _DEADLINE else 20.0
            if polish_time > 5.0:
                try:
                    h_pol, c_pol = polish_slsqp(best_h, n, polish_time)
                    if h_pol is not None and math.isfinite(c_pol):
                        update_state(h_pol, n)
                except Exception:
                    pass

    # Final: try one more LP refinement on the overall best
    if _DEADLINE is not None and time.time() < _DEADLINE - 5.0 and best_h is not None:
        try:
            h_fin, c_fin = lp_refine(best_h, best_n, max_iter=20, delta_init=0.3)
            update_state(h_fin, best_n)
        except Exception:
            pass

    if best_h is None:
        h_final = as_feasible(h0, n0)
        c_final = c5_value(h_final, n0)
        n_final = n0
    else:
        h_final = as_feasible(best_h, best_n)
        c_final = c5_value(h_final, best_n)
        n_final = best_n

    return h_final, float(c_final), int(n_final)
