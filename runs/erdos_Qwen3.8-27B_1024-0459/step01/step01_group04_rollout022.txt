def smooth_grad(x, alpha):
    check_deadline()
    n = x.size
    if n == 0:
        return np.zeros(0)
    h = np.clip(x, 0.0, 1.0)
    scores = compute_scores(h, n)
    m = float(np.max(scores))
    
    if alpha > 1e-12:
        sh = np.clip((scores - m) / alpha, -40.0, 0.0)
        w = np.exp(sh)
        tot = float(np.sum(w))
        if tot < 1e-300:
            return np.zeros(n)
        probs = w / tot
    else:
        tol = max(1e-10, 1e-7 * max(1.0, m))
        active = scores >= m - tol
        if not np.any(active):
            active[int(np.argmax(scores))] = True
        probs = active.astype(float) / float(np.sum(active))
    
    grad = np.zeros(n, dtype=float)
    scale = 2.0 / float(n)
    
    # Only process lags with significant probability
    thresh = 1e-10
    for k in range(2 * n - 1):
        p = probs[k]
        if p < thresh:
            continue
        d = k - (n - 1)
        if d == 0:
            grad += p * scale * (1.0 - 2.0 * h)
        elif d > 0:
            # s_d = (2/n) * sum_{i=0}^{n-d-1} h[i+d]*(1-h[i])
            # grad[j] += (2/n)*p * [(1-h[j-d]) for j>=d] - (2/n)*p * [h[j+d] for j<=n-d-1]
            m_eff = n - d
            grad[d:] += p * scale * (1.0 - h[:m_eff])
            grad[:m_eff] -= p * scale * h[d:]
        else:
            d_pos = -d
            m_eff = n - d_pos
            # s_{-d'} = (2/n) * sum_{i=0}^{n-d'-1} h[i]*(1-h[i+d'])
            # grad[j] += (2/n)*p * [(1-h[j+d']) for j < n-d'] - (2/n)*p * [h[j-d'] for j>=d']
            grad[:m_eff] += p * scale * (1.0 - h[d_pos:])
            grad[d_pos:] -= p * scale * h[:m_eff]
    
    if not np.all(np.isfinite(grad)):
        return np.zeros(n)
    return grad

def coordinate_descent(h, n, alpha, state, max_sweeps=10, time_limit=30.0):

                break
            # Optimize h[j] with others fixed, subject to integral constraint
            # After changing h[j], we need to adjust one other coordinate
            # Use 1D optimization: minimize over h[j] and compensate in a free variable
            # Simpler: use projected gradient step in coordinate direction
            h_old = h[j]
            # Try small perturbations
            best_val = compute_c5(h)
            for delta in (-0.1, -0.05, -0.02, 0.02, 0.05, 0.1):
                h[j] = np.clip(h_old + delta, 0.0, 1.0)
                # Rebalance
                h2 = as_feasible(h, n)
                c = compute_c5(h2)
                if c < best_val - 1e-12:
                    best_val = c
                    h = h2

                        h = h2
                        improved = True

def binary_scores(h, n):
    n = int(n)
    scores = np.zeros(2 * n - 1, dtype=float)
    hf = h.astype(float)
    for d in range(1, n):
        scores[2 * d - 1] = float(np.dot(hf[:n-d], 1.0 - hf[d:]))
        scores[2 * d] = float(np.dot(hf[d:], 1.0 - hf[:n-d]))
    return scores

def binary_flip_update(scores, h, idx, new_val, n):
    idx = int(idx)
    new_val = int(new_val)
    i = idx
    for d in range(1, n):
        pidx = 2 * d - 1
        midx = 2 * d
        if i - d >= 0:
            hj = int(h[i - d])
            if new_val == 1:
                scores[pidx] -= hj
                scores[midx] += 1.0 - hj
            else:
                scores[pidx] += hj
                scores[midx] -= 1.0 - hj
        if i + d < n:
            hj = int(h[i + d])
            if new_val == 1:
                scores[pidx] += 1.0 - hj
                scores[midx] -= hj
            else:
                scores[pidx] -= 1.0 - hj
                scores[midx] += hj
    h[i] = new_val

def binary_swap_apply(scores, h, i, j, n):
    i = int(i)
    j = int(j)
    if i == j:
        return False
    hi = int(h[i])
    hj = int(h[j])
    if hi == hj:
        return False
    if hi == 0 and hj == 1:
        binary_flip_update(scores, h, i, 1, n)
        binary_flip_update(scores, h, j, 0, n)
    else:
        binary_flip_update(scores, h, i, 0, n)
        binary_flip_update(scores, h, j, 1, n)
    return True

def propose_binary_swap(h, scores, cur, rng, n):
    if rng.random() < 0.6:
        active = np.flatnonzero(scores >= cur - 1e-9)
        if active.size > 0:
            idx = int(active[int(rng.integers(active.size))])
            if idx != 0:
                if idx % 2 == 1:
                    d = (idx + 1) // 2
                    mask = (h[:n-d] == 1) & (h[d:] == 0)
                else:
                    d = idx // 2
                    mask = (h[:n-d] == 0) & (h[d:] == 1)
                cand = np.nonzero(mask)[0]
                if cand.size > 0:
                    pos = int(cand[int(rng.integers(cand.size))])
                    return pos, pos + d
    
    zeros = np.flatnonzero(h == 0)
    ones = np.flatnonzero(h == 1)
    if zeros.size == 0 or ones.size == 0:
        return -1, -1
    i = int(zeros[int(rng.integers(zeros.size))])
    j = int(ones[int(rng.integers(ones.size))])
    return i, j

def binary_local_search(start_h, n, state, time_limit, rng, max_restarts=3):
    n = int(n)
    best_c = 1e9
    best_h = None
    restart_time = max(3.0, time_limit / float(max_restarts + 1))
    start_t = time.time()
    
    for rs in range(max_restarts):
        if time.time() - start_t > time_limit:
            break
        if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE:
            break
        rl = min(time.time() + restart_time, start_t + time_limit)
        if _RUN_DEADLINE is not None:
            rl = min(rl, _RUN_DEADLINE)
        
        if rs == 0 and start_h is not None and np.size(start_h) == n:
            vals = np.asarray(start_h, dtype=float)
            h = (np.asarray(vals, dtype=float) >= 0.5).astype(np.int8)
            # Ensure exactly n//2 ones
            target = n // 2
            n_ones = int(h.sum())
            if n_ones > target:
                ones_idx = np.flatnonzero(h == 1)
                extra = ones_idx[target:]
                h[extra] = 0
            elif n_ones < target:
                zeros_idx = np.flatnonzero(h == 0)
                need = target - n_ones
                h[zeros_idx[:need]] = 1
        else:
            h = np.zeros(n, dtype=np.int8)
            target = n // 2
            idx = rng.choice(n, size=target, replace=False)
            h[idx] = 1
        
        scores = binary_scores(h, n)
        cur = float(np.max(scores))
        best_local = cur
        best_local_h = h.copy()
        
        ops = 0
        stall = 0
        max_ops = 50000
        while ops < max_ops and time.time() < rl:
            ops += 1
            i, j = propose_binary_swap(h, scores, cur, rng, n)
            if i < 0:
                stall += 100
                continue
            if not binary_swap_apply(scores, h, i, j, n):
                continue
            new_max = float(np.max(scores))
            if new_max < cur - 1e-9:

                stall = 0

        if c_local < best_c:
            best_c = c_local
            best_h = best_local_h.astype(float)

def subgradient_method(h, n, state, time_limit, alpha=0.05, lr_init=0.01):
    h = as_feasible(h, n).copy()
    start_t = time.time()
    lr = lr_init
    patience = 0
    best_c = compute_c5(h)
    best_h = h.copy()
    
    while time.time() - start_t < time_limit:
        if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE:
            break
        try:
            g = smooth_grad(h, alpha)
        except BudgetError:
            break
        if not np.all(np.isfinite(g)) or np.max(np.abs(g)) < 1e-14:
            break
        # Projected gradient step
        h_new = h - lr * g
        h_new = as_feasible(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            state.update(h_new, c_new, n)
            h = h_new
            patience = 0
            lr = min(lr * 1.1, 0.05)
        else:
            patience += 1
            if patience > 50:
                lr *= 0.7
                patience = 0
                if lr < 1e-6:
                    break
            h = h_new
        # Adaptive alpha: decrease over time
        alpha = max(0.005, alpha * 0.995)
    
    return best_h, best_c

    remain = float(global_deadline - time.time())
    if remain > 20:
        t_limit = min(80.0, remain * 0.5)
        run_slsqp(state.h, state.n, state, t_limit, alpha_init=0.02, alpha_final=0.0, n_stages=3)
    
    # Validate and return
    h_final = as_feasible(state.h, state.n)
    c_final = compute_c5(h_final)
    n_final = int(state.n)
    
    if not is_valid_candidate(h_final):
        h_final = as_feasible(h0, n0)
        c_final = compute_c5(h_final)
        n_final = n0
    
    return h_final, float(c_final), int(n_final)
