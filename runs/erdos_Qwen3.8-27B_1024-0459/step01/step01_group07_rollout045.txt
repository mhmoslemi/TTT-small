def adam_step(h, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
    return h_new, m, v

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = h < 1.0
            if not np.any(free): break
            h[free] = np.minimum(h[free] + diff / np.sum(free), 1.0)
        else:
            free = h > 0.0
            if not np.any(free): break
            h[free] = np.maximum(h[free] + diff / np.sum(free), 0.0)
    h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_soft_grad(h, n, temp):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(conv)
    exps = np.exp((conv - m) / temp)
    weights = exps / exps.sum()
    
    grad = np.zeros(n)

    return grad
