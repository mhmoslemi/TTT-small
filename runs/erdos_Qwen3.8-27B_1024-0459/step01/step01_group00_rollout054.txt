    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value(h, n, beta):
    dx = 2.0 / n
    corr = corr_array(h)
    vals = corr * dx
    m = np.max(vals)
    shifted = beta * (vals - m)
    shifted = np.clip(shifted, -50, 50)
    return m + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))

def smooth_max_gradient(h, n, beta):
    dx = 2.0 / n
    corr = corr_array(h)
    vals = corr * dx
    m = np.max(vals)
    shifted = beta * (vals - m)
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    p = exp_vals / np.sum(exp_vals)
    
    grad = np.ones(n) * dx
    for d_idx in range(2 * n - 1):
        if p[d_idx] < 1e-18:
            continue
        d = d_idx - (n - 1)
        w = p[d_idx] * dx
        if d >= 0:
            if d < n:
                grad[:n - d] -= w * h[d:]
                grad[d:] -= w * h[:n - d]
        else:
            d2 = -d
            if d2 < n:
                grad[d2:] -= w * h[:n - d2]
                grad[:n - d2] -= w * h[d2:]

def projected_gradient_smooth(h, n, beta_start, beta_end, n_betas, lr_init, max_iter, start_time, budget_s):
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for b_idx in range(n_betas):
        frac = b_idx / max(n_betas - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = lr_init / (1.0 + 0.3 * b_idx)
        
        for it in range(max_iter // n_betas):
            if time.time() - start_time > budget_s - 30:
                return best_h, best_c5
            h = make_feasible(h, n)
            grad = smooth_max_gradient(h, n, beta)
            h_new = h - lr * grad
            h_new = make_feasible(h_new, n)
            # Backtracking line search
            f_new = smooth_max_value(h_new, n, beta)
            if not np.isfinite(f_new):

            h = h_new
            c5_current = compute_c5(h)
            if c5_current < best_c5 - 1e-12:
                best_c5 = c5_current
                best_h = h.copy()
            lr *= 0.98

def slsq_optimize(h, n, max_iter=2000, ftol=1e-14, start_time=None, budget_s=None):
    n_local = n
    
    def obj(x):
        return compute_c5(x)
    
    def obj_grad(x):
        vals = corr_array(x) * (2.0 / n_local)
        idx_max = np.argmax(vals)
        # Subgradient
        g = np.zeros(n_local)
        d = idx_max - (n_local - 1)
        if d >= 0:
            if d < n_local:
                g[:n_local - d] += (1.0 - x[d:]) * (2.0 / n_local)
                g[d:] += (1.0 - x[:n_local - d]) * (2.0 / n_local)
        else:
            d2 = -d
            if d2 < n_local:
                g[d2:] += (1.0 - x[:n_local - d2]) * (2.0 / n_local)
                g[:n_local - d2] += (1.0 - x[d2:]) * (2.0 / n_local)
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    h0 = make_feasible(h.copy(), n)
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': ftol})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

        h3[n // 2:n // 2 + n // 4] = 1.0
        inits.append(make_feasible(h3.copy(), n))
    # Thue-Morse
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    inits.append(make_feasible(tm, n))
    # Gold sequence (m-sequence based)
    if n % 2 == 0:
        # Simple pseudo-random with good autocorrelation
        h_gold = np.zeros(n)
        pos = set()
        seed_val = 1
        for j in range(n // 2):
            p = (seed_val * 7919 + 42) % n
            while p in pos:
                p = (p + 1) % n
            pos.add(p)
            h_gold[p] = 1.0
            seed_val = (seed_val * 3 + 7) % 1000003
        inits.append(make_feasible(h_gold, n))
    # Smooth transition
    x = np.linspace(0, 2, n, endpoint=False)
    h_smooth = 0.5 + 0.4 * np.sin(2 * np.pi * x) + 0.1 * np.sin(4 * np.pi * x + 0.5)
    inits.append(make_feasible(h_smooth, n))
    # Random
    inits.append(make_feasible(rng.rand(n), n))
    return inits

def multi_start_optimize(n, h_starts, budget_per, start_time, budget_s):
    best_h = None
    best_c5 = np.inf
    for h0 in h_starts:
        if time.time() - start_time > budget_s - 60:
            break
        h0 = make_feasible(h0, n)
        c5_0 = compute_c5(h0)
        if c5_0 < best_c5:
            best_c5 = c5_0
            best_h = h0.copy()
        
        # Projected gradient with smooth max annealing
        h_pg, c5_pg = projected_gradient_smooth(
            h0, n, beta_start=5.0, beta_end=200.0,
            n_betas=4, lr_init=0.02, max_iter=3000,
            start_time=start_time, budget_s=budget_s
        )
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
        
        # SLSQP polish
        if time.time() - start_time < budget_s - 30:
            h_sl, c5_sl = slsq_optimize(h_pg, n, max_iter=1000, ftol=1e-14)
            if h_sl is not None and c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()
        
        # Local perturbation
        if time.time() - start_time < budget_s - 20:
            h_lp, c5_lp = local_perturbation(best_h, n, 500, 0.03, start_time, budget_s)
            if c5_lp < best_c5:
                best_c5 = c5_lp
                best_h = h_lp.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    # Get parent
    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is None or len(h_parent) == 0:
            h_parent = None
    except Exception:
        h_parent = None
    
    # Determine resolutions to try
    n_target = 256
    if h_parent is not None:
        # Start with parent resolution and target
        res_list = [len(h_parent), n_target, 192, 128]
    else:
        res_list = [160, n_target, 128, 192]
    res_list = sorted(set(r for r in res_list if 32 <= r <= 512))
    
    best_h_global = None
    best_c5_global = np.inf
    best_n_global = n_target
    
    # Phase 1: Low-resolution exploration for diverse patterns
    # Try n=64, 80, 96 for quick structural discovery
    phase1_end = start_time + 0.08 * budget_s
    low_res_best = []
    for n_lr in [64, 80, 96, 128]:
        if time.time() > phase1_end:
            break
        inits_lr = generate_structured(n_lr, rng)
        for h0 in inits_lr[:4]:
            h_opt, c5_val = projected_gradient_smooth(
                h0, n_lr, beta_start=5.0, beta_end=100.0,
                n_betas=3, lr_init=0.03, max_iter=1500,
                start_time=start_time, budget_s=phase1_end
            )
            if c5_val < 0.45:
                low_res_best.append((h_opt, c5_val, n_lr))
    
    low_res_best.sort(key=lambda x: x[1])
    
    # Phase 2: Main optimization at various resolutions
    phase2_end = start_time + 0.7 * budget_s
    
    for n in res_list:
        if time.time() > phase2_end:
            break
        
        # Build diverse initializations for this resolution
        inits = []
        
        # From parent (interpolated)
        if h_parent is not None and len(h_parent) != n:
            h_p = interpolate_to_n(h_parent, n)
            inits.append(make_feasible(h_p, n))
        elif h_parent is not None:
            inits.append(h_parent.copy())
        
        # From current best global (interpolated)
        if best_h_global is not None and len(best_h_global) != n:
            h_bg = interpolate_to_n(best_h_global, n)
            inits.append(make_feasible(h_bg, n))
        
        # From low-res best solutions (upsampled)
        for (h_lr, c5_lr, n_lr) in low_res_best[:4]:
            if n_lr != n:
                h_up = interpolate_to_n(h_lr, n)
                inits.append(make_feasible(h_up, n))
        
        # Structured
        structured = generate_structured(n, rng)
        inits.extend(structured[:5])
        
        # Random with structure
        for _ in range(2):
            x = np.linspace(0, 2, n, endpoint=False)
            h_r = 0.5 + rng.uniform(-0.3, 0.3) * np.sin(2 * np.pi * rng.randint(1, 8) * x + rng.uniform(0, 2*np.pi))
            inits.append(make_feasible(h_r, n))
        
        # Remove near-duplicates
        unique_inits = [inits[0]] if inits else []
        for h in inits[1:]:
            is_dup = False
            for u in unique_inits:
                if np.max(np.abs(h - u)) < 0.03:
                    is_dup = True
                    break
            if not is_dup:
                unique_inits.append(h)
        inits = unique_inits[:10]
        
        # Optimize each initialization
        for h0 in inits:
            if time.time() > phase2_end:
                break
            h0 = make_feasible(h0, n)
            
            # Projected gradient with smooth max
            h_pg, c5_pg = projected_gradient_smooth(
                h0, n, beta_start=5.0, beta_end=300.0,
                n_betas=5, lr_init=0.015, max_iter=4000,
                start_time=start_time, budget_s=phase2_end
            )
            if c5_pg < best_c5_global:
                best_c5_global = c5_pg
                best_h_global = h_pg.copy()
                best_n_global = n
            
            # SLSQP polish on the projected gradient result
            if time.time() < phase2_end - 20:
                h_sl, c5_sl = slsq_optimize(h_pg, n, max_iter=1500, ftol=1e-14)
                if h_sl is not None and c5_sl < best_c5_global:
                    best_c5_global = c5_sl
                    best_h_global = h_sl.copy()
                    best_n_global = n
            
            # Local perturbation
            if time.time() < phase2_end - 10:
                h_lp, c5_lp = local_perturbation(best_h_global, best_n_global, 400, 0.025, start_time, phase2_end)
                if c5_lp < best_c5_global:
                    best_c5_global = c5_lp
                    best_h_global = h_lp.copy()
        
        # After optimizing at this resolution, update low_res_best
        if best_h_global is not None:
            low_res_best.append((best_h_global.copy(), best_c5_global, n))
            low_res_best.sort(key=lambda x: x[1])
            low_res_best = low_res_best[:6]
    
    # Phase 3: Refinement at best resolution and neighbors
    phase3_end = start_time + 0.9 * budget_s
    if best_h_global is not None:
        n_best = best_n_global
        # Try best resolution and nearby
        for n_try in [n_best, n_best + 16, n_best - 16, n_best + 32]:
            if n_try < 48 or n_try > 512:
                continue
            if time.time() > phase3_end:
                break
            h_interp = interpolate_to_n(best_h_global, n_try)
            h_interp = make_feasible(h_interp, n_try)
            
            # Aggressive projected gradient
            h_opt, c5_opt = projected_gradient_smooth(
                h_interp, n_try, beta_start=10.0, beta_end=500.0,
                n_betas=6, lr_init=0.01, max_iter=5000,
                start_time=start_time, budget_s=phase3_end
            )
            if c5_opt < best_c5_global:
                best_c5_global = c5_opt
                best_h_global = h_opt.copy()
                best_n_global = n_try
            
            # SLSQP
            if time.time() < phase3_end - 30:
                h_sl, c5_sl = slsq_optimize(h_opt, n_try, max_iter=2000, ftol=1e-15)
                if h_sl is not None and c5_sl < best_c5_global:
                    best_c5_global = c5_sl
                    best_h_global = h_sl.copy()
                    best_n_global = n_try
    
    # Phase 4: Final polish
    if best_h_global is not None and time.time() < start_time + budget_s - 15:
        n_fin = best_n_global
        # Simulated annealing for discrete moves
        h_sa, c5_sa = sim_anneal_discrete(
            best_h_global, n_fin, n_iter=3000,
            T_start=0.05, T_end=0.001,
            start_time=start_time, budget_s=start_time + budget_s - 20
        )
        if c5_sa < best_c5_global:
            best_c5_global = c5_sa
            best_h_global = h_sa.copy()
        
        # Local perturbation fine-tuning
        n_trials = min(5000, int(max(0, remaining() - 10) / 0.005))
        h_lp, c5_lp = local_perturbation(
            best_h_global, n_fin,
            n_trials=max(n_trials, 2000), step_size=0.015,
            start_time=start_time, budget_s=start_time + budget_s - 10
        )
        if c5_lp < best_c5_global:
            best_c5_global = c5_lp
            best_h_global = h_lp.copy()
        
        # Final SLSQP
        if time.time() < start_time + budget_s - 10:
            h_sl, c5_sl = slsq_optimize(
                best_h_global, n_fin, max_iter=3000, ftol=1e-15
            )
            if h_sl is not None and c5_sl < best_c5_global:
                best_c5_global = c5_sl
                best_h_global = h_sl.copy()
    
    # Final feasibility check
    n_final = best_n_global if best_h_global is not None else 160
    if best_h_global is None:
        best_h_global = np.ones(n_final) * 0.5
        best_c5_global = compute_c5(best_h_global)
    
    best_h_global = make_feasible(best_h_global, n_final)
    best_c5_global = compute_c5(best_h_global)
    
    return (best_h_global, best_c5_global, n_final)
