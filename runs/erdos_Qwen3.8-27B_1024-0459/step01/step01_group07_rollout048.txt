    n = n_points
    target = n / 2.0
    h = h0.copy()
    h_prev = h.copy()
    m = np.zeros(n)  # momentum
    best_h = h.copy()
    best_val = compute_c5(h, n)
    lr = lr_init
    beta = 0.9  # Nesterov momentum
    
    for it in range(n_iter):
        # Nesterov step
        h_nest = h + beta * m
        _, grad = smooth_max_and_grad(h_nest, n, temp)
        # Grad descent
        m = beta * m - lr * grad
        h_new = h + m
        h_new = project_feasible(h_new, n)
        
        val = compute_c5(h_new, n)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        
        # Simple adaptive: reduce lr if no improvement
        h = h_new
        lr *= 0.999  # slow decay
        if lr < 1e-6:
            lr = 1e-6
    
    return best_h, best_val

    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

    grad = dx * (W1 @ (1.0 - h) - W2 @ h)
    return grad

def nesterov_optimize(h0, n_points, n_iter, temp_start, temp_end, lr_init):
    n = n_points
    target = n / 2.0
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_val = compute_c5(h, n)
    
    m = np.zeros(n)
    beta = 0.9
    lr = lr_init
    no_improve = 0
    
    for it in range(n_iter):
        # Temperature schedule
        frac = it / max(n_iter - 1, 1)
        temp = temp_start * (temp_end / temp_start) ** frac
        
        # Nesterov lookahead
        h_nest = h + beta * m
        h_nest = np.clip(h_nest, 0, 1)
        _, # not needed
        grad = smooth_max_grad(h_nest, n, temp)
        
        # Update
        m = beta * m - lr * grad
        h_new = h + m
        h_new = project_feasible(h_new, n)
        
        val = compute_c5(h_new, n)
        if val < best_val - 1e-14:
            best_val = val
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        h = h_new
        
        # Adapt lr
        if no_improve > 50:
            lr *= 0.9
            no_improve = 0
        if no_improve > 200:
            break
        
        # Adaptive lr: if loss increasing, reduce
        if it % 100 == 99:
            # backtracking
            if compute_c5(h, n) > best_val * 1.01:
                lr *= 0.8
                h = best_h.copy()
                m = np.zeros(n)
    
    return best_h, best_val

def binary_local_search(h0, n_points, n_iter=50000, t_init=0.01, t_end=0.001, rng=None):
    """Simulated annealing on binary h."""
    n = n_points
    if rng is None:
        rng = np.random.RandomState(0)
    
    # Start from binary rounding
    target = n // 2
    idx = np.argsort(-h0)[:target]
    h = np.zeros(n)
    h[idx] = 1.0
    h = project_feasible(h, n)
    
    best_h = h.copy()
    best_val = compute_c5(h, n)
    curr_val = best_val
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    t = t_init
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        temp_sa = t_init * (t_end / t_init) ** frac
        
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        if a == b:
            continue
        
        # Try swap
        h[a] = 0.0
        h[b] = 1.0
        new_val = compute_c5(h, n)
        
        if new_val < curr_val or rng.random() < np.exp(-(new_val - curr_val) / max(temp_sa, 1e-10)):
            curr_val = new_val
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # Revert
            h[a] = 1.0
            h[b] = 0.0
    
    return best_h, best_val
