def project_density(x,target):
    x=np.array(x,dtype=float,copy=True)
    n=x.size
    target=float(target)
    if n==0: return x
    x=np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x=np.clip(x,0.0,1.0)
    s=float(np.sum(x))
    if abs(s-target) <= 1e-12*max(1,n): return x
    lo,hi=-1.0,1.0
    for _ in range(64):
       mid=0.5*(lo+hi)
       h=np.clip(x+mid,0.0,1.0)
       s2=float(np.sum(h))
       if s2 < target: lo=mid
       else: hi=mid
    h=np.clip(x+0.5*(lo+hi),0.0,1.0)
    res=target - float(np.sum(h))
    # try one coordinate with enough slack
    if abs(res) > 0.0:
       tol=1e-12
       if res > 0:
          space=1.0-h
          idxs=np.where(space>tol)[0]
          if idxs.size:
             idx=int(idxs[np.argmax(space[idxs])])
             adj=min(res, space[idx])
             h[idx]+=adj
             res-=adj
          if abs(res)>tol:
             # distribute among all with space
             idxs=np.where(space>tol)[0]
             if idxs.size:
                h[idxs]+=res/idxs.size
                h=np.clip(h,0,1)
       else:
          room=h
          idxs=np.where(room>tol)[0]
          if idxs.size:
             idx=int(idxs[np.argmax(room[idxs])])
             adj=max(res, -room[idx])  # res negative
             h[idx]+=adj
             res-=adj
          if abs(res)>tol:
             idxs=np.where(room>tol)[0]
             if idxs.size:
                h[idxs]+=res/idxs.size
                h=np.clip(h,0,1)
    h=np.clip(h,0,1)
    res=target - float(np.sum(h))
    if abs(res)>1e-12:
       free=np.where((h>1e-15)&(h<1.0-1e-15))[0]
       if free.size:
          h[free[0]] += res
          h=np.clip(h,0,1)
       else:
          # last resort choose any with room
          if res>0: idx=int(np.argmax(1-h))
          else: idx=int(np.argmax(h))
          h[idx]+=res
          h=np.clip(h,0,1)
    return h

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)


def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    n = x.size
    target = float(target)
    if n == 0:
        return x

    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)

    s = float(np.sum(x))
    if abs(s - target) <= 1e-13 * max(1, n):
        return x

    lo, hi = -1.0, 1.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        y = np.clip(x + mid, 0.0, 1.0)
        sy = float(np.sum(y))
        if sy < target:
            lo = mid
        else:
            hi = mid

    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))

    if abs(res) > 1e-14:
        tol = 1e-12
        if res > 0.0:
            space = 1.0 - h
            idxs = np.where(space > tol)[0]
            if idxs.size:
                idx = int(idxs[np.argmax(space[idxs])])
                adj = min(res, space[idx])
                h[idx] += adj
                res -= adj
            if abs(res) > tol:
                idxs = np.where(space > tol)[0]
                if idxs.size:
                    h[idxs] += res / float(idxs.size)
        else:
            room = h
            idxs = np.where(room > tol)[0]
            if idxs.size:
                idx = int(idxs[np.argmax(room[idxs])])
                adj = max(res, -room[idx])
                h[idx] += adj
                res -= adj
            if abs(res) > tol:
                idxs = np.where(room > tol)[0]
                if idxs.size:
                    h[idxs] += res / float(idxs.size)

    h = np.clip(h, 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-13:
        free = np.where((h > 1e-14) & (h < 1.0 - 1e-14))[0]
        if free.size:
            h[free[0]] += res
            h = np.clip(h, 0.0, 1.0)
        else:
            if res > 0.0:
                idx = int(np.argmax(1.0 - h))
            else:
                idx = int(np.argmax(h))
            h[idx] += res
            h = np.clip(h, 0.0, 1.0)

    return h


def vector_close(a, b):

def upsample_step(parent, n):

def active_gradient(h, indices, weights=None):
    h = np.asarray(h, dtype=float)
    n = h.size
    g = np.zeros(n, dtype=float)
    if n == 0 or len(indices) == 0:
        return g

    center = n - 1
    if weights is None:
        wscale = 1.0 / float(len(indices))
    else:
        warr = np.asarray(weights, dtype=float)

    for ii, t in enumerate(indices):
        if weights is None:
            wt = wscale
        else:
            wt = float(warr[ii])
        if wt == 0.0:
            continue
        t = int(t)
        k = t - center
        if k == 0:
            g += wt * (1.0 - 2.0 * h)
        elif k > 0:
            ell = k
            g[ell:] += wt * (1.0 - h[:n - ell])
            g[:n - ell] -= wt * h[ell:]
        else:
            ell = -k
            g[:n - ell] += wt * (1.0 - h[ell:])
            g[ell:] -= wt * h[:n - ell]

    return g


def pair_linear_delta(h, i, j):
    n = h.size
    center = n - 1
    L = np.zeros(2 * n - 1, dtype=float)
    a = h
    br = (1.0 - a)[::-1]

    L[i:i + n] += br
    L[center - i:center - i + n] -= a

    L[j:j + n] -= br
    L[center - j:center - j + n] += a

    return L


def search_delta(corr, L, d, cap, n, current_best):
    if cap <= 1e-13:
        return None, float(current_best)

    center = n - 1
    Lsz = corr.size
    fracs = (1.0, 0.99, 0.95, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.22, 0.15, 0.08, 0.04, 0.02, 0.01)
    tried = []
    best_delta = None
    best_val = float(current_best)

    for ff in fracs:
        delta = float(cap) * float(ff)
        if delta <= 1e-12:
            continue
        if delta >= cap - 1e-14:
            delta = float(cap)
        dup = False
        for td in tried:
            if abs(delta - td) < 1e-13 * max(1.0, cap):
                dup = True
                break
        if dup:
            continue
        tried.append(delta)

        v = corr + delta * L
        if d != 0:
            d2 = delta * delta
            ip = center + d
            im = center - d
            if 0 <= ip < Lsz:
                v[ip] += d2
            if 0 <= im < Lsz:
                v[im] += d2
            v[center] -= 2.0 * d2

        m = float(np.max(v))
        if m < best_val - 1e-12:
            best_val = m
            best_delta = float(delta)

    if best_delta is None:
        return None, float(current_best)
    return best_delta, float(best_val)

        if len(res) >= count:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        t1, t2 = (i, j) if i < j else (j, i)
        t = (t1, t2)
        if t in seen:
            continue
        seen.add(t)
        res.append((i, j))
    return res


def local_pair_fast(h, deadline, rng, active_k=8, top_pair=4, max_iter=2000):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 1:
        c = evaluate_erdos(h)
        return h, c

    target = 0.5 * float(n)
    h = project_density(h, target)
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_raw = float(np.max(corr))

    full_counter = 0
    stall = 0
    iter_no = 0

    while iter_no < int(max_iter) and time.perf_counter() < float(deadline):
        if full_counter % 80 == 0:
            corr = np.correlate(h, 1.0 - h, mode="full")
            best_raw = float(np.max(corr))
            full_counter = 0
        else:
            full_counter += 1

        if corr.size <= int(active_k):
            active_idx = np.arange(corr.size)
        else:
            ak = min(int(active_k), corr.size)
            active_idx = np.argpartition(corr, -ak)[-ak:]

        g = active_gradient(h, active_idx, None)
        spread = float(np.max(g) - np.min(g))

        pairs = []
        if spread > 1e-12 and top_pair > 0:
            lows = np.argsort(g)[:top_pair]
            highs = np.argsort(-g)[:top_pair]
            for ii in lows:
                for jj in highs:
                    ii = int(ii)
                    jj = int(jj)
                    if ii != jj:
                        pairs.append((ii, jj))
        pairs.extend(random_pairs(n, rng, 4))

        found = False
        a_now = h
        for i, j in pairs:
            if time.perf_counter() >= float(deadline):
                break
            if i == j:
                continue

            cap_pos = min(1.0 - float(h[i]), float(h[j]))
            cap_neg = min(float(h[i]), 1.0 - float(h[j]))
            if cap_pos <= 1e-11 and cap_neg <= 1e-11:
                continue

            d = abs(i - j)
            L = pair_linear_delta(a_now, i, j)

            if cap_pos > 1e-11:
                cap_pos *= (1.0 - 1e-13)
                best_del, mval = search_delta(corr, L, d, cap_pos, n, best_raw)
                if best_del is not None and mval < best_raw - 1e-12:
                    h[i] = float(h[i]) + float(best_del)
                    h[j] = float(h[j]) - float(best_del)
                    if h[i] < 0.0 or h[i] > 1.0 or h[j] < 0.0 or h[j] > 1.0:
                        h = project_density(h, target)
                    corr = np.correlate(h, 1.0 - h, mode="full")
                    best_raw = float(np.max(corr))
                    found = True
                    break

            if cap_neg > 1e-11:
                cap_neg *= (1.0 - 1e-13)
                best_del, mval = search_delta(corr, -L, d, cap_neg, n, best_raw)
                if best_del is not None and mval < best_raw - 1e-12:
                    h[i] = float(h[i]) - float(best_del)
                    h[j] = float(h[j]) + float(best_del)
                    if h[i] < 0.0 or h[i] > 1.0 or h[j] < 0.0 or h[j] > 1.0:
                        h = project_density(h, target)
                    corr = np.correlate(h, 1.0 - h, mode="full")
                    best_raw = float(np.max(corr))
                    found = True
                    break

        if not found and iter_no % 6 == 0 and float(deadline) - time.perf_counter() > 2.0:
            dvec = -g
            dvec = dvec - float(np.mean(dvec))
            normd = float(np.linalg.norm(dvec))
            if normd > 1e-12:
                dvec /= normd
                for step in (0.03, 0.08, 0.16, 0.28):
                    cand = project_density(h + step * dvec, target)
                    ccorr = np.correlate(cand, 1.0 - cand, mode="full")
                    cm = float(np.max(ccorr))
                    if cm < best_raw - 1e-12:
                        h = cand
                        corr = ccorr
                        best_raw = cm
                        found = True
                        break

        if found:
            stall = 0
        else:
            stall += 1
            if stall >= 90:
                if float(deadline) - time.perf_counter() > 5.0:
                    eps = 0.05 if n <= 220 else 0.025
                    cand = project_density(h + eps * (rng.random(n) - 0.5), target)
                    ccand = evaluate_erdos(cand)
                    cur_c = 2.0 * best_raw / float(n)
                    if ccand < cur_c + 0.004:
                        h = cand
                        corr = np.correlate(h, 1.0 - h, mode="full")
                        best_raw = float(np.max(corr))
                        stall = 0
                        iter_no += 1
                        continue
                break

        iter_no += 1

    h = project_density(h, target)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = float(np.max(corr) * (2.0 / float(n)))
    return h, float(c)


def pen_objective(x):
    x = np.asarray(x, dtype=float)
    n = _NC
    if n <= 1:
        return 1e100

    if not np.all(np.isfinite(x)):
        return 1e100
    x = np.clip(x, 0.0, 1.0)

    corr = np.correlate(x, 1.0 - x, mode="full")
    if corr.size == 0 or not np.all(np.isfinite(corr)):
        return 1e100

    m = float(np.max(corr))
    y = _ALPHA * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return 1e100

    f = (2.0 / float(n)) * (m + np.log(z) / _ALPHA)
    s = float(np.sum(x) - _TARGET)
    return float(f + 0.5 * _RHO * s * s)


def pen_jacobian(x):
    x = np.asarray(x, dtype=float)
    n = _NC
    if n <= 1:
        return np.zeros(max(1, n), dtype=float)

    if not np.all(np.isfinite(x)):
        return np.zeros(n, dtype=float)
    x = np.clip(x, 0.0, 1.0)

    corr = np.correlate(x, 1.0 - x, mode="full")
    if corr.size == 0 or not np.all(np.isfinite(corr)):
        return np.zeros(n, dtype=float)

    m = float(np.max(corr))
    y = _ALPHA * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return np.zeros(n, dtype=float)

    w = e / z
    idx = np.where(w > 1e-11)[0]
    if idx.size == 0:
        idx = np.argmax(w)
        idx = np.array([idx], dtype=int)

    gg = active_gradient(x, idx, w[idx])
    s = float(np.sum(x) - _TARGET)
    gg += _RHO * s
    return (2.0 / float(n)) * gg


def time_callback(xk):
    return time.perf_counter() >= float(_LB_DEADLINE)


def smooth_lgbfgs(h, n, deadline):
    h = project_density(h, 0.5 * float(n))
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)

    if n <= 130:
        alpha_rho_iter = [
            (8.0, 1.0e4, 280),
            (30.0, 1.0e5, 220),
            (90.0, 1.0e6, 180),
        ]
    elif n <= 300:
        alpha_rho_iter = [
            (10.0, 1.0e5, 190),
            (40.0, 1.0e6, 150),
            (100.0, 1.0e7, 110),
        ]
    else:
        alpha_rho_iter = [
            (12.0, 1.0e6, 130),
            (50.0, 1.0e7, 95),
        ]

    bounds = [(0.0, 1.0)] * int(n)

    for alpha, rho, maxiter in alpha_rho_iter:
        if time.perf_counter() >= float(deadline) - 2.0:
            break

        set_weight_context(n, alpha, 0.5 * float(n), rho)
        global _LB_DEADLINE
        _LB_DEADLINE = float(deadline)

        start_h = project_density(h, 0.5 * float(n))
        try:
            res = minimize(
                pen_objective,
                start_h,
                method="L-BFGS-B",
                jac=pen_jacobian,
                bounds=bounds,
                options={
                    "maxiter": int(maxiter),
                    "ftol": 1e-9,
                    "gtol": 1e-4,
                    "maxcor": 20,
                    "maxls": 30,
                },
                callback=time_callback,
            )
            xg = getattr(res, "x", None)
            if xg is not None:
                xg = np.asarray(xg, dtype=float)
                if np.all(np.isfinite(xg)):
                    cand = project_density(xg, 0.5 * float(n))
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-12:
                        best_h = cand
                        best_c = c
                    if c <= best_c + 0.03:
                        h = cand
                    else:
                        h = best_h
        except Exception:
            h = best_h

    best_h = project_density(best_h, 0.5 * float(n))
    best_c = evaluate_erdos(best_h)
    return best_h, float(best_c)


def optimize_one(h0, n, sub_deadline, rng):
    n = int(n)
    h = project_density(h0, 0.5 * float(n))
    best_c = evaluate_erdos(h)
    best_h = h.copy()

    now = time.perf_counter()
    rem = float(sub_deadline) - now

    if 2 <= n <= 600 and rem > 90.0:
        lbfg_time = min(0.45 * rem, 135.0)
        if lbfg_time > 25.0:
            lbfg_deadline = now + lbfg_time
            h2, c2 = smooth_lgbfgs(h, n, lbfg_deadline)
            if c2 < best_c - 1e-12:
                best_h = h2
                best_c = c2
            h = best_h

    now = time.perf_counter()
    if float(sub_deadline) - now > 12.0:
        if n <= 160:
            ak, tp, mi = 9, 4, 3000
        elif n <= 320:
            ak, tp, mi = 11, 3, 1600
        else:
            ak, tp, mi = 13, 2, 750

        h3, c3 = local_pair_fast(h, float(sub_deadline), rng, active_k=ak, top_pair=tp, max_iter=mi)
        if c3 < best_c - 1e-12:
            best_h = h3
            best_c = c3

    best_h = project_density(best_h, 0.5 * float(n))
    best_c = evaluate_erdos(best_h)
    return best_h, float(best_c)


def tuple_score(t):
    return t[1]


def build_starts(n, rng, parent, best_h, max_starts):
    n = int(n)
    raw_starts = []

    if parent is not None:
        raw_starts.append(("parent", upsample_step(parent, n)))

    if best_h is not None:
        if best_h.size == n:
            raw_starts.append(("best", np.asarray(best_h, dtype=float).copy()))
        else:
            raw_starts.append(("best", upsample_step(best_h, n)))

    extra_count = max(1, min(5, max_starts * 2))
    kinds = [int(k) for k in rng.permutation(6)]
    for kind in kinds[:extra_count]:
        h = make_candidate(n, rng, kind, parent)
        raw_starts.append(("rand", h))

    scored = []
    for tag, hv in raw_starts:
        hp = project_density(hv, 0.5 * float(n))
        c = evaluate_erdos(hp)
        scored.append((tag, float(c), hp))

    scored.sort(key=tuple_score)

    final_starts = []

    def add_unique(tag_h):
        tag, c, h = tag_h
        for _, _, u in final_starts:
            if vector_close(h, u):
                return
        final_starts.append((tag, c, h))

    for item in scored:
        if item[0] == "parent":
            if len(final_starts) < max_starts:
                add_unique(item)
            break

    for item in scored:
        if item[0] == "best":
            if len(final_starts) < max_starts:
                add_unique(item)
            break

    for item in scored:
        if len(final_starts) >= max_starts:
            break
        add_unique(item)

    return [h for _, _, h in final_starts]


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.perf_counter()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if not np.isfinite(budget) or budget <= 0:
        budget = 1300.0

    deadline = start_time + max(1.0, budget - 8.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")

    parent = None
    try:
        raw = initial_h_values
        raw = np.asarray(raw, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            p = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0)
            p = np.clip(p, 0.0, 1.0)
            parent = p.copy()
            tp = 0.5 * float(parent.size)
            pp = project_density(parent, tp)
            cp = evaluate_erdos(pp)
            if cp < best_c:
                best_h = pp
                best_c = cp
    except Exception:
        parent = None

    if best_h is None:
        n0 = 90
        best_h = make_candidate(n0, rng, 0, None)
        best_c = evaluate_erdos(best_h)

    base_n = parent.size if parent is not None else int(best_h.size)
    if base_n < 2:
        base_n = 90

    nset = set()
    for mult in (1, 2, 3, 4, 6, 8):
        nn = int(base_n * mult)
        if 2 <= nn <= 720:
            nset.add(nn)
    for nn in (90, 120, 180, 240, 360, 540):
        if 2 <= nn <= 720:
            nset.add(nn)

    nlist = sorted(int(x) for x in nset if 2 <= int(x) <= 720)

    final_check = project_density(best_h, 0.5 * float(best_h.size))
    final_c = evaluate_erdos(final_check)
    if final_c <= _TARGET_C:
        return final_check, float(final_c), int(final_check.size)

    for n in nlist:
        now = time.perf_counter()
        if float(deadline) - now < 55.0:
            break

        if n <= 150:
            max_starts = 3
            per_cap = 150.0
        elif n <= 320:
            max_starts = 2
            per_cap = 115.0
        else:
            max_starts = 1
            per_cap = 90.0

        starts = build_starts(n, rng, parent, best_h, max_starts)
        if not starts:
            continue

        for si, st in enumerate(starts):
            now = time.perf_counter()
            rem = float(deadline) - now
            if rem < 25.0:
                break

            left = max(1, len(starts) - si)
            per = min(rem / float(left), per_cap)
            per = max(14.0, per)
            sub_deadline = time.perf_counter() + min(per, rem - 4.0)

            h_out, c_out = optimize_one(st, n, sub_deadline, rng)
            if h_out is not None:
                h_out = project_density(h_out, 0.5 * float(h_out.size))
                c_out = evaluate_erdos(h_out)
                if c_out < best_c - 1e-12:
                    best_h = h_out
                    best_c = c_out

            final_check = project_density(best_h, 0.5 * float(best_h.size))
            final_c = evaluate_erdos(final_check)
            if final_c <= _TARGET_C:
                return final_check, float(final_c), int(final_check.size)

    now = time.perf_counter()
    if best_h is not None and float(deadline) - now > 25.0:
        if best_h.size <= 300:
            ak, tp, mi = 10, 4, 2500
        elif best_h.size <= 500:
            ak, tp, mi = 12, 3, 1200
        else:
            ak, tp, mi = 14, 2, 600
        h_fin, c_fin = local_pair_fast(best_h, float(deadline), rng, active_k=ak, top_pair=tp, max_iter=mi)
        if c_fin < best_c - 1e-12:
            best_h = h_fin
            best_c = c_fin

    final_h = project_density(best_h, 0.5 * float(best_h.size))
    final_c = evaluate_erdos(final_h)
    if final_c >= best_c:
        final_h = project_density(best_h, 0.5 * float(best_h.size))
        final_c = evaluate_erdos(final_h)

    return final_h, float(final_c), int(final_h.size)
