import time
import numpy as np
from scipy.optimize import minimize

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)

def project_density(x, target):
    # Project x onto {0 <= x <= 1, sum(x) = target}
    n = x.size
    if n == 0:
        return x.copy()
    x = np.clip(x, 0.0, 1.0)
    s = float(np.sum(x))
    if abs(s - target) < 1e-12:
        return x
    
    # Bisection for the shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-14:
        free = np.where((h > 1e-15) & (h < 1.0 - 1e-15))[0]
        if free.size > 0:
            h[free] += res / float(free.size)
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n:
        return p.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)

def make_random_binary(n, rng):
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n//2, replace=False)
    h[ones] = 1.0
    return h

def make_fourier_smooth(n, rng, max_freq=8):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.full(n, 0.5, dtype=float)
    for m in range(1, max_freq + 1):
        amp = rng.uniform(0.05, 0.3) * (0.6 ** m)
        freq = m
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
        phase2 = rng.uniform(0, 2 * np.pi)
        h += amp * 0.5 * np.cos(2 * np.pi * freq * x / 2.0 + phase2)
    return project_density(h, 0.5 * n)

def pair_descent(h, n, target, active_k, max_iter, deadline):
    best = h.copy()
    best_c = evaluate_erdos(best)
    corr = np.correlate(best, 1-best, mode="full")
    center = n - 1
    
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        
        # Compute gradient of max approximated by top-k average
        idxs = np.argsort(corr)[-active_k:]
        g = np.zeros(n)
        for t in idxs:
            k = int(t) - center
            if k == 0:
                g += (1 - 2*best)
            elif k > 0:
                # d/dh[j] = (1-h[j+k]) for j in [0, n-1-k] - h[j-k] for j in [k, n-1]
                g[:n-k] += (1 - best[k:])
                g[k:] -= best[:n-k]
            else:
                k2 = -k
                g[k2:] += (1 - best[:n-k2])
                g[:n-k2] -= best[k2:]
        g /= active_k
        
        # Find best pair
        min_idx = int(np.argmin(g))
        max_idx = int(np.argmax(g))
        if g[min_idx] >= g[max_idx] - 1e-12:
            break
        
        # Line search
        cap = min(best[min_idx], 1.0 - best[max_idx])
        if cap < 1e-12:
            # Try next best pair
            g_sorted = np.argsort(g)
            found = False
            for a in g_sorted[:10]:
                for b in g_sorted[-10:]:
                    if a == b: continue
                    if g[a] >= g[b]: continue
                    cap2 = min(best[a], 1.0 - best[b])
                    if cap2 < 1e-12: continue
                    a, b, cap = int(a), int(b), cap2
                    found = True
                    break
                if found: break
            if not found:
                break
        
        # Try several step sizes
        improved = False
        for frac in (0.5, 0.25, 0.1, 0.05, 0.01):
            step = frac * cap
            if step < 1e-12: continue
            h_new = best.copy()
            h_new[min_idx] += step
            h_new[max_idx] -= step
            # (or h_new[a] += step, h_new[b] -= step if we found alternative)
            c_new = evaluate_erdos(h_new)
            if c_new < best_c - 1e-13:
                best = h_new
                best_c = c_new
                corr = np.correlate(best, 1-best, mode="full")
                improved = True
                break
        
        if not improved:
            # Try a few more pairs
            g_sorted_min = np.argsort(g)[:5]
            g_sorted_max = np.argsort(-g)[:5]
            found_any = False

                        step = frac * cap2

import time
import numpy as np
from scipy.optimize import minimize

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)

def project_density(x, target):
    n = x.size
    if n == 0:
        return x.copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    s = float(np.sum(x))
    if abs(s - target) < 1e-14:
        return x
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-14:
        free = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if free.size > 0:
            h[free] += res / float(free.size)
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(n, 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)

def make_random_init(n, rng):
    mode = int(rng.integers(0, 4))
    if mode == 0:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, size=n // 2, replace=False)
        h[ones] = 1.0
    elif mode == 1:
        x = (np.arange(n) + 0.5) * (2.0 / float(n))
        h = np.full(n, 0.5)
        for m in range(1, 7):
            amp = rng.uniform(0.05, 0.3) * (0.65 ** m)
            freq = int(m * rng.uniform(0.5, 1.5)) + 1
            phase = rng.uniform(0, 2 * np.pi)
            h += amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    elif mode == 2:
        h = np.empty(n, dtype=float)
        pos = 0
        levels = [0.0, 0.25, 0.5, 0.75, 1.0]
        while pos < n:
            rem = n - pos
            length = int(rng.integers(1, max(2, rem + 1)))
            val = float(rng.choice(levels))
            h[pos:pos + length] = val
            pos += length
    else:
        u = rng.random(n)
        thresh = float(np.median(u))
        h = (u > thresh).astype(float)
    return project_density(h, 0.5 * n)

def compute_active_gradient(h, active_indices):
    n = h.size
    center = n - 1
    g = np.zeros(n)
    for t in active_indices:
        t = int(t)
        k = t - center
        if k == 0:
            g += (1.0 - 2.0 * h)
        elif k > 0:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
        else:
            k2 = -k
            g[k2:] += (1.0 - h[:n - k2])
            g[:n - k2] -= h[k2:]
    return g / float(len(active_indices))

def pair_descent(h, deadline, max_iter=5000, active_k=5, search_width=5):
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    corr = np.correlate(best, 1.0 - best, mode="full")
    stall = 0

    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break

        if not np.all(np.isfinite(corr)):
            break

        idxs = np.argsort(corr)[-active_k:]
        g = compute_active_gradient(best, idxs)

        if (float(np.max(g)) - float(np.min(g))) < 1e-12:
            break

        neg_order = np.argsort(g)[:search_width]
        pos_order = np.argsort(-g)[:search_width]

        found = False
        for a in neg_order:
            a = int(a)
            for b in pos_order:
                b = int(b)
                if a == b:
                    continue
                if g[a] >= g[b] - 1e-12:
                    continue
                cap = min(best[a], 1.0 - best[b])
                if cap < 1e-12:
                    continue
                for frac in (0.4, 0.2, 0.1, 0.05, 0.02):
                    step = frac * cap
                    if step < 1e-14:
                        continue
                    h_new = best.copy()
                    h_new[a] += step
                    h_new[b] -= step
                    c_new = evaluate_erdos(h_new)
                    if c_new < best_c - 1e-14:
                        best = h_new
                        best_c = c_new
                        corr = np.correlate(best, 1.0 - best, mode="full")
                        found = True
                        break
                if found:
                    break
            if found:
                break

        if not found:

            d = -g
            d -= float(np.mean(d))
            norm = float(np.linalg.norm(d))
            if norm > 1e-13:
                d /= norm
                improved = False
                for step in (0.05, 0.1, 0.2, 0.4, 0.7):
                    cand = project_density(best + step * d, target)

                        best = cand
                        best_c = c_new
                        corr = np.correlate(best, 1.0 - best, mode="full")
                        improved = True
                        break
                if improved:
                    found = True

        if found:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break

    best = project_density(best, target)
    best_c = evaluate_erdos(best)
    return best, best_c

def slsqp_smooth(n):
    h = np.asarray(h, dtype=float)
    if n < 2:
        return 1e100
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    alpha = 40.0
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0 or not np.isfinite(z):
        return 1e100
    return float((2.0 / n) * (m + np.log(z) / alpha))

def slsqp_jacobian(h):
    n = h.size
    if n < 2:
        return np.zeros(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    alpha = 40.0
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0 or not np.isfinite(z):
        return np.zeros(n)
    w = e / z
    idx = np.where(w > 1e-10)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(w))])
    g = np.zeros(n)
    center = n - 1
    for t in idx:
        t = int(t)
        p = float(w[t])
        k = t - center
        if k == 0:
            g += p * (1.0 - 2.0 * h)
        elif k > 0:
            g[:n - k] += p * (1.0 - h[k:])
            g[k:] -= p * h[:n - k]
        else:
            k2 = -k
            g[k2:] += p * (1.0 - h[:n - k2])
            g[:n - k2] -= p * h[k2:]
    return (2.0 / n) * g

    except Exception:

def multi_start_optimize(n, starts, deadline):
