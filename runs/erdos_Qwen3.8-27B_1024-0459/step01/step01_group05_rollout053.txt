import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_val(x, beta):
    m = np.max(x)
    ez = np.exp(np.clip(beta * (x - m), -700, 700))
    return m + np.log(np.sum(ez)) / beta

def smooth_max_w(x, beta):
    m = np.max(x)
    ez = np.exp(np.clip(beta * (x - m), -700, 700))
    return ez / np.sum(ez)

def grad_smooth_max(h, beta):
    """Analytic gradient of smooth_max(corr(h), beta) w.r.t. h"""
    n = len(h)
    corr = compute_corr(h)
    w = smooth_max_w(corr, beta)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-12:
            continue
        s = idx - (n - 1)
        if s >= 0:
            grad[s:] += wi * (1.0 - h[:n - s])
            grad[:n - s] -= wi * h[s:]
        else:
            t = -s
            grad[:n - t] += wi * (1.0 - h[t:])
            grad[t:] -= wi * h[:n - t]
    return grad

def project_sum(h, target):
    """Project h onto {sum(h)=target, 0<=h<=1}"""
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def proj_grad_optimize(h0, beta, target, max_steps=3000, step0=0.5, tol=1e-13):
    """Projected gradient descent with analytic gradient and line search"""
    n = len(h0)
    h = project_sum(h0.copy(), target)
    f_curr = smooth_max_val(compute_corr(h), beta)
    best_h = h.copy()
    best_f = f_curr
    alpha = step0

    for step in range(max_steps):
        g = grad_smooth_max(h, beta)
        g_dir = g - np.mean(g)
        gnorm = np.linalg.norm(g_dir)
        if gnorm < 1e-14:
            break

        improved = False
        for ls in range(15):
            h_try = h - alpha * g_dir
            h_try = project_sum(h_try, target)
            f_try = smooth_max_val(compute_corr(h_try), beta)
            if f_try < f_curr - 1e-15:
                h = h_try
                f_curr = f_try
                improved = True
                if f_curr < best_f:
                    best_f = f_curr
                    best_h = h_try.copy()
                break
            alpha *= 0.5

        if not improved:
            alpha = alpha * 2.0
            if alpha > step0:
                alpha = step0
            if step > 200 and abs(best_f - f_curr) < tol:
                break
            if step > 500:
                break

    return best_h, best_f

def binary_start(n, seed_val):
    """Create a random binary start with n/2 ones"""
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return project_sum(h, n / 2.0)

def sine_start(n, freq, phase):
    """Smooth parametric start"""
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.5 * np.sin(freq * np.pi * x + phase)
    return project_sum(h, n / 2.0)

def threshold_push(h, target, push=0.1):
    """Push values toward 0 or 1"""
    h2 = h.copy()
    h2 = np.where(h2 < 0.5 - push, h2 * 0.5, h2)
    h2 = np.where(h2 > 0.5 + push, 1.0 - (1.0 - h2) * 0.5, h2)
    return project_sum(h2, target)
