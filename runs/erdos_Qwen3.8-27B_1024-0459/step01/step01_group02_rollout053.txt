import numpy as np
from scipy.optimize import minimize
import time

_target_sum_global = 0.0
_obj_n_global = 0
_obj_beta_global = 50.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr_dx(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = (h < 1.0 - 1e-12)
        else:
            mask = (h > 1e-12)
        count = np.sum(mask)
        if count == 0:
            h = target / n * np.ones(n)
            break
        h[mask] += diff / count
        h = np.clip(h, 0.0, 1.0)
    return h

def constraint_fun(x):
    return np.sum(x) - _target_sum_global

def smooth_max_obj(x):
    n = _obj_n_global
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    beta = _obj_beta_global
    shifted = beta * (corr - m)
    shifted[shifted < -500] = -500
    exps = np.exp(shifted)
    return m + np.log(np.sum(exps)) / beta

def set_binary_sum(h, n, rng):
    n_want = n // 2
    n_ones = int(np.sum(h > 0.5))
    if n_ones > n_want:
        idx = np.where(h > 0.5)[0]
        flip = rng.choice(idx, n_ones - n_want, replace=False)
        h[flip] = 0.0
        h[h > 0.5] = 1.0
    elif n_ones < n_want:
        idx = np.where(h <= 0.5)[0]
        flip = rng.choice(idx, n_want - n_ones, replace=False)
        h[flip] = 1.0
        h[h > 0.5] = 1.0
    else:
        h[h > 0.5] = 1.0
        h[h <= 0.5] = 0.0
    return h

def slsqp_refine(h, n, beta, maxiter, time_limit):

    _obj_n_global = n
    _obj_beta_global = beta

        h_out = h.copy()
    return h_out

def simulated_annealing_binary(h, n, n_iter, T_init, rng, time_limit):
    t_start = time.time()
    h = h.copy()
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    ones_idx = np.where(h == 1.0)[0]
    zeros_idx = np.where(h == 0.0)[0]
    
    for it in range(n_iter):
        if time.time() - t_start > time_limit:
            break
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = ones_idx[rng.randint(len(ones_idx))]
