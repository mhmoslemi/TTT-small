from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    N = 2 * n
    H = fft(h, N)
    NH = fft(1.0 - h, N)
    corr = np.real(ifft(H * np.conj(NH)))
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def rand_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    return h

def binary_sa(n, h_init, t_start, t_end, n_steps, rng, time_limit=None, t_start_time=None):
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5_nt(h, n)
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for step in range(n_steps):
        if t_start_time is not None and time.time() - t_start_time > time_limit:
            break
        T = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        
        # Pick random one and zero to swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        if i == j:

        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5_nt(h, n)

            h[j] = 0.0
            continue
        
        # Update ones/zeros

        zeros = np.where(h == 0.0)[0]

    except Exception:
        pass
    return None, None

def softmax_objective(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    return cmax + (1.0 / beta) * np.log(np.sum(np.exp(beta * (corr - cmax))))

def smooth_anneal_solve(n, h_init, betas=(20, 80, 400, 2000, 8000)):
    current = h_init.copy()
    best_c5 = compute_c5(current)
    best_h = current.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    for beta in betas:
        def obj_s(x):
            return softmax_objective(x, n, beta)
        try:
            res = minimize(obj_s, current, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-14, 'iprint': 0})
            if res.fun < 1e10:
                current = make_feasible(res.x, n)
                c5 = compute_c5(current)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current.copy()
        except Exception:
            pass
    return best_h, best_c5

def pair_swap_local(h, n, n_rounds, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_rounds):
        improved = False
        for i in range(n - 1):
            if h[i] <= h[i+1]:
                continue
            for step in [0.01, 0.02, 0.05, 0.1]:
                h_try = h.copy()
                amt = min(step, (h[i] - h[i+1]) / 2.0)
                if amt < 1e-6:
                    continue
                h_try[i] -= amt
                h_try[i+1] += amt
                h_try = np.clip(h_try, 0.0, 1.0)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def random_perturb_improve(h, n, n_trials, base_step, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    step = base_step
    for _ in range(n_trials):
        h_try = h.copy()
        n_mod = rng.randint(1, 7)
        idxs = rng.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += rng.uniform(-step, step)
        h_try = np.clip(h_try, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h_try)
        mask = (h_try > 0.005) & (h_try < 0.995)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h_try[mask] += diff / cnt
        h_try = np.clip(h_try, 0.0, 1.0)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            h = h_try
            best_c5 = c5_try
        elif rng.random() < 0.1:
            step *= 0.9
    return h, best_c5

def binary_sa(n, rng, n_steps=40000, t_start=1.0, t_end=0.005):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(n_steps):
        T = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
        elif rng.random() < np.exp(-(c5_new - best_c5) / max(T, 1e-15)):
            pass
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5

def structured_init(n, rng):
    inits = []
    if n % 4 == 0:
        h1 = np.tile([1.0, 1.0, 0.0, 0.0], n // 4)
        inits.append(h1.copy())
    if n % 2 == 0:
        h2 = np.tile([1.0, 0.0], n // 2)
        inits.append(h2.copy())
    # Wiedemann sequence approximation
    h3 = np.zeros(n)
    for i in range(n):
        h3[i] = 1.0 if (i * (i + 1) // 2) % n < n // 2 else 0.0
    inits.append(make_feasible(h3, n))
    # Random
    h4 = rng.rand(n)
    inits.append(make_feasible(h4, n))
    return inits

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is None or len(h_parent) == 0:
            h_parent = None
    except Exception:
        h_parent = None

    if h_parent is not None:
        best_h = h_parent.copy()
        best_n = len(best_h)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5

    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP direct on parent
    if remaining() > 120:
        t0 = time.time()
        h_opt, c5_opt = slsqp_run(best_n, best_h, max_iter=4000, ftol=1e-15)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Smooth annealing from parent
    if remaining() > 120:
        h_opt, c5_opt = smooth_anneal_solve(best_n, best_h, betas=(30, 150, 800, 5000))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 3: Pair-swap local (exploits sum constraint naturally)
    if remaining() > 30:
        n_rounds = min(80, int(remaining() / 0.5))
        best_h, best_c5 = pair_swap_local(best_h, best_n, n_rounds, rng)

    # Phase 4: Random perturbation improvement
    if remaining() > 30:
        n_trials = min(15000, int((remaining() - 10) / 0.012))
        best_h, best_c5 = random_perturb_improve(best_h, best_n, n_trials, 0.04, rng)

    # Phase 5: Binary SA at several resolutions
    for n_try in [best_n, 120, 140, 160, 180, 200, 256]:
        if remaining() < 25:
            break
        if n_try < 40:
            continue
        if n_try % 2 != 0:
            n_try += 1
        h_bin, c5_bin = binary_sa(n_try, rng, n_steps=30000, t_start=0.8, t_end=0.003)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.astype(float)
            best_n = n_try

    # Phase 6: Resolution sweep with SLSQP
    for n_try in [140, 150, 160, 170, 180, 200, 240, 256]:
        if remaining() < 25:
            break
        h_init = make_feasible(interpolate_to_n(best_h, n_try), n_try)
        h_opt, c5_opt = slsqp_run(n_try, h_init, max_iter=2500, ftol=1e-14)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_try

    # Phase 7: Multi-start SLSQP from perturbations
    for trial in range(5):
        if remaining() < 20:
            break
        h_try = best_h + rng.randn(best_n) * 0.02
        h_try = make_feasible(h_try, best_n)
        h_opt, c5_opt = slsqp_run(best_n, h_try, max_iter=2000, ftol=1e-14)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 8: Final aggressive polish
    if remaining() > 20:
        h_opt, c5_opt = slsqp_run(best_n, best_h, max_iter=6000, ftol=1e-16)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    if remaining() > 15:
        n_rounds = min(50, int(remaining() / 0.5))
        best_h, best_c5 = pair_swap_local(best_h, best_n, n_rounds, rng)

    if remaining() > 10:
        n_trials = min(20000, int((remaining() - 5) / 0.008))
        best_h, best_c5 = random_perturb_improve(best_h, best_n, n_trials, 0.015, rng)

    # Final feasibility check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1 + 1e-12)
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-8

    return (best_h, best_c5, best_n)
