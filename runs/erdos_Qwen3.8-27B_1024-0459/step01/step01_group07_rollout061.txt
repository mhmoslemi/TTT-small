import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_fft(h, n):
    """Compute C5 using FFT for speed."""
    dx = 2.0 / n
    # Use np.correlate for correctness (fast enough for n <= 500)
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def compute_c5_and_argmax(h, n):
    """Compute C5 and the argmax lag."""
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    k_max = np.argmax(conv)
    return conv[k_max] * dx, k_max

def compute_gradient(h, n, k_max):
    """Compute the gradient of the C5 objective at the argmax lag."""
    dx = 2.0 / n
    grad = np.zeros(n)
    # k_max is index in the full correlation output (length 2n-1)
    # Center is at n-1
    m = k_max - (n - 1)
    if m >= 0:
        # corr = sum_{j=0}^{n-1-m} h[j+m] * (1-h[j])
        # d/dh[i] = (1-h[i-m]) for m <= i <= n-1
        #           - h[i+m] for 0 <= i <= n-1-m
        # Also there's a self-term if i = j and i = j+m simultaneously (when m=0)
        # For m >= 1: ranges don't overlap in a problematic way
        # For m = 0: corr = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
        #   d/dh[i] = 1 - 2*h[i]
        if m == 0:
            grad[:] = (1.0 - 2.0 * h) * dx
        else:
            if m < n:
                # (1-h[i-m]) for i in [m, n-1]
                grad[m:] += (1.0 - h[:n-m]) * dx
            # -h[i+m] for i in [0, n-1-m]
            if n - m > 0:
                grad[:n-m] += (-h[m:]) * dx
    else:
        p = -m
        # corr = sum_{j=p}^{n-1} h[j] * (1-h[j-p])
        # d/dh[i] = (1-h[i-p]) for p <= i <= n-1
        #           -h[i+p] for 0 <= i <= n-1-p
        if p == 0:
            grad[:] = (1.0 - 2.0 * h) * dx
        else:
            if p < n:
                grad[p:] += (1.0 - h[:n-p]) * dx
            if n - p > 0:
                grad[:n-p] += (-h[p:]) * dx
    return grad

def project_feasible(h, n):
    """Project h onto {x : 0<=x[i]<=1, sum(x)=n/2}."""
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0
            cnt = np.sum(free)
            if cnt == 0:
                break
            step = diff / cnt
            h[free] = np.minimum(h[free] + step, 1.0)
        else:
            free = h > 0.0
            cnt = np.sum(free)
            if cnt == 0:
                break
            step = diff / cnt
            h[free] = np.maximum(h[free] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - target) > 1e-10:

        h = np.clip(h, 0.0, 1.0)
    return h

def adam_optimize(h0, n, max_iter=5000, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, 
                  time_limit=None, t_start=None):
    """Adam optimizer with projection for the C5 problem."""
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5_fft(h, n)
    m_v = np.zeros(n)  # first moment
    v_v = np.zeros(n)  # second moment
    t = 0
    
    for it in range(max_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        
        # Compute gradient at current best lag
        val, k_max = compute_c5_and_argmax(h, n)
        grad = compute_gradient(h, n, k_max)
        
        # Adam update
        t += 1
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad * grad
        m_hat = m_v / (1 - beta1 ** t)
        v_hat = v_v / (1 - beta2 ** t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        
        # Evaluate
        new_val = compute_c5_fft(h_new, n)
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_val

def slsqp_optimize(h0, n, maxiter=300, ftol=1e-10):
    """SLSQP optimization."""
    target = n / 2.0
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return compute_c5_fft(x, n)
    
    def obj_grad(x):
        _, k_max = compute_c5_and_argmax(x, n)
        return compute_gradient(x, n, k_max)
    
    try:
        res = minimize(obj, h0.copy(), jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x, res.fun
    except Exception:
        return h0, None

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def compute_c5_argmax(h, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx, np.argmax(conv)

def compute_grad_at_lag(h, n, k_max):
    dx = 2.0 / n
    grad = np.zeros(n)
    m = k_max - (n - 1)
    if m == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif m > 0:
        if m < n:
            grad[m:] += (1.0 - h[:n-m]) * dx
        if n - m > 0:
            grad[:n-m] += (-h[m:]) * dx
    else:
        p = -m
        if n - p > 0:
            grad[:n-p] += (1.0 - h[p:]) * dx
        if p < n:
            grad[p:] += (-h[:n-p]) * dx
    return grad

def compute_avg_grad(h, n, num_lags=3):
    """Average gradient over top lags for smoother optimization."""
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    # Get top num_lags indices
    top_idx = np.argsort(-conv)[:num_lags]
    top_vals = conv[top_idx]
    # Weight by softmax
    weights = np.exp((top_vals - top_vals[0]) / (np.std(top_vals) + 1e-10))
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        grad += w * compute_grad_at_lag(h, n, k)
    return grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-12
            cnt = np.sum(free)
            if cnt == 0:
                h[free] = np.minimum(h[free] + diff / max(1, cnt), 1.0)
                break
            step = diff / cnt
            h[free] = np.minimum(h[free] + step, 1.0)
        else:
            free = h > 1e-12
            cnt = np.sum(free)
            if cnt == 0:
                break
            step = diff / cnt
            h[free] = np.maximum(h[free] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - target) > 1e-10:
        h += (target - np.sum(h)) / n
        h = np.clip(h, 0.0, 1.0)
    return h

                  eps=1e-8, time_limit=None, t_start=None, use_avg_grad=True):
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h, n)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    patience = 0
    patience_limit = 500
    
    for it in range(max_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break

            grad = compute_grad_at_lag(h, n, k_max)

        m_hat = m_v / (1 - beta1 ** t)
        v_hat = v_v / (1 - beta2 ** t)

def binary_local_search(h_bin, n, max_sweeps=100):
    """Local search on binary solutions using swap moves."""
    h = h_bin.copy().astype(float)
    best_val = compute_c5(h, n)
    target = n // 2  # exactly n/2 ones
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for sweep in range(max_sweeps):
        improved = False
        for i in list(ones):
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                val = compute_c5(h, n)
                if val < best_val:
                    best_val = val
                    ones.remove(i)
                    ones.append(j)
                    zeros.remove(j)
                    zeros.append(i)
                    improved = True
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        if not improved:
            break
    
    return h, best_val

        return compute_c5(x, n)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx


def compute_c5_argmax(h, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx, np.argmax(conv)


def compute_grad_at_lag(h, n, k_max):
    dx = 2.0 / n
    grad = np.zeros(n)
    m = k_max - (n - 1)
    if m == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif m > 0:
        if m < n:
            grad[m:] += (1.0 - h[:n - m]) * dx
        if n - m > 0:
            grad[:n - m] += (-h[m:]) * dx
    else:
        p = -m
        if n - p > 0:
            grad[:n - p] += (1.0 - h[p:]) * dx
        if p < n:
            grad[p:] += (-h[:n - p]) * dx
    return grad


def compute_avg_grad(h, n, num_lags=3):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-conv)[:min(num_lags, len(conv))]
    top_vals = conv[top_idx]
    spread = top_vals[0] - top_vals[-1]
    if spread < 1e-12:
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        weights = np.exp((top_vals - top_vals[0]) / spread)
        weights /= np.sum(weights)
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        grad += w * compute_grad_at_lag(h, n, k)
    return grad


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-12
            cnt = int(np.sum(free))
            if cnt == 0:
                break
            step = diff / cnt
            h[free] = np.minimum(h[free] + step, 1.0)
        else:
            free = h > 1e-12
            cnt = int(np.sum(free))
            if cnt == 0:
                break
            step = diff / cnt
            h[free] = np.maximum(h[free] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - target) > 1e-10:
        h += (target - np.sum(h)) / n
        h = np.clip(h, 0.0, 1.0)
    return h

                  eps=1e-8, time_limit=None, t_start=None, use_avg_grad=True,
                  binary_pressure=0.0):

    best_val = compute_c5(h, n)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    patience = 0
    patience_limit = 600
    n_improved = 0

    for it in range(max_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break

        else:

            grad = compute_grad_at_lag(h, n, k_max)

        if binary_pressure > 0:
            bp = 2.0 * h * (1.0 - h) * (1.0 - 2.0 * h)
            grad = grad + binary_pressure * bp

        t += 1

        return compute_c5(x, n)

        return compute_avg_grad(x, n, num_lags=3)

    def con_fun(x):
        return np.sum(x) - target

    def con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fun, h0.copy(), jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        hx = project_feasible(res.x, n)
        return hx, compute_c5(hx, n)
    except Exception:
        return h0, compute_c5(h0, n)
