    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = conv - np.max(conv)
    return np.max(conv) * dx + temp * np.log(np.sum(np.exp(shifted / temp)))

    n = n_points
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    log_weights = conv / temp - np.max(conv) / temp

Actually, one more important consideration: the `np.correlate(h, 1-h, mode='full')` computes the LINEAR correlation, not circular. This means the function is defined on [0, 2] without periodic boundary conditions. The overlap at shift k only includes pairs (x, x+k) where both x and x+k are in [0, 2].

This is important because it means the "effective" correlation at shift k is only over the range [0, 2-k] (for k≥0) or [k, 2] (for k<0). The maximum will typically be near k=0 where the overlap is largest.

For k=0: C₅(0) = Σ h[i](1-h[i]) * dx. For binary h, this is 0. For continuous h with mean 0.5, this is n/4 * 2/n = 1/2.

So for binary h, the max is NOT at k=0 (it would be 0 there). It must be at some other shift. This is a key structural insight.

For a random binary sequence with n/2 ones and n/2 zeros, the expected correlation at shift k (for |k| < n) is approximately:
- Number of overlapping pairs: n - |k|
- Expected (1,0) pairs: (n-|k|) * (n/2)/(n) * (n/2)/(n) = (n-|k|)/4

So the max correlation for random binary would be about (n-1)/4 (at k=0, but that's 0 for binary). Actually at k=1: (n-1)/4 ≈ n/4. With dx = 2/n, C₅ ≈ 1/2.

To get 0.38, we need the max correlation to be about 0.38 * n/2 = 0.19n. For n=70, that's about 13.3 (in the summed correlation before multiplying by dx). Since the total number of (1,0) pairs is (n/2)^2 = 1225, and there are about n "effective" shifts (say 70), the average is about 17.5 per shift. So 13.3 is below average, meaning we need a pattern where the (1,0) pairs are concentrated at shifts with LESS overlap (larger |k|) and fewer at shifts with more overlap (smaller |k|).

This is the key optimization challenge: arrange the 1s and 0s so that 1s tend to be followed by 1s (at small shifts) and 0s tend to be followed by 0s (at small shifts), reducing the (1,0) count at small shifts.

This is essentially a "clustering" problem: group the 1s together. But you can't just put all 1s in one block because then at shift n/2, you'd have a lot of (1,0) or (0,1) pairs.

The optimal pattern is likely something like: a block of 1s, then a block of 0s, but with some mixing at the boundary to balance the correlations.

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv)) * dx


def smooth_obj(x, n, temp):
    dx = 2.0 / n
    conv = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(conv)
    shifted = (conv - m) / temp
    shifted -= np.max(shifted)
    return m + temp * np.log(np.sum(np.exp(shifted)))


def smooth_grad(x, n, temp):
    dx = 2.0 / n
    conv = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(conv)
    log_w = (conv - m) / temp
    log_w -= np.max(log_w)
    weights = np.exp(log_w)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        s = k - (n - 1)
        wd = w * dx
        if s == 0:
            grad += wd * (1.0 - 2.0 * x)
        elif s > 0:
            grad[s:] += wd * (1.0 - x[:n - s])
            if n - s > 0:
                grad[:n - s] += wd * (-x[s:])
        else:
            p = -s
            if n - p > 0:
                grad[:n - p] += wd * (1.0 - x[p:])
            if p < n:
                grad[p:] += wd * (-x[:n - p])
    return grad


def exact_obj(x, n):
    return compute_c5(x, n)


def exact_grad(x, n):
    dx = 2.0 / n
    conv = np.correlate(x, 1.0 - x, mode='full')
    k_max = int(np.argmax(conv))
    s = k_max - (n - 1)
    grad = np.zeros(n)
    if s == 0:
        grad[:] = dx * (1.0 - 2.0 * x)
    elif s > 0:
        grad[s:] += dx * (1.0 - x[:n - s])

    else:
        p = -s
        if n - p > 0:

    return grad


def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.any(interior):
            h[interior] += (target - s) / np.sum(interior)
            h = np.clip(h, 0.0, 1.0)
    return h


class SumConstraint:
    def __init__(self, n):
        self.n = n

    def fun(self, x):
        return np.sum(x) - self.n / 2.0

    def jac(self, x):
        return np.ones(self.n)


def do_slsqp_smooth(h0, n, temp, maxiter):
    target = n / 2.0
    sc = SumConstraint(n)
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        smooth_obj, h0.copy(), args=(n, temp),
        jac=smooth_grad,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': sc.fun, 'jac': sc.jac}],
        options={'maxiter': maxiter, 'ftol': 1e-15}
    )
    x = project_feasible(res.x, n)
    return x, compute_c5(x, n)


def do_slsqp_exact(h0, n, maxiter):
    target = n / 2.0
    sc = SumConstraint(n)
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        exact_obj, h0.copy(), args=(n,),
        jac=exact_grad,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': sc.fun, 'jac': sc.jac}],
        options={'maxiter': maxiter, 'ftol': 1e-15}
    )
    x = project_feasible(res.x, n)
    return x, compute_c5(x, n)


def project_binary(h, n):
    target = int(n / 2)
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin


def subgradient_step(h, n, lr, n_active=5):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(conv)[-n_active:]
    grad = np.zeros(n)
    for k in top_k:
        s = k - (n - 1)
        if s == 0:
            grad += dx * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += dx * (1.0 - h[:n - s])
            if n - s > 0:
                grad[:n - s] += dx * (-h[s:])
        else:
            p = -s
            if n - p > 0:
                grad[:n - p] += dx * (1.0 - h[p:])
            if p < n:
                grad[p:] += dx * (-h[:n - p])
    grad /= n_active
    return h - lr * grad
