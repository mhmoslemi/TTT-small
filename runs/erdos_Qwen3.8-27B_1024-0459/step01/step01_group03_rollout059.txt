import time
import numpy as np
from scipy.optimize import minimize


def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)


def project_density(x, target):
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-13 * max(1, n):
        return x.copy()
    lo, hi = -2.0, 2.0
    for _ in range(90):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-14:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            h[free] += res / float(free.size)
            h = np.clip(h, 0.0, 1.0)
        res = target - float(np.sum(h))
        if abs(res) > 1e-14:
            free = np.where((h > 0.0) & (h < 1.0))[0]
            if free.size > 0:
                h[int(free[0])] += res
                h = np.clip(h, 0.0, 1.0)
    return h


def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    n = int(n)
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)


def make_periodic_init(n, period=1, duty=0.5, n_periods=2):
    h = np.zeros(n, dtype=float)
    x = np.arange(n) * (2.0 / n)
    period_len = period * n_periods
    for m in range(int(n_periods)):
        start = int(m * period_len)
        end = int((m + 1) * period_len)
        ones_start = start
        ones_end = start + int(duty * period_len)
        h[ones_start:ones_end] = 1.0
    return project_density(h, 0.5 * n)


def make_random_init(n, rng, mode=0):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    if mode == 0:
        h = np.full(n, 0.5, dtype=float)
        for m in range(1, 8):
            amp = float(rng.uniform(0.05, 0.3)) * (0.7 ** (m - 1))
            freq = int(max(1, round(m * float(rng.uniform(0.5, 2.0)))))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            h += amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    elif mode == 1:
        h = np.empty(n, dtype=float)
        pos = 0
        while pos < n:
            length = int(rng.integers(2, max(3, n // 20)))
            val = float(rng.choice([0.0, 0.3, 0.5, 0.7, 1.0]))
            h[pos:pos + length] = val
            pos += length
        if pos < n:
            h[pos:] = 0.5
    else:
        h = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.0, 1.0)
    return project_density(h, 0.5 * n)


def compute_corr_and_gradient(h, active_k):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    top_k = np.argsort(corr)[-active_k:]
    w = np.ones(active_k) / float(active_k)
    g = np.zeros(n)
    for idx_t in range(active_k):
        t = int(top_k[idx_t])
        k = t - center
        if k == 0:
            g += (1.0 - 2.0 * h)
        else:
            lag = abs(k)
            if lag >= n:
                continue
            if k > 0:
                g[lag:] += (1.0 - h[:n - lag])
                g[:n - lag] -= h[lag:]
            else:
                g[:n - lag] += (1.0 - h[lag:])
                g[lag:] -= h[:n - lag]
    g *= (1.0 / float(active_k))
    return corr, top_k, g


def aggressive_local_search(h, deadline, active_k=8, max_iter=5000, pair_size=8):

    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    if n < 2:
        return best, best_c
    stall = 0
    total_iters = 0

    for it in range(max_iter):

def _set_ctx(n, alpha, target):

def _smooth_objective(x):
    n = _global_n
    alpha = _global_alpha
    target = _global_target
    h = project_density(x, target)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0 or not np.all(np.isfinite(corr)):
        return 1e100
    m = float(np.max(corr))
    dx = 2.0 / float(n)
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return 1e100
    return dx * (m + np.log(z) / alpha)


def _smooth_jacobian(x):
    n = _global_n
    alpha = _global_alpha
    target = _global_target
    h = project_density(x, target)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0 or not np.all(np.isfinite(corr)):
        return np.zeros(n)
    m = float(np.max(corr))
    dx = 2.0 / float(n)
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return np.zeros(n)
    w = e / z
    center = n - 1
    g = np.zeros(n)
    for t in range(len(corr)):
        p = float(w[t])
        if p < 1e-10:
            continue
        k = t - center
        if k == 0:
            g += p * (1.0 - 2.0 * h)
        else:
            lag = abs(k)
            if lag >= n:
                continue
            if k > 0:
                g[lag:] += p * (1.0 - h[:n - lag])
                g[:n - lag] -= p * h[lag:]
            else:
                g[:n - lag] += p * (1.0 - h[lag:])
                g[lag:] -= p * h[:n - lag]
    g -= float(np.mean(g))
    return dx * g
