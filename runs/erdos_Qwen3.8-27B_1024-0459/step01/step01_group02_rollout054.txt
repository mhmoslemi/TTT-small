        if n_de % 2 != 0:
            n_de += 1
        _g_n = n_de
        bounds_de = [(0.0, 1.0)] * n_de
        start_de = _interp_to_n(best_h, best_n, n_de)
        try:
            de_res = differential_evolution(_de_obj, bounds_de,
                                            seed=seed, maxiter=30,
                                            popsize=15, tol=1e-8,
                                            polish=True, mutation=(0.4, 1.0),
                                            recombination=0.8,
                                            updating='deferred',
                                            init='halton')
            h_de = project_feasible(de_res.x, n_de)
            c5_de = compute_c5(h_de)
            if c5_de < best_c5:
                best_c5 = c5_de
                best_h = h_de.copy()
                best_n = n_de
                print("DE: n={}, C5={:.10f}".format(n_de, best_c5))
                # Polish DE result
                h_pol = _slsqp_exact(h_de, n_de, 200)
                c5_pol = compute_c5(h_pol)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()
                    print("DE polish: C5={:.10f}".format(best_c5))
        except Exception:
            pass

    # Phase 6: Perturbation + reoptimize on best
    if time_left() > 30:
        for attempt in range(5):
            h_pert = best_h + rng.normal(0, 0.02, best_n)
            h_pert = project_feasible(np.clip(h_pert, 0, 1), best_n)
            _g_n = best_n
            _g_beta = 500
            h_new = _slsqp_opt(h_pert, best_n, 500, 200, 1e-14)
            h_new = _slsqp_exact(h_new, best_n, 150)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("Perturb{}: C5={:.10f}".format(attempt, best_c5))

    # Binary rounding
    if time_left() > 5:
        h_r = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_r))
        n_want = best_n // 2
        if n_ones != n_want:
            if n_ones > n_want:
                ones_idx = np.where(h_r == 1.0)[0]
                order = np.argsort(best_h[ones_idx])[:n_ones - n_want]
                h_r[ones_idx[order]] = 0.0
            else:
                zeros_idx = np.where(h_r == 0.0)[0]
                order = np.argsort(-best_h[zeros_idx])[:n_want - n_ones]
                h_r[zeros_idx[order]] = 1.0
        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            print("Binary: C5={:.10f}".format(best_c5))

    # Final projection and verification
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        ec = evaluate_erdos_solution(best_h)
        if ec < final_c5:
            final_c5 = ec
    except Exception:
        pass

    print("FINAL: n={}, C5={:.10f}, time={:.0f}s".format(best_n, final_c5, time.time() - start_time))
    return (best_h, final_c5, best_n)
