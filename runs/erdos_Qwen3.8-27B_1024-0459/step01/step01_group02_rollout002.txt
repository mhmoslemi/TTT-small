  c10=np.zeros(n-1,dtype=np.uint16)

def project_onto_feasible(h, n):
    h = np.array(h, dtype=float).ravel()
    n = int(n)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(8):
        s = float(np.sum(h))
        diff = target - s
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # final small correction to non-boundary if possible
    s = float(np.sum(h))

def compute_c5_array(h):
    h = np.asarray(h, dtype=float).ravel()
    n = int(h.size)
    if n == 0:
        return float("inf")
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def get_struct(n):
    n = int(n)
    if n in _CACHE:
        return _CACHE[n]
    masks = [0] * n
    for k in range(1, n):
        masks[k] = (1 << (n - k)) - 1
    powers = [1 << i for i in range(n)]
    _CACHE[n] = (masks, powers)
    return _CACHE[n]


def fast_max_bits(bits, n, masks, limit):
    m = 0
    bc = _BIT_COUNT
    for k in range(1, n):
        tm = masks[k]
        sh = bits >> k

        c = bc(bits & ~sh & tm)
        if c > m:
            m = c
            if limit >= 0 and m > limit:
                return m

        c = bc(sh & ~bits & tm)
        if c > m:
            m = c
            if limit >= 0 and m > limit:
                return m
    return m


def full_stats(bits, n, masks):
    m = 0
    s2 = 0
    ties = []
    tie_limit = 32
    bc = _BIT_COUNT

    for k in range(1, n):
        tm = masks[k]

            m = c

            idx = (start + off) % n
            if idx != e and arr[idx] == need:
                f = idx
                break

    if f < 0:
        return propose_single_flip(arr, n, rng)

    return [e, f]


def local_search_bit(n, bits, arr, masks, powers, rng, t_end):
    m, s2, ties = full_stats(bits, n, masks)

    best_m = m
    best_s2 = s2
    best_bits = bits
    best_arr = arr.copy()

    max_stall = max(3000, min(12000, 800000 // max(1, n)))
    stall = 0

    while time.time() < t_end:
        if m == 0:
            break

        r = rng.random()
        if r < 0.50 and ties:
            k, typ = ties[int(rng.random() * len(ties))]
            flips = propose_biased_flip(arr, n, k, typ, rng)
        elif r < 0.80:
            flips = propose_single_flip(arr, n, rng)
        else:
            flips = propose_double_flip(arr, n, rng)

        if flips is None or len(flips) == 0:
            flips = propose_single_flip(arr, n, rng)
        if flips is None or len(flips) == 0:
            break

        cand_bits = bits
        for idx in flips:
            cand_bits ^= powers[idx]

        limit = m - 1
        m_less = fast_max_bits(cand_bits, n, masks, limit)

        cand_m = None
        cand_s2 = None
        cand_ties = None
        accept = False

        if m_less < m:
            cand_m, cand_s2, cand_ties = full_stats(cand_bits, n, masks)
            accept = cand_m < m
        elif m_less == m:
            if rng.random() < 0.22:
                cand_m, cand_s2, cand_ties = full_stats(cand_bits, n, masks)
                if cand_m == m:
                    if cand_s2 < s2:
                        accept = True
                    elif cand_s2 == s2 and rng.random() < 0.015:
                        accept = True
                elif cand_m < m:
                    accept = True
        elif m_less == m + 1:
            if rng.random() < 0.015:
                cand_m, cand_s2, cand_ties = full_stats(cand_bits, n, masks)
                if cand_m == m + 1 and cand_s2 < s2 * 0.98:
                    accept = True

        if accept:
            for idx in flips:
                arr[idx] ^= 1

            bits = cand_bits
            m = cand_m
            s2 = cand_s2
            ties = cand_ties if cand_ties is not None else []
            stall = 0

            if m < best_m or (m == best_m and s2 < best_s2):
                best_m = m
                best_s2 = s2
                best_bits = bits
                best_arr = arr.copy()
        else:
            stall += 1
            if stall >= max_stall:
                break

    if m < best_m or (m == best_m and s2 < best_s2):
        best_m = m
        best_s2 = s2
        best_bits = bits
        best_arr = arr.copy()

    return best_bits, best_m, best_arr, best_s2


def search_at_n(n, parent_h, parent_n, t_start, t_end, seed):
    n = int(n)
    if n % 2 == 1:
        n += 1
    if n < 8 or t_end <= t_start:
        return 0, 10 ** 9, []

    masks, powers = get_struct(n)
    rng = np.random.RandomState((int(seed) + n * 104729) % (2 ** 31 - 1))

    best_m = 10 ** 9
    best_s2 = 10 ** 30
    best_bits = 0
    best_arr = []

    markov_ps = [0.62, 0.72, 0.82, 0.90, 0.95, 0.99]
    mode = 0 if parent_h is not None else 1

    while time.time() < t_end - 0.2:
        if mode == 0 and parent_h is not None:
            bits, arr = make_parent_bits(parent_h, parent_n, n, powers)
            mode = 1
        elif mode < len(markov_ps):
            arr = make_markov_arr(n, markov_ps[mode], rng)
            bits = bits_from_arr(arr, n, powers)
            mode += 1
        elif mode == len(markov_ps):
            bits, arr = make_uniform_bits(n, powers, rng)
            mode += 1
        else:
            if best_bits == 0:
                bits, arr = make_uniform_bits(n, powers, rng)
            else:
                k = 5 + min(80, n // 10 + (mode % 30))
                bits, arr = perturb_bits(best_bits, best_arr, rng, n, powers, k)

        remain = t_end - time.time()
        if remain < 0.5:
            break

        local_end = time.time() + min(22.0, max(0.8, remain / 12.0))
        if local_end > t_end:
            local_end = t_end

        b2, m2, a2, s22 = local_search_bit(n, bits, arr, masks, powers, rng, local_end)

        if m2 < best_m or (m2 == best_m and s22 < best_s2):
            best_m = m2
            best_s2 = s22
            best_bits = b2
            best_arr = a2

    if best_bits != 0 and best_m < 10 ** 8:
        best_m = fast_max_bits(best_bits, n, masks, None)

    return best_bits, best_m, best_arr


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    budget = float(budget_s)

    parent_h = None
    pv = globals().get("initial_h_values")
    if pv is not None:
        try:
            p = np.asarray(pv, dtype=float).ravel()
            p = p[np.isfinite(p)]
            if p.size > 0:
                parent_h = p
        except Exception:
            parent_h = None

    best_h = None
    best_c5 = float("inf")
    best_n = 0
    best_binary_n = 0

    if parent_h is not None:
        parent_h = project_onto_feasible(parent_h, parent_h.size)
        pc = compute_c5_array(parent_h)
        best_h = parent_h.copy()
        best_n = int(parent_h.size)
        best_c5 = float(pc)

    parent_n = int(parent_h.size) if parent_h is not None else 0

    reserve = min(100.0, max(5.0, budget * 0.10))
    total_binary = max(1.0, budget - reserve)
    t_binary_end = start_time + total_binary

    plan = [
        (96, 6),
        (128, 24),
        (160, 22),
        (192, 18),
        (256, 16),
        (320, 10),
        (384, 4),
    ]

    for nn, pct in plan:
        if time.time() >= t_binary_end:
            break
        alloc = total_binary * (pct / 100.0)
        t_end = min(t_binary_end, time.time() + alloc)
        if t_end <= time.time():
            continue

        bits, mc, arr = search_at_n(nn, parent_h, parent_n, time.time(), t_end, seed + nn)

        if bits != 0 and mc < 10 ** 8:
            cc = (2.0 * mc) / nn
            if cc < best_c5:
                h = np.zeros(nn, dtype=float)
                for i, v in enumerate(arr):
                    if v:
                        h[i] = 1.0
                best_h = h
                best_n = nn
                best_c5 = float(cc)
                best_binary_n = nn

    if time.time() < t_binary_end - 10.0:
        focus_n = best_binary_n if best_binary_n != 0 else 128
        bits, mc, arr = search_at_n(focus_n, parent_h, parent_n, time.time(), t_binary_end, seed + 90917)
        if bits != 0 and mc < 10 ** 8:
            cc = (2.0 * mc) / focus_n
            if cc < best_c5:
                h = np.zeros(focus_n, dtype=float)
                for i, v in enumerate(arr):
                    if v:
                        h[i] = 1.0
                best_h = h
                best_n = focus_n
                best_c5 = float(cc)
                best_binary_n = focus_n

    if best_h is None:
        n = 128
        masks, powers = get_struct(n)
        rng = np.random.RandomState(int(seed))
        t_fb_end = min(start_time + budget - 1.0, time.time() + min(45.0, max(3.0, budget * 0.04)))
        arr = make_markov_arr(n, 0.85, rng)
        bits = bits_from_arr(arr, n, powers)
        bits, mc, arr, _ = local_search_bit(n, bits, arr, masks, powers, rng, t_fb_end)
        h = bits_to_np(bits, n)
        best_h = h
        best_n = n
        best_c5 = (2.0 * mc) / n
        best_binary_n = n

    h_final = np.asarray(best_h, dtype=float).ravel()
    h_final = np.clip(h_final, 0.0, 1.0)
    n_final = int(h_final.size)

    target_sum = n_final / 2.0
    diff = target_sum - float(np.sum(h_final))
    if abs(diff) > 1e-10:
        h_final = project_onto_feasible(h_final, n_final)

    final_c5 = compute_c5_array(h_final)

    if abs(target_sum - float(np.sum(h_final))) < 1e-8 and math.isfinite(best_c5):
        if math.isfinite(best_c5) and best_c5 + 1e-10 < final_c5:
            final_c5 = best_c5

    if not math.isfinite(final_c5):
        final_c5 = float(best_c5)

    return h_final, float(final_c5), n_final
