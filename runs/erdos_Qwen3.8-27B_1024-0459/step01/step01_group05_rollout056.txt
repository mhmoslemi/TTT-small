from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    """Project h onto {sum(h)=target, 0<=h<=1}."""
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        free = (h > 0.0) & (h < 1.0)
        if np.any(free):
            h[free] += diff / np.sum(free)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_weights(corr, beta):
    """Compute smooth max value and softmax weights."""
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_s = np.exp(shifted)
    log_sum = np.log(np.sum(exp_s))
    val = m + log_sum / beta
    w = exp_s / np.sum(exp_s)

def project_sum(h, target):
