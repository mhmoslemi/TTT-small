   The kink points are where two or more conv[k] values are equal. We can find the minimum by:
   - Evaluating the objective at all kink points and boundaries
   - Or using a ternary search (since it's convex)
   - Or using the fact that the minimum of max of linear functions is where the "active" set changes

   Actually, for a 1D convex piecewise-linear function, the minimum is where the subgradient contains 0. We can find this by:
   - Computing the objective and its "slope" (which changes at kink points)
   - Binary search on the slope

   But the simplest approach is to just do a fine grid search over t, or use scipy's scalar minimizer.

   Let me implement this:

```python
def paired_cd_step(h, i, j, n_points, temp=0.01):
    """One paired coordinate descent step: optimize h[i] and h[j] jointly."""
    n = n_points
    h_i = h[i]
    h_j = h[j]
    
    # h[i] = t, h[j] = h_j + h_i - t
    # Bounds: 0 <= t <= 1, 0 <= h_j + h_i - t <= 1
    # So t >= h_i + h_j - 1, t <= 1, t >= 0, t <= h_i + h_j
    t_lo = max(0.0, h_i + h_j - 1.0)
    t_hi = min(1.0, h_i + h_j)
    
    if t_hi - t_lo < 1e-15:
        return h
    
    # Try multiple values of t
    # For efficiency, use a small grid + refinement
    best_t = t_lo  # Start at current value
    best_val = None
    h[i] = h_i
    h[j] = h_j
    current_val = smooth_max_obj(h, n, temp)
    best_val = current_val
    best_t = h_i
    
    # Coarse grid
    n_grid = 20
    for t in np.linspace(t_lo, t_hi, n_grid):
        h[i] = t
        h[j] = h_j + h_i - t
        val = smooth_max_obj(h, n, temp)
        if val < best_val:
            best_val = val
            best_t = t
    
    # Refine around best
    if best_t > t_lo and best_t < t_hi:
        for t in np.linspace(best_t - (t_hi-t_lo)/n_grid, best_t + (t_hi-t_lo)/n_grid, 20):
            if t < t_lo or t > t_hi:
                continue
            h[i] = t
            h[j] = h_j + h_i - t
            val = smooth_max_obj(h, n, temp)
            if val < best_val:
                best_val = val
                best_t = t
    
    h[i] = best_t
    h[j] = h_j + h_i - best_t
    return h
```

This is O(n²) per step (due to the correlation computation), and we do it n times per sweep (for n pairs). So one sweep is O(n³). For n=200, that's 8*10⁶ operations per sweep, which should be fast.

But wait, I'm computing smooth_max_obj from scratch for each t value, which involves a full correlation. That's O(n²) per evaluation. With 40 evaluations per pair and n pairs per sweep, that's O(40*n³) per sweep = 40*8*10⁶ = 3.2*10⁸ for n=200. That might be too slow.

Let me optimize: instead of computing the full correlation each time, I can compute only the change. When I change h[i] from h_i to t and h[j] from h_j to h_j + h_i - t, the change in conv[k] is:
Δconv[k] = (t - h_i)*(1-h[k-i]) + (h_j + h_i - t - h_j)*(1-h[k-j])
         = (t - h_i)*((1-h[k-i]) - (1-h[k-j]))
         = (t - h_i)*(h[k-j] - h[k-i])

Wait, let me redo this. 
conv[k] = sum_m h[m]*(1-h[k-m])

When we change h[i] → t and h[j] → h_j' = h_j + h_i - t:
conv_new[k] - conv_old[k] = (t - h_i)*(1-h[k-i]) + (h_j' - h_j)*(1-h[k-j])
                           = (t - h_i)*(1-h[k-i]) + (h_i - t)*(1-h[k-j])
                           = (t - h_i)*((1-h[k-i]) - (1-h[k-j]))
                           = (t - h_i)*(h[k-j] - h[k-i])

So Δconv[k] = (t - h_i) * (h[k-j] - h[k-i])

This is very efficient to compute! We precompute d[k] = h[k-j] - h[k-i] (a vector of length 2n-1), and then conv_new = conv_old + (t-h_i) * d.

Then we only need to compute the smooth max of conv_new, which is O(n).

This makes each evaluation O(n) instead of O(n²), and each sweep is O(40*n²) instead of O(40*n³).

Let me implement this optimized version:

```python
def paired_cd_sweep(h, n_points, temp=0.01, n_pairs=None, order=None):
    """One full sweep of paired coordinate descent."""
    n = n_points
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx  # current conv values
    
    if n_pairs is None:
        n_pairs = n
    if order is None:
        order = list(range(n))
    
    # Shuffle pairs
    pairs = [(order[i % n], order[(i + n//2) % n]) for i in range(n_pairs)]
    
    for i, j in pairs:
        h_i = h[i]
        h_j = h[j]
        t_lo = max(0.0, h_i + h_j - 1.0)
        t_hi = min(1.0, h_i + h_j)
        
        if t_hi - t_lo < 1e-15:
            continue
        
        # Precompute d[k] = h[(k-j) mod valid] - h[(k-i) mod valid]
        # For conv[k] = sum_m h[m]*(1-h[k-m]), the valid range of k-m is [0, n-1]
        # h[k-i] is valid when 0 <= k-i < n, i.e., i <= k <= i+n-1
        # h[k-j] is valid when 0 <= k-j < n, i.e., j <= k <= j+n-1
        
        d = np.zeros(2*n - 1)
        # h[k-i] contribution: valid for k in [i, i+n-1]
        for k in range(i, min(i+n, 2*n-1)):
            idx = k - i  # 0 to n-1
            d[k] -= h[idx]  # subtract h[k-i]
        # h[k-j] contribution: valid for k in [j, j+n-1]
        for k in range(j, min(j+n, 2*n-1)):
            idx = k - j  # 0 to n-1
            d[k] += h[idx]  # add h[k-j]
        
        # Now try different t values
        best_t = h_i
        best_val = smooth_max(conv, temp)
        
        n_grid = 15
        for t in np.linspace(t_lo, t_hi, n_grid):
            conv_new = conv + (t - h_i) * d
            val = smooth_max(conv_new, temp)
            if val < best_val:
                best_val = val
                best_t = t
        
        # Refine
        if best_t > t_lo + 1e-10 and best_t < t_hi - 1e-10:
            span = (t_hi - t_lo) / n_grid
            for t in np.linspace(max(t_lo, best_t - span), min(t_hi, best_t + span), 15):
                conv_new = conv + (t - h_i) * d
                val = smooth_max(conv_new, temp)
                if val < best_val:
                    best_val = val
                    best_t = t
        
        # Update
        if best_t != h_i:
            h[i] = best_t
            h[j] = h_j + h_i - best_t
            conv = conv + (best_t - h_i) * d
    
    return h, conv
```

Wait, I'm making the d computation O(n) in a loop, which is O(n) per pair. Let me vectorize:

```python
# d[k] for k = 0 to 2n-2:
# d[k] = h[k-j] (if 0<=k-j<n) - h[k-i] (if 0<=k-i<n)
k_vals = np.arange(2*n-1)
d = np.zeros(2*n-1)
valid_j = (k_vals >= j) & (k_vals < j+n)
valid_i = (k_vals >= i) & (k_vals < i+n)

d[valid_i] -= h[k_vals[valid_i] - i]
```

This is O(n) vectorized. Good.

```python
def smooth_max(conv, temp):
```

Let me now put together the full algorithm:

**Algorithm:**
1. Initialize from initial_h_values (n=70) or random
2. Phase 1: Paired coordinate descent at n=70 (many sweeps)
3. Phase 2: Upscale to higher resolution (n=140, 200) and continue CD
4. Phase 3: Gradient-based refinement (FISTA or L-BFGS-B)
5. Phase 4: More coordinate descent at the best resolution
6. Return the best solution found

Let me also think about the temperature schedule. Starting with high temp (e.g., 0.1) helps explore, and decreasing to low temp (e.g., 0.001) helps exploit.

Actually, I realize there's a subtlety: the coordinate descent minimizes the smooth max, not the exact max. As the temperature decreases, it gets closer to the exact max. So I should do:
- Many sweeps at high temp (exploration)
- Many sweeps at medium temp
- Many sweeps at low temp (exploitation)
- Final: minimize the exact max (which is just evaluating at the kink points)

For the exact max (temp → 0), the coordinate descent step becomes: minimize max_k (conv[k] + (t-h_i)*d[k]) over t ∈ [t_lo, t_hi]. This is a 1D convex piecewise-linear minimization, which can be solved exactly by finding where the subgradient contains 0.

The function g(t) = max_k (a_k + t*b_k) where a_k = conv[k] - h_i*d[k] and b_k = d[k].
g'(t) = b_{k*(t)} where k*(t) is the active constraint (the one achieving the max).
As t increases, the active constraint can only change to one with larger b_k.
So g is piecewise linear with non-decreasing slopes (convex).

The minimum is where the slope changes from negative to positive (or is zero). We can find this by:
1. Sort the constraints by b_k
2. For each "interval" of t where a particular constraint is active, check if the minimum is in that interval
3. Or more simply: use binary search on the subgradient

Actually, the simplest approach for the exact max is:
- The minimum of max_k (a_k + t*b_k) occurs where the "upper envelope" has a minimum.
- This can be found by: for each k, the line a_k + t*b_k has minimum at t = t_lo if b_k > 0, t = t_hi if b_k < 0, or is constant if b_k = 0.
- The overall minimum is the lowest point on the upper envelope.

A practical approach: evaluate g(t) at O(n) points (the intersection points of pairs of lines) and take the minimum. But that's O(n²) points.

A simpler approach: use the fact that g is convex and unimodal. Use golden section search or ternary search on [t_lo, t_hi]. Since g is convex, ternary search works in O(log(range/epsilon)) evaluations, each O(n).

Let me use a hybrid: fine grid + ternary refinement.

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def smooth_max_fn(conv, temp):
    m = np.max(conv)
    if temp < 1e-12:
        return m
    return m + temp * np.log(np.sum(np.exp((conv - m) / temp)))

def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    h_proj = np.clip(h - mid, 0.0, 1.0)
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        mask = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
        if np.any(mask):
            h_proj[mask] += diff / np.sum(mask)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def paired_cd_sweep(h, conv, n_points, temp, n_sweeps_pairs=None):
    """One sweep of paired coordinate descent."""
    n = n_points
    dx = 2.0 / n
    k_vals = np.arange(2 * n - 1)
    
    if n_sweeps_pairs is None:
        n_sweeps_pairs = n
    
    # Create pairs: (i, i + n//2) mod n
    pairs = [(i, (i + n // 2) % n) for i in range(n_sweeps_pairs)]
    
    for i, j in pairs:
        h_i = h[i]
        h_j = h[j]
        t_lo = max(0.0, h_i + h_j - 1.0)
        t_hi = min(1.0, h_i + h_j)
        
        if t_hi - t_lo < 1e-14:
            continue
        
        # Compute d[k] = h[k-j] (if valid) - h[k-i] (if valid)
        d = np.zeros(2 * n - 1)
        valid_j = (k_vals >= j) & (k_vals < j + n)
        valid_i = (k_vals >= i) & (k_vals < i + n)
        d[valid_j] += h[k_vals[valid_j] - j]
        d[valid_i] -= h[k_vals[valid_i] - i]
        
        # Check if d is all zeros (no change possible)
        if np.all(np.abs(d) < 1e-15):
            continue
        
        # Find best t
        best_t = h_i
        best_val = smooth_max_fn(conv, temp)
        
        # Grid search
        n_grid = 20
        t_vals = np.linspace(t_lo, t_hi, n_grid)
        for t in t_vals:
            val = smooth_max_fn(conv + (t - h_i) * d, temp)
            if val < best_val:
                best_val = val
                best_t = t
        
        # Refine with ternary search
        if best_t > t_lo and best_t < t_hi:
            a = best_t - (t_hi - t_lo) / n_grid
            b = best_t + (t_hi - t_lo) / n_grid
            a = max(a, t_lo)
            b = min(b, t_hi)
            for _ in range(30):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                f1 = smooth_max_fn(conv + (m1 - h_i) * d, temp)
                f2 = smooth_max_fn(conv + (m2 - h_i) * d, temp)
                if f1 < f2:
                    b = m2
                else:
                    a = m1
                if b - a < 1e-10:
                    break
            t_mid = (a + b) / 2.0
            val = smooth_max_fn(conv + (t_mid - h_i) * d, temp)
            if val < best_val:
                best_val = val
                best_t = t_mid
        
        # Apply update
        if abs(best_t - h_i) > 1e-12:
            h[i] = best_t
            h[j] = h_j + h_i - best_t
            conv = conv + (best_t - h_i) * d
    
    return h, conv

def gradient_descent_step(h, n_points, temp, lr=0.01):
    """One projected gradient descent step."""
    n = n_points
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(conv)
    logits = (conv - m) / temp
    logits -= np.max(logits)
    exp_logits = np.exp(logits)
    weights = exp_logits / np.sum(exp_logits)
    grad = dx * sig_correlate(weights, 1.0 - 2.0 * h, mode='valid')
    
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new
