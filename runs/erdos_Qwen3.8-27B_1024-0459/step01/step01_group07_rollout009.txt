import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def compute_shifts(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_val(g, T):
    gm = np.max(g)
    if T < 1e-10:
        return gm
    s = g - gm
    s = np.where(s < -50, -50, s)
    return gm + np.log(np.sum(np.exp(T * s))) / T

def obj_grad(h, n, T, penalty=500.0):
    g = compute_shifts(h, n)
    val = smooth_max_val(g, T)
    
    # Softmax
    gm = np.max(g)
    s = g - gm
    s = np.where(s < -50, -50, s)
    w = np.exp(T * s)
    w /= np.sum(w)
    
    dx = 2.0 / n
    target = n / 2.0
    grad = np.zeros(n)
    
    for j in range(n):
        # (a) h[j] as first factor: i = j - k' + (n-1), valid 0<=i<n
        lo_a = max(0, j)
        hi_a = min(2*n-2, j + n - 1)
        if lo_a <= hi_a:
            i_vals = j - np.arange(lo_a, hi_a+1) + (n-1)
            grad[j] += dx * np.dot(w[lo_a:hi_a+1], 1.0 - h[i_vals])
        
        # (b) h[j] in (1-h[j]): need 0<=j+k'-(n-1)<n
        lo_b = max(0, n-1-j)
        hi_b = min(2*n-2, 2*n-2-j)
        if lo_b <= hi_b:
            idx = j + np.arange(lo_b, hi_b+1) - (n-1)
            grad[j] -= dx * np.dot(w[lo_b:hi_b+1], h[idx])
    
    # Penalty gradient
    sum_diff = np.sum(h) - target
    grad += 2.0 * penalty * sum_diff
    
    obj_val = val + penalty * sum_diff * sum_diff
    return obj_val, grad

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-14
            cnt = np.sum(free)
            if cnt == 0:
                break
            h[free] = np.minimum(h[free] + diff / cnt, 1.0)
        else:
            free = h > 1e-14
            cnt = np.sum(free)
            if cnt == 0:
                break
            h[free] = np.maximum(h[free] + diff / cnt, 0.0)

        return v

        _, g = obj_grad(x, n, T, penalty)
        return g
    
    bounds = [(0.0, 1.0)] * n

def binary_local_search(h, n, max_iter=5000, rng=None):

    if rng is None:

        h_bin[excess] = 0

        h_bin[deficit] = 1

    best_h = h_bin.copy()

        a = rng.choice(ones)

            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
    
    return best_h, best_val

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except:
        h_parent = None
    
    if h_parent is not None and len(h_parent) > 0:
        n = len(h_parent)
        h0 = project_feasible(h_parent, n)
    else:
        n = 70
        h0 = np.full(n, 0.5)
    
    best_h = h0.copy()
    best_val = compute_c5(best_h, n)
    best_n = n
    
    # Phase 1: Smooth max optimization from parent (increasing T)
    for T in [20, 50, 100, 200, 400]:
        if time_left() < 100:
            break
        h_try, v_try = optimize_smooth(best_h, best_n, T=T, penalty=500, maxiter=200)
        if v_try < best_val:
            best_h, best_val = h_try, v_try
    
    # Phase 2: Multi-start
    n_starts = 20
    for s in range(n_starts):
        if time_left() < 200:
            break
        
        strat = s % 5
        if strat == 0:
            h_rand = rng.uniform(0, 1, best_n)
        elif strat == 1:
            h_rand = rng.binomial(1, 0.5, best_n).astype(float)
        elif strat == 2:
            p = rng.randint(2, 12)
            pat = rng.binomial(1, 0.5, p).astype(float)
            h_rand = np.tile(pat, (best_n // p) + 1)[:best_n]
        elif strat == 3:
            h_rand = np.clip(best_h + rng.normal(0, 0.1, best_n), 0, 1)
        else:
            h_rand = np.tile([0.2, 0.8, 0.5, 0.5], (best_n // 4) + 1)[:best_n] + rng.normal(0, 0.1, best_n)
            h_rand = np.clip(h_rand, 0, 1)
        
        h_rand = project_feasible(h_rand, best_n)
        
        h_try, v_try = optimize_smooth(h_rand, best_n, T=50, penalty=500, maxiter=150)
        if v_try < best_val:
            best_h, best_val = h_try, v_try
            # Follow up with higher T
            h_try2, v_try2 = optimize_smooth(h_try, best_n, T=150, penalty=500, maxiter=150)
            if v_try2 < best_val:
                best_h, best_val = h_try2, v_try2
    
    # Phase 3: Binary local search
    if time_left() > 150:
        h_bin, v_bin = binary_local_search(best_h, best_n, max_iter=3000, rng=rng)
        # Use binary as starting point for continuous optimization
        h_try, v_try = optimize_smooth(h_bin, best_n, T=50, penalty=500, maxiter=200)
        if v_try < best_val:
            best_h, best_val = h_try, v_try
        h_try, v_try = optimize_smooth(h_try, best_n, T=200, penalty=500, maxiter=20
