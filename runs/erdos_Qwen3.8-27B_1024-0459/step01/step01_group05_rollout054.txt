```python
def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_weights(corr, beta):
    """Compute smooth max and softmax weights."""
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted = np.clip(shifted, -200.0, 200.0)
    sm = np.sum(np.exp(shifted))
    if sm < 1e-300:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
    else:
        weights = np.exp(shifted) / sm
    f = (max_c + np.log(sm) / beta)
    return f, weights

    f, _ = smooth_max_and_weights(corr, beta)
    return f * dx
```

```python
def c5_grad_analytical(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted = np.clip(shifted, -200.0, 200.0)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    
    g = 1.0 - 2.0 * h  # length n
    full_corr = np.correlate(g, w, mode='full')  # length 3n-2
    grad = dx * full_corr[2*n-2 : 3*n-2]
    return grad
```
