    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    obj = cmax + (1.0/beta) * np.log(np.sum(np.exp(shifted)))
    
    # Gradient
    grad = np.zeros(n)
    for idx in range(2*n - 1):
        s = idx - (n - 1)  # actual shift
        if abs(weights[idx]) < 1e-15:
            continue
        w = weights[idx] / beta
        for j in range(n):

            js = j - s
            if 0 <= js < n:
                grad[j] += w * (1.0 - h[js])
            # -h[j+s] * 1[0 <= j+s < n]
            js2 = j + s
            if 0 <= js2 < n:

    return obj, grad

def get_corr(h):

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:

        else:

            cnt = np.count_nonzero(mask)
            if cnt == 0:

def project_feasible(h, n):

        else:

def projected_subgradient_step(h, n, step_size, n_active=5):
    """One step of projected subgradient descent on the exact max."""
    corr = get_corr(h)
    # Find the worst shifts
    top_idx = np.argsort(corr)[-n_active:]
    # Subgradient: average of gradients of worst shifts
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for idx in top_idx:
        s = idx - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += (1.0 - h[:n - s]) / n_active
            if n - s > 0:
                grad[:n - s] -= h[s:] / n_active
        else:
            a = -s
            if n - a > 0:
                grad[:n - a] += (1.0 - h[a:]) / n_active
            if a < n:
                grad[a:] -= h[:n - a] / n_active
    
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

def local_search(h, n, n_trials, step_size, rng):
    """Local perturbation search."""
    h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_trials):
        h_try = h.copy()
        n_mod = rng.randint(1, 8)
        idxs = rng.choice(n, n_mod, replace=False)
        h_try[idxs] += rng.uniform(-step_size, step_size, n_mod)
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            h = h_try
            best_c5 = c5_try
    return h, best_c5

def compute_gradient(h, w, n):
    """w are softmax weights over the 2n-1 correlation values."""
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-18:
            continue
        s = idx - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += wi * (1.0 - h[:n - s])
            if n - s > 0:
                grad[:n - s] -= wi * h[s:]
        else:
            a = -s
            if n - a > 0:
                grad[:n - a] += wi * (1.0 - h[a:])
            if a < n:
                grad[a:] -= wi * h[:n - a]
    return grad
