       s=np.clip(y-mid,0,1).sum()

def resample_step(h_old,n_new):
    n_old=len(h_old)
    if n_old==n_new: return h_old.copy()
    if n_new > n_old and n_new % n_old ==0: return np.repeat(...)
    x_old=np.linspace(0,2,n_old,endpoint=False)
    x_new=np.linspace(0,2,n_new,endpoint=False)
    idx=np.searchsorted(x_old, x_new, side='right') -1
    idx=np.clip(idx,0,n_old-1)
    return h_old[idx]

import numpy as np
from scipy.optimize import minimize
import time

OPT_N = 0
OPT_TARGET = 0.0
OPT_TAU = 1.0
OPT_PEN = 0.0
OPT_M2 = 0

OPT_DX = 1.0


def set_opt_params(n, tau_c5, pen):
    global OPT_N, OPT_TARGET, OPT_TAU, OPT_PEN, OPT_M2, OPT_M3, OPT_DX
    n = int(n)
    dx = 2.0 / float(n)
    target = 0.5 * float(n)
    tau = max(float(tau_c5), 1e-8) / dx
    pen = float(pen)

    m2 = 1
    while m2 < (2 * n - 1):
        m2 <<= 1
    m3 = 1
    while m3 < (3 * n - 2):
        m3 <<= 1

    OPT_N = n
    OPT_TARGET = target
    OPT_TAU = tau
    OPT_PEN = pen
    OPT_M2 = m2
    OPT_M3 = m3
    OPT_DX = dx


def project_box_sum(y, target):
    y = np.asarray(y, dtype=float).ravel()
    target = float(target)
    n = y.size
    if n == 0:
        return y

    low = float(np.min(y)) - 1.0
    high = float(np.max(y)) + 1.0

    for _ in range(60):
        mid = 0.5 * (low + high)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            low = mid
        else:
            high = mid

    t = 0.5 * (low + high)
    val = np.clip(y - t, 0.0, 1.0)

    for _ in range(20):
        diff = target - float(np.sum(val))
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            free = np.where(val < 1.0 - 1e-12)[0]
        else:
            free = np.where(val > 1e-12)[0]
        if free.size == 0:
            break
        val[free] += diff / float(free.size)
        val[free] = np.clip(val[free], 0.0, 1.0)

    diff = target - float(np.sum(val))
    if abs(diff) > 1e-8:
        order = np.argsort(y)
        val = np.zeros(n, dtype=float)
        if diff >= 0:
            for i in range(n):
                idx = order[i]
                room = 1.0 - val[idx]
                add = min(room, diff)
                if add <= 0.0:
                    break
                val[idx] += add
                diff -= add
        else:
            for i in range(n - 1, -1, -1):
                idx = order[i]
                room = val[idx]
                sub = min(room, -diff)
                if sub <= 0.0:
                    break
                val[idx] -= sub
                diff += sub

    val = np.clip(val, 0.0, 1.0)
    return val


def corr_fft(x, m):

    b = (1.0 - x)[::-1]
    c = np.fft.irfft(np.fft.rfft(x, m) * np.fft.rfft(b, m), m)[:2 * n - 1]
    return c.real


def corr_opt(x):

    if n <= 1024:
        return np.correlate(x, 1.0 - x, mode="full")
    return corr_fft(x, OPT_M2)


def fun_scalar(x):
    x = np.asarray(x, dtype=float).ravel()
    c = corr_opt(x)
    m = float(np.max(c))
    u = np.clip((c - m) / OPT_TAU, -745.0, 0.0)
    e = np.exp(u)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        f = m
    else:
        f = m + OPT_TAU * float(np.log(s))

    res = float(np.sum(x) - OPT_TARGET)
    f += OPT_PEN * res * res
    return float(f)


def grad_scalar(x):
    x = np.asarray(x, dtype=float).ravel()
    n = OPT_N

    c = corr_opt(x)
    m = float(np.max(c))
    u = np.clip((c - m) / OPT_TAU, -745.0, 0.0)
    e = np.exp(u)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        w = np.full(2 * n - 1, 1.0 / (2.0 * n - 1.0))
    else:
        w = e / s

    pref = np.empty(2 * n, dtype=float)
    pref[0] = 0.0
    pref[1:] = np.cumsum(w)
    B = pref[n:2 * n] - pref[0:n]

    fW = np.fft.rfft(w, OPT_M3)
    fh = np.fft.rfft(x, OPT_M3)

    conv1 = np.fft.irfft(fW * fh, OPT_M3)
    S1 = conv1[n - 1:2 * n - 1].real

    wr = w[::-1]
    fW2 = np.fft.rfft(wr, OPT_M3)
    conv2 = np.fft.irfft(fW2 * fh, OPT_M3)
    S2 = conv2[n - 1:2 * n - 1].real

    g = B - S1 - S2
    if OPT_PEN != 0.0:
        res = float(np.sum(x) - OPT_TARGET)
        g += 2.0 * OPT_PEN * res
    return g


def run_lbbfgs(x0, n, tau_c5, pen, maxiter):
    x0 = np.asarray(x0, dtype=float).ravel()
    x0 = project_box_sum(np.clip(x0, 0.0, 1.0), 0.5 * float(n))
    maxiter = int(maxiter)
    if maxiter <= 0:
        return x0

    set_opt_params(n, tau_c5, pen)
    bounds = [(0.0, 1.0)] * n
    options = {
        "maxiter": maxiter,
        "maxfun": int(max(100, 8 * maxiter)),
        "gtol": 1e-5,
        "ftol": 1e-9,
        "disp": False,
    }

    try:
        res = minimize(
            fun_scalar,
            x0,
            method="L-BFGS-B",
            bounds=bounds,
            jac=grad_scalar,
            options=options,
        )
        x = res.x
    except Exception:
        x = x0

    return np.clip(np.asarray(x, dtype=float), 0.0, 1.0)


def objective_feasible(h):
    h = np.asarray(h, dtype=float).ravel()
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_and_eval(h):
    h = np.asarray(h, dtype=float).ravel()
    if h.size == 0:
        return h, 1e100
    h = project_box_sum(h, 0.5 * float(h.size))
    h = np.clip(h, 0.0, 1.0)
    return h, objective_feasible(h)


def resample_step(h_old, n_new):
    h_old = np.asarray(h_old, dtype=float).ravel()
    n_old = h_old.size
    n_new = int(n_new)
    if n_old == n_new:
        return h_old.copy()
    if n_new > n_old and (n_new % n_old) == 0:
        return np.repeat(h_old, n_new // n_old).astype(float)

    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    idx = np.searchsorted(x_old, x_new, side="right") - 1
    idx = np.clip(idx, 0, n_old - 1)
    return h_old[idx].astype(float)


def resample_linear(h_old, n_new):
    h_old = np.asarray(h_old, dtype=float).ravel()
    n_old = h_old.size
    n_new = int(n_new)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old).astype(float)


def rank_binarize(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    out = np.zeros(n, dtype=float)
    if n == 0:
        return out
    order = np.argsort(h)[::-1]
    k = n // 2
    if k > 0:
        out[order[:k]] = 1.0
    if (n % 2) == 1:
        out[order[k]] = 0.5
    return out


def random_binary_start(n, rng):
    n = int(n)
    x = rng.random(n)
    return rank_binarize(x)


def sine_start(n, freq, amp, phase):
    n = int(n)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    y = np.sin(2.0 * np.pi * float(freq) * x + float(phase))
    if abs(float(amp)) >= 0.999:
        return rank_binarize(0.5 + 0.5 * y)
    h = np.clip(0.5 + float(amp) * y, 0.0, 1.0)
    return project_box_sum(h, 0.5 * float(n))


def pattern_start(n, period, duty_count, offset=0):
    n = int(n)
    period = max(1, int(period))
    duty_count = max(0, int(duty_count))
    offset = int(offset) % period
    h = ((np.arange(n) + offset) % period) < float(duty_count)
    h = h.astype(float)
    return project_box_sum(h, 0.5 * float(n))


def make_stage_starts(base_h, n, rng, extra):
    n = int(n)
    starts = []
    target = 0.5 * float(n)

    if base_h is not None and np.size(base_h) > 0:
        hs = resample_step(base_h, n)
        starts.append(hs)

        hb = rank_binarize(hs)
        starts.append(hb)

        pert = project_box_sum(hs + rng.normal(0.0, 0.08, n), target)
        starts.append(pert)

        hl = project_box_sum(resample_linear(base_h, n), target)
        starts.append(hl)

    if extra:
        starts.append(sine_start(n, 2.0, 1.0, 0.0))
        starts.append(sine_start(n, 3.0, 1.0, 0.7853981633974483))
        starts.append(pattern_start(n, 5, 2, 0))
        starts.append(pattern_start(n, 4, 2, 1))
        starts.append(random_binary_start(n, rng))

    return starts


def polish_candidate(start_h, n, time_limit, short=False):
    n = int(n)
    local_h, local_c = project_and_eval(start_h)

    if short:
        steps = [
            (0.020, 1.0e4, 220),
            (0.006, 1.0e5, 300),
            (0.0015, 1.0e6, 360),
        ]
    else:
        steps = [
            (0.050, 1.0e4, 260),
            (0.020, 1.0e4, 360),
            (0.008, 1.0e5, 460),
            (0.003, 1.0e5, 560),
            (0.001, 1.0e6, 640),
            (0.0003, 1.0e6, 720),
            (0.0001, 1.0e6, 700),
        ]

    scale = max(0.35, min(2.2, 800.0 / float(max(n, 1))))
    cost = max(0.006, 0.0015 + n * 2.0e-5)

    for tau, pen, base_iter in steps:
        remaining = time_limit - time.time()
        if remaining < 12.0:
            break

        maxit = int(base_iter * scale)
        maxit_time = int(remaining * 0.15 / cost)
        maxit = min(maxit, maxit_time)
        if maxit < 30:
            continue

        h2 = run_lbbfgs(local_h, n, tau, pen, maxit)
        h2, c2 = project_and_eval(h2)

        if c2 < local_c:
            local_h, local_c = h2, c2
        elif c2 < local_c + 1e-5:
            local_h = h2

    hb = rank_binarize(local_h)
    cb = objective_feasible(hb)
    if cb < local_c:
        local_h, local_c = hb, cb

    local_h, local_c = project_and_eval(local_h)
    return local_h, local_c


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget_s = float(max(20.0, budget_s))
    deadline = t0 + budget_s - 2.0
    rng = np.random.default_rng(seed)

    parent = globals().get("initial_h_values", None)
    if parent is None:
        parent_h = np.full(180, 0.5, dtype=float)
    else:
        parent_arr = np.asarray(parent, dtype=float)
        if np.size(parent_arr) == 0:
            parent_h = np.full(180, 0.5, dtype=float)
        else:
            parent_h = parent_arr.ravel().astype(float, copy=True)

    parent_h = np.clip(parent_h, 0.0, 1.0)
    parent_h = project_box_sum(parent_h, 0.5 * float(parent_h.size))
    if parent_h.size < 32:
        parent_h = project_box_sum(resample_linear(parent_h, 32), 16.0)

    best_h = parent_h.copy()
    best_c = objective_feasible(best_h)

    hb = rank_binarize(best_h)
    cb = objective_feasible(hb)
    if cb < best_c:
        best_h, best_c = hb, cb

    if time.time() > deadline - 10.0:
        best_h = project_box_sum(best_h, 0.5 * float(best_h.size))
        best_c = objective_feasible(best_h)
        return (best_h, best_c, int(best_h.size))

    base_n = best_h.size
    ns = [base_n]
    for t in [180, 360, 720, 1440, 2880, 4096]:
        if t > ns[-1] and t <= 4096:
            ns.append(t)

    total_available = max(20.0, budget_s - 40.0)
    base_fracs = [0.08, 0.15, 0.22, 0.24, 0.20, 0.11]
    if len(ns) <= len(base_fracs):
        fr = base_fracs[:len(ns)]
    else:
        fr = base_fracs[-len(ns):]
    ss = float(sum(fr))
    fr = [0.95 * (f / ss) for f in fr]

    for si, n in enumerate(ns):
        if time.time() > deadline - 35.0:
            break
        stage_alloc = total_available * fr[si]
        if stage_alloc < 20.0:
            continue

        starts = make_stage_starts(best_h, n, rng, extra=(si == 0))
        if n <= 360:
            max_starts = 6
        elif n <= 1440:
            max_starts = 4
        else:
            max_starts = 3
        starts = starts[:max_starts]

        stage_start = time.time()
        stage_deadline = min(deadline, stage_start + stage_alloc)

        for sti, st in enumerate(starts):
            rem = stage_deadline - time.time()
            if rem < 18.0:
                break
            alloc = min(rem, max(18.0, stage_alloc / float(max(1, len(starts) - sti))))
            if time.time() + 12.0 > deadline:
                break

            h2, c2 = polish_candidate(st, n, time.time() + alloc, short=(n >= 2880))
            if c2 < best_c - 1e-14:
                best_h, best_c = h2, c2

        if time.time() > deadline - 25.0:
            break

    if time.time() < deadline - 30.0:
        if best_h.size <= 2880:
            n_rand = best_h.size
        else:
            n_rand = 2880
        if n_rand < 360 and (deadline - time.time()) > 180.0:
            n_rand = 720

        base_rand = resample_step(best_h, n_rand) if best_h.size != n_rand else best_h.copy()
        target_rand = 0.5 * float(n_rand)
        sigmas = [0.20, 0.11, 0.06, 0.025]
        it = 0

        while time.time() < deadline - 18.0:
            it += 1
            choice = it % 5

            if choice == 0:
                st = project_box_sum(base_rand + rng.normal(0.0, sigmas[it % len(sigmas)], n_rand), target_rand)
            elif choice == 1:
                st = rank_binarize(base_rand + rng.normal(0.0, 0.05, n_rand))
            elif choice == 2:
                f = float(rng.integers(1, 7))
                ph = float(rng.random() * 2.0 * np.pi)
                st = sine_start(n_rand, f, 1.0, ph)
            elif choice == 3:
                st = random_binary_start(n_rand, rng)
            else:
                pr = int(rng.integers(3, 14))
                du = max(1, int(round(rng.random() * pr)))
                of = int(rng.integers(0, pr))
                st = pattern_start(n_rand, pr, du, of)

            st_pt, c0 = project_and_eval(st)
            if c0 > best_c + 0.07:
                continue

            alloc = min(60.0, max(20.0, 0.25 * (deadline - time.time())))
            if time.time() + alloc > deadline - 5.0:
                continue

            h2, c2 = polish_candidate(st_pt, n_rand, time.time() + alloc, short=True)
            if c2 < best_c - 1e-14:
                best_h, best_c = h2, c2
                base_rand = resample_step(best_h, n_rand) if best_h.size != n_rand else best_h.copy()

    final_ns = set()
    final_ns.add(best_h.size)
    final_ns.add(base_n)
    if best_h.size <= 720:
        final_ns.add(min(1440, max(720, best_h.size * 2)))
    if best_h.size >= 1440:
        final_ns.add(max(720, best_h.size // 2))

    for n in sorted(final_ns):
        if n < 32:
            continue
        if time.time() > deadline - 8.0:
            break
        try:
            h0 = resample_step(best_h, n)
            hb = rank_binarize(h0)
            cb = objective_feasible(hb)
            if cb < best_c - 1e-14:
                best_h, best_c = hb, cb
        except Exception:
            pass

    best_h = project_box_sum(best_h, 0.5 * float(best_h.size))
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = objective_feasible(best_h)

    return (best_h, float(best_c), int(best_h.size))
