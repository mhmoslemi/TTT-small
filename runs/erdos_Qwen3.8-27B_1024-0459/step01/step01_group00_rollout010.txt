    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(10000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    m = np.max(x)
    return m + (1.0/beta) * np.log(np.sum(np.exp(beta*(x - m))))

def smooth_max_obj(x, beta):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max_val(corr, beta)

def smooth_max_obj_constrained(x, beta, n):
    """Objective for SLSQP: smooth max of correlation."""
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max_val(corr, beta)

def sum_func(x, n):
    return np.sum(x) - n / 2.0

def optimize_slsqp_smooth(n, h_init, betas=None, max_iter_per_beta=1000):
    """Multi-beta SLSQP optimization."""
    if betas is None:
        betas = [10.0, 50.0, 200.0, 800.0]
    
    h = make_feasible(h_init.copy(), n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: sum_func(x, n)}]
    
    for beta in betas:
        res = minimize(
            lambda x: smooth_max_obj_constrained(x, beta, n),
            h, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': max_iter_per_beta, 'ftol': 1e-16}
        )
        if res.fun < 1e10:
            h = make_feasible(res.x, n)
    
    return h

def local_search_targeted(h, n, n_trials=2000, step=0.05):
    """Local search that targets the active (maximizing) shifts."""
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    target_sum = n / 2.0
    
    for trial in range(n_trials):
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Find the top few active shifts
        top_idx = np.argpartition(corr, -5)[-5:]
        
        # Pick a random active shift
        k_idx = top_idx[np.random.randint(len(top_idx))]
        k = k_idx - (n - 1)
        
        # Find positions that contribute most to this shift
        if k >= 0:
            valid_range = range(0, n - k)
        else:
            valid_range = range(-k, n)
        
        # Sample a position from the valid range
        positions = list(valid_range)
        if not positions:
            continue
        j = positions[np.random.randint(len(positions))]
        
        # Try to reduce h[j]*(1-h[j+k]) or h[j-k]*(1-h[j])
        h_try = h.copy()
        
        # Reduce contribution: decrease h at position where h=1 and h_shift=0
        # or increase h at position where h=0 and h_shift=1
        if 0 <= j + k < n:
            if h[j] > h[j+k]:
                h_try[j] -= step
                h_try[j+k] += step
            else:
                h_try[j] += step
                h_try[j+k] -= step
        elif 0 <= j - k < n:
            if h[j-k] > h[j]:
                h_try[j-k] -= step
                h_try[j] += step
            else:
                h_try[j-k] += step
                h_try[j] -= step
        
        h_try = np.clip(h_try, 0.0, 1.0)
        # Fix sum
        diff = target_sum - np.sum(h_try)
        if abs(diff) > 1e-10:
            mask = (h_try > 0.001) & (h_try < 0.999)
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h_try[mask] += diff / cnt
            h_try = np.clip(h_try, 0.0, 1.0)
        
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            h = h_try
            best_c5 = c5_try
    
    return h, best_c5

def coordinate_descent(h, n, n_sweeps=20):
    """Coordinate descent: optimize one variable at a time."""
    h = make_feasible(h.copy(), n)
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    
    for sweep in range(n_sweeps):
        for j in range(n):
            # Optimize h[j] by trying a range of values
            # The constraint is sum = target_sum, so changing h[j] by d means
            # we need to adjust the sum of others by -d. For simplicity, just
            # try different values of h[j] and see which gives best C5.
            
            # Binary search / golden section on h[j]
            # Actually, for a fixed h[j] = v, the rest must sum to target_sum - v.
            # This is complicated. Instead, just try a few values.
            
            best_v = h[j]
            best_c5_local = best_c5
            
            for dv in [-0.1, -0.05, -0.02, 0.02, 0.05, 0.1]:
                v_new = h[j] + dv
                if v_new < 0 or v_new > 1:
                    continue
                h[j] = v_new
                # Distribute the difference to others
                diff = target_sum - np.sum(h)
                mask = np.arange(n) != j
                mask &= (h > 0.001) & (h < 0.999)
                cnt = np.count_nonzero(mask)
                if cnt > 0:
                    h[mask] += diff / cnt
                    h = np.clip(h, 0.0, 1.0)
                c5_local = compute_c5(h)
                if c5_local < best_c5_local:
                    best_c5_local = c5_local
                    best_v = v_new
                else:
                    h[j] = best_v  # revert
                    # Redo the sum fix
                    diff = target_sum - np.sum(h)
                    mask2 = np.arange(n) != j
                    mask2 &= (h > 0.001) & (h < 0.999)
                    cnt2 = np.count_nonzero(mask2)
                    if cnt2 > 0:
                        h[mask2] += diff / cnt2
                        h = np.clip(h, 0.0, 1.0)
            
            if best_c5_local < best_c5:
                best_c5 = best_c5_local
        
        if compute_c5(h) < best_c5:
            best_c5 = compute_c5(h)
            h = make_feasible(h, n)
    
    return h, best_c5

        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Phase 8: Very fine local search
    if time_left() > 15:
        n_trials = min(20000, int(time_left() * 40))
        best_h, best_c5 = targeted_local_search(best_h, best_n, n_trials=n_trials, step=0.008)

    # Phase 9: Tiny perturbations
    if time_left() > 10:
        n_trials = min(30000, int(time_left() * 60))
        best_h, best_c5 = targeted_local_search(best_h, best_n, n_trials=n_trials, step=0.003)

    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
