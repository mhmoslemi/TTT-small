```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum(h, target):
    """Project h onto {sum(h)=target, 0<=h<=1}."""
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        # Distribute diff to free variables
        if diff > 0:
            free = h < 1.0
        else:
            free = h > 0.0
        if not np.any(free):
            break
        per = diff / np.sum(free)
        h[free] = np.clip(h[free] + per, 0.0, 1.0)
    # Final exact correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            free = (h < 1.0)
        else:
            free = (h > 0.0)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, beta):
    """Compute smooth max of corr and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Smooth max
    bcorr = beta * corr
    bcorr_max = np.max(bcorr)
    shifted = bcorr - bcorr_max
    shifted = np.clip(shifted, -500, 500)
    log_sum = bcorr_max + np.log(np.sum(np.exp(shifted)))
    f = log_sum / beta  # This is the smooth max of corr (not yet scaled by dx)
    
    # Weights
    p = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # Gradient of each corr[k] w.r.t. h
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # term1: I[lo≤m≤hi] * (1-h[m+n-1-k])
        m1 = np.arange(lo, hi + 1)
        idx1 = m1 + n - 1 - k
        grad[m1] += p[k] * (1.0 - h[idx1])
        # term2: I[lo-k+n-1≤m≤hi+n-1-k] * (-h[m-n+1+k])
        lo2 = lo - k + n - 1
        hi2 = hi + n - 1 - k
        lo2 = max(0, lo2)
        hi2 = min(n - 1, hi2)
        if lo2 <= hi2:
            m2 = np.arange(lo2, hi2 + 1)
            idx2 = m2 - n + 1 + k
            # Need to check idx2 is valid (0 ≤ idx2 < n)
            valid = (idx2 >= 0) & (idx2 < n)
            if np.any(valid):
                grad[m2[valid]] += p[k] * (-h[idx2[valid]])
    
    return f * dx, grad * dx
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        count = np.sum(mask)
        if count == 0:
            break
        h[mask] = np.clip(h[mask] + diff / count, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    b = beta * corr
    bmax = np.max(b)
    shifted = np.clip(b - bmax, -500, 500)
    e = np.exp(shifted)
    s = np.sum(e)
    return (bmax + np.log(s)) / beta * dx

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    b = beta * corr
    bmax = np.max(b)
    shifted = np.clip(b - bmax, -500, 500)
    e = np.exp(shifted)
    s = np.sum(e)
    p = e / s  # softmax weights
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # term1: for m in [lo, hi], contribution (1 - h[m + n - 1 - k])
        m1 = np.arange(lo, hi + 1)
        idx1 = m1 + n - 1 - k
        grad[m1] += p[k] * (1.0 - h[idx1])
        # term2: for m in [lo-k+n-1, hi+n-1-k] (clamped to [0,n-1]), contribution -h[m-n+1+k]
        lo2 = max(0, lo - k + n - 1)
        hi2 = min(n - 1, hi + n - 1 - k)
        if lo2 <= hi2:
            m2 = np.arange(lo2, hi2 + 1)
            idx2 = m2 - n + 1 + k
            valid = (idx2 >= 0) & (idx2 < n)
            if np.any(valid):
                grad[m2[valid]] -= p[k] * h[idx2[valid]]
    
    return grad * dx

def run_slsqp(h0, n, target, beta, max_iter=300, ftol=1e-12):
    h0 = project_sum(h0, target)
    
    def obj(h):
        return smooth_max_obj(h, beta)
    
    def grad(h):
        return smooth_max_grad(h, beta)
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': max_iter, 'ftol': ftol})
        if res.success or res.fun < obj(h0):
            return project_sum(res.x, target)
    except:
        pass
    return h0
```

```python
    return (bmax + np.log(e.sum() + 1e-300)) / _beta * dx

def _grad(h):
    n = len(h)
    dx = 2.0 / n
    hc = np.clip(h, 0.0, 1.0)
    corr = np.correlate(hc, 1.0 - hc, mode='full')
    b = _beta * corr
    bmax = b.max()
    sh = np.clip(b - bmax, -500.0, 0.0)
    e = np.exp(sh)
    p = e / (e.sum() + 1e-300)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        s = n - 1 - k
        m1 = np.arange(lo, hi + 1)
        g[m1] += w * (1.0 - hc[m1 + s])
        lo2 = max(0, lo + s)
        hi2 = min(n - 1, hi + s)
        if lo2 <= hi2:
            m2 = np.arange(lo2, hi2 + 1)
            g[m2] -= w * hc[m2 - s]
    return g * dx

def _cfun(x):
    return np.sum(x) - _ct

def _cjac(x):
    return np.ones(len(x))

def do_optimize_sl(h0, n, beta_list, iter_list, time_budget):
    global _ct, _beta
    target = n / 2.0
    _ct = target
    h = project_sum(h0, target)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    record_c5 = compute_c5(h)
    t_start = time.time()
    for i in range(len(beta_list)):
        if time.time() - t_start > time_budget:
            break
        _beta = beta_list[i]
        mi = iter_list[i] if i < len(iter_list) else iter_list[-1]
        try:
            res = minimize(_obj, h, jac=_grad, method='SLSQP',
                         bounds=bounds, cons=cons,
                         options={'maxiter': mi, 'ftol': 1e-12})
            h_new = project_sum(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < record_c5 + 1e-10:
                h = h_new
                record_c5 = c5_new
        except Exception:
            pass
    return h, record_c5

def do_optimize_lbfgs(h0, n, beta_list, iter_list, time_budget):
    global _ct, _beta
    target = n / 2.0
    _ct = target
    h = project_sum(h0, target)
    bounds = [(0.0, 1.0)] * n
    record_c5 = compute_c5(h)
    t_start = time.time()
    for i in range(len(beta_list)):
        if time.time() - t_start > time_budget:
            break
        _beta = beta_list[i]
        mi = iter_list[i] if i < len(iter_list) else iter_list[-1]
        
        def _pen_obj(x):
            base = _obj(x)
            pen = 500.0 * (np.sum(x) - target) ** 2
            return base + pen
        
        def _pen_grad(x):
            base_g = _grad(x)
            pen_g = 1000.0 * (np.sum(x) - target)
            return base_g + pen_g * np.ones(n)
        
        try:
            res = minimize(_pen_obj, h, jac=_pen_grad, method='L-BFGS-B',
                         bounds=bounds, options={'maxiter': mi, 'ftol': 1e-12})
            h_new = project_sum(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < record_c5 + 1e-10:
                h = h_new
                record_c5 = c5_new
        except Exception:
            pass
    return h, record_c5

def make_block_init(n, n_blocks):
    h = np.zeros(n)
    block_size = max(1, n // n_blocks)
    for i in range(n_blocks):
        start = i * block_size
        end = min(start + block_size, n)
        if i % 2 == 0:
            h[start:end] = 0.8
        else:
            h[start:end] = 0.2
    return h

def make_sin_init(n, freq):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x / 2.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _ct, _beta
    t0 = time.time()
    np.random.seed(seed)
    
    h_p = np.array(initial_h_values, dtype=float)
    np_p = len(h_p)
    c5_p = compute_c5(h_p)
    
    best_h = h_p.copy()
    best_c5 = c5_p
    best_n = np_p
    
    def elapsed():
        return time.time() - t0
    
    # Phase 1: Optimize at parent resolution with SLSQP + analytical grad
    if elapsed() < budget_s * 0.15:
        h_new, c5_new = do_optimize_sl(h_p, np_p,
            [5, 20, 50, 150, 400], [200, 250, 250, 200, 150], budget_s * 0.12)
        if c5_new < best_c5:
            best_h = project_sum(h_new, np_p / 2.0)
            best_c5 = c5_new
    
    # Phase 2: Try higher resolutions
    resolutions = [100, 130, 160, 200, 250, 300]
    for n_try in resolutions:
        if elapsed() > budget_s * 0.80:
            break
        remaining = budget_s * 0.80 - elapsed()
        if remaining < 40:
            break
        time_alloc = min(remaining * 0.85, 180.0)
        
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_start = project_sum(h_interp, n_try / 2.0)
        
        # SLSQP from interpolated
        h_opt, c5_opt = do_optimize_sl(h_start, n_try,
            [5, 20, 60, 200, 500], [150, 200, 200, 200, 150], time_alloc)
        if c5_opt < best_c5:
            best_h = project_sum(h_opt, n_try / 2.0)
            best_c5 = c5_opt
            best_n = n_try
        
        # L-BFGS-B from same start if time allows
        if elapsed() < budget_s * 0.78:
            h_opt2, c5_opt2 = do_optimize_lbfgs(h_start, n_try,
                [10, 50, 200, 500], [300, 300, 200, 150], 60.0)
            if c5_opt2 < best_c5:
                best_h = project_sum(h_opt2, n_try / 2.0)
                best_c5 = c5_opt2
                best_n = n_try
    
    # Phase 3: Diverse starting points at best resolution
    if elapsed() < budget_s * 0.88:
        n_b = best_n
        starts = []
        # Block pattern
        starts.append(project_sum(make_block_init(n_b, 6), n_b / 2.0))
        starts.append(project_sum(make_block_init(n_b, 10), n_b / 2.0))
        starts.append(project_sum(make_sin_init(n_b, 3), n_b / 2.0))
        starts.append(project_sum(make_sin_init(n_b, 5), n_b / 2.0))
        # Random
        starts.append(project_sum(np.random.beta(2, 2, n_b), n_b / 2.0))
        starts.append(project_sum(np.random.beta(1, 1, n_b), n_b / 2.0))
        
        for h_st in starts:
            if elapsed() > budget_s * 0.92:
                break
            h_opt, c5_opt = do_optimize_sl(h_st, n_b,
                [10, 50, 200, 500], [150, 200, 200, 150], 50.0)
            if c5_opt < best_c5:
                best_h = project_sum(h_opt, n_b / 2.0)
                best_c5 = c5_opt
    
    # Phase 4: Perturbation around best
    if elapsed() < budget_s * 0.94:
        for scale in [0.03, 0.01, 0.005]:
            if elapsed() > budget_s * 0.95:
                break
            h_pert = project_sum(best_h + np.random.randn(best_n) * scale, best_n / 2.0)
            h_opt, c5_opt = do_optimize_sl(h_pert, best_n,
                [100, 300, 800], [200, 200, 150], 40.0)
            if c5_opt < best_c5:
                best_h = project_sum(h_opt, best_n / 2.0)
                best_c5 = c5_opt
    
    # Phase 5: Final sharp refinement
    if elapsed() < budget_s * 0.97:
        h_ref, c5_ref = do_optimize_sl(best_h, best_n,
            [500, 1000, 2000], [150, 100, 80], budget_s * 0.025)
        if c5_ref < best_c5:
            best_h = project_sum(h_ref, best_n / 2.0)
            best_c5 = c5_ref
    
    # Final exact projection and evaluation
    best_h = project_sum(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)
    
    return (best_h, final_c5, best_n)
```
