    oness = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return None

    # Δ from a: 1->0 is 1[h[a+km]=1]*1[a+km<n] - 1[h[a-km]=0]*1[a>=km]
    costs_a = np.zeros(len(ones))
    for idx, a in enumerate(ones):
        delta = 0.0
        if a + km < n and h[a + km] == 1.0:
            delta += 1.0
        if a >= km and h[a - km] == 0.0:
            delta -= 1.0
        costs_a[idx] = delta

    c = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * c
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / beta * (2.0 / n))


def soft_grad(h, n, beta):
    c = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * c
    m = np.max(scaled)
    w = np.exp(scaled - m)

            grad_h[lag:] += wk * (1.0 - h[:n - lag])

        else:

def binary_local_search(h, n, rng, t0, budget_s, reserve_time, directed_fraction=0.4, sa_t_initial=0.02, sa_t_final=0.001):
    h = h.copy()
    c_arr = np.correlate(h, 1.0 - h, mode="full")
    c5 = float(np.max(c_arr) * (2.0 / n))
    best_c5 = c5
    best_h = h.copy()
    
    deadline = t0 + budget_s - reserve_time
    it = 0
    
    while time.time() < deadline:
        it += 1
        # Annealing temperature
        progress = min(1.0, it / 200000.0)
        T = sa_t_initial * (sa_t_final / sa_t_initial) ** progress
        
        r = rng.random()
        
        if r < directed_fraction:
            # Directed move: target worst lag
            k_idx = np.argmax(c_arr)
            km = k_idx - (n - 1)
            if km < 0:
                km = -km
            if km == 0:
                # k=0 is always 0 for binary, so pick next worst
                sorted_idx = np.argsort(-c_arr)
                for si in sorted_idx:
                    candidate_km = si - (n - 1)
                    if abs(candidate_km) > 0:
                        km = abs(candidate_km)
                        break
                else:
                    km = 1
            
            # Compute costs
            cost_1to0 = np.zeros(n)
            cost_0to1 = np.zeros(n)
            if km < n:
                cost_1to0[:n-km] += h[km:]
                cost_0to1[:n-km] -= h[km:]
                cost_1to0[km:] -= (1.0 - h[:n-km])
                cost_0to1[km:] += (1.0 - h[:n-km])
            
            # Find best a (1→0) and b (0→1)
            mask_1 = (h == 1.0)
            mask_0 = (h == 0.0)
            
            if mask_1.sum() == 0 or mask_0.sum() == 0:
                break
            
            # Among positions with h=1, find the one with minimum cost_1to0
            costs_a = np.where(mask_1, cost_1to0, np.inf)
            costs_b = np.where(mask_0, cost_0to1, np.inf)
            
            a = np.argmin(costs_a)
            b = np.argmin(costs_b)
            
            if a == b:
                # Try second best
                costs_a[a] = np.inf
                a = np.argmin(costs_a)
                if a == b:
                    costs_b[b] = np.inf
                    b = np.argmin(costs_b)
                    if a == b:
                        continue
            
            # Perform swap
            h[a] = 0.0
            h[b] = 1.0
            c_arr_new = np.correlate(h, 1.0 - h, mode="full")
            c5_new = float(np.max(c_arr_new) * (2.0 / n))
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                c_arr = c_arr_new
            elif T > 1e-10:
                delta = c5_new - best_c5
                if delta < 0.1 and rng.random() < np.exp(-delta / T):
                    c_arr = c_arr_new
                else:
                    h[a] = 1.0
                    h[b] = 0.0
            else:
                h[a] = 1.0
                h[b] = 0.0
        else:
            # Random move
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
