    n = len(h)
    h = np.array(h, dtype=float)
    lo, hi = -1.0, 1.0  # range for lambda
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid

So minimizing max_k f_k ≈ maximizing min_k sum_{valid} d[i]*d[i+k]

This is like maximizing the minimum autocorrelation of d. For d with zero mean, the autocorrelation at k=0 is sum d[i]² > 0, and we want it to stay high at all k.

This suggests d should be a "smooth" function (like a low-frequency sinusoid) to maintain high autocorrelation at all lags.

d[i] = a*cos(2π*p*i/n) for some low p would give:
sum d[i]*d[i+k] ≈ (n/2)*a²*cos(2π*p*k/n)

The minimum over k occurs at k = n/(2p), giving -n*a²/2.
So f_k ≈ n/4 + n*a²/2 at worst.
C₅ ≈ (n/4 + n*a²/2) * 2/n = 1/2 + a²

That's worse! We want d[i]*d[i+k] to be positive on average, not to go negative.

What about d[i] = a*cos(π*i/n)? (half period)
sum d[i]*d[i+k] involves sum cos(π*i/n)*cos(π*(i+k)/n)
= (1/2) sum [cos(2π*i/n + π*k/n) + cos(π*k/n)]
≈ (n/2)*cos(π*k/n) / 2 + n*cos(π*k/n)/2 ...

def projected_smooth_descent(h0, n_points, max_iter=3000, tau_schedule=None, lr0=0.005):
    n = n_points
    target = n / 2.0
    h = project_box_sum(h0.copy(), target)
    best_h = h.copy()

    if tau_schedule is None:

    lr = lr0

        iters_per_tau = max_iter // len(tau_schedule)
        for it in range(iters_per_tau):

            gnorm = np.linalg.norm(grad)

                break
            step = lr / (1.0 + 0.1 * gnorm)

                h_new = project_box_sum(h - step * grad, target)
                new_true = compute_c5(h_new, n)

                if new_true < best_val - 1e-12 or new_smooth < val - 1e-12 * step:
                    break
                step *= 0.5
            h = h_new
            true_val = compute_c5(h, n)
            if true_val < best_val:
                best_val = true_val
                best_h = h.copy()
            if it % 100 == 99:

    return best_h, best_val


def slsqp_polish(h0, n_points, maxiter=500):
    n = n_points
    target = n / 2.0

    def obj(x):
        return compute_c5(x, n)

    def jac(x):
        dx = 2.0 / n
        conv = np.correlate(x, 1.0 - x, mode='full')
        k_max = int(np.argmax(conv))
        m = k_max - (n - 1)
        grad = np.zeros(n)
        if m == 0:
            grad[:n] = (1.0 - 2.0 * x) * dx
        elif m > 0:
            grad[:n - m] += (1.0 - x[m:]) * dx
            grad[m:] += (-x[:n - m]) * dx
        else:
            p = -m
            grad[p:] += (1.0 - x[:n - p]) * dx
            grad[:n - p] += (-x[p:]) * dx
        return grad

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    h_start = project_box_sum(h0.copy(), target)
    try:
        res = minimize(obj, h_start, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_box_sum(res.x, target)
        return h_out, compute_c5(h_out, n)
    except Exception:
        return h_start, compute_c5(h_start, n)


def random_binary_init(n, rng):
    h = np.zeros(n)
    ones = rng.permutation(n)[:n // 2]
    h[ones] = 1.0
    return h


def structured_init(n, rng, strategy):
    h = np.zeros(n)
    if strategy == 0:
        half = n // 2
        h[:half] = 1.0
        rng.shuffle(h)
    elif strategy == 1:
        h = 0.5 + 0.4 * np.cos(np.linspace(0, np.pi, n))
    elif strategy == 2:
        period = max(4, n // 10)
        for i in range(n):
            phase = (i % period) / period
            h[i] = 0.5 + 0.3 * np.cos(2 * np.pi * phase)
    elif strategy == 3:
        h = 0.5 + rng.uniform(-0.2, 0.2, n)
    elif strategy == 4:
        h = random_binary_init(n, rng) + rng.normal(0, 0.1, n)
    else:
        h = rng.uniform(0.2, 0.8, n)
    return project_box_sum(h, n / 2.0)


def try_swap_perturb(h, n_points, rng, n_swaps=5):
    target = n / 2.0
    h_new = h.copy()
    for _ in range(n_swaps):

            h_new[i] = swap_val
            h_new[j] = dj - (swap_val - di)
    return project_box_sum(h_new, target)
