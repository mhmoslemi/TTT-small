    """Run SLSQP with a sequence of alpha values (soft-max smoothing)."""
    n = int(n)
    deadline = time.time() + max(5.0, float(time_limit))
    if _RUN_DEADLINE is not None:
        deadline = min(deadline, _RUN_DEADLINE - 3.0)

    h = as_feasible(h, n).copy()
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": constraint_eq, "jac": constraint_jac}]

    for alpha in alphas:
        remaining = deadline - time.time()
        if remaining < 10.0:
            break
        if _RUN_DEADLINE is not None and time.time() >= _RUN_DEADLINE - 5.0:
            break

        maxiter = int(max(30, min(300, 100.0 * (100.0 / float(n)) * math.sqrt(remaining / 60.0))))
        if alpha < 1e-14:
            maxiter = int(max(50, min(500, maxiter * 1.5)))

        old_dl = _RUN_DEADLINE
        _RUN_DEADLINE = min(deadline, _RUN_DEADLINE if _RUN_DEADLINE is not None else deadline)
        try:
            res = minimize(
                slsqp_objective, h, args=(alpha,),
                jac=slsqp_gradient, method="SLSQP",
                bounds=bounds, constraints=constraints,
                options={"maxiter": maxiter, "ftol": 1e-10, "disp": 0}
            )
        except BudgetError:
            res = None
        except Exception:
            res = None
        finally:
            _RUN_DEADLINE = old_dl

        if res is not None and getattr(res, "x", None) is not None:
            h_new = as_feasible(res.x, n)
            c_new = compute_c5(h_new)
            if math.isfinite(c_new) and c_new < compute_c5(h) + 1e-10:
                h = h_new
                state.update(h, n)
        else:
            # Keep current h if optimization failed
            pass

    return h


def synthetic_init(n, rng):
    """Create a reasonable starting point."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = []
    for f in (1, 2, 3, 4, 5):
        for ph in (0.0, 0.3, 0.5 * math.pi):
            h = 0.5 + 0.3 * np.cos(2.0 * math.pi * f * x / 2.0 + ph)
            h = as_feasible(h, n)
            candidates.append(h)
    # Binary random
    for _ in range(5):
        h = make_binary(None, n, rng).astype(float)
        h = as_feasible(h, n)
        candidates.append(h)
    best = candidates[0]
    best_c = compute_c5(best)
    for h in candidates[1:]:
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best = h
    return best


def run(seed=42, budget_s=1300, **kwargs):
    global _RUN_DEADLINE
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0.0:
        budget_s = 30.0
    reserve = 15.0 if budget_s > 60.0 else max(2.0, 0.1 * budget_s)
    _RUN_DEADLINE = t0 + max(1.0, budget_s - reserve)

    rng = np.random.default_rng(seed)
    state = BestState()

    # Get parent
    parent = get_parent()
    if parent is not None and parent.size > 0:
        n0 = int(parent.size)
        h0 = as_feasible(parent, n0)
        c0 = compute_c5(h0)
    else:
        n0 = 70
        h0 = synthetic_init(n0, rng)
        c0 = compute_c5(h0)
    state.update(h0, n0)

    # Phase 1: Binary local search at n0 for a good discrete starting point
    if n0 % 2 == 0:
        bin_time = min(30.0, max(5.0, (_RUN_DEADLINE - time.time()) * 0.05))
        try:
            bh, bc = binary_local_search(h0, n0, state, rng, bin_time)
            if bh is not None:
                bh_f = as_feasible(bh, n0)
                bh_f = 0.5 * bh_f + 0.5 * h0  # Blend binary with original
                state.update(bh_f, n0)
        except Exception:
            pass

    # Phase 2: SLSQP at n0 with alpha decay
    h_best = state.h.copy() if state.h is not None else h0.copy()
    n_try = int(n0)
    slsqp_time = min(400.0, max(60.0, (_RUN_DEADLINE - time.time()) * 0.45))
    alphas = [1.0, 0.5, 0.2, 0.1, 0.05, 0.0]
    h_best = run_slsqp_phase(h_best, n_try, state, slsqp_time, alphas)

    # Phase 3: Targeted reduction at n0
    tgt_time = min(60.0, max(10.0, (_RUN_DEADLINE - time.time()) * 0.1))
    h_best = targeted_reduction(h_best, n_try, state, tgt_time)

    # Phase 4: SLSQP at 2*n0
    if _RUN_DEADLINE is not None and time.time() < _RUN_DEADLINE - 120.0:
        n2 = 2 * n_try
        h2 = upsample_linear(state.h if state.h is not None else h_best, n2)
        h2 = as_feasible(h2, n2)
        c2 = compute_c5(h2)
        if c2 < state.c:
            state.update(h2, n2)

        slsqp_time2 = min(300.0, max(40.0, (_RUN_DEADLINE - time.time()) * 0.35))
        alphas2 = [0.5, 0.2, 0.1, 0.05, 0.0]
        h2 = run_slsqp_phase(h2, n2, state, slsqp_time2, alphas2)

        tgt_time2 = min(40.0, max(10.0, (_RUN_DEADLINE - time.time()) * 0.08))
        h2 = targeted_reduction(h2, n2, state, tgt_time2)

    # Phase 5: Try n=140 with different initialization
    if _RUN_DEADLINE is not None and time.time() < _RUN_DEADLINE - 100.0:
        n3 = 140
        # Upsample from best
        h3 = upsample_linear(state.h if state.h is not None else h_best, n3)
        h3 = as_feasible(h3, n3)
        c3 = compute_c5(h3)
        # Also try from parent upsampled
        if parent is not None:
            h3p = upsample_linear(parent, n3)
            h3p = as_feasible(h3p, n3)
            c3p = compute_c5(h3p)
            if c3p < c3:
                h3 = h3p
                c3 = c3p
        state.update(h3, n3)

        slsqp_time3 = min(200.0, max(30.0, (_RUN_DEADLINE - time.time()) * 0.3))
        alphas3 = [0.3, 0.1, 0.05, 0.0]
        h3 = run_slsqp_phase(h3, n3, state, slsqp_time3, alphas3)

        if _RUN_DEADLINE is not None and time.time() < _RUN_DEADLINE - 20.0:
            tgt_time3 = min(30.0, max(5.0, (_RUN_DEADLINE - time.time()) * 0.05))
            h3 = targeted_reduction(h3, n3, state, tgt_time3)

    # Phase 6: Final SLSQP polish at the best resolution
    if state.h is not None and _RUN_DEADLINE is not None and time.time() < _RUN_DEADLINE - 30.0:
        n_best = int(state.n)
        polish_time = min(120.0, max(20.0, (_RUN_DEADLINE - time.time()) * 0.2))
        alphas_polish = [0.02, 0.01, 0.0]
        h_polish = run_slsqp_phase(state.h.copy(), n_best, state, polish_time, alphas_polish)

    # Final
    if state.h is not None:
        h_final = as_feasible(state.h, state.n)
        c_final = compute_c5(h_final)
        n_final = int(state.n)
    else:
        h_final = as_feasible(h0, n0)
        c_final = compute_c5(h_final)
        n_final = n0

    if not is_valid_candidate(h_final):
        h_final = as_feasible(h_final, h_final.size)
        c_final = compute_c5(h_final)
        n_final = int(h_final.size)

    return h_final, float(c_final), int(n_final)
