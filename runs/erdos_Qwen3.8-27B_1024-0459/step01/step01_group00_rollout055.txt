            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_max

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)

def perturbation_search(h0, n, n_trials=10000, step_sizes=None, rng=None):
    if rng is None:
        rng = np.random.RandomState(777)
    h = make_feasible(h0.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    if step_sizes is None:
        step_sizes = [0.003, 0.008, 0.02, 0.05, 0.1]
    
    for _ in range(n_trials):
        h_try = h.copy()
        n_mod = rng.randint(1, 8)
        idxs = rng.choice(n, n_mod, replace=False)
        ss = step_sizes[rng.randint(len(step_sizes))]
        for idx in idxs:
            h_try[idx] += rng.uniform(-ss, ss)
        
        h_try = np.clip(h_try, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h_try)
        free = (h_try > 0.005) & (h_try < 0.995)
        cnt = np.count_nonzero(free)
        if cnt > 0:
            h_try[free] += diff / cnt
        h_try = np.clip(h_try, 0.0, 1.0)
        
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try.copy()
    
    return best_h, best_c5

def coordinate_descent(h0, n, n_sweeps=20, rng=None):
    if rng is None:
        rng = np.random.RandomState(99)
    h = make_feasible(h0.copy(), n)
    best_c5 = compute_c5(h)
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            # Try changing h[i] to optimal value given others
            # Fix sum: h[i] = target - sum(others)
            others_sum = np.sum(h) - h[i]
            target = n / 2.0
            # h[i] is fixed by the sum constraint!
            # So instead, do a 2-swap: change h[i] and h[j]
            j = rng.randint(n)
            if j == i:
                continue
            # Find optimal delta: h[i] += delta, h[j] -= delta
            # corr changes linearly in delta
            # Find the delta that minimizes max(corr + delta * grad)
            # Approximate: try a few candidate deltas
            old_hi = h[i]
            old_hj = h[j]
            for delta in [0.01, -0.01, 0.05, -0.05, 0.1, -0.1]:
                new_hi = old_hi + delta
                new_hj = old_hj - delta
                if new_hi < 0 or new_hi > 1 or new_hj < 0 or new_hj > 1:
                    continue
                h[i] = new_hi
                h[j] = new_hj
                c5_try = compute_c5(h)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                else:
                    h[i] = old_hi
                    h[j] = old_hj
                    break
            else:
                pass
        h = make_feasible(h, n)
    
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if len(h_parent) == 0:
            h_parent = None
    except Exception:
        h_parent = None
    
    if h_parent is not None:
        best_n = len(h_parent)
        best_h = make_feasible(h_parent.copy(), best_n)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5
    
    best_c5 = compute_c5(best_h)
    
    # === Phase 1: SLSQP smooth max annealing at parent resolution ===
    n = best_n
    for beta in [30, 60, 120, 250, 500, 1000, 2000]:
        if time_left() < 20:
            break
        h_opt = slsqp_optimize(best_h, n, beta=beta, maxiter=2000)
        if h_opt is not None:
            c5_new = compute_c5(h_opt)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_opt
    
    # === Phase 2: Binary search ===
    if time_left() > 80:
        # Multiple binary starting points
        binary_starts = [binary_round(best_h)]
        binary_starts.append(thue_morse_binary(best_n))
        binary_starts.append(random_binary(best_n, rng))
        
        for b_start in binary_starts:
            if time_left() < 30:
                break
            # Pair swap local search
            h_bin_opt, bin_max = binary_pair_swap_search(b_start, best_n,
                                                         max_rounds=50,
                                                         n_trials_per_round=4000,
                                                         rng=rng)
            c5_bin = bin_max * 2.0 / best_n
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin_opt.copy()
                best_n = best_n
            
            # Simulated annealing from the best binary
            if time_left() > 30:
                h_bin_sa, sa_max = binary_simulated_annealing(h_bin_opt, best_n,
                                                             max_iters=30000,
                                                             t_start=3.0, t_end=0.05,
                                                             rng=rng)
                c5_sa = sa_max * 2.0 / best_n
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_bin_sa.copy()
        
        # SLSQP refinement from best binary
        if time_left() > 40:
            for beta in [50, 200, 800]:
                if time_left() < 15:
                    break
                h_opt = slsqp_optimize(best_h, best_n, beta=beta, maxiter=2000)
                if h_opt is not None:
                    c5_new = compute_c5(h_opt)
                    if c5_new < best_c5 - 1e-12:
                        best_c5 = c5_new
                        best_h = h_opt
    
    # === Phase 3: Try nearby resolutions ===
    res_list = [best_n]
    for dn in [-4, -3, -2, -1, 1, 2, 3, 4, 8, 16, 40, 80]:
        res_list.append(best_n + dn)
    # Filter valid
    res_list = sorted(set(r for r in res_list if r >= 20 and r <= 512 and r % 2 == 0))
    
    for n_try in res_list:
        if time_left() < 30:
            break
        if n_try == best_n:
            continue
        
        h_interp = interpolate_to_n(best_h, n_try)
        
        # SLSQP at new resolution
        for beta in [50, 200, 800]:
            if time_left() < 15:
                break
            h_opt = slsqp_optimize(h_interp, n_try, beta=beta, maxiter=1500)
            if h_opt is not None:
                c5_new = compute_c5(h_opt)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h_opt
                    best_n = n_try
        
        # Binary search at new resolution
        if time_left() > 30:
            h_bin_try = binary_round(h_interp)
            h_bin_opt, bin_max = binary_pair_swap_search(h_bin_try, n_try,
                                                         max_rounds=30,
                                                         n_trials_per_round=3000,
                                                         rng=rng)
            c5_bin = bin_max * 2.0 / n_try
            if c5_bin < best_c5 - 1e-12:
                best_c5 = c5_bin
                best_h = h_bin_opt.copy()
                best_n = n_try
    
    # === Phase 4: COBYLA on the best solution ===
    if time_left() > 30:
        h_cobyla = cobyla_optimize(best_h, best_n, maxiter=2000, rhobeg=0.008)
        if h_cobyla is not None:
            c5_cob = compute_c5(h_cobyla)
            if c5_cob < best_c5 - 1e-12:
                best_c5 = c5_cob
                best_h = h_cobyla
    
    # === Phase 5: Fine perturbation polish ===
    if time_left() > 30:
        n_trials = min(15000, int((time_left() - 10) * 300))
        best_h, best_c5 = perturbation_search(best_h, best_n,
                                              n_trials=n_trials,
                                              step_sizes=[0.002, 0.005, 0.01, 0.02, 0.04],
                                              rng=rng)
    
    # === Phase 6: Final SLSQP tight optimization ===
    if time_left() > 25:
        for beta in [500, 1000, 2000, 5000]:
            if time_left() < 10:
                break
            h_opt = slsqp_optimize(best_h, best_n, beta=beta, maxiter=5000)
            if h_opt is not None:
                c5_new = compute_c5(h_opt)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_opt
    
    # === Phase 7: Ultra-fine perturbation ===
    if time_left() > 20:
        n_trials = min(10000, int((time_left() - 5) * 500))
        best_h, best_c5 = perturbation_search(best_h, best_n,
                                              n_trials=n_trials,
                                              step_sizes=[0.001, 0.003, 0.006],
                                              rng=rng)
    
    # Final feasibility check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
