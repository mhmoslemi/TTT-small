    return c if math.isfinite(c) else 1e6

def compute_scores(h):
    return np.correlate(h, 1.0 - h, mode="full")

def as_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = max(1, int(np.count_nonzero(mask)))
        if mask.any():
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def score_vector(h):
    n = h.size
    return np.correlate(h, 1.0 - h, mode="full")

def gradient_of_max(h, scores, alpha):
    """Compute gradient of smooth max (or exact max subgradient) of scores."""
    n = h.size
    m = float(np.max(scores))
    if alpha > 1e-14:
        w = np.exp((scores - m) / alpha)
        A = float(np.sum(w))
        if A <= 0 or not math.isfinite(A):
            return np.zeros(n), m
        w = w / A
    else:
        # Exact max: average over active constraints
        tol = max(1e-8, 1e-6 * max(1.0, m))
        active = np.flatnonzero(scores >= m - tol)
        if len(active) == 0:
            active = np.array([int(np.argmax(scores))])
        w = np.zeros(len(scores))
        w[active] = 1.0 / len(active)
    
    # Compute gradient
    grad = np.zeros(n)
    # For each active (or weighted) lag j:
    # k = j - (n-1)
    for j in range(len(scores)):
        wk = w[j]
        if wk < 1e-14:
            continue
        k = j - (n - 1)
        if k >= 0:
            # score = sum_{n=0}^{n-1-k} h[n+k]*(1-h[n])
            # d/dh[i] = (1-h[i-k]) * [k<=i<n] - h[i+k] * [0<=i<n-k]
            if k == 0:
                grad += wk * (1.0 - 2.0 * h)
            else:
                # +term: i from k to n-1
                grad[k:] += wk * (1.0 - h[:n-k])
                # -term: i from 0 to n-k-1
                grad[:n-k] += wk * (-h[k:])
        else:
            # k < 0, d = -k
            d = -k
            # score = sum_{j=0}^{n-1-d} h[j]*(1-h[j+d])
            # d/dh[i] = (1-h[i+d]) * [0<=i<n-d] - h[i-d] * [d<=i<n]
            grad[:n-d] += wk * (1.0 - h[d:])
            grad[d:] += wk * (-h[:n-d])
    return grad, m

def slsqp_objective(x, alpha):
    check_deadline()
    h = np.clip(x, 0.0, 1.0)
    scores = score_vector(h)
    m = float(np.max(scores))
    if alpha > 1e-14:
        w = np.exp((scores - m) / alpha)
        A = float(np.sum(w))
        if math.isfinite(A) and A > 0:
            val = m + alpha * math.log(A)
        else:
            val = m
    else:
        val = m
    return float(val) * (2.0 / len(h))

def slsqp_gradient(x, alpha):
    check_deadline()
    h = np.clip(x, 0.0, 1.0)
    scores = score_vector(h)
    grad, m = gradient_of_max(h, scores, alpha)
    return grad * (2.0 / len(h))

def constraint_eq(x, *args):
    n = max(1, int(x.size))
    return (float(np.sum(x)) - 0.5 * float(n)) / float(n)

def constraint_jac(x, *args):
    n = max(1, int(x.size))
    return np.full((1, n), 1.0 / float(n))

def upsample_linear(parent, n):
    n0 = parent.size
    if n == n0:
        return parent.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x, x0, parent).astype(float)

class BestState:
    def __init__(self):
        self.h = None
        self.c = 1e9
        self.n = 0
    def update(self, h, n):
        c = compute_c5(h)
        if math.isfinite(c) and c < self.c:
            self.h = h.copy()
            self.c = c
            self.n = n
