    n = len(x)
    target = n / 2.0
    
    # Bisection on lambda
    lo, hi = -2.0, 2.0

        projected = np.clip(x - mid, 0, 1)
        s = np.sum(projected)
        if s > target:
            lo = mid
        else:
            hi = mid
    projected = np.clip(x - (lo + hi) / 2, 0, 1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13 * n:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            step = min(diff / cnt, (1.0 - h[mask]).max() * 0.9)
            h[mask] += step
        else:
            mask = h > 1e-15
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            step = min(-diff / cnt, h[mask].min() * 0.9)
            h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if np.any(mask):
            h[mask] += diff / np.count_nonzero(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def _pen_fun(x, n, beta, mu):
    corr = np.correlate(x, 1.0 - x, 'full')
    val = smooth_max_val(corr, beta)
    val += 0.5 * mu * (np.sum(x) - n / 2.0) ** 2

def penalty_optimize(h0, n, beta, mu, maxiter=500, gtol=1e-10):
    h0 = make_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_pen_fun, h0, args=(n, beta, mu), jac=_pen_jac,
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'gtol': gtol, 'ftol': 1e-15})

def subgradient_step(h, n, n_active=5, step_size=0.01):
    """Take a subgradient step for the non-smooth max."""
    corr = np.correlate(h, 1.0 - h, 'full')
    idx_sorted = np.argsort(corr)[::-1]
    active = idx_sorted[:n_active]
    # Average gradient over active shifts
    grad = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        for j in range(n):
            i1 = j - s
            i2 = j + s
            term1 = (1.0 - h[i1]) if (0 <= i1 < n) else 0.0
            term2 = h[i2] if (0 <= i2 < n) else 0.0
            grad[j] += term1 - term2
    grad /= n_active
    # Step
    h_new = h - step_size * grad
    h_new = make_feasible(h_new, n)
    return h_new

def binary_init_mseq(n):
    """Generate a maximal-length sequence (m-sequence) of length n (if n=2^m-1)."""
    # Find closest 2^m-1
    m = int(np.log2(n + 1))
    L = (1 << m) - 1
    if L != n:
        return None
    # Generate m-sequence using LFSR
    # For m bits, use primitive polynomial
    polynomials = {3: 0b110, 4: 0b1011, 5: 0b10011, 6: 0b101101, 7: 0b1000111}
    if m not in polynomials:
        return None
    poly = polynomials[m]
    state = 1
    seq = []
    for _ in range(L):
        bit = state & 1
        seq.append(bit)
        feedback = (state ^ (state >> (m-1))) & 1
        # Apply XOR of taps
        fb = 0
        t = state
        for i in range(m):
            if (poly >> i) & 1:
                fb ^= (t >> i) & 1
        state = (state >> 1) | (fb << (m - 1))
        state &= (1 << m) - 1
    h = np.array(seq, dtype=float)
    return h

                break
            for _ in range(200):
                if remaining() < 2:
                    break
                h_sub = subgradient_step(h_sub, best_n, n_active=5, step_size=step_size)
                c5_sub = compute_c5(h_sub)
                if c5_sub < best_c5 - 1e-12:
                    best_c5 = c5_sub
                    best_h = h_sub.copy()
    
    # Phase 3: Targeted local search
    if remaining() > 30:
        t_budget = min(remaining() * 0.3, 200)
        h_tg, c5_tg = targeted_local_search(best_h, best_n, t_budget)
        if c5_tg < best_c5 - 1e-12:
            best_c5 = c5_tg
            best_h = h_tg.copy()
    
    # Phase 4: Try other resolutions
    other_ns = [120, 140, 180, 200, 240, 256]
    for n_try in other_ns:
        if remaining() < 60:
            break
        if n_try % 2 != 0:
            continue
        
        # Interpolate best to this resolution
        h_interp = make_feasible(interpolate_to_n(best_h, n_try), n_try)
        
        # Penalty optimize
        for beta in [100, 500]:
            if remaining() < 20:
                break
            for mu in [100, 1000]:
                if remaining() < 10:
                    break
                h_opt, c5_opt = penalty_optimize(h_interp, n_try, beta, mu, maxiter=200)
                if h_opt is not None and c5_opt < best_c5 - 1e-12:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_try
                    h_interp = make_feasible(h_opt.copy(), n_try)
    
    # Phase 5: Diverse starts at best_n
    if remaining() > 60:
        inits = diverse_inits(best_n, rng)
        # Also try cross between parent and random
        h_cross = 0.5 * best_h + 0.5 * rng.rand(best_n)
        inits.append(make_feasible(h_cross, best_n))
        
        for h0 in inits[:4]:
            if remaining() < 20:
                break
            h_opt, c5_opt = penalty_optimize(h0, best_n, 200, 500, maxiter=300)
            if h_opt is not None and c5_opt < best_c5 - 1e-12:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    
    # Phase 6: Extended local search
    if remaining() > 30:
        ls_budget = min(remaining() * 0.5, 300)
        h_ls, c5_ls = local_search(best_h, best_n, ls_budget)
        if c5_ls < best_c5 - 1e-12:
            best_c5 = c5_ls
            best_h = h_ls.copy()
    
    # Phase 7: Final fine-tuning
    if remaining() > 15:
        for beta in [500, 2000, 5000]:
            if remaining() < 5:
                break
            h_opt, c5_opt = penalty_optimize(best_h, best_n, beta, 5000, maxiter=100, gtol=1e-12)
            if h_opt is not None and c5_opt < best_c5 - 1e-14:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    
    # Phase 8: Very fine local search
    if remaining() > 10:
        h_fine, c5_fine = local_search(best_h, best_n, remaining() - 5, step_sizes=[0.01, 0.005, 0.002, 0.001])
        if c5_fine < best_c5 - 1e-14:
            best_c5 = c5_fine
            best_h = h_fine.copy()
    
    # Final feasibility check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
