def project_feasible(h, n):
    """Project onto {0 <= h <= 1, sum(h) = n/2}."""
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = h.sum() - target
        if abs(diff) < 1e-15:

        if diff > 0:
            # Need to decrease sum: reduce values that are > 0
            mask = h > 1e-12
            if not mask.any():

            # Proportional reduction

    dx = 2.0 / n
    lags = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = np.max(lags)
    # Active lags: those within tau of max
    active = np.where(lags >= cmax - tau)[0]

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def compute_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def c5_of(h, n):
    return float(np.max(compute_lags(h, n)))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-12
            if not mask.any():
                break
            h[mask] *= (1.0 - diff / h[mask].sum())
        else:
            mask = h < 1.0 - 1e-12
            if not mask.any():
                break
            h[mask] += (-diff) * (1.0 - h[mask]) / (1.0 - h[mask]).sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def sgradient(h, n, tol_frac=1e-4):
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    cmax_raw = np.max(corr_raw)
    if cmax_raw < 1e-14:
        return np.zeros(n)
    tol = tol_frac * cmax_raw
    active = np.where(corr_raw >= cmax_raw - tol)[0]
    g = np.zeros(n)
    for m in active:
        offset = m - (n - 1)
        if offset >= 0:
            g[:n - offset] += (1.0 - h[offset:]) * dx
            g[offset:] -= h[:n - offset] * dx
        else:
            mm = -offset
            g[mm:] += (1.0 - h[:n - mm]) * dx
            g[:n - mm] -= h[mm:] * dx
    g /= len(active)
    return g


def slsqp_fun(x, nn):
    return c5_of(x, nn)


def slsqp_grad(x, nn):
    return sgradient(x, nn)


def sum_eq(x, nn):
    return np.sum(x) - nn / 2.0


def sum_jac(x, nn):
    return np.ones(nn)

    h_arr[j_val] = t
    val = float(np.max(np.correlate(h_arr, 1.0 - h_arr, mode='full')))

def coord_descend(h, n, max_rounds=15, xatol=1e-12):
    improved = True
    for rnd in range(max_rounds):
        if not improved:
            break
        improved = False
        old_c5 = c5_of(h, n)
        for j in range(n):
            orig = h[j]
            res = minimize_scalar(
                _coord_eval, args=(h, j),
                bounds=(0.0, 1.0), method='bounded',
                options={'xatol': xatol}
            )
            h[j] = res.x
            h = project_feasible(h, n)
            if c5_of(h, n) < old_c5 - 1e-13:
                old_c5 = c5_of(h, n)
                improved = True
    return h


def subgradient_descent(h, n, max_steps=5000, tol=1e-14):
    h = h.copy()
    step = 0.5
    best_local = c5_of(h, n)
    best_h_local = h.copy()
    for step_i in range(max_steps):
        if c5_of(h, n) < best_local:
            best_local = c5_of(h, n)
            best_h_local = h.copy()
        g = sgradient(h, n)
        g_norm = np.linalg.norm(g)
        if g_norm < tol:
            break
        for attempt in range(50):
            h_new = h - step * g / g_norm
            h_new = project_feasible(h_new, n)
            c_new = c5_of(h_new, n)
            if c_new < c5_of(h, n):
                h = h_new
                step = min(step * 1.05, 5.0)
                break
            else:
                step *= 0.7
        if step < 1e-8:
            break
    if c5_of(best_h_local, n) < c5_of(h, n):
        h = best_h_local
    return h


def upsample(h, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)


def slsqp_polish(h, n, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac, 'args': (n,)}]
    try:
        res = minimize(
            fun=slsqp_fun, x0=h, jac=slsqp_grad,
            args=(n,), method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h
