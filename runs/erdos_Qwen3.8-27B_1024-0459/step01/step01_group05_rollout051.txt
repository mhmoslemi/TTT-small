        return (time.time() - t_start) < time_limit
    
    result = minimize(func, h0, jac=jac, method='L-BFGS-B',
                      bounds=bounds, callback=time_check,
                      options={'maxiter': max_iter, 'ftol': ftol, 'gtol': 1e-7, 'maxls': 40})
    
    if result is not None and result.x is not None:
        return np.clip(result.x, 0.0, 1.0)
    return h0

def interpolate_h(h_src, n_src, n_dst):
    x_old = np.linspace(0, 2, n_src, endpoint=False)
    x_new = np.linspace(0, 2, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return np.clip(h_interp, 0.0, 1.0)

def make_random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.rand(n)
    threshold = 0.5
    h = (h > threshold).astype(float)
    # Adjust to get exactly n/2 ones (or as close as possible)
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if diff > 0:
        zero_idx = np.where(h == 0)[0]
        rng.shuffle(zero_idx)
        n_to_flip = int(diff)
        h[zero_idx[:n_to_flip]] = 1.0
    elif diff < 0:
        one_idx = np.where(h == 1)[0]
        rng.shuffle(one_idx)
        n_to_flip = int(-diff)
        h[one_idx[:n_to_flip]] = 0.0
    # Smooth it a bit
    h = 0.7 * h + 0.3 * np.clip(np.convolve(h, np.ones(5)/5, mode='same'), 0, 1)
    return h

def make_structured_h(n, seed):
    rng = np.random.RandomState(seed)
    # Create a pattern based on primes (pseudo-random binary)
    h = np.zeros(n)
    p1 = rng.randint(3, max(4, n//4))
    p2 = rng.randint(p1+1, max(p1+2, n//3))
    for i in range(n):
        h[i] = 1.0 if ((i % p1 < p1//2) != (i % p2 < p2//2)) else 0.0
    # Smooth
    kernel = np.ones(3)/3
    h = np.convolve(h, kernel, mode='same')
    h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    
    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)
    
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_parent
    
    def elapsed():
        return time.time() - t0
    
    def try_update(h_cand, n_cand):
        nonlocal best_h, best_c5, best_n
        if len(h_cand) != n_cand:
            return
        h_cand = project_sum(h_cand, n_cand / 2.0)
        c5_val = compute_c5(h_cand)
        if c5_val < best_c5 - 1e-13:
            best_h = h_cand.copy()
            best_c5 = c5_val
            best_n = n_cand
            return True
        return False
    
    # Phase 1: Optimize at parent resolution with progressive beta
    n = n_parent
    target = n / 2.0
    h_cur = project_sum(h_parent.copy(), target)
    
    beta_schedule_1 = [5, 15, 40, 100, 300]
    for beta in beta_schedule_1:
        if elapsed() > budget_s * 0.20:
            break
        remaining = max(30, int((budget_s * 0.20 - elapsed()) * 0.9))
        max_it = min(300, max(50, int(remaining / 5)))
        h_cur = run_lbgfsb(h_cur, n, beta, penalty_c=500.0, max_iter=max_it,
                          ftol=1e-13, time_limit=remaining)
        h_cur = project_sum(h_cur, target)
        try_update(h_cur, n)
    
    # Phase 2: Try multiple resolutions
    resolutions = sorted(set([n_parent, 100, 120, 150, 200]))
    # Remove parent (already done) and keep diverse
    resolutions = [r for r in resolutions if r != n_parent]
    resolutions.sort()
    
    for n_try in resolutions:
        if elapsed() > budget_s * 0.75:
            break
        
        target_try = n_try / 2.0
        # Interpolate from current best
        h_interp = interpolate_h(best_h, best_n, n_try)
        h_interp = project_sum(h_interp, target_try)
        
        per_res_time = min(150, max(40, int((budget_s * 0.75 - elapsed()) / max(1, len(resolutions)))))
        
        # Progressive beta at this resolution
        for beta in [10, 50, 150]:
            if elapsed() > budget_s * 0.75:
                break
            tl = min(per_res_time // 3 - 5, 80)
            if tl < 15:
                break
            max_it = max(50, int(tl / 3))
            h_interp = run_lbgfsb(h_interp, n_try, beta, penalty_c=500.0,
                                  max_iter=max_it, ftol=1e-12, time_limit=tl)
            h_interp = project_sum(h interp = project_sum(h_interp, target_try)
            try_update(h_interp, n_try)
        
        # Try from a structured/random start at this resolution
        if elapsed() < budget_s * 0.70:
            h_struct = make_structured_h(n_try, seed + n_try)
            h_struct = project_sum(h_struct, target_try)
            tl2 = min(60, max(20, int((budget_s * 0.70 - elapsed()) * 0.8)))
            if tl2 > 15:
                h_struct = run_lbgfsb(h_struct, n_try, 30, penalty_c=500.0,
                                      max_iter=150, ftol=1e-11, time_limit=tl2)
                h_struct = project_sum(h_struct, target_try)
                try_update(h_struct, n_try)
    
    # Phase 3: Diversification with perturbations at best resolution
    if elapsed() < budget_s * 0.90:
        n_best = best_n
        tgt_best = n_best / 2.0
        n_perturb = 4
        per_pert_time = max(15, int((budget_s * 0.90 - elapsed()) / n_perturb))
        
        for p in range(n_perturb):
            if elapsed() > budget_s * 0.90:
                break
            scale = 0.03 * (1 - p * 0.15)
            h_pert = best_h + np.random.randn(n_best) * scale
            h_pert = np.clip(h_pert, 0.0, 1.0)
            h_pert = project_sum(h_pert, tgt_best)
            
            tl_p = min(per_pert_time - 5, 70)
            if tl_p < 10:
                break
            h_pert = run_lbgfsb(h_pert, n_best, 50, penalty_c=500.0,
                               max_iter=200, ftol=1e-12, time_limit=tl_p)
            h_pert = project_sum(h_pert, tgt_best)
            try_update(h_pert, n_best)
    
    # Phase 4: Final sharp refinement
    if elapsed() < budget_s * 0.97:
        n_fin = best_n
        tgt_fin = n_fin / 2.0
        for beta_fin in [200, 500, 1000]:
            if elapsed() > budget_s * 0.97:
                break
            tl_fin = max(15, int((budget_s * 0.97 - elapsed()) * 0.6))
            if tl_fin < 10:
                break
            best_h = run_lbgfsb(best_h, n_fin, beta_fin, penalty_c=1000.0,
                               max_iter=200, ftol=1e-14, time_limit=tl_fin)
            best_h = project_sum(best_h, tgt_fin)
            c5_chk = compute_c5(best_h)
            if c5_chk < best_c5:
                best_c5 = c5_chk
    
    # Final projection and verification
    best_h = project_sum(best_h, best_n / 2.0)
    # Ensure bounds are exact
    best_h = np.clip(best_h, 0.0, 1.0)
    # Fix sum exactly
    s = np.sum(best_h)
    diff = best_n / 2.0 - s
    if abs(diff) > 1e-15:
        free = (best_h > 1e-12) & (best_h < 1.0 - 1e-12)
        if np.any(free):
            best_h[free] += diff / np.sum(free)
            best_h = np.clip(best_h, 0.0, 1.0)
    
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, best_n)
