```python
import time
import numpy as np
from scipy.optimize import minimize

_nc = 0
_alpha = 10.0
_target = 0.0


def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_density(x, target):
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    s = float(np.sum(x))
    if abs(s - target) < 1e-13 * max(1, n):
        return x
    lo, hi = -1.0, 1.0
    for _ in range(65):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-12:
        free = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if free.size > 0:
            h[free] += res / float(free.size)
            h = np.clip(h, 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-12:
        if res > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + res, 0.0, 1.0))
    return h


def set_ctx(n, alpha, target):
    global _nc, _alpha, _target
    _nc = n
    _alpha = alpha
    _target = target


def smooth_obj(x):
    n = _nc
    alpha = _alpha
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return 1e100
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return 1e100
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return 1e100
    return (2.0 / n) * (m + np.log(z) / alpha)


def smooth_jac(x):
    n = _nc
    alpha = _alpha
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return np.zeros(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return np.zeros(n)
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return np.zeros(n)
    w = e / z
    g = np.zeros(n)
    center = n - 1
    sig = w > 1e-10
    if not np.any(sig):
        sig[int(np.argmax(w))] = True
    for t in np.where(sig)[0]:
        t = int(t)
        p = float(w[t])
        k = t - center
        if k == 0:
            g += p * (1.0 - 2.0 * h)
        else:
            lag = abs(k)
            if k > 0:
                g[lag:] += p
                g[:n - lag] -= p * h[lag:]
                g[lag:] -= p * h[:n - lag]
            else:
                g[:n - lag] += p
                g[:n - lag] -= p * h[lag:]
                g[lag:] -= p * h[:n - lag]
    return (2.0 / n) * g


def sum_fun(x):
    return float(np.sum(x) - _target)


def sum_jac(x):
    return np.ones_like(x)


def upsample(h, n):
    p = np.asarray(h, dtype=float)
    n0 = p.size
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return np.full(n, 0.5)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    h_new = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h_new, 0.5 * n)


def slsqp_optimize(h0, n, deadline, alphas=None, maxiter=None):
    if alphas is None:
        if n <= 120:
            alphas = (8.0, 20.0, 50.0, 120.0, 300.0)
        elif n <= 300:
            alphas = (10.0, 30.0, 80.0, 200.0)
        else:
            alphas = (10.0, 30.0, 80.0)
    if maxiter is None:
        if n <= 120:
            maxiter = 200
        elif n <= 300:
            maxiter = 120
        else:
            maxiter = 80

    target = 0.5 * n
    h = project_density(h0, target)
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)

    bounds = [(0.0, 1.0)] * n
    constraints = ({"type": "eq", "fun": sum_fun, "jac": sum_jac},)

    for alpha in alphas:
        if time.perf_counter() >= deadline:
            break
        set_ctx(n, alpha, target)
        try:
            res = minimize(
                smooth_obj, h, method="SLSQP", jac=smooth_jac,
                bounds=bounds, constraints=constraints,
                options={"maxiter": maxiter, "ftol": 1e-10, "disp": False}
            )
            xg = getattr(res, "x", None)
            if xg is not None:
                xg = np.asarray(xg, dtype=float)
                if np.all(np.isfinite(xg)):
                    cand = project_density(xg, target)
                    c = evaluate_erdos(cand)
                    if c < best_c:
                        best_h = cand
                        best_c = c
                    if c <= best_c + 0.01:
                        h = cand
                    else:
                        h = best_h
        except Exception:
            h = best_h

    return best_h, best_c


def full_pair_search(h, deadline, max_sweeps=200, verbose=False):
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    rng = np.random.default_rng(12345)

    sweep = 0
    no_improve = 0
    while time.perf_counter() < deadline and sweep < max_sweeps:
        sweep += 1
        improved = False
        order = list(range(n))
        rng.shuffle(order)

        for i in order:
            if time.perf_counter() >= deadline:
                return best, best_c
            hi_val = best[i]
            for j in order:
                if i == j:
                    continue
                hj_val = best[j]
                # Direction 1: h[i] += d, h[j] -= d
                max_d1 = min(1.0 - hi_val, hj_val)
                if max_d1 > 1e-12:
                    for frac in (0.5, 1.0):
                        d = frac * max_d1
                        cand = best.copy()
                        cand[i] = hi_val + d
                        cand[j] = hj_val - d
                        c = evaluate_erdos(cand)
                        if c < best_c - 1e-12:
                            best = cand
                            best_c = c
                            improved = True
                            hi_val = cand[i]
                            hj_val = cand[j]
                            break
                # Direction 2: h[j] += d, h[i] -= d
                max_d2 = min(1.0 - hj_val, hi_val)
                if max_d2 > 1e-12:
                    for frac in (0.5, 1.0):
                        d = frac * max_d2
                        cand = best.copy()
                        cand[j] = hj_val + d
                        cand[i] = hi_val - d
                        c = evaluate_erdos(cand)
                        if c < best_c - 1e-12:
                            best = cand
                            best_c = c
                            improved = True
                            hi_val = cand[i]
                            hj_val = cand[j]
                            break

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

    return best, best_c


def gradient_polish(h, deadline, max_iter=1000, top_k=8):
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    center = n - 1
    stagnation = 0

    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break

        corr = np.correlate(best, 1.0 - best, mode="full")
        c_max = float(np.max(corr))
        threshold = c_max - 0.002 * n * 0.5
        active = np.where(corr >= threshold)[0]
        if active.size == 0:
            active = np.argsort(corr)[-top_k:]
        active = active[:top_k]

        g = np.zeros(n)
        for t in active:
            t = int(t)
            k = t - center
            if k == 0:
                g += 1.0 - 2.0 * best
            else:
                lag = abs(k)
                if k > 0:
                    g[lag:] += 1.0
                    g[:n - lag] -= best[lag:]
                    g[lag:] -= best[:n - lag]
                else:
                    g[:n - lag] += 1.0
                    g[:n - lag] -= best[lag:]
                    g[lag:] -= best[:n - lag]

        d = -g
        d -= float(np.mean(d))
        norm = float(np.linalg.norm(d))
        if norm < 1e-14:
            break
        d /= norm

        improved = False
        for step in (0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.3):
            cand = project_density(best + step * d, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-12:
                best = cand
                best_c = c
                improved = True
                break

        if improved:
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 20:
                break

    return best, best_c


def multi_direction_search(h, deadline, max_iter=500):

    n = h.size

        if active.size == 0:
            active = np.argsort(corr)[-6:]

        for t in active:
            t = int(t)
            k = t - center
            if k == 0:
                g += 1.0 - 2.0 * best
            else:
                lag = abs(k)
                if k > 0:
                    g[lag:] += 1.0
                    g[:n - lag] -= best[lag:]
                    g[lag:] -= best[:n - lag]
                else:

                    g[lag:] -= best[:n - lag]

        d = -g
        d -= float(np.mean(d))
        norm = float(np.linalg.norm(d))
        if norm < 1e-14:
            break
        d /= norm

        improved = False
        # Also try perturbed directions
        dirs = [d]
        for _ in range(3):

                    break
```
