2. Adjust to fix the sum (distribute the error among non-saturated coordinates)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_overlaps(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h

def project_onto_feasible(h, n):
    """Project h onto {0≤h≤1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    h = fix_sum(h, n)
    return h

def compute_subgradient(h, active_k_indices, n):
    """
    Compute the (average) subgradient at the active shifts.
    active_k_indices: indices in the correlate output (0 to 2n-2)
    """
    dx = 2.0 / n
    g = np.zeros(n)
    
    for ci in active_k_indices:
        # ci is the index in the full correlation output
        # The shift k = ci - (n-1)
        k = ci - (n - 1)
        
        # C(k) = sum_{valid i} h[i] * (1-h[i+k]) * dx
        # ∂C(k)/∂h[i]:
        #   From h[i]*(1-h[i+k]): (1-h[i+k])*dx if i+k is valid
        #   From h[i-k]*(1-h[i]): -h[i-k]*dx if i-k is valid (i.e., the term where h[i] is the (1-h[...]) part)
        
        if k >= 0:
            # i+k valid: i from 0 to n-1-k
            for i in range(n - k):
                g[i] += (1.0 - h[i+k]) * dx
            # i-k valid: i from k to n-1
            for i in range(k, n):
                g[i] += -h[i-k] * dx
        else:
            # i+k valid: i from -k to n-1
            for i in range(-k, n):
                g[i] += (1.0 - h[i+k]) * dx
            # i-k valid: i from 0 to n-1+k
            for i in range(0, n + k):
                g[i] += -h[i-k] * dx
    
    g /= len(active_k_indices)
    return g
