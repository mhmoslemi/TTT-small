    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    
    # Smooth max weights
    max_corr = np.max(corr)
    shifted = beta * (corr - max_corr)
    shifted = np.clip(shifted, -60, 60)
    weights = np.exp(shifted)
    weights = weights / np.sum(weights)
    
    # Gradient
    grad = np.zeros(n)
    # For each k in corr (lag k), m = k - (n-1)
    # ∂corr[k]/∂h[j] = (1-h[j+m])*1_{0≤j+m<n} - h[j-m]*1_{0≤j-m<n}
    
    for k in range(len(corr)):
        m = k - (n - 1)
        w = weights[k]
        if w < 1e-15:
            continue
        # Contribution to grad[j] from this k:
        # (1-h[j+m]) for j+m in [0,n-1]
        # -h[j-m] for j-m in [0,n-1]
        if m >= 0:

def compute_grad_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    
    # Smooth max weights
    max_corr = np.max(corr)
    shifted = beta * (corr - max_corr)
    shifted = np.clip(shifted, -60, 60)
    weights = np.exp(shifted)
    weights = weights / np.sum(weights)
    
    # Gradient: ∂f/∂h[j] = dx * sum_k p_k * [(1-h[j-k])*1_{0≤j-k<n} - h[j+k]*1_{0≤j+k<n}]
    # where corr[k+(n-1)] corresponds to lag k
    
    grad = np.zeros(n)
    
    for k_idx in range(2*n-1):
        k = k_idx - (n - 1)  # actual lag
        w = weights[k_idx]
        if w < 1e-15:
            continue
        # (1-h[j-k]) for j-k in [0, n-1], i.e., j in [k, n-1+k] intersect [0, n-1]
        # -h[j+k] for j+k in [0, n-1], i.e., j in [-k, n-1-k] intersect [0, n-1]
        
        if k >= 0:
            # j-k in [0, n-1] => j in [k, n-1+k], intersect [0,n-1] => j in [k, n-1]
            # (1-h[j-k]) for j=k..n-1: h[j-k] = h[0..n-1-k]
            grad[k:n] += w * (1.0 - h[0:n-k])
            # j+k in [0, n-1] => j in [-k, n-1-k], intersect [0,n-1] => j in [0, n-1-k]
            # -h[j+k] for j=0..n-1-k: h[j+k] = h[k..n-1]

    # Fix any floating point error

    if abs(diff) > 1e-14:
        if diff > 0:
            mask = proj < 1.0
        else:
            mask = proj > 0.0
        if np.sum(mask) > 0:
            proj[mask] += diff / np.sum(mask)
            proj = np.clip(proj, 0, 1)
    return proj

def optimize_pg(h0, target, beta, max_iter, lr_init):
    """Projected gradient descent with backtracking line search."""
    n = len(h0)
    h = project_box_sum(h0, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    
    for it in range(max_iter):
        grad = compute_grad_c5(h, beta)
        
        # Backtracking line search
        alpha = lr
        f_cur = compute_c5(h)
        for ls in range(30):
            h_new = project_box_sum(h - alpha * grad, target)
            f_new = compute_c5(h_new)
            if f_new < f_cur - 1e-14:
                break
            alpha *= 0.5
        
        if alpha < 1e-10:
            break
            
        h = h_new
        if f_new < best_c5:
            best_c5 = f_new
            best_h = h.copy()
        else:
            h = best_h.copy()
            break
        
        # Adaptive learning rate
        if f_new < f_cur * 0.999:
            lr = min(lr * 1.1, 0.5)
        else:
            lr = lr * 0.8
    
    return best_h, best_c5

def coordinate_descent(h, target, max_sweeps=10):
    """Coordinate descent: for each j, find optimal h[j]."""
    n = len(h)
    h = h.copy()
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            # Try a small range of values for h[j]
            best_val = h[j]
            best_c5 = compute_c5(h)
            
            # Try perturbation
            for delta in [-0.1, -0.05, -0.01, 0.01, 0.05, 0.1]:
                h_test = h.copy()
                h_test[j] = np.clip(h[j] + delta, 0, 1)
                # Re-project to maintain sum
                h_test = project_box_sum(h_test, target)
                c5_test = compute_c5(h_test)
                if c5_test < best_c5 - 1e-12:
                    best_c5 = c5_test
                    best_val = h_test[j]
            
            if best_val != h[j]:
                h[j] = best_val
                h = project_box_sum(h, target)
                improved = True
        
        if not improved:
            break
    
    return h

Actually, for the coordinate descent, I can be smarter. For a given j, the optimal h[j] (ignoring the sum constraint) can be found analytically since the objective is the max of linear functions of h[j].

Let me precompute for each j:
- A_k(j) and B_k(j) such that corr[k] = A_k(j) + B_k(j) * h[j]

Then the optimal h[j] minimizes max_k (A_k(j) + B_k(j) * h[j]).

This is a 1D minimax problem. The optimal value is the smallest v such that there exists h[j] in [0,1] with A_k(j) + B_k(j) * h[j] ≤ v for all k.

This can be solved by bisection on v. For a given v, we need:
- For k with B_k(j) > 0: h[j] ≤ (v - A_k(j)) / B_k(j)
- For k with B_k(j) < 0: h[j] ≥ (v - A_k(j)) / B_k(j)
- For k with B_k(j) = 0: need A_k(j) ≤ v

So the feasible range for h[j] is:
lower = max(0, max_{B_k<0} (v - A_k(j)) / B_k(j))
upper = min(1, min_{B_k>0} (v - A_k(j)) / B_k(j))

And we need lower ≤ upper.

This is O(n * 2n) = O(n²) per coordinate per bisection step. With log(1/eps) bisection steps, total is O(n² * log(1/eps)) per coordinate, and O(n³ * log(1/eps)) for a full sweep. For n=200, that's about 8M * 30 = 240M operations per sweep. That's too slow for Python.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def compute_grad_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    max_corr = np.max(corr)
    shifted = beta * (corr - max_corr)
    shifted = np.clip(shifted, -60, 60)
    weights = np.exp(shifted)
    weights = weights / np.sum(weights)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        w = weights[k_idx]
        if w < 1e-15:
            continue
        if k >= 0:
            grad[k:n] += w * (1.0 - h[0:n - k])
            grad[0:n - k] -= w * h[k:n]
        else:
            kk = -k
            grad[0:n - kk] += w * (1.0 - h[kk:n])
            grad[kk:n] -= w * h[0:n - kk]
    return grad * dx

def project_box_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = proj < 1.0 - 1e-12
        else:
            mask = proj > 1e-12
        cnt = np.sum(mask)
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def optimize_nesterov(h0, target, beta, max_iter, lr_init, momentum):
    n = len(h0)
    h = project_box_sum(h0, target)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    no_improve = 0
    
    for it in range(max_iter):
        y = project_box_sum(h + momentum * v, target)
        grad = compute_grad_c5(y, beta)
        v = momentum * v - lr * grad
        
        alpha = 1.0
        f_cur = compute_c5(h)
        h_new = None
        for _ in range(40):
            h_try = project_box_sum(h + alpha * v, target)
            f_new = compute_c5(h_try)
            if f_new < f_cur + 1e-14:
                h_new = h_try
                break
            alpha *= 0.5
        if h_new is None:
            break
        h = h_new
        if f_new < best_c5 - 1e-13:
            best_c5 = f_new
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break
        if lr < 1e-8:
            break
    return best_h, best_c5

def binary_round(h, target):
    """Round h to nearest binary-like (values in {0,1} plus one fractional if needed)."""
    n = len(h)
    h_sort_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    running_sum = 0.0
    for i in range(n):
        idx = h_sort_idx[i]
        if running_sum + 1.0 <= target + 1e-10:
            h_bin[idx] = 1.0
            running_sum += 1.0
        else:
            frac = target - running_sum
            if 0 <= frac <= 1:
                h_bin[idx] = frac
            running_sum = target
            break
    return h_bin

def two_flip_local_search(h_bin, target, max_sweeps=50):
    """2-flip local search on binary-like solution."""
    n = len(h)
    h = h_bin.copy()
    c5_cur = compute_c5(h)

                b = zeros_list[j]
                if abs(h[a] - h[b]) < 1e-9:
                    continue
                h_test = h.copy()
                h_test[a] -= 1.0
                h_test[b] += 1.0
                h_test = project_box_sum(h_test, target)
                c5_test = compute_c5(h_test)
                if c5_test < c5_cur - 1e-12:
                    h = h_test
                    c5_cur = c5_test
                    improved = True

            break
