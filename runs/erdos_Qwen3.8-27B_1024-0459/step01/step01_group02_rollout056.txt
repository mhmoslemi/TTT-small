def _obj_smooth(x):
    n = _g_n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    sh = _g_beta * (c - m)
    sh = np.clip(sh, -500.0, 0.0)
    smax = m + np.log(np.sum(np.exp(sh))) / _g_beta
    return smax * (2.0 / n)
