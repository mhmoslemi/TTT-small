    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)

    grad = np.zeros(n)
    if k_star < n:

        #          from (1-h[i]) term: -h[i+s] where i=j, valid if 0<=j<=n-1-s
        for j in range(n):

                grad[j] += (1 - h[j - s])
            if j <= n - 1 - s:
                grad[j] += (-h[j + s])

                grad[j] += (1 - h[j + s])
            if j >= s and j - s <= n - 1 - s:
                grad[j] += (-h[j - s])
    
    # Normalize gradient
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-10:
        return h
    
    h_new = h - lr * grad / gnorm
    return project(h_new, n)

    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):

        if abs(diff) < 1e-14:
            break

        n_free = np.sum(free)
        if n_free == 0:
            free = np.ones(n, dtype=bool)
            n_free = n
        h[free] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value(h, beta, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    # numerically stable log-sum-exp
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = m + np.log(np.sum(np.exp(shifted))) / beta
    return val * dx

        k_star = int(np.argmax(corr))

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.sum(free)
        if nf == 0:
            free = np.ones(n, dtype=bool)
            nf = n
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, beta, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 0.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def raw_max_obj(x, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_gradient_at_max(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    grad = np.zeros(n)
    d = k_star - (n - 1)
    lo1 = max(0, d)
    hi1 = min(n, n + d)
    if hi1 > lo1:
        grad[lo1:hi1] += (1.0 - h[lo1 - d:hi1 - d])
    lo2 = max(0, -d)
    hi2 = min(n, n - d)
    if hi2 > lo2:
        grad[lo2:hi2] += (-h[lo2 + d:hi2 + d])
    return corr, k_star, grad

def subgradient_descent(h_init, n, iters, lr_start, lr_end):
    h = h_init.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    stall = 0
    for it in range(iters):
        t = it / max(1, iters - 1)
        lr = lr_start * (lr_end / lr_start) ** t
        _, _, grad = compute_gradient_at_max(h, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        h_new = h - lr * grad / gnorm
        h_new = project_feasible(h_new, n)
        val = compute_c5(h_new)
        if val < best_val - 1e-10:
            best_val = val
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        h = h_new
        if stall > 100:
            break
    return best_h, best_val

def slsqp_polish(h0, n, beta, maxiter):
    def obj(x):
        return smooth_max_obj(x, beta, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        hp = project_feasible(res.x, n)
        return hp
    except Exception:
        return h0.copy()
