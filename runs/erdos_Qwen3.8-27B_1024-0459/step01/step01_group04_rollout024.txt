                grad[d:] += c * (1.0 - h[:m2])
    grad *= (2.0 / n)
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n)
    return grad

def eq_constraint(x):
    n = x.size
    return (float(np.sum(x)) - 0.5 * n) / n

def eq_jacobian(x):
    n = x.size
    return np.full((1, n), 1.0 / n)

def upsample_linear(parent, n):
    n0 = parent.size
    if n == n0:
        return parent.copy()
    if n0 <= 0:
        return np.full(n, 0.5)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x, x0, parent)

def binary_best_swap_search(h_init, n, deadline, max_iters=5000):
    """Exhaustive best-swap binary local search."""
    target = n // 2
    h = np.zeros(n, dtype=np.int8)
    vals = np.asarray(h_init, dtype=float)
    order = np.argsort(vals)
    h[order[target:]] = 1
    
    def compute_scores_bin(hb):
        hf = hb.astype(float)
        corr = np.correlate(hf, 1.0 - hf, mode="full")
        return corr
    
    scores = compute_scores_bin(h)
    cur = float(np.max(scores))
    best_cur = cur
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        best_delta = 0.0
        best_pair = None
        # For moderate n, try all pairs
        if len(zeros) * len(ones) <= 20000:
            for i in zeros:
                for j in ones:
                    # Simulate swap i->1, j->0
                    h[i] = 1
                    h[j] = 0
                    new_scores = compute_scores_bin(h)
                    new_max = float(np.max(new_scores))
                    delta = new_max - cur
                    h[i] = 0
                    h[j] = 1
                    if delta < best_delta - 1e-12:
                        best_delta = delta
                        best_pair = (i, j)
        else:
            # Sampled search for large n
            n_samples = 5000
            for _ in range(n_samples):
                i = zeros[rng_global % len(zeros)]
                j = ones[(rng_global + 7) % len(ones)]
                rng_global += 1
                h[i] = 1
                h[j] = 0
                new_scores = compute_scores_bin(h)
                new_max = float(np.max(new_scores))
                delta = new_max - cur
                h[i] = 0
                h[j] = 1
                if delta < best_delta - 1e-12:
                    best_delta = delta
                    best_pair = (i, j)
        
        if best_pair is None:
            break
        
        i, j = best_pair
        h[i] = 1
        h[j] = 0
        scores = compute_scores_bin(h)
        cur = float(np.max(scores))
        
        if cur < best_cur - 1e-12:
            best_cur = cur
            best_h = h.copy()
    
    return best_h.astype(float), best_cur * (2.0 / n)

rng_global = 0

def subgradient_polish(h, n, deadline, max_iters=2000):
    """Subgradient descent with Polyak step for the max objective."""
    global _RUN_DEADLINE
    h = project_feasible(h, n).copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    
    # Compute initial gap for Polyak step
    best_c_gap = best_c  # we want to push below this
    
    for it in range(max_iters):
        _check_dl()
        if time.time() > deadline:
            break
        scores = get_scores(h, n)
        m = float(np.max(scores))
        c_cur = m * (2.0 / n)
        
        if c_cur < best_c - 1e-12:
            best_c = c_cur
            best_h = h.copy()
        
        # Subgradient
        tol = max(1e-6, 1e-5 * m)
        active = scores >= m - tol
        if not active.any():
            active[np.argmax(scores)] = True
        
        grad = np.zeros(n)
        for k in np.where(active)[0]:
            c = 1.0 / active.sum()
            if k == 0:
                grad += c * (1.0 - 2.0 * h)
            else:
                if k % 2 == 1:
                    d = (k + 1) // 2
                else:
                    d = k // 2
                m2 = n - d
                if m2 <= 0:
                    continue
                if k % 2 == 1:
                    grad[:m2] += c * (1.0 - h[d:])
                    grad[d:] += c * (-h[:m2])
                else:
                    grad[:m2] += c * (-h[d:])
                    grad[d:] += c * (1.0 - h[:m2])
        
        g_norm = float(np.linalg.norm(grad))
        if g_norm < 1e-12:
            break
        
        # Polyak step
        gap = c_cur - best_c * 0.999
        if gap > 1e-10:
            step = gap / (g_norm * (2.0 / n) + 1e-12)
        else:
            step = 0.5 / (g_norm + 1e-12)
        step = min(step, 2.0)
        
        h_new = project_feasible(h - step * grad, n)
        
        # Check improvement
        c_new = compute_c5(h_new)
        if c_new <= c_cur + 1e-10:
            h = h_new
        else:
            # Line search
            h_new = project_feasible(h - 0.5 * step * grad, n)
            c_new = compute_c5(h_new)
            if c_new <= c_cur + 1e-10:
                h = h_new
    
    return best_h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    global _RUN_DEADLINE, rng_global
    rng_global = 0
    
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0:
        budget_s = 1300.0
    reserve = 25.0
    deadline = t0 + budget_s - reserve
    _RUN_DEADLINE = deadline
    
    rng = np.random.default_rng(seed)
    
    # Get parent
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        if parent.ndim != 1 or parent.size < 4:
            raise ValueError
    except Exception:
        parent = np.full(70, 0.5)
        rng.shuffle(parent)
        parent = (parent < 0.5).astype(float)
        parent = project_feasible(parent, parent.size)
    
    n0 = int(parent.size)
    h0 = project_feasible(parent, n0)
    c0 = compute_c5(h0)
    
    best_h = h0.copy()
    best_c = c0
    best_n = n0
    
    def update_best(h, c, n):
        nonlocal best_h, best_c, best_n
        if c < best_c - 1e-14:
            best_h = h.copy()
            best_c = c
            best_n = n
    
    # === Phase 1: Binary search at n0 ===
    if time.time() < deadline - 200:
        bin_deadline = min(t0 + 30, deadline - 200)
        try:
            bh, bc = binary_best_swap_search(h0, n0, bin_deadline, max_iters=3000)
            update_best(bh, bc, n0)
        except Exception:
            pass
    
    # === Phase 2: SLSQP at n0 with progressive alpha ===
    if time.time() < deadline - 80:
        alpha_schedule = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005]
        h_cur = best_h.copy() if best_n == n0 else project_feasible(upsample_linear(best_h, n0), n0)
        
        total_slsqp_time = min(450.0, max(60.0, deadline - time.time() - 120))
        n_stages = len(alpha_schedule)
        time_per_stage = total_slsqp_time / n_stages
        # Weight later stages more
        weights = [0.5, 0.7, 1.0, 1.2, 1.5, 2.0, 2.5]
        total_w = sum(weights)
        
        for si, alpha in enumerate(alpha_schedule):
            if time.time() >= deadline - 100:
                break
            stage_time = time_per_stage * weights[si] / total_w
            maxiter = int(80 + 120 * weights[si] / total_w * 2)
            maxiter = max(30, min(250, maxiter))
            
            stage_dl = time.time() + stage_time
            if stage_dl > deadline - 100:
                stage_dl = deadline - 100
            old_dl = _RUN_DEADLINE
            _RUN_DEADLINE = stage_dl
            try:
                bounds = [(0.0, 1.0)] * n0
                constraints = [{"type": "eq", "fun": eq_constraint, "jac": eq_jacobian}]
                res = minimize(
                    softmax_obj, h_cur, args=(alpha,),
                    jac=softmax_grad,
                    method="SLSQP",
                    bounds=bounds,
                    constraints=constraints,
                    options={"maxiter": maxiter, "ftol": 1e-9, "disp": 0}
                )
                if res is not None and getattr(res, "x", None) is not None:
                    h_new = project_feasible(res.x, n0)
                    c_new = compute_c5(h_new)
                    update_best(h_new, c_new, n0)
                    if c_new <= compute_c5(h_cur) + 1e-10:
                        h_cur = h_new
            except Exception:
                pass
            finally:
                _RUN_DEADLINE = old_dl
        
        # Final SLSQP with alpha=0 (exact max)
        if time.time() < deadline - 60:
            for restart in range(3):
                if time.time() >= deadline - 60:
                    break
                alpha = 0.0
                maxiter = 200
                stage_dl = min(time.time() + 80, deadline - 60)
                old_dl = _RUN_DEADLINE
                _RUN_DEADLINE = stage_dl
                try:
                    bounds = [(0.0, 1.0)] * n0
                    constraints = [{"type": "eq", "fun": eq_constraint, "jac": eq_jacobian}]
                    start_h = h_cur.copy()
                    if restart > 0:
                        noise = rng.normal(0, 0.02, n0)
                        start_h = project_feasible(start_h + noise, n0)
                    res = minimize(
                        softmax_obj, start_h, args=(alpha,),
                        jac=softmax_grad,
                        method="SLSQP",
                        bounds=bounds,
                        constraints=constraints,
                        options={"maxiter": maxiter, "ftol": 1e-10, "disp": 0}
                    )
                    if res is not None and getattr(res, "x", None) is not None:
                        h_new = project_feasible(res.x, n0)
                        c_new = compute_c5(h_new)
                        update_best(h_new, c_new, n0)
                        if c_new < compute_c5(h_cur):
                            h_cur = h_new
                except Exception:
                    pass
                finally:
                    _RUN_DEADLINE = old_dl
    
    # === Phase 3: Subgradient polish at n0 ===
    if time.time() < deadline - 60:
        try:
            sg_dl = min(time.time() + 60, deadline - 60)
            h_sg, c_sg = subgradient_polish(best_h, n0, sg_dl, max_iters=1500)
            update_best(h_sg, c_sg, n0)
        except Exception:
            pass
    
    # === Phase 4: Try higher resolution 2*n0 ===
    n2 = 2 * n0
    if time.time() < deadline - 80 and n2 <= 300:
        h2 = project_feasible(upsample_linear(best_h, n2), n2)
        c2 = compute_c5(h2)
        
        # Quick SLSQP at n2
        alpha2_schedule = [0.2, 0.1, 0.05, 0.02, 0.0]
        total_time2 = min(200.0, max(40.0, deadline - time.time() - 40))
        
        for alpha in alpha2_schedule:
            if time.time() >= deadline - 40:
                break
            maxiter = 100 if alpha > 0 else 150
            stage_dl = min(time.time() + total_time2 / len(alpha2_schedule), deadline - 40)
            old_dl = _RUN_DEADLINE
            _RUN_DEADLINE = stage_dl
            try:
                bounds = [(0.0, 1.0)] * n2
                constraints = [{"type": "eq", "fun": eq_constraint, "jac": eq_jacobian}]
                res = minimize(
                    softmax_obj, h2, args=(alpha,),
                    jac=softmax_grad,
                    method="SLSQP",
                    bounds=bounds,
                    constraints=constraints,
                    options={"maxiter": maxiter, "ftol": 1e-9, "disp": 0}
                )
                if res is not None and getattr(res, "x", None) is not None:
                    h_new = project_feasible(res.x, n2)
                    c_new = compute_c5(h_new)
                    update_best(h_new, c_new, n2)
                    if c_new < c2:
                        h2 = h_new
                        c2 = c_new
            except Exception:
                pass
            finally:
                _RUN_DEADLINE = old_dl
    
    # === Phase 5: If time remains, do more polish at best resolution ===
    if time.time() < deadline - 30:
        best_h_local = best_h.copy()
        best_n_local = best_n
        remaining = deadline - time.time() - 25
        
        # Subgradient at best resolution
        if remaining > 30:
            try:
                sg_dl = time.time() + min(80, remaining * 0.5)
                h_sg, c_sg = subgradient_polish(best_h_local, best_n_local, sg_dl, max_iters=int(remaining * 2))
                update_best(h_sg, c_sg, best_n_local)
            except Exception:
                pass
        
        # One more SLSQP burst at best resolution
        if time.time() < deadline - 30:
            remaining2 = deadline - time.time() - 25
            if remaining2 > 40:
                try:
                    alpha_f = 0.0
                    maxiter = int(min(200, remaining2 * 0.5))
                    stage_dl = time.time() + min(remaining2 * 0.6, 120)
                    old_dl = _RUN_DEADLINE
                    _RUN_DEADLINE = stage_dl
                    bounds = [(0.0, 1.0)] * best_n
                    constraints = [{"type": "eq", "fun": eq_constraint, "jac": eq_jacobian}]
                    res = minimize(
                        softmax_obj, best_h, args=(alpha_f,),
                        jac=softmax_grad,
                        method="SLSQP",
                        bounds=bounds,
                        constraints=constraints,
                        options={"maxiter": maxiter, "ftol": 1e-10, "disp": 0}
                    )
                    if res is not None and getattr(res, "x", None) is not None:
                        h_new = project_feasible(res.x, best_n)
                        c_new = compute_c5(h_new)
                        update_best(h_new, c_new, best_n)
                except Exception:
                    pass
                finally:
                    _RUN_DEADLINE = old_dl
    
    # Final verification
    _RUN_DEADLINE = None
    h_final = project_feasible(best_h, best_n)
    c_final = compute_c5(h_final)
    
    # Ensure parent is not worse
    if c0 < c_final + 1e-12:
        h_final = h0
        c_final = c0
        best_n = n0
    
    return h_final, float(c_final), int(best_n)
