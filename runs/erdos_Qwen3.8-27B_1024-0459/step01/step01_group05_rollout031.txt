                h[free] = np.clip(h[free] + per, 0.0, 1.0)
        else:
            free = np.where(h > 1e-15)[0]
            if len(free) > 0:
                per = diff / len(free)
                h[free] = np.clip(h[free] + per, 0.0, 1.0)
    return h

def obj_and_grad(h, beta):
    n = len(h)
    hc = np.clip(h, 0.0, 1.0)
    corr = np.correlate(hc, 1.0 - hc, mode='full')
    obj = smooth_max_val(corr, beta)
    
    p = get_softmax_weights(corr, beta)
    comp = 1.0 - hc
    
    # Compute gradient analytically
    # grad[i] = sum_{s: 0<=i+s<n} p[s+n-1] * comp[i+s] - sum_{s: 0<=i-s<n} p[s+n-1] * hc[i-s]
    
    grad = np.zeros(n)
    
    # Vectorize the computation
    # For first sum: for each i, sum p[n-1-i : n-1+n-1-i] * comp[0 : n-i]
    #   Actually: s ranges from -i to n-1-i, p index = s+n-1, comp index = i+s = 0 to n-1
    #   p indices: n-1-i to 2n-2-i... no, n-1+i+s_low to n-1+i+s_high
    #   s_low = -i → p index = -i + n - 1
    #   s_high = n-1-i → p index = n-1-i + n-1 = 2n-2-i
    #   comp indices: 0 to n-1-i+... wait, i+s for s=-i to n-1-i gives 0 to n-1-2i? No.
    #   i + (-i) = 0
    #   i + (n-1-i) = n-1
    #   So comp[0:n-1+1] = comp[0:n], but only the first (n-i) elements... 
    #   No wait: the number of s values is (n-1-i) - (-i) + 1 = n
    #   comp indices: 0, 1, 2, ..., n-1 (all of them!)
    #   p indices: n-1-i, n-i, ..., 2n-2-i
    #   Hmm, but p has length 2n-1 (indices 0 to 2n-2)
    #   2n-2-i ≤ 2n-2 ✓
    #   n-1-i ≥ 0 (since i ≤ n-1) ✓
    
    # So first[i] = dot(p[n-1-i : 2n-1-i], comp)
    # But p[n-1-i : 2n-1-i] has length (2n-1-i) - (n-1-i) = n ✓
    # And comp has length n ✓
    
    # For second sum: s ranges from i-n+1 to i
    #   p indices: i-n+1+n-1 to i+n-1 = i to 2n-2
    #   Wait: s+n-1 for s=i-n+1 to i gives (i-n+1+n-1) to (i+n-1) = i to i+n-1
    #   But p only has indices 0 to 2n-2. So we need i+n-1 ≤ 2n-2, i.e., i ≤ n-1 ✓
    #   hc indices: i-s for s=i-n+1 to i gives i-(i-n+1) to i-i = n-1 to 0
    #   So second[i] = dot(p[i : i+n], hc[n-1::-1])... 
    #   Wait: p indices i to i+n-1 (length n), hc indices n-1, n-2, ..., 0 (reversed)
    
    for i in range(n):
        first_slice = p[n-1-i : 2n-1-i]  # length n
        second_slice = p[i : i+n]  # length n
        grad[i] = np.dot(first_slice, comp) - np.dot(second_slice, hc[::-1])
    
    return obj, grad

Let me also consider: the problem has a nice structure. The overlap at shift k is:
∫ h(x)(1-h(x+k)) dx = ∫ h(x) dx - ∫ h(x)h(x+k) dx = 1 - ∫ h(x)h(x+k) dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def get_softmax_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(corr), 1.0 / len(corr))
    return e / s

def project_sum(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-14:
        if diff > 0:
            free = np.where(h < 1.0 - 1e-14)[0]
            if len(free) > 0:
                per = diff / len(free)
                h[free] = np.clip(h[free] + per, 0.0, 1.0)
        else:
            free = np.where(h > 1e-14)[0]
            if len(free) > 0:
                per = diff / len(free)
                h[free] = np.clip(h[free] + per, 0.0, 1.0)
    return h

def obj_and_grad(h, beta):
    n = len(h)
    hc = np.clip(h, 0.0, 1.0)
    corr = np.correlate(hc, 1.0 - hc, mode='full')
    obj = smooth_max_val(corr, beta)
    p = get_softmax_weights(corr, beta)
    comp = 1.0 - hc
    hc_rev = hc[::-1]
    grad = np.zeros(n)
    for i in range(n):
        first_slice = p[n - 1 - i : 2 * n - 1 - i]
        second_slice = p[i : i + n]
        grad[i] = np.dot(first_slice, comp) - np.dot(second_slice, hc_rev)
    return obj, grad

def sum_eq_fun(h, target):
    return np.sum(h) - target

def sum_eq_jac(h, target):
    return np.ones(len(h))

def run_slsqp(h0, n, target, beta, max_iter, ftol):
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: sum_eq_fun(x, target),
                    'jac': lambda x: sum_eq_jac(x, target)}]
    
    def obj_fn(h):
        return obj_and_grad(h, beta)[0]
    
    def grad_fn(h):
        return obj_and_grad(h, beta)[1]
    
    try:
        res = minimize(obj_fn, h0, jac=grad_fn, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': ftol})
        if res.success or res.x is not None:
            return project_sum(res.x, target)
    except Exception:
        pass
    return h0

def local_search(h, n, target, max_rounds=50, improvement_tol=1e-10):
    """Try local coordinate moves to reduce max overlap."""
    best = h.copy()
    best_c5_raw = float(np.max(np.correlate(best, 1.0 - best, mode='full')))
    
    for rnd in range(max_rounds):
        improved = False
        corr = np.correlate(best, 1.0 - best, mode='full')
        max_idx = np.argmax(corr)
        
        for i in range(n):
            if best[i] <= 1e-10 or best[i] >= 1.0 - 1e-10:
                continue
            for shift in range(-n + 1, n):
                j = i + shift
                if j < 0 or j >= n:
                    continue
                if j == i:
                    continue
                # Try moving small amount from i to j
                for delta in [0.02, 0.01, 0.005]:
                    if best[i] < delta or best[j] > 1.0 - delta:
                        continue
                    trial = best.copy()
                    trial[i] -= delta
                    trial[j] += delta
                    trial_clipped = np.clip(trial, 0.0, 1.0)
                    corr_t = np.correlate(trial_clipped, 1.0 - trial_clipped, mode='full')
                    max_t = float(np.max(corr_t))
                    if max_t < best_c5_raw - improvement_tol:
                        best = trial_clipped
                        best_c5_raw = max_t
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best

def local_search2(h, n, target, max_rounds=30):
    """Try adding small amount to one position and subtracting from another."""
    best = h.copy()
    best_c5_raw = float(np.max(np.correlate(best, 1.0 - best, mode='full')))
    
    for rnd in range(max_rounds):
        improved = False
        for i in range(0, n, 2):
            if best[i] <= 1e-10:
                continue
            for delta in [0.01, 0.02, 0.03]:
                if best[i] < delta:
                    continue
                # Find best j to move to
                best_j = -1
                best_val = best_c5_raw
                for j in range(i + 1, min(i + 20, n)):
                    if best[j] >= 1.0 - delta:
                        continue
                    trial = best.copy()
                    trial[i] -= delta
                    trial[j] += delta
                    if np.any(trial < -1e-12) or np.any(trial > 1.0 + 1e-12):
                        continue
                    corr_t = np.correlate(trial, 1.0 - trial, mode='full')
                    max_t = float(np.max(corr_t))
                    if max_t < best_val:
                        best_val = max_t
                        best_j = j
                if best_j >= 0:
                    best[i] -= delta
                    best[best_j] += delta
                    best_c5_raw = best_val
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    
    return project_sum(best, target)

def interpolate_to_n(h_src, n_target):
    n_src = len(h_src)
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
    return np.interp(x_new, x_src, h_src)

def make_binary_like(h, n):
    """Nudge values toward 0 or 1 while preserving sum."""
    target = n / 2.0
    h2 = h.copy()
    # Sort by distance from 0.5
    dist = np.abs(h2 - 0.5)
    order = np.argsort(dist)
    # Set the most extreme ones to 0 or 1
    for idx in order:
        if h2[idx] > 0.5:
            h2[idx] = 1.0
        else:
            h2[idx] = 0.0
    # Now fix the sum
    s = np.sum(h2)
    diff = target - s
    if diff > 0:
        # Need to add: find 0s closest to 0.5 in original
        zeros = np.where(h2 == 0.0)[0]
        zeros_sorted = zeros[np.argsort(-h[zeros])]
        for idx in zeros_sorted[:int(np.ceil(diff))]:
            h2[idx] = 1.0
        # Fine tune
        s2 = np.sum(h2)
        diff2 = target - s2
        frac_idx = zeros_sorted[int(np.ceil(diff)) : int(np.ceil(diff)) + 1]
        if len(frac_idx) > 0 and abs(diff2) > 1e-10:
            if diff2 > 0:
                h2[frac_idx[0]] = min(1.0, diff2)
    elif diff < 0:
        ones = np.where(h2 == 1.0)[0]
        ones_sorted = ones[np.argsort(h[ones])]
        for idx in ones_sorted[:int(np.ceil(-diff))]:
            h2[idx] = 0.0
        s2 = np.sum(h2)
        diff2 = target - s2
        if abs(diff2) > 1e-10 and int(np.ceil(-diff)) < len(ones_sorted):
            idx = ones_sorted[int(np.ceil(-diff))]
            h2[idx] = max(0.0, h2[idx] + diff2)
    return project_sum(h2, target)
