```
def random_lowpass_binary(n,rng,alpha):
    z=rng.normal(size=n)
    F=np.fft.rfft(z)
    freq=np.fft.rfftfreq(n)
    F = F / (1.0 + (np.abs(freq)*n)**alpha)
    y=np.fft.irfft(F,n)
    if y.max()-y.min() < 1e-12:
       y=rng.normal(size=n)
    target=n//2
    order=np.argsort(y)
    threshold=y[order[target-1]] if target>0 else y.min()+1
    b=(y>threshold).astype(float)
    ones=int(np.sum(b))
    if ones>target:
       one_idx=np.flatnonzero(b==1.0)
       # remove the ones with smallest y
       vals=y[one_idx]
       rem=np.argsort(vals)[:ones-target]
       b[one_idx[rem]]=0.0
    elif ones<target:
       zero_idx=np.flatnonzero(b==0.0)
       vals=y[zero_idx]
       add=np.argsort(vals)[::-1][:target-ones]
       b[zero_idx[add]]=1.0
```

```
def project_box_sum(y, target):
    h = np.asarray(y, dtype=float).copy()
    n = len(h)
    if n == 0:
        return h
    s = float(np.sum(h))
    if h.min() >= -1e-12 and h.max() <= 1.0 + 1e-12 and abs(s - target) < 1e-10:
        return np.clip(h, 0.0, 1.0)
    lo = min(-1.0, float(np.min(h)) - 1.0)
    hi = max(2.0, float(np.max(h)) + 1.0)
    mid = 0.5*(lo+hi)
    for _ in range(50):
        val = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if val < target:
            lo = mid
        else:
            hi = mid
        mid = 0.5*(lo+hi)
    out = np.clip(h + mid, 0.0, 1.0)
    diff = target - float(np.sum(out))
    if abs(diff) > 1e-10:
        free = (out > 1e-9) & (out < 1.0 - 1e-9)
        cnt = int(np.count_nonzero(free))
        if cnt > 0:
            out[free] += diff / cnt
        else:
            out += diff / n
        out = np.clip(out, 0.0, 1.0)
    return out
```

```
def random_lowpass_binary(n, rng, alpha):
    if n <=0: return np.zeros(0)
    z=rng.normal(size=n)
    F=np.fft.rfft(z)
    freq=np.fft.rfftfreq(n)
    F = F / (1.0 + (np.abs(freq)*float(n))**alpha)
    y=np.fft.irfft(F,n)
    trend = (np.arange(n, dtype=float)/max(1,n-1) - 0.5)*0.2
    y = y + trend + 0.05*rng.normal(size=n)
    target=n//2
    if target<=0:
       return np.zeros(n)
    order=np.argsort(y)
    threshold=float(y[order[target-1]])
    b=(y>threshold).astype(float)
    ones=int(np.sum(b))
    if ones>target:
       one_idx=np.flatnonzero(b==1.0)
       vals=y[one_idx]
       rem=vals.argsort()[:ones-target]
       b[one_idx[rem]]=0.0
    elif ones<target:
       zero_idx=np.flatnonzero(b==0.0)
       vals=y[zero_idx]
       add=vals.argsort()[::-1][:target-ones]
       b[zero_idx[add]]=1.0
    return b
```

```python
import numpy as np
import time
import math


def project_box_sum(y, target):
    h = np.asarray(y, dtype=float).copy()
    n = len(h)
    if n == 0:
        return h

    s = float(np.sum(h))
    if h.min() >= -1e-12 and h.max() <= 1.0 + 1e-12 and abs(s - target) < 1e-10:
        return np.clip(h, 0.0, 1.0)

    lo = min(-1.0, float(np.min(h)) - 1.0)
    hi = max(2.0, float(np.max(h)) + 1.0)
    mid = 0.5 * (lo + hi)

    for _ in range(50):
        val = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if val < target:
            lo = mid
        else:
            hi = mid
        mid = 0.5 * (lo + hi)

    out = np.clip(h + mid, 0.0, 1.0)
    diff = target - float(np.sum(out))
    if abs(diff) > 1e-10:
        free = (out > 1e-9) & (out < 1.0 - 1e-9)
        cnt = int(np.count_nonzero(free))
        if cnt > 0:
            out[free] += diff / cnt
        else:
            out += diff / n
        out = np.clip(out, 0.0, 1.0)
    return out


def project_box_sum_fast(y, target):
    h = np.clip(np.asarray(y, dtype=float), 0.0, 1.0).copy()
    n = len(h)
    if n == 0:
        return h

    for _ in range(12):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-10:
            break
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = int(np.count_nonzero(free))
        if cnt > 0:
            h[free] += diff / cnt
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)

    diff = target - float(np.sum(h))
    if abs(diff) > 1e-7:
        return project_box_sum(h, target)
    return h


def maxcorr_with_corr(h):
    h = np.asarray(h, dtype=float)
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full")
    return float(np.max(corr)), corr


def fast_c5(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return float("inf")
    mc, _ = maxcorr_with_corr(h)
    return float(mc) * (2.0 / float(n))


def interpolate_to_n(h0, n):
    h0 = np.asarray(h0, dtype=float)
    if n <= 0:
        return np.zeros(0, dtype=float)
    if h0.size == 0:
        return project_box_sum_fast(np.zeros(n, dtype=float), 0.5 * n)

    m = len(h0)
    if m == 1:
        vals = np.full(n, float(h0[0]), dtype=float)
    else:
        xp = (np.arange(m) + 0.5) * (2.0 / m)
        xn = (np.arange(n) + 0.5) * (2.0 / n)
        vals = np.interp(xn, xp, h0)
    return project_box_sum_fast(vals, 0.5 * n)


def to_binary_feasible(h, n):
    target = n // 2
    h = np.asarray(h, dtype=float)
    b = (h >= 0.5).astype(float)
    ones = int(np.sum(b))

    if ones > target:
        idx = np.flatnonzero(b == 1.0)
        order = np.argsort(h[idx])
        remove = order[: ones - target]
        b[idx[remove]] = 0.0
    elif ones < target:
        idx = np.flatnonzero(b == 0.0)
        order = np.argsort(h[idx])[::-1]
        add = order[: target - ones]
        b[idx[add]] = 1.0
    return b


def random_binary(n, rng):
    h = np.zeros(n, dtype=float)
    if n % 2 == 0 and n > 0:
        target = n // 2
        if target > 0:
            idx = rng.choice(n, target, replace=False)
            h[idx] = 1.0
        return h
    return project_box_sum_fast((rng.rand(n) > 0.5).astype(float), n, 0.5 * n)


def random_lowpass_binary(n, rng, alpha):
    if n <= 0:
        return np.zeros(0, dtype=float)

    z = rng.normal(size=n)
    F = np.fft.rfft(z)
    freq = np.fft.rfftfreq(n)
    F = F / (1.0 + (np.abs(freq) * float(n)) ** float(alpha))
    y = np.fft.irfft(F, n)

    trend = (np.arange(n, dtype=float) / max(1, n - 1) - 0.5) * 0.2
    y = y + trend + 0.05 * rng.normal(size=n)

    target = n // 2
    if target <= 0:
        return np.zeros(n, dtype=float)

    order = np.argsort(y)
    threshold = float(y[order[target - 1]])
    b = (y > threshold).astype(float)
    ones = int(np.sum(b))

    if ones > target:
        one_idx = np.flatnonzero(b == 1.0)
        vals = y[one_idx]
        rem = vals.argsort()[: ones - target]
        b[one_idx[rem]] = 0.0
    elif ones < target:
        zero_idx = np.flatnonzero(b == 0.0)
        vals = y[zero_idx]
        add = vals.argsort()[::-1][: target - ones]
        b[zero_idx[add]] = 1.0

    return b


def random_smooth_float(n, rng, order=4):

                        break
                    eta = mult / norm
                    hn = project_box_sum_fast(h - eta * grad, n, target)
                    cn, _ = maxcorr_with_corr(hn)
                    cn5 = cn * dx
                    if cn5 < c5 + 1e-13:
                        h = hn
                        accepted = True
                        cand_c5 = cn5
                        break

            if not accepted:
                break

            if cand_c5 < best_c5 - 1e-15:
                best_c5 = cand_c5
                best_h = h.copy()

        if time.time() >= deadline - 2.0:
            break

        h = best_h + 0.015 * rng.normal(size=n)
        h = project_box_sum_fast(h, n, target)
        cc, _ = maxcorr_with_corr(h)
        hc5 = cc * dx
        if hc5 < best_c5 - 1e-15:
            best_h = h.copy()
            best_c5 = hc5

    return best_h, best_c5


def local_search_binary(h0, n, time_limit, rng):
    if n <= 0 or time_limit <= 0.0:
        return (None if h0 is None else np.asarray(h0, dtype=float)), (fast_c5(h0) if h0 is not None else float("inf"))

    n = int(n)
    if n % 2 != 0:
        n -= 1

    b0 = to_binary_feasible(h0, n).astype(np.float32)
    cur = b0.copy()
    comp = (1.0 - cur).astype(np.float32)

    cur_corr = np.correlate(cur, comp, mode="full")
    cur_count = int(np.max(cur_corr))

    dx = 2.0 / float(n)
    best_b = cur.astype(float).copy()
    best_count = cur_count
    best_c5 = float(cur_count) * dx

    t0 = time.time()
    deadline = t0 + max(1.0, float(time_limit))

    temp = 2.0
    no_improve = 0
    last_improve = t0

    while time.time() < deadline:
        use_targeted = rng.random() < 0.70
        i = -1
        j = -1

        if use_targeted:
            L = len(cur_corr)
            k = min(12, L)
            if k == L:
                top = np.arange(L)
            else:
                top = np.argpartition(cur_corr, L - k)[-k:]

            valid = []
            for idx in top:
                idx = int(idx)
                if cur_corr[idx] > 0.0 and idx != n - 1:
                    valid.append(idx)

            if len(valid) > 0:
                idx = valid[int(rng.randint(len(valid)))]
                s = n - 1 - idx

                if s > 0 and s < n:
                    mask = np.where((cur[: n - s] == 1.0) & (comp[s:] == 1.0))[0]
                    if len(mask) > 0:
                        rel = int(mask[int(rng.randint(len(mask)))])
                        i = rel
                        j = rel + s
                elif s < 0 and -s < n:
                    d = -s
                    mask = np.where((cur[d:] == 1.0) & (comp[: n - d] == 1.0))[0]
                    if len(mask) > 0:
                        rel = int(mask[int(rng.randint(len(mask)))])
                        i = rel + d
                        j = rel

        if i < 0 or j < 0 or i == j:
            ones = np.flatnonzero(cur)
            zeros = np.flatnonzero(comp)
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = int(ones[int(rng.randint(len(ones)))])
            j = int(zeros[int(rng.randint(len(zeros)))])
            if i == j:
                continue

        cur[i] = 0.0
        comp[i] = 1.0
        cur[j] = 1.0
        comp[j] = 0.0

        new_corr = np.correlate(cur, comp, mode="full")
        new_count = int(np.max(new_corr))

        accept = False
        if new_count < best_count:
            accept = True
        elif new_count == best_count:
            if rng.random() < 0.25:
                accept = True
        else:
            delta = new_count - cur_count
            if delta <= 0:
                accept = True
            else:
                if temp > 1e-3:
                    ratio = float(delta) / temp
                    if ratio < 700.0:
                        if rng.random() < math.exp(-ratio):
                            accept = True

        if accept:
            cur_count = new_count
            cur_corr = new_corr
            if new_count < best_count:
                best_count = new_count
                best_b = cur.astype(float).copy()
                best_c5 = float(best_count) * dx
                no_improve = 0
                last_improve = time.time()
                temp = max(0.10, temp * 0.97)
            else:
                no_improve += 1
            temp = max(0.05, temp * 0.9995)
        else:
            cur[i] = 1.0
            comp[i] = 0.0
            cur[j] = 0.0
            comp[j] = 1.0
            no_improve += 1
            if no_improve > 300:
                temp = min(6.0, temp * 1.4)

        if time.time() - last_improve > 18.0:
            cur = best_b.astype(np.float32).copy()
            comp = (1.0 - cur).astype(np.float32)

            mut = int(max(2, n * (0.01 + 0.05 * rng.rand())))
            for _ in range(mut):
                ones = np.flatnonzero(cur)
                zeros = np.flatnonzero(comp)
                if len(ones) == 0 or len(zeros) == 0:
                    break
                ii = int(ones[int(rng.randint(len(ones)))])
                jj = int(zeros[int(rng.randint(len(zeros)))])
                if ii == jj:
                    continue
                cur[ii] = 0.0
                comp[ii] = 1.0
                cur[jj] = 1.0
                comp[jj] = 0.0

            cur_corr = np.correlate(cur, comp, mode="full")
            cur_count = int(np.max(cur_corr))
            last_improve = time.time()
            no_improve = 0

            if cur_count < best_count:
                best_count = cur_count
                best_b = cur.astype(float).copy()
                best_c5 = float(best_count) * dx

    return best_b, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    budget_s = float(budget_s)
    deadline = t_start + max(5.0, budget_s - 30.0)
    rng = np.random.RandomState(int(seed))

    best_h = None
    best_c5 = float("inf")
    best_n = 0

    parent_h = None
    parent_n = 0
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        if parent_n > 0:
            parent_h = project_box_sum(parent_h, parent_n, 0.5 * parent_n)
            best_h = parent_h.copy()
            best_n = parent_n
            best_c5 = fast_c5(best_h)
            print("parent: n={}, C5={:.8f}".format(best_n, best_c5))
    except Exception:
        parent_h = None
        parent_n = 0

    if best_h is None:
        best_n = 128
        best_h = random_lowpass_binary(best_n, rng, 1.0)
        best_c5 = fast_c5(best_h)
        print("fallback initial: n={}, C5={:.8f}".format(best_n, best_c5))

    all_ns = []
    for nn in (best_n if best_n > 0 else 0, 96, 128, 160, 192, 224, 256, 288, 320):
        if nn > 0 and 32 <= nn <= 384 and nn not in all_ns:
            all_ns.append(nn)

    cont_order = []
    for nn in (best_n if best_n > 0 else 0, 128, 192, 256, 160, 96, 224):
        if nn in all_ns and nn not in cont_order:
            cont_order.append(nn)

    cont_deadline = min(deadline, t_start + max(10.0, 0.18 * budget_s - 10.0))

    for idx_cont, n in enumerate(cont_order):
        if time.time() >= cont_deadline - 5.0:
            break

        starts = []

        if parent_h is not None and parent_n > 0:
            if n == parent_n:
                starts.append(parent_h.copy())
            else:
                starts.append(interpolate_to_n(parent_h, n))

        if best_h is not None and best_n > 0:
            if n == best_n:
                cand = best_h.copy()
            else:
                cand = interpolate_to_n(best_h, n)
            starts.append(cand)

        starts.append(random_smooth_float(n, rng))
        starts = starts[:2]

        if time.time() >= cont_deadline - 5.0:
            break

        rem = max(1, len(cont_order) - idx_cont)
        per = max(6.0, min(30.0, (cont_deadline - time.time()) / max(1, rem * max(1, len(starts)))))

        for st in starts:
            if time.time() >= cont_deadline:
                break
            h2, c2 = smooth_projected_gradient(st, n, per, rng)
            if h2 is not None and c2 < best_c5:
                best_h = np.asarray(h2, dtype=float)
                best_c5 = float(c2)
                best_n = n
                print("continuous improved: n={}, C5={:.8f}".format(best_n, best_c5))

    local_order = []
    for nn in (128, 192, 160, 256, 96, 224, 288, 320):
        if nn % 2 == 0 and nn in all_ns and nn not in local_order:
            local_order.append(nn)

    if best_n > 0 and best_n % 2 == 0 and best_n not in local_order:
        ins = 0
        if len(local_order) > 1:
            ins = 1
        local_order.insert(ins, best_n)

    for idx_local, n in enumerate(local_order):
        if time.time() >= deadline - 30.0:
            break

        starts = []

        if best_h is not None and best_n > 0:
            if n == best_n:
                starts.append(best_h.copy())
            else:
                starts.append(interpolate_to_n(best_h, n))

        if parent_h is not None and parent_n > 0 and n != parent_n:
            starts.append(interpolate_to_n(parent_h, n))

        starts.append(random_lowpass_binary(n, rng, 0.9))
        starts.append(random_lowpass_binary(n, rng, 1.6))
        if len(starts) < 4:
            starts.append(random_binary(n, rng))

        starts = starts[:4]

        rem = max(1, len(local_order) - idx_local)
        rem_time = max(5.0, deadline - 40.0 - time.time())
        per = max(8.0, min(45.0, rem_time / max(1, rem * max(1, len(starts)))))
        if per < 5.0:
            break

        for st in starts:
            if time.time() >= deadline - 30.0:
                break
            b, c = local_search_binary(st, n, per, rng)
            if b is not None and c < best_c5:
                best_h = np.asarray(b, dtype=float)
                best_c5 = float(c)
                best_n = n
                print("local improved: n={}, C5={:.8f}".format(best_n, best_c5))

        if best_n == n and time.time() < deadline - 60.0:
            extra = min(30.0, max(5.0, deadline - 60.0 - time.time()))
            b, c = local_search_binary(best_h, n, extra, rng)
            if b is not None and c < best_c5:
                best_h = np.asarray(b, dtype=float)
                best_c5 = float(c)
                best_n = n
                print("local polish improved: n={}, C5={:.8f}".format(best_n, best_c5))

    if time.time() < deadline - 25.0 and best_h is not None and best_n > 0:
        rem = min(25.0, max(5.0, deadline - 30.0 - time.time()))
        if best_n % 2 == 0:
            b, c = local_search_binary(best_h, best_n, rem, rng)
            if b is not None and c < best_c5:
                best_h = np.asarray(b, dtype=float)
                best_c5 = float(c)
                best_n = best_n
                print("final binary polish: C5={:.8f}".format(best_c5))
        else:
            h2, c2 = smooth_projected_gradient(best_h, best_n, rem, rng)
            if h2 is not None and c2 < best_c5:
                best_h = np.asarray(h2, dtype=float)
                best_c5 = float(c2)
                print("final continuous polish: C5={:.8f}".format(best_c5))

    if time.time() < deadline - 20.0 and best_h is not None and best_n > 0 and best_n % 2 == 0 and best_n <= 320:
        rem = min(18.0, max(5.0, deadline - 25.0 - time.time()))
        h2, c2 = smooth_projected_gradient(best_h, best_n, rem, rng)
        if h2 is not None and c2 < best_c5:
            best_h = np.asarray(h2, dtype=float)
            best_c5 = float(c2)
            print("post-continuous refinement: C5={:.8f}".format(best_c5))

    if best_h is None:
        best_h = random_lowpass_binary(128, rng, 1.0)
        best_n = 128

    best_h = project_box_sum(best_h, best_n, 0.5 * best_n)
    final_c5 = fast_c5(best_h)

    if best_n > 0 and best_n % 2 == 0:
        hb = to_binary_feasible(best_h, best_n)
        cb = fast_c5(hb)
        if cb < final_c5:
            best_h = hb
            final_c5 = float(cb)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, best_n, 0.5 * best_n)
    final_c5 = fast_c5(best_h)

    try:
        ev = float(evaluate_erdos_solution(best_h))
        if np.isfinite(ev) and ev < final_c5:
            final_c5 = ev
    except Exception:
        pass

    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (np.asarray(best_h, dtype=float), float(final_c5), int(best_n))
```
