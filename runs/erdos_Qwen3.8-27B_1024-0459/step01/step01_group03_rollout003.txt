def project_density(x, target):
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    if n == 0:
        return x
    target = float(target)
    s = float(np.sum(x))
    if abs(s - target) <= 1e-13 * max(1, n):
        return x

    lo, hi = -1.0, 1.0
    for _ in range(75):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid

    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))

    if abs(res) > 1e-13:
        free = np.where((h > 1e-14) & (h < 1.0 - 1e-14))[0]
        if free.size:
            if res > 0.0:
                idx = int(free[np.argmax(1.0 - h[free])])
                h[idx] = min(1.0, h[idx] + res)
            else:
                idx = int(free[np.argmax(h[free])])
                h[idx] = max(0.0, h[idx] + res)

    res = target - float(np.sum(h))
    if abs(res) > 1e-13:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size:
            h[free] += res / float(free.size)
            h = np.clip(h, 0.0, 1.0)

    res = target - float(np.sum(h))
    if abs(res) > 1e-13:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size:
            h[free[0]] += res
            h = np.clip(h, 0.0, 1.0)
        else:
            free = np.where(h < 1.0 if res > 0.0 else h > 0.0)[0]
            if free.size:
                h[free[0]] += res
                h = np.clip(h, 0.0, 1.0)

    return h


def corr_values(h):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return np.array([], dtype=float)
    c = np.correlate(h, 1.0 - h, mode="full")
    return c


def eval_c5(h):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return float("inf")
    c = corr_values(h)
    if c.size == 0 or not np.all(np.isfinite(c)):
        return 1e100
    return float(np.max(c) * (2.0 / float(h.size)))


def make_smooth_candidate(n, rng, mode):
    n = int(n)
    if n <= 1:
        return project_density(np.array([0.5], dtype=float), 0.5 * n)

    if mode == 1:
        u = rng.random(n)
        thresh = float(np.median(u))
        h = (u > thresh).astype(float)
    elif mode == 2:

        levels = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
        pos = 0
        while pos < n:

            length = int(rng.integers(1, rem + 1))

            h[pos:pos + length] = val

def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    p = np.nan_to_num(p, nan=0.5, posinf=1.0, neginf=0.0)
    p = np.clip(p, 0.0, 1.0)
    p0 = p.size
    n = int(n)
    if p0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    if p0 == n:
        return p.copy()
    if n % p0 == 0:
        return project_density(np.repeat(p, n // p0), 0.5 * n)

    x0 = (np.arange(p0) + 0.5) * (2.0 / float(p0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)


def round_binary(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    k = n // 2
    out = np.zeros(n, dtype=float)
    idx = np.argsort(h)[::-1][:k]
    out[idx] = 1.0
    return out

def vec_close(a, b):
    if a.size != b.size:
        return False
    return bool(np.allclose(a, b, rtol=0.0, atol=1e-8))


def build_deriv(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    g = np.zeros((n, 2 * n - 1), dtype=float)
    ctr = n - 1
    g[np.arange(n), ctr] = 1.0 - 2.0 * h

    for t in range(1, n):
        idx1 = np.arange(n - t)
        idx2 = idx1 + t
        pos = ctr + t
        neg = ctr - t
        g[idx2, pos] += 1.0 - h[idx1]
        g[idx1, pos] += -h[idx2]
        g[idx1, neg] += 1.0 - h[idx2]
        g[idx2, neg] += -h[idx1]
    return g


def add_delta(ds, x, cap):
    if not np.isfinite(x):
        return ds
    if x < cap:
        if x <= 1e-12:
            return ds
        if x > cap:
            x = cap
    else:
        x = cap
    if cap - x <= 1e-13:
        x = cap
    for y in ds:
        if abs(x - y) <= 1e-9:
            return ds
    return ds + [float(x)]

        a = float(gdiff[ell])
        b = 1.0 if (ell == lpos or ell == lneg) else 0.0
        if b > 0.0 and abs(a) > 1e-12:
            opt = -a / (2.0 * b)
            if 0.0 < opt < cap:
                opts.append(opt)
        if a < 0.0:
            opts.append(0.67 * cap)
            opts.append(0.33 * cap)

    for opt in opts:
        if len(ds) >= max_samples - 2:
            break
        ds = add_delta(ds, opt, cap)

    for frac in (0.5, 0.25, 0.125, 0.05):
        if len(ds) >= max_samples:
            break
        ds = add_delta(ds, frac * cap, cap)

    return ds[:max_samples]


def pair_descent(h, deadline, rng, max_iters=500, top_lags=12, max_samples=4):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n < 2:
        return h, eval_c5(h)
    target = 0.5 * n
    h = project_density(h, target)
    best = h.copy()
    best_c = eval_c5(h)
    dx = 2.0 / float(n)
    max_stag = 0
    ctr = n - 1

    for _ in range(int(max_iters)):
        if time.perf_counter() >= deadline:
            break

        c = corr_values(h)
        curM = float(np.max(c))
        curC5 = curM * dx
        curSum = float(np.sum(c))

        if curC5 < best_c - 1e-13:
            best = h.copy()
            best_c = curC5

        if curM <= 1e-13:
            break

        g = build_deriv(h, n)
        top = np.argsort(c)[-int(max(3, top_lags)):]
        bestM = curM + 1e-12
        bestSum = curSum
        best_move = None

        done_time = False
        for i in range(n - 1):
            if done_time:
                break
            ai = float(h[i])
            for j in range(i + 1, n):
                if (j & 255) == 0 and time.perf_counter() >= deadline:
                    done_time = True
                    break
                bj = float(h[j])
                d = j - i
                lpos = ctr + d
                lneg = ctr - d

                cap_up = min(1.0 - ai, bj)
                if cap_up > 1e-12:
                    gdiff = g[i, :] - g[j, :]
                    deltas = make_delta_candidates(gdiff, top, cap_up, lpos, lneg, max_samples)
                    if deltas:
                        darr = np.asarray(deltas, dtype=float)
                        cand = c[None, :] + darr[:, None] * gdiff[None, :]
                        cand[:, lpos] += darr * darr
                        if 0 <= lneg < c.size:
                            cand[:, lneg] += darr * darr
                        mns = np.max(cand, axis=1)
                        minm = float(np.min(mns))
                        if minm < bestM - 1e-10:
                            close = np.where(np.abs(mns - minm) <= 1e-10)[0]
                            if close.size == 1:
                                idx = int(close[0])
                            else:
                                sums = cand[close, :].sum(axis=1)
                                idx = int(close[int(np.argmin(sums))])
                            bestM = minm
                            bestSum = float(np.sum(cand[idx, :]))
                            best_move = (i, j, float(darr[idx]))
                        elif minm < curM + 1e-10:
                            close = np.where(np.abs(mns - minm) <= 1e-10)[0]
                            if close.size:
                                sums = cand[close, :].sum(axis=1)
                                idx = int(close[int(np.argmin(sums))])
                                if float(sums[idx]) < bestSum - 1e-8:
                                    bestM = float(minm)
                                    bestSum = float(sums[idx])
                                    best_move = (i, j, float(darr[idx]))

                cap_dn = min(1.0 - bj, ai)
                if cap_dn > 1e-12:
                    gdiff = g[j, :] - g[i, :]
                    deltas = make_delta_candidates(gdiff, top, cap_dn, lpos, lneg, max_samples)
                    if deltas:
                        darr = np.asarray(deltas, dtype=float)
                        cand = c[None, :] + darr[:, None] * gdiff[None, :]
                        cand[:, lpos] += darr * darr
                        if 0 <= lneg < c.size:
                            cand[:, lneg] += darr * darr
                        mns = np.max(cand, axis=1)
                        minm = float(np.min(mns))
                        if minm < bestM - 1e-10:
                            close = np.where(np.abs(mns - minm) <= 1e-10)[0]
                            if close.size == 1:
                                idx = int(close[0])
                            else:
                                sums = cand[close, :].sum(axis=1)
                                idx = int(close[int(np.argmin(sums))])
                            bestM = minm
                            bestSum = float(np.sum(cand[idx, :]))
                            best_move = (j, i, float(darr[idx]))
                        elif minm < curM + 1e-10:
                            close = np.where(np.abs(mns - minm) <= 1e-10)[0]
                            if close.size:
                                sums = cand[close, :].sum(axis=1)
                                idx = int(close[int(np.argmin(sums))])
                                if float(sums[idx]) < bestSum - 1e-8:
                                    bestM = float(minm)
                                    bestSum = float(sums[idx])
                                    best_move = (j, i, float(darr[idx]))

        if best_move is not None:
            i, j, dval = best_move
            h2 = h.copy()
            h2[i] += dval
            h2[j] -= dval
            h2 = project_density(np.clip(h2, 0.0, 1.0), target)
            c5r = eval_c5(h2)
            if c5r <= curC5 + 1e-10:
                h = h2
                if c5r < best_c - 1e-13:
                    best = h.copy()
                    best_c = c5r
                    max_stag = 0
                else:
                    max_stag += 1
            else:
                max_stag = min(50, max_stag + 1)
        else:
            if max_stag < 35 and time.perf_counter() < deadline - 2.0:
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                tries = 0
                while i == j and tries < 20:
                    i = int(rng.integers(0, n))
                    j = int(rng.integers(0, n))
                    tries += 1
                if i != j:
                    if rng.random() < 0.5:
                        cap = min(1.0 - float(h[i]), float(h[j]))
                        iu, ju = i, j
                    else:
                        cap = min(1.0 - float(h[j]), float(h[i]))
                        iu, ju = j, i
                    if cap > 1e-9:
                        dval = cap * float(rng.uniform(0.01, 0.06))
                        h2 = h.copy()
                        h2[iu] += dval
                        h2[ju] -= dval
                        h2 = project_density(np.clip(h2, 0.0, 1.0), target)
                        c5r = eval_c5(h2)
                        if c5r < best_c + 0.01:
                            h = h2
                            if c5r < best_c - 1e-13:
                                best = h.copy()
                                best_c = c5r
                                max_stag = 0
                            else:
                                max_stag += 1
                        else:
                            max_stag += 1
                    else:
                        max_stag += 1
                else:
                    max_stag += 1
            else:
                max_stag += 1

        if max_stag > 35:
            break

    h = project_density(best, target)
    return h, eval_c5(h)


def binary_score(c, T):
    m = float(np.max(c))
    bad = float(np.sum(np.maximum(c - float(T), 0.0)))
    s = float(np.sum(c))
    return m, bad, s


def binary_score_better(m1, b1, s1, m2, b2, s2):
    if m1 < m2 - 1e-9:
        return True
    if m1 > m2 + 1e-9:
        return False
    if b1 < b2 - 1e-9:
        return True
    if b1 > b2 + 1e-9:
        return False
    return s1 < s2 - 1e-8


def binary_hill(h, T, deadline, rng):
    h = round_binary(h)
    n = h.size
    T = int(T)
    best = h.copy()
    bc = eval_c5(best)
    bm, bb, bs = binary_score(corr_values(best), T)

    stalls = 0
    while time.perf_counter() < deadline:
        c = corr_values(h)
        m, bad, s = binary_score(c, T)
        if m < bm - 1e-9 or (abs(m - bm) <= 1e-9 and bad < bb - 1e-9 and s < bs - 1e-8):
            best = h.copy()
            bm, bb, bs = m, bad, s
            bc = eval_c5(best)

        if m <= T and bad <= 1e-8:
            break

        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1.0 - h)
        if ones.size == 0 or zeros.size == 0:
            break

        found = False
        for ii in range(len(ones)):
            i = int(ones[ii])
            for jj in range(len(zeros)):
                j = int(zeros[jj])
                if i == j:
                    continue
                hc = h.copy()
                hc[i] = 0.0
                hc[j] = 1.0
                cc = corr_values(hc)
                cmax = float(np.max(cc))
                if cmax > m + 1e-9:
                    continue
                bad2 = float(np.sum(np.maximum(cc - float(T), 0.0)))
                sum2 = float(np.sum(cc))
                if binary_score_better(cmax, bad2, sum2, m, bad, s):
                    h = hc
                    m, bad, s = cmax, bad2, sum2
                    found = True

        if found:
            stalls = 0
        else:
            stalls += 1
            kick = int(rng.integers(1, 5))
            for _ in range(kick):
                ones = np.flatnonzero(h)
                zeros = np.flatnonzero(1.0 - h)
                if ones.size == 0 or zeros.size == 0:
                    break
                i = int(ones[int(rng.integers(len(ones)))])
                j = int(zeros[int(rng.integers(len(zeros)))])
                h[i] = 0.0
                h[j] = 1.0

            if stalls > 22:
                break

    return best, eval_c5(best)


def binary_local_search(current_h, n, deadline, rng):
    n = int(n)
    if n < 4 or time.perf_counter() >= deadline - 5.0:
        return None, float("inf")
    target_m = min(int(np.floor(_TARGET_C * n / 2.0)), 17 if n == 90 else n // 8 + 5)
    if target_m < 1:
        return None, float("inf")

    starts = []
    if current_h is not None:
        if current_h.size == n:
            starts.append(round_binary(current_h))
        else:
            starts.append(round_binary(upsample_h(current_h, n)))
    for _ in range(12):
        starts.append(random_binary(n, rng))
    for mode in range(3):
        starts.append(round_binary(make_smooth_candidate(n, rng, mode)))

    uniq = []
    for st in starts:
        st = round_binary(st)
        dup = False
        for u in uniq:
            if vec_close(st, u):
                dup = True
                break
        if not dup:
            uniq.append(st)

    best_h = None
    best_c = float("inf")
    for st in uniq:
        if time.perf_counter() >= deadline - 8.0:
            break
        sub_dead = min(deadline, time.perf_counter() + 25.0)
        h2, c2 = binary_hill(st, target_m, sub_dead, rng)
        if h2 is not None and c2 < best_c:
            best_h = h2
            best_c = c2
        if best_c <= 0.378:
            break
    return best_h, best_c


def fourier_set_context(basis, n, alpha):
    global _FOURIER_BASIS, _FOURIER_N, _FOURIER_ALPHA
    _FOURIER_BASIS = basis
    _FOURIER_N = int(n)
    _FOURIER_ALPHA = float(alpha)


def fourier_build_h(a):
    a = np.asarray(a, dtype=float)
    a = np.nan_to_num(a, nan=0.0, posinf=4.0, neginf=-4.0)
    a = np.clip(a, -4.0, 4.0)
    basis = _FOURIER_BASIS
    n = _FOURIER_N
    if n <= 0 or basis is None:
        return project_density(np.zeros(1), 0.0)

    z = np.dot(basis.T, a)
    t = np.tanh(z)
    y = t - float(np.mean(t))
    amax = float(np.max(np.abs(y)))
    if amax > 1e-12:
        y = y / amax
    else:
        y = np.zeros(n, dtype=float)

    h = 0.5 - 0.5 * y
    return project_density(h, 0.5 * n)


def fourier_obj(a):
    try:
        h = fourier_build_h(a)
        c = corr_values(h)
        if c.size == 0 or not np.all(np.isfinite(c)):
            return 1e100
        m = float(np.max(c))
        alpha = _FOURIER_ALPHA
        if alpha <= 1e-6:
            return (2.0 / float(_FOURIER_N)) * m
        y = alpha * (c - m)
        y = np.clip(y, -745.0, 0.0)
        e = np.exp(y)
        z = float(np.sum(e))
        if z <= 0.0 or not np.isfinite(z):
            return 1e100
        return float((2.0 / float(_FOURIER_N)) * (m + np.log(z) / alpha))
    except Exception:
        return 1e100


def fourier_optimize(n, deadline, rng):
    n = int(n)
    if n < 20 or time.perf_counter() >= deadline - 20.0:
        return None, float("inf")

    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    mlist = (1, 2, 3, 4, 8, 12)
    d = min(6, max(2, n // 15))
    basis = np.empty((d, n), dtype=float)
    for k, m in enumerate(mlist[:d]):
        basis[k] = np.cos(float(m) * np.pi * (x - 1.0))

    fourier_set_context(basis, n, 50.0)

    best_h = None
    best_c = float("inf")

    starts = [
        np.array([1.0, -0.25, 0.1, 0.0, 0.0, 0.0], dtype=float)[:d],
        rng.standard_normal(d) * 0.9,
        rng.standard_normal(d) * 1.8,
        np.array([0.0, 1.0, 0.0, -0.5, 1.0, 0.0], dtype=float)[:d],
    ]

    for a0 in starts:
        if time.perf_counter() >= deadline - 20.0:
            break
        try:
            res = minimize(
                fourier_obj,
                a0,
                method="Powell",
                options={"maxiter": 130, "xtol": 1e-5, "fatol": 1e-8, "disp": False},
            )
            a_best = getattr(res, "x", None)
            if a_best is None:
                a_best = a0
            h2 = fourier_build_h(a_best)
            c2 = eval_c5(h2)
            if h2 is not None and c2 < best_c:
                best_h = h2
                best_c = c2
        except Exception:
            continue

    return best_h, best_c


def logsumexp_polish(h, deadline, rng, max_iters=500):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n < 2:
        return h, eval_c5(h)
    target = 0.5 * n
    h = project_density(h, target)
    best = h.copy()
    best_c = eval_c5(h)
    dx = 2.0 / float(n)

    alphas = (18.0, 40.0, 80.0, 150.0)
    worst_stag = 0

    for alpha in alphas:
        stag = 0
        for _ in range(int(max_iters)):
            if time.perf_counter() >= deadline:
                return best, best_c

            c = corr_values(h)
            m = float(np.max(c))
            if m <= 1e-13:
                break
            e = np.exp(np.clip(alpha * (c - m), -745.0, 0.0))
            w = e / float(np.sum(e))
            g = build_deriv(h, n)
            grad = np.sum(g * w[None, :], axis=1) * dx
            d = -grad
            d -= float(np.mean(d))
            nn = float(np.linalg.norm(d))
            if nn < 1e-14:
                break
            d /= nn

            improved = False
            for step in (0.18, 0.08, 0.04, 0.02, 0.01, 0.005):
                if time.perf_counter() >= deadline:
                    break
                cand = project_density(h + step * d, target)
                cc = eval_c5(cand)
                if cc < best_c - 1e-12:
                    h = cand
                    best = cand.copy()
                    best_c = cc
                    improved = True
                    break

            if improved:
                stag = 0
                worst_stag = 0
            else:
                stag += 1
                if stag > 18:
                    break

        if stag > 18:
            worst_stag += 1
        else:
            worst_stag = 0
        if worst_stag > 2:
            break

    return best, eval_c5(best)


def feasible_enough(h, tol=1e-6):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    target = 0.5 * h.size
    return abs(float(np.sum(h)) - target) <= max(1e-7, tol * h.size)


def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    if budget_s is None:
        budget = 1300.0
    else:
        budget = float(budget_s)
    budget = max(20.0, budget)
    deadline = start + max(1.0, budget - 7.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")

    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        pp = project_density(parent, 0.5 * float(parent.size))
        cp = eval_c5(pp)
        if cp < best_c:
            best_h = pp
            best_c = cp

    if best_h is None:
        n0 = 90
        best_h = make_smooth_candidate(n0, rng, 0)
        best_c = eval_c5(best_h)

    if not feasible_enough(best_h):
        best_h = project_density(best_h, 0.5 * float(best_h.size))
        best_c = eval_c5(best_h)

    # Heavy pairwise local descent on the parent resolution.
    if best_h.size <= 220 and time.perf_counter() < deadline - 110.0:
        cap_time = 190.0 if best_h.size <= 100 else 210.0
        sub_dead = min(deadline, time.perf_counter() + cap_time)
        h2, c2 = pair_descent(
            best_h,
            sub_dead,
            rng,
            max_iters=1200 if best_h.size <= 100 else 600,
            top_lags=14 if best_h.size <= 100 else 12,
            max_samples=4 if best_h.size <= 100 else 3,
        )
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Smooth subgradient polish.
    if best_h.size <= 220 and time.perf_counter() < deadline - 80.0:
        sub_dead = min(deadline, time.perf_counter() + 70.0)
        h3, c3 = logsumexp_polish(best_h, sub_dead, rng, max_iters=350)
        if c3 < best_c:
            best_h = h3
            best_c = c3

    # Higher resolution from the current best.
    if best_h.size < 180 and time.perf_counter() < deadline - 240.0:
        n2 = 180
        h4 = upsample_h(best_h, n2)
        sub_dead = min(deadline, time.perf_counter() + 235.0)
        h4, c4 = pair_descent(
            h4,
            sub_dead,
            rng,
            max_iters=420,
            top_lags=11,
            max_samples=3,
        )
        if c4 < best_c:
            best_h = h4
            best_c = c4

        if best_h.size == 180 and time.perf_counter() < deadline - 55.0:
            sub_dead = min(deadline, time.perf_counter() + 45.0)
            h5, c5 = logsumexp_polish(best_h, sub_dead, rng, max_iters=180)
            if c5 < best_c:
                best_h = h5
                best_c = c5

    # Parametric symmetric-family optimizer.
    if time.perf_counter() < deadline - 100.0:
        nf = 120
        hf, cf = fourier_optimize(nf, min(deadline, time.perf_counter() + 90.0), rng)
        if hf is not None and cf < best_c:
            best_h = hf
            best_c = cf

        if best_h is not None and best_h.size == nf and time.perf_counter() < deadline - 55.0:
            sub_dead = min(deadline, time.perf_counter() + 45.0)
            h6, c6 = pair_descent(
                best_h,
                sub_dead,
                rng,
                max_iters=260,
                top_lags=12,
                max_samples=4,
            )
            if c6 < best_c:
                best_h = h6
                best_c = c6

    # Binary min-conflicts search at the parent resolution.
    if time.perf_counter() < deadline - 150.0:
        hb, cb = binary_local_search(best_h, 90, min(deadline, time.perf_counter() + 145.0), rng)
        if hb is not None and cb < best_c:
            best_h = hb
            best_c = cb

    # Final polishing on the best discovered vector.
    if best_h is not None and best_h.size <= 260 and time.perf_counter() < deadline - 25.0:
        sub_dead = min(deadline - 2.0, time.perf_counter() + 20.0)
        h7, c7 = pair_descent(
            best_h,
            sub_dead,
            rng,
            max_iters=220,
            top_lags=14,
            max_samples=4,
        )
        if c7 < best_c:
            best_h = h7
            best_c = c7

    if best_h is None:
        best_h = project_density(np.full(90, 0.5, dtype=float), 45.0)
        best_c = eval_c5(best_h)

    final_h = project_density(best_h, 0.5 * float(best_h.size))
    final_c = eval_c5(final_h)
    if not np.all(np.isfinite(final_h)):
        final_h = np.clip(np.nan_to_num(final_h, 0.5, 1.0, 0.0), 0.0, 1.0)
        final_h = project_density(final_h, 0.5 * float(final_h.size))
        final_c = eval_c5(final_h)

    return final_h, float(final_c), int(final_h.size)
