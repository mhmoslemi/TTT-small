                    if c5_new < c5_cur:
                        h0 = h_new
                        c5_cur = c5_new
                except Exception:
                    pass

            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h0.copy()
                best_n = n
                print("  Improved n={}, C5={:.6f}".format(n, best_c5))

    # Subgradient targeting: reduce the argmax correlation
    if remaining_time() > 25:
        n = best_n
        _N = n
        _TGT = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_sum}]
        h = best_h.copy()

        for it in range(300):
            if remaining_time() < 4:
                break
            dx_val = 2.0 / n
            corr = np.correlate(h, 1.0 - h, mode='full')
            k_max = int(np.argmax(corr))
            offset = k_max - (n - 1)

            # Find contributing positions
            if offset >= 0:
                n_pairs = n - offset
                # h[i+offset] * (1-h[i]) for i=0..n_pairs-1
                h_shift = h[offset:offset + n_pairs]
                h_base = 1.0 - h[:n_pairs]
                # grad[j] from h_shift part: j in [offset, offset+n_pairs), contrib = (1-h[j-offset])
                # grad[j] from h_base part: j in [0, n_pairs), contrib = -h[j+offset]
                grad = np.zeros(n)
                grad[offset:offset + n_pairs] += h_base
                grad[:n_pairs] -= h_shift
            else:
                abs_off = -offset
                n_pairs = n - abs_off
                h_shift = h[:n_pairs]
                h_base = 1.0 - h[abs_off:abs_off + n_pairs]
                grad = np.zeros(n)
                grad[:n_pairs] += h_base
                grad[abs_off:abs_off + n_pairs] -= h_shift

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm

            improved_this_iter = False
            for step in [0.03, 0.07, 0.15, 0.02, 0.1]:
                h_trial = h - step * grad
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h = h_trial.copy()
                    improved_this_iter = True
                    break

            if not improved_this_iter and remaining_time() > 12:
                _BETA = 300
                try:
                    res = minimize(obj_smooth, h, method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 200, 'ftol': 1e-15, 'iprint': 0})
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h = h_new.copy()
                except Exception:
                    pass

    # Coordinate descent / bit-flipping
    if remaining_time() > 12:
        n = best_n
        _N = n
        _TGT = n / 2.0
        h = best_h.copy()

        for sweep in range(20):
            if remaining_time() < 4:
                break
            any_imp = False
            order = rng.permutation(n)
            for i in order:
                if remaining_time() < 2:
                    break
                old_val = h[i]
                for new_val in [0.0, 1.0]:
                    if abs(new_val - old_val) < 1e-6:
                        continue
                    h[i] = new_val
                    h = project_feasible(h, n)
                    c5_test = compute_c5(h)
                    if c5_test < best_c5:
                        best_c5 = c5_test
                        best_h = h.copy()
                        any_imp = True
                    else:
                        h = best_h.copy()
            if not any_imp:
                break

    # Final polish with high-beta SLSQP
    if remaining_time() > 10:
        n = best_n
        _N = n
        _TGT = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_sum}]
        for beta_val in [1000, 2000]:
            if remaining_time() < 5:
                break
            _BETA = beta_val
            try:
                res = minimize(obj_smooth, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-16, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
