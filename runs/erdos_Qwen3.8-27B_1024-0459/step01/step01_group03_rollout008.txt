    if idx.size == 0:
        idx = [int(np.argmax(w))]
    g = _corr_grad_avg(h, idx)
    # Re-weight by w
    g2 = np.zeros(n)
    center = n - 1
    for t in idx:
        t = int(t)
        p = w[t]
        k = t - center
        if k == 0:
            g2 += p * (1.0 - 2.0 * h)
        else:
            lag = abs(k)
            if k > 0:
                g2[lag:] += p
            else:
                g2[:n - lag] += p
            g2[:n - lag] -= p * h[lag:]
            g2[lag:] -= p * h[:n - lag]
    return (2.0 / float(n)) * g2

def _lsq_optimize(h0, n, deadline):
    target = 0.5 * n
    h = _project(h0, target)
    best_h = h.copy()
    best_c = _evaluate(best_h)
    bounds = [(0.0, 1.0)] * n

    alphas = [50, 150, 350, 600]
    maxiters = [80, 60, 50, 40]

    for i, (alpha, miter) in enumerate(zip(alphas, maxiters)):
        if time.perf_counter() >= deadline - 2:
            break
        try:
            res = minimize(
                lambda x: _lsq_smooth_objective(x, n, alpha, target),
                h,
                method="SLSQP",
                jac=lambda x: _lsq_smooth_jac(x, n, alpha, target),
                bounds=bounds,
                constraints=[{"type": "eq", "fun": lambda x: np.sum(x) - target,
                              "jac": lambda x: np.ones(n)}],
                options={"maxiter": miter, "ftol": 1e-10, "disp": False}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                cand = _project(res.x, target)
                c = _evaluate(cand)
                if c < best_c:
                    best_h = cand
                    best_c = c
                h = cand if c <= best_c + 0.01 else best_h
        except Exception:
            pass

    return best_h, best_c

def _chebyshev_search(n, M, deadline, rng):
    """Global search using Chebyshev parameterization."""
    target = 0.5 * n
    xpts = (np.arange(n) + 0.5) * (2.0 / n) - 1.0  # in [-1, 1]

    def cheb_eval(coeffs):
        val = np.zeros(n)
        T0 = np.ones(n)
        T1 = xpts.copy()
        val += coeffs[0] * T0
        if M >= 1:
            val += coeffs[1] * T1
        for j in range(2, M + 1):
            Tj = 2.0 * xpts * T1 - T0
            val += coeffs[j] * Tj
            T0, T1 = T1, Tj
        return val

    best_h = None
    best_c = float("inf")

    def obj(coeffs):
        h = _project(cheb_eval(coeffs), target)
        return _evaluate(h)

    # Multiple random restarts with Nelder-Mead
    n_restarts = max(3, int(60.0 / (time.perf_counter() - 0 + 1)))  # won't work, use fixed
    n_restarts = 8
    per_restart_budget = (deadline - time.perf_counter()) / n_restarts

    for r in range(n_restarts):
        rd = time.perf_counter() + per_restart_budget
        if time.perf_counter() >= rd:
            break
        scale = 0.05 * (1.0 + 0.5 * r)
        c0 = rng.normal(0, scale, M + 1)
        c0[0] = 0.0  # Keep mean at 0 for now, projection handles it
        try:
            res = minimize(obj, c0, method="Nelder-Mead",
                          options={"maxiter": int(2000 * per_restart_budget / 5.0),
                                   "xatol": 1e-8, "fatol": 1e-9, "adaptive": True})
            if res.x is not None:
                h_cand = _project(cheb_eval(res.x), target)
                c_cand = _evaluate(h_cand)
                if c_cand < best_c:
                    best_c = c_cand
                    best_h = h_cand
        except Exception:
            pass

    if best_h is not None:
        return best_h, best_c
    return None, float("inf")

def _upsample(h, n_target):
    n0 = h.size
    if n0 == n_target:
        return h.copy()
    target = 0.5 * n_target
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    h_interp = np.interp(x1, x0, h, left=h[0], right=h[-1])
    return _project(h_interp, target)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start_time + max(1.0, budget - 10.0)

    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    # Load parent
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        npart = parent.size
        tp = 0.5 * npart
        pp = _project(parent, tp)
        cp = _evaluate(pp)
        if cp < best_c:
            best_h = pp
            best_c = cp
            best_n = npart

    if best_h is None:
        n0 = 90
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
        best_h = _project(h_init, 0.5 * n0)
        best_c = _evaluate(best_h)
        best_n = n0

    base_n = best_n

    # Phase 1: Intensive optimization at base resolution
    n1 = base_n
    t1_dead = start_time + 0.40 * (budget - 10.0)

    starts = []
    if parent is not None:
        starts.append(_upsample(parent, n1))
    starts.append(best_h.copy() if best_h.size == n1 else _upsample(best_h, n1))
    # Perturbed versions
    for s in range(3):
        pert = best_h.copy() + rng.normal(0, 0.05, n1)
        starts.append(_project(pert, 0.5 * n1))
    # Symmetric random
    sym_h = rng.random(n1)
    sym_h = np.maximum(sym_h, sym_h[::-1])
    starts.append(_project(sym_h, 0.5 * n1))

    seen = []
    uniq_starts = []
    for s in starts:
        dup = False
        for u in seen:
            if np.allclose(s, u, atol=1e-6):
                dup = True
                break
        if not dup:
            uniq_starts.append(s)
            seen.append(s)

    for si, s in enumerate(uniq_starts):
        rem = t1_dead - time.perf_counter()
        if rem < 10:
            break
        per = min(rem / max(1, len(uniq_starts) - si), 90.0)
        per = max(8.0, per)
        sub_dl = time.perf_counter() + per

        # Subgradient
        h_a, c_a = _subgrad_method(s, sub_dl, n_active=8, max_iter=2000)
        if c_a < best_c:
            best_h, best_c, best_n = h_a, c_a, n1

        # Pair exchange
        if time.perf_counter() < sub_dl - 3:
            h_b, c_b = _pair_exchange(h_a, sub_dl, n_active=6, max_iter=2000)
            if c_b < best_c:
                best_h, best_c, best_n = h_b, c_b, n1

        # Log-sum-exp SLSQP
        if time.perf_counter() < sub_dl - 5:
            h_c, c_c = _lsq_optimize(h_b if c_b < c_a else h_a, n1, sub_dl)
            if c_c < best_c:
                best_h, best_c, best_n = h_c, c_c, n1

    # Chebyshev global search at n1
    if time.perf_counter() < t1_dead - 5:
        M = min(12, n1 - 1)
        hc, cc = _chebyshev_search(n1, M, t1_dead, rng)
        if hc is not None and cc < best_c:
            best_h, best_c, best_n = hc, cc, n1

    # Phase 2: Upsample and optimize at higher resolution
    n2 = min(n1 * 2, 200)
    if n2 != n1 and time.perf_counter() < deadline - 60:
        t2_dl = start_time + 0.70 * (budget - 10.0)
        h_up = _upsample(best_h, n2)
        # Subgradient at n2
        h2a, c2a = _subgrad_method(h_up, t2_dl, n_active=10, max_iter=1500)
        if c2a < best_c:
            best_h, best_c, best_n = h2a, c2a, n2
        # Pair exchange at n2
        if time.perf_counter() < t2_dl - 5:
            h2b, c2b = _pair_exchange(h_up if c2a >= best_c else h2a, t2_dl, n_active=8, max_iter=1500)
            if c2b < best_c:
                best_h, best_c, best_n = h2b, c2b, n2
        # SLSQP at n2
        if time.perf_counter() < t2_dl - 5:
            h2c, c2c = _lsq_optimize(best_h if best_n == n2 else h_up, n2, t2_dl)
            if c2c < best_c:
                best_h, best_c, best_n = h2c, c2c, n2

    # Phase 3: Even higher resolution if time allows
    n3 = min(n2 * 2, 400)
    if n3 != n2 and time.perf_counter() < deadline - 40:
        t3_dl = start_time + 0.88 * (budget - 10.0)
        h_up3 = _upsample(best_h, n3)
        h3a, c3a = _subgrad_method(h_up3, t3_dl, n_active=10, max_iter=800)
        if c3a < best_c:
            best_h, best_c, best_n = h3a, c3a, n3
        if time.perf_counter() < t3_dl - 5:
            h3b, c3b = _pair_exchange(best_h, t3_dl, n_active=8, max_iter=800)
            if c3b < best_c:
                best_h, best_c, best_n = h3b, c3b, n3

    # Phase 4: Final polish at best resolution
    if time.perf_counter() < deadline - 5:
        h_final, c_final = _subgrad_method(best_h, deadline - 2, n_active=10, max_iter=2000)
        if c_final < best_c:
            best_h, best_c = h_final, c_final
        if time.perf_counter() < deadline - 2:
            h_final2, c_final2 = _pair_exchange(best_h, deadline - 1, n_active=8, max_iter=1000)
            if c_final2 < best_c:
                best_h, best_c = h_final2, c_final2

    # Final projection and evaluation
    final_h = _project(best_h, 0.5 * best_h.size)
    final_c = _evaluate(final_h)
    return final_h, float(final_c), int(final_h.size)
