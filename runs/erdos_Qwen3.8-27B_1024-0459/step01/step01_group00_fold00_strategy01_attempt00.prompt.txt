def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def c5_objective(x):

def sum_constraint(x, n):

def smooth_max_objective(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    return cmax + (1.0 / beta) * np.log(np.sum(np.exp(beta * (corr - cmax))))

def optimize_slsq(n, h_init, max_iter=2000, ftol=1e-15):

    cons = [{'type': 'eq', 'fun': lambda x: sum_constraint(x, n)}]
