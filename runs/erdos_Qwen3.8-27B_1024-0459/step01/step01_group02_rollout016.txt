def compute_c5(h):
    n = len(h)

def compute_c5_vals(h):

    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)

    mask = (h > 1e-10) & (h < 1.0 - 1e-10)
    if np.sum(mask) > 0:
        h[mask] += diff / np.sum(mask)
    h = np.clip(h, 0.0, 1.0)

def project_binary(h, n):

            best_c5 = c5_g2
            best_h = h_g2.copy()
            print("  ExtGreedy: C5={:.6f}".format(best_c5))

        # A few more SA restarts from perturbed best
        for restart in range(3):
            if time.time() > phase2_end - 5:
                break
            h_pert = best_h.copy()
            n_flips = max(2, best_n // 20)
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
            for _ in range(n_flips):
                fl = ones[rng.randint(len(ones))]
                fr = zeros[rng.randint(len(zeros))]
                h_pert[fl] = 0.0
                h_pert[fr] = 1.0
                ones = np.where(h_pert == 1.0)[0]
                zeros = np.where(h_pert == 0.0)[0]

            h_sr, c5_sr = sa_binary(h_pert, best_n, rng, 50000, 0.2, 0.001,
                                    min(15.0, phase2_end - time.time() - 5), t0)
            if c5_sr < best_c5:
                best_c5 = c5_sr
                best_h = h_sr.copy()
                print("  Restart {}: C5={:.6f}".format(restart, best_c5))

    # Phase 3: Continuous SLSQP polish
    if rem() > 20:
        n_p = best_n
        if n_p % 2 != 0:
            n_p = n_p + 1

        h0 = best_h.copy()
        if len(h0) != n_p:
            xp = (np.arange(len(h0)) + 0.5) * (2.0 / len(h0))
            xn = (np.arange(n_p) + 0.5) * (2.0 / n_p)
            h0 = np.interp(xn, xp, h0)
            h0 = project_feasible(h0, n_p)

        bounds = [(0.0, 1.0)] * n_p
        cons = [{'type': 'eq', 'fun': _con_eq}]

        for beta_val in [20, 50, 100, 200]:
            if rem() < 12:
                break
            try:
                res = minimize(lambda x: obj_smooth(x, beta_val), h0, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 500, 'ftol': 1e-14, 'iprint': 0})
                hp = project_feasible(res.x, n_p)
                cp = compute_c5(hp)
                if cp < best_c5:
                    best_c5 = cp
                    best_h = hp.copy()
                    best_n = n_p
                    print("  SLSQP beta={}: C5={:.6f}".format(beta_val, best_c5))
                    h0 = hp.copy()
            except Exception:
                pass

        # Raw max polish
        if rem() > 10:
            try:
                res2 = minimize(lambda x: np.max(np.correlate(x, 1.0 - x, mode='full')) * (2.0 / len(x)),
                                h0, method='SLSQP', bounds=bounds, constraints=cons,
                                options={'maxiter': 300, 'ftol': 1e-14, 'iprint': 0})
                hp2 = project_feasible(res2.x, n_p)
                cp2 = compute_c5(hp2)
                if cp2 < best_c5:
                    best_c5 = cp2
                    best_h = hp2.copy()
                    best_n = n_p
                    print("  SLSQP raw: C5={:.6f}".format(best_c5))
            except Exception:
                pass

    # Phase 4: Final binary rounding
    if rem() > 3:
        hr = project_binary(best_h, best_n)
        cr = compute_c5(hr)
        if cr < best_c5:
            best_c5 = cr
            best_h = hr.copy()
            print("  BinRound: C5={:.6f}".format(best_c5))

    # Ensure valid output
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    print("FINAL: n={}, C5={:.6f}".format(best_n, best_c5))
    return (best_h, best_c5, best_n)
