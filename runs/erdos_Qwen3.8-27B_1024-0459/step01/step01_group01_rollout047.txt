        if time.time() - t0 > budget_s * 0.98:
            break
        noise = rng.normal(0, 0.008, n_use)
        x_pert = fix_sum(np.clip(best_h + noise, 0.0, 1.0), n_use)
        try:
            res = minimize(
                functools.partial(obj_smooth, n=n_use, dx=dx_use, tau=0.0005, bp=0.0),
                x_pert, method='SLSQP', bounds=bounds_use, constraints=cons_use,
                options={'maxiter': 1000, 'ftol': 1e-15, 'disp': False}
            )
            h_new = fix_sum(np.clip(res.x, 0.0, 1.0), n_use)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
