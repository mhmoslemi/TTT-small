```python
import numpy as np
from scipy.optimize import minimize
import functools
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h


def project_feasible(y, n):
    target = n / 2.0
    lo = -y.max() - 2.0
    hi = -y.min() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(y + lam, 0.0, 1.0)


def grad_f_k(h, n, k):
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[k:n] = (1.0 - h[0:n - k])
        g[0:n - k] -= h[k:n]
    else:
        m = -k
        g[0:n - m] = (1.0 - h[m:n])
        g[m:n] -= h[0:n - m]
    return g


def smooth_max_grad(h, n, tau, top_k):
    corr = np.correlate(h, 1.0 - h, mode="full")
    nc = len(corr)
    if top_k < nc:
        top_idx = np.argsort(corr)[-top_k:]
    else:
        top_idx = np.arange(nc)
    m = corr.max()
    sub_w = np.exp((corr[top_idx] - m) / tau)
    sub_w /= sub_w.sum()
    g = np.zeros(n)
    for idx in range(len(top_idx)):
        w = sub_w[idx]
        if w < 1e-3:
            continue
        k = top_idx[idx] - (n - 1)
        g += w * grad_f_k(h, n, k)
    return g, corr


def upsample_to(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def smooth_max_obj(x, n, dx, tau):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mv = np.max(corr)
    return float((mv + tau * np.log(np.sum(np.exp((corr - mv) / tau)))) * dx)


def eq_con(x, n):
    return np.array([x.sum() - n / 2.0])


def eq_jac(x, n):
    return np.ones((1, n))


def binary_local_search(h, n, n_sweeps, rng, time_deadline):
    h = np.round(h)
    h = np.clip(h, 0.0, 1.0)
    h = fix_sum(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for sweep in range(n_sweeps):
        if time.time() > time_deadline:
            break
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_try = min(2000, len(ones) * len(zeros))
        for _ in range(n_try):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            c5_val = compute_c5(h)
            if c5_val < best_c5 - 1e-15:
                best_c5 = c5_val
                best_h = h.copy()
                improved = True
            else:
                h[i], h[j] = h[j], h[i]
        if not improved:
            break
    h = fix_sum(best_h, n)
    return best_h, compute_c5(h)


def subgradient_phase(h, n, tau_list, alpha_start, momentum, time_deadline, top_k=30):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    no_improve = 0
    for tau in tau_list:
        if time.time() > time_deadline:
            break
        alpha = alpha_start
        for it in range(5000):
            if time.time() > time_deadline:
                break
            g, _ = smooth_max_grad(h, n, tau, top_k=top_k)
            v = momentum * v - alpha * g
            h_new = project_feasible(h + v, n)
            c5_val = compute_c5(h_new)
            if c5_val < best_c5 - 1e-15:
                best_c5 = c5_val
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            h = h_new
            alpha = max(alpha * 0.99985, alpha_start * 0.1)
            if no_improve > 1500:
                v = np.zeros(n)
                h = best_h.copy() + np.random.normal(0, 0.03, n)
                h = fix_sum(h, n)
                no_improve = 0
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None and np.size(initial_h_values) > 0:
        parent_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        parent_h = np.full(180, 0.5)

    best_h = parent_h.copy()
    best_c5 = compute_c5(best_h)
    n_parent = len(best_h)

    # Phase 1: Subgradient at parent resolution
    t_deadline1 = t0 + budget_s * 0.25
    taus1 = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
    h1, c5_1 = subgradient_phase(best_h, n_parent, taus1, 0.05, 0.90, t_deadline1, top_k=30)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Upsample and subgradient at 2x resolution
    n2 = 2 * n_parent
    if time.time() < t0 + budget_s * 0.28:
        h2 = upsample_to(best_h, n2)
        h2 = fix_sum(h2, n2)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

        t_deadline2 = t0 + budget_s * 0.52
        taus2 = [0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
        h2, c5_2 = subgradient_phase(h2, n2, taus2, 0.03, 0.90, t_deadline2, top_k=40)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Binary local search at best resolution
    t_deadline3 = t0 + budget_s * 0.60
    n_bs = len(best_h)
    h_bin, c5_bin = binary_local_search(best_h, n_bs, n_sweeps=50, rng=rng, time_deadline=t_deadline3)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 4: SLSQP polish
    t_deadline4 = t0 + budget_s * 0.82
    n_pol = len(best_h)
    dx_pol = 2.0 / n_pol
    bounds_pol = [(0.0, 1.0)] * n_pol

    for tau in [0.005, 0.002, 0.001, 0.0005]:
        if time.time() > t_deadline4:
            break
        obj_p = functools.partial(smooth_max_obj, n=n_pol, dx=dx_pol, tau=tau)
        con_p = functools.partial(eq_con, n=n_pol)
        jac_p = functools.partial(eq_jac, n=n_pol)
        try:
            res = minimize(
                fun=obj_p, x0=best_h.copy(), method='SLSQP',
                bounds=bounds_pol,
                constraints=[{'type': 'eq', 'fun': con_p, 'jac': jac_p}],
                options={'maxiter': 3000, 'ftol': 1e-12, 'disp': False}
            )
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = fix_sum(h_new, n_pol)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    # Phase 5: Multi-start SLSQP with perturbations
    t_deadline5 = t0 + budget_s * 0.93
    for r in range(12):
        if time.time() > t_deadline5:
            break
        noise_scale = 0.04 if r < 6 else 0.015
        noise = rng.normal(0, noise_scale, n_pol)
        x_pert = np.clip(best_h + noise, 0.0, 1.0)
        x_pert = fix_sum(x_pert, n_pol)
        tau = 0.001 if r < 6 else 0.0003
        obj_p = functools.partial(smooth_max_obj, n=n_pol, dx=dx_pol, tau=tau)
        con_p = functools.partial(eq_con, n=n_pol)
        jac_p = functools.partial(eq_jac, n=n_pol)
        try:
            res = minimize(
                fun=obj_p, x0=x_pert, method='SLSQP',
                bounds=bounds_pol,
                constraints=[{'type': 'eq', 'fun': con_p, 'jac': jac_p}],
                options={'maxiter': 1500, 'ftol': 1e-13, 'disp': False}
            )
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = fix_sum(h_new, n_pol)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    # Phase 6: Final subgradient micro-polish
    t_deadline6 = t0 + budget_s * 0.97
    if time.time() < t_deadline6:
        taus_final = [0.001, 0.0005, 0.0003, 0.0002]
        h_final, c5_final = subgradient_phase(best_h, n_pol, taus_final, 0.01, 0.85, t_deadline6, top_k=20)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
```
