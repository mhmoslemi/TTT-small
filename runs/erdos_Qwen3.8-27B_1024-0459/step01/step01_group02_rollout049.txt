    if np.sum(mask) > 0 and abs(diff) > 1e-12:
        h[mask] += diff / np.sum(mask)
    h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n):
    idx = np.argsort(-h)[:n // 2]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def obj_smooth(h):
    dx = 2.0 / _G_n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    shifted = corr - m
    exps = np.exp(_G_beta * shifted)
    return (m + float(np.log(np.sum(exps))) / _G_beta) * dx

def obj_raw(h):
    dx = 2.0 / _G_n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def constraint_sum(h):
    return np.sum(h) - _G_n / 2.0

def get_remaining(start_time, budget_s):
    return max(0.0, budget_s - (time.time() - start_time) - 8.0)

def greedy_swap_search(h, n, rng, max_rounds=50):
    h = h.copy()
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    exhaustive = (n <= 110)

    for rnd in range(max_rounds):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_swap_c5 = cur_c5
        best_i = -1
        best_j = -1
        if exhaustive:
            for i in ones:
                for j in zeros:
                    h[i] = 0.0
                    h[j] = 1.0
                    c5_t = compute_c5(h)
                    h[i] = 1.0
                    h[j] = 0.0
                    if c5_t < best_swap_c5 - 1e-14:
                        best_swap_c5 = c5_t
                        best_i = i
                        best_j = j
        else:
            n_try = min(3000, len(ones) * len(zeros))
            for _ in range(n_try):
                i = int(rng.choice(ones))
                j = int(rng.choice(zeros))
                h[i] = 0.0
                h[j] = 1.0
                c5_t = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                if c5_t < best_swap_c5 - 1e-14:
                    best_swap_c5 = c5_t
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            cur_c5 = best_swap_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            break
    return best_h, best_c5

def sa_swap_search(h, n, rng, max_iters=5000):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    T = 0.0004
    T_min = 1e-9
    cool = 0.9995

    for it in range(max_iters):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, T_min)):
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T * cool, T_min)
    return best_h, best_c5

def sa_pair_swap(h, n, rng, max_iters=2000):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    T = 0.0003
    T_min = 1e-9
    cool = 0.9995

    for it in range(max_iters):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i1, i2 = rng.choice(ones, size=2, replace=False)
        j1, j2 = rng.choice(zeros, size=2, replace=False)
        h[i1] = 0.0
        h[i2] = 0.0
        h[j1] = 1.0
        h[j2] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, T_min)):
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i1] = 1.0
            h[i2] = 1.0
            h[j1] = 0.0
            h[j2] = 0.0
        T = max(T * cool, T_min)
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _G_n, _G_beta
    start_time = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    parent_h = None
    parent_n = 0
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        if parent_n > 0:
            parent_c5 = compute_c5(parent_h)
            best_h = parent_h.copy()
            best_c5 = parent_c5
            best_n = parent_n
            print("Parent: n={}, C5={:.6f}".format(parent_n, parent_c5))
    except Exception:
        pass

    if best_h is None:
        best_n = 96
        best_h = make_binary(rng.rand(best_n), best_n)
        best_c5 = compute_c5(best_h)
        print("Random init: n={}, C5={:.6f}".format(best_n, best_c5))

    n_list = sorted(set([96, 100, 128, 160, 192, 256]))
    n_list = [nn for nn in n_list if 64 <= nn <= 256 and nn % 2 == 0]

    for n in n_list:
        rem = get_remaining(start_time, budget_s)
        if rem < 25:
            break

        _G_n = n
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': constraint_sum}]

        inits = []
        if parent_h is not None and parent_n > 0:
            xp = (np.arange(parent_n) + 0.5) * (2.0 / parent_n)
            x_new = (np.arange(n) + 0.5) * (2.0 / n)
            interp = np.interp(x_new, xp, parent_h)
            interp = project_feasible(interp, n)
            inits.append(interp)
            inits.append(make_binary(interp, n))

        for _ in range(2):
            inits.append(make_binary(rng.rand(n), n))

        h_alt = np.zeros(n)
        h_alt[0::2] = 1.0
        inits.append(h_alt)

        for h0 in inits:
            rem = get_remaining(start_time, budget_s)
            if rem < 10:
                break

            h0 = project_feasible(h0, n)
            h_bin = make_binary(h0, n)

            # Greedy swap
            h_g, c5_g = greedy_swap_search(h_bin, n, rng, max_rounds=40)
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g.copy()
                best_n = n
                print("  Greedy: n={}, C5={:.6f}".format(n, best_c5))

            # SA single swap
            rem = get_remaining(start_time, budget_s)
            if rem > 10:
                sa_it = min(5000, int(rem * 300))
                h_sa, c5_sa = sa_swap_search(h_g, n, rng, max_iters=sa_it)
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
                    best_n = n
                    print("  SA1: n={}, C5={:.6f}".format(n, best_c5))

            # SA pair swap
            rem = get_remaining(start_time, budget_s)
            if rem > 10:
                h_sp, c5_sp = sa_pair_swap(h_g, n, rng, max_iters=2000)
                if c5_sp < best_c5:
                    best_c5 = c5_sp
                    best_h = h_sp.copy()
                    best_n = n
                    print("  SA2: n={}, C5={:.6f}".format(n, best_c5))

            # SLSQP continuous
            rem = get_remaining(start_time, budget_s)
            if rem > 6:
                h_c = h_g.copy()
                for bv in [25.0, 75.0]:
                    _G_beta = bv
                    try:
                        r = minimize(obj_smooth, h_c, method='SLSQP',
                                     bounds=bounds, constraints=cons,
                                     options={'maxiter': 200, 'ftol': 1e-13, 'iprint': 0})
                        h_c = project_feasible(r.x, n)
                        c5_c = compute_c5(h_c)
                        if c5_c < best_c5:
                            best_c5 = c5_c
                            best_h = h_c.copy()
                            best_n = n
                            print("  SLSQP b={}: n={}, C5={:.6f}".format(int(bv), n, best_c5))
                    except Exception:
                        pass

                # Round + greedy
                h_rb = make_binary(h_c, n)
                h_g2, c5_g2 = greedy_swap_search(h_rb, n, rng, max_rounds=30)
                if c5_g2 < best_c5:
                    best_c5 = c5_g2
                    best_h = h_g2.copy()
                    best_n = n
                    print("  R+G: n={}, C5={:.6f}".format(n, best_c5))

        # Polish at best n
        if best_n == n:
            rem = get_remaining(start_time, budget_s)
            if rem > 15:
                h_p = best_h.copy()
                for bv in [40.0, 150.0, 400.0]:
                    rem = get_remaining(start_time, budget_s)
                    if rem < 5:
                        break
                    _G_beta = bv
                    try:
                        r = minimize(obj_smooth, h_p, method='SLSQP',
                                     bounds=bounds, constraints=cons,
                                     options={'maxiter': 400, 'ftol': 1e-14, 'iprint': 0})
                        h_p = project_feasible(r.x, n)
                        c5_p = compute_c5(h_p)
                        if c5_p < best_c5:
                            best_c5 = c5_p
                            best_h = h_p.copy()
                            print("  Pol b={}: C5={:.6f}".format(int(bv), best_c5))
                    except Exception:
                        pass
                rem = get_remaining(start_time, budget_s)
                if rem > 5:
                    h_bp = make_binary(h_p, n)
                    h_pp, c5_pp = greedy_swap_search(h_bp, n, rng, max_rounds=50)
                    if c5_pp < best_c5:
                        best_c5 = c5_pp
                        best_h = h_pp.copy()
                        print("  Pol G: C5={:.6f}".format(best_c5))

    # Random restarts at best n
    rem = get_remaining(start_time, budget_s)
    if rem > 25 and best_n > 0:
        _G_n = best_n
        bounds = [(0.0, 1.0)] * best_n
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        for ri in range(4):
            rem = get_remaining(start_time, budget_s)
            if rem < 10:
                break
            h_r = make_binary(rng.rand(best_n), best_n)
            hr_g, c5_rg = greedy_swap_search(h_r, best_n, rng, max_rounds=40)
            if c5_rg < best_c5:
                best_c5 = c5_rg
                best_h = hr_g.copy()
                print("  Rest{}: C5={:.6f}".format(ri, best_c5))
            rem = get_remaining(start_time, budget_s)
            if rem > 8:
                h_sr, c5_sr = sa_swap_search(hr_g, best_n, rng, max_iters=3000)
                if c5_sr < best_c5:
                    best_c5 = c5_sr
                    best_h = h_sr.copy()
                    print("  Rest{} SA: C5={:.6f}".format(ri, best_c5))

    # Final: perturb + re-optimize at best
    rem = get_remaining(start_time, budget_s)
    if rem > 10 and best_n > 0:
        _G_n = best_n
        bounds = [(0.0, 1.0)] * best_n
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        h_nb = best_h.copy()
        flip_mask = rng.rand(best_n) < 0.1
        h_nb[flip_mask] = 1.0 - h_nb[flip_mask]
        h_nb = project_feasible(h_nb, best_n)
        h_nb_bin = make_binary(h_nb, best_n)
        h_NB, c5_NB = greedy_swap_search(h_nb_bin, best_n, rng, max_rounds=40)
        if c5_NB < best_c5:
            best_c5 = c5_NB
            best_h = h_NB.copy()
            print("  Perturb: C5={:.6f}".format(best_c5))

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
