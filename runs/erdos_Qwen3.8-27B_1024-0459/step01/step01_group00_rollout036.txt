def slsqp_optimize(h_init, n, max_iter=3000, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    
    def constraint_fun(x):
        return np.sum(x) - n / 2.0
    
    cons = [{'type': 'eq', 'fun': constraint_fun}]
    
    try:
        res = minimize(obj, h_init, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': ftol})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

    x = np.linspace(0, 2, n, endpoint=False)
    for freq in [1.0, 1.5, 2.0, 2.5, 3.0]:
        h = 0.5 + 0.4 * np.sin(np.pi * freq * x)
        inits.append(make_feasible(h, n))
    
    # Random
    for _ in range(3):
        h = rng.rand(n)
        inits.append(make_feasible(h, n))
    
    # Greedy balanced
    h4 = np.zeros(n)
    positions = sorted(range(n), key=lambda i: (i * 7919) % n)
    for j in range(n // 2):
        h4[positions[j]] = 1.0
    inits.append(make_feasible(h4, n))
    
    return inits


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    # Get parent
    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is None or len(h_parent) == 0:
            h_parent = None
    except Exception:
        h_parent = None
    
    if h_parent is not None:
        best_h = h_parent.copy()
        best_n = len(best_h)
    else:
        best_n = 160
        best_h = thue_morse_init(best_n)
    
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Subgradient optimization at parent resolution
    if remaining() > 200:
        h_opt, c5_opt = subgradient_optimize(best_h, best_n, n_iter=8000,
                                             beta=12.0, top_frac=0.1,
                                             step_init=0.12, step_decay=0.9998)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 2: SLSQP polish at parent resolution
    if remaining() > 150:
        h_opt, c5_opt = slsqp_optimize(best_h, best_n, max_iter=3000, ftol=1e-15)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 3: Smooth SLSQP at parent resolution
    if remaining() > 100:
        h_opt, c5_opt = smooth_slsqp(best_h, best_n, beta=40.0, max_iter=2000)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 4: Try higher resolutions
    higher_resolutions = []
    for nr in [200, 240, 280, 320]:
        if nr > best_n:
            higher_resolutions.append(nr)
    
    for nr in higher_resolutions:
        if remaining() < 120:
            break
        
        # Upsample
        h_interp = interpolate_to_n(best_h, nr)
        h_interp = make_feasible(h_interp, nr)
        
        # Subgradient at new resolution
        h_opt, c5_opt = subgradient_optimize(h_interp, nr, n_iter=5000,
                                             beta=15.0, top_frac=0.08,
                                             step_init=0.1, step_decay=0.9997)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = nr
        
        # SLSQP polish
        if remaining() > 60:
            h_opt2, c5_opt2 = slsqp_optimize(best_h, best_n, max_iter=2000, ftol=1e-14)
            if h_opt2 is not None and c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
    
    # Phase 5: Try a few different initializations at best_n
    if remaining() > 200:
        alts = structured_inits(best_n, rng)
        if h_parent is not None and len(h_parent) != best_n:
            h_p = make_feasible(interpolate_to_n(h_parent, best_n), best_n)
            alts.append(h_p)
        
        for h0 in alts[:4]:
            if remaining() < 50:
                break
            
            # Quick subgradient
            h_opt, c5_opt = subgradient_optimize(h0, best_n, n_iter=3000,
                                                 beta=10.0, top_frac=0.1,
                                                 step_init=0.15, step_decay=0.9999)
            if c5_opt < best_c5 - 0.005:
                # Promising, do SLSQP
                h_opt2, c5_opt2 = slsqp_optimize(h_opt, best_n, max_iter=2000)
                if h_opt2 is not None and c5_opt2 < best_c5:
                    best_c5 = c5_opt2
                    best_h = h_opt2.copy()
    
    # Phase 6: Intensive local search
    if remaining() > 100:
        n_trials = min(8000, int((remaining() - 20) / 0.002))
        best_h, best_c5 = local_perturbation(best_h, best_n,
                                             n_trials=n_trials, step_size=0.025)
    
    # Phase 7: Final SLSQP
    if remaining() > 30:
        h_opt, c5_opt = slsqp_optimize(best_h, best_n, max_iter=5000, ftol=1e-16)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 8: Final local perturbation
    if remaining() > 30:
        n_trials = min(5000, int((remaining() - 10) / 0.002))
        best_h, best_c5 = local_perturbation(best_h, best_n,
                                             n_trials=n_trials, step_size=0.015)
    
    # Final feasibility
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
