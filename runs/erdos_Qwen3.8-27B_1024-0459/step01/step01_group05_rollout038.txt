For the 1D optimization (finding optimal δ for a given delta vector):

```python
def find_optimal_step(corr, dvec, delta_lo, delta_hi):
    """Find δ that minimizes max_k(corr[k] + dvec[k]*δ) over [delta_lo, delta_hi]."""
    # Convex piecewise linear function, use binary search on subgradient
    lo, hi = delta_lo, delta_hi
    if lo >= hi:
        return lo
    best_val = np.inf
    best_d = lo
    for _ in range(60):
        mid = (lo + hi) / 2
        vals = corr + dvec * mid
        max_val = np.max(vals)
        if max_val < best_val:
            best_val = max_val
            best_d = mid
        active = vals >= max_val - 1e-12
        min_slope = np.min(dvec[active])
        max_slope = np.max(dvec[active])
        if min_slope > 1e-14:
            hi = mid  # f increasing, go left
        elif max_slope < -1e-14:
            lo = mid  # f decreasing, go right
        else:
            break  # 0 in subgradient
    return best_d
```

- h[m] - δ ∈ [0, 1] → δ ∈ [h[m]-1, h[m]]

So: δ ∈ [max(-h[j], h[m]-1), min(1-h[j], h[m])]

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_delta(h, j, N):
    """Compute change in corr[i] per unit change in h[j]."""
    delta = np.zeros(2*N - 1)
    # First part: h[j] in h[n+lag], i in [j, j+N-1]
    for i in range(j, min(2*N-1, j+N)):
        n_idx = j - i + N - 1
        delta[i] += (1.0 - h[n_idx])
    # Second part: h[j] in (1-h[n]) where n=j, i in [max(0,N-1-j), 2N-j-2]
    lo = max(0, N - 1 - j)
    hi = 2*N - j - 2
    for i in range(lo, min(2*N-1, hi+1)):
        n_plus_lag = j + i - N + 1
        delta[i] -= h[n_plus_lag]
    return delta

def find_optimal_step(corr, dvec, delta_lo, delta_hi):
    lo, hi = delta_lo, delta_hi
    if lo >= hi:
        return lo
    best_val = np.inf
    best_d = lo
    for _ in range(50):
        mid = (lo + hi) / 2
        vals = corr + dvec * mid
        max_val = np.max(vals)
        if max_val < best_val:
            best_val = max_val
            best_d = mid
        active = vals >= max_val - 1e-10
        min_slope = dvec[active].min()
        max_slope = dvec[active].max()
        if min_slope > 1e-12:
            hi = mid
        elif max_slope < -1e-12:
            lo = mid
        else:
            best_val = max_val
            best_d = mid
            break
    return best_d

def coordinate_descent_pairwise(h, max_sweeps=1000, time_limit=None, t0=None):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * (2.0/N)
    
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        improved = False
        for j in range(N):
            delta_j = compute_delta(h, j, N)
            for m in range(j+1, N):
                if h[j] <= 0 and h[m] <= 0:
                    continue
                if h[j] >= 1 and h[m] >= 1:
                    continue
                dvec = delta_j - compute_delta(h, m, N)
                d_lo = max(-h[j], h[m]-1)
                d_hi = min(1-h[j], h[m])
                if d_lo >= d_hi:
                    continue
                opt_d = find_optimal_step(corr, dvec, d_lo, d_hi)
                new_max = np.max(corr + dvec * opt_d)
                if new_max < np.max(corr) - 1e-14:
                    h[j] += opt_d
                    h[m] -= opt_d
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    improved = True
                    # Recompute delta_j since h changed
                    delta_j = compute_delta(h, j, N)
        if not improved:
            break
    
    final_c5 = compute_c5(h)
    return h, final_c5
```

So yes, C₅ = max(corr) * dx where corr = np.correlate(h, 1-h, mode='full').

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_delta(h, j, N):
    delta = np.zeros(2 * N - 1)
    i_lo = j
    i_hi = min(2 * N - 1, j + N)
    if i_lo < i_hi:
        i_idx = np.arange(i_lo, i_hi)
        n_idx = j - i_idx + N - 1
        delta[i_idx] += (1.0 - h[n_idx])
    i_lo2 = max(0, N - 1 - j)
    i_hi2 = min(2 * N - 1, 2 * N - j - 1)
    if i_lo2 < i_hi2:
        i_idx2 = np.arange(i_lo2, i_hi2)
        npl_idx = j + i_idx2 - N + 1
        delta[i_idx2] -= h[npl_idx]
    return delta

def precompute_all_deltas(h):
    N = len(h)
    deltas = np.zeros((N, 2 * N - 1))
    for j in range(N):
        deltas[j] = compute_delta(h, j, N)
    return deltas

def find_optimal_step(corr, dvec, d_lo, d_hi):
    if d_lo >= d_hi:
        return d_lo
    lo, hi = d_lo, d_hi
    best_val = np.inf
    best_d = d_lo
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = corr + dvec * mid
        mv = np.max(vals)
        if mv < best_val:
            best_val = mv
            best_d = mid
        active = vals >= mv - 1e-10
        if np.min(dvec[active]) > 1e-14:
            hi = mid
        elif np.max(dvec[active]) < -1e-14:
            lo = mid
        else:
            best_d = mid
            break
    return best_d

def pairwise_cd(h, max_time, max_sweeps=5000):
    N = len(h)
    t_start = time.time()
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = np.max(corr)
    
    for sweep in range(max_sweeps):
        if time.time() - t_start > max_time:
            break
        deltas = precompute_all_deltas(h)
        any_update = False
        
        for j in range(N):
            if time.time() - t_start > max_time:
                return h
            for m in range(j + 1, N):
                d_lo = max(-h[j], h[m] - 1.0)
                d_hi = min(1.0 - h[j], h[m])
                if d_lo >= d_hi - 1e-14:
                    continue
                dvec = deltas[j] - deltas[m]
                opt_d = find_optimal_step(corr, dvec, d_lo, d_hi)
                if abs(opt_d) < 1e-13:
                    continue
                new_max = np.max(corr + dvec * opt_d)
                if new_max < current_max - 1e-13:
                    h[j] += opt_d
                    h[m] -= opt_d
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    current_max = np.max(corr)
                    deltas = precompute_all_deltas(h)
                    any_update = True
        
        if not any_update:
            break
    
    return h

def project_sum(h, target):
    h = h.copy().astype(float)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    s = np.sum(result)
    diff = target - s
    if abs(diff) > 1e-14:
        free = (result > 1e-10) & (result < 1.0 - 1e-10)
        if np.sum(free) > 0:
            result[free] += diff / np.sum(free)
            result = np.clip(result, 0.0, 1.0)
    return result

def lgbfgs_optimize(h0, beta, penalty, maxiter=500, ftol=1e-14):
    """L-BFGS-B optimization with penalty for sum constraint."""
    N = len(h0)
    target = N / 2.0
    
    def fun_grad(h):
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        exp_vals = np.exp(beta * (corr - max_corr))
        sum_exp = np.sum(exp_vals)
        p = exp_vals / sum_exp
        
        # Smooth max value (scaled by 2/N)
        f_smooth = (2.0 / N) * (max_corr + np.log(sum_exp) / beta)
        
        # Gradient of smooth max
        deltas = precompute_all_deltas(h)
        grad_smooth = (2.0 / N) * (deltas.T @ p)
        
        # Penalty
        s = np.sum(h)
        diff = s - target
        f_pen = penalty * diff * diff
        grad_pen = 2.0 * penalty * diff
        
        return f_smooth + f_pen, grad_smooth + grad_pen
    
    result = minimize(fun_grad, h0, jac=True, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * N,
                      options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
    
    if result is not None and result.x is not None:
        return result.x
    return h0
```
