def _grad_loop(h, idx, weights):

def exact_c5(h):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return float("inf")
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def fix_sum(h):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    if n == 0:
        return h
    target = 0.5 * n
    for _ in range(250):
        h = np.clip(h, 0.0, 1.0)
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    h = np.clip(h, 0.0, 1.0)
    return h


def symmetrize(h):
    h = np.asarray(h, dtype=float)
    hs = 0.5 * (h + 1.0 - h[::-1])
    return fix_sum(hs)


def upsample(h, n_new):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n_old = len(h)
    if n_old == n_new:
        return fix_sum(h.copy())
    if n_old == 1:
        h_new = np.full(n_new, float(h[0]), dtype=float)
        return fix_sum(h_new)
    x_old = (np.arange(n_old, dtype=float) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new, dtype=float) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h, left=float(h[0]), right=float(h[-1]))
    return fix_sum(h_new)


def ramp_h(alpha, n):
    alpha = float(alpha)
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
    if alpha <= 0.0:
        h = 0.5 * x

        h = (x - alpha) / (2.0 * q)

def tile_no_fix(a, n):
    p = len(a)
    reps = n // p
    if reps <= 0:
        return np.asarray(a, dtype=float).copy()

    if len(h) < n:

    return h

def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _corr_fast(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if not _G_FFT or n <= 260:
        return np.correlate(h, 1.0 - h, mode="full")
    L = 2 * n - 1
    N = _next_pow2(L)
    b = (1.0 - h)[::-1]
    fa = np.fft.rfft(h, N)
    fb = np.fft.rfft(b, N)
    out = np.fft.irfft(fa * fb, N)
    return out[:L]


def _conv_term(w, h, reverse_b):
    n = _G_N
    out_len = 3 * n - 2
    if not _G_FFT or n <= 450:
        b = h[::-1] if reverse_b else (1.0 - h)
        full = np.convolve(w, b, mode="full")
        return full[n - 1:2 * n - 1]
    N = _next_pow2(out_len)
    fw = np.fft.rfft(w, N)
    if reverse_b:
        fb = np.fft.rfft(h[::-1], N)
    else:
        fb = np.fft.rfft(1.0 - h, N)
    return np.fft.irfft(fw * fb, N)[:out_len][n - 1:2 * n - 1]


def _grad_loop(h, idx, weights):
    n = _G_N
    center = n - 1
    g = np.zeros(n, dtype=float)
    for ii in range(len(idx)):
        w = float(weights[ii])
        if w <= 0.0:
            continue
        p = int(idx[ii])
        s = p - center
        if s >= 0:
            L = n - s
            if L <= 0:
                continue
            g[:L] -= w * h[s:s + L]
            g[s:s + L] += w * (1.0 - h[:L])
        else:
            t = -s
            L = n - t
            if L <= 0:
                continue
            g[:L] += w * (1.0 - h[t:t + L])
            g[t:t + L] -= w * h[:L]
    return _G_DX * g


def _softmax(vals, tau):
    m = float(np.max(vals))
    e = np.exp((vals - m) / tau)
    s = float(e.sum())
    if s <= 0.0:
        w = np.zeros_like(vals, dtype=float)
        w[int(np.argmax(vals))] = 1.0
        return w
    return e / s


def _smooth_obj(x):
    h = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    c = _corr_fast(h)
    if _G_LAGS is None:
        vals = c
    else:
        vals = c[_G_LAGS]
    m = float(np.max(vals))
    e = np.exp((vals - m) / _G_TAU)
    s = float(e.sum())
    if s <= 0.0:
        return float(_G_DX * m)
    return float(_G_DX * (m + _G_TAU * np.log(s)))


def _smooth_grad(x):
    h = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    c = _corr_fast(h)
    n = _G_N
    if _G_LAGS is None:
        vals = c
        w = _softmax(vals, _G_TAU)
        sig = np.where(w > 1e-10)[0]
        if len(sig) <= max(24, n // 4):
            g = _grad_loop(h, sig, w[sig])

    return g

def sym_from_y(y):

    h[k] = 0.5

    return h

        x, c_x = update_if_better(xs, x, exact_c5(x))
        best_h, best_c = update_if_better(x, best_h, best_c)

        base_best_x = x.copy()
        t180s = [0.12, 0.06, 0.03, 0.016, 0.008]
        mit180s = [420, 320, 260, 200, 160]
        for tau, mit in zip(t180s, mit180s):
            if time.time() >= base_deadline - 20:
                break
            x2 = run_slsqp_stage(x, n_base, 2.0 / n_base, tau, None, mit)
            x2 = fix_sum(np.clip(x2, 0.0, 1.0))
            x3 = symmetrize(x2)
            x2, c2 = update_if_better(x3, x2, exact_c5(x2))
            best_h, best_c = update_if_better(x2, best_h, best_c)
            if c2 < exact_c5(base_best_x) + 0.01:
                base_best_x = x2

            if tau <= 0.02 and time.time() < base_deadline - 20:
                lags = select_active_lags(base_best_x, 32)
                x4 = run_slsqp_stage(base_best_x, n_base, 2.0 / n_base, 0.0035, lags, 180)
                x4 = fix_sum(np.clip(x4, 0.0, 1.0))
                x5 = symmetrize(x4)
                x4, c4 = update_if_better(x5, x4, exact_c5(x4))
                best_h, best_c = update_if_better(x4, best_h, best_c)
                if c4 < exact_c5(base_best_x) + 0.008:
                    base_best_x = x4

        for r in range(3):
            if time.time() >= base_deadline:
                break
            noise = rng.normal(0.0, 0.035, n_base)
            xp = np.clip(base_best_x + noise, 0.0, 1.0)
            xp = fix_sum(xp)
            x2 = run_slsqp_stage(xp, n_base, 2.0 / n_base, 0.02, None, 120)
            x2 = fix_sum(np.clip(x2, 0.0, 1.0))
            x3 = symmetrize(x2)
            x2, c2 = update_if_better(x3, x2, exact_c5(x2))
            best_h, best_c = update_if_better(x2, best_h, best_c)
            if c2 < exact_c5(base_best_x) + 0.008:
                base_best_x = x2

    if time.time() < full_deadline:
        n_med = 360
        x = upsample(best_h, n_med)
        xs = symmetrize(x)
        x, c_x = update_if_better(xs, x, exact_c5(x))
        best_h, best_c = update_if_better(x, best_h, best_c)

        base_best_x = x.copy()
        tmeds = [0.08, 0.04, 0.02, 0.01]
        mitmeds = [220, 180, 140, 100]
        for tau, mit in zip(tmeds, mitmeds):
            if time.time() >= full_deadline - 30:
                break
            x2 = run_slsqp_stage(x, n_med, 2.0 / n_med, tau, None, mit)
            x2 = fix_sum(np.clip(x2, 0.0, 1.0))
            x3 = symmetrize(x2)
            x2, c2 = update_if_better(x3, x2, exact_c5(x2))

            if c2 < exact_c5(base_best_x) + 0.008:

            if tau <= 0.02 and time.time() < full_deadline - 30:
                lags = select_active_lags(base_best_x, 36)
                x4 = run_slsqp_stage(base_best_x, n_med, 2.0 / n_med, 0.0035, lags, 130)
                x4 = fix_sum(np.clip(x4, 0.0, 1.0))
                x5 = symmetrize(x4)
                x4, c4 = update_if_better(x5, x4, exact_c5(x4))
                best_h, best_c = update_if_better(x4, best_h, best_c)
                if c4 < exact_c5(base_best_x) + 0.006:
                    base_best_x = x4

        for r in range(2):
            if time.time() >= full_deadline:
                break
            noise = rng.normal(0.0, 0.03, n_med)
            xp = np.clip(base_best_x + noise, 0.0, 1.0)
            xp = fix_sum(xp)
            x2 = run_slsqp_stage(xp, n_med, 2.0 / n_med, 0.018, None, 110)
            x2 = fix_sum(np.clip(x2, 0.0, 1.0))
            x3 = symmetrize(x2)
            x2, c2 = update_if_better(x3, x2, exact_c5(x2))
            best_h, best_c = update_if_better(x2, best_h, best_c)
            if c2 < exact_c5(base_best_x) + 0.007:
                base_best_x = x2

    final_deadline = t0 + budget_s * 0.97
    for n_sym in [181, 361]:
        if time.time() >= final_deadline:
            break
        k = (n_sym - 1) // 2
        h0 = upsample(best_h, n_sym)
        h0 = symmetrize(h0)
        h0[k] = 0.5
        h0 = fix_sum(h0)
        y = h0[:k].copy() - 0.5
        y = np.clip(y, -0.5, 0.5)

        y_best = y.copy()
        h_best = sym_from_y(y_best)
        h_best, c_best = update_if_better(h_best, h_best, exact_c5(h_best))

                if c4 < exact_c5(sym_from_y(y_best)) + 0.006:
                    y_best = y4
                    h_best = h4

    if time.time() < final_deadline:
        n_best = len(best_h)
        for r in range(14):
            if time.time() >= final_deadline:
                break
            sigma = 0.008 if n_best <= 220 else 0.004
            noise = rng.normal(0.0, sigma, n_best)
            hp = np.clip(best_h + noise, 0.0, 1.0)
            hp = fix_sum(hp)
            best_h, best_c = update_if_better(hp, best_h, best_c)

            if time.time() >= final_deadline:
                break
            lags = select_active_lags(best_h, 24)
            h2 = run_slsqp_stage(best_h, n_best, 2.0 / n_best, 0.002, lags, 70)
            h2 = fix_sum(np.clip(h2, 0.0, 1.0))
            h3 = symmetrize(h2)
            best_h, best_c = update_if_better(h3, best_h, best_c)

    best_h = fix_sum(np.clip(best_h, 0.0, 1.0))
    best_c = exact_c5(best_h)
    return best_h, float(best_c), int(len(best_h))
