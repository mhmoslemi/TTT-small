def proj(x, target):

    lo, hi = -1.0, 1.0

        if s < target:

    res = target - float(np.sum(h))

def eval_c(h):

def upsample(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    n = int(n)

        return p.copy()
    if n0 <= 0:

    h = np.interp(x1, x0, p, left=p[0], right=p[-1])
    return proj(h, 0.5 * n)


def make_mseq(n, seed_val=7):
    state = seed_val + 1
    if state == 0:
        state = 1
    bits = []
    for _ in range(max(n, 256)):
        bits.append(state & 1)
        state ^= (state << 13) & 0xFFFFFFFF
        state ^= (state >> 17)
        state ^= (state << 5) & 0xFFFFFFFF
        state &= 0xFFFFFFFF
    b = np.array(bits[:n], dtype=float)
    return b

def soft_obj(x, n, alpha):
    corr = c_full(x)
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0 or not np.isfinite(z):
        return 1e100
    return (2.0 / n) * (m + np.log(z) / alpha)


def soft_jac(x, n, alpha):
    corr = c_full(x)
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0 or not np.isfinite(z):
        return np.zeros(n)
    w = e / z
    idx = np.where(w > 1e-10)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(w))])
    g = np.zeros(n)
    center = n - 1
    for t in idx:
        t = int(t)
        p = float(w[t])
        k = t - center
        if k == 0:
            g += p * (1.0 - 2.0 * x)
        else:
            lag = abs(k)
            if k > 0:
                g[lag:] += p
                g[lag:] -= p * x[:n - lag]
                g[:n - lag] -= p * x[lag:]
            else:
                g[:n - lag] += p
                g[:n - lag] -= p * x[lag:]
                g[lag:] -= p * x[:n - lag]
    return (2.0 / n) * g


def slsqp_solve(h, n, deadline):
    target = 0.5 * n
    h = proj(h, target)
    best_h = h.copy()
    best_c = eval_c(best_h)
    alphas = (15.0, 40.0, 100.0)
    bounds = [(0.0, 1.0)] * n
    if n <= 360:
        maxiter = 200
    elif n <= 720:
        maxiter = 120
    else:
        maxiter = 60
    for alpha in alphas:
        if time.perf_counter() >= deadline - 2:
            break
        x0 = best_h.copy()

        def obj_fn(x):
            return soft_obj(x, n, alpha)

        def jac_fn(x):
            return soft_jac(x, n, alpha)

        def cons_fn(x):
            return float(np.sum(x) - target)

        def cons_jac(x):
            return np.ones(n)

        try:
            res = minimize(
                obj_fn, x0, method='SLSQP', jac=jac_fn,
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': cons_fn, 'jac': cons_jac}],
                options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False}
            )
            xg = np.asarray(res.x, dtype=float) if res.x is not None else None
            if xg is not None and np.all(np.isfinite(xg)):
                cand = proj(xg, target)
                c = eval_c(cand)
                if c < best_c:
                    best_h = cand
                    best_c = c
                else:
                    if c < best_c + 0.05:
                        best_h = cand
        except Exception:
            pass
        if time.perf_counter() >= deadline - 2:
            break
    return best_h, best_c


def refine_coarse_to_fine(coarse_h, fine_n, deadline):
    target = 0.5 * fine_n
    h = upsample(coarse_h, fine_n)
    h = proj(h, target)
    c = eval_c(h)
    h2, c2 = active_set_search(h, deadline, max_iter=500, K=5)
    if c2 < c:
        h, c = h2, c2
    if time.perf_counter() < deadline - 5:
        h3, c3 = pair_blocks_search(h, deadline, max_iter=300, K=5)
        if c3 < c:
            h, c = h3, c3
    return h, c


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    t_end = t_start + max(1.0, budget - 8.0)

    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float('inf')

    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        np_ = parent.size
        h_p = proj(parent, 0.5 * np_)
        c_p = eval_c(h_p)
        if c_p < best_c:
            best_h = h_p
            best_c = c_p
    else:
        h_p = make_structured(90, rng, 0)
        best_h = h_p
        best_c = eval_c(h_p)

    phase_deadline = t_start + 25.0
    if best_c > 0.39:
        tries = [0, 1, 2, 3, 0, 1]
        for tm in tries:
            if time.perf_counter() >= phase_deadline:
                break
            h_try = make_structured(90, rng, tm)
            c_try = eval_c(h_try)
            if c_try < best_c:
                best_h = h_try
                best_c = c_try

    h90, c90 = active_set_search(best_h, phase_deadline, max_iter=500, K=5)
    if c90 < best_c:
        best_h, best_c = h90, c90

    n_list = [180, 360, 720]
    time_alloc = {}
    remaining = t_end - time.perf_counter()
    if remaining < 30:
        n_list = []
    else:
        for i, ni in enumerate(n_list):
            frac = [0.2, 0.35, 0.45]
            time_alloc[ni] = max(10.0, remaining * frac[i]) if i < len(frac) else 10.0

    for ni in n_list:
        if time.perf_counter() >= t_end - 15:
            break
        sub_dl = time.perf_counter() + time_alloc.get(ni, 30.0)
        sub_dl = min(sub_dl, t_end - 10.0)

        h_start = upsample(best_h, ni)
        h_start = proj(h_start, 0.5 * ni)
        c_start = eval_c(h_start)

        h_opt, c_opt = active_set_search(h_start, sub_dl, max_iter=800, K=6)
        if c_opt < c_start:
            h_start, c_start = h_opt, c_opt

        if sub_dl - time.perf_counter() > 10:
            h_pb, c_pb = pair_blocks_search(h_start, sub_dl, max_iter=400, K=6)
            if c_pb < c_start:
                h_start, c_start = h_pb, c_pb

        if c_start < best_c:
            best_h, best_c = h_start, c_start

    if time.perf_counter() < t_end - 20 and best_h is not None:
        bn = best_h.size
        if bn <= 360:
            h_up = upsample(best_h, min(720, bn * 2))
            h_up = proj(h_up, 0.5 * h_up.size)
            sub_dl = t_end - 15
            h_fine, c_fine = slsqp_solve(h_up, h_up.size, sub_dl)
            if c_fine < best_c:
                best_h, best_c = h_fine, c_fine
                bn = best_h.size
        if time.perf_counter() < t_end - 10:
            h_f2, c_f2 = active_set_search(best_h, t_end - 5, max_iter=1000, K=8)
            if c_f2 < best_c:
                best_h, best_c = h_f2, c_f2
        if time.perf_counter() < t_end - 5:
            h_f3, c_f3 = pair_blocks_search(best_h, t_end - 2, max_iter=500, K=8)
            if c_f3 < best_c:
                best_h, best_c = h_f3, c_f3

    if time.perf_counter() < t_end - 3 and best_h is not None:
        bn = best_h.size
        if bn < 720 and bn >= 2:
            h_up2 = upsample(best_h, min(1440, bn * 2))
            h_up2 = proj(h_up2, 0.5 * h_up2.size)
            c_up2 = eval_c(h_up2)
            if c_up2 < best_c:
                h_f4, c_f4 = active_set_search(h_up2, t_end - 1, max_iter=200, K=6)
                if c_f4 < best_c:
                    best_h, best_c = h_f4, c_f4

    if best_h is None:
        fallback = proj(np.full(90, 0.5), 45.0)
        return fallback, eval_c(fallback), 90

    final_h = proj(best_h, 0.5 * best_h.size)
    final_c = eval_c(final_h)
    return final_h, float(final_c), int(final_h.size)
