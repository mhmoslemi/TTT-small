        g = subgradient(h, top_idx, K)
        
        # Direction: -g (to reduce the max)
        d = -g
        # Remove mean to help with sum constraint
        d -= np.mean(d)
        if np.abs(np.mean(d)) > 0:
            pass  # Already handled by projection
        
        norm_d = np.linalg.norm(d)
        if norm_d < 1e-14:
            break
        d /= norm_d
        
        # Line search
        improved = False
        for trial_step in [step, step*0.5, step*0.25, step*0.1, 0.01]:
            h_trial = project_density(h + trial_step * d, target)
            c_trial = compute_c5(h_trial)

                best_h = h_trial.copy()

            if step < 1e-6:

    best_c = compute_c5(best_h)
    return best_h, best_c

import time
import numpy as np


def _project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    n = x.size
    if n == 0:
        return x
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    s = float(np.sum(x))
    if abs(s - target) < 1e-13:
        return x
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s_mid = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s_mid < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-14:
        free = np.where((h > 1e-14) & (h < 1.0 - 1e-14))[0]
        if free.size > 0:
            if res > 0:
                idx = int(free[np.argmax(1.0 - h[free])])
            else:
                idx = int(free[np.argmax(h[free])])
            h[idx] = min(1.0, max(0.0, h[idx] + res))
    return h


def _c5(h):
    n = h.size
    if n < 1:
        return 1e9
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _top_corr_indices(h, K):
    corr = np.correlate(h, 1.0 - h, mode='full')
    K = min(K, corr.size)
    return np.argsort(corr)[-K:], corr


def _subgrad(h, idx_list):
    n = h.size
    center = n - 1
    g = np.zeros(n)
    for t in idx_list:
        t = int(t)
        k = t - center
        if k == 0:
            g += 1.0 - 2.0 * h
        else:
            lag = abs(k)
            if k > 0:
                g[lag:] += 1.0
            else:
                g[:n - lag] += 1.0
            g[:n - lag] -= h[lag:]
            g[lag:] -= h[:n - lag]
    g /= float(len(idx_list))
    return g


def _pgd_optimize(h0, target, max_iter, K_init, K_max, deadline, step_init=0.3):
    n = h0.size
    h = _project_density(h0.copy(), target)
    best_h = h.copy()
    best_c = _c5(h)
    step = step_init
    K = min(K_init, 2 * n - 1)
    stall = 0
    K_increase = 0

    for it in range(max_iter):
        if deadline is not None and time.perf_counter() >= deadline:
            break

        top_idx, corr_arr = _top_corr_indices(h, K)
        g = _subgrad(h, top_idx)
        d = -g
        d -= np.mean(d)
        nd = float(np.linalg.norm(d))
        if nd < 1e-14:
            break
        d /= nd

        improved = False
        for s_t in [step, step * 0.5, step * 0.25, step * 0.1, 0.005]:
            h_t = _project_density(h + s_t * d, target)
            c_t = _c5(h_t)
            if c_t < best_c - 1e-13:
                h = h_t
                best_h = h_t.copy()
                best_c = c_t
                step = min(step * 1.15, 1.0)
                improved = True
                break

        if improved:
            stall = 0
            K_increase += 1
            if K_increase % 50 == 0 and K < K_max:
                K = min(K + max(1, K // 5), K_max)
        else:
            stall += 1
            step *= 0.7
            if stall > 30:
                if K < K_max:
                    K = min(K + 3, K_max)
                    stall = 15
                else:
                    break

    best_h = _project_density(best_h, target)
    best_c = _c5(best_h)
    return best_h, best_c


def _pair_polish(h0, target, deadline, max_iter=2000, K_lags=8):
    n = h0.size
    h = _project_density(h0.copy(), target)
    best_h = h.copy()
    best_c = _c5(h)
    K = min(K_lags, 2 * n - 1)
    stall = 0

    for _ in range(max_iter):
        if deadline is not None and time.perf_counter() >= deadline:
            break

        top_idx, corr_arr = _top_corr_indices(h, K)
        g = _subgrad(h, top_idx)
        if (float(np.max(g)) - float(np.min(g))) < 1e-12:
            break

        neg_idx = np.argsort(g)[:4]
        pos_idx = np.argsort(-g)[:4]
        found = False

        for ii in neg_idx:
            ii = int(ii)
            for jj in pos_idx:
                jj = int(jj)
                if ii == jj:
                    continue
                cap_pos = 1.0 - h[ii]
                cap_neg = h[jj]
                cap = min(cap_pos, cap_neg)
                if cap < 1e-12:
                    continue
                a = g[ii] - g[jj]
                if a >= -1e-12:
                    continue

                for frac in (0.3, 0.6, 0.9, 1.0):
                    d = frac * cap
                    h_t = h.copy()
                    h_t[ii] = min(1.0, h_t[ii] + d)
                    h_t[jj] = max(0.0, h_t[jj] - d)
                    c_t = _c5(h_t)
                    if c_t < best_c - 1e-13:
                        h = h_t
                        best_h = h_t.copy()
                        best_c = c_t
                        found = True
                        break

        if not found:
            d = -g
            d -= np.mean(d)
            nd = float(np.linalg.norm(d))
            if nd > 1e-13:
                d /= nd
                for s in (0.02, 0.05, 0.1):
                    h_t = _project_density(h + s * d, target)
                    c_t = _c5(h_t)
                    if c_t < best_c - 1e-13:
                        h = h_t
                        best_h = h_t.copy()
                        best_c = c_t
                        found = True
                        break

        if found:
            stall = 0
        else:
            stall += 1
            if stall > 40:
                break

    best_h = _project_density(best_h, target)
    best_c = _c5(best_h)
    return best_h, best_c


def _upsample(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    n = int(n)
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return _project_density(np.full(n, 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.interp(x1, x0, p, left=p[0], right=p[-1])
    return _project_density(h, 0.5 * n)


def _structured_init(n, rng, mode):
    target = 0.5 * n
    if mode == 0:
        h = np.zeros(n)
        n_ones = n // 2
        indices = rng.choice(n, size=n_ones, replace=False)
        h[indices] = 1.0
    elif mode == 1:
        period = int(rng.integers(max(2, n // 20), n // 5 + 1))
        duty = float(rng.uniform(0.3, 0.7))
        h = np.zeros(n)
        for i in range(n):
            if (i % period) < int(duty * period):
                h[i] = 1.0
    elif mode == 2:
        h = np.full(n, 0.5)
        n_terms = int(rng.integers(2, 6))
        for _ in range(n_terms):
            freq = int(rng.integers(1, n // 10 + 1))
            amp = float(rng.uniform(0.1, 0.35))
            phase = float(rng.uniform(0, 2 * np.pi))
            x = (np.arange(n) + 0.5) * (2.0 / n)
            h += amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    else:
        h = np.zeros(n)
        n_blocks = int(rng.integers(3, n // 10 + 1))
        for b in range(n_blocks):
            start = int(b * n / n_blocks)
            end = int((b + 1) * n / n_blocks)
            h[start:end] = float(rng.choice([0.0, 0.5, 1.0]))
    return _project_density(h, target)


def _momentum_pgd(h0, target, max_iter, K, deadline, step_init=0.2, beta=0.5):
    n = h0.size
    h = _project_density(h0.copy(), target)
    best_h = h.copy()
    best_c = _c5(h)
    step = step_init
    v = np.zeros(n)
    stall = 0

    for it in range(max_iter):
        if deadline is not None and time.perf_counter() >= deadline:
            break

        top_idx, _ = _top_corr_indices(h, K)
        g = _subgrad(h, top_idx)
        v = beta * v - step * g
        h_t = _project_density(h + v, target)
        c_t = _c5(h_t)
        if c_t < best_c - 1e-13:
            h = h_t
            best_h = h_t.copy()
            best_c = c_t
            stall = 0
        else:
            v = 0.5 * v
            step *= 0.8
            stall += 1
            if stall > 40:
                break

    best_h = _project_density(best_h, target)
    best_c = _c5(best_h)
    return best_h, best_c
