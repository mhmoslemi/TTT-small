import time

    c = np.correlate(h, 1.0 - h, mode="full")

def soft_max_value(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * c
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / beta * dx)

def soft_max_grad(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * c
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w_sum = np.sum(w)
    w /= w_sum
    grad = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k]
        if wk < 1e-14:

        lag = k - (n - 1)
        if lag >= 0:
            if lag == 0:
                grad += wk * (1.0 - 2.0 * h)
            else:
                grad[:n-lag] += wk * (1.0 - h[lag:])
                grad[lag:] -= wk * h[:n-lag]
        else:
            d = -lag
            grad[d:] += wk * (1.0 - h[:n-d])
            grad[:n-d] -= wk * h[d:]
    return grad * dx

def project_feasible(h, target):
    """Project h to {x: 0<=x<=1, sum(x)=target} via bisection on lambda."""
    h = np.asarray(h, dtype=float).copy()
    
    def sum_clipped(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0))
    
    # Bisection
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = sum_clipped(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    lam = (lo + hi) / 2
    result = np.clip(h - lam, 0.0, 1.0)
    # Fine-tune
    diff = target - result.sum()
    if abs(diff) > 1e-10:
        # Adjust slightly
        if diff > 0:
            mask = result < 1.0 - 1e-12
        else:
            mask = result > 1e-12
        if mask.any():
            result[mask] += diff / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

    grad = soft_max_grad(h, n, beta)
    h_new = h - alpha * grad

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def soft_max_obj(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * c
    m = np.max(scaled)
    val = (m + np.log(np.sum(np.exp(scaled - m)))) / beta
    return val * dx


def soft_max_grad(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * c
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if lag == 0:
                grad += wk * (1.0 - 2.0 * h)
            else:
                grad[lag:] += wk * (1.0 - h[:n - lag])
                grad[:n - lag] -= wk * h[lag:]
        else:
            d = -lag
            grad[:n - d] += wk * (1.0 - h[d:])
            grad[d:] -= wk * h[:n - d]
    return grad * dx
