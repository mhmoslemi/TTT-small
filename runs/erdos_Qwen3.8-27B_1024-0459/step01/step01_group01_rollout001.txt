def project_box_sum(x, S):
    x=np.asarray(x,dtype=float).ravel()
    n=x.size
    if n==0: return x
    S=float(S)
    if S<=0: return np.zeros(n)
    if S>=n: return np.ones(n)
    lo=float(np.min(x))-1.0
    hi=float(np.max(x))+1.0
    # At lo, x-lo >=1? if lo=min-1 => max clip 1, sum n >=S.
    # At hi, clip 0, sum0 <=S.
    mid=0.5*(lo+hi)
    for _ in range(80):
        mid=0.5*(lo+hi)
        s=np.sum(np.clip(x-mid,0.0,1.0))
        if s > S:
            lo=mid
        else:
            hi=mid
    h=np.clip(x-mid,0.0,1.0)
    # correct sum
    err=S-h.sum()
    for _ in range(3):
       if abs(err)<1e-13: break
       if err>0:
          idx=np.where(h<1.0-1e-15)[0]
       else:
          idx=np.where(h>1e-15)[0]
       if idx.size:
          per=err/idx.size
          h[idx]+=per
          h=np.clip(h,0,1)
          err=S-h.sum()
    # final exact adjust
    err=S-h.sum()
    if err>0:
       # add to elements with most room
       i=np.argmax(1.0-h)
       d=min(min(err,1.0-h[i]), err)
       if d>0: h[i]+=d

       i=np.argmax(h)
       d=max(max(err,-h[i]), err) # err negative
       if d!=0: h[i]+=d
    return h

def project_box_sum(x, S):
    x = np.asarray(x, dtype=float).ravel()
    n = x.size
    if n == 0: return x.copy()
    S = float(S)
    if S <= 0.0: return np.zeros(n, dtype=float)
    if S >= n: return np.ones(n, dtype=float)
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    mid = 0.5*(lo+hi)
    for _ in range(70):
        mid = 0.5*(lo+hi)
        s = float(np.sum(np.clip(x-mid, 0.0, 1.0)))
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(x-mid, 0.0, 1.0)
    for _ in range(5):
        err = S - float(h.sum())
        if abs(err) < 1e-14: break
        if err > 0:
            idx = np.nonzero(h < 1.0 - 1e-15)[0]
        else:
            idx = np.nonzero(h > 1e-15)[0]
        if idx.size:
            h[idx] += err / idx.size
            h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        err = S - float(h.sum())
        if abs(err) < 1e-14: break
        if err > 0:
            i = int(np.argmax(1.0 - h))
            d = min(err, 1.0 - h[i])
            if d <= 0: break
            h[i] += d
        else:
            i = int(np.argmax(h))
            d = max(err, -h[i])
            if d >= 0: break
            h[i] += d
    return h

import math
import time

import numpy as np
from scipy.optimize import minimize


_SUM_TARGET = 0.5


def project_box_sum(x, S):
    """Project x onto {0 <= h <= 1, sum(h) = S}."""
    x = np.asarray(x, dtype=float).ravel()
    n = x.size
    if n == 0:
        return x.copy()
    S = float(S)
    x = np.nan_to_num(x, nan=0.0, posinf=1.0, neginf=0.0)
    if S <= 0.0:
        return np.zeros(n, dtype=float)
    if S >= n:
        return np.ones(n, dtype=float)

    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    mid = 0.5 * (lo + hi)

    for _ in range(55):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > S:
            lo = mid
        else:
            hi = mid

    h = np.clip(x - mid, 0.0, 1.0)

    for _ in range(5):
        err = S - float(h.sum())
        if abs(err) < 1e-14:
            break
        if err > 0.0:
            idx = np.nonzero(h < 1.0 - 1e-15)[0]
        else:
            idx = np.nonzero(h > 1e-15)[0]
        if idx.size:
            h[idx] += err / float(idx.size)
            h = np.clip(h, 0.0, 1.0)

    for _ in range(12):
        err = S - float(h.sum())
        if abs(err) < 1e-14:
            break
        if err > 0.0:
            i = int(np.argmax(1.0 - h))
            d = min(err, 1.0 - h[i])
            if d <= 0.0:
                break
            h[i] += d
        else:
            i = int(np.argmax(h))
            d = max(err, -h[i])
            if d >= 0.0:
                break
            h[i] += d
    return h


def compute_c5(h):
    """Exact discrete Erdos overlap objective used by the evaluator."""
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return 1e9
    if not np.all(np.isfinite(h)):
        return 1e9
    T = np.correlate(h, 1.0 - h, mode="full")
    val = float(np.max(T) * (2.0 / float(n)))
    return val if np.isfinite(val) else 1e9


def resample_h(h_old, n_new):
    """Resample a cell-centered function on [0,2] to n_new cells."""
    h_old = np.asarray(h_old, dtype=float).ravel()
    if n_new <= 0:
        return np.zeros(0, dtype=float)
    if h_old.size == 0:
        return project_box_sum(np.full(n_new, 0.5), 0.5 * n_new)
    if h_old.size == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, h_old.size, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_box_sum(h_new, 0.5 * n_new)


def smoothmax_obj(x, n, dx, tau):
    """Smooth approximation to dx * max_lag corr."""
    x = np.asarray(x, dtype=float).ravel()
    if x.size != n:
        x = resample_h(x, n)
    if not np.all(np.isfinite(x)):
        x = np.nan_to_num(x, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(x, 0.0, 1.0)
    T = np.correlate(h, 1.0 - h, mode="full")
    if not np.all(np.isfinite(T)):
        return 1e9
    m = float(np.max(T))
    if tau <= 1e-12:
        val = m
    else:
        z = np.exp((T - m) / tau)
        s = float(np.sum(z))
        if not np.isfinite(s) or s <= 0.0:
            val = m
        else:
            val = m + tau * np.log(s)
    return float(val * dx)


def smoothmax_jac(x, n, dx, tau):
    """Gradient of smoothmax_obj."""
    x = np.asarray(x, dtype=float).ravel()
    if x.size != n:
        h = resample_h(x, n)
    else:
        h = x.copy()
    if not np.all(np.isfinite(h)):
        return np.zeros(max(0, int(n)), dtype=float)
    h = np.clip(h, 0.0, 1.0)

    T = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(T))

    if tau <= 1e-12:
        w = np.zeros(T.size, dtype=float)
        act = np.where(T >= m - 1e-12)[0]
        if act.size == 0:
            act = np.array([int(np.argmax(T))], dtype=int)
        w[act] = 1.0 / float(act.size)
    else:
        z = np.exp((T - m) / tau)
        s = float(np.sum(z))
        if not np.isfinite(s) or s <= 0.0:
            w = np.zeros(T.size, dtype=float)
            act = np.where(T >= m - 1e-12)[0]
            if act.size == 0:
                act = np.array([int(np.argmax(T))], dtype=int)
            w[act] = 1.0 / float(act.size)
        else:
            w = z / s

    grad = np.zeros(n, dtype=float)
    if n == 0:
        return grad

    q = 1.0 - h
    ww = w
    hh = h
    qq = q
    L = n - 1
    for i in range(n):
        grad[i] = float(np.dot(ww[i:i + n][::-1], qq) - np.dot(ww[L - i:2 * L - i + 1], hh))

    grad *= dx
    if not np.all(np.isfinite(grad)):
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return grad


def sum_eq_fun(x):
    return np.array([float(np.sum(x)) - _SUM_TARGET])


def sum_eq_jac(x):
    return np.ones((1, x.size), dtype=float)


def slsqp_optimize(h0, n, tau, maxiter, deadline):
    """Run SLSQP for smoothmax with exact box and equality constraints."""
    global _SUM_TARGET
    n = int(n)
    if n <= 0:
        return h0
    if time.time() >= deadline:
        return project_box_sum(h0, 0.5 * n)

    x0 = project_box_sum(h0, 0.5 * n)
    maxiter = int(maxiter)
    if maxiter < 5:
        return x0

    _SUM_TARGET = 0.5 * n
    bounds = [(0.0, 1.0) for _ in range(n)]
    try:
        res = minimize(
            fun=smoothmax_obj,
            x0=x0,
            method="SLSQP",
            bounds=bounds,
            jac=smoothmax_jac,
            constraints=[{"type": "eq", "fun": sum_eq_fun, "jac": sum_eq_jac}],
            args=(n, 2.0 / n, tau),
            options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
        )
        x1 = res.x
    except Exception:
        x1 = x0
    return project_box_sum(x1, 0.5 * n)


def improve_candidate(h, n, best_c, best_h):
    """Project a candidate, try reversal symmetry, and update best."""
    n = int(n)
    if n <= 0:
        return h, best_c, best_c, best_h
    h = np.asarray(h, dtype=float).ravel()
    if h.size != n:
        h = resample_h(h, n)
    h = project_box_sum(h, 0.5 * n)
    c = compute_c5(h)

    h_use = h
    c_use = c

    if n > 1:
        hs = project_box_sum(0.5 * (h + h[::-1]), 0.5 * n)
        cs = compute_c5(hs)
        if cs < c_use:
            h_use = hs
            c_use = cs

    if c_use < best_c:
        best_c = c_use
        best_h = h_use.copy()

    return h_use, c_use, best_c, best_h


def polish_with_slsqp(h_start, n, tau_list, deadline, best_c, best_h, maxiter_base):
    """Successive SLSQP smoothing from a starting point."""
    h, c, best_c, best_h = improve_candidate(h_start, n, best_c, best_h)
    for tau in tau_list:
        if time.time() >= deadline:
            break
        h2 = slsqp_optimize(h, n, tau, maxiter_base, deadline)
        h_alt, c_alt, best_c, best_h = improve_candidate(h2, n, best_c, best_h)
        if c_alt <= c + 1e-14:
            h = h_alt
            c = c_alt
    return best_c, best_h


def random_smooth_h(n, rng, amp=0.35):
    """Low-frequency random smooth starting point, centered at 0.5."""
    if n <= 0:
        return np.zeros(0, dtype=float)
    freq = np.fft.rfftfreq(n) * float(n)
    fall = np.exp(-(freq * freq) / 9.0)
    f = rng.normal(size=freq.size) * fall
    f[0] = 0.0
    h = np.fft.irfft(f, n=n)
    mn = float(np.mean(h))
    sd = float(np.std(h))
    if sd > 1e-12:
        h = amp * (h - mn) / sd
    else:
        h = np.zeros(n, dtype=float)
    h = 0.5 + h
    return project_box_sum(h, 0.5 * n)


def periodic_h(n, period, phase, amp):

    if n <= 0:

    if period <= 0.0 or not np.isfinite(period):

def block_obj(x, m, L, tau):
    """Smoothmax objective for a block-constant function."""
    x = np.asarray(x, dtype=float).ravel()
    if x.size != m:
        return 1e9
    h = np.repeat(x, L)
    n = h.size
    if n == 0:
        return 1e9
    return smoothmax_obj(h, n, 2.0 / float(n), tau)


def block_jac(x, m, L, tau):
    """Gradient with respect to block values."""
    x = np.asarray(x, dtype=float).ravel()
    if x.size != m:
        return np.zeros(max(0, int(m)), dtype=float)
    h = np.repeat(x, L)
    n = h.size
    if n == 0:
        return np.zeros(max(0, int(m)), dtype=float)
    g = smoothmax_jac(h, n, 2.0 / float(n), tau)
    if n == m * L:
        gb = g.reshape(m, L).sum(axis=1)
    else:
        gb = np.zeros(m, dtype=float)
        for i in range(m):
            a = i * L
            b = min(n, (i + 1) * L)
            gb[i] = float(np.sum(g[a:b]))
    return gb


def optimize_block(b0, m, L, tau, maxiter, deadline):

    global _SUM_TARGET
    if time.time() >= deadline:

def exact_random_polish(h0, n, deadline, best_c, best_h, rng, step_init=0.25):

        d = d / nrm

        trial = project_box_sum(h + step * d, 0.5 * n)
        c2 = compute_c5(trial)
        if c2 < c - 1e-14:
            h = trial
            c = c2
            step = min(step * 1.05, 4.0)
            stall = 0
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            delta = c2 - c
            if delta < 0.0015 and np.isfinite(delta):
                p = math.exp(-float(delta) / max(1e-12, temp))
            else:
                p = 0.0
            if rng.random() < p:
                h = trial
                c = c2
                stall += 1
            else:
                stall += 1
                if stall >= 22:
                    step *= 0.6
                    stall = 0
                    if step < 0.015:
                        break
        temp = max(1e-6, temp * 0.995)

    c = compute_c5(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_c, best_h


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    if budget <= 0.0:
        try:
            parent_h = np.asarray(initial_h_values, dtype=float).ravel()
        except Exception:
            parent_h = np.full(180, 0.5)
        if parent_h.size == 0:
            parent_h = np.full(180, 0.5)
        h = project_box_sum(parent_h, 0.5 * len(parent_h))
        return (h, compute_c5(h), len(h))

    deadline = t0 + budget
    rng = np.random.default_rng(seed)

    try:
        parent_h = np.asarray(initial_h_values, dtype=float).ravel()
    except Exception:
        parent_h = np.full(180, 0.5)
    if parent_h.size == 0:
        parent_h = np.full(180, 0.5)

    n_initial = parent_h.size
    n0 = n_initial
    if n0 > 720:
        n0 = 720
    parent_work = resample_h(parent_h, n0)

    best_h = project_box_sum(parent_work, 0.5 * n0)
    best_c = compute_c5(best_h)

    scale = max(0.25, min(1.0, budget / 1300.0))

    block_deadline = t0 + budget * 0.18
    low_deadline = t0 + budget * 0.55
    mid_deadline = t0 + budget * 0.75
    high_deadline = t0 + budget * 0.90
    final_deadline = t0 + budget * 0.98

    # 1. Coarse block-constant search. This is cheap in dimension and often
    #    finds qualitatively good low-overlap shapes.
    if n0 >= 24:
        best_c, best_h = block_phase(best_h, n0, rng, block_deadline, best_c, best_h, scale)

    # 2. Full-resolution SLSQP polishing at the original/working resolution.
    low_maxit = maxiter_for_n(n0, scale)
    low_starts = [best_h.copy(), parent_work.copy()]

    for sigma in (0.03, 0.06, 0.10, 0.16):
        if time.time() >= low_deadline:
            break
        low_starts.append(project_box_sum(best_h + rng.normal(0.0, sigma, size=n0), 0.5 * n0))

    for amp in (0.25, 0.40, 0.55):
        if time.time() >= low_deadline:
            break
        low_starts.append(random_smooth_h(n0, rng, amp))

    for period in (1.0, 0.5, 2.0 / 3.0, 0.4, 0.8):
        if time.time() >= low_deadline:
            break
        low_starts.append(periodic_h(n0, period, float(rng.uniform(0.0, 2.0 * np.pi)), 0.35))
        low_starts.append(periodic_h(n0, period, float(rng.uniform(0.0, 2.0 * np.pi)), 0.50))

    low_starts = low_starts[:18]
    tau_low = [0.05, 0.018, 0.006]

    for idx, hst in enumerate(low_starts):
        if time.time() >= low_deadline:
            break
        cst = compute_c5(hst)
        if idx >= 4 and cst > best_c + 0.08:
            continue
        if idx < 2:
            mi = low_maxit
            tr = tau_low
        elif idx < 7:
            mi = int(max(30, 0.80 * low_maxit))
            tr = tau_low[1:]
        else:
            mi = int(max(30, 0.60 * low_maxit))
            tr = tau_low[1:2] if len(tau_low) > 1 else tau_low
        best_c, best_h = polish_with_slsqp(hst, n0, tr, low_deadline, best_c, best_h, mi)

    # 3. Resolution doubling.
    n1 = 2 * n0 if n0 <= 360 else n0
    if n1 > 900:
        n1 = n0
    if n1 > n0 and time.time() < mid_deadline:
        h_up1 = resample_h(best_h, n1)
        h_up2 = resample_h(parent_work, n1)
        mid_maxit = maxiter_for_n(n1, scale * 0.85)
        best_c, best_h = polish_with_slsqp(h_up1, n1, [0.035, 0.012, 0.005], mid_deadline, best_c, best_h, mid_maxit)
        if time.time() < mid_deadline:
            best_c, best_h = polish_with_slsqp(
                h_up2, n1, [0.020, 0.008], mid_deadline, best_c, best_h, int(max(30, 0.70 * mid_maxit))
            )

    # 4. Optional higher resolution for the best shape found so far.
    n2 = 4 * n0 if n0 <= 180 else (2 * n0 if n0 <= 360 else n0)
    if n2 <= 720 and n2 > len(best_h) and time.time() < high_deadline:
        h_up = resample_h(best_h, n2)
        high_maxit = maxiter_for_n(n2, scale * 0.65)
        best_c, best_h = polish_with_slsqp(h_up, n2, [0.018, 0.006], high_deadline, best_c, best_h, high_maxit)
        if time.time() < high_deadline:
            h_up2 = resample_h(parent_work, n2)
            best_c, best_h = polish_with_slsqp(
                h_up2, n2, [0.010], high_deadline, best_c, best_h, int(max(20, 0.55 * high_maxit))
            )

    # 5. Deterministic gradient polish with increasingly sharp smoothmax.
    if time.time() < final_deadline:
        n_best = len(best_h)
        for tau in (0.020, 0.008, 0.003):
            if time.time() >= final_deadline:
                break
            if n_best <= 200:
                git = 3500
            elif n_best <= 400:
                git = 1200
            else:
                git = 450
            git = int(max(60, git * scale))
            best_c, best_h = gradient_polish(
                best_h, len(best_h), tau, final_deadline, best_c, best_h,
                max_iter=git, step_init=max(0.2, 0.012 * len(best_h))
            )

    # 6. Very short exact-objective random polishing.
    if time.time() < final_deadline:
        rem_deadline = min(deadline - 0.05, final_deadline)
        if rem_deadline > time.time():
            best_c, best_h = exact_random_polish(
                best_h, len(best_h), rem_deadline, best_c, best_h, rng,
                step_init=max(0.05, 0.012 * len(best_h))
            )

    # Final exact projection and evaluation.
    best_h = project_box_sum(best_h, 0.5 * len(best_h))
    best_c = compute_c5(best_h)

    return (best_h, float(best_c), int(len(best_h)))
