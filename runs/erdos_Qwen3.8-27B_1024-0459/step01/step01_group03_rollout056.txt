def project_density(x, target):
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    if n == 0:
        return x
    target = float(target)
    s = float(np.sum(x))
    if abs(s - target) <= 1e-14 * max(1, n):
        return x
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-12:
        free = np.where((h > 1e-15) & (h < 1.0 - 1e-15))[0]
        if free.size:
            if res > 0:
                idx = int(free[np.argmax(1.0 - h[free])])
            else:
                idx = int(free[np.argmax(h[free])])
            h[idx] += res
            h = np.clip(h, 0.0, 1.0)
    return h

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float('inf')
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if corr.size == 0:
        return float('inf')
    return float(np.max(corr) * dx)

def coord_sweep(h, target, max_sweeps=100, grid_size=4001, verbose=False):
    n = len(h)
    center = n - 1
    h_work = h.copy()
    best_h = h_work.copy()
    best_c = evaluate_erdos(best_h)
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            # Compute c and d for all lags
            c = np.zeros(2 * n - 1)
            d = np.zeros(2 * n - 1)
            
            # For m > 0 (correlate index center + m):
            for m in range(1, n):
                k = center + m
                if j >= m:
                    c[k] += (1.0 - h_work[j - m])
                if j + m < n:
                    c[k] += (-h_work[j + m])
                    d[k] += h_work[j + m]
            
            # For m > 0 (correlate index center - m):
            for m in range(1, n):
                k = center - m
                if j + m < n:
                    c[k] += (1.0 - h_work[j + m])
                if j >= m:
                    c[k] += (-h_work[j - m])
                    d[k] += h_work[j - m]
            
            # Zero lag: B0 + t - t^2
            B0 = 0.0
            for i in range(n):
                if i != j:
                    B0 += h_work[i] * (1.0 - h_work[i])
            
            # Minimize max(B0 + t - t^2, max_k(c[k]*t + d[k])) over t in [0,1]
            t_grid = np.linspace(0.0, 1.0, grid_size)
            zero_lag = B0 + t_grid - t_grid * t_grid
            # For non-zero lags
            c_nz = c.copy()
            d_nz = d.copy()
            c_nz[center] = -np.inf
            # Evaluate max of linear functions
            non_zero = np.max(c_nz[None, :] * t_grid[:, None] + d_nz[None, :], axis=1)

            t_star = t_grid[np.argmin(f)]
            
            if t_star != h_work[j]:
                h_work[j] = t_star
                improved = True
        
        # Project to sum constraint
        h_work = project_density(h_work, target)
        c_val = evaluate_erdos(h_work)
        if c_val < best_c - 1e-12:
            best_c = c_val
            best_h = h_work.copy()
        
        if not improved:
            break
    
    return best_h, best_c

import time
import numpy as np
from scipy.optimize import minimize


def project_density(x, target):
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    if n == 0:
        return x
    target = float(target)
    s = float(np.sum(x))
    if abs(s - target) <= 1e-14 * max(1, n):
        return x
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-12:
        free = np.where((h > 1e-15) & (h < 1.0 - 1e-15))[0]
        if free.size:
            if res > 0:
                idx = int(free[np.argmax(1.0 - h[free])])
            else:
                idx = int(free[np.argmax(h[free])])
            h[idx] += res
            h = np.clip(h, 0.0, 1.0)
    return h


def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float('inf')
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if corr.size == 0:
        return float('inf')
    return float(np.max(corr) * dx)


def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    n = int(n)
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)

def coord_descent_sweeps(h_init, target, deadline, max_sweeps=5000, grid_size=4001):
    n = len(h_init)
    center = n - 1
    h_work = h_init.copy()
    best_h = h_init.copy()
    best_c = evaluate_erdos(best_h)
    t_grid = np.linspace(0.0, 1.0, grid_size)
    no_improve_count = 0
    rng = np.random.default_rng(12345)

    for sweep in range(max_sweeps):
        if time.perf_counter() >= deadline:
            break
        sweep_improved = False
        idx_order = np.arange(n)
        if sweep % 3 == 1:
            idx_order = idx_order[::-1]
        elif sweep % 3 == 2:
            rng.shuffle(idx_order)

        for j in idx_order:
            j = int(j)
            B0 = float(np.sum(h_work * (1.0 - h_work))) - h_work[j] * (1.0 - h_work[j])

            c = np.zeros(2 * n - 1)
            d = np.zeros(2 * n - 1)

            for m in range(1, n):
                k_pos = center + m
                k_neg = center - m
                if j + m < n:
                    c[k_pos] += (1.0 - h_work[j + m])
                    c[k_neg] += (-h_work[j + m])
                    d[k_neg] += h_work[j + m]
                if j >= m:
                    c[k_pos] += (-h_work[j - m])
                    d[k_pos] += h_work[j - m]
                    c[k_neg] += (1.0 - h_work[j - m])

            zero_lag = B0 + t_grid - t_grid * t_grid
            c[center] = -np.inf
            vals = t_grid[:, None] * c[None, :] + d[None, :]
            non_zero = np.max(vals, axis=1)
            f = np.maximum(zero_lag, non_zero)
            t_star = float(t_grid[np.argmin(f)])

            if abs(t_star - h_work[j]) > 1e-10:
                h_work[j] = t_star
                sweep_improved = True

        h_work = project_density(h_work, target)
        c_val = evaluate_erdos(h_work)
        if c_val < best_c - 1e-13:
            best_c = c_val
            best_h = h_work.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count > 80:
            break
        if not sweep_improved and no_improve_count > 20:
            break

    return best_h, best_c


def polished_local_search(h_init, target, deadline, max_iter=500, active_k=6):
    n = len(h_init)
    best = h_init.copy()
    best_c = evaluate_erdos(best)
    if n < 2:
        return best, best_c
    k = min(int(active_k), 2 * n - 1)
    stall = 0

    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode='full')
        if not np.all(np.isfinite(corr)):
            break
        idxs = np.argsort(corr)[-k:]
        center = n - 1

        g = np.zeros(n)
        for t in idxs:
            t = int(t)
            kk = t - center
            if kk == 0:
                g += 1.0 - 2.0 * best
            else:
                lag = abs(kk)
                if kk > 0:
                    g[lag:] += 1.0
                else:
                    g[:n - lag] += 1.0
                g[:n - lag] -= best[lag:]
                g[lag:] -= best[:n - lag]
        g /= float(k)

        neg = np.argsort(g)[:4]
        pos = np.argsort(-g)[:4]
        found = False

        for ii in neg:
            ii = int(ii)
            for jj in pos:
                jj = int(jj)
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap <= 1e-12:
                    continue
                for frac in (0.3, 0.5, 0.7, 1.0):
                    dv = frac * cap
                    if dv <= 1e-12 or dv > cap:
                        continue
                    cand = best.copy()
                    cand[ii] += dv
                    cand[jj] -= dv
                    cand = np.clip(cand, 0.0, 1.0)
                    cand = project_density(cand, target)
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-12:
                        best = cand
                        best_c = c
                        found = True

        if not found:
            d = -g
            d -= float(np.mean(d))
            norm = float(np.linalg.norm(d))
            if norm > 1e-13:
                d /= norm
                for step in (0.03, 0.1, 0.3, 0.6):
                    cand = project_density(best + step * d, target)
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-12:
                        best = cand
                        best_c = c
                        found = True
                        break

        if found:
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break

    best = project_density(best, target)
    best_c = evaluate_erdos(best)
    return best, best_c


def smooth_objective(h, n, alpha, target):
    if not np.all(np.isfinite(h)):
        return 1e100
    corr = np.correlate(h, 1.0 - h, mode='full')
    if corr.size == 0 or not np.all(np.isfinite(corr)):
        return 1e100
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return 1e100
    return float((2.0 / float(n)) * (m + np.log(z) / alpha))


def make_random_h(n, rng):

        h += amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    return project_density(h, 0.5 * n)
