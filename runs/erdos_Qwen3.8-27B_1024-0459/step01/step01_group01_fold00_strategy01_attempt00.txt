def fix_integral(h, n, target=1.0):
    # Maintains integral constraint without resampling
    current = np.sum(h) * (2.0/n)
    error = current - target
    while abs(error) > 1e-6:
        signs = np.random.choice([-1, 1], size=len(h))
        change = (0.5 * error) * (signs + 0.0)
        h += change
        # Project to bounds
        h = np.clip(h, 0, 1)
        # Maintain integral
