def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = (h < 1.0)
            if not np.any(free):
                break
            h[free] = np.minimum(h[free] + diff / np.sum(free), 1.0)
        else:
            free = (h > 0.0)
            if not np.any(free):
                break
            h[free] = np.maximum(h[free] + diff / np.sum(free), 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def fast_c5_binary(h_bin, n):
    """Fast C5 computation for binary h"""
    dx = 2.0 / n
    conv = np.correlate(h_bin.astype(float), 1.0 - h_bin.astype(float), mode='full')
    return np.max(conv) * dx

def sa_optimize_binary(h0, n, t_start, t_end, rng, T_init=10.0, T_final=0.01):
    """Simulated annealing on binary sequences"""
    h = h0.copy()
    best_h = h.copy()
    best_cost = compute_c5(h, n)
    cost = best_cost
    
    # SA parameters
    alpha = (T_final / T_init) ** (1.0 / max(1, int((t_end - t_start) / 0.5)))
    
    t = t_start
    T = T_init
    while t < t_end:
        # Pick a swap
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            t += 0.01
            continue
        if h[i] == h[j]:
            t += 0.01
            continue
        
        # Swap
        h[i], h[j] = h[j], h[i]
        new_cost = compute_c5(h, n)
        delta = new_cost - cost
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cost = new_cost
            if cost < best_cost:
                best_cost = cost
                best_h = h.copy()
        else:
            # Revert
            h[i], h[j] = h[j], h[i]
        
        T *= alpha
        t += 0.01  # approximate time step
    
    return best_h, best_cost

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv)) * dx


def compute_conv(h, n_points):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = float(np.sum(h))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = (h < 1.0 - 1e-15)
            cnt = int(np.sum(free))
            if cnt == 0:
                break
            h[free] = np.minimum(h[free] + diff / cnt, 1.0)
        else:
            free = (h > 1e-15)
            cnt = int(np.sum(free))
            if cnt == 0:
                break
            h[free] = np.maximum(h[free] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    s = float(np.sum(h))
    if abs(s - target) > 1e-10:
        h += (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h


def soft_max_value(h, n, beta):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = float(np.max(conv))
    shifted = (conv - max_c) * beta
    shifted -= float(np.max(shifted))
    return max_c + (1.0 / beta) * float(np.log(np.sum(np.exp(shifted))))


def soft_max_grad(h, n, beta):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    max_c = float(np.max(conv))
    shifted = (conv - max_c) * beta
    shifted -= float(np.max(shifted))
    weights = np.exp(shifted)
    weights /= float(np.sum(weights))
    grad = np.zeros(n)
    shifts = np.arange(2 * n - 1) - (n - 1)
    for k_full in range(2 * n - 1):
        w = weights[k_full]
        if w < 1e-12:
            continue
        s = int(shifts[k_full])
        if s >= 0:
            i_max = n - s
            idx = np.arange(0, i_max)
            grad[idx] += (1.0 - h[idx + s]) * w
            grad[idx + s] += (-h[idx]) * w
        else:
            p = -s
            idx = np.arange(p, n)
            grad[idx] += (1.0 - h[idx - p]) * w
            grad[idx - p] += (-h[idx]) * w
    return grad * dx

        grad[idx] += (1.0 - h[idx - p]) * dx

                delta_arr[mask_i_ms] += h[i_ms[mask_i_ms]]
                delta_arr[mask_j_ps] += (1.0 - h[j_ps[mask_j_ps]])
                delta_arr[mask_j_ms] -= h[j_ms[mask_j_ms]]

                conv_new = conv + delta_arr
                new_max = float(np.max(conv_new))
                d = new_max - cur_max
                if d < best_delta:
                    best_delta = d
                    best_swap = (i, j)

        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            # Recompute conv (safer)
            conv = compute_conv(h, n)
            cur_max = float(np.max(conv))
        else:
            break

    return h, cur_max * dx


def multi_start_slsqp(h, n, rng, n_starts, maxiter):
    best_h = h.copy()
    best_val = compute_c5(h, n)
    for _ in range(n_starts):
        h_p = h + rng.normal(0, 0.05, n)
        h_p = project_feasible(h_p, n)
        h_try, v_try = slsqp_polish(h_p, n, maxiter=maxiter)
        if v_try < best_val:
            best_h = h_try.copy()
            best_val = v_try
    return best_h, best_val


def upsample_binary(h_coarse, n_coarse, n_fine):
    x_c = np.linspace(0, 2, n_coarse, endpoint=False) + 1.0 / n_coarse
    x_f = np.linspace(0, 2, n_fine, endpoint=False) + 1.0 / n_fine
    h_fine = np.interp(x_f, x_c, h_coarse)
    return round_to_binary(h_fine, n_fine)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    # Get initial solution
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if h0 is None or len(h0) == 0:
        n_init = 70
        h0 = np.full(n_init, 0.5)
    else:
        n_init = len(h0)

    h_init = project_feasible(h0, n_init)
    best_h = h_init.copy()
    best_val = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Soft-max gradient descent at initial resolution
    if time_left() > 120:
        h_work = best_h.copy()
        for beta in [2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 400.0, 800.0]:
            if time_left() < 30:
                break
            n_steps = 200
            lr = 0.3 / (1.0 + 0.1 * beta)
            for step in range(n_steps):
                if time_left() < 10:
                    break
                val = soft_max_value(h_work, best_n, beta)
                grad = soft_max_grad(h_work, best_n, beta)
                h_new = h_work - lr * grad
                h_new = project_feasible(h_new, best_n)
                new_val = soft_max_value(h_new, best_n, beta)
                if new_val < val:
                    h_work = h_new
                else:
                    lr *= 0.7
                if step % 50 == 49:
                    lr *= 0.9
            actual_val = compute_c5(h_work, best_n)
            if actual_val < best_val:
                best_val = actual_val
                best_h = h_work.copy()

    # Phase 2: SLSQP polish on current best
    if time_left() > 100:
        h_try, v_try = slsqp_polish(best_h, best_n, maxiter=200)
        if v_try < best_val:
            best_h = h_try
            best_val = v_try

    # Phase 3: Binary SA + local search at multiple resolutions
    resolutions = []
    for nr in [70, 100, 140, 200, 280]:
        if nr not in resolutions:
            resolutions.append(nr)

    for n_res in resolutions:
        if time_left() < 120:
            break

        # Initialize binary at this resolution
        if n_res == best_n:
            h_bin = round_to_binary(best_h, n_res)
        else:
            h_bin = upsample_binary(best_h, best_n, n_res)

        # Also try a few random initializations
        h_candidates = [h_bin]
        for _ in range(3):
            h_rand = np.zeros(n_res, dtype=float)
            idx = rng.choice(n_res, size=n_res // 2, replace=False)
            h_rand[idx] = 1.0
            h_candidates.append(h_rand)

        for h_cand in h_candidates:
            if time_left() < 60:
                break
            # SA phase
            sa_iters = max(5000, min(50000, int((time_left() - 30) / (2 * n_res / 100.0))))
            sa_iters = min(sa_iters, 30000)
            h_sa, val_sa = sa_binary_fast(h_cand, n_res, sa_iters,
                                         0.05, 0.0001, rng, time.time(), time.time() + time_left() - 30)
            # Local search
            t_ls_start = time.time()
            t_ls_limit = tLS_start = t_ls_start + min(60, time_left() - 20) if time_left() > 40 else t_ls_start
            h_ls, val_ls = local_search_binary(h_sa, n_res, max_sweeps=30,
                                              t_start=t_ls_start, t_limit=t_ls_limit)
            if val_ls < best_val and n_res == n_res:
                best_val = val_ls
                best_h = h_ls.copy()
                best_n = n_res

            # SLSQP polish on binary solution
            if time_left() > 40:
                h_pol, val_pol = slsqp_polish(h_ls, n_res, maxiter=100)
                if val_pol < best_val:
                    best_val = val_pol
                    best_h = h_pol.copy()
                    best_n = n_res

    # Phase 4: Additional SLSQP multi-start on best
    if time_left() > 60:
        h_try, v_try = multi_start_slsqp(best_h, best_n, rng, 5, 150)
        if v_try < best_val:
            best_h = h_try
            best_val = v_try

    # Phase 5: Final polish
    if time_left() > 20:
        h_try, v_try = slsqp_polish(best_h, best_n, maxiter=300)
        if v_try < best_val:
            best_h = h_try
            best_val = v_try

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_val = compute_c5(best_h, best_n)

    return (best_h, best_val, best_n)
