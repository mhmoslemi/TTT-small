```python
    sum_A = np.zeros(n)
    sum_B = np.zeros(n)
    for j in range(n):
        # A contribution: h[j+s] for s where 0<=j+s<n, weight w[n-1+s]
        s_min = max(-(n-1), -j)  # s >= -(n-1) and s >= -j
        s_max = min(n-1, n-1-j)  # s <= n-1 and s <= n-1-j
```

```python
def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        if abs(s - target) < 1e-15:
            break
        # Binary search for lambda
        lo, hi = -1.0, 1.0
        for _ in range(50):
            mid = (lo + hi) / 2
            s_mid = np.sum(np.clip(h + mid, 0.0, 1.0))
            if s_mid > target:
                hi = mid
            else:
                lo = mid
        h = np.clip(h + (lo+hi)/2, 0.0, 1.0)
    return h
```
