    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()

def smooth_max_obj(h, n, beta):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    s = np.sum(np.exp(beta * (c - m)))
    return float((m + math.log(s) / beta) * (2.0 / n))

def smooth_max_grad(h, n, beta):
    c = np.correlate(h, 1.0 - h, mode="full")

    w = np.exp(beta * (c - m))

        x_new = np.arange(n_new) * 2.0 / n_new + 1.0 / n_new
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_new)

        for beta in [5.0, 20.0, 80.0, 300.0]:
            if time_left() < 150:
                break
            try:
                h_new = slsqp_optimize(h_interp, n_new, beta=beta, maxiter=400)
                c5 = compute_c5(h_new, n_new)
                if c5 < best_c5:
                    best_h, best_c5, best_n = h_new, c5, n_new
                h_interp = project_feasible(h_new, n_new)
            except Exception:
                pass

        if time_left() > 80:
            try:
                h_new = slsqp_optimize(h_interp, n_new, beta=None, maxiter=500, ftol=1e-13)
                c5 = compute_c5(h_new, n_new)
                if c5 < best_c5:
                    best_h, best_c5, best_n = h_new, c5, n_new
            except Exception:
                pass

            # Subgradient refinement
            if time_left() > 60:
                h_sg, c5_sg = projected_subgrad_descent(h_interp, n_new, max_iter=1500)
                if c5_sg < best_c5:
                    best_h, best_c5, best_n = h_sg, c5_sg, n_new

    # Phase 3: Binary SA at even resolutions for diverse structures
    for n_sa in [96, 192, 288, 384]:
        if time_left() < 60:
            break
        h_bin = make_block_init(n_sa, rng)
        c5_bin = compute_c5(h_bin, n_sa)
        h_sa, c5_sa = binary_sa(h_bin, n_sa, n_steps=8000, T_start=0.4, T_end=0.003, seed=seed + n_sa)
        if c5_sa < best_c5:
            best_h, best_c5, best_n = h_sa, c5_sa, n_sa

        # Refine the SA result with gradient methods
        if time_left() > 80:
            h_ref = slsqp_optimize(h_sa, n_sa, beta=200.0, maxiter=300)
            c5_ref = compute_c5(h_ref, n_sa)
            if c5_ref < best_c5:
                best_h, best_c5, best_n = h_ref, c5_ref, n_sa

    # Phase 4: Structured initializations + optimization
    n_struct = best_n if best_n >= 100 else 190
    x_struct = np.arange(n_struct) * 2.0 / n_struct + 1.0 / n_struct

    inits = []
    # Block structures
    for bi in range(4):
        inits.append(make_block_init(n_struct, np.random.RandomState(seed + 100 + bi)))
    # Sine structures
    for freq in [1, 2, 3, 4]:
        inits.append(make_sine_init(n_struct, freq, 0.6, np.random.RandomState(seed + 200 + freq)))
    # Interpolated best
    if best_n != n_struct:
        h_interp_s = np.interp(x_struct, np.arange(best_n) * 2.0 / best_n + 1.0 / best_n, best_h)
        inits.append(project_feasible(h_interp_s, n_struct))

    for h_s in inits:
        if time_left() < 40:
            break
        try:
            h_opt = slsqp_optimize(h_s, n_struct, beta=50.0, maxiter=300)
            c5_s = compute_c5(h_opt, n_struct)
            if c5_s < best_c5:
                best_h, best_c5, best_n = h_opt, c5_s, n_struct
                h_s_ref = h_opt
                if time_left() > 30:
                    h_opt2 = slsqp_optimize(h_opt, n_struct, beta=None, maxiter=300, ftol=1e-13)
                    c5_s2 = compute_c5(h_opt2, n_struct)
                    if c5_s2 < best_c5:
                        best_h, best_c5 = h_opt2, c5_s2
        except Exception:
            pass

    # Phase 5: Perturbation restarts at best resolution
    bounds_opt = [(0.0, 1.0)] * best_n
    cons_opt = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (best_n,)}]

    n_pert = max(0, int((time_left() - 40) / 6))
    for i in range(min(n_pert, 60)):
        if time_left() < 30:
            break
        scale = 0.02 + 0.04 * rng.random()
        h_pert = best_h + rng.randn(best_n) * scale
        h_pert = project_feasible(h_pert, best_n)
        try:
            res = minimize(
                fun=hard_max_obj, x0=h_pert, jac=hard_max_subgrad,
                args=(best_n,), method='SLSQP',
                bounds=bounds_opt, constraints=cons_opt,
                options={'maxiter': 250, 'ftol': 1e-11}
            )
            h_new = project_feasible(res.x, best_n)
            c5 = compute_c5(h_new, best_n)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5
        except Exception:
            pass

    # Phase 6: Final refinement
    if time_left() > 25:
        try:
            h_fin = slsqp_optimize(best_h, best_n, beta=200.0, maxiter=300)
            c5_fin = compute_c5(h_fin, best_n)
            if c5_fin < best_c5:
                best_h, best_c5 = h_fin, c5_fin
        except Exception:
            pass

    if time_left() > 15:
        try:
            h_fin2, c5_fin2 = projected_subgrad_descent(best_h, best_n, max_iter=1000, init_alpha=0.005)
            if c5_fin2 < best_c5:
                best_h, best_c5 = h_fin2, c5_fin2
        except Exception:
            pass

    if time_left() > 10:
        try:
            res = minimize(
                fun=hard_max_obj, x0=best_h, jac=hard_max_subgrad,
                args=(best_n,), method='SLSQP',
                bounds=bounds_opt, constraints=cons_opt,
                options={'maxiter': 400, 'ftol': 1e-14}
            )
            h_new = project_feasible(res.x, best_n)
            c5 = compute_c5(h_new, best_n)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5
        except Exception:
            pass

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    try:
        val = evaluate_erdos_solution(best_h, best_n)
        if val is not None and 0.0 < val < 1.0:
            best_c5 = min(best_c5, float(val))
    except Exception:
        pass

    return (best_h, best_c5, best_n)
