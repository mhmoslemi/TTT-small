def compute_c5(h):

def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = beta * (x - m)
    shifted = np.clip(shifted, -60.0, 60.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

    """Compute analytical gradient of smooth_max(corr*dx, beta) w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    out = corr * dx

    weights = np.exp(shifted)

    weights /= wsum

    out_A = np.correlate(weights, v, mode='full')

    B = out_B[n-1 : 2*n-1]  # indices n-1, n, ..., 2n-2 (n elements)

def project_sum(h, target):
    """Project h onto {sum(h)=target, 0<=h<=1}"""
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)

    diff = target - s
