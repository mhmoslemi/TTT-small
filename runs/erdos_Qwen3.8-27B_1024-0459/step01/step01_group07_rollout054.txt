import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = h < 1.0 - 1e-13
            cnt = int(np.sum(free))
            if cnt == 0:
                break
            h[free] = np.minimum(h[free] + diff / cnt, 1.0)
        else:
            free = h > 1e-13
            cnt = int(np.sum(free))
            if cnt == 0:
                break
            h[free] = np.maximum(h[free] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def snap_binary(h, n_points):
    """Snap h to binary: top n/2 values become 1, rest 0"""
    n = n_points
    target = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def binary_local_search(h, n_points, max_passes=100):
    """Local search: swap 1s and 0s to reduce max correlation"""
    n = n_points
    dx = 2.0 / n
    h = np.array(h, dtype=float).copy()
    h = snap_binary(h, n)
    
    for _pass in range(max_passes):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Try all swaps
        best_delta = 0
        best_i, best_j = -1, -1
        cur_c5 = compute_c5(h, n)
        
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0, 1
                new_c5 = compute_c5(h, n)
                h[i], h[j] = 1, 0
                delta = new_c5 - cur_c5

                    best_i, best_j = i, j

    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])

    for _pass in range(max_passes):
        best_delta = 1e-15
        best_a, best_b = -1, -1
        for a in ones:
            for b in zeros:
                h[a], h[b] = 0.0, 1.0
                new_c5 = compute_c5(h, n)
                h[a], h[b] = 1.0, 0.0
                delta = new_c5 - cur_c5
                if delta < best_delta - 1e-15:
                    best_delta = delta
                    best_a, best_b = a, b
        if best_a >= 0:
            h[best_a], h[best_b] = 0.0, 1.0
            ones.remove(best_a)
            zeros.remove(best_b)
            ones.append(best_b)
            zeros.append(best_a)
            cur_c5 += best_delta
        else:
            break
    return h, compute_c5(h, n)


def simulated_anneal_binary(h, n_points, rng, n_iter=5000, t0=0.02, t1=0.001):
    n = n_points
    h = np.array(h, dtype=float).copy()
    h = snap_binary(h, n)
    cur_c5 = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = cur_c5
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    best_iter = 0

    for it in range(n_iter):
        temp = t0 * (t1 / t0) ** (it / n_iter)
        a = rng.choice(ones)
        b = rng.choice(zeros)
        h[a], h[b] = 0.0, 1.0
        new_c5 = compute_c5(h, n)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            cur_c5 = new_c5
            ones[ones.index(a)] = b
            zeros[zeros.index(b)] = a
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                best_iter = it
        else:
            h[a], h[b] = 1.0, 0.0
        if it - best_iter > n_iter // 3:
            break
    return best_h, best_c5


def compute_smooth_grad(h, n_points, temp):
    n = n_points
    dx = 2.0 / n
    conv_raw = np.correlate(h, 1.0 - h, mode='full')
    conv = conv_raw * dx
    cmax = np.max(conv)
    exps = np.exp((conv - cmax) / temp)
    w = exps / np.sum(exps)

    n2 = 2 * n - 1
    grad = np.zeros(n)
    h1 = 1.0 - h
    for j in range(n):
        t1 = 0.0
        k_lo = max(0, j)
        k_hi = min(n2 - 1, j + n - 1)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi + 1)
            ia = j - ks + n - 1
            t1 = np.sum(w[ks] * h1[ia])
        t2 = 0.0
        k_lo2 = max(0, n - 1 - j)
        k_hi2 = min(n2 - 1, n2 - 1 - j)
        if k_lo2 <= k_hi2:
            ks = np.arange(k_lo2, k_hi2 + 1)
            ib = j + ks - (n - 1)
            t2 = np.sum(w[ks] * h[ib])
        grad[j] = dx * (t1 - t2)
    return grad


def adam_smoothed_optimize(h0, n_points, rng, n_iter=2000, temp_init=0.05, temp_end=0.005, lr=0.05):
    n = n_points
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    best_h = h.copy()
    best_val = compute_c5(h, n)

    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        temp = temp_init * (temp_end / temp_init) ** frac
        loss = temp * np.log(np.sum(np.exp((compute_conv(h, n) - np.max(compute_conv(h, n))) / temp))) + np.max(compute_conv(h, n))
        g = compute_smooth_grad(h, n, temp)
        t = it + 1
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = project_feasible(h_new, n)
        val = compute_c5(h_new, n)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new

    return best_h, best_val


def structured_init(n, rng, pattern_type=0):
    h = np.zeros(n)
    target = n // 2
    if pattern_type == 0:
        for i in range(n):
            h[i] = 1.0 if (i % 2 == 0) else 0.0
    elif pattern_type == 1:
        block = n // 7
        ones_per = target // 7
        rem = target - ones_per * 7
        count = 0
        for start in range(0, n, block):
            end = min(start + block, n)
            seg = h[start:end]
            n_ones_here = ones_per + (1 if count < rem else 0)
            seg_idx = np.arange(len(seg))
            positions = np.sort(rng.choice(len(seg), n_ones_here, replace=False))
            seg[positions] = 1.0
            h[start:end] = seg
            count += 1
    elif pattern_type == 2:
        mask = np.zeros(n, dtype=bool)
        for i in range(target):
            mask[rng.randint(n)] = True
        h[mask] = 1.0
    else:
        h[:] = 0.5
        h[rng.choice(n, target, replace=False)] = 1.0
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    best_h = None
    best_val = 1.0
    best_n = 70

    if h0 is not None and len(h0) > 0:
        best_n = len(h0)
        h_init = project_feasible(h0, best_n)
        best_h = h_init.copy()
        best_val = compute_c5(best_h, best_n)
    else:
        best_n = 70
        best_h = project_feasible(rng.uniform(0, 1, best_n), best_n)
        best_val = compute_c5(best_h, best_n)

    # Phase 1: Binary local search from initial
    if time_left() > 100:
        h_bin, v_bin = binary_local_search(best_h, best_n, max_passes=100, rng=rng)
        if v_bin < best_val:
            best_val = v_bin
            best_h = h_bin.copy()

    # Phase 2: Continuous optimization from binary
    if time_left() > 200:
        h_cont, v_cont = adam_smoothed_optimize(best_h, best_n, rng, n_iter=3000, temp_init=0.04, temp_end=0.003, lr=0.08)
        if v_cont < best_val:
            best_val = v_cont
            best_h = h_cont.copy()
    if time_left() > 100:
        h_bin2, v_bin2 = binary_local_search(best_h, best_n, max_passes=100, rng=rng)
        if v_bin2 < best_val:
            best_val = v_bin2
            best_h = h_bin2.copy()

    # Phase 3: Multi-resolution exploration
    resolutions = [70, 100, 140, 200, 350]
    for n_try in resolutions:
        if time_left() < 200:
            break
        n_start = n_try
        attempts = 0
        while time_left() > 150 and attempts < 4:
            attempts += 1
            # Different init strategies
            if attempts == 1:
                # Interpolate best
                h_candidate = np.interp(np.linspace(0, 2, n_start, endpoint=False),
                                        np.linspace(0, 2, best_n, endpoint=False), best_h)
                h_candidate = project_feasible(h_candidate, n_start)
            elif attempts == 2:
                h_candidate = structured_init(n_start, rng, 0)
            elif attempts == 3:
                h_candidate = structured_init(n_start, rng, 1)
            else:
                h_candidate = structured_init(n_start, rng, rng.randint(3))

            # Binary local search
            h_b, v_b = binary_local_search(h_candidate, n_start, max_passes=150, rng=rng)
            if v_b < best_val:
                best_val = v_b
                best_h = h_b.copy()
                best_n = n_start

            # Continuous polish
            if time_left() > 120:
                h_c, v_c = adam_smoothed_optimize(h_b, n_start, rng, n_iter=2000, temp_init=0.03, temp_end=0.002, lr=0.06)
                if v_c < best_val:
                    best_val = v_c
                    best_h = h_c.copy()
                    best_n = n_start
                # Re-snap to binary
                h_b2, v_b2 = binary_local_search(h_c, n_start, max_passes=80, rng=rng)
                if v_b2 < best_val:
                    best_val = v_b2
                    best_h = h_b2.copy()
                    best_n = n_start

    # Phase 4: Simulated annealing on best binary solution
    if time_left() > 150:
        h_bin_final = snap_binary(best_h, best_n)
        h_sa, v_sa = simulated_anneal_binary(h_bin_final, best_n, rng, n_iter=8000, t0=0.015, t1=0.0005)
        if v_sa < best_val:
            best_val = v_sa
            best_h = h_sa.copy()

    # Phase 5: Another round of continuous + binary at best resolution
    if time_left() > 200:
        h_c2, v_c2 = adam_smoothed_optimize(best_h, best_n, rng, n_iter=3000, temp_init=0.02, temp_end=0.001, lr=0.04)
        if v_c2 < best_val:
            best_val = v_c2
            best_h = h_c2.copy()
        h_b3, v_b3 = binary_local_search(best_h, best_n, max_passes=200, rng=rng)
        if v_b3 < best_val:
            best_val = v_b3
            best_h = h_b3.copy()

    # Phase 6: Random restarts at best resolution
    n_best = best_n
    n_restarts = 0
    while time_left() > 80 and n_restarts < 20:
        n_restarts += 1
        strat = n_restarts % 4
        if strat == 0:
            h_r = project_feasible(rng.uniform(0, 1, n_best), n_best)
        elif strat == 1:
            h_r = project_feasible(best_h + rng.normal(0, 0.15, n_best), n_best)
        elif strat == 2:
            h_r = structured_init(n_best, rng, rng.randint(2))
        else:
            h_r = project_feasible(rng.binomial(1, 0.5, n_best).astype(float), n_best)
        h_r = snap_binary(h_r, n_best)
        h_r2, v_r2 = binary_local_search(h_r, n_best, max_passes=100, rng=rng)
        if v_r2 < best_val:
            best_val = v_r2
            best_h = h_r2.copy()
        if time_left() > 120:
            h_rc, v_rc = adam_smoothed_optimize(h_r2, n_best, rng, n_iter=1500, temp_init=0.03, temp_end=0.002, lr=0.06)
            if v_rc < best_val:
                best_val = v_rc
                best_h = h_rc.copy()
            h_rb, v_rb = binary_local_search(h_rc, n_best, max_passes=80, rng=rng)
            if v_rb < best_val:
                best_val = v_rb
                best_h = h_rb.copy()

    # Phase 7: Final polish
    if time_left() > 60:
        h_fin, v_fin = adam_smoothed_optimize(best_h, best_n, rng, n_iter=2000, temp_init=0.01, temp_end=0.0005, lr=0.03)
        if v_fin < best_val:
            best_val = v_fin
            best_h = h_fin.copy()
        h_fb, v_fb = binary_local_search(best_h, best_n, max_passes=200, rng=rng)
        if v_fb < best_val:
            best_val = v_fb
            best_h = h_fb.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_val = compute_c5(best_h, best_n)

    return (best_h, best_val, best_n)
