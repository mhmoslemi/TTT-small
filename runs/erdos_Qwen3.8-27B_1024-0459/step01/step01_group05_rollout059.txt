import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    n = len(h)
    comp = 1.0 - h
    return np.correlate(h, comp, mode='full')

def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50, 50)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def get_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50, 50)
    e = np.exp(shifted)
    s = e.sum()
    return e / s

def obj_func(x, beta):
    corr = compute_corr(x)
    return smooth_max(corr, beta)

def grad_func(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = compute_corr(x)
    w = get_weights(corr, beta)
    s_vals = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    one_minus_x = 1.0 - x
    for j in range(n):
        idx1 = j + s_vals
        v1 = (idx1 >= 0) & (idx1 < n)
        t1 = np.dot(w[v1], one_minus_x[idx1[v1]])
        idx2 = j - s_vals
        v2 = (idx2 >= 0) & (idx2 < n)
        t2 = np.dot(w[v2], x[idx2[v2]])
        grad[j] = dx * (t1 - t2)
    return grad

def project_sum(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-13:
        if diff > 0:
            free = np.where(h < 1.0 - 1e-15)[0]
            if len(free) > 0:
                per = diff / len(free)
                h[free] = np.minimum(h[free] + per, 1.0)

            if len(free) > 0:
                per = -diff / len(free)
                h[free] = np.maximum(h[free] - per, 0.0)
    # Final exact correction
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-15 and len(h) > 0:
        mid = len(h) // 2
        h[mid] = np.clip(h[mid] + diff, 0.0, 1.0)
    return h

def ceq_fun(x, target):
    return np.sum(x) - target

def ceq_jac(x, target):
    return np.ones(len(x))

    bounds = [(0.0, 1.0)] * n

def binary_swap_search(h_bin, n, max_sweeps):
    """Local search: try all pair swaps, accept best improvement per sweep."""
    target = n // 2
    h = h_bin.copy()

        best_improvement = 0
        best_oi = -1
        best_zi = -1

        # Then for each swap, compute the change
