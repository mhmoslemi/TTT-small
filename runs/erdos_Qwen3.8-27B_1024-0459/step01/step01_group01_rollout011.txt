import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, tau):
    m = np.max(corr)
    return float(m + tau * np.log(np.sum(np.exp((corr - m) / tau))))

def smooth_max_and_weights(corr, tau):
    m = np.max(corr)
    w = np.exp((corr - m) / tau)
    w /= w.sum()
    val = m + tau * np.log(np.sum(np.exp((corr - m) / tau)))
    return val, w

def compute_grad(h, tau):
    n = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    _, w = smooth_max_and_weights(corr, tau)
    shifts = np.arange(2 * n - 1) - (n - 1)
    M = np.arange(n)[:, None]
    S = shifts[None, :]
    j1 = np.clip(M + S, 0, n - 1)
    valid1 = (M + S >= 0) & (M + S < n)
    term1 = (valid1 * w[None, :] * one_minus_h[j1]).sum(axis=1)
    j2 = np.clip(M - S, 0, n - 1)
    valid2 = (M - S >= 0) & (M - S < n)
    term2 = (valid2 * w[None, :] * h[j2]).sum(axis=1)
    return term1 - term2

def project_onto_set(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.astype(float).sum()
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def upsample_to(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def nesterov_optimize(h0, n, budget_frac, rng, t0, budget_s):
    deadline = t0 + budget_frac * budget_s
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_val = smooth_max_val(np.correlate(h, 1.0 - h, mode="full"), 0.1)
    mu = 0.9
    tau = 0.1
    step = 1.0
    n_iter = 0
    stalled = 0
    
    while time.time() < deadline:
        tau = max(0.002, tau * 0.999)
        corr = np.correlate(h, 1.0 - h, mode="full")
        val, _ = smooth_max_and_weights(corr, tau)
        
        if val < best_val:
            best_val = val
            best_h = h.copy()
            stalled = 0
        else:
            stalled += 1
            if stalled > 200:
                h = best_h.copy()
                v *= 0.5
                tau *= 1.5
                tau = min(tau, 0.1)
                stalled = 0
        
        grad = compute_grad(h, tau)
        v = mu * v + grad
        gnorm = np.linalg.norm(grad)
        
        if gnorm < 1e-8:
            break
        
        yk = h - step * v
        yk = project_onto_set(yk, n)
        new_val = smooth_max_val(np.correlate(yk, 1.0 - yk, mode="full"), tau)
        
        tries = 0
        while new_val > val + 1e-10 and tries < 15:

def slsqp_optimize(h0, n, maxiter, tau):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        return float(m + tau * np.log(np.sum(np.exp((corr - m) / tau)))) * 1.0
    
    def con_eq(x):
        return np.array([x.sum() - n / 2.0])
    
    def con_jac(x):
        return np.ones((1, n))
    
    try:
        res = minimize(
            fun=obj, x0=h0, method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': con_eq, 'jac': con_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False}
        )
        h = np.clip(res.x, 0.0, 1.0)
        return project_onto_set(h, n)
    except Exception:
        return h0

    sorted_idx = np.argsort(h)
    half = n // 2
    h_bin = np.zeros(n)
    for i in range(half, n):
        h_bin[sorted_idx[i]] = 1.0
    return h_bin

    return (best_h, best_c5, len(best_h))
