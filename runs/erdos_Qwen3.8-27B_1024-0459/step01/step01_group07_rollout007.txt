def c5_smooth_gradient_part(h, eps, topk=0):
    n=len(h)
    dx=2.0/n
    c=np.correlate(h, 1.0-h, mode='full')
    mx=c.max()
    if not np.isfinite(mx):
        return np.zeros(n)
    Larr=c.size
    if topk is None or topk <=0 or topk >= Larr:
        idx=np.arange(Larr)
        vals=c[idx]
    else:
        k=int(min(topk,Larr))
        idx=np.argpartition(c, Larr-k)[-k:]
        vals=c[idx]
    w=np.exp((vals - mx) / float(eps))
    s=w.sum()
    if s <= 0 or not np.isfinite(s):
        # uniform over idx
        w=np.ones(idx.size)/idx.size
    else:
        w=w/s
    grad=np.zeros(n)
    for ii in range(idx.size):
        wi=float(w[ii])
        if wi <= 1e-18: continue
        m=int(idx[ii]) - (n-1)
        if m >= 0:
            L=n-m
            if L <= 0: continue
            # source m:n, target 0:L
            grad[m:n] += wi * dx * (1.0 - h[0:L])
            grad[0:L] += -wi * dx * h[m:n]
        else:
            p=-m
            L=n-p
            if L <=0: continue
            grad[0:L] += wi * dx * (1.0 - h[p:n])
            grad[p:n] += -wi * dx * h[0:L]
    return grad

def fix_full_sum(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) <= 1e-14:
            return h
        if diff > 0:
            free = np.where(h < 1.0 - 1e-14)[0]
            if free.size == 0:
                break
            room = 1.0 - h[free]
            add = np.minimum(room, diff / float(free.size))
            h[free] += add
        else:
            free = np.where(h > 1e-14)[0]

        else:
            idx = int(np.argmax(h))

def fix_full_sum(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(100):
        diff = target - float(np.sum(h))
        if abs(diff) <= 1e-14:
            return h
        if diff > 0.0:
            free = np.where(h < 1.0 - 1e-14)[0]
            if free.size == 0:
                break
            add = diff / float(free.size)
            room = 1.0 - h[free]
            add = min(add, float(np.min(room)))
            h[free] += add
        else:
            free = np.where(h > 1e-14)[0]
            if free.size == 0:
                break

            room = h[free]
            sub = min(sub, float(np.min(room)))

    diff = target - float(np.sum(h))

            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)

            else:
                if u[L] > 1e-14:
                    dec = min(u[L], -diff)
                    u[L] -= dec
                    continue
                free = np.where(u[:L] > 1e-14)[0]
                if free.size == 0:
                    break
                sub = -diff / (2.0 * float(free.size))
                room = u[free]
                sub = min(sub, float(np.min(room)))
                u[free] -= sub
        else:
            s = 2.0 * float(np.sum(u))
            diff = target - s
            if abs(diff) <= 1e-14:
                return u

            if diff > 0.0:
                free = np.where(u < 1.0 - 1e-14)[0]
                if free.size == 0:
                    break
                add = diff / (2.0 * float(free.size))
                room = 1.0 - u[free]
                add = min(add, float(np.min(room)))
                u[free] += add
            else:
                free = np.where(u > 1e-14)[0]
                if free.size == 0:
                    break
                sub = -diff / (2.0 * float(free.size))
                room = u[free]
                sub = min(sub, float(np.min(room)))
                u[free] -= sub

    if u.size == L + 1:
        diff = target - (2.0 * float(np.sum(u[:L])) + float(u[L]))
    else:
        diff = target - 2.0 * float(np.sum(u))

    if abs(diff) > 1e-12:
        if u.size == L + 1:
            if diff > 0.0:
                if u[L] < 1.0:
                    u[L] = min(1.0, u[L] + diff)
                else:
                    idx = int(np.argmax(1.0 - u[:L]))
                    u[idx] = min(1.0, u[idx] + 0.5 * diff)
            else:
                if u[L] > 0.0:
                    u[L] = max(0.0, u[L] + diff)
                else:
                    idx = int(np.argmax(u[:L]))
                    u[idx] = max(0.0, u[idx] + 0.5 * diff)
        else:
            if diff > 0.0:
                idx = int(np.argmax(1.0 - u))
                u[idx] = min(1.0, u[idx] + 0.5 * diff)
            else:
                idx = int(np.argmax(u))
                u[idx] = max(0.0, u[idx] + 0.5 * diff)
    return u


def project_sym(z, L, target):
    z = np.asarray(z, dtype=float)
    z = np.where(np.isfinite(z), z, 0.5)

    if L == 0:
        if z.size == 1:
            u = np.array([min(1.0, max(0.0, float(z[0])))], dtype=float)
        else:
            u = np.array([min(1.0, max(0.0, float(target)))], dtype=float)
        return fix_sym_sum(u, 0, target)

    if z.size == L + 1:
        pairs = z[:L]
        center = z[L]
        lo = float(np.min(z)) - 1.0
        hi = float(np.max(z)) + 1.0
        for _ in range(70):
            lam = 0.5 * (lo + hi)
            s = (2.0 * float(np.sum(np.clip(pairs - 2.0 * lam, 0.0, 1.0))) +
                 float(np.clip(center - lam, 0.0, 1.0)))
            if s > target:
                lo = lam
            else:
                hi = lam
        lam = 0.5 * (lo + hi)
        u = np.empty(L + 1, dtype=float)
        u[:L] = np.clip(pairs - 2.0 * lam, 0.0, 1.0)
        u[L] = np.clip(center - lam, 0.0, 1.0)
    else:
        pairs = z
        lo = float(np.min(pairs)) - 1.0
        hi = float(np.max(pairs)) + 1.0
        for _ in range(70):
            lam = 0.5 * (lo + hi)
            s = 2.0 * float(np.sum(np.clip(pairs - 2.0 * lam, 0.0, 1.0)))
            if s > target:
                lo = lam
            else:
                hi = lam
        lam = 0.5 * (lo + hi)
        u = np.clip(pairs - 2.0 * lam, 0.0, 1.0)

    return fix_sym_sum(u, L, target)


def full_from_sym(u, n):
    u = np.asarray(u, dtype=float)
    m = n // 2
    h = np.empty(n, dtype=float)
    if n % 2 == 1:
        h[:m] = u[:m]
        h[m] = u[m]
        h[m + 1:] = u[:m][::-1]
    else:
        h[:m] = u
        h[m:] = u[::-1]
    return h


def sym_from_full(h, n):
    h = np.asarray(h, dtype=float)
    m = n // 2
    target = 0.5 * float(n)
    if n % 2 == 1:
        u = np.empty(m + 1, dtype=float)
        if m > 0:
            u[:m] = 0.5 * (h[:m] + h[m + 1:][::-1])
        u[m] = h[m]
    else:
        u = 0.5 * (h[:m] + h[m:][::-1])
    return project_sym(u, m, target)


def resample_full(h_old, n_new):
    h_old = np.asarray(h_old, dtype=float)
    n_old = h_old.size
    if n_old == 0:
        return np.full(n_new, 0.5)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def sum_full_fun(x, target):
    return float(np.sum(np.asarray(x, dtype=float))) - target


def sum_full_jac(x, target):
    return np.ones_like(np.asarray(x, dtype=float), dtype=float)


def sym_sum_even(x, target):
    x = np.asarray(x, dtype=float)
    return 2.0 * float(np.sum(x)) - target


def sym_sum_even_jac(x, target):
    x = np.asarray(x, dtype=float)
    return 2.0 * np.ones_like(x, dtype=float)


def sym_sum_odd(x, target):
    x = np.asarray(x, dtype=float)
    if x.size == 0:
        return -target
    return 2.0 * float(np.sum(x[:-1])) + float(x[-1]) - target


def sym_sum_odd_jac(x, target):
    x = np.asarray(x, dtype=float)
    j = np.full(x.size, 2.0, dtype=float)
    if x.size > 0:
        j[-1] = 1.0
    return j


def c5_logsum_value(h, eps):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 1e100
    dx = 2.0 / float(n)
    c = c5_count_array(h) * dx
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return 1e100
    eps = max(float(eps), 1e-8)
    val = eps * (float(np.log(np.sum(np.exp((c - mx) / eps)))) + mx / eps)
    return val if np.isfinite(val) else 1e100


def c5_smooth_gradient_part(h, eps, topk=0):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return np.zeros(0, dtype=float)
    dx = 2.0 / float(n)
    c_raw = c5_count_array(h)
    c = c_raw * dx
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.zeros(n, dtype=float)

    size = c.size
    if topk is None or topk <= 0 or topk >= size:
        idx = np.arange(size)
        vals = c[idx]
    else:
        k = int(min(max(1, int(topk)), size))
        idx = np.argpartition(c, size - k)[-k:]
        vals = c[idx]

    eps = max(float(eps), 1e-8)
    w = np.exp((vals - mx) / eps)
    sw = float(np.sum(w))
    if sw <= 0.0 or not np.isfinite(sw):
        w = np.ones(idx.size, dtype=float) / float(idx.size)
    else:
        w = w / sw

    grad = np.zeros(n, dtype=float)
    for ii in range(idx.size):
        wi = float(w[ii])
        if wi <= 1e-18:
            continue
        m = int(idx[ii]) - (n - 1)
        if m >= 0:
            L = n - m
            if L <= 0:
                continue
            grad[m:n] += wi * dx * (1.0 - h[0:L])
            grad[0:L] += -wi * dx * h[m:n]
        else:
            p = -m
            L = n - p
            if L <= 0:
                continue
            grad[0:L] += wi * dx * (1.0 - h[p:n])
            grad[p:n] += -wi * dx * h[0:L]
    return grad


def sym_gradient_from_full(gfull, n):
    gfull = np.asarray(gfull, dtype=float)
    m = n // 2
    if n % 2 == 1:
        g = np.empty(m + 1, dtype=float)
        if m > 0:
            g[:m] = gfull[:m] + gfull[m + 1:][::-1]
        g[m] = gfull[m]
    else:
        g = gfull[:m] + gfull[m:][::-1]
    return g


def sym_smooth_obj(u, n, eps):
    u = np.asarray(u, dtype=float)
    u = np.clip(u, 0.0, 1.0)
    h = full_from_sym(u, n)
    return c5_logsum_value(h, eps)


def sym_smooth_grad(u, n, eps, topk=0):
    u = np.asarray(u, dtype=float)
    u = np.clip(u, 0.0, 1.0)
    h = full_from_sym(u, n)
    gfull = c5_smooth_gradient_part(h, eps, topk)
    return sym_gradient_from_full(gfull, n)


def sym_slsqp(u0, n, maxiter=100, eps=0.05, topk=0):
    L = n // 2
    target = 0.5 * float(n)
    u = project_sym(u0, L, target)
    bounds = [(0.0, 1.0)] * u.size

    if n % 2 == 0:
        constr = [
            {
                "type": "eq",
                "fun": sym_sum_even,
                "jac": sym_sum_even_jac,
                "args": (target,),
            }
        ]
    else:
        constr = [
            {
                "type": "eq",
                "fun": sym_sum_odd,
                "jac": sym_sum_odd_jac,
                "args": (target,),
            }
        ]

    try:
        res = minimize(
            sym_smooth_obj,
            u,
            method="SLSQP",
            jac=sym_smooth_grad,
            args=(n, eps, topk),
            bounds=bounds,
            constraints=constr,
            options={"maxiter": int(maxiter), "ftol": 1e-12, "maxls": 30},
        )
        u = project_sym(res.x, L, target)
    except Exception:
        u = project_sym(u, L, target)
    return u


def proposed_sym_gradient(u0, n, deadline, iters=1000, lr_init=0.2, lr_max=5.0, topk=0, eps=0.02):
    L = n // 2
    target = 0.5 * float(n)
    u = project_sym(u0, L, target)
    h = full_from_sym(u, n)
    val = c5_exact(h)
    best_h = h.copy()
    best_val = val

    lr = float(min(max(lr_init, 1e-8), lr_max))
    stall = 0

    for _ in range(int(iters)):
        if time.time() >= float(deadline):
            break

        gfull = c5_smooth_gradient_part(h, eps, topk)
        g = sym_gradient_from_full(gfull, n)

        accepted = False
        local_lr = lr
        for _line in range(8):
            u_new = u - local_lr * g
            u_new = project_sym(u_new, L, target)
            h_new = full_from_sym(u_new, n)
            val_new = c5_exact(h_new)
            if np.isfinite(val_new) and val_new < val - 1e-14:
                u = u_new
                h = h_new
                val = val_new
                if val < best_val - 1e-14:
                    best_val = val
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                lr = float(min(local_lr * 1.15, lr_max))
                accepted = True
                break
            local_lr *= 0.45
            if local_lr < 1e-9:
                break

        if not accepted:
            lr = float(max(1e-9, lr * 0.5))
            stall += 1
            if stall > 250:
                break
        if stall > 1000:
            break

    return best_h, float(best_val)

def sym_binary_from_values(vals, L, need, rng):
    vals = np.asarray(vals, dtype=float)
    if vals.size < L:
        vals = rng.uniform(0.0, 1.0, L)
    vals = vals[:L].copy()
    vals = np.where(np.isfinite(vals), vals, 0.5)

    if need <= 0:
        return np.zeros(L, dtype=float)
    if need >= L:
        return np.ones(L, dtype=float)

    score = vals + rng.normal(0.0, 1e-7, size=L)
    order = np.argsort(-score)
    u = np.zeros(L, dtype=float)
    u[order[:need]] = 1.0
    return u


def run_template_values(L, a, b):
    if L <= 0:
        return np.zeros(0, dtype=float)
    a = max(1, int(a))
    b = max(1, int(b))
    period = a + b
    unit = np.concatenate((np.ones(a, dtype=float), np.zeros(b, dtype=float)))
    reps = (L // period) + 2
    return np.tile(unit, reps)[:L]


def make_structured_pairs(L, need, mode, rng):
    if L <= 0:
        return np.zeros(0, dtype=float)
    if need <= 0:
        return np.zeros(L, dtype=float)
    if need >= L:
        return np.ones(L, dtype=float)

    if mode == 0:
        u = np.zeros(L, dtype=float)
        u[:need] = 1.0
        return u

    if mode == 1:
        u = np.zeros(L, dtype=float)
        idx = rng.choice(L, need, replace=False)
        u[idx] = 1.0
        return u

    if mode == 2:
        vals = (np.arange(L, dtype=float) % 2.0 == 0.0).astype(float)
        return sym_binary_from_values(vals, L, need, rng)

    if mode == 3:
        vals = np.zeros(L, dtype=float)
        for i in range(L):
            vals[i] = float(bin(i).count("1") % 2)
        return sym_binary_from_values(vals, L, need, rng)

    if mode == 4:
        t = np.arange(L, dtype=float)
        vals = 0.5 + 0.5 * np.cos(2.0 * np.pi * t / float(L))
        return sym_binary_from_values(vals, L, need, rng)

    if mode == 5:
        t = np.arange(L, dtype=float)
        ph1 = rng.uniform(0.0, 2.0 * np.pi)
        ph2 = rng.uniform(0.0, 2.0 * np.pi)
        f1 = float(rng.randint(2, 6))
        f2 = float(rng.randint(6, 13))
        vals = 0.5 + 0.30 * np.cos(2.0 * np.pi * f1 * t / float(L) + ph1) + \
               0.20 * np.sin(2.0 * np.pi * f2 * t / float(L) + ph2)
        return sym_binary_from_values(vals, L, need, rng)

    if mode == 6:
        vals = run_template_values(L, 3, 3)
        return sym_binary_from_values(vals, L, need, rng)

    if mode == 7:
        vals = run_template_values(L, 4, 4)
        return sym_binary_from_values(vals, L, need, rng)

    if mode == 8:
        vals = run_template_values(L, 5, 5)
        return sym_binary_from_values(vals, L, need, rng)

    if mode == 9:
        vals = run_template_values(L, 2, 3)
        return sym_binary_from_values(vals, L, need, rng)

    if mode == 10:
        vals = run_template_values(L, 5, 4)
        return sym_binary_from_values(vals, L, need, rng)

    u = np.zeros(L, dtype=float)
    idx = rng.choice(L, need, replace=False)
    u[idx] = 1.0
    return u


def pos_to_pair(p, n):
    L = n // 2
    if p < L:
        return int(p)
    if p > L:
        return int(n - 1 - p)
    return -1


def choose_directed_sym(u, n, cur_c, h, rng):
    L = n // 2
    if L <= 0:
        return -1, -1

    size = cur_c.size
    if size == 0:
        return -1, -1

    k = min(8, size)
    if k < size:
        idxs = np.argpartition(cur_c, size - k)[-k:]
    else:
        idxs = np.arange(size)

    for _ in range(4):
        idx = int(idxs[int(rng.random() * idxs.size)])
        m = int(idx - (n - 1))

        if m >= 0:
            Lg = n - m
            if Lg <= 0:
                continue
            tgt = np.arange(Lg)
            src = m + tgt
        else:
            p = -m
            Lg = n - p
            if Lg <= 0:
                continue
            src = np.arange(Lg)
            tgt = p + src

        mask = (h[src] > 0.5) & (h[tgt] < 0.5)
        if not np.any(mask):
            continue

        nz = np.nonzero(mask)[0]
        r0 = int(nz[int(rng.random() * nz.size)])
        s = int(src[r0])
        t = int(tgt[r0])

        pi = pos_to_pair(s, n)
        pj = pos_to_pair(t, n)

        if pi >= 0 and pj >= 0 and pi != pj and u[pi] > 0.5 and u[pj] < 0.5:
            return pi, pj

    return -1, -1


def perturb_sym(u_prev, n, strength, rng):
    L = n // 2
    u = np.array(u_prev, dtype=float, copy=True)
    if L <= 0:
        h = full_from_sym(u, n)
        return u, h, c5_count_array(h)

    ones = np.where(u[:L] > 0.5)[0]
    zeros = np.where(u[:L] < 0.5)[0]
    k = min(int(strength), ones.size, zeros.size)
    if k > 0:
        oi = ones[rng.choice(ones.size, k, replace=False)]
        oz = zeros[rng.choice(zeros.size, k, replace=False)]
        for a, b in zip(oi, oz):
            u[int(a)] = 0.0
            u[int(b)] = 1.0

    u = project_sym(u, L, 0.5 * float(n))
    h = full_from_sym(u, n)
    return u, h, c5_count_array(h)


def sym_binary_search(pairs0, n, rng, deadline, max_evals=20000, temp0=0.02, tempf=1e-5):
    L = n // 2
    need = L // 2
    if n % 2 != 1 or L % 2 != 0:
        if n % 2 == 1:
            need = L // 2
        else:
            need = int(round(0.5 * float(n) // 2))
        u = np.empty(n // 2 + (1 if n % 2 else 0), dtype=float)
        if pairs0 is not None:
            pairs = sym_binary_from_values(pairs0, L, need, rng)
            u[:L] = pairs
            if n % 2 == 1:
                u[L] = 0.5
        else:
            u[:] = 0.5
        h = full_from_sym(u, n)
        h = project_full(h)
        return h, c5_exact(h)

    if pairs0 is None or len(pairs0) != L:
        pairs = make_structured_pairs(L, need, 1, rng)
    else:
        pairs = np.asarray(pairs0, dtype=float)
        pairs = np.where(np.isfinite(pairs), pairs, 0.0)
        rounded = np.round(np.clip(pairs, 0.0, 1.0))
        if int(round(pairs.sum())) == need:
            pairs = rounded
        else:
            pairs = sym_binary_from_values(pairs, L, need, rng)

    u = np.empty(L + 1, dtype=float)
    u[:L] = pairs
    u[L] = 0.5

    h = full_from_sym(u, n)
    cur_c = c5_count_array(h)
    curC = float(np.max(cur_c) * (2.0 / float(n)))

    best_h = h.copy()
    best_u = u.copy()
    bestC = curC

    cool = max(0.90, (float(tempf) / float(temp0)) ** (1.0 / float(max(1, int(max_evals)))))
    T = float(temp0)
    max_evals = int(max(500, int(max_evals)))
    attempts = 0
    no_best_improve = 0
    perturb_tries = 0
    strength = 3

    while time.time() < float(deadline) and attempts < max_evals:
        attempts += 1
        pairs = u[:L]

        ones = np.where(pairs > 0.5)[0]
        zeros = np.where(pairs < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break

        i = -1
        j = -1
        if rng.random() < 0.78:
            for _ in range(4):
                ii, jj = choose_directed_sym(u, n, cur_c, h, rng)
                if ii >= 0 and jj >= 0 and ii != jj:
                    i = ii
                    j = jj
                    break

        if i < 0 or j < 0:
            i = int(ones[int(rng.random() * ones.size)])
            j = int(zeros[int(rng.random() * zeros.size)])

        if i == j:
            continue

        old_i = u[i]
        old_j = u[j]
        if old_i <= 0.5 or old_j >= 0.5:
            if old_j < 0.5 and old_i > 0.5:
                pass
            elif old_i < 0.5 and old_j > 0.5:
                i, j = j, i
                old_i, old_j = old_j, old_i
            else:
                continue

        u[i] = 0.0
        u[j] = 1.0

        pi1 = i
        pi2 = n - 1 - i
        pj1 = j
        pj2 = n - 1 - j

        h[pi1] = 0.0
        h[pi2] = 0.0
        h[pj1] = 1.0
        h[pj2] = 1.0

        cand_c = c5_count_array(h)
        candC = float(np.max(cand_c) * (2.0 / float(n)))
        delta = candC - curC

        accept = False
        if delta < -1e-15:
            accept = True
        else:
            if T > tempf * 0.5:
                if rng.random() < np.exp(-max(0.0, delta) / T):
                    accept = True

        if accept:
            cur_c = cand_c
            curC = candC
            if candC < bestC - 1e-14:
                bestC = candC
                best_h = h.copy()
                best_u = u.copy()
                no_best_improve = 0
            else:
                no_best_improve += 1
            T = max(tempf, T * cool)
        else:
            u[i] = old_i
            u[j] = old_j
            h[pi1] = old_i
            h[pi2] = old_i
            h[pj1] = old_j
            h[pj2] = old_j
            T = max(tempf, T * cool)

        if perturb_tries < 6 and no_best_improve > max(2500, attempts // 4):
            strength = min(int(strength) + 2, max(2, L // 4))
            u, h, cur_c = perturb_sym(best_u, n, strength, rng)
            pairs = u[:L]
            if n % 2 == 1:
                u[L] = project_sym(u, L, 0.5 * float(n))[L]
            h = full_from_sym(u, n)
            cur_c = c5_count_array(h)
            curC = float(np.max(cur_c) * (2.0 / float(n)))
            no_best_improve = 0
            T = max(T, temp0 * 0.75)
            perturb_tries += 1
        elif no_best_improve > 60000:
            break

    h_out = full_from_sym(best_u, n)
    c_out = c5_exact(h_out)
    return h_out, float(c_out)


def polish_symmetry(n, h_best, global_deadline, alloc, rng):
    if n <= 0 or alloc <= 1.0:
        h_best = project_full(np.asarray(h_best, dtype=float))
        return h_best, c5_exact(h_best)

    L = n // 2
    target = 0.5 * float(n)

    u = sym_from_full(h_best, n)
    h = full_from_sym(project_sym(u, L, target), n)
    best_h = h.copy()
    best_c = c5_exact(h)

    start = time.time()
    phase_end = min(float(global_deadline), start + float(alloc))

    base_iters = 1400 if n <= 80 else 700 if n <= 160 else 350
    if n <= 160:
        gd_settings = [
            (0, base_iters, 0.005, 0.2),
            (0, base_iters, 0.01, 0.5),
            (0, base_iters // 2, 0.05, 1.0),
            (0, base_iters // 2, 0.02, 0.3),
        ]
    else:
        gd_settings = [
            (80, base_iters, 0.005, 0.2),
            (80, base_iters, 0.01, 0.5),
            (80, base_iters // 2, 0.05, 1.0),
            (100, base_iters // 2, 0.02, 0.3),
        ]

    for topk, iters, eps, lr in gd_settings:
        now = time.time()
        if now >= phase_end or phase_end - now <= 2.0:
            break
        inner_allow = min(12.0, max(1.0, phase_end - now))
        dl = min(float(global_deadline), now + inner_allow)
        try:
            h2, v2 = proposed_sym_gradient(
                u, n, dl, iters=iters, lr_init=lr, lr_max=5.0, topk=topk, eps=eps
            )
            if v2 < best_c - 1e-14:
                best_c = v2
                best_h = h2.copy()
                u = project_sym(sym_from_full(h2, n), L, target)
        except Exception:
            pass

    slsqp_settings = [(0.10, 70), (0.05, 90), (0.02, 110), (0.01, 130)]
    slsqp_topk = 0
    for eps, maxit in slsqp_settings:
        now = time.time()
        if now >= phase_end - 4.0:
            break
        if global_deadline - now < 12.0:
            break
        if n > 260 and maxit > 80:
            maxit = 70
        try:
            u2 = sym_slsqp(u, n, maxiter=int(maxit), eps=eps, topk=slsqp_topk)
            h2 = full_from_sym(u2, n)
            v2 = c5_exact(h2)
            if v2 < best_c - 1e-14:
                best_c = v2
                best_h = h2.copy()
                u = project_sym(u2, L, target)
        except Exception:
            pass

    best_h = project_full(best_h)
    best_c = c5_exact(best_h)
    return best_h, float(best_c)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget_s = float(max(5.0, budget_s))
    global_deadline = t0 + max(1.0, budget_s - 5.0)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = np.inf
    best_n = 0

    parent = None
    try:
        parent = np.asarray(initial_h_values, dtype=float)
    except Exception:
        parent = None

    if parent is not None and parent.size > 0:
        h0 = project_full(parent)
        c0 = c5_exact(h0)
        if np.isfinite(c0):
            best_h = h0.copy()
            best_c = c0
            best_n = h0.size
    else:
        h0 = np.full(70, 0.5, dtype=float)
        best_h = h0.copy()
        best_c = c5_exact(h0)
        best_n = h0.size

    if best_h is None or not np.isfinite(best_c):
        best_n = 70
        best_h = np.full(best_n, 0.5, dtype=float)
        best_c = c5_exact(best_h)

    if remaining_time(t0, budget_s) > 55.0:
        try:
            h2, v2 = projected_full_gradient(
                best_h, best_n,
                deadline=min(global_deadline, time.time() + 35.0),
                iters=2500, lr_init=0.25, topk=0, eps=0.02
            )
            if v2 < best_c - 1e-13:
                best_c = v2
                best_h = h2.copy()
        except Exception:
            pass

    if remaining_time(t0, budget_s) > 55.0:
        try:
            u2 = sym_from_full(best_h, best_n)
            h2 = full_from_sym(u2, best_n)
            h3, v3 = polish_symmetry(best_n, h2, global_deadline, 35.0, rng)
            if v3 < best_c - 1e-13:
                best_c = v3
                best_h = h3.copy()
        except Exception:
            pass

    odd_ns = [73, 145, 217, 289]
    for n in odd_ns:
        rem = remaining_time(t0, budget_s)
        if rem < 55.0:
            break

        L = n // 2
        if L % 2 != 0:
            continue

        if n <= 80:
            alloc = min(rem * 0.34, rem - 25.0)
            num_structures = 10
        elif n <= 160:
            alloc = min(rem * 0.24, rem - 25.0)
            num_structures = 8
        else:
            alloc = min(rem * 0.16, rem - 20.0)
            num_structures = 6
        if alloc < 18.0:
            continue

        local_h = None
        local_c = np.inf

        starts = []
        if best_h is not None:
            h_r = resample_full(best_h, n)
            u_r = sym_from_full(h_r, n)
            starts.append(sym_binary_from_values(u_r[:L], L, L // 2, rng))

        for mode in range(num_structures):
            starts.append(make_structured_pairs(L, L // 2, mode, rng))

        per_start = max(2.0, float(alloc) / float(max(1, len(starts))))
        for start_pairs in starts:
            now = time.time()
            if now >= global_deadline:
                break
            this_alloc = min(per_start, global_deadline - now)
            if this_alloc < 1.0:
                break
            deadline = now + this_alloc
            max_evals = max(1500, int(2500000 // n))
            try:
                h_c, v_c = sym_binary_search(
                    start_pairs, n, rng, deadline,
                    max_evals=max_evals, temp0=0.02, tempf=1e-5
                )
                if np.isfinite(v_c) and v_c < local_c:
                    local_c = v_c
                    local_h = h_c
            except Exception:
                pass

        if local_h is not None:
            if local_c < best_c - 1e-14:
                best_c = local_c
                best_h = local_h.copy()
                best_n = n

            if local_c < best_c + 0.004:
                rem2 = remaining_time(t0, budget_s)
                if rem2 > 35.0:
                    try:
                        h_p, v_p = polish_symmetry(n, local_h, global_deadline,
                                                   max(10.0, min(60.0, rem2 * 0.35)), rng)
                        if v_p < best_c - 1e-14:
                            best_c = v_p
                            best_h = h_p.copy()
                            best_n = n
                    except Exception:
                        pass

    rem = remaining_time(t0, budget_s)
    if rem > 70.0 and best_n > 0:
        try:
            u_final = sym_from_full(best_h, best_n)
            h_final = full_from_sym(u_final, best_n)
            hh, vv = polish_symmetry(best_n, h_final, global_deadline,
                                     max(20.0, min(180.0, rem * 0.55)), rng)
            if vv < best_c - 1e-14:
                best_c = vv
                best_h = hh.copy()
                best_n = best_n
        except Exception:
            pass

    try:
        best_h = np.asarray(best_h, dtype=float).copy()
        best_h = project_full(best_h)
        best_c = c5_exact(best_h)
        best_n = int(best_h.size)
    except Exception:
        best_n = 70
        best_h = np.full(best_n, 0.5, dtype=float)
        best_c = c5_exact(best_h)

    return best_h, float(best_c), int(best_n)
