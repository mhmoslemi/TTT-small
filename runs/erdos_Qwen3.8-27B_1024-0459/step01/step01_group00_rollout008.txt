def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    for _ in range(5):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            mask = h < 1.0 - 1e-14
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            h[mask] += diff / float(cnt)
        else:
            mask = h > 1e-14
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            h[mask] += diff / float(cnt)
        h = np.clip(h, 0.0, 1.0)
    diff = target - float(np.sum(h))
    guard=0
    while abs(diff) > 1e-14 and guard < 10:
       guard +=1
       if diff > 0:
          idxs = np.nonzero(h < 1.0 - 1e-14)[0]
       else:
          idxs = np.nonzero(h > 1e-14)[0]
       if idxs.size == 0: break
       for idx in idxs:
          if abs(diff) <= 1e-14: break
          if diff > 0:
             room = 1.0 - h[idx]
             if room <= 1e-15: continue
             add = room if room < diff else diff
             if add > 1e-16:
                h[idx] += add
                diff -= add
          else:
             room = h[idx]
             if room <= 1e-15: continue
             sub = room if room < -diff else -diff
             if sub > 1e-16:
                h[idx] -= sub
                diff += sub
    return h

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)

    for _ in range(5):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            mask = h < 1.0 - 1e-14
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            h[mask] += diff / float(cnt)
        else:
            mask = h > 1e-14
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            h[mask] += diff / float(cnt)
        h = np.clip(h, 0.0, 1.0)

    diff = target - float(np.sum(h))
    guard = 0
    while abs(diff) > 1e-14 and guard < 12:
        guard += 1
        if diff > 0.0:
            idxs = np.nonzero(h < 1.0 - 1e-14)[0]
        else:
            idxs = np.nonzero(h > 1e-14)[0]
        if idxs.size == 0:
            break

        for idx in idxs:
            if abs(diff) <= 1e-14:
                break
            if diff > 0.0:
                room = 1.0 - h[idx]
                if room <= 1e-15:
                    continue
                add = room if room < diff else diff
                if add > 1e-16:
                    h[idx] += add
                    diff -= add
            else:
                room = h[idx]
                if room <= 1e-15:
                    continue
                sub = room if room < (-diff) else (-diff)
                if sub > 1e-16:
                    h[idx] -= sub
                    diff += sub

    return h


def eval_h(h, n):
    hh = make_feasible(h, n)
    c = np.correlate(hh, 1.0 - hh, mode="full")
    m = float(np.max(c))
    c5 = m * (2.0 / float(n))
    return hh, c, m, c5


def fallback_h(n):
    x = np.zeros(n, dtype=float)
    vals = ((np.arange(n, dtype=float) * 7919.0) % float(n)) / float(max(n, 1))
    order = np.argsort(vals)
    k = int(round(0.5 * float(n)))
    x[order[:k]] = 1.0
    return make_feasible(x, n)


def make_binary_from_frac(h, n):
    x = np.zeros(n, dtype=float)
    k = int(n // 2)
    if k > 0:
        order = np.argsort(h)
        x[order[-k:]] = 1.0
    return x


def upsample_nearest(h, n):
    m = len(h)
    if m == 1:
        return np.full(n, float(h[0]), dtype=float)
    idx = (np.arange(n, dtype=np.int64) * int(m)) // int(n)
    return h[idx].astype(float).copy()


def upsample_repeat(h, n):
    m = len(h)
    if m > 0 and n % m == 0:
        return np.repeat(h, int(n // m)).astype(float)
    return upsample_nearest(h, n)


def upsample_linear(h, n):
    m = len(h)
    if m == 1:
        return np.full(n, float(h[0]), dtype=float)
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(x_new, x_old, h).astype(float)


def guided_search(x0, c0, n, rng, deadline, alpha=1.0, temp=0.0,
                  top_prob=0.25, recompute_every=1000, retry=12):
    n = int(n)
    x = np.array(x0, dtype=float).copy()
    c = np.array(c0, dtype=float).copy()
    y = 1.0 - x

    two_n1 = 2 * n - 1
    L = two_n1
    target = 0.5 * float(n)

    lmax = int(np.argmax(c))
    M = float(c[lmax])
    bestM = M
    bestx = x.copy()
    bestc = c.copy()

    K = min(L, 32)
    if K >= L:
        top_lags = np.arange(L)
        top_len = int(top_lags.size)
    else:
        top_lags = np.argpartition(c, -K)[-K:].astype(int)
        top_len = int(top_lags.size)

    it = 0
    while time.time() < deadline:
        if temp == 0.0 and M > bestM + 1e-8:
            x = bestx.copy()
            c = bestc.copy()
            y = 1.0 - x
            M = bestM

        lmax = int(np.argmax(c))
        M = float(c[lmax])

        if top_prob > 0.0 and top_len > 0 and rng.random_sample() < top_prob:
            lag = int(top_lags[int(rng.randint(top_len))])
        else:
            lag = lmax

        d = lag - (n - 1)
        ok = False
        inc = -1
        dec = -1
        maxd = 0.0

        if d == 0:
            for _ in range(retry):
                a = int(rng.randint(n))
                b = int(rng.randint(n))
                if a == b:
                    continue
                if x[a] >= x[b]:
                    inc = a
                    dec = b
                else:
                    inc = b
                    dec = a
                md = min(1.0 - x[inc], x[dec])
                if md > 1e-10:
                    ok = True
                    maxd = md
                    break
        else:
            ad = abs(d)
            m = n - ad
            if m > 0:
                for _ in range(retry):
                    r = int(rng.randint(m))
                    if d > 0:
                        dec = ad + r
                        inc = r
                    else:
                        dec = r
                        inc = ad + r
                    md = min(x[dec], 1.0 - x[inc])
                    if md > 1e-10:
                        ok = True
                        maxd = md
                        break

        if not ok:
            for _ in range(4):
                a = int(rng.randint(n))
                b = int(rng.randint(n))
                if a == b:
                    continue
                md1 = min(1.0 - x[a], x[b])
                md2 = min(1.0 - x[b], x[a])
                if md1 >= md2:
                    if md1 > 1e-10:
                        inc = a
                        dec = b
                        maxd = md1
                        ok = True
                        break
                else:
                    if md2 > 1e-10:
                        inc = b
                        dec = a
                        maxd = md2
                        ok = True
                        break

        if not ok:
            it += 1
            continue

        u = rng.random_sample()
        if u < 0.2:
            frac = 1.0 - 1e-13
        else:
            if alpha <= 0.0:
                rr = rng.random_sample()
            else:
                rr = rng.random_sample() ** alpha
            frac = 0.02 + 0.98 * rr
            if frac >= 1.0:
                frac = 1.0 - 1e-13

        delta = maxd * frac
        if delta <= 1e-14:
            it += 1
            continue

        # Apply inc += delta.
        e_inc = two_n1 - inc
        c[inc:inc + n] += delta * y[::-1]
        c[n - 1 - inc:e_inc] -= delta * x
        x[inc] += delta
        y[inc] -= delta

        # Apply dec -= delta.
        e_dec = two_n1 - dec
        c[dec:dec + n] -= delta * y[::-1]
        c[n - 1 - dec:e_dec] += delta * x
        x[dec] -= delta
        y[dec] += delta

        lnew = int(np.argmax(c))
        Mnew = float(c[lnew])

        accept = False
        if Mnew < M - 1e-12:
            accept = True
        elif Mnew <= M + 1e-11:
            if rng.random_sample() < 0.15:
                accept = True
        elif temp > 0.0:
            dv = Mnew - M
            if dv < 4.0 * temp:
                if rng.random_sample() < np.exp(-dv / temp):
                    accept = True

        if accept:
            M = Mnew
            if M < bestM - 1e-13:
                bestM = M
                bestx = x.copy()
                bestc = c.copy()
        else:
            # Undo in reverse order: dec += delta, then inc -= delta.
            c[dec:dec + n] += delta * y[::-1]
            c[n - 1 - dec:e_dec] -= delta * x
            x[dec] += delta
            y[dec] -= delta

            c[inc:inc + n] -= delta * y[::-1]
            c[n - 1 - inc:e_inc] += delta * x
            x[inc] -= delta
            y[inc] += delta

        it += 1
        if it % recompute_every == 0:
            s = float(np.sum(x))
            if abs(s - target) > 1e-8:
                x = make_feasible(x, n)
            direct = np.correlate(x, 1.0 - x, mode="full")
            c = direct
            y = 1.0 - x
            lmax = int(np.argmax(c))
            M = float(c[lmax])
            if M < bestM - 1e-13:
                bestM = M
                bestx = x.copy()
                bestc = c.copy()
            if K >= L:
                top_lags = np.arange(L)
                top_len = int(top_lags.size)
            else:
                top_lags = np.argpartition(c, -K)[-K:].astype(int)
                top_len = int(top_lags.size)

    bestx = np.array(bestx, dtype=float).copy()
    bestc = np.correlate(bestx, 1.0 - bestx, mode="full")
    bestM = float(np.max(bestc))
    return bestx, bestc, bestM


def random_search(x0, c0, n, rng, deadline, alpha=1.0, temp=0.0,
                  recompute_every=1000, spread_prob=0.35):
    n = int(n)
    x = np.array(x0, dtype=float).copy()
    c = np.array(c0, dtype=float).copy()
    y = 1.0 - x

    two_n1 = 2 * n - 1
    L = two_n1
    target = 0.5 * float(n)

    lmax = int(np.argmax(c))
    M = float(c[lmax])

    bestx = x.copy()
    bestc = c.copy()

    it = 0
    while time.time() < deadline:
        if temp == 0.0 and M > bestM + 1e-8:

        lmax = int(np.argmax(c))
        M = float(c[lmax])

        ok = False
        inc = -1
        dec = -1

                    continue

                    inc = b
                    dec = a
                md = min(1.0 - x[inc], x[dec])
                if md > 1e-10:
                    ok = True
                    maxd = md
                    break
        else:
            for _ in range(8):
                a = int(rng.randint(n))
                b = int(rng.randint(n))
                if a == b:
                    continue
                md1 = min(1.0 - x[a], x[b])
                md2 = min(1.0 - x[b], x[a])
                if md1 >= md2:
                    if md1 > 1e-10:
                        inc = a
                        dec = b
                        maxd = md1
                        ok = True
                        break
                else:
                    if md2 > 1e-10:
                        inc = b
                        dec = a
                        maxd = md2
                        ok = True
                        break

        if not ok:
            for _ in range(4):
                a = int(rng.randint(n))
                b = int(rng.randint(n))
                if a == b:
                    continue
                md1 = min(1.0 - x[a], x[b])
                md2 = min(1.0 - x[b], x[a])
                if md1 >= md2:
                    if md1 > 1e-10:
                        inc = a
                        dec = b
                        maxd = md1
                        ok = True
                        break
                else:
                    if md2 > 1e-10:
                        inc = b
                        dec = a
                        maxd = md2
                        ok = True
                        break

        if not ok:
            it += 1
            continue

        u = rng.random_sample()
        if u < 0.1:
            frac = 1.0 - 1e-13
        else:
            if alpha <= 0.0:
                rr = rng.random_sample()
            else:
                rr = rng.random_sample() ** alpha
            frac = 0.02 + 0.98 * rr
            if frac >= 1.0:
                frac = 1.0 - 1e-13

        delta = maxd * frac
        if delta <= 1e-14:
            it += 1
            continue

        e_inc = two_n1 - inc
        c[inc:inc + n] += delta * y[::-1]
        c[n - 1 - inc:e_inc] -= delta * x
        x[inc] += delta
        y[inc] -= delta

        e_dec = two_n1 - dec
        c[dec:dec + n] -= delta * y[::-1]
        c[n - 1 - dec:e_dec] += delta * x
        x[dec] -= delta
        y[dec] += delta

        lnew = int(np.argmax(c))
        Mnew = float(c[lnew])

        accept = False
        if Mnew < M - 1e-12:
            accept = True
        elif Mnew <= M + 1e-11:
            if rng.random_sample() < 0.1:
                accept = True
        elif temp > 0.0:
            dv = Mnew - M
            if dv < 4.0 * temp:
                if rng.random_sample() < np.exp(-dv / temp):
                    accept = True

        if accept:
            M = Mnew
            if M < bestM - 1e-13:
                bestM = M
                bestx = x.copy()
                bestc = c.copy()
        else:
            c[dec:dec + n] += delta * y[::-1]
            c[n - 1 - dec:e_dec] -= delta * x
            x[dec] += delta
            y[dec] -= delta

            c[inc:inc + n] -= delta * y[::-1]
            c[n - 1 - inc:e_inc] += delta * x
            x[inc] -= delta
            y[inc] += delta

        it += 1
        if it % recompute_every == 0:
            s = float(np.sum(x))
            if abs(s - target) > 1e-8:
                x = make_feasible(x, n)
            direct = np.correlate(x, 1.0 - x, mode="full")
            c = direct
            y = 1.0 - x
            lmax = int(np.argmax(c))
            M = float(c[lmax])
            if M < bestM - 1e-13:
                bestM = M
                bestx = x.copy()
                bestc = c.copy()

    bestx = np.array(bestx, dtype=float).copy()
    bestc = np.correlate(bestx, 1.0 - bestx, mode="full")
    bestM = float(np.max(bestc))
    return bestx, bestc, bestM


def optimize_candidate(h, n, c, rng, limit_s, alpha1=1.0, alpha2=0.8,
                       temp1=0.0):
    n = int(n)
    limit_s = float(limit_s)
    if limit_s <= 5.0:
        return h.copy(), c.copy(), float(np.max(c))

    now = time.time()
    end = now + limit_s

    if limit_s > 40.0:
        end_g = min(end, now + 0.6 * limit_s)
    else:
        end_g = end

    x1, c1, M1 = guided_search(h, c, n, rng, end_g,
                               alpha=alpha1, temp=temp1,
                               top_prob=0.25, recompute_every=1000)

    if time.time() < end - 8.0:
        x2, c2, M2 = random_search(x1, c1, n, rng, end,
                                   alpha=alpha2, temp=0.0,
                                   recompute_every=1000,
                                   spread_prob=0.35)
        c5_1 = M1 * (2.0 / float(n))
        c5_2 = M2 * (2.0 / float(n))
        if c5_2 < c5_1:
            x1, c1, M1 = x2, c2, M2

    return x1, c1, M1


def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    budget_s = float(budget_s)

    reserve = min(45.0, max(10.0, 0.05 * budget_s))
    usable = budget_s - reserve
    if usable < 30.0:
        usable = max(5.0, budget_s - 2.0)
    final_deadline = start + usable

    rng = np.random.RandomState(seed)

    h_parent = None
    try:
        h_parent = np.asarray(initial_h_values, dtype=float)
    except Exception:
        h_parent = None

    if h_parent is not None and h_parent.size > 0:
        n0 = int(len(h_parent))
        best_x, best_c, best_M, best_c5 = eval_h(h_parent, n0)
    else:
        n0 = 160
        best_x = fallback_h(n0)
        best_c, best_M, best_c5 = eval_h(best_x, n0)[1:]
    best_n = int(n0)

    # First, polish the supplied parent/current best at its own resolution.
    if time.time() < final_deadline - 60.0:
        rem = final_deadline - time.time()
        lim = min(300.0, max(60.0, 0.20 * rem))
        xg, cg, Mg = optimize_candidate(best_x, best_n, best_c, rng, lim,
                                        alpha1=1.0, alpha2=0.8, temp1=0.0)
        c5g = Mg * (2.0 / float(best_n))
        if c5g < best_c5 - 1e-12:
            best_c5 = c5g
            best_x = xg.copy()
            best_c = cg.copy()
            best_M = Mg
            best_n = int(best_n)

    # Build a modest candidate pool from the improved best.
    candidates = []
    counter = 0

    if best_n % 2 == 0:
        hb = make_binary_from_frac(best_x, best_n)
        hb, hc, hm, hc5 = eval_h(hb, best_n)
        candidates.append((hc5, hm, best_n, counter, hb, hc, "bin"))
        counter += 1

    ns = set()
    for base in (160, 240, 256, 320, 480, 640, 768):
        if 64 <= base <= 1024 and base % 2 == 0:
            ns.add(base)
    if best_n % 2 == 0:
        for mult in (2, 4):
            nn = best_n * mult
            if 64 <= nn <= 1024:
                ns.add(nn)
    for base in (160, 256, 320, 480):
        nn = base + 1
        if 64 <= nn <= 1024:
            ns.add(nn)
        nn = base - 1
        if 64 <= nn <= 1024:
            ns.add(nn)
    for nn in (best_n + 1, best_n - 1):
        if 64 <= nn <= 1024:
            ns.add(nn)

    for nn in sorted(ns):
        if time.time() > final_deadline - 80.0:
            break
        if nn == best_n:
            continue

        h_rep = upsample_repeat(best_x, nn)
        hh, cc, mm, cc5 = eval_h(h_rep, nn)
        candidates.append((cc5, mm, nn, counter, hh, cc, "rep"))
        counter += 1

        if time.time() > final_deadline - 60.0:
            break

        h_lin = upsample_linear(best_x, nn)
        hh, cc, mm, cc5 = eval_h(h_lin, nn)
        candidates.append((cc5, mm, nn, counter, hh, cc, "lin"))
        counter += 1

    if candidates:
        candidates.sort()

    processed = 0
    cand_limit_fracs = [0.18, 0.14, 0.11, 0.09, 0.07]

    for cand in candidates:
        if processed >= 5:
            break
        if time.time() > final_deadline - 90.0:
            break

        cc5, mm, nn, cnt, hh, hc, label = cand
        nn = int(nn)
        if cc5 > best_c5 + 0.035:
            break

        rem = final_deadline - time.time()
        if rem < 45.0:
            break

        frac = cand_limit_fracs[min(processed, len(cand_limit_fracs) - 1)]
        limit = min(170.0, max(25.0, rem * frac))
        x1, c1, M1 = optimize_candidate(hh, nn, hc, rng, limit,
                                        alpha1=1.0, alpha2=0.8, temp1=0.0)
        c5_1 = M1 * (2.0 / float(nn))
        if c5_1 < best_c5 - 1e-12:
            best_c5 = c5_1
            best_x = x1.copy()
            best_c = c1.copy()
            best_M = M1
            best_n = nn
        processed += 1

    # Second wave: re-upsample the improved best at nearby / doubled resolutions.
    second_ns = []
    for nn in (best_n + 1, best_n + 2, best_n * 2, best_n * 4):
        if 64 <= nn <= 1024:
            second_ns.append(int(nn))

    for nn in second_ns:
        if time.time() > final_deadline - 110.0:
            break
        if nn == best_n:
            continue
        h_up = upsample_linear(best_x, nn)
        hh, cc, mm, cc5 = eval_h(h_up, nn)
        if cc5 > best_c5 + 0.03:
            continue
        if time.time() > final_deadline - 80.0:
            break
        limit = min(130.0, max(30.0, final_deadline - time.time() - 70.0))
        x2, c2, M2 = optimize_candidate(hh, nn, cc, rng, limit,
                                        alpha1=1.0, alpha2=0.8, temp1=0.0)
        c5_2 = M2 * (2.0 / float(nn))
        if c5_2 < best_c5 - 1e-12:
            best_c5 = c5_2
            best_x = x2.copy()
            best_c = c2.copy()
            best_M = M2
            best_n = nn

    # Binary-like restart on the current best resolution.
    if best_n % 2 == 0 and time.time() < final_deadline - 110.0:
        hb = make_binary_from_frac(best_x, best_n)
        hb, hc, hm, hc5 = eval_h(hb, best_n)
        if hc5 < best_c5 + 0.03:
            limit = min(100.0, max(30.0, final_deadline - time.time() - 80.0))
            x3, c3, M3 = optimize_candidate(hb, best_n, hc, rng, limit,
                                            alpha1=0.7, alpha2=0.4, temp1=0.0)
            c5_3 = M3 * (2.0 / float(best_n))
            if c5_3 < best_c5 - 1e-12:
                best_c5 = c5_3
                best_x = x3.copy()
                best_c = c3.copy()
                best_M = M3
                best_n = int(best_n)

    # Small noisy restart on the current best resolution.
    if time.time() < final_deadline - 100.0:
        h_no = best_x + rng.uniform(-0.12, 0.12, size=best_n)
        h_no, hc_no, hm_no, hc5_no = eval_h(h_no, best_n)
        if hc5_no < best_c5 + 0.05:
            limit = min(100.0, max(30.0, final_deadline - time.time() - 80.0))
            x4, c4, M4 = optimize_candidate(h_no, best_n, hc_no, rng, limit,
                                            alpha1=0.9, alpha2=0.7, temp1=0.0)
            c5_4 = M4 * (2.0 / float(best_n))
            if c5_4 < best_c5 - 1e-12:
                best_c5 = c5_4
                best_x = x4.copy()
                best_c = c4.copy()
                best_M = M4
                best_n = int(best_n)

    # Mild annealing phase to escape shallow local optima.
    if time.time() < final_deadline - 120.0:
        limit = min(120.0, max(30.0, final_deadline - time.time() - 80.0))
        temp = max(0.4, 0.003 * float(best_n))
        x5, c5v, M5 = guided_search(best_x, best_c, best_n, rng,
                                    time.time() + limit,
                                    alpha=0.7, temp=temp,
                                    top_prob=0.35, recompute_every=800)
        c5_5 = M5 * (2.0 / float(best_n))
        if c5_5 < best_c5 - 1e-12:
            best_c5 = c5_5
            best_x = x5.copy()
            best_c = c5v.copy()
            best_M = M5
            best_n = int(best_n)

    # Final fine polish.
    if time.time() < final_deadline - 30.0:
        limit = min(140.0, max(20.0, final_deadline - time.time() - 10.0))
        x6, c6, M6 = optimize_candidate(best_x, best_n, best_c, rng, limit,
                                        alpha1=3.0, alpha2=2.0, temp1=0.0)
        c5_6 = M6 * (2.0 / float(best_n))
        if c5_6 < best_c5 - 1e-12:
            best_c5 = c5_6
            best_x = x6.copy()
            best_c = c6.copy()
            best_M = M6
            best_n = int(best_n)

    final_h, final_c, final_M, final_c5 = eval_h(best_x, best_n)
    return final_h, float(final_c5), int(best_n)
