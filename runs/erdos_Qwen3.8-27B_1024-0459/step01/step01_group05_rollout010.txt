    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    obj = corr * dx
    
    m = np.max(obj)
    shifted = np.clip(beta * (obj - m), -50, 50)
    exp_s = np.exp(shifted)

    softmax = exp_s / sum_exp

Hmm, but this is pure gradient descent which might be slow. Let me think about using L-BFGS-B with a trick.

Actually, the cleanest approach for the equality constraint with L-BFGS-B:
- Use a Lagrange multiplier for the sum constraint
- At each iteration, the effective gradient is grad + lambda * ones
- Update lambda to maintain the constraint

This is basically what projected gradient does, but with L-BFGS memory for the curvature.

Let me try a different approach: use scipy's minimize with method='COBYLA' or 'Nelder-Mead' for the nonconvex part, or use a penalty method.

Actually, you know what, let me just go with a well-tuned SLSQP approach but with:
1. Better initial points (try multiple)
2. Adaptive beta for smooth max
3. Resolution sweeping
4. The key improvement: use the analytic gradient I derived above (much faster than numerical)

The previous code used numerical gradient (200+ function evaluations per gradient step). With analytic gradient, each SLSQP iteration will be much faster, allowing many more iterations in the same time.

Let me also try a completely different angle: **simulated annealing / randomized search**.

For the simulated annealing:
- State: h vector
- Move: pick 2-3 random positions, adjust them while maintaining sum constraint
- Accept: with probability exp(-ΔC₅/T)

This could escape local optima that gradient methods get stuck in.

Let me combine approaches:
1. Phase 1: Analytic-gradient SLSQP from parent (bestใช้ speed up)
2. Phase 2: Multi-resolution with different starting points
3. Phase 3: Local search with random perturbations + re-optimization
4. Phase 4: Simulated annealing on the best solution

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_val_grad(h, beta):
    """Compute smooth max of corr*dx and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    obj = corr * dx
    
    m = np.max(obj)
    shifted = np.clip(beta * (obj - m), -50.0, 50.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    sm = exp_s / sum_exp
    val = m + np.log(sum_exp) / beta
    
    # Analytic gradient
    grad = np.zeros(n)
    center = n - 1
    for k in range(2 * n - 1):
        L = abs(k - center)
        wk = sm[k] * dx
        if wk < 1e-16:
            continue
        if L == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            grad[L:] += wk * (1.0 - h[:n - L])
            grad[:n - L] -= wk * h[L:]
    
    return val, grad

def project_h(h, target):
    """Project h onto {sum(h)=target, 0<=h<=1}."""
    h = np.array(h, dtype=float)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            free = np.where(h < 0.9999999)[0]
            if len(free) > 0:
                h[free] += diff / len(free)
        else:
            free = np.where(h > 1e-7)[0]
            if len(free) > 0:
                h[free] += diff / len(free)
    h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_con(h, beta, target):
    val, grad = smooth_max_val_grad(h, beta)
    return val, grad

def make_constraints(target):
    def cfun(h):
        return np.sum(h) - target
    def cjac(h):
        return np.ones(len(h))
    return cfun, cjac

def optimize_slsqp(h0, n, target, beta, max_iter, ftol, time_limit=None, t_start=None):
    h = project_h(h0, target)
    
    def objective(h_in):
        return smooth_max_val_grad(np.clip(h_in, 0, 1), beta)[0]
    
    def gradient(h_in):
        h_c = np.clip(h_in, 0, 1)
        return smooth_max_val_grad(h_c, beta)[1]
    
    cfun, cjac = make_constraints(target)
    constraints = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bounds = [(0.0, 1.0)] * n
    
    result = None
    try:
        opts = {'maxiter': max_iter, 'ftol': ftol}
        if time_limit is not None and t_start is not None:
            remaining = time_limit - (time.time() - t_start)
            if remaining < 5:
                return h
        result = minimize(objective, h, method='SLSQP', jac=gradient,
                         bounds=bounds, constraints=constraints, options=opts)
    except Exception:
        pass
    
    if result is not None and result.x is not None:
        h_opt = project_h(result.x, target)
        return h_opt
    return h

def simulated_annealing(h0, n, target, t_start, time_limit, t_initial=0.01, t_final=0.0001):
    h = project_h(h0.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    c5_cur = best_c5
    
    n_steps = 5000
    n_total = 2 * n - 1
    
    for step in range(n_steps):
        if time.time() - t_start > time_limit:
            break
        T = t_initial * (t_final / t_initial) ** (step / n_steps)
        
        # Random move: pick 2-4 positions and adjust
        n_move = np.random.randint(2, 5)
        idx = np.random.choice(n, n_move, replace=False)
        
        h_new = h.copy()
        # Add random perturbation
        deltas = np.random.randn(n_move) * 0.05
        # Maintain sum constraint
        total_delta = np.sum(deltas)
        if abs(total_delta) > 1e-12:
            other_idx = np.setdiff1d(np.arange(n), idx)
            if len(other_idx) > 0:
                total_delta /= len(other_idx)
                delta_other = -total_delta
                h_new[other_idx] += delta_other * np.random.uniform(0.5, 1.5, len(other_idx))
                # Re-normalize
                h_new = project_h(h_new, target)
        
        h_new[idx] += deltas
        h_new = project_h(h_new, target)
        
        c5_new = compute_c5(h_new)
        delta = c5_new - c5_cur
        
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    
    return best_h, best_c5
