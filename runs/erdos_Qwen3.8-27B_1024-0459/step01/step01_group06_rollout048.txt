import numpy as np
from scipy.optimize import minimize
import time

def project_box_sum(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum} using bisection on shift."""
    h = np.asarray(h, float)
    n = len(h)
    # We need sum(clip(h - tau, 0, 1)) = target_sum
    # At tau = -max(h): sum = n (all clipped to 1)
    # At tau = min(h): sum = 0 (all clipped to 0)
    # Actually at tau = max(h): some are 0
    # Let's think: clip(h[i] - tau, 0, 1) is decreasing in tau
    # We want it to equal target_sum/n on average
    
    lo, hi = -1.0, 2.0  # wide range for tau
    for _ in range(60):
        tau = (lo + hi) / 2
        s = np.sum(np.clip(h - tau, 0, 1))
        if s > target_sum:
            lo = tau  # need to reduce, increase tau
        else:
            hi = tau
    tau = (lo + hi) / 2
    return np.clip(h - tau, 0, 1)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_active_lag(h, k_max, n):
    """Gradient of c_{k_max} with respect to h."""
    n = len(h)
    dx = 2.0 / n
    lag = k_max - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        # c = dx * sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
        # d/dh[j] = dx * [(1-h[j+lag]) if j <= n-1-lag] + [-h[j-lag] if j >= lag]
        for j in range(n):
            if j <= n - 1 - lag:
                g[j] += (1.0 - h[j + lag])
            if j >= lag:
                g[j] -= h[j - lag]
    else:
        # lag < 0, let m = -lag
        m = -lag
        # c = dx * sum_{i=m}^{n-1} h[i]*(1-h[i-m])
        # d/dh[j] = dx * [(1-h[j-m]) if j >= m] + [-h[j+m] if j+m <= n-1 and j+m >= m]
        #         = dx * [(1-h[j-m]) if j >= m] + [-h[j+m] if j <= n-1-m]
        for j in range(n):
            if j >= m:
                g[j] += (1.0 - h[j - m])
            if j <= n - 1 - m:
                g[j] -= h[j + m]
    return g * dx

def grad_active_lag_vec(h, k_max, n):
    """Vectorized gradient of c_{k_max}."""
    n = len(h)
    dx = 2.0 / n
    lag = k_max - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        # d/dh[j] = (1-h[j+lag]) for j in [0, n-1-lag]
        g[:n-lag] += (1.0 - h[lag:n])
        # d/dh[j] -= h[j-lag] for j in [lag, n-1]
        g[lag:] -= h[:n-lag]
    else:
        m = -lag
        # d/dh[j] += (1-h[j-m]) for j in [m, n-1]
        g[m:] += (1.0 - h[:n-m])
        # d/dh[j] -= h[j+m] for j in [0, n-1-m]
        g[:n-m] -= h[m:n]
    return g * dx

def soft_max_obj(h, n, beta):
    """Soft approximation of max c_k."""
    c = compute_correlations(h)
    scaled = beta * c
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / beta * (2.0 / n))

def soft_max_grad(h, n, beta):
    """Gradient of soft max objective."""
    c = compute_correlations(h)
    scaled = beta * c
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= np.sum(w)
    dx = 2.0 / n
    g = np.zeros(n)
    n_bridge = n - 1
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-15:
            continue
        lag = k_idx - n_bridge
        if lag >= 0:
            g[:n-lag] += wk * (1.0 - h[lag:n])
            g[lag:] -= wk * h[:n-lag]
        else:
            m_abs = -lag
            g[m_abs:] += wk * (1.0 - h[:n-m_abs])
            g[:n-m_abs] -= h[m_abs:n]
    return g * dx

import numpy as np
from scipy.optimize import minimize
import time

def project_box_sum(h, target_sum):
    n = len(h)
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - tau, 0.0, 1.0))
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    tau = 0.5 * (lo + hi)
    return np.clip(h - tau, 0.0, 1.0)

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5_val(h):
    n = len(h)
    c = compute_correlations(h)
    return float(np.max(c) * (2.0 / n))

def gradient_single_lag(h, k_idx, n):
    dx = 2.0 / n
    lag = k_idx - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        g[:n - lag] += (1.0 - h[lag:n])
        g[lag:] -= h[:n - lag]
    else:
        m = -lag
        g[m:] += (1.0 - h[:n - m])
        g[:n - m] -= h[m:n]
    return g * dx

def gradient_topk(h, K=10):
    n = len(h)
    c = compute_correlations(h)
    top_idx = np.argsort(c)[-K:]
    top_idx = top_idx[np.argsort(c[top_idx][::-1])[::-1]]
    g = np.zeros(n)
    for k_idx in top_idx:
        g += gradient_single_lag(h, k_idx, n)
    return g, c

def gradient_softmax(h, beta):
    n = len(h)
    c = compute_correlations(h)
    scaled = beta * c
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= np.sum(w)
    dx = 2.0 / n
    active = w > 1e-12
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            g[:n - lag] += wk * (1.0 - h[lag:n])
            g[lag:] -= wk * h[:n - lag]
        else:
            m = -lag
            g[m:] += wk * (1.0 - h[:n - m])
            g[:n - m] -= h[m:n]
    return g * dx, c

def nested_gradient_opt(h, n, target, max_iter=3000, base_lr=0.02, momentum=0.9,
                        topk=5, use_soft=False, beta=20.0, tol=1e-10):
    dx = 2.0 / n
    h_cur = h.copy()
    v = np.zeros(n)
    best_c = compute_c5_val(h_cur)
    best_h = h_cur.copy()
    c_history = np.zeros(min(max_iter, 500))
    
    for it in range(max_iter):
        if use_soft:
            g, c = gradient_softmax(h_cur, beta)
        else:
            g, c = gradient_topk(h_cur, K=topk)
        
        c_max = float(np.max(c) * dx)
        if c_max < best_c - 1e-12:
            best_c = c_max
            best_h = h_cur.copy()
        
        if it < len(c_history):
            c_history[it] = c_max
        
        if it > 200 and np.std(c_history[it-200:it]) < tol:
            break
        
        lr = base_lr / (1.0 + 0.01 * it)
        v = momentum * v - lr * g
        h_new = project_box_sum(h_cur + v, target)
        
        c_new = compute_c5_val(h_new)
        if c_new > c_max * 1.05 + 0.05:
            v = 0.5 * v
            h_new = project_box_sum(h_cur + v, target)
            c_new = compute_c5_val(h_new)
            if c_new > c_max * 1.1 + 0.1:
                v *= 0.5
                h_new = project_box_sum(h_cur + v, target)
        
        h_cur = h_new
    
    return h_cur, best_c, best_h

def binary_project(h, target):
    n = len(h)
    target_int = int(round(target))
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target_int]] = 1.0
    return h_bin

def binary_local_search(h_bin, n, max_flips=200, seed=42):
    rng = np.random.RandomState(seed)
    target = n // 2
    ones = np.where(h_bin == 1)[0]

    best_c = compute_c5_val(h_bin)

    for _ in range(max_flips):
        if len(ones) < 2 or len(zeros) < 2:

        c_new = compute_c5_val(h_bin)

            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
        else:
            h_bin[a] = 1.0
            h_bin[b] = 0.0
    
    return best_h, best_c
