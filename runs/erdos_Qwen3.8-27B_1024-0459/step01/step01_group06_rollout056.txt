def hard_max_obj_grad(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    k_max = np.argmax(c)
    obj = float(c[k_max] * dx)
    lag = k_max - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        cnt = n - lag
        if cnt > 0:
            g[lag:] += (1.0 - h[:cnt])
            g[:cnt] -= h[lag:]
    else:
        m_ = -lag
        cnt = n - m_
        if cnt > 0:
            g[:cnt] += (1.0 - h[m_:])
            g[m_:] -= h[:cnt]
    g *= dx
    return obj, g


def projected_gradient(h, n, beta, max_iter=5000, tol=1e-12, use_soft=True):
    h = h.copy()
    step = 0.01
    prev_obj = soft_max_obj(h, n, beta) if use_soft else compute_c5(h, n)
    for it in range(max_iter):
        if use_soft:
            obj, grad = soft_max_obj_grad(h, n, beta)
        else:
            obj, grad = hard_max_obj_grad(h, n)
        
        # Backtracking line search
        s = step
        for _ in range(40):
            h_try = project_feasible(h - s * grad, n)
            if use_soft:
                obj_try = soft_max_obj(h_try, n, beta)
            else:
                obj_try = compute_c5(h_try, n)
            if obj_try < obj - 1e-15:
                break
            s *= 0.5
        else:
            break
        h = h_try
        step = min(s * 1.5, 0.1)
        if abs(obj - obj_try) < tol:
            break
        obj = obj_try
    return h


def slsqp_polish(h, n, maxiter=300):
    def fun(x):
        return compute_c5(x, n)
    
    def jac(x):
        _, g = hard_max_obj_grad(x, n)
        return g
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(fun, x0=h, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.success or compute_c5(np.clip(res.x, 0, 1), n) < compute_c5(h, n):
            return project_feasible(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        pass
    return h


def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)


def make_block_init(n, nblocks=4):
    h = np.zeros(n)
    sizes = []
    remaining = n
    for i in range(nblocks - 1):
        sz = max(1, int(remaining / (nblocks - i)))
        sizes.append(sz)
        remaining -= sz
    sizes.append(remaining)
    pos = 0
    for i, sz in enumerate(sizes):
        if i % 2 == 0:
            h[pos:pos + sz] = 1.0
        pos += sz
    # Adjust to get exactly n/2 ones
    n_ones = int(h.sum())
    need = n // 2 - n_ones
    if need > 0:
        zero_idx = np.where(h == 0)[0]
        random.shuffle(zero_idx.tolist()) if hasattr(random, 'shuffle') else None
        for i in range(need):
            h[zero_idx[i]] = 1.0
    elif need < 0:
        one_idx = list(np.where(h == 1)[0])
        import random as _r
        _r.shuffle(one_idx)
        for i in range(-need):
            h[one_idx[i]] = 0.0
    return project_feasible(h, n)


def make_smooth_init(n, seed_val=0):
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    h = 0.5 + 0.35 * np.sin(np.pi * x + seed_val)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)


def make_periodic_init(n, period=6):
    h = np.zeros(n)
    for i in range(n):
        pos_in_period = i % period
        if pos_in_period < period // 2:
            h[i] = 1.0
        else:
            h[i] = 0.0
    # Fix sum
    h = project_feasible(h, n)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial values from parent
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        n_parent = len(h_parent)
    except Exception:
        h_parent = np.full(95, 0.5)
        n_parent = 95
    
    h_parent = project_feasible(h_parent, n_parent)
    best_h = h_parent.copy()
    best_n = n_parent
    best_c5 = compute_c5(best_h, best_n)
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    # Phase 1: SLSQP polish on parent resolution
    if time_left() > 120:
        h_pol = slsqp_polish(best_h, best_n, maxiter=200)
        c5_pol = compute_c5(h_pol, best_n)
        if c5_pol < best_c5:
            best_h = h_pol
            best_c5 = c5_pol
    
    # Phase 2: Projected gradient at parent resolution with soft-max
    for beta in [5.0, 20.0, 80.0, 300.0]:
        if time_left() < 80:
            break
        h_opt = projected_gradient(best_h, best_n, beta, max_iter=3000)
        c5_opt = compute_c5(h_opt, best_n)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    
    # Phase 3: SLSQP polish again after projection gradient
    if time_left() > 100:
        h_pol = slsqp_polish(best_h, best_n, maxiter=300)
        c5_pol = compute_c5(h_pol, best_n)
        if c5_pol < best_c5:
            best_h = h_pol
            best_c5 = c5_pol
    
    # Phase 4: Try higher resolutions
    resolutions = []
    if best_n < 100:
        resolutions = [2 * best_n, 4 * best_n, 6 * best_n]
    else:
        resolutions = [2 * best_n, 3 * best_n]
    
    for n_new in resolutions:
        if time_left() < 150:
            break
        
        # Interpolate current best to new resolution
        h_new = interpolate_h(best_h, best_n, n_new)
        
        # Soft-max projected gradient at new resolution
        for beta in [5.0, 30.0, 150.0, 500.0]:
            if time_left() < 60:
                break
            max_it = min(8000, int(time_left() * 8))
            if max_it < 500:
                max_it = 500
            h_opt = projected_gradient(h_new, n_new, beta, max_iter=max_it)
            c5_opt = compute_c5(h_opt, n_new)
            if c5_opt < best_c5:
                best_h = h_opt
                best_n = n_new
                best_c5 = c5_opt
                h_new = h_opt.copy()
        
        # SLSQP polish at this resolution if time permits
        if time_left() > 80:
            h_pol = slsqp_polish(h_new, n_new, maxiter=200)
            c5_pol = compute_c5(h_pol, n_new)
            if c5_pol < best_c5:
                best_h = h_pol
                best_n = n_new
                best_c5 = c5_pol
                h_new = h_pol.copy()
    
    # Phase 5: Hard-max projected gradient refinement
    if time_left() > 40:
        max_it = min(5000, int(time_left() * 10))
        h_opt = projected_gradient(best_h, best_n, 1.0, max_iter=max_it, use_soft=False)
        c5_opt = compute_c5(h_opt, best_n)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    
    # Phase 6: Random restarts with structured initializations
    if time_left() > 60:
        n_try = max(best_n, 190)
        if n_try > 300:
            n_try = 300
        inits = []
        # Interpolated parent
        inits.append(interpolate_h(best_h, best_n, n_try))
        # Periodic patterns
        for p in [5, 7, 8, 10, 12]:
            inits.append(make_periodic_init(n_try, p))
        # Smooth
        for sp in [0.0, 1.0, 2.0]:
            inits.append(make_smooth_init(n_try, sp))
        # Random
        for _ in range(3):
            hr = rng.rand(n_try) * 0.3
            hr = project_feasible(hr, n_try)
            inits.append(hr)
        
        for h_init in inits:
            if time_left() < 30:
                break
            # Quick soft-max optimization
            for beta in [10.0, 100.0]:
                if time_left() < 20:
                    break
                max_it = min(3000, int(time_left() * 5))
                if max_it < 200:
                    break
                h_opt = projected_gradient(h_init, n_try, beta, max_iter=max_it)
                c5_opt = compute_c5(h_opt, n_try)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_n = n_try
                    best_c5 = c5_opt
                    h_init = h_opt.copy()
                    break  # Go to next init
    
    # Phase 7: Final SLSQP polish
    if time_left() > 20:
        h_pol = slsqp_polish(best_h, best_n, maxiter=500)
        c5_pol = compute_c5(h_pol, best_n)
        if c5_pol < best_c5:
            best_h = h_pol
            best_c5 = c5_pol
    
    # Final hard-max refinement
    if time_left() > 10:
        h_opt = projected_gradient(best_h, best_n, 1.0, max_iter=2000, use_soft=False)
        c5_opt = compute_c5(h_opt, best_n)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    
    # Validate
    try:
        val = evaluate_erdos_solution(best_h, best_n)
        if val is not None and 0.0 < val < 1.0:
            best_c5 = min(best_c5, float(val))
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
