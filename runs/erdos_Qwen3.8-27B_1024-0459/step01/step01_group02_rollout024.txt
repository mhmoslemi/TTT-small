def project_feasible(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

        else:

def corr_gradient(h, n, active_weights):
    """
    Compute gradient of weighted sum of correlations.
    active_weights: array of shape (2n-1,) with weights for each correlation
    Returns: gradient of sum_j active_weights[j] * corr[j] w.r.t. h

    grad = np.zeros(n)

                grad[:n_valid] += w * (1.0 - h[L:])

    weights = exps / np.sum(exps)
    obj = (m + np.log(np.sum(exps)) / beta) * (2.0 / n)
    grad = corr_gradient(h, n, weights) * (2.0 / n)
    return obj, grad

def subgradient_step(h, n, lr, tol=0.005):

    cmax = np.max(corr)
    active = corr >= cmax - tol

    # Use weights proportional to (corr - (cmax-tol)) for active

    else:

    grad = corr_gradient(h, n, weights)
    h_new = h - lr * grad
    return project_feasible(h_new, n)

    target = n / 2.0

def soft_max_and_grad(h, beta):
    n = len(h)
    corr = correlate_full(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)

    obj = (m + np.log(s) / beta) * (2.0 / n)
