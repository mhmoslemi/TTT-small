def projected_gradient_step(h, eta, alpha, n):
    """One step of projected gradient descent on the smooth max."""
    scores = np.correlate(h, 1-h, mode="full") * (2.0/n)
    m = np.max(scores)
    
    if alpha < 1e-14:
        # Use subgradient: uniform over active set
        active = np.flatnonzero(scores >= m - 1e-10)
        if len(active) == 0:
            active = np.array([np.argmax(scores)])
        w = np.zeros(len(scores))
        w[active] = 1.0/len(active)
    else:
        w = np.exp((scores - m) / alpha)
        A = np.sum(w)
        w /= A
    
    # Compute gradient
    grad = np.zeros(n)
    for idx in range(len(scores)):
        wi = w[idx]
        if wi < 1e-14:
            continue
        if idx == n - 1:
            grad += wi * (1.0 - 2.0*h)
        elif idx > n - 1:
            d = idx - (n-1)
            grad[d:] += wi * (1.0 - h[:n-d])
            grad[:n-d] -= wi * h[:n-d]
        else:
            d = (n-1) - idx
            grad[:n-d] += wi * (1.0 - h[d:])
            grad[d:] -= wi * h[d:]
    
    # Gradient step
    h_new = h - eta * grad
    # Project to feasible set
    h_new = as_feasible(h_new, n)
    return h_new

def full_project(h, n):
    """Exact projection via bisection (slower but exact)"""
    h = np.array(h, dtype=float, copy=True)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    # Fine fix
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def compute_scores_scaled(h, n):
    """Return all shift scores scaled by 2/n"""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv * (2.0 / n)

def smooth_max_val(scores, alpha):
    m = float(np.max(scores))
    if alpha < 1e-14:
        return m
    shifted = (scores - m) / alpha
    shifted = np.clip(shifted, -500.0, 500.0)
    w = np.exp(shifted)
    A = float(np.sum(w))
    if not math.isfinite(A) or A <= 0:
        return m
    return m + alpha * math.log(A)

def compute_pgd_gradient(h, n, alpha):
    """Gradient of the smooth max of scores w.r.t. h"""
    conv_raw = np.correlate(h, 1.0 - h, mode="full")
    scores = conv_raw * (2.0 / n)
    m = float(np.max(scores))
    nz = 2 * n - 1
    
    if alpha < 1e-14:
        tol = max(1e-10, 1e-8 * max(1.0, m))
        active = np.flatnonzero(scores >= m - tol)
        if len(active) == 0:
            active = np.array([int(np.argmax(scores))])
        w = np.zeros(nz)
        w[active] = 1.0 / len(active)
    else:
        shifted = (scores - m) / alpha
        shifted = np.clip(shifted, -500.0, 500.0)
        w = np.exp(shifted)
        A = float(np.sum(w))
        if not math.isfinite(A) or A <= 0:
            return np.zeros(n)
        w /= A
    
    grad = np.zeros(n)
    for idx in range(nz):
        wi = w[idx]
        if wi < 1e-14:
            continue
        if idx == n - 1:
            grad += wi * (1.0 - 2.0 * h)
        elif idx > n - 1:
            d = idx - (n - 1)
            if d < n:
                grad[d:] += wi * (1.0 - h[:n - d])
                grad[:n - d] -= wi * h[:n - d]
        else:
            d = (n - 1) - idx
            if d < n:
                grad[:n - d] += wi * (1.0 - h[d:])
                grad[d:] -= wi * h[d:]
    
    return grad

def projected_gradient(h, n, alpha, eta, rng, iters, progress_callback=None):
    """Run projected gradient descent for a fixed alpha"""
    h = np.array(h, dtype=float, copy=True)
    best_h = h.copy()
    best_c = compute_c5(h)
    
    for t in range(iters):
        check_budget()
        if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE - 5:
            break
        
        grad = compute_pgd_gradient(h, n, alpha)
        h_new = h - eta * grad
        h_new = fast_project(h_new, n)
        
        # Occasionally add perturbation for exploration
        if alpha > 0.01 and t % 200 == 100:
            noise = rng.normal(0, 0.01, size=n)
            h_new = fast_project(h_new + noise, n)
        
        h = h_new
        
        if t % 50 == 0 or t == iters - 1:
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        
        if progress_callback is not None:
            progress_callback(t, h, best_h, best_c)
    
    # Final check
    c = compute_c5(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    
    return best_h, best_c

def slsqp_obj(x, alpha, n):
    check_budget()
    h = np.clip(x, 0.0, 1.0)
    scores = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    return smooth_max_val(scores, alpha)

def slsqp_grad(x, alpha, n):
    check_budget()
    h = np.clip(x, 0.0, 1.0)
    return compute_pgd_gradient(h, n, alpha)

def slsqp_polish(h, n, time_limit):
    """Final SLSQP polish"""
    if time_limit < 5:
        return h
    h = full_project(h, n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 0.5 * n}]
    c = compute_c5(h)
    
    alpha = 0.005
    maxiter = min(200, max(50, int(time_limit * 3)))
    
    try:
        res = minimize(slsqp_obj, h, args=(alpha, n),
                       jac=slsqp_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': 0})
        if res.success or res.fun < c:
            h2 = full_project(res.x, n)
            c2 = compute_c5(h2)
            if c2 < c:
                return h2
    except Exception:
        pass
    return h

def upsample(parent, n):
    n0 = parent.size
    if n0 == n:
        return parent.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x, x0, parent)

def downsample(h, n):
    n0 = h.size
    if n0 == n:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x, x0, h)

def make_sin_init(n, rng):
    """Create diverse sinusoidal initializations"""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    cands = []
    for f in [1, 2, 3]:
        for a in [0.3, 0.5, 0.7]:
            for ph in [0, 0.5 * math.pi, math.pi]:
                h = 0.5 + a * np.cos(2.0 * math.pi * f * x / 2.0 + ph)
                h = full_project(h, n)
                c = compute_c5(h)
                cands.append((h, c))
    cands.sort(key=lambda x: x[1])
    return cands[:3]

def make_random_init(n, rng):
    """Create random smooth initializations"""
    cands = []
    for _ in range(4):
        # Low-freq random
        freqs = rng.integers(1, 6, size=5)
        phases = rng.uniform(0, 2 * math.pi, size=5)
        amps = rng.uniform(0.1, 0.4, size=5)
        x = (np.arange(n) + 0.5) * (2.0 / n)
        h = 0.5 + sum(a * np.cos(2 * math.pi * f * x / 2.0 + p) for a, f, p in zip(amps, freqs, phases))
        h = full_project(h, n)
        c = compute_c5(h)
        cands.append((h, c))
    cands.sort(key=lambda x: x[1])
    return cands[:2]

def get_parent():
    try:
        p = np.asarray(initial_h_values, dtype=float)
        if p.ndim != 1 and p.size >= 2:
            return None
        return p.copy()
    except Exception:
        return None

def run(seed=42, budget_s=1300, **kwargs):
    global _RUN_DEADLINE
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0:
        budget_s = 30.0
    _RUN_DEADLINE = t0 + max(1.0, budget_s - 20.0)
    
    rng = np.random.default_rng(seed)
    
    # Get initial solution
    parent = get_parent()
    if parent is not None and parent.size >= 2:
        n0 = parent.size
        h_best = full_project(parent, n0)
        c_best = compute_c5(h_best)
    else:
        n0 = 70
        h_best = full_project(np.full(n0, 0.5), n0)
        c_best = compute_c5(h_best)
    
    # Multi-resolution optimization
    resolutions = []
    for mult in [1, 2, 3, 4]:
        nn = ((mult * n0) // 2) * 2  # ensure even
        if nn >= 20 and nn <= 4 * n0 and nn not in resolutions:
            resolutions.append(nn)
    if n0 not in resolutions:
        resolutions.append(n0)
    resolutions.sort()
    
    total_time = budget_s - 40.0
    # Allocate time: more to lower resolutions
    weights = [1.0 / (1.0 + 0.3 * i) for i in range(len(resolutions))]
    wsum = sum(weights)
    times = [total_time * w / wsum for w in weights]
    
    for ri, n in enumerate(resolutions):
        if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE - 30:
            break
        t_limit = min(times[ri], _RUN_DEADLINE - time.time() - 20) if _RUN_DEADLINE else times[ri]
        if t_limit < 30:
            break
        
        # Generate candidates for this resolution
        cands = []
        
        # From parent/state
        if n == n0:
            cands.append((h_best.copy(), c_best))
        else:
            src = parent if parent is not None else h_best
            h_up = full_project(upsample(src, n), n)
            c_up = compute_c5(h_up)
            cands.append((h_up, c_up))
            if c_best < c_up:
                h_up2 = full_project(upsample(h_best, n), n)
                c_up2 = compute_c5(h_up2)
                if c_up2 < c_up:
                    cands.append((h_up2, c_up2))
        
        # Sinusoidal
        sin_cands = make_sin_init(n, rng)
        for h, c in sin_cands:
            if len(cands) < 8:
                cands.append((h, c))
        
        # Random
        rand_cands = make_random_init(n, n) if False else make_random_init(n, rng)
        for h, c in rand_cands:
            if len(cands) < 8:
                cands.append((h, c))
        
        cands.sort(key=lambda x: x[1])
        cands = cands[:4]
        
        # Time per candidate
        n_cands = len(cands)
        time_per_cand = t_limit / n_cands - 5.0
        if time_per_cand < 15:
            time_per_cand = 15
            n_cands = max(1, int(t_limit / time_per_cand))
            cands = cands[:n_cands]
        
        for ci, (h_init, c_init) in enumerate(cands):
            if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE - 15:
                break
            cand_time = time_per_cand - ci * 2
            if cand_time < 10:
                break
            
            h_work = h_init.copy()
            
            # Phase 1: Coarse soft-max (large alpha)
            alpha_list = [0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005]
            alpha_list = [a for a in alpha_list if a > 1e-6]
            
            # Distribute time across alphas (more time for smaller alpha)
            iters_total = max(200, int(cand_time * 800.0 / n * max(1.0, 40.0 / n) * 10))
            iters_total = min(iters_total, 50000)
            
            # Proportionally allocate
            n_alpha = len(alpha_list)
            fold = 0.6  # prefer later (smaller alpha) stages
            w_alpha = [(1.0 + fold * i) for i in range(n_alpha)]
            w_alpha_sum = sum(w_alpha)
            iters_per_alpha = [max(20, int(iters_total * w / w_alpha_sum)) for w in w_alpha]
            
            eta_base = 0.8 * n / 2.0  # step in h-space
            
            for ai, alpha in enumerate(alpha_list):
                if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE - 10:
                    break
                T = iters_per_alpha[ai]
                eta = min(eta_base, 2.0 * alpha * n)
                eta = max(eta, 0.5)
                
                h_work, c_local = projected_gradient(
                    h_work, n, alpha, eta, rng, T,
                    progress_callback=None
                )
                if c_local < c_best:
                    c_best = c_local
                    h_best = h_work.copy()
                
                if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE - 60:
                    break
            
            # Phase 2: SLSQP polish
            if _RUN_DEADLINE is not None and time.time() < _RUN_DEADLINE - 30:
                slq_time = min(40.0, max(5.0, (_RUN_DEADLINE - time.time()) - 20))
                h_polished = slsqp_polish(h_work, n, slq_time)
                c_polished = compute_c5(h_polished)
                if c_polished < c_best:
                    c_best = c_polished
                    h_best = h_polished.copy()
        
        # After finishing this resolution, record
        if c_best < c_init_cands_best if 'c_init_cands_best' in dir() else True:
            pass
    
    # Final validation
    h_final = full_project(h_best, int(h_best.size))
    c_final = compute_c5(h_final)
    n_final = int(h_best.size)
    
    return h_final, float(c_final), n_final
