def smooth_max_value(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted = np.clip(shifted, -700, 700)
    return max_c + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted = np.clip(shifted, -700, 700)

        s2 = 0.0

def subgradient_step(h, target, alpha):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    lag = k_star - (n - 1)
    # subgradient of corr[k_star]: (1-h[j+lag]) - h[j-lag]
    sg = np.zeros(n)
    for j in range(n):

        if 0 <= j + lag < n:
            val += 1.0 - h[j + lag]

            best_val = best_c5
            best_h_i = h[i]

                for sign in [1, -1]:
                    h_try = h.copy()

                    if new_val < 0 or new_val > 1:

- w_full: array of length 2n-1, where w_full[m] = w[m] (the softmax weight at lag m-(n-1))
- h: array of length n

Then:
s1[j] = sum_m w_full[m] * h[j + m - (n-1)] = (h ⊛ w_full)[j - (n-1) + (2n-2)] = (h ⊛ w_full)[j + n - 1]

Wait, np.correlate(h, w_full, 'full') has length n + (2n-1) - 1 = 3n - 2.
The formula is: result[m] = sum_k h[k] * w_full[k - m + (n-1)]

Hmm, this is getting too complicated. Let me just use the loop. For n=200, it's fast enough.

Actually, let me try a different vectorization. I can compute s1 for all j at once using the fact that:
s1[j] = sum_{k: 0≤j+k<n} w[k] * h[j+k]

Let lags = [-（n-1), -(n-2), ..., 0, 1, ..., n-1] (length 2n-1)
w[k_idx] is the weight at lag k = k_idx - (n-1)

For each lag k, the contribution to s1[j] is w[k_idx] * h[j+k] for j in [max(0,-k), min(n-1, n-1-k)].

I can accumulate this using slicing:
```python
s1 = np.zeros(n)
s2 = np.zeros(n)
for idx in range(2*n-1):
    k = idx - (n-1)
    w_k = w[idx]
    if w_k < 1e-14:
        continue
    if k >= 0:
        s1[:n-k] += w_k * h[k:]
        s2[k:] += w_k * h[:n-k]
    else:
        kk = -k
        s1[kk:] += w_k * h[:n-kk]
        s2[:n-kk] += w_k * h[kk:]
```

This is still O(n²) total but with numpy vectorization (each operation is O(n) in C). For n=200, that's 399 numpy operations on arrays of length ~200. Very fast (microseconds each).

import numpy as np
from scipy.optimize import minimize
import time

# Global params for L-BFGS-B objective (avoids closures)
_GP = {'beta': 10.0, 'target': 47.5, 'lam': 1e6}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    one_m_h = 1.0 - h
    corr = np.correlate(h, one_m_h, mode='full')
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    val = max_c + np.log(sum_exp) / beta
    w = exp_s / sum_exp
    
    # Gradient vectorized
    s1 = np.zeros(n)
    s2 = np.zeros(n)
    for idx in range(2*n-1):
        wk = w[idx]
        if wk < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            s1[:n-k] += wk * h[k:]
            s2[k:] += wk * h[:n-k]
        else:
            kk = -k
            s1[kk:] += wk * h[:n-kk]
            s2[:n-kk] += wk * h[kk:]
    grad = (1.0 - s1 - s2) * dx
    return val, grad

def project_sum(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < (1.0 - 1e-12)
        else:
            mask = h > 1e-12
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def opt_func(x):
    beta = _GP['beta']
    target = _GP['target']
    lam = _GP['lam']
    hv = np.clip(x, 0.0, 1.0)
    v, _ = smooth_max_and_grad(hv, beta)
    v += lam * (np.sum(hv) - target) ** 2
    return v

def opt_grad(x):
    beta = _GP['beta']
    target = _GP['target']
    lam = _GP['lam']
    hv = np.clip(x, 0.0, 1.0)
    _, g = smooth_max_and_grad(hv, beta)
    g += 2.0 * lam * (np.sum(hv) - target)
    return g

def run_lbfgs(h0, beta, target, max_iter, ftol, lam=1e6):
    _GP['beta'] = beta
    _GP['target'] = target
    _GP['lam'] = lam
    n = len(h0)
    result = minimize(opt_func, h0, jac=opt_grad, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': max_iter, 'ftol': ftol, 'gtol': 1e-7})
    if result.x is not None:
        return project_sum(result.x, target)
    return project_sum(h0, target)

def subgradient_step(h, target, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    tol = 0.01 * max_val + 1e-6
    active = np.where(corr >= max_val - tol)[0]
    sg_total = np.zeros(n)
    for ak in active:
        lag = ak - (n - 1)
        sg = np.zeros(n)
        for j in range(n):
            v = 0.0
            if 0 <= j + lag < n:
                v += 1.0 - h[j + lag]
            if 0 <= j - lag < n:
                v -= h[j - lag]
            sg[j] = v * dx
        sg_total += sg
    sg_total /= len(active)
    h_new = project_sum(h - alpha * sg_total, target)
    return h_new

def coordinate_descent(h, target, n_sweeps, max_time):
    n = len(h)
    best_c5 = compute_c5(h)
    t0 = time.time()
    for sweep in range(n_sweeps):
        if time.time() - t0 > max_time:
            break
        improved = False
        for i in range(n):
            best_val = best_c5
            best_delta = 0.0
            for di in [0.002, 0.005, 0.01, 0.02, 0.05]:
                for sign in [1.0, -1.0]:
                    new_v = h[i] + sign * di
                    if new_v < 0.0 or new_v > 1.0:
                        continue
                    h_try = h.copy()
                    h_try[i] = new_v
                    h_try = project_sum(h_try, target)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_val - 1e-13:
                        best_val = c5_try
                        best_delta = sign * di
            if best_delta != 0.0:
                h[i] += best_delta
                h = project_sum(h, target)
                best_c5 = best_val
                improved = True
        if not improved:
            break
    return h, best_c5
