```python
def slsqp_polish(h, n, max_iter=3000):
    def obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.success or res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def subgradient_step(h, n, alpha, n_active=5):
    """One subgradient step for max of correlations."""
    corr = get_corr(h)
    max_val = np.max(corr)
    # Find most active lags
    thresholds = np.sort(corr)[-n_active:]
    active = corr >= thresholds[0] - 1e-14
    ells = np.arange(-(n-1), n)
    
    # Compute subgradient
    grad = np.zeros(n)
    for idx in np.where(active)[0]:
        ell = ells[idx]
        for j in range(n):
            # d c[idx] / d h[j]
            val = 0.0
            if 0 <= j - ell < n:
                val += (1.0 - h[j - ell])
            if 0 <= j + ell < n:
                val += (-h[j + ell])
            grad[j] += val
    grad /= len(np.where(active)[0])
    
    h_new = h - alpha * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = make_feasible(h_new, n)
    return h_new
```

Actually, for the subgradient step, I can compute it more efficiently:
grad[j] = (1/|A|) * sum_{m active} [(1-h[j-ℓ_m])*I(valid(j-ℓ_m)) + (-h[j+ℓ_m])*I(valid(j+ℓ_m))]

For each active lag ℓ_m:
- Term 1: (1-h[j-ℓ_m]) for j ∈ [max(0,ℓ_m), min(n-1, n-1+ℓ_m)]
- Term 2: -h[j+ℓ_m] for j ∈ [max(0,-ℓ_m), min(n-1, n-1-ℓ_m)]

```python
def subgradient_step(h, n, alpha, n_active=5):
    corr = get_corr(h)
    thresh = np.sort(corr)[-n_active:]
    active_idx = np.where(corr >= thresh[0] - 1e-14)[0]
    ells = np.arange(-(n-1), n)[active_idx]
    
    grad = np.zeros(n)
    for ell in ells:
        # Term 1: (1-h[j-ell]) for valid j
        j_start = max(0, int(ell))
        j_end = min(n, int(n + ell))
        if j_start < j_end:
            grad[j_start:j_end] += (1.0 - h[j_start-int(ell):j_end-int(ell)])
        
        # Term 2: -h[j+ell] for valid j
        j_start2 = max(0, int(-ell))
        j_end2 = min(n, int(n - ell))
        if j_start2 < j_end2:
            grad[j_start2:j_end2] += -h[j_start2+int(ell):j_end2+int(ell)]
    
    grad /= len(ells)
    h_new = h - alpha * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = make_feasible(h_new, n)
    return h_new
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def coord_opt_j(h, j, n, corr, ells):
    idx1 = j - ells
    idx2 = j + ells
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    a = np.zeros(2 * n - 1)
    if np.any(valid1):
        a[valid1] += 1.0 - h[idx1[valid1]]
    if np.any(valid2):
        a[valid2] += -h[idx2[valid2]]
    b = corr - h[j] * a
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = a * mid + b
        mx = np.max(vals)
        active = vals >= mx - 1e-14
        sg = np.mean(a[active])
        if sg > 1e-13:
            hi = mid
        elif sg < -1e-13:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = get_corr(h)
    ells = np.arange(-(n - 1), n)
    start = time.time()

            break
        for j in range(n):
            t_opt = coord_opt_j(h, j, n, corr, ells)
            h[j] = t_opt
        h = make_feasible(h, n)
        corr = get_corr(h)
        c5 = np.max(corr) * (2.0 / n)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)


def sum_constraint_fn(x, n):
    return np.sum(x) - n / 2.0


def c5_obj_slq(x):
    return np.max(np.correlate(x, 1.0 - x, mode='full'))


def slsqp_polish(h, n, max_iter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint_fn, 'args': (n,)}]
    try:
        res = minimize(c5_obj_slq, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None


def subgradient_step(h, n, alpha, n_active=5):
    corr = get_corr(h)
    thresh = np.sort(corr)[-n_active:]
    active_idx = np.where(corr >= thresh[0] - 1e-13)[0]
    ells = np.arange(-(n - 1), n)[active_idx]
    grad = np.zeros(n)
    for ell in ells:
        e = int(ell)
        js1 = max(0, e)
        je1 = min(n, n + e)
        if js1 < je1:
            grad[js1:je1] += (1.0 - h[js1 - e:je1 - e])
        js2 = max(0, -e)
        je2 = min(n, n - e)
        if js2 < je2:
            grad[js2:je2] += -h[js2 + e:je2 + e]
    grad /= max(len(ells), 1)
    h_new = h - alpha * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    return make_feasible(h_new, n)
```
