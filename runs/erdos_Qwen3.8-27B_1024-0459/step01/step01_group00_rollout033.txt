import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(500):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.full(n_dst, 0.5)
    x_old = np.arange(len(h_src)) / len(h_src) * 2.0
    x_new = np.arange(n_dst) / n_dst * 2.0
    return np.interp(x_new, x_old, h_src)

def smooth_beta_objective(h, beta):
    corr = compute_corr_full(h)
    cmax = corr.max()
    return cmax + (1.0/beta) * np.log(np.sum(np.exp(beta*(corr - cmax))))

def run_slsqp_smooth(h0, n, beta_schedule, max_iter_per_step=2000):
    h = make_feasible(h0.copy(), n)
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    
    for beta in beta_schedule:
        def obj(x):
            return smooth_beta_objective(x, beta)
        def con_fun(x):
            return x.sum() - target
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': max_iter_per_step, 'ftol': 1e-14})
            h_new = make_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
            h = h_new
        except Exception:
            pass
    return h, best_c5

def pair_swap_search(h, n, n_trials=5000, base_step=0.05):
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    for _ in range(n_trials):

        j = np.random.randint(n)

def smooth_obj_and_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exp_corr = np.exp(beta * (corr - m))
    sum_exp = exp_corr.sum()
    obj = m + np.log(sum_exp) / beta
    
    # Softmax weights
    w = exp_corr / sum_exp  # shape (2n-1,), indexed from -(n-1) to n-1
    
    # Gradient: for each i, compute ∑_k w_k * ∂corr_k/∂h[i]
    grad = np.zeros(n)
    
    # Positive lags k = 0, 1, ..., n-1 (indices 0 to n-1 in corr, which is offset by n-1)
    # corr[n-1 + k] corresponds to lag k
    for i in range(n):
        g = 0.0
        # k = 0
        g += w[n-1] * (1 - 2*h[i])
        # k = 1 to n-1 (correspond to corr[n-1+k])
        for k in range(1, n):
            if i < n - k:
                g += w[n-1+k] * (1 - h[i+k])
            if i >= k:
                g -= w[n-1+k] * h[i-k]
            # Negative lag -k (correspond to corr[n-1-k])
            if i >= k:
                g += w[n-1-k] * (1 - h[i-k])
            if i < n - k:
                g -= w[n-1-k] * h[i+k]
        grad[i] = g
    
    return obj, grad

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exp_corr = np.exp(beta * (corr - m))
    sum_exp = exp_corr.sum()
    obj = m + np.log(sum_exp) / beta
    w = exp_corr / sum_exp
    
    # w[n-1+k] for k=-n+1 to n-1
    # For positive lag k (0..n-1): w_pos[k] = w[n-1+k]
    # For negative lag -k (k=1..n-1): w_neg[k] = w[n-1-k]
    
    w_pos = w[n-1:]  # length n, w_pos[k] = w[n-1+k] for k=0..n-1
    w_neg = w[:n-1]  # length n-1, w_neg[k] = w[n-1-1-k] for k=0..n-2
                     # w_neg[k] corresponds to lag -(k+1)
    
    grad = np.zeros(n)
    
    for i in range(n):
        g = 0.0
        # k=0 term
        g += w_pos[0] * (1 - 2*h[i])
        # Positive k=1..n-1
        max_k = n - 1 - i  # max k such that i+k < n
        for k in range(1, min(max_k+1, n)):
            g += w_pos[k] * (1 - h[i+k])
        for k in range(1, min(i+1, n)):
            g -= w_pos[k] * h[i-k]
        # Negative k=1..n-1 (using w_neg[k-1] for lag -k)
        for k in range(1, min(i+1, n)):
            g += w_neg[k-1] * (1 - h[i-k])
        for k in range(1, min(max_k+1, n)):
            g -= w_neg[k-1] * h[i+k]
        grad[i] = g
    
    return obj, grad
