import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def get_correlations(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def gradient_lag_raw(h, n, k_idx):
    """Raw gradient of C_{k_idx} w.r.t. h (without dx factor)."""
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            n_valid = n - lag
            grad[:n_valid] += (1.0 - h[lag:lag + n_valid])
            grad[lag:lag + n_valid] -= h[:n_valid]
    else:
        l = -lag
        grad[l:] += (1.0 - h[:n - l])
        grad[:n - l] -= h[l:]
    return grad

def soft_obj_and_grad(h, n, beta):
    """Compute soft objective and its gradient."""
    dx = 2.0 / n
    C = get_correlations(h, n)
    s = beta * dx * C
    m = np.max(s)
    exp_s = np.exp(s - m)
    w = exp_s / np.sum(exp_s)
    obj = (m + np.log(np.sum(exp_s)))
    # Gradient: dx * sum_k w[k] * dC_k/dh
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        grad += w[k] * gradient_lag_raw(h, n, k)
    grad *= dx
    return obj, grad

def hard_obj_and_grad(h, n):
    """Compute hard objective and subgradient."""
    dx = 2.0 / n
    C = get_correlations(h, n)
    k_max = np.argmax(C)
    obj = C[k_max] * dx
    grad = gradient_lag_raw(h, n, k_max) * dx
    return obj, grad, k_max, C

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = h.sum() - target
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h > 1e-14
        else:
            mask = h < 1.0 - 1e-14
        if mask.sum() == 0:
            break
        h[mask] -= diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def projected_gradient_step(h, n, beta, lr):
    """One projected gradient step with soft objective."""
    obj, grad = soft_obj_and_grad(h, n, beta)
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new

def slsqp_wrapper_fun(h, n, beta):
    obj, _ = soft_obj_and_grad(h, n, beta)
    return obj

def slsqp_wrapper_jac(h, n, beta):
    _, grad = soft_obj_and_grad(h, n, beta)
    return grad

def slsqp_cons_fun(h, n):
    return h.sum() - n / 2.0

def slsqp_cons_jac(h, n):
    return np.ones(n)

def interpolate_h(h_old, n_old, n_new):
    """Interpolate h from old grid to new grid."""
    x_old = np.linspace(0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old, period=2.0)  # periodic interp?
    # Actually, not periodic. Let's use linear interp with clamping.
    h_new = np.interp(x_new, x_old, h_old)
    # For values beyond the range, use nearest
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new)

            h_new = slsqp_slsqp_run(best_h, n_best, beta, 400, 1e-14)
            c5_new = compute_c5(h_new, n_best)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    # Phase 2: Hard SLSQP passes
    for _pass in range(4):
        if time.time() - t0 > budget_s - 400:
            break
        try:
            h_new = slsqp_hard_run(best_h, n_best, 600, 1e-14)
            c5_new = compute_c5(h_new, n_best)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    # Phase 3: Diversified restarts at original resolution
    for i in range(25):
        if time.time() - t0 > budget_s - 350:
            break
        if i % 3 == 0:
            h_pert = block_perturb(best_h, n_best, rng, 3, 0.15)
        elif i % 3 == 1:
            h_pert = gaussian_perturb(best_h, n_best, rng, 0.04 * (1 + i / 10.0))
        else:
            h_pert = block_perturb(best_h, n_best, rng, 5, 0.1)
        try:
            h_new = slsqp_slsqp_run(h_pert, n_best, 100.0, 300, 1e-13)
            c5_new = compute_c5(h_new, n_best)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    # Phase 4: Upscale to higher resolution and optimize with projected gradient
    for n_new in [2 * n_orig, 3 * n_orig]:
        if time.time() - t0 > budget_s - 200:
            break
        if n_new > 600:
            break
        h_up = interpolate_to_resolution(best_h, n_best, n_new)
        # Projected gradient with soft objective
        h_pg, c5_pg = projected_gradient_loop(
            h_up, n_new, 2000, 5.0, 200.0, 0.1, t0, budget_s, 100
        )
        if c5_pg < best_c5:
            best_h = h_pg
            best_c5 = c5_pg
            n_best = n_new
        else:
            # Try from best_h interpolated
            h_up2 = interpolate_to_resolution(best_h, n_best, n_new)
            h_pg2, c5_pg2 = projected_gradient_loop(
                h_up2, n_new, 1500, 10.0, 300.0, 0.08, t0, budget_s, 80
            )
            if c5_pg2 < best_c5:
                best_h = h_pg2
                best_c5 = c5_pg2
                n_best = n_new

    # Phase 5: Subgradient refinement at current best resolution
    if time.time() - t0 < budget_s - 80:
        h_sg, c5_sg = pg_subgradient_loop(best_h, n_best, 3000, t0, budget_s, 40)
        if c5_sg < best_c5:
            best_h = h_sg
            best_c5 = c5_sg

    # Phase 6: More restarts at higher resolution if available
    if n_best > n_orig and time.time() - t0 < budget_s - 100:
        for i in range(15):
            if time.time() - t0 > budget_s - 60:
                break
            if i % 2 == 0:
                h_pert = block_perturb(best_h, n_best, rng, 2, 0.1)
            else:
                h_pert = gaussian_perturb(best_h, n_best, rng, 0.03)
            h_pg, c5_pg = projected_gradient_loop(
                h_pert, n_best, 800, 20.0, 400.0, 0.05, t0, budget_s, 30
            )
            if c5_pg < best_c5:
                best_h = h_pg
                best_c5 = c5_pg

    # Phase 7: Final SLSQP polish
    if time.time() - t0 < budget_s - 15:
        try:
            h_final = slsqp_hard_run(best_h, n_best, 300, 1e-15)
            c5_final = compute_c5(h_final, n_best)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
        except Exception:
            pass

    # Phase 8: One more soft SLSQP pass
    if time.time() - t0 < budget_s - 10:
        try:
            h_final2 = slsqp_slsqp_run(best_h, n_best, 500.0, 200, 1e-14)
            c5_final2 = compute_c5(h_final2, n_best)
            if c5_final2 < best_c5:
                best_h = h_final2
                best_c5 = c5_final2
        except Exception:
            pass

    # Final feasibility check
    best_h = project_feasible(best_h, n_best)
    best_c5 = compute_c5(best_h, n_best)

    # Validate with evaluator if available
    try:
        val = evaluate_erdos_solution(best_h, n_best)
        if val is not None and 0.0 < val < 2.0:
            best_c5 = min(best_c5, float(val))
    except Exception:
        pass

    return (best_h, best_c5, n_best)
