                grad[:no] += wk * (1.0 - x[-lag:])
                grad[-lag:] -= wk * x[:no]
    grad *= dx
    s = x.sum() - n / 2.0
    grad += 2.0 * mu * s
    return grad

def hm_obj(x):
    n = _current_n
    c = np.correlate(x, 1.0 - x, mode="full")
    return float(np.max(c) * (2.0 / n))

def hm_grad(x):
    n = _current_n
    c = np.correlate(x, 1.0 - x, mode="full")
    ki = int(np.argmax(c))
    lag = ki - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag >= 0:
        rem = n - lag
        if rem > 0:
            grad[lag:] += (1.0 - x[:rem])
            grad[:rem] -= x[lag:]
    else:
        no = n + lag
        if no > 0:
            grad[:no] += (1.0 - x[-lag:])
            grad[-lag:] -= x[:no]
    return grad * dx

def sum_eq_fun(x):
    return x.sum() - _current_n / 2.0

def sum_eq_jac(x):
    return np.ones(len(x))
