from scipy.optimize import minimize
from scipy.special import logsumexp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def compute_correlations(h):
    """Compute all correlation values."""
    comp = 1.0 - h
    return np.correlate(h, comp, mode='full')

def smooth_max(x, beta):
    """Compute log-sum-exp based smooth max."""
    x_shifted = beta * (x - np.max(x))
    x_shifted = np.clip(x_shifted, -500, 500)
    return np.max(x) + np.log(np.sum(np.exp(x_shifted))) / beta

def smooth_max_grad_weights(x, beta):
    """Compute weights for smooth max gradient: w_m = exp(beta*x_m) / sum(exp(beta*x_l))"""
    x_shifted = beta * (x - np.max(x))
    x_shifted = np.clip(x_shifted, -500, 500)
    exps = np.exp(x_shifted)
    return exps / np.sum(exps)

def smooth_max_objective(h, beta):
    corr = compute_correlations(h)
    return smooth_max(corr, beta)

def smooth_max_gradient(h, beta):
    """Analytical gradient of smooth max of correlations."""
    corr = compute_correlations(h)
    weights = smooth_max_grad_weights(corr, beta)
    n = len(h)
    grad = np.zeros(n)
    for m in range(2*n - 1):
        if weights[m] < 1e-15:
            continue
        shift = m - (n - 1)
        # ∂corr[m]/∂h[j] = (1-h[j-shift]) * I[0≤j-shift≤n-1] - h[j+shift] * I[0≤j+shift≤n-1]
        # For each valid j:
        # term1: j - shift in [0, n-1] → j in [shift, shift+n-1], contribution (1-h[j-shift])
        # term2: j + shift in [0, n-1] → j in [-shift, n-1-shift], contribution -h[j+shift]
        j_min1 = max(0, shift)
        j_max1 = min(n-1, shift + n - 1)
        if j_min1 <= j_max1:
            grad[j_min1:j_max1+1] += weights[m] * (1.0 - h[j_min1 - shift : j_max1 - shift + 1])
        j_min2 = max(0, -shift)

        if j_min2 <= j_max2:
            grad[j_min2:j_max2+1] += weights[m] * (-h[j_min2 + shift : j_max2 + shift + 1])
    return grad

def project_sum(h, target):

    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:

        if np.any(free):
            per = diff / np.count_nonzero(free)
            h[free] = np.clip(h[free] + per, 0.0, 1.0)
    return h

        return smooth_max_objective(h, beta)

        return smooth_max_gradient(h, beta)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def compute_correlations(h):
    comp = 1.0 - h
    return np.correlate(h, comp, mode='full')

def smooth_max_value(x, beta):
    mx = np.max(x)
    shifted = np.clip(beta * (x - mx), -500, 500)
    return mx + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(x, beta):
    mx = np.max(x)
    shifted = np.clip(beta * (x - mx), -500, 500)
    exps = np.exp(shifted)
    s = np.sum(exps)
    if s == 0:
        return np.ones(len(x)) / len(x)
    return exps / s

def smo_max_obj(h, beta):
    corr = compute_correlations(h)
    return smooth_max_value(corr, beta)

def smo_max_grad(h, beta):
    n = len(h)
    corr = compute_correlations(h)
    w = smooth_max_weights(corr, beta)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-16:
            continue
        shift = m - (n - 1)
        # term1: (1 - h[j-shift]) for j in [max(0,shift), min(n-1, shift+n-1)]
        lo1 = max(0, shift)
        hi1 = min(n - 1, shift + n - 1)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[idx] += wm * (1.0 - h[idx - shift])
        # term2: -h[j+shift] for j in [max(0,-shift), min(n-1, n-1-shift)]
        lo2 = max(0, -shift)
        hi2 = min(n - 1, n - 1 - shift)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            grad[idx] += wm * (-h[idx + shift])
    return grad

def project_sum(h, target):
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:

        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s

        if diff > 0:
            free = h < 0.999999
        else:
            free = h > 1e-6
        if np.any(free):
            per = diff / np.count_nonzero(free)
            h[free] = np.clip(h[free] + per, 0.0, 1.0)
    return h

def slsqp_opt(h0, n, beta, max_iter=250, ftol=1e-13):
    target = n / 2.0
    h0 = project_sum(h0, target)

    def obj(h):
        return smo_max_obj(h, beta)

    def grad(h):
        return smo_max_grad(h, beta)

    def con_fun(h):
        return np.sum(h) - target

    def con_jac(h):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': ftol})
        if res.x is not None:
            return project_sum(res.x, target)
    except Exception:
        pass
    return h0

def projected_grad_step(h, grad, step, target):
    h_new = h - step * grad
    h_new = project_sum(h_new, target)
    return h_new

def projected_gradient_opt(h0, n, beta, n_iter=2000, lr_init=0.01):
    target = n / 2.0
    h = project_sum(h0, target)
    best_h = h.copy()
    best_val = smo_max_obj(h, beta)
    lr = lr_init

    for it in range(n_iter):
        grad = smo_max_grad(h, beta)
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad_norm = grad / gnorm
        # Line search
        found = False
        for ls in range(50):
            step = lr / (1.0 + ls * 0.5)
            h_try = projected_gradient_step(h, grad_norm, step, target)
            val_try = smo_max_obj(h_try, beta)
            if val_try < best_val - 1e-14 * (it + 1):
                h = h_try
                found = True
                if val_try < best_val:
                    best_val = val_try
                    best_h = h.copy()
                break
        if not found:
            lr *= 0.5
            if lr < 1e-12:
                break
        else:
            lr = min(lr * 1.1, 0.1)
    return best_h, best_val
