def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_cort_full(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def smooth_obj(h, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / len(h)
    m = np.max(corr)
    # Log-sum-exp for numerical stability
    shifted = corr - m
    exps = np.exp(beta * shifted)
    s = np.sum(exps)
    if s > 0:
        return (m + np.log(s) / beta) * dx
    return m * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final adjustment on free variables
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

        ones_list = np.array(sorted(ones_set))
        zeros_list = np.array(sorted(set(range(n)) - ones_set))
        
        if len(ones_list) == 0 or len(zeros_list) == 0:

                    idx_pert = rng.choice(n, size=n_pert, replace=False)
                    hp[idx_pert] = np.clip(hp[idx_pert] + 0.1 * rng.randn(n_pert), 0, 1)
                    hp = project_feasible(hp, n)
                    try:
                        res = minimize(smooth_objective, hp, method='SLSQP',
                                       bounds=bounds, constraints=cons,
                                       options={'maxiter': 250, 'ftol': 1e-13, 'iprint': 0})
                        hp = project_feasible(res.x, n)
                        pc5 = compute_c5(hp)
                        if pc5 < best_c5:
                            best_c5 = pc5
                            best_h = hp.copy()
                            best_n = n
                            print("  n={} pert{}: C5={:.6f}".format(n, pert, best_c5))
                    except Exception:
                        pass

        # Binary local search
        if time_left() > 30:
            h_bi = binary_adjust(best_h, n)
            hbl, c5b = binary_flip_search(h_bi, n, rng, max_iters=4000)
            if c5b < best_c5:
                best_c5 = c5b
                best_h = hbl.copy()
                best_n = n
                print("  n={} binFLS: C5={:.6f}".format(n, best_c5))

        # Full sweep binary (only for smaller n)
        if n <= 128 and time_left() > 20:
            h_bs, c5bs = full_sweep_binary(best_h, n, max_sweeps=2)
            if c5bs < best_c5:
                best_c5 = c5bs
                best_h = h_bs.copy()
                best_n = n
                print("  n={} binSW: C5={:.6f}".format(n, best_c5))

        # Simulated annealing
        if time_left() > 60:
            ha = binary_adjust(best_h, n)
            hs, c5s = anneal_binary(ha, n, rng, iters=6000, T_init=0.05, T_final=0.001)
            if c5s < best_c5:
                best_c5 = c5s
                best_h = hs.copy()
                best_n = n
                print("  n={} SA: C5={:.6f}".format(n, best_c5))
            # Do a quick local search on the SA result
            if time_left() > 20:
                h_sl, c5_sl = binary_flip_search(hs, n, rng, max_iters=2000)
                if c5_sl < best_c5:
                    best_c5 = c5_sl
                    best_h = h_sl.copy()
                    best_n = n
                    print("  n={} SA+LS: C5={:.6f}".format(n, best_c5))

    # Final: try rounding
    if time_left() > 5:
        hr = binary_adjust(best_h, best_n)
        cr = compute_c5(hr)
        if cr < best_c5:
            best_c5 = cr
            best_h = hr.copy()
            print("  Rounded: C5={:.6f}".format(best_c5))
        # Quick local search on rounded
        if time_left() > 15:
            h_rf, c5_rf = binary_flip_search(hr, best_n, rng, max_iters=3000)
            if c5_rf < best_c5:
                best_c5 = c5_rf
                best_h = h_rf.copy()
                print("  Rounded+LS: C5={:.6f}".format(best_c5))

    # Final continuous polish at best_n
    if time_left() > 15 and best_n <= 200:
        _n = best_n
        _beta = 500.0
        bounds = [(0.0, 1.0)] * best_n
        cons = [{'type': 'eq', 'fun': sum_constraint}]
        h_fp = project_feasible(best_h, best_n)
        try:
            res = minimize(smooth_objective, h_fp, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-14, 'iprint': 0})
            h_fp = project_feasible(res.x, best_n)
            cf = compute_c5(h_fp)
            if cf < best_c5:
                best_c5 = cf
                best_h = h_fp.copy()
                print("  Final polish: C5={:.6f}".format(best_c5))
        except Exception:
            pass

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        ec = evaluate_erdos_solution(best_h)
        if ec < final_c5:
            final_c5 = ec
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, float(final_c5), best_n)
