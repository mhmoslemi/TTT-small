```python
def compute_lag_change(h, i, j, delta, k, n):
    """Compute change in c_t when h[i]+=delta, h[j]-=delta, for lag k=t-(n-1)."""
    # c_t = sum_{j_elem: valid} h[j_elem+k]*(1-h[j_elem])
    # The j_elem values that matter are: i (term A), i-k (term B), j (term C), j-k (term D)
    # But some of these may be the same, so use a set of unique j_elem values.
    
    j_elems = set()
    if 0 <= i < n and 0 <= i+k < n:
        j_elems.add(i)  # Term A
    if 0 <= i-k < n and 0 <= i < n:
        j_elems.add(i-k)  # Term B
    if 0 <= j < n and 0 <= j+k < n:
        j_elems.add(j)  # Term C
    if 0 <= j-k < n and 0 <= j < n:
        j_elems.add(j-k)  # Term D
    
    old_val = 0.0
    new_val = 0.0
    for je in j_elems:
        old_val += h[je+k] * (1-h[je])
        h_new_jek = h[je+k]
        if je+k == i:
            h_new_jek = h[je+k] + delta
        elif je+k == j:
            h_new_jek = h[je+k] - delta
        h_new_je = h[je]
        if je == i:
            h_new_je = h[je] + delta
        elif je == j:
            h_new_je = h[je] - delta
        new_val += h_new_jek * (1 - h_new_je)
    
    return new_val - old_val
```

```python
def _evaluate(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)

def _project(x, target):
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-13:
        return x
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:

    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)

            if abs(res2) > 1e-13:

def _upsample(h, n_new):
    h = np.asarray(h, dtype=float)
    n_old = h.size
    if n_old == n_new:
        return h.copy()
    x0 = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x1 = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x1, x0, h, left=h[0], right=h[-1])
    return _project(h_new, 0.5 * n_new)

def _compute_d(h, k, n):
    """Compute d_k(i) = 1 - h[i-|k|] - h[i+|k|] for all i (0 for k=0: 1-2*h[i])."""
    if k == 0:
        return 1.0 - 2.0 * h
    lag = abs(k)
    if lag >= n:
        return np.full(n, 1.0, dtype=float)
    h_im = np.zeros(n, dtype=float)
    h_ip = np.zeros(n, dtype=float)
    h_im[lag:] = h[:n - lag]
    h_ip[:n - lag] = h[lag:]
    return 1.0 - h_im - h_ip

def _lag_change(h, i, j, delta, k, n):
    """Exact change in c_t when h[i]+=delta, h[j]-=delta for lag k."""
    je_list = []
    if 0 <= i < n and 0 <= i + k < n:
        je_list.append(i)
    if 0 <= i - k < n and 0 <= i < n:
        if i - k not in je_list:
            je_list.append(i - k)
    if 0 <= j < n and 0 <= j + k < n:
        if j not in je_list:
            je_list.append(j)
    if 0 <= j - k < n and 0 <= j < n:
        if j - k not in je_list:
            je_list.append(j - k)
    old = 0.0
    new = 0.0
    for je in je_list:
        old += h[je + k] * (1.0 - h[je])
        a = h[je + k]
        b = h[je]
        a_new = a
        if je + k == i:
            a_new = a + delta
        elif je + k == j:
            a_new = a - delta
        b_new = b
        if je == i:
            b_new = b + delta
        elif je == j:
            b_new = b - delta
        new += a_new * (1.0 - b_new)
    return new - old

def _pair_search(h, deadline, m=8, max_rounds=20000):
    """Aggressive pair-transfer local search using exact linear formula."""
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    target = 0.5 * n
    h = _project(h, target)
    best_c = _evaluate(h)
    center = n - 1
    stall = 0

    for rnd in range(max_rounds):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        n_corr = corr.size
        top_m = min(m, n_corr)
        top_idx = np.argsort(corr)[-top_m:]
        top_vals = corr[top_idx].copy()
        top_max = top_vals.max()

        # Compute d for each top lag
        d_list = []
        for t in top_idx:
            k = t - center
            d_list.append(_compute_d(h, k, n))
        d_arr = np.array(d_list)  # shape (top_m, n)
        d_avg = d_arr.mean(axis=0)

        # Find candidate pairs: increase i, decrease j
        # i should have low d_avg and h[i] < 1
        # j should have high d_avg and h[j] > 0
        candidates_i = np.where((d_avg < 0) & (h < 1.0 - 1e-6))[0]
        candidates_j = np.where((d_avg > 0) & (h > 1e-6))[0]

        if candidates_i.size == 0 or candidates_j.size == 0:
            stall += 1
            if stall > 20:
                break
            continue

        # Limit candidates for speed
        if candidates_i.size > 40:
            candidates_i = candidates_i[np.argmin(d_avg[candidates_i])][:40] if len(d_avg[candidates_i]) >= 40 else candidates_i
            idx_i = np.argsort(d_avg[candidates_i])[:40]
            candidates_i = candidates_i[idx_i]
        if candidates_j.size > 40:
            idx_j = np.argsort(-d_avg[candidates_j])[:40]
            candidates_j = candidates_j[idx_j]

        found_improve = False
        for ii in candidates_i:
            ii = int(ii)
            if h[ii] >= 1.0 - 1e-7:
                continue
            for jj in candidates_j:
                jj = int(jj)
                if ii == jj:
                    continue
                if h[jj] <= 1e-7:
                    continue

                # Compute s_k = d_k(i) - d_k(j) for each top lag
                s_vals = d_arr[:, ii] - d_arr[:, jj]
                # Predicted new values: top_vals + delta * s_vals
                # Find optimal delta in [0, min(1-h[ii], h[jj])]
                delta_max = min(1.0 - h[ii], h[jj])
                if delta_max <= 1e-10:
                    continue

                # Min of max of linear functions: find breakpoints
                # f(d) = max_k (top_vals[k] + d * s_vals[k])
                # Breakpoints where top_vals[k] + d*s_vals[k] = top_vals[l] + d*s_vals[l]
                best_delta = 0.0
                best_val = float(np.max(top_vals))  # f(0)
                deltas = [0.0, delta_max]
                for a in range(top_m):
                    for b in range(a + 1, top_m):
                        den = s_vals[a] - s_vals[b]
                        if abs(den) > 1e-14:
                            dd = (top_vals[b] - top_vals[a]) / den
                            if 0.0 < dd < delta_max:
                                deltas.append(dd)
                deltas = sorted(set(deltas))
                for dd in deltas:
                    val = float(np.max(top_vals + dd * s_vals))
                    if val < best_val - 1e-12:
                        best_val = val
                        best_delta = dd

                if best_delta < 1e-7 or best_val >= top_max - 1e-10:
                    continue

                # Verify with exact computation
                new_h = h.copy()
                new_h[ii] += best_delta
                new_h[jj] -= best_delta
                if new_h[ii] < 0: new_h[ii] = 0
                if new_h[ii] > 1: new_h[ii] = 1
                if new_h[jj] < 0: new_h[jj] = 0
                if new_h[jj] > 1: new_h[jj] = 1

                # Check actual top-m values
                corr_new = np.correlate(new_h, 1.0 - new_h, mode="full")
                new_max_at_top = float(np.max(corr_new[top_idx]))
                new_global_max = float(np.max(corr_new))

                if new_global_max < float(np.max(corr)) - 1e-12:
                    h = new_h
                    found_improve = True
                    # Recompute top
                    stall = 0
                    break
            if found_improve:
                break
        if not found_improve:
            stall += 1
            if stall > 30:
                break

    h = _project(h, target)
    final_c = _evaluate(h)
    return h, final_c

def _lbgfb_optimize(h, deadline, alpha=25.0, maxiter=200):
    """L-BFGS-B with log-sum-exp smoothing."""
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    target = 0.5 * n
    h = _project(h, target)
    best_h = h.copy()
    best_c = _evaluate(best_h)

    def obj(x):
        h_t = np.asarray(x, dtype=float)
        if not np.all(np.isfinite(h_t)):
            return 1e100
        corr = np.correlate(h_t, 1.0 - h_t, mode="full")
        if corr.size == 0:
            return 1e100
        mx = float(np.max(corr))
        y = alpha * (corr - mx)
        y = np.clip(y, -700.0, 0.0)
        e = np.exp(y)
        z = float(np.sum(e))
        if z <= 0:
            return 1e100
        return (2.0 / n) * (mx + np.log(z) / alpha)

    def obj_grad(x):
        h_t = np.asarray(x, dtype=float)
        n_t = h_t.size
        corr = np.correlate(h_t, 1.0 - h_t, mode="full")
        mx = float(np.max(corr))
        y = alpha * (corr - mx)
        y = np.clip(y, -700.0, 0.0)
        e = np.exp(y)
        z = float(np.sum(e))
        if z <= 0:
            return np.zeros(n_t), 0.0
        w = e / z
        g = np.zeros(n_t, dtype=float)
        center = n_t - 1
        for t in range(corr.size):
            p = w[t]
            if p < 1e-14:
                continue
            k = t - center
            if k == 0:
                g += p * (1.0 - 2.0 * h_t)
            else:
                lag = abs(k)
                if lag >= n_t:
                    continue
                d = _compute_d(h_t, k, n_t)
                g += p * d
        return (2.0 / n_t) * g, obj(h_t)

    try:
        res = minimize(
            lambda x: obj(x),
            h,
            jac=lambda x: obj_grad(x)[0],
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8, 'disp': False},
        )
        if res.x is not None:
            cand = _project(res.x, target)
            c = _evaluate(cand)
            if c < best_c:
                best_h = cand
                best_c = c
    except Exception:
        pass

    # Second pass with higher alpha
    if time.perf_counter() < deadline - 5:
        alpha2 = alpha * 3
        try:
            def obj2(x):
                h_t = np.asarray(x, dtype=float)
                corr = np.correlate(h_t, 1.0 - h_t, mode="full")
                mx = float(np.max(corr))
                y = alpha2 * (corr - mx)
                y = np.clip(y, -700.0, 0.0)
                e = np.exp(y)
                z = float(np.sum(e))
                if z <= 0:
                    return 1e100
                return (2.0 / n) * (mx + np.log(z) / alpha2)

            def obj2_grad(x):
                h_t = np.asarray(x, dtype=float)
                n_t = h_t.size
                corr = np.correlate(h_t, 1.0 - h_t, mode="full")
                mx = float(np.max(corr))
                y = alpha2 * (corr - mx)
                y = np.clip(y, -700.0, 0.0)
                e = np.exp(y)
                z = float(np.sum(e))
                if z <= 0:
                    return np.zeros(n_t)
                w = e / z
                g = np.zeros(n_t, dtype=float)
                center = n_t - 1
                for t in range(corr.size):
                    p = w[t]
                    if p < 1e-14:
                        continue
                    k = t - center
                    if k == 0:
                        g += p * (1.0 - 2.0 * h_t)
                    else:
                        d = _compute_d(h_t, k, n_t)
                        g += p * d
                return (2.0 / n_t) * g

            res2 = minimize(
                obj2, best_h,
                jac=obj2_grad,
                method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': maxiter // 2, 'ftol': 1e-14, 'gtol': 1e-8, 'disp': False},
            )
            if res2.x is not None:
                cand2 = _project(res2.x, target)
                c2 = _evaluate(cand2)
                if c2 < best_c:
                    best_h = cand2
                    best_c = c2
        except Exception:
            pass

    return best_h, best_c

def _random_perturb(h, rng, n_perturb):
    """Perturb n_perturb random coordinates to escape local optima."""
    n = h.size
    target = 0.5 * n
    h_new = h.copy()
    idx = rng.choice(n, size=n_perturb * 2, replace=False)
    for i in range(0, n_perturb * 2, 2):
        a, b = int(idx[i]), int(idx[i + 1])
        amt = float(rng.uniform(0.05, 0.3))
        h_new[a] += amt
        h_new[b] -= amt
    h_new = _project(h_new, target)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 20.0)

    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    # Load parent
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        np_ = parent.size
        target_ = 0.5 * np_
        if abs(float(np.sum(parent)) - target_) < 1e-5:
            cp = _evaluate(parent)
            best_h = parent.copy()
            best_c = cp
            best_n = np_
        pproj = _project(parent, target_)
        cp = _evaluate(pproj)
        if cp < best_c:
            best_h = pproj
            best_c = cp
            best_n = np_

    if best_h is None:
        n0 = 90
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_init = 0.5 + 0.3 * np.sin(2.0 * np.pi * 3.0 * x / 2.0 + 0.5) + 0.15 * np.sin(2.0 * np.pi * 5.0 * x / 2.0 + 1.2)
        best_h = _project(h_init, 0.5 * n0)
        best_c = _evaluate(best_h)
        best_n = n0

    resolutions = [int(best_n), int(best_n) * 2, int(best_n) * 3]
    resolutions = sorted(set(r for r in resolutions if 2 <= r <= 500))

    for res_idx, n_res in enumerate(resolutions):
        if deadline - time.perf_counter() < 60.0:
            break

        # Get solution at this resolution
        if int(best_h.size) == n_res:
            h_cur = best_h.copy()
        else:
            h_cur = _upsample(best_h, n_res)

        c_cur = _evaluate(h_cur)
        if c_cur < best_c:
            best_h = h_cur.copy()
            best_c = c_cur
            best_n = n_res

        # L-BFGS-B optimization
        lbfgs_deadline = min(deadline - 120.0, time.perf_counter() + min(80.0, 0.35 * (deadline - time.perf_counter())))
        if n_res <= 200:
            h_lb, c_lb = _lbgfb_optimize(h_cur, lbfgs_deadline, alpha=25.0, maxiter=150)
        else:
            h_lb, c_lb = _lbgfb_optimize(h_cur, lbfgs_deadline, alpha=20.0, maxiter=80)
        if c_lb < best_c:
            best_h = h_lb.copy()
            best_c = c_lb
            best_n = n_res

        # Pair search (main workhorse)
        ps_deadline = min(deadline - 40.0, time.perf_counter() + min(500.0, 0.5 * (deadline - time.perf_counter())))
        m_lags = 8 if n_res <= 180 else 10
        h_ps, c_ps = _pair_search(h_lb, ps_deadline, m=m_lags, max_rounds=15000)
        if c_ps < best_c:
            best_h = h_ps.copy()
            best_c = c_ps
            best_n = n_res

        # Try a few random restarts
        if deadline - time.perf_counter() > 80.0:
            for attempt in range(3):
                if deadline - time.perf_counter() < 40.0:
                    break
                h_pert = _random_perturb(h_ps, rng, n_perturb=max(3, n_res // 20))
                h_pr, c_pr = _pair_search(h_pert, min(deadline - 30.0, time.perf_counter() + 60.0), m=m_lags, max_rounds=3000)
                if c_pr < best_c:
                    best_h = h_pr.copy()
                    best_c = c_pr
                    best_n = n_res

    # Final polish on the best solution
    if best_h is not None and deadline - time.perf_counter() > 30.0:
        n_best = int(best_h.size)
        h_fin, c_fin = _pair_search(best_h, deadline - 5.0, m=10, max_rounds=20000)
        if c_fin < best_c:
            best_h = h_fin
            best_c = c_fin
            best_n = n_best

    if best_h is None:
        fallback = _project(np.full(90, 0.5, dtype=float), 45.0)
        return fallback, _evaluate(fallback), 90

    final_h = _project(best_h, 0.5 * int(best_h.size))
    final_c = _evaluate(final_h)
    return final_h, float(final_c), int(final_h.size)
```
