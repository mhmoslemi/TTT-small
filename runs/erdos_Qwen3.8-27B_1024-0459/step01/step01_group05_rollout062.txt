    n = len(h)
    C = np.correlate(h, 1.0-h, mode='full')  # length 2n-1
    m = np.max(C)
    shifted = beta * (C - m)
    shifted = np.clip(shifted, -50, 50)
    exp_s = np.exp(shifted)
    W = exp_s / exp_s.sum()  # softmax weights, length 2n-1
    value = m + np.exp(np.log(exp_s.sum())) / beta  # = m + log(sum(exp_s))/beta
    
    grad = np.zeros(n)
    # W[k] corresponds to lag k-(n-1)
    for j in range(n):
        g = 0.0
        for lag in range(-(n-1), n):
            k = lag + (n-1)
            # h[j] appears as h[i+lag] when i = j-lag: contribution (1-h[j-lag]) if 0<=j-lag<n
            # h[j] appears as h[i] when i = j: contribution -h[j+lag] if 0<=j+lag<n

            if 0 <= i1 < n:
                g += W[k] * (1.0 - h[i1])
            i2 = j + lag
            if 0 <= i2 < n:
                g -= W[k] * h[i2]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_and_grad(h, beta):
    """Compute smooth max of correlation values and its gradient."""
    n = len(h)
    C = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    m = np.max(C)
    shifted = beta * (C - m)
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_s = np.exp(shifted)
    sum_exp = exp_s.sum()
    W = exp_s / sum_exp  # softmax weights
    
    value = m + np.log(sum_exp) / beta
    
    # Gradient: grad[j] = sum_{lag} W[k] * [(1-h[j-lag])*1[0<=j-lag<n] - h[j+lag]*1[0<=j+lag<n]]
    grad = np.zeros(n)
    for j in range(n):
        for lag in range(-(n-1), n):
            k = lag + (n - 1)
            i1 = j - lag
            if 0 <= i1 < n:
                grad[j] += W[k] * (1.0 - h[i1])
            i2 = j + lag
            if 0 <= i2 < n:
                grad[j] -= W[k] * h[i2]
    
    return value, grad

def project_to_constraint(h, target):
    """Project h onto {sum(h)=target, 0<=h<=1}."""
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        # Find free variables
        if diff > 0:
            free = (h < 1.0 - 1e-15)
            if not free.any():
                break
            step = min(diff / free.sum(), 1e-6)
            h[free] += step
        else:
            free = (h > 1e-15)
            if not free.any():
                break
            step = min(-diff / free.sum(), 1e-6)
            h[free] += step
        h = np.clip(h, 0.0, 1.0)
    # Final exact adjustment
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            free = (h < 1.0)
            if free.any():
                per = diff / free.sum()
                h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = (h > 0.0)
            if free.any():
                per = diff / free.sum()
                h[free] = np.maximum(h[free] + per, 0.0)
    return h

def make_objective(h, beta):
    hc = np.clip(h, 0.0, 1.0)
    val, _ = smooth_max_and_grad(hc, beta)
    return val

def make_gradient(h, beta):
    hc = np.clip(h, 0.0, 1.0)
    _, grad = smooth_max_and_grad(hc, beta)
    # Zero out gradient where bounds are active
    grad[hc <= 0.0] = 0.0
    grad[hc >= 1.0] = 0.0
    return grad

def optimize_slsqp(h0, target, beta, maxiter, ftol):
    n = len(h0)
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return make_objective(x, beta)
    
    def grad(x):
        return make_gradient(x, beta)
    
    result = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})
    
    if result.success and result.x is not None:
        return project_to_constraint(result.x, target)
    elif result.x is not None:
        return project_to_constraint(result.x, target)
    return project_to_constraint(h0, target)

    value = m + np.log(sum_exp) / beta
    return value, W

def smooth_max_gradient(h, beta):
    n = len(h)
    C = np.correlate(h, 1.0 - h, mode='full')
    value, W = smooth_max_and_weights(C, beta)
    one_minus_h = 1.0 - h
    term1 = np.convolve(W, one_minus_h, mode='full')
    term2 = np.correlate(h, W, mode='full')
    grad = term1[n - 1:2 * n - 1] - term2[n - 1:2 * n - 1]
    return value, grad

def obj_fn(x, beta):
    xc = np.clip(x, 0.0, 1.0)
    val, _ = smooth_max_gradient(xc, beta)
    return val

def grad_fn(x, beta):
    xc = np.clip(x, 0.0, 1.0)
    _, g = smooth_max_gradient(xc, beta)
    mask_lo = xc <= 1e-15
    mask_hi = xc >= 1.0 - 1e-15
    if np.any(mask_lo):
        g[mask_lo] = np.maximum(g[mask_lo], 0.0)
    if np.any(mask_hi):
        g[mask_hi] = np.minimum(g[mask_hi], 0.0)
    return g

def sum_eq(x, target):
    return x.sum() - target

def sum_eq_jac(x, target):
    return np.ones(len(x))

def aug_obj(x, beta, lam, mu, target):
    xc = np.clip(x, 0.0, 1.0)
    f, _ = smooth_max_gradient(xc, beta)
    s = x.sum() - target
    return f + lam * s + 0.5 * mu * s * s

def aug_grad(x, beta, lam, mu, target):
    xc = np.clip(x, 0.0, 1.0)
    _, gf = smooth_max_gradient(xc, beta)
    s = x.sum() - target
    n = len(x)
    g = gf + (lam + mu * s) * np.ones(n)
    mask_lo = xc <= 1e-15
    mask_hi = xc >= 1.0 - 1e-15
    if np.any(mask_lo):
        g[mask_lo] = np.maximum(g[mask_lo], 0.0)
    if np.any(mask_hi):
        g[mask_hi] = np.minimum(g[mask_hi], 0.0)
    return g

def project_sum(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            free = (h < 1.0 - 1e-14)
            if not free.any():
                break
            h[free] += diff / free.sum()
        else:
            free = (h > 1e-14)
            if not free.any():
                break
            h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            free = np.where(h < 1.0)[0]
            if len(free) > 0:
                per = diff / len(free)
                for i in free:
                    h[i] = min(h[i] + per, 1.0)
        else:
            free = np.where(h > 0.0)[0]
            if len(free) > 0:
                per = diff / len(free)
                for i in free:
                    h[i] = max(h[i] + per, 0.0)
    return h

def run_slsqp(h0, beta, maxiter, ftol):
    n = len(h0)
    target = n / 2.0
    h0 = project_sum(h0, target)
    cons = [{'type': 'eq', 'fun': partial(sum_eq, target=target),
             'jac': partial(sum_eq_jac, target=target)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(partial(obj_fn, beta=beta), h0,
                      jac=partial(grad_fn, beta=beta),
                      method='SLSQP', bounds=bounds,
                      constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return project_sum(res.x, target)
    except Exception:
        pass
    return h0

def run_lbfgsb_auglag(h0, beta, n_outer, maxiter_inner):
    n = len(h0)
    target = n / 2.0
    h = project_sum(h0, target).copy()
    lam = 0.0
    mu = 1.0
    bounds = [(0.0, 1.0)] * n
    for _ in range(n_outer):
        try:
            res = minimize(partial(aug_obj, beta=beta, lam=lam, mu=mu, target=target),
                          h, jac=partial(aug_grad, beta=beta, lam=lam, mu=mu, target=target),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter_inner, 'ftol': 1e-14, 'gtol': 1e-10})
            if res.x is not None:
                h = res.x
        except Exception:
            pass
        s = h.sum() - target
        lam += mu * s
        mu = min(mu * 2.0, 1000.0)
        if abs(s) < 1e-12:
            break
    return project_sum(h, target)

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def try_update(best_state, hc):
    n = len(hc)
    tgt = n / 2.0
    hc = project_sum(hc, tgt)
    c5 = compute_c5(hc)
    if c5 < best_state[1] - 1e-15:
        best_state[0] = hc.copy()
        best_state[1] = c5
        best_state[2] = n
    return c5

def elapsed_time(t0):
    return time.time() - t0

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)
    c5_parent = compute_c5(h_parent)
    best_state = [h_parent.copy(), c5_parent, n_parent]

    # Phase 1: Optimize at parent resolution with progressive beta
    h_cur = best_state[0].copy()
    betas_phase1 = [5.0, 15.0, 40.0, 100.0, 300.0, 800.0]
    for i_beta in range(len(betas_phase1)):
        if elapsed_time(t0) > budget_s * 0.28:
            break
        b = betas_phase1[i_beta]
        mx = 400 if i_beta < 3 else 300
        h_cur = run_slsqp(h_cur, b, maxiter=mx, ftol=1e-14)
        try_update(best_state, h_cur)

    # Also try L-BFGS-B augmented Lagrangian at parent resolution
    if elapsed_time(t0) < budget_s * 0.35:
        h_alt = run_lbfgsb_auglag(best_state[0], 100.0, n_outer=8, maxiter_inner=300)
        try_update(best_state, h_alt)

    # Phase 2: Try other resolutions
    resolutions = [n_parent, 100, 120, 150, 200]
    for n_try in resolutions:
        if elapsed_time(t0) > budget_s * 0.68:
            break
        if n_try == best_state[2]:
            h_interp = best_state[0].copy()
        else:
            h_interp = interpolate_h(best_state[0], best_state[2], n_try)
            h_interp = project_sum(h_interp, n_try / 2.0)

        for b in [20.0, 100.0, 400.0]:
            if elapsed_time(t0) > budget_s * 0.68:
                break
            h_interp = run_slsqp(h_interp, b, maxiter=350, ftol=1e-14)
            try_update(best_state, h_interp)

        if elapsed_time(t0) < budget_s * 0.72 and n_try != best_state[2]:
            h_interp = run_lbfgsb_auglag(h_interp, 200.0, n_outer=6, maxiter_inner=200)
            try_update(best_state, h_interp)

    # Phase 3: Random perturbations + reoptimize at best resolution
    n_best = best_state[2]
    tgt_best = n_best / 2.0
    noise_scales = [0.03, 0.02, 0.015, 0.01, 0.02, 0.01]
    for i in range(len(noise_scales)):
        if elapsed_time(t0) > budget_s * 0.88:
            break
        noise = np.random.randn(n_best) * noise_scales[i]
        h_pert = project_sum(best_state[0] + noise, tgt_best)
        h_pert = run_slsqp(h_pert, 100.0, maxiter=300, ftol=1e-14)
        try_update(best_state, h_pert)

    # Phase 4: Subgradient fine-tuning on the best solution
    if elapsed_time(t0) < budget_s * 0.94:
        n_sub = best_state[2]
        tgt_sub = n_sub / 2.0
        h_sub = best_state[0].copy()
        lr = 0.01
        for it in range(500):
            if elapsed_time(t0) > budget_s * 0.94:
                break
            C = np.correlate(h_sub, 1.0 - h_sub, mode='full')
            val, W = smooth_max_and_weights(C, 200.0)
            one_minus = 1.0 - h_sub
            t1 = np.convolve(W, one_minus, mode='full')
            t2 = np.correlate(h_sub, W, mode='full')
            g = t1[n_sub - 1:2 * n_sub - 1] - t2[n_sub - 1:2 * n_sub - 1]
            h_sub = h_sub - lr * g
            h_sub = project_sum(h_sub, tgt_sub)
            lr = max(lr * 0.995, 0.001)
        try_update(best_state, h_sub)

    # Phase 5: Final sharp SLSQP refinement
    if elapsed_time(t0) < budget_s * 0.98:
        h_fin = run_slsqp(best_state[0], 500.0, maxiter=200, ftol=1e-15)
        try_update(best_state, h_fin)
        h_fin = run_slsqp(best_state[0], 1000.0, maxiter=150, ftol=1e-15)
        try_update(best_state, h_fin)

    # Final projection to ensure exact constraint
    best_state[0] = project_sum(best_state[0], best_state[2] / 2.0)
    final_c5 = compute_c5(best_state[0])

    return (best_state[0], final_c5, best_state[2])
