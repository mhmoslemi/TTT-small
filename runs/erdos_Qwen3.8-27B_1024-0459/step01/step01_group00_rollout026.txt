        h = project_sum_boundary(h, n / 2.0)
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
    
    return h, best_c5

def subgradient_descent(h, n_steps=3000, beta_start=5.0, beta_end=200.0,
                        time_limit=60.0, start_time=None):
    """Subgradient descent with smooth max and momentum."""
    n = len(h)
    h = h.copy()
    if start_time is None:
        start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    mu = 0.9
    
    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        # Log-linear beta schedule
        frac = step / max(n_steps - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Compute smooth gradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = np.max(corr)
        exp_c = np.exp(beta * (corr - max_c))
        w = exp_c / np.sum(exp_c)
        
        # Gradient: grad[i] = (1/beta) * sum_m w[m+n-1] * [(1-h[i+m]) - h[i-m]]
        grad = np.zeros(n)
        for m in range(-(n - 1), n):
            wm = w[m + n - 1]
            if wm < 1e-16:
                continue
            # (1-h[i+m]) for valid i
            lo1 = max(0, -m)
            hi1 = min(n, n + m)
            if lo1 < hi1:
                grad[lo1:hi1] += wm * (1.0 - h[lo1 + m:hi1 + m])
            # -h[i-m] for valid i
            lo2 = max(0, m)
            hi2 = min(n, m + n)
            if lo2 < hi2:
                grad[lo2:hi2] -= wm * h[lo2 - m:hi2 - m]
        grad /= beta
        
        # Adaptive step size
        step_size = 1.0 / (beta * (1.0 + 0.1 * beta))
        
        # Update with momentum
        momentum = mu * momentum - step_size * grad
        h_new = h + momentum
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_sum_boundary(h_new, n / 2.0)
        
        h = h_new
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    
    return best_h, best_c5

def slsqp_polish(h, n, max_iter=5000, ftol=1e-15):
    """SLSQP local optimization."""
    def obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': ftol})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.clip(np.interp(x_new, x_old, h_src), 0.0, 1.0)

def local_perturbation(h, n, n_trials=3000, step_size=0.03, time_limit=30.0, start_time=None):
    """Random perturbation local search."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    if start_time is None:
        start_time = time.time()
    rng = np.random.RandomState(12345)
    
    for _ in range(n_trials):
        if time.time() - start_time > time_limit:
            break
        h_try = best_h.copy()
        n_mod = rng.randint(1, 5)
        idxs = rng.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += rng.uniform(-step_size, step_size)
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_sum_boundary(h_try, n / 2.0)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try
    
    return best_h, best_c5

def generate_inits(n, rng, parent_h=None):
    """Generate diverse initial points."""
    inits = []
    
    # From parent
    if parent_h is not None:
        h_p = interpolate_to_n(parent_h, n)
        inits.append(make_feasible(h_p, n))
    
    # Structured: two-block
    h1 = np.zeros(n)
    h1[:n//4] = 1.0
    h1[n//2:n//2+n//4] = 1.0
    inits.append(make_feasible(h1, n))
    
    # Structured: four-block
    if n % 4 == 0:
        h2 = np.tile([1.0, 0.0, 1.0, 0.0], n // 4)
        inits.append(make_feasible(h2, n))
        h3 = np.tile([1.0, 1.0, 0.0, 0.0], n // 4)
        inits.append(make_feasible(h3, n))
    
    # Thue-Morse
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    inits.append(make_feasible(tm, n))
    
    # Random
    for _ in range(2):
        h_r = rng.rand(n) * 1.2
        inits.append(make_feasible(h_r, n))
    
    # Symmetric noise
    half = n // 2
    h_sym = rng.rand(half) * 1.0
    h_full = np.concatenate([h_sym, h_sym[::-1]])
    if n % 2 == 1:
        h_full = np.concatenate([h_full, [0.5]])
    inits.append(make_feasible(h_full[:n], n))
    
    return inits

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    # Load parent
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if len(h_parent) == 0:
            h_parent = None
    except Exception:
        h_parent = None
    
    if h_parent is not None:
        h_parent = make_feasible(h_parent, len(h_parent))
        best_h = h_parent.copy()
        best_n = len(best_h)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5
    
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Quick SLSQP polish on parent at its resolution
    if remaining() > 60:
        h_pol, c5_pol = slsqp_polish(best_h, best_n, max_iter=2000)
        if h_pol is not None and c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    
    # Phase 2: Coordinate descent at parent resolution
    if remaining() > 60:
        h_cd, c5_cd = coord_descent(best_h, n_sweeps=200,
                                     time_limit=min(50, remaining() - 20),
                                     start_time=time.time())
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Phase 3: Try higher resolutions
    target_resolutions = [200, 240, 280, 320, 400]
    # Add best_n if not in list
    if best_n not in target_resolutions:
        target_resolutions.insert(0, best_n)
    target_resolutions.sort()
    
    for n_res in target_resolutions:
        if remaining() < 30:
            break
        
        # Interpolate current best to this resolution
        h_interp = make_feasible(interpolate_to_n(best_h, n_res), n_res)
        
        # Coordinate descent at this resolution
        t_lim_cd = min(40, remaining() - 20)
        if t_lim_cd > 10:
            h_cd, c5_cd = coord_descent(h_interp, n_sweeps=300,
                                         time_limit=t_lim_cd,
                                         start_time=time.time())
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n_res
        
        # SLSQP polish
        if remaining() > 30:
            h_sl, c5_sl = slsqp_polish(h_cd if c5_cd < best_c5 else h_interp, n_res,
                                         max_iter=3000)
            if h_sl is not None and c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()
                best_n = n_res
    
    # Phase 4: Subgradient descent with annealing on best solution
    if remaining() > 40:
        h_sub, c5_sub = subgradient_descent(best_h, n_steps=2000,
                                             beta_start=3.0, beta_end=150.0,
                                             time_limit=min(50, remaining() - 20),
                                             start_time=time.time())
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()
    
    # Phase 5: Coordinate descent polish on final best
    if remaining() > 30:
        h_cd2, c5_cd2 = coord_descent(best_h, n_sweeps=500,
                                       time_limit=min(50, remaining() - 15),
                                       start_time=time.time())
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
    
    # Phase 6: SLSQP final polish
    if remaining() > 25:
        h_fin, c5_fin = slsqp_polish(best_h, best_n, max_iter=8000, ftol=1e-16)
        if h_fin is not None and c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
    
    # Phase 7: Local perturbation
    if remaining() > 15:
        h_pert, c5_pert = local_perturbation(best_h, best_n,
                                              n_trials=2000, step_size=0.02,
                                              time_limit=min(20, remaining() - 5),
                                              start_time=time.time())
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
        # Final tiny coordinate descent
        if remaining() > 10:
            h_cd3, c5_cd3 = coord_descent(best_h, n_sweeps=100,
                                           time_limit=min(8, remaining() - 3),
                                           start_time=time.time())
            if c5_cd3 < best_c5:
                best_c5 = c5_cd3
                best_h = h_cd3.copy()
    
    # Phase 8: Final feasibility and verification
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
