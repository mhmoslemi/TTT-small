    h = h.copy()

    # Fix sum
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

        h[idx] = 0
        ones = ones.tolist(); ones.remove(idx); ones = np.array(ones)
        zeros = np.append(zeros, idx)
    while len(ones) < target:

        # Pick a random 1 and a random 0, try swap

  - Term h[a]*h[a+m]: was 1*h[a+m], now 0 → change = -h[a+m]
  - Term h[a-m]*h[a]: was h[a-m]*1, now 0 → change = -h[a-m]
  - Term h[b]*h[b+m]: was 0, now 1*h[b+m] → change = h[b+m]
  - Term h[b-m]*h[b]: was h[b-m]*0, now h[b-m]*1 → change = h[b-m]

So Δ(sum_i h[i]*h[i+m]) = -h[a+m] - h[a-m] + h[b+m] + h[b-m] (for valid indices)

And Δc_k = -Δ(sum_i h[i]*h[i+m]) = h[a+m] + h[a-m] - h[b+m] - h[b-m]

    shifted = np.clip(beta * (x - m), -100, 100)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_sum(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            free = h < 1.0
        else:
            free = h > 0.0
        if not np.any(free):
            break

def sa_binary_optimize(h_init, n, max_iter=300000, t0=0.02):
    """Simulated annealing on binary patterns, minimizing max correlation."""
    h = np.round(h_init).astype(float)
    # Enforce exact sum
    target = n // 2

        ones = np.where(h == 1)[0]
        idx = np.random.choice(ones)
        h[idx] = 0

        zeros = np.where(h == 0)[0]
        idx = np.random.choice(zeros)
        h[idx] = 1

    current_max = np.max(corr)
    best_h = h.copy()
    best_max = current_max
    
    T = t0
    cool_rate = (1e-6 / T) ** (1.0 / max_iter)
    
    for it in range(max_iter):
        T = max(t0 * cool_rate ** it, 1e-8)
        a = np.random.randint(n)
        b = np.random.randint(n)
        if h[a] == 0 or h[b] == 1:
            continue
        h[a] = 0
        h[b] = 1
        new_corr = np.correlate(h, 1.0 - h, mode='full')

def coord_descent(h, n, target, max_rounds=50, step_sizes=None):
    """Coordinate descent: adjust one coordinate at a time to reduce max corr."""
    h = project_sum(h.copy(), target)
    current_c5 = compute_c5(h)
    
    if step_sizes is None:
        step_sizes = [0.3, 0.1, 0.05, 0.02, 0.01, 0.005]
    
    for s in step_sizes:
        for _ in range(max_rounds):
            improved = False
            perm = np.random.permutation(n)
            for j in perm:
                for direction in [1, -1]:
                    h_new = h.copy()
                    h_new[j] = np.clip(h[j] + direction * s, 0, 1)
                    h_new = project_sum(h_new, target)
                    c5_new = compute_c5(h_new)
                    if c5_new < current_c5 - 1e-14:
                        h = h_new
                        current_c5 = c5_new
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
    return h, current_c5

def slsqp_optimize(h_start, n, target, beta, max_iter, ftol):
    """SLSQP with smooth max objective."""
    h0 = project_sum(h_start.copy(), target)
    
    def obj(hv):
        hv = np.clip(hv, 0.0, 1.0)
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        return smooth_max(corr, beta)
    
    def obj_grad(hv):
        hv = np.clip(hv, 0.0, 1.0)
        f0 = obj(hv)
        g = np.zeros(len(hv))
        eps = 1e-7
        for i in range(len(hv)):
            hv2 = hv.copy()
            hv2[i] = min(1.0, max(0.0, hv2[i] + eps))
            g[i] = (obj(hv2) - f0) / eps
        return g
    
    def sum_fun(hv):
        return np.sum(hv) - target
    
    def sum_jac(hv):
        return np.ones(len(hv))
    
    constraints = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        result = minimize(obj, h0, jac=obj_grad, method='SLSQP', bounds=bounds,
                         constraints=constraints,
                         options={'maxiter': max_iter, 'ftol': ftol})
        if result.success or result.x is not None:
            return project_sum(result.x, target)
    except Exception:
        pass
    return h0
