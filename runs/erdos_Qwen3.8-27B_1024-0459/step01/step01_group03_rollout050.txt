def project_density(x, target):
    """Project x onto {0<=y<=1, sum(y)=target}"""
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)

    # Water-filling to achieve exact sum
    s = float(np.sum(x))
    if abs(s - target) < 1e-14:
        return x
    
    # Binary search for the shift
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:

        else:
            hi = mid

        grad = sens.mean(axis=0)

        neg_candidates = sorted_idx[:min(15, n//2)]  # low gradient -> increase h

            for j in pos_candidates:

                    new_h[j] = np.clip(h[j] - d, 0.0, 1.0)
                    # Re-project to maintain sum

                        best_corr_max = c
