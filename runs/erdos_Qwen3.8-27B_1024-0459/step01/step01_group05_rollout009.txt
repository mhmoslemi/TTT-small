        # Second term: -h[j+k] for valid j+k
        j2_start = max(0, -k)
        j2_end = min(n, n - k)
        if j2_start < j2_end:
            j2_idx = np.arange(j2_start, j2_end)
            h2_idx = j2_idx + k
            grad[j2_idx] -= pk * h[h2_idx]
    grad *= dx
    return grad

def project_sum(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        free = (h > 0.0) & (h < 1.0)
        if np.sum(free) == 0:
            break
        h[free] += diff / np.sum(free)
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-14:
        if diff > 0:
            free = (h < 1.0)
            if np.any(free):
                per = diff / np.sum(free)
                h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = (h > 0.0)
            if np.any(free):
                per = diff / np.sum(free)
                h[free] = np.maximum(h[free] + per, 0.0)
    return h

def run_slsqp(h0, n, target, beta, max_iter, ftol):
    dx = 2.0 / n
    
    def obj(h):
        o, _, _ = compute_corr_and_weights(h, beta)
        return o
    
    def jac(h):
        _, w, _ = compute_corr_and_weights(h, beta)
        return compute_gradient(h, w, dx)
    
    def eq_fun(h):
        return np.sum(h) - target
    
    def eq_jac(h):
        return np.ones(n)
    
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        result = minimize(obj, h0, method='SLSQP', bounds=bounds,
                         constraints=constraints, jac=jac,
                         options={'maxiter': max_iter, 'ftol': ftol})
        if result.success or result.x is not None:
            return project_sum(result.x, target)
    except Exception:
        pass
    return h0

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx


def compute_corr_and_weights(h, beta):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full') * dx
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    obj = m + np.log(np.sum(exp_shifted)) / beta
    return obj, weights, corr


def compute_gradient(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for k in range(-(n - 1), n):
        pk = weights[k + n - 1]
        if pk < 1e-15:
            continue
        if k >= 0:
            j_idx = np.arange(k, n)
            h_idx = np.arange(0, n - k)
            grad[j_idx] += pk * (1.0 - h[h_idx])
        else:
            j_idx = np.arange(0, n + k)
            h_idx = np.arange(-k, n)
            grad[j_idx] += pk * (1.0 - h[h_idx])
        j2_start = max(0, -k)
        j2_end = min(n, n - k)
        if j2_start < j2_end:
            j2_idx = np.arange(j2_start, j2_end)
            h2_idx = j2_idx + k
            grad[j2_idx] -= pk * h[h2_idx]
    grad *= dx
    return grad

    for _ in range(500):
        s = np.sum(h)
        diff = target - s

        free = (h > 0.0) & (h < 1.0)
        nfree = np.sum(free)
        if nfree == 0:

            free = (h > 0.0)
            if np.any(free):
                per = diff / np.sum(free)
                h[free] = np.maximum(h[free] + per, 0.0)
    return h

    o, _, _ = compute_corr_and_weights(h, beta)
    return o

    n = len(h)
    dx = 2.0 / n
    return compute_gradient(h, w, dx)

    return np.sum(h) - target


def eq_jac(h, target):
    return np.ones(len(h))


def run_slsqp(h0, n, target, beta, max_iter, ftol):

        return slsqp_objective(h, beta)

    def jac(h):
        return slsqp_gradient(h, beta)

    def cfun(h):
        return eq_fun(h, target)

    def cjac(h):
        return eq_jac(h, target)

    constraints = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bounds = [(0.0, 1.0)] * n
    h_start = project_sum(h0, target)

    except Exception:
