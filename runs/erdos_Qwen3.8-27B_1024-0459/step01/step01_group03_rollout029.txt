corr[0] = ∑ h[i]*(1-h[i]) = ∑ h[i] - ∑ h[i]²
∂/∂h[j] = 1 - 2*h[j]

For ℓ < 0, let k = -ℓ > 0:
corr[-k] = ∑_{i=k}^{n-1} h[i] * (1-h[i-k])
= ∑_{i=k}^{n-1} h[i] - ∑_{i=k}^{n-1} h[i]*h[i-k]

∂/∂h[j]:
- From first sum: 1 if k ≤ j ≤ n-1
- From second sum: h[i-k] if j = i (i.e., k ≤ j ≤ n-1, contributes h[j-k])
                     + h[i] if j = i-k (i.e., j+k = i, so 0 ≤ j ≤ n-1-k, contributes h[j+k])
  So: -h[j-k]*I(k≤j≤n-1) - h[j+k]*I(0≤j≤n-1-k)

Total: I(k≤j≤n-1) - h[j-k]*I(k≤j≤n-1) - h[j+k]*I(0≤j≤n-1-k)

Great, so I can compute the gradient for any single lag efficiently.

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)

def project_density(x, target):
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=-1.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) <= 1e-12 * max(1, n):
        return x
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s2 = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s2 < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-13:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            if res > 0:
                idx = int(free[np.argmax(1.0 - h[free])])
            else:
                idx = int(free[np.argmax(h[free])])
            h[idx] = float(np.clip(h[idx] + res, 0.0, 1.0))
    res = target - float(np.sum(h))
    if abs(res) > 1e-13:

        if free.size > 0:
            h[free] += res / float(free.size)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):

def lag_gradient_single(h, center, t):
    """Gradient of corr[t] with respect to h. t is index in corr array (0 to 2n-2)."""
    n = h.size
    lag = t - center  # actual lag, -(n-1) to n-1
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n - lag] += 1.0
        g[:n - lag] -= h[lag:]
        g[lag:] -= h[:n - lag]
    else:
        la = -lag
        g[la:] += 1.0
        g[la:] -= h[:n - la]
        g[:n - la] -= h[la:]
    return g

def pair_optimal_transfer(h, i, j, corr, top_indices):
    """Find optimal transfer δ from j to i (h[i]+=δ, h[j]-=δ) to minimize max of top lags."""
    n = h.size
    center = n - 1
    
    # Compute d_lag for each top lag
    # d_lag = grad_i_lag - grad_j_lag
    # For lag k: grad_i - grad_j
    
    # Bounds on delta
    L = max(-h[i], h[j] - 1.0)
    U = min(1.0 - h[i], h[j])
    if U - L < 1e-14:
        return 0.0, False
    
    # For each top lag t, compute the slope d_t and intercept
    slopes = np.zeros(len(top_indices))
    for idx, t in enumerate(top_indices):
        lag = t - center
        if lag == 0:
            # grad = 1 - 2h[j], so d = -2h[i] + 2h[j]... 
            # Actually grad_lag0[j] = 1 - 2*h[j]
            # d = grad[i] - grad[j] = (1-2h[i]) - (1-2h[j]) = 2*(h[j]-h[i])
            slopes[idx] = 2.0 * (h[j] - h[i])
        elif lag > 0:
            gi = 0.0
            if 0 <= i <= n - 1 - lag:
                gi = 1.0 - h[i + lag]
            if lag <= i <= n - 1:
                gi -= h[i - lag]
            gj = 0.0
            if 0 <= j <= n - 1 - lag:
                gj = 1.0 - h[j + lag]
            if lag <= j <= n - 1:
                gj -= h[j - lag]
            slopes[idx] = gi - gj
        else:
            la = -lag
            gi = 0.0
            if la <= i <= n - 1:
                gi = 1.0 - h[i - la]
            if 0 <= i <= n - 1 - la:
                gi -= h[i + la]
            gj = 0.0
            if la <= j <= n - 1:
                gj = 1.0 - h[j - la]
            if 0 <= j <= n - 1 - la:
                gj -= h[j + la]
            slopes[idx] = gi - gj
    
    # Current values of top lags
    vals = corr[top_indices]
    
    # We want to minimize max_k (vals[k] + slopes[k] * delta)
    # over delta in [L, U]
    
    # Candidate points: L, U, and all breakpoints where vals[a] + slopes[a]*d = vals[b] + slopes[b]*d

    best_delta = 0.0

        mx = float(np.max(new_vals))

            best_max = mx
            best_delta = d
    
    # Check if there's improvement
    if best_delta != 0.0 and best_max < float(np.max(vals)) - 1e-13:
        return best_delta, True
    return 0.0, False

def subgradient_optimize(h, deadline, max_iter=200, top_m=8):
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    top_m = min(top_m, 2*n-1)
    
    step_size = 0.5  # initial
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        
        corr = np.correlate(best, 1.0 - best, mode="full")
        center = n - 1
        top_idx = np.argsort(corr)[-top_m:]
        
        # Subgradient: average gradient of top_m lags
        g = np.zeros(n)
        for t in top_idx:
            lag = t - center
            if lag == 0:
                g += 1.0 - 2.0 * best
            elif lag > 0:
                g[:n-lag] += 1.0
                g[:n-lag] -= best[lag:]
                g[lag:] -= best[:n-lag]
            else:
                la = -lag
                g[la:] += 1.0
                g[la:] -= best[:n-la]
                g[:n-la] -= best[la:]
        g /= float(top_m)
        
        # Remove mean (to preserve sum constraint in the direction)
        g -= float(np.mean(g))
        
        # Line search
        improved = False
        for ss in [step_size, step_size*0.5, step_size*0.25, step_size*0.1]:
            cand = project_density(best - ss * g, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-12:
                best = cand
                best_c = c
                improved = True
                break
        
        if improved:
            step_size = min(2.0, step_size * 1.2)
        else:
            step_size *= 0.7
            if step_size < 0.01:
                break
    
    return best, best_c
