```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c_values(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return c * dx

def compute_c5(h, n):
    return float(np.max(compute_c_values(h, n)))

def grad_single(h, n, m):
    dx = 2.0 / n
    L = m - (n - 1)
    grad = np.zeros(n)
    if L >= 0:
        grad[:n-L] += (1.0 - h[L:]) * dx
        grad[L:] -= h[:n-L] * dx
    else:
        M = -L
        grad[M:] += (1.0 - h[:n-M]) * dx
        grad[:n-M] -= h[M:] * dx
    return grad

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-12
        else:
            mask = h < 1.0 - 1e-12
        if mask.sum() == 0:
            break
        h[mask] -= diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h
```

```python
from scipy.optimize import minimize

def compute_c_array(h, n):

    c = np.correlate(h, 1.0 - h, mode="full")

    c = compute_c_array(h, n)
    return float(np.max(c))

def grad_c_index(h, n, m):
    """Gradient of c[m] (the m-th full correlate times dx)"""
    dx = 2.0 / n
    L = m - (n - 1)
    grad = np.zeros(n)
    if L >= 0:
        grad[:n-L] += (1.0 - h[L:]) * dx
        grad[L:] -= h[:n-L] * dx
    else:
        M = -L
        grad[M:] += (1.0 - h[:n-M]) * dx
        grad[:n-M] -= h[M:] * dx
    return grad

def c5_obj(h, n, top_m_indices):
    """Objective: sum of top correlation values (for active set)"""
    c = compute_c_array(h, n)
    return float(np.sum(c[top_m_indices]))

def c5_obj_grad(h, n, top_m_indices):
    """Gradient of sum of top correlation values"""
    grad = np.zeros(n)
    for m in top_m_indices:
        grad += grad_c_index(h, n, m)
    return grad

def smooth_max_obj(h, n, beta):
    """Smooth max approximation using log-sum-exp"""
    c = compute_c_array(h, n)
    m = np.max(c)
    return float((m + np.log(np.sum(np.exp(beta * c - beta * m)))) / beta)

def smooth_max_grad(h, n, beta):
    """Gradient of smooth max"""
    c = compute_c_array(h, n)
    m = np.max(c)
    w = np.exp(beta * c - beta * m)
    w /= np.sum(w)
    grad = np.zeros(n)
```

```python
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def time_left():
        return budget_s - elapsed()

    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float)
        n = len(h_init)
    except Exception:
        n = 95
        h_init = np.full(n, 0.5)

    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, n)
    best_n = n

    # Stage 1: Active-set SLSQP at initial resolution (multiple passes with updated active set)
    for iteration in range(8):
        if time_left() < 120:
            break
        c_arr = compute_c_array(best_h, n)
        top_k = 5
        top_idx = np.argsort(c_arr)[-top_k:]

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n,)}]

        try:
            res = minimize(
                active_set_obj, best_h,
                jac=active_set_grad,
                args=(n, top_idx),
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 1500, 'ftol': 1e-15}
            )
            h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n)
            c5_new = compute_c5(h_new, n)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    # Stage 2: Smooth max with L-BFGS-B (penalty for sum)
    for beta in [15, 40, 80, 160, 300]:
        if time_left() < 100:
            break
        lam = 5.0
        bounds_lfsb = [(0.0, 1.0)] * n
        try:
            res = minimize(
                penalty_smooth_obj, best_h,
                jac=penalty_smooth_grad,
                args=(n, beta, lam),
                method='L-BFGS-B', bounds=bounds_lfsb,
                options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-12}
            )
            h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n)
            c5_new = compute_c5(h_new, n)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    # Stage 3: More SLSQP passes with tight active set (top 3)
    for iteration in range(6):
        if time_left() < 100:
            break
        c_arr = compute_c_array(best_h, n)
        top_idx = np.argsort(c_arr)[-3:]

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n,)}]

        try:
            res = minimize(
                active_set_obj, best_h,
                jac=active_set_grad,
                args=(n, top_idx),
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 2000, 'ftol': 1e-16}
            )
            h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n)
            c5_new = compute_c5(h_new, n)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    # Stage 4: Higher resolution (2x)
    n2 = 2 * n
    if time_left() > 300 and n2 <= 600:
        x_old = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False) + 1.0 / n2
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n2)

        n_cur = n2
        h_cur = h_interp.copy()
        c5_cur = compute_c5(h_cur, n_cur)

        # SLSQP at new resolution
        for iteration in range(4):
            if time_left() < 150:
                break
            c_arr = compute_c_array(h_cur, n_cur)
            top_idx = np.argsort(c_arr)[-5:]
            bounds_c = [(0.0, 1.0)] * n_cur
            cons_c = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n_cur,)}]
            try:
                res = minimize(
                    active_set_obj, h_cur,
                    jac=active_set_grad,
                    args=(n_cur, top_idx),
                    method='SLSQP', bounds=bounds_c, constraints=cons_c,
                    options={'maxiter': 800, 'ftol': 1e-14}
                )
                h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n_cur)
                c5_new = compute_c5(h_new, n_cur)
                if c5_new < c5_cur:
                    h_cur = h_new
                    c5_cur = c5_new
            except Exception:
                pass

        # L-BFGS-B at new resolution
        for beta in [40, 100, 200]:
            if time_left() < 100:
                break
            bounds_l = [(0.0, 1.0)] * n_cur
            try:
                res = minimize(
                    penalty_smooth_obj, h_cur,
                    jac=penalty_smooth_grad,
                    args=(n_cur, beta, 5.0),
                    method='L-BFGS-B', bounds=bounds_l,
                    options={'maxiter': 1000, 'ftol': 1e-15, 'gtol': 1e-11}
                )
                h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n_cur)
                c5_new = compute_c5(h_new, n_cur)
                if c5_new < c5_cur:
                    h_cur = h_new
                    c5_cur = c5_new
            except Exception:
                pass

        if c5_cur < best_c5:
            best_h = h_cur copy if False else h_cur
            best_c5 = c5_cur
            best_n = n_cur

    # Stage 5: Even higher resolution (4x original) if time permits
    n3 = 4 * n
    if time_left() > 250 and n3 <= 800:
        n_src = best_n
        x_old = np.linspace(0.0, 2.0, n_src, endpoint=False) + 1.0 / n_src
        x_new = np.linspace(0.0, 2.0, n3, endpoint=False) + 1.0 / n3
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n3)

        n_cur = n3
        h_cur = h_interp.copy()
        c5_cur = compute_c5(h_cur, n_cur)

        for iteration in range(3):
            if time_left() < 120:
                break
            c_arr = compute_c_array(h_cur, n_cur)
            top_idx = np.argsort(c_arr)[-4:]
            bounds_c = [(0.0, 1.0)] * n_cur
            cons_c = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n_cur,)}]
            try:
                res = minimize(
                    active_set_obj, h_cur,
                    jac=active_set_grad,
                    args=(n_cur, top_idx),
                    method='SLSQP', bounds=bounds_c, constraints=cons_c,
                    options={'maxiter': 600, 'ftol': 1e-13}
                )
                h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n_cur)
                c5_new = compute_c5(h_new, n_cur)
                if c5_new < c5_cur:
                    h_cur = h_new
                    c5_cur = c5_new
            except Exception:
                pass

        for beta in [80, 160]:
            if time_left() < 80:
                break
            bounds_l = [(0.0, 1.0)] * n_cur
            try:
                res = minimize(
                    penalty_smooth_obj, h_cur,
                    jac=penalty_smooth_grad,
                    args=(n_cur, beta, 5.0),
                    method='L-BFGS-B', bounds=bounds_l,
                    options={'maxiter': 800, 'ftol': 1e-14, 'gtol': 1e-11}
                )
                h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n_cur)
                c5_new = compute_c5(h_new, n_cur)
                if c5_new < c5_cur:
                    h_cur = h_new
                    c5_cur = c5_new
            except Exception:
                pass

        if c5_cur < best_c5:
            best_h = h_cur
            best_c5 = c5_cur
            best_n = n_cur

    # Stage 6: Perturbation polish at best resolution
    n_pol = best_n
    for i in range(300):
        if time_left() < 20:
            break
        scale = 0.015 * (1.0 + 0.3 * (i / 100.0))
        h_pert = best_h + rng.randn(n_pol) * scale
        h_pert = project_feasible(h_pert, n_pol)

        c_arr = compute_c_array(h_pert, n_pol)
        top_idx = np.argsort(c_arr)[-3:]

        bounds_p = [(0.0, 1.0)] * n_pol
        cons_p = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n_pol,)}]

        try:
            res = minimize(
                active_set_obj, h_pert,
                jac=active_set_grad,
                args=(n_pol, top_idx),
                method='SLSQP', bounds=bounds_p, constraints=cons_p,
                options={'maxiter': 300, 'ftol': 1e-14}
            )
            h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n_pol)
            c5_new = compute_c5(h_new, n_pol)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    # Stage 7: Final tight SLSQP on best solution
    if time_left() > 20:
        n_fin = best_n
        c_arr = compute_c_array(best_h, n_fin)
        top_idx = np.argsort(c_arr)[-3:]
        bounds_f = [(0.0, 1.0)] * n_fin
        cons_f = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n_fin,)}]
        try:
            res = minimize(
                active_set_obj, best_h,
                jac=active_set_grad,
                args=(n_fin, top_idx),
                method='SLSQP', bounds=bounds_f, constraints=cons_f,
                options={'maxiter': 3000, 'ftol': 1e-17}
            )
            h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n_fin)
            c5_new = compute_c5(h_new, n_fin)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    # Validate
    try:
        val = evaluate_erdos_solution(best_h, best_n)
        if val is not None and 0.0 < val < 1.0:
            best_c5 = min(best_c5, float(val))
    except Exception:
        pass

    return (best_h, best_c5, best_n)
```
