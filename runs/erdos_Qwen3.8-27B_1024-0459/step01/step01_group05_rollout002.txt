def _project_box_sum(a, target):

def _adjust_sum(p, target):
    p = np.clip(p,0,1).copy()
    n=len(p)
    for _ in range(30):
        diff = target - float(np.sum(p))
        if abs(diff) < 1e-14: break
        if diff > 0:
            idx = np.where(p < 1.0 - 1e-15)[0]
        else:
            idx = np.where(p > 1e-15)[0]
        if idx.size == 0: break
        step = diff / float(idx.size)
        p[idx] += step
        np.clip(p,0,1,out=p)
    diff = target - float(np.sum(p))
    if abs(diff) > 1e-15 and n > 0:
        # pick index with capacity
        if diff > 0:
            cand = np.where(p < 1.0)[0]
        else:
            cand = np.where(p > 0.0)[0]
        if cand.size:
            # choose one where p+diff in [0,1]
            ok = (p + diff >= -1e-14) & (p + diff <= 1.0 + 1e-14)
            if np.any(ok):
                idx = np.where(ok)[0][0]
            else:
                idx = cand[len(cand)//2]
            p[idx] = float(np.clip(p[idx] + diff, 0.0, 1.0))

    out = np.empty_like(x)
    pos = x >= 0.0
    out[pos] = 1.0 / (1.0 + np.exp(-x[pos]))
    neg = ~pos
    ex = np.exp(x[neg])
    out[neg] = ex / (1.0 + ex)
    return out


def _raw_corr(h):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode='full')


def _eval_c5(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return 0.0
    c = _raw_corr(h)
    return float(np.max(c) * (2.0 / float(n)))


def _adjust_sum(p, target):
    p = np.clip(np.asarray(p, dtype=float).copy(), 0.0, 1.0)
    n = len(p)
    target = float(target)
    if n == 0:
        return p

    for _ in range(40):
        diff = target - float(np.sum(p))
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            idx = np.where(p < 1.0 - 1e-15)[0]
        else:
            idx = np.where(p > 1e-15)[0]
        if idx.size == 0:
            break
        step = diff / float(idx.size)
        p[idx] += step
        np.clip(p, 0.0, 1.0, out=p)

    diff = target - float(np.sum(p))
    if abs(diff) > 1e-15:
        if diff > 0.0:
            cand = np.where(p < 1.0)[0]
        else:
            cand = np.where(p > 0.0)[0]
        if cand.size > 0:
            ok = (p + diff >= -1e-14) & (p + diff <= 1.0 + 1e-14)
            if np.any(ok):
                i = int(np.where(ok)[0][0])
            else:
                i = int(cand[cand.size // 2])
            p[i] = float(np.clip(p[i] + diff, 0.0, 1.0))
    return p


def _project_box_sum(a, target):
    a = np.clip(np.asarray(a, dtype=float).copy(), 0.0, 1.0)
    n = len(a)
    target = float(target)
    if n == 0:
        return a
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    s0 = float(np.sum(a))
    if abs(s0 - target) <= 1e-9:
        return _adjust_sum(a, target)

    lo = float(np.min(a)) - 1.0
    hi = float(np.max(a)) + 1.0

    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(a - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    p = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    return _adjust_sum(p, target)


def _interpolate_to_n(h, n_new):
    h = np.asarray(h, dtype=float)
    n_old = len(h)
    n_new = int(n_new)
    if n_old == n_new:
        return _project_box_sum(h, 0.5 * float(n_new))
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return _project_box_sum(h_new, 0.5 * float(n_new))


def _weights_from_corr(v, gamma):
    v = np.asarray(v, dtype=float)
    m = float(np.max(v))
    gamma = float(gamma)
    z = gamma * (v - m)
    z = np.clip(z, -80.0, 0.0)
    ez = np.exp(z)
    s = float(np.sum(ez))
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros_like(v, dtype=float)
        w[int(np.argmax(v))] = 1.0
        return w
    return ez / s


def _gradient_from_weights(h, w):
    h = np.asarray(h, dtype=float)
    w = np.asarray(w, dtype=float)
    n = len(h)
    g = np.zeros(n, dtype=float)

    w0 = float(w[n - 1])
    if w0 > 1e-16:
        g += w0 * (1.0 - 2.0 * h)

    for r in range(1, n):
        lap = n - r

        # Positive lag in the correlate output index n-1-r:
        # A_r = sum_i h[i] * (1 - h[i+r])
        wa = float(w[n - 1 - r])
        if wa > 1e-16:
            g[:lap] += wa * (1.0 - h[r:])
            g[r:] -= wa * h[:lap]

        # Negative lag in the correlate output index n-1+r:
        # B_r = sum_i h[i+r] * (1 - h[i])
        wb = float(w[n - 1 + r])
        if wb > 1e-16:
            g[:lap] -= wb * h[r:]
            g[r:] += wb * (1.0 - h[:lap])

    return g


def _z_to_h(u):
    u = np.asarray(u, dtype=float)
    m = len(u)
    n = m + 1
    z = np.empty(n, dtype=float)
    if m > 0:
        z[:m] = u
        z[-1] = -float(np.sum(u))
    else:
        z[0] = 0.0

    target = float(_STATE['target'])

    zmin = float(np.min(z))
    zmax = float(np.max(z))
    lo = zmin - 80.0
    hi = zmax + 80.0

    for _ in range(8):
        if float(np.sum(_sigmoid(z + lo))) < target:
            break
        lo -= 400.0
    for _ in range(8):
        if float(np.sum(_sigmoid(z + hi))) > target:
            break
        hi += 400.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.sum(_sigmoid(z + mid))) < target:
            lo = mid
        else:
            hi = mid

    return _sigmoid(z + 0.5 * (lo + hi))


def _h_to_u(h, n):
    h = np.asarray(h, dtype=float)
    eps = _H_EPS
    p = np.clip(h, eps, 1.0 - eps)
    z = np.log(p / (1.0 - p))
    z = z - float(np.mean(z))
    m = int(n) - 1
    if m <= 0:
        return np.zeros(0, dtype=float)
    u = z[:m].astype(float).copy()
    return np.clip(u, -_Z_BOUND, _Z_BOUND)


def _z_obj(u):
    u = np.asarray(u, dtype=float)
    h = _z_to_h(u)
    v = _raw_corr(h)
    gamma = float(_STATE['gamma'])
    m = float(np.max(v))
    if gamma <= 1e-12:
        return m
    z = gamma * (v - m)
    z = np.clip(z, -80.0, 0.0)
    ez = np.exp(z)
    s = float(np.sum(ez))
    if s <= 0.0 or not np.isfinite(s):
        return m
    return float(m + np.log(s) / gamma)


def _z_grad(u):
    u = np.asarray(u, dtype=float)
    h = _z_to_h(u)
    v = _raw_corr(h)
    gamma = float(_STATE['gamma'])
    w = _weights_from_corr(v, gamma)
    G = _gradient_from_weights(h, w)

    q = h * (1.0 - h)
    Q = float(np.sum(q))
    if Q < 1e-12:
        Q = 1e-12
    W = float(np.sum(G * q)) / Q

    gz = q * (G - W)
    gz -= float(np.mean(gz))

    m = len(u)
    if m == 0:
        return np.zeros(0, dtype=float)
    return gz[:m] - gz[-1]


def _u_lfgs(h0, n, gamma, maxiter, gtol=1e-6, ftol=1e-12):
    target = 0.5 * float(n)
    h0 = _project_box_sum(h0, target)
    u0 = _h_to_u(h0, n)
    if len(u0) == 0:
        return h0

    _STATE['target'] = target
    _STATE['gamma'] = float(gamma)

    bounds = [(-_Z_BOUND, _Z_BOUND)] * len(u0)
    maxiter = int(max(1, maxiter))

    try:
        res = minimize(
            _z_obj,
            u0,
            jac=_z_grad,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol, 'maxls': 60},
        )
        if res is not None and getattr(res, 'x', None) is not None:
            h = _z_to_h(res.x)
        else:
            h = h0
    except Exception:
        h = h0

    return _project_box_sum(h, target)


def _subgrad_refine(h, n, gamma, max_iter, initial_step, time_end):
    target = 0.5 * float(n)
    h = _project_box_sum(h, target)
    best_h = h.copy()
    best_c = _eval_c5(h)

    step = float(initial_step)
    min_step = 1e-9
    stall = 0
    target_time = float(time_end)

    for _it in range(int(max_iter)):
        if time.time() >= target_time:
            break

        v = _raw_corr(h)
        M = float(np.max(v))
        w = _weights_from_corr(v, float(gamma))
        g = _gradient_from_weights(h, w)
        nrm = float(np.linalg.norm(g))
        if not np.isfinite(nrm) or nrm < 1e-13:
            break

        s = step
        best_cand = None
        best_M = M

        for _ls in range(25):
            cand = _project_box_sum(h - s * g, target)
            v2 = _raw_corr(cand)
            M2 = float(np.max(v2))

            if M2 < best_M - 1e-12:
                best_M = M2
                best_cand = cand

            if best_M < M - 1e-11:
                break

            s *= 0.5
            if s < min_step:
                break

        if best_cand is None:
            stall += 1
            step *= 0.6
            if step < min_step or stall > 30:
                break
            continue

        h = best_cand
        c = float(best_M * (2.0 / float(n)))
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        step = min(step * 1.05, 2.0 * float(initial_step))

    return best_h, best_c


def _record_if_better(state, h, n):
    if state is None or h is None:
        return
    n = int(n)
    target = 0.5 * float(n)
    h = _project_box_sum(h, target)
    c = _eval_c5(h)
    if not np.isfinite(c):
        return
    cur = state.get('c5', None)
    if cur is None or c < float(cur) - 1e-12:
        state['h'] = h
        state['c5'] = float(c)
        state['n'] = n


def _try_reverse_symmetry(state):
    if state is None or 'h' not in state:
        return
    h = state['h']
    n = int(state['n'])
    _record_if_better(state, h[::-1].copy(), n)
    hs = 0.5 * (h + h[::-1].copy())
    _record_if_better(state, hs, n)


def _try_snap_improve(state):
    if state is None or 'h' not in state:
        return
    h = state['h']
    n = int(state['n'])
    for tol in (1e-8, 1e-6, 1e-5, 5e-4):
        hs = np.where(h < tol, 0.0, np.where(h > 1.0 - tol, 1.0, h))
        _record_if_better(state, hs, n)


def _refine_resolution(state, h_start, n, phase_end, rng):
    h_start = _project_box_sum(h_start, 0.5 * float(n))
    _record_if_better(state, h_start, n)
    _try_reverse_symmetry(state)

    schedule = [
        (10.0, 90),
        (40.0, 80),
        (150.0, 60),
        (600.0, 45),
        (1500.0, 35),
    ]

    for gamma, base_iter in schedule:
        if time.time() > float(phase_end) - 140.0:
            break
        left = float(phase_end) - time.time()
        if left < 110.0:
            break
        maxiter = int(min(base_iter, max(18, (left - 60.0) / 5.0)))
        if maxiter < 16:
            break
        h_new = _u_lfgs(state['h'], n, gamma, maxiter, gtol=1e-6, ftol=1e-12)
        _record_if_better(state, h_new, n)
        _try_reverse_symmetry(state)

    sub_schedule = [
        (30.0, 80, 0.030),
        (120.0, 90, 0.016),
        (600.0, 100, 0.006),
    ]
    for gamma, iters, st in sub_schedule:
        if time.time() > float(phase_end) - 70.0:
            break
        h2, _c2 = _subgrad_refine(state['h'], n, gamma, iters, st, float(phase_end) - 20.0)
        _record_if_better(state, h2, n)

    if time.time() < float(phase_end) - 180.0 and n <= 300:
        for scale in (0.025, 0.012, 0.006, 0.003):
            if time.time() > float(phase_end) - 80.0:
                break
            h_p = state['h'] + float(scale) * rng.randn(n)
            h_p = _project_box_sum(h_p, 0.5 * float(n))
            left = float(phase_end) - time.time()
            if left < 100.0:
                break
            maxiter = int(min(80, max(16, (left - 60.0) / 8.0)))
            if maxiter < 14:
                break
            h_new = _u_lfgs(h_p, n, 400.0, maxiter, gtol=1e-6, ftol=1e-12)
            _record_if_better(state, h_new, n)
            h2, _c2 = _subgrad_refine(state['h'], n, 1200.0, 45, 0.004, float(phase_end) - 10.0)
            _record_if_better(state, h2, n)

    _try_snap_improve(state)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s)
    rng = np.random.RandomState(int(seed))

    try:
        parent = np.asarray(initial_h_values, dtype=float)
    except Exception:
        parent = None

    if parent is None:
        n0 = 96
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        parent = 0.5 + 0.45 * np.sin(np.pi * x0)
        parent = _project_box_sum(parent, 0.5 * float(n0))

    parent = np.clip(parent, 0.0, 1.0)
    n_parent = int(len(parent))
    parent = _project_box_sum(parent, 0.5 * float(n_parent))

    state = {'h': parent.copy(), 'n': n_parent, 'c5': _eval_c5(parent)}

    resolutions = []
    base_candidates = [n_parent, 128, 160, 200, 240]
    for nn in base_candidates:
        nn = int(nn)
        if nn > 10 and nn not in resolutions:
            resolutions.append(nn)

    if n_parent >= 200:
        extra = min(n_parent + 32, 288)
        if extra not in resolutions:
            resolutions.append(extra)

    resolutions.sort()
    if n_parent in resolutions:
        resolutions.remove(n_parent)
        resolutions.insert(0, n_parent)

    m = len(resolutions)
    if m == 0:
        frac = [0.0]
    else:
        frac = [0.30]
        rem = 0.70
        for _ii in range(1, m):
            frac.append(rem / float(m - 1))

    for idx, nn in enumerate(resolutions):
        if time.time() >= deadline - 70.0:
            break
        budget_n = frac[idx] * float(budget_s)
        phase_end = min(deadline - 35.0, time.time() + budget_n)
        if time.time() >= phase_end:
            continue
        if nn == int(state['n']):
            h_start = state['h']
        else:
            h_start = _interpolate_to_n(state['h'], nn)
        _refine_resolution(state, h_start, nn, phase_end, rng)

    if time.time() < deadline - 80.0:
        _refine_resolution(state, state['h'], int(state['n']), deadline - 25.0, rng)

    _try_reverse_symmetry(state)
    _try_snap_improve(state)

    final_h = _project_box_sum(state['h'], 0.5 * float(state['n']))
    final_c5 = _eval_c5(final_h)

    return (final_h, float(final_c5), int(state['n']))
