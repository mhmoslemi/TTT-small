        w = weights[j]
        if w < 1e-12:

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx


def compute_c5_smooth(h, n, dx, temperature):
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(conv)
    exp_vals = np.exp((conv - m) / temperature)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    obj = m + temperature * np.log(sum_exp)
    return obj, weights, conv


def compute_c5_grad(h, n, dx, temperature):
    obj, weights, conv = compute_c5_smooth(h, n, dx, temperature)
    grad = np.zeros(n)
    nconv = 2 * n - 1
    for j in range(nconv):
        w = weights[j]
        if w < 1e-12:
            continue
        k = j - (n - 1)
        ak = abs(k)
        if ak == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        else:
            if ak < n:
                grad[ak:n] += w * (1.0 - h[0:n-ak]) * dx
                grad[0:n-ak] += w * (-h[ak:n]) * dx
    return obj, grad


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-14
            if not np.any(free):
                break
            step = diff / np.sum(free)
            h[free] = np.minimum(h[free] + step, 1.0)
        else:
            free = h > 0.0 + 1e-14
            if not np.any(free):
                break
            step = diff / np.sum(free)
            h[free] = np.maximum(h[free] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-12:
        h += (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h


def lbfgs_smooth(h0, n, dx, temperature, maxiter=500):
    def obj_and_grad(x):
        val, g = compute_c5_grad(x, n, dx, temperature)
        return val, g
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_and_grad, h0.copy(), jac=True, method='L-BFGS-B',
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-15,
                            'gtol': 1e-10, 'maxls': 30,
                            'maxfun': maxiter * 20})
    return res.x, res.fun

                   eps=1e-8, n_iter=1000, temperature=0.1):

    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h, n)
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    no_improve = 0
    for it in range(n_iter):
        t += 1
        _, grad = compute_c5_grad(h, n, dx, temperature)
        m_vec = beta1 * m_vec + (1 - beta1) * grad
        v_vec = beta2 * v_vec + (1 - beta2) * grad * grad
        m_hat = m_vec / (1 - beta1 ** t)

            no_improve = 0
        else:

            if no_improve > 300:
                break

def try_structured_init(n, rng):
    candidates = []
    m_half = n // 2
    # Alternating blocks
    for block_size in [2, 3, 4, 5, 7, 11]:
        h = np.zeros(n)
        pos = 0
        fill = True
        while pos < n and np.sum(h) < m_half:
            end = min(pos + block_size, n)
            if fill:
                h[pos:end] = 1.0
            pos = end
            fill = not fill
        if np.sum(h) > m_half:
            h[-int(np.sum(h) - m_half):] = 0.0

            candidates.append(h.copy())

            pos += p

            h[idxs[-int(np.sum(h) - m_half):]] = 0.0
        if np.sum(h) == m_half:
            candidates.append(h.copy())
    # Sines
