import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    # Exact fix
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
                np.clip(h[mask], 0.0, 1.0, out=h[mask])
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
                np.clip(h[mask], 0.0, 1.0, out=h[mask])
    return h

def smooth_max_obj_and_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    obj = m + np.log(exp_sum) / beta
    w = exp_vals / exp_sum
    
    grad = np.zeros(n)

        # -h[j+d] term

        je2 = min(n-1, n-1-d)
        if js2 <= je2:

    def obj(h):
        h_c = np.clip(h, 0, 1)
        v, _ = smooth_max_obj_and_grad(h_c, beta)
        return v

        _, g = smooth_max_obj_and_grad(h_c, beta)
        return g
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,

def project_sum(h, target):
