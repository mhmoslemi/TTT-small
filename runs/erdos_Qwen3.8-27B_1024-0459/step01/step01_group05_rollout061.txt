import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def compute_D(h):
    """Compute D[k] = sum_i h[i+k]*(1-h[i]) for k=0..n-1"""
    n = len(h)
    D = np.zeros(n)
    for k in range(n):
        D[k] = np.sum(h[k:] * (1.0 - h[:n-k]))
    return D

def project_sum(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-14:
        if diff > 0:
            free = (h < 1.0 - 1e-12)
            if np.any(free):
                per = diff / np.sum(free)
                h[free] = np.minimum(1.0, h[free] + per)
        else:
            free = (h > 1e-12)
            if np.any(free):
                per = diff / np.sum(free)
                h[free] = np.maximum(0.0, h[free] + per)
    return h

def smooth_corr_val_grad(h, beta):
    """Compute smooth max of correlation and analytical gradient."""
    n = len(h)
    D = np.zeros(n)
    for k in range(n):
        D[k] = np.sum(h[k:] * (1.0 - h[:n-k]))
    
    # Full correlation array (2n-1 values)
    C = np.empty(2*n-1)
    C[n-1] = D[0]
    for k in range(1, n):
        C[n-1+k] = D[k]
        C[n-1-k] = D[k]
    
    # Smooth max
    c_max = np.max(C)
    shifted = beta * (C - c_max)
    shifted = np.clip(shifted, -60.0, 60.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    val = c_max + log_sum / beta
    
    # Softmax weights
    p = np.exp(shifted - log_sum)
    
    # Analytical gradient
    grad = np.zeros(n)
    for m in range(2*n-1):
        pm = p[m]
        if pm < 1e-14:
            continue
        k = abs(m - (n-1))
        # dD[k]/dh[j] = (1-h[j-k]) for j>=k, minus h[j+k] for j<=n-1-k
        # Part 1: for j = k to n-1: add (1 - h[j-k])
        #   Indices: j-k = 0 to n-1-k
        if k < n:
            grad[k:] += pm * (1.0 - h[:n-k])
        # Part 2: for j = 0 to n-1-k: add -h[j+k]
        #   Indices: j+k = k to n-1
        if k > 0:
            grad[:n-k] -= pm * h[k:]
    
    return val, grad

def make_binary_pattern(n, pattern_type, seed):
    """Create a binary starting pattern with n/2 ones."""
    rng = np.random.RandomState(seed)
    if n % 2 != 0:
        n = n - 1
    half = n // 2
    
    if pattern_type == 0:  # Random
        h = np.zeros(n)
        idx = rng.choice(n, half, replace=False)
        h[idx] = 1.0
    elif pattern_type == 1:  # Grouped (111...000...)
        h = np.zeros(n)
        h[:half] = 1.0

def compute_D(h):
    n = len(h)

        D[k] = np.sum(h[k:] * (1.0 - h[:n-k]))

def project_sum(h, target):
