```python
def linear_oracle(g):
    """Minimize g·h over {h: 0≤h≤1, sum(h)=n/2}"""
    n = len(g)
    idx = np.argsort(g)[:n//2]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def frank_wolfe_smooth(h, tau, t_start, t_end, n_iter):
    n = len(h)
    dx = 2.0 / n
    h_c = h.copy()
    v = np.zeros(n)  # Nesterov velocity
    for t in range(n_iter):
        step = (t_end - t_start) * (t / n_iter) + t_start
        # Lookahead
        h_la = h_c + v
        h_la = np.clip(h_la, 0, 1)
        # Gradient of smooth max at h_la
        corr = np.correlate(h_la, 1-h_la, mode="full")
        max_c = np.max(corr)
        p = np.exp((corr - max_c) / tau)
        p /= p.sum()
        # Gradient
        g = np.zeros(n)
        # Vectorized: g[j] = sum_k p[k] * (1 - h_la[j+d_k]) - sum_k p[k] * h_la[j-d_k]
        # This is a correlation-like operation
        # g1[j] = sum_k p[k] * h_la[j + k - (n-1)]
        # = np.correlate(h_la, p, "full")[j + (n-1)] (approximately)
        # Actually let me just do it with the few significant p values
        significant = np.where(p > 1e-8)[0]
        for k in significant:
            d = k - (n-1)
            if d >= 0:
                valid = np.arange(0, n-d)
                g[valid] += p[k] * (1.0 - h_la[valid + d])
                if d > 0:
                    valid2 = np.arange(d, n)
                    g[valid2] -= p[k] * h_la[valid2 - d]
            else:
                dd = -d
                valid = np.arange(dd, n)
                g[valid] += p[k] * (1.0 - h_la[valid - dd])
                valid2 = np.arange(0, n-dd)
                g[valid2] -= p[k] * h_la[valid2 + dd]
        # Linear oracle
        v_new = linear_oracle(g)
        # Update
        h_c = h_c + step * (v_new - h_c)
        h_c = fix_sum(h_c, n)
        h_c = np.clip(h_c, 0, 1)
        v = h_c - h_la  # or some other Nesterov update
    return h_c
```

```python
import functools
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = h.copy()
    target = n / 2.0
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, tau):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    dx = 2.0 / len(h)
    return float((m + tau * np.log(np.sum(np.exp((corr - m) / tau)))) * dx)

def grad_smooth(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    p = np.exp((corr - max_c) / tau)
    p /= p.sum()
    g = np.zeros(n)
    sig = np.where(p > 1e-10)[0]
    for k in sig:
        d = k - (n - 1)
        wk = p[k]
        if d >= 0:
            if d < n:
                g[:n-d] += wk * (1.0 - h[d:])
            if d > 0:
                g[d:] -= wk * h[:n-d]
        else:
            dd = -d
            if dd < n:
                g[dd:] += wk * (1.0 - h[:n-dd])
            g[:n-dd] -= wk * h[dd:]
    return g * dx

def slsqp_wrap(x, n, tau):
    h = np.clip(x, 0.0, 1.0)
    return smooth_obj(h, tau)

def slsqp_constraint(x):
    return np.array([x.sum() - len(x) / 2.0])

def slsqp_jac(x):
    return np.ones((1, len(x)))

    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return fix_sum(np.interp(x_new, x_old, h_old), n_new)
```

```python
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def smooth_val(h, tau):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    dx = 2.0 / len(h)
    return float((m + tau * np.log(np.sum(np.exp((corr - m) / tau)))) * dx)


def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    p = np.exp((corr - max_c) / tau)
    p /= p.sum()
    g = np.zeros(n)
    sig = np.where(p > 1e-12)[0]
    for k in sig:
        d = int(k - (n - 1))
        wk = p[k]
        if d >= 0:
            if d < n:
                g[:n - d] += wk * (1.0 - h[d:])
            if d > 0:
                g[d:] -= wk * h[:n - d]
        else:
            dd = -d
            if dd < n:
                g[dd:] += wk * (1.0 - h[:n - dd])
            g[:n - dd] -= wk * h[dd:]
    return g * dx


def penalized_val(x, tau, lam):
    h = np.clip(x, 0.0, 1.0)
    n = len(h)
    sv = smooth_val(h, tau)
    pen = lam * (h.sum() - n / 2.0) ** 2
    return sv + pen


def penalized_jac(x, tau, lam):
    h = np.clip(x, 0.0, 1.0)
    n = len(h)
    g = smooth_grad(h, tau)
    g += 2.0 * lam * (h.sum() - n / 2.0)
    mask = (h < 1e-14) | (h > 1.0 - 1e-14)
    g[mask] = 0.0
    return g


def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return fix_sum(np.interp(x_new, x_old, h_old), n_new)


def eq_con(x, n):
    return np.array([x.sum() - n / 2.0])


def eq_jac(x, n):
    return np.ones((1, n))


def slsqp_fun(x, tau, n):
    h = np.clip(x, 0.0, 1.0)
    return smooth_val(h, tau)


def subgrad_nesterov(h, alpha0, n_iter, cooling=0.995):
    n = len(h)
    h_curr = h.copy()
    v = np.zeros(n)
    for i in range(n_iter):
        alpha = alpha0 * (cooling ** i)
        # Nesterov lookahead
        h_la = h_curr + v
        h_la = np.clip(h_la, 0.0, 1.0)
        h_la = fix_sum(h_la, n)
        g = smooth_grad(h_la, 0.001)
        h_new = h_curr - alpha * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = fix_sum(h_new, n)
        v = h_new - h_curr
        h_curr = h_new
    return h_curr


def fw_step(h, tau, gamma):
    n = len(h)
    g = smooth_grad(h, tau)
    # Linear oracle: minimize g·h over {0<=h<=1, sum=h n/2}
    idx = np.argsort(g)[:n // 2]
    v = np.zeros(n)
    v[idx] = 1.0
    h_new = (1.0 - gamma) * h + gamma * v
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = fix_sum(h_new, n)
    return h_new


def fw_nesterov(h, tau, n_iter):
    n = len(h)
    h_curr = h.copy()
    w = np.zeros(n)
    for t in range(n_iter):
        gamma = 2.0 / (t + 2)
        # Lookahead
        if t == 0:
            h_la = h_curr
        else:
            h_la = h_curr + (t / (t + 2)) * (h_curr - w)
            h_la = np.clip(h_la, 0.0, 1.0)
            h_la = fix_sum(h_la, n)
        g = smooth_grad(h_la, tau)
        idx = np.argsort(g)[:n // 2]
        v = np.zeros(n)
        v[idx] = 1.0
        h_new = (1.0 - gamma) * h_curr + gamma * v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = fix_sum(h_new, n)
        w = h_curr.copy()
        h_curr = h_new
    return h_curr


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None and np.size(initial_h_values) > 0:
        parent_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        parent_h = np.full(180, 0.5)

    best_h = parent_h.copy()
    best_c5 = compute_c5(best_h)
    n_parent = len(parent_h)

    # Phase 1: L-BFGS-B with penalty at parent resolution
    n = n_parent
    h_curr = best_h.copy()
    taus = [0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    for tau in taus:
        if time.time() - t0 > budget_s * 0.25:
            break
        lam = 1e4
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(
                fun=penalized_val, x0=h_curr,
                jac=penalized_jac,
                args=(tau, lam),
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-12, 'disp': False}
            )
            h_new = fix_sum(np.clip(res.x, 0.0, 1.0), n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_curr = h_new
        except Exception:
            pass

    # Phase 2: SLSQP polish at parent resolution
    for tau in [0.005, 0.002, 0.001]:
        if time.time() - t0 > budget_s * 0.35:
            break
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(
                fun=slsqp_fun, x0=h_curr,
                args=(tau, n),
                method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': eq_con, 'jac': eq_jac, 'args': (n,)}],
                options={'maxiter': 1000, 'ftol': 1e-14, 'disp': False}
            )
            h_new = fix_sum(np.clip(res.x, 0.0, 1.0), n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_curr = h_new
        except Exception:
            pass

    # Phase 3: Random restarts at parent resolution
    n_restarts = 8
    for r in range(n_restarts):
        if time.time() - t0 > budget_s * 0.45:
            break
        noise = rng.normal(0, 0.06, n)
        x_pert = fix_sum(np.clip(h_curr + noise, 0.0, 1.0), n)
        tau = 0.003
        lam = 1e5
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(
                fun=penalized_val, x0=x_pert,
                jac=penalized_jac, args=(tau, lam),
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': 3000, 'ftol': 1e-15, 'disp': False}
            )
            h_new = fix_sum(np.clip(res.x, 0.0, 1.0), n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_curr = h_new
        except Exception:
            pass

    # Phase 4: Subgradient + Nesterov at parent resolution
    if time.time() - t0 < budget_s * 0.55:
        h_sub = subgrad_nesterov(h_curr, alpha0=0.005, n_iter=500, cooling=0.998)
        c5_sub = compute_c5(h_sub)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()
        # Polish the subgradient result
        tau = 0.002
        lam = 1e5
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(
                fun=penalized_val, x0=h_sub,
                jac=penalized_jac, args=(tau, lam),
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': 2000, 'ftol': 1e-15, 'disp': False}
            )
            h_new = fix_sum(np.clip(res.x, 0.0, 1.0), n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    # Phase 5: Upsample to higher resolution
    for n_new in [2 * n_parent, 4 * n_parent]:
        if time.time() - t0 > budget_s * 0.80:
            break
        h_hi = upsample(best_h, n_new)
        n_hi = n_new
        bounds_hi = [(0.0, 1.0)] * n_hi

        for tau in [0.1, 0.05, 0.02, 0.01, 0.005]:
            if time.time() - t0 > budget_s * 0.85:
                break
            lam = 1e4
            try:
                res = minimize(
                    fun=penalized_val, x0=h_hi,
                    jac=penalized_jac, args=(tau, lam),
                    method='L-BFGS-B', bounds=bounds_hi,
                    options={'maxiter': 3000, 'ftol': 1e-15, 'disp': False}
                )
                h_new = fix_sum(np.clip(res.x, 0.0, 1.0), n_hi)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_hi = h_new
            except Exception:
                pass

        # SLSQP at higher res
        if time.time() - t0 < budget_s * 0.88:
            for tau in [0.003, 0.001]:
                if time.time() - t0 > budget_s * 0.90:
                    break
                try:
                    res = minimize(
                        fun=slsqp_fun, x0=h_hi, args=(tau, n_hi),
                        method='SLSQP', bounds=bounds_hi,
                        constraints=[{'type': 'eq', 'fun': eq_con, 'jac': eq_jac, 'args': (n_hi,)}],
                        options={'maxiter': 500, 'ftol': 1e-13, 'disp': False}
                    )
                    h_new = fix_sum(np.clip(res.x, 0.0, 1.0), n_hi)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                    h_hi = h_new
                except Exception:
                    pass

    # Phase 6: Frank-Wolfe exploration at parent resolution
    if time.time() - t0 < budget_s * 0.92:
        n_fw = n_parent
        for r in range(3):
            if time.time() - t0 > budget_s * 0.94:
                break
            h_fw0 = fix_sum(best_h + rng.normal(0, 0.1, n_fw), n_fw)
            h_fw = fw_nesterov(h_fw0, tau=0.05, n_iter=300)
            c5_fw = compute_c5(h_fw)
            if c5_fw < best_c5:
                best_c5 = c5_fw
                best_h = h_fw.copy()
            # Polish
            tau = 0.003
            lam = 1e5
            bounds = [(0.0, 1.0)] * n_fw
            try:
                res = minimize(
                    fun=penalized_val, x0=h_fw,
                    jac=penalized_jac, args=(tau, lam),
                    method='L-BFGS-B', bounds=bounds,
                    options={'maxiter': 2000, 'ftol': 1e-15, 'disp': False}
                )
                h_new = fix_sum(np.clip(res.x, 0.0, 1.0), n_fw)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    # Phase 7: Final fine polish
    if time.time() - t0 < budget_s * 0.97:
        n_fin = len(best_h)
        bounds_fin = [(0.0, 1.0)] * n_fin
        for r in range(15):
            if time.time() - t0 > budget_s * 0.98:
                break
            noise = rng.normal(0, 0.008, n_fin)
            x_pert = fix_sum(np.clip(best_h + noise, 0.0, 1.0), n_fin)
            tau = 0.001
            lam = 1e5
            try:
                res = minimize(
                    fun=penalized_val, x0=x_pert,
                    jac=penalized_jac, args=(tau, lam),
                    method='L-BFGS-B', bounds=bounds_fin,
                    options={'maxiter': 5000, 'ftol': 1e-16, 'disp': False}
                )
                h_new = fix_sum(np.clip(res.x, 0.0, 1.0), n_fin)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
            except Exception:
                pass
        # SLSQP final
        if time.time() - t0 < budget_s * 0.99:
            try:
                res = minimize(
                    fun=slsqp_fun, x0=best_h, args=(0.0005, n_fin),
                    method='SLSQP', bounds=bounds_fin,
                    constraints=[{'type': 'eq', 'fun': eq_con, 'jac': eq_jac, 'args': (n_fin,)}],
                    options={'maxiter': 2000, 'ftol': 1e-15, 'disp': False}
                )
                h_new = fix_sum(np.clip(res.x, 0.0, 1.0), n_fin)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
            except Exception:
                pass

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
```
