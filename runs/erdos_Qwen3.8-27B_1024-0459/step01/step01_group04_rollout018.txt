    lo, hi = -30.0, 30.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)

        else:

    n = z.size
    lam = find_lambda(z, n)
    return sigmoid(z + lam)

    n = h.size
    scores = np.empty(2*n - 1)
    scores[n-1] = np.sum(h * (1.0 - h))
    for d in range(1, n):
        m = n - d
        scores[n-1+d] = np.dot(h[d:], 1.0 - h[:m])
        scores[n-1-d] = np.dot(h[:m], 1.0 - h[d:])
    return scores

def smooth_max_obj_grad(z, alpha):
    global _RUN_DEADLINE
    if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE:
        return 1e10, np.zeros_like(z)
    n = z.size
    h = h_from_z(z)
    scores = compute_scores(h)
    # Smooth max
    m = np.max(scores)
    shifted = (scores - m) / alpha
    # For numerical stability, shift by max
    exps = np.exp(shifted)
    A = np.sum(exps)
    if A < 1e-300:
        return 1e10, np.zeros_like(z)
    w = exps / A
    obj = (2.0 / n) * (m + alpha * math.log(A))
    
    # Gradient w.r.t. h
    dh = np.zeros(n)
    # Zero lag
    dh += w[n-1] * (1.0 - 2.0 * h)
    # All other lags
    for d in range(1, n):
        wd = w[n-1+d]
        m_count = n - d
        dh[:m_count] -= wd * h[d:]       # (1-h[j]) derivative
        dh[d:] += wd * (1.0 - h[:m_count])  # h[j+d] derivative
        
        wd2 = w[n-1-d]
        dh[:m_count] += wd2 * (1.0 - h[d:])  # h[j] derivative
        dh[d:] -= wd2 * h[:m_count]          # (1-h[j+d]) derivative

    denom = np.sum(h_deriv)
