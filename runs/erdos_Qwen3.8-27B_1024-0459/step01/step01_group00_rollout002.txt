import numpy as np

def positive_autocorr(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return np.empty(0, dtype=float)
    if n <= 384:
        full = np.correlate(h, h, mode='full')
        return full[n - 1:n].copy()
    need = 2 * n - 1
    m = 1
    while m < need:
        m <<= 1
    F = np.fft.rfft(h, m)
    A = np.fft.irfft(F * np.conjugate(F), m)
    return A[:n].copy()


def get_c_all(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return np.empty(0, dtype=float)
    total = float(np.sum(h))
    S = positive_autocorr(h)
    if n == 1:
        return np.array([total - S[0]], dtype=float)
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    c = np.empty(2 * n - 1, dtype=float)
    c[0] = total - S[0]
    t = np.arange(1, n, dtype=int)
    St = S[1:1 + (n - 1)]
    c[1::2] = pref[n - t] - St
    c[2::2] = total - pref[t] - St
    return c


def compute_c5_fast(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return 0.0
    c = get_c_all(h)
    if c.size == 0:
        return 0.0
    return float(np.max(c) * (2.0 / float(n)))


def compute_c5_direct(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return 0.0
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / float(n)))


def fast_project_box_sum(v, target=None):
    v = np.asarray(v, dtype=float).copy().ravel()
    n = v.size
    if n == 0:
        return v
    if target is None:
        target = 0.5 * float(n)
    target = float(target)
    v = np.clip(v, 0.0, 1.0)

def project_box_sum(v, target=None):

    if n == 0:
        return v
    if target is None:
        target = 0.5 * float(n)

    if abs(float(np.sum(v)) - target) < 1e-12:
        return v
    vmin = float(np.min(v))
    vmax = float(np.max(v))

def enforce_exact_sum(h):

def interpolate_to_n(h, n):
    h = np.asarray(h, dtype=float).ravel()
    n = int(n)
    if h.size == 0:
        return project_box_sum(np.full(n, 0.5))
    if h.size == n:
        return project_box_sum(h)
    x_old = np.arange(h.size, dtype=float) * (2.0 / float(h.size))
    x_new = np.arange(n, dtype=float) * (2.0 / float(n))
    y = np.interp(x_new, x_old, h)
    return project_box_sum(y)


def upsample_repeat(h, n):
    h = np.asarray(h, dtype=float).ravel()
    n = int(n)
    if h.size == 0:
        return project_box_sum(np.full(n, 0.5))
    if h.size == n:
        return project_box_sum(h)
    if n % h.size == 0:
        return project_box_sum(np.repeat(h, n // h.size))
    return interpolate_to_n(h, n)


def normalize_binary(h, half):
    h = np.asarray(h, dtype=float).copy()
    cnt = int(np.rint(float(np.sum(h))))
    half = int(half)
    if cnt > half:
        idx = np.flatnonzero(h > 0.5)
        if idx.size > 0:
            h[idx[:cnt - half]] = 0.0
    elif cnt < half:
        idx = np.flatnonzero(h < 0.5)
        if idx.size > 0:
            h[idx[:half - cnt]] = 1.0
    return h


def binary_init(n, mode, rng):
    n = int(n)
    half = n // 2
    idx = np.arange(n, dtype=int)
    h = np.zeros(n, dtype=float)
    if mode == 0:
        h[:half] = 1.0
    elif mode == 1:
        q = max(1, n // 4)
        h[:q] = 1.0
        h[n // 2:n // 2 + q] = 1.0
    elif mode == 2:
        perm = rng.permutation(n)
        h[perm[:half]] = 1.0
    elif mode == 3:
        order = np.argsort((idx * 7919) % n)
        h[order[:half]] = 1.0
    elif mode == 4:
        h = np.where((idx % 8) < 4, 1.0, 0.0)
    elif mode == 5:
        h = np.where((idx % 16) < 8, 1.0, 0.0)
    elif mode == 6:
        order = np.argsort(np.abs(np.sin(0.318 * idx + 1.117)))
        h[order[:half]] = 1.0
    elif mode == 7:
        order = np.argsort(np.abs(np.sin(1.7 * idx * idx + 0.5)))
        h[order[:half]] = 1.0
    else:
        perm = rng.permutation(n)
        h[perm[:half]] = 1.0
    return normalize_binary(h, half)


def random_sinusoid_init(n, rng):
    n = int(n)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    y = np.full(n, 0.5, dtype=float)
    max_modes = 2 + int(rng.randint(0, 3))
    for m in range(1, max_modes + 1):
        amp = float(rng.uniform(0.03, 0.28)) / max(1.0, 0.7 * float(np.sqrt(m)))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        a = float(rng.uniform(0.3, 1.0))
        b = float(rng.uniform(0.0, 0.6))
        y += amp * (a * np.cos(np.pi * float(m) * x + phase) +
                    b * np.sin(np.pi * float(m) * x + phase))
    return project_box_sum(y)


def make_initial_guesses(n, rng, reference, max_candidates=8):
    n = int(n)
    out = []
    if reference is not None and len(reference) > 0:
        out.append(interpolate_to_n(reference, n))
        if n % len(reference) == 0 and n > len(reference):
            out.append(upsample_repeat(reference, n))
    for mode in range(8):
        out.append(binary_init(n, mode, rng))
    for _ in range(4):
        out.append(random_sinusoid_init(n, rng))
    out.append(project_box_sum(rng.rand(n)))
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    out.append(project_box_sum(0.5 + 0.42 * np.cos(np.pi * x)))
    out.append(project_box_sum(0.5 + 0.38 * np.cos(np.pi * 2.0 * x)))
    out.append(project_box_sum(0.5 + 0.30 * np.sin(np.pi * x)))
    unique = []
    for h in out:
        h = project_box_sum(h)
        ok = True
        for u in unique:
            if float(np.max(np.abs(h - u))) < 0.045:
                ok = False
                break
        if ok:
            unique.append(h)
        if len(unique) >= int(max_candidates):
            break
    if not unique:
        unique.append(project_box_sum(rng.rand(n)))
    return unique


def choose_quick_best(cands):
    best_h = None
    best_c = np.inf
    for h in cands:
        h2 = project_box_sum(h)
        c = compute_c5_fast(h2)
        if c < best_c:
            best_c = c
            best_h = h2.copy()
    if best_h is None:
        return project_box_sum(np.full(128, 0.5)), compute_c5_fast(np.full(128, 0.5))
    return best_h, float(best_c)


def pair_score(item):
    return item[0]


def scored_candidates(cands):
    lst = []
    for h in cands:
        h2 = project_box_sum(h)
        lst.append((compute_c5_fast(h2), h2))
    lst.sort(key=pair_score)
    return lst


def gradient_from_weights(h, w):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    g = np.zeros(n, dtype=float)
    if n == 0:
        return g
    if w.size > 0:
        w0 = float(w[0])
        if w0 > 1e-15:
            g += (1.0 - 2.0 * h) * w0
    if n > 1 and w.size >= 2 * n - 1:
        wp = w[1::2]
        wn = w[2::2]
        for i in range(n - 1):
            t = i + 1
            a = float(wp[i])
            b = float(wn[i])
            if a > 1e-15:
                g[:n - t] += a * (1.0 - h[t:])
                g[t:] -= a * h[:n - t]
            if b > 1e-15:
                g[t:] += b * (1.0 - h[:n - t])
                g[:n - t] -= b * h[t:]
    return g


def smooth_value_grad(h, beta, max_active=None):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return 0.0, np.empty(0)
    c = get_c_all(h)
    cmax = float(np.max(c))
    z = float(beta) * (c - cmax)
    zm = float(np.max(z))
    expz = np.exp(z - zm)
    s = float(np.sum(expz))
    if s <= 0.0:
        w = np.zeros_like(c)
    else:
        w = expz / s
    if max_active is not None and max_active < len(w):
        k = int(min(max_active, len(w)))
        if k > 0 and k < len(w):
            idx = np.argpartition(w, -k)[-k:]
            w2 = np.zeros_like(w)
            w2[idx] = w[idx]
            sw = float(np.sum(w2))
            if sw > 0.0:
                w = w2 / sw
            else:
                ridx = int(np.argmax(w))
                w = np.zeros_like(w)
                w[ridx] = 1.0
    grad = gradient_from_weights(h, w)
    val = cmax + (zm + float(np.log(s))) / float(beta)
    return float(val), grad


def get_bounds(n):
    return [(0.0, 1.0) for _ in range(int(n))]


def sum_constraint(x):
    return float(np.sum(x) - _TARGET_GLOBAL)


def smooth_obj(x):
    x = np.asarray(x, dtype=float)
    x = np.clip(x, 0.0, 1.0)
    val, _ = smooth_value_grad(x, _BETA_GLOBAL, _MAX_ACTIVE_GLOBAL)
    return float(val)


def smooth_jac(x):
    x = np.asarray(x, dtype=float)
    x = np.clip(x, 0.0, 1.0)
    _, g = smooth_value_grad(x, _BETA_GLOBAL, _MAX_ACTIVE_GLOBAL)
    return g


def slsqp_smooth(h, n, beta, maxiter):
    global _TARGET_GLOBAL, _BETA_GLOBAL, _MAX_ACTIVE_GLOBAL
    h = np.asarray(h, dtype=float).ravel()
    n = int(n)
    h = project_box_sum(h, 0.5 * float(n))
    _TARGET_GLOBAL = 0.5 * float(n)
    _BETA_GLOBAL = float(beta)
    _MAX_ACTIVE_GLOBAL = (2 * n - 1) if n <= 192 else 48
    try:
        res = minimize(smooth_obj,
                       h,
                       jac=smooth_jac,
                       method='SLSQP',
                       bounds=get_bounds(n),
                       constraints=[{'type': 'eq', 'fun': sum_constraint}],
                       options={'maxiter': int(maxiter), 'ftol': 1e-12, 'maxls': 40})
        h2 = project_box_sum(res.x, 0.5 * float(n))
        c2 = compute_c5_fast(h2)
        return h2, c2
    except Exception:
        return h, compute_c5_fast(h)


def random_pair_move(h, n, rng):
    h = np.asarray(h, dtype=float).ravel()
    ht = h.copy()
    i = int(rng.randint(0, n))
    j = int(rng.randint(0, max(1, n - 1)))
    if i == j:
        j = (j + 1) % n
    lo = max(-float(ht[i]), float(ht[j]) - 1.0)
    hi = min(1.0 - float(ht[i]), float(ht[j]))
    if hi <= lo:
        return None
    delta = 0.5 * (lo + hi) + 0.8 * (float(rng.rand()) - 0.5) * (hi - lo)
    ht[i] += delta
    ht[j] -= delta
    return fast_project_box_sum(ht)


def gradient_direction_polish(h, n, max_iters, rng):
    h = np.asarray(h, dtype=float).ravel()
    n = int(n)
    if n < 2 or max_iters <= 0:
        return project_box_sum(h), compute_c5_fast(h)
    h = project_box_sum(h)
    best_h = h.copy()
    best_c = compute_c5_fast(h)
    fail = 0
    alphas = [1e-1, 5e-2, 3e-2, 1e-2, 5e-3, 3e-3, 1e-3]
    for it in range(int(max_iters)):
        if fail > 45:
            break
        if it < max_iters // 3:
            beta = 20.0
        elif it < 2 * max_iters // 3:
            beta = 80.0
        else:
            beta = 250.0
        max_active = None if n <= 160 else 32
        _, g = smooth_value_grad(h, beta, max_active)
        g = g - float(np.mean(g))
        gmax = float(np.max(np.abs(g)))
        if gmax < 1e-12:
            break
        d = -g / gmax
        improved = False
        for alpha in alphas:
            cand = fast_project_box_sum(h + alpha * d)
            c = compute_c5_fast(cand)
            if c < best_c - 1e-12:
                h = cand
                best_h = cand.copy()
                best_c = c
                fail = 0
                improved = True
                break
        if not improved:
            fail += 1
            if fail % 5 == 0:
                cand = random_pair_move(best_h, n, rng)
                if cand is not None:
                    c = compute_c5_fast(cand)
                    if c < best_c - 1e-12:
                        h = cand.copy()
                        best_h = cand.copy()
                        best_c = c
                        fail = 0
            if fail > 45:
                break
    return best_h, float(best_c)


def random_perturb_search(h, n, n_trials, rng, step=0.05):
    h = np.asarray(h, dtype=float).ravel()
    n = int(n)
    if n < 2 or n_trials <= 0:
        return project_box_sum(h), compute_c5_fast(h)
    h = project_box_sum(h)
    best_h = h.copy()
    best_c = compute_c5_fast(h)
    n_trials = int(n_trials)
    for trial in range(n_trials):
        cur_step = step / (1.0 + 0.02 * float(trial // 100))
        ht = best_h.copy()
        k = int(rng.randint(1, 5))
        k = min(k, n)
        idx = rng.choice(n, k, replace=False)
        ht[idx] += cur_step * (rng.rand(k) - 0.5)
        ht = fast_project_box_sum(ht)
        c = compute_c5_fast(ht)
        if c < best_c - 1e-12:
            best_c = c
            best_h = ht.copy()
    return best_h, float(best_c)


def direct_perturb_search(h, n, n_trials, rng, step=0.02):
    h = np.asarray(h, dtype=float).ravel()
    n = int(n)
    if n < 2 or n_trials <= 0:
        return project_box_sum(h), compute_c5_direct(h)
    h = project_box_sum(h)
    best_h = h.copy()
    best_c = compute_c5_direct(h)
    n_trials = int(n_trials)
    for trial in range(n_trials):
        ht = best_h.copy()
        k = int(rng.randint(1, 4))
        k = min(k, n)
        idx = rng.choice(n, k, replace=False)
        ht[idx] += step * (rng.rand(k) - 0.5) / (1.0 + 0.02 * float(trial // 50))
        ht = project_box_sum(ht)
        c = compute_c5_direct(ht)
        if c < best_c - 1e-12:
            best_c = c
            best_h = ht.copy()
    return best_h, float(best_c)


def local_refine(h, n, rng, runtime):
    h = np.asarray(h, dtype=float).ravel()
    n = int(h.size)
    if n < 2:
        return h, compute_c5_fast(h)
    h = project_box_sum(h)
    best_h = h.copy()
    best_c = compute_c5_fast(h)
    t0 = time.time()
    runtime = float(max(0.0, runtime))
    if runtime <= 0.0:
        return best_h, float(best_c)

    if n <= 192 and runtime > 25.0:
        slq_budget = min(runtime * 0.35, 90.0)
        betas = [20.0, 80.0, 300.0]
        for beta in betas:
            if time.time() - t0 > slq_budget:
                break
            maxiter = int(max(30, min(140, slq_budget * 1.5)))
            h2, c2 = slsqp_smooth(h, n, beta, maxiter)
            if c2 < best_c - 1e-12:
                best_c = c2
                best_h = h2.copy()
                h = h2.copy()

    elapsed = time.time() - t0
    grad_alloc = max(0.0, runtime * 0.45 - elapsed)
    if grad_alloc > 5.0:
        iters = int(max(60, min(3000, grad_alloc * 7.0)))
        h2, c2 = gradient_direction_polish(h, n, iters, rng)
        if c2 < best_c - 1e-12:
            best_c = c2
            best_h = h2.copy()
            h = h2.copy()

    elapsed = time.time() - t0
    rem = max(0.0, runtime - elapsed)
    if rem > 3.0:
        if n <= 256:
            trials = int(min(5000, max(150, rem * 30.0)))
            step = 0.06
        elif n <= 640:
            trials = int(min(1600, max(60, rem * 10.0)))
            step = 0.04
        else:
            trials = int(min(500, max(20, rem * 3.0)))
            step = 0.025
        h2, c2 = random_perturb_search(h, n, trials, rng, step)
        if c2 < best_c - 1e-12:
            best_c = c2
            best_h = h2.copy()
            h = h2.copy()

    elapsed = time.time() - t0
    rem = max(0.0, runtime - elapsed)
    if rem > 3.0 and n <= 220:
        trials = int(min(900, max(60, rem * 12.0)))
        h2, c2 = direct_perturb_search(h, n, trials, rng, 0.018)
        if c2 < best_c - 1e-12:
            best_c = c2
            best_h = h2.copy()
            h = h2.copy()

    return best_h, float(best_c)


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    deadline = time.time() + float(budget_s)

    parent = None
    try:
        p = initial_h_values
        if p is not None:
            p = np.asarray(p, dtype=float).ravel()
            if p.size > 0:
                parent = project_box_sum(p)
    except Exception:
        parent = None

    if parent is not None:
        if parent.size % 2 == 1:
            best_h = interpolate_to_n(parent, parent.size + 1)
        else:
            best_h = parent.copy()
        best_n = int(best_h.size)
    else:
        best_n = 128
        cands = make_initial_guesses(best_n, rng, None, 8)
        best_h, _ = choose_quick_best(cands)
        best_n = int(best_h.size)

    best_c = compute_c5_fast(best_h)

    rem = deadline - time.time()
    if rem > 120.0:
        rt = min(260.0, rem * 0.25)
        h2, c2 = local_refine(best_h, best_n, rng, rt)
        if c2 < best_c:
            best_h = h2
            best_c = c2
            best_n = int(h2.size)

    low_budget = min(320.0, max(0.0, (deadline - time.time()) * 0.32))
    low_start = time.time()
    if low_budget > 60.0:
        low_ns = []
        for nn in (64, 96, 128, 160):
            if nn % 2:
                nn += 1
            if nn not in low_ns:
                low_ns.append(nn)
        for n in low_ns:
            if time.time() - low_start >= low_budget:
                break
            cands = make_initial_guesses(n, rng, best_h, 7)
            scored = scored_candidates(cands)
            for idx in range(min(2, len(scored))):
                if time.time() - low_start >= low_budget:
                    break
                rt = min(55.0, max(10.0, low_budget - (time.time() - low_start)))
                h2, c2 = local_refine(scored[1][idx], n, rng, rt)
                if c2 < best_c:
                    best_h = h2
                    best_c = c2
                    best_n = n

    high_budget = min(480.0, max(0.0, (deadline - time.time()) * 0.45))
    high_start = time.time()
    if high_budget > 100.0:
        high_ns = []
        for f in (2, 3, 4):
            nn = int(best_n) * int(f)
            if nn % 2:
                nn += 1
            if nn > best_n and nn <= 1024:
                high_ns.append(nn)
        for base in (256, 320, 512, 640, 768, 1024):
            if base > best_n and base <= 1024:
                high_ns.append(base)
        high_ns = sorted(set(high_ns))
        for n in high_ns[:3]:
            if time.time() - high_start >= high_budget:
                break
            cands = []
            cands.append(interpolate_to_n(best_h, n))
            cands.append(upsample_repeat(best_h, n))
            if parent is not None:
                cands.append(interpolate_to_n(parent, n))
                cands.append(upsample_repeat(parent, n))
            h0, c0 = choose_quick_best(cands)
            rt = min(280.0, max(50.0, high_budget - (time.time() - high_start)))
            h2, c2 = local_refine(h0, n, rng, rt)
            if c2 < best_c:
                best_h = h2
                best_c = c2
                best_n = int(h2.size)

    rem = deadline - time.time()
    if rem > 20.0:
        rt = min(220.0, rem * 0.55)
        h2, c2 = local_refine(best_h, best_n, rng, rt)
        if c2 < best_c:
            best_h = h2
            best_c = c2
            best_n = int(h2.size)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h)
    best_h = enforce_exact_sum(best_h)
    best_c_direct = compute_c5_direct(best_h)

    if best_c_direct > 0.38085 and best_n <= 230 and deadline - time.time() > 8.0:
        trials = int(min(900, max(50, (deadline - time.time()) * 8)))
        h2, c2 = direct_perturb_search(best_h, best_n, trials, rng, 0.012)
        if c2 < best_c_direct:
            best_h = h2
            best_c_direct = c2

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h)
    best_h = enforce_exact_sum(best_h)
    final_c = compute_c5_direct(best_h)
    return (best_h, float(final_c), int(best_h.size))
