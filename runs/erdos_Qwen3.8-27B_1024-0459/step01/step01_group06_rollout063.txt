                    best_h = h_new
                    best_c5 = c5_new
            except Exception:
                pass

    # Phase 7: Structured restarts (oscillatory patterns)
    if remaining() > 40:
        for freq in [3, 5, 7, 11]:
            if remaining() < 20:
                break
            x = np.arange(n_use, dtype=float) / n_use * 2.0
            h_struc = 0.5 + 0.45 * np.cos(np.pi * freq * x)
            h_struc = np.clip(h_struc, 0.0, 1.0)
            h_struc = project_feasible(h_struc, n_use)
            try:
                res = minimize(
                    smooth_max_obj, h_struc, jac=smooth_max_grad,
                    args=(n_use, 400.0), method='SLSQP', bounds=bounds_use,
                    constraints=cons_use, options={'maxiter': 400, 'ftol': 1e-12}
                )
                h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n_use)
                c5_new = compute_c5(h_new, n_use)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
            except Exception:
                pass

    # Phase 8: Final hard-max polish
    if remaining() > 15:
        try:
            res = minimize(
                hard_max_obj, best_h, jac=hard_max_grad,
                args=(best_n,), method='SLSQP', bounds=bounds_use,
                constraints=cons_use, options={'maxiter': 1000, 'ftol': 1e-15}
            )
            h_final = project_feasible(np.clip(res.x, 0.0, 1.0), best_n)
            c5_final = compute_c5(h_final, best_n)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
        except Exception:
            pass

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    # Validate with evaluator
    try:
        val = evaluate_erdos_solution(best_h, best_n)
        if val is not None and 0.0 < float(val) < 1.0:
            best_c5 = min(best_c5, float(val))
    except Exception:
        pass

    return (best_h, best_c5, best_n)
