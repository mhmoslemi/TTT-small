                best = val0
                for d in range(1, n):
                    vl = a_left[d-1] * t + (scores_cur[2*d-1] - old_vi * a_left[d-1])
                    vr = a_right[d-1] * t + (scores_cur[2*d] - old_vi * a_right[d-1])
                    if vl > best:
                        best = vl
                    if vr > best:
                        best = vr
                return best
            
            # Ternary search on [0, 1]
            lo, hi = 0.0, 1.0
            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if f_of_t(m1) < f_of_t(m2):
                    hi = m2
                else:
                    lo = m1
            t_opt = 0.5 * (lo + hi)
            
            # Round to avoid numerical issues
            if t_opt < 1e-10:
                t_opt = 0.0
            elif t_opt > 1.0 - 1e-10:
                t_opt = 1.0
            
            if abs(t_opt - old_vi) > tol:
                h[i] = t_opt
                improved = True
        
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, n)
        c = float(np.max(compute_scores_raw(h, n)) * dx)
        if c < best_c:
            best_c = c
        if not improved:
            break
    return h, best_c


def projected_gradient(h, n, max_iter, deadline, momentum=0.9):
    """Projected gradient with momentum and adaptive step."""
    h = np.array(h, dtype=float, copy=True)
    h = project_feasible(h, n)
    dx = 2.0 / n
    scores = compute_scores_raw(h, n)
    m = float(np.max(scores))
    best_c = m * dx
    best_h = h.copy()
    
    v = np.zeros(n)
    step = 0.1
    stall = 0
    
    for it in range(max_iter):
        check_deadline()
        if time.time() > deadline:
            break
        
        m = float(np.max(scores))
        
        # Compute subgradient (active set)
        tol = max(1e-6 * m, 1e-8)
        active = np.where(scores >= m - tol)[0]
        if len(active) > 20:
            # Sample if too many active
            idx_sample = np.random.choice(len(active), size=20, replace=False)
            active = active[idx_sample]
        
        grad = np.zeros(n)
        for idx in active:
            grad += grad_single(h, n, idx)
        grad /= float(len(active))
        grad *= dx
        
        # Momentum
        v = momentum * v - step * grad
        h_new = h + v
        h_new = project_feasible(h_new, n)
        
        # Evaluate
        scores_new = compute_scores_raw(h_new, n)
        m_new = float(np.max(scores_new))
        c_new = m_new * dx
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0
        elif c_new < m * dx + step * np.dot(grad, grad) * 0.5:
            # Accept by Wolfe-like condition (approximate)
            pass
        else:
            step *= 0.5
            if step < 1e-8:
                step = 0.1
                v = np.zeros(n)
            h_new = h + 0.5 * v
            h_new = project_feasible(h_new, n)
            scores_new = compute_scores_raw(h_new, n)
            m_new = float(np.max(scores_new))
            c_new = m_new * dx
            if c_new > best_c * 1.001:
                # Reject, try smaller step
                step *= 0.5
                continue
        
        if c_new <= m * dx:
            h = h_new
            scores = scores_new
            m = m_new
            stall = 0
            if it % 200 == 0 and step < 0.5:
                step = min(step * 1.1, 0.5)
        else:
            stall += 1
            if stall > 500:
                # Restart momentum
                v = np.zeros(n)
                step = 0.1
                stall = 0
        
        # Periodic nudge to escape flat regions
        if it % 500 == 499 and stall > 100:
            h[::7] += 0.01 * np.random.randn(n // 7 + 1)[:h[::7].size] * 0.5
            h = project_feasible(h, n)
    
    return project_feasible(best_h, n), best_c


def slsqp_polish(h, n, deadline, maxiter=80):
    """SLSQP polish with log-sum-exp."""
    h = np.array(h, dtype=float, copy=True)
    n = int(n)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x, alpha):
        x = np.clip(x, 0.0, 1.0)
        sc = compute_scores_raw(x, n)
        m = float(np.max(sc))
        if alpha > 1e-14:
            w = np.exp((sc - m) / alpha)
            val = m + alpha * float(np.sum(w)) # wrong, should be log(sum(exp))
            # Redo:
            w = np.exp((sc - m) / alpha)
            A = float(np.sum(w))
            if math.isfinite(A) and A > 0:
                val = m + alpha * math.log(A)
            else:
                val = m
        else:
            val = m
        return val * (2.0 / n)
    
    def obj_grad(x, alpha):
        x = np.clip(x, 0.0, 1.0)
        sc = compute_scores_raw(x, n)
        m = float(np.max(sc))
        grad = np.zeros(n)
        if alpha > 1e-14:
            w = np.exp((sc - m) / alpha)
            A = float(np.sum(w))
            if math.isfinite(A) and A > 0:
                thresh = 1e-12
                for i in range(len(sc)):
                    c_i = w[i] / A
                    if c_i >= thresh:
                        grad += c_i * grad_single(x, n, i)
        else:
            tol = max(1e-7, 1e-6 * m)
            active = np.where(sc >= m - tol)[0]
            if len(active) == 0:
                active = np.array([int(np.argmax(sc))])
            for i in active:
                grad += grad_single(x, n, i) / float(len(active))
        grad *= (2.0 / n)
        return grad
    
    def con_eq(x, *args):
        return (float(np.sum(x)) - 0.5 * n) / n
    
    def con_jac(x, *args):
        return np.full((1, n), 1.0 / n)
    
    constraints = [{"type": "eq", "fun": con_eq, "jac": con_jac}]
    
    alpha = max(0.1, 0.003 * n)
    
    old_deadline = _RUN_DEADLINE
    _RUN_DEADLINE = deadline
    try:
        res = minimize(obj, h, args=(alpha,), jac=lambda x: obj_grad(x, alpha),
                       method="SLSQP", bounds=bounds, constraints=constraints,
                       options={"maxiter": maxiter, "ftol": 1e-10, "disp": 0})
    except Exception:
        res = None
    finally:
        _RUN_DEADLINE = old_deadline
    
    if res is not None and res.x is not None:
        h_new = project_feasible(res.x, n)
        c = compute_c5(h_new)
        c_old = compute_c5(h)
        if c < c_old:
            return h_new, c
    return h, compute_c5(h)


def get_parent():
    try:
        p = np.asarray(initial_h_values, dtype=float)
        if p.ndim != 1 or p.size < 2:
            return None
        return p.copy()
    except Exception:
        return None


def synthetic_init(n, rng):
    """Create a diverse synthetic initialization."""
    x = (np.arange(n) + 0.5) * (2.0 / n)
    candidates = []
    # Cosine patterns
    for f in (1, 2, 3, 4, 5, 6, 7, 8):
        for ph in (0.0, 0.5 * math.pi, 0.25 * math.pi, 0.75 * math.pi):
            h = 0.5 + 0.45 * np.cos(f * math.pi * x + ph)
            candidates.append(h)
    # Polynomial
    for c in (-0.3, -0.5, 0.3, 0.5):
        h = 0.5 + c * (x - 1.0) ** 2
        candidates.append(h)
    # Smooth random
    freqs = rng.integers(1, 6, size=4)
    phases = rng.uniform(0, 2 * math.pi, size=4)
    amps = rng.uniform(0.1, 0.3, size=4)
    h = 0.5 + sum(a * np.cos(f * math.pi * x + p) for a, f, p in zip(amps, freqs, phases))
    candidates.append(h)
    
    best_h = np.full(n, 0.5)
    best_c = 1e9
    for hc in candidates:
        hf = project_feasible(hc, n)
        c = compute_c5(hf)
        if c < best_c:
            best_c = c
            best_h = hf
    return best_h


class State:
    def __init__(self, h, c, n):
        self.h = np.array(h, dtype=float)
        self.c = float(c)
        self.n = int(n)
    
    def try_update(self, h, n):
        if h is None:
            return
        h = np.asarray(h, dtype=float)
        if h.ndim != 1 or h.size != n:
            return
        hf = project_feasible(h, n)
        c = compute_c5(hf)
        if math.isfinite(c) and c < self.c - 1e-14:
            self.h = hf
            self.c = c
            self.n = n


def run(seed=42, budget_s=1300, **kwargs):
    global _RUN_DEADLINE
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 10:
        budget_s = 30.0
    reserve = 30.0
    deadline = t0 + max(5.0, budget_s - reserve)
    _RUN_DEADLINE = deadline
    
    rng = np.random.default_rng(seed)
    
    parent = get_parent()
    if parent is not None:
        n_parent = int(parent.size)
        h_parent = project_feasible(parent, n_parent)
        c_parent = compute_c5(h_parent)
    else:
        n_parent = 70
        h_parent = synthetic_init(n_parent, rng)
        c_parent = compute_c5(h_parent)
    
    state = State(h_parent.copy(), c_parent, n_parent)
    
    # Resolutions to try
    res_list = [n_parent]
    for mult in [1.2, 1.5, 2.0]:
        nn = int(round(mult * n_parent / 2.0) * 2)
        if nn >= 4 and nn <= 4 * n_parent and nn not in res_list:
            res_list.append(nn)
    res_list.sort()
    
    # Time allocation per resolution
    total_opt_time = max(50.0, budget_s - reserve - 60.0)
    # Weight: more time for parent resolution, less for larger
    weights = []
    for nn in res_list:
        w = 1.0 / (1.0 + 0.01 * (nn - n_parent))
        weights.append(w)
    wsum = sum(weights)
    time_alloc = {res_list[i]: total_opt_time * weights[i] / wsum for i in range(len(res_list))}
    
    for n in res_list:
        if time.time() > deadline - 60:
            break
        alloc = min(time_alloc[n], max(20.0, deadline - time.time() - 30.0))
        if alloc < 25:
            break
        
        # Get starting points
        if n == n_parent:
            starts = [h_parent.copy()]
        else:
            starts = [upsample_linear(h_parent, n)]
            starts.append(upsample_linear(state.h, n))
            if len(starts) < 3:
                starts.append(synthetic_init(n, rng))
        
        # Remove duplicates
        unique_starts = []
        for s in starts:
            sf = project_feasible(s, n)
            dup = False
            for u in unique_starts:
                if np.allclose(sf, u, atol=1e-6):
                    dup = True
                    break
            if not dup:
                unique_starts.append(sf)
        
        for h_start in unique_starts:
            if time.time() > deadline - 40:
                break
            remaining = deadline - time.time()
            if remaining < 30:
                break
            
            h_cur = h_start.copy()
            c_cur = compute_c5(h_cur)
            state.try_update(h_cur, n)
            
            # Phase 1: Coordinate descent
            cd_time = min(alloc * 0.3, max(10.0, remaining * 0.3))
            cd_deadline = time.time() + cd_time
            if cd_time > 10:
                try:
                    h_cd, c_cd = coord_descent(h_cur, n, max_passes=int(cd_time * 8),
                                               deadline=cd_deadline)
                    state.try_update(h_cd, n)
                    if compute_c5(h_cd) < c_cur:
                        h_cur = h_cd
                        c_cur = compute_c5(h_cd)
                except (BudgetError, Exception):
                    pass
            
            # Phase 2: Projected gradient
            pg_time = min(alloc * 0.4, max(10.0, remaining * 0.4 - 20))
            pg_deadline = time.time() + pg_time
            if pg_time > 10 and time.time() < deadline - 50:
                try:
                    h_pg, c_pg = projected_gradient(h_cur, n,
                                                    max_iter=int(pg_time * 200),
                                                    deadline=pg_deadline)
                    state.try_update(h_pg, n)
                    if compute_c5(h_pg) < c_cur:
                        h_cur = h_pg
                        c_cur = compute_c5(h_pg)
                except (BudgetError, Exception):
                    pass
            
            # Phase 3: SLSQP polish
            sl_time = min(alloc * 0.3, max(5.0, remaining * 0.25 - 20))
            if sl_time > 10 and time.time() < deadline - 35:
                try:
                    h_sl, c_sl = slsqp_polish(h_cur, n, deadline=time.time() + sl_time,
                                               maxiter=int(sl_time * 2))
                    state.try_update(h_sl, n)
                except (BudgetError, Exception):
                    pass
            
            # Update best
            if state.n == n:
                pass
    
    # Final validation
    h_final = project_feasible(state.h, state.n)
    c_final = compute_c5(h_final)
    n_final = int(state.n)
    
    # Also check parent
    c_parent_check = compute_c5(h_parent)
    if c_parent_check < c_final:
        h_final = h_parent
        c_final = c_parent_check
        n_final = n_parent
    
    return h_final, float(c_final), int(n_final)
