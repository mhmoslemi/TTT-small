```python
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr

def subgradient_pos(h, k):
    """Subgradient of positive-lag correlation (index n-1+k in full array)."""
    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        return 1.0 - 2.0 * h
    grad[:k] = -h[k:]
    if n - 2 * k > 0:
        grad[k:n-k] = (1.0 - h[:n-2*k]) - h[2*k:]
    if k > 0 and n - k < n:
        grad[n-k:] = 1.0 - h[n-2*k:n-k]
    return grad

def subgradient_neg(h, k):
    """Subgradient of negative-lag correlation (index n-1-k in full array)."""
    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        return 1.0 - 2.0 * h
    if k < n:
        grad[:k] = 1.0 - h[k:2*k]
    if n - 2 * k > 0:
        grad[k:n-k] = (1.0 - h[2*k:]) - h[:n-2*k]
    if k > 0:
        grad[n-k:] = -h[n-2*k:n-k]
    return grad

def fix_sum(h, n):
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:

        else:

        if len(idx) == 0:
            break
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h

    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)

    corr = np.correlate(h, 1.0 - h, mode="full")

    """Optimize using projected subgradient with momentum."""
    h = h0.copy()
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for t in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        max_val = np.max(corr)
        min_val = np.min(corr)  # not needed but for info
        
        # Find indices in top 1% of correlations
        threshold = max_val - 0.01 * np.max(np.abs(corr))
        top_mask = corr >= threshold
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        n_idx = n - 1
        for idx in np.where(top_mask)[0]:
            w = (corr[idx] - threshold) ** 2 + 0.01
            shift = idx - n_idx
            if shift >= 0:
                sg = subgradient_pos(h, shift)
            else:
                sg = subgradient_neg(h, -shift)
            grad += w * sg
        
        grad_mean = grad.mean()
        grad = grad - grad_mean  # center to preserve sum
        
        # Step size
        alpha = alpha0 / (1.0 + 0.01 * t)
        
        # Momentum update
        v = momentum * v + (1 - momentum) * alpha * grad
        h_new = h - v
        
        # Project
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = fix_sum(h_new, n)
        
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
```

Actually, you know what, let me try a completely different approach that's much simpler and potentially more effective:

**Alternating Direction Method / Bisection on target T:**

1. Set a target T (initially the current C₅ value)
2. Solve: minimize sum_k max(0, corr_k - T)² subject to constraints
   This is a smooth problem (piecewise quadratic) that can be solved with standard methods
3. If the solution has all corr_k ≤ T, decrease T and repeat
4. If not, increase T and repeat

This is like a barrier method. The advantage is that at each step, we're solving a smooth convex problem (or at least a well-behaved non-convex problem) that standard optimizers handle well.

```python
def compute_smooth_max_grad(h, tau, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    weights = np.exp((corr - m) / tau)
    weights /= weights.sum()
    
    n_idx = n - 1
    grad = np.zeros(n)
    
    # Zero shift
    w0 = weights[n_idx]
    grad += w0 * (1 - 2*h)
    
    # For efficiency, only process significant weights
    significant = np.where(weights > 1e-10)[0]
    
    for idx in significant:
        w = weights[idx]
        shift = idx - n_idx
        if shift > 0:
            # positive lag
            grad[:shift] += -w * h[shift:]
            mid_start = shift
            mid_end = n - shift
            if mid_end > mid_start:
                grad[mid_start:mid_end] += w * ((1 - h[:n-2*shift]) - h[2*shift:])
            if n - shift < n:
                grad[n-shift:] += w * (1 - h[n-2*shift:-shift])
        elif shift < 0:
            s = -shift
            if s < n:
                grad[:s] += w * (1 - h[s:2*s])
            mid_start = s
            mid_end = n - s
            if mid_end > mid_start:
                grad[mid_start:mid_end] += w * ((1 - h[2*s:]) - h[:n-2*s])
            if s > 0:
                grad[n-s:] += -w * h[n-2*s:n-s]
    
    return grad * dx
```

This should work. The number of significant weights depends on tau: small tau means only a few weights are significant (close to the max), making it fast.

import numpy as np
from scipy.optimize import minimize
import time
import functools

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h

def upsample_to(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def thue_morse_init(n):
    """Create Thue-Morse inspired initialization."""
    # Generate Thue-Morse sequence
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    h = tm.astype(float)
    h = fix_sum(h, n)
    return h

def periodic_init(n, period, duty):
    """Create periodic initialization with given period and duty cycle."""
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = ((x % period) / period < duty).astype(float)
    h = fix_sum(h, n)
    return h

def smooth_max_value_and_weights(h, tau):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    weights = np.exp((corr - m) / tau)
    weights /= weights.sum()
    return m, weights, corr

def smooth_max_grad_vec(h, tau, dx):
    n = len(h)
    m, weights, corr = smooth_max_value_and_weights(h, tau)
    n_idx = n - 1
    grad = np.zeros(n)
    
    # Only process weights above threshold
    thresh = 1e-10
    significant = np.where(weights > thresh)[0]
    
    for idx in significant:
        w = weights[idx]
        shift = idx - n_idx
        if shift == 0:
            grad += w * (1.0 - 2.0 * h)
        elif shift > 0:
            if shift < n:
                grad[:shift] += -w * h[shift:]
                mid_len = n - 2 * shift
                if mid_len > 0:
                    grad[shift:shift+mid_len] += w * ((1.0 - h[:mid_len]) - h[2*shift:2*shift+mid_len])
                if n - shift < n:
                    tail_len = shift
                    grad[n-shift:] += w * (1.0 - h[n-2*shift:n-shift])
        else:
            s = -shift
            if s < n:
                head_len = min(s, n)
                grad[:head_len] += w * (1.0 - h[s:s+head_len])
                mid_len = n - 2 * s
                if mid_len > 0:
                    grad[s:s+mid_len] += w * ((1.0 - h[2*s:2*s+mid_len]) - h[:mid_len])
                if s > 0 and n - s < n:
                    tail_len = min(s, n)
                    grad[n-s:] += -w * h[n-2*s:n-s]
    
    return (m + tau * np.log(np.sum(np.exp((corr - m) / tau)))) * dx, grad * dx

def smooth_max_obj(x, n, dx, tau):
    h = np.clip(x, 0.0, 1.0)
    val, _ = smooth_max_grad_vec(h, tau, dx)
    return val

def smooth_max_grad_obj(x, n, dx, tau):
    h = np.clip(x, 0.0, 1.0)
    _, g = smooth_max_grad_vec(h, tau, dx)
    return g

def smooth_max_obj_pen(x, n, dx, tau, mu):
    h = np.clip(x, 0.0, 1.0)
    val, _ = smooth_max_grad_vec(h, tau, dx)
    penalty = mu * (h.sum() - n/2.0)**2
    return val + penalty

def smooth_max_grad_pen(x, n, dx, tau, mu):
    h = np.clip(x, 0.0, 1.0)
    _, g = smooth_max_grad_vec(h, tau, dx)
    g = g + 2.0 * mu * (h.sum() - n/2.0)
    return g

                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-8}

            h_new = np.clip(res.x, 0.0, 1.0)

            c5_new = compute_c5(h_new)
