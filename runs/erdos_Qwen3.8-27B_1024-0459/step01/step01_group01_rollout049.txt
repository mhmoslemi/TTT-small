    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None and np.size(initial_h_values) > 0:
        parent_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        parent_h = np.full(180, 0.5)

    n_parent = len(parent_h)
    best_h = fix_sum(parent_h, n_parent)
    best_c5 = compute_c5(best_h)

    # ============ Phase 1: Binary 2-swap SA with directed moves ============
    n = n_parent
    h_bin = binary_from_continuous(best_h, rng)
    if h_bin.sum() != n / 2:
        while h_bin.sum() < n / 2:
            z = np.where(h_bin == 0)[0]
            if len(z) == 0:
                break
            h_bin[rng.integers(len(z))] = 1
        while h_bin.sum() > n / 2:
            o = np.where(h_bin == 1)[0]
            if len(o) == 0:
                break
            h_bin[rng.integers(len(o))] = 0

    best_bin_h = h_bin.copy()
    best_bin_c5 = compute_c5(h_bin)
    if best_bin_c5 < best_c5:
        best_c5 = best_bin_c5
        best_h = h_bin.copy()

    h_cur = h_bin.copy()
    c5_cur = best_bin_c5
    T = 0.004
    T_min = 1e-9
    cool = 0.9965
    phase1_end = t0 + budget_s * 0.35

    while time.time() < phase1_end and T > T_min:
        # Directed vs random move
        if rng.random() < 0.45:
            corr_arr = get_corr_array(h_cur)
            worst_idx = np.argmax(corr_arr)
            shift = worst_idx - (n - 1)
            found = False
            if shift > 0 and shift < n:
                for _ in range(20):
                    i = rng.integers(0, n - shift)
                    if h_cur[i] == 1 and h_cur[i + shift] == 0:
                        zeros = np.where(h_cur == 0)[0]
                        j = zeros[rng.integers(len(zeros))]
                        found = True
                        break
            elif shift < 0 and abs(shift) < n:
                k = -shift
                for _ in range(20):
                    i = rng.integers(0, n - k)
                    if h_cur[i] == 0 and h_cur[i + k] == 1:
                        ones = np.where(h_cur == 1)[0]
                        j = ones[rng.integers(len(ones))]
                        found = True
                        break
            if not found:
                ones = np.where(h_cur == 1)[0]
                zeros = np.where(h_cur == 0)[0]
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
        else:
            ones = np.where(h_cur == 1)[0]
            zeros = np.where(h_cur == 0)[0]
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]

        h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
        c5_new = compute_c5(h_cur)
        delta = c5_new - c5_cur
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            c5_cur = c5_new
            if c5_new < best_bin_c5:
                best_bin_c5 = c5_new
                best_bin_h = h_cur.copy()
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_cur.copy()
        else:
            h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
        T *= cool

    # ============ Phase 2: Continuous optimization (L-BFGS-B with penalty) ============
    n_resolutions = [n, 2 * n]
    if time.time() - t0 < t0 + budget_s * 0.70:
        n_resolutions.append(3 * n)

    start_points = [best_h.copy(), best_bin_h.copy()]
    # Add a smoothed version
    kernel = np.array([1, 2, 3, 2, 1], dtype=float)
    kernel /= kernel.sum()
    h_s = np.convolve(best_h, kernel, mode='same')
    h_s = np.clip(h_s, 0, 1)
    h_s = fix_sum(h_s, n)
    start_points.append(h_s)

    phase2_end = t0 + budget_s * 0.88

    for n_opt in n_resolutions:
        if time.time() > phase2_end:
            break
        dx_opt = 2.0 / n_opt

        for sp in start_points:
            if time.time() > phase2_end:
                break
            if len(sp) == n_opt:
                x0 = sp.copy()
            elif len(sp) < n_opt:
                x0 = upsample(sp, n_opt)
            else:
                x0 = downsample(sp, n_opt)
            x0 = np.clip(x0, 0, 1)
            x0 = fix_sum(x0, n_opt)

            for tau in [0.05, 0.025, 0.012, 0.006, 0.003, 0.001]:
                if time.time() > phase2_end:
                    break

                def obj_fn(x, _tau=tau, _dx=dx_opt, _n=n_opt):
                    h = np.clip(x, 0, 1)
                    corr = np.correlate(h, 1.0 - h, mode="full")
                    m = np.max(corr)
                    z = (corr - m) / _tau
                    z = np.clip(z, -60, 0)
                    val = (m + _tau * np.log(np.sum(np.exp(z)))) * _dx
                    sv = h.sum() - _n / 2.0
                    val += 1e8 * sv * sv
                    return val

                try:
                    maxiter = 3000 if n_opt <= 360 else 1500
                    res = minimize(
                        obj_fn, x0, method='L-BFGS-B',
                        bounds=[(0.0, 1.0)] * n_opt,
                        options={'maxiter': maxiter, 'ftol': 1e-16,
                                 'maxfun': maxiter * 5, 'gtol': 1e-12,
                                 'iprint': -1}
                    )
                    h_new = np.clip(res.x, 0, 1)
                    h_new = fix_sum(h_new, n_opt)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                    x0 = h_new
                except Exception:
                    pass

    # ============ Phase 3: Fine polish at best resolution ============
    phase3_end = t0 + budget_s * 0.99
    n_f = len(best_h)
    dx_f = 2.0 / n_f

    trial = 0
    while time.time() < phase3_end:
        trial += 1
        scale = 0.003 + 0.015 * max(0, 1.0 - trial / 40.0)
        x_pert = np.clip(best_h + rng.normal(0, scale, n_f), 0, 1)
        x_pert = fix_sum(x_pert, n_f)

        def obj_fine(x, _scale=scale):
            h = np.clip(x, 0, 1)
            corr = np.correlate(h, 1.0 - h, mode="full")
            m = np.max(corr)
            tau_f = 0.0005
            z = (corr - m) / tau_f
            z = np.clip(z, -60, 0)
            val = (m + tau_f * np.log(np.sum(np.exp(z)))) * dx_f
            sv = h.sum() - n_f / 2.0
            val += 1e8 * sv * sv
            return val

        try:
            res = minimize(
                obj_fine, x_pert, method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n_f,
                options={'maxiter': 800, 'ftol': 1e-16,
                         'maxfun': 4000, 'gtol': 1e-12, 'iprint': -1}
            )
            h_new = np.clip(res.x, 0, 1)
            h_new = fix_sum(h_new, n_f)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass

    # ============ Final verification ============
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
