            res_list.append(candidate)
    res_list.sort()

    for n in res_list:
        if tleft() < 60:
            break

        h_interp = make_feasible(upsample(best_h, n), n)

        # SLSQP at this resolution
        if tleft() > 40:
            h_o, c5_o = slqp_optimize(h_interp, n, max_iter=5000, ftol=1e-18)
            if h_o is not None and c5_o < best_c5:
                best_c5, best_h, best_n_final = c5_o, h_o, n

        # Smooth max
        if tleft() > 40:
            h_start = make_feasible(upsample(best_h, n), n)
            h_o, c5_o = smooth_max_optimize(h_start, n, [20, 60, 150, 400, 1000])
            if c5_o < best_c5:
                best_c5, best_h, best_n_final = c5_o, h_o, n

        # Try a structured initialization at this resolution
        if tleft() > 50:
            for ptype in [0, 2]:
                h_struct = generate_binary_init(n, seed + ptype * 1000, pattern_type=ptype)
                h_o, c5_o = smooth_max_optimize(h_struct, n, [30, 100, 300])
                if c5_o < best_c5:
                    best_c5, best_h, best_n_final = c5_o, h_o, n
                    break

        # Random perturbation
        if tleft() > 25:
            h_start = make_feasible(upsample(best_h, n), n)
            nt = min(2000, int(tleft() / 0.003))
            h_o, c5_o = random_perturb_search(h_start, n, nt, 0.005, seed + n)
            if c5_o < best_c5:
                best_c5, best_h, best_n_final = c5_o, h_o, n

    # Phase 3: Coordinate local search at best resolution
    if tleft() > 40:
        nr = min(200, int(tleft() / 0.02))
        h_o, c5_o = coord_local_search(best_h, best_n_final, nr, seed + 500)
        if c5_o < best_c5:
            best_c5, best_h = c5_o, h_o

    # Phase 4: Final SLSQP polish (very tight)
    if tleft() > 30:
        h_o, c5_o = slqp_optimize(best_h, best_n_final, max_iter=10000, ftol=1e-20)
        if h_o is not None and c5_o < best_c5:
            best_c5, best_h = c5_o, h_o

    # Phase 5: Extended random perturbation
    if tleft() > 20:
        nt = min(8000, int((tleft() - 10) / 0.002))
        h_o, c5_o = random_perturb_search(best_h, best_n_final, nt, 0.003, seed + 999)
        if c5_o < best_c5:
            best_c5, best_h = c5_o, h_o

    # Phase 6: Try one higher resolution if time permits
    if tleft() > 60:
        n_new = best_n_final + 40 if best_n_final < 400 else best_n_final + 20
        h_up = make_feasible(upsample(best_h, n_new), n_new)
        h_o, c5_o = slqp_optimize(h_up, n_new, max_iter=3000, ftol=1e-18)
        if h_o is not None and c5_o < best_c5:
            best_c5, best_h, best_n_final = c5_o, h_o, n_new
            if tleft() > 20:
                h_o2, c5_o2 = random_perturb_search(best_h, best_n_final, 2000, 0.004, seed + 7777)
                if c5_o2 < best_c5:
                    best_c5, best_h = c5_o2, h_o2

    # Phase 7: Final coordinate search
    if tleft() > 15:
        h_o, c5_o = coord_local_search(best_h, best_n_final, 100, seed + 1234)
        if c5_o < best_c5:
            best_c5, best_h = c5_o, h_o

    # Phase 8: Last-chance perturbation
    if tleft() > 10:
        nt = min(5000, int((tleft() - 5) / 0.002))
        h_o, c5_o = random_perturb_search(best_h, best_n_final, max(nt, 500), 0.002, seed + 4321)
        if c5_o < best_c5:
            best_c5, best_h = c5_o, h_o

    # Final feasibility enforcement
    best_h = make_feasible(best_h, best_n_final)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n_final)
