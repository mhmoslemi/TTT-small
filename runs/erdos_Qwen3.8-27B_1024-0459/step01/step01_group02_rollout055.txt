def project_feasible(x, n):
    target = n / 2.0
    # Bisection on lambda: h_i = clip(x_i - lambda, 0, 1)
    lo, hi = -max(x) - 1, max(x) + 1  # or tighter bounds
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0, 1)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo+hi)/2, 0, 1)
    # Small correction
    diff = target - h.sum()
    # Distribute among non-clipped elements
    mask = (h > 0) & (h < 1)
    if mask.sum() > 0:
        h[mask] += diff / mask.sum()
    return np.clip(h, 0, 1)

import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _corr_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _grad_corr(h, k_star):
    """Gradient of corr[k_star] * dx with respect to h."""
    n = len(h)
    dx = 2.0 / n
    d = k_star - n + 1  # shift
    grad = np.zeros(n)
    # Term from h[i] when i=j: (1-h[j+d]) if valid
    lo1 = max(0, -d)
    hi1 = min(n, n - d)
    if lo1 < hi1:
        grad[lo1:hi1] += (1.0 - h[lo1+d:hi1+d])
    # Term from -h[i+d] when i+d=j, i=j-d: -h[j-d] if valid
    lo2 = max(0, d)
    hi2 = min(n, n + d)
    if lo2 < hi2:
        grad[lo2:hi2] += -h[lo2-d:hi2-d]
    return grad * dx

def _project_feasible(x, n):
    target = n / 2.0
    # Bisection for exact projection onto {0<=h<=1, sum(h)=n/2}
    lo = -10.0
    hi = 10.0
    # Tighten bounds
    lo = -x.max() - 1.0
    hi = x.max() - x.min() + 0.0  # not tight, just safe
    hi = x.max() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        h = np.clip(x - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    # Final fine correction
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h

def _obj_smooth(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    # For numerical stability, only keep terms within reasonable range
    max_exp = np.max(shifted) * beta
    if max_exp > 500:
        exps = np.exp(beta * shifted - max_exp)
        return (m + np.log(np.sum(exps)) / beta + max_exp / beta) * dx
    exps = np.exp(beta * shifted)
    return (m + np.log(np.sum(exps)) / beta) * dx

def _obj_raw(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return np.max(corr) * dx

def _projected_subgradient(h0, n, iters, lr_init=0.02, lr_decay=0.999, 
                           stagnation_limit=500, wobble=0.01, rng=None):
    """Projected subgradient descent on max of correlations."""
    h = h0.copy().astype(float)
    dx = 2.0 / n
    best_local = _compute_c5(h)
    lr = lr_init
    stagnation = 0
    momentum = np.zeros(n)
    mu = 0.9
    
    for it in range(iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = np.max(corr) * dx
        
        if c5_val < best_local - 1e-12:
            best_local = c5_val
            stagnation = 0
        else:
            stagnation += 1
        
        # Find worst lag
        k_star = np.argmax(corr)
        grad = _grad_corr(h, k_star)
        
        # Momentum
        momentum = mu * momentum + (1 - mu) * grad
        h_try = h - lr * momentum
        
        # Project
        h_try = _project_feasible(h_try, n)
        
        # Check improvement
        corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
        c5_try = np.max(corr_try) * dx
        
        if c5_try < c5_val:
            h = h_try
            lr = min(lr * 1.05, 0.1)
        else:
            # Try smaller step
            h_try2 = h - lr * 0.3 * momentum
            h_try2 = _project_feasible(h_try2, n)
            corr_try2 = np.correlate(h_try2, 1.0 - h_try2, mode='full')
            c5_try2 = np.max(corr_try2) * dx
            if c5_try2 < c5_val:
                h = h_try2
            else:
                # Wobble to escape
                wobble_amt = lr * wobble
                idx = rng.randint(n) if rng is not None else 0
                for attempt in range(5):
                    ii = rng.randint(n) if rng is not None else (it + attempt) % n
                    jj = rng.randint(n) if rng is not None else (it + attempt + 1) % n
                    pert = h.copy()
                    pert[ii] += wobble_amt
                    pert[jj] -= wobble_amt
                    pert = _project_feasible(pert, n)
                    corr_p = np.correlate(pert, 1.0 - pert, mode='full')
                    c5_p = np.max(corr_p) * dx
                    if c5_p < c5_val:
                        h = pert
                        break
            lr *= lr_decay
        
        if stagnation > stagnation_limit:
            lr = lr_init * 0.5
            stagnation = 0
            # Perturb
            if rng is not None:
                pert = h + 0.005 * (rng.rand(n) - 0.5)
                h = _project_feasible(pert, n)
    
    return h, _compute_c5(h)

def _hill_climb(h0, n, iters=5000, step=0.05, rng=None):
    """Coordinate descent / hill climbing."""
    h = h0.copy().astype(float)
    best = _compute_c5(h)
    
    for it in range(iters):
        if rng is None:
            j = it % n
            k = (it + n//2) % n
        else:
            j = rng.randint(n)
            k = rng.randint(n)
            if j == k:
                k = (k + 1) % n
        
        for delta in [step, -step, step*0.5, -step*0.5]:
            h_try = h.copy()
            h_try[j] += delta
            h_try[k] -= delta
            h_try = _project_feasible(h_try, n)
            c5_try = _compute_c5(h_try)
            if c5_try < best - 1e-14:
                h = h_try
                best = c5_try
                break
    
    return h, best

def _slsqp_polish(h0, n, beta=100.0, maxiter=300):
    """SLSQP optimization with smooth or raw objective."""
    h0 = _project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    
    def constraint_fun(x):
        return np.sum(x) - target
    
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': lambda x: np.ones(n)}]
    
    if beta < 1000:
        def obj(x):
            return _obj_smooth(x, beta)
    else:
        def obj(x):
            return _obj_raw(x)
    
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h_out = _project_feasible(res.x, n)
        return h_out, _compute_c5(h_out)
    except Exception:
        return h0, _compute_c5(h0)

def _try_binary_local_search(h0, n, rng, max_flips=500):
    """Try to improve a near-binary solution by flipping bits."""
    h = h0.copy()
    # Round to binary
    h_bin = (h > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    n_want = n // 2
    
    # Fix count
    if n_ones > n_want:
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h0[ones_idx])[:n_ones - n_want]
        h_bin[ones_idx[order]] = 0.0
    elif n_ones < n_want:
        zeres_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h0[zeres_idx])[:n_want - n_ones]
        h_bin[zeres_idx[order]] = 1.0
    
    best = _compute_c5(h_bin)
    
    for _ in range(max_flips):
        # Try swapping a 1 and a 0
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c5_try = _compute_c5(h_bin)
        if c5_try < best - 1e-14:
            best = c5_try
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    
    return h_bin, best

def _fourier_init(n, rng, k_modes=5):
    """Create initial h from superposition of Fourier modes."""
    t = np.arange(n) / n * 2.0  # position in [0, 2)
    h = np.zeros(n)
    for k in range(1, k_modes + 1):
        amp = rng.uniform(0.1, 0.4) / np.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2 * np.pi * k * t / 2.0 + phase)
    h = 0.5 + h
    h = np.clip(h, 0, 1)
    h = _project_feasible(h, n)
    return h
