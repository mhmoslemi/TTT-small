import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_binary_swap(h, i, j):
    """Swap: h[i] 1->0, h[j] 0->1 (or vice versa)"""
    h[i] = 1.0 - h[i]
    h[j] = 1.0 - h[j]
    return h

def local_search_binary(h, n, time_limit, rng):
    """Greedy 2-flip local search for binary h."""
    start = time.time()
    best_c5 = compute_c5(h)
    h = h.copy()
    
    for iteration in range(10000):
        if time.time() - start > time_limit:
            break
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        # Shuffle for randomization
        rng.shuffle(ones)
        rng.shuffle(zeros)
        
        for i in ones:
            if time.time() - start > time_limit:
                break
            for j in zeros:
                # Try swap
                h[i] = 0.0
                h[j] = 1.0
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    improved = True
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            if improved:
                break
        if not improved:
            break
    
    return h, best_c5

    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_c5 = best_c5

        j = zeros[j]

                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        T *= cooling
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)

    x_new = (np.arange(n) + 0.5) * (2.0 / n)
    interp = np.interp(x_new, xp, parent_h)
    # Round to binary with sum = n/2
    sorted_idx = np.argsort(-interp)
    h = np.zeros(n)
    h[sorted_idx[:n // 2]] = 1.0
    return h

        beta = 50.0
        exps = np.exp(beta * (corr - m))
        return (m + np.log(np.sum(exps)) / beta) * dx
