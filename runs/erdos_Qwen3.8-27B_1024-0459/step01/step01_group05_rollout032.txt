import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def smooth_max_val(c, beta):
    m = np.max(c)
    shifted = np.clip(beta * (c - m), -60.0, 60.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(c, beta):
    m = np.max(c)
    shifted = np.clip(beta * (c - m), -60.0, 60.0)
    e = np.exp(shifted)
    w = e / np.sum(e)
    return w

def c5_grad_analytical(h, beta):
    """Analytical gradient of smooth max of correlate(h, 1-h)."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    w = smooth_max_weights(c, beta)
    
    grad = np.zeros(n)
    idx = np.arange(n)
    
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        d = k - (n - 1)
        if d >= 0:
            # c[k] = sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
            # dc/dh[j] = (1-h[j-d]) for j in [d, n-1]
            #           -h[j+d] for j in [0, n-1-d]
            if d == 0:
                # dc/dh[j] = (1-h[j]) - h[j] = 1 - 2*h[j]
                grad += w[k] * (1.0 - 2.0 * h)
            else:
                mask1 = idx >= d  # j-d valid
                mask2 = idx < (n - d)  # j+d valid
                grad[mask1] += w[k] * (1.0 - h[idx[mask1] - d])
                grad[mask2] += w[k] * (-h[idx[mask2] + d])
        else:
            dd = -d
            # c[k] = sum_{i=dd}^{n-1} h[i-dd]*(1-h[i])
            # Let j' = i-dd, j' ranges 0 to n-1-dd
            # c[k] = sum_{j'=0}^{n-1-dd} h[j']*(1-h[j'+dd])
            # dc/dh[j] = (1-h[j+dd]) for j in [0, n-1-dd]
            #           -h[j-dd] for j in [dd, n-1]
            if dd == 0:
                grad += w[k] * (1.0 - 2.0 * h)
            else:
                mask1 = idx < (n - dd)
                mask2 = idx >= dd
                grad[mask1] += w[k] * (1.0 - h[idx[mask1] + dd])
                grad[mask2] += w[k] * (-h[idx[mask2] - dd])
    
    return grad

def project_sum(h, target):
    """Project h onto {sum(h)=target, 0<=h<=1}."""
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        # Distribute among free variables
        if diff > 0:
            free = h < 1.0
        else:
            free = h > 0.0
        if not np.any(free):
            # Fallback: distribute among all
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)

_sum_target = 0.0
_beta_global = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_val(c, beta):
    m = np.max(c)
    shifted = np.clip(beta * (c - m), -60.0, 60.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(c, beta):
    m = np.max(c)
    shifted = np.clip(beta * (c - m), -60.0, 60.0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s == 0:
        return np.ones(len(e)) / len(e)
    return e / s

def _obj_func(h):
    n = len(h)
    cur = np.clip(h, 0.0, 1.0)
    corr = np.correlate(cur, 1.0 - cur, mode='full')
    return smooth_max_val(corr, _beta_global)

def _jac_func(h):
    n = len(h)
    cur = np.clip(h, 0.0, 1.0)
    corr = np.correlate(cur, 1.0 - cur, mode='full')
    w = smooth_max_weights(corr, _beta_global)
    grad = np.zeros(n)
    grad += w[n - 1] * (1.0 - 2.0 * cur)
    for d in range(1, n):
        wp = w[n - 1 + d]
        wm = w[n - 1 - d]
        if abs(wp) < 1e-18 and abs(wm) < 1e-18:
            continue
        if wp != 0.0:
            grad[d:] += wp * (1.0 - cur[:n - d])

        if wm != 0.0:
            grad[:n - d] += wm * (1.0 - cur[d:])
            grad[d:] -= wm * cur[:n - d]
    return grad

            free = h < 1.0
        else:
            free = h > 0.0

            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)

        if diff > 0:

            order = np.argsort(h)

            else:
                adj = max(diff, -h[i])
            h[i] += adj
            diff -= adj
            if abs(diff) < 1e-15:
                break
    return h

def interp_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return h_new

def run_slsqp(h0, target, beta, max_iter, ftol):

    _sum_target = target
    _beta_global = beta
