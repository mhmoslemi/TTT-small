        best_h = _random_binary(best_n, rng)
        best_c5 = _compute_c5(best_h)

    print("Initial: n={}, C5={:.6f}".format(best_n, best_c5))

    def _time_left():
        return max(0.0, budget_s - (time.time() - start_time) - 5.0)

    # Phase 1: Binary sequence search at n=96 and n=192
    # This explores the combinatorial landscape
    for n in [96, 192]:
        if _time_left() < 50:
            break
        # Also load parent at this resolution
        candidates = []
        if parent_h is not None and parent_n > 0:
            if n == parent_n:
                candidates.append(parent_h.copy())
            else:
                candidates.append(_interpolate(parent_h, parent_n, n))
        
        # Random binary starts
        for _ in range(8):
            if _time_left() < 15:
                break
            h_bin = _random_binary(n, rng)
            candidates.append(h_bin)
        
        for h0 in candidates:
            if _time_left() < 10:
                break
            h0 = _project(h0, n)
            if np.allclose(h0, (h0 > 0.5).astype(float)):
                h_opt, c5_opt = _hill_climb_binary(h0, n, rng, max_iters=2000)
            else:
                h_opt, c5_opt = _slsqp_optimize(h0, n, 10.0, 150, 1e-12)
            if c5_opt < best_c5 - 1e-10:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
                print("  Phase1 n={}, C5={:.6f}".format(n, best_c5))

    print("After Phase 1: n={}, C5={:.6f}".format(best_n, best_c5))

    # Phase 2: Multi-resolution SLSQP refinement with increasing beta
    # Focus on the best resolution and a couple near it
    res_list = sorted(set([best_n, 96, 128]))
    res_list = [r for r in res_list if 64 <= r <= 256 and r % 2 == 0]
    
    for n in res_list:
        if _time_left() < 25:
            break
        _G_N = n
        _G_BETA = 1.0
        _G_DX = 2.0 / n
        
        # Get starting point
        if n == best_n:
            h0 = _project(best_h.copy(), n)
        elif parent_h is not None:
            h0 = _interpolate(parent_h, parent_n, n)
        else:
            h0 = _project(best_h[:n] if len(best_h) >= n else best_h, n)
            if len(h0) != n:
                h0 = _random_binary(n, rng)
        
        h_curr = h0.copy()
        c5_curr = _compute_c5(h_curr)
        
        for beta in [5.0, 15.0, 40.0, 100.0, 300.0, 1e8]:
            if _time_left() < 12:
                break
            h_new, c5_new = _slsqp_optimize(h_curr, n, beta, 250, 1e-14)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n
                print("  Phase2 n={}, beta={:.0f}, C5={:.6f}".format(n, beta, best_c5))
            if c5_new < c5_curr - 1e-12:
                h_curr = h_new
                c5_curr = c5_new

    print("After Phase 2: n={}, C5={:.6f}".format(best_n, best_c5))

    # Phase 3: Basin hopping for escape from local optima
    if _time_left() > 60:
        n = best_n
        h_curr = _project(best_h.copy(), n)
        c5_curr = _compute_c5(h_curr)
        
        scale_outer = 0.15
        n_pert = max(5, n // 8)
        
        for outer_iter in range(15):
            if _time_left() < 15:
                break
            h_pert = _perturb_reoptimize(h_curr, n, rng, n_pert, scale_outer)
            h_opt, c5_opt = _slsqp_optimize(h_pert, n, 40.0, 200, 1e-13)
            if c5_opt < best_c5 - 1e-12:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
                h_curr = h_opt.copy()
                c5_curr = c5_opt
                print("  Basin{} n={}, C5={:.6f}".format(outer_iter, n, best_c5))
            # Slowly reduce perturbation
            scale_outer *= 0.85
    
    # Phase 4: Fine polish at very high beta + raw
    if _time_left() > 20:
        n = best_n
        h_curr = _project(best_h.copy(), n)
        
        for beta in [200.0, 500.0, 1000.0, 1e8]:
            if _time_left() < 8:
                break
            h_new, c5_new = _slsqp_optimize(h_curr, n, beta, 400, 1e-15)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_curr = h_new.copy()
                print("  Polish beta={:.0f}, C5={:.6f}".format(beta, best_c5))
            elif c5_new < _compute_c5(h_curr):
                h_curr = h_new

    # Phase 5: Coordinate descent on active lags
    if _time_left() > 15:
        n = best_n
        h_curr = _project(best_h.copy(), n)
        corr = _compute_corr(h_curr)
        max_val = np.max(corr)
        active = np.where(corr >= max_val * 0.995)[0]
        
        # Try single-coordinate adjustments
        for coord in active:
            if _time_left() < 3:
                break
            s = coord - (n - 1)
            # Which h values affect this lag?
            L = max(0, -s)
            U = min(n, n - s)
            affected = list(range(L + s, min(n, U + s))) + list(range(L, U))
            affected = list(set(affected))
            for j in affected[:20]:
                for delta in [0.01, -0.01, 0.005, -0.005]:
                    h_try = h_curr.copy()
                    h_try[j] += delta
                    h_try = _project(h_try, n)
                    c5_try = _compute_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h_curr = h_try.copy()
                        break
                else:
                    continue
                break

    # Phase 6: Try a different resolution for the final push
    if _time_left() > 25:
        # Try n=best_n * 2 for potential improvement
        n2 = best_n * 2
        if n2 <= 400:
            h2 = _interpolate(best_h, best_n, n2)
            h2_opt, c5_2 = _slsqp_optimize(h2, n2, 20.0, 200, 1e-13)
            if c5_2 < best_c5 - 1e-10:
                best_c5 = c5_2
                best_h = h2_opt.copy()
                best_n = n2
                print("  Higher-res n={}, C5={:.6f}".format(n2, best_c5))
                # Polish at new resolution
                if _time_left() > 15:
                    h2_opt2, c5_2b = _slsqp_optimize(h2_opt, n2, 200.0, 200, 1e-14)
                    if c5_2b < best_c5:
                        best_c5 = c5_2b
                        best_h = h2_opt2.copy()
                        print("  Higher-res polish C5={:.6f}".format(best_c5))

    # Final projection and evaluation
    best_h = _project(best_h, best_n)
    final_c5 = _compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
