**Idea: Coordinate descent with optimal 1D solution**

For each i, given all other h's fixed, the overlap at lag k is:
corr_k = A_k + B_k * h[i]

where A_k and B_k are constants (depending on other h values):
A_k = corr_k without the h[i] term
B_k = d corr_k / d h[i] (as computed above)

So the max overlap as a function of h[i] (with h[i] ∈ [0,1]) is:
max_k (A_k + B_k * h[i])

This is the max of linear functions of h[i], so it's a convex piecewise-linear function. The minimum over h[i] ∈ [0,1] can be found exactly!

If B_k > 0 for all active k: minimum at h[i] = 0
If B_k < 0 for all active k: minimum at h[i] = 1
Otherwise: find the intersection point where the max changes.

**Pair coordinate descent**: Pick two indices (i, j) and optimize h[i] and h[j] jointly while keeping h[i] + h[j] = constant (to maintain the integral). This is a 2D problem but the feasible set is a 1D line segment.

For the pair (i,j) with h[i] + h[j] = c (constant):
Set h[j] = c - h[i], then the overlap at lag k becomes:
corr_k = A_k + B_k[i] * h[i] + B_k[j] * (c - h[i])
= A_k + B_k[j]*c + (B_k[i] - B_k[j]) * h[i]

So it's still linear in h[i]! The max over k is the max of linear functions of h[i], which is convex. We can find the exact minimizer.

   g. Update h[i] and h[j]
3. Repeat until convergence or budget exhausted

The 1D minimization of max of linear functions:
Given lines f_i(t) = a_i + b_i * t for i=1..m, find t* that minimizes max_i f_i(t) subject to t ∈ [t_min, t_max].

The minimum of the max of lines occurs at a "breakpoint" where two (or more) lines cross, or at a boundary. We can find it by:
- Compute all pairwise intersections: t_ij = (a_i - a_j) / (b_j - b_i) for b_i ≠ b_j
- Filter to those in [t_min, t_max]
- Evaluate max_i f_i(t) at all candidate points
- Pick the best

- Compute gradient for top 2 lags: O(n) = 180 ops per lag

import time
import numpy as np
from scipy.optimize import minimize


def _eval(h):
    n = h.size
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_density(x, target):
    x = np.asarray(x, dtype=float)
    x = np.clip(np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-13:
        return x.copy()
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        sm = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if sm < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-12:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            h[free[0]] += res
            h = np.clip(h, 0.0, 1.0)
    return h


def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(n, 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.interp(x1, x0, p)
    return project_density(h, 0.5 * n)


def _grad_lag(h, m, L, n):
    """Gradient of corr at physical lag L w.r.t. h[m]."""
    val = 0.0
    if max(0, L) <= m <= min(n - 1, n - 1 + L):
        val += (1.0 - h[m - L])
    if max(0, -L) <= m <= min(n - 1, n - 1 - L):
        val -= h[m + L]
    return val


def _grad_all_lags(h, L, n):
    """Gradient of corr at physical lag L for all positions."""
    g = np.zeros(n)
    # First term: (1-h[m-L]) for m in [max(0,L), min(n-1, n-1+L)]
    lo1 = max(0, L)
    hi1 = min(n - 1, n - 1 + L)
    if lo1 <= hi1:
        g[lo1:hi1 + 1] += (1.0 - h[lo1 - L:hi1 + 1 - L])
    # Second term: -h[m+L] for m in [max(0,-L), min(n-1, n-1-L)]
    lo2 = max(0, -L)
    hi2 = min(n - 1, n - 1 - L)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] -= h[lo2 + L:hi2 + 1 + L]
    return g


def _minimax_lines(A, D, t_lo, t_hi, t_init):
    """Minimize max_k(A[k] + D[k]*t) for t in [t_lo, t_hi]."""
    m = len(A)
    if m == 0:
        return t_init
    candidates = [t_lo, t_hi, t_init]
    for i in range(m):
        for j in range(i + 1, m):
            dD = D[i] - D[j]
            if abs(dD) > 1e-14:
                t_int = (A[j] - A[i]) / dD
                if t_lo - 1e-12 <= t_int <= t_hi + 1e-12:
                    candidates.append(t_int)
    best_t = t_init
    best_val = np.inf
    for t in candidates:
        t = max(t_lo, min(t_hi, t))
        val = float(np.max(A + D * t))
        if val < best_val - 1e-14:
            best_val = val
            best_t = t
    return best_t


def gradient_swap_descent(h_in, target, deadline, n_active=6, max_iters=500000, rng=None):
    """Main optimizer: gradient-directed pair swaps with exact 1D minimax."""
    n = h_in.size
    if rng is None:
        rng = np.random.default_rng(12345)
    best_h = project_density(h_in, target).copy()
    best_c = _eval(best_h)
    center = n - 1
    stall = 0
    it = 0

    while it < max_iters and time.perf_counter() < deadline:
        it += 1
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        n_corr = len(corr)
        k_use = min(n_active, n_corr)
        top_idx = np.argpartition(corr, -k_use)[-k_use:]
        top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]

        # Compute combined gradient from top 2-3 lags
        n_grad = min(3, k_use)
        g = np.zeros(n)
        for ai in range(n_grad):
            L = int(top_idx[ai]) - center
            g += _grad_all_lags(best_h, L, n)
        g /= float(n_grad)

        # Find best productive pair: high g → decrease, low g → increase
        # Use top candidates for efficiency
        neg_cands = np.argsort(g)[:12]
        pos_cands = np.argsort(-g)[:12]

        improved = False
        lags_arr = top_idx - center  # physical lags

        for ii in neg_cands:
            ii_i = int(ii)
            if best_h[ii_i] >= 1.0 - 1e-12:
                continue
            for jj in pos_cands:
                jj_i = int(jj)
                if jj_i == ii_i or best_h[jj_i] <= 1e-12:
                    continue
                s_pair = best_h[ii_i] + best_h[jj_i]
                t_lo = max(0.0, s_pair - 1.0)
                t_hi = min(1.0, s_pair)
                if t_hi - t_lo < 1e-12:
                    continue

                # Exact minimax over active lags
                D_vals = np.zeros(k_use)
                A_vals = np.zeros(k_use)
                for ai in range(k_use):
                    L = lags_arr[ai]
                    gi = _grad_lag(best_h, ii_i, int(L), n)
                    gj = _grad_lag(best_h, jj_i, int(L), n)
                    D_vals[ai] = gi - gj
                    A_vals[ai] = corr[top_idx[ai]] + (gj - gi) * best_h[ii_i]

                t_opt = _minimax_lines(A_vals, D_vals, t_lo, t_hi, best_h[ii_i])
                if abs(t_opt - best_h[ii_i]) < 1e-11:
                    continue

                new_ii = float(t_opt)
                new_jj = s_pair - new_ii
                if new_ii < -1e-12 or new_ii > 1.0 + 1e-12:
                    continue
                if new_jj < -1e-12 or new_jj > 1.0 + 1e-12:
                    continue

                best_h[ii_i] = max(0.0, min(1.0, new_ii))
                best_h[jj_i] = max(0.0, min(1.0, new_jj))
                improved = True
                break
            if improved:
                break

        if improved:
            stall = 0
            if it % 200 == 0:
                c = _eval(best_h)
                if c < best_c:
                    best_c = c
        else:
            stall += 1
            if stall > 300:
                # Random perturbation to escape
                for _ in range(30):
                    a = rng.integers(n)
                    b = rng.integers(n)
                    if a == b:
                        continue
                    if best_h[a] <= 0 and best_h[b] >= 1:
                        continue
                    delta = rng.uniform(0.005, 0.05)
                    if best_h[a] + delta <= 1 and best_h[b] - delta >= 0:
                        best_h[a] += delta
                        best_h[b] -= delta
                    elif best_h[a] - delta >= 0 and best_h[b] + delta <= 1:
                        best_h[a] -= delta
                        best_h[b] += delta
                    else:
                        continue
                    best_h = project_density(best_h, target)
                    c = _eval(best_h)
                    if c < best_c:
                        best_c = c
                        stall = 0
                        break
                else:
                    # Try binary rounding
                    h_bin = (best_h > 0.5).astype(float)
                    h_bin = project_density(h_bin, target)
                    c_bin = _eval(h_bin)
                    if c_bin < best_c:
                        best_h = h_bin
                        best_c = c_bin
                        stall = 0
                    if stall > 2000:
                        break

    best_h = project_density(best_h, target)
    best_c = _eval(best_h)
    return best_h, best_c


def full_pair_cd(h_in, target, deadline, n_active=8, max_passes=500, rng=None):
    """Full pair coordinate descent (tries all pairs)."""
    n = h_in.size
    if rng is None:
        rng = np.random.default_rng(777)
    best_h = project_density(h_in, target).copy()
    best_c = _eval(best_h)
    center = n - 1
    stall = 0

    for p in range(max_passes):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        k_use = min(n_active, len(corr))
        top_idx = np.argpartition(corr, -k_use)[-k_use:]
        top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
        lags_arr = (top_idx - center).astype(int)
        current_max = float(corr[top_idx[0]])

        improved = False
        order = rng.permutation(n)

        for idx_i in range(n):
            i = int(order[idx_i])
            for idx_j in range(idx_i + 1, n):
                j = int(order[idx_j])
                s_pair = best_h[i] + best_h[j]
                t_lo = max(0.0, s_pair - 1.0)
                t_hi = min(1.0, s_pair)
                if t_hi - t_lo < 1e-12:
                    continue

                D_vals = np.zeros(k_use)
                A_vals = np.zeros(k_use)
                any_diff = False
                for ai in range(k_use):
                    L = lags_arr[ai]
                    gi = _grad_lag(best_h, i, L, n)
                    gj = _grad_lag(best_h, j, L, n)
                    d = gi - gj
                    D_vals[ai] = d
                    A_vals[ai] = corr[top_idx[ai]] + (gj - gi) * best_h[i]
                    if abs(d) > 1e-12:
                        any_diff = True
                if not any_diff:
                    continue

                t_opt = _minimax_lines(A_vals, D_vals, t_lo, t_hi, best_h[i])
                if abs(t_opt - best_h[i]) > 1e-10:
                    new_i = max(0.0, min(1.0, float(t_opt)))
                    new_j = max(0.0, min(1.0, s_pair - new_i))
                    best_h[i] = new_i
                    best_h[j] = new_j
                    improved = True

        if improved:
            stall = 0
            c = _eval(best_h)
            if c < best_c:
                best_c = c
        else:
            stall += 1
            if stall > 5:
                break

    best_h = project_density(best_h, target)
    best_c = _eval(best_h)
    return best_h, best_c


def slsqp_smooth(h_in, n, target, deadline, alphas=(15.0, 50.0, 150.0), maxiter=60):
    """SLSQP with smooth max approximation."""
    best_h = project_density(h_in, target)
    best_c = _eval(best_h)
    bounds = [(0.0, 1.0)] * n
    constraint = ({"type": "eq", "fun": lambda x: float(np.sum(x)) - target,
                   "jac": lambda x: np.ones(n)},)

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        if not np.all(np.isfinite(corr)):
            return 1e100
        m = float(np.max(corr))
        dx = 2.0 / n
        return m * dx

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        cm = float(np.max(corr))
        dx = 2.0 / n
        # Subgradient of max
        top = np.where(np.abs(corr - cm) < 1e-10)[0]
        g = np.zeros(n)
        for t in top[:5]:
            L = int(t) - (n - 1)
            g += _grad_all_lags(x, L, n)
        g /= max(1, len(top[:5]))
        return dx * g

    h = best_h.copy()
    for alpha in alphas:
        if time.perf_counter() >= deadline:
            break
        try:
            res = minimize(obj, h, method="SLSQP", jac=jac, bounds=bounds,
                           constraints=constraint,
                           options={"maxiter": maxiter, "ftol": 1e-10, "disp": False})
            if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
                cand = project_density(res.x, target)
                c = _eval(cand)
                if c < best_c:
                    best_h = cand
                    best_c = c
                h = cand if c < _eval(h) + 0.01 else best_h
        except Exception:
            pass

    return best_h, best_c


def make_binary_candidate(h, target, rng):
    """Try to find a good binary solution near h."""
    n = h.size
    # Round and fix
    h_bin = (h + rng.uniform(-0.1, 0.1, n) > 0.5).astype(float)
    h_bin = project_density(h_bin, target)
    _ = _eval(h_bin)
    return h_bin

    deadline_phase3 = start + min(budget * 0.55, hard_deadline - 120)

        best_c = final_c
    best_h = final_h
