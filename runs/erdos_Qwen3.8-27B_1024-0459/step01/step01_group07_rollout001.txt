def project_robust(y, n):
    target = n/2.0
    # initial clip
    x=np.clip(y,0,1)
    # use waterfilling from y
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    # We want sum clamp(y - alpha,0,1) = target. x(alpha)=sum(clip(y-alpha,0,1)) decreases with alpha.
    for _ in range(60):
       mid=(lo+hi)/2
       s=np.sum(np.clip(y-mid,0.0,1.0))
       if s > target: lo=mid  # need larger alpha to reduce sum
       else: hi=mid
    alpha=(lo+hi)/2
    x=np.clip(y-alpha,0,1)
    # correct residual
    diff=target-x.sum()
    free=(x>0)&(x<1)
    if free.any(): x[free]=np.clip(x[free]+diff/np.sum(free),0,1)
    # few fast iter
    return x
```

def project_robust(h,n):
    target = 0.5*n
    x = np.array(h, dtype=float)
    if x.size != n: ... # maybe resize? Not needed.
    # if not finite, replace
    x[~np.isfinite(x)] = 0.5
    if target <= 1e-15: return np.zeros(n)
    if target >= n-1e-15: return np.ones(n)
    # Quick bisection on alpha to minimize ||x - clip(y-alpha)||
    y=np.clip(x,-10,10)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(50):
       mid=0.5*(lo+hi)
       s = np.sum(np.clip(y-mid,0.0,1.0))
       if s > target: lo=mid
       else: hi=mid
    alpha=0.5*(lo+hi)
    z=np.clip(y-alpha,0.0,1.0)
    # if residual large due to discontinuity, fallback top rounding
    if abs(np.sum(z)-target) > 1e-8:
       # use sort fallback (exact but not necessarily Euclidean projection)
       order=np.argsort(-y)
       z=np.zeros(n)
       m=int(target)
       r=target-m
       z[order[:m]]=1.0
       if r > 1e-12:
           z[order[m]]=r
       # if m==target no fractional
       # This handles degenerate; for normal fail maybe not optimal but feasible
       return z
    # adjust residual with free vars
    d=target-z.sum()
    free=(z>1e-12)&(z<1-1e-12)
    if free.any():
       z[free]=np.clip(z[free]+d/np.count_nonzero(free),0,1)
    # final few
    for _ in range(5):
       d=target-z.sum()
       if abs(d)<1e-13: break
       free=(z>0)&(z<1)
       if not free.any():
          break
       z[free]=np.clip(z[free]+d/np.count_nonzero(free),0,1)
    return z

def compute_c(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode='full') * dx


def eval_c5(h, n):
    c = compute_c(h, n)
    return float(np.max(c))


def top_indices(c, k):
    m = c.size
    if k <= 0:
        k = 1
    if k >= m:
        return np.arange(m)
    return np.argpartition(c, m - k)[-k:]


def smooth_weights(c, temp, topk):
    mx = float(np.max(c))
    idx = top_indices(c, topk)
    vals = c[idx]
    sh = (vals - mx) / temp
    sh = np.where(sh > -50.0, sh, -50.0)
    w = np.exp(sh)
    sw = float(np.sum(w))

    if sw <= 1e-300:
        w = np.zeros_like(w)
        w[int(np.argmax(vals))] = 1.0
        sw = 1.0
    else:
        w = w / sw

    val = mx + temp * np.log(sw)
    return val, idx, w


def active_weights(c, mx, tol, topk):
    idx = np.flatnonzero(c >= mx - tol)
    if idx.size == 0:
        idx = np.array([int(np.argmax(c))])
    if idx.size > topk:
        idx = top_indices(c, topk)
    w = np.full(idx.size, 1.0 / float(idx.size))
    return mx, idx, w


def grad_weights(h, idx, w, n):
    grad = np.zeros(n, dtype=float)
    dx = 2.0 / float(n)
    base = n - 1

    for t, wj in zip(idx, w):
        if wj <= 0.0:
            continue
        m = int(t) - base
        wdx = wj * dx

        if m == 0:
            grad += wdx * (1.0 - 2.0 * h)
        elif m > 0:
            seg = n - m
            if seg <= 0:
                continue
            grad[m:] += wdx * (1.0 - h[:seg])
            grad[:seg] += -wdx * h[m:]
        else:
            p = -m
            seg = n - p
            if seg <= 0:
                continue
            grad[:seg] += wdx * (1.0 - h[p:])
            grad[p:] += -wdx * h[:seg]

    return grad


def resample_linear(h, n_src, n_dst):
    if n_dst == n_src:
        return np.array(h, dtype=float).copy()
    if n_src == 1:
        return np.full(n_dst, float(h[0]))
    old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(new, old, h)


def upsample(h_src, n_src, n_dst):
    h_src = np.asarray(h_src, dtype=float).reshape(-1)
    if n_dst == n_src:
        return h_src.copy()
    if n_src > 0 and n_dst % n_src == 0:
        r = n_dst // n_src
        if r <= 12:
            return np.repeat(h_src, r).copy()
    return resample_linear(h_src, n_src, n_dst)


def make_sum_exact(h, n):
    target = 0.5 * float(n)
    x = np.asarray(h, dtype=float).reshape(-1).copy()
    if x.size != n:
        x = project_robust(x, n)
    x = np.clip(x, 0.0, 1.0)

    for _ in range(n + 5):
        d = target - float(np.sum(x))
        if abs(d) < 1e-14:
            break

        if d > 0.0:
            idx = int(np.argmin(x))
            room = 1.0 - x[idx]
            if room > 0.0:
                add = min(room, d)
                x[idx] += add
            else:
                changed = False
                for i in range(n):
                    if x[i] < 1.0:
                        add = min(1.0 - x[i], d)
                        x[i] += add
                        d -= add
                        changed = True
                        if abs(d) < 1e-14:
                            break
                if not changed:
                    break
        else:
            idx = int(np.argmax(x))
            head = x[idx]
            if head > 0.0:
                sub = min(head, -d)
                x[idx] -= sub
            else:
                changed = False
                for i in range(n):
                    if x[i] > 0.0:
                        sub = min(x[i], -d)
                        x[i] -= sub
                        d += sub
                        changed = True
                        if abs(d) < 1e-14:
                            break
                if not changed:
                    break

    d = target - float(np.sum(x))
    if abs(d) > 1e-15:
        if d > 0.0:
            for i in range(n):
                if x[i] < 1.0:
                    add = min(1.0 - x[i], d)
                    x[i] += add
                    d -= add
                    if abs(d) < 1e-15:
                        break
        else:
            for i in range(n):
                if x[i] > 0.0:
                    sub = min(x[i], -d)
                    x[i] -= sub
                    d += sub
                    if abs(d) < 1e-15:
                        break

    return x


def pattern_cos(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    return 0.5 + 0.5 * np.cos(np.pi * x)


def pattern_center(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    return 0.5 - 0.5 * np.cos(np.pi * x)


def pattern_tri(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.minimum(x, 2.0 - x)


def pattern_interval(n):
    h = np.zeros(n, dtype=float)
    s = int(0.25 * n)
    e = s + int(0.5 * n)
    h[s:e] = 1.0
    return h


def anneal_optimize(h0, n, deadline, rng, topk, initial_temp=0.10):
    h = project_robust(h0, n)
    c0 = compute_c(h, n)
    best_h = h.copy()
    best_v = float(np.max(c0))

    mom = np.zeros(n, dtype=float)
    beta = 0.70
    step = 0.18
    temp = float(initial_temp)
    min_temp = 0.0018
    max_temp = 0.20

    no_smooth = 0
    no_exact = 0
    it = 0

    while time.time() < deadline:
        c = compute_c(h, n)
        mx = float(np.max(c))

        if mx < best_v - 1e-12:
            best_v = mx
            best_h = h.copy()
            no_exact = 0
        else:
            no_exact += 1

        if temp < 0.0035:
            val, idx, w = active_weights(c, mx, 1e-6, topk)
        else:
            val, idx, w = smooth_weights(c, temp, topk)

        grad = grad_weights(h, idx, w, n)
        gnorm = float(np.linalg.norm(grad))

        if gnorm < 1e-13:
            h = project_fast(best_h + rng.normal(0.0, 0.10, n), n)
            mom *= 0.3
            no_smooth = 0
            it += 1
            continue

        mom = beta * mom + (1.0 - beta) * grad
        d = -mom
        dnorm = float(np.linalg.norm(d))

        if dnorm < 1e-13:
            d = -grad
            dnorm = gnorm

        if dnorm < 1e-13:
            h = project_fast(best_h + rng.normal(0.0, 0.10, n), n)
            mom *= 0.3
            no_smooth = 0
            it += 1
            continue

        d = d / dnorm
        accepted = False

        for fac in (1.0, 0.60, 0.35, 0.20, 0.10):
            alpha = step * fac
            if alpha < 1e-10:
                break

            cand = h + alpha * d
            cand = project_fast(cand, n)
            cn = compute_c(cand, n)
            mxn = float(np.max(cn))

            if mxn > best_v + 0.15 + 0.5 * temp:
                continue

            if mxn < mx - 1e-11:
                h = cand
                accepted = True
                break

            if temp >= 0.0035:
                valn, _, _ = smooth_weights(cn, temp, topk)
                if valn < val - 1e-10:
                    h = cand
                    accepted = True
                    break

        if accepted:
            no_smooth = 0
            if step < 0.8:
                step = min(0.8, step * 1.04)
            if temp > min_temp:
                temp = max(min_temp, temp * 0.997)
        else:
            no_smooth += 1
            if step > 0.02:
                step = max(0.02, step * 0.85)
            if temp < max_temp:
                temp = min(max_temp, temp * 1.015)

        if no_smooth > 40 or no_exact > 140:
            scale = 0.04 + 0.35 * temp + 0.01
            if rng.random() < 0.6:
                h = project_fast(best_h + rng.normal(0.0, scale, n), n)
            else:
                h = best_h.copy()
                nblocks = 1 + int(rng.randint(0, 3))
                for _ in range(nblocks):
                    blen = max(2, int(n * 0.04 + rng.randint(0, max(1, int(n * 0.08)))))
                    if blen > n:
                        blen = n
                    start = int(rng.randint(0, max(1, n - blen)))
                    h[start:start + blen] += rng.normal(0.0, scale * 1.5, blen)
                h = project_fast(h, n)

            mom *= 0.25
            step = 0.20
            no_smooth = 0
            if no_exact > 140:
                no_exact = 60

        it += 1

    h = project_fast(h, n)
    cv = float(np.max(compute_c(h, n)))
    if cv < best_v:
        best_v = cv
        best_h = h.copy()

    best_h = project_fast(best_h, n)
    return best_h, best_v


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    deadline_global = max(t0 + 10.0, t0 + budget - 25.0)
    rng = np.random.RandomState(seed)

    h_parent = None
    try:
        arr = np.asarray(initial_h_values, dtype=float)
        if arr.size > 0:
            h_parent = arr.reshape(-1)
    except Exception:
        h_parent = None

    if h_parent is not None:
        parent_n = int(h_parent.size)
        if parent_n < 4:
            n_base = max(4, parent_n + (parent_n % 2))
        else:
            n_base = parent_n if (parent_n % 2 == 0) else (parent_n - 1)
        if n_base < 2:
            n_base = 2

        if n_base != parent_n:
            h0b = resample_linear(h_parent, parent_n, n_base)
        else:
            h0b = h_parent.copy()
        h0b = project_robust(h0b, n_base)
    else:
        n_base = 140
        candidates = [
            pattern_cos(n_base),
            pattern_center(n_base),
            pattern_tri(n_base),
            pattern_interval(n_base),
            project_robust(rng.uniform(0.0, 1.0, n_base), n_base),
        ]
        best_v_pat = 1e9
        h0b = candidates[0]
        for p in candidates:
            p = project_robust(p, n_base)
            v = eval_c5(p, n_base)
            if v < best_v_pat:
                best_v_pat = v
                h0b = p

    best_h = h0b.copy()
    best_n = int(n_base)
    best_v = eval_c5(best_h, best_n)

    ns = []
    seen = set()
    for mult in (1, 2, 3, 4, 5, 6, 8):
        n = mult * best_n
        if 10 <= n <= 600 and n not in seen:
            ns.append(n)
            seen.add(n)

    if len(ns) == 0:
        ns.append(best_n)

    if max(ns) < 280:
        for n in (best_n + 60, best_n + 120, best_n + 180, 280, 420):
            if best_n <= n <= 600 and n not in seen:
                ns.append(n)
                seen.add(n)

    ns.sort()
    if best_n not in seen:
        ns.append(best_n)
        ns.sort()

    reserve = 150.0
    total_opt = max(60.0, budget - reserve)

    profiles = []
    sumw = 0.0
    for n in ns:
        if n <= 80:
            w = 220.0
        elif n <= 160:
            w = 280.0
        elif n <= 240:
            w = 300.0
        elif n <= 320:
            w = 300.0
        elif n <= 440:
            w = 260.0
        else:
            w = 180.0
        profiles.append((n, w))
        sumw += w

    scale = total_opt / sumw if sumw > 0.0 else 1.0

    for n, w in profiles:
        if time.time() >= deadline_global - 10.0:
            break

        alloc = w * scale
        if alloc < 15.0:
            if n == best_n or best_v > 0.38085 + 1e-6:
                alloc = max(15.0, min(45.0, deadline_global - time.time()))
            else:
                alloc = max(20.0, min(60.0, deadline_global - time.time()))

        local_deadline = min(deadline_global, time.time() + alloc)
        if local_deadline - time.time() < 8.0:
            continue

        topk = int(min(2 * n - 1, max(24, int(n * 0.35))))
        start = upsample(best_h, best_n, n)

        init_temp = 0.09 if n == best_n else 0.07
        if best_v <= 0.38085:
            init_temp = min(init_temp, 0.06)

        first_frac = 0.70 if n == best_n else 0.60
        first_deadline = min(local_deadline, time.time() + first_frac * alloc)
        res_h, res_v = anneal_optimize(start, n, first_deadline, rng, topk, init_temp)
        if res_v < best_v - 1e-12:
            best_h = res_h.copy()
            best_n = int(n)
            best_v = res_v

        if local_deadline - time.time() > 12.0:
            start2 = upsample(best_h, best_n, n)
            noise = 0.05 if best_v <= 0.38085 else 0.10
            start2 = project_fast(start2 + rng.normal(0.0, noise, n), n)
            res_h2, res_v2 = anneal_optimize(start2, n, local_deadline, rng, topk, 0.055)
            if res_v2 < best_v - 1e-12:
                best_h = res_h2.copy()
                best_n = int(n)
                best_v = res_v2

    if time.time() < deadline_global - 5.0:
        topk = int(min(2 * best_n - 1, max(24, int(best_n * 0.35))))
        final_deadline = min(deadline_global, time.time() + max(12.0, deadline_global - time.time()))
        res_h, res_v = anneal_optimize(upsample(best_h, best_n, best_n), best_n, final_deadline, rng, topk, 0.005)
        if res_v < best_v - 1e-12:
            best_h = res_h.copy()
            best_v = res_v

    if time.time() < deadline_global - 30.0:
        n2 = best_n + max(2, int(best_n * 0.5))
        if n2 % 2 == 1:
            n2 += 1
        if best_n < n2 <= 700:
            topk2 = int(min(2 * n2 - 1, max(24, int(n2 * 0.35))))
            res_h2, res_v2 = anneal_optimize(
                upsample(best_h, best_n, n2),
                n2,
                min(deadline_global, time.time() + 45.0),
                rng,
                topk2,
                0.06
            )
            if res_v2 < best_v - 1e-12:
                best_h = res_h2.copy()
                best_n = int(n2)
                best_v = res_v2

    best_h = make_sum_exact(best_h, best_n)
    best_v = float(np.max(compute_c(best_h, best_n)))

    if not np.isfinite(best_v) or best_v > 0.9:
        best_n = max(2, int(n_base))
        if best_n % 2 == 1:
            best_n += 1
        best_h = project_robust(np.full(best_n, 0.5), best_n)
        best_v = eval_c5(best_h, best_n)

    return (best_h, float(best_v), int(best_n))
