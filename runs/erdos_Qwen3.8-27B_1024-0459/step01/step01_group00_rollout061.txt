def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Exact fix
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-14:
        # Add to the entry closest to 0.5
        idx = np.argmin(np.abs(h - 0.5))
        if 0 <= h[idx] + diff <= 1:
            h[idx] += diff
        else:
            # Distribute
            mask = (h > 0) & (h < 1)
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h[mask] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
    return h

    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted = np.clip(shifted, -500, 500)

def smooth_max_gradient(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted = np.clip(shifted, -500, 500)
    p = np.exp(shifted)
    p /= np.sum(p)
    
    grad = np.zeros(n)
    # Only loop over significant p values
    significant = np.where(p > 1e-8)[0]
    for k in significant:
        s = k - (n - 1)
        w = p[k]
        if s >= 0:
            # (1-h[j+s]) for j in [0, n-1-s]
            if n - 1 - s >= 0:
                grad[:n-s] += w * (1.0 - h[s:])
            # -h[j-s] for j in [s, n-1]
            if s <= n-1:
                grad[s:] -= w * h[:n-s]
        else:
            s2 = -s  # s2 > 0
            # (1-h[j-s2]) for j in [s2, n-1]
            if s2 <= n-1:
                grad[s2:] += w * (1.0 - h[:n-s2])
            # -h[j+s2] for j in [0, n-1-s2]
            if n-1-s2 >= 0:
                grad[:n-s2] -= w * h[s2:]
    return grad

def projected_gradient_descent(h_init, n, beta, max_iter=5000, lr=None, verbose=False):
    h = h_init.copy()
    if lr is None:
        lr = 1.0 / (beta * n)
    
    best_h = h.copy()
    best_val = smooth_max_value(h, beta)
    
    for it in range(max_iter):
        grad = smooth_max_gradient(h, beta)
        
        # Backtracking line search
        step = lr
        h_new = h - step * grad
        h_new = make_feasible(h_new, n)
        val_new = smooth_max_value(h_new, beta)
        
        tries = 0

            step *= 0.5
            h_new = h - step * grad
            h_new = make_feasible(h_new, n)
            val_new = smooth_max_value(h_new, beta)

        h = h_new

            best_h = h.copy()
        
        if verbose and it % 500 == 0:
            print(f"  iter {it}, val={best_val:.8f}, step={step:.2e}")
    
    return best_h, best_val, best_h

            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02, 0.001, -0.001]:

                h_new[j] = h_save + delta

                        adj = -diff
                        if 0 <= h_new[idx2] + adj <= 1:

                        else:

                    else:

                h_new = make_feasible(h_new, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    h = h_new
                    best_c5 = c5_new
                    improved = True
                    break
        if not improved:

    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-15
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-14:
        idx = np.argmin(np.abs(h - 0.5))
        if 0 <= h[idx] + diff <= 1:
            h[idx] += diff
        else:
            mask = (h > 1e-8) & (h < 1.0 - 1e-8)
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h[mask] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def smooth_max_value(h, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted = np.clip(shifted, -500, 500)
    return cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))

def smooth_max_gradient(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted = np.clip(shifted, -500, 500)
    p = np.exp(shifted)
    p /= np.sum(p)
    grad = np.zeros(n)
    significant = np.where(p > 1e-10)[0]
    for k in significant:
        s = k - (n - 1)
        w = p[k]
        if s >= 0:
            if n - 1 - s >= 0:
                grad[:n - s] += w * (1.0 - h[s:])
            if s <= n - 1:
                grad[s:] -= w * h[:n - s]
        else:
            s2 = -s
            if s2 <= n - 1:
                grad[s2:] += w * (1.0 - h[:n - s2])
            if n - 1 - s2 >= 0:
                grad[:n - s2] -= w * h[s2:]
    return grad

def projected_gradient_descent(h_init, n, beta, max_iter=3000, lr_scale=1.0):
    h = h_init.copy()
    lr = lr_scale / (beta * n)
    best_h = h.copy()
    best_val = smooth_max_value(h, beta)
    stalled = 0
    for it in range(max_iter):
        grad = smooth_max_gradient(h, beta)
        step = lr
        h_new = h - step * grad
        h_new = make_feasible(h_new, n)
        val_new = smooth_max_value(h_new, beta)
        tries = 0
        while val_new > best_val - 1e-15 and tries < 25 and step > 1e-12:
            step *= 0.5
            h_new = h - step * grad
            h_new = make_feasible(h_new, n)
            val_new = smooth_max_value(h_new, beta)
            tries += 1
        if tries >= 25:
            break
        h = h_new.copy()
        if val_new < best_val - 1e-15:
            best_val = val_new
            best_h = h.copy()
            stalled = 0
        else:
            stalled += 1
        if stalled > 200:
            break
    return best_h, best_val

def coordinate_descent_polish(h, n, max_rounds=80):
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    target = float(n) / 2.0
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            h_save = h[j]
            for delta in [0.005, -0.005, 0.01, -0.01, 0.02, -0.02, 0.002, -0.002]:
                h_new = h.copy()
                h_new[j] = h_save + delta
                if h_new[j] < -0.01 or h_new[j] > 1.01:
                    continue
                h_new = np.clip(h_new, 0.0, 1.0)
                diff = target - np.sum(h_new)
                if abs(diff) > 1e-10:
                    mask = (h_new > 0.02) & (h_new < 0.98)
                    mask[j] = False
                    if not np.any(mask):
                        continue
                    idxs = np.where(mask)[0]
                    idx2 = idxs[np.argmin(np.abs(h_new[idxs] - 0.5))]
                    adj = -diff
                    if -0.01 <= h_new[idx2] + adj <= 1.01:
                        h_new[idx2] += adj
                        h_new = np.clip(h_new, 0.0, 1.0)
                    else:
                        continue
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    h = h_new
                    best_c5 = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def random_local_search(h, n, n_trials, step_size, rng):
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    target = float(n) / 2.0

        n_mod = rng.randint(1, 5)
