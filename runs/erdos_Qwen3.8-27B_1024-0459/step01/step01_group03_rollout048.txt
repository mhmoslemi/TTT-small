import numpy as np
from scipy.optimize import minimize

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float('inf')
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float('inf')
    return float(np.max(corr) * dx)

def project_density(x, target):
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) <= 1e-13 * max(1, n):
        return x
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        free = np.where((h > 1e-14) & (h < 1.0 - 1e-14))[0]
        if free.size > 0:
            if res > 0:
                idx = int(free[np.argmax(1.0 - h[free])])
            else:
                idx = int(free[np.argmax(h[free])])
            adjust = min(res, 1.0 - h[idx]) if res > 0 else max(res, -h[idx])
            h[idx] += adjust
            res -= adjust
    if abs(res) > 1e-10:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size > 0:
            per = res / float(free.size)
            h[free] = np.clip(h[free] + per, 0.0, 1.0)
    return h

def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    n = int(n)
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)

def lag_gradient(h, lag):
    n = h.size
    g = np.zeros(n, dtype=float)
    if abs(lag) >= n:
        return g
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
        return g
    p = abs(lag)
    if lag > 0:
        g[p:] += (1.0 - h[:n - p])
        g[:n - p] -= h[p:]
    else:
        g[:n - p] += (1.0 - h[p:])
        g[p:] -= h[:n - p]
    return g

def topk_gradient(h, k=5):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[-k:]
    center = n - 1
    g = np.zeros(n, dtype=float)
    for t in idx:
        lag = int(t) - center
        g += lag_gradient(h, lag)
    return g / float(k)

def smooth_max_obj(x, n, alpha):

    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    return m + float(np.sum(np.exp(y))) / alpha

def smooth_max_grad(x, n, alpha):
    h = np.asarray(x, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    w = e / z
    center = n - 1
    g = np.zeros(n, dtype=float)
    for t in range(len(corr)):

                g[:n - p] -= wt * h[p:]
            else:
                g[:n - p] += wt * (1.0 - h[p:])
                g[p:] -= wt * h[:n - p]
    return g

def sum_fun(x, target):
    return float(np.sum(x) - target)

            eta = min(eta * 1.1, 0.5)
        else:
            stall += 1
            eta *= 0.7
            if eta < 1e-5:
                break
            if stall > 20:
                break
    
    return best, best_c

def coordinate_descent(h, target, deadline, max_sweeps=50):
    n = h.size
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    order = np.random.default_rng(12345).permutation(n)
    
    for sweep in range(int(max_sweeps)):
        if time.perf_counter() >= deadline:
            break
        for i in order:
            i = int(i)
            corr = np.correlate(best, 1.0 - best, mode="full")
            k_args = set()
            for kk in range(2 * n - 1):
                lag = kk - (n - 1)
                a = 0.0
                b_rest = 0.0
                # corr[lag] = sum_j h[j+lag]*(1-h[j])
                # terms with h[i]:
                # h[i] as h[j+lag]: j = i-lag, contrib (1-h[i-lag]) if valid
                # h[i] as (1-h[j]): j = i, contrib -h[i+lag] if valid
                # a = coeff of h[i]
                # For j = i-lag (h[i]*(1-h[j])): need 0<=i-lag<=n-1 and 0<=i<=n-1
                #   For this j to be in valid range: 0<=i-lag<=n-1 and lag valid
                val1 = 0.0
                val2 = 0.0
                if lag == 0:
                    a = 1.0 - 2.0 * best[i]
                else:
                    p = abs(lag)
                    if lag > 0:
                        if i >= p and i < n:
                            a = (1.0 - best[i - p])
                        if i < n - p:
                            a += (-best[i + p])
                    else:
                        if i < n - p:
                            a = (1.0 - best[i + p])
                        if i >= p:
                            a += (-best[i - p])
                k_args.add(float(a))
            
            # Simple subgradient step
            amax_lag = int(np.argmax(corr))
            lag_best = amax_lag - (n - 1)
            a_best = 0.0
            if lag_best == 0:
                a_best = 1.0 - 2.0 * best[i]
            else:
                p = abs(lag_best)
                if lag_best > 0:
                    if i >= p:
                        a_best = (1.0 - best[i - p])
                    if i < n - p:
                        a_best += (-best[i + p])
                else:
                    if i < n - p:
                        a_best = (1.0 - best[i + p])
                    if i >= p:
                        a_best += (-best[i - p])
            
            if a_best > 0.01:
                delta = -0.1 * a_best
            elif a_best < -0.01:
                delta = -0.1 * a_best
            else:
                continue
            
            cand = best.copy()
            cand[i] += delta
            cand = project_density(cand, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-13:
                best = cand
                best_c = c
    
    return best, best_c

def pair_swap_search(h, target, deadline, max_iter=3000, k_top=5):
    n = h.size
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    rng = np.random.default_rng(777)
    stall = 0
    
    for it in range(int(max_iter)):
        if time.perf_counter() >= deadline:
            break
        
        g = topk_gradient(best, k_top)
        g_centered = g - float(np.mean(g))
        
        neg_idx = np.argsort(g_centered)[:5]
        pos_idx = np.argsort(-g_centered)[:5]
        
        found = False
        for ii in neg_idx:
            ii = int(ii)
            for jj in pos_idx:
                jj = int(jj)
                if ii == jj:
                    continue
                cap_inc = min(1.0 - best[ii], best[jj])
                if cap_inc < 1e-10:
                    continue
                for frac in (1.0, 0.5, 0.25, 0.7):
                    d = frac * cap_inc
                    cand = best.copy()
                    cand[ii] += d
                    cand[jj] -= d
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-12:
                        best = cand
                        best_c = c
                        found = True
                        break
                if found:
                    break
            if found:
                break
        
        if not found:
            # Try random pair
            ii = int(rng.integers(0, n))
            jj = int(rng.integers(0, n))
            if ii != jj:
                direction = 1 if g_centered[ii] < g_centered[jj] else -1
                cap = min((1.0 - best[ii]) if direction > 0 else best[ii],
                          best[jj] if direction > 0 else (1.0 - best[jj]))
                if cap > 1e-10:
                    d = 0.3 * cap * direction
                    cand = best.copy()
                    cand[ii] += d
                    cand[jj] -= d
                    cand = project_density(cand, target)
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-12:
                        best = cand
                        best_c = c
                        found = True
        
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
    
    return best, best_c

def slsqp_optimize(h0, n, sub_deadline, target):
    h = project_density(h0, target)
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)
    now = time.perf_counter()
    rem = max(1.0, sub_deadline - now)
    
    bounds = [(0.0, 1.0)] * n
    constraints = ({"type": "eq", "fun": lambda x: sum_fun(x, target),
                    "jac": lambda x: sum_jac(x)},)
    
    alphas = [30.0, 80.0, 200.0]
    maxiters = [80, 60, 40]
    
    for idx_a, alpha in enumerate(alphas):
        if time.perf_counter() >= sub_deadline - 5.0:
            break
        try:
            res = minimize(
                lambda x: smooth_max_obj(x, n, alpha),
                h,
                method="SLSQP",
                jac=lambda x: smooth_max_grad(x, n, alpha),
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": maxiters[idx_a], "ftol": 1e-10, "disp": False}
            )
            xg = getattr(res, 'x', None)
            if xg is not None:
                cand = project_density(np.asarray(xg, dtype=float), target)
                c = evaluate_erdos(cand)
                if c < best_c:
                    best_h = cand
                    best_c = c
                if c <= best_c + 0.05:
                    h = cand
        except Exception:
            pass
    
    return best_h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 15.0)
    rng = np.random.default_rng(seed)
    
    best_h = None
    best_c = float('inf')
    best_n = 0
    
    # Load parent
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None
    
    if parent is not None:
        nparent = parent.size
        targetp = 0.5 * nparent
        pp = project_density(parent, targetp)
        cp = evaluate_erdos(pp)
        if cp < best_c:
            best_h = pp.copy()
            best_c = cp
            best_n = nparent
    
    if best_h is None:
        n0 = 90
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h0 = 0.5 + 0.3 * np.sin(2 * np.pi * 2 * x / 2) + 0.15 * np.sin(2 * np.pi * 5 * x / 2)
        best_h = project_density(h0, 0.5 * n0)
        best_c = evaluate_erdos(best_h)
        best_n = n0
    
    # Phase 1: Optimize at parent resolution (n=90)
    n1 = int(best_n)
    if deadline - time.perf_counter() > 80:
        t1_dead = min(deadline - (deadline - time.perf_counter()) * 0.15, time.perf_counter() + 70)
        h1, c1 = slsqp_optimize(best_h, n1, t1_dead, 0.5 * n1)
        if c1 < best_c:
            best_h, best_c, best_n = h1, c1, n1
        # Quick topk gradient
        if deadline - time.perf_counter() > 15:
            h1b, c1b = topk_gradient_descent(best_h, 0.5 * n1, time.perf_counter() + 15, k=5, max_iter=500)
            if c1b < best_c:
                best_h, best_c, best_n = h1b, c1b, n1
    
    # Phase 2: Optimize at 2x resolution (n=180)
    n2 = n1 * 2
    if n2 <= 400 and deadline - time.perf_counter() > 200:
        h2_init = upsample_h(best_h, n2)
        t2_dead = min(deadline - 100, time.perf_counter() + 250)
        h2, c2 = slsqp_optimize(h2_init, n2, t2_dead, 0.5 * n2)
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, n2
        # Top-k gradient descent
        if deadline - time.perf_counter() > 50:
            t2b = min(deadline - 50, time.perf_counter() + 80)
            h2b, c2b = topk_gradient_descent(best_h, 0.5 * n2, t2b, k=3, max_iter=3000)
            if c2b < best_c:
                best_h, best_c, best_n = h2b, c2b, n2
        # Pair swap
        if deadline - time.perf_counter() > 40:
            t2c = min(deadline - 30, time.perf_counter() + 60)
            h2c, c2c = pair_swap_search(best_h, 0.5 * n2, t2c, max_iter=2000, k_top=3)
            if c2c < best_c:
                best_h, best_c, best_n = h2c, c2c, n2
        # More gradient with different k
        if deadline - time.perf_counter() > 40:
            t2d = min(deadline - 20, time.perf_counter() + 50)
            h2d, c2d = topk_gradient_descent(best_h, 0.5 * n2, t2d, k=8, max_iter=2000)
            if c2d < best_c:
                best_h, best_c, best_n = h2d, c2d, n2
    
    # Phase 3: Try 4x resolution (n=360)
    n3 = n1 * 4
    if n3 <= 720 and deadline - time.perf_counter() > 150:
        h3_init = upsample_h(best_h, n3)
        t3_dead = min(deadline - 60, time.perf_counter() + 150)
        h3, c3 = slsqp_optimize(h3_init, n3, t3_dead, 0.5 * n3)
        if c3 < best_c:
            best_h, best_c, best_n = h3, c3, n3
        if deadline - time.perf_counter() > 40:
            t3b = min(deadline - 30, time.perf_counter() + 50)
            h3b, c3b = topk_gradient_descent(best_h, 0.5 * n3, t3b, k=3, max_iter=2000)
            if c3b < best_c:
                best_h, best_c, best_n = h3b, c3b, n3
    
    # Phase 4: Try different starting points at best resolution
    if deadline - time.perf_counter() > 100:
        nb = int(best_n)
        # Random smooth start
        for restart in range(3):
            if deadline - time.perf_counter() < 25:
                break
            x = (np.arange(nb) + 0.5) * (2.0 / nb)
            h_rand = 0.5
            for m in range(1, 8):
                amp = rng.uniform(0.05, 0.3) * (0.6 ** m)
                freq = int(max(1, round(m * rng.uniform(0.8, 1.5))))
                phase = rng.uniform(0, 2 * np.pi)
                h_rand = h_rand + amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
            h_rand = project_density(h_rand, 0.5 * nb)
            
            t_r = min(deadline - 20, time.perf_counter() + 25)
            hr, cr = slsqp_optimize(h_rand, nb, t_r, 0.5 * nb)
            if cr < best_c:
                best_h, best_c, best_n = hr, cr, nb
            if deadline - time.perf_counter() > 15:
                hr2, cr2 = topk_gradient_descent(hr, 0.5 * nb, time.perf_counter() + 15, k=4, max_iter=500)
                if cr2 < best_c:
                    best_h, best_c, best_n = hr2, cr2, nb
    
    # Phase 5: Final aggressive polish
    if best_h is not None and deadline - time.perf_counter() > 30:
        nb = int(best_h.size)
        target = 0.5 * nb
        # Pair swap final
        if deadline - time.perf_counter() > 25:
            tf = min(deadline - 10, time.perf_counter() + 25)
            hf, cf = pair_swap_search(best_h, target, tf, max_iter=5000, k_top=3)
            if cf < best_c:
                best_h, best_c, best_n = hf, cf, nb
        # Top-k gradient final
        if deadline - time.perf_counter() > 15:
            tf2 = min(deadline - 5, time.perf_counter() + 15)
            hf2, cf2 = topk_gradient_descent(best_h, target, tf2, k=3, max_iter=3000)
            if cf2 < best_c:
                best_h, best_c, best_n = hf2, cf2, nb
    
    # Final: ensure feasibility and return
    if best_h is None:
        fallback = project_density(np.full(90, 0.5), 45.0)
        return fallback, evaluate_erdos(fallback), 90
    
    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
