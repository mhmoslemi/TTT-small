    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def to_binary(h, n):
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:n // 2]] = 1.0
    return b

def thue_morse_binary(n):
    tm = np.zeros(n)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    # Adjust to have exactly n/2 ones
    return to_binary(tm, n)

def periodic_binary(n, period):
    h = np.zeros(n)
    half = period // 2
    for i in range(n):
        h[i] = 1.0 if (i % period) < half else 0.0
    return to_binary(h, n)

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def fix_sum(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0

def smooth_max_obj(x, n, dx, tau):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / tau
    shifted = np.clip(shifted, -50.0, 50.0)
    s = np.sum(np.exp(shifted))
    return float((m + tau * np.log(s)) * dx)


def eq_con(x, n):
    return np.array([x.sum() - n / 2.0])


def eq_jac(x, n):
    return np.ones((1, n))

        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def to_binary(h, n):

    return b
