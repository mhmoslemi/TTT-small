from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_onto_set(v, n):
    target = n / 2.0
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2, 0.0, 1.0)
    return h

def smooth_max_val_and_weights(x, beta):
    """Returns (smooth_max_value, probability_weights) for the log-sum-exp based smooth max"""
    bx = beta * x
    m = np.max(bx)
    e = np.exp(bx - m)
    p = e / np.sum(e)
    val = (m + np.log(np.sum(e))) / beta
    return val, p

def compute_smooth_c5_and_grad(h, beta):
    """Compute smooth C5 and its gradient using FFT-based convolution"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val, p = smooth_max_val_and_weights(corr, beta)
    
    # Gradient computation via convolution
    # grad[t] = dx * (conv(1-h, p)[t+n-1] - conv(h, p)[t+n-1])
    conv_1h = np.convolve(1.0 - h, p, mode='full')
    conv_h = np.convolve(h, p, mode='full')
    
    # We need indices n-1 to 2n-2
    grad = (conv_1h[n-1:2*n-1] - conv_h[n-1:2*n-1]) * dx
    
    return val * dx, grad

def subgradient_method(h, n, max_iter=5000, learning_rate=0.01, decay=0.999):
    """Subgradient optimization for the max correlation problem"""
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = learning_rate
    
    for it in range(max_iter):
        c5, corr = compute_c5(h), np.correlate(h, 1.0-h, mode='full')
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Subgradient: gradient of the active constraint
        k = np.argmax(corr)
        j = k - (n - 1)  # the lag
        n = len(h)
        
        # Compute gradient of corr[k] wrt h
        # corr[k] = sum_i h[i+j] * (1-h[i]) for valid i
        # d/dh[t] = (1-h[t-j])*[0<=t-j<n] - h[t+j]*[0<=t+j<n] (where j=k-(n-1))
        grad = np.zeros(n)
        for t in range(n):
            if 0 <= t - j < n:
                grad[t] += (1.0 - h[t-j])
            if 0 <= t + j < n:
                grad[t] -= h[t+j]
        
        h_new = project_onto_set(h - lr * grad, n)
        h = h_new
        lr *= decay
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_onto_set(v, n):
    """Project v onto {h : 0<=h[i]<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo = v.min() - 3.0
    hi = v.max() + 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def softplus_max_and_grad(x, beta):
    """Compute smooth max and gradient (probability weights)"""
    bx = beta * x
    m = np.max(bx)
    e = np.exp(bx - m)
    p = e / np.sum(e)
    val = (m + np.log(np.sum(e))) / beta
    return val, p

def smooth_obj_and_grad(h, n, beta):
    """Smooth C5 objective and gradient"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val, p = softplus_max_and_grad(corr, beta)
    
    # Gradient via convolution
    conv_1h = np.convolve(1.0 - h, p, mode='full')
    conv_h = np.convolve(h, p, mode='full')
    grad = (conv_1h[n-1:2*n-1] - conv_h[n-1:2*n-1]) * dx
    
    return val * dx, grad

def run_lbfgs(h_init, n, beta_seq, time_budget, start_time):
    """Run L-BFGS-B with annealed beta"""
    h = make_feasible(h_init.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    t_alloc = time_budget / len(beta_seq)
    
    for beta in beta_seq:
        t_start = time.time()
        max_iter = int(t_alloc / 0.001)  # estimate
        
        def obj_func(x):
            val, _ = smooth_obj_and_grad(x, n, beta)
            return val
        
        def grad_func(x):
            _, g = smooth_obj_and_grad(x, n, beta)
            return g
        
        # Wrap with constraint handling
        # L-BFGS-B handles box constraints natively
        # For the sum constraint, we use a penalty or reparameterize
        
        # Actually, let's add the sum constraint as a penalty
        def obj_constrained(x):
            x_c = np.clip(x, 0, 1)
            sum_pen = 100.0 * (np.sum(x_c) - n/2.0)**2
            val, _ = smooth_obj_and_grad(x_c, n, beta)
            return val + sum_pen
        
        def grad_constrained(x):
            x_c = np.clip(x, 0, 1)
            _, g = smooth_obj_and_grad(x_c, n, beta)
            sum_grad = 200.0 * (np.sum(x_c) - n/2.0)
            return g + sum_grad * np.ones_like(x_c)
        
        try:
            res = minimize(obj_constrained, h, jac=grad_constrained,
                         method='L-BFGS-B',
                         bounds=[(0,1)]*n,
                         options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12})
            h_new = make_feasible(res.x, n)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
        
        # Check time
        if time.time() - start_time > budget_s - 30:
            break
    
    return best_h, best_c5

```python
def compute_gradient_direct(h, p):
    """Compute gradient of sum_m p[m]*corr[m] wrt h"""
    n = len(h)
    grad = np.zeros(n)
    for m in range(2*n-1):
        if p[m] == 0:
            continue
        j = m - (n-1)  # the lag
        # d corr[m]/dh[t] = (1-h[t-j])*[0<=t-j<n] - h[t+j]*[0<=t+j<n]
        # First term: (1-h[t-j]) for t-j in [0,n-1], i.e., t in [j, j+n-1] intersected with [0,n-1]
        t_lo = max(0, j)
        t_hi = min(n-1, j+n-1)
        if t_lo <= t_hi:
            grad[t_lo:t_hi+1] += p[m] * (1.0 - h[t_lo-j:t_hi-j+1])
        # Second term: -h[t+j] for t+j in [0,n-1], i.e., t in [-j, n-1-j] intersected with [0,n-1]
        t_lo2 = max(0, -j)
        t_hi2 = min(n-1, n-1-j)
        if t_lo2 <= t_hi2:
            grad[t_lo2:t_hi2+1] -= p[m] * h[t_lo2+j:t_hi2+j+1]
    return grad
```
