Now let me also think about the `smooth_max_grad` more carefully. The formula I derived:

For k ≥ 0 (k_idx = n-1+k):
  grad_k[j] = dx * [(1-h[j+k]) for j ≤ n-1-k] - dx * [h[j-k] for j ≥ k]
  
  In array form:
  term1 = zeros(n); term1[0:n-k] = 1 - h[k:n]
  term2 = zeros(n); term2[k:n] = h[0:n-k]
  grad_k = dx * (term1 - term2)

For k < 0 (p = -k, k_idx = n-1-k = n-1+p):
  grad_{-p}[j] = dx * [(1-h[j-p]) for j ≥ p] - dx * [h[j+p] for j ≤ n-1-p]
  
  In array form:
  term1 = zeros(n); term1[p:n] = 1 - h[0:n-p]
  term2 = zeros(n); term2[0:n-p] = h[p:n]
  grad_{-p} = dx * (term1 - term2)

The total gradient is:
grad = sum_k w_k * grad_k

where w_k are the softmax weights.

```python
def smooth_max_grad(h, n, temp):
    dx = 2.0 / n
    shifts = compute_all_shifts(h, n)
    # softmax weights
    s = (shifts - np.max(shifts)) / temp
    w = np.exp(s)
    w /= w.sum()
    
    grad = np.zeros(n)
    for k_idx in range(2*n - 1):
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            if n - k > 0:
                grad[:n-k] += wk * dx * (1.0 - h[k:])
                grad[k:] -= wk * dx * h[:n-k]
        else:
            p = -k
            if n - p > 0:
                grad[p:] += wk * dx * (1.0 - h[:n-p])
                grad[:n-p] -= wk * dx * h[p:]
    
    return grad
```

```python
def compute_all_shifts(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def compute_c5(h, n):
    return np.max(compute_all_shifts(h, n))


def smooth_max_val(values, temp):
    m = np.max(values)
    s = (values - m) / temp
    return m + temp * np.log(np.exp(s).sum())


def smooth_max_grad(h, n, temp):
    dx = 2.0 / n
    shifts = compute_all_shifts(h, n)
    s = (shifts - np.max(shifts)) / temp

        wk = w[k_idx]

            p = -k

    target = n / 2.0
    h = np.array(h, dtype=float).copy()

        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] = np.maximum(h[mask] + diff / mask.sum(), 0.0)
    h = np.clip(h, 0.0, 1.0)
    h += (target - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    return h

                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14,
                                'gtol': 1e-12, 'maxfun': maxiter * 3})

    except Exception:

def active_grad_slsqp_obj(x, n):

    conv = np.correlate(x, 1.0 - x, mode='full')
    k_max = int(np.argmax(conv))

def active_grad_eval(x, n):

def slsqp_exact(h0, n, maxiter=300, ftol=1e-12):
    target = n / 2.0

    def c_fun(x):
        return x.sum() - target

    def c_jac(x):
        return np.ones(n)
```
