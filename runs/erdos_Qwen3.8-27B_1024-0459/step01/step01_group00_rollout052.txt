- corr[n-1-k] = Σ_{i=0}^{n-1-k} h[i](1-h[i+k])  [shift -k]

These are the two "directions" of overlap. The maximum C₅ takes the max over all of these.

Now, the gradient:
∂corr[j]/∂h[m]:
For j = n-1 (zero shift): 
corr[n-1] = Σ_i h[i](1-h[i])
∂corr[n-1]/∂h[m] = (1-2h[m])

For j = n-1+k (k>0):
corr[n-1+k] = Σ_{i=0}^{n-1-k} h[i+k](1-h[i])
∂corr[n-1+k]/∂h[m]:
- h[m] appears as h[i+k] when i = m-k (if 0≤m-k≤n-1-k, i.e., k≤m≤n-1): contributes (1-h[m-k])
- h[m] appears as h[i] when i = m (if 0≤m≤n-1-k): contributes -h[m+k]
So ∂corr[n-1+k]/∂h[m] = (1-h[m-k])*I(k≤m≤n-1) - h[m+k]*I(0≤m≤n-1-k)

For j = n-1-k (k>0):
corr[n-1-k] = Σ_{i=0}^{n-1-k} h[i](1-h[i+k])
∂corr[n-1-k]/∂h[m]:
- h[m] appears as h[i] when i = m (if 0≤m≤n-1-k): contributes (1-h[m+k])
- h[m] appears as h[i+k] when i = m-k (if 0≤m-k≤n-1-k, i.e., k≤m≤n-1): contributes -h[m-k]
So ∂corr[n-1-k]/∂h[m] = (1-h[m+k])*I(0≤m≤n-1-k) - h[m-k]*I(k≤m≤n-1)

Great, now I can compute the gradient efficiently.

grad[m] = Σ_j w[j] * ∂corr[j]/∂h[m]

Let me split this:
grad[m] = w[n-1]*(1-2*h[m]) + Σ_{k=1}^{n-1} w[n-1+k]*((1-h[m-k])*I(k≤m≤n-1) - h[m+k]*I(0≤m≤n-1-k))

This can be computed in O(n²) with numpy vectorization, or potentially O(n log n) with FFT.

    return np.interp(x_new, x_old, h_src)

def gen_structured_inits(n, rng):
    inits = []
    if n % 4 == 0:
        h1 = np.tile([1.0, 1.0, 0.0, 0.0], n // 4)
        inits.append(make_feasible(h1, n))
        h2 = np.tile([1.0, 0.0, 1.0, 0.0], n // 4)
        inits.append(make_feasible(h2, n))
    if n % 2 == 0:
        h3 = np.zeros(n)
        q = n // 4
        h3[:q] = 1.0
        h3[n//2:n//2+q] = 1.0
        inits.append(make_feasible(h3, n))
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    inits.append(make_feasible(tm, n))
    h_freq = 0.5 + 0.5 * np.cos(2.0 * np.pi * np.arange(n) * 3.0 / n)
    inits.append(make_feasible(h_freq, n))
    h_wave = 0.5 + 0.4 * np.sin(2.0 * np.pi * np.arange(n) * 5.0 / n) + 0.1 * np.sin(2.0 * np.pi * np.arange(n) * 11.0 / n)
    inits.append(make_feasible(h_wave, n))
    return inits

def local_perturb(h, n, n_trials, step_sz, t_start, t_limit):
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    for _ in range(n_trials):
        if time.time() - t_start > t_limit:
            break
        h_try = h.copy()
        n_mod = np.random.randint(1, 7)
        idxs = np.random.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += np.random.uniform(-step_sz, step_sz)
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = fix_sum(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            h = h_try
    return h, best_c5

def slsqp_optimize(h_init, n, max_iter):
    h0 = make_feasible(h_init.copy(), n)
    bounds = [(0.0, 1.0)] * n
    sum_val = np.sum(h0) - n / 2.0
    beta = 100.0
    def objective(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        cm = np.max(corr)
        return cm + (1.0/beta) * np.log(np.sum(np.exp(beta * (corr - cm))))
    def obj_grad(x):
        f, g = sm_obj_grad(x, beta)
        return f, np.concatenate([g, [np.sum(x) - n/2.0]], axis=0)
    try:
        res = minimize(objective, h0, method='L-BFGS-B', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}],
                      options={'maxiter': max_iter, 'ftol': 1e-14})
        h_out = make_feasible(res.x, n)
        return h_out, compute_c5(h_out)
    except Exception:
        return None, None

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()

    def remaining():
        return budget_s - (time.time() - t_start)

    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is not None and len(h_parent) == 0:
            h_parent = None
    except Exception:
        h_parent = None

    if h_parent is not None:
        best_h = h_parent.copy()
        best_n = len(best_h)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5
    best_c5 = compute_c5(best_h)

    t_limit_main = t_start + budget_s * 0.82
    t_limit_polish = t_start + budget_s * 0.95

    resolutions = [160, 192, 200, 224, 256]
    if h_parent is not None and len(h_parent) not in resolutions:
        resolutions.insert(0, len(h_parent))
    resolutions = sorted(set(resolutions))

    for n in resolutions:
        if remaining() < 60:
            break
        t_limit_res = min(t_start + remaining() * 0.6, t_limit_main)

        inits = []
        h_interp = interpolate_to_n(best_h, n)
        inits.append(make_feasible(h_interp, n))

        if h_parent is not None and len(h_parent) != n:
            h_p = interpolate_to_n(h_parent, n)
            h_p = make_feasible(h_p, n)
            if np.max(np.abs(h_p - inits[0])) > 0.02:
                inits.append(h_p)

        struct = gen_structured_inits(n, rng)
        for s in struct[:4]:
            inits.append(s)

        for _ in range(2):
            h_r = rng.rand(n)
            inits.append(make_feasible(h_r, n))

        unique_inits = [inits[0]]
        for h_i in inits[1:]:
            if all(np.max(np.abs(h_i - u)) > 0.03 for u in unique_inits):
                unique_inits.append(h_i)
        inits = unique_inits[:6]

        for h0 in inits:
            if remaining() < 30:
                break
            t_lim = min(t_start + remaining() * 0.4, t_limit_res)
            h_ad, c5_ad = adam_optimize(h0, n, 3000, 5.0, 200.0, t_start, t_lim)
            if c5_ad < best_c5 - 1e-12:
                best_c5 = c5_ad
                best_h = h_ad.copy()
                best_n = n

        if remaining() < 40:
            break
        t_lim2 = min(t_start + remaining() * 0.35, t_limit_main)
        h_best_at_n = interpolate_to_n(best_h, n)
        h_ad2, c5_ad2 = adam_optimize(h_best_at_n, n, 4000, 10.0, 300.0, t_start, t_lim2)
        if c5_ad2 < best_c5 - 1e-12:
            best_c5 = c5_ad2
            best_h = h_ad2.copy()
            best_n = n

    if remaining() < 20:
        best_h = make_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)
        return (best_h, best_c5, best_n)

    t_polish_start = time.time()
    t_polish = min(remaining() * 0.4, budget_s * 0.12)

    h_fin, c5_fin = sgd_optimize(best_h, best_n, 5000, 50.0, 500.0, t_start, t_polish_start + t_polish)
    if c5_fin < best_c5:
        best_c5 = c5_fin
        best_h = h_fin.copy()

    if remaining() > 30:
        t_pert = min(time.time() + remaining() * 0.3, t_limit_polish)
        n_trials = min(8000, int(remaining() * 200))
        best_h, best_c5 = local_perturb(best_h, best_n, n_trials, 0.015, t_start, t_pert)

    if remaining() > 40:
        t_polish2 = min(time.time() + remaining() * 0.35, t_limit_polish)
        h_fin2, c5_fin2 = adam_optimize(best_h, best_n, 3000, 100.0, 800.0, t_start, t_polish2)
        if c5_fin2 < best_c5:
            best_c5 = c5_fin2
            best_h = h_fin2.copy()

    if remaining() > 20:
        t_pert2 = min(time.time() + remaining() * 0.4, t_limit_polish)
        n_trials2 = min(5000, int(remaining() * 150))
        best_h, best_c5 = local_perturb(best_h, best_n, n_trials2, 0.008, t_start, t_pert2)

    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
