    corr = np.correlate(h, 1.0 - h, mode="full")

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h

def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def obj_and_grad(x, n, dx, tau, n_top=50):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / tau
    # For numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    val = (m + tau * np.log(sum_exp)) * dx
    
    if tau < 1e-8:
        # Use exact max
        top_idx = np.where(corr == np.max(corr))[0]
        weights = np.zeros(len(corr))
        for i in top_idx:
            weights[i] = 1.0 / len(top_idx)
    else:
        weights = exp_shifted / sum_exp
        # Only keep top n_top
        if np.sum(weights > 1e-6) > n_top:
            threshold = np.sort(weights)[::-1][n_top-1]
            weights[weights < threshold] = 0.0

    for k in active:

        s = k - (n - 1)
        if s >= 0:

            if n - s2 > 0:
                grad[s2:] += w * (1.0 - h[:n-s2])
                grad[:n-s2] += w * (-h[s2:])
    grad *= dx
    return val, grad

    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    stalled = 0
    
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = np.max(corr)
        # Find active lags
        threshold = max_corr - 0.001
        active = np.where(corr >= threshold)[0]
        
        # Subgradient
        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            if s >= 0:
                if n - s > 0:
                    grad[:n-s] += (1.0 - h[s:])
                    grad[s:] += -h[:n-s]
            else:
                s2 = -s
                if n - s2 > 0:
                    grad[s2:] += (1.0 - h[:n-s2])
                    grad[:n-s2] += -h[s2:]
        grad /= len(active)
        grad *= dx

            h_new = fix_sum(h_new, n)

                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best = h.copy()
                    stalled = 0
                break
            lr *= 0.5
        else:
            if max_corr * dx - best_c5 < 1e-10:
                stalled += 1
                lr *= 0.8
            if stalled > 50:
                break
    
    return best, best_c5

def periodic_c5(pattern, p):
    """Compute C5 for a periodic pattern of period p"""
    # For large n relative to p, C5 ≈ 2/p * max_s sum_i pattern[i]*(1-pattern[(i+s)%p])
    max_val = 0
    for s in range(p):
        val = 0
        for i in range(p):
            val += pattern[i] * (1 - pattern[(i + s) % p])
        if val > max_val:
            max_val = val
    return 2.0 / p * max_val

def periodic_c5_grad(pattern, p, n_top=10):
    """Gradient of periodic C5 w.r.t. pattern"""

    for s in range(p):
        for i in range(p):

    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def corr_values(h, p):
    """Compute the p correlation values for periodic pattern"""
    vals = np.zeros(p)
    for s in range(p):
        vals[s] = np.sum(h * (1.0 - np.roll(h, -s)))
    return vals

def periodic_c5_val(pattern, p):
    vals = corr_values(pattern, p)
    return (2.0 / p) * np.max(vals)

def periodic_c5_grad(pattern, p):
    vals = corr_values(pattern, p)
    m = np.max(vals)
    active_mask = vals >= m - 1e-10
    n_active = np.sum(active_mask)
    grad = np.zeros(p)
    for s in range(p):
        if not active_mask[s]:
            continue
        shifted_pos = np.roll(pattern, -s)  # shifted_pos[j] = pattern[(j+s)%p]
        shifted_neg = np.roll(pattern, s)    # shifted_neg[j] = pattern[(j-s)%p]
        grad += (1.0 - shifted_pos) - shifted_neg
    grad /= n_active
    grad *= 2.0 / p
    return grad

def periodic_eq_jac(pattern, p):
    return np.ones((1, p))

def full_slqp_obj(x, n, dx, tau):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip((corr - m) / tau, -50, 0)
    val = m + tau * np.log(np.sum(np.exp(shifted)))
    return val * dx

def full_slqp_jac(x, n, dx, tau, n_top=30):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    if tau < 1e-8:
        top_mask = corr >= m - 1e-10
        n_top_actual = np.sum(top_mask)
        if n_top_actual == 0:
            top_mask[np.argmax(corr)] = True
        n_top_actual = np.sum(top_mask)
    else:
        shifted = np.clip((corr - m) / tau, -50, 0)
        exp_s = np.exp(shifted)
        weights = exp_s / np.sum(exp_s)
        # Keep top n_top
        if np.sum(weights > 0.01) > n_top:
            sorted_idx = np.argsort(weights)[::-1]
            weights = np.zeros_like(weights)
            weights[sorted_idx[:n_top]] = exp_s[sorted_idx[:n_top]] / np.sum(exp_s[sorted_idx[:n_top]])
        n_top_actual = np.sum(weights > 1e-8)
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k] if tau >= 1e-8 else (1.0 / n_top_actual)
        if w < 1e-10:
            continue
        s = k - (n - 1)
        if s >= 0:
            if n - s > 0:
                grad[:n-s] += w * (1.0 - h[s:])
                grad[s:] += w * (-h[:n-s])
        else:
            s2 = -s
            if n - s2 > 0:
                grad[s2:] += w * (1.0 - h[:n-s2])
                grad[:n-s2] += w * (-h[s2:])
    return grad * dx
