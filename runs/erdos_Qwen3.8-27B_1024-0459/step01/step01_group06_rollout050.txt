    c = _corr_full(h, n)

def _proj(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    for _ in range(200):
        d = h.sum() - target
        if abs(d) < 1e-14:

        if d > 0:
            m = h > 1e-14
        else:
            m = h < 1.0 - 1e-14

            break

    return h


def _grad_lag(h, n, lag):
    dx = 2.0 / n
    g = np.zeros(n)
    if lag >= 0:
        r = n - lag
        if r > 0:
            g[:r] += 1.0 - h[lag:lag + r]
            g[lag:] -= h[:r]
    else:
        m = -lag
        r = n - m
        if r > 0:
            g[m:] += 1.0 - h[:r]
            g[:r] -= h[m:]
    return g * dx

    c = _corr_full(h, n)
    k_max = int(np.argmax(c))
    lag = k_max - (n - 1)
    return _grad_lag(h, n, lag)

    c = _corr_full(h, n)
    idx = np.argpartition(c, -top_k)[-top_k:]
    c_top = c[idx]
    w = np.exp(beta * (c_top - c_top.max()))
    s = w.sum()
    if s < 1e-30:
        w = np.ones(len(idx)) / len(idx)
    else:
        w /= s

        val = evaluate_erdos_solution(best_h, len(best_h))
        if val is not None and 0.0 < val < 1.0:
            best_c5 = min(best_c5, float(val))
    except Exception:
        pass

    return (best_h, best_c5, len(best_h))
