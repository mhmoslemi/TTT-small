The key observation is that the optimal solution likely has few transitions between 0 and 1, which corresponds to a piecewise constant function with a small number of segments. This approach leverages the observation that sharp transitions (i.e., discontinuities in the function) tend to minimize the maximum overlap integral.

Unlike the previous methods that maintain continuous values, we discretize the solution space by allowing only two possible values for the function: 0 and 1. We then use a binary representation (where each value is either 0 or 1) with a fixed number of segments. The function will have a specified number of continuous regions where it remains constant (either 0 or 1).

This method addresses the optimization problem by:

3. Assigning a binary value (0 or 1) to each segment such that the integral constraint is satisfied
4. Evaluating the overlap integral for all possible shifts

def project_feasible_binary(h_flat, n_points, target_sum):
    # Ensure values are 0 or 1
    h_clip = np.clip(h_flat, 0, 1)
    h_01 = (np.round(h_clip) == 1).astype(float)
    
    # Adjust segment lengths to satisfy integral constraint
    current_sum = np.sum(h_01) * (2.0 / n_points)
    adjustment_factor = (1.0 - current_sum) / (n_points/2.0)
    
    # Create sorted indices by value
    idx = np.argsort(h_01)
    h_rounded = np.zeros(n_points)
    h_rounded[idx[:-int(n_points * adjustment_factor)]] = 1.0
    
    return h_rounded

from scipy.optimize import dual_annealing, shgo

def objective_function_optimize(x, n_points, N_segments):
    # x: binary decision variables (segment values)
    total_segments = N_segments
    
    # Create h function via interpolation
    x_interp = np.linspace(0, 2, n_points)
    h_vals = np.zeros(n_points)
    
    # Compute segment boundaries
    boundaries = np.linspace(0, 2, total_segments + 1)
    
    # Accumulate segment contributions
    cumulative_sum = 0
    for i in range(n_points):
        if x[i] > 0.5:
            # Find which segment this point falls in
            seg = np.searchsorted(boundaries, x_interp[i])
            if seg >= total_segments:
                seg = total_segments - 1
            
            # If last segment, add full value
            if seg == total_segments - 1:
                h_vals[i] = 1.0
            else:
                seg_length = (boundaries[seg+1] - boundaries[seg]) / n_points
                cumulative = cumulative_sum + h_vals[i] * seg_length
                h_vals[i] = min(1.0, (1.0 - cumulative) / seg_length)
                cumulative_sum = cumulative
    # project and then update
    h_vals = project_feasible_binary(x, n_points, N_segments)
    
    # Compute the correlation
    conv = np.correlate(h_vals, 1.0 - h_vals, mode='full')
    c5_val = np.max(conv) * (2.0 / n_points)
    return c5_val
