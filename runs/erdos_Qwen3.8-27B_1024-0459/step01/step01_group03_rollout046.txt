def _project(x, target):
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) <= 1e-14 * max(1, n):
        return x
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-12:
        free = np.where((h > 1e-15) & (h < 1.0 - 1e-15))[0]
        if free.size > 0:
            h[free] += res / float(free.size)
            h = np.clip(h, 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-15:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size > 0:
            idx = int(free[np.argmax(np.abs(h[free] - 0.5))])
            h[idx] += res
            h = np.clip(h, 0.0, 1.0)
    return h


def _eval(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _corr_grad(h, idxs):
    n = h.size
    g = np.zeros(n, dtype=float)
    center = n - 1
    for t in idxs:
        t = int(t)
        k = t - center
        if k == 0:
            g += (1.0 - 2.0 * h)
        else:
            lag = abs(k)
            if k > 0:
                g[lag:] += 1.0
            else:
                g[:n - lag] += 1.0
            g[:n - lag] -= h[lag:]
            g[lag:] -= h[:n - lag]
    return g


def _active_set(h, deadline, max_iter=400, k_act=4):
    n = h.size
    target = 0.5 * n
    h = _project(h, target)
    best = h.copy()
    best_c = _eval(best)
    if n < 2:
        return best, best_c
    stall = 0
    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        if not np.all(np.isfinite(corr)):
            break
        idxs = np.argsort(corr)[-k_act:]
        g = _corr_grad(best, idxs) / float(k_act)
        d = -g
        d -= float(np.mean(d))
        nv = float(np.linalg.norm(d))
        if nv < 1e-14:
            break
        d /= nv
        improved = False
        for step in (0.003, 0.008, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
            cand = _project(best + step * d, target)
            c = _eval(cand)
            if c < best_c - 1e-13:
                best = cand
                best_c = c
                improved = True
                break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall == 25 and k_act < 10:
                k_act = min(k_act + 2, 2 * n - 1)
            elif stall == 60 and k_act < 20:
                k_act = min(k_act + 5, 2 * n - 1)
            if stall > 80:
                break
    best = _project(best, target)
    best_c = _eval(best)
    return best, best_c


def _pair_search(h, deadline, max_iter=1500, k_act=5):
    n = h.size
    target = 0.5 * n
    h = _project(h, target)
    best = h.copy()
    best_c = _eval(best)
    if n < 2:

    stall = 0

        idxs = np.argsort(corr)[-k_act:]
        g = _corr_grad(best, idxs) / float(k_act)

        pos = np.argsort(-g)[:6]
        found = False
        for ii in neg:

                if ii == jj:

                    cand[ii] = min(1.0, best[ii] + dd)
                    cand[jj] = max(0.0, best[jj] - dd)
                    c = _eval(cand)
                    if c < best_c - 1e-13:

                        best_c = c
                        found = True
                        break
                if found:

            if found:

        if found:

            ok = False

                        found = True
                        ok = True

def _slsqp_obj(x):
    n = _G_N
    alpha = _G_ALPHA
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return 1e100
    corr = np.correlate(h, 1.0 - h, mode="full")
    if not np.all(np.isfinite(corr)):
        return 1e100
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return 1e100
    return float((2.0 / n) * (m + np.log(z) / alpha))


def _slsqp_jac(x):
    n = _G_N
    alpha = _G_ALPHA
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return np.zeros(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if not np.all(np.isfinite(corr)):
        return np.zeros(n)
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return np.zeros(n)
    w = e / z
    g = np.zeros(n, dtype=float)
    center = n - 1
    for t in range(2 * n - 1):
        p = w[t]
        if p < 1e-14:
            continue
        k = t - center
        if k == 0:
            g += p * (1.0 - 2.0 * h)
        else:
            lag = abs(k)
            if k > 0:
                g[lag:] += p
            else:
                g[:n - lag] += p
            g[:n - lag] -= p * h[lag:]
            g[lag:] -= p * h[:n - lag]
    return (2.0 / n) * g


def _slsqp_con(x):
    return float(np.sum(np.asarray(x, dtype=float)) - _G_TARGET)


def _slsqp_con_jac(x):
    return np.ones(_G_N)


def _slsqp_opt(x0, n, deadline, alphas=(10.0, 30.0, 80.0), maxiter=60):
    target = 0.5 * n
    best = _project(x0, target)
    best_c = _eval(best)
    _G_N = n
    _G_TARGET = target
    bounds = [(0.0, 1.0)] * n
    cos_arr = ({"type": "eq", "fun": _slsqp_con, "jac": _slsqp_con_jac},)
    for alpha in alphas:
        if time.perf_counter() >= deadline:
            break
        _G_ALPHA = alpha
        try:

                _slsqp_obj, best, method="SLSQP", jac=_slsqp_jac,

                    c = _eval(cand)

                        best_c = c
        except Exception:

def _upsample(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    n = int(n)
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return _project(np.full(n, 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return _project(h, 0.5 * n)


def _make_binary_periodic(n, period_len, rng):
    h = np.zeros(n)
    n_ones = period_len // 2
    if n_ones < 1:
        n_ones = 1
    ones = list(rng.choice(period_len, size=min(n_ones, period_len - 1), replace=False))
    ones_set = set(ones)
    for i in range(n):
        h[i] = 1.0 if (i % period_len) in ones_set else 0.0
    return _project(h, 0.5 * n)


def _make_random_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return _project(h, 0.5 * n)


def _make_smooth_random(n, rng):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.full(n, 0.5)
    for m in range(1, 7):
        amp = 0.3 * (0.65 ** (m - 1))
        freq = int(max(1, np.rint(m * rng.uniform(0.7, 1.4))))
        phase = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    return _project(h, 0.5 * n)


def _annealed_steps(h, deadline, max_iter=500, k_act=4, temp=0.002):

    target = 0.5 * n
    h = _project(h, target)
    best = h.copy()
    best_c = _eval(best)
    cur = h.copy()
    cur_c = best_c
    t = temp
    for _ in range(max_iter):

        corr = np.correlate(cur, 1.0 - cur, mode="full")

        d = -g
        d -= float(np.mean(d))
        nv = float(np.linalg.norm(d))
        if nv < 1e-14:

        d /= nv

        t = max(0.0001, t * 0.98)

    return best_h, float(best_c), int(best_h.size)
