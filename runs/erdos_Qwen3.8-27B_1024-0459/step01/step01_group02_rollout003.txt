def project_sum_bounds(x, target):
    x = np.asarray(x, dtype=float).reshape(-1)
    n = x.size
    if n == 0:
        return x
    target = float(target)

    x = np.clip(x, 0.0, 1.0)
    s0 = float(x.sum())
    if abs(s0 - target) <= 1e-12:
        return x

    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        s = float(y.sum())
        if s > target:
            lo = mid
        else:
            hi = mid

    lamb = 0.5 * (lo + hi)
    y = np.clip(x - lamb, 0.0, 1.0)
    diff = target - float(y.sum())

    if abs(diff) > 1e-12:
        free = (y > 1e-10) & (y < 1.0 - 1e-10)
        nf = int(np.count_nonzero(free))
        if nf > 0:
            y[free] += diff / float(nf)
        else:
            y += diff / float(n)
        y = np.clip(y, 0.0, 1.0)

        diff2 = target - float(y.sum())
        if abs(diff2) > 1e-13:
            slack = np.minimum(y, 1.0 - y)
            idx = int(np.argmax(slack))
            if slack[idx] > 1e-12:
                y[idx] += float(np.clip(diff2, -y[idx], 1.0 - y[idx]))
            else:
                y[idx] += diff2
            y = np.clip(y, 0.0, 1.0)

    return y


def compute_c5(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    val = float(np.max(corr)) * dx
    if not math.isfinite(val):
        return float("inf")
    return val


def soft_obj_grad(h, beta):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0:
        return float("inf"), np.zeros(0), np.zeros(0), np.zeros(0)

    if not np.all(np.isfinite(h)):
        return float("inf"), np.zeros(n), np.zeros(2 * n - 1), np.zeros(2 * n - 1)

    dx = 2.0 / float(n)
    b = 1.0 - h
    corr = np.correlate(h, b, mode="full")
    c = corr * dx
    m = float(np.max(c))

    beta = float(beta)
    if beta < 1e-8:
        weights = np.full(2 * n - 1, 1.0 / float(2 * n - 1))
        f = m
    else:
        scaled = beta * (c - m)
        scaled = np.clip(scaled, -700.0, 0.0)
        exps = np.exp(scaled)
        s = float(np.sum(exps))
        if s <= 0.0 or not math.isfinite(s):
            weights = np.full(2 * n - 1, 1.0 / float(2 * n - 1))
            f = m
        else:
            weights = exps / s
            f = m + math.log(s) / beta

    M = 2 * n - 1
    center = n - 1

    S1 = np.convolve(weights, b, mode="full")[center:center + n]
    wr = weights[::-1]
    S2 = np.convolve(h, wr, mode="full")[center:center + n]

    g = dx * (S1 - S2)
    g -= float(np.mean(g))
    return float(f), g, weights, c


def _slsqp_con(x):
    x = np.asarray(x, dtype=float).reshape(-1)
    return float(x.sum() - 0.5 * _slsqp_eq_n)


def _slsqp_fun(x):

    x = np.asarray(x, dtype=float).reshape(-1)

def slsqp_polish(h0, n, beta_seq, maxiter=100, time_limit=60.0):

    end = time.time() + float(time_limit)
    bounds = [(0.0, 1.0)] * n

    mac = int(maxiter)

        _slsqp_beta = float(beta)

        _slsqp_cache_g = None

        try:

                _slsqp_fun,
                h,

                bounds=bounds,

def resample_init(parent_h, n):
    parent_h = np.asarray(parent_h, dtype=float).reshape(-1)
    n = int(n)
    m = parent_h.size
    if n <= 0:
        return np.zeros(0)
    if m == 0:
        return np.full(n, 0.5)
    if m == n:
        return project_sum_bounds(parent_h, 0.5 * n)

    xp = (np.arange(m) + 0.5) * (2.0 / float(m))
    xn = (np.arange(n) + 0.5) * (2.0 / float(n))
    y = np.interp(xn, xp, parent_h)
    return project_sum_bounds(y, 0.5 * n)


def scores_to_binary(scores, n):
    scores = np.asarray(scores, dtype=float).reshape(-1)
    n = int(n)
    h = np.zeros(n, dtype=float)
    if n <= 0:
        return h

    k = n // 2
    order = np.argsort(scores)
    if k > 0:
        h[order[k:]] = 1.0
    if n % 2 == 1 and k < n:
        h[order[k]] = 0.5
    return h


def threshold_to_binary(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float).reshape(-1)
    if h.size != n:
        h = resample_init(h, n)
    return scores_to_binary(h, n)


def make_random_binary(n, seed, style):
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=float)
    rng = np.random.RandomState(int(seed) % (2**31 - 1))
    idx = np.arange(n, dtype=float)

    if style == 0:
        scores = rng.rand(n)
    elif style == 1:
        x = rng.randn(n)
        x = np.convolve(x, np.ones(3) / 3.0, mode="same")
        x = np.convolve(x, np.ones(5) / 5.0, mode="same")
        scores = x
    elif style == 2:
        x = rng.randn(n)
        f = 1.0 + float(rng.rand() * 3.0)
        ph = rng.rand() * 2.0 * math.pi
        scores = 0.6 * np.sin(2.0 * math.pi * f * idx / float(n) + ph) + 0.4 * x
    else:
        x = rng.randn(n)
        f1 = 1.0 + float(rng.rand() * 8.0)
        f2 = 1.0 + float(rng.rand() * 12.0)
        ph1 = rng.rand() * 2.0 * math.pi
        ph2 = rng.rand() * 2.0 * math.pi
        scores = (
            0.4 * np.sin(2.0 * math.pi * f1 * idx / float(n) + ph1)
            + 0.3 * np.sin(2.0 * math.pi * f2 * idx / float(n) + ph2)
            + 0.3 * x
        )

    return scores_to_binary(scores, n)


def build_neighbors(n):
    n = int(n)
    center = n - 1
    neigh = []
    for i in range(n):
        idx1 = []
        idx2 = []
        ks = []
        for k in range(n):
            if k == i:
                continue
            idx1.append(center + i - k)
            idx2.append(center + k - i)
            ks.append(k)
        neigh.append((idx1, idx2, ks))
    return neigh


def get_neighbors(n):
    n = int(n)
    if n not in _neigh_cache:
        _neigh_cache[n] = build_neighbors(n)
    return _neigh_cache[n]


def flip_binary_list(c, h, i, delta, neigh):
    idx1, idx2, ks = neigh[i]
    if delta > 0:
        for t in range(len(ks)):
            hk = h[ks[t]]
            if hk > 0.5:
                c[idx2[t]] -= 1.0
            else:
                c[idx1[t]] += 1.0
    else:

            hk = h[ks[t]]
            if hk > 0.5:
                c[idx2[t]] += 1.0
            else:
                c[idx1[t]] -= 1.0
    h[i] = 1.0 - h[i]


def binary_local_search(h0, n, time_limit, seed):

    current_max = best_max

    neigh = get_neighbors(n)
    ones = [i for i in range(n) if h[i] > 0.5]

    if len(ones) == 0 or len(zeros) == 0:
        return np.array(best_h, dtype=float), best_val

    rng = np.random.RandomState(int(seed) % (2**31 - 1))
    deadline = time.time() + float(time_limit)

        best_j = -1
        best_move_max = current_max
        broken = False

                broken = True
                break
            for j in zeros:

def best_variant(h, n):
    h = project_sum_bounds(np.asarray(h, dtype=float).reshape(-1), 0.5 * int(n))
    n = int(n)

    best_h = h.copy()
    best_val = compute_c5(best_h)

    hs = 0.5 * (h + h[::-1])
    hs = project_sum_bounds(hs, 0.5 * n)
    vs = compute_c5(hs)
    if vs < best_val - 1e-12:
        best_val = vs
        best_h = hs

    hc = 1.0 - h
    hc = project_sum_bounds(hc, 0.5 * n)
    vc = compute_c5(hc)
    if vc < best_val - 1e-12:
        best_val = vc
        best_h = hc

    hcs = 0.5 * (hc + hc[::-1])
    hcs = project_sum_bounds(hcs, 0.5 * n)
    vcs = compute_c5(hcs)
    if vcs < best_val - 1e-12:
        best_val = vcs
        best_h = hcs

    return best_h, best_val


def consider_candidate(h, n, best_h, best_c5, best_n):

            time_limit=min(120.0, rem * 0.55),
        )
        best_h, best_c5, best_n = consider_candidate(h2, best_n, best_h, best_c5, best_n)

        rem = hard_end - time.time()
        if rem > 60.0:
            h2, _ = slsqp_polish(
                best_h,
                best_n,
                [20.0, 100.0, 500.0, 2000.0],
                maxiter=130,
                time_limit=min(130.0, rem * 0.6),
            )
            best_h, best_c5, best_n = consider_candidate(h2, best_n, best_h, best_c5, best_n)

        rem = hard_end - time.time()
        if best_n * 2 <= 256 and rem > 80.0:
            n2 = int(best_n) * 2
            h2 = resample_init(best_h, n2)
            h2, _ = slsqp_polish(
                h2,
                n2,
                [10.0, 50.0, 200.0, 800.0],
                maxiter=130,
                time_limit=min(90.0, rem * 0.7),
            )
            best_h, best_c5, best_n = consider_candidate(h2, n2, best_h, best_c5, best_n)

    if best_h is None:
        best_n = 96
        best_h = project_sum_bounds(np.full(best_n, 0.5), 0.5 * best_n)
        best_c5 = compute_c5(best_h)

    best_h = project_sum_bounds(np.asarray(best_h, dtype=float).reshape(-1), 0.5 * int(best_n))
    final_h, final_c5 = best_variant(best_h, best_n)

    if "evaluate_erdos_solution" in globals():
        try:
            ev = evaluate_erdos_solution(final_h)
            if isinstance(ev, (tuple, list)):
                ev = ev[0]
            evf = float(ev)
            if math.isfinite(evf) and evf < final_c5:
                final_c5 = evf
        except Exception:
            pass

    if not math.isfinite(final_c5):
        final_c5 = compute_c5(final_h)

    return final_h, float(final_c5), int(best_n)
