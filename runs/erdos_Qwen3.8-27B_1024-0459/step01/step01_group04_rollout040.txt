def compute_c5(h):

    n = h.size

    h = np.clip(h, 0.0, 1.0)

def project_feasible(h, n):

    lo, hi = -2.0, 2.0
    for _ in range(60):

        s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s < target:

        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

    # fm(t) = corr[m] + (t-old)*c[m]  [linear]

    # = (corr[n-1] - old + old²) + t - t²

    B = corr.copy()

    # Find upper envelope of lines B[m] + c[m]*t for m ≠ n-1
    # and the quadratic A0 + t - t²
    
    # Generate candidate points
    candidates = [0.0, 1.0]
    
    # Quadratic vertex
    candidates.append(0.5)
    
    # Line-line intersections
    # For efficiency, only consider lines that could be on the upper envelope
    # For now, do all pairs (n is small)
    indices = np.concatenate([np.arange(n-1), np.arange(n, 2*n-1)])
    M = len(indices)

    t_samples = np.linspace(0.0, 1.0, 101)
    
    # Add quadratic-line intersections
    for m in indices:
        cm = c[m]
        Bm = B[m]
        # A0 + t - t² = Bm + cm*t
        # -t² + (1-cm)*t + (A0-Bm) = 0
        # t² - (1-cm)*t - (A0-Bm) = 0
        a_coef = 1.0
        b_coef = -(1.0 - cm)
        c_coef = -(A0 - Bm)
        disc = b_coef*b_coef - 4*a_coef*c_coef

            sqrt_disc = math.sqrt(disc)
            t1 = (-b_coef - sqrt_disc) / (2*a_coef)
            t2 = (-b_coef + sqrt_disc) / (2*a_coef)

                candidates.append(t1)
            if 0 <= t2 <= 1:
                candidates.append(t2)
    
    candidates = np.array(candidates)
    candidates = candidates[(candidates >= 0) & (candidates <= 1)]
    
    # Evaluate g at all candidates
    best_t = old
    best_g = float('inf')
    
    for t in candidates:
        f0 = A0 + t - t*t
        f_lin = B + c * t
        f_lin[n-1] = -1e6
        g_val = max(f0, float(np.max(f_lin)))
        if g_val < best_g - 1e-14:
            best_g = g_val
            best_t = float(t)
    
    return best_t
