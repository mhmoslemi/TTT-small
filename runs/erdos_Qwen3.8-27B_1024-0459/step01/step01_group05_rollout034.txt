    return m + np.log(sum_exp) / beta

def corr_smooth_max_grad(h, beta):

    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    scaled = corr * dx
    m = np.max(scaled)
    shifted = np.clip(beta * (scaled - m), -500, 500)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    value = m + np.log(sum_exp) / beta
    softmax_w = exp_s / sum_exp
    
    grad_h = np.zeros(n)
    for s_idx in range(2 * n - 1):
        s = s_idx - (n - 1)
        w = softmax_w[s_idx] * dx
        if abs(w) < 1e-16:
            continue
        j_lo = max(0, s)
        j_hi = min(n - 1, n - 1 + s)
        if j_lo <= j_hi:
            i_lo = j_lo - s
            i_hi = j_hi - s
            grad_h[j_lo:j_hi + 1] += w * (1.0 - h[i_lo:i_hi + 1])
        j_lo2 = max(-s, 0)
        j_hi2 = min(n - 1 - s, n - 1)
        if j_lo2 <= j_hi2:
            i_lo2 = j_lo2 + s
            i_hi2 = j_hi2 + s
            grad_h[j_lo2:j_hi2 + 1] -= w * h[i_lo2:i_hi2 + 1]
    return value, grad_h

    return corr_smooth_max_value(h, beta)

    # Phase 3: Try n=190 and n=300 if time allows
    for n_try in [190, 300]:
        if budget_s - (time.time() - t0) < 80:
            break
        h_interp = interp_to_resolution(best_h, best_n, n_try)
        h_interp = project_sum(h_interp, n_try / 2.0)
        h_opt = do_optimize(h_interp, n_try, 30.0, 300, 1e-10, t0, 60)
        h_opt = project_sum(h_opt, n_try / 2.0)
        c5_try = compute_c5(h_opt)
        if c5_try < best_c5:
            best_h = h_opt
            best_c5 = c5_try
            best_n = n_try
            # Quick refine
            if budget_s - (time.time() - t0) > 60:
                h_opt2 = do_optimize(best_h, best_n, 150.0, 300, 1e-11, t0, 50)
                h_opt2 = project_sum(h_opt2, best_n / 2.0)
                c5_try2 = compute_c5(h_opt2)
                if c5_try2 < best_c5:
                    best_h = h_opt2
                    best_c5 = c5_try2

    # Phase 4: Intensive refinement at best resolution
    for beta in [50.0, 150.0, 500.0, 1000.0]:
        if budget_s - (time.time() - t0) < 80:
            break
        h_opt = do_optimize(best_h, best_n, beta, 500, 1e-12, t0, 60)
        h_opt = project_sum(h_opt, best_n / 2.0)
        c5_try = compute_c5(h_opt)
        if c5_try < best_c5:
            best_h = h_opt
            best_c5 = c5_try

    # Phase 5: Multi-start to escape local optima
    n_best = best_n
    if budget_s - (time.time() - t0) > 200:
        # Random restarts
        n_starts = 4
        for attempt in range(n_starts):
            if budget_s - (time.time() - t0) < 60:
                break
            h_rand = np.random.uniform(0, 1, n_best)
            h_rand = project_sum(h_rand, n_best / 2.0)
            h_opt = do_optimize(h_rand, n_best, 25.0, 350, 1e-10, t0, 45)
            h_opt = project_sum(h_opt, n_best / 2.0)
            c5_try = compute_c5(h_opt)
            if c5_try < best_c5:
                best_h = h_opt
                best_c5 = c5_try
                best_n = n_best

    # Phase 6: Structured initializations
    if budget_s - (time.time() - t0) > 120:
        n_try = best_n
        inits = []
        inits.append(make_sine_init(n_try, 1.0, 0.35, 0.0))
        inits.append(make_sine_init(n_try, 2.0, 0.35, 0.0))
        inits.append(make_sine_init(n_try, 3.0, 0.3, 0.5))
        inits.append(make_sine_init(n_try, 5.0, 0.25, 0.0))
        inits.append(make_two_wave_init(n_try, 1.0, 3.0, 0.3))
        inits.append(make_two_wave_init(n_try, 2.0, 5.0, 0.25))

        for h_init in inits:
            if budget_s - (time.time() - t0) < 40:
                break
            h_init = project_sum(h_init, n_try / 2.0)
            h_opt = do_optimize(h_init, n_try, 25.0, 300, 1e-10, t0, 35)
            h_opt = project_sum(h_opt, n_try / 2.0)
            c5_try = compute_c5(h_opt)
            if c5_try < best_c5:
                best_h = h_opt
                best_c5 = c5_try
                best_n = n_try
                # Quick refine
                if budget_s - (time.time() - t0) > 30:
                    h_opt2 = do_optimize(best_h, best_n, 100.0, 300, 1e-11, t0, 25)
                    h_opt2 = project_sum(h_opt2, best_n / 2.0)
                    c5_try2 = compute_c5(h_opt2)
                    if c5_try2 < best_c5:
                        best_h = h_opt2
                        best_c5 = c5_try2

    # Phase 7: Perturbation around best
    if budget_s - (time.time() - t0) > 100:
        for attempt in range(3):
            if budget_s - (time.time() - t0) < 40:
                break
            noise = 0.02 * (1.0 - attempt * 0.25)
            h_pert = best_h + np.random.randn(best_n) * noise
            h_pert = project_sum(h_pert, best_n / 2.0)
            h_opt = do_optimize(h_pert, best_n, 50.0, 400, 1e-11, t0, 30)
            h_opt = project_sum(h_opt, best_n / 2.0)
            c5_try = compute_c5(h_opt)
            if c5_try < best_c5:
                best_h = h_opt
                best_c5 = c5_try

    # Phase 8: Final very-fine optimization
    if budget_s - (time.time() - t0) > 40:
        h_opt = do_optimize(best_h, best_n, 2000.0, 500, 1e-13, t0, 30)
        h_opt = project_sum(h_opt, best_n / 2.0)
        c5_try = compute_c5(h_opt)
        if c5_try < best_c5:
            best_h = h_opt
            best_c5 = c5_try

    # Final cleanup
    best_h = project_sum(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    return (best_h, final_c5, best_n)
