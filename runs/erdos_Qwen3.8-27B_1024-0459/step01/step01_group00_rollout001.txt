def smooth_weights(F, beta):
    F = np.asarray(F, dtype=float)
    M = float(np.max(F))
    if beta <= 1e-6:
        w = np.full(F.shape, 1.0 / F.size, dtype=float)
        return float(np.mean(F)), w, M
    shift = beta * (F - M)
    # exp all; since shift<=0, no overflow. Underflow okay.
    ex = np.exp(shift)
    s = float(np.sum(ex))
    if s <= 0.0 or not np.isfinite(s):
       w = np.zeros(F.shape, dtype=float)
       w[int(np.argmax(F))] = 1.0
       return M, w, M
    w = ex / s
    G = M + float(np.log(s)) / beta
    return G, w, M
```

def grad_by_weights(h, w):
    n = h.size
    grad = np.zeros(n, dtype=float)
    dx = 2.0 / float(n)
    for l in range(1, n):
       wp = w[n-1-l] * dx
       if wp > 1e-18:
           grad[:n-l] += wp * (1.0 - h[l:])
           grad[l:]   -= wp * h[:n-l]
       wn = w[n-1+l] * dx
       if wn > 1e-18:
           grad[l:]   += wn * (1.0 - h[:n-l])
           grad[:n-l] -= wn * h[l:]
    wc = w[n-1]*dx
    grad += wc * (1.0 - 2.0*h)
    return grad

def project_feasible(y, target):
    y = np.asarray(y, dtype=float)
    if not np.all(np.isfinite(y)):
       y = np.full(y.shape, 0.5, dtype=float)
    lo = float(np.min(-y)) - 1.0
    hi = float(np.max(1.0 - y)) + 1.0
    # If target outside [0,n], clip
    if target < 0: target = 0
    if target > y.size: target = y.size
    for _ in range(80):
       mid = 0.5*(lo+hi)
       s = float(np.sum(np.clip(y+mid, 0.0, 1.0)))
       if s < target:
           lo = mid
       else:
           hi = mid
    z = np.clip(y + 0.5*(lo+hi), 0.0, 1.0)
    # adjust tiny diff
    diff = target - float(np.sum(z))
    if abs(diff) > 1e-15:
       # distribute among non-saturated
       if diff > 0:
          mask = z < 1.0 - 1e-15
       else: mask = z > 1e-15
       if mask.any():
          z[mask] += diff / float(np.sum(mask))
       z = np.clip(z,0.0,1.0)
    return z

```python
import time
import numpy as np
from scipy.optimize import minimize

_TIME_END = 0.0
_SLS_BETA = 10.0
_SLS_N = 1


def time_left():
    return float(_TIME_END - time.time())


def eval_full(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return np.array([1e100], dtype=float)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def c5_value(h):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return 1e100
    return float(np.max(eval_full(h)))


def project_feasible(y, target):
    y = np.asarray(y, dtype=float)
    if y.size == 0:
        return y
    target = float(target)
    if not np.all(np.isfinite(y)):
        y = np.full(y.shape, 0.5, dtype=float)
    lo = float(np.min(-y)) - 1.0
    hi = float(np.max(1.0 - y)) + 1.0
    if target < 0.0:
        target = 0.0
    if target > float(y.size):
        target = float(y.size)
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    z = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(np.sum(z))
    if abs(diff) > 1e-14:
        if diff > 0.0:
            mask = z < 1.0 - 1e-14
        else:
            mask = z > 1e-14
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            z = np.asarray(z, dtype=float)
            z[mask] += diff / float(cnt)
            z = np.clip(z, 0.0, 1.0)
    return z


def quick_project(y, target):
    y = np.asarray(y, dtype=float)
    target = float(target)
    if y.size == 0:
        return y
    if not np.all(np.isfinite(y)):
        y = np.full(y.shape, 0.5, dtype=float)
    y = np.clip(y, 0.0, 1.0)
    for _ in range(200):
        diff = target - float(np.sum(y))
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            mask = y < 1.0 - 1e-13
        else:
            mask = y > 1e-13
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        y[mask] += diff / float(cnt)
        y = np.clip(y, 0.0, 1.0)
    return y


def make_feasible(h, n):
    return project_feasible(np.asarray(h, dtype=float), 0.5 * float(n))


def make_feasible_fast(h, n):
    return quick_project(np.asarray(h, dtype=float), 0.5 * float(n))


def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    if n_dst <= 1:
        return make_feasible_fast(np.full(n_dst, 0.5), n_dst)
    if h_src.size == 0:
        return make_feasible_fast(np.full(n_dst, 0.5), n_dst)
    if h_src.size == n_dst:
        return make_feasible_fast(h_src, n_dst)
    x_old = np.linspace(0.0, 2.0, h_src.size, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    y = np.interp(x_new, x_old, h_src)
    return make_feasible_fast(y, n_dst)


def downsample_h(h, m):
    h = np.asarray(h, dtype=float)
    m = int(m)
    if m <= 1:
        return make_feasible_fast(np.array([0.5 * float(h.size)], dtype=float), 1)
    if m >= h.size:
        return make_feasible_fast(h, m)
    out = np.empty(m, dtype=float)
    for i in range(m):
        a = i * h.size // m
        b = (i + 1) * h.size // m
        out[i] = float(np.mean(h[a:b]))
    return make_feasible_fast(out, m)


def smooth_weights(F, beta):
    F = np.asarray(F, dtype=float)
    if F.size == 0:
        return 1e100, np.array([], dtype=float), 1e100
    M = float(np.max(F))
    if not np.isfinite(M):
        return 1e100, np.zeros(F.shape, dtype=float), 1e100
    if beta <= 1e-8:
        w = np.full(F.shape, 1.0 / float(F.size), dtype=float)
        return float(np.mean(F)), w, M
    shift = beta * (F - M)
    ex = np.exp(shift)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros(F.shape, dtype=float)
        w[int(np.argmax(F))] = 1.0
        return M, w, M
    w = ex / s
    G = M + float(np.log(s)) / float(beta)
    if not np.isfinite(G):
        G = M
    return G, w, M


def active_weights(F, tol):
    F = np.asarray(F, dtype=float)
    M = float(np.max(F))
    act = F >= (M - float(tol))
    cnt = int(np.count_nonzero(act))
    w = np.zeros(F.shape, dtype=float)
    if cnt == 0 or not np.any(act):
        w[int(np.argmax(F))] = 1.0
        cnt = 1
    else:
        w[act] = 1.0 / float(cnt)
    return M, w, M


def grad_by_weights(h, w):
    h = np.asarray(h, dtype=float)
    n = h.size
    grad = np.zeros(n, dtype=float)
    if n == 0 or w.size != 2 * n - 1:
        return grad
    dx = 2.0 / float(n)
    for l in range(1, n):
        wp = w[n - 1 - l] * dx
        if wp > 1e-18:
            grad[:n - l] += wp * (1.0 - h[l:])
            grad[l:] -= wp * h[:n - l]
        wn = w[n - 1 + l] * dx
        if wn > 1e-18:
            grad[l:] += wn * (1.0 - h[:n - l])
            grad[:n - l] -= wn * h[l:]
    wc = w[n - 1] * dx
    grad += wc * (1.0 - 2.0 * h)
    return grad


def optimize_projected(h_init, n, beta, max_iter, max_step=0.7,
                       use_active=False, active_tol=1e-9):
    n = int(n)
    h = make_feasible(np.array(h_init, dtype=float).copy(), n)
    best = h.copy()
    best_F = eval_full(best)
    best_c = float(np.max(best_F))

    current = h.copy()
    cur_c = best_c
    last_step = float(max_step)
    no_improve = 0
    max_iter = int(max(1, max_iter))

    for _ in range(max_iter):
        if time_left() < 2.0:
            break

        F = eval_full(current)
        cur_c = float(np.max(F))
        if cur_c < best_c - 1e-15:
            best_c = cur_c
            best = current.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 80:
            current = best.copy()
            cur_c = best_c
            no_improve = 0
            last_step = float(max_step)

        if use_active:
            M, w, _ = active_weights(F, active_tol)
            G = float(np.dot(w, F))
        else:
            G, w, M = smooth_weights(F, float(beta))

        if not np.all(np.isfinite(w)):
            break

        grad = grad_by_weights(current, w)
        if not np.all(np.isfinite(grad)):
            break
        grad -= float(np.mean(grad))
        norm = float(np.linalg.norm(grad))
        if norm < 1e-14:
            break

        d = grad / norm
        step = min(float(last_step), float(max_step))
        improved = False

        for __ in range(34):
            y = current - step * d
            hp = make_feasible(y, n)
            Fp = eval_full(hp)
            cp = float(np.max(Fp))

            if cp < cur_c - 1e-15:
                current = hp
                cur_c = cp
                last_step = step
                improved = True
                break

            if use_active:
                Gp = float(np.dot(w, Fp))
            else:
                Gp, _wp, _Mp = smooth_weights(Fp, float(beta))

            if Gp < G - 1e-11 and cp <= cur_c + 1e-12 and np.isfinite(Gp):
                current = hp
                cur_c = cp
                last_step = step
                improved = True
                break

            step *= 0.5
            if step < 1e-10:
                break

        if not improved:
            break

    current = make_feasible(current, n)
    cf = float(np.max(eval_full(current)))
    if cf < best_c:
        best = current
        best_c = cf
    else:
        best = make_feasible(best, n)
    return best, best_c


def slsq_obj(x):
    if not np.all(np.isfinite(x)):
        return 1e100
    F = eval_full(x)
    G, _w, _M = smooth_weights(F, float(_SLS_BETA))
    if not np.isfinite(G):
        return 1e100
    return float(G)


def slsq_jac(x):
    if not np.all(np.isfinite(x)):
        return np.zeros(x.shape, dtype=float)
    F = eval_full(x)
    _G, w, _M = smooth_weights(F, float(_SLS_BETA))
    g = grad_by_weights(x, w)
    if not np.all(np.isfinite(g)):
        g = np.zeros(x.shape, dtype=float)
    return g


def slsq_sum_con(x):
    return float(np.sum(x) - 0.5 * float(_SLS_N))


def attempt_slsqp(h_init, n, beta, maxiter=250, ftol=1e-14):
    global _SLS_BETA, _SLS_N
    n = int(n)
    _SLS_BETA = float(beta)
    _SLS_N = n
    h = make_feasible(np.array(h_init, dtype=float).copy(), n)
    best = h.copy()
    best_c = c5_value(best)

    if time_left() < 8.0:
        return best, best_c

    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": slsq_sum_con}]

    try:
        res = minimize(
            slsq_obj,
            h,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            jac=slsq_jac,
            options={"maxiter": int(max(10, maxiter)), "ftol": float(ftol)}
        )
        if hasattr(res, "x") and res.x is not None:
            hp = make_feasible(res.x, n)
            c = c5_value(hp)
            if np.isfinite(c) and c < best_c:
                best = hp
                best_c = c
    except Exception:
        pass

    best = make_feasible(best, n)
    best_c = c5_value(best)
    return best, best_c

            best_c = c
            h = hp

    return make_feasible(best, n), best_c


def generate_fallback_inits(n, rng, count=8):
    n = int(n)
    inits = []
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    inits.append(make_feasible(np.full(n, 0.5, dtype=float), n))

    for freq in (0.5, 1.0, 1.5, 2.0, 3.0):
        for amp in (0.45, 0.75, 1.0):
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            h = 0.5 + 0.5 * amp * np.cos(np.pi * freq * x + ph)
            inits.append(make_feasible(h, n))

    for _ in range(count):
        a1 = float(rng.uniform(0.0, 2.0 * np.pi))
        b1 = float(rng.uniform(0.0, 2.0 * np.pi))
        a2 = float(rng.uniform(0.0, 2.0 * np.pi))
        h = 0.5 + 0.35 * np.cos(np.pi * x + a1) + 0.25 * np.sin(2.0 * np.pi * x + b1) + 0.20 * np.cos(3.0 * np.pi * x + a2)
        inits.append(make_feasible(h, n))

    return inits


def run(seed=42, budget_s=1300, **kwargs):
    global _TIME_END
    _TIME_END = time.time() + float(budget_s)

    np.random.seed(seed)
    rng = np.random.RandomState(seed)

    TARGET = 0.38085

    h_parent = None
    try:
        hv = initial_h_values
        hv = np.asarray(hv, dtype=float)
        if hv is not None and hv.size > 1 and np.all(np.isfinite(hv)):
            h_parent = hv.copy()
    except Exception:
        h_parent = None

    if h_parent is not None:
        n_best = int(h_parent.size)
        if n_best < 2:
            n_best = 160
            h_best = make_feasible(np.full(n_best, 0.5, dtype=float), n_best)
        else:
            h_best = make_feasible(h_parent, n_best)
    else:
        n_best = 160
        h_best = make_feasible(np.full(n_best, 0.5, dtype=float), n_best)
        inits = generate_fallback_inits(n_best, rng, 12)
        c_best = c5_value(h_best)
        for h0 in inits:
            c0 = c5_value(h0)
            if c0 < c_best:
                c_best = c0
                h_best = make_feasible(h0, n_best)

    c_best = c5_value(h_best)

    # Initial local optimization at the parent resolution.
    for beta, iters, ms in ((4.0, 160, 0.65), (15.0, 220, 0.55), (60.0, 260, 0.45)):
        if time_left() < 700.0:
            break
        h2, c2 = optimize_projected(h_best, n_best, beta, iters, max_step=ms)
        if c2 < c_best:
            h_best, c_best = h2, c2

    for beta, iters in ((100.0, 300), (400.0, 300)):
        if time_left() < 650.0:
            break
        h2, c2 = attempt_slsqp(h_best, n_best, beta, maxiter=iters)
        if c2 < c_best:
            h_best, c_best = h2, c2

    for beta, iters, ms in ((250.0, 260, 0.35), (900.0, 280, 0.22)):
        if time_left() < 560.0:
            break
        h2, c2 = optimize_projected(h_best, n_best, beta, iters, max_step=ms)
        if c2 < c_best:
            h_best, c_best = h2, c2

    for tol, iters, ms in ((1e-6, 180, 0.12), (1e-8, 220, 0.08), (1e-9, 260, 0.05)):
        if time_left() < 520.0:
            break
        h2, c2 = optimize_projected(h_best, n_best, 1.0, iters, max_step=ms,
                                    use_active=True, active_tol=tol)
        if c2 < c_best:
            h_best, c_best = h2, c2

    for beta, iters in ((800.0, 280), (2500.0, 220)):
        if time_left() < 460.0:
            break
        h2, c2 = attempt_slsqp(h_best, n_best, beta, maxiter=iters, ftol=1e-15)
        if c2 < c_best:
            h_best, c_best = h2, c2

    # If the target is already met, do only light polishing to avoid unnecessary
    # risk from exploratory movements.
    if c_best < TARGET - 1e-10:
        for tol, iters, ms in ((1e-9, 120, 0.03), (5e-10, 140, 0.02), (1e-10, 160, 0.015)):
            if time_left() < 15.0:
                break
            h2, c2 = optimize_projected(h_best, n_best, 1000.0, iters, max_step=ms,
                                        use_active=True, active_tol=tol)
            if c2 < c_best:
                h_best, c_best = h2, c2

        if time_left() > 35.0:
            h2, c2 = attempt_slsqp(h_best, n_best, 3500.0, maxiter=220, ftol=1e-15)
            if c2 < c_best:
                h_best, c_best = h2, c2

        if time_left() > 15.0:
            n_trials = int(min(20000, max(1000, 500.0 * max(1.0, time_left() - 5.0))))
            h2, c2 = local_perturb(h_best, n_best, n_trials, 0.004, rng)
            if c2 < c_best:
                h_best, c_best = h2, c2

        h_best = make_feasible(h_best, n_best)
        c_best = c5_value(h_best)
        return (h_best, c_best, n_best)

    # Multi-start perturbations from the current best.
    if time_left() > 360.0:
        for sigma in (0.004, 0.008, 0.016, 0.035, 0.07):
            if time_left() < 260.0:
                break
            for _ in range(2):
                if time_left() < 120.0:
                    break
                h0 = make_feasible_fast(h_best + rng.normal(0.0, sigma, n_best), n_best)
                h2, c2 = attempt_slsqp(h0, n_best, 700.0, maxiter=220)
                if c2 < c_best:
                    h_best, c_best = h2, c2
                if time_left() < 120.0:
                    break

    # Coarse-restart / multi-resolution basin hopping.
    if time_left() > 280.0:
        for coarse_n in (64, 48, 36):
            if time_left() < 140.0:
                break
            h0 = downsample_h(h_best, coarse_n)
            for attempt in range(3):
                if time_left() < 60.0:
                    break
                if attempt == 1:
                    h_cur = make_feasible_fast(h0 + rng.normal(0.0, 0.03, coarse_n), coarse_n)
                elif attempt == 2:
                    h_cur = make_feasible_fast(h0 + rng.normal(0.0, 0.07, coarse_n), coarse_n)
                else:
                    h_cur = h0.copy()

                for beta in (8.0, 40.0, 160.0):
                    if time_left() < 30.0:
                        break
                    h_cur, _c = optimize_projected(h_cur, coarse_n, beta, 70, max_step=0.7)

                h_up = interpolate_to_n(h_cur, n_best)
                c_up = c5_value(h_up)
                if c_up < c_best:
                    h_best, c_best = h_up, c_up

                h2, c2 = optimize_projected(h_up, n_best, 220.0, 60, max_step=0.35)
                if c2 < c_best:
                    h_best, c_best = h2, c2

                if time_left() > 70.0:
                    h2, c2 = attempt_slsqp(h_best, n_best, 800.0, maxiter=180)
                    if c2 < c_best:
                        h_best, c_best = h2, c2

    # Try a higher resolution candidate if enough time remains.
    if time_left() > 240.0:
        n_hi = int(min(384, max(256, 2 * n_best)))
        if n_hi > n_best:
            h_hi = interpolate_to_n(h_best, n_hi)
            ch = c5_value(h_hi)
            if ch < c_best:
                h_best, c_best, n_best = h_hi, ch, n_hi

            if time_left() > 170.0:
                h_hi, ch = optimize_projected(h_best, n_best, 120.0, 70, max_step=0.4)
                if ch < c_best:
                    h_best, c_best = h_hi, ch
                if time_left() > 100.0:
                    h_hi, ch = attempt_slsqp(h_best, n_best, 600.0, maxiter=200, ftol=1e-14)
                    if ch < c_best:
                        h_best, c_best = h_hi, ch
                if time_left() > 60.0:
                    h_hi, ch = optimize_projected(h_best, n_best, 1.0, 70, max_step=0.1,
                                                  use_active=True, active_tol=1e-8)
                    if ch < c_best:
                        h_best, c_best = h_hi, ch

    # Final high-precision active-set polishing.
    for tol, iters, ms in ((1e-7, 120, 0.05), (1e-8, 180, 0.03),
                           (3e-9, 220, 0.02), (1e-9, 260, 0.012),
                           (3e-10, 300, 0.008), (1e-10, 340, 0.005)):
        if time_left() < 12.0:
            break
        h2, c2 = optimize_projected(h_best, n_best, 1.0, iters, max_step=ms,
                                    use_active=True, active_tol=tol)
        if c2 < c_best:
            h_best, c_best = h2, c2

    # Final SLSQP polish.
    if time_left() > 30.0:
        h2, c2 = attempt_slsqp(h_best, n_best, 4000.0, maxiter=260, ftol=1e-15)
        if c2 < c_best:
            h_best, c_best = h2, c2

    # Small random local search for last-kilometer improvements.
    if time_left() > 12.0:
        n_trials = int(min(60000, max(2000, 900.0 * max(1.0, time_left() - 4.0))))
        h2, c2 = local_perturb(h_best, n_best, n_trials, 0.0025, rng)
        if c2 < c_best:
            h_best, c_best = h2, c2

    if time_left() > 18.0:
        h2, c2 = attempt_slsqp(h_best, n_best, 2000.0, maxiter=180, ftol=1e-15)
        if c2 < c_best:
            h_best, c_best = h2, c2

    h_best = make_feasible(h_best, n_best)
    c_best = c5_value(h_best)
    if not np.isfinite(c_best):
        h_best = make_feasible(np.full(n_best, 0.5, dtype=float), n_best)
        c_best = c5_value(h_best)

    return (h_best, c_best, n_best)
