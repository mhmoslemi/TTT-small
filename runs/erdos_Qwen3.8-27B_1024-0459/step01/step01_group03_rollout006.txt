def project_density(x, target=None):
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0).copy()
    n = x.size
    if n == 0:
        return x
    if target is None:
        target = 0.5 * float(n)
    target = float(np.clip(target, 0.0, float(n)))
    if target <= 0.0:
        x.fill(0.0); return x
    if target >= float(n):
        x.fill(1.0); return x
    s = float(np.sum(x))
    if abs(s - target) > 1e-13 * max(1, n):
        lo = -1.0; hi = 1.0
        for _ in range(70):
            mid = 0.5 * (lo + hi)
            sm = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
            if sm < target:
                lo = mid
            else:
                hi = mid
        x = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(3):
        res = target - float(np.sum(x))
        if abs(res) <= 1e-14 * max(1, n):
            break
        if res > 0.0:
            cap = 1.0 - x
            cs = float(np.sum(cap))
            if cs <= 0.0: break
            x = x + res * cap / cs
        else:
            cap = x
            cs = float(np.sum(cap))
            if cs <= 0.0: break
            x = x + res * cap / cs
        x = np.clip(x, 0.0, 1.0)
    res = target - float(np.sum(x))
    if res > 0.0:
        idx = int(np.argmax(1.0 - x))
        val = x[idx] + res
        if val > 1.0: val = 1.0

def accumulate_grad(h, indices, weights, dx):
    n = h.size
    g = np.zeros(n, dtype=float)
    if n == 0: return g
    center = n - 1
    for t, w in zip(indices, weights):

def project_density(x, target=None):
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0).copy()
    n = x.size
    if n == 0:
        return x
    if target is None:
        target = 0.5 * float(n)
    target = float(np.clip(target, 0.0, float(n)))

    if target <= 0.0:
        x.fill(0.0)
        return x
    if target >= float(n):
        x.fill(1.0)
        return x

    s = float(np.sum(x))
    if abs(s - target) > 1e-13 * max(1, n):
        lo = -1.0
        hi = 1.0
        for _ in range(70):
            mid = 0.5 * (lo + hi)
            sm = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
            if sm < target:
                lo = mid
            else:
                hi = mid
        x = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(3):
        res = target - float(np.sum(x))
        if abs(res) <= 1e-14 * max(1, n):
            break
        if res > 0.0:
            cap = 1.0 - x
            cs = float(np.sum(cap))
            if cs <= 0.0:
                break
            x = x + res * cap / cs
        else:
            cap = x
            cs = float(np.sum(cap))
            if cs <= 0.0:
                break
            x = x + res * cap / cs
        x = np.clip(x, 0.0, 1.0)

    res = target - float(np.sum(x))
    if res > 0.0:
        idx = int(np.argmax(1.0 - x))
        val = x[idx] + res
        if val > 1.0:
            val = 1.0
        if val < 0.0:
            val = 0.0
        x[idx] = val
    elif res < 0.0:
        idx = int(np.argmax(x))
        val = x[idx] + res
        if val < 0.0:
            val = 0.0
        if val > 1.0:
            val = 1.0
        x[idx] = val
    return x


def simple_feasible(h, target, tol=1e-7):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    return abs(float(np.sum(h)) - float(target)) <= max(1e-8, tol * max(1, h.size))


def cw_eval(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)


def cw_corr(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return np.array([], dtype=float)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def accumulate_grad(h, indices, weights, dx):
    n = h.size
    g = np.zeros(n, dtype=float)
    if n == 0:
        return g
    center = n - 1
    for t, w in zip(indices, weights):
        w = float(w)
        if w <= 0.0:
            continue
        t = int(t)
        k = t - center
        if k == 0:
            g += w * (1.0 - 2.0 * h)
        elif k > 0:
            lag = k
            if lag < n:
                g[lag:] += w
                g[:n - lag] -= w * h[lag:]
                g[lag:] -= w * h[:n - lag]
        else:
            lag = -k
            if lag < n:
                g[:n - lag] += w
                g[:n - lag] -= w * h[lag:]
                g[lag:] -= w * h[:n - lag]
    g *= dx
    return g


def gradient_matrix(h, indices, dx):
    n = h.size
    m = int(len(indices))
    G = np.zeros((m, n), dtype=float)
    center = n - 1
    for r, t in enumerate(indices):
        g = np.zeros(n, dtype=float)
        t = int(t)
        k = t - center
        if k == 0:
            g += 1.0 - 2.0 * h
        elif k > 0:
            lag = k
            if lag < n:
                g[lag:] += 1.0
                g[:n - lag] -= h[lag:]
                g[lag:] -= h[:n - lag]
        else:
            lag = -k
            if lag < n:
                g[:n - lag] += 1.0
                g[:n - lag] -= h[lag:]
                g[lag:] -= h[:n - lag]
        G[r, :] = g
    G *= dx
    return G


def select_active_indices(c, max_count, margin=1e-6):
    c = np.asarray(c, dtype=float)
    total = c.size
    if total == 0:
        return np.array([], dtype=int)
    mc = int(max_count)
    if mc < 1:
        mc = 1
    if mc > total:
        mc = total
    idx = np.argsort(c)[-mc:]
    if margin and margin > 0.0:
        m = float(np.max(c))
        near = np.where(c >= m - margin)[0]
        idx = np.unique(np.concatenate((idx, near)))
        cap = 220
        if idx.size > cap:
            order = np.argsort(c[idx])[::-1]
            idx = idx[order[:cap]]
    return idx


def _set_opt(n, alpha, target):
    global _OPT_N, _OPT_DX, _OPT_ALPHA, _OPT_TARGET
    _OPT_N = int(n)
    if _OPT_N <= 0:
        _OPT_N = 1
    _OPT_DX = 2.0 / float(_OPT_N)
    _OPT_ALPHA = float(alpha)
    _OPT_TARGET = float(0.5 * _OPT_N) if target is None else float(target)


def _opt_sum(x):
    return float(np.sum(np.asarray(x, dtype=float)) - _OPT_TARGET)


def _opt_sum_jac(x):
    return np.ones(np.asarray(x, dtype=float).size, dtype=float)


def _logsum_value(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    if corr.size == 0 or not np.all(np.isfinite(corr)):
        return None, 0.0, None
    cmax = float(np.max(corr))
    a = _OPT_ALPHA
    y = a * (corr - cmax)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return None, cmax, None
    val = _OPT_DX * (cmax + np.log(z) / a)
    w = e / z
    return val, cmax, w


def _opt_fun(x):
    x = np.asarray(x, dtype=float)
    if x.size != _OPT_N or not np.all(np.isfinite(x)):
        return 1e100
    out = _logsum_value(x)
    if out[0] is None:
        return 1e100
    return float(out[0])


def _opt_jac(x):
    x = np.asarray(x, dtype=float)
    if x.size != _OPT_N or not np.all(np.isfinite(x)):
        return np.zeros(x.size, dtype=float)
    out = _logsum_value(x)
    if out[0] is None:
        return np.zeros(x.size, dtype=float)
    w = out[2]
    thr = max(1e-10, 1e-8 / max(1, w.size))
    idx = np.where(w > thr)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(w))], dtype=int)
    else:
        idx = idx.astype(int)
    vals = w[idx]
    return accumulate_grad(x, idx, vals, _OPT_DX)


def slsqp_smooth(h0, deadline, alphas=None, maxiter=None):
    h = project_density(h0)
    n = h.size
    if n == 0:
        return h, float("inf")
    target = 0.5 * float(n)
    best = h.copy()
    best_c = cw_eval(best)
    if deadline - time.perf_counter() < 5.0:
        return best, best_c

    if alphas is None:
        if n <= 120:
            alphas = (15.0, 45.0, 120.0, 300.0, 800.0)
        elif n <= 240:
            alphas = (20.0, 60.0, 180.0)
        elif n <= 450:
            alphas = (20.0, 60.0)
        else:
            alphas = (30.0,)
    if maxiter is None:
        if n <= 120:
            maxiter = 300
        elif n <= 240:
            maxiter = 180
        else:
            maxiter = 90

    bounds = [(0.0, 1.0)] * n
    constraints = ({"type": "eq", "fun": _opt_sum, "jac": _opt_sum_jac},)
    x0 = best.copy()
    same = 0

    for alpha in alphas:
        if time.perf_counter() >= deadline:
            break
        _set_opt(n, alpha, target)
        try:
            res = minimize(
                _opt_fun,
                x0,
                method="SLSQP",
                jac=_opt_jac,
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": int(maxiter), "ftol": 1e-10, "disp": False},
            )
            xg = getattr(res, "x", None)
            if xg is not None:
                xg = np.asarray(xg, dtype=float)
                if np.all(np.isfinite(xg)):
                    cand = project_density(xg, target)
                    ca = cw_eval(cand)
                    if ca < best_c:
                        best = cand
                        best_c = ca
                        same = 0
                    else:
                        same += 1
                    if ca <= best_c + 0.02:
                        x0 = cand
                    else:
                        x0 = best
                else:
                    x0 = best
            else:
                x0 = best
        except Exception:
            x0 = best
        if same >= 3:
            break

    h = project_density(x0, target)
    ca = cw_eval(h)
    if ca < best_c:
        best = h
        best_c = ca
    return best, best_c


def active_gradient_descent(h0, deadline, rng=None):
    h = project_density(h0)
    n = h.size
    if n == 0:
        return h, float("inf")
    best = h.copy()
    best_c = cw_eval(best)
    dx = 2.0 / float(n)
    target = 0.5 * float(n)
    if rng is None:
        rng = np.random.default_rng(987654)

    if n <= 150:
        K = 48
        taus = (0.08, 0.02, 0.005, 0.0015)
    elif n <= 360:
        K = 64
        taus = (0.08, 0.02, 0.005)
    else:
        K = 80
        taus = (0.05, 0.01, 0.003)

    for tau in taus:
        stall = 0
        for _ in range(300):
            if time.perf_counter() >= deadline:
                break
            c = cw_corr(best)
            m = float(np.max(c))
            active = select_active_indices(c, K, margin=0.001)
            if active.size == 0:
                break
            y = (c[active] - m) / tau
            y = np.clip(y, -745.0, 0.0)
            w = np.exp(y)
            z = float(np.sum(w))
            if z <= 0.0:
                break
            w = w / z
            g = accumulate_grad(best, active, w, dx)
            d = -g
            d -= float(np.mean(d))
            norm = float(np.linalg.norm(d))
            if norm < 1e-13:
                break
            d = d / norm

            improved = False
            for ratio in (1.0, 0.5, 2.0, 0.2, 0.1, 0.05, 5.0, 0.02):
                cand = project_density(best + (0.05 * ratio) * d, target)
                ca = cw_eval(cand)
                if ca < best_c:
                    best = cand
                    best_c = ca
                    improved = True
                    break

            if improved:
                stall = 0
            else:
                stall += 1
                if stall >= 8:
                    esc = False
                    for amp in (0.02, 0.05, 0.10):
                        noise = rng.random(n) - 0.5
                        noise -= float(np.mean(noise))
                        cand = project_density(best + amp * noise, target)
                        ca = cw_eval(cand)
                        if ca < best_c:
                            best = cand
                            best_c = ca
                            esc = True
                            break
                    if esc:
                        stall = 0
                    else:
                        break

    best = project_density(best, target)
    return best, cw_eval(best)


def pair_search(h0, deadline, rng=None, M=64, ngrad=16, nrand=120, surrogate_points=49):
    h = project_density(h0)
    n = h.size
    if n < 2:
        return h, cw_eval(h)

    target = 0.5 * float(n)
    dx = 2.0 / float(n)
    if rng is None:
        rng = np.random.default_rng(24680)

    best = h.copy()
    best_c = cw_eval(best)

    if surrogate_points is None:
        P = 49
    else:
        P = int(surrogate_points)
    if n <= 120:
        P = max(P, 65)
        max_pairs = 550
        max_eval = 60
    elif n <= 360:
        P = max(P, 45)
        max_pairs = 350
        max_eval = 35
    else:
        P = max(P, 31)
        max_pairs = 200
        max_eval = 20
    P = max(17, min(P, 121))

    M_use = max(1, int(M))
    ng = max(4, int(ngrad))
    nr = max(20, int(nrand))
    ng_use = min(ng, n)
    nr_use = min(nr, 520)

    stall = 0
    max_stall = 14
    escape_tol = 0.004

    while time.perf_counter() < deadline:
        c = cw_corr(h)
        cur_c = float(np.max(c))
        m = cur_c

        active = select_active_indices(c, M_use, margin=1e-4)
        if active.size == 0:
            break

        C0 = c[active].copy()
        G = gradient_matrix(h, active, dx)
        lags = np.abs(active - (n - 1)).astype(int)

        y = (C0 - m) / 0.005
        y = np.clip(y, -745.0, 0.0)
        w = np.exp(y)
        wz = float(np.sum(w))
        if wz > 0.0:
            w = w / wz
        else:
            w = np.full(active.size, 1.0 / float(active.size), dtype=float)
        g = G.T @ w

        pos = np.argsort(-g)[:ng_use]
        neg = np.argsort(g)[:ng_use]

        pairs_set = set()
        for ii in neg:
            ii = int(ii)
            for jj in pos:
                jj = int(jj)
                if ii != jj:
                    if ii < jj:
                        pairs_set.add((ii, jj))
                    else:
                        pairs_set.add((jj, ii))

        for _ in range(nr_use):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            while j == i:
                j = int(rng.integers(0, n))
            if i < j:
                pairs_set.add((i, j))
            else:
                pairs_set.add((j, i))

        pairs = list(pairs_set)
        if len(pairs) > max_pairs:
            pairs = pairs[:max_pairs]

        candidates = []
        for (i, j) in pairs:
            hi_i = 1.0 - float(h[i])
            lo_i = -float(h[i])
            hi_j = float(h[j])
            lo_j = float(h[j]) - 1.0
            lo = max(lo_i, lo_j)
            hi = min(hi_i, hi_j)
            if hi - lo < 1e-10:
                continue
            a = lo + 1e-12
            b = hi - 1e-12
            if b <= a:
                continue

            deltas = np.linspace(a, b, P)
            d2 = deltas * deltas
            L = G[:, i] - G[:, j]
            absd = abs(i - j)
            Q = np.where(lags == 0, -2.0 * dx, 0.0)
            if absd > 0:
                Q = np.where(lags == absd, dx, Q)

            vals = C0[:, None] + L[:, None] * deltas[None, :] + Q[:, None] * d2[None, :]
            surr = np.max(vals, axis=0)
            minv = float(np.min(surr))

            if minv < m - 1e-9 or minv < best_c - 1e-9:
                idxmin = int(np.argmin(surr))
                candidates.append((minv, i, j, float(deltas[idxmin])))

        if candidates:
            candidates.sort()
            found = False
            evaluated = 0
            for (_cv, i, j, dv) in candidates:
                if time.perf_counter() >= deadline:
                    break
                if evaluated >= max_eval:
                    break
                evaluated += 1

                cand = h.copy()
                cand[i] = h[i] + dv
                cand[j] = h[j] - dv
                cand = project_density(cand, target)
                ca = cw_eval(cand)

                if ca < best_c:
                    h = cand
                    best = cand
                    best_c = ca
                    found = True
                    break
                elif ca < m - 1e-10 and ca <= best_c + escape_tol:
                    h = cand
                    found = True
                    break

            if found:
                stall = 0
                continue

        stall += 1
        if stall >= max_stall:
            escaped = False

            for _ in range(30):
                if time.perf_counter() >= deadline:
                    break
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                lo = max(-float(h[i]), float(h[j]) - 1.0)
                hi = min(1.0 - float(h[i]), float(h[j]))
                if hi <= lo:
                    continue
                q = float(rng.random())
                dv = lo + q * (hi - lo)
                cand = h.copy()
                cand[i] = h[i] + dv
                cand[j] = h[j] - dv
                cand = project_density(cand, target)
                ca = cw_eval(cand)
                if ca < best_c:
                    h = cand
                    best = cand
                    best_c = ca
                    escaped = True
                    break
                elif ca < cur_c - 1e-10 and ca <= best_c + escape_tol:
                    h = cand
                    escaped = True
                    break

            if not escaped:
                for amp in (0.02, 0.05, 0.10):
                    if time.perf_counter() >= deadline:
                        break
                    noise = rng.random(n) - 0.5
                    noise -= float(np.mean(noise))
                    cand = project_density(h + amp * noise, target)
                    ca = cw_eval(cand)
                    if ca < best_c:
                        h = cand
                        best = cand
                        best_c = ca
                        escaped = True
                        break
                    elif ca <= best_c + escape_tol:
                        h = cand
                        escaped = True
                        break

            if not escaped:
                break
            stall = 0

    best = project_density(best, target)
    return best, cw_eval(best)


def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    n = int(n)
    if n0 == n:
        return p.copy()
    if n0 <= 0 or n <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * float(max(1, n)))

    if n % n0 == 0:
        h = np.repeat(p, n // n0)
    else:
        x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
        x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
        h = np.interp(x1, x0, p, left=p[0], right=p[-1])
    return project_density(h, 0.5 * float(n))


def make_random_start(n, rng, mode, parent=None):
    n = int(n)
    if n <= 1:
        return np.array([0.5], dtype=float)

    if mode == 1:
        h = np.zeros(n, dtype=float)
        m = int(round(0.5 * float(n)))
        pos = np.arange(n)
        rng.shuffle(pos)
        if m > 0:
            h[pos[:m]] = 1.0
        h += 0.08 * (rng.random(n) - 0.5)
        return project_density(h, 0.5 * float(n))

    if mode == 2:
        h = np.empty(n, dtype=float)
        pos = 0
        while pos < n:
            rem = n - pos
            length = int(rng.integers(1, rem + 1))
            val = float(rng.choice([0.0, 1.0, 0.5, 0.25, 0.75]))
            h[pos:pos + length] = val
            pos += length
        return project_density(h, 0.5 * float(n))

    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.full(n, 0.5, dtype=float)
    for m in range(1, 7):
        amp = 0.18 * (0.65 ** (m - 1)) * float(rng.uniform(0.3, 1.2))
        freq = int(max(1, round(m * float(rng.uniform(0.5, 1.5)))))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        h += amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    h += 0.02 * (rng.random(n) - 0.5)
    return project_density(h, 0.5 * float(n))


def round_start(h, n):
    h = project_density(h, 0.5 * float(n))
    n = int(n)
    b = np.zeros(n, dtype=float)
    m = int(round(0.5 * float(n)))
    if m > 0:
        order = np.argsort(-h)
        b[order[:m]] = 1.0
    return project_density(b, 0.5 * float(n))


def canonicalize_solution(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    target = 0.5 * float(n)
    best_h = h.copy()
    best_c = cw_eval(h)

    for eps in (1e-12, 1e-7, 1e-5, 1e-3):
        h2 = h.copy()
        if eps > 0.0:
            h2[h2 < eps] = 0.0
            h2[1.0 - h2 < eps] = 1.0
        h2 = project_density(h2, target)
        if not simple_feasible(h2, target):
            continue
        c2 = cw_eval(h2)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    return best_h, best_c


def optimize_pipeline(h0, n, deadline, rng):
    n = int(n)
    h = project_density(h0, 0.5 * float(n))
    c = cw_eval(h)
    best = h.copy()
    best_c = c

    now = time.perf_counter()
    span = deadline - now
    if span < 5.0:
        return best, best_c

    if n <= 450 and span > 40.0:
        if n <= 120:
            slq = min(260.0, 0.5 * span)
        elif n <= 240:
            slq = min(150.0, 0.5 * span)
        else:
            slq = min(70.0, 0.5 * span)
        slq_dead = now + slq
        try:
            h2, c2 = slsqp_smooth(best, slq_dead)
            if c2 < best_c:
                best = h2
                best_c = c2
        except Exception:
            pass

    rem = deadline - time.perf_counter()
    if rem > 20.0:
        if n <= 200:
            gd = min(120.0, 0.30 * rem)
        else:
            gd = min(60.0, 0.25 * rem)
        gd_dead = time.perf_counter() + gd
        try:
            h3, c3 = active_gradient_descent(best, gd_dead, rng)
            if c3 < best_c:
                best = h3
                best_c = c3
        except Exception:
            pass

    rem = deadline - time.perf_counter()
    if rem > 20.0:
        if n <= 120:
            M = 80
            ngrad = 20
            nrand = 250
            pts = 73
        elif n <= 360:
            M = 80
            ngrad = 16
            nrand = 180
            pts = 49
        else:
            M = 96
            ngrad = 14
            nrand = 100
            pts = 33
        try:
            h4, c4 = pair_search(best, deadline, rng, M=M, ngrad=ngrad, nrand=nrand, surrogate_points=pts)
            if c4 < best_c:
                best = h4
                best_c = c4
        except Exception:
            pass

    best = project_density(best, 0.5 * float(n))
    return best, cw_eval(best)


def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if not np.isfinite(budget) or budget <= 30.0:
        parent = safe_parent()
        if parent is not None:
            h = project_density(parent, 0.5 * float(len(parent)))
            return h, cw_eval(h), int(h.size)
        h = project_density(np.full(45, 0.5), 22.5)
        return h, cw_eval(h), 45

    deadline = start + max(10.0, budget - 25.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = safe_parent()
    if parent is not None:
        pp = project_density(parent, 0.5 * float(len(parent)))
        cp = cw_eval(pp)
        best_h = pp
        best_c = cp
        best_n = int(pp.size)

    if best_h is None:
        n0 = 90
        best_h = make_random_start(n0, rng, 0, None)
        best_c = cw_eval(best_h)
        best_n = n0

    base_n = int(best_n)
    if base_n < 2:
        base_n = 90

    max_n = 900
    ns = set()
    if base_n >= 2:
        ns.add(base_n)
    for mult in (2, 4, 8):
        n = 2 * int(round(base_n * mult / 2.0))
        if 2 <= n <= max_n:
            ns.add(n)
    if not ns:
        ns.add(90)
    nlist = sorted(ns)

    weights = []
    for n in nlist:
        if n <= max(120, base_n):
            weights.append(3.0)
        elif n <= 2 * base_n:
            weights.append(2.0)
        elif n <= 4 * base_n:
            weights.append(1.2)
        else:
            weights.append(0.7)
    total_w = float(sum(weights))

    reserve_final = min(120.0, max(30.0, 0.08 * (deadline - start)))
    alloc_total = max(1.0, (deadline - time.perf_counter()) - reserve_final)

    for idx, n in enumerate(nlist):
        now = time.perf_counter()
        rem = deadline - now
        if rem < 30.0:
            break

        frac = weights[idx] / total_w
        sub = frac * alloc_total
        if n <= 180:
            cap = 520.0
        elif n <= 360:
            cap = 360.0
        else:
            cap = 240.0
        sub = max(20.0, min(sub, cap, rem - 20.0))
        if sub <= 1.0:
            break

        starts = []
        if best_h.size == n:
            starts.append(np.asarray(best_h, dtype=float).copy())
        else:
            starts.append(upsample_h(best_h, n))

        extras = []
        if n <= 240 and sub > 120.0:
            extras.append(make_random_start(n, rng, 0, parent))
        if n <= 240 and sub > 200.0 and parent is not None:
            extras.append(round_start(upsample_h(parent, n), n))
        if n <= 240 and sub > 300.0:
            extras.append(make_random_start(n, rng, 1, parent))

        first_share = 0.6 if len(extras) > 0 else 1.0

        for si, s in enumerate(starts + extras):
            if time.perf_counter() >= deadline - 20.0:
                break
            if si == 0:
                share = sub * first_share
            else:
                share = (sub * (1.0 - first_share)) / float(len(extras))
            share = max(10.0, min(share, deadline - time.perf_counter() - 20.0))
            if si > 0 and share < 10.0:
                continue
            sub_dead = time.perf_counter() + share
            try:
                h_out, c_out = optimize_pipeline(s, n, sub_dead, rng)
            except Exception:
                continue
            if h_out is None or int(h_out.size) == 0:
                continue
            h_out = project_density(h_out, 0.5 * float(n))
            c_out = cw_eval(h_out)
            if c_out < best_c:
                best_h = h_out
                best_c = c_out
                best_n = int(n)

    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 40.0:
        nbest = int(best_h.size)

        if nbest <= 360 and rem > 80.0:
            gd_dead = time.perf_counter() + min(50.0, 0.30 * rem)
            try:
                h2, c2 = active_gradient_descent(best_h, gd_dead, rng)
                if c2 < best_c:
                    best_h = h2
                    best_c = c2
                    best_n = int(h2.size)
            except Exception:
                pass

        rem = deadline - time.perf_counter()
        if rem > 25.0:
            if nbest <= 120:
                M = 96
                ngrad = 20
                nrand = 250
                pts = 73
            elif nbest <= 360:
                M = 96
                ngrad = 18
                nrand = 200
                pts = 49
            elif nbest <= 900:
                M = 110
                ngrad = 16
                nrand = 120
                pts = 33
            else:
                M = 80
                ngrad = 10
                nrand = 50
                pts = 25
            try:
                h3, c3 = pair_search(
                    best_h,
                    time.perf_counter() + max(15.0, 0.6 * rem),
                    rng,
                    M=M,
                    ngrad=ngrad,
                    nrand=nrand,
                    surrogate_points=pts,
                )
                if c3 < best_c:
                    best_h = h3
                    best_c = c3
                    best_n = int(h3.size)
            except Exception:
                pass

    if best_h is None:
        h = project_density(np.full(45, 0.5), 22.5)
        return h, cw_eval(h), 45

    best_h, best_c = canonicalize_solution(best_h)
    if best_c >= float("inf") or not np.isfinite(best_c):
        h = project_density(np.full(45, 0.5), 22.5)
        return h, cw_eval(h), 45

    return best_h, float(best_c), int(best_h.size)
