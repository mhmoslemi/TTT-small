import numpy as np
from scipy.optimize import minimize
import functools
import time
import math


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max_val(corr, tau):
    m = float(np.max(corr))
    shifted = (corr - m) / tau
    shifted = np.clip(shifted, -500.0, 500.0)
    log_sum = m / tau + np.log(np.sum(np.exp(shifted)))
    return float(tau * log_sum)


def fix_sum(h, n):
    target = float(n) / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        per = diff / cnt
        h[mask] = np.clip(h[mask] + per, 0.0, 1.0)
    return h


def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max_val(corr, tau) * dx

    return [x.sum() - float(n) / 2.0]


def made_eq_jac(x, n):
    return np.ones((1, n))

    bounds = [(0.0, 1.0)] * n

    eq = functools.partial(made_eq, n=n)
    eq_jac = functools.partial(made_eq_jac, n=n)
    res = minimize(
        fun=obj, x0=x0, method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': eq, 'jac': eq_jac}],
        options={'maxiter': maxiter, 'ftol': ftol, 'disp': False}
    )
    return res

    bounds = [(0.0, 1.0)] * n

        h = np.clip(x, 0.0, 1.0)

        s = h.sum() - float(n) / 2.0
        return sm + lam * s * s

    res = minimize(
        fun=pen_obj, x0=x0, method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': maxiter, 'ftol': ftol, 'disp': False}
    )
    return res

    """Generate Thue-Morse binary sequence of length n."""
    t = [0]
    for _ in range(18):
        t += [1 - x for x in t]
        if len(t) >= n:
            break
    return (t[:n]).astype(float)

    """Generate binary pattern based on golden ratio rotation."""
    phi = (1.0 + math.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    alpha = 1.0 / phi
    for i in range(n):
        h[i] = 1.0 if ((alpha * i) % 1.0) < 0.5 else 0.0
    return h

    return h
