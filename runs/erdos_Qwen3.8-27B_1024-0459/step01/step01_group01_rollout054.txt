    target = n / 2.0
    lo, hi = v.min() - 2.0, v.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo+hi)/2, 0.0, 1.0)

    if abs(diff) > 1e-10:

        else:

            h = np.clip(h, 0.0, 1.0)
    return h

Hmm, the gradient computation with the loop over all k (up to 2n-1 ≈ 720 for n=360) with numpy operations inside is going to be slow. Let me think of a better way.

Actually, the gradient can be computed as:
grad = sum_k weights[k] * ((1-h) shifted by s_k) - (h shifted by s_k)

where the shifts are s_k = k - (n-1).

This is a weighted sum of shifted versions of h and (1-h). I can compute it as:
grad = weighted_shift_sum(1-h, {-s_k for k}) - weighted_shift_sum(h, {s_k for k})

where weighted_shift_sum(f, {s_k}) = sum_k weights[k] * pad_and_shift(f, s_k)

This is essentially a convolution! Specifically:
grad = convolve(1-h, w_neg) - convolve(h, w_pos)

where w_neg[s] = sum of weights[k] for which -s_k = s (i.e., s_k = -s)
and w_pos[s] = sum of weights[k] for which s_k = s

But the shifts s_k range from -(n-1) to (n-1), and there's at most one k for each s_k. So:
w_pos[s] = weights[n-1+s] for s in [0, n-1] (shifts 0 to n-1)
w_neg[s] = weights[n-1-s] for s in [0, n-1] (shifts 0 to -(n-1), negated to 0 to n-1)

So:
grad = convolve(1-h, w_neg, mode='same') - convolve(h, w_pos, mode='same')

Wait, I need to be careful with what "shifted by s" means and how convolve works.

Let me define: for a shift s, the contribution to grad from h is:
- If s ≥ 0: h[j+s] for j in [0, n-1-s], which is h shifted right by s (or left in convolve terminology)
- If s < 0: h[j+s] for j in [-s, n-1], which is h shifted left by |s|

This is getting complicated. Let me just use the direct loop but optimize it:

For each shift s (from -(n-1) to (n-1)), with weight w_s:
- Add w_s * (1-h[j-s]) to grad[j] for j where 0≤j-s<n
- Subtract w_s * h[j+s] from grad[j] for j where 0≤j+s<n

The first part: for shift s, we're adding w_s*(1-h) shifted by s to grad.
The second part: for shift s, we're subtracting w_s*h shifted by -s from grad.

So: grad = sum_s w_s * shift(1-h, s) - sum_s w_s * shift(h, -s)

where shift(f, s)[j] = f[j-s] if 0≤j-s<n, else 0.

This is: grad = convolve(w, 1-h) - convolve(w, h) (in some appropriate mode)

Actually, let me define shift(f, s) more carefully:
shift(f, s)[j] = f[j-s] for j where 0≤j-s<n, else 0.

This is the same as: pad f with zeros, then extract the window starting at position s.

In numpy: if s ≥ 0: np.roll(f, s) with the right end zeroed.
If s < 0: np.roll(f, s) with the left end zeroed.

Or more simply: 
shift(f, s) = f shifted so that element at position j of shift(f,s) = f[j-s] if valid.

This is: np.correlate(f, impulse_at_s) or just indexing.

from scipy.optimize import minimize
import time

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_feasible(v, n):
    target = n / 2.0
    # Bisection for the correct offset
    lo = -5.0
    hi = 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h = np.interp(x_new, x_old, h_old)
    return project_feasible(h, n_new)

def grad_shift(h, s, n):
    """Gradient of C[s] w.r.t. h, where s is shift offset (m - (n-1))."""
    grad = np.zeros(n)
    # Term 1: (1-h[j-s]) for j where 0 <= j-s < n
    lo1 = max(0, s)
    hi1 = min(n, n + s)
    if lo1 < hi1:
        grad[lo1:hi1] += (1.0 - h[lo1 - s:hi1 - s])
    # Term 2: -h[j+s] for j where 0 <= j+s < n
    lo2 = max(0, -s)
    hi2 = min(n, n - s)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2 + s:hi2 + s]
    return grad

def lse_objective(x, n, dx, tau):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + tau * np.log(np.sum(np.exp((corr - m) / tau)))

def lse_gradient(x, n, dx, tau):
    h = np.clip(x, 0.0, 1.0)
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr = corr_raw * dx
    m = np.max(corr)
    w = np.exp((corr - m) / tau)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-12:
            continue
        s = k - (n - 1)
        grad += w[k] * grad_shift(h, s, n)
    return grad

def subgradient_optimize(h, n, n_iter, rng, time_limit):
    """Projected subgradient with multi-active set."""
    t_start = time.time()
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)
    
    best_local = h.copy()
    best_local_c5 = compute_c5(h)
    
    # Annealing schedule: start with large step, decrease
    for it in range(n_iter):
        if time.time() - t_start > time_limit:
            break
        if it % 2000 == 0 and it > 0:
            # Check time budget
            pass
        
        cvals = compute_c5_array(h)
        cmax = np.max(cvals)
        
        # Active set: top (1-eps) of cmax, or top 20%
        threshold = cmax - 0.02  # within 0.02 of max
        active = np.where(cvals >= threshold)[0]
        
        # Also always include the argmax
        k_max = np.argmax(cvals)
        if k_max not in active:
            active = np.append(active, k_max)
        
        # Compute averaged gradient
        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            grad += grad_shift(h, s, n)
        grad /= len(active)
        
        # Step size: sublinear decay
        step = 0.8 / (1.0 + 0.1 * it ** 0.5)
        
        # Update

        h_new = project_feasible(h_new, n)
        
        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new < best_local_c5:
            best_local_c5 = c5_new
            best_local = h_new.copy()
        
        h = h_new
    
    return best_local, best_local_c5

def lse_polish(h, n, tau, maxiter, time_limit):

    dx = 2.0 / n

        return np.ones(n)

        )
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new)
    except Exception:

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_c5_array(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def project_feasible(v, n):
    target = n / 2.0
    lo = -5.0
    hi = 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h


def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return project_feasible(h_old.copy(), n_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h = np.interp(x_new, x_old, h_old)
    return project_feasible(h, n_new)


def grad_shift(h, s, n):
    grad = np.zeros(n)
    lo1 = max(0, s)
    hi1 = min(n, n + s)
    if lo1 < hi1:
        grad[lo1:hi1] += (1.0 - h[lo1 - s:hi1 - s])
    lo2 = max(0, -s)
    hi2 = min(n, n - s)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2 + s:hi2 + s]
    return grad


def lse_objective(x, n, dx, tau):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + tau * np.log(np.sum(np.exp((corr - m) / tau)))


def lse_gradient(x, n, dx, tau):
    h = np.clip(x, 0.0, 1.0)
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr = corr_raw * dx
    m = np.max(corr)
    w = np.exp((corr - m) / tau)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(w)) / len(w)
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-11:
            continue
        s = k - (n - 1)
        grad += w[k] * grad_shift(h, s, n)
    return grad


def subgradient_optimize(h_init, n, time_limit, rng):
    t_start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), n)
    best_local = h.copy()
    best_local_c5 = compute_c5(h)
    no_improve = 0

    step0 = 1.2
    decay = 0.9995
    it = 0
    while time.time() - t_start < time_limit:
        it += 1
        cvals = compute_c5_array(h)
        cmax = np.max(cvals)

        if cmax < best_local_c5 - 1e-14:
            best_local_c5 = cmax
            best_local = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 2000 and it > 500:
            h = project_feasible(h + rng.normal(0, 0.05, n), n)
            no_improve = 0

        threshold = cmax - 0.03
        active = np.where(cvals >= threshold)[0]
        k_max = int(np.argmax(cvals))
        if k_max not in active:
            active = np.append(active, k_max)

        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            grad += grad_shift(h, s, n)
        grad /= max(1, len(active))

        step = step0 * (decay ** it) / (1.0 + 0.3 * it ** 0.3)
        h_new = h - step * grad
        h = project_feasible(h_new, n)

    return best_local, best_local_c5


def slsqp_polish(h, n, tau, maxiter, time_limit):
    t_start = time.time()
    dx = 2.0 / n
    bounds_list = [(0.0, 1.0)] * n

    def obj_func(x):
        if time.time() - t_start > time_limit:
            raise RuntimeError("timeout")
        return lse_objective(x, n, dx, tau)

    con_fun = lambda x: np.array([x.sum() - n / 2.0])
    con_jac = lambda x: np.array([np.ones(n)])

    try:
        res = minimize(
            fun=obj_func, x0=h, method='SLSQP',
            bounds=bounds_list,
            constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-13, 'disp': False}
        )
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)


def mirror_descent_step(h, grad, step, n):
    h_new = h * np.exp(-step * grad)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n)


def mirror_subgradient(h_init, n, time_limit, rng):
    t_start = time.time()
    h = project_feasible(h_init.copy(), n)
    best_local = h.copy()
    best_local_c5 = compute_c5(h)
    no_improve = 0
    it = 0
    while time.time() - t_start < time_limit:
        it += 1
        cvals = compute_c5_array(h)
        cmax = np.max(cvals)
        if cmax < best_local_c5 - 1e-14:
            best_local_c5 = cmax
            best_local = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 3000 and it > 1000:
            h = project_feasible(h + rng.normal(0, 0.08, n), n)
            no_improve = 0
        threshold = cmax - 0.04
        active = np.where(cvals >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(cvals))])
        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            grad += grad_shift(h, s, n)
        grad /= max(1, len(active))
        step = 0.5 / (1.0 + 0.15 * it ** 0.4)
        h = mirror_descent_step(h, grad, step, n)
    return best_local, best_local_c5


def binary_pattern_random(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    h = project_feasible(h, n)
    return h


def binary_smooth(h_bin, n, sigma, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    from scipy.ndimage import gaussian_filter1d
    h_s = gaussian_filter1d(h_bin, sigma=sigma)
    h_s = rng.normal(h_s, 0.02, n)
    return project_feasible(h_s, n)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None and np.size(initial_h_values) > 0:

    if time_left() > 80:

        # Mirror descent

               (pol_time = min(40.0, time_left() * 0.2))
