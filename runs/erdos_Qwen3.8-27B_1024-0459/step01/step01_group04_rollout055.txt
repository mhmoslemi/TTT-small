    h = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    # Fix residual
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() > 0:
            per = diff / mask.sum()
            h[mask] += per
            h = np.clip(h, 0.0, 1.0)
    return h

def scores_from_h(h, n):
    """Compute all 2n-1 overlap scores."""
    corr = np.correlate(h, h, mode="full")
    ac = corr[n-1:n-1+n]
    prefix = np.zeros(n+1)
    np.cumsum(h, out=prefix[1:])
    total = prefix[n]
    scores = np.zeros(2*n-1)
    scores[0] = total - ac[0]
    for d in range(1, n):
        scores[2*d-1] = prefix[n-d] - ac[d]
        scores[2*d] = total - prefix[d] - ac[d]
    return scores
