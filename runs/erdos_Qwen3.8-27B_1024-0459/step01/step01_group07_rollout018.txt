def compute_max_conv(h, n_points):

def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0
            count = np.sum(free)
            if count == 0:
                break
            h[free] = np.minimum(h[free] + diff / count, 1.0)
        else:
            free = h > 0.0
            count = np.sum(free)
            if count == 0:
                break
            h[free] = np.maximum(h[free] + diff / count, 0.0)
    h = np.clip(h, 0.0, 1.0)

        free = (h > 0) & (h < 1)
        if np.sum(free) > 0:
            h[free] += (target - s) / np.sum(free)
            h = np.clip(h, 0, 1)
    return h

def mod_binary(n, p):

def m_sequence_binary(n, poly):

        feedback = sum(state[k] * poly[k] for k in range(len(poly))) % 2
        state = [feedback] + state[:-1]

    h_out = project_feasible(res.x, n)
