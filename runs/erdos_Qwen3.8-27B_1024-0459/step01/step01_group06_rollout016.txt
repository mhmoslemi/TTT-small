```python
def compute_delta(h, a, b, n):
    """Compute the change in C = correlate(h, 1-h, 'full') when swapping h[a] and h[b]."""
    # Assume h[a]=1, h[b]=0
    dC = np.zeros(2*n-1)
    
    for k_idx in range(2*n-1):
        # k_idx corresponds to lag = k_idx - (n-1)
        lag = k_idx - (n-1)
        delta = 0.0
        
        # a as i+k: i = a-k = a-lag, valid if 0 <= a-lag < n
        if 0 <= a - lag < n:
            delta -= (1.0 - h[a - lag])  # h[a] goes 1→0
        
        # a as i: i = a, term is h[a+lag], valid if 0 <= a+lag < n
        if 0 <= a + lag < n:
            delta += h[a + lag]  # (1-h[a]) goes 0→1
        
        # b as i+k: i = b-lag, valid if 0 <= b-lag < n
        if 0 <= b - lag < n:
            delta += (1.0 - h[b - lag])  # h[b] goes 0→1
        
        # b as i: i = b, term is h[b+lag], valid if 0 <= b+lag < n
        if 0 <= b + lag < n:
            delta -= h[b + lag]  # (1-h[b]) goes 1→0
        
        dC[k_idx] = delta
    
    return dC
```

```python
def compute_delta_fast(h, a, b, n):
    """Vectorized delta computation."""
    lags = np.arange(-(n-1), n)  # length 2n-1
    
    # f(a, lag) = -(1-h[a-lag]) [valid if 0<=a-lag<n] + h[a+lag] [valid if 0<=a+lag<n]
    idx1 = a - lags  # positions a-lag
    mask1 = (idx1 >= 0) & (idx1 < n)
    idx2 = a + lags  # positions a+lag
    mask2 = (idx2 >= 0) & (idx2 < n)
    
    f = np.zeros(2*n-1)
    f[mask1] -= (1.0 - h[idx1[mask1]])
    f[mask2] += h[idx2[mask2]]
    
    # g(b, lag) = +(1-h[b-lag]) [valid if 0<=b-lag<n] + (-h[b+lag]) [valid if 0<=b+lag<n]
    idx3 = b - lags
    mask3 = (idx3 >= 0) & (idx3 < n)
    idx4 = b + lags
    mask4 = (idx4 >= 0) & (idx4 < n)
    
    g = np.zeros(2*n-1)
    g[mask3] += (1.0 - h[idx3[mask3]])
    g[mask4] -= h[idx4[mask4]]
    
    return f + g
```
