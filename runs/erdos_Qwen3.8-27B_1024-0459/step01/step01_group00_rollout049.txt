```python
def compute_gradient_vec(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    w = np.exp(beta * (corr - cmax))
    w /= np.sum(w)
    
    grad = np.zeros(n)
    
    # For each lag k (positive and negative), add weighted gradient
    # Positive lags: m = n-1+k for k=0,...,n-1
    # Negative lags: m = n-1-k for k=1,...,n-1
    
    for k in range(n):  # positive lag k (m = n-1+k)
        wk = w[n-1+k]
        if wk < 1e-15:
            continue
        if k == 0:
            # S(0) = sum h[i](1-h[i]), d/dh[j] = 1-2h[j]
            grad += wk * (1.0 - 2.0*h)
        else:
            # d/dh[j] = (1-h[j-k]) for j>=k, -h[j+k] for j<=n-1-k
            grad[k:] += wk * (1.0 - h[:-k])
            grad[:n-k] -= wk * h[k:]
    
    for k in range(1, n):  # negative lag -k (m = n-1-k)
        wk = w[n-1-k]
        if wk < 1e-15:
            continue
        # corr = sum h[i](1-h[i+k]) for i=0..n-1-k
        # d/dh[j] = (1-h[j+k]) for j<=n-1-k, -h[j-k] for j>=k
        grad[:n-k] += wk * (1.0 - h[k:])
        grad[k:] -= wk * h[:n-k]
    
    return grad * dx
```

This is O(n²) with numpy vectorized inner operations. For n=256, that's about 256*256 = 65k operations per gradient, which is very fast.

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_c5(x, beta):
    x = np.clip(x, 0.0, 1.0)
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    if cmax < 0:
        cmax = 0
    return (cmax + (1.0/beta) * np.log(np.sum(np.exp(beta * (corr - cmax))))) * dx

def sum_constraint(x, n):
    return np.sum(x) - n / 2.0

def compute_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    w = np.exp(beta * (corr - cmax))
    w /= np.sum(w)
    
    grad = np.zeros(n)
    
    # Positive lags (including 0)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-14:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            grad[k:] += wk * (1.0 - h[:n-k])
            grad[:n-k] -= wk * h[k:]
    
    # Negative lags
    for k in range(1, n):
        wk = w[n - 1 - k]
        if wk < 1e-14:
            continue
        grad[:n-k] += wk * (1.0 - h[k:])
        grad[k:] -= wk * h[:n-k]
    
    return grad * dx

def project_feasible(y, n):
    """Project y onto {0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    # Bisection on shift
    y = np.asarray(y, dtype=float)
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def generate_binary_init(n, seed):
    rng = np.random.RandomState(seed)
    # Random binary with n/2 ones
    idx = rng.choice(n, n // 2, replace=False)

def generate_structured_init(n, seed):
    rng = np.random.RandomState(seed)
    inits = []
    
    # Binary: two blocks
    h1 = np.zeros(n)
    h1[:n//4] = 1.0
    h1[n//2:n//2+n//4] = 1.0
    inits.append(h1.copy())
    
    # Binary: alternating
    if n % 2 == 0:
        h2 = np.zeros(n)
        h2[::2] = 1.0
        inits.append(h2.copy())
    
    # Binary: periodic with period p
    for p in [3, 5, 7, 11]:
        if n % p == 0:
            h3 = np.zeros(n)
            # pattern: 1 for first floor(p/2) elements of each period
            block = np.zeros(p)
            block[:p//2] = 1.0
            h3 = np.tile(block, n // p)
            inits.append(h3.copy())
    
    # Greedy: spread ones as evenly as possible
    h4 = np.zeros(n)
    positions = np.linspace(0, n-1, n//2, dtype=int)
    h4[positions] = 1.0
    inits.append(h4.copy())
    
    # Thue-Morse
    h5 = np.zeros(n)
    for i in range(n):
        h5[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    # Fix to have exactly n/2 ones
    ones = np.sum(h5).astype(int)
    if ones > n // 2:
        # Remove excess
        ones_idx = np.where(h5 == 1)[0]
        remove = ones_idx[:ones - n//2]
        h5[remove] = 0
    elif ones < n // 2:
        zeros_idx = np.where(h5 == 0)[0]
        add = zeros_idx[:n//2 - ones]
        h5[add] = 1
    inits.append(h5.copy())
    
    # Random binary
    for _ in range(3):
        inits.append(generate_binary_init(n, rng.randint(100000)))

def interpolate_to(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_new = np.interp(x_new, x_old, h_src)
    return project_feasible(h_new, n_dst)

def slsqp_optimize(h, n, beta, max_iter=500):
    h0 = np.clip(h, 0, 1).copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}]
    try:
        res = minimize(smooth_max_c5, h0, args=(beta,), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.fun < 1e10:
            h_out = project_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None
