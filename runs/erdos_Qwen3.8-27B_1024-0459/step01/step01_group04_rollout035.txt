        T *= cool

    return best_h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    global _RUN_DEADLINE
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0:
        budget_s = 30.0
    reserve = 30.0
    global_deadline = t0 + max(1.0, budget_s - reserve)
    _RUN_DEADLINE = global_deadline

    rng = np.random.default_rng(seed)

    # Get parent solution
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        if parent.ndim != 1 or parent.size < 2:
            raise ValueError
    except Exception:
        parent = None

    if parent is not None and parent.size > 0:
        n0 = int(parent.size)
        h0 = fix_integral(parent.copy())
        c0 = compute_c5(h0)
    else:
        n0 = 70
        h0 = fix_integral(np.full(n0, 0.5))
        c0 = compute_c5(h0)

    best_h = h0.copy()
    best_c = c0
    best_n = n0

    def try_update(h, c, n):
        nonlocal best_h, best_c, best_n
        if h is None or not math.isfinite(c):
            return
        h = fix_integral(h)
        c2 = compute_c5(h)
        if c2 < best_c - 1e-14:
            best_h = h.copy()
            best_c = float(c2)
            best_n = int(n)

    # Phase 1: Subgradient descent at parent resolution
    remaining = float(global_deadline - time.time())
    if remaining > 60:
        t_limit = min(remaining * 0.35, 300.0)
        try:
            h_sub, c_sub = subgradient_descent(h0, n0, t_limit, rng, h0, c0)
            try_update(h_sub, c_sub, n0)
        except Exception:
            pass

    # Phase 2: Coordinate descent at parent resolution
    remaining = float(global_deadline - time.time())
    if remaining > 40:
        t_limit = min(remaining * 0.3, 200.0)
        try:
            h_cd, c_cd = coordinate_descent(best_h, best_n, t_limit, rng, best_h, best_c)
            try_update(h_cd, c_cd, best_n)
        except Exception:
            pass

    # Phase 3: Pair descent
    remaining = float(global_deadline - time.time())
    if remaining > 30:
        t_limit = min(remaining * 0.2, 120.0)
        try:
            h_pd, c_pd = pair_descent(best_h, best_n, t_limit, rng)
            try_update(h_pd, c_pd, best_n)
        except Exception:
            pass

    # Phase 4: Try higher resolution
    for n_new in (int(n0 * 1.5), n0 * 2, int(n0 * 2.5)):
        n_new = int(n_new)
        if n_new < 4 or n_new > 400:
            continue
        remaining = float(global_deadline - time.time())
        if remaining < 80:
            break
        t_limit = min(remaining * 0.25, 200.0)
        if t_limit < 50:
            continue
        try:
            h_up = upsample(best_h, n_new)
            h_up = fix_integral(h_up)
            c_up = compute_c5(h_up)

            # Subgradient at higher resolution
            t_sub = t_limit * 0.5
            t_sub = min(t_sub, max(20.0, float(global_deadline - time.time()) - 30.0))
            if t_sub > 20:
                h_sub, c_sub = subgradient_descent(h_up, n_new, t_sub, rng, h_up, c_up)
                try_update(h_sub, c_sub, n_new)

            # Coordinate descent at higher resolution
            remaining2 = float(global_deadline - time.time())
            if remaining2 > 50:
                t_cd = min(remaining2 * 0.3, 100.0)
                if t_cd > 30:
                    h_cd, c_cd = coordinate_descent(best_h, best_n, t_cd, rng, best_h, best_c)
                    try_update(h_cd, c_cd, best_n)

        except Exception:
            pass

    # Phase 5: Final refinement at best resolution
    remaining = float(global_deadline - time.time())
    if remaining > 30:
        t_limit = min(remaining * 0.6, 200.0)
        # Subgradient again
        t_sub = t_limit * 0.5
        if t_sub > 20:
            try:
                h_sub, c_sub = subgradient_descent(best_h, best_n, t_sub, rng, best_h, best_c)
                try_update(h_sub, c_sub, best_n)
            except Exception:
                pass
        # Anneal
        remaining2 = float(global_deadline - time.time())
        if remaining2 > 30:
            t_ann = min(remaining2 * 0.6, 100.0)
            if t_ann > 20:
                try:
                    h_an, c_an = anneal_refine(best_h, best_n, t_ann, rng)
                    try_update(h_an, c_an, best_n)
                except Exception:
                    pass
        # Final coordinate descent
        remaining3 = float(global_deadline - time.time())
        if remaining3 > 20:
            try:
                h_cd, c_cd = coordinate_descent(best_h, best_n, min(remaining3 - 5, 80.0), rng, best_h, best_c)
                try_update(h_cd, c_cd, best_n)
            except Exception:
                pass

    # Phase 6: Smooth and refine
    if best_n >= 10:
        remaining = float(global_deadline - time.time())
        if remaining > 20:
            try:
                h_smooth = smooth_h(best_h, sigma=1.0)
                h_smooth = fix_integral(h_smooth)
                c_smooth = compute_c5(h_smooth)
                if c_smooth < best_c:
                    h_iters = min(remaining * 0.5, 60.0)
                    if h_iters > 10:
                        h_ref, c_ref = subgradient_descent(h_smooth, best_n, h_iters, rng, h_smooth, c_smooth)
                        try_update(h_ref, c_ref, best_n)
                    else:
                        try_update(h_smooth, c_smooth, best_n)
            except Exception:
                pass

    # Validate and return
    best_h = fix_integral(best_h)
    best_c = compute_c5(best_h)
    if not math.isfinite(best_c):
        best_h = fix_integral(h0)
        best_c = compute_c5(best_h)
        best_n = n0

    # Ensure we don't return worse than the original parent
    if parent is not None:
        h0_check = fix_integral(parent.copy())
        c0_check = compute_c5(h0_check)
        if c0_check < best_c:
            best_h = h0_check
            best_c = c0_check
            best_n = n0

    return best_h, float(best_c), int(best_n)
