import time
import numpy as np
from scipy.optimize import minimize

def project_density(x, target):
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-13:
        return x
    # Bisection for the correct offset
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-14:
        free = np.where((h > 1e-15) & (h < 1.0 - 1e-15))[0]
        if free.size > 0:
            per = res / float(free.size)
            h[free] = np.clip(h[free] + per, 0.0, 1.0)
    return h
