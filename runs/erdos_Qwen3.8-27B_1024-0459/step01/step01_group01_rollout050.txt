    n = len(h)

    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_up = np.interp(x_new, x_old, h_old)
    return project_feasible(h_up, n_new)


def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    z = np.clip((corr - m) / tau, -50, 50)
    exp_z = np.exp(z)
    sum_exp = np.sum(exp_z)
    obj = (m + tau * np.log(sum_exp)) * dx
    weights = exp_z / sum_exp
    
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w_m = weights[m_idx]
        if w_m < 1e-12:
            continue

        lo = max(0, k)

        if hi > lo:
            grad[lo:hi] += w_m * (1.0 - h[lo - k:hi - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
