_DEADLINE = None

class BudgetError(Exception):

def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 1e6
    h = np.clip(h, 0.0, 1.0)
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * (2.0 / n))

def as_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    if h.size != n:
        h = np.resize(h, n)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5*(lo+hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h

def upsample(parent, n):
    n0 = parent.size
    if n == n0:
        return parent.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0/n0)
    x1 = (np.arange(n) + 0.5) * (2.0/n)
    return np.interp(x1, x0, parent)

def coord_sweep(h, n):
    """One full sweep of coordinate descent."""
    for i in range(n):
        check_dl()
        # Compute A and C for this coordinate
        lags = np.arange(-(n-1), n, dtype=int)
        A = np.zeros(2*n-1)
        C = np.zeros(2*n-1)
        for k in range(2*n-1):
            l = k - (n-1)
            a_k = 0.0
            c_k = 0.0
            if 0 <= i+l < n:
                a_k += (1.0 - h[i+l])
            if 0 <= i-l < n:
                a_k -= h[i-l]
                c_k += h[i-l]
            A[k] = a_k
            C[k] = c_k
        
        # Minimize max(C + A*x) over x in [0,1]
        # Ternary search (convex function)
        lo, hi = 0.0, 1.0
        for _ in range(40):
            m1 = lo + (hi-lo)/3.0
            m2 = hi - (hi-lo)/3.0
            f1 = float(np.max(C + A*m1))
            f2 = float(np.max(C + A*m2))
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        h[i] = 0.5*(lo+hi)
    return h

def coord_sweep(h, n):
    conv = np.correlate(h, 1.0-h, mode="full")
    for i in range(n):
        check_dl()
        # Compute A[k] = coefficient of h[i] in conv[k]
        A = np.zeros(2*n-1)
        # Condition: i+l in [0,n-1] where l = k-(n-1)
        #   k in [n-1-i, 2n-2-i]
        lo1, hi1 = n-1-i, 2*n-2-i
        if lo1 <= hi1:
            ls = np.arange(lo1, hi1+1) - (n-1)  # l values
            A[lo1:hi1+1] += (1.0 - h[i + ls])
        # Condition: i-l in [0,n-1]
        #   k in [i, n-1+i]
        lo2, hi2 = i, n-1+i
        if lo2 <= hi2:
            ls = np.arange(lo2, hi2+1) - (n-1)  # l values
            A[lo2:hi2+1] += (-h[i - ls])
        
        # C[k] = conv[k] - h[i]*A[k]
        C = conv - h[i] * A
        
        # Minimize max(C + A*x) over x in [0,1]
        lo, hi = 0.0, 1.0
        for _ in range(40):

            m2 = hi - (hi-lo)/3.0

            f2 = float(np.max(C + A*m2))
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        x_new = 0.5*(lo+hi)
        
        # Update conv
        delta = x_new - h[i]
        if abs(delta) > 1e-14:
            conv += A * delta
            h[i] = x_new
    return h

import numpy as np
import time
import math
from scipy.optimize import minimize

_DEADLINE = None

class BudgetError(Exception):
    pass

def check_dl():
    if _DEADLINE is not None and time.time() > _DEADLINE:
        raise BudgetError

def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 1e6
    h = np.clip(h, 0.0, 1.0)
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * (2.0 / n))

def as_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    if h.size != n:
        h = np.resize(h, n)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h

def upsample(parent, n):
    n0 = parent.size
    if n == n0:
        return parent.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x1, x0, parent)

def coord_sweep(h, n, max_sweeps_inner=1):
    """One or more full coordinate descent sweeps."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    for sweep in range(max_sweeps_inner):
        for i in range(n):
            check_dl()
            A = np.zeros(2 * n - 1)
            lo1, hi1 = n - 1 - i, 2 * n - 2 - i
            if lo1 <= hi1:
                ls = np.arange(lo1, hi1 + 1) - (n - 1)
                A[lo1:hi1 + 1] += (1.0 - h[i + ls])
            lo2, hi2 = i, n - 1 + i
            if lo2 <= hi2:
                ls = np.arange(lo2, hi2 + 1) - (n - 1)
                A[lo2:hi2 + 1] += (-h[i - ls])
            C = conv - h[i] * A
            lo, hi = 0.0, 1.0
            for _ in range(45):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = float(np.max(C + A * m1))
                f2 = float(np.max(C + A * m2))
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            x_new = 0.5 * (lo + hi)
            delta = x_new - h[i]
            if abs(delta) > 1e-14:
                conv += A * delta
                h[i] = x_new
    return h

def coord_sweep_shuffled(h, n, rng, max_sweeps_inner=1):
    """Coordinate descent with random order."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    for sweep in range(max_sweeps_inner):
        order = rng.permutation(n)
        for i in order:
            check_dl()
            i = int(i)
            A = np.zeros(2 * n - 1)
            lo1, hi1 = n - 1 - i, 2 * n - 2 - i
            if lo1 <= hi1:
                ls = np.arange(lo1, hi1 + 1) - (n - 1)
                A[lo1:hi1 + 1] += (1.0 - h[i + ls])
            lo2, hi2 = i, n - 1 + i
            if lo2 <= hi2:
                ls = np.arange(lo2, hi2 + 1) - (n - 1)
                A[lo2:hi2 + 1] += (-h[i - ls])
            C = conv - h[i] * A
            lo, hi = 0.0, 1.0
            for _ in range(45):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = float(np.max(C + A * m1))
                f2 = float(np.max(C + A * m2))
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            x_new = 0.5 * (lo + hi)
            delta = x_new - h[i]
            if abs(delta) > 1e-14:
                conv += A * delta
                h[i] = x_new
    return h

def get_scores(h, n):
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv

def slsqp_obj(x, alpha, n):
    check_dl()
    h = np.clip(x, 0.0, 1.0)
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(conv))
    if alpha > 1e-14:
        w = np.exp((conv - m) / alpha)
        s = float(np.sum(w))
        if math.isfinite(s) and s > 0:
            val = m + alpha * math.log(s)
        else:
            val = m
    else:
        val = m
    return val * (2.0 / n)

def slsqp_grad(x, alpha, n):
    check_dl()
    h = np.clip(x, 0.0, 1.0)
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(conv))
    grad = np.zeros(n)
    if alpha > 1e-14:
        w = np.exp((conv - m) / alpha)
        A = float(np.sum(w))
        if math.isfinite(A) and A > 0:
            pw = w / A
            for k in range(2 * n - 1):
                if pw[k] < 1e-12:
                    continue
                l = k - (n - 1)
                lo1, hi1 = n - 1 - 0, 2 * n - 2 - 0
                for i in range(n):
                    coeff = 0.0
                    if 0 <= i + l < n:
                        coeff += (1.0 - h[i + l])
                    if 0 <= i - l < n:
                        coeff -= h[i - l]
                    grad[i] += pw[k] * coeff
    else:
        tol = max(1e-8, 1e-7 * max(1.0, m))
        active = np.flatnonzero(conv >= m - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(conv))])
        coef = 1.0 / float(len(active))
        for k in active:
            l = k - (n - 1)
            for i in range(n):
                c = 0.0
                if 0 <= i + l < n:
                    c += (1.0 - h[i + l])
                if 0 <= i - l < n:
                    c -= h[i - l]
                grad[i] += coef * c
    grad *= (2.0 / n)
    return grad

def slsqp_eq(x, n):
    return (float(np.sum(x)) - 0.5 * float(n)) / float(n)

def slsqp_jac(x, n):
    return np.full((1, len(x)), 1.0 / float(len(x)))

def run_slsqp(h_start, n, time_limit):
    start_t = time.time()
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": lambda x: slsqp_eq(x, n),
                    "jac": lambda x: slsqp_jac(x, n)}]
    
    alpha_start = max(0.3, 0.005 * n)
    maxiter1 = max(30, min(200, int(80.0 * (100.0 / n))))
    
    dl1 = start_t + 0.5 * time_limit
    if _DEADLINE is not None:
        dl1 = min(dl1, _DEADLINE)
    old_dl = _DEADLINE
    _DEADLINE = dl1
    try:
        res = minimize(slsqp_obj, h_start.copy(), args=(alpha_start, n),
                       jac=lambda x: slsqp_grad(x, alpha_start, n),
                       method="SLSQP", bounds=bounds, constraints=constraints,
                       options={"maxiter": maxiter1, "ftol": 1e-10, "disp": 0})
    except BudgetError:
        res = None
    except Exception:
        res = None
    finally:
        _DEADLINE = old_dl
    
    if res is not None and getattr(res, 'x', None) is not None:
        h_mid = as_feasible(res.x, n)
    else:
        h_mid = h_start.copy()
    
    if time.time() < start_t + time_limit - 5.0:
        dl2 = start_t + time_limit
        if _DEADLINE is not None:
            dl2 = min(dl2, _DEADLINE)
        _DEADLINE = dl2
        maxiter2 = max(40, min(300, int(120.0 * (100.0 / n))))
        try:
            res2 = minimize(slsqp_obj, h_mid, args=(0.0, n),
                            jac=lambda x: slsqp_grad(x, 0.0, n),
                            method="SLSQP", bounds=bounds, constraints=constraints,
                            options={"maxiter": maxiter2, "ftol": 1e-10, "disp": 0})
        except BudgetError:
            res2 = None
        except Exception:
            res2 = None
        finally:
            _DEADLINE = old_dl
        if res2 is not None and getattr(res2, 'x', None) is not None:
            h_mid = as_feasible(res2.x, n)
    
    _DEADLINE = old_dl
    return h_mid

def synthetic_starts(n, rng, parent):
    """Generate diverse starting points."""
    starts = []
    x = (np.arange(n) + 0.5) * (2.0 / n)
    
    if parent is not None:
        starts.append(as_feasible(upsample(parent, n), n))
        h_noisy = upsample(parent, n) + 0.05 * rng.standard_normal(n)
        starts.append(as_feasible(h_noisy, n))
    
    for f in (1, 2, 3):
        for ph in (0.0, math.pi / 2):
            h = 0.5 + 0.3 * np.cos(math.pi * f * x + ph)
            starts.append(as_feasible(h, n))
    
    if n % 2 == 0:
        target = n // 2
        h_bin = np.zeros(n)
        idx = rng.choice(n, size=target, replace=False)
        h_bin[idx] = 1.0
        h_smooth = h_bin.copy()
        for _ in range(3):
            new_h = h_smooth.copy()
            if n > 2:
                new_h[1:-1] = 0.25 * h_smooth[:-2] + 0.5 * h_smooth[1:-1] + 0.25 * h_smooth[2:]
            h_smooth = new_h
        starts.append(as_feasible(h_smooth, n))
    
    return starts

def optimize_at_resolution(n, starts, state_best, state_c, time_budget, rng):
    """Run coordinate descent + SLSQP at given resolution."""
    start_t = time.time()
    best_h = state_best if state_best is not None and state_best.size == n else None
    best_c = state_c
    
    for si, s in enumerate(starts):
        if time.time() > start_t + time_budget - 30.0:
            break
        h = as_feasible(s, n)
        c = compute_c5(h)
        if not math.isfinite(c):
            continue
        if best_h is None or c < best_c:
            best_h = h.copy()
            best_c = c
        
        remaining = start_t + time_budget - 30.0 - time.time()
        if remaining < 10.0:
            break
        
        # Coordinate descent phase
        cd_time = min(remaining * 0.6, 200.0)
        cd_end = time.time() + cd_time
        old_dl = _DEADLINE
        _DEADLINE = min(cd_end, _DEADLINE) if _DEADLINE is not None else cd_end
        
        prev_c = c
        stall = 0
        sweeps_done = 0
        max_sweeps = 2000
        while time.time() < cd_end and sweeps_done < max_sweeps:
            h = coord_sweep(h, n, max_sweeps_inner=3)
            h = as_feasible(h, n)
            c = compute_c5(h)
            sweeps_done += 1
            if c < best_c - 1e-10:
                best_c = c
                best_h = h.copy()
            if prev_c - c < 1e-12:
                stall += 1
                if stall > 20:
                    break
            else:
                stall = 0
            prev_c = c
            # Occasionally shuffle
            if sweeps_done % 50 == 0:
                h = coord_sweep_shuffled(h, n, rng, max_sweeps_inner=3)
                h = as_feasible(h, n)
        _DEADLINE = old_dl
        
        if math.isfinite(c) and c < best_c:
            best_c = c
            best_h = h.copy()
    
    if best_h is not None and not np.allclose(best_h, h, atol=1e-7):
        best_h = h.copy()
        best_c = compute_c5(best_h)
    
    # SLSQP refinement on best_h
    if best_h is not None:
        remaining = start_t + time_budget - 10.0 - time.time()
        if remaining > 30.0:
            sl_time = min(remaining * 0.5, 120.0)
            h_sl = run_slsqp(best_h, n, sl_time)
            c_sl = compute_c5(h_sl)
            if c_sl < best_c:
                best_c = c_sl
                best_h = h_sl
    
    # Final coordinate descent polish
    if best_h is not None:
        remaining = start_t + time_budget - 5.0 - time.time()
        if remaining > 10.0:
            old_dl = _DEADLINE
            end_t = time.time() + remaining * 0.7
            if _DEADLINE is not None:
                end_t = min(end_t, _DEADLINE)
            _DEADLINE = end_t
            sweeps = 0
            prev_c = compute_c5(best_h)
            while time.time() < end_t and sweeps < 500:
                best_h = coord_sweep(best_h, n, max_sweeps_inner=2)
                best_h = as_feasible(best_h, n)
                c = compute_c5(best_h)
                sweeps += 1
                if prev_c - c < 1e-13:
                    if sweeps > 100:
                        break
                else:
                    prev_c = c
            _DEADLINE = old_dl
            c_final = compute_c5(best_h)
            if c_final < best_c:
                best_c = c_final
                best_h = best_h.copy()
    
    return best_h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    global _DEADLINE
    
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0:
        budget_s = 30.0
    reserve = 25.0
    global_deadline = t0 + max(5.0, budget_s - reserve)
    _DEADLINE = global_deadline
    
    rng = np.random.default_rng(seed)
    
    parent = None
    n_parent = 70
    try:
        p = np.asarray(initial_h_values, dtype=float)
        if p.ndim == 1 and p.size >= 4:
            parent = p.copy()
            n_parent = p.size
    except Exception:
        pass
    
    if parent is None:
        x = (np.arange(n_parent) + 0.5) * (2.0 / n_parent)
        parent = as_feasible(0.5 + 0.3 * np.cos(math.pi * x), n_parent)
        n_parent = parent.size
    
    c_parent = compute_c5(parent)
    best_h = parent.copy()
    best_c = c_parent
    
    resolutions = [n_parent]
    for mult in (1.4, 2.0, 2.8):
        nn = int(round(n_parent * mult / 2.0) * 2)
        if nn >= 20 and nn <= 400 and nn not in resolutions:
            resolutions.append(nn)
    resolutions.sort()
    
    total_time = max(50.0, budget_s - reserve - 50.0)
    n_res = len(resolutions)
    # Allocate time: more to smaller n (parent resolution gets most)
    weights = []
    for r in resolutions:
        w = 1.0 / (1.0 + 0.3 * (r - n_parent) / n_parent)
        weights.append(w)
    total_w = sum(weights)
    time_allocs = [total_time * w / total_w for w in weights]
    
    for ri, n in enumerate(resolutions):
        if time.time() > global_deadline - 60.0:
            break
        tb = min(time_allocs[ri], max(30.0, global_deadline - time.time() - 40.0))
        if tb < 25.0:
            break
        
        if n == n_parent:
            starts = [parent.copy()]
            starts.append(as_feasible(parent + 0.08 * rng.standard_normal(n_parent), n))
        else:
            starts = synthetic_starts(n, rng, parent)
            if best_h is not None and best_h.size != n:
                starts.append(as_feasible(upsample(best_h, n), n))
            elif best_h is not None and best_h.size == n:
                starts.append(best_h.copy())
        
        h_new, c_new = optimize_at_resolution(n, starts, best_h if best_h.size == n else None,
                                              compute_c5(best_h) if best_h.size == n else 1e9,
                                              tb, rng)
        if h_new is not None and c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
    
    # Final verification
    if best_h is not None:
        best_h = as_feasible(best_h, best_h.size)
        best_c = compute_c5(best_h)
    
    if not math.isfinite(best_c):
        best_h = as_feasible(parent, n_parent)
        best_c = compute_c5(best_h)
    
    _DEADLINE = None
    return best_h, float(best_c), int(best_h.size)
