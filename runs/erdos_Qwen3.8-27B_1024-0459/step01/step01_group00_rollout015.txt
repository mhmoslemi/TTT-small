    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def projected_subgradient(n, h0, n_iters, lr_start=0.01, lr_end=0.001, rng=None):
    h = make_feasible(h0.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iters):
        t = it / max(n_iters - 1, 1)
        lr = lr_start * (lr_end / lr_start) ** t
        val, grad = c5_max_obj_grad(h, 1e-5)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.arange(len(h_src)) * 2.0 / len(h_src)
    x_new = np.arange(n_dst) * 2.0 / n_dst
    return np.interp(x_new, x_old, h_src)

def local_perturb(h, n, n_trials, step_size, rng):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for _ in range(n_trials):
        h_try = best_h.copy()
        k = int(rng.randint(1, 6))
        idxs = rng.choice(n, k, replace=False)
        delta = rng.uniform(-step_size, step_size)
        h_try[idxs] += delta
        h_try = np.clip(h_try, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h_try)
        mask = (h_try > 0.001) & (h_try < 0.999)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h_try[mask] += diff / cnt
        h_try = np.clip(h_try, 0.0, 1.0)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()
    return best_h, best_c5

def gradient_local_search(h, n, n_trials, lr=0.005, rng=None):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for _ in range(n_trials):
        val, grad = c5_max_obj_grad(best_h, 1e-5)
        if rng is not None:
            noise = rng.randn(n) * 0.001
            direction = -grad + noise
        else:
            direction = -grad
        h_try = best_h + lr * direction
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()
    return best_h, best_c5

def generate_structured_inits(n, rng):
    inits = []
    if n % 4 == 0:
        inits.append(np.tile([1.0, 1.0, 0.0, 0.0], n // 4))
        inits.append(np.tile([1.0, 0.0, 1.0, 0.0], n // 4))
    if n % 2 == 0:
        h3 = np.zeros(n)
        h3[:n // 4] = 1.0
        h3[n // 2:n // 2 + n // 4] = 1.0
        inits.append(h3)
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    inits.append(make_feasible(tm, n))
    h4 = np.zeros(n)
    positions = sorted(range(n), key=lambda i: (i * 7919) % n)
    for j in range(n // 2):
        h4[positions[j]] = 1.0
    inits.append(make_feasible(h4, n))
    # Von Neumann / golden ratio
    for frac in [0.382, 0.618, 0.5]:
        hv = np.zeros(n)
        for i in range(n):
            pos = (i * frac * n) % n
            hv[int(pos)] = 1.0
        for j in range(n // 2):
            idx = int((j * frac * n) % n)
            hv[idx] = 1.0
        inits.append(make_feasible(hv, n))
    return inits

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is not None and len(h_parent) == 0:
            h_parent = None
    except Exception:
        h_parent = None

    if h_parent is not None:
        best_h = h_parent.copy()
        best_n = len(best_h)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5

    best_c5 = compute_c5(best_h)

    # Phase 1: Refine parent at its resolution with analytical-gradient SLSQP
    if remaining() > 80:
        h_opt, c5_val = slsqp_max(best_n, best_h, max_iter=3000, eps_active=5e-7)
        if h_opt is not None and c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h_opt.copy()

    # Smooth anneal at parent resolution
    if remaining() > 50:
        h_opt2, c5_val2 = smooth_anneal(best_n, best_h, betas=[5, 15, 50, 150], max_iter_each=800)
        if c5_val2 < best_c5 - 1e-12:
            best_c5 = c5_val2
            best_h = h_opt2.copy()

    # Projected subgradient at parent
    if remaining() > 30:
        h_sub, c5_sub = projected_subgradient(best_n, best_h, n_iters=500, lr_start=0.02, lr_end=0.002, rng=rng)
        if c5_sub < best_c5 - 1e-12:
            best_c5 = c5_sub
            best_h = h_sub.copy()

    # Phase 2: Try higher resolutions
    res_tried = [best_n]
    for n_try in [200, 320, 480, 640, 800]:
        if remaining() < 80:
            break
        if n_try in res_tried:
            continue
        res_tried.append(n_try)

        h_interp = make_feasible(interpolate_to_n(best_h, n_try), n_try)

        # SLSQP with gradient at this resolution
        h_opt, c5_val = slsqp_max(n_try, h_interp, max_iter=1500, eps_active=5e-7)
        if h_opt is not None and c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h_opt.copy()
            best_n = n_try

        # Smooth anneal at this resolution
        if remaining() > 40:
            h_opt2, c5_val2 = smooth_anneal(n_try, make_feasible(interpolate_to_n(best_h, n_try), n_try),
                                             betas=[3, 10, 30, 100], max_iter_each=600)
            if c5_val2 < best_c5 - 1e-12:
                best_c5 = c5_val2
                best_h = h_opt2.copy()
                best_n = n_try

    # Phase 3: Diverse initializations at a moderate resolution
    n_div = max(best_n, 160)
    if remaining() > 60 and n_div <= 320:
        inits = generate_structured_inits(n_div, rng)
        # Also try the parent interpolated
        if h_parent is not None and len(h_parent) != n_div:
            inits.append(make_feasible(interpolate_to_n(h_parent, n_div), n_div))
        # Random inits
        for _ in range(2):
            inits.append(make_feasible(rng.rand(n_div) * 1.5, n_div))

        for h0 in inits[:6]:
            if remaining() < 20:
                break
            h_opt, c5_val = slsqp_max(n_div, h0, max_iter=800, eps_active=1e-5)
            if h_opt is not None and c5_val < best_c5 - 1e-10:
                best_c5 = c5_val
                best_h = h_opt.copy()
                best_n = n_div
                # Continue refining the new best
                if remaining() > 30:
                    h_opt2, c5_val2 = slsqp_max(n_div, best_h, max_iter=1500, eps_active=5e-7)
                    if h_opt2 is not None and c5_val2 < best_c5 - 1e-12:
                        best_c5 = c5_val2
                        best_h = h_opt2.copy()

    # Phase 4: Gradient-based local search at best resolution
    if remaining() > 30:
        n_trials = min(2000, int((remaining() - 10) * 0.4 / 0.008))
        best_h, best_c5 = gradient_local_search(best_h, best_n, n_trials, lr=0.003, rng=rng)

    # Phase 5: Perturbation local search
    if remaining() > 20:
        n_trials = min(3000, int((remaining() - 10) * 0.5 / 0.004))
        best_h, best_c5 = local_perturb(best_h, best_n, n_trials, 0.02, rng)

    # Phase 6: Final SLSQP polish
    if remaining() > 25:
        h_opt, c5_val = slsqp_max(best_n, best_h, max_iter=5000, ftol=1e-17, eps_active=1e-7)
        if h_opt is not None and c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h_opt.copy()

    # Phase 7: One more smooth anneal pass
    if remaining() > 20:
        h_opt2, c5_val2 = smooth_anneal(best_n, best_h, betas=[20, 60, 200], max_iter_each=500)
        if c5_val2 < best_c5 - 1e-13:
            best_c5 = c5_val2
            best_h = h_opt2.copy()

    # Phase 8: Final tiny perturbations
    if remaining() > 15:
        n_trials = min(1000, int((remaining() - 5) * 0.6 / 0.003))
        best_h, best_c5 = local_perturb(best_h, best_n, n_trials, 0.008, rng)

    # Final feasibility check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
