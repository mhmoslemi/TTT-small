def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def project_sum(h, target):

            break

def smooth_max_value_grad(h, beta):
    n = len(h)
    hc = np.clip(h, 0.0, 1.0)
    one_minus = 1.0 - hc
    
    # Compute c_m = sum_{i=0}^{n-1-m} hc[i]*(1-hc[i+m]) for m=0..n-1
    c = np.zeros(n)
    for m in range(n):
        c[m] = np.dot(hc[:n-m], one_minus[m:])
    
    # Softmax weights
    c_max = np.max(c)
    exp_c = np.exp(beta * (c - c_max))
    exp_sum = np.sum(exp_c)
    p = exp_c / exp_sum
    
    val = c_max + np.log(exp_sum) / beta
    
    # Gradient: df/dh[j] = sum_{m=0}^{n-1-j} p[m]*(1-hc[j+m]) - sum_{m=0}^{j} p[m]*hc[j-m]
    grad = np.zeros(n)
    for j in range(n):
        # First term: sum_{m=0}^{n-1-j} p[m]*(1-hc[j+m])
        s1 = np.dot(p[:n-j], one_minus[j:])

        else:
            s2 = p[0] * hc[0]
        grad[j] = s1 - s2
    
    return val, grad
