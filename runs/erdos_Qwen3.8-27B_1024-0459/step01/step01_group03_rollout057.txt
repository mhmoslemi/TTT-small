def binary_sa_search(h, target, deadline, max_iter=1000000, seed=123):
    """Simulated annealing on near-binary values."""
    n = h.size
    if n < 4 or n % 2 != 0:
        return project_density(h, target), evaluate_erdos(project_density(h, target))
    rng_local = np.random.default_rng(seed)
    # Start: round to nearest binary
    best = (h > 0.5).astype(float)
    best = project_density(best, target)
    best_c = evaluate_erdos(best)
    # Also try the original
    orig = project_density(h, target)
    orig_c = evaluate_erdos(orig)
    if orig_c < best_c:
        best = orig
        best_c = orig_c
    T = 0.002
    T_min = 1e-7
    cool_rate = (T_min / T) ** (1.0 / max(1, max_iter))
    cur = best.copy()
    cur_c = best_c
    ones = np.where(cur > 0.5)[0]
    zeros = np.where(cur <= 0.5)[0]
    it = 0
    while it < max_iter and time.perf_counter() < deadline:
        it += 1
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Pick random 1 and random 0
        i = int(ones[rng_local.integers(len(ones))])
        j = int(zeros[rng_local.integers(len(zeros))])
        # Swap
        cur[i] = 0.0
        cur[j] = 1.0
        cur_c_new = evaluate_erdos(cur)
        delta = cur_c_new - cur_c
        if delta < 0 or (T > T_min and rng_local.random() < np.exp(-delta / T)):
            cur_c = cur_c_new
            ones = np.where(cur > 0.5)[0]
            zeros = np.where(cur <= 0.5)[0]
            if cur_c < best_c:
                best = cur.copy()
                best_c = cur_c
        else:
            cur[i] = 1.0
            cur[j] = 0.0
        T *= cool_rate
    # Relax: allow small deviations from binary
    best = project_density(best, target)
    best_c = evaluate_erdos(best)
    return best, best_c
