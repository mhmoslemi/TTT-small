import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_direct(h, n_points):
    dx = 2.0 / n_points
    n = n_points
    H = np.fft.fft(h)
    autocorr = np.fft.ifft(H * np.conj(H)).real
    I_k = dx * (n / 2.0 - autocorr)
    return np.max(I_k)

def compute_I_k(h, dx):
    n = len(h)
    H = np.fft.fft(h)
    autocorr = np.fft.ifft(H * np.conj(H)).real
    return dx * (n / 2.0 - autocorr)

def smooth_max_and_grad(h, n_points, tau):
    n = n_points
    dx = 2.0 / n
    I_k = compute_I_k(h, dx)
    
    # Smooth max
    max_I = np.max(I_k)
    shifted = (I_k - max_I) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    probs = exp_shifted / sum_exp
    obj = max_I + tau * np.log(sum_exp)
    
    # Gradient
    H = np.fft.fft(h)
    p_fft = np.fft.fft(probs)
    p_flip_arr = probs[np.zeros(1, dtype=int).tolist() + [(-k) % n for k in range(1, n)]]
    # Actually p_flip[k] = probs[(-k) % n]
    p_flip_arr = np.array([probs[(-k) % n] for k in range(n)])
    
    B = np.fft.ifft(p_fft * H).real  # B[j] = sum_k probs[k] * h[(j+k)%n]
    C = np.fft.ifft(np.fft.fft(p_flip_arr) * H).real  # C[j] = sum_k probs[(-k)%n] * h[(j+k)%n]
    
    grad = -dx * (B + C)
    return obj, grad

def project_box_hyperplane(y, target):
    """Project y onto {x: 0<=x[i]<=1, sum(x)=target}."""
    n = len(y)
    # Binary search for lambda
    lo, hi = 0.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) < 1e-15:
            break
    lam = (lo + hi) / 2.0
    x = np.clip(y - lam, 0.0, 1.0)
    # Correct tiny numerical errors
    diff = target - np.sum(x)
    if abs(diff) > 1e-15:
        # Distribute to free variables
        free = (x > 1e-10) & (x < 1.0 - 1e-10)
        if np.any(free):
            x[free] += diff / np.sum(free)
            x = np.clip(x, 0.0, 1.0)
    return x

    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5_direct(h, n)
    
    m = np.zeros(n)  # momentum
    v = np.zeros(n)  # velocity
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    tau = tau_init
    lr = 0.05
    
    while budget_param() > 10:
        t += 1
        obj, grad = smooth_max_and_grad(h, n, tau)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_box_hyperplane(h_new, target)
        
        val = compute_c5_direct(h_new, n)

            best_h = h_new.copy()
        
        h = h_new
        
        # Decay temperature

            tau = max(tau_min, tau * 0.95)
            lr = max(0.001, lr * 0.97)
    
    return best_h, best_val

def slsqp_smooth(h0, n_points, tau=0.05, maxiter=200):
    """SLSQP with smooth max objective."""
    n = n_points
    target = n / 2.0
    
    def objective(x):
        val, _ = smooth_max_and_grad(x, n, tau)
        return val
    
    def gradient(x):
        _, g = smooth_max_and_grad(x, n, tau)
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    res = minimize(objective, h0.copy(), jac=gradient, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return res.x, compute_c5_direct(res.x, n)

    _SCP_TAU = tau
    n = n_points
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [('eq', _scp_cons_fun, _scp_cons_jac)]
    try:
        res = minimize(_scp_obj, h0.copy(), jac=_scp_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        h_res = project_onto_feasible(res.x, target)
        val = compute_c5_direct(h_res, n)
        return h_res, val
    except Exception:
        return project_onto_feasible(h0, target), compute_c5_direct(h0, n)


def generate_init(n, rng, strategy, best_h):
    target = n / 2.0
    if strategy == 0:
        h = rng.uniform(0, 1, n)
    elif strategy == 1:
        h = (rng.binomial(1, 0.5, n) + rng.normal(0, 0.2, n) * 0.5).astype(float)
    elif strategy == 2:
        period = int(rng.choice([3, 4, 5, 6, 7, 8, 9]))
        n_ones = max(1, period // 2)
        pattern = np.zeros(period)
        pattern[:n_ones] = 1.0
        h = np.tile(pattern, n // period + 1)[:n] + rng.normal(0, 0.1, n)
    elif strategy == 3:
        freq = int(rng.randint(2, 7))
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.35 * np.sin(2.0 * np.pi * freq * np.linspace(0, 1, n) + phase)
    elif strategy == 4:
        h = best_h + rng.normal(0, 0.25, n)
    elif strategy == 5:
        h = np.abs(rng.normal(0, 0.4, n))
    elif strategy == 6:
        n_blocks = int(rng.randint(3, 12))
        block_size = n // n_blocks
        h = np.zeros(n)
        for b in range(n_blocks):
            if rng.random() < 0.5:
                h[b * block_size:(b + 1) * block_size] = 1.0
        h += rng.normal(0, 0.1, n)
    else:
        h = rng.exponential(0.6, n)
    return project_onto_feasible(h, target)


def interpolate_h(h_coarse, n_coarse, n_fine):
    x_c = np.linspace(0.0, 2.0, n_coarse, endpoint=False)
    x_f = np.linspace(0.0, 2.0, n_fine, endpoint=False)
    h_fine = np.interp(x_f, x_c, h_coarse)
    return project_onto_feasible(h_fine, n_fine / 2.0)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    # Get initial from parent
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 100
        h0 = np.full(n0, 0.5)

    h0 = project_onto_feasible(h0, n0 / 2.0)
    best_h = h0.copy()
    best_val = compute_c5_direct(best_h, n0)
    best_n = n0

    # Phase 1: Adam optimization at parent resolution
    if budget_s - (time.time() - t0) > 120:
        h_try, v_try = adam_projected(h0, n0, t0, budget_s - (budget_s - (time.time() - t0)) + 0, 0.08, 0.005, 0.03, 3000)
        # Simplify: use remaining time
        remaining = budget_s - (time.time() - t0)
        if remaining > 120:
            h_try, v_try = adam_projected(h0, n0, t0, budget_s - 100, 0.08, 0.005, 0.03, 5000)
            if v_try < best_val:
                best_h, best_val = h_try, v_try

    # Phase 2: SLSQP polish with smooth max
    if budget_s - (time.time() - t0) > 100:
        try:
            h_try, v_try = slsqp_smooth(best_h, best_n, 0.03, 800)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
        except Exception:
            pass

    # Phase 3: More Adam with tighter tau
    if budget_s - (time.time() - t0) > 80:
        h_try, v_try = adam_projected(best_h, best_n, t0, budget_s - (budget_s - (time.time() - t0)) + 0, 0.02, 0.002, 0.02, 5000)
        remaining = budget_s - (time.time() - t0)
        if remaining > 80:
            h_try, v_try = adam_projected(best_h, best_n, t0, budget_s - 80, 0.02, 0.002, 0.02, 5000)
            if v_try < best_val:
                best_h, best_val = h_try, v_try

    # Phase 4: Random restarts at current resolution
    n_restarts = 0
    max_restarts = 40
    while budget_s - (time.time() - t0) > 250 and n_restarts < max_restarts:
        strategy = n_restarts % 7
        h_rand = generate_init(best_n, rng, strategy, best_h)
        h_try, v_try = adam_projected(h_rand, best_n, t0, budget_s - (budget_s - (time.time() - t0)) + 0, 0.06, 0.01, 0.025, 1500)
        remaining = budget_s - (time.time() - t0)
        if remaining > 250:
            h_try, v_try = adam_projected(h_rand, best_n, t0, budget_s - 250, 0.06, 0.01, 0.025, 2000)
        if v_try < best_val:
            best_h, best_val = h_try, v_try
        n_restarts += 1

    # Phase 5: SLSQP polish again
    if budget_s - (time.time() - t0) > 60:
        try:
            h_try, v_try = slsqp_smooth(best_h, best_n, 0.015, 1000)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
        except Exception:
            pass

    # Phase 6: Higher resolution attempts
    for n_fine in [100, 140, 200, 280]:
        if budget_s - (time.time() - t0) < 200:
            break
        if n_fine <= best_n and compute_c5_direct(best_h, best_n) <= best_val:
            # Skip if same or lower res
            if n_fine <= best_n:
                continue
        h_fine = interpolate_h(best_h, best_n, n_fine)
        # Adam at higher res
        h_try, v_try = adam_projected(h_fine, n_fine, t0, budget_s - (budget_s - (time.time() - t0)) + 0, 0.05, 0.008, 0.02, 3000)
        remaining = budget_s - (time.time() - t0)
        if remaining > 180:
            h_try, v_try = adam_projected(h_fine, n_fine, t0, budget_s - 180, 0.05, 0.008, 0.02, 4000)
        if v_try < best_val:
            best_h, best_val, best_n = h_try, v_try, n_fine
        # SLSQP at higher res
        if budget_s - (time.time() - t0) > 120 and v_try < best_val + 0.01:
            try:
                h_try2, v_try2 = slsqp_smooth(h_try, n_fine, 0.02, 500)
                if v_try2 < best_val:
                    best_h, best_val, best_n = h_try2, v_try2, n_fine
            except Exception:
                pass

    # Phase 7: Final Adam polish
    if budget_s - (time.time() - t0) > 30:
        h_try, v_try = adam_projected(best_h, best_n, t0, budget_s - (budget_s - (time.time() - t0)) + 0, 0.01, 0.001, 0.015, 5000)
        remaining = budget_s - (time.time() - t0)
        if remaining > 30:
            h_try, v_try = adam_projected(best_h, best_n, t0, budget_s - 20, 0.01, 0.001, 0.015, 5000)
            if v_try < best_val:
                best_h, best_val = h_try, v_try

    # Phase 8: Final SLSQP
    if budget_s - (time.time() - t0) > 20:
        try:
            h_try, v_try = slsqp_smooth(best_h, best_n, 0.005, 500)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
        except Exception:
            pass

    # Final validation and cleanup
    best_h = project_onto_feasible(best_h, best_n / 2.0)
    best_val = compute_c5_direct(best_h, best_n)
    # Verify constraints
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10)
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-6

    return (best_h, best_val, best_n)
