def smooth_jacobian(x):
    n = _SM_N
    if n <= 1:
        return np.zeros(max(1,n), dtype=float)
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return np.zeros(n, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if corr.size == 0:
        return np.zeros(n, dtype=float)
    if not np.all(np.isfinite(corr)):
        return np.zeros(n, dtype=float)
    m = float(np.max(corr))
    alpha = _SM_ALPHA
    if alpha <= 0:
        alpha = 1.0
    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not math.isfinite(z):
        return np.zeros(n, dtype=float)
    w = e / z
    thr = 1e-10
    idx = np.where(w > thr)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(w))], dtype=int)
    g = np.zeros(n, dtype=float)
    center = n - 1
    for s in idx:
        s = int(s)
        p = float(w[s])
        lag = s - center
        if lag == 0:
            g += p * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += p
            g[lag:] -= p * h[:n - lag]
            g[:n - lag] -= p * h[lag:]
        else:
            r = -lag
            g[:n - r] += p
            g[:n - r] -= p * h[r:]
            g[r:] -= p * h[:n - r]
    return (2.0 / float(n)) * g
```

_CORR_REV = 2


def get_corr_rev():
    global _CORR_REV
    if _CORR_REV != 2:
        return _CORR_REV
    h = np.array([0.0, 0.0, 1.0, 1.0], dtype=float)
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = 3
    if abs(float(corr[center + 1]) - 1.0) <= 1e-12:
        _CORR_REV = 0
    else:
        _CORR_REV = 1
    return _CORR_REV


def correlate_oriented(h):
    h = np.asarray(h, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if get_corr_rev() == 1:
        corr = corr[::-1].copy()
    return corr


def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    corr = correlate_oriented(h)
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * (2.0 / float(n)))


def compute_q(h):
    h = np.asarray(h, dtype=float)
    q = correlate_oriented(h)
    if q.size == 0:
        return q, float("inf")
    return q, float(np.max(q))


def project_density(x, target):
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x

    s = float(np.sum(x))
    if abs(s - target) <= 1e-13 * max(1, n):
        return x

    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid

    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)

    res = target - float(np.sum(h))
    if abs(res) > 1e-15:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            if res > 0.0:
                idx = int(free[np.argmax(1.0 - h[free])])
            else:
                idx = int(free[np.argmax(h[free])])
            h[idx] += res

    res = target - float(np.sum(h))
    if abs(res) > 1e-15:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size > 0:
            h[free] += res / float(free.size)
            h = np.clip(h, 0.0, 1.0)

    res = target - float(np.sum(h))
    if abs(res) > 1e-15:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size > 0:
            h[int(free[0])] += res

    return np.clip(h, 0.0, 1.0)


def is_feasible(h, target, tol=1e-6):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    if abs(float(np.sum(h)) - float(target)) > max(1e-8, tol * max(1, h.size)):
        return False
    return True


def upsample_h(parent, n):

    n0 = p.size
    n = int(n)

        return p.copy()

    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)


def make_interval_sample(n, m, rng):
    n = int(n)
    m = int(m)
    if m <= 0:
        m = 1

    raw = rng.random(2 * m).astype(float) + 1e-3
    lengths = raw[:m]
    gaps = raw[m:]
    lengths *= 1.0 / float(np.sum(lengths))
    gaps *= 1.0 / float(np.sum(gaps))

    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.zeros(n, dtype=float)

    start = 0.0

        end = start + float(lengths[i])
        mask = (x >= start) & (x < end)
        h[mask] = 1.0
        start = end

    return project_density(h, 0.5 * n)

    elif mode == 3:

def set_smooth_context(n, alpha, target):
    global _SM_N, _SM_ALPHA, _SM_TARGET
    _SM_N = int(n)
    _SM_ALPHA = float(alpha)
    _SM_TARGET = float(target)


def sum_constraint(x):
    return float(np.sum(np.asarray(x, dtype=float)) - _SM_TARGET)


def sum_constraint_jac(x):
    x = np.asarray(x, dtype=float)
    return np.ones_like(x, dtype=float)


def smooth_objective(x):
    n = _SM_N
    if n <= 1:
        return 0.0

    h = np.asarray(x, dtype=float)
    if h.size != n or not np.all(np.isfinite(h)):
        return 1e100

    corr = correlate_oriented(h)
    if corr.size == 0 or not np.all(np.isfinite(corr)):
        return 1e100

    m = float(np.max(corr))
    alpha = _SM_ALPHA if _SM_ALPHA > 1e-12 else 1.0

    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not math.isfinite(z):
        return 1e100

    return float((2.0 / float(n)) * (m + math.log(z) / alpha))


def smooth_jacobian(x):
    n = _SM_N
    if n <= 1:
        return np.zeros(max(1, n), dtype=float)

    h = np.asarray(x, dtype=float)
    if h.size != n or not np.all(np.isfinite(h)):
        return np.zeros(n, dtype=float)

    corr = correlate_oriented(h)
    if corr.size == 0 or not np.all(np.isfinite(corr)):
        return np.zeros(n, dtype=float)

    m = float(np.max(corr))
    alpha = _SM_ALPHA if _SM_ALPHA > 1e-12 else 1.0

    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not math.isfinite(z):
        return np.zeros(n, dtype=float)

    w = e / z
    thr = 1e-10
    idx = np.where(w > thr)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(w))], dtype=int)

    g = np.zeros(n, dtype=float)
    center = n - 1

    for s in idx:
        s = int(s)
        p = float(w[s])
        lag = s - center

        if lag == 0:
            g += p * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += p
            g[lag:] -= p * h[:n - lag]
            g[:n - lag] -= p * h[lag:]
        else:
            r = -lag
            g[:n - r] += p
            g[:n - r] -= p * h[r:]
            g[r:] -= p * h[:n - r]

    return (2.0 / float(n)) * g


def run_smooth_minimize(h0, n, deadline, alphas, maxiter):
    n = int(n)
    target = 0.5 * n
    h = project_density(np.asarray(h0, dtype=float), target)
    best_h = h.copy()
    best_c = evaluate_erdos(h)

    bounds = [(0.0, 1.0)] * n
    constraints = ({"type": "eq", "fun": sum_constraint, "jac": sum_constraint_jac},)

    for alpha in alphas:
        if time.perf_counter() >= deadline:
            break

        set_smooth_context(n, alpha, target)
        try:
            res = minimize(
                smooth_objective,
                h,
                method="SLSQP",
                jac=smooth_jacobian,
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": int(maxiter), "ftol": 1e-9, "disp": False},
            )
            xg = getattr(res, "x", None)
            if xg is not None:
                xg = np.asarray(xg, dtype=float)
                if xg.shape == (n,) and np.all(np.isfinite(xg)):
                    cand = project_density(xg, target)
                    c = evaluate_erdos(cand)
                    if c < best_c:
                        best_h = cand.copy()
                        best_c = c
                        h = cand
                    else:
                        h = best_h
        except Exception:
            h = best_h

    return best_h, best_c


def active_gradient(h, active_indices):
    n = h.size
    g = np.zeros(n, dtype=float)
    if len(active_indices) == 0:
        return g

    center = n - 1
    for s in active_indices:
        s = int(s)
        lag = s - center
        if lag == 0:
            g += 1.0 - 2.0 * h
        elif lag > 0:
            g[lag:] += 1.0
            g[lag:] -= h[:n - lag]
            g[:n - lag] -= h[lag:]
        else:
            r = -lag
            g[:n - r] += 1.0
            g[:n - r] -= h[r:]
            g[r:] -= h[:n - r]

    return g / float(len(active_indices))


def polish_gradient(h0, deadline, active_k=8, max_iter=100):
    h = np.asarray(h0, dtype=float)
    n = int(h.size)
    target = 0.5 * n
    h = project_density(h, target)

    q, maxq = compute_q(h)
    dx = 2.0 / float(n)
    c = maxq * dx
    best_h = h.copy()
    best_c = c

    if n < 2:
        return best_h, best_c

    k = max(1, min(int(active_k), 2 * n - 1))
    stall = 0

    for _ in range(int(max_iter)):
        if time.perf_counter() >= deadline:
            break

        if 2 * n - 1 > k:
            top = np.argpartition(q, -k)[-k:]
        else:
            top = np.arange(2 * n - 1)

        g = active_gradient(h, top)
        d = -g
        d -= float(np.mean(d))
        norm = float(np.linalg.norm(d))
        if norm < 1e-13:
            break
        d /= norm

        improved = False
        for step in (0.01, 0.03, 0.08, 0.18, 0.35):
            cand = project_density(h + step * d, target)
            qc, mac = compute_q(cand)
            cn = mac * dx
            if cn < best_c - 1e-12:
                h = cand
                q = qc
                c = cn
                best_h = h.copy()
                best_c = c
                improved = True
                break

        if not improved:
            stall += 1
            if stall > 12:
                break

    final_h = project_density(best_h, target)
    qf, macf = compute_q(final_h)
    return final_h, macf * dx


def coord_slope(h, inc, dec):
    n = h.size
    center = n - 1
    m = 2 * n - 1
    ar = np.arange(n)

    s = np.zeros(m, dtype=float)

    s[center + inc - ar] += 1.0 - h
    s[center - inc + ar] -= h

    s[center + dec - ar] -= 1.0 - h
    s[center - dec + ar] += h

    return s


def pair_move_max(q, slope, c1, c2, p, dx):
    if p <= 0.0:
        return float(np.max(q)) * dx

    tmp = q + p * slope
    val = p * p
    center = q.size // 2

    tmp[center] -= 2.0 * val
    if c1 != c2:
        tmp[c1] += val
        tmp[c2] += val
    else:
        tmp[c1] += 2.0 * val

    return float(np.max(tmp)) * dx


def refine_pair_p(q, slope, c1, c2, dx, b, init_p):
    if b <= 1e-15:
        return 0.0, float(np.max(q)) * dx

    safe_b = b * (1.0 - 1e-12)
    best_p = 0.0
    best_f = pair_move_max(q, slope, c1, c2, 0.0, dx)

    for i in range(1, 25):
        p = safe_b * float(i) / 24.0
        f = pair_move_max(q, slope, c1, c2, p, dx)
        if f < best_f:
            best_f = f
            best_p = p

    if init_p > 0.0:
        p = float(init_p)
        if p > safe_b:
            p = safe_b
        f = pair_move_max(q, slope, c1, c2, p, dx)
        if f < best_f:
            best_f = f
            best_p = p

        for fr in (0.75, 0.85, 0.95, 1.05, 1.15, 1.30):
            p2 = float(init_p) * fr
            if p2 <= 1e-15:
                continue
            if p2 > safe_b:
                p2 = safe_b
            f2 = pair_move_max(q, slope, c1, c2, p2, dx)
            if f2 < best_f:
                best_f = f2
                best_p = p2

    return float(best_p), float(best_f)


def local_pair_search_q(h0, deadline, rng, active_k=10, max_iter=100000, max_stall=100):
    h = np.asarray(h0, dtype=float)
    n = int(h.size)
    target = 0.5 * n
    h = project_density(h, target)

    q, maxq = compute_q(h)
    dx = 2.0 / float(n)
    c = maxq * dx
    best_h = h.copy()
    best_c = c
    best_q = q.copy()

    if n < 2:
        return best_h, best_c

    center = n - 1
    m = 2 * n - 1
    k = max(1, min(int(active_k), m))
    stall = 0
    recompute = 0
    it = 0
    max_cand = 28 if n <= 180 else 20

    while time.perf_counter() < deadline and it < int(max_iter):
        it += 1

        if recompute >= 100 or not math.isfinite(c):
            q, maxq = compute_q(h)
            c = maxq * dx
            recompute = 0
            if c > best_c + 1e-8:
                h = best_h.copy()
                q = best_q.copy()
                c = best_c

        if m > k:
            top = np.argpartition(q, -k)[-k:]
        else:
            top = np.arange(m)

        g = active_gradient(h, top)
        candidates = []

        lows = np.argsort(g)[:6]
        highs = np.argsort(g)[-6:]
        for ii in lows:
            for jj in highs:
                ii = int(ii)
                jj = int(jj)
                if ii != jj:
                    candidates.append((jj, ii))
                    if len(candidates) < 64:
                        candidates.append((ii, jj))

        high_h = np.argsort(-h)[:4]
        low_h = np.argsort(h)[:4]
        for a in high_h:
            for b in low_h:
                a = int(a)
                b = int(b)
                if a != b:
                    candidates.append((a, b))

        top_sorted = top[np.argsort(-q[top])]
        for s in top_sorted[:6]:
            s = int(s)
            lag = s - center

            if lag > 0:
                r = lag
                w = h[r:] * (1.0 - h[:n - r])
                if w.size >= 2:
                    order = np.argpartition(w, -2)[-2:]
                elif w.size == 1:
                    order = np.array([0], dtype=int)
                else:
                    order = np.array([], dtype=int)

                for idx in order:
                    idx = int(idx)
                    candidates.append((idx + r, idx))

            elif lag < 0:
                r = -lag
                w = h[:n - r] * (1.0 - h[r:])
                if w.size >= 2:
                    order = np.argpartition(w, -2)[-2:]
                elif w.size == 1:
                    order = np.array([0], dtype=int)
                else:
                    order = np.array([], dtype=int)

                for idx in order:
                    idx = int(idx)
                    candidates.append((idx, idx + r))

        for _ in range(6):
            a = int(rng.integers(n))
            b = int(rng.integers(n))
            if a != b:
                candidates.append((a, b))
                candidates.append((b, a))

        seen = set()
        cand_list = []
        for a, b in candidates:
            key = int(a) * n + int(b)
            if key in seen:
                continue
            seen.add(key)
            cand_list.append((int(a), int(b)))
            if len(cand_list) >= max_cand:
                break

        best_pair = None
        best_cand_c = c
        best_sample_p = 0.0
        best_slope = None
        best_c1 = 0
        best_c2 = 0

        for dec, inc in cand_list:
            cap = min(float(h[dec]), float(1.0 - h[inc]))
            if cap <= 1e-13:
                continue

            slope = coord_slope(h, inc, dec)
            c1 = center + dec - inc
            c2 = center + inc - dec
            safe_cap = cap * (1.0 - 1e-12)

            local_best_p = 0.0
            local_best_c = c

            for fr in (0.05, 0.2, 0.5, 0.8, 1.0):
                p = fr * safe_cap
                if p <= 1e-14:
                    continue
                cn = pair_move_max(q, slope, c1, c2, p, dx)
                if cn < local_best_c:
                    local_best_c = cn
                    local_best_p = p

            if local_best_c < best_cand_c - 1e-13:
                best_cand_c = local_best_c
                best_pair = (dec, inc)
                best_sample_p = local_best_p
                best_slope = slope
                best_c1 = c1
                best_c2 = c2

        if best_pair is not None:
            dec, inc = best_pair
            cap = min(float(h[dec]), float(1.0 - h[inc]))
            p_opt, gc = refine_pair_p(q, best_slope, best_c1, best_c2, dx, cap, best_sample_p)

            if gc < best_cand_c:
                best_cand_c = gc

            if p_opt <= 1e-14:
                p_opt = best_sample_p

            if p_opt > 1e-14:
                h[dec] -= p_opt
                h[inc] += p_opt
                q += p_opt * best_slope

                val = p_opt * p_opt
                q[center] -= 2.0 * val
                if best_c1 != best_c2:
                    q[best_c1] += val
                    q[best_c2] += val
                else:
                    q[best_c1] += 2.0 * val

                if h[dec] < 0.0:
                    h[dec] = 0.0
                    q, maxq = compute_q(h)
                    c = maxq * dx
                elif h[inc] > 1.0:
                    h[inc] = 1.0
                    q, maxq = compute_q(h)
                    c = maxq * dx
                else:
                    c = float(np.max(q)) * dx

                recompute = 0
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                    best_q = q.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if stall > int(max_stall):
            break

    final_h = project_density(best_h, target)
    qf, macf = compute_q(final_h)
    return final_h, macf * dx


def improve_candidate(h0, deadline, rng, do_smooth=False, do_polish=True):
    h = np.asarray(h0, dtype=float).copy()
    n = int(h.size)
    if n < 1:
        h = project_density(np.full(90, 0.5), 45.0)
        n = 90

    target = 0.5 * n
    h = project_density(h, target)
    c = evaluate_erdos(h)

    now = time.perf_counter()
    if deadline - now <= 5.0 or n < 2:
        return h, c

    if do_smooth and n <= 200 and deadline - now > 35.0:
        smooth_dur = min(0.22 * (deadline - now), 120.0)
        if smooth_dur > 10.0:
            hd = now + smooth_dur
            alphas = (10.0, 30.0)
            maxit = 35 if n <= 100 else 16
            h2, c2 = run_smooth_minimize(h, n, hd, alphas, maxit)
            if c2 < c:
                h, c = h2, c2

    if do_polish and deadline - time.perf_counter() > 8.0:
        polish_dead = min(deadline - 8.0, time.perf_counter() + max(5.0, 0.12 * (deadline - time.perf_counter())))
        if polish_dead > time.perf_counter():
            h3, c3 = polish_gradient(h, polish_dead, active_k=min(12, max(6, n // 12)), max_iter=60)
            if c3 < c:
                h, c = h3, c3

    if deadline - time.perf_counter() > 5.0:
        ak = min(16, max(8, n // 10))
        h4, c4 = local_pair_search_q(h, deadline, rng, active_k=ak, max_iter=200000, max_stall=120)
        if c4 < c:
            h, c = h4, c4

    return h, c


def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    if budget < 20.0:
        budget = 20.0
    deadline = start + max(1.0, budget - 15.0)

    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1.0).astype(float, copy=True)
    except Exception:
        parent = None

    if parent is not None:
        nparent = int(parent.size)
        pr = project_density(parent, 0.5 * nparent)
        cp = evaluate_erdos(pr)
        best_h = pr.copy()
        best_c = cp
        best_n = nparent
    else:
        n0 = 90
        best_h = make_candidate(n0, rng, 0)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    target = 0.38085

    if deadline - time.perf_counter() > 40.0:
        dur = min(0.32 * budget, deadline - time.perf_counter() - 30.0)
        if dur > 25.0:
            h1, c1 = improve_candidate(best_h, time.perf_counter() + dur, rng, True, True)
            if c1 < best_c:
                best_h = h1
                best_c = c1
                best_n = int(h1.size)

    if best_c > target and deadline - time.perf_counter() > 150.0:
        n2 = int(best_n) * 2
        if 2 <= n2 <= 720 and n2 != int(best_n):
            rem = deadline - time.perf_counter()
            dur = min(0.30 * rem, 300.0, rem - 30.0)
            if dur > 30.0:
                hstart = upsample_h(best_h, n2)
                h2, c2 = improve_candidate(hstart, time.perf_counter() + dur, rng, n2 <= 220, True)
                if c2 < best_c:
                    best_h = h2
                    best_c = c2
                    best_n = int(h2.size)

    if best_c > target and deadline - time.perf_counter() > 100.0:
        rem = deadline - time.perf_counter()
        dur = min(0.22 * rem, 180.0, rem - 30.0)
        if dur > 25.0:
            noise = 0.02 if int(best_n) <= 200 else 0.01
            hstart = project_density(best_h + noise * rng.normal(size=int(best_n)), 0.5 * int(best_n))
            h3, c3 = improve_candidate(hstart, time.perf_counter() + dur, rng, False, True)
            if c3 < best_c:
                best_h = h3
                best_c = c3
                best_n = int(h3.size)

    if best_c > target and deadline - time.perf_counter() > 80.0:
        rem = deadline - time.perf_counter()
        dur = min(0.18 * rem, 120.0, rem - 30.0)
        if dur > 20.0:
            hstart = make_candidate(int(best_n), rng, 3)
            h4, c4 = improve_candidate(hstart, time.perf_counter() + dur, rng, False, True)
            if c4 < best_c:
                best_h = h4
                best_c = c4
                best_n = int(h4.size)

    if best_c > target and deadline - time.perf_counter() > 150.0:
        n4 = int(best_n) * 4
        if 2 <= n4 <= 720 and n4 > int(best_n):
            rem = deadline - time.perf_counter()
            dur = min(0.30 * rem, 300.0, rem - 30.0)
            if dur > 30.0:
                hstart = upsample_h(best_h, n4)
                h5, c5 = improve_candidate(hstart, time.perf_counter() + dur, rng, False, True)
                if c5 < best_c:
                    best_h = h5
                    best_c = c5
                    best_n = int(h5.size)

    if deadline - time.perf_counter() > 30.0:
        final_dead = deadline - 5.0
        ak = min(18, max(8, int(best_n) // 8))
        hfin, cfin = local_pair_search_q(best_h, final_dead, rng, active_k=ak, max_iter=500000, max_stall=250)
        if cfin < best_c:
            best_h = hfin
            best_c = cfin
            best_n = int(hfin.size)

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
