- The overlap at shift k is the number of positions i where h[i]=1 and h[i+k]=0 (for the linear correlation)
- We want to minimize the maximum overlap over all k
- Local search: pick a position p where h[p]=0 and a position q where h[q]=1, swap them (set h[p]=1, h[q]=0)
- Check if the max overlap decreased

import numpy as np

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def compute_conv_dx(h, n_points):
    dx = 2.0 / n_points
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min(diff / cnt, (1.0 - h[mask]).min() if cnt > 0 else 0)
            # Actually, just distribute evenly
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    # Final fix
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        # Distribute the remaining difference
        diff = target - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(conv_dx, T):
    m = np.max(conv_dx)
    return m + T * np.log(np.sum(np.exp((conv_dx - m) / T)))

def smooth_max_grad(h, n, T):
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode='full')
    C = R * dx
    C_max = np.max(C)
    exp_C = np.exp((C - C_max) / T)
    p = exp_C / np.sum(exp_C)
    
    grad = np.zeros(n)
    
    # k >= 0: p[n-1+k] for k = 0, ..., n-1
    for k in range(n):
        w = p[n - 1 + k]
        if w < 1e-13:
            continue
        # (1-h[j-k]) for j in [k, n-1]
        grad[k:] += w * (1.0 - h[:n - k])
        # -h[j+k] for j in [0, n-1-k]
        grad[:n - k] += w * (-h[k:])
    
    # k < 0: p[n-1-k'] for k' = 1, ..., n-1
    for kp in range(1, n):
        w = p[n - 1 - kp]
        if w < 1e-13:
            continue
        # (1-h[j+kp]) for j in [0, n-1-kp]
        grad[:n - kp] += w * (1.0 - h[kp:])
        # -h[j-kp] for j in [kp, n-1]
        grad[kp:] += w * (-h[:n - kp])
    
    return dx * grad

def smooth_max_func_and_grad(h, n, T):
    conv_dx = compute_conv_dx(h, n)
    val = smooth_max_val(conv_dx, T)
    grad = smooth_max_grad(h, n, T)
    return val, grad

def slsqp_smooth(h0, n, T, maxiter=300):
    target = n / 2.0
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_max_val(compute_conv_dx(x, n), T)
    
    def jac(x):
        return smooth_max_grad(x, n, T)
    
    res = minimize(obj, h0.copy(), jac=jac, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return res.x, res.fun

def binary_local_search(h, n, n_iters=5000, rng=None):

    return best_h, best_val
