def projected_subgradient(h, n, t_end, K=5, alpha0=0.05):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    t = 0
    alpha = alpha0
    while time.time() < t_end:
        c = compute_correlations(h, n)
        g = topk_subgradient(h, n, K, c)
        h_new = h - alpha * g
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
        t += 1
        alpha = alpha0 / (1 + 0.01 * t)

        g[:n - shift] += (1.0 - h[shift:])
        if shift > 0:
            g[shift:] -= h[:n - shift]
    else:

        g[t:] += (1.0 - h[:n - t])

        for idx in top_idx:
            shift = idx - (n - 1)
            g += grad_single_shift(h, n, shift)
        g /= K
        h_new = h - alpha * g
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5_val(h_new, n)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
    return best_h, best_c5


def random_binary_init(n, rng):
    h = np.zeros(n)
    positions = rng.choice(n, size=n // 2, replace=False)
    h[positions] = 1.0
    return h


def periodic_init(n, period, ones_per_period):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < ones_per_period:
            h[i] = 1.0
    # Adjust to get exactly n/2 ones
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        one_positions = np.where(h == 1.0)[0]
        remove = rng_global.choice(one_positions, size=n_ones - target, replace=False)
        h[remove] = 0.0
    elif n_ones < target:
        zero_positions = np.where(h == 0.0)[0]
        add = rng_global.choice(zero_positions, size=target - n_ones, replace=False)
        h[add] = 1.0
    return h


def subgradient_then_lp(h, n, sg_time=20, lp_time=60):
    h, _ = subgradient_refine(h, n, sg_time)
    h, c5 = lp_refine(h, n, lp_time)
    return h, c5


def run(seed=42, budget_s=1300, **kwargs):
    global rng_global
    t0 = time.time()
    rng_global = np.random.RandomState(seed)

    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float)
        n = len(h_init)
    except Exception:
        n = 95
        h_init = np.full(n, 0.5)

    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5_val(best_h, n)

    # Phase 1: LP refine from parent (300s)
    time_left = budget_s - (time.time() - t0)
    if time_left > 100:
        t1 = time.time()
        best_h, best_c5 = lp_refine(best_h, n, time_limit=min(280, time_left - 100))
        elapsed = time.time() - t1

    # Phase 2: Subgradient + LP from parent with perturbations (250s)
    time_left = budget_s - (time.time() - t0)
    if time_left > 80:
        for attempt in range(8):
            if time.time() - t0 > budget_s - 150:
                break
            scale = 0.05 + 0.03 * attempt
            h_pert = best_h + rng_global.randn(n) * scale
            h_pert = project_feasible(h_pert, n)
            rem = budget_s - (time.time() - t0) - 100
            if rem < 20:
                break
            split = rem / max(rem / 40, 1)
            h_try, c5_try = subgradient_then_lp(h_pert, n, sg_time=15, lp_time=40)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try

    # Phase 3: Diverse initializations (300s)
    time_left = budget_s - (time.time() - t0)
    if time_left > 100:
        inits = []
        # Random binary
        inits.append(random_binary_init(n, rng_global))
        # Periodic patterns
        for period in [5, 7, 11, 13]:
            if n >= period * 3:
                inits.append(periodic_init(n, period, period // 2))
        # Another random
        h_rand = rng_global.rand(n)
        inits.append(project_feasible(h_rand, n))
        # Smeared version of best
        from scipy.ndimage import gaussian_filter1d
        h_smeared = gaussian_filter1d(best_h, sigma=2)
        inits.append(project_feasible(h_smeared, n))

        for h_in in inits:
            if time.time() - t0 > budget_s - 80:
                break
            h_in = project_feasible(h_in, n)
            c5_in = compute_c5_val(h_in, n)
            if c5_in < best_c5 + 0.05:  # Only optimize promising starts
                rem = budget_s - (time.time() - t0) - 60
                if rem < 20:
                    break
                h_try, c5_try = subgradient_then_lp(h_in, n, sg_time=10, lp_time=min(50, rem - 10))
                if c5_try < best_c5:
                    best_h = h_try
                    best_c5 = c5_try

    # Phase 4: Higher resolution (300s)
    time_left = budget_s - (time.time() - t0)
    if time_left > 120:
        for n_alt in [120, 150]:
            if time.time() - t0 > budget_s - 80:
                break
            x_old = np.linspace(0.0, 2.0, n, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_alt, endpoint=False)
            h_alt = np.interp(x_new, x_old, best_h)
            h_alt = project_feasible(h_alt, n_alt)
            c5_alt = compute_c5_val(h_alt, n_alt)
            rem = budget_s - (time.time() - t0) - 60
            if rem > 30:
                h_alt, c5_alt = lp_refine(h_alt, n_alt, time_limit=min(rem - 10, 250))
                if c5_alt < best_c5:
                    best_h = h_alt
                    best_c5 = c5_alt
                    n = n_alt

    # Phase 5: Final fine-tuning (80s)
    time_left = budget_s - (time.time() - t0)
    if time_left > 20:
        best_h, best_c5 = lp_refine(best_h, n, time_limit=min(time_left - 10, 70),
                                     K_init=5, K_max=20, delta_init=0.03, delta_min=0.002)

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5_val(best_h, n)

    # Validate
    try:
        val = evaluate_erdos_solution(best_h, n)
        if val is not None and 0.0 < val < 1.0:
            best_c5 = min(best_c5, float(val))
    except Exception:
        pass

    return (best_h, best_c5, n)
