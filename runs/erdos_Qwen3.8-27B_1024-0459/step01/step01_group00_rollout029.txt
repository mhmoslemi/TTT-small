        new_energy = np.max(new_corr)

            # Update ones/zeros

            if new_energy < best_energy:
                best_energy = new_energy
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_energy

    best_h = h.copy()

    # Initial step size
    alpha = 0.01
    start = time.time()

                g -= h[j - s]

            h_new = np.clip(h_new, 0.0, 1.0)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        
        # Decrease step size
        alpha *= 0.9999
    
    return best_h, best_c5

        # j ranges from s to n-1

    else:
        # j ranges from 0 to n-1+s
        grad[:n+s] -= h[-s:]
    return grad

Actually, a simpler coordinate descent:
For each i (one at a time):
- Fix all h[j] for j≠i
- We need to redistribute the "sum deficit" by changing h[i]
- But we can't change just one coordinate without violating the sum constraint
- So: change h[i] to h[i] + δ, and distribute -δ uniformly over all other coordinates
- This is a 1D problem in δ

The objective as a function of δ:
c_k(δ) = c_k + δ·g_k where g_k = ∂c_k/∂h[i] - (1/(n-1))·∑_j ∂c_k/∂h[j]

We want to minimize max_k c_k(δ) over δ such that h[i]+δ ∈ [0,1] and all h[j] - δ/(n-1) ∈ [0,1].

This is a 1D convex optimization (min of max of linear functions), solvable by ternary search or by finding the upper envelope intersection.

Let me implement this:

def coordinate_descent(h, n, n_passes=100, time_limit=60):
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    start = time.time()
    
    for pfx in range(n_passes):
        if time.time() - start > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - start > time_limit:
                break
            # Compute gradient of each c_k w.r.t. h[i] (and the uniform adjustment)
            corr = np.correlate(h, 1.0-h, mode='full')
            # d(c_k)/d(h[i]) for each k
            # c[m] = sum_j h[j] * (1-h[j+m-n+1]) 
            # d(c[m])/d(h[i]) = (1-h[i+m-n+1]) [if 0<=i+m-n+1<n] - h[i-m+n-1] [if 0<=i-m+n-1<n]
            
            # Instead of computing for all m, let's just do a numerical 1D search
            # Try small changes to h[i]
            hi = h[i]
            # Range of valid δ
            delta_max = min(1.0 - hi, (n/2.0 - sum(h) + hi) / 1.0)  # simplified
            # Actually, if we change h[i] by δ, we need sum to stay n/2
            # So we add -δ/(n-1) to each other element
            # Valid range: 0 <= hi+δ <= 1, and 0 <= h[j]-δ/(n-1) <= 1 for all j≠i
            
            delta_lo = max(-hi, -min(h[j] for j in range(n) if j != i) * (n-1))
            delta_hi = min(1.0-hi, (1.0 - max(h[j] for j in range(n) if j != i)) * (n-1))
            
            if delta_hi <= delta_lo:
                continue
            
            # Binary search for optimal δ
            for _ in range(30):
                d1 = delta_lo + (delta_hi - delta_lo) * 0.33
                d2 = delta_lo + (delta_hi - delta_lo) * 0.67
                h_test1 = h.copy()
                h_test2 = h.copy()
                h_test1[i] += d1
                h_test1[:i] -= d1/(n-1)
                h_test1[i+1:] -= d1/(n-1)
                h_test2[i] += d2
                h_test2[:i] -= d2/(n-1)
                h_test2[i+1:] -= d2/(n-1)
                
                c5_1 = compute_c5(np.clip(h_test1, 0, 1))  # might need feasibility fix
                c5_2 = compute_c5(np.clip(h_test2, 0, 1))
                
                if c5_1 < c5_2:
                    delta_hi = d2
                else:
                    delta_lo = d1
            
            # Apply optimal δ
            delta_opt = (delta_lo + delta_hi) / 2
            h[i] += delta_opt
            mask = np.ones(n, dtype=bool)
            mask[i] = False
            h[mask] -= delta_opt / (n-1)
            h = np.clip(h, 0, 1)
            h = make_feasible(h, n)
            
            c5_new = compute_c5(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                improved = True
        if not improved and pfx > 5:
            break
    
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(10000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgrad(h, n, s):
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1.0 - h[s:n])
        grad[s:] -= h[:n - s]
    else:
        grad[-s:] += (1.0 - h[:n + s])
        grad[:n + s] -= h[-s:]
    return grad

def subgradient_optimize(h, n, n_steps=80000, time_limit=80):
    h = make_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    alpha = 0.5
    decay = 0.99995
    for it in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corr)
        s = k_star - (n - 1)
        grad = compute_subgrad(h, n, s)
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h_new)
        mask = (h_new > 1e-12) & (h_new < 1.0 - 1e-12)
        if np.any(mask):
            h_new[mask] += diff / np.count_nonzero(mask)
            h_new = np.clip(h_new, 0.0, 1.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-16:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        alpha *= decay
    return best_h, best_c5

def binary_sa(h_bin, n, time_limit=60, n_restarts=3):
    best_overall_h = h_bin.copy()
    best_overall_c = np.max(np.correlate(h_bin, 1.0 - h_bin, mode='full'))
    start_time = time.time()
    for restart in range(n_restarts):
        if time.time() - start_time > time_limit:
            break
        if restart == 0:
            h = h_bin.astype(float).copy()
        else:
            h = best_overall_h.astype(float).copy()
            n_flips = n // 10
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            for _ in range(n_flips):
                i = ones[np.random.randint(len(ones))] if len(ones) > 0 else 0
                j = zeros[np.random.randint(len(zeros))] if len(zeros) > 0 else 0
                h[i], h[j] = h[j], h[i]
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        corr = np.correlate(h, 1.0 - h, mode='full')
        curr_c = np.max(corr)
        local_best_h = h.copy()
        local_best_c = curr_c
        start_temp = 2.0
        end_temp = 0.005
        it = 0
        total_iters = max(10000, int((time_limit - (time.time() - start_time)) / 0.001))
        for it in range(total_iters):
            if time.time() - start_time > time_limit:
                break
            if len(ones) == 0 or len(zeros) == 0:
                break
            temp = start_temp * (end_temp / start_temp) ** (it / total_iters)

                ones[i] = j
                zeros[j] = i

            else:
                h[i], h[j] = h[j], h[i]
        if local_best_c < best_overall_c:

def directed_local_search(h, n, time_limit=30):
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    start_time = time.time()
    no_improve = 0
    while time.time() - start_time < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corr)
        s = k_star - (n - 1)
        grad = compute_subgrad(h, n, s)
        # Find coordinate with most negative gradient (most beneficial to decrease)
        # and coordinate with most positive gradient (beneficial to increase)
        # Try adjusting the pair
        i_dec = np.argmin(grad)
        i_inc = np.argmax(grad)
        if i_dec == i_inc:
            break
        # Small step
        step = 0.01
        h[i_dec] -= step
        h[i_inc] += step
        h = np.clip(h, 0.0, 1.0)
        h = make_feasible(h, n)
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            no_improve = 0
        else:
            h[i_dec] += step
            h[i_inc] -= step
            h = np.clip(h, 0.0, 1.0)
            h = make_feasible(h, n)
            no_improve += 1
            if no_improve > 500:
                break
    return h, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def slsqp_optimize(h_init, n, max_iter=3000):
    def obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    def con_fun(x):
        return np.sum(x) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

    # Phase 6: Try n=400 if time permits
    if remaining() > 150 and best_n != 400:
        n3 = 400
        h_interp3 = interpolate_to_n(best_h, n3)
        h_interp3 = make_feasible(h_interp3, n3)
        c_interp3 = compute_c5(h_interp3)
        if c_interp3 < best_c5 + 0.03:
            if remaining() > 100:
                h_bin3 = (h_interp3 >= 0.5).astype(float)
                n_ones3 = int(np.sum(h_bin3))
                target_ones3 = n3 // 2
                if n_ones3 != target_ones3:
                    ones_idx = np.where(h_bin3 == 1)[0]
                    zeros_idx = np.where(h_bin3 == 0)[0]
                    if n_ones3 > target_ones3:
                        to_flip = np.random.choice(ones_idx, n_ones3 - target_ones3, replace=False)
                        h_bin3[to_flip] = 0
                    else:
                        to_flip = np.random.choice(zeros_idx, target_ones3 - n_ones3, replace=False)
                        h_bin3[to_flip] = 1
                h_sa3, c_sa3 = binary_sa(h_bin3, n3, time_limit=40, n_restarts=2)
                c_sa3_full = compute_c5(make_feasible(h_sa3, n3))
                if c_sa3_full < best_c5:
                    best_c5 = c_sa3_full
                    best_h = make_feasible(h_sa3, n3).copy()
                    best_n = n3

            if remaining() > 60:
                h_sg3, c_sg3 = subgradient_optimize(best_h, best_n, n_steps=30000, time_limit=30)
                if c_sg3 < best_c5:
                    best_c5 = c_sg3
                    best_h = h_sg3.copy()

    # Phase 7: Final polish - subgradient with very small steps
    if remaining() > 20:
        h_fin, c_fin = subgradient_optimize(best_h, best_n, n_steps=20000, time_limit=15)
        if c_fin < best_c5:
            best_c5 = c_fin
            best_h = h_fin.copy()

    # Phase 8: Final directed local search
    if remaining() > 15:
        h_d2, c_d2 = directed_local_search(best_h, best_n, time_limit=10)
        if c_d2 < best_c5:
            best_c5 = c_d2
            best_h = h_d2.copy()

    # Final feasibility check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
