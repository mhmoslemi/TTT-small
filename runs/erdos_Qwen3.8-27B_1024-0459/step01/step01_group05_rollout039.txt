import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def project_simplex_box(h, target):
    """Project h onto {sum(h)=target, 0<=h<=1} via bisection on lambda."""
    h = h.astype(float).copy()
    # Bisect on lambda: find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if abs(s - target) < 1e-12:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def project_simplex_box(h, target):
    """Project h onto {sum(h)=target, 0<=h<=1}."""
    h = h.astype(float).copy()
    lo = h.min() - 2.0
    hi = h.max() + 2.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def project_simplex_box(h, target):
    h = h.astype(float).copy()
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_value(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def compute_gradient(h, beta):
    n = len(h)
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    w = w / w_sum

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        i_lo = max(0, n - 1 - k)
        i_hi = min(n - 1, 2 * n - 2 - k)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            jdx = idx + k - n + 1
            grad[idx] += wk * (1.0 - h[jdx])

        if i_lo2 <= i_hi2:
            idx2 = np.arange(i_lo2, i_hi2 + 1)
            jdx2 = idx2 - k + n - 1
            grad[idx2] -= wk * h[jdx2]

    # Final: ensure exact constraint satisfaction
    best_h = project_simplex_box(best_h, best_n / 2.0)
    
    # Verify and fix sum if needed
    s = np.sum(best_h)
    tgt = best_n / 2.0
    if abs(s - tgt) > 1e-10:
        diff = tgt - s
        if diff > 0:
            free = np.where(best_h < 1.0)[0]
            if len(free) > 0:
                best_h[free] += diff / len(free)
        else:
            free = np.where(best_h > 0.0)[0]
            if len(free) > 0:
                best_h[free] += diff / len(free)
        best_h = np.clip(best_h, 0.0, 1.0)
    
    final_c5 = compute_c5(best_h)
    
    return (best_h, final_c5, best_n)
