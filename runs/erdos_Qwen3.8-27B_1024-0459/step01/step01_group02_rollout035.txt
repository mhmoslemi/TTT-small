            ind[ones[perm[:n_ones - target]]] = 0.0
        elif n_ones < target:
            zeros = np.where(ind == 0.0)[0]
            perm = rng.permutation(len(zeros))
            ind[zeros[perm[:target - n_ones]]] = 1.0
        return ind

    population = [make_individual(h_init)]
    for _ in range(pop_size - 1):
        h_rand = rng.rand(n)
        population.append(make_individual(h_rand))

    scores = np.array([compute_c5(p) for p in population])
    best_idx = np.argmin(scores)
    best_h = population[best_idx].copy()
    best_c5 = scores[best_idx]

    for gen in range(generations):
        # Selection: tournament
        new_pop = []
        # Elite: keep top 3
        order = np.argsort(scores)
        for e in range(min(3, pop_size)):
            new_pop.append(population[order[e]].copy())

        while len(new_pop) < pop_size:
            # Tournament selection
            idx1 = rng.randint(pop_size)
            idx2 = rng.randint(pop_size)
            parent1 = population[idx1] if scores[idx1] < scores[idx2] else population[idx2]
            idx3 = rng.randint(pop_size)
            idx4 = rng.randint(pop_size)
            parent2 = population[idx3] if scores[idx3] < scores[idx4] else population[idx4]

            # Uniform crossover
            mask = rng.rand(n) < 0.5
            child = np.where(mask, parent1, parent2)

            # Mutation: flip 1-2 bits (as swap to maintain count)
            n_mut = rng.randint(1, 3)
            for _ in range(n_mut):
                ones = np.where(child == 1.0)[0]
                zeros = np.where(child == 0.0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    a = ones[rng.randint(len(ones))]
                    b = zeros[rng.randint(len(zeros))]
                    child[a], child[b] = child[b], child[a]

            new_pop.append(child.astype(float))

        population = new_pop
        scores = np.array([compute_c5(p) for p in population])
        new_best = np.argmin(scores)
        if scores[new_best] < best_c5 - 1e-15:
            best_c5 = scores[new_best]
            best_h = population[new_best].copy()

    return best_h, best_c5

def coordinate_descent(h, n, max_rounds=50, rng=None):
    """Cycle through positions, adjust each to minimize objective."""
    if rng is None:
        rng = np.random.RandomState(7)
    dx = 2.0 / n
    target = n / 2.0
    h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for rnd in range(max_rounds):
        if rng.rand() < 0.5:
            order = np.arange(n)
        else:
            order = rng.permutation(n)

        for idx in order:
            orig = h[idx]
            # Try a few values
            candidates = [0.0, 1.0, max(0, min(1, orig + 0.1)), max(0, min(1, orig - 0.1))]
            best_val = orig
            best_local_c5 = compute_c5(h)

            for cv in candidates:
                if abs(cv - orig) < 1e-10:
                    continue
                h[idx] = cv
                h = project_feasible(h, n)
                c5_val = compute_c5(h)
                if c5_val < best_local_c5 - 1e-12:
                    best_local_c5 = c5_val
                    best_val = cv

            h[idx] = best_val
            h = project_feasible(h, n)

        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-12:
            best_c5 = c5_now
            best_h = h.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_beta
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time) - 3.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    parent_h = None
    parent_n = 0
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        if parent_n > 0:
            parent_c5 = compute_c5(parent_h)
            best_h = parent_h.copy()
            best_c5 = parent_c5
            best_n = parent_n
            print("Parent: n={}, C5={:.6f}".format(parent_n, parent_c5))
    except Exception:
        pass

    if best_h is None:
        best_n = 200
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)
        best_n = len(best_h)
        print("Default init: n={}, C5={:.6f}".format(best_n, best_c5))

    def upsample_to(parent, n_new):
        n_old = len(parent)
        xp = (np.arange(n_old) + 0.5) * (2.0 / n_old)
        x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
        interp = np.interp(x_new, xp, parent)
        return project_feasible(interp, n_new)

    resolutions = [200, 400, 600]

    for n in resolutions:
        if remaining() < 60:
            break
        print("\n=== Resolving n={} ===".format(n))

        # --- Strategy A: Continuous SLSQP from upsampled parent ---
        if remaining() > 40:
            h0 = upsample_to(best_h, n)
            _g_n = n
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': slsqp_eq_con}]

            for beta_val in [20, 60, 150]:
                if remaining() < 15:
                    break
                _g_beta = beta_val
                try:
                    res = minimize(slsqp_smooth_obj, h0, method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 300, 'ftol': 1e-13, 'iprint': 0})
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        best_n = n
                        print("  SLSQP beta={}: C5={:.6f}".format(beta_val, c5_new))
                except Exception:
                    pass

            # Raw max polish
            _g_beta = 500
            try:
                res = minimize(slsqp_raw_obj, h0, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 200, 'ftol': 1e-14, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n
                    print("  SLSQP raw: C5={:.6f}".format(c5_new))
            except Exception:
                pass

        # --- Strategy B: Binary local search ---
        if remaining() > 50:
            h_bin = (best_h > 0.5).astype(float)
            if best_n != n:
                h_bin = (upsample_to(best_h, n) > 0.5).astype(float)
            # Also try random binary inits
            binary_inits = [h_bin]
            for _ in range(3):
                hr = rng.rand(n)
                binary_inits.append((hr > 0.5).astype(float))

            for bi in binary_inits:
                if remaining() < 20:
                    break
                h_ls, c5_ls = binary_local_search(bi, n, iters=80000, rng=rng)
                if c5_ls < best_c5:
                    best_c5 = c5_ls
                    best_h = h_ls.copy()
                    best_n = n
                    print("  Binary LS: C5={:.6f}".format(c5_ls))

        # --- Strategy C: Genetic algorithm ---
        if remaining() > 40:
            h_ga, c5_ga = genetic_algorithm(best_h, n, pop_size=50, generations=200, rng=rng)
            if c5_ga < best_c5:
                best_c5 = c5_ga
                best_h = h_ga.copy()
                best_n = n
                print("  GA: C5={:.6f}".format(c5_ga))

        # --- Strategy D: Polish best at this resolution ---
        if best_n == n and remaining() > 30:
            _g_n = n
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': slsqp_eq_con}]
            h0 = project_feasible(best_h, n)
            _g_beta = 100
            try:
                res = minimize(slsqp_smooth_obj, h0, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-14, 'iprint': 0})
                h_p = project_feasible(res.x, n)
                c5_p = compute_c5(h_p)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p.copy()
                    print("  Polish: C5={:.6f}".format(c5_p))
            except Exception:
                pass

    # --- Strategy E: Final high-resolution polish ---
    if remaining() > 30:
        n_final = max(best_n, 400)
        if n_final % 2 != 0:
            n_final += 1
        h_f = upsample_to(best_h, n_final)
        _g_n = n_final
        bounds = [(0.0, 1.0)] * n_final
        cons = [{'type': 'eq', 'fun': slsqp_eq_con}]

        for beta_val in [50, 200]:
            if remaining() < 15:
                break
            _g_beta = beta_val
            try:
                res = minimize(slsqp_smooth_obj, h_f, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-14, 'iprint': 0})
                h_new = project_feasible(res.x, n_final)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n_final
                    print("  Final polish n={} beta={}: C5={:.6f}".format(n_final, beta_val, c5_new))
            except Exception:
                pass

        # Binary LS at final resolution
        if remaining() > 30:
            h_bin = (best_h > 0.5).astype(float)
            h_ls, c5_ls = binary_local_search(h_bin, best_n if best_n % 2 == 0 else best_n + 1,
                                              iters=100000, rng=rng)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
                best_n = len(h_ls)
                print("  Final binary LS: C5={:.6f}".format(c5_ls))

    # --- Final: coordinate descent polish ---
    if remaining() > 15:
        h_cd, c5_cd = coordinate_descent(best_h, best_n, max_rounds=10, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            print("  Coord descent: C5={:.6f}".format(c5_cd))

    # Final evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
