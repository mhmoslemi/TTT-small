import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    mask = (h > 1e-12) & (h < 1.0 - 1e-12)
    if np.sum(mask) > 0 and abs(diff) > 1e-12:
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_obj(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    return (m + np.log(np.sum(exps)) / beta) * dx

                    best_h = h_sa2.copy()
                    best_n = n
                    print("  SA2: n={}, C5={:.6f}".format(n, best_c5))

        # Final SLSQP on best binary/semi-binary
        if remaining() > 30:
            h_fin, c_fin = slsqp_polish(best_h.copy() if best_n == n else np.zeros(n), n,
                                         min(40.0, remaining() * 0.3))
            if best_n == n and c_fin < best_c5:
                best_c5 = c_fin
                best_h = h_fin.copy()
                print("  Final SLSQP: n={}, C5={:.6f}".format(n, best_c5))

        phase_time = time.time() - t_phase
        print("  Phase n={} done in {:.0f}s, best C5={:.6f}".format(n, phase_time, best_c5))

    # Binary rounding on final best
    if remaining() > 5 and best_h is not None:
        h_br = (best_h > 0.5).astype(float)
        n_ones = int(h_br.sum())
        n_want = best_n // 2
        if n_ones > n_want:
            ones = np.where(h_br == 1.0)[0]
            order = np.argsort(best_h[ones])[:n_ones - n_want]
            h_br[ones[order]] = 0.0
        elif n_ones < n_want:
            zeros = np.where(h_br == 0.0)[0]
            order = np.argsort(-best_h[zeros])[:n_want - n_ones]
            h_br[zeros[order]] = 1.0
        c5_br = compute_c5(h_br)
        if c5_br < best_c5:
            best_c5 = c5_br
            best_h = h_br.copy()
            print("  Binary rounding: C5={:.6f}".format(best_c5))

    # Final SA on very small timescale for last tweaks
    if remaining() > 10 and best_h is not None and best_n % 2 == 0:
        h_last, c_last = sa_binary(best_h, best_n, 200000, 0.001, 1e-8, seed + 9999)
        if c_last < best_c5:
            best_c5 = c_last
            best_h = h_last.copy()
            print("  Last SA: C5={:.6f}".format(best_c5))

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
