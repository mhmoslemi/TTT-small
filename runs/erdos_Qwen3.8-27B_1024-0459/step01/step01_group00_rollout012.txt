```python
def compute_gradient(h, weights):
    """Compute gradient of sum_k weights[k]*out[k] w.r.t. h.
    
    weights: array of shape (2n-1,)
    Returns: gradient array of shape (n,)
    """
    n = len(h)
    grad = np.zeros(n)
    
    for k in range(2*n-1):
        l = k - (n-1)
        # Valid i range: max(0,-l) <= i <= min(n-1, n-1-l)
        i_min = max(0, -l)
        i_max = min(n-1, n-1-l)
        
        for i in range(i_min, i_max+1):
            # First term: ∂(h[i+l])/∂h[j] at j=i+l, contributes weights[k]*(1-h[i])
            j1 = i + l
            grad[j1] += weights[k] * (1.0 - h[i])
            # Second term: ∂(1-h[i])/∂h[j] at j=i, contributes -weights[k]*h[i+l]
            j2 = i
            grad[j2] -= weights[k] * h[i+l]
    
    return grad
```

This is O(n²) per call. For n=160, that's 25600 inner loop iterations. In Python, that's probably ~50-100 microseconds. Fast enough.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def smooth_max_weights(corr, beta):
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    return weights

def compute_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        l = k - (n - 1)
        i_min = max(0, -l)
        i_max = min(n - 1, n - 1 - l)
        if i_min > i_max:
            continue
        i_range = np.arange(i_min, i_max + 1)
        idx1 = i_range + l
        idx2 = i_range
        w = weights[k]
        grad[idx1] += w * (1.0 - h[i_range])
        grad[idx2] -= w * h[idx1]
    return grad

def projected_subgradient(h, n, n_iters, lr_start, lr_end, beta_start, beta_end, 
                          momentum=0.0, ftol=1e-15):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    lr = lr_start
    n_range = max(n_iters, 1)
    
    for it in range(n_iters):
        frac = it / n_range
        lr = lr_start * (lr_end / lr_start) ** frac
        beta = beta_start * (beta_end / beta_start) ** frac
        
        corr = compute_corr(h)
        weights = smooth_max_weights(corr, beta)
        grad = compute_gradient(h, weights)
        
        v = momentum * v + lr * grad
        h_new = h - v
        h_new = np.clip(h_new, 0.0, 1.0)
        
        # Fix sum constraint
        diff = n / 2.0 - np.sum(h_new)
        if abs(diff) > 1e-14:
            mask = (h_new > 0.01) & (h_new < 0.99)
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h_new[mask] += diff / cnt
                h_new = np.clip(h_new, 0.0, 1.0)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - ftol:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = h_new  # still move (momentum helps escape)
            v *= 0.5
    
    return make_feasible(best_h, n), compute_c5(make_feasible(best_h, n))

def coordinate_descent(h, n, n_passes, n_inner=10):
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    step_sizes = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    for p in range(n_passes):
        improved = False
        for j in range(n):
            best_val = h[j]
            best_local_c5 = best_c5
            for ss in step_sizes:
                for delta in [-ss, ss]:
                    h_try = h.copy()
                    h_try[j] = np.clip(h[j] + delta, 0.0, 1.0)
                    # Fix sum
                    diff = n / 2.0 - np.sum(h_try)
                    if abs(diff) > 1e-12:
                        # Redistribute to closest non-boundary coords
                        mask = (h_try > 0.01) & (h_try < 0.99)
                        if j not in np.where(mask)[0]:
                            mask[j] = False
                        cnt = np.count_nonzero(mask)
                        if cnt > 0:
                            h_try[mask] += diff / cnt
                            h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_local_c5 - 1e-14:
                        best_local_c5 = c5_try
                        best_val = h_try[j]
            if abs(best_val - h[j]) > 1e-10:
                h[j] = best_val
                # Fix sum after update
                diff = n / 2.0 - np.sum(h)
                mask = (h > 0.01) & (h < 0.99)
                cnt = np.count_nonzero(mask)
                if cnt > 0:
                    h[mask] += diff / cnt
                    h = np.clip(h, 0.0, 1.0)
                best_c5 = best_local_c5
                improved = True
        if not improved:
            break
    
    return make_feasible(h, n), compute_c5(make_feasible(h, n))

def random_perturbation(h, n, n_trials, step_size, seed_val=42):

    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)

        diff = n / 2.0 - np.sum(h_try)
        mask = (h_try > 0.01) & (h_try < 0.99)

def simulated_anneal(h, n, n_iters, temp_start, temp_end, step_scale, seed_val=42):
    rng = np.random.RandomState(seed_val)
    h = make_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    
    for it in range(n_iters):
        frac = it / max(n_iters - 1, 1)
        temp = temp_start * (temp_end / temp_start) ** frac
        ss = step_scale * (1.0 - 0.8 * frac)
        
        h_try = h.copy()
        n_mod = rng.randint(1, 4)
        idxs = rng.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += rng.normal(0, ss)
        h_try = np.clip(h_try, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h_try)
        mask = (h_try > 0.01) & (h_try < 0.99)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h_try[mask] += diff / cnt
            h_try = np.clip(h_try, 0.0, 1.0)
        
        c5_try = compute_c5(h_try)
        if c5_try < cur_c5 or rng.random() < np.exp(-(c5_try - cur_c5) / max(temp, 1e-10)):
            h = h_try
            cur_c5 = c5_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
    
    return make_feasible(best_h, n), compute_c5(make_feasible(best_h, n))

def refine_with_annealed_subgradient(h, n, time_s, seed_val=42):
    """Multi-phase subgradient with increasing smoothness."""
    start_t = time.time()
    current = make_feasible(h.copy(), n)
    best_c5 = compute_c5(current)
    
    phases = [
        (5000, 0.05, 0.01, 20.0, 80.0, 0.0),
        (5000, 0.03, 0.005, 80.0, 200.0, 0.0),
        (5000, 0.02, 0.002, 200.0, 500.0, 0.1),
        (5000, 0.01, 0.001, 500.0, 1000.0, 0.1),
    ]
    
    for (n_ops, lr_s, lr_e, b_s, b_e, mom) in phases:
        if time.time() - start_t > time_s * 0.85:
            break
        t0 = time.time()
        # Estimate how many iterations we can afford
        # First do a timing run
        corr = compute_corr(current)
        weights = smooth_max_weights(corr, b_s)
        grad = compute_gradient(current, weights)
        per_iter = max(time.time() - t0, 1e-6)
        affordable = int(min(n_ops, (time_s * 0.15) / per_iter))
        affordable = max(affordable, 100)
        
        h_new, c5_new = projected_subgradient(current, n, affordable, lr_s, lr_e, b_s, b_e, momentum=mom)
        if c5_new < best_c5:
            best_c5 = c5_new
            current = h_new
    
    return make_feasible(current, n), best_c5, compute_c5(make_feasible(current, n))

                                        temp_start=0.01, temp_end=0.0001,
                                        step_scale=0.05, seed_val=seed + 555)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
    
    # Phase 5: Final fine subgradient
    if remaining() > 30:
        h_fin, c5_fin, c5_fcheck = refine_with_annealed_subgradient(
            best_h, best_n, remaining() * 0.6, seed + 1234)
        if c5_fcheck < best_c5:
            best_c5 = c5_fcheck
            best_h = h_fin
    
    # Phase 6: Final coordinate + random polish
    if remaining() > 20:
        h_cd2, c5_cd2 = coordinate_descent(best_h, best_n, n_passes=3)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2
    
    if remaining() > 10:
        n_trials = int(min(50000, remaining() * 80))
        h_rp, c5_rp = random_perturbation(best_h, best_n, n_trials, 0.003, seed + 4242)
        if c5_rp < best_c5:
            best_c5 = c5_rp
            best_h = h_rp
    
    # Final check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
