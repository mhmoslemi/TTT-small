def polish_gradient(h, deadline, max_iter=500, active_k=6):
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)

    if n < 2:

        if time.perf_counter() >= deadline:
            break
        
        corr = np.correlate(best, 1.0 - best, mode="full")
        if not np.all(np.isfinite(corr)):
            break
        
        # Get top-k lags
        idxs = np.argsort(corr)[-k:]

        g = np.zeros(n)
        center = n - 1
        for t_idx in idxs:
            t_idx = int(t_idx)

            if kk == 0:
                g += 1.0 - 2.0 * best
            else:
                lag = abs(kk)
                if kk > 0:

                    g[lag:] -= best[:n - lag]

                    g[:n - lag] += 1.0

                    g[lag:] -= best[:n - lag]
        g /= float(k)

            if stall > 20:
                break

    if n == 0:
        return x
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)

def cheb_eval(coeffs):
    n = _CG['n']
    Tmat = _CG['Tmat']
    h = np.zeros(n)
    for j in range(len(coeffs)):
        c = float(coeffs[j])
        if abs(c) > 1e-15:
            h += c * Tmat[j]
    return h

def cheb_project(coeffs):
    h = cheb_eval(coeffs)
    h = np.clip(h, 0.0, 1.0)
    return project_density(h, 0.5 * _CG['n'])

def cheb_soft_obj(coeffs):
    n = _CG['n']
    alpha = _CG['alpha']
    h = cheb_project(coeffs)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return 1e100
    return (2.0 / n) * (m + np.log(z) / alpha)

def grid_smooth_jac(x):
    n = _CG['n']
    alpha = _CG['alpha']
    h = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return np.zeros(n)
    w = e / z
    g = np.zeros(n)
    center = n - 1
    for t_idx in np.where(w > 1e-12)[0]:
        p = float(w[t_idx])
        if p <= 0.0:
            continue
        t_idx = int(t_idx)
        kk = t_idx - center
        if kk == 0:
            g += p * (1.0 - 2.0 * h)
        else:
            lag = abs(kk)
            if kk > 0:
                g[lag:] += p
                g[:n - lag] -= p * h[lag:]
                g[lag:] -= p * h[:n - lag]
            else:
                g[:n - lag] += p
                g[:n - lag] -= p * h[lag:]
                g[lag:] -= p * h[:n - lag]
    return (2.0 / n) * g

def sum_constraint(x):
    return float(np.sum(np.asarray(x, dtype=float)) - _CG['target'])

def sum_constraint_jac(x):
    return np.ones_like(np.asarray(x, dtype=float), dtype=float)

def corr_gradient(h, indices):
    n = h.size
    g = np.zeros(n)
    center = n - 1
    for t_idx in indices:
        t_idx = int(t_idx)
        kk = t_idx - center
        if kk == 0:
            g += 1.0 - 2.0 * h
        else:
            lag = abs(kk)
            if kk > 0:
                g[lag:] += 1.0
                g[:n - lag] -= h[lag:]
                g[lag:] -= h[:n - lag]
            else:
                g[:n - lag] += 1.0
                g[:n - lag] -= h[lag:]
                g[lag:] -= h[:n - lag]
    return g

def polish_gradient(h, deadline, max_iter=500, active_k=8):
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    if n < 2:
        return best, best_c
    k = min(int(active_k), 2 * n - 1)
    stall = 0
    for it in range(int(max_iter)):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        if not np.all(np.isfinite(corr)):
            break
        idxs = np.argsort(corr)[-k:]
        g = corr_gradient(best, idxs) / float(k)
        d = -g
        d -= float(np.mean(d))
        norm = float(np.linalg.norm(d))
        if norm < 1e-14:
            break
        d /= norm
        improved = False
        for step in (0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5):
            cand = project_density(best + step * d, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-13:
                best = cand
                best_c = c
                improved = True
                break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 25:
                break
    best = project_density(best, target)
    best_c = evaluate_erdos(best)
    return best, best_c
