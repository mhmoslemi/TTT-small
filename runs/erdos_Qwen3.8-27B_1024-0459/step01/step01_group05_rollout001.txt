def project_box_sum(v,target):
    v=np.array(v,dtype=float,copy=True)
    n=len(v)
    if n==0: return v
    if target<0: target=0
    if target>n: target=float(n)
    if not np.all(np.isfinite(v)):
        v=np.clip(v,0,1)
    # quick if already
    s=np.sum(v)
    if np.all(v>=0) and np.all(v<=1) and abs(s-target)<1e-13:
        return v
    lo=float(v.min())-1.0
    hi=float(v.max())+1.0
    # If target close 0/n, bisection works.
    for _ in range(55):
       mid=0.5*(lo+hi)
       s=np.sum(np.clip(v-mid,0.0,1.0))
       if s>target: lo=mid
       else: hi=mid
    tau=0.5*(lo+hi)
    out=np.clip(v-tau,0.0,1.0)

import numpy as np
import time
from scipy.optimize import minimize

CACHE_H = None
CACHE_W = None
CACHE_BETA = -1.0
CACHE_OK = False


def invalidate_cache():
    global CACHE_OK
    CACHE_OK = False


def default_initial():
    n = 95
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.25 * np.cos(np.pi * x)
    h = np.clip(h, 0.0, 1.0)
    return project_box_sum(h, 0.5 * n)


def get_parent_initial():
    vals = globals().get("initial_h_values")
    if vals is not None:
        h = np.array(vals, dtype=float)
    else:
        h = default_initial()
    n = len(h)
    return project_box_sum(h, 0.5 * n)


def project_box_sum(v, target):
    v = np.array(v, dtype=float, copy=True)
    n = len(v)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.5, posinf=1.0, neginf=0.0)
    target = float(target)
    if target < 0.0:
        target = 0.0
    if target > float(n):
        target = float(n)

    if np.all(v >= 0.0) and np.all(v <= 1.0) and abs(float(np.sum(v)) - target) < 1e-13:
        return v

    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0

    for _ in range(55):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    tau = 0.5 * (lo + hi)
    out = np.clip(v - tau, 0.0, 1.0)

    for _ in range(20):
        s = float(np.sum(out))
        diff = target - s
        if abs(diff) <= 1e-14:
            break
        if diff > 0.0:
            mask = out < 1.0
        else:
            mask = out > 0.0
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        out[mask] += diff / float(cnt)
        out = np.clip(out, 0.0, 1.0)

    return out


def next_power_two(x):
    p = 1
    while p < int(x):
        p <<= 1
    return p


def erdos_r(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    dx = 2.0 / n
    N = next_power_two(2 * n - 1)

    fh = np.fft.fft(h, N)
    a = np.fft.ifft(fh * np.conj(fh)).real

    total = float(np.sum(h))
    pre = np.empty(n + 1, dtype=float)
    pre[0] = 0.0
    pre[1:] = np.cumsum(h)

    L = 2 * n - 1
    mid = n - 1
    S = np.empty(L, dtype=float)
    A = np.empty(L, dtype=float)

    if n > 1:
        S[mid + 1:] = pre[n - 1:0:-1]
        S[0:n - 1] = total - pre[n - 1:0:-1]
        A[mid + 1:] = a[1:n]
        A[0:n - 1] = a[N - n + 1:N]

    S[mid] = total
    A[mid] = a[0]

    r = dx * (S - A)
    return r


def smooth_max_weights(r, beta):
    beta = float(beta)
    m = float(np.max(r))
    if beta <= 0.0:
        w = np.zeros_like(r, dtype=float)
        w[int(np.argmax(r))] = 1.0
        return m, w

    sh = beta * (r - m)
    sh = np.clip(sh, -50.0, 0.0)
    ex = np.exp(sh)
    s = float(np.sum(ex))

    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(r, dtype=float)
        w[int(np.argmax(r))] = 1.0
        return m, w

    w = ex / s
    val = m + np.log(s) / beta
    return float(val), w


def gradient_from_weights(h, w):
    h = np.asarray(h, dtype=float)
    w = np.asarray(w, dtype=float)
    n = len(h)
    L = 2 * n - 1
    mid = n - 1
    N = next_power_two(2 * n - 1)

    P = np.zeros(N, dtype=float)
    P[mid:mid + L] = w

    Pw = np.zeros(N, dtype=float)
    Pw[mid:mid + L] = w[::-1]

    U = 1.0 - h
    H = h

    t1 = np.fft.ifft(np.fft.fft(Pw) * np.fft.fft(U)).real
    t2 = np.fft.ifft(np.fft.fft(P) * np.fft.fft(H)).real

    sl = slice(mid, mid + n)
    return (2.0 / n) * (t1[sl] - t2[sl])


def erdos_smooth_objective(h, beta):
    global CACHE_H, CACHE_W, CACHE_BETA, CACHE_OK
    h = np.asarray(h, dtype=float)
    if not np.all(np.isfinite(h)):
        return 1e100
    r = erdos_r(h)
    val, w = smooth_max_weights(r, float(beta))
    CACHE_H = h.copy()
    CACHE_W = w
    CACHE_BETA = float(beta)
    CACHE_OK = True
    return val


def erdos_smooth_gradient(h, beta):
    global CACHE_H, CACHE_W, CACHE_BETA, CACHE_OK
    h = np.asarray(h, dtype=float)
    beta = float(beta)
    if not np.all(np.isfinite(h)):
        return np.zeros_like(h, dtype=float)

    if (not CACHE_OK or CACHE_BETA != beta or CACHE_H is None or
            CACHE_H.shape != h.shape or not np.array_equal(CACHE_H, h)):
        r = erdos_r(h)
        _, w = smooth_max_weights(r, beta)
        CACHE_H = h.copy()
        CACHE_W = w
        CACHE_BETA = beta
        CACHE_OK = True

    w = CACHE_W
    return gradient_from_weights(h, w)


def sum_constraint(h, target):
    return float(np.sum(h) - target)


def sum_jacobian(h, target):
    return np.ones_like(h, dtype=float)


def objective_wrapper(h, beta, target):
    return erdos_smooth_objective(h, beta)


def gradient_wrapper(h, beta, target):
    return erdos_smooth_gradient(h, beta)


def exact_c5(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return 1e100
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full")
    return float(np.max(corr) * (2.0 / n))


def fast_c5(h):
    return float(np.max(erdos_r(h)))


def circular_shift(h, shift):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return h.copy()
    shift = int(shift) % n
    return np.roll(h, -shift).copy()


def fourier_lowpass(h, modes):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n < 2 or modes <= 0:
        return h.copy()
    modes = min(int(modes), max(1, n // 2))
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    d = h - 0.5
    rec = 0.5 * np.ones(n, dtype=float)
    norm = 2.0 / n
    for k in range(1, modes + 1):
        th = np.pi * k * x
        c = norm * float(np.dot(d, np.cos(th)))
        s = norm * float(np.dot(d, np.sin(th)))
        rec += c * np.cos(th) + s * np.sin(th)
    return project_box_sum(rec, 0.5 * n)


def fourier_perturb(h, modes, sigma):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n < 2 or modes <= 0:
        return h.copy()
    modes = min(int(modes), max(1, n // 2))
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    noise = np.zeros(n, dtype=float)
    for k in range(1, modes + 1):
        th = np.pi * k * x
        a = float(np.random.randn()) * sigma
        b = float(np.random.randn()) * sigma
        noise += a * np.cos(th) + b * np.sin(th)
    noise -= float(np.mean(noise))
    return project_box_sum(h + noise, 0.5 * n)


def resample_to_n(h, n_target):
    h = np.asarray(h, dtype=float)
    n_base = len(h)
    n_target = int(n_target)
    if n_target <= 0:
        return h.copy()
    if n_target == n_base:
        return project_box_sum(h, 0.5 * n_target)
    if n_target % n_base == 0:
        out = np.repeat(h, n_target // n_base)
    elif n_base % n_target == 0:
        out = h.reshape(n_target, n_base // n_target).mean(axis=1)
    else:
        x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
        pos = x_new / (2.0 / float(n_base))
        pos = np.mod(pos, float(n_base))
        i0 = np.floor(pos).astype(int)
        i1 = (i0 + 1) % n_base
        frac = pos - i0
        out = h[i0] * (1.0 - frac) + h[i1] * frac
    return project_box_sum(out, 0.5 * n_target)


def run_slsqp(h0, target, beta, maxiter):
    h0 = project_box_sum(h0, target)
    n = len(h0)
    if n == 0:
        return h0

    invalidate_cache()

    bounds = [(0.0, 1.0)] * n
    constraints = [
        {"type": "eq", "fun": sum_constraint, "jac": sum_jacobian, "args": (target,)}
    ]
    options = {
        "maxiter": int(maxiter),
        "ftol": 1e-12,
        "maxls": 15,
        "disp": False,
    }

    try:
        res = minimize(
            objective_wrapper,
            h0.copy(),
            method="SLSQP",
            jac=gradient_wrapper,
            args=(float(beta), float(target)),
            bounds=bounds,
            constraints=constraints,
            options=options,
        )
    except Exception:
        return h0

    if res is not None and res.x is not None and np.all(np.isfinite(res.x)):
        return project_box_sum(res.x, target)
    return h0


def local_refine(h, max_iters, beta=250.0):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return h
    target = 0.5 * n
    best = project_box_sum(h, target)
    best_c = fast_c5(best)
    alpha = 0.05
    stall = 0

    for _ in range(int(max_iters)):
        r = erdos_r(best)
        _, w = smooth_max_weights(r, float(beta))
        grad = gradient_from_weights(best, w)
        gscale = float(np.max(np.abs(grad)))
        if not np.isfinite(gscale) or gscale < 1e-14:
            break
        step = grad / gscale
        new = project_box_sum(best - alpha * step, target)
        c = fast_c5(new)
        if c < best_c - 1e-13:
            best = new
            best_c = c
            stall = 0
            if alpha < 0.12:
                alpha *= 1.05
        else:
            alpha *= 0.6
            stall += 1
            if stall > 25 or alpha < 1e-5:
                break

    return best


def final_polish(h, max_iters=45):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return h
    target = 0.5 * n
    best = project_box_sum(h, target)
    best_c = exact_c5(best)
    alpha = 0.01

    for _ in range(int(max_iters)):
        r = erdos_r(best)
        t = float(np.max(r))
        tol = max(1e-8, 1e-5 * abs(t))
        active = r >= (t - tol)
        cnt = int(np.count_nonzero(active))
        if cnt == 0:
            break
        w = np.zeros_like(r, dtype=float)
        w[active] = 1.0 / float(cnt)
        grad = gradient_from_weights(best, w)
        gscale = float(np.max(np.abs(grad)))
        if not np.isfinite(gscale) or gscale < 1e-14:
            break
        step = grad / gscale

        improved = False
        step_size = alpha
        for __ in range(10):
            new = project_box_sum(best - step_size * step, target)
            c = exact_c5(new)
            if c < best_c - 1e-14:
                best = new
                best_c = c
                alpha = min(0.06, alpha * 1.1)
                improved = True
                break
            step_size *= 0.5

        if not improved:
            alpha *= 0.5
            if alpha < 1e-6:
                break

    return best


def optimize_resolution(start_h, n, beta_list, maxiter_base, deadline, floor):
    n = int(n)
    target = 0.5 * n
    cur = project_box_sum(start_h, target)
    best_local = cur.copy()
    best_local_c = exact_c5(cur)

    for beta in beta_list:
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < floor:
            break

        mi = int(maxiter_base)
        if n <= 120:
            mi = min(mi, 350)
        elif n <= 220:
            mi = min(mi, 220)
        elif n <= 320:
            mi = min(mi, 130)
        else:
            mi = min(mi, 80)

        if remaining < 120.0:
            mi = max(30, int(round(mi * max(0.2, remaining / 120.0))))

        h_opt = run_slsqp(cur, target, float(beta), mi)
        c = exact_c5(h_opt)

        if c < best_local_c - 1e-14:
            best_local = h_opt
            best_local_c = c
            cur = h_opt
        else:
            if c <= best_local_c + 0.02:
                cur = h_opt
            else:
                cur = best_local

    if time.time() + 5.0 < deadline:
        iters = 250 if n <= 120 else 120 if n <= 220 else 60
        cand = local_refine(best_local, iters, beta=250.0)
        c = exact_c5(cand)
        if c < best_local_c - 1e-14:
            best_local = cand
            best_local_c = c

    return best_local, best_local_c


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget_s = max(1.0, float(budget_s))
    np.random.seed(seed)

    h_initial = get_parent_initial()
    orig_n = int(len(h_initial))

    best_h = h_initial.copy()
    best_c = exact_c5(best_h)
    best_n = orig_n
    parent_h = h_initial.copy()

    work_deadline = t0 + max(10.0, budget_s * 0.90)
    floor = 30.0 if budget_s > 200.0 else 5.0

    if time.time() < work_deadline:
        candidates = []
        candidates.append(best_h[::-1].copy())
        if best_n > 1:
            step = max(1, best_n // 4)
            for m in (1, 2, 3):
                sh = circular_shift(best_h, m * step)
                candidates.append(sh)
                candidates.append(0.5 * (best_h + sh))
        candidates.append(0.5 * (best_h + best_h[::-1].copy()))
        for md in (4, 8, 16):
            if best_n >= 2 * md:
                candidates.append(fourier_lowpass(best_h, md))

        for cand in candidates:
            cand = project_box_sum(cand, 0.5 * best_n)
            cc = exact_c5(cand)
            if cc + 1e-15 < best_c:
                best_h = cand
                best_c = cc

    max_n = 380 if budget_s > 600.0 else 260 if budget_s > 300.0 else 190
    raw_res = []
    for frac in (0.35, 0.5, 0.75, 1.0, 2.0, 3.0):
        nr = int(round(orig_n * frac))
        if 12 <= nr <= max_n:
            if nr not in raw_res:
                raw_res.append(nr)
    if orig_n not in raw_res and orig_n >= 12:
        raw_res.append(orig_n)

    order_tmp = []
    for nr in raw_res:
        order_tmp.append((abs(nr - orig_n), nr))
    order_tmp.sort()
    res_order = []
    for _, nr in order_tmp:
        res_order.append(nr)

    for nr in res_order:
        if time.time() > work_deadline:
            break
        remaining = work_deadline - time.time()
        if remaining < floor:
            break

        if nr == orig_n:
            beta_list = (4.0, 15.0, 50.0, 180.0)
            maxit = 350 if nr <= 120 else 220
        elif nr < orig_n:
            beta_list = (15.0, 80.0)
            maxit = 260 if nr <= 120 else 140
        else:
            beta_list = (12.0, 60.0, 220.0)
            maxit = 260 if nr <= 130 else 160 if nr <= 260 else 90

        if len(best_h) == nr:
            start = best_h.copy()
        else:
            start = resample_to_n(best_h, nr)

        cand, cc = optimize_resolution(start, nr, beta_list, maxit, work_deadline, floor)
        if cc + 1e-15 < best_c:
            best_h = cand
            best_c = cc
            best_n = nr

    if time.time() < work_deadline:
        for scale in (0.012, 0.025):
            remaining = work_deadline - time.time()
            if remaining < max(floor, 15.0):
                break
            start = fourier_perturb(best_h, 12, scale)
            mi = 200 if best_n <= 120 else 120 if best_n <= 220 else 80
            h_opt = run_slsqp(start, best_n, 0.5 * best_n, 100.0, mi)
            c = exact_c5(h_opt)
            if c + 1e-15 < best_c:
                best_h = h_opt
                best_c = c

            if time.time() + 8.0 < work_deadline:
                cand = local_refine(h_opt, 80, beta=250.0)
                c2 = exact_c5(cand)
                if c2 + 1e-15 < best_c:
                    best_h = cand
                    best_c = c2

    if time.time() + 20.0 < t0 + budget_s * 0.97:
        iters = 300 if best_n <= 120 else 150 if best_n <= 220 else 80
        cand = local_refine(best_h, iters, beta=300.0)
        c = exact_c5(cand)
        if c + 1e-15 < best_c:
            best_h = cand
            best_c = c

    if time.time() + 25.0 < t0 + budget_s * 0.99:
        cand = final_polish(best_h, 45)
        c = exact_c5(cand)
        if c + 1e-15 < best_c:
            best_h = cand
            best_c = c

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = exact_c5(best_h)
    return best_h, best_c, best_n
