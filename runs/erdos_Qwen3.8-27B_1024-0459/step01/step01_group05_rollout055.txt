    corr = np.correlate(h, 1.0 - h, mode='full')

    # idx corresponds to lag k = idx - (n-1)
    grad = np.zeros(n)

        if wi < 1e-14:

            # For j from 0 to n-1-k: dr/dh[j] -= h[j+k]
            j_range = np.arange(k, n)

def project_to_constraint(h, target):

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(beta * (corr - m))

            best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_constraint(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -5.0, 5.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    result = np.clip(h - lam, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-13:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        if mask.sum() > 0:
            result[mask] += diff / float(mask.sum())
            result = np.clip(result, 0.0, 1.0)
    return result

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -50.0, 50.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(corr)] = 1.0
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for idx in range(len(corr)):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            j1 = np.arange(k, n)
            grad[j1] += wi * (1.0 - h[j1 - k])
            j2 = np.arange(0, n - k)
            grad[j2] -= wi * h[j2 + k]
        else:
            k2 = -k
            j1 = np.arange(0, n - k2)
            grad[j1] += wi * (1.0 - h[j1 + k2])
            j2 = np.arange(k2, n)
            grad[j2] -= wi * h[j2 - k2]
    return grad * dx

def run_proj_grad(h_init, n, max_iter, beta, C0, decay):
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_f = compute_c5(h)
    for t in range(max_iter):
        f_exact = compute_c5(h)
        if f_exact < best_f:
            best_f = f_exact
            best_h = h.copy()
        g = smooth_grad(h, beta)
        step = C0 / (1.0 + decay * t)
        alpha = step
        f_cur = smooth_obj(h, beta)
        for _ in range(25):
            h_trial = project_to_constraint(h - alpha * g, target)
            f_trial = smooth_obj(h_trial, beta)
            if f_trial < f_cur - 1e-14:
                break
            alpha *= 0.5
        h = project_to_constraint(h - alpha * g, target)
    return best_h, best_f

def run_slsqp_opt(h_init, n, beta, max_iter, ftol):
    target = n / 2.0
    h0 = project_to_constraint(h_init, target)
    
    def obj_fn(h):
        return smooth_obj(h, beta)
    
    def grad_fn(h):
        return smooth_grad(h, beta)
    
    def cons_fun(h):
        return np.sum(h) - target
    
    def cons_jac(h):
        return np.ones(n)
    
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(obj_fn, h0, method='SLSQP', jac=grad_fn,
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': max_iter, 'ftol': ftol, 'eps': 1e-8})
        if result.x is not None:
            return project_to_constraint(result.x, target)
    except Exception:
        pass
    return h0

def run_fista(h_init, n, max_iter, beta, L_init):
    target = n / 2.0
    h = h_init.copy()
    y = h.copy()
    t_k = 1.0
    L = L_init
    best_h = h.copy()
    best_f = compute_c5(h)
    for it in range(max_iter):
        f_y = smooth_obj(y, beta)
        g_y = smooth_grad(y, beta)
        alpha = 1.0 / L
        f_candidate = f_y
        for _ in range(30):
            h_new = project_to_constraint(y - alpha * g_y, target)
            f_candidate = smooth_obj(h_new, beta)
            if f_candidate <= f_y - 0.5 * alpha * np.dot(g_y, h_new - y):
                break
            alpha *= 0.5
        else:
            h_new = project_to_constraint(y - alpha * g_y, target)
        diff = h_new - h
        y_new = h_new + ((t_k - 1.0) / ((1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0)) * diff
        y_new = project_to_constraint(y_new, target)
        h = h_new
        y = y_new
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        L = max(L * 1.05, 0.1)
        f_exact = compute_c5(h)
        if f_exact < best_f:
            best_f = f_exact
            best_h = h.copy()
    return best_h, best_f
