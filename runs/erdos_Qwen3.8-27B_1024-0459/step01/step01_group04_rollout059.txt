def project_feasible(y, n):
    """Project y onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Bisection for lambda
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        s = float(np.sum(h))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(y - lam, 0.0, 1.0)
    # Fix numerical issues
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h

Now for the binary local search. Let me adapt the previous code's approach but make it faster:

For binary h (all 0s and 1s), the score at shift d is:
- scores[n-1+d] = count of i where h[i+d]=1 AND h[i]=0 (for i=0,...,n-1-d)
- scores[n-1-d] = count of i where h[i]=1 AND h[i+d]=0

When we swap h[i] and h[j] (one is 0, one is 1), we update the scores:
For each shift d:
- If i+d is valid: the pair (h[i], h[i+d]) changes
- If j+d is valid: the pair (h[j], h[j+d]) changes
- If i-d is valid: the pair (h[i-d], h[i]) changes
- If j-d is valid: the pair (h[j-d], h[j]) changes

The previous code's binary_flip_update does this. Let me keep it.

def _c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0 or not np.all(np.isfinite(h)):
        return 1e6
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    v = float(np.max(corr) * (2.0 / n))
    return v if math.isfinite(v) else 1e6

def _proj(y, n):
    target = 0.5 * float(n)
    y = np.asarray(y, dtype=float)
    lo = float(np.min(y)) - 2.0
    hi = float(np.max(y)) + 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    d = target - float(np.sum(h))
    if d > 1e-13:
        idx = int(np.argmax(1.0 - h))
        h[idx] = min(1.0, h[idx] + d)
    elif d < -1e-13:
        idx = int(np.argmax(h))
        h[idx] = max(0.0, h[idx] + d)
    return h

def _scores(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _sgrad(h, n, alpha, sc):
    m = float(np.max(sc))
    e = np.exp((sc - m) / alpha)
    Z = float(np.sum(e))
    if Z <= 0 or not math.isfinite(Z):
        return np.zeros(n)
    w = e / Z
    g = np.zeros(n)
    g += w[n-1] * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = w[n-1+d]
        wm = w[n-1-d]
        nd = n - d
        if wp > 1e-17:
            g[d:] += wp * (1.0 - h[:nd])
            g[:nd] -= wp * h[d:]
        if wm > 1e-17:
            g[:nd] += wm * (1.0 - h[d:])
            g[d:] -= wm * h[:nd]
    return g

def _sobj(h, n, alpha, sc):
    m = float(np.max(sc))
    return float(alpha * (m / alpha + math.log(float(np.sum(np.exp((sc - m) / alpha))))))

def _slsqp_obj(x, n, alpha):

    h = np.clip(x, 0.0, 1.0)
    sc = _scores(h)
    m = float(np.max(sc))
    val = alpha * (m / alpha + math.log(float(np.sum(np.exp((sc - m) / alpha)))))

    h = np.clip(x, 0.0, 1.0)

    g = _sgrad(h, n, alpha, sc)
    return g * (2.0 / n)

def _upsample(parent, n):
    n0 = parent.size
    if n0 == n:
        return parent.copy()
    if n0 < 2:
        return np.full(n, 0.5)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x, x0, parent)

def _binary_flip(sc, h, i, nv, n):
    i = int(i)
    nv = int(nv)
    for d in range(1, n):
        pi = 2 * d - 1
        mi = 2 * d
        if i - d >= 0:
            hj = int(h[i - d])
            if nv == 1:
                sc[pi] -= hj
                sc[mi] += 1.0 - hj
            else:
                sc[pi] += hj
                sc[mi] -= 1.0 - hj
        if i + d < n:
            hj = int(h[i + d])
            if nv == 1:
                sc[pi] += 1.0 - hj
                sc[mi] -= hj
            else:
                sc[pi] -= 1.0 - hj
                sc[mi] += hj
    h[i] = nv

def _binary_scores(h, n):
    sc = np.zeros(2 * n - 1)
    hf = h.astype(float)
    for d in range(1, n):
        sc[2*d-1] = float(np.dot(hf[:n-d], 1.0 - hf[d:]))
        sc[2*d] = float(np.dot(hf[d:], 1.0 - hf[:n-d]))
    return sc

    best_h = h.copy()

    stall = 0
    max_stall = 5000

        it += 1

                    else:

                        i, j = pos, pos + d
                        _binary_flip(sc, h, i, 0, n)
                        _binary_flip(sc, h, j, 1, n)
