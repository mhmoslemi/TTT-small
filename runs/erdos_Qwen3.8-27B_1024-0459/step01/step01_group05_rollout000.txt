def _overlap_c_full(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return np.zeros(0)
    total = float(np.sum(h))
    pref = np.empty(n+1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    idx = np.arange(n)
    auto = np.correlate(h, h, mode='full')
    A = auto[n-1:n-1+n]
    c_neg = pref[n-idx] - A
    c_pos = total - pref[idx] - A
    c = np.empty(2*n-1, dtype=float)
    c[0] = c_neg[0]
    if n > 1:
        c[1:n] = c_neg[1:]
        c[n:] = c_pos[1:]
    return c

def _smooth_gradient(h,beta,dx):
    h=np.asarray(h,dtype=float)
    n=h.size
    if n==0: return np.zeros(0)
    c=_overlap_c_full(h)
    m=float(np.max(c))
    z=np.exp(np.clip(beta*(c-m), -745.0, 0.0))
    s=float(np.sum(z))
    if s <=0 or not finite:
       w=np.zeros_like(c); w[c==m]=1.0/max(1,np.count_nonzero(c==m))
    else: w=z/s
    g0 = w[0]*(1.0 - 2.0*h)
    if n > 1:
       a = np.empty(n); a[0]=0; a[1:]=w[1:n]
       b = np.empty(n); b[0]=0; b[1:]=w[n:2*n-1]
       cmat = a+b
       idx=np.arange(n)
       prefA=np.cumsum(a)
       prefB=np.cumsum(b)
       const = prefA[n-1-idx] + prefB[idx]
       corr = np.correlate(h, a, mode='full')
       term1 = corr[n-1:]
       term3 = np.convolve(cmat, h, mode='full')[:n]
       tailB = prefB[n-1-idx]
       g = const - term1 - h*tailB - term3 + g0
    else:
       g = g0
    return (g * float(dx))

```python
import numpy as np
import time
from scipy.optimize import minimize


_REDOJ_BETA = 1.0
_REDOJ_DX = 1.0

_LB_BETA = 1.0
_LB_DX = 1.0
_LB_RHO = 1.0


def _overlap_c_full(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return np.zeros(0)

    total = float(np.sum(h))
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])

    idx = np.arange(n)
    auto = np.correlate(h, h, mode='full')
    auto = auto[n - 1:n - 1 + n]

    c_neg = pref[n - idx] - auto
    c_pos = total - pref[idx] - auto

    c = np.empty(2 * n - 1, dtype=float)
    c[0] = c_neg[0]
    if n > 1:
        c[1:n] = c_neg[1:]
        c[n:] = c_pos[1:]
    return c


def _compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _smooth_objective(h, beta, dx):
    c = _overlap_c_full(h)
    m = float(np.max(c))
    z = np.exp(np.clip(float(beta) * (c - m), -745.0, 0.0))
    s = float(np.sum(z))
    if s <= 0.0 or not np.isfinite(s):
        return m * float(dx)
    return float((m + np.log(s) / float(beta)) * dx)


def _smooth_gradient(h, beta, dx):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return np.zeros(0)

    c = _overlap_c_full(h)
    m = float(np.max(c))
    z = np.exp(np.clip(float(beta) * (c - m), -745.0, 0.0))
    s = float(np.sum(z))

    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros_like(c)
        am = (c == m)
        cnt = int(np.count_nonzero(am))
        if cnt == 0:
            w[0] = 1.0
        else:
            w[am] = 1.0 / float(cnt)
    else:
        w = z / s

    g0 = w[0] * (1.0 - 2.0 * h)

    if n > 1:
        a = np.empty(n, dtype=float)
        b = np.empty(n, dtype=float)
        a[0] = 0.0
        b[0] = 0.0
        a[1:] = w[1:n]
        b[1:] = w[n:2 * n - 1]

        cmat = a + b
        idx = np.arange(n)

        pref_a = np.cumsum(a)
        pref_b = np.cumsum(b)

        const = pref_a[n - 1 - idx] + pref_b[idx]

        corr_a = np.correlate(h, a, mode='full')
        term1 = corr_a[n - 1:]

        term3 = np.convolve(cmat, h, mode='full')[:n]

        tail_b = pref_b[n - 1 - idx]

        g = const - term1 - h * tail_b - term3 + g0
    else:
        g = g0

    return g * float(dx)


def _slsqp_obj(h):
    return _smooth_objective(h, _REDOJ_BETA, _REDOJ_DX)


def _slsqp_grad(h):
    return _smooth_gradient(h, _REDOJ_BETA, _REDOJ_DX)


def _sum_eq(h):
    h = np.asarray(h, dtype=float)
    return float(np.sum(h) - 0.5 * h.size)


def _sum_eq_jac(h):
    return np.ones(np.asarray(h).size, dtype=float)


def _lb_obj(h):
    h = np.asarray(h, dtype=float)
    val = _smooth_objective(h, _LB_BETA, _LB_DX)
    s = float(np.sum(h) - 0.5 * h.size)
    val += float(_LB_RHO) * s * s
    return float(val)


def _lb_grad(h):
    h = np.asarray(h, dtype=float)
    g = _smooth_gradient(h, _LB_BETA, _LB_DX)
    s = float(np.sum(h) - 0.5 * h.size)
    return g + (2.0 * float(_LB_RHO) * s) * np.ones_like(g)


def _fix_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    target = float(target)
    if n == 0:
        return h

    for _ in range(40):
        s = float(np.sum(h))
        diff = target - s
        if abs(diff) <= 1e-13:

                h[i] += add
        else:
            if np.any(h > 0.0):
                i = int(np.argmax(h))
                add = max(diff, -h[i])
                h[i] += add
    return h


def _project_box_sum(h, target):
    h = np.asarray(h, dtype=float)
    n = h.size
    target = float(target)
    if n == 0:
        return h.copy()
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    h = np.clip(h, 0.0, 1.0)
    lo = -1.0
    hi = 1.0

    for _ in range(48):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid

    hp = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    return _fix_sum(hp, target)


def _casevaluate(h):
    if h is None:
        return None, 0.0, 0, False
    arr = np.asarray(h, dtype=float)
    if arr.size == 0:
        return None, 0.0, 0, False
    if not np.all(np.isfinite(arr)):
        return None, 0.0, 0, False
    hp = _project_box_sum(arr, 0.5 * float(arr.size))
    c5 = _compute_c5(hp)
    return hp, float(c5), int(hp.size), True


def _optimize_lbfgsb(h_start, beta, rho, maxfun, ftol=1e-11):
    global _LB_BETA, _LB_DX, _LB_RHO

    h_start = np.asarray(h_start, dtype=float)
    h = _project_box_sum(h_start, 0.5 * h_start.size)
    n = h.size
    if n <= 1:
        return h

    _LB_BETA = float(beta)
    _LB_DX = 2.0 / float(n)
    _LB_RHO = float(rho)

    bounds = [(0.0, 1.0)] * n
    options = {
        'maxfun': int(max(20, maxfun)),
        'maxiter': int(max(20, maxfun)),
        'ftol': float(ftol),
        'gtol': 1e-8,
        'disp': 0
    }

    try:
        res = minimize(_lb_obj, h, method='L-BFGS-B', bounds=bounds,
                       jac=_lb_grad, options=options)
        if res.x is not None:
            return _project_box_sum(res.x, 0.5 * n)
    except Exception:
        pass
    return h


def _optimize_slsqp(h_start, beta, maxiter, ftol=1e-12):
    global _REDOJ_BETA, _REDOJ_DX

    h_start = np.asarray(h_start, dtype=float)
    h = _project_box_sum(h_start, 0.5 * h_start.size)
    n = h.size
    if n <= 1:
        return h

    _REDOJ_BETA = float(beta)
    _REDOJ_DX = 2.0 / float(n)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]
    options = {
        'maxiter': int(max(20, maxiter)),
        'ftol': float(ftol),
        'iprint': 0
    }

    try:
        res = minimize(_slsqp_obj, h, method='SLSQP', bounds=bounds,
                       constraints=constraints, jac=_slsqp_grad, options=options)
        if res.x is not None:
            return _project_box_sum(res.x, 0.5 * n)
    except Exception:
        pass
    return h


def _make_ramp_desc(n):
    n = max(2, int(n))
    x = (np.arange(n) + 0.5) / float(n)
    return 1.0 - x


def _make_ramp_asc(n):
    n = max(2, int(n))
    x = (np.arange(n) + 0.5) / float(n)
    return x


def _make_spectral(n, seed):
    n = max(2, int(n))
    rng = np.random.RandomState(int(seed) & 0x7fffffff)
    x = 2.0 * (np.arange(n) + 0.5) / float(n)
    h = 0.5 * np.ones(n, dtype=float)

    for m in range(1, 12):
        amp = 0.55 / float(m * m)
        ac = rng.uniform(-amp, amp)
        sa = rng.uniform(-amp, amp)
        h += ac * np.cos(np.pi * m * x)
        h += sa * np.sin(np.pi * m * x)

    return _project_box_sum(h, 0.5 * float(n))


def _make_beatty(n, seed):
    n = max(2, int(n))
    rng = np.random.RandomState(int(seed) & 0x7fffffff)
    alpha = (0.25 + rng.uniform(0.0, 1.0) + np.sqrt(2.0)) % 1.0
    phase = rng.uniform(0.0, 1.0)
    idx = np.arange(n)
    vals = np.mod(idx * alpha + phase, 1.0)
    h = (vals < 0.5).astype(float)
    return _project_box_sum(h, 0.5 * float(n))


def _make_blocks(n, seed):
    n = max(2, int(n))
    rng = np.random.RandomState(int(seed) & 0x7fffffff)
    h = np.empty(n, dtype=float)
    i = 0
    val = 1.0
    min_len = 2
    max_len = max(3, n // 25)
    while i < n:
        ln = int(rng.randint(min_len, max_len + 1))
        ln = min(ln, n - i)
        h[i:i + ln] = val
        i += ln
        if i < n and rng.random() < 0.75:
            val = 1.0 - val
    return _project_box_sum(h, 0.5 * float(n))


def _interpolate_h(h_old, n_new):
    h_old = np.asarray(h_old, dtype=float)
    n_old = h_old.size
    n_new = max(2, int(n_new))
    if n_old == 1:
        return np.full(n_new, float(h_old[0]))
    x_old = (np.arange(n_old) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    return np.interp(x_new, x_old, h_old, left=float(h_old[0]), right=float(h_old[-1]))


def _coordinate_local_search(h_initial, deadline, seed):
    h_initial = np.asarray(h_initial, dtype=float)
    n = h_initial.size
    if n <= 1:
        h = _project_box_sum(h_initial, 0.5 * n)
        return h, _compute_c5(h)

    target = 0.5 * float(n)
    dx = 2.0 / float(n)

    h = _project_box_sum(h_initial, target)
    best_h = h.copy()
    best_c = _compute_c5(h)

    rng = np.random.RandomState(int(seed) & 0x7fffffff)

    high_idx = np.arange(n)
    low_idx = np.arange(n)

    stall = 0
    it = 0

    while time.time() < deadline:
        it += 1

        if it % 50 == 0:
            g = _smooth_gradient(h, 80.0, dx)
            if not np.all(np.isfinite(g)):
                g = np.zeros(n)
            med = float(np.median(g))
            high_idx = np.where(g > med)[0]
            low_idx = np.where(g < med)[0]
            if high_idx.size == 0:
                high_idx = np.arange(n)
            if low_idx.size == 0:
                low_idx = np.arange(n)

        src = int(rng.choice(high_idx))
        dst = int(rng.choice(low_idx))
        if src == dst:
            continue

        max_eps = min(float(h[src]), 1.0 - float(h[dst]))
        if max_eps < 1e-7:
            continue

        lo_eps = 0.002
        hi_eps = max(0.04, min(0.25, max_eps))
        if lo_eps > max_eps:
            lo_eps = 0.5 * max_eps
        eps = float(rng.uniform(lo_eps, hi_eps))
        if eps > max_eps:
            eps = max_eps
        if eps <= 0.0:
            continue

        h2 = h.copy()
        h2[src] -= eps
        h2[dst] += eps

        c2 = _compute_c5(h2)
        if c2 < best_c - 1e-14:
            best_c = c2
            best_h = h2
            h = h2
            stall = 0
        else:
            stall += 1
            if stall > 1500:
                h = _project_box_sum(best_h + rng.normal(0.0, 0.02, size=n), target)
                c = _compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                else:
                    h = best_h.copy()
                stall = 0

    return best_h, float(best_c)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    np.random.seed(int(seed) & 0x7fffffff)

    budget_s = float(budget_s)
    if budget_s <= 0.0:
        budget_s = 1e-6

    h_best = None
    c_best = np.inf
    n_best = 0

    try:
        h_par = np.asarray(initial_h_values, dtype=float)
    except Exception:
        h_par = None

    if h_par is not None and h_par.size > 0 and np.all(np.isfinite(h_par)):
        hp, cp, np_, ok = _casevaluate(h_par)
        if ok:
            h_best = hp
            c_best = float(cp)
            n_best = np_

    h0, c0, n0, ok0 = _casevaluate(_make_ramp_desc(120))
    if ok0 and c0 < c_best:
        h_best = h0
        c_best = float(c0)
        n_best = n0

    h0, c0, n0, ok0 = _casevaluate(_make_spectral(120, 1009))
    if ok0 and c0 < c_best:
        h_best = h0
        c_best = float(c0)
        n_best = n0

    full_scale = max(0.25, min(1.0, budget_s / 600.0))
    full_deadline = t0 + min(320.0, 0.32 * budget_s)

    if h_best is not None and n_best > 1 and n_best <= 220:
        h_work = h_best.copy()
        for beta, lbfun0, sliter0, rho in (
            (20.0, 300, 80, 25.0),
            (60.0, 250, 100, 60.0),
            (160.0, 160, 80, 100.0),
            (320.0, 100, 60, 160.0)
        ):
            if time.time() >= full_deadline:
                break

            h1 = _optimize_lbfgsb(h_work, beta, rho, int(max(30, lbfun0 * full_scale)), 1e-10)
            hc, cc, nc, ok1 = _casevaluate(h1)
            if ok1:
                if cc < c_best:
                    h_best = hc
                    c_best = float(cc)
                    n_best = nc
                h_work = hc
                if time.time() < full_deadline:
                    h2 = _optimize_slsqp(h_work, beta, int(max(30, sliter0 * full_scale)), 1e-12)
                    hc, cc, nc, ok2 = _casevaluate(h2)
                    if ok2:
                        if cc < c_best:
                            h_best = hc
                            c_best = float(cc)
                            n_best = nc
                        h_work = hc
            time.sleep(0)

    alt_scale = max(0.25, min(1.0, budget_s / 900.0))
    alt_deadline = t0 + min(1080.0, 0.76 * budget_s)

    if h_best is not None:
        n_trials = [80, 95, 100, 120, 140, 170, 210, 260]
        for n_try in n_trials:
            if n_try <= 1:
                continue
            if time.time() >= alt_deadline:
                break

            starts = [_make_ramp_desc(n_try)]

            if n_try <= 170:
                starts.append(_make_ramp_asc(n_try))

            starts.append(_make_spectral(n_try, 10000 + 17 * n_try + int(seed)))

            if n_try <= 150:
                starts.append(_make_beatty(n_try, 20000 + 23 * n_try + int(seed)))
                starts.append(_make_blocks(n_try, 30000 + 29 * n_try + int(seed)))

            local_h = None
            local_c = np.inf

            for st in starts:
                if time.time() >= alt_deadline:
                    break

                h0, c0, n0, ok0 = _casevaluate(st)
                if not ok0:
                    continue

                if c0 < local_c:
                    local_c = float(c0)
                    local_h = h0
                if c0 < c_best:
                    h_best = h0
                    c_best = float(c0)
                    n_best = n0

                lbfun = int(max(50, 260 * alt_scale))
                if n_try > 180:
                    lbfun = int(max(30, 160 * alt_scale))

                h1 = _optimize_lbfgsb(h0, 30.0, 40.0, lbfun, 1e-10)
                hc, cc, nc, ok1 = _casevaluate(h1)
                if ok1:
                    if cc < local_c:
                        local_c = float(cc)
                        local_h = hc
                    if cc < c_best:
                        h_best = hc
                        c_best = float(cc)
                        n_best = nc
                    h0 = hc

            if local_h is not None and local_c < c_best + 0.06 and n_try <= 130:
                if time.time() < alt_deadline - 45.0:
                    sliter = int(max(30, 70 * alt_scale))
                    h2 = _optimize_slsqp(local_h, 80.0, sliter, 1e-11)
                    hc, cc, nc, ok2 = _casevaluate(h2)
                    if ok2:
                        if cc < c_best:
                            h_best = hc
                            c_best = float(cc)
                            n_best = nc
                        local_h = hc
                        local_c = float(cc)

                    if time.time() < alt_deadline - 25.0:
                        sliter2 = int(max(25, 55 * alt_scale))
                        h3 = _optimize_slsqp(local_h, 220.0, sliter2, 1e-12)
                        hc, cc, nc, ok3 = _casevaluate(h3)
                        if ok3 and cc < c_best:
                            h_best = hc
                            c_best = float(cc)
                            n_best = nc

    scale_deadline = t0 + min(1200.0, 0.88 * budget_s)
    if h_best is not None and n_best > 1 and time.time() < scale_deadline:
        for factor in (1.5, 2.0, 0.75):
            if time.time() >= scale_deadline:
                break
            n_new = int(round(float(n_best) * factor))
            if n_new < 40 or n_new > 340:
                continue

            h_new = _interpolate_h(h_best, n_new)
            hc, cc, nc, ok0 = _casevaluate(h_new)
            if not ok0:
                continue
            if cc < c_best:
                h_best = hc
                c_best = float(cc)
                n_best = nc
                h_new = hc

            lbfun = int(max(80, 320 * alt_scale))
            if n_new > 220:
                lbfun = int(max(40, 180 * alt_scale))

            if time.time() < scale_deadline - 35.0:
                h1 = _optimize_lbfgsb(h_new, 80.0, 80.0, lbfun, 1e-11)
                hc, cc, nc, ok1 = _casevaluate(h1)
                if ok1:
                    if cc < c_best:
                        h_best = hc
                        c_best = float(cc)
                        n_best = nc
                    h_new = hc

                    if n_new <= 180 and time.time() < scale_deadline - 30.0:
                        sliter = int(max(35, 80 * alt_scale))
                        h2 = _optimize_slsqp(h_new, 170.0, sliter, 1e-11)
                        hc, cc, nc, ok2 = _casevaluate(h2)
                        if ok2 and cc < c_best:
                            h_best = hc
                            c_best = float(cc)
                            n_best = nc

    coord_deadline = t0 + min(1250.0, 0.94 * budget_s)
    if h_best is not None and n_best > 1 and n_best <= 240 and time.time() < coord_deadline:
        hc, cc = _coordinate_local_search(h_best, coord_deadline, int(seed) + 991)
        if cc < c_best:
            h_best = hc
            c_best = float(cc)
            n_best = int(hc.size)

    final_polish_deadline = t0 + 0.985 * budget_s
    if h_best is not None and n_best > 1 and n_best <= 220 and time.time() < final_polish_deadline - 25.0:
        h1 = _optimize_slsqp(h_best, 320.0, int(max(30, 70 * full_scale)), 1e-12)
        hc, cc, nc, ok1 = _casevaluate(h1)
        if ok1 and cc < c_best:
            h_best = hc
            c_best = float(cc)
            n_best = nc

    if h_best is None or h_best.size == 0:
        h_best, c_best, n_best, ok = _casevaluate(_make_ramp_desc(120))
        if not ok:
            h_best = np.full(100, 0.5)
            c_best = _compute_c5(_project_box_sum(h_best, 50.0))
            n_best = 100

    h_best = _project_box_sum(h_best, 0.5 * float(h_best.size))
    c_best = _compute_c5(h_best)

    return (h_best, float(c_best), int(h_best.size))
```
