def interpolate_h(h_src, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_new = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_interp = np.interp(x_new, xp, h_src)
    return project_feasible(h_interp, n_dst)

# Global variables for objective (to avoid closures)
_g_n = 128
_g_beta = 50.0

def _lse_obj(x):
    dx = 2.0 / _g_n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(_g_beta * (corr - m), -300, 0)
    return (m + np.log(np.exp(shifted).sum()) / _g_beta) * dx

def _con_sum(x):
    return np.sum(x) - _g_n / 2.0

def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def multi_slsqp(h0, n, betas, maxiters, ftol, time_limit):
    """Run SLSQP sequentially with different betas."""
    global _g_n, _g_beta
    _g_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_sum}]
    
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    
    for beta, maxit in zip(betas, maxiters):
        if time.time() > time_limit:
            break
        _g_beta = beta
        try:
            res = minimize(_lse_obj, h_cur, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': maxit, 'ftol': ftol, 'iprint': 0})
            h_try = project_feasible(res.x, n)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur:
                h_cur = h_try
                c5_cur = c5_try
        except Exception:
            pass
    
    return h_cur, c5_cur

def binary_sa(h_init, n, duration, seed):
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    # Round to binary
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_want = n // 2
    if n_ones != n_want:
        if n_ones > n_want:
            ones = np.where(h_bin == 1)[0]
            to_flip = rng.choice(ones, n_ones - n_want, replace=False)
            h_bin[to_flip] = 0.0
        else:
            zeros = np.where(h_bin == 0)[0]
            to_flip = rng.choice(zeros, n_want - n_ones, replace=False)
            h_bin[to_flip] = 1.0
    
    c5_cur = compute_c5(h_bin)
    best_c5 = c5_cur
    best_h = h_bin.copy()
    
    t_end = time.time() + duration
    temp = 0.05
    cool = 0.99999
    
    ones_arr = np.where(h_bin == 1)[0].copy()
    zeros_arr = np.where(h_bin == 0)[0].copy()
    
    for it in range(500000):
        if time.time() > t_end:
            break
        t = temp * (cool ** it)
        if t < 1e-14:
            break
        
        i_pos = rng.randint(len(ones_arr))
        j_pos = rng.randint(len(zeros_arr))
        i = ones_arr[i_pos]
        j = zeros_arr[j_pos]
        
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        ones_arr[i_pos] = j
        zeros_arr[j_pos] = i
        
        c5_new = compute_c5(h_bin)
        d = c5_new - c5_cur
        
        if d < 0 or rng.rand() < np.exp(-d / max(t, 1e-15)):
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            ones_arr[i_pos] = i
            zeros_arr[j_pos] = j
    
    return best_h, best_c5

def greedy_continuous(h0, n, steps=5000):
    """Move one coordinate at a time to reduce max overlap."""
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    
    for step in range(steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the worst lag
        k_star = np.argmax(corr)
        s = k_star - (n - 1)  # shift
        
        best_improve = 0
        best_i = -1
        best_val = 0
        
        for i in range(n):
            orig = h[i]
            for delta in [-0.1, 0.1]:
                new_val = np.clip(orig + delta, 0, 1)
                if new_val == orig:
                    continue
                h[i] = new_val
                # Fix sum
                diff = n/2.0 - np.sum(h)
                # For now just evaluate (approximate)
                corr_new = np.correlate(h, 1.0 - h, mode='full')
                c5_new = np.max(corr_new) * dx
                if c5_new < c5 - 1e-15:
                    improve = c5 - c5_new
                    if improve > best_improve:
                        best_improve = improve
                        best_i = i
                        best_val = new_val
            h[i] = orig
        
        if best_i >= 0:
            h[best_i] = best_val
            h = project_feasible(h, n)
        else:
            break  # No improvement found
    
    return best_h, best_c5
