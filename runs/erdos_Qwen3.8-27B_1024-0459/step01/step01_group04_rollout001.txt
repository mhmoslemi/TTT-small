def time_left_global():
    if _GLOBAL_DEADLINE is None:
        return 1.0e9
    return max(0.0, _GLOBAL_DEADLINE - time.time())


def is_valid_candidate(h):
    h = np.asarray(h)
    if h.ndim != 1 or h.size == 0:
        return False
    if not np.all(np.isfinite(h)):
        return False
    lo = float(h.min())
    hi = float(h.max())
    if lo < -1.0e-7 or hi > 1.0 + 1.0e-7:
        return False
    if abs(float(h.sum()) - 0.5 * float(h.size)) > 1.0e-6:
        return False
    return True


def project_sum(arr, target=None):
    arr = np.asarray(arr, dtype=float)
    if arr.ndim != 1:
        arr = arr.ravel()
    n = int(arr.size)
    if n == 0:
        return arr

    arr = np.nan_to_num(arr, nan=0.5, posinf=1.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1.0)

    if target is None:
        target = 0.5 * float(n)
    target = float(target)

    s = float(arr.sum())
    if abs(s - target) <= 1.0e-12:
        return arr

    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s_mid = float(np.sum(np.clip(arr + mid, 0.0, 1.0)))
        if s_mid < target:
            lo = mid
        else:
            hi = mid

    arr = np.clip(arr + 0.5 * (lo + hi), 0.0, 1.0)

    diff = target - float(arr.sum())
    guard = 0
    while abs(diff) > 1.0e-11 and guard < n + 30:
        guard += 1
        if diff > 0.0:
            mask = arr < 1.0 - 1.0e-12
            if not np.any(mask):
                break
            cnt = int(np.count_nonzero(mask))
            max_inc = float(np.min(1.0 - arr[mask]))
            inc = min(diff / float(cnt), max_inc)
            if inc <= 1.0e-15:
                break
            arr[mask] += inc
        else:
            mask = arr > 1.0e-12
            if not np.any(mask):
                break
            cnt = int(np.count_nonzero(mask))
            max_dec = float(np.min(arr[mask]))
            dec = min(-diff / float(cnt), max_dec)
            if dec <= 1.0e-15:
                break
            arr[mask] -= dec
        diff = target - float(arr.sum())

    diff = target - float(arr.sum())
    if abs(diff) > 1.0e-13:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - arr))
            arr[idx] = min(1.0, arr[idx] + diff)
        else:
            idx = int(np.argmax(arr))
            arr[idx] = max(0.0, arr[idx] + diff)

    return arr

def score_array(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n == 0:
        return np.zeros(0, dtype=float)
    h = np.clip(h, 0.0, 1.0)

    total = float(np.sum(h))
    hh = float(np.dot(h, h))
    scores = np.empty(2 * n - 1, dtype=float)
    scores[0] = total - hh

    if n > 1:
        ac = np.correlate(h, h, mode="full")
        pref = np.empty(n + 1, dtype=float)
        pref[0] = 0.0

    return scores

    if m <= 0 or w == 0.0:
        return
    d = int(d)

    if m <= 0 or w == 0.0:
        return
    d = int(d)
    grad[:m] += -w * h[d:]

        h = project_sum(h, n)
    if not is_valid_candidate(h):
        return state
    if c is None:
        c = compute_c5(h)
    if not math.isfinite(c):
        return state
    if c < state["c"] - 1.0e-15:
        state["h"] = h
        state["c"] = float(c)
        state["n"] = n
    return state


def get_h_at_n(state, n):
    n = int(n)
    if state is None:
        return project_sum(np.full(n, 0.5, dtype=float), n)
    h = np.asarray(state["h"], dtype=float)
    if h.size == n:
        return h.copy()
    return project_sum(upsample_linear(h, n), n)


def smooth_direction(n, rng, style):
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=float)
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    out = np.zeros(n, dtype=float)

    style = int(style) % 3
    if style == 0:
        maxf = max(1, min(10, n // 4))
        for f in range(1, maxf + 1):
            amp = float(rng.normal(0.0, 1.0)) / math.sqrt(float(f + 1))
            ph = float(rng.uniform(0.0, 2.0 * math.pi))
            out += amp * np.sin(math.pi * float(f) * x + ph)
            amp2 = float(rng.normal(0.0, 1.0)) / math.sqrt(float(f + 1))
            ph2 = float(rng.uniform(0.0, 2.0 * math.pi))
            out += amp2 * np.cos(math.pi * float(f) * x + ph2)
    elif style == 1:
        raw = rng.normal(0.0, 1.0, n).astype(float)
        out = 0.0
        out += 0.10 * np.roll(raw, 2)
        out += 0.25 * np.roll(raw, 1)
        out += 0.30 * raw
        out += 0.25 * np.roll(raw, -1)
        out += 0.10 * np.roll(raw, -2)
    else:
        seg = int(rng.integers(4, max(5, min(18, n // 3))))
        vals = rng.uniform(0.0, 1.0, seg).astype(float)
        idx = np.floor(np.arange(n) * float(seg) / float(n)).astype(int)
        idx = np.clip(idx, 0, seg - 1)
        out = vals[idx]

    out -= float(np.mean(out))
    s = float(np.std(out))
    if s < 1.0e-12:
        out = rng.normal(0.0, 1.0, n).astype(float)
        out -= float(np.mean(out))
        s = float(np.std(out))
        if s < 1.0e-12:
            return np.zeros(n, dtype=float)
    out = out / s
    return out


def candidate_duplicate(cands, h):
    h = np.asarray(h, dtype=float)
    for ex, cc in cands:
        if ex.size == h.size:
            if np.allclose(ex, h, rtol=1.0e-7, atol=1.0e-7):
                return True
    return False


def add_to_cands(cands, h):
    h = np.asarray(h, dtype=float)
    if h.ndim != 1 or h.size == 0:
        return
    n = int(h.size)
    h = project_sum(h, n)
    if not is_valid_candidate(h):
        return
    if candidate_duplicate(cands, h):
        return
    c = compute_c5(h)
    if not math.isfinite(c):
        return
    cands.append((h, float(c)))


def random_start_list(n, parent, state, rng, count):
    n = int(n)
    count = int(count)
    if n <= 0 or count <= 0:
        return []

    if parent is not None:
        base = project_sum(upsample_linear(parent, n), n)
    elif state is not None:
        base = project_sum(get_h_at_n(state, n), n)
    else:
        base = project_sum(np.full(n, 0.5, dtype=float), n)

    cands = []

    amplitudes = [0.04, 0.08, 0.14, 0.22]
    for style in (0, 1, 2):
        for amp in amplitudes:
            d = smooth_direction(n, rng, style)
            add_to_cands(cands, base + amp * d)

    shift = int(rng.integers(1, max(2, n // 4)))
    add_to_cands(cands, np.roll(base, shift))

    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    for f in (1, 2, 3):
        amp = 0.18 + 0.32 * float(rng.random())
        ph = float(rng.uniform(0.0, 2.0 * math.pi))
        add_to_cands(cands, 0.5 + amp * np.sin(math.pi * float(f) * x + ph))

        amp2 = 0.10 + 0.22 * float(rng.random())
        ph2 = float(rng.uniform(0.0, 2.0 * math.pi))
        add_to_cands(cands, 0.5 + amp * np.cos(math.pi * float(f) * x + ph) + amp2 * np.sin(math.pi * float(f + 1) * x + ph2))

    keyed = []
    for h, cc in cands:
        keyed.append((cc, h))
    keyed.sort()

    out = []
    for cc, h in keyed:
        out.append(h)
        if len(out) >= count:
            break
    return out


def make_candidates(n, parent, state, rng, limit):
    n = int(n)
    limit = max(1, int(limit))
    cands = []

    if parent is not None:
        add_to_cands(cands, upsample_linear(parent, n))

    if state is not None:
        add_to_cands(cands, get_h_at_n(state, n))

    rnds = random_start_list(n, parent, state, rng, max(4, limit + 2))
    for h in rnds:
        add_to_cands(cands, h)

    if not cands:
        add_to_cands(cands, np.full(n, 0.5, dtype=float))

    keyed = []
    for h, cc in cands:
        keyed.append((cc, h))
    keyed.sort()

    out = []
    for cc, h in keyed:
        out.append((h, cc))
        if len(out) >= limit:
            break
    return out


def slsqp_maxiter(n):
    n = int(n)
    if n <= 60:
        return 90
    if n <= 100:
        return 65
    if n <= 160:
        return 45
    if n <= 240:
        return 32
    return 24


def lbfgs_maxiter(n):
    n = int(n)
    if n <= 80:
        return 180
    if n <= 140:
        return 120
    if n <= 220:
        return 80
    return 55


def run_slsqp(h, n, limit, alpha, state):
    global _DEADLINE

    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=float)
    h = project_sum(h, n)
    c = compute_c5(h)
    state_update(state, h, c, n)

    if limit <= 5.0:
        return h

    t_end = time.time() + float(limit)
    if _GLOBAL_DEADLINE is not None:
        t_end = min(t_end, _GLOBAL_DEADLINE)

    if state is not None and state["n"] == n and state["c"] < c:
        h = state["h"].copy()
        c = state["c"]

    current_c = c

    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": eq_fun, "jac": eq_jac, "args": (n,)}]
    mini_iter = slsqp_maxiter(n)
    no_improve = 0

    while time.time() < t_end - 3.0:
        stage_time = min(35.0, t_end - time.time())
        if stage_time < 5.0:
            break

        old_deadline = _DEADLINE
        stage_deadline = time.time() + stage_time
        if old_deadline is not None:
            stage_deadline = min(stage_deadline, old_deadline)
        _DEADLINE = stage_deadline

        res = None
        try:
            res = minimize(
                slsqp_obj,
                h,
                args=(float(alpha),),
                jac=slsqp_grad,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": mini_iter, "ftol": 1.0e-9, "disp": 0},
            )
        except BudgetError:
            res = None
        except Exception:
            res = None
        finally:
            _DEADLINE = old_deadline

        if res is not None and getattr(res, "x", None) is not None:
            h2 = project_sum(res.x, n)
            c2 = compute_c5(h2)
        else:
            h2 = h
            c2 = current_c

        improved = c2 < current_c - 1.0e-14
        if improved:
            h = h2
            current_c = c2
            no_improve = 0
            state_update(state, h, c2, n)
        else:
            no_improve += 1
            if state is not None and state["n"] == n and state["c"] < current_c - 1.0e-15:
                h = state["h"].copy()
                current_c = state["c"]
                no_improve = 0

            success = bool(getattr(res, "success", False))
            if success and no_improve >= 2:
                break
            if no_improve >= 5:
                break

    if state is not None and state["n"] == n and state["c"] < compute_c5(h):
        h = state["h"].copy()
    return h


def run_lbfgs(h, n, limit, state):
    global _DEADLINE

    n = int(n)
    if n <= 0 or limit <= 8.0:
        return project_sum(h, n)

    h = project_sum(h, n)
    c = compute_c5(h)
    state_update(state, h, c, n)

    if state is not None and state["n"] == n and state["c"] < c:
        h = state["h"].copy()
        c = state["c"]

    t_end = time.time() + float(limit)
    if _GLOBAL_DEADLINE is not None:
        t_end = min(t_end, _GLOBAL_DEADLINE)

    bounds = [(0.0, 1.0)] * n
    stages = (
        (max(1.0, 0.10 * float(n)), 30.0),
        (max(0.2, 0.025 * float(n)), 250.0),
        (max(0.02, 0.005 * float(n)), 1200.0),
    )
    stage_time = max(5.0, float(limit) / float(len(stages)))

    for j in range(len(stages)):
        if time.time() >= t_end - 4.0:
            break
        alpha, pen = stages[j]
        old_deadline = _DEADLINE
        stage_deadline = min(t_end, time.time() + stage_time)
        if old_deadline is not None:
            stage_deadline = min(stage_deadline, old_deadline)
        _DEADLINE = stage_deadline

        res = None
        try:
            res = minimize(
                penalty_obj,
                h,
                args=(n, float(alpha), float(pen)),
                jac=penalty_grad,
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": lbfgs_maxiter(n),
                    "ftol": 1.0e-12,
                    "gtol": 1.0e-8,
                    "maxls": 20,
                    "disp": 0,
                },
            )
        except BudgetError:
            res = None
        except Exception:
            res = None
        finally:
            _DEADLINE = old_deadline

        if res is not None and getattr(res, "x", None) is not None:
            h2 = project_sum(res.x, n)
            c2 = compute_c5(h2)
            if math.isfinite(c2):
                state_update(state, h2, c2, n)
                if c2 <= c + 1.0e-4:
                    h = h2
                    c = min(c, c2)

    return h


def refine_candidate(h, n, limit, state, rng):
    n = int(n)
    if n <= 0 or limit <= 5.0:
        return

    h = project_sum(h, n)
    c = compute_c5(h)
    state_update(state, h, c, n)
    baseline_c = state["c"]

    t_end = time.time() + float(limit)
    if _GLOBAL_DEADLINE is not None:
        t_end = min(t_end, _GLOBAL_DEADLINE)

    rem = t_end - time.time()
    if rem > 70.0:
        lb_time = min(0.32 * float(rem), 85.0)
        hlb = run_lbfgs(h, n, lb_time, state)
        c_lb = compute_c5(hlb)
        if state is not None and state["n"] == n and state["c"] <= c_lb + 1.0e-5:
            h_soft = get_h_at_n(state, n)
        else:
            h_soft = hlb
    else:
        h_soft = get_h_at_n(state, n)

    rem = t_end - time.time()
    if rem > 35.0:
        sl_time = min(0.30 * float(rem), 80.0)
        soft_alpha = max(0.5, 0.01 * float(n))
        h_soft = run_slsqp(h_soft, n, sl_time, soft_alpha, state)

    rem = t_end - time.time()
    if rem > 18.0:
        ex_time = max(0.0, min(rem - 2.0, 260.0))
        h_exact = run_slsqp(get_h_at_n(state, n), n, ex_time, 0.0, state)

    rem = t_end - time.time()
    if rem > 35.0 and state is not None and state["c"] >= baseline_c - 1.0e-10:
        amps = (0.02, 0.045, 0.075)
        for amp in amps:
            if time.time() >= t_end - 25.0:
                break
            base = get_h_at_n(state, n)
            d = smooth_direction(n, rng, int(rng.integers(0, 3)))
            cand = project_sum(base + float(amp) * d, n)
            cc = compute_c5(cand)
            state_update(state, cand, cc, n)
            tt = max(10.0, min(35.0, (t_end - time.time()) * 0.32))
            run_slsqp(cand, n, tt, 0.0, state)
            if state is not None and state["c"] < baseline_c - 1.0e-12:
                break


def run(seed=42, budget_s=1300, **kwargs):
    global _DEADLINE, _GLOBAL_DEADLINE

    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0.0:
        budget_s = 30.0

    reserve = 30.0 if budget_s > 120.0 else max(3.0, 0.15 * budget_s)
    global_deadline = t0 + max(1.0, budget_s - reserve)
    _GLOBAL_DEADLINE = global_deadline
    _DEADLINE = global_deadline

    rng = np.random.default_rng(seed)

    parent = get_parent_array()
    if parent is not None:
        n0 = int(parent.size)
        h_raw = parent.copy()
        if is_valid_candidate(h_raw):
            h0 = h_raw
            c0 = compute_c5(h0)
        else:
            h0 = project_sum(h_raw, n0)
            c0 = compute_c5(h0)
    else:
        n0 = 70
        lst = random_start_list(n0, None, None, rng, 12)
        if lst:
            h0 = lst[0]
        else:
            h0 = project_sum(np.full(n0, 0.5, dtype=float), n0)
        c0 = compute_c5(h0)

    state = make_initial_state(h0, c0, n0)

    if not is_valid_candidate(state["h"]):
        h0 = project_sum(state["h"], int(state["n"]))
        c0 = compute_c5(h0)
        state = make_initial_state(h0, c0, int(state["n"]))

    base_n = max(8, int(state["n"]))
    max_n = int(max(140, min(300, 3.4 * float(base_n))))
    mults = (1.0, 1.12, 1.30, 1.55, 1.90, 2.25, 2.65)
    ns_raw = []
    for mm in mults:
        nn = int(round(float(base_n) * float(mm) / 2.0) * 2)
        if nn < 12:
            nn = 12
        if nn > max_n:
            continue
        if nn not in ns_raw:
            ns_raw.append(nn)

    nn2 = 2 * base_n
    if nn2 <= max_n and nn2 not in ns_raw:
        ns_raw.append(nn2)

    if base_n not in ns_raw:
        ns_raw.insert(0, base_n)

    keyed = []
    for nn in ns_raw:
        keyed.append((abs(nn - base_n), nn, nn))
    keyed.sort()
    ns = []
    for _, _, nn in keyed:
        if nn not in ns:
            ns.append(nn)

    total_opt = max(60.0, global_deadline - t0)
    fracs = (0.34, 0.21, 0.15, 0.11, 0.09, 0.06, 0.04)

    for idx, nn in enumerate(ns):
        rem_global = time_left_global()
        if rem_global < 75.0:
            break
        frac = fracs[idx] if idx < len(fracs) else 0.03
        alloc = frac * total_opt
        alloc = min(alloc, rem_global - 35.0)
        if alloc < 45.0:
            break

        cand_limit = 3 if alloc > 210.0 else 2
        cands = make_candidates(int(nn), parent, state, rng, cand_limit)
        if not cands:
            continue

        rem_alloc = float(alloc)
        for ci in range(len(cands)):
            if time_left_global() < 45.0:
                break
            if rem_alloc <= 1.0:
                break
            cand_time = rem_alloc / float(len(cands) - ci)
            cand_time = min(cand_time, time_left_global() - 25.0)
            if cand_time < 28.0:
                break

            h, _ = cands[ci]
            ts = time.time()
            refine_candidate(h, int(nn), cand_time, state, rng)
            used = min(float(cand_time), max(0.0, time.time() - ts))
            rem_alloc -= used

    rem = time_left_global()
    if rem > 35.0:
        hbest = state["h"].copy()
        nb = int(state["n"])
        project_sum(hbest, nb)
        run_slsqp(hbest, nb, min(170.0, rem - 12.0), 0.0, state)

    h_final = state["h"].copy()
    n_final = int(state["n"])
    h_final = tighten_sum(h_final, n_final)
    c_final = compute_c5(h_final)

    if not is_valid_candidate(h_final) or not math.isfinite(c_final):
        h_final = project_sum(state["h"], n_final)
        c_final = compute_c5(h_final)

    if state is not None and state["h"].size == n_final:
        c_state = compute_c5(tighten_sum(state["h"].copy(), n_final))
        if math.isfinite(c_state) and c_state < c_final:
            h_final = tighten_sum(state["h"].copy(), n_final)
            c_final = c_state

    if not math.isfinite(c_final):
        n_final = max(12, int(n_final))
        h_final = project_sum(np.full(n_final, 0.5, dtype=float), n_final)
        c_final = compute_c5(h_final)

    return h_final, float(c_final), int(n_final)
