Let me also add one more trick: **line search refinement**. After SLSQP converges, I can try small perturbations along specific directions (e.g., the gradient of the max overlap) to squeeze out a tiny bit more.

Actually, for the "exact max" gradient (alpha=0), the subgradient is not unique when multiple shifts achieve the max. SLSQP handles this by using the average subgradient over the active set. But we can also try: for each shift in the active set, compute the gradient and do a line search in that direction, keeping the best.

import numpy as np
import time
import math
from scipy.optimize import minimize

_RUN_DEADLINE = None

class BudgetError(Exception):
    pass

def check_deadline():
    if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE:
        raise BudgetError

def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 1e6
    h = np.clip(h, 0.0, 1.0)
    conv = np.correlate(h, 1.0 - h, mode="full")
    c = float(np.max(conv) * (2.0 / n))
    if not math.isfinite(c):
        return 1e6
    return c

def as_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    if h.size != n:
        if h.size > n:
            h = h[:n]
        else:
            tmp = np.full(n, 0.5, dtype=float)
            m = min(n, h.size)
            if m > 0:
                tmp[:m] = h[:m]
            h = tmp
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    s = float(h.sum())
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s_mid = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s_mid < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-10
            if np.any(mask):
                h[mask] += diff / float(np.count_nonzero(mask))
        else:
            mask = h > 1e-10
            if np.any(mask):
                h[mask] += diff / float(np.count_nonzero(mask))
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h

def upsample_linear(parent, n):
    n0 = int(parent.size)
    if n == n0:
        return parent.copy()
    if n0 <= 0:
        return np.full(n, 0.5)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x, x0, parent.astype(float))

def score_vector(h, n):
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, h, mode="full")
    ac = corr[n-1:n-1+n]
    prefix = np.empty(n+1, dtype=float)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])
    total = prefix[n]
    scores = np.empty(2*n-1, dtype=float)
    scores[0] = total - ac[0]
    for d in range(1, n):
        scores[2*d-1] = prefix[n-d] - ac[d]
        scores[2*d] = total - prefix[d] - ac[d]
    return scores

def obj_func(x, alpha):
    check_deadline()
    n = x.size
    h = np.clip(x, 0.0, 1.0)
    scores = score_vector(h, n)
    m = float(np.max(scores))
    if alpha > 1e-14:
        shifted = np.clip(scores - m, -60.0, 0.0)
        w = np.exp(shifted / alpha)
        A = float(np.sum(w))
        if not math.isfinite(A) or A <= 0:
            val = m
        else:
            val = m + alpha * math.log(A)
    else:
        val = m
    if not math.isfinite(val):
        return 1e6
    return float(val) * (2.0 / n)

def grad_func(x, alpha):
    check_deadline()
    n = x.size
    h = np.clip(x, 0.0, 1.0)
    scores = score_vector(h, n)
    m = float(np.max(scores))
    grad = np.zeros(n, dtype=float)
    if alpha <= 1e-14:
        tol = max(1e-7, 1e-7 * max(1.0, m))
        active = np.flatnonzero(scores >= m - tol)
        if len(active) == 0:
            active = np.array([int(np.argmax(scores))])
        n_act = len(active)
        for i in active:
            coef = 1.0 / n_act
            if i == 0:
                grad += coef * (1.0 - 2.0 * h)
            else:
                if i % 2 == 1:
                    d = (i + 1) // 2
                else:
                    d = i // 2
                m_len = n - d
                if m_len <= 0:
                    continue
                if i % 2 == 1:
                    grad[:m_len] += coef * (1.0 - h[d:])
                    grad[d:] += coef * (-h[:m_len])
                else:
                    grad[:m_len] += coef * (-h[d:])
                    grad[d:] += coef * (1.0 - h[:m_len])
    else:
        shifted = np.clip(scores - m, -60.0, 0.0)
        w = np.exp(shifted / alpha)
        A = float(np.sum(w))
        if not math.isfinite(A) or A <= 0:
            A = 1.0
        w = w / A
        for i in range(len(w)):
            coef = w[i]
            if coef < 1e-10:
                continue
            if i == 0:
                grad += coef * (1.0 - 2.0 * h)
            else:
                if i % 2 == 1:
                    d = (i + 1) // 2
                else:
                    d = i // 2
                m_len = n - d
                if m_len <= 0:
                    continue
                if i % 2 == 1:
                    grad[:m_len] += coef * (1.0 - h[d:])
                    grad[d:] += coef * (-h[:m_len])
                else:
                    grad[:m_len] += coef * (-h[d:])
                    grad[d:] += coef * (1.0 - h[:m_len])
    grad *= 2.0 / n
    return grad

def constraint_eq(x, *args):
    n = max(1, int(x.size))
    return (float(np.sum(x)) - 0.5 * float(n)) / float(n)

def constraint_jac(x, *args):
    n = max(1, int(x.size))
    return np.full((1, n), 1.0 / float(n), dtype=float)

def run_slsqp_stage(h, n, alpha, maxiter, time_limit):
    start_t = time.time()
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": constraint_eq, "jac": constraint_jac}]
    stage_deadline = start_t + time_limit
    if _RUN_DEADLINE is not None:
        stage_deadline = min(stage_deadline, _RUN_DEADLINE)
    old_dl = _RUN_DEADLINE
    _RUN_DEADLINE = stage_deadline
    try:
        res = minimize(
            obj_func, h, args=(alpha,),
            jac=grad_func, method="SLSQP",
            bounds=bounds, constraints=constraints,
            options={"maxiter": maxiter, "ftol": 1e-10, "disp": 0}
        )
    except BudgetError:
        _RUN_DEADLINE = old_dl
        return h
    except Exception:
        _RUN_DEADLINE = old_dl
        return h
    final_dl = _RUN_DEADLINE
    _RUN_DEADLINE = old_dl
    if res is not None and getattr(res, "x", None) is not None:
        return as_feasible(res.x, n)
    return h

def multi_stage_optimize(h, n, state, time_limit):
    stages = [
        (1.0, 40, 0.15),
        (0.3, 60, 0.20),
        (0.05, 80, 0.25),
        (0.005, 100, 0.25),
        (0.0, 150, 0.15),
    ]
    h = as_feasible(h, n)
    c = compute_c5(h)
    state.h = h.copy()
    state.c = c
    state.n = n
    elapsed = 0.0
    for alpha, iters, frac in stages:
        remaining = time_limit - elapsed
        if remaining < 10.0:
            break
        if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE - 5:
            break
        tl = min(frac * time_limit, remaining * 0.6)
        if tl < 5.0:
            break
        t_start = time.time()
        h = run_slsqp_stage(h, n, alpha, iters, tl)
        elapsed += time.time() - t_start
        c = compute_c5(h)
        state.h = h.copy()
        state.c = float(c)
        state.n = n
    return h, float(compute_c5(h))

class OptimizerState:
    def __init__(self):
        self.h = None
        self.c = 1e9
        self.n = 0

    def try_update(self, h, n):
        if h is None:
            return False
        h = np.asarray(h, dtype=float)
        if h.ndim != 1 or h.size == 0:
            return False
        n = int(n)
        if h.size != n:
            return False
        try:
            h2 = as_feasible(h, n)
            c2 = compute_c5(h2)
        except Exception:
            return False
        if math.isfinite(c2) and c2 < self.c - 1e-14:
            self.h = h2
            self.c = float(c2)
            self.n = n
            return True
        return False

def make_binary(h, n):
    target = n // 2
    hb = np.zeros(n, dtype=np.int8)
    order = np.argsort(-h)
    hb[order[:target]] = 1
    return hb

def binary_scores_fast(hb, n):
    scores = np.zeros(2*n-1, dtype=float)
    hf = hb.astype(float)
    corr = np.correlate(hf, 1.0-hf, mode="full")
    scores[:] = corr
    return scores

def binary_local_search(hb, n, state, deadline, max_ops):
    hb = hb.astype(np.int8).copy()
    scores = binary_scores_fast(hb, n)
    cur = float(np.max(scores))
    best = cur
    best_hb = hb.copy()
    dx = 2.0 / n
    state.try_update(hb.astype(float), n)

    stall = 0
    ops = 0
    while ops < max_ops:
        if time.time() > deadline:
            break
        if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE:
            break
        ops += 1
        if ops % 500 == 0:
            check_deadline()

        active = np.flatnonzero(scores >= cur - 1e-9)
        if len(active) == 0:
            active = np.array([int(np.argmax(scores))])
        idx = int(active[int(np.random.default_rng(ops).integers(len(active)))])
        if idx == 0:
            continue
        if idx % 2 == 1:
            d = (idx + 1) // 2
        else:
            d = idx // 2
        m_len = n - d
        if m_len <= 0:
            continue
        if idx % 2 == 1:
            mask = (hb[:m_len] == 1) & (hb[d:] == 0)
        else:
            mask = (hb[:m_len] == 0) & (hb[d:] == 1)
        cand = np.nonzero(mask)[0]
        if len(cand) == 0:
            zeros = np.flatnonzero(hb == 0)
            ones = np.flatnonzero(hb == 1)
            if len(zeros) == 0 or len(ones) == 0:
                stall += 10
                continue
            i = int(zeros[int(np.random.default_rng(ops).integers(len(zeros)))])
            j = int(ones[int(np.random.default_rng(ops).integers(len(ones)))])
        else:
            pos = int(cand[int(np.random.default_rng(ops).integers(len(cand)))])
            i = pos
            j = pos + d

        hi = int(hb[i])
        hj = int(hb[j])
        if hi == hj:
            stall += 1
            continue

        # Apply flip
        old_scores_snapshot = scores.copy()
        hb[i] = 1 - hi
        hb[j] = 1 - hj
        new_scores = binary_scores_fast(hb, n)
        new_cur = float(np.max(new_scores))

        if new_cur < cur - 1e-12:
            scores = new_scores
            cur = new_cur
            stall = 0
            if cur < best:
                best = cur
                best_hb = hb.copy()
                state.try_update(hb.astype(float), n)
        else:
            # Revert
            scores = old_scores_snapshot
            hb[i] = hi
            hb[j] = hj
            if stall > 3000 and np.random.default_rng(ops).random() < 0.05:
                # Accept with probability (SA-like)
                if np.random.default_rng(ops).random() < 0.3:
                    scores = new_scores
                    cur = new_cur
            stall += 1

        if stall > 5000:
            break

    state.try_update(best_hb.astype(float), n)
    return best_hb.astype(float), float(best * dx)

    return h_final, float(c_final), int(n_final)
