def project_density(x, target):

    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)

    target=float(target)
    if n==0: return x

       mid=0.5*(lo+hi)
       s=float(np.sum(np.clip(x+mid,0,1)))
       if s < target: lo=mid
       else: hi=mid
    h=np.clip(x+0.5*(lo+hi),0,1)
    # exact fix using bisection residual

       if abs(res) <= 1e-14*max(1,n): break

          mask = h < 1.0 - 1e-15
          if not np.any(mask): break
          # distribute among all mask? Use equal add min
          add = res / float(np.sum(mask))
          # if some would exceed 1, do iterative cap

          h[mask] = np.clip(h[mask] + add, 0,1)

          h[mask] = np.clip(h[mask] + add, 0,1)

def fix_sum(h, target):
    h = np.array(h, dtype=float, copy=True)
    n = h.size
    target = float(target)
    if n == 0:
        return h

    for _ in range(40):
        res = target - float(np.sum(h))
        if abs(res) <= 1e-14 * max(1, n):
            return h
        if res > 0.0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            add = res / float(cnt)
            if add > 0.0:
                h[mask] = np.minimum(1.0, h[mask] + add)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            add = res / float(cnt)

                h[mask] = np.maximum(0.0, h[mask] + add)

    res = target - float(np.sum(h))

            order = np.argsort(h)
            for idx in order:

                if slack > 1e-15:

                        res -= amt
        else:
            order = np.argsort(-h)

                idx = int(idx)

                room = float(h[idx])

        return x

    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid

    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    return fix_sum(h, target)


def is_feasible(h, target, tol=1e-9):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    if abs(float(np.sum(h)) - float(target)) > max(1e-8, 1e-12 * max(1, h.size)):
        return False
    return True


def vector_close(a, b):
    if a.size != b.size:
        return False
    return bool(np.allclose(a, b, rtol=0.0, atol=1e-7))


def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    n = int(n)
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5, dtype=float), 0.5 * n)
    if n0 == n:
        return project_density(p, 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)


def make_candidate(n, rng, mode):
    n = int(n)
    if n <= 1:
        return project_density(np.array([0.5], dtype=float), 0.5 * n)
    x = (np.arange(n) + 0.5) * (2.0 / float(n))

    mode = int(mode) % 4
    if mode == 0:
        h = np.full(n, 0.5, dtype=float)
        for m in range(1, 6):
            amp = float(rng.uniform(0.08, 0.38)) * (0.78 ** (m - 1))
            freq = int(max(1, np.rint(max(1, m) * float(rng.uniform(0.7, 1.5)))))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            h += amp * np.sin(2.0 * np.pi * float(freq) * x / 2.0 + phase)
    elif mode == 1:
        h = np.full(n, 0.5, dtype=float)
        for m in range(1, 7):
            amp = float(rng.uniform(0.05, 0.40)) * (0.72 ** (m - 1))
            freq = int(max(1, np.rint(max(1, m) * float(rng.uniform(0.9, 1.7)))))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            h += amp * np.sin(2.0 * np.pi * float(freq) * x / 1.0 + phase)
    elif mode == 2:
        h = np.empty(n, dtype=float)
        levels = np.array([0.0, 0.25, 0.5, 0.75, 1.0], dtype=float)
        pos = 0
        while pos < n:
            rem = n - pos
            length = int(rng.integers(1, rem + 1))
            val = float(rng.choice(levels))
            h[pos:pos + length] = val
            pos += length
    else:
        u = rng.random(n)
        thresh = float(np.median(u))
        h = (u > thresh).astype(float)

    return project_density(h, 0.5 * n)


def read_parent():
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            raw = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0)
            raw = np.clip(raw, 0.0, 1.0)
            return raw.copy()
    except Exception:
        pass
    return None


def top_counts_indices(corr, m):
    corr = np.asarray(corr, dtype=float)
    m = int(m)
    if corr.size == 0 or m <= 0:
        return np.array([], dtype=int)
    if m >= corr.size:
        return np.arange(corr.size, dtype=int)
    k = corr.size - m
    idx = np.argpartition(corr, k)[k:]
    idx = idx[np.argsort(corr[idx])[::-1]]
    return idx.astype(int, copy=False)


def lag_gradient(h, idx):
    h = np.asarray(h, dtype=float)
    n = h.size
    g = np.zeros(n, dtype=float)
    if n <= 0:
        return g
    idx = int(idx)
    center = n - 1
    if idx < 0 or idx > 2 * n - 2:
        return g

    lag = idx - center
    if lag == 0:
        return 1.0 - 2.0 * h

    if lag > 0:
        m = n - lag
        g[:m] += 1.0 - h[lag:]
        g[lag:] -= h[:m]
    else:
        p = -lag
        m = n - p
        g[p:] += 1.0 - h[:m]
        g[:m] -= h[p:]
    return g


def add_var_grad(g, h, v, sign):
    n = h.size
    if n <= 0:
        return
    center = n - 1
    v = int(v)

    ist = center - v
    ien = center + (n - 1 - v)
    if ist <= ien:
        lags = np.arange(ist, ien + 1, dtype=int)
        partner = v + (lags - center)
        g[ist:ien + 1] += sign * (1.0 - h[partner])

    ist2 = center + v - (n - 1)
    ien2 = center + v
    if ist2 <= ien2:
        lags2 = np.arange(ist2, ien2 + 1, dtype=int)
        partner2 = v + center - lags2
        g[ist2:ien2 + 1] += sign * (-h[partner2])


def pair_lin_quad(h, i, j):
    n = h.size
    m = 2 * n - 1
    center = n - 1
    lin = np.zeros(m, dtype=float)
    quad = np.zeros(m, dtype=float)
    add_var_grad(lin, h, int(i), 1.0)
    add_var_grad(lin, h, int(j), -1.0)
    quad[center] = -2.0
    p = abs(int(i) - int(j))
    if p > 0:
        quad[center - p] = 1.0
        quad[center + p] = 1.0
    return lin, quad


def pair_line_search(base_c, lin, quad, cap, n_alpha, local_alpha):
    base_c = np.asarray(base_c, dtype=float)
    lin = np.asarray(lin, dtype=float)
    quad = np.asarray(quad, dtype=float)
    cap = float(cap)
    if base_c.size != lin.size or base_c.size != quad.size:
        return 0.0, float(np.max(base_c)) if base_c.size else 0.0
    if cap <= 1e-12:
        return 0.0, float(np.max(base_c))

    a = np.linspace(0.0, cap, max(2, int(n_alpha)))
    center = base_c.size // 2
    if quad[center] != 0.0:
        v = -lin[center] / (2.0 * quad[center])
        if 0.0 < v < cap:
            a = np.concatenate((a, np.array([v], dtype=float)))
    a = np.concatenate((a, np.array([cap], dtype=float)))
    a = np.unique(a)

    best_a = 0.0
    best_val = float(np.max(base_c))
    chunk = 32

    for s in range(0, a.size, chunk):
        an = a[s:s + chunk]
        aa = an[:, None]
        cnt = base_c[None, :] + lin[None, :] * aa + quad[None, :] * (aa * aa)
        vals = cnt.max(axis=1)
        ii = int(vals.argmin())
        if vals[ii] < best_val:
            best_val = float(vals[ii])
            best_a = float(an[ii])

    radius = max(cap / float(max(1, int(n_alpha))), 1e-6) * 0.5
    lo = max(0.0, best_a - radius)
    hi = min(cap, best_a + radius)
    if hi > lo:
        a2 = np.linspace(lo, hi, max(2, int(local_alpha)))
        for s in range(0, a2.size, chunk):
            an = a2[s:s + chunk]
            aa = an[:, None]
            cnt = base_c[None, :] + lin[None, :] * aa + quad[None, :] * (aa * aa)
            vals = cnt.max(axis=1)
            ii = int(vals.argmin())
            if vals[ii] < best_val:
                best_val = float(vals[ii])
                best_a = float(an[ii])

    return best_a, best_val


def smooth_obj(x):
    n = int(_SMOOTH_N)
    if n <= 1:
        return 0.0
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return 1e100
    corr = corr_counts(h)
    if corr.size == 0:
        return 1e100
    m = float(np.max(corr))
    y = _SMOOTH_ALPHA * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return 1e100
    return float(_SMOOTH_DX * (m + np.log(z) / _SMOOTH_ALPHA))


def smooth_obj_jac(x):
    n = int(_SMOOTH_N)
    if n <= 1:
        return np.zeros(max(1, n), dtype=float)
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return np.zeros(n, dtype=float)
    corr = corr_counts(h)
    if corr.size == 0:
        return np.zeros(n, dtype=float)
    m = float(np.max(corr))
    y = _SMOOTH_ALPHA * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return np.zeros(n, dtype=float)

    w = e / z
    idxs = np.where(w > 1e-12)[0]
    if idxs.size == 0:
        idxs = np.array([int(np.argmax(w))], dtype=int)
    elif idxs.size > 80:
        k = idxs.size - 80
        idxs = np.argpartition(w[idxs], k)[k:]
        idxs = idxs[np.argsort(w[idxs])[::-1]]

    g = np.zeros(n, dtype=float)
    for idx in idxs:
        idx = int(idx)
        p = float(w[idx])
        if p <= 0.0:
            continue
        g += p * lag_gradient(h, idx)
    return _SMOOTH_DX * g


def smooth_sum_fun(x):
    return float(np.sum(np.asarray(x, dtype=float)) - _SMOOTH_TARGET)


def smooth_sum_jac(x):
    return np.ones_like(np.asarray(x, dtype=float), dtype=float)


def smooth_slsqp_step(h, alpha, maxiter):
    global _SMOOTH_N, _SMOOTH_ALPHA, _SMOOTH_DX, _SMOOTH_TARGET
    n = int(h.size)
    target = 0.5 * float(n)
    h = project_density(h, target)
    if n <= 1:
        return h

    _SMOOTH_N = n
    _SMOOTH_ALPHA = float(alpha)
    _SMOOTH_DX = 2.0 / float(n)
    _SMOOTH_TARGET = target

    bounds = [(0.0, 1.0)] * n
    constraints = (
        {"type": "eq", "fun": smooth_sum_fun, "jac": smooth_sum_jac},
    )
    try:
        res = minimize(
            smooth_obj,
            h,
            jac=smooth_obj_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": int(maxiter), "ftol": 1e-11, "disp": False},
        )
        xg = getattr(res, "x", None)
        if xg is not None:
            xg = np.asarray(xg, dtype=float)
            if np.all(np.isfinite(xg)):
                return project_density(xg, target)
    except Exception:
        pass
    return h


def active_obj(z):
    return float(np.asarray(z, dtype=float)[-1])


def active_obj_jac(z):
    g = np.zeros(int(_ACTIVE_N) + 1, dtype=float)
    if g.size:
        g[-1] = 1.0
    return g


def active_sum_fun(z):
    z = np.asarray(z, dtype=float)
    n = int(_ACTIVE_N)
    return float(np.sum(z[:n]) - _ACTIVE_TARGET)


def active_sum_jac(z):
    n = int(_ACTIVE_N)
    g = np.zeros(n + 1, dtype=float)
    g[:n] = 1.0
    return g


def active_con_fun(z):
    n = int(_ACTIVE_N)
    idxs = _ACTIVE_IDX
    if n <= 0 or idxs.size == 0:
        return np.array([], dtype=float)
    z = np.asarray(z, dtype=float)
    h = z[:n]
    t = float(z[-1])
    corr = corr_counts(h)
    return t - _ACTIVE_DX * corr[idxs]


def active_con_jac(z):
    n = int(_ACTIVE_N)
    idxs = _ACTIVE_IDX
    m = int(idxs.size)
    if n <= 0 or m == 0:
        return np.zeros((0, n + 1), dtype=float)
    z = np.asarray(z, dtype=float)
    h = z[:n]
    rows = np.zeros((m, n + 1), dtype=float)
    for r, idx in enumerate(idxs):
        rows[r, :n] = -_ACTIVE_DX * lag_gradient(h, int(idx))
        rows[r, n] = 1.0
    return rows


def active_slsqp_step(h, idxs, maxiter):
    global _ACTIVE_N, _ACTIVE_IDX, _ACTIVE_DX, _ACTIVE_TARGET
    n = int(h.size)
    target = 0.5 * float(n)
    h = project_density(h, target)
    if n <= 0:
        return h

    _ACTIVE_N = n
    _ACTIVE_IDX = np.asarray(idxs, dtype=int)
    _ACTIVE_DX = 2.0 / float(n)
    _ACTIVE_TARGET = target

    z0 = np.empty(n + 1, dtype=float)
    z0[:n] = h
    z0[n] = min(1.0, evaluate_c5(h) + 1e-6)

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    constraints = (
        {"type": "ineq", "fun": active_con_fun, "jac": active_con_jac},
        {"type": "eq", "fun": active_sum_fun, "jac": active_sum_jac},
    )
    try:
        res = minimize(
            active_obj,
            z0,
            jac=active_obj_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": int(maxiter), "ftol": 1e-11, "disp": False},
        )
        xg = getattr(res, "x", None)
        if xg is not None:
            xg = np.asarray(xg, dtype=float)
            if np.all(np.isfinite(xg)):
                return project_density(xg[:n], target)
    except Exception:
        pass
    return h


def active_refine(h, deadline, max_rounds=4, m_base=6, m_max=16, maxiter=None):
    n = int(h.size)
    target = 0.5 * float(n)
    h = project_density(h, target)
    c = evaluate_c5(h)
    if n < 2:
        return h, c

    no_improve = 0
    for r in range(int(max_rounds)):
        if time.perf_counter() >= deadline:
            break
        total = 2 * n - 1
        m = min(int(m_max), int(m_base) + 2 * r, total)
        if m <= 0:
            break
        corr = corr_counts(h)
        idxs = top_counts_indices(corr, m)
        if idxs.size == 0:
            break

        if maxiter is None:
            mi = 45 if n <= 120 else 28 if n <= 250 else 18
        else:
            mi = int(maxiter)
        if time.perf_counter() >= deadline:
            break

        cand = active_slsqp_step(h, idxs, mi)
        cc = evaluate_c5(cand)
        if cc < c - 1e-12:
            h = cand
            c = cc
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                break

    h = project_density(h, target)
    c = evaluate_c5(h)
    return h, c


def gradient_refine(h, deadline, K=6, Kmax=14, max_iter=120):
    n = int(h.size)
    target = 0.5 * float(n)
    h = project_density(h, target)
    c = evaluate_c5(h)
    if n < 2:
        return h, c

    K = int(max(2, min(K, 2 * n - 1)))
    Kmax = int(max(K, min(Kmax, 2 * n - 1)))
    stall = 0
    kstreak = 0

    for _ in range(int(max_iter)):
        if time.perf_counter() >= deadline:
            break
        corr = corr_counts(h)
        idxs = top_counts_indices(corr, K)
        if idxs.size == 0:
            break

        g = np.zeros(n, dtype=float)
        for idx in idxs:
            g += lag_gradient(h, int(idx))
        g /= float(idxs.size)

        d = -g
        d -= float(np.mean(d))
        norm = float(np.linalg.norm(d))
        if norm < 1e-14:
            break
        d /= norm

        best_h = h
        best_c = c
        for step in (1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 1e-1):
            cand = project_density(h + step * d, target)
            cc = evaluate_c5(cand)
            if cc < best_c - 1e-12:
                best_c = cc
                best_h = cand

        if best_c < c - 1e-12:
            h = best_h
            c = best_c
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                stall = 0
                if K < Kmax:
                    K = min(Kmax, K + 2)
                else:
                    kstreak += 1
                    if kstreak >= 3:
                        break

    h = project_density(h, target)
    c = evaluate_c5(h)
    return h, c


def add_pair_candidates(cands, h, g, num):
    n = int(h.size)
    if n < 2 or g.size != n:
        return
    num = int(max(1, num))
    if num > n:
        num = n

    neg = np.argsort(g)[:num]
    pos = np.argsort(-g)[:num]
    for ii_val in neg:
        ii = int(ii_val)
        gi = float(g[ii])
        for jj_val in pos:
            jj = int(jj_val)
            if ii == jj:
                continue
            gj = float(g[jj])
            diff = gj - gi
            if diff <= 1e-14:
                continue
            cap = min(1.0 - float(h[ii]), float(h[jj]))
            if cap <= 1e-12:
                continue
            score = cap * diff
            cands.append((score, ii, jj, cap))


def pair_local(h, deadline, max_iter=300, K=6, Kmax=16, max_pairs=24):
    n = int(h.size)
    target = 0.5 * float(n)
    dx = 2.0 / float(n)
    h = project_density(h, target)
    if n < 2:
        return h, evaluate_c5(h)

    base_c = corr_counts(h)
    best_count = float(np.max(base_c))
    best_c = best_count * dx
    accept = 0
    stall = 0
    kstreak = 0

    K = int(max(2, min(K, 2 * n - 1)))
    Kmax = int(max(K, min(Kmax, 2 * n - 1)))
    max_pairs = int(max_pairs)
    if n > 600:
        max_pairs = min(max_pairs, 12)
    elif n > 300:
        max_pairs = min(max_pairs, 16)

    n_alpha = 80 if n <= 240 else 48
    local_alpha = 160 if n <= 240 else 90
    tol_count = 1e-10 * max(1.0, float(n))

    for it in range(int(max_iter)):
        if time.perf_counter() >= deadline:
            break

        idxs = top_counts_indices(base_c, K)
        if idxs.size == 0:
            break

        g_avg = np.zeros(n, dtype=float)
        for idx in idxs:
            g_avg += lag_gradient(h, int(idx))
        g_avg /= float(idxs.size)

        cands = []
        num = 5 if n <= 300 else 4
        add_pair_candidates(cands, h, g_avg, num)
        for idx in idxs[:3]:
            g_one = lag_gradient(h, int(idx))
            add_pair_candidates(cands, h, g_one, num)

        if not cands:
            low = np.argsort(h)[:num]
            high = np.argsort(-h)[:num]
            for ii_val in low:
                ii = int(ii_val)
                for jj_val in high:
                    jj = int(jj_val)
                    if ii == jj:
                        continue
                    cap = min(1.0 - float(h[ii]), float(h[jj]))
                    if cap <= 1e-12:
                        continue
                    score = cap + 0.2 * abs(float(h[jj]) - float(h[ii]))
                    cands.append((score, ii, jj, cap))

        if not cands:
            break

        if len(cands) > max_pairs:
            cands.sort(reverse=True)
            cands = cands[:max_pairs]
        else:
            cands.sort(reverse=True)

        best_choice = None
        best_val = best_count

        for score, ii, jj, cap in cands:
            if time.perf_counter() >= deadline:
                break
            lin, quad = pair_lin_quad(h, ii, jj)
            alpha, val = pair_line_search(base_c, lin, quad, cap, n_alpha, local_alpha)
            if val < best_val - tol_count:
                best_val = val
                best_choice = (ii, jj, float(alpha), lin, quad)

        if best_choice is not None:
            ii, jj, alpha, lin, quad = best_choice
            old_i = float(h[ii])
            old_j = float(h[jj])
            h = np.array(h, dtype=float, copy=True)
            h[ii] = old_i + alpha
            h[jj] = old_j - alpha
            h = np.clip(h, 0.0, 1.0)

            base_c = base_c + lin * alpha + quad * (alpha * alpha)
            best_count = float(np.max(base_c))
            if best_count > best_val + 1e-7 * max(1.0, float(n)):
                base_c = corr_counts(h)
                best_count = float(np.max(base_c))
            if old_i + alpha >= 1.0 - 1e-12 or old_j - alpha <= 1e-12:
                base_c = corr_counts(h)
                best_count = float(np.max(base_c))

            best_c = best_count * dx
            accept += 1
            if accept % 8 == 0:
                base_c = corr_counts(h)
                best_count = float(np.max(base_c))
                best_c = best_count * dx
            stall = 0
        else:
            d = -g_avg
            d -= float(np.mean(d))
            norm = float(np.linalg.norm(d))
            improved = False
            if norm > 1e-13:
                d /= norm
                for step in (1e-4, 1e-3, 3e-3, 1e-2):
                    cand = project_density(h + step * d, target)
                    cc = evaluate_c5(cand)
                    if cc < best_c - 1e-12:
                        h = cand
                        base_c = corr_counts(h)
                        best_count = float(np.max(base_c))
                        best_c = best_count * dx
                        improved = True
                        break
            if improved:
                stall = 0
            else:
                stall += 1
                if stall >= 3:
                    stall = 0
                    if K < Kmax:
                        K = min(Kmax, K + 2)
                    else:
                        kstreak += 1
                        if kstreak >= 3:
                            break

    h = project_density(h, target)
    best_c = evaluate_c5(h)
    return h, best_c


def build_starts(n, rng, parent, best_h, n_rand):
    starts = []
    n = int(n)
    if best_h is not None:
        if int(best_h.size) == n:
            starts.append(np.asarray(best_h, dtype=float).copy())
        else:
            starts.append(upsample_h(best_h, n))
    if parent is not None:
        starts.append(upsample_h(parent, n))

    for mm in range(int(n_rand)):
        starts.append(make_candidate(n, rng, mm % 4))

    uniq = []
    for s in starts:
        s = project_density(s, 0.5 * n)
        dup = False
        for u in uniq:
            if vector_close(s, u):
                dup = True
                break
        if not dup:
            uniq.append(s)
    return uniq


def polish_candidate(h0, n, sub_deadline, do_smooth=True, do_active=True,
                     do_gradient=True, do_pair=True):
    n = int(n)
    target = 0.5 * float(n)
    h = project_density(h0, target)
    c = evaluate_c5(h)
    if n < 2:
        return h, c

    maxiter_smooth = 32 if n <= 120 else 22

    if do_smooth and n <= 240 and time.perf_counter() < sub_deadline - 12.0:
        if n <= 120:
            alphas = (25.0, 80.0)
        else:
            alphas = (18.0, 55.0)
        for alpha in alphas:
            if time.perf_counter() >= sub_deadline - 4.0:
                break
            cand = smooth_slsqp_step(h, alpha, maxiter_smooth)
            cc = evaluate_c5(cand)
            if cc < c - 1e-12:
                h = cand
                c = cc

    if do_active and n <= 700 and time.perf_counter() < sub_deadline - 8.0:
        if n <= 120:
            h, c = active_refine(h, sub_deadline, max_rounds=4, m_base=6, m_max=24,
                                 maxiter=40)
        elif n <= 300:
            h, c = active_refine(h, sub_deadline, max_rounds=3, m_base=6, m_max=12,
                                 maxiter=26)
        else:
            h, c = active_refine(h, sub_deadline, max_rounds=2, m_base=5, m_max=8,
                                 maxiter=16)

    if do_gradient and n <= 500 and time.perf_counter() < sub_deadline - 5.0:
        h, c = gradient_refine(h, sub_deadline, K=6, Kmax=14,
                               max_iter=70 if n <= 240 else 40)

    if do_pair and time.perf_counter() < sub_deadline - 1.0:
        mi = 500 if n <= 150 else 300 if n <= 300 else 150
        mp = 24 if n <= 240 else 12
        h, c = pair_local(h, sub_deadline, max_iter=mi, K=6, Kmax=16, max_pairs=mp)

    return h, c


def choose_resolutions(base_n, max_n=900):
    base_n = int(base_n)
    out = set()

    def add(x):
        x = int(x)
        if 6 <= x <= int(max_n):
            out.add(x)

    for m in (1, 2, 3, 4):
        add(base_n * m)
    add(720)

    common = [60, 90, 120, 180, 240, 360, 480, 600, 720, 900]
    for x in common:
        if len(out) >= 5:
            break
        if base_n <= x <= int(max_n):
            out.add(x)

    res = sorted(out)
    if base_n in res:
        res.remove(base_n)
    res.insert(0, int(base_n))
    return res[:5]


def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    if budget_s is None:
        budget = 1300.0
    else:
        try:
            budget = float(budget_s)
        except Exception:
            budget = 1300.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 1300.0

    reserve = 12.0 if budget > 30.0 else 2.0
    hard_deadline = start + max(1.0, budget - reserve)

    rng = np.random.default_rng(int(seed) if seed is not None else 42)

    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = read_parent()
    if parent is not None:
        nparent = int(parent.size)
        targetp = 0.5 * float(nparent)
        if is_feasible(parent, targetp, 1e-8):
            cp = evaluate_c5(parent)
            if cp < best_c:
                best_h = parent.copy()
                best_c = cp
                best_n = nparent
        pproj = project_density(parent, targetp)
        cp = evaluate_c5(pproj)
        if cp < best_c:
            best_h = pproj
            best_c = cp
            best_n = int(pproj.size)

    if best_h is None:
        n0 = 90
        best_h = make_candidate(n0, rng, 2)
        best_c = evaluate_c5(best_h)
        best_n = n0

    base_n = int(best_h.size) if int(best_h.size) >= 6 else 90
    nlist = choose_resolutions(base_n, max_n=900)

    plan = []
    for n in nlist:
        if int(n) <= max(120, 2 * base_n):
            cnt = 2
            cap = 110.0
        elif int(n) <= max(300, 4 * base_n):
            cnt = 1
            cap = 70.0
        else:
            cnt = 1
            cap = 45.0
        plan.append((int(n), int(cnt), float(cap)))

    for pi, (n, expected_cnt, default_cap) in enumerate(plan):
        if hard_deadline - time.perf_counter() < 25.0:
            break

        n_rand = 2 if n <= 120 else 1
        starts = build_starts(n, rng, parent, best_h, n_rand)
        if not starts:
            continue

        future = 0
        for q in range(pi + 1, len(plan)):
            future += int(plan[q][1])
        remaining = len(starts) + future

        now = time.perf_counter()
        rem = hard_deadline - now
        if rem < 18.0:
            break

        for s in starts:
            now = time.perf_counter()
            rem = hard_deadline - now
            if rem < 15.0:
                break
            per = min(float(default_cap), max(10.0, rem * 0.85 / float(max(1, remaining))))
            sub_deadline = min(now + per, hard_deadline - 2.0)
            remaining -= 1

            try:
                h_out, c_out = polish_candidate(s, n, sub_deadline)
            except Exception:
                h_out, c_out = project_density(s, 0.5 * n), evaluate_c5(s)

            if h_out is None:
                continue
            h_out = project_density(h_out, 0.5 * int(h_out.size))
            c_out = evaluate_c5(h_out)
            if c_out < best_c:
                best_h = h_out
                best_c = c_out
                best_n = int(h_out.size)

    rem = hard_deadline - time.perf_counter()
    if best_h is not None and rem > 35.0:
        nf = int(best_h.size)
        final_deadline = hard_deadline - 1.0

        if nf <= 120 and final_deadline - time.perf_counter() > 180.0:
            try:
                allidx = np.arange(2 * nf - 1, dtype=int)
                cand = active_slsqp_step(best_h, allidx, 28)
                cc = evaluate_c5(cand)
                if cc < best_c:
                    best_h = cand
                    best_c = cc
                    best_n = int(cand.size)
            except Exception:
                pass

        try:
            h_out, c_out = polish_candidate(
                best_h, best_n, final_deadline,
                do_smooth=(best_n <= 200 and final_deadline - time.perf_counter() > 80.0),
                do_active=True,
                do_gradient=True,
                do_pair=True,
            )
            if h_out is not None:
                h_out = project_density(h_out, 0.5 * int(h_out.size))
                c_out = evaluate_c5(h_out)
                if c_out < best_c:
                    best_h = h_out
                    best_c = c_out
                    best_n = int(h_out.size)
        except Exception:
            pass

    if best_h is None:
        fallback = project_density(np.full(90, 0.5, dtype=float), 45.0)
        return fallback, evaluate_c5(fallback), 90

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_c5(final_h)
    return final_h, float(final_c), int(final_h.size)
