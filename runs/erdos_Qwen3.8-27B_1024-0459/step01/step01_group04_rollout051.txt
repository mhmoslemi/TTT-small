import numpy as np
import time
import math
from scipy.optimize import minimize

_DEADLINE = None

class _Timeout(Exception):
    pass

def _check():
    if _DEADLINE is not None and time.time() > _DEADLINE:
        raise _Timeout()

def _compute_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * 2.0 / n)

def _get_scores(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _score_grad_all(h, weights):
    """
    Compute gradient of weighted sum of correlation scores.
    weights: array of length 2n-1, same as scores.
    Returns gradient of length n.
    """
    n = len(h)
    grad = np.zeros(n)
    for k, w in enumerate(weights):
        if abs(w) < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[:n-s] += w * (1.0 - h[s:])
            grad[s:] += w * (-h[:n-s])
        else:
            t = -s
            grad[t:] += w * (1.0 - h[:n-t])
            grad[:n-t] += w * (-h[t:])
    return grad

def _objective(x, alpha):
    _check()
    n = len(x)
    h = np.clip(x, 0.0, 1.0)
    scores = _get_scores(h)
    m = float(np.max(scores))
    if alpha > 1e-14:
        exp_val = np.exp((scores - m) / alpha)
        A = float(np.sum(exp_val))
        if math.isfinite(A) and A > 0:
            val = m + alpha * math.log(A)
        else:
            val = m
    else:
        val = m
    return val * (2.0 / n)

def _gradient(x, alpha):
    _check()
    n = len(x)
    h = np.clip(x, 0.0, 1.0)
    scores = _get_scores(h)
    m = float(np.max(scores))
    if alpha <= 1e-14:
        # Subgradient of max: uniform over active set
        tol = max(1e-7, 1e-6 * m)
        active = scores >= (m - tol)
        count = float(np.sum(active))
        if count == 0:
            active[int(np.argmax(scores))] = True
            count = 1.0
        weights = np.where(active, 1.0/count, 0.0)
    else:
        exp_val = np.exp((scores - m) / alpha)
        A = float(np.sum(exp_val))
        if math.isfinite(A) and A > 0:
            weights = exp_val / A
        else:
            weights = np.zeros(len(scores))
            weights[int(np.argmax(scores))] = 1.0
    grad = _score_grad_all(h, weights)
    grad *= (2.0 / n)
    return grad

def _constraint_f(x):
    n = len(x)
    return (float(np.sum(x)) - 0.5 * n) / n

def _constraint_jac(x):
    n = len(x)
    return np.full((1, n), 1.0 / n)

def _project(h, n):
    """Project h to feasible set: 0<=h<=1, sum(h)=n/2"""
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    # Bisection for additive shift
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    # Fine adjustment
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def _run_slsqp(h0, n, alphas, time_limit, state):
    """Run SLSQP with a sequence of alpha values."""
    global _DEADLINE
    start_t = time.time()
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": _constraint_f, "jac": _constraint_jac}]
    
    for alpha in alphas:
        remaining = time_limit - (time.time() - start_t)
        if remaining < 5.0:
            break
        if _DEADLINE and time.time() > _DEADLINE - 10:
            break
        maxiter = min(300, max(30, int(remaining / 0.5)))
        stage_deadline = time.time() + max(5.0, remaining)
        if _DEADLINE:
            stage_deadline = min(stage_deadline, _DEADLINE)
        
        old_dl = _DEADLINE
        _DEADLINE = stage_deadline
        try:
            res = minimize(
                _objective, h0, args=(alpha,),
                jac=_gradient,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": maxiter, "ftol": 1e-10, "disp": 0}
            )
        except _Timeout:
            res = None
        except Exception:
            res = None
        finally:
            _DEADLINE = old_dl
        
        if res is not None and hasattr(res, 'x') and res.x is not None:
            h0 = _project(res.x, n)
            c = _compute_c5(h0)
            if c < state.c:
                state.h = h0.copy()
                state.c = c
                state.n = n
        else:
            # Keep current best
            h0 = state.h.copy() if state.h is not None else h0
            if h0.size != n:
                h0 = _project(h0, n)
    
    return h0

class _State:
    def __init__(self):
        self.h = None
        self.c = float('inf')
        self.n = 0

def _upsample(parent, n):
    """Linearly upsample parent to n points."""

    if n0 == n:

    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x, x0, parent).astype(float)

import numpy as np
import time
import math
from scipy.optimize import minimize

_DEADLINE = None

class _TMO(Exception):
    pass

def _chk():
    if _DEADLINE is not None and time.time() > _DEADLINE:
        raise _TMO()

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def _scores(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _grad_weighted(h, w):
    n = len(h)
    g = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if abs(wk) < 1e-13:
            continue
        s = k - (n - 1)
        if s >= 0:
            g[:n - s] += wk * (1.0 - h[s:])
            g[s:] -= wk * h[:n - s]
        else:
            t = -s
            g[t:] += wk * (1.0 - h[:n - t])
            g[:n - t] -= wk * h[t:]
    return g

def _obj(x, alpha):
    _chk()
    n = len(x)
    h = np.clip(x, 0.0, 1.0)
    sc = _scores(h)
    m = float(np.max(sc))
    if alpha > 1e-14:
        e = np.exp((sc - m) / alpha)
        A = float(np.sum(e))
        if math.isfinite(A) and A > 0:
            v = m + alpha * math.log(A)
        else:
            v = m
    else:
        v = m
    if not math.isfinite(v):
        return 1e6
    return v * (2.0 / n)

def _grad(x, alpha):
    _chk()
    n = len(x)
    h = np.clip(x, 0.0, 1.0)
    sc = _scores(h)
    m = float(np.max(sc))
    if alpha <= 1e-14:
        tol = max(1e-7, 5e-7 * max(1.0, m))
        act = sc >= (m - tol)
        cnt = float(np.sum(act))
        if cnt == 0:
            act[int(np.argmax(sc))] = True
            cnt = 1.0
        w = np.where(act, 1.0 / cnt, 0.0)
    else:
        e = np.exp((sc - m) / alpha)
        A = float(np.sum(e))
        if math.isfinite(A) and A > 0:
            w = e / A
        else:
            w = np.zeros(len(sc))
            w[int(np.argmax(sc))] = 1.0
    g = _grad_weighted(h, w)
    g *= (2.0 / n)
    if not np.all(np.isfinite(g)):
        g = np.zeros(n)
    return g

def _cf(x):
    n = len(x)
    return (float(np.sum(x)) - 0.5 * n) / n

def _cj(x):
    n = len(x)
    return np.full((1, n), 1.0 / n)

def _proj(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    if h.size != n:
        h = h[:n] if h.size > n else np.pad(h, (0, n - h.size))
    tgt = 0.5 * n
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s < tgt:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = tgt - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.any(mask):
            per = diff / float(np.sum(mask))
            h[mask] += per
    return np.clip(h, 0.0, 1.0)

class _St:
    def __init__(self):
        self.h = None
        self.c = 1e9
        self.n = 0
    def upd(self, h, n):
        if h is None:
            return
        h = _proj(h, n)
        c = _c5(h)
        if math.isfinite(c) and c < self.c - 1e-14:
            self.h = h.copy()
            self.c = c
            self.n = n

def _slsqp(h0, n, alphas, tlim, st):
    global _DEADLINE
    t0 = time.time()
    bnds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _cf, "jac": _cj}]
    for al in alphas:
        rem = tlim - (time.time() - t0)
        if rem < 5.0:
            break
        if _DEADLINE is not None and time.time() > _DEADLINE - 15:
            break
        mi = min(400, max(40, int(rem * 2)))
        dl = time.time() + max(5.0, rem)
        if _DEADLINE is not None:
            dl = min(dl, _DEADLINE)
        old = _DEADLINE
        _DEADLINE = dl
        res = None
        try:
            res = minimize(_obj, h0, args=(al,), jac=_grad,
                           method="SLSQP", bounds=bnds, constraints=cons,
                           options={"maxiter": mi, "ftol": 1e-10, "disp": 0})
        except _TMO:
            res = None
        except Exception:
            res = None
        finally:
            _DEADLINE = old
        if res is not None and getattr(res, "x", None) is not None:
            h0 = _proj(res.x, n)
            st.upd(h0, n)
        if st.h is not None and st.n == n:
            h0 = st.h.copy()
    return _proj(h0, n)

def _upsample(par, n):
    n0 = len(par)
    if n0 == n:
        return par.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x, x0, par.astype(float))

def _synth(n, rng):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    best_h = _proj(np.full(n, 0.5), n)
    best_c = _c5(best_h)
    for f in range(1, 6):
        for ph in (0.0, 0.5 * math.pi, math.pi):
            h = 0.5 + 0.45 * np.cos(f * math.pi * x + ph)
            h = _proj(h, n)
            c = _c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    for amp in (0.2, 0.3, 0.4):
        h = 0.5 + amp * np.sin(3 * math.pi * x + 0.3) + 0.1 * np.cos(7 * math.pi * x)
        h = _proj(h, n)
        c = _c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h

def _binary_init(n, rng):
    target = n // 2
    h = np.zeros(n)
    idx = rng.choice(n, size=target, replace=False)
    h[idx] = 1.0
    # Smooth it
    out = h.copy()
    out[1:-1] = 0.25 * h[:-2] + 0.5 * h[1:-1] + 0.25 * h[2:]
    out[0] = 0.75 * h[0] + 0.25 * h[1]
    out[-1] = 0.25 * h[-2] + 0.75 * h[-1]
    return _proj(out, n)
