    """Project h onto {0<=x<=1, sum(x)=n/2}."""
    target = 0.5 * n

    m = float(np.max(s))
    active = np.where(s >= m - tol)[0]
    if len(active) == 0:

    g = np.zeros(n)

        elif k < n - 1:

    _DEADLINE = old_dl
    return best_h, best_c

def _perturb(h, n, rng, scale):
    h_new = h.copy()
    n_pert = max(3, n // 7)
    idx = rng.choice(n, size=n_pert, replace=False)
    h_new[idx] += rng.normal(0, scale, size=n_pert)
    h_new = _feasible(h_new, n)
    return h_new

def _upsample(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    x0 = (np.arange(n_from) + 0.5) * (2.0 / n_from)
    x1 = (np.arange(n_to) + 0.5) * (2.0 / n_to)
    h_new = np.interp(x1, x0, h)
    return _feasible(h_new, n_to)

def _cosine_init(n, f, phase, A):
    x = np.arange(n)
    h = 0.5 + A * np.cos(2.0 * math.pi * f * x / n + phase)
    h = np.clip(h, 0.0, 1.0)
    return _feasible(h, n)

def _try_resolution(h_start, n, budget, rng, state):
    if n < 4:
        return
    deadline = time.time() + budget
    if _DEADLINE is not None:
        deadline = min(deadline, _DEADLINE)
    if time.time() > deadline - 10:
        return
    h = _feasible(h_start, n)
    c = _c5(h)
    if c < state[0] + 1e-12:
        pass
    h_opt, c_opt = _lbfgs_full(h, n, deadline)
    if c_opt < state[0]:
        state[0] = c_opt
        state[1] = h_opt.copy()
        state[2] = n
    for restart in range(3):
        if time.time() > deadline - 15:
            break
        h_p = _perturb(h_opt, n, rng, 0.05)
        h_r, c_r = _projected_subgrad(h_p, n, 2000, 0.015, deadline)
        if c_r < state[0]:
            state[0] = c_r
            state[1] = h_r.copy()
            state[2] = n
            h_opt = h_r.copy()

def run(seed=42, budget_s=1300, **kwargs):
    global _DEADLINE
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0:
        budget_s = 30.0
    reserve = max(10.0, 0.02 * budget_s)
    _DEADLINE = t0 + budget_s - reserve

    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        parent = None

    state = [1.0, None, 0]

    if parent is not None and parent.size >= 4:
        n0 = int(parent.size)
        h0 = _feasible(parent, n0)
        c0 = _c5(h0)
        state = [c0, h0.copy(), n0]
    else:
        n0 = 70
        h0 = _feasible(np.full(n0, 0.5), n0)
        state = [_c5(h0), h0.copy(), n0]

    elapsed = time.time() - t0
    remaining = budget_s - reserve - elapsed

    # Phase 1: Main optimization at parent resolution
    if remaining > 60:
        b1 = min(0.35 * budget_s, remaining - 30)
        _try_resolution(state[1], state[2], b1, rng, state)

    # Phase 2: Try cosine initializations at n0
    elapsed = time.time() - t0
    remaining = budget_s - reserve - elapsed
    if remaining > 40:
        n_c = state[2]
        for f in [1, 2, 3]:
            if remaining < 30:
                break
            for ph in [0.0, math.pi / 3, 2 * math.pi / 3]:
                if remaining < 20:
                    break
                h_cos = _cosine_init(n_c, f, ph, 0.4)
                c_cos = _c5(h_cos)
                if c_cos < state[0] + 0.05:
                    b_c = min(25.0, remaining - 10)
                    if b_c > 15:
                        _try_resolution(h_cos, n_c, b_c, rng, state)
                remaining = budget_s - reserve - (time.time() - t0)

    # Phase 3: Try higher resolutions
    elapsed = time.time() - t0
    remaining = budget_s - reserve - elapsed
    n_best = state[2]
    higher_ns = []
    for mult in [1.5, 2.0, 2.5, 3.0, 4.0]:
        nn = int(round(mult * n_best / 2.0) * 2)
        if nn >= 4 and nn <= 12 * n_best and nn not in higher_ns:
            higher_ns.append(nn)
    higher_ns.sort()
    for nn in higher_ns[:4]:
        elapsed = time.time() - t0
        remaining = budget_s - reserve - elapsed
        if remaining < 50:
            break
        h_up = _upsample(state[1], state[2], nn)
        b_h = min(0.20 * budget_s, remaining - 20)
        if b_h > 30:
            _try_resolution(h_up, nn, b_h, rng, state)

    # Phase 4: Try a few random restarts at the best resolution
    elapsed = time.time() - t0
    remaining = budget_s - reserve - elapsed
    if remaining > 30:
        n_best = state[2]
        for _ in range(5):
            if remaining < 20:
                break
            h_r = _perturb(state[1], n_best, rng, 0.08)
            b_r = min(30.0, remaining - 10)
            if b_r > 15:
                deadline = time.time() + b_r
                h_opt, c_opt = _projected_subgrad(h_r, n_best, 3000, 0.015, deadline)
                if c_opt < state[0]:
                    state[0] = c_opt
                    state[1] = h_opt.copy()
                    state[2] = n_best
            remaining = budget_s - reserve - (time.time() - t0)

    # Final: ensure feasibility
    h_final = _feasible(state[1], state[2])
    c_final = _c5(h_final)
    n_final = int(state[2])

    # Verify
    if not np.all(np.isfinite(h_final)):
        h_final = np.zeros(n_final)
        h_final[n_final // 2:] = 1.0
        h_final = _feasible(h_final, n_final)
        c_final = _c5(h_final)

    return h_final, float(c_final), n_final
