from scipy.optimize import minimize

def compute_corr_values(h):

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        frac = (h > 1e-12) & (h < 1.0 - 1e-12)
        nfrac = np.sum(frac)
        if nfrac == 0:
            break
        h[frac] += diff / nfrac
    h = np.clip(h, 0.0, 1.0)
    # Final adjustment for exact sum
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Add to the element with most room
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    vals = corr * dx
    m = np.max(vals)
    # Log-sum-exp for numerical stability
    shifted = vals - m
    smooth = m + np.log(np.sum(np.exp(beta * shifted))) / beta
    return smooth

def binary_local_search(h, n, max_iters=10000, seed=42):
    """Greedy bit-flip local search for binary solutions"""
    rng = np.random.RandomState(seed)
    h = (h > 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h))
    n_want = n // 2
    if n_ones > n_want:
        ones = np.where(h == 1)[0]
        order = np.random.default_rng(seed).permutation(len(ones))
        for i in range(n_ones - n_want):
            h[ones[order[i]]] = 0
    elif n_ones < n_want:
        zeros = np.where(h == 0)[0]
        order = np.random.default_rng(seed).permutation(len(zeros))
        for i in range(n_want - n_ones):
            h[zeros[order[i]]] = 1
    
    best_c5 = compute_c5(h)
    
    for iteration in range(max_iters):
        improved = False
        # Try flipping each bit
        indices = np.arange(n)
        rng.shuffle(indices)
        for j in indices:
            old_val = h[j]
            h[j] = 1.0 - old_val
            # Check sum constraint
            s = np.sum(h)
            if abs(s - n/2) > 1e-10:
                h[j] = old_val
                continue
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                improved = True
            else:
                h[j] = old_val
        if not improved:
            break
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_g_beta = 50.0
_g_n = 0

def _smooth_obj(x):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    vals = corr * dx
    m = np.max(vals)
    shifted = vals - m
    return (m + np.log(np.sum(np.exp(_g_beta * shifted))) / _g_beta)

def _eq_con(x):
    return np.sum(x) - _g_n / 2.0

def _de_obj(x):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c5 = np.max(corr) * dx
    penalty = 100.0 * abs(np.sum(x) - n / 2.0)
    return c5 + penalty

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.astype(float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        frac = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.sum(frac)
        if nf == 0:
            break
        h[frac] += diff / nf
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        room = 1.0 - h[idx] if diff > 0 else h[idx]
        adj = min(abs(diff), room) if diff > 0 else max(diff, -room)
        h[idx] += adj
    return h

def interp_res(h_src, n_src, n_dst):
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    xs = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    xd = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    return project_feasible(np.interp(xd, xs, h_src), n_dst)

    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def make_thue_morse(n):
    h = np.array([float(bin(i).count('1') % 2) for i in range(n)], dtype=float)
    return project_feasible(h, n)

def make_sinusoidal(n, seed):
    rng = np.random.RandomState(seed)
    freq = rng.uniform(1.5, 4.0)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    return project_feasible(h, n)

def make_block_pattern(n, period):
    h = np.zeros(n)
    on = period // 2
    for i in range(n):
        if (i % period) < on:
            h[i] = 1.0
    return project_feasible(h, n)

def binary_swap_ls(h, n, n_swaps=100000, seed=42):
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    h = (h > 0.5).astype(float)
    n_want = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

        extra = rng.choice(ones, size=len(ones) - n_want, replace=False)
        h[extra] = 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    elif len(ones) < n_want:
        extra = rng.choice(zeros, size=n_want - len(ones), replace=False)
        h[extra] = 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        h[j] = 1.0

            h[i] = 1.0
            h[j] = 0.0

                fit[i] = ft
                if ft < best_f:
                    best_f = ft
                    best_x = trial.copy()
                    best_i = i
    return best_x, best_f
