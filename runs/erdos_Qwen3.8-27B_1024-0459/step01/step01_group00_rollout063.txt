import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, beta):
    x = np.clip(x, 0.0, 1.0)
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    # Numerically stable soft-max
    log_sum = np.log(np.sum(np.exp(beta * shifted))) + cmax * beta
    return log_sum / beta

def con_sum(x, n):
    return np.sum(x) - n / 2.0

def corr_gradient(h, k, n):
    """Gradient of the k-th element of np.correlate(h, 1-h, mode='full')."""
    d = n - 1 - k  # shift
    grad = np.zeros(n)
    m = np.arange(n)
    j1 = m + d
    j2 = m - d
    mask1 = (j1 >= 0) & (j1 < n)
    mask2 = (j2 >= 0) & (j2 < n)
    grad[mask1] += (1.0 - h[j1[mask1]])
    grad[mask2] -= h[j2[mask2]]
    return grad

def subgradient_refine(h, n, n_iter, alpha_init=0.1):
    """Subgradient descent on the max correlation with adaptive step."""
    h = make_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_lag = int(np.argmax(corr))
    
    alpha = alpha_init
    for it in range(n_iter):
        if (it % 200 == 0 and it > 0):
            # Halve step if no recent improvement
            alpha *= 0.95
        
        # Compute gradient for the active lag
        grad = corr_gradient(h, best_lag, n)
        
        # Update
        h_new = h - alpha * grad
        h_new = make_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            # Recompute active lag
            corr = np.correlate(h, 1.0 - h, mode='full')
            best_lag = int(np.argmax(corr))
            alpha = min(alpha * 1.05, 1.0)  # slight increase
        else:
            h = h_new  # still move, but track best
            alpha *= 0.98
        
        # Also periodically check if a different lag became active
        if it % 50 == 0:
            corr = np.correlate(h, 1.0 - h, mode='full')
            best_lag = int(np.argmax(corr))
    
    return best_h, best_c5

def interpolate_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    if n_src < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def generate_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return h

    return inits

def slsqp_optimize(n, h_init, beta, max_iter=3000):
    """SLSQP optimization of smooth max correlation."""
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': con_sum, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h_init, method='SLSQP',
                      args=(n, beta),
                      bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': 1e-18})
        if res.success or res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            c5 = compute_c5(h_out)
            return h_out, c5
    except Exception:
        pass
    return None, None

def coordinate_descent(h, n, n_rounds=50):
    """Simple coordinate descent: perturb one coordinate at a time."""
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    target = n / 2.0
    
    for rnd in range(n_rounds):
        improved = False
        order = np.random.permutation(n)
        for m in order:
            # Try to adjust h[m]
            # The objective depends on h[m] through correlations where m appears
            # Simple: try a few values
            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                h[m] += delta
                if h[m] < 0 or h[m] > 1:
                    h[m] -= delta
                    continue
                # Adjust sum: distribute to another random element
                # Actually, let's just check if the sum is still close
                s = np.sum(h)
                if abs(s - target) > 0.01:
                    h[m] -= delta

                c5_try = compute_c5(h)
                if c5_try < best_c5 - 1e-13:
                    best_c5 = c5_try
                    improved = True
                    break
                h[m] -= delta
        if not improved:
            break
    
    return make_feasible(h, n), best_c5

    from scipy.optimize import linprog

    grad[mask1] += (1.0 - h[j1[mask1]])
    grad[mask2] -= h[j2[mask2]]

    betas = [8, 20, 50, 100, 250, 600, 1500]
    for beta in betas:
        if _remaining() < 80:
            break
        max_iter = 4000 if _remaining() > 400 else 2000
        h_opt, c5_opt = slsqp_run(n, h_work, beta, max_iter=max_iter)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h_work = h_opt.copy()
    
    # Phase 2: Top-lags focused optimization
    if _remaining() > 60:
        h_tl, c5_tl = top_lags_optimize(best_h, best_n, top_m=5, max_iter=2000)
        if c5_tl < best_c5:
            best_c5 = c5_tl
            best_h = h_tl.copy()
    
    # Phase 3: Subgradient refinement
    if _remaining() > 60:
        n_iter = min(8000, int((_remaining() - 30) / 0.004))
        h_sub, c5_sub = subgradient_refine(best_h, best_n, n_iter, alpha_init=0.04)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()
    
    # Phase 4: Cutting plane LP
    if _remaining() > 50:
        h_cp, c5_cp = cutting_plane_lp(best_h, best_n, max_iter=80)
        if c5_cp < best_c5:
            best_c5 = c5_cp
            best_h = h_cp.copy()
    
    # Phase 5: Higher resolution (n=256)
    n_high = 256
    if _remaining() > 150:
        h_interp = interpolate_to_n(best_h, n_high)
        h_interp = make_feasible(h_interp, n_high)
        
        for beta in [15, 50, 150, 500, 1200]:
            if _remaining() < 50:
                break
            max_iter = 2500 if _remaining() > 200 else 1000
            h_opt, c5_opt = slsqp_run(n_high, h_interp, beta, max_iter=max_iter)
            if h_opt is not None and c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_high
                h_interp = h_opt.copy()
        
        if _remaining() > 50:
            h_tl, c5_tl = top_lags_optimize(best_h, best_n, top_m=5, max_iter=1500)
            if c5_tl < best_c5:
                best_c5 = c5_tl
                best_h = h_tl.copy()
        
        if _remaining() > 40:
            n_iter = min(5000, int((_remaining() - 20) / 0.004))
            h_sub, c5_sub = subgradient_refine(best_h, best_n, n_iter, alpha_init=0.025)
            if c5_sub < best_c5:
                best_c5 = c5_sub
                best_h = h_sub.copy()
        
        if _remaining() > 40:
            h_cp, c5_cp = cutting_plane_lp(best_h, best_n, max_iter=50)
            if c5_cp < best_c5:
                best_c5 = c5_cp
                best_h = h_cp.copy()
    
    # Phase 6: Higher resolution (n=320) if time allows
    n_high2 = 320
    if _remaining() > 120:
        h_interp2 = interpolate_to_n(best_h, n_high2)
        h_interp2 = make_feasible(h_interp2, n_high2)
        
        for beta in [30, 100, 400]:
            if _remaining() < 40:
                break
            max_iter = 1500
            h_opt, c5_opt = slsqp_run(n_high2, h_interp2, beta, max_iter=max_iter)
            if h_opt is not None and c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_high2
                h_interp2 = h_opt.copy()
        
        if _remaining() > 40:
            n_iter = min(4000, int((_remaining() - 20) / 0.005))
            h_sub, c5_sub = subgradient_refine(best_h, best_n, n_iter, alpha_init=0.02)
            if c5_sub < best_c5:
                best_c5 = c5_sub
                best_h = h_sub.copy()
    
    # Phase 7: Random perturbation local search
    if _remaining() > 30:
        n_trials = min(15000, int((_remaining() - 15) / 0.004))
        h_rs, c5_rs = random_perturbation_search(best_h, best_n, n_trials, step_size=0.02)
        if c5_rs < best_c5:
            best_c5 = c5_rs
            best_h = h_rs.copy()
    
    # Phase 8: Final polish - SLSQP + subgradient
    if _remaining() > 25:
        h_fin, c5_fin = slsqp_run(best_n, best_h, 800, max_iter=5000)
        if h_fin is not None and c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
    
    if _remaining() > 20:
        h_sub, c5_sub = subgradient_refine(best_h, best_n, 3000, alpha_init=0.01)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()
    
    if _remaining() > 15:
        h_cp, c5_cp = cutting_plane_lp(best_h, best_n, max_iter=30, near_frac=0.9)
        if c5_cp < best_c5:
            best_c5 = c5_cp
            best_h = h_cp.copy()
    
    # Final feasibility
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
