def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -60, 60)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_grad(x, beta):

def project_onto_simplex_box(h, target):

    h = h.copy()

        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        free_lo = h < 0.9999999
        free_hi = h > 1e-7
        free = free_lo & free_hi
        if not np.any(free):

            h = h + diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / np.sum(free)

    h = np.clip(h, 0.0, 1.0)

        # Vectorize: for each lag L, the contribution to grad is:
        # w_m * [(1-h[shifted by L]) - h[shifted by -L]]

        c5 = compute_c5(h)
        if c5 < best_val - 1e-12:

            best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700, 700)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_grad_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700, 700)
    exps = np.exp(shifted)
    s = np.sum(exps)
    return exps / s

def corr_grad(h, weights, dx):
    """Compute gradient of sum_m weights[m] * corr[m] w.r.t. h.
    corr = np.correlate(h, 1-h, 'full')
    corr[m] = sum_i h[i] * (1-h[i - (m - (n-1))])
    Let L = m - (n-1). Then corr[m] = sum_i h[i] * (1-h[i-L])
    d(corr[m])/d(h[j]) = (1-h[j-L]) [if valid] - h[j+L] [if valid]
    """
    n = len(h)
    lags = np.arange(-(n-1), n)  # L values for m=0,...,2n-2
    
    grad = np.zeros(n)
    for idx, L in enumerate(lags):
        w = weights[idx] * dx
        if abs(w) < 1e-15:
            continue
        if L >= 0:
            # (1-h[j+L]) for j = 0,...,n-1-L
            if n - L > 0:
                grad[:n-L] += w * (1.0 - h[L:])
            # -h[j-L] for j = L,...,n-1
            if L < n:
                grad[L:] -= w * h[:n-L]
        else:
            La = -L
            # (1-h[j-La]) for j = La,...,n-1
            grad[La:] += w * (1.0 - h[:n-La])
            # -h[j+La] for j = 0,...,n-1-La
            grad[:n-La] -= w * h[La:]
    
    return grad

def prox_sum_box(x, target, tol=1e-12):
    """Project onto {sum(x)=target, 0<=x<=1} via bisection on shift."""
    x = np.asarray(x, dtype=float)
    n = len(x)
    # Find lambda such that sum(clip(x - lambda, 0, 1)) = target
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    
    for _ in range(100):
        mid = (lo + hi) / 2
        x_proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(x_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    
    x_proj = np.clip(x - (lo + hi) / 2, 0.0, 1.0)
    # Fix residual
    s = np.sum(x_proj)
    diff = target - s
    if abs(diff) > 1e-14:
        # Distribute among non-boundary elements
        free = (x_proj > 0.001) & (x_proj < 0.999)
        if free.sum() > 0:
            x_proj[free] += diff / free.sum()

import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def lag_gradients(h):
    """Return array of shape (2n-1, n) where [m, j] = d(corr[m])/d(h[j]).
    This is O(n²) memory. For n=300, that's 599*300 = 179700 floats = 1.4MB. Fine.
    """
    n = len(h)
    lags = np.arange(-(n-1), n)  # L = m - (n-1)
    grads = np.zeros((2*n-1, n))
    
    for m, L in enumerate(lags):
        if L >= 0:
            grads[m, :n-L] += (1.0 - h[L:])       # (1-h[j+L]) for j=0..n-1-L
            grads[m, L:] -= h[:n-L]               # -h[j-L] for j=L..n-1
        else:
            La = -L
            grads[m, La:] += (1.0 - h[:n-La])     # (1-h[j-La]) for j=La..n-1
            grads[m, :n-La] -= h[La:]             # -h[j+La] for j=0..n-1-La
    
    return grads

def project_sum(h, target):
    """Project h onto {sum(h)=target, 0<=h<=1}."""
    h = h.astype(float).copy()
    # Bisection on shift
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(hp)
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    diff = target - np.sum(hp)
    if abs(diff) > 1e-14:
        free = (hp > 1e-6) & (hp < 1 - 1e-6)
        if free.sum() > 0:
            hp[free] += diff / free.sum()
            hp = np.clip(hp, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum(h, target):
    h = h.astype(float).copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(hp)
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-14:
        free = (hp > 1e-8) & (hp < 1.0 - 1e-8)
        if free.sum() > 0:
            hp[free] += diff / float(free.sum())
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    exps = np.exp(shifted)
    w = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        wm = w[mm] * dx
        if abs(wm) < 1e-16:
            continue
        L = mm - (n - 1)
        if L >= 0:
            if n - L > 0:
                grad[:n-L] += wm * (1.0 - h[L:])
            if L < n:
                grad[L:] -= wm * h[:n-L]
        else:
            La = -L
            grad[La:] += wm * (1.0 - h[:n-La])
            grad[:n-La] -= wm * h[La:]
    return grad

def topk_smooth_max_grad(h, beta, k):
    """Gradient using only top-k lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top k indices
    if k >= len(corr):
        idx = np.argsort(corr)[::-1]
    else:

    top_corr = corr[idx]
    m = np.max(top_corr)
    shifted = np.clip(beta * (top_corr - m), -700.0, 700.0)
    exps = np.exp(shifted)
    w_top = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for ii, mm in enumerate(idx):
        wm = w_top[ii] * dx
        if abs(wm) < 1e-16:

        L = mm - (n - 1)
        if L >= 0:
            if n - L > 0:
                grad[:n-L] += wm * (1.0 - h[L:])
            if L < n:
                grad[L:] -= wm * h[:n-L]
        else:
            La = -L
            grad[La:] += wm * (1.0 - h[:n-La])
            grad[:n-La] -= wm * h[La:]
    return grad

def fista_optimize(h_init, target, beta, step, max_iter, t_end, topk=0):
    n = len(h_init)
    x = project_sum(np.clip(h_init, 0.0, 1.0), target)
    y = x.copy()
    x_prev = x.copy()
    t_k = 1.0
    
    best_val = compute_c5(x)
    best_x = x.copy()
    patience_count = 0
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        if it > 0:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = x + ((t_k - 1.0) / t_new) * (x - x_prev)
            y = np.clip(y, 0.0, 1.0)
            t_k = t_new
        
        if topk > 0:
            grad = topk_smooth_max_grad(y, beta, topk)
        else:
            grad = smooth_max_grad(y, beta)
        
        x_new = y - step * grad
        x_new = project_sum(x_new, target)
        
        c5 = compute_c5(x_new)
        if c5 < best_val - 1e-12:
            best_val = c5
            best_x = x_new.copy()
            patience_count = 0
        else:
            patience_count += 1
        
        x_prev = x
        x = x_new
        
        if patience_count > 300:
            break
    
    return best_x, best_val

def slsqp_refine(h_init, target, beta, max_iter, t_end):
    n = len(h_init)
    h0 = project_sum(h_init, target)
    
    def obj_fn(h):
        hc = np.clip(h, 0.0, 1.0)
        corr = np.correlate(hc, 1.0 - hc, mode='full')
        mx = np.max(corr)
        sh = np.clip(beta * (corr - mx), -700.0, 700.0)
        return (mx + np.log(np.sum(np.exp(sh))) / beta) * (2.0 / n)
    
    def grad_fn(h):
        return smooth_max_grad(np.clip(h, 0.0, 1.0), beta)
    
    def cons_fn(h):
        return np.sum(h) - target
    
    def cons_jac_fn(h):
        return np.ones(n)
    
    try:
        res = minimize(obj_fn, h0, jac=grad_fn, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': cons_fn, 'jac': cons_jac_fn}],
                      options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None:
            return project_sum(res.x, target)
    except Exception:
        pass
    return h0

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)

    threshold = m - 50.0 / beta  # exp(-50) ≈ 2e-22
    active = corr >= threshold

        active = np.zeros(len(corr), dtype=bool)
        active[np.argmax(corr)] = True
    
    corr_active = corr[active]

    exps = np.exp(shifted)
