    m = len(idx_block)  # number of variables in the block
    h_fixed = h[idx_fixed]
    h_block_current = h[idx_block].copy()
    
    # Target for the block
    target_block = target - float(np.sum(h_fixed))
    target_block = np.clip(target_block, 0.0, float(m))
    
    # Compute all correlations
    corr = compute_correlations(h)
    num_lags = len(corr)  # 2n-1
    center = n - 1
    
    # For each lag, compute the linear approximation of Q_k in terms of h_block
    # A matrix: num_lags x m (coefficients of h_block)
    # b vector: num_lags (constant terms)
    
    # Precompute gradients for each lag
    # For lag index t (0 to 2n-2), the actual lag is k = t - center
    A = np.zeros((num_lags, m))
    b = np.zeros(num_lags)
    
    for t in range(num_lags):
        k = t - center
        g = lag_gradient(h, k)  # gradient w.r.t. all of h
        g_block = g[idx_block]  # gradient w.r.t. h_block
        A[t, :] = g_block
        b[t] = corr[t] - float(np.dot(g_block, h_block_current))
    
    # LP: minimize t
    # Variables: [h_block[0], ..., h_block[m-1], t] (m+1 vars)
    c = np.zeros(m + 1)
    c[m] = 1.0  # minimize t
    
    # A_ub: A - [0,...,0, 1] (each row: A[t] @ h_block - t <= -b[t])
    A_ub = np.zeros((num_lags, m + 1))
    A_ub[:, :m] = A
    A_ub[:, m] = -1.0
    b_ub = -b
    
    # A_eq: sum(h_block) = target_block
    A_eq = np.zeros((1, m + 1))
    A_eq[0, :m] = 1.0
    b_eq = np.array([target_block])
    
    # Bounds: 0 <= h_block <= 1, t >= 0
    bounds = [(0.0, 1.0)] * m + [(0.0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs')
        if res.success:
            h_block_new = res.x[:m]
            h_block_new = np.clip(h_block_new, 0.0, 1.0)
            # Ensure sum constraint
            h_block_new = project_density(h_block_new, target_block)
            h_new = h.copy()
            h_new[idx_block] = h_block_new
            return h_new
    except Exception:
        pass
    
    return h.copy()

def project_density(x, target):

    elif mode == 1:
        u = rng.random(n)
        thresh = float(np.median(u))
        h = (u > thresh).astype(float)
        h += rng.uniform(-0.1, 0.1, n)
    elif mode == 2:
        h = np.empty(n)
        pos = 0
        while pos < n:
            length = int(rng.integers(2, min(12, n - pos + 1)))
            val = float(rng.choice([0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0]))
            h[pos:pos + length] = val
            pos += length
    else:
        h = np.full(n, 0.5)
        h[:n // 3] = 0.8
        h[2 * n // 3:] = 0.2
        h += rng.uniform(-0.05, 0.05, n)
    return project_density(h, 0.5 * n)

def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 15.0)
    rng = np.random.default_rng(seed)
    best_h = None
    best_c = float("inf")
    best_n = 0
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        pass
    if parent is not None:
        pp = project_density(parent, 0.5 * parent.size)
        cp = evaluate_erdos(pp)
        best_h = pp
        best_c = cp
        best_n = parent.size
    else:
        best_h = make_start(90, rng, 0)
        best_c = evaluate_erdos(best_h)
        best_n = 90
    base_n = best_n if best_n > 1 else 90
    n_list = sorted(set([base_n, base_n * 2, base_n // 2 if base_n > 2 else base_n]))
    n_list = [n for n in n_list if 20 <= n <= 400]
    if not n_list:
        n_list = [90]
    for ni in n_list:
        if deadline - time.perf_counter() < 40.0:
            break
        starts = []
        if best_h is not None:
            if best_h.size == ni:
                starts.append(best_h.copy())
            else:
                starts.append(upsample_h(best_h, ni))
        if parent is not None and parent.size != ni:
            starts.append(upsample_h(parent, ni))
        elif parent is not None and parent.size == ni:
            starts.append(parent.copy())
        for mm in range(3):
            starts.append(make_start(ni, rng, mm))
        seen = []
        uniq_starts = []
        for s in starts:
            s = project_density(s, 0.5 * ni)
            dup = any(np.allclose(s, u, atol=1e-6) for u in seen)
            if not dup:
                uniq_starts.append(s)
                seen.append(s)
        n_starts = len(uniq_starts)
        for idx_s, s in enumerate(uniq_starts):
            rem = deadline - time.perf_counter()
            if rem < 20.0:
                break
            per = min(rem / max(1, n_starts - idx_s), 80.0)
            per = max(8.0, per)
            sub_dl = time.perf_counter() + per
            if time.perf_counter() < sub_dl - 5:
                c0 = evaluate_erdos(s)
                if c0 < best_c:
                    best_h = s.copy()
                    best_c = c0
                    best_n = ni
                h_lp, c_lp = run_block_lp(s, max_iters=150, deadline=sub_dl - 3)
                if c_lp < best_c:
                    best_h = h_lp.copy()
                    best_c = c_lp
                    best_n = ni
                if time.perf_counter() < sub_dl - 3:
                    h_sqp, c_sqp = slsqp_optimize(h_lp, ni, sub_dl - 2)
                    if c_sqp < best_c:
                        best_h = h_sqp.copy()
                        best_c = c_sqp
                        best_n = ni
        rem = deadline - time.perf_counter()
        if best_h is not None and rem > 30:
            h_ls, c_ls = local_search(best_h, deadline, max_iter=3000, active_k=5)
            if c_ls < best_c:
                best_h = h_ls.copy()
                best_c = c_ls
                best_n = best_h.size
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 20:
        h_ls2, c_ls2 = local_search(best_h, deadline, max_iter=5000, active_k=6)
        if c_ls2 < best_c:
            best_h = h_ls2.copy()
            best_c = c_ls2
            best_n = best_h.size
    if best_h is not None and deadline - time.perf_counter() > 10:
        h_lp2, c_lp2 = run_block_lp(best_h, max_iters=100, deadline=deadline)
        if c_lp2 < best_c:
            best_h = h_lp2.copy()
            best_c = c_lp2
            best_n = best_h.size
    if best_h is None:
        fallback = project_density(np.full(90, 0.5), 45.0)
        return fallback, evaluate_erdos(fallback), 90
    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
