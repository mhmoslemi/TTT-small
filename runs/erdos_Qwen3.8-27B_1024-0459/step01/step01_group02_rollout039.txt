    xp = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, xp, h_old)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    parent_h = None
    parent_n = 0
    
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        if parent_n > 0:
            parent_c5 = compute_c5(parent_h)
            best_h = parent_h.copy()
            best_c5 = parent_c5
            best_n = parent_n
    except Exception:
        pass
    
    if best_h is None:
        best_n = 100
        h0 = np.zeros(best_n)
        h0[::2] = 1.0
        best_h = project_feasible(h0, best_n)
        best_c5 = compute_c5(best_h)
    
    print("Initial: n={}, C5={:.6f}".format(best_n, best_c5))
    
    def time_left():
        return max(0.0, budget_s - (time.time() - start_time) - 3.0)
    
    # Phase 1: Continuous optimization at multiple resolutions
    resolutions = sorted(set([best_n] + [100, 128, 160]))
    resolutions = [n for n in resolutions if 48 <= n <= 256]
    
    for n in resolutions:
        if time_left() < 40:
            break
        
        print("\n=== Continuous optimization at n={} ===".format(n))
        dx = 2.0 / n
        target_sum = n / 2.0
        
        # Build initial guesses
        inits = []
        
        if parent_n > 0:
            h_interp = interpolate_to_n(parent_h, parent_n, n)
            inits.append(project_feasible(h_interp, n))
        
        # Alternating
        h_alt = np.zeros(n)
        h_alt[::2] = 1.0
        inits.append(project_feasible(h_alt, n))
        
        # From best so far
        if best_n != n and best_n > 0:
            h_from_best = interpolate_to_n(best_h, best_n, n)
            inits.append(project_feasible(h_from_best, n))
        
        # Random with structure
        for _ in range(2):
            h_r = rng.rand(n)
            inits.append(project_feasible(h_r, n))
        
        # Block patterns with different block sizes
        for bs in [max(2, n//16), max(2, n//8), max(3, n//5)]:
            h_blk = np.zeros(n)
            for b in range(0, n, 2*bs):
                h_blk[b:b+bs] = 1.0
            inits.append(project_feasible(h_blk, n))
        
        # Arithmetic progression
        for step in [3, 4, 5]:
            h_ap = np.zeros(n)
            h_ap[::step] = 1.0
            inits.append(project_feasible(h_ap, n))
        
        # Thue-Morse inspired
        h_tm = np.zeros(n)
        for i in range(n):
            h_tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
        inits.append(project_feasible(h_tm, n))
        
        # Run SLSQP for each init
        for init_idx, h0 in enumerate(inits):
            if time_left() < 15:
                break
            
            h0 = project_feasible(h0, n)
            h_curr = h0.copy()
            c5_curr = compute_c5(h_curr)
            
            # Multi-beta SLSQP
            for beta in [15, 50, 200, 500, 1500]:
                if time_left() < 8:
                    break
                try:
                    h_new = slsqp_optimize(h_curr, n, beta, maxiter=250, ftol=1e-14)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_curr:
                        h_curr = h_new
                        c5_curr = c5_new
                except Exception:
                    pass
            
            if c5_curr < best_c5 - 1e-10:
                best_c5 = c5_curr
                best_h = h_curr.copy()
                best_n = n
                print("  Init {} at n={}: C5={:.6f}".format(init_idx, n, best_c5))
        
        # Perturbation-based global search
        n_perturbs = min(8, max(3, int(time_left() // 30)))
        for p in range(n_perturbs):
            if time_left() < 8:
                break
            
            if best_n == n:
                h_pert = best_h.copy()
            else:
                h_pert = interpolate_to_n(best_h, best_n, n)
            
            mask = rng.rand(n) < 0.4
            h_pert[mask] += rng.randn(np.sum(mask)) * 0.15
            h_pert = project_feasible(h_pert, n)
            
            try:
                h_new = slsqp_optimize(h_pert, n, 100, maxiter=200, ftol=1e-13)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-10:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n
                    print("  Perturb {} at n={}: C5={:.6f}".format(p, n, best_c5))
            except Exception:
                pass
    
    # Phase 2: Binary optimization
    print("\n=== Binary optimization ===")
    binary_ns = sorted(set([n for n in [96, 100, 128, 160] if n % 2 == 0 and 48 <= n <= 256]))
    
    for n in binary_ns:
        if time_left() < 20:
            break
        
        print("\n--- Binary at n={} ---".format(n))
        dx = 2.0 / n
        
        # Get initial binary solution
        if best_n == n:
            h_bin = (best_h > 0.5).astype(float)
        elif best_n > 0:
            h_interp = interpolate_to_n(best_h, best_n, n)
            h_bin = (h_interp > 0.5).astype(float)
        else:
            h_bin = np.zeros(n)
            h_bin[::2] = 1.0
        
        # Fix sum to exactly n/2
        n_want = n // 2
        n_ones = int(np.sum(h_bin))
        if n_ones != n_want:
            if n_ones > n_want:
                ones_idx = np.where(h_bin == 1.0)[0]
                # Flip the ones with lowest fractional values
                if best_n > 0 and best_n == n:
                    order = np.argsort(best_h[ones_idx])[:n_ones - n_want]
                else:
                    interp_for_order = interpolate_to_n(best_h, best_n, n) if best_n > 0 and best_n != n else (best_h > 0.5).astype(float)
                    order = np.argsort(-ones_idx)[:n_ones - n_want]
                h_bin[ones_idx[order]] = 0.0
            else:
                zeros_idx = np.where(h_bin == 0.0)[0]
                need = n_want - n_ones
                h_bin[zeros_idx[:need]] = 1.0
        
        c5_bin = compute_c5(h_bin)
        print("  Initial binary C5={:.6f}".format(c5_bin))
        
        # Simulated annealing
        c5_sa = c5_bin
        h_sa = h_bin.copy()
        n_sa_iter = min(150000, max(20000, int(time_left() * 3000)))
        if time_left() > 15:
            h_sa, c5_sa = binary_sa(h_bin, n_iter=n_sa_iter, T_start=0.04, T_end=0.0005, seed=seed + n)
            print("  After SA: C5={:.6f}".format(c5_sa))
        
        # Steepest descent local search
        h_ls = h_sa.copy()
        time_for_ls = min(60.0, time_left() * 0.5)
        h_ls, _ = binary_local_search(h_ls, max_passes=300, time_limit=time_for_ls, start_time=start_time)
        c5_ls = compute_c5(h_ls)
        print("  After local search: C5={:.6f}".format(c5_ls))
        
        if c5_ls < best_c5 - 1e-10:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            best_n = n
            print("  NEW BEST: n={}, C5={:.6f}".format(n, best_c5))
    
    # Phase 3: Final polish on best solution
    if time_left() > 15:
        print("\n=== Final polish ===")
        h_final = best_h.copy()
        c5_final = compute_c5(h_final)
        
        for beta in [500, 1000, 3000]:
            if time_left() < 5:
                break
            try:
                h_new = slsqp_optimize(h_final, best_n, beta, maxiter=500, ftol=1e-15)
                c5_new = compute_c5(h_new)
                if c5_new < c5_final - 1e-10:
                    h_final = h_new
                    c5_final = c5_new
                    print("  Beta {} polish: C5={:.6f}".format(beta, c5_final))
            except Exception:
                pass
        
        if c5_final < best_c5 - 1e-10:
            best_c5 = c5_final
            best_h = h_final
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    # Try evaluate_erdos_solution if available
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if isinstance(eval_c5, (float, int)) and eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
