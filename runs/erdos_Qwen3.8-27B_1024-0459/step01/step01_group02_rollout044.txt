def project_feasible(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)

def subgradient_step(h, n, lr, n_inner=3):
    """Subgradient method: reduce the max correlation component"""
    h = h.copy()
    for _ in range(n_inner):
        corr = compute_corr(h)
        k_max = np.argmax(corr)
        m = k_max - (n - 1)
        # Gradient of corr[k_max] w.r.t. h[i]
        grad = np.zeros(n)
        for i in range(n):
            # corr[k] = sum_j h[j]*(1-h[j-m]) for m>=0, or sum_j h[j+m]*(1-h[j]) for m<0
            if m >= 0:
                # h[i] appears with (1-h[i-m]) if i-m in [0,n) AND with -h[i+m] if i+m in [0,n)
                term1 = 0.0
                term2 = 0.0
                if m == 0:
                    term1 = 1.0 - h[i]
                    term2 = -h[i]
                else:
                    if i - m >= 0:
                        term1 = 1.0 - h[i - m]
                    if i + m < n:
                        term2 = -h[i + m]
                grad[i] = term1 + term2
            else:
                m2 = -m
                if i + m2 < n:
                    grad[i] = 1.0 - h[i + m2]
                if i - m2 >= 0:
                    grad[i] += -h[i - m2]
        h = h - lr * grad
        h = project_feasible(h, n)
    return h

def to_binary(h, n, rng):
    """Convert continuous h to binary with exact sum constraint"""
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_want = n // 2
    if n_ones > n_want:
        ones = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones])
        for idx in order[:n_ones - n_want]:
            h_bin[ones[idx]] = 0.0
    elif n_ones < n_want:

        order = np.argsort(-h[zeros])
        for idx in order[:n_want - n_ones]:
            h_bin[zeros[idx]] = 1.0
    return h_bin

        if remaining() > 15 and n % 2 == 0:
            h_fin_bin = to_binary(_best_h, n, _rng)
            h_fin_ls, c5_fin = binary_local_search(h_fin_bin, n, _rng, n_iter=2000)
            update_best(h_fin_ls, n)

    # Final verification
    _best_h = project_feasible(_best_h, _best_n)
    _best_c5 = compute_c5(_best_h)
    
    # Try evaluate_erdos_solution if available
    try:
        eval_c5 = evaluate_erdos_solution(_best_h)
        if eval_c5 < _best_c5:
            _best_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(_best_n, _best_c5))
    return (_best_h, _best_c5, _best_n)
