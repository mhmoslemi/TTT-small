def compute_c5(h):

    if n == 0:

    h = np.clip(h, 0.0, 1.0)

    c = float(np.max(corr) * (2.0 / n))

def get_scores(h):
    """Returns raw correlation values (length 2n-1)."""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    return np.correlate(h, 1.0 - h, mode='full')

def shift_grad(h, k, n):
    """Gradient of raw correlation at shift index n-1+k w.r.t. h."""
    grad = np.zeros(n)
    if k >= 0:
        m = n - k
        if m <= 0:
            return grad
        grad[:m] += (1.0 - h[k:k+m])
        grad[k:] += -h[:m]
    else:
        d = -k
        m = n - d
        if m <= 0:
            return grad
        grad[d:] += (1.0 - h[:m])
        grad[:m] += -h[d:d+m]
    return grad

    dx = 2.0 / n
    corr = get_scores(h)  # raw, length 2n-1
