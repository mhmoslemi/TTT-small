def project_density(x, target):
    """Project x onto {0<=x<=1, sum(x)=target}"""
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:

    # Water-filling

    # Binary search for the threshold
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5*(lo+hi)

        if s < target:

    h = np.clip(x + 0.5*(lo+hi), 0.0, 1.0)

    if abs(res) > 1e-10:
