    for k in range(n_overlaps):

        w = p[k] * dx
        if w < 1e-14:
            continue

        j_lo = max(0, -s)
        j_hi = min(n - 1, n - 1 - s)
        if j_lo <= j_hi:

            i_hi = n - 1
        else:
            i_lo = 0
            i_hi = n - 1 + s
        if i_lo <= i_hi:
            grad[i_lo:i_hi+1] += -w * h[i_lo-s:i_hi+1-s]
    
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return np.max(compute_overlaps(h, n))

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-10
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        h += (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(h, n, T):
    ov = compute_overlaps(h, n)
    m = ov.max()
    return m + T * np.log(np.sum(np.exp((ov - m) / T)))

def smooth_max_grad(h, n, T):
    ov = compute_overlaps(h, n)
    m = ov.max()
    p = np.exp((ov - m) / T)
    p /= p.sum()
    
    # Only compute for significant weights
    threshold = T * np.log(len(ov))  # below this, contribution is negligible
    active = p > threshold / ov.max()  # heuristic
    # Actually just use all, it's O(n) per k
    
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for k in range(2 * n - 1):
        w = p[k] * dx
        if w < 1e-13:
            continue
        s = k - (n - 1)
        
        # Term 1: i in [max(0,-s), min(n-1, n-1-s)]
        j_lo = max(0, -s)
        j_hi = min(n - 1, n - 1 - s)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += w * (1.0 - h[j_lo+s:j_hi+1+s])
        
        # Term 2: i in [max(0, s), min(n-1, n-1+s)]
        i_lo = max(0, s)
        i_hi = min(n - 1, n - 1 + s)
        if i_lo <= i_hi:
            grad[i_lo:i_hi+1] += -w * h[i_lo-s:i_hi+1-s]
    
    return grad

def slsqp_smooth(h0, n, T, maxiter=300):
    h = h0.copy()
    res = minimize(
        lambda x: smooth_max_val(x, n, T),
        h,
        jac=lambda x: smooth_max_grad(x, n, T),
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{
            'type': 'eq',
            'fun': lambda x: np.sum(x) - n / 2.0,
            'jac': lambda x: np.ones(n)
        }],
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out, n)

            rng_sa_f = np.random.RandomState(seed + 50000 + n_fine)
            try:
                n_iter_sa = 40000 if n_fine <= 140 else 25000
                h_bin, v_bin = binary_sa(h_fine, n_fine, rng_sa_f,
                                          n_iter=n_iter_sa, T_start=0.003, T_end=0.0002)
                if v_bin < best_val - 1e-10:
                    best_h, best_val, best_n = h_bin.copy(), v_bin, n_fine
            except Exception:
            pass

        # Another SA from random
        if time_left() > 100 and n_fine <= 200:
            rng_sa_r = np.random.RandomState(seed + 90000 + n_fine)
            h_rand_bin = project_feasible(rng_sa_r.uniform(0, 1, n_fine), n_fine)
            try:
                h_bin, v_bin = binary_sa(h_rand_bin, n_fine, rng_sa_r,
                                          n_iter=20000, T_start=0.005, T_end=0.0003)
                if v_bin < best_val - 1e-10:
                    best_h, best_val, best_n = h_bin.copy(), v_bin, n_fine
            except Exception:
                pass

    # Phase 4: Iterated local search at best resolution
    if time_left() > 150:
        for il in range(20):
            if time_left() < 40:
                break
            # Different perturbation strategies
            strategy = il % 3
            if strategy == 0:
                h_pert = best_h + rng.normal(0, 0.1, best_n)
            elif strategy == 1:
                h_pert = best_h.copy()
                idx = rng.choice(best_n, size=max(3, best_n // 7), replace=False)
                h_pert[idx] = np.clip(h_pert[idx] + rng.normal(0, 0.4, len(idx)), 0, 1)
            else:
                h_pert = best_h.copy()
                # Flip a block
                start = rng.randint(0, best_n // 2)
                length = rng.randint(3, best_n // 6)
                h_pert[start:start + length] *= 0.3

            h_pert = project_feasible(h_pert, best_n)

            # SLSQP
            try:
                h_try, v_try = slsqp_optimize(h_pert, best_n, 0.01, maxiter=400)
                if v_try < best_val - 1e-10:
                    best_h, best_val = h_try.copy(), v_try
            except Exception:
                pass

            # SA from perturbed
            if time_left() > 40:
                rng_sa_il = np.random.RandomState(seed + 100000 + il * 7)
                try:
                    h_bin, v_bin = binary_sa(h_pert, best_n, rng_sa_il,
                                              n_iter=10000, T_start=0.003, T_end=0.0002)
                    if v_bin < best_val - 1e-10:
                        best_h, best_val = h_bin.copy(), v_bin
                except Exception:
                    pass

    # Phase 5: Intensive SA on the best solution (late annealing)
    if time_left() > 60:
        rng_sa_final = np.random.RandomState(seed + 777777)
        try:
            h_bin, v_bin = binary_sa(best_h, best_n, rng_sa_final,
                                      n_iter=40000, T_start=0.002, T_end=0.0001)
            if v_bin < best_val - 1e-12:
                best_h, best_val = h_bin.copy(), v_bin
        except Exception:
            pass

    # Phase 6: Final SLSQP polish
    if time_left() > 30:
        for T in [0.005, 0.002, 0.001]:
            if time_left() < 15:
                break
            try:
                h_try, v_try = slsqp_optimize(best_h, best_n, T, maxiter=600, ftol=1e-14)
                if v_try < best_val - 1e-12:
                    best_h, best_val = h_try.copy(), v_try
                else:
                    break
            except Exception:
                break

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_val = compute_c5(best_h, best_n)

    return (best_h, best_val, best_n)
