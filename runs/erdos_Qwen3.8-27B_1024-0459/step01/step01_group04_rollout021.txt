def structured_starts(n, rng):
    """Generate diverse structured binary starting points."""
    starts = []
    target = n // 2
    
    # Golden ratio placement
    phi = (1 + math.sqrt(5)) / 2
    positions = set()

        if pos not in positions:
            positions.add(pos)
            positions_list.append(pos)

        b = np.zeros(n, dtype=np.int8)
        b[sorted(positions_list[:target])] = 1
        starts.append(b.copy())
    
    # Periodic patterns
    for period in range(max(2, n // 20), min(12, n // 4) + 1):
        fill = max(1, int(target * period / n))
        if fill > period:
            fill = period - 1
        if fill <= 0 or fill >= period:
            continue
        b = np.zeros(n, dtype=np.int8)
        for i in range(n):
            if i % period < fill:
                b[i] = 1
        # Adjust to exact count
        while b.sum() > target:
            idx = np.flatnonzero(b == 1)
            b[idx[rng.integers(len(idx))]] = 0
        while b.sum() < target:
            idx = np.flatnonzero(b == 0)
            b[idx[rng.integers(len(idx))]] = 1
        if b.sum() == target:
            starts.append(b.copy())
    
    # Thue-Morse-like
    tm = np.zeros(n, dtype=np.int8)
    for i in range(n):
        cnt = bin(i).count('1')
        tm[i] = cnt % 2
    if tm.sum() != target:
        # Adjust
        if tm.sum() > target:
            excess = tm.sum() - target
            ones = np.flatnonzero(tm == 1)
            rng.shuffle(ones)
            tm[ones[:excess]] = 0
        else:
            deficit = target - tm.sum()
            zeros = np.flatnonzero(tm == 0)
            rng.shuffle(zeros)
            tm[zeros[:deficit]] = 1
    starts.append(tm.copy())
    
    # Random
    for _ in range(3):
        idx = rng.choice(n, size=target, replace=False)
        b = np.zeros(n, dtype=np.int8)
        b[idx] = 1
        starts.append(b.copy())
    
    return starts


def upsample_binary(parent, n):
    """Upsample a binary array to new length."""
    n0 = parent.size
    if n0 == 0:
        return np.zeros(n, dtype=np.int8)
    x0 = np.arange(n0, dtype=float)
    x = np.linspace(0, n0 - 1, n)
    # Nearest neighbor upsample
    idx = np.round(x).astype(int)
    idx = np.clip(idx, 0, n0 - 1)
    return parent[idx].astype(np.int8)

        pass
    except Exception:
        pass
    
    # Stage 2: Exact max with smaller alpha
    if _RUN_DEADLINE is not None and time.time() + 10 < _RUN_DEADLINE:
        alpha = 0.1 / n
        
        def obj2(x):
            check_deadline()
            x = np.clip(x, 0.0, 1.0)
            s = np.correlate(x, 1.0 - x, mode="full")
            m = float(np.max(s))
            shifted = (s - m) / alpha
            shifted = np.clip(shifted, -50, 0)
            logsum = m + alpha * np.log(np.sum(np.exp(shifted)))
            return float(logsum) * (2.0 / n)
        
        def obj2_grad(x):
            check_deadline()
            x = np.clip(x, 0.0, 1.0)
            s = np.correlate(x, 1.0 - x, mode="full")
            m = float(np.max(s))
            shifted = (s - m) / alpha
            shifted = np.clip(shifted, -50, 0)
            w = np.exp(shifted)
            A = w.sum()
            if A < 1e-30:
                w = np.zeros(len(w))
                w[np.argmax(s)] = 1.0
            else:
                w /= A
            grad = np.zeros(n)
            n_full = 2 * n - 1
            for k in range(n_full):
                if w[k] < 1e-12:
                    continue
                ck = w[k] * (2.0 / n)
                if k < n:
                    m_val = n - k
                    grad[k:] += ck * (1.0 - x[:m_val])
                    grad[:m_val] += ck * (-x[k:])
                else:
                    d = k - n + 1
                    m_val = n - d
                    grad[:m_val] += ck * (1.0 - x[d:])
                    grad[d:] += ck * (-x[:m_val])
            if not np.all(np.isfinite(grad)):
                grad = np.zeros(n)
            return grad
        
        start_t2 = time.time()
        tl2 = min(time_limit * 0.5, max(10.0, time_limit * 0.4))
        d2 = start_t2 + tl2
        if _RUN_DEADLINE is not None:
            d2 = min(d2, _RUN_DEADLINE)
        
        try:
            res2 = minimize(
                obj2, best_h, method="SLSQP", bounds=bounds,
                constraints=constraints,
                options={"maxiter": 300, "ftol": 1e-11, "disp": 0,
                         "eps": 1e-7}
            )
            if res2.x is not None:
                h2 = as_feasible(res2.x, n)
                c2 = compute_c5(h2)
                if c2 < best_c:
                    best_c = c2
                    best_h = h2.copy()
        except BudgetError:
            pass
        except Exception:
            pass
    
    return best_h, best_c


def coordinate_descent(h, n, deadline, max_iters=5000):
    """Simple coordinate descent: adjust one element at a time."""
    h = as_feasible(h, n).copy()
    best_c = compute_c5(h)
    step = 0.01
    
    for it in range(max_iters):
        check_deadline()
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            # Try increasing
            old = h[i]
            h[i] = min(1.0, h[i] + step)
            # Adjust to maintain sum
            s = h.sum()
            target = 0.5 * n
            diff = s - target
            if abs(diff) > 1e-12:
                others = np.arange(n) != i
                h[others] -= diff / others.sum()
                h = np.clip(h, 0.0, 1.0)
                s2 = h.sum()
                # Re-correct
                rem = target - s2
                if abs(rem) > 1e-9:
                    h[i] = old  # revert if can't fix
                    continue
            c = compute_c5(h)
            if c < best_c - 1e-12:
                best_c = c
                improved = True
            else:
                h[i] = old
                h = as_feasible(h, n)
        
        if not improved:
            # Try smaller step
            step *= 0.7
            if step < 1e-5:
                break
            # Reset step occasionally
            if it % 500 == 499:
                step = 0.01
    
    return as_feasible(h, n), best_c


def get_parent():
    try:
        p = np.asarray(initial_h_values, dtype=float)
        if p.ndim != 1 or p.size < 4:
            return None
        return p.copy()
    except Exception:
        return None


def run(seed=42, budget_s=1300, **kwargs):
    global _RUN_DEADLINE
    
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0:
        budget_s = 1300.0
    reserve = min(30.0, 0.1 * budget_s)
    deadline = t0 + max(5.0, budget_s - reserve)
    _RUN_DEADLINE = deadline
    
    rng = np.random.default_rng(seed)
    
    # Get parent
    parent = get_parent()
    if parent is not None:
        n0 = int(parent.size)
        h0 = parent.copy()
        h0 = as_feasible(h0, n0)
        c0 = compute_c5(h0)
    else:
        n0 = 70
        h0 = np.full(n0, 0.5)
        h0 = as_feasible(h0, n0)
        c0 = compute_c5(h0)
    
    # Best solution tracker
    best_h = h0.copy()
    best_c = c0
    best_n = n0
    
    def update_best(h, c, n):
        nonlocal best_h, best_c, best_n
        if math.isfinite(c) and c < best_c:
            best_c = float(c)
            best_h = np.array(h, dtype=float)
            best_n = int(n)
    
    print(f"Initial: n={n0}, C5={best_c:.6f}")
    
    # === Phase 1: Binary search at n0 ===
    if n0 % 2 == 0:
        n_b = n0
        bin_starts = []
        
        # From parent
        bin0 = binary_from_continuous(parent, n_b)
        if bin0.sum() == n_b // 2:
            bin_starts.append(bin0)
        
        # Structured starts
        structured = structured_starts(n_b, rng)
        bin_starts.extend(structured[:5])
        
        print(f"Binary search at n={n_b}: {len(bin_starts)} starts")
        
        bl_deadline = min(time.time() + 80.0, deadline - 60.0)
        for si, bs in enumerate(bin_starts):
            if time.time() > bl_deadline:
                break
            remaining = max(5.0, bl_deadline - time.time())
            try:
                bh, bc = binary_hill_climb(bs, n_b, rng, bl_deadline, max_rounds=80)
                if bc < best_c:
                    update_best(bh.astype(float), bc, n_b)
                    print(f"  Binary start {si}: C5={bc:.6f}")
            except BudgetError:
                break
            except Exception:
                pass
        
        # 3-swap refinement on best binary
        if time.time() < deadline - 60.0:
            # Get best binary
            bvals = (best_h > 0.5).astype(np.int8)
            if bvals.sum() == n_b // 2:
                try:
                    bh3, bc3 = binary_3swap(bvals, n_b, rng, deadline, iters=10000)
                    if bc3 < best_c:
                        update_best(bh3.astype(float), bc3, n_b)
                        print(f"  3-swap: C5={bc3:.6f}")
                except Exception:
                    pass
    
    # === Phase 2: Continuous refinement at n0 ===
    if time.time() < deadline - 80.0:
        tl_cont = min(200.0, max(30.0, deadline - time.time() - 100.0))
        try:
            h_ref, c_ref = continuous_refine(best_h, best_n, deadline, tl_cont)
            if c_ref < best_c:
                update_best(h_ref, c_ref, best_n)
                print(f"Continuous refine n={best_n}: C5={c_ref:.6f}")
        except Exception:
            pass
    
    # === Phase 3: Try higher resolution ===
    for mult in [2, 3]:
        n_new = n0 * mult
        if n_new % 2 != 0:
            n_new += 1
        if n_new > 280:
            continue
        if time.time() > deadline - 100.0:
            break
        
        # Upsample
        x_old = (np.arange(best_n) + 0.5) * (2.0 / best_n)
        x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = as_feasible(h_up, n_new)
        c_up = compute_c5(h_up)
        print(f"Upsampled to n={n_new}: C5={c_up:.6f}")
        
        # Binary search at new resolution
        bin_new = binary_from_continuous(h_up, n_new)
        if time.time() < deadline - 60.0:
            try:
                bh, bc = binary_hill_climb(bin_new, n_new, rng, deadline, max_rounds=50)
                if bc < best_c:
                    update_best(bh.astype(float), bc, n_new)
                    print(f"  Binary n={n_new}: C5={bc:.6f}")
            except Exception:
                pass
        
        # Continuous refinement
        if time.time() < deadline - 40.0:
            tl2 = min(120.0, max(20.0, deadline - time.time() - 40.0))
            try:
                h_r, c_r = continuous_refine(best_h, best_n, deadline, tl2)
                if c_r < best_c:
                    update_best(h_r, c_r, best_n)
                    print(f"  Cont refine n={best_n}: C5={c_r:.6f}")
            except Exception:
                pass
    
    # === Phase 4: Coordinate descent polish ===
    if time.time() < deadline - 20.0:
        try:
            h_cd, c_cd = coordinate_descent(best_h, best_n, deadline, max_iters=2000)
            if c_cd < best_c:
                update_best(h_cd, c_cd, best_n)
                print(f"Coord descent: C5={c_cd:.6f}")
        except Exception:
            pass
    
    # === Phase 5: Final SLSQP on best ===
    if time.time() < deadline - 15.0:
        tl_final = max(10.0, deadline - time.time() - 10.0)
        try:
            h_f, c_f = continuous_refine(best_h, best_n, deadline, min(60.0, tl_final))
            if c_f < best_c:
                update_best(h_f, c_f, best_n)
                print(f"Final refine: C5={c_f:.6f}")
        except Exception:
            pass
    
    # Final validation
    h_final = as_feasible(best_h, best_n)
    c_final = compute_c5(h_final)
    if c_final > best_c + 1e-6:
        h_final = best_h
        c_final = best_c
    final_n = int(best_n)
    
    if not is_valid_candidate(h_final):
        h_final = as_feasible(h_final, h_final.size)
        c_final = compute_c5(h_final)
        final_n = h_final.size
    
    print(f"\nFinal: n={final_n}, C5={c_final:.6f}")
    return h_final, float(c_final), int(final_n)
