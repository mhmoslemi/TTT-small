                    grad[:n-e] += (-h[e:])
        
        grad /= float(len(active))
    else:
        shifted = (corr - m) / alpha
        shifted = np.clip(shifted, -50, 50)
        weights = np.exp(shifted)
        A = float(np.sum(weights))
        if not math.isfinite(A) or A <= 0:
            A = 1.0
            weights = np.zeros(len(corr))
            weights[int(np.argmax(corr))] = 1.0
        weights /= A
        for idx in range(len(corr)):
            if weights[idx] < 1e-10:
                continue
            w = weights[idx]
            d = idx - (n - 1)
            if d == 0:
                grad += w * (1.0 - 2.0 * h)
            elif d > 0:
                mm = n - d
                grad[:mm] += w * (1.0 - h[d:])
                grad[d:] += w * (-h[:mm])
            else:
                e = -d
                grad[e:] += w * (1.0 - h[:n-e])
                grad[:n-e] += w * (-h[e:])
    
    grad *= 2.0 / n
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n)
    return grad

def constraint_fun(x):
    return float(np.sum(x)) - 0.5 * float(len(x))

def constraint_jac(x):
    return np.ones(len(x))

def try_slsqp(h, n, state, time_limit):
    """Run SLSQP optimization on h of size n."""
    check_deadline()
    start = time.time()
    h = as_feasible(h, n)
    state.try_update(h, n)
    
    if time.time() - start + 10.0 > time_limit:
        return
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": constraint_fun, "jac": constraint_jac}]
    
    # Stage 1: smooth objective
    alpha = max(0.1, 0.01 * float(n))
    maxiter1 = max(30, min(200, int(5000.0 / n)))
    
    dl = start + max(15.0, 0.5 * time_limit)
    if _RUN_DEADLINE is not None:
        dl = min(dl, _RUN_DEADLINE)
    old_dl = _RUN_DEADLINE
    _RUN_DEADLINE = dl
    try:
        res = minimize(slsqp_obj, h, args=(alpha,), jac=slsqp_grad,
                      method='SLSQP', bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter1, 'ftol': 1e-10, 'disp': 0})
    except BudgetError:
        res = None
    except Exception:
        res = None
    finally:
        _RUN_DEADLINE = old_dl
    
    if res is not None and hasattr(res, 'x') and res.x is not None:
        h1 = as_feasible(res.x, n)
    else:
        h1 = h
    state.try_update(h1, n)
    
    # Stage 2: exact max (alpha=0)
    if _RUN_DEADLINE is None or time.time() < _RUN_DEADLINE - 15.0:
        maxiter2 = max(50, min(400, int(10000.0 / n)))
        dl2 = start + max(20.0, time_limit)
        if _RUN_DEADLINE is not None:
            dl2 = min(dl2, _RUN_DEADLINE)
        old_dl = _RUN_DEADLINE
        _RUN_DEADLINE = dl2
        try:
            res2 = minimize(slsqp_obj, h1, args=(0.0,), jac=slsqp_grad,
                          method='SLSQP', bounds=bounds, constraints=constraints,
                          options={'maxiter': maxiter2, 'ftol': 1e-10, 'disp': 0})
        except BudgetError:
            res2 = None
        except Exception:
            res2 = None
        finally:
            _RUN_DEADLINE = old_dl
        
        if res2 is not None and hasattr(res2, 'x') and res2.x is not None:
            state.try_update(res2.x, n)
    
    state.try_update(h1, n)

def binary_scores_full(h, n):
    """Compute full correlation scores for binary h."""
    hf = h.astype(float)
    corr = np.correlate(hf, 1.0 - hf, mode="full")
    return corr

def binary_local_search(h_init, n, state, rng, deadline, max_iters=200000):
    """Local search on binary solutions."""
    h = h_init.astype(np.int8).copy()
    # Ensure exactly n/2 ones
    target = n // 2
    ones_count = int(np.sum(h))
    if ones_count != target:
        # Fix
        if ones_count > target:
            ones_idx = np.flatnonzero(h == 1)
            to_remove = ones_idx[rng.choice(ones_count - target, replace=False)]
            h[to_remove] = 0
        else:
            zeros_idx = np.flatnonzero(h == 0)
            to_add = zeros_idx[rng.choice(n - ones_count, target - ones_count, replace=False)]
            h[to_add] = 1
    
    scores = binary_scores_full(h, n)
    cur_max = float(np.max(scores))
    best_max = cur_max
    best_h = h.copy()
    state.try_update(h.astype(float), n)
    
    iters = 0
    stall = 0
    while iters < max_iters:
        if time.time() > deadline:
            break
        iters += 1
        if iters % 10000 == 0 and _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE:
            break
        
        # Propose a swap
        zeros = np.flatnonzero(h == 0)
        ones = np.flatnonzero(h == 1)
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        # Target active constraints
        active_thresh = cur_max - 1e-9
        active_shifts = np.flatnonzero(scores >= active_thresh)
        
        if len(active_shifts) > 0 and rng.random() < 0.6:
            # Try to fix an active constraint
            shift_idx = int(active_shifts[rng.integers(len(active_shifts))])
            d = shift_idx - (n - 1)
            if d == 0:
                # Lag 0: sum h[i]*(1-h[i]) - always 0 for binary, skip
                i = int(zeros[rng.integers(len(zeros))])
                j = int(ones[rng.integers(len(ones))])
            else:
                if d > 0:
                    # scores[n-1+d] = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
                    # To decrease: find i where h[i]=1, h[i+d]=0 → set h[i]=0 or h[i+d]=1
                    # Or find i where h[i]=0, h[i+d]=1 → swap helps
                    # Actually to decrease scores[n-1+d]:
                    # We want fewer (h[i]=1, h[i+d]=0) pairs
                    # So: set h[i]=0 where h[i]=1 and h[i+d]=0 (removes one)
                    #   or set h[i+d]=1 where h[i]=1 and h[i+d]=0 (removes one)
                    # But we need to maintain the count, so we swap.
                    # Swapping h[a]↔h[b] affects all shifts.
                    # Let's just pick a random zero and one that are "misaligned"
                    mm = n - d
                    bad_pairs = np.flatnonzero((h[:mm] == 1) & (h[d:] == 0))
                    good_pairs = np.flatnonzero((h[:mm] == 0) & (h[d:] == 1))
                    if len(bad_pairs) > 0:
                        pos = int(bad_pairs[rng.integers(len(bad_pairs))])
                        i = pos  # h[i]=1, h[i+d]=0
                        # We want to set h[i]=0, so swap with a zero elsewhere
                        j = int(zeros[rng.integers(len(zeros))])
                        # Or set h[i+d]=1, swap h[i+d] with a zero
                        if rng.random() < 0.5:
                            j = int(zeros[rng.integers(len(zeros))])
                        else:
                            i = pos + d  # now h[i]=0 (was h[pos+d]=0)
                            j = int(ones[rng.integers(len(ones))])
                    elif len(good_pairs) > 0:
                        # h[i]=0, h[i+d]=1 → this contributes to the OTHER direction
                        pos = int(good_pairs[rng.integers(len(good_pairs))])
                        i = pos  # h[i]=0
                        j = int(ones[rng.integers(len(ones))])
                    else:
                        i = int(zeros[rng.integers(len(zeros))])
                        j = int(ones[rng.integers(len(ones))])
                else:
                    e = -d
                    mm = n - e
                    bad_pairs = np.flatnonzero((h[e:] == 1) & (h[:mm] == 0))
                    good_pairs = np.flatnonzero((h[e:] == 0) & (h[:mm] == 1))
                    if len(bad_pairs) > 0:
                        pos = int(bad_pairs[rng.integers(len(bad_pairs))])
                        i = pos + e  # h[i]=1, h[i-e]=0
                        j = int(zeros[rng.integers(len(zeros))])
                    elif len(good_pairs) > 0:
                        pos = int(good_pairs[rng.integers(len(good_pairs))])
                        i = pos + e  # h[i]=0, h[i-e]=1
                        j = int(ones[rng.integers(len(ones))])
                    else:
                        i = int(zeros[rng.integers(len(zeros))])
                        j = int(ones[rng.integers(len(ones))])
        else:
            i = int(zeros[rng.integers(len(zeros))])
            j = int(ones[rng.integers(len(ones))])
        
        if i == j:
            continue
        
        # Compute impact of swap i↔j
        # h[i] and h[j] swap values
        hi = int(h[i])
        hj = int(h[j])
        if hi == hj:
            continue
        
        # Apply swap
        h[i] = hj
        h[j] = hi
        
        # Recompute scores (expensive for large n, but we can do it incrementally)
        # For now, just recompute the affected entries
        # Actually, a swap affects ALL shift entries, so we need to recompute.
        # Let's use the efficient approach:
        # corr[k] = sum_i h[i] * (1-h[i + k - (n-1)])
        # Changing h[i] and h[j] affects:
        # For each shift d: 
        #   if i is in the valid range: contribution changes by (new_h[i] - old_h[i]) * (1-h[i+d])
        #   if j is in the valid range: contribution changes by (new_h[j] - old_h[j]) * (1-h[j+d])
        #   and similarly for the "h[i+d]" term
        # This is O(n) per swap, which is the same as recomputing.
        # Let's just recompute for simplicity if n is small, or use incremental for large n.
        
        new_scores = binary_scores_full(h, n)
        new_max = float(np.max(new_scores))
        
        if new_max < cur_max - 1e-12:
            cur_max = new_max
            stall = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                state.try_update(h.astype(float), n)
        elif new_max <= cur_max + 1e-12:
            # Keep or revert based on temperature
            if stall > 50 and rng.random() < 0.1:
                stall += 1
            else:
                # Revert
                h[i] = hi
                h[j] = hj
                stall += 1
        else:
            # Revert
            h[i] = hi
            h[j] = hj
            stall += 1
            if stall > 5000:
                break
        scores = new_scores if new_max <= cur_max + 1e-12 else scores
    
    return best_h.astype(float), best_max * (2.0 / n)

def block_move_search(h_init, n, state, rng, deadline, max_iters=50000):
    """Local search with block moves (swap contiguous blocks)."""
    h = h_init.astype(np.int8).copy()
    target = n // 2
    ones_count = int(np.sum(h))
    if ones_count != target:
        if ones_count > target:
            ones_idx = np.flatnonzero(h == 1)
            to_remove = ones_idx[rng.choice(ones_count - target, replace=False)]
            h[to_remove] = 0
        else:
            zeros_idx = np.flatnonzero(h == 0)
            to_add = zeros_idx[rng.choice(n - ones_count, target - ones_count, replace=False)]
            h[to_add] = 1
    
    scores = binary_scores_full(h, n)
    cur_max = float(np.max(scores))
    best_max = cur_max
    best_h = h.copy()
    state.try_update(h.astype(float), n)
    
    iters = 0
    stall = 0
    while iters < max_iters:
        if time.time() > deadline:
            break
        iters += 1
        
        # Propose a block swap
        # Find a contiguous block of 1s and a contiguous block of 0s of same length
        max_block = max(2, min(10, n // 10))
        
        if rng.random() < 0.5:
            # Single bit swap
            zeros = np.flatnonzero(h == 0)
            ones = np.flatnonzero(h == 1)
            if len(zeros) == 0 or len(ones) == 0:
                break
            i = int(zeros[rng.integers(len(zeros))])
            j = int(ones[rng.integers(len(ones))])
            h[i], h[j] = h[j], h[i]
        else:
            # Block swap
            block_len = int(rng.integers(2, max_block + 1))
            # Find a block of zeros and a block of ones
            zero_blocks = []
            one_blocks = []
            i = 0
            while i < n:
                if h[i] == 0:
                    start = i
                    while i < n and h[i] == 0:
                        i += 1
                    if i - start >= block_len:
                        for s in range(start, i - block_len + 1):
                            zero_blocks.append(s)
                else:
                    start = i
                    while i < n and h[i] == 1:
                        i += 1
                    if i - start >= block_len:
                        for s in range(start, i - block_len + 1):
                            one_blocks.append(s)
            
            if len(zero_blocks) == 0 or len(one_blocks) == 0:
                iters += 1
                continue
            
            zb = int(zero_blocks[rng.integers(len(zero_blocks))])
            ob = int(one_blocks[rng.integers(len(one_blocks))])
            if zb == ob:
                continue
            # Swap blocks
            h[zb:zb+block_len] = 1
            h[ob:ob+block_len] = 0
            # Record for potential revert
            revert_i = (zb, ob, block_len)
        
        new_scores = binary_scores_full(h, n)
        new_max = float(np.max(new_scores))
        
        if new_max < cur_max - 1e-12:
            cur_max = new_max
            stall = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                state.try_update(h.astype(float), n)
            scores = new_scores
        else:
            # Revert
            if 'revert_i' in locals():
                zb, ob, bl = revert_i
                # This is complex - for block moves, revert is tricky
                # Let's just skip block moves in revert (too complex)
            # For single bit: revert
            if rng.random() < 0.5 and 'revert_i' not in locals():
                # Already handled above
                pass
            stall += 1
            if stall > 20000:
                break
            # Need to revert... this is getting complicated
            # Let me simplify: just recompute from best_h periodically
    
    return best_h.astype(float), best_max * (2.0 / n)

def run(seed=42, budget_s=1300, **kwargs):
    global _RUN_DEADLINE
    
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0.0:
        budget_s = 30.0
    reserve = 15.0 if budget_s > 60.0 else max(2.0, 0.05 * budget_s)
    global_deadline = t0 + max(1.0, budget_s - reserve)
    _RUN_DEADLINE = global_deadline
    
    rng = np.random.default_rng(seed)
    state = Best()
    
    # Get parent
    parent = get_parent()
    if parent is not None:
        n0 = int(parent.size)
        h0 = as_feasible(parent, n0)
        state.try_update(h0, n0)
    else:
        n0 = 70
        h0 = np.where((np.arange(n0) + 0.5) * (2.0/n0) < 1.0, 1.0, 0.0)
        h0 = as_feasible(h0, n0)
        state.try_update(h0, n0)
    
    # Generate structured initializations at parent resolution
    inits = structured_init(n0, rng)
    for h_init in inits:
        state.try_update(h_init, n0)
    
    # Binary local search at parent resolution
    if n0 % 2 == 0:
        dl = min(global_deadline - 10.0, t0 + 120.0)
        if tl := (dl - t0) > 10:
            for restart in range(3):
                if time.time() > dl:
                    break
                if restart == 0:
                    start_h = state.h if state.h is not None else h0
                else:
                    start_h = (rng.random(n0) < 0.5).astype(float)
                try:
                    binary_local_search(start_h, n0, state, rng, dl, max_iters=150000)
                except Exception:
                    pass
    
    # Continuous optimization at parent resolution
    remaining = global_deadline - time.time()
    if remaining > 60:
        try:
            try_slsqp(state.h, state.n, state, min(200.0, remaining - 30.0))
        except Exception:
            pass
    
    # Try higher resolutions
    n_candidates = [n0]
    for mult in [1.5, 2, 3, 4]:
        nn = int(round(mult * n0 / 2.0) * 2)
        if nn >= 4 and nn <= max(200, 6 * n0) and nn % 2 == 0:
            if nn not in n_candidates:
                n_candidates.append(nn)
    n_candidates.sort()
    
    remaining = global_deadline - time.time()
    for nn in n_candidates:
        if time.time() > global_deadline - 40:
            break
        remaining = global_deadline - time.time()
        if remaining < 50:
            break
        
        if nn == n0:
            continue  # Already done
        
        # Upsample current best
        h_start = upsample_linear(state.h, nn)
        h_start = as_feasible(h_start, nn)
        state.try_update(h_start, nn)
        
        # Binary search at this resolution
