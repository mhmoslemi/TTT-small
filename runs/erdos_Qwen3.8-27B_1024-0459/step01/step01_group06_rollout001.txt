import time
import math
import numpy as np
from scipy.optimize import minimize


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    if h.size < n:
        h = np.pad(h, (0, n - h.size), mode="constant")
    elif h.size > n:
        h = h[:n]
    h = np.clip(h, 0.0, 1.0)

    target = 0.5 * float(n)
    if n <= 0:
        return h

    lo = float(np.min(h)) - 1.0 - 1e-12
    hi = float(np.max(h)) + 1.0 + 1e-12
    mid = 0.5 * (lo + hi)

    for _ in range(90):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    p = np.clip(h - mid, 0.0, 1.0)

    for _ in range(12):
        s = math.fsum(p)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            idx = int(np.argmin(p))
            room = 1.0 - p[idx]
            if room < 0.0:
                room = 0.0
            add = min(diff, room)
            p[idx] += add
        else:
            idx = int(np.argmax(p))
            room = p[idx]
            if room < 0.0:
                room = 0.0
            sub = min(-diff, room)
            p[idx] -= sub
    return p


def exact_sum_adjust(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * float(n)
    for _ in range(8):
        s = math.fsum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            idx = int(np.argmin(h))
            add = min(diff, max(0.0, 1.0 - h[idx]))
            h[idx] += add
        else:
            idx = int(np.argmax(h))
            sub = min(-diff, max(0.0, h[idx]))
            h[idx] -= sub
    return h


def compute_c5_values(h):
    n = int(len(h))
    c = np.correlate(h, 1.0 - h, mode="full")
    return c, float(np.max(c) * (2.0 / float(n)))


def compute_c5(h):
    return compute_c5_values(h)[1]

    return float(np.sum(h) - 0.5 * float(n))


def sum_jac(h, n):

    return b


def make_eq_cons(n):
    return [{"type": "eq", "fun": sum_fun, "jac": sum_jac, "args": (int(n),)}]


def make_t_bounds(n):
    b = []
    for _ in range(int(n)):
        b.append((0.0, 1.0))
    b.append((0.0, 2.0))
    return b


def soft_obj(h, n, beta):
    n = int(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    scaled = float(beta) * c
    m = float(np.max(scaled))
    return float((m + math.log(float(np.sum(np.exp(scaled - m))))) / float(beta) * (2.0 / float(n)))


def soft_grad(h, n, beta):
    n = int(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    scaled = float(beta) * c
    m = float(np.max(scaled))
    w = np.exp(scaled - m)
    s = float(np.sum(w))
    if s <= 0.0:
        w = np.zeros_like(w)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= s

    dx = 2.0 / float(n)
    g = np.zeros(n, dtype=float)
    base = n - 1

    for idx in range(len(c)):
        wg = float(w[idx])
        if wg <= 1e-14:
            continue
        lag = int(idx) - base
        if lag >= 0:
            mlen = n - lag
            if mlen > 0:
                g[lag:] += wg * (1.0 - h[:mlen])
                g[:mlen] -= wg * h[lag:]
        else:
            d = -lag
            mlen = n - d
            if mlen > 0:
                g[:mlen] += wg * (1.0 - h[d:])
                if d < n:
                    g[d:] -= wg * h[:mlen]
    return g * dx


def hard_obj(h, n):
    return compute_c5(h)


def hard_grad(h, n):
    n = int(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    k = int(np.argmax(c))
    lag = k - (n - 1)
    dx = 2.0 / float(n)
    g = np.zeros(n, dtype=float)
    if lag >= 0:
        mlen = n - lag
        if mlen > 0:
            g[lag:] += (1.0 - h[:mlen])
            g[:mlen] -= h[lag:]
    else:
        d = -lag
        mlen = n - d
        if mlen > 0:
            g[:mlen] += (1.0 - h[d:])
            if d < n:
                g[d:] -= h[:mlen]
    return g * dx


def slsqp_soft_call(h0, n, beta, maxiter, bounds, cons):
    res = minimize(
        soft_obj,
        x0=h0,
        jac=soft_grad,
        args=(int(n), float(beta)),
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": int(maxiter), "ftol": 1e-12},
    )
    x = np.asarray(res.x, dtype=float)
    return project_feasible(np.clip(x, 0.0, 1.0), int(n))


def active_t_obj(x, n, active):
    return float(x[-1])


def active_t_jac(x, n, active):
    n = int(n)
    j = np.zeros(n + 1, dtype=float)
    j[-1] = 1.0
    return j


def active_t_cons_fun(x, n, active):
    n = int(n)
    h = x[:n]
    t = float(x[-1])
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    act = [int(k) for k in active]
    return t - dx * c[act]


def active_t_cons_jac(x, n, active):
    n = int(n)
    h = x[:n]
    dx = 2.0 / float(n)
    m = int(len(active))
    jac = np.zeros((m, n + 1), dtype=float)
    base = n - 1

    for r, k in enumerate(active):
        k = int(k)
        lag = k - base
        g = np.zeros(n, dtype=float)
        if lag >= 0:
            mlen = n - lag
            if mlen > 0:
                g[lag:] += (1.0 - h[:mlen])
                g[:mlen] -= h[lag:]
        else:
            d = -lag
            mlen = n - d
            if mlen > 0:
                g[:mlen] += (1.0 - h[d:])
                if d < n:
                    g[d:] -= h[:mlen]
        jac[r, :n] = -dx * g
        jac[r, n] = 1.0
    return jac


def select_active_list(c, K=12):
    c = np.asarray(c, dtype=float)
    M = int(len(c))
    K = max(1, int(K))
    order = np.argsort(c)
    top = order[-K:]
    rawmax = float(np.max(c))
    tol = max(1e-10, 1e-6 * max(1.0, rawmax))
    near = np.where(c >= rawmax - tol)[0]
    arr = np.union1d(top, near)
    if arr.size > 30:
        arr = order[-30:]
    return arr


def active_slsqp_call(h, n, active, t0, bounds_t):
    n = int(n)
    x0 = np.concatenate((np.asarray(h, dtype=float), np.array([t0], dtype=float)))
    cons = [{"type": "ineq", "fun": active_t_cons_fun, "jac": active_t_cons_jac, "args": (n, active)}]
    res = minimize(
        active_t_obj,
        x0,
        jac=active_t_jac,
        args=(n, active),
        method="SLSQP",
        bounds=bounds_t,
        constraints=cons,
        options={"maxiter": 300, "ftol": 1e-12},
    )
    h_new = project_feasible(np.clip(np.asarray(res.x[:n], dtype=float), 0.0, 1.0), n)
    return h_new, float(res.x[-1])


def optimize_active_slsqp(h, n, time_limit):
    t0 = time.time()
    deadline = t0 + float(max(0.0, time_limit))
    n = int(n)
    best = np.asarray(h, dtype=float).copy()
    best_c = float(compute_c5(best))
    bounds_t = make_t_bounds(n)
    no_improve = 0

    while time.time() < deadline:
        c, raw = compute_c5_values(best)
        active = select_active_list(c, 12)
        t0active = float(raw + 1e-8)
        h_new, _ = active_slsqp_call(best, n, active, t0active, bounds_t)
        h_new = project_feasible(np.clip(h_new, 0.0, 1.0), n)
        c_new = float(compute_c5(h_new))
        if c_new < best_c - 1e-10:
            best = h_new
            best_c = c_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                break
    return best, float(best_c)


def line_roots(A, B, C, lo, hi):
    eps = 1e-13
    if abs(A) < eps:
        if abs(B) < eps:
            return []
        r = -C / B
        if lo < r < hi and math.isfinite(r):
            return [float(r)]
        return []

    disc = B * B - 4.0 * A * C
    if disc < -1e-10:
        return []
    if disc < 0.0:
        disc = 0.0
    sq = math.sqrt(disc)
    out = []
    denom = 2.0 * A
    r1 = (-B - sq) / denom
    r2 = (-B + sq) / denom
    if lo < r1 < hi and math.isfinite(r1):
        out.append(float(r1))
    if lo < r2 < hi and math.isfinite(r2):
        out.append(float(r2))
    return out


def eval_line_delta(c, L, Q, delta):
    return float(np.max(c + delta * L + delta * delta * Q))


def line_best_delta(c, L, Q, lo, hi):
    M = int(len(c))
    if hi <= lo or M <= 0:
        return 0.0, float(np.max(c))

    cands = [float(lo), float(hi)]
    if lo < 0.0 < hi:
        cands.append(0.0)

    cands.extend(np.linspace(lo, hi, 81).tolist())

    qpos = Q > 1e-11
    if np.any(qpos):
        verts = -L[qpos] / (2.0 * Q[qpos])
        for v in verts:
            v = float(v)
            if lo < v < hi and math.isfinite(v):
                cands.append(v)

    K = min(24, M)
    if M >= 2:
        order = np.argsort(c)
        idxs = order[-K:]
        nidx = len(idxs)
        for ii in range(nidx):
            i = int(idxs[ii])
            ci = float(c[i])
            Li = float(L[i])
            Qi = float(Q[i])
            for jj in range(ii + 1, nidx):
                j = int(idxs[jj])
                A = Qi - float(Q[j])
                B = Li - float(L[j])
                C = ci - float(c[j])
                roots = line_roots(A, B, C, lo, hi)
                cands.extend(roots)

    cands = [float(x) for x in cands if math.isfinite(x)]
    arr = np.array(cands, dtype=float)
    if arr.size == 0:
        return 0.0, float(np.max(c))

    mat = c[None, :] + arr[:, None] * L[None, :] + (arr * arr)[:, None] * Q[None, :]
    vals = mat.max(axis=1)

    bi = int(np.argmin(vals))
    best = float(arr[bi])
    bestv = float(vals[bi])

    step = max((hi - lo) / 90.0, 1e-12)
    a = max(lo, best - 2.0 * step)
    b = min(hi, best + 2.0 * step)
    if b - a > 1e-14:
        for _ in range(30):
            m1 = a + (b - a) * 0.38196601125
            m2 = b - (b - a) * 0.38196601125
            f1 = eval_line_delta(c, L, Q, m1)
            f2 = eval_line_delta(c, L, Q, m2)
            if f1 <= f2:
                b = m2
            else:
                a = m1
        mid = 0.5 * (a + b)
        fm = eval_line_delta(c, L, Q, mid)
        if fm < bestv:
            best = float(mid)
            bestv = float(fm)

    if best > hi:
        best = hi
    if best < lo:
        best = lo
    return best, bestv


def pair_line_coeffs(h, c, a, b):
    n = int(len(h))
    M = 2 * n - 1
    L = np.zeros(M, dtype=float)
    Q = np.zeros(M, dtype=float)

    ha = float(h[a])
    hb = float(h[b])
    lo = max(-ha, hb - 1.0)
    hi = min(1.0 - ha, hb)
    if hi <= lo + 1e-15:
        return None

    for s in range(-(n - 1), n):
        idx = s + (n - 1)
        terms = []

        if s >= 0:
            lag = s
            mlen = n - lag
            if mlen > 0:
                if a >= lag and a - lag < mlen:
                    terms.append((a, a - lag))
                if b >= lag and b - lag < mlen:
                    terms.append((b, b - lag))
                if a < mlen and a + lag < n:
                    terms.append((a + lag, a))
                if b < mlen and b + lag < n:
                    terms.append((b + lag, b))
        else:
            d = -s
            mlen = n - d
            if mlen > 0:
                if a < mlen and a + d < n:
                    terms.append((a, a + d))
                if b < mlen and b + d < n:
                    terms.append((b, b + d))
                if a >= d and a - d < mlen:
                    terms.append((a - d, a))
                if b >= d and b - d < mlen:
                    terms.append((b - d, b))

        for u, v in terms:
            dup = False
            for uu, vv in terms[:terms.index((u, v))]:
                if uu == u and vv == v:
                    dup = True
                    break
            if dup:
                continue

            du = 1.0
            if u == b:
                du = -1.0
            elif u != a:
                du = 0.0

            dv = 1.0
            if v == b:
                dv = -1.0
            elif v != a:
                dv = 0.0

            if du != 0.0 or dv != 0.0:
                L[idx] += du * (1.0 - h[v]) - h[u] * dv
                Q[idx] += -du * dv

    return L, Q, lo, hi


def top_gradient(h, c, K=8):
    n = int(len(h))
    K = max(1, int(K))
    order = np.argsort(c)[-K:]
    g = np.zeros(n, dtype=float)
    w = 1.0 / float(len(order))
    base = n - 1

    for k in order:
        k = int(k)
        lag = k - base
        if lag >= 0:
            mlen = n - lag
            if mlen > 0:
                g[lag:] += w * (1.0 - h[:mlen])
                g[:mlen] -= w * h[lag:]
        else:
            d = -lag
            mlen = n - d
            if mlen > 0:
                g[:mlen] += w * (1.0 - h[d:])
                if d < n:
                    g[d:] -= w * h[:mlen]
    return g


def coordinate_descent_run(h0, c0, seed, time_limit):
    h = np.asarray(h0, dtype=float).copy()
    c = np.asarray(c0, dtype=float).copy()
    n = int(len(h))
    t0 = time.time()
    deadline = t0 + float(max(0.0, time_limit))
    rng = np.random.RandomState(int(seed))

    best_h = h.copy()
    best_c = float(compute_c5(h))
    last_best = t0
    scale = 0.02
    counter = 0

    while time.time() < deadline:
        if time.time() - last_best > 30.0 and time.time() < deadline - 30.0:
            h = project_feasible(h + scale * rng.randn(n), n)
            c = np.correlate(h, 1.0 - h, mode="full")
            scale = min(0.25, scale * 1.35)

        counter += 1
        if counter % 3 == 0:
            g = top_gradient(h, c, 8)
            a = int(np.argmin(g))
            b = int(np.argmax(g))
            tries = 0
            while a == b and tries < 12:
                a = int(rng.randint(n))
                b = int(rng.randint(n))
                tries += 1
        else:
            a = int(rng.randint(n))
            b = int(rng.randint(n))
            tries = 0
            while a == b and tries < 12:
                a = int(rng.randint(n))
                b = int(rng.randint(n))
                tries += 1

        res = pair_line_coeffs(h, c, a, b)
        if res is None:
            counter += 1
            if counter % 25 == 0:
                c = np.correlate(h, 1.0 - h, mode="full")
            continue

        L, Q, lo, hi = res
        if hi <= lo + 1e-13:
            continue

        cur_raw = float(np.max(c))
        delta, val = line_best_delta(c, L, Q, lo, hi)

        if val < cur_raw - 1e-11:
            ha_new = float(h[a]) + float(delta)
            hb_new = float(h[b]) - float(delta)
            if ha_new < -1e-12 or ha_new > 1.0 + 1e-12 or hb_new < -1e-12 or hb_new > 1.0 + 1e-12:
                continue

            h[a] = ha_new
            h[b] = hb_new
            c = c + float(delta) * L + float(delta) * float(delta) * Q
            new_raw = float(np.max(c))

            if new_raw < cur_raw - 1e-12:
                new_c = float(new_raw * (2.0 / float(n)))
                if new_c < best_c - 1e-12:
                    best_h = h.copy()
                    best_c = float(new_c)
                    last_best = time.time()
                    scale = 0.02

            if counter % 25 == 0:
                c = np.correlate(h, 1.0 - h, mode="full")

    try:
        _, final_c = compute_c5_values(best_h)
    except Exception:
        final_c = best_c
    return best_h, float(final_c)


def optimize_continuous(h0, n, seed, time_limit):
    t_start = time.time()
    n = int(n)
    time_limit = float(max(0.0, time_limit))

    h = project_feasible(h0, n)
    _, cur = compute_c5_values(h)
    best_h = h.copy()
    best_c = float(cur)

    bounds = make_bounds(n)
    cons = make_eq_cons(n)

    if n <= 100:
        betas = [4.0, 12.0, 50.0, 200.0, 700.0]
        maxiter = 300
    elif n <= 150:
        betas = [5.0, 20.0, 100.0, 350.0]
        maxiter = 200
    else:
        betas = [5.0, 20.0, 80.0]
        maxiter = 120

    for beta in betas:
        if time.time() - t_start > time_limit - 30.0:
            break
        try:
            h1 = slsqp_soft_call(best_h, n, beta, maxiter, bounds, cons)
            c1 = float(compute_c5(h1))
            if c1 < best_c:
                best_h = h1
                best_c = c1
        except Exception:
            pass

    if time.time() - t_start < time_limit - 30.0:
        try:
            h1, c1 = optimize_active_slsqp(best_h, n, min(50.0, max(5.0, time_limit - 30.0)))
            if c1 < best_c:
                best_h = h1
                best_c = float(c1)
        except Exception:
            pass

    rem = time_limit - (time.time() - t_start)
    if rem > 15.0:
        try:
            c_raw, _ = compute_c5_values(best_h)
            h1, c1 = coordinate_descent_run(best_h, c_raw, seed + 29, min(rem - 5.0, 200.0))
            if c1 < best_c:
                best_h = h1
                best_c = float(c1)
        except Exception:
            pass

    rem = time_limit - (time.time() - t_start)
    if rem > 20.0:
        try:
            h1, c1 = optimize_active_slsqp(best_h, n, min(20.0, rem - 10.0))
            if c1 < best_c:
                best_h = h1
                best_c = float(c1)
        except Exception:
            pass

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, n)
    best_h = exact_sum_adjust(best_h, n)
    _, best_c = compute_c5_values(best_h)
    return best_h, float(best_c)


def interpolate_h(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float).ravel()
    n_src = int(len(h_src))
    n_dst = int(n_dst)
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 0:
        return np.full(n_dst, 0.5, dtype=float)
    if n_src == 1:
        return np.full(n_dst, float(h_src[0]), dtype=float)
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False) + 1.0 / float(n_src)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False) + 1.0 / float(n_dst)
    return np.interp(x_dst, x_src, h_src)


def bits_from_h(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float).ravel()
    if n <= 0:
        return 0
    if len(h) != n:
        h = np.interp(np.linspace(0.0, 1.0, n), np.linspace(0.0, 1.0, len(h)), h)
    order = np.argsort(h)
    bits = 0
    start = n - (n // 2)
    for i in range(start, n):
        bits |= (1 << int(order[i]))
    return bits


def random_balanced_bits(n, rng):
    n = int(n)
    perm = rng.permutation(n)
    bits = 0
    for i in perm[: n // 2]:
        bits |= (1 << int(i))
    return bits


def random_fixed_bits(bits, n, shifts, k, rng):
    bits = int(bits)
    for _ in range(int(k)):
        ok = False
        for _ in range(20):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i != j and (((bits >> i) & 1) ^ ((bits >> j) & 1)):
                bits ^= shifts[i] ^ shifts[j]
                ok = True
                break
        if not ok:
            break
    return bits


def bin_masks(n):
    n = int(n)
    masks = [0] * n
    for l in range(1, n):
        masks[l] = (1 << (n - l)) - 1
    return masks


def binary_max(bits, n, masks, limit=None):
    n = int(n)
    bits = int(bits)
    m = 0
    for l in range(1, n):
        mask = masks[l]
        hi = bits & mask
        sh = (bits >> l) & mask
        c10 = (hi & ~sh).bit_count()
        if c10 > m:
            m = c10
            if limit is not None and m >= int(limit):
                return m
        c01 = (~hi & sh).bit_count()
        if c01 > m:
            m = c01
            if limit is not None and m >= int(limit):
                return m
    return m


def bits_to_array(bits, n):
    n = int(n)
    arr = np.zeros(n, dtype=float)
    for i in range(n):
        if (bits >> i) & 1:
            arr[i] = 1.0
    return arr


def binary_search_run(n, seed, time_limit, targetC, start_bits=None):
    n = int(n)
    if n < 8 or (n % 2 != 0):
        return None, None, None
    t0 = time.time()
    deadline = t0 + float(max(0.0, time_limit))

    masks = bin_masks(n)
    shifts = []
    for i in range(n):
        shifts.append(1 << i)

    rng = np.random.RandomState(int(seed))
    if start_bits is None:
        best_bits = random_balanced_bits(n, rng)
    else:
        best_bits = int(start_bits)
        if bin_zones = None:
            pass

    bestM = binary_max(best_bits, n, masks)

    if targetC is None:
        targetM = -1
    else:
        targetM = int(math.floor(float(targetC) * n / 2.0 - 1e-12))
        if targetM < 0:
            targetM = -1

    pert_sizes = [0, 1, 2, 3, 5, 8, 13, 21]
    pi = 0

    while time.time() < deadline:
        psize = pert_sizes[pi % len(pert_sizes)]
        pi += 1

        work = int(best_bits)
        if psize > 0:
            work = random_fixed_bits(work, n, shifts, psize, rng)

        curM = binary_max(work, n, masks)
        fail = 0

        while time.time() < deadline and fail < 6000:
            i = -1
            j = -1
            for _ in range(25):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i != j and (((work >> i) & 1) ^ ((work >> j) & 1)):
                    break
            if i < 0 or j < 0:
                fail += 1
                continue

            cand = work ^ shifts[i] ^ shifts[j]
            newM = binary_max(cand, n, masks, limit=curM)

            if newM < curM:
                work = cand
                curM = int(newM)
                fail = 0
                if curM < bestM:
                    bestM = curM
                    best_bits = work
                    if targetM >= 0 and bestM <= targetM:
                        break
            else:
                fail += 1

        if targetM >= 0 and bestM <= targetM:
            break

    h = bits_to_array(best_bits, n)
    c5 = 2.0 * float(bestM) / float(n)
    return h, float(c5), int(best_bits)


def choose_ns(candidates, best_n, max_count=3):
    seen = []
    for x in candidates:
        x = int(x)
        if x not in seen:
            seen.append(x)

    valid = []
    for x in seen:
        if 16 <= x <= 300:
            valid.append(x)

    chosen = []
    while valid and len(chosen) < max_count:
        best_d = 1e100
        best_x = valid[0]
        for x in valid:
            d = abs(x - int(best_n))
            if d < best_d:
                best_d = d
                best_x = x
        chosen.append(best_x)
        valid.remove(best_x)
    return chosen


def make_smooth_start(n, seed):
    n = int(n)
    rng = np.random.RandomState(int(seed))
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / float(n)
    h = 0.5 * np.ones(n, dtype=float)

    for _ in range(4):
        f = float(rng.randint(1, 9))
        amp = 0.12 + 0.35 * float(rng.rand())
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        h = h + amp * np.sin(2.0 * np.pi * f * x / 2.0 + ph)

    h = 0.5 + 0.5 * np.tanh(4.0 * (h - 0.5))
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    t_global = time.time()
    budget_s = float(max(60.0, budget_s))

    parent = None
    try:
        pv = initial_h_values
        arr = np.asarray(pv, dtype=float).ravel()
        if arr.size > 0:
            parent = arr
    except Exception:
        parent = None

    if parent is not None:
        best_n = int(len(parent))
        best_h = project_feasible(parent, best_n)
    else:
        best_n = 95
        best_h = np.full(best_n, 0.5, dtype=float)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = exact_sum_adjust(best_h, best_n)
    best_c = float(compute_c5(best_h))

    rem = budget_s - (time.time() - t_global)
    if rem > 80.0:
        init_limit = float(min(500.0, max(40.0, 0.32 * budget_s)))
        try:
            h1, c1 = optimize_continuous(best_h, best_n, seed, init_limit)
            if c1 < best_c:
                best_h = h1
                best_c = float(c1)
                best_n = int(len(h1))
        except Exception:
            pass

    binary_starts = []
    rem = budget_s - (time.time() - t_global)
    if rem > 200.0:
        cand = []
        base = best_n if (best_n % 2 == 0) else best_n + 1
        cand.append(base)
        cand.append(best_n + 1)
        dbl = 2 * best_n
        cand.append(dbl)
        for nx in [96, 112, 128, 144, 160, 190, 192, 224, 256]:
            cand.append(nx)

        chosen = choose_ns(cand, best_n, max_count=3)
        if chosen:
            per = max(10.0, min(55.0, (rem - 80.0) / float(len(chosen))))
            for n2 in chosen:
                rem = budget_s - (time.time() - t_global)
                if rem < 30.0 or per < 8.0:
                    break
                try:
                    start_h = interpolate_h(best_h, int(n2))
                    start_bits = bits_from_h(start_h, int(n2))
                    hb, cb, _ = binary_search_run(int(n2), seed + 101, per, best_c, start_bits=start_bits)
                    if hb is not None and cb is not None:
                        binary_starts.append((hb, int(n2), float(cb)))
                        if cb < best_c:
                            best_h = hb
                            best_n = int(n2)
                            best_c = float(cb)
                except Exception:
                    pass

    if binary_starts:
        binary_starts.sort(key=None)
        top_binary = []
        for item in binary_starts:
            if len(top_binary) < 2:
                top_binary.append(item)
            else:
                worst = 0
                wc = top_binary[0][2]
                for idx, it in enumerate(top_binary):
                    if it[2] > wc:
                        wc = it[2]
                        worst = idx
                if item[2] < wc:
                    top_binary[worst] = item

        for hb, nb, cb in top_binary:
            rem = budget_s - (time.time() - t_global)
            if rem < 120.0:
                break
            if cb <= best_c + 0.02:
                try:
                    h1, c1 = optimize_continuous(hb, nb, seed + nb + 7, min(110.0, rem - 100.0))
                    if c1 < best_c:
                        best_h = h1
                        best_n = nb
                        best_c = float(c1)
                except Exception:
                    pass

    rem = budget_s - (time.time() - t_global)
    if rem > 220.0:
        n_smooth = best_n if best_n >= 60 else 95
        for st in [seed + 17, seed + 31]:
            rem = budget_s - (time.time() - t_global)
            if rem < 130.0:
                break
            try:
                h2 = make_smooth_start(n_smooth, st)
                h1, c1 = optimize_continuous(h2, n_smooth, st, min(110.0, rem - 100.0))
                if c1 < best_c:
                    best_h = h1
                    best_n = n_smooth
                    best_c = float(c1)
            except Exception:
                pass

    rem = budget_s - (time.time() - t_global)
    if rem > 250.0 and best_n < 256:
        n2 = 2 * best_n
        if 128 <= n2 <= 288:
            try:
                h2 = project_feasible(interpolate_h(best_h, n2), n2)
                h1, c1 = optimize_continuous(h2, n2, seed + 277, min(210.0, rem - 120.0))
                if c1 < best_c:
                    best_h = h1
                    best_n = n2
                    best_c = float(c1)
            except Exception:
                pass

    rem = budget_s - (time.time() - t_global)
    if rem > 35.0:
        try:
            h1, c1 = optimize_active_slsqp(best_h, best_n, min(25.0, rem - 15.0))
            if c1 < best_c:
                best_h = h1
                best_c = float(c1)
        except Exception:
            pass

    rem = budget_s - (time.time() - t_global)
    if rem > 25.0:
        try:
            c_raw, _ = compute_c5_values(best_h)
            h1, c1 = coordinate_descent_run(best_h, c_raw, seed + 911, min(18.0, rem - 10.0))
            if c1 < best_c:
                best_h = h1
                best_c = float(c1)
        except Exception:
            pass

    best_h = np.asarray(best_h, dtype=float).ravel()
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, best_n)
    best_h = exact_sum_adjust(best_h, best_n)
    _, best_c = compute_c5_values(best_h)
    best_c = float(best_c)

    try:
        val = evaluate_erdos_solution(best_h, best_n)
        if val is not None:
            val = float(val)
            if 0.0 < val < 1.0 and val < best_c:
                best_c = val
    except Exception:
        try:
            val = evaluate_erdos_solution(best_h)
            if val is not None:
                val = float(val)
                if 0.0 < val < 1.0 and val < best_c:
                    best_c = val
        except Exception:
            pass

    return (best_h, best_c, int(best_n))
