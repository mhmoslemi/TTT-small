    ops, stall = 0, 0
    while ops < max_ops and time.time() < deadline:
        if _DEADLINE is not None and time.time() > _DEADLINE:
            break
        ops += 1
        improved = False
        act = np.flatnonzero(scores >= cur - 1e-9)
        if act.size > 0:
            k = int(act[int(rng.integers(act.size))])
            if k != 0:
                if k % 2 == 1:
                    d = (k+1)//2
                    mask = (hb[:n-d]==1) & (hb[d:]==0)
                else:
                    d = k//2
                    mask = (hb[:n-d]==0) & (hb[d:]==1)
                cand = np.nonzero(mask)[0]
                if cand.size > 0:
                    p = int(cand[int(rng.integers(cand.size))])
                    j = p + d
                    if hb[p] != hb[j]:
                        hi, hj = hb[p], hb[j]
                        binary_apply(scores, hb, p, 1.0-hi, n)
                        binary_apply(scores, hb, j, 1.0-hj, n)
                        nm = float(np.max(scores))
                        if nm + 1e-9 < cur:
                            cur = nm
                            stall = 0
                            improved = True
                            if cur*(2.0/n) < best_c:
                                best_c = cur*(2.0/n)
                                best_h = hb.copy()
                                if state is not None:
                                    state.update(best_h, best_c, n)
                            zeros = np.nonzero(hb==0)[0]
                            ones = np.nonzero(hb==1)[0]
                        else:
                            binary_apply(scores, hb, p, hi, n)
                            binary_apply(scores, hb, j, hj, n)
                            stall += 1
                        continue
        if len(zeros) > 0 and len(ones) > 0:
            i = int(zeros[int(rng.integers(len(zeros)))])
            j = int(ones[int(rng.integers(len(ones)))])
            hi, hj = hb[i], hb[j]
            binary_apply(scores, hb, i, hj, n)
            binary_apply(scores, hb, j, hi, n)
            nm = float(np.max(scores))
            if nm + 1e-9 < cur:
                cur = nm
                stall = 0
                improved = True
                if cur*(2.0/n) < best_c:
                    best_c = cur*(2.0/n)
                    best_h = hb.copy()
                    if state is not None:
                        state.update(best_h, best_c, n)
                zeros = np.nonzero(hb==0)[0]
                ones = np.nonzero(hb==1)[0]
            else:
                binary_apply(scores, hb, i, hi, n)
                binary_apply(scores, hb, j, hj, n)
                stall += 1
        if stall > stall_limit:
            break
    if state is not None:
        state.update(best_h, best_c, n)
    return best_h, best_c

def binary_apply(scores, h, idx, new_val, n):
    i = idx
    for d in range(1, n):
        pidx, midx = 2*d-1, 2*d
        if i - d >= 0:
            hj = h[i-d]
            if new_val > 0.5:
                scores[pidx] -= hj
                scores[midx] += 1.0 - hj
            else:
                scores[pidx] += hj
                scores[midx] -= 1.0 - hj
        if i + d < n:
            hj = h[i+d]
            if new_val > 0.5:
                scores[pidx] += 1.0 - hj
                scores[midx] -= hj
            else:
                scores[pidx] -= 1.0 - hj
                scores[midx] += hj
    h[i] = new_val

def coord_descent(h, n, state, max_rounds=200, deadline=None):
    h = as_feasible(h, n)
    for r in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        if _DEADLINE is not None and time.time() > _DEADLINE:
            break
        improved = False
        idx = np.random.randint(n)
        old = h[idx]
        for delta in (-0.05, -0.02, -0.01, 0.01, 0.02, 0.05):
            nv = old + delta
            if nv < 0 or nv > 1:
                continue
            h[idx] = nv
            hf = as_feasible(h, n)
            c = compute_c5(hf)
            if c < state.c - 1e-10:
                h = hf
                state.update(h, c, n)
                improved = True
                break
        if not improved:
            h[idx] = old
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _DEADLINE, _rng_bin
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0:
        budget_s = 30.0
    reserve = min(30.0, max(5.0, 0.05 * budget_s))
    deadline = t0 + max(1.0, budget_s - reserve)
    _DEADLINE = deadline
    rng = np.random.default_rng(seed)
    _rng_bin = rng

    # Get parent
    parent = None
    try:
        p = np.asarray(initial_h_values, dtype=float)
        if p.ndim == 1 and p.size >= 4:
            parent = p.copy()
    except Exception:
        pass

    if parent is not None:
        n0 = len(parent)
        h0 = as_feasible(parent, n0)
    else:
        n0 = 70
        h0 = make_synthetic(n0, rng)

    c0 = compute_c5(h0)
    state = State(h0, c0, n0)
    print(f"Initial: n={n0}, C5={c0:.8f}", flush=True)

    # Phase 1: Binary search at n0 (quick, ~30s)
    if n0 % 2 == 0 and time.time() < deadline - 60:
        blim = min(30.0, deadline - time.time())
        bh, bc = binary_search(h0, n0, state, rng, blim, max_ops=40000, stall_limit=4000)
        if bh is not None and bc < c0:
            c0 = bc
            print(f"Binary n={n0}: C5={c0:.8f}", flush=True)

    # Phase 2: SLSQP at n0 with alpha schedule
    n = n0
    h_work = state.h.copy() if state.n == n0 else h0

    alpha_schedule = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.0]
    iters_schedule = [80, 80, 80, 80, 100, 100, 120, 150]
    total_slsqp_time = min(500.0, deadline - time.time() - 100.0)
    per_stage = total_slsqp_time / len(alpha_schedule)

    for i, (al, mi) in enumerate(zip(alpha_schedule, iters_schedule)):
        if time.time() > deadline - 40:
            break
        stage_dl = time.time() + max(15.0, per_stage)
        if _DEADLINE is not None:
            stage_dl = min(stage_dl, _DEADLINE - 5.0)
        h_work = slsqp_stage(h_work, n, state, al, mi, stage_dl)
        print(f"SLSQP n={n} alpha={al}: C5={state.c:.8f}", flush=True)

    # Phase 3: Coordinate descent refinement
    if time.time() < deadline - 30:
        h_cd = state.h.copy()
        coord_descent(h_cd, n, state, max_rounds=2000, deadline=deadline - 30)
        print(f"After coord descent: C5={state.c:.8f}", flush=True)

    # Phase 4: Try n=2*n0 with upsampled starting point
    n2 = 2 * n0
    if time.time() < deadline - 80 and n2 <= 4 * n0:
        h2 = upsample_linear(state.h, n2)
        h2 = as_feasible(h2, n2)
        c2 = compute_c5(h2)
        state.update(h2, c2, n2)
        alphas2 = [0.3, 0.1, 0.05, 0.0, 0.0]
        iters2 = [60, 60, 80, 100, 100]
        rem2 = deadline - time.time() - 40
        per2 = rem2 / len(alphas2)
        for al, mi in zip(alphas2, iters2):
            if time.time() > deadline - 40:
                break
            stage_dl2 = time.time() + max(10.0, per2)
            if _DEADLINE is not None:
                stage_dl2 = min(stage_dl2, _DEADLINE - 5.0)
            h2 = slsqp_stage(h2, n2, state, al, mi, stage_dl2)
            print(f"SLSQP n={n2} alpha={al}: C5={state.c:.8f}", flush=True)

    # Phase 5: Multi-start at n0 with perturbations
    if time.time() < deadline - 50:
        n_best = state.n
        h_best = state.h.copy()
        c_best = state.c
        for trial in range(5):
            if time.time() > deadline - 40:
                break
            np.random.seed(seed + 100 * (trial + 1))
            h_pert = h_best.copy() + 0.05 * np.random.randn(n_best)
            h_pert = as_feasible(h_pert, n_best)
            c_pert = compute_c5(h_pert)
            if c_pert < c_best:
                c_best = c_pert
            stage_dl = min(time.time() + 30.0, _DEADLINE - 5.0)
            h_pert = slsqp_stage(h_pert, n_best, state, 0.05, 60, stage_dl)
            h_pert = slsqp_stage(h_pert, n_best, state, 0.0, 80, min(time.time()+20, _DEADLINE-5))
            print(f"Multi-start {trial}: C5={state.c:.8f}", flush=True)

    # Final validation
    h_final = as_feasible(state.h, state.n)
    c_final = compute_c5(h_final)
    n_final = int(state.n)
    print(f"Final: n={n_final}, C5={c_final:.8f}", flush=True)
    return h_final, float(c_final), n_final
