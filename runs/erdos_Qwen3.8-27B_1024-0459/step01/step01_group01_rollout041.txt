import numpy as np
from scipy.optimize import minimize
import time
import functools

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        for i in idx:
            h[i] = np.clip(h[i] + per, 0.0, 1.0)
    return h

def smooth_max_obj(x, n, dx, tau):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return float((m + tau * np.log(np.sum(np.exp((corr - m) / tau)))) * dx)

def eq_con(x, n):
    return np.array([x.sum() - n / 2.0])

def upsample_to(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def binary_local_search(h, n, max_iter, rng, temperature=0.0):
    """Swap-based local search on binary vector."""
    dx = 2.0 / n
    # Ensure binary
    h = (h > 0.5).astype(float)
    # Fix sum
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    target = n // 2
    if len(ones) > target:
        excess = len(ones) - target
        to_flip = rng.choice(ones, excess, replace=False)
        h[to_flip] = 0
    elif len(ones) < target:
        deficit = target - len(ones)
        to_flip = rng.choice(zeros, deficit, replace=False)
        h[to_flip] = 1
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    current_c5 = best_c5
    current_h = h.copy()
    
    for it in range(max_iter):
        ones = np.where(current_h == 1)[0]
        zeros = np.where(current_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        current_h[i] = 0
        current_h[j] = 1
        new_c5 = compute_c5(current_h)
        if new_c5 < current_c5 or (temperature > 0 and rng.random() < np.exp(-(new_c5 - current_c5) / temperature)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
        else:
            current_h[i] = 1
            current_h[j] = 0
    
    return best_h, best_c5

    bounds = [(0.0, 1.0)] * n
    obj_fn = functools.partial(smooth_max_obj, n=n, dx=dx, tau=tau)
    con_fun = functools.partial(slsqp_con_fun, n=n)
    con_jac = functools.partial(slsqp_con_jac, n=n)
    try:
        res = minimize(
            fun=obj_fn, x0=x0, method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False}
        )
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return x0.copy()


def binary_sa(h_init, n, n_iter, rng, T_start=0.003, T_end=0.00003,
              reheat_every=8000, reheat_T=0.002):
    target = n // 2
    h = (h_init > 0.5).astype(float).copy()
    ones_count = int(h.sum())
    if ones_count > target:
        excess = ones_count - target
        one_idx = np.where(h == 1)[0]
        flip = rng.choice(one_idx, excess, replace=False)
        h[flip] = 0
    elif ones_count < target:
        deficit = target - ones_count
        zero_idx = np.where(h == 0)[0]
        flip = rng.choice(zero_idx, deficit, replace=False)
        h[flip] = 1

    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for it in range(n_iter):
        progress = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** progress

        if no_improve > reheat_every:
            T = reheat_T
            no_improve = 0

        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        i = one_idx[rng.integers(len(one_idx))]
        j = zero_idx[rng.integers(len(zero_idx))]

        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)

        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import functools
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h


def upsample_to(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def smooth_max_obj(x, n, dx, tau):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return float((m + tau * np.log(np.sum(np.exp((corr - m) / tau)))) * dx)


def slsqp_con_fun(x, n):
    return np.array([x.sum() - n / 2.0])

    return np.ones((1, n))

def coordinate_descent_binary(h, n, max_passes=5):
    """For each 1-position, try moving it to each 0-position (best improvement)."""
    h = h.copy()
    best_c5 = compute_c5(h)
    for _pass in range(max_passes):
        improved = False
        ones = np.where(h == 1)[0]

                    zeros = np.where(h == 0)[0]
                    improved = True
                else:
                    h[i] = 1.0
                    h[j] = 0.0

            break
    return h, best_c5
