    h = np.clip(h.copy(), 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = h.sum() - tgt
        if abs(d) < 1e-14:
            break

            m = h > 1e-14
        else:
            m = h < 1.0 - 1e-14
        if not m.any():

        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_active(h, n):
    """Subgradient at the active (max) lag."""
    c = c5_vals(h, n)
    k = np.argmax(c)
    lag = k - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    if lag >= 0:
        ramp = n - lag
        if ramp > 0:

            g[lag:] += 1.0 - h[:ramp]
    else:
        la = -lag
        ramp = n - la
        if ramp > 0:
            g[:ramp] += 1.0 - h[la:]
            g[la:] -= h[:ramp]
    return g * dx

    """Weighted subgradient using top-K active lags."""
    c = c5_vals(h, n)
    dx = 2.0 / n
    # Top K lags

    """Gradient of c_k w.r.t. h for a given lag."""
    g = np.zeros(n)
    if lag >= 0:
        r = n - lag
        if r > 0:
            g[:r] -= h[lag:]
            g[lag:] += 1.0 - h[:r]
    else:
        la = -lag
        r = n - la
        if r > 0:
            g[:r] += 1.0 - h[la:]
            g[la:] -= h[:r]

def weighted_lag_grad(h, n, idx, weights):
    """Weighted sum of gradients at specified lags."""
    dx = 2.0 / n
    g = np.zeros(n)
    n1 = n - 1
    for j, k in enumerate(idx):
        w = weights[j] * dx
        lag = k - n1
        if lag >= 0:
            r = n - lag
            if r > 0:
                g[:r] -= w * h[lag:]
                g[lag:] += w * (1.0 - h[:r])
        else:
            la = -lag
            r = n - la
            if r > 0:
                g[:r] += w * (1.0 - h[la:])
                g[la:] -= w * h[:r]
    return g

    dx = 2.0 / n
    s = beta * c * dx
    m = np.max(s)
