def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

        diff = h.sum() - target

            break
        if diff > 0:
            mask = h > 1e-15
        else:
            mask = h < 1.0 - 1e-15
        if mask.sum() == 0:
            break
        h[mask] -= diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def project_simplex_ball(h, n):
    target = n / 2.0
    lo = -1.0
    hi = 1.0

        else:

    w = np.exp(scaled - m)
    w /= np.sum(w)
    dx = 2.0 / n
    grad = np.zeros(n)
    n_half = n - 1
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-16:
            continue
        lag = k_idx - n_half
        if lag >= 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            if n - lag > 0:
                grad[:n - lag] -= wk * h[lag:]
        else:
            grad[:n + lag] += wk * (1.0 - h[-lag:])
            if -lag < n:
                grad[-lag:] -= wk * h[:n + lag]
    return grad * dx

    val = float(c[k_max] * (2.0 / n))
    return val, k_max


def hardmax_grad(h, n, k_max):
    dx = 2.0 / n
    grad = np.zeros(n)
    n_half = n - 1
    lag = k_max - n_half
    if lag >= 0:
        grad[lag:] += (1.0 - h[:n - lag])
        if n - lag > 0:
            grad[:n - lag] -= h[lag:]
    else:
        grad[:n + lag] += (1.0 - h[-lag:])
        if -lag < n:
            grad[-lag:] -= h[:n + lag]
    return grad * dx
