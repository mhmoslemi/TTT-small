    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def smooth_max(c, temp=0.01):
    m = np.max(c)
    return temp * np.log(np.sum(np.exp((c - m) / temp))) + m

def compute_smooth_c5(h, n_points, temp=0.01):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    return smooth_max(conv, temp)

    dx = 2.0 / n

    m = np.max(conv)

def project_h(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min(diff / cnt, 1.0 - h[mask].min() if cnt > 0 else 0)
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)

    s = np.sum(h)

        else:
            mask = h > 0.0
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, n, temp, penalty=0.0):
    """Compute smooth max objective and its gradient, plus penalty for sum constraint."""
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    # Log-sum-exp
    shifted = conv / temp
    shifted -= np.max(shifted)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    obj = temp * np.log(np.sum(exp_s)) + np.max(conv)
    
    # Gradient of smooth max
    grad = np.zeros(n)
    for ki in range(len(conv)):
        wk = w[ki] * dx
        if wk < 1e-14:
            continue
        k = ki - (n - 1)
        if k >= 0:
            grad[k:] += wk * (1.0 - h[:n-k])
            grad[:n-k] += wk * (-h[k:])
        else:
            p = -k
            grad[:n-p] += wk * (1.0 - h[p:])
            grad[p:] += wk * (-h[:n-p])
    
    # Add penalty gradient
    if penalty > 0:
        s = np.sum(h) - n / 2.0
        obj += penalty * s * s
        grad += 2.0 * penalty * s * np.ones(n)
    
    return obj, grad

def slsqp_smooth(h0, n, temp, maxiter=200, ftol=1e-12):
    """Optimize using SLSQP with smooth max objective."""
    target = n / 2.0
    
    def obj_func(x):
        return smooth_obj_and_grad(x, n, temp)[0]
    
    def grad_func(x):
        return smooth_obj_and_grad(x, n, temp)[1]
    
    res = minimize(obj_func, h0.copy(), jac=grad_func, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                                 'jac': lambda x: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x, res.fun

def anneal_optimize(h0, n, temps, maxiter_per_temp=100):

    h = h0.copy()

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx


def project_h(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        diff = target - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_grad(h, n, temp):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = conv / temp
    shifted -= np.max(shifted)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    obj = temp * (np.log(np.sum(exp_s)) + np.max(shifted))
    grad = np.zeros(n)
    for ki in range(len(conv)):
        wk = w[ki] * dx
        if wk < 1e-15:
            continue
        k = ki - (n - 1)
        if k >= 0:
            grad[k:] += wk * (1.0 - h[:n-k])
            grad[:n-k] += wk * (-h[k:])
        else:
            p = -k
            grad[:n-p] += wk * (1.0 - h[p:])
            grad[p:] += wk * (-h[:n-p])
    return obj, grad


def slsqp_smooth(h0, n, temp, maxiter=200, ftol=1e-12):
    target = n / 2.0
    state = {'obj': 0.0, 'grad': None}

    def obj_func(x):
        state['obj'], state['grad'] = smooth_obj_grad(x, n, temp)
        return state['obj']

    def grad_func(x):
        return state['grad']

    res = minimize(obj_func, h0.copy(), jac=grad_func, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq',
                                 'fun': lambda x: np.sum(x) - target,
                                 'jac': lambda x: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x, res.fun


def lbfgs_smooth(h0, n, temp, penalty=500.0, maxiter=500):
    target = n / 2.0

    def obj_func(x):
        o, g = smooth_obj_grad(x, n, temp)
        s = np.sum(x) - target
        o += 0.5 * penalty * s * s
        g += penalty * s
        return o, g

    def obj_only(x):
        return obj_func(x)[0]

    def grad_only(x):
        return obj_func(x)[1]

    res = minimize(obj_only, h0.copy(), jac=grad_only, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
    return res.x, res.fun


def anneal_optimize(h0, n, temps, method='slsqp', maxiter_per_temp=100):
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h, n)
    for temp in temps:
        try:
            if method == 'slsqp':
                h_new, _ = slsqp_smooth(h, n, temp, maxiter=maxiter_per_temp)
            else:
                h_new, _ = lbfgs_smooth(h, n, temp, penalty=1000.0, maxiter=maxiter_per_temp)
            h_new = project_h(h_new, n)
            v = compute_c5(h_new, n)
            if v < best_val:
                best_val = v
                best_h = h_new.copy()
            h = h_new
        except Exception:
            h = project_h(h, n)
    return best_h, best_val


def binary_local_search(h, n, max_rounds=50):
    h_bin = np.zeros(n)
    target = int(n / 2)
    idx = np.argsort(-h)[:target]
    h_bin[idx] = 1.0
    best_val = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            h_try = best_h.copy()
            h_try[j] = 1.0 - h_try[j]
            if np.sum(h_try) != target:
                continue
            v = compute_c5(h_try, n)
            if v < best_val - 1e-12:
                best_val = v
                best_h = h_try
                improved = True
        if not improved:
            break
    return best_h, best_val


def coordinate_refine(h, n, n_sweeps=30):
    h = project_h(h, n).copy()
    best_val = compute_c5(h, n)
    for _ in range(n_sweeps):
        improved = False
        for j in range(n):
            best_j_val = best_val
            best_j_h = h[j]
            for frac in [0.0, 0.25, 0.5, 0.75, 1.0,
                         h[j] - 0.1, h[j] + 0.1,
                         h[j] - 0.05, h[j] + 0.05]:
                val = min(1.0, max(0.0, frac))
                if abs(val - h[j]) < 1e-8:
                    continue
                h_try = h.copy()
                h_try[j] = val
                h_try = project_h(h_try, n)
                v = compute_c5(h_try, n)
                if v < best_j_val:
                    best_j_val = v
                    best_j_h = val
            if best_j_h != h[j]:
                h[j] = best_j_h
                h = project_h(h, n)
                if best_j_val < best_val:
                    best_val = best_j_val
                    improved = True
        if not improved:
            break
    return h, best_val
