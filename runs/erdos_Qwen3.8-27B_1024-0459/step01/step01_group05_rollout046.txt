    best_h = h.copy()

def coordinate_descent(h0, target, n_passes=5):

    h = project_simplex(h0, target)

        n_cd_passes = 8 if n_best <= 100 else 4
        h_cd, c5_cd = coordinate_descent(best_h, tgt, n_passes=n_cd_passes)
        try_update(h_cd, n_best)
    
    # Phase 4: Try other resolutions (15% time)
    resolutions_to_try = []
    for n_try in [100, 96, 120, 150, 200, 80, 110, 140]:
        if n_try != best_n:
            resolutions_to_try.append(n_try)
    
    # Sort by priority: prefer resolutions close to parent, then others
    resolutions_to_try.sort(key=lambda x: abs(x - best_n))
    
    for n_try in resolutions_to_try[:5]:
        if elapsed() > budget_s * 0.90:
            break
        x_src = np.linspace(0, 2, best_n, endpoint=False)
        x_dst = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_dst, x_src, best_h)
        target_try = n_try / 2.0
        h_interp = project_simplex(h_interp, target_try)
        
        # Subgradient phase
        if time_left() > 60:
            iters = min(30000, int(time_left() * 0.5 / 0.0005))
            h_sg, _ = subgradient_optimize(h_interp, target_try,
                                           max_iter=iters, lr_start=0.03, lr_end=0.0001)
        else:
            h_sg = h_interp
        try_update(h_sg, n_try)
        
        # Quick SLSQP polish
        if elapsed() < budget_s * 0.88:
            n_cur = best_n
            tgt_cur = n_cur / 2.0
            h_pol = slsqp_optimize(best_h, tgt_cur, beta=50.0, max_iter=200, ftol=1e-12)
            try_update(h_pol, n_cur)
    
    # Phase 5: Final refinement (last 10% time)
    if elapsed() < budget_s * 0.97:
        n_best = best_n
        tgt = n_best / 2.0
        
        # Final SLSQP with high beta
        h_final = slsqp_optimize(best_h, tgt, beta=300.0, max_iter=300, ftol=1e-13)
        try_update(h_final, n_best)
        
        # Final coordinate descent
        h_cd2, _ = coordinate_descent(best_h, tgt, n_passes=3)
        try_update(h_cd2, n_best)
    
    # Ensure exact constraint
    n_final = best_n
    tgt_final = n_final / 2.0
    best_h = project_simplex(best_h, tgt_final)
    
    # Verify
    s = np.sum(best_h)
    if abs(s - tgt_final) > 1e-10:
        best_h[n_final // 2] += (tgt_final - s)
        best_h = np.clip(best_h, 0.0, 1.0)
    
    final_c5 = compute_c5(best_h)
    
    return (best_h, final_c5, best_n)
