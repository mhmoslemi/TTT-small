def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):

            # c1[n-1] - c0[n-1] = 1 - 0 = 1, but actual is t - t^2
            # So for lag 0: value = A[0]*... let me handle separately
            # Actually: c(lag=0) at h[i]=t: the h[i] term contributes t*(1-t)
            # c0 has h[i]=0: contributes 0
            # c1 has h[i]=1: contributes 0
            # So A[n-1] = c1[n-1] - c0[n-1] doesn't capture the quadratic
            # The true value at lag 0 as function of t: (t - t^2) + (c0 - 0 + 0) ... no
            # c0 = corr with h[i]=0: the lag-0 term for i is 0*(1-0)=0
            # c1 = corr with h[i]=1: the lag-0 term for i is 1*(1-1)=0
            # True value at lag 0 with h[i]=t: t*(1-t)
            # So the lag-0 correlation as function of t: c0[n-1] + t*(1-t) - 0 - 0
            # Actually c0[n-1] already includes all other terms at lag 0, plus 0 for i
            # So true_c0(t) = c0[n-1] + t*(1-t)
            # And for other lags: true_c_k(t) = c0[k] + A[k]*t

            # Build objective on t_grid
            # For lag != 0: c0[k] + A[k]*t
            # For lag == 0: c0[n-1] + t*(1-t)
            c0_arr = c0.astype(float)
            c0_arr_sq = c0_arr.copy()
            c0_arr_sq[n-1] = c0_arr_sq[n-1]  # base for lag 0
            A_arr = A.astype(float)

            # Linear part for all lags (treating lag 0 as linear approximation for now)
            lin_vals = c0_arr[None, :] + A_arr[None, :] * t_grid[:, None]  # (num_t, 2n-1)
            # Fix lag 0: replace with true quadratic
            quad_val = t_grid - t_grid**2 + c0_arr[n-1]
            lin_vals[:, n-1] = quad_val

            obj_vals = np.max(lin_vals, axis=1)  # (num_t,)
            # Scale by dx
            obj_vals_scaled = obj_vals * dx

            i_best = int(np.argmin(obj_vals_scaled))
            t_new = t_grid[i_best]

            # Refine with bisection around best
            lo = t_grid[max(0, i_best-1)]
            hi = t_grid[min(num_t-1, i_best+1)]
            for _ in range(20):
                mid1 = lo + (hi - lo) / 3.0
                mid2 = hi - (hi - lo) / 3.0
                # Evaluate at mid1
                h[i] = mid1
                c_m1 = np.max(get_correlations(h, n)) * dx
                h[i] = mid2
                c_m2 = np.max(get_correlations(h, n)) * dx
                h[i] = old_val
                if c_m1 < c_m2:
                    hi = mid2
                else:
                    lo = mid1

            t_new = (lo + hi) / 2.0
            t_new = np.clip(t_new, 0.0, 1.0)

            if abs(t_new - old_val) > 1e-10:
                h[i] = t_new
                h = project_feasible(h, n)
                c5_val = compute_c5(h, n)
                if c5_val < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_val
                    improved = True
                if c5_val > best_c5 + 0.01:
                    h = best_h.copy()

        if not improved:
            break

    return best_h, best_c5


def slsqp_optimize(h, n, method='soft', beta=50.0, maxiter=1000, ftol=1e-12, t0=None, budget_s=1300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 'jac': lambda x: np.ones(n)}]

    if method == 'soft':
        fun = lambda x: subgradient_h_fn(x, n, beta)
        jac = lambda x: subgradient_h_grad_fn(x, n, beta)
    else:
        fun = lambda x: hard_max_fn(x, n)
        jac = lambda x: hard_max_grad_fn(x, n)

    try:
        res = minimize(
            fun, h, jac=jac, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n)
        return h_new
    except Exception:
        return h.copy()


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    # Get initial values from parent
    try:
        h_init = np.array(initial_h_values, dtype=float)
        n0 = len(h_init)
    except Exception:
        n0 = 95
        h_init = np.full(n0, 0.5)

    h_init = project_feasible(h_init, n0)

    # Upsample to higher resolution
    n = n0 * 2
    h = upsample_linear(h_init, n0, n)
    h = project_feasible(h, n)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)

    # Stage 1: SLSQP soft-max with increasing beta
    for beta in [10.0, 50.0, 200.0]:
        if time.time() - t0 > budget_s - 300:
            break
        h_new = slsqp_optimize(h, n, method='soft', beta=beta, maxiter=600, ftol=1e-11, t0=t0, budget_s=budget_s)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        h = h_new

    # Stage 2: Subgradient method
    if time.time() - t0 < budget_s - 200:
        h_sg, c5_sg = subgradient_method(best_h.copy(), n, num_steps=500, rng=rng, t0=t0, budget_s=budget_s)
        if c5_sg < best_c5:
            best_h = h_sg
            best_c5 = c5_sg

    # Stage 3: Coordinate descent
    if time.time() - t0 < budget_s - 150:
        h_cd, c5_cd = coordinate_descent(best_h.copy(), n, num_sweeps=40, t0=t0, budget_s=budget_s, num_t=50)
        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd
        else:
            # Also try CD from the SLSQP result
            h_cd2, c5_cd2 = coordinate_descent(h.copy(), n, num_sweeps=30, t0=t0, budget_s=budget_s, num_t=50)
            if c5_cd2 < best_c5:
                best_h = h_cd2
                best_c5 = c5_cd2

    # Stage 4: SLSQP hard-max refinement
    for _pass in range(3):
        if time.time() - t0 > budget_s - 80:
            break
        h_new = slsqp_optimize(best_h.copy(), n, method='hard', maxiter=1500, ftol=1e-13, t0=t0, budget_s=budget_s)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Stage 5: Try coordinate descent again after SLSQP
    if time.time() - t0 < budget_s - 60:
        h_cd3, c5_cd3 = coordinate_descent(best_h.copy(), n, num_sweeps=30, t0=t0, budget_s=budget_s, num_t=60)
        if c5_cd3 < best_c5:
            best_h = h_cd3
            best_c5 = c5_cd3

    # Stage 6: Random restarts from best solution
    if time.time() - t0 < budget_s - 100:
        for restart in range(30):
            if time.time() - t0 > budget_s - 50:
                break
            scale = 0.02 + 0.03 * (restart / 10.0)
            h_pert = best_h + rng.randn(n) * scale
            h_pert = project_feasible(h_pert, n)

            # Quick subgradient + coordinate descent
            h_sg_r, c5_sg_r = subgradient_method(h_pert.copy(), n, num_steps=100, rng=rng, t0=t0, budget_s=budget_s)
            if c5_sg_r < best_c5:
                best_h = h_sg_r
                best_c5 = c5_sg_r

            h_cd_r, c5_cd_r = coordinate_descent(h_sg_r.copy() if c5_sg_r < compute_c5(h_pert, n) else h_pert.copy(),
                                                   n, num_sweeps=20, t0=t0, budget_s=budget_s, num_t=40)
            if c5_cd_r < best_c5:
                best_h = h_cd_r
                best_c5 = c5_cd_r

    # Stage 7: Try different resolution
    if time.time() - t0 < budget_s - 120:
        n_alt = n0 * 3  # 3x original
        h_alt = upsample_linear(best_h, n, n_alt)
        h_alt = project_feasible(h_alt, n_alt)

        for beta in [30.0, 150.0]:
            if time.time() - t0 > budget_s - 40:
                break
            h_alt_new = slsqp_optimize(h_alt, n_alt, method='soft', beta=beta, maxiter=500, t0=t0, budget_s=budget_s)
            c5_alt = compute_c5(h_alt_new, n_alt)
            if c5_alt < best_c5:
                best_h = h_alt_new
                best_c5 = c5_alt
                n = n_alt
            h_alt = h_alt_new

        if time.time() - t0 < budget_s - 80:
            h_alt_new = slsqp_optimize(h_alt, n, method='hard', maxiter=1000, ftol=1e-12, t0=t0, budget_s=budget_s)
            c5_alt = compute_c5(h_alt_new, n)
            if c5_alt < best_c5:
                best_h = h_alt_new
                best_c5 = c5_alt

    # Final SLSQP polish
    if time.time() - t0 < budget_s - 20:
        h_final = slsqp_optimize(best_h, n, method='hard', maxiter=2000, ftol=1e-14, t0=t0, budget_s=budget_s)
        c5_final = compute_c5(h_final, n)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)

    # Validate with evaluator if available
    try:
        val = evaluate_erdos_solution(best_h, n)
        if val is not None and 0.0 < val < 1.0:
            best_c5 = min(best_c5, float(val))
    except Exception:
        pass

    return (best_h, best_c5, n)
