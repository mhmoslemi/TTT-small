    n = len(h)
    grad = np.zeros(n)
    shift = k  # can be negative
    
    if shift >= 0:

            grad[i] += (1.0 - h[i - shift])

    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = n / 2.0 - s

        grad[:n-k] -= h[k:]
    else:

Wait, the c_k values are already in "integral units" (they're sums that get multiplied by dx). Let me be careful.

The raw correlation values are c_k = Σ_i h_i(1-h_{i-k}). The actual integral values are c_k * dx. The max is max_k(c_k) * dx.

For the softmax, I should work with the actual values:
f_β = (1/β) log Σ_k exp(β * c_k * dx)

Or equivalently (absorbing dx):
f_β = (1/β) log Σ_k exp(β * c_k) (then multiply by dx at the end)

Actually for SLSQP, I just need a smooth function. Let me use:
f_β(h) = max(corr) + (1/β) * log(Σ exp(β*(corr - max(corr))))

This is the log-sum-exp trick, giving a smooth approximation to max. For large β, it's close to max.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(2000):
        s = np.sum(h)
        diff = n / 2.0 - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_at_shift(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] += (1.0 - h[:n - k])
        grad[:n - k] -= h[k:]
    else:
        ks = -k
        grad[:n - ks] += (1.0 - h[ks:])
        grad[ks:] -= h[:n - ks]
    return grad

def soft_max_objective(h, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    return cmax + (1.0 / beta) * np.log(np.sum(np.exp(beta * (corr - cmax))))

def interpolate_h(h_src, n_dst):
    if len(h_src) == n_dst:
        return h_src.copy()
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def projected_gradient_optimize(h0, n, time_limit, use_softmax=False, beta=20.0):
    h = project(h0.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    step_size = 0.05
    momentum = np.zeros(n)
    mu = 0.3
    no_improve = 0
    max_no_improve = 500

    while time.time() - t0 < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        if use_softmax:
            cmax = np.max(corr)
            w = np.exp(beta * (corr - cmax))
            w /= np.sum(w)
            grad = np.zeros(n)
            for j in range(2 * n - 1):
                if w[j] < 1e-12:
                    continue
                k = j - (n - 1)
                grad += w[j] * gradient_at_shift(h, k)
        else:
            active_idx = np.argmax(corr)
            active_shift = active_idx - (n - 1)
            grad = gradient_at_shift(h, active_shift)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        
        momentum = mu * momentum + (1 - mu) * grad
        direction = -momentum / (np.linalg.norm(momentum) + 1e-15)
        
        # Backtracking line search
        accepted = False
        for trial_step in [step_size, step_size * 0.5, step_size * 0.1, step_size * 0.01]:
            h_new = project(h + trial_step * direction * grad_norm, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
                step_size = min(0.2, step_size * 1.1)
                accepted = True
                break
        if not accepted:
            no_improve += 1
            step_size *= 0.8
            if no_improve > max_no_improve:
                # Try different active shifts
                top_k = np.argsort(corr)[-8:]
                found = False
                for idx in top_k:
                    k = idx - (n - 1)
                    g = gradient_at_shift(h, k)
                    h_try = project(h - 0.1 * g, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5:
                        h = h_try
                        best_c5 = c5_try
                        best_h = h.copy()
                        no_improve = 0
                        max_no_improve = 500
                        step_size = 0.05
                        found = True
                        break
                if not found:
                    break
        if time.time() - t0 > time_limit:
            break

    return best_h, best_c5

def local_perturbation(h, n, time_limit, base_step=0.05, n_mod_min=1, n_mod_max=7):
    h = project(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    step = base_step

    while time.time() - t0 < time_limit:
        h_try = best_h.copy()
        n_mod = np.random.randint(n_mod_min, n_mod_max + 1)
        idxs = np.random.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += np.random.uniform(-step, step)
        h_try = project(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            best_h = h_try.copy()
            step = max(0.005, step * 0.97)
        else:
            step = min(base_step * 2, step * 1.01)

    return best_h, best_c5

def slsqp_optimize(h0, n, time_limit, beta=100.0):
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        if time.time() - t0 > time_limit:
            raise TimeoutError()
        return soft_max_objective(x, beta)
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                      constraints=cons,
                      options={'maxiter': 5000, 'ftol': 1e-15})
        if time.time() - t0 < time_limit + 5:
            h_out = project(res.x, n)
            c5 = compute_c5(h_out)
            return h_out, c5
    except (TimeoutError, Exception):
        pass
    return None, None

def generate_structured_inits(n, rng):
    inits = []
    # Periodic patterns
    if n % 4 == 0:
        h1 = np.tile([1.0, 1.0, 0.0, 0.0], n // 4)
        inits.append(project(h1, n))
        h2 = np.tile([1.0, 0.0, 1.0, 0.0], n // 4)
        inits.append(project(h2, n))
    if n % 3 == 0:
        h3 = np.tile([1.0, 1.0, 0.0], n // 3)
        inits.append(project(h3, n))
    # Thue-Morse
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    inits.append(project(tm, n))
    # Cosine modulation
    x = np.linspace(0, 2, n, endpoint=False)
    h4 = np.clip(0.5 + 0.4 * np.cos(2 * np.pi * x), 0, 1)
    inits.append(project(h4, n))
    h5 = np.clip(0.5 + 0.3 * np.sin(4 * np.pi * x + 0.5), 0, 1)
    inits.append(project(h5, n))
    # Random
    inits.append(project(rng.rand(n) * 1.5, n))
    inits.append(project(rng.rand(n) * 1.5, n))
    return inits

def coordinate_descent_shift(h, n, k, max_iter=50):
    """Coordinate descent minimizing overlap at shift k (with sum constraint)."""
    h = h.copy()
    for _ in range(max_iter):
        # Compute coefficients a_i = (1 - h_{i-k}) - h_{i+k}
        a = np.zeros(n)
        if k >= 0:
            a[k:] += (1.0 - h[:n - k])
            a[:n - k] -= h[k:]
        else:
            ks = -k
            a[:n - ks] += (1.0 - h[ks:])
            a[ks:] -= h[:n - ks]
        
        # LP: min a·h s.t. 0<=h<=1, sum(h)=n/2
        # Solution: sort a, set h=1 for smallest a, h=0 for largest, one fractional
        order = np.argsort(a)
        h_new = np.zeros(n)
        remaining = n / 2.0
        for idx in order:
            if remaining <= 1e-15:
                break
            h_new[idx] = min(1.0, remaining)
            remaining -= h_new[idx]
        h_new = project(h_new, n)
        
        # Check improvement at this shift
        if k >= 0:
            c_old = np.sum(h[k:] * (1.0 - h[:n - k]))
            c_new = np.sum(h_new[k:] * (1.0 - h_new[:n - k]))
        else:
            ks = -k
            c_old = np.sum(h[:n - ks] * (1.0 - h[ks:]))
            c_new = np.sum(h_new[:n - ks] * (1.0 - h_new[ks:]))
        
        if c_new < c_old - 1e-12:
            h = h_new
    
    return h
