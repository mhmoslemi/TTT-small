from scipy.optimize import minimize, linprog
import time

# Top-level helper functions

def get_c_values(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    c = get_c_values(h)
    return float(np.max(c) * (2.0 / n))

def soft_max_val(h, beta):
    n = len(h)
    c = get_c_values(h)
    scaled = beta * c
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / beta * (2.0 / n))

def soft_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = get_c_values(h)
    scaled = beta * c
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[lag:] += wk * (1.0 - h[:n - lag])
                grad[:n - lag] -= wk * h[lag:]
        else:
            m_neg = -lag
            if m_neg < n:
                grad[m_neg:] += wk * (1.0 - h[:n - m_neg])
                grad[:n - m_neg] -= wk * h[m_neg:]
    return grad * dx

def project_feasible(h, target):
    """Water-filling projection onto {0<=x<=1, sum(x)=target}"""
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def projected_grad_descent(h0, n, beta, max_iter, time_limit, tol=1e-14):
    """PGD with log-sum-exp and water-filling projection"""
    h = project_feasible(h0, n / 2.0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        
        val = soft_max_val(h, beta)
        grad = soft_max_grad(h, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < tol:
            break
        
        # Backtracking line search (Armijo)
        step = 1.0 / (beta * 0.5)  # initial step
        alpha = step
        c1 = 1e-4
        for _ in range(40):
            h_trial = project_feasible(h - alpha * grad, n / 2.0)
            val_trial = soft_max_val(h_trial, beta)
            if val_trial <= val - c1 * alpha * np.dot(grad, h_trial - h):
                break
            alpha *= 0.5
        
        h = project_feasible(h - alpha * grad, n / 2.0)
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    
    return best_h, best_c5

    # c_obj = [0,...,0, 1]

Let me also think about whether the water-filling projection is correct.

We want to project h onto {x : 0 ≤ x_i ≤ 1, sum(x_i) = target}.

The projection is: x_i* = clip(h_i - τ, 0, 1) where τ is chosen so that sum(x_i*) = target.

Since sum(clip(h_i - τ, 0, 1)) is a continuous, non-increasing function of τ, we can find τ by bisection.

When τ → -∞: all x_i → 1, sum → n
When τ → +∞: all x_i → 0, sum → 0

So for target ∈ [0, n], there exists a unique τ. Bisection on τ ∈ [min(h)-1, max(h)+1] should work.

import numpy as np
from scipy.optimize import minimize, linprog
import time


def get_c_values(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h):
    n = len(h)
    c = get_c_values(h)
    return float(np.max(c) * (2.0 / n))


def soft_max_val(h, beta):
    n = len(h)
    c = get_c_values(h)
    scaled = beta * c
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / beta * (2.0 / n))


def soft_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = get_c_values(h)
    scaled = beta * c
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-15:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[lag:] += wk * (1.0 - h[:n - lag])
                grad[:n - lag] -= wk * h[lag:]
        else:
            m_neg = -lag
            if m_neg < n:
                grad[m_neg:] += wk * (1.0 - h[:n - m_neg])
                grad[:n - m_neg] -= wk * h[m_neg:]
    return grad * dx


def project_feasible(h, target):
    h = np.asarray(h, dtype=float)
    lo = -5.0
    hi = 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)


def pgd_optimize(h0, n, beta, max_iter, time_limit):
    h = project_feasible(h0, n / 2.0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    stagnant = 0
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        val = soft_max_val(h, beta)
        grad = soft_max_grad(h, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = 1.0 / (beta * 0.3)
        alpha = step
        c1 = 1e-4
        improved = False
        for _ in range(50):
            h_trial = project_feasible(h - alpha * grad, n / 2.0)
            val_trial = soft_max_val(h_trial, beta)
            if val_trial <= val - c1 * alpha * (grad @ (h_trial - h)):
                improved = True
                break
            alpha *= 0.5
        if not improved:
            stagnant += 1
            if stagnant > 10:
                break
            h = h + grad * 1e-6
            h = project_feasible(h, n / 2.0)
            continue
        stagnant = 0
        h = h_trial
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    return best_h, best_c5


def sum_eq_fun(x, n):
    return np.sum(x) - n / 2.0


def sum_eq_jac(x, n):
    return np.ones(n)


def soft_obj_args(x, n, beta):
    return soft_max_val(x, beta)


def soft_grad_args(x, n, beta):
    return soft_max_grad(x, beta)


def hard_obj_args(x, n):
    c = get_c_values(x)
    return float(np.max(c) * (2.0 / n))


def hard_grad_args(x, n):
    c = get_c_values(x)
    k_max = np.argmax(c)
    lag = k_max - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag >= 0:
        if n - lag > 0:
            grad[lag:] += (1.0 - x[:n - lag])
            grad[:n - lag] -= x[lag:]
    else:
        m = -lag
        grad[m:] += (1.0 - x[:n - m])
        grad[:n - m] -= x[m:]
    return grad * dx


def lp_based_step(h, n):
    c = get_c_values(h)
    n_lags = 2 * n - 1
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    b_ub = np.zeros(n_lags)
    for k_idx in range(n_lags):
        lag = k_idx - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            if n - lag > 0:
                g[lag:] += (1.0 - h[:n - lag])
                g[:n - lag] -= h[lag:]
        else:
            m = -lag
            if m < n:
                g[m:] += (1.0 - h[:n - m])
                g[:n - m] -= h[m:]
        A_ub[k_idx, :n] = g
        A_ub[k_idx, n] = -1.0
        b_ub[k_idx] = g @ h - c[k_idx]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def lp_improve(h, n, max_iters=10):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(max_iters):
        x_target = lp_based_step(best_h, n)
        if x_target is None:
            break
        best_found = False
        for t in [0.1, 0.25, 0.4, 0.5, 0.65, 0.8, 0.9, 1.0]:

            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-12:
                best_c5 = c5_trial
                best_h = h_trial
                best_found = True
                break
        if not best_found:
            break
    return best_h, best_c5


def binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.rand(n)
    indices = np.argsort(h)
    h_sorted = np.zeros(n)
    h_sorted[indices[:n // 2]] = 1.0
    return project_feasible(h_sorted, n / 2.0)

    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / 2.0 + phase)

def slsqp_polish(h, n, beta, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(
            fun=soft_obj_args,
            x0=h,
            jac=soft_grad_args,
            args=(n, beta),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-14}

        h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n / 2.0)
        c5_new = compute_c5(h_new)
        return h_new, c5_new
    except Exception:
        return h, compute_c5(h)
