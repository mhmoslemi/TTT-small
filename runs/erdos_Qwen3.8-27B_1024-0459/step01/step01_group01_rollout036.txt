    where A_h[k] = g shifted by k (the correlation kernel)
    """

    # sum(h) = n/2

    # A matrix: n_corrs x (n+1)
    # For row m, column i: g[i + (m - (n-1))] if valid, else 0

import numpy as np
from scipy.optimize import minimize, linprog
import time
import functools


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        for i in idx:
            h[i] = np.clip(h[i] + per, 0.0, 1.0)
    return h


def upsample_to(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old, kind='linear')


def smooth_max_obj(x, n, dx, tau):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return float((m + tau * np.log(np.sum(np.exp((corr - m) / tau)))) * dx)


def topk_obj(x, n, dx, k, tau):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    smooth_m = m + tau * np.log(np.sum(np.exp((corr - m) / tau)))
    sorted_c = np.sort(corr)
    topk_sum = np.sum(sorted_c[-k:])
    return float((smooth_m + 0.1 * topk_sum) * dx)


def eq_con_func(x, n):
    return np.array([x.sum() - n / 2.0])


def eq_jac_func(x, n):
    return np.ones((1, n))


def solve_lp_fixed_g(g, n):
    n_corrs = 2 * n - 1
    A_ub = np.zeros((n_corrs, n + 1))
    b_ub = np.zeros(n_corrs)
    for m in range(n_corrs):
        l = m - (n - 1)
        i_min = max(0, -l)
        i_max = min(n, n - l)
        if i_max > i_min:
            A_ub[m, i_min:i_max] = g[i_min + l:i_max + l]
        A_ub[m, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(0, None)]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None


def alternating_lp(h_init, n, max_iters=15):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        g = 1.0 - h
        h_new = solve_lp_fixed_g(g, n)
        if h_new is None:
            break
        h_new = fix_sum(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if np.max(np.abs(h_new - h)) < 1e-8:
            break
        h = 0.5 * h + 0.5 * h_new
        h = fix_sum(h, n)
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
    return best_h, best_c5


def slsqp_optimize(x0, n, tau, maxiter, ftol):
    dx = 2.0 / n
    bounds_list = [(0.0, 1.0)] * n
    obj_p = functools.partial(smooth_max_obj, n=n, dx=dx, tau=tau)
    con_p = functools.partial(eq_con_func, n=n)
    jac_p = functools.partial(eq_jac_func, n=n)
    try:
        res = minimize(
            fun=obj_p, x0=x0, method='SLSQP',
            bounds=bounds_list,
            constraints=[{'type': 'eq', 'fun': con_p, 'jac': jac_p}],
            options={'maxiter': maxiter, 'ftol': ftol, 'disp': False}
        )
        h_new = np.clip(res.x, 0.0, 1.0)
        h_new = fix_sum(h_new, n)
        return h_new
    except Exception:
        return x0


def subgradient_step(h, n, step_size, n_steps=100):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current = h.copy()
    for step in range(n_steps):
        corr = np.correlate(current, 1.0 - current, mode="full")
        m_idx = np.argmax(corr)
        l = m_idx - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            val = 0.0
            j_plus = j + l
            if 0 <= j_plus < n:
                val += (1.0 - current[j_plus])
            j_minus = j - l
            if 0 <= j_minus < n:
                val -= current[j_minus]
            grad[j] = val
        grad = grad * dx
        current = current - step_size * grad
        current = np.clip(current, 0.0, 1.0)
        current = fix_sum(current, n)
        c5_cur = compute_c5(current)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = current.copy()
            step_size = min(step_size * 1.05, 0.1)
        else:
            step_size = step_size * 0.7
    return best_h, best_c5


def periodic_init(n, period_frac, duty_frac):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    period = 2.0 * period_frac
    h = np.zeros(n)
    for i in range(n):
        phase = (x[i] % period) / period
        h[i] = 1.0 if phase < duty_frac else 0.0
    h = fix_sum(h, n)
    return h
