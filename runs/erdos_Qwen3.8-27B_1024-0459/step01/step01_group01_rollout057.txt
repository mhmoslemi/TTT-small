The projection onto {0≤h≤1, sum(h)=n/2} is:
Find λ such that sum(clip(h_new + λ, 0, 1)) = n/2.
This can be done by bisection on λ.

Wait, that's not quite right. The projection of a point v onto {0≤h≤1, sum(h)=n/2} is:
h* = argmin ||h-v||² subject to 0≤h≤1, sum(h)=n/2

By KKT: h*_i = clip(v_i + λ, 0, 1) for some λ.
Find λ such that sum(clip(v_i + λ, 0, 1)) = n/2.

This is a 1D problem in λ, solvable by bisection since sum(clip(v_i+λ, 0, 1)) is monotonically increasing in λ.

**Let me also think about a completely different approach: CVXPY with a reformulation.**

The problem is:

subject to:

So CVXPY won't directly work for the exact problem. However, we could use a convex relaxation:

One more thought: the problem has a natural symmetry. If we consider h on the circle (periodic extension), the correlation structure is simpler. But the problem as stated uses [0,2] with the full correlation (not circular). However, since the function is only defined on [0,2] and we're looking at all shifts, the "ends" matter.

Actually wait - looking at the problem again: h: [0,2] → [0,1], and the integral is ∫ h(x)(1-h(x+k)) dx. The domain of integration would be [0, 2-k] for k>0 or [k, 2-k]... Actually with the discretization using np.correlate, it handles the finite support naturally.

Key implementation details:
1. `compute_c5(h)`: np.correlate(h, 1-h, mode='full') * dx, take max
2. `smooth_max_and_grad(h, tau)`: returns (value, gradient)
3. `project(v, n)`: project v onto {0≤h≤1, sum(h)=n/2}
4. `projected_gradient(h, n, tmax, tau, ...)`: main optimization loop

For the gradient computation, I'll use the convolution approach:
- T1 = np.convolve(1-h, W_rev, 'full')[n-1:2*n-1]
- T2 = np.convolve(h, W, 'full')[n-1:2*n-1]
- grad = dx * (T1 - T2)

where W is the softmax weights over the 2n-1 correlation values.

import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve
import functools
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h


def project_onto_simplex_box(v, n):
    """Project v onto {h : 0<=h_i<=1, sum(h)=n/2} via bisection on lambda."""
    target = n / 2.0
    lo = v.min() - 1.0
    hi = 1.0 - v.max()
    if hi - lo < 1e-15:
        return np.clip(v, 0.0, 1.0)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h_try = np.clip(v + mid, 0.0, 1.0)
        s = h_try.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx] = np.clip(h[idx] + diff / len(idx), 0.0, 1.0)
    return h


def upsample_to(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def smooth_max_and_grad(h, tau):
    """Compute smooth max of correlations and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / tau
    exp_shifted = np.exp(shifted - np.max(shifted))
    Z = np.sum(exp_shifted)
    weights = exp_shifted / Z  # length 2n-1
    soft_max_val = (m + tau * np.log(np.sum(np.exp((corr - m) / tau)))) * dx

    # Gradient via convolution
    A = 1.0 - h
    W = weights  # length 2n-1
    W_rev = W[::-1]

    # T1[j] = sum_m W[m-j+n-1] * A[m] = (A * W_rev)[j + n - 1]
    conv1 = fftconvolve(A, W_rev, mode="full")
    T1 = conv1[n - 1 : 2 * n - 1]

    # T2[j] = sum_m h[m] * W[j-m+n-1] = (h * W)[j + n - 1]
    conv2 = fftconvolve(h, W, mode="full")
    T2 = conv2[n - 1 : 2 * n - 1]

    grad = dx * (T1 - T2)
    return soft_max_val, grad


def projected_gradient_optimize(h_init, n, tmax, tau_list, lr_init=0.01, momentum_init=0.9):
    """Projected gradient with momentum and adaptive step size. Returns best h found."""
    h = h_init.copy()
    best_h = h.copy()
    best_obj = smooth_max_and_grad(h, tau_list[0])[0]
    best_c5 = compute_c5(h)

    v = np.zeros(n)  # momentum velocity
    momentum = momentum_init
    lr = lr_init

    t_start = time.time()
    tau_idx = 0
    patience = 0
    max_patience = 300

    while time.time() - t_start < tmax:
        tau = tau_list[min(tau_idx, len(tau_list) - 1)]
        obj, grad = smooth_max_and_grad(h, tau)

        # Adaptive step size: scale by inverse of gradient norm
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-12:
            step_scale = lr_init / (gnorm / n)
        else:
            step_scale = lr_init

        # Momentum update
        v = momentum * v - step_scale * grad
        h_new = h + v

        # Project
        h_new = project_onto_simplex_box(h_new, n)

        # Accept/reject with simple improvement check
        obj_new, _ = smooth_max_and_grad(h_new, tau)
        if obj_new < obj:
            h = h_new
            patience = 0
        else:
            # Try smaller step
            v = 0.5 * v
            h_new = project_onto_simplex_box(h + v, n)
            obj_new2, _ = smooth_max_and_grad(h_new, tau)
            if obj_new2 < obj:
                h = h_new
                patience = 0
            else:
                v = np.zeros(n)
                patience += 1
                lr *= 0.8
                if lr < 1e-6:
                    lr = lr_init * 0.3

        # Update best
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()

        # Decay momentum slowly
        momentum = min(momentum * 1.0001, 0.95)

        # Advance tau schedule
        if patience > max_patience:
            if tau_idx < len(tau_list) - 1:
                tau_idx += 1
            else:
                # Restart with random perturbation
                h = project_onto_simplex_box(h + np.random.randn(n) * 0.05, n)
                v = np.zeros(n)
                patience = 0
                lr = lr_init
            max_patience += 200

    return h, best_h, best_c5


def multi_tau_pg_optimize(h_init, n, tmax, rng):
    """Run projected gradient with multiple tau values and restarts."""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    tau_schedule = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.003]

    # Multiple passes with decreasing tau
    for tau in tau_schedule:
        remaining = tmax - (time.time() - (tmax_t0 := time.time()))
        if remaining <= 5:
            break
        # Allocate time proportionally
        budget_for_tau = min(remaining * 0.25, max(30, tmax * 0.12))
        h_new, best_h_new, best_c5_new = projected_gradient_optimize(
            h, n, budget_for_tau, [tau], lr_init=0.005, momentum_init=0.85
        )
        h = h_new
        if best_c5_new < best_c5:
            best_c5 = best_c5_new
            best_h = best_h_new

    # Perturbation restarts
    for r in range(5):
        remaining = tmax - (time.time() - tmax_t0)
        if remaining < 30:
            break
        noise_scale = 0.03 + 0.02 * r
        h_pert = project_onto_simplex_box(best_h + rng.standard_normal(n) * noise_scale, n)
        h_new, best_h_new, best_c5_new = projected_gradient_optimize(
            h_pert, n, min(remaining * 0.3, 100),
            [0.02, 0.01, 0.005], lr_init=0.003, momentum_init=0.8
        )
        if best_c5_new < best_c5:
            best_c5 = best_c5_new
            best_h = best_h_new

    return best_h, best_c5


def slsqp_local(h_init, n, tmax):
    """Local SLSQP polish for small n."""
    if n > 400:
        return h_init, compute_c5(h_init)

    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def obj_func(x):
        h_c = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h_c, 1.0 - h_c, mode="full")
        m = np.max(corr)
        tau = 0.005
        return float((m + tau * np.log(np.sum(np.exp((corr - m) / tau)))) * dx)

    def con_fun(x):
        return np.array([x.sum() - n / 2.0])

    def con_jac(x):
        return np.ones((1, n))

    try:
        res = minimize(
            obj_func, h_init, method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
            options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
        )
        h_new = fix_sum(np.clip(res.x, 0.0, 1.0), n)
        c5_new = compute_c5(h_new)
        return h_new, c5_new
    except Exception:
        return h_init, compute_c5(h_init)


def binary_local_search(h_init, n, tmax, rng):
    """Try binary rounding and local swaps to improve."""
    h = h_init.copy()
    # Round to binary
    thresh = 0.5
    h_bin = (h > thresh).astype(float)
    # Fix sum
    ones = np.sum(h_bin)
    target = n // 2
    if ones > target:
        # Turn off some 1s -> 0s (those with lowest h values)
        one_indices = np.where(h_bin == 1.0)[0]
        rng.shuffle(one_indices)
        for i in one_indices[:int(ones - target)]:
            h_bin[i] = 0.0
    elif ones < target:
        zero_indices = np.where(h_bin == 0.0)[0]
        rng.shuffle(zero_indices)
        for i in zero_indices[:int(target - ones)]:
            h_bin[i] = 1.0
    h_bin = fix_sum(h_bin, n)
    c5_bin = compute_c5(h_bin)

    if c5_bin >= compute_c5(h):
        return h_init, compute_c5(h_init)

    # Local search: swap pairs of 1 and 0
    best_h = h_bin.copy()
    best_c5 = c5_bin
    t_start = time.time()
    improving = True
    max_rounds = 10
    for _ in range(max_rounds):
        if time.time() - t_start > tmax:
            break
        if not improving:
            break
        improving = False
        one_idx = np.where(best_h == 1.0)[0]
        zero_idx = np.where(best_h == 0.0)[0]
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break
        # Try random swaps
        n_trials = min(5000, len(one_idx) * len(zero_idx))
        for _t in range(n_trials):
            if time.time() - t_start > tmax:
                break
            i = one_idx[rng.integers(len(one_idx))]
            j = zero_idx[rng.integers(len(zero_idx))]
            best_h[i] = 0.0
            best_h[j] = 1.0
            c5_try = compute_c5(best_h)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                improving = True
            else:
                best_h[i] = 1.0
                best_h[j] = 0.0

    return best_h, best_c5


def periodic_init(n, period_frac, duty_frac):

    for i in range(n):
        phase = (x[i] % period) / period
        h[i] = 1.0 if phase < duty_frac else 0.0
    h = fix_sum(h, n)
    return h
