def score_vector(h):
    h = np.asarray(h, dtype=float)

    h = np.clip(h, 0.0, 1.0)

    corr = np.correlate(h, h, mode="full")
    ac = corr[n - 1:2 * n - 1]
    prefix = np.empty(n + 1, dtype=float)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])
    scores = np.empty(2 * n - 1, dtype=float)
    scores[0] = total - ac[0]
    if n > 1:
        d = np.arange(1, n, dtype=int)
        scores[2 * d - 1] = prefix[n - d] - ac[1:]

    return scores

def random_fourier_start(n, rng):

def block_start(n, rng):
    n = int(n)
    if n < 4:
        return np.full(n, 0.5, dtype=float)
    L = int(min(12, max(3, rng.integers(3, max(4, n // 5 + 1)))))

    fill = 0
    block = 0
    while fill < n:
        amp = float(rng.uniform(0.18, 0.36))
        val = 0.5 + amp if (block % 2 == 0) else 0.5 - amp
        h[fill:fill + L] = val
        fill += L
        block += 1
    h = h[:n]
    h += float(rng.normal(0.0, 0.015)) * rng.normal(0.0, 1.0, size=n)
    return h

def soft_objective(x, alpha):
    check_deadline()
    x = np.asarray(x, dtype=float)
    n = int(x.size)
    if n == 0:
        return 1e10
    if not np.all(np.isfinite(x)):
        return 1e10
    h = np.clip(x, 0.0, 1.0)
    s = score_vector(h)
    m = float(np.max(s))

        else:
            val = m
    else:
        val = m
    if not math.isfinite(val):
        return 1e10
    return float(val)


def soft_gradient(x, alpha):
    check_deadline()
    x = np.asarray(x, dtype=float)
    n = int(x.size)
    if n == 0:
        return np.zeros(0, dtype=float)
    if not np.all(np.isfinite(x)):
        return np.zeros(n, dtype=float)
    h = np.clip(x, 0.0, 1.0)
    s = score_vector(h)
    m = float(np.max(s))
    w = np.zeros(2 * n - 1, dtype=float)
    alpha = float(alpha)
    if alpha > 1e-10:
        z = (s - m) / alpha
        ww = np.exp(z)
        A = float(ww.sum())
        if math.isfinite(A) and A > 0.0:
            w = ww / A
        else:
            w[int(np.argmax(s))] = 1.0
    else:
        tol = max(1e-7, 1e-6 * max(1.0, m))
        active = np.flatnonzero(s >= m - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(s))], dtype=int)
        w[active] = 1.0 / float(active.size)

    grad = np.zeros(n, dtype=float)
    w0 = float(w[0])
    if w0 > 0.0:
        grad += w0 * (1.0 - 2.0 * h)

    for d in range(1, n):
        p = float(w[2 * d - 1])
        q = float(w[2 * d])
        if p <= 1e-14 and q <= 1e-14:
            continue
        mlen = n - d
        if mlen <= 0:
            continue
        if p > 0.0:
            grad[:mlen] += p * (1.0 - h[mlen:])
            grad[mlen:] -= p * h[:mlen]
        if q > 0.0:
            grad[:mlen] -= q * h[mlen:]
            grad[mlen:] += q * (1.0 - h[:mlen])

    if not np.all(np.isfinite(grad)):
        return np.zeros(n, dtype=float)
    return grad

    return np.full((1, n), 1.0 / float(n), dtype=float)

def run_slsqp_chain(h_start, n, state, time_limit):

        finally:

            break

    if best_h is not None:

def grad_lag_single(h, idx):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    g = np.zeros(n, dtype=float)
    idx = int(idx)
    if idx == 0:
        return 1.0 - 2.0 * h
    if idx % 2 == 1:
        d = (idx + 1) // 2
        plus = True
    else:
        d = idx // 2
        plus = False
    m = n - d
    if m <= 0:
        return g
    if plus:
        g[:m] = 1.0 - h[d:]
        g[d:] -= h[:m]
    else:
        g[:m] = -h[d:]
        g[d:] += 1.0 - h[:m]
    return g


def lp_polish(h_in, n, state, time_limit):
    n = int(n)
    time_limit = float(time_limit)
    if n < 4 or time_limit < 5.0:
        return
    try:
        h = as_feasible(h_in, n)
    except Exception:
        return
    best_c = compute_c5(h)
    state_update(state, h, best_c, n)

    start = time.time()
    if n > 150:
        max_iters = 6
    elif n > 100:
        max_iters = 10
    else:
        max_iters = 18
    stall = 0
    it = 0
    while it < max_iters and (time.time() - start) < time_limit:
        check_deadline()
        scores = score_vector(h)
        if not np.all(np.isfinite(scores)):
            break
        m = float(np.max(scores))
        rows = []
        bvals = []
        for idx in range(int(scores.size)):
            g = grad_lag_single(h, idx)
            if not np.any(g):
                continue
            offset = float(scores[idx]) - float(np.dot(g, h))
            row = np.empty(n + 1, dtype=float)
            row[:n] = g
            row[n] = 0.0
            rows.append(row)
            bvals.append(offset)
        if len(rows) == 0:
            break
        A = np.vstack(rows)
        b = np.asarray(bvals, dtype=float)
        cvec = np.zeros(n + 1, dtype=float)
        cvec[-1] = 1.0
        Aeq = np.zeros((1, n + 1), dtype=float)
        Aeq[0, :n] = 1.0
        beq = np.array([0.5 * float(n)], dtype=float)
        bounds = [(0.0, 1.0)] * n
        bounds.append((0.0, m + 1.0))

        improved = False
        try:
            res = linprog(
                c=cvec,
                A_ub=A,
                b_ub=b,
                A_eq=Aeq,
                b_eq=beq,
                bounds=bounds,
                method="highs",
            )
        except BudgetError:
            break
        except Exception:
            res = None

        if res is not None and int(getattr(res, "status", -1)) == 0 and getattr(res, "x", None) is not None:
            h_lp = np.asarray(res.x[:n], dtype=float)
            if np.all(np.isfinite(h_lp)):
                for beta in (1.0, 0.75, 0.5, 0.35, 0.2, 0.1):
                    hx = (1.0 - beta) * h + beta * h_lp
                    try:
                        hx = as_feasible(hx, n)
                        cx = compute_c5(hx)
                    except Exception:
                        continue
                    if math.isfinite(cx) and cx < best_c - 1e-13:
                        h = hx
                        best_c = cx
                        improved = True
                        state_update(state, h, best_c, n)
                        break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break
        it += 1

    if h is not None:
        state_update(state, h, best_c, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _RUN_DEADLINE

    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0.0:
        budget_s = 60.0
    reserve = 25.0 if budget_s > 100.0 else max(3.0, 0.06 * budget_s)
    global_deadline = t0 + max(1.0, budget_s - reserve)
    _RUN_DEADLINE = global_deadline

    rng = np.random.default_rng(int(seed) if seed is not None else 42)

    parent = get_parent()
    if parent is not None:
        n0 = int(parent.size)
        h0 = parent.copy()
    else:
        n0 = 80
        h0 = random_fourier_start(n0, rng)
        h0 = as_feasible(h0, n0)

    try:
        h0 = as_feasible(h0, n0)
    except Exception:
        h0 = np.full(max(2, n0), 0.5, dtype=float)
    c0 = compute_c5(h0)
    state = {"h": h0, "c": float(c0), "n": int(n0)}
    initial_h = h0.copy()
    initial_c = float(c0)
    initial_n = int(n0)

    def time_left():
        if _RUN_DEADLINE is None:
            return max(0.0, budget_s - (time.time() - t0))
        return max(0.0, float(_RUN_DEADLINE) - time.time())

    rem = time_left()
    phase_time0 = min(430.0, max(80.0, 0.31 * float(budget_s)))
    phase_time0 = min(phase_time0, rem - 35.0)
    if phase_time0 >= 45.0:
        try:
            loc_h, loc_c = run_slsqp_chain(h0, n0, state, phase_time0)
            if loc_h is not None and time_left() > 70.0 and float(loc_c) < float(state["c"]) + 0.01:
                lp_polish(loc_h, n0, state, min(120.0, time_left() - 40.0))
        except Exception:
            pass

    max_n = max(120, min(3 * max(1, int(n0)), 210))
    n_list = []
    deltas = (10, 4, 20, -10, 30, 40, -20, 60, -30, 80, 50, -40)
    seen = set([int(n0)])
    for dt in deltas:
        nn = int(n0) + int(dt)
        if 20 <= nn <= max_n and nn not in seen:
            seen.add(nn)
            n_list.append(nn)
    for nn in (2 * int(n0), int(1.5 * int(n0))):
        if nn is not None and 20 <= int(nn) <= max_n and int(nn) not in seen:
            seen.add(int(nn))
            n_list.append(int(nn))
    n_list = [int(x) for x in n_list]

    scenario_times = (150.0, 125.0, 105.0, 90.0, 78.0, 65.0, 55.0, 50.0, 45.0, 40.0)
    for si, nn in enumerate(n_list):
        rem = time_left()
        if rem < 85.0:
            break
        T = scenario_times[si] if si < len(scenario_times) else 40.0
        T = min(T, rem - 45.0)
        if T < 40.0:
            break
        try:
            starts = build_starts(nn, parent, state, rng, limit=3 if T > 110.0 else 2)
            if not starts:
                continue
            for k0, hstart in enumerate(starts):
                if k0 == 0:
                    lt = 0.62 * T
                elif k0 == 1:
                    lt = 0.28 * T
                else:
                    lt = 0.10 * T
                lt = min(lt, time_left() - 35.0)
                if lt < 25.0:
                    break
                try:
                    loc_h, loc_c = run_slsqp_chain(hstart, nn, state, lt)
                    if loc_h is not None and time_left() > 50.0 and float(loc_c) < float(state["c"]) + 0.006:
                        lp_polish(loc_h, nn, state, min(0.28 * T, time_left() - 25.0))
                except Exception:
                    pass
        except Exception:
            pass
        if time_left() < 170.0:
            break

    rem = time_left()
    if rem > 60.0 and state.get("h") is not None:
        try:
            lp_polish(state["h"], state["n"], state, min(rem - 25.0, 300.0))
        except Exception:
            pass

    rem = time_left()
    if rem > 90.0 and state.get("h") is not None:
        try:
            lt = min(rem - 30.0, 140.0)
            loc_h, loc_c = run_slsqp_chain(state["h"], state["n"], state, lt)
            if loc_h is not None and time_left() > 40.0:
                lp_polish(loc_h, state["n"], state, min(time_left() - 20.0, 90.0))
        except Exception:
            pass

    rem = time_left()
    if rem > 50.0 and state.get("h") is not None:
        try:
            lp_polish(state["h"], state["n"], state, min(rem - 10.0, 120.0))
        except Exception:
            pass

    if initial_c <= state["c"] + 1e-14:
        h_final = initial_h
        c_final = initial_c
        n_final = initial_n
    else:
        h_final = state.get("h")
        c_final = state.get("c")
        n_final = state.get("n")

    if h_final is None or c_final is None or not math.isfinite(c_final):
        h_final = initial_h
        c_final = initial_c
        n_final = initial_n
    n_final = int(n_final)
    try:
        h_final = np.asarray(h_final, dtype=float)
        if h_final.size != n_final:
            h_final = resample_linear(h_final, n_final)
        h_final = as_feasible(h_final, n_final)
        c_final = compute_c5(h_final)
    except Exception:
        h_final = as_feasible(initial_h, initial_n)
        c_final = compute_c5(h_final)
        n_final = int(initial_n)

    if (not math.isfinite(c_final)) or (h_final.min() < -1e-9) or (h_final.max() > 1.0 + 1e-9):
        h_final = as_feasible(initial_h, initial_n)
        c_final = compute_c5(h_final)
        n_final = int(initial_n)

    return h_final, float(c_final), int(n_final)
