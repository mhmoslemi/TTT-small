import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-15
        else:
            mask = h < 1.0 - 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] -= diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / n))


def obj_smooth(h, beta, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    lse = m + np.log(np.sum(np.exp(c - m))) / beta
    return float(lse * (2.0 / n))


def grad_smooth(h, beta, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    w = np.exp(c - m)
    w /= np.sum(w)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-15:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            if n - lag > 0:
                grad[:n - lag] -= wk * h[lag:]
        else:
            l = -lag
            grad[:n - l] += wk * (1.0 - h[l:])
            if n - l > 0:
                grad[l:] -= wk * h[:n - l]
    return grad * dx


def obj_hard(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / n))


def grad_hard(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")

    lag = k_max - (n - 1)
    dx = 2.0 / n

    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:])
        if n - l > 0:
            grad[l:] -= h[:n - l]
    return grad * dx


def obj_penalized(h, n, lam):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    lse = m + np.log(np.sum(np.exp(c - m))) / 50.0
    base = lse * (2.0 / n)
    penalty = lam * (np.sum(h) - n / 2.0) ** 2
    return float(base + penalty)


def grad_penalized(h, n, lam):
    grad_c5 = grad_smooth(h, 50.0, n)
    penalty_grad = 2.0 * lam * (np.sum(h) - n / 2.0) * np.ones(n)
    return grad_c5 + penalty_grad


def subgradient_polish(h, n, max_iter, t0, budget_s, margin_s):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.05
    n_active = 10
    for it in range(max_iter):
        if time.time() - t0 > budget_s - margin_s:
            break
        c = np.correlate(h, 1.0 - h, mode="full")
        # Use average of top n_active subgradients
        top_k = np.argsort(c)[::-1][:n_active]
        top_k = top_k[top_k > 0]  # exclude lag 0 (not meaningful)
        if len(top_k) == 0:
            top_k = np.argsort(c)[::-1][:n_active]
        grad = np.zeros(n)
        for k_idx in top_k:
            lag = k_idx - (n - 1)
            g = np.zeros(n)
            if lag >= 0:
                g[lag:] += (1.0 - h[:n - lag])
                if n - lag > 0:
                    g[:n - lag] -= h[lag:]
            else:
                l = -lag
                g[:n - l] += (1.0 - h[l:])
                if n - l > 0:
                    g[l:] -= h[:n - l]
            grad += g
        grad /= len(top_k)
        grad *= (2.0 / n)
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)

            best_c5 = c5_new
            alpha = min(0.1, alpha * 1.05)
