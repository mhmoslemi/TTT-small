import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx


def c5_all_values(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv * dx


def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-15
            if not np.any(free):
                break
            step = diff / np.sum(free)
            h[free] = np.minimum(h[free] + step, 1.0)
        else:
            free = h > 1e-15
            if not np.any(free):
                break
            step = diff / np.sum(free)
            h[free] = np.maximum(h[free] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    residual = target - s
    if abs(residual) > 1e-12:
        h += residual / n
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_smooth_gradient(h, n_points, temp=0.05):
    n = n_points
    dx = 2.0 / n
    c_vals = np.correlate(h, 1.0 - h, mode='full') * dx
    
    c_max = np.max(c_vals)
    weights = np.exp((c_vals - c_max) / temp)
    weights /= np.sum(weights)
    
    w = weights * dx
    grad = np.zeros(n)
    
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-18:
            continue
        d = n - 1 - m
        
        # First term: (1-h[j+d]) for valid j where 0<=j<n and 0<=j+d<n
        if d >= 0:
            j_lo = 0
            j_hi = n - 1 - d
        else:
            j_lo = -d
            j_hi = n - 1
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            jd = idx + d
            grad[j_lo:j_hi + 1] += wm * (1.0 - h[jd])
        
        # Second term: -h[j-d] for valid j where 0<=j<n and 0<=j-d<n
        if d >= 0:
            j_lo2 = d
            j_hi2 = n - 1
        else:
            j_lo2 = 0
            j_hi2 = n - 1 + d
        if j_lo2 <= j_hi2:
            idx2 = np.arange(j_lo2, j_hi2 + 1)
            jd2 = idx2 - d
            grad[j_lo2:j_hi2 + 1] -= wm * h[jd2]
    
    return grad


def adam_optimize_smooth(h0, n_points, temp_start=0.1, temp_end=0.01,
                         n_iter=500, lr=0.01, beta1=0.9, beta2=0.999,
                         eps=1e-8, time_fn=None, time_limit=0):
    n = n_points
    h = h0.copy()
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    best_h = h.copy()
    best_val = compute_c5(h, n)
    
    for t in range(1, n_iter + 1):
        if time_fn is not None:
            remaining = time_limit - time_fn()
            if remaining < 5:
                break
            
        # Anneal temperature
        frac = t / n_iter
        temp = temp_start * (temp_end / temp_start) ** frac
        
        # Compute gradient of smooth objective
        grad = compute_smooth_gradient(h, n, temp)
        
        # Adam update
        m_hat = beta1 * m_hat + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * grad ** 2
        m_hat_corr = m_hat / (1 - beta1 ** t)
        v_hat_corr = v_hat / (1 - beta2 ** t)
        
        h_new = h - lr * m_hat_corr / (np.sqrt(v_hat_corr) + eps)
        h_new = project_feasible(h_new, n)
        
        # Evaluate
        val = compute_c5(h_new, n)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_val


def slsqp_smooth(h0, n_points, temp=0.02, maxiter=300):
    n = n_points
    target = n / 2.0
    
    def obj(x):
        c_vals = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(c_vals)
        return c_max + temp * np.log(np.sum(np.exp((c_vals - c_max) / temp)))
    
    def obj_grad(x):
        return compute_smooth_gradient(x, n, temp)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0.copy(), jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        x = project_feasible(res.x, n)
        return x, compute_c5(x, n)
    except Exception:
        return h0, compute_c5(h0, n)

def interpolate_solution(h_coarse, n_coarse, n_fine):

            h_bin[remove_idx] = 0.0

            zero_mask = h_bin == 0.0
            zero_vals = h_bin[zero_mask]
            n_add = target_ones - n_ones
            add_idx = np.where(zero_mask)[0][np.argsort(-zero_vals)[:n_add]]
            h_bin[add_idx] = 1.0
        
        h_bin, v_bin = binary_local_search(h_bin, best_n, n_iter=3000)
        if v_bin < best_val:
            best_h, best_val = h_bin, v_bin
        
        # Also try multiple random binaries
        for _ in range(10):
            if time_left() < 30:
                break
            h_rbin = random_binary(best_n, rng)
            h_rbin, v_rbin = binary_local_search(h_rbin, best_n, n_iter=2000)
            if v_rbin < best_val:
                best_h, best_val = h_rbin, v_rbin
    
    # Phase 6: Fine-tune around best with small perturbations
    if time_left() > 80:
        for scale in [0.02, 0.05, 0.1]:
            if time_left() < 30:
                break
            for _ in range(5):
                if time_left() < 15:
                    break
                h_pert = best_h + rng.normal(0, scale, best_n)
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = project_feasible(h_pert, best_n)
                h_try, v_try = adam_optimize_smooth(
                    h_pert, best_n, temp_start=0.03, temp_end=0.003,
                    n_iter=100, lr=0.005)
                h_try = project_feasible(h_try, best_n)
                v_try = compute_c5(h_try, best_n)
                if v_try < best_val:
                    best_h, best_val = h_try, v_try
    
    # Phase 7: Final SLSQP polish
    if time_left() > 30:
        try:
            h_try, v_try = slsqp_smooth(best_h, best_n, temp=0.005, maxiter=500)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
        except Exception:
            pass
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_val = compute_c5(best_h, best_n)
    
    # Ensure constraints are met exactly
    assert np.all(best_h >= -1e-15) and np.all(best_h <= 1.0 + 1e-15)
    best_h = np.clip(best_h, 0.0, 1.0)
    s = np.sum(best_h)
    target = best_n / 2.0
    if abs(s - target) > 1e-10:
        best_h = project_feasible(best_h, best_n)
    
    best_val = compute_c5(best_h, best_n)
    
    return (best_h, best_val, best_n)
