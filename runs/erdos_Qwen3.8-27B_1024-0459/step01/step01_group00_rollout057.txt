import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h_prime, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        h_test = np.clip(h_prime - mid, 0.0, 1.0)
        s = np.sum(h_test)
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h_prime - tau, 0.0, 1.0)

def compute_gradient_v2(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = corr - cmax
    exp_terms = np.exp(np.clip(beta * shifted, -500.0, 500.0))
    p = exp_terms / np.sum(exp_terms)
    one_minus_h = 1.0 - h
    jj = np.arange(n)[:, np.newaxis]
    kk = np.arange(2 * n - 1)[np.newaxis, :]
    idx1 = jj + kk - (n - 1)
    mask1 = (idx1 >= 0) & (idx1 < n)
    idx2 = jj - kk + (n - 1)
    mask2 = (idx2 >= 0) & (idx2 < n)
    p_row = p[np.newaxis, :]
    t1 = (p_row * mask1.astype(float) * one_minus_h[np.clip(idx1, 0, n - 1)]).sum(axis=1)
    t2 = (p_row * mask2.astype(float) * h[np.clip(idx2, 0, n - 1)]).sum(axis=1)
    grad = dx * (t1 - t2)
    return corr, p, grad

def pgd_optimize(h_init, n, beta_start, beta_end, n_beta_steps, max_iter_per_beta, step_base):
    h = make_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for b_step in range(n_beta_steps):
        if n_beta_steps <= 1:
            beta = beta_end
        else:
            frac = b_step / (n_beta_steps - 1)
            beta = beta_start * (beta_end / beta_start) ** frac
        step = step_base / max(beta, 1.0)
        no_improve = 0
        for it in range(max_iter_per_beta):
            corr, p, grad = compute_gradient_v2(h, n, beta)
            h_try = h - step * grad
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            c5_cur = compute_c5(h)
            if c5_try < c5_cur:
                h = h_try
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                step = min(step * 1.05, step_base / max(beta, 1.0))
            else:
                step *= 0.7
                no_improve += 1
                if no_improve > 100:
                    break
            if step < 1e-12:
                break
    return best_h, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def generate_binary_init(n, rng, method):
    if method == 'random':
        positions = rng.choice(n, n // 2, replace=False)
        h = np.zeros(n)
        h[positions] = 1.0
        return h
    elif method == 'thue_morse':
        counts = np.array([bin(i).count('1') for i in range(n)])
        even = np.where(counts % 2 == 0)[0]
        odd = np.where(counts % 2 == 1)[0]
        h = np.zeros(n)
        need = n // 2
        if len(even) >= need:
            h[even[:need]] = 1.0
        else:
            h[even] = 1.0
            h[odd[:need - len(even)]] = 1.0
        return h
    elif method == 'golden':
        phi = 0.6180339887
        positions = sorted(int(round(((i + 0.5) / n) * phi * n)) % n for i in range(n))
        h = np.zeros(n)
        h[positions[:n // 2]] = 1.0
        if h.sum() < n // 2:
            h[rng.choice(np.where(h == 0)[0], int(n // 2 - h.sum()), replace=False)] = 1.0
        elif h.sum() > n // 2:
            extra = int(h.sum() - n // 2)
            h[rng.choice(np.where(h == 1)[0], extra, replace=False)] = 0.0
        return h
    elif method == 'segments':
        seg = n // 4
        h = np.zeros(n)
        off = rng.randint(0, n)
        h[(off:off + seg) % n] = 1.0
        h[(off + n // 2:off + n // 2 + seg) % n] = 1.0
        if h.sum() != n // 2:
            flat = h.ravel()
            flat = (flat > 0.5).astype(float)
            if flat.sum() > n // 2:
                flat[rng.choice(np.where(flat == 1)[0], int(flat.sum() - n // 2), replace=False)] = 0
            elif flat.sum() < n // 2:
                flat[rng.choice(np.where(flat == 0)[0], int(n // 2 - flat.sum()), replace=False)] = 1
            h = flat
        return h
    else:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        return h

def coordinate_polish(h, n, max_rounds, step_sizes):
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            for ds in step_sizes:
                for delta in [ds, -ds]:
                    h_try = h.copy()
                    h_try[j] = np.clip(h[j] + delta, 0.0, 1.0)
                    h_try = make_feasible(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-12:
                        h = h_try
                        best_c5 = c5_try
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return h, best_c5

def swap_polish(h, n, max_rounds):
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        n_swaps = min(5000, n * n // 4)
        for _ in range(n_swaps):
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            if i == j:
                continue
            if abs(h[i] - h[j]) < 1e-8:
                continue
            h_try = h.copy()
            h_try[i], h_try[j] = h[j], h[i]
            h_try = make_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                h = h_try
                best_c5 = c5_try
                improved = True
                if improved and c5_try < best_c5:
                    pass
        if not improved:
            break
    return h, best_c5
