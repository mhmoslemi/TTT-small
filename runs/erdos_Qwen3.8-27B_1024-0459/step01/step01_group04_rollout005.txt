def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0 or not np.all(np.isfinite(h)):
        return 1e6
    h = np.clip(h, 0.0, 1.0)
    conv = np.correlate(h, 1.0 - h, mode="full")
    c = float(np.max(conv) * (2.0 / float(n)))
    if not math.isfinite(c):
        return 1e6
    return c


def is_valid_candidate(h):

        return False
    if not np.all(np.isfinite(h)):
        return False
    if float(h.min()) < -1e-7 or float(h.max()) > 1.0 + 1e-7:
        return False
    if abs(float(h.sum()) - 0.5 * float(n)) > 1e-5:
        return False
    return True


def _adjust_residual(h, target):
    h = np.asarray(h, dtype=float).copy()
    guard = 0
    diff = target - float(h.sum())
    n = h.size
    while abs(diff) > 1e-12 and guard <= n + 5:
        guard += 1
        if diff > 0.0:
            mask = h < 1.0
            if not np.any(mask):
                break
            cnt = int(np.count_nonzero(mask))
            per = diff / float(cnt)
            inc = min(per, float((1.0 - h[mask]).min()))
            if inc <= 1e-15:
                break
            h[mask] += inc
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            cnt = int(np.count_nonzero(mask))
            per = -diff / float(cnt)
            dec = min(per, float(h[mask].min()))
            if dec <= 1e-15:
                break
            h[mask] -= dec
        diff = target - float(h.sum())

    diff = target - float(h.sum())
    if abs(diff) > 1e-13:
        if diff > 0.0:

def score_vector(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return np.zeros(0, dtype=float), 0.0
    h = np.clip(h, 0.0, 1.0)
    oh = 1.0 - h
    sc = np.empty(2 * n - 1, dtype=float)
    sc[0] = float(np.dot(h, oh))
    for d in range(1, n):
        m = n - d
        sc[2 * d - 1] = float(np.dot(h[:m], oh[d:]))
        sc[2 * d] = float(np.dot(h[d:], oh[:m]))
    mx = float(np.max(sc))
    return sc, mx


def add_lag_grad(g, idx, coef, h, n):
    if coef == 0.0:
        return
    if idx == 0:
        g += coef * (1.0 - 2.0 * h)
        return
    if idx % 2 == 1:
        d = (idx + 1) // 2
    else:
        d = idx // 2
    m = n - d
    if m <= 0:
        return
    if idx % 2 == 1:
        g[:m] += coef * (1.0 - h[d:])
        g[d:] += coef * (-h[:m])
    else:
        g[:m] += coef * (-h[d:])
        g[d:] += coef * (1.0 - h[:m])


def soft_score_gradient(h, alpha):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 0.0, np.zeros(0, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    sc, m = score_vector(h)
    g = np.zeros(n, dtype=float)

    if alpha > 1e-12:
        z = (sc - m) / alpha
        np.minimum(z, 0.0, out=z)
        np.maximum(z, -50.0, out=z)
        w = np.exp(z)
        A = float(np.sum(w))
        if not math.isfinite(A) or A <= 0.0:
            w.fill(0.0)
            w[int(np.argmax(sc))] = 1.0
        else:
            w /= A
            val = m + alpha * math.log(A)
    else:
        val = m
        tol = max(1e-9, 1e-7 * max(1.0, m))
        active = sc >= (m - tol)
        if not np.any(active):
            active = np.zeros_like(sc, dtype=bool)
            active[int(np.argmax(sc))] = True
        w = np.where(active, 1.0 / float(np.count_nonzero(active)), 0.0)

    for idx in range(sc.size):
        wi = float(w[idx])
        if wi > 1e-10:
            add_lag_grad(g, idx, wi, h, n)

    g *= (2.0 / float(n))
    if not math.isfinite(val):
        val = 1e6
    if not np.all(np.isfinite(g)):
        g.fill(0.0)
    return float(val), g

def u_calc(u, alpha):

def optimize_u_start(u0, n, state, end_time, time_limit):
    global _STAGE_END
    n = int(n)
    if n <= 0:
        return
    u = np.nan_to_num(np.asarray(u0, dtype=float), nan=0.0, posinf=0.0, neginf=0.0)
    u = np.clip(u, -_MAX_U, _MAX_U)
    u = u - np.mean(u)
    start_t = time.time()

    scale = max(1.0, float(n) / 140.0)
    alphas = (0.6 * scale, 0.2 * scale, 0.07 * scale, 0.02 * scale)
    stage_time = max(6.0, float(time_limit) / float(len(alphas)))
    bounds = [(-_MAX_U, _MAX_U)] * n

    for alpha in alphas:
        if time.time() >= end_time - 8.0 or time.time() >= start_t + float(time_limit) - 2.0:
            break
        stage_deadline = min(start_t + float(time_limit), time.time() + stage_time, end_time - 5.0)
        if stage_deadline <= time.time() + 2.0:
            continue
        if n <= 120:
            max_iter = int(max(25, min(300, 12000.0 / float(n))))
        elif n <= 220:
            max_iter = int(max(18, min(180, 8000.0 / float(n))))
        else:
            max_iter = int(max(12, min(120, 5000.0 / float(n))))

        _STAGE_END = stage_deadline
        res = None
        try:
            res = minimize(
                u_objective,
                u,
                args=(alpha,),
                method="L-BFGS-B",
                jac=True,
                bounds=bounds,
                options={"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-8, "maxls": 18, "disp": 0},
                callback=budget_callback,
            )
        except BudgetError:
            res = None
        except Exception:
            res = None
        finally:
            _STAGE_END = None

        if res is not None and getattr(res, "x", None) is not None:
            u = np.asarray(res.x, dtype=float)
            u = np.nan_to_num(u, nan=0.0, posinf=0.0, neginf=0.0)
            u = np.clip(u, -1.5 * _MAX_U, 1.5 * _MAX_U)
            u = u - np.mean(u)

        p = p_from_centered_u(u)
        state.update(p, None, n)

    if time.time() < min(start_t + float(time_limit), end_time) - 8.0:
        rem = min(5.0, min(start_t + float(time_limit), end_time) - time.time())
        if rem > 3.0:
            direct_polish(p, n, state, min(start_t + float(time_limit), end_time), rem)


def run_lbgfs_starts(starts, n, state, end_time, max_starts=8):
    if not starts:
        return
    n = int(n)
    if n <= 0 or time.time() >= end_time - 35.0:
        return
    starts = starts[:max_starts]
    available = float(end_time - time.time())
    if available < 35.0:
        return
    cap = 120.0 if n <= 120 else 170.0
    per = max(25.0, min(cap, available / (float(len(starts)) + 0.25)))
    for _, u in starts:
        if time.time() >= end_time - 40.0:
            break
        tlim = min(per, float(end_time - time.time() - 40.0))
        if tlim < 15.0:
            break
        try:
            optimize_u_start(u, n, state, end_time, tlim)
        except Exception:
            pass


def slsqp_polish(h, n, state, end_time, time_limit):
    n = int(n)
    h = as_feasible(h, n)
    if time.time() >= end_time - 25.0:
        state.update(h, None, n)
        return
    available = min(float(time_limit), float(end_time - time.time() - 25.0))
    if available < 15.0 or n > 220:
        state.update(h, None, n)
        return

    alphas = (0.25, 0.08, 0.02)
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": constraint_eq, "jac": constraint_jac}]

    for alpha in alphas:
        if time.time() >= end_time - 12.0:
            break
        max_iter = int(max(12, min(60, 3200.0 / float(n))))
        res = None
        try:
            res = minimize(
                slsqp_objective,
                h,
                args=(alpha,),
                jac=slsqp_gradient,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": max_iter, "ftol": 1e-9, "disp": 0},
            )
        except Exception:
            res = None
        if res is not None and getattr(res, "x", None) is not None:
            state.update(res.x, None, n)
        if state is not None and int(state.n) == n:
            h = state.h.copy()
        else:
            h = as_feasible(h, n)
    state.update(h, None, n)


def direct_polish(h, n, state, end_time, time_limit):
    n = int(n)
    h = as_feasible(h, n)
    best = float(compute_c5(h))
    state.update(h, best, n)
    stage_end = min(float(end_time) - 2.0, time.time() + float(time_limit))
    step = 0.05
    while time.time() < stage_end - 1.0:
        sc, _ = score_vector(h)
        _, g = soft_score_gradient(h, 0.0)
        if g is None or not np.all(np.isfinite(g)) or float(np.linalg.norm(g)) < 1e-14:
            break
        improved = False
        for fac in (1.0, 0.5, 0.25, 0.1, 0.05):
            q = as_feasible(h - step * fac * g, n)
            c = float(compute_c5(q))
            if c < best - 1e-12:
                h = q
                best = c
                state.update(h, best, n)
                improved = True
                break
        if not improved:
            step *= 0.7
            if step < 1e-5:
                break
    state.update(h, None, n)


def make_binary_from_h(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float)
    if h.size != n:
        h = upsample_linear(h, n)
    b = (h >= 0.5).astype(np.int8)
    target = n // 2
    cnt = int(np.count_nonzero(b))
    if cnt > target:
        ones = np.flatnonzero(b == 1)
        if ones.size > 0:
            order = ones[np.argsort(h[ones])][: max(0, cnt - target)]
            b[order] = 0
    elif cnt < target:
        zeros = np.flatnonzero(b == 0)
        need = target - cnt
        if zeros.size > 0:
            order = zeros[np.argsort(-h[zeros])][: max(0, min(need, zeros.size))]
            b[order] = 1
    return b.astype(float)


def binary_search(h_start, n, state, end_time, time_limit, rng):
    n = int(n)
    if n <= 0 or n % 2 != 0 or n > 220:
        return
    if time.time() >= end_time - 25.0:
        return
    stage_end = min(float(end_time) - 3.0, time.time() + float(time_limit))
    h = make_binary_from_h(h_start, n).astype(float)
    cur = float(compute_c5(h))
    best_local = cur
    state.update(h, cur, n)

    temperature = 0.02
    since_best = 0
    active_idx = None
    it = 0
    while time.time() < stage_end - 1.0:
        it += 1
        if it % 20 == 0:
            sc, mx = score_vector(h)
            tol = max(1e-9, 1e-7 * max(1.0, mx))
            active_idx = np.flatnonzero(sc >= mx - tol)
            if active_idx.size == 0:
                active_idx = np.array([int(np.argmax(sc))], dtype=int)

        pair = None
        if rng.random() < 0.72 and active_idx is not None and active_idx.size > 0:
            idx = int(active_idx[int(rng.integers(active_idx.size))])
            if idx > 0:
                if idx % 2 == 1:
                    d = (idx + 1) // 2
                    cand = np.flatnonzero((h[:n - d] == 1.0) & (h[d:] == 0.0))
                else:
                    d = idx // 2
                    cand = np.flatnonzero((h[:n - d] == 0.0) & (h[d:] == 1.0))
                if cand.size > 0:
                    a = int(cand[int(rng.integers(cand.size))])
                    pair = (a, a + d)

        if pair is None:
            zeros = np.flatnonzero(h == 0.0)
            ones = np.flatnonzero(h == 1.0)
            if zeros.size == 0 or ones.size == 0:
                break
            i = int(zeros[int(rng.integers(zeros.size))])
            j = int(ones[int(rng.integers(ones.size))])
            pair = (i, j)

        i, j = pair
        if i == j:
            continue

        h[i], h[j] = h[j], h[i]
        c = float(compute_c5(h))
        accept = False
        improved_cur = False
        if c < cur - 1e-12:
            accept = True
            improved_cur = True
        else:
            delta = c - cur
            if delta <= 1e-12:
                if since_best > 400 and rng.random() < 0.25:
                    accept = True
            elif delta < 8.0 * temperature:
                if rng.random() < math.exp(-delta / max(1e-6, temperature)):
                    accept = True

        if not accept:
            h[i], h[j] = h[j], h[i]
        else:
            cur = c
            if improved_cur:
                temperature = max(0.002, temperature * 0.995)
            else:
                temperature = max(0.002, temperature * 0.985)

        if c < best_local - 1e-12:
            best_local = c
            state.update(h, c, n)
            since_best = 0
        else:
            since_best += 1

        if since_best > 4000:
            for _ in range(5):
                zeros = np.flatnonzero(h == 0.0)
                ones = np.flatnonzero(h == 1.0)
                if zeros.size == 0 or ones.size == 0:
                    break
                a = int(zeros[int(rng.integers(zeros.size))])
                b = int(ones[int(rng.integers(ones.size))])
                h[a], h[b] = h[b], h[a]
            cur = float(compute_c5(h))
            if cur < best_local - 1e-12:
                best_local = cur
                since_best = 0
            else:
                since_best = 0
            temperature = 0.025

    state.update(h, None, n)


def synthetic_h(n, rng):
    n = int(n)
    x = (np.arange(n) + 0.5) / float(n)
    best_h = as_feasible(np.full(n, 0.5), n)
    best_c = compute_c5(best_h)
    for f in (1, 2, 3, 4, 5):
        for ph in (0.0, 0.5 * math.pi):
            h = 0.5 + 0.38 * np.cos(2.0 * math.pi * float(f) * x + ph)
            h = as_feasible(h, n)
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h
    hb = np.zeros(n, dtype=np.int8)
    target = n // 2
    idx = rng.choice(n, size=target, replace=False)
    hb[idx] = 1
    hs = np.convolve(hb.astype(float), np.array([0.25, 0.5, 0.25]), mode="same")
    hs = as_feasible(hs, n)
    c = compute_c5(hs)
    if c < best_c:
        best_c = c
        best_h = hs
    return best_h


def evaluate_ulsampled_ns(parent, state, n0, end_time):
    n0 = int(n0)
    if n0 <= 0:
        return
    seen = set()
    for mult in (1.0, 1.5, 2.0, 3.0, 4.0):
        n = int(round(float(mult) * float(n0)))
        if n % 2 != 0:
            n += 1
        if n < 6 or n > max(10, 5 * n0):
            continue
        if n in seen:
            continue
        seen.add(n)
        if time.time() >= end_time - 15.0:
            break
        base = parent if parent is not None else state.h
        if base is None:
            continue
        h = base.copy() if int(base.size) == n else upsample_linear(base, n)
        h = as_feasible(h, n)
        state.update(h, None, n)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0.0:
        budget_s = 1300.0
    reserve = 25.0 if budget_s > 60.0 else max(3.0, 0.1 * budget_s)
    end_time = t0 + max(1.0, budget_s - reserve)

    try:
        seed_int = int(seed)
    except Exception:
        seed_int = 42
    rng = np.random.default_rng(seed_int)

    parent = get_parent()
    if parent is not None:
        n0 = int(parent.size)
        h0 = as_feasible(parent, n0)
    else:
        n0 = 70
        h0 = synthetic_h(n0, rng)

    c0 = float(compute_c5(h0))
    if not math.isfinite(c0) or c0 > 2.0:
        h0 = as_feasible(np.full(n0, 0.5), n0)
        c0 = float(compute_c5(h0))

    state = BestState(h0.copy(), c0, n0)
    evaluate_ulsampled_ns(parent, state, n0, end_time)

    if time.time() < end_time - 130.0:
        starts = generate_u_starts(n0, parent, state, rng, 10)
        if not starts:
            u = make_u_from_h(state.h if int(state.n) == n0 else upsample_linear(state.h, n0), n0)
            p = as_feasible(p_from_centered_u(u), n0)
            starts = [(float(compute_c5(p)), u)]
        run_lbgfs_starts(starts, n0, state, end_time, 10)

        if time.time() < end_time - 85.0:
            slsqp_polish(
                state.h if int(state.n) == n0 else upsample_linear(state.h, n0),
                n0,
                state,
                end_time,
                max(18.0, min(80.0, float(end_time - time.time() - 45.0))),
            )

        if n0 % 2 == 0 and time.time() < end_time - 70.0:
            binary_search(
                state.h if int(state.n) == n0 else upsample_linear(state.h, n0),
                n0,
                state,
                end_time,
                max(30.0, min(110.0, float(end_time - time.time() - 40.0))),
                rng,
            )

    n2 = 2 * n0 if n0 % 2 == 0 else 2 * (n0 + 1)
    if n2 > 0 and n2 <= max(6, 6 * n0) and time.time() < end_time - 260.0:
        starts = generate_u_starts(n2, parent, state, rng, 7)
        run_lbgfs_starts(starts, n2, state, end_time, 5)

        if time.time() < end_time - 90.0 and n2 <= 200:
            slsqp_polish(
                state.h if int(state.n) == n2 else upsample_linear(state.h, n2),
                n2,
                state,
                end_time,
                max(15.0, min(55.0, float(end_time - time.time() - 45.0))),
            )

        if n2 % 2 == 0 and time.time() < end_time - 70.0 and n2 <= 200:
            binary_search(
                state.h if int(state.n) == n2 else upsample_linear(state.h, n2),
                n2,
                state,
                end_time,
                max(25.0, min(80.0, float(end_time - time.time() - 45.0))),
                rng,
            )

    n4 = 4 * n0 if n0 % 2 == 0 else 4 * (n0 + 1)
    if n4 > 0 and n4 > n2 and n4 <= max(14, 8 * n0) and n4 <= 320 and time.time() < end_time - 260.0:
        starts = generate_u_starts(n4, parent, state, rng, 5)
        run_lbgfs_starts(starts, n4, state, end_time, 3)

    if time.time() < end_time - 25.0:
        nf = int(state.n)
        hf = state.h.copy()
        direct_polish(hf, nf, state, end_time, min(25.0, max(5.0, float(end_time - time.time() - 18.0))))

        if nf <= 160 and time.time() < end_time - 30.0:
            slsqp_polish(
                state.h,
                nf,
                state,
                end_time,
                min(35.0, max(8.0, float(end_time - time.time() - 18.0))),
            )

    h_final = as_feasible(state.h, state.n)
    c_final = float(compute_c5(h_final))
    if not is_valid_candidate(h_final):
        h_final = as_feasible(h_final, h_final.size)
        c_final = float(compute_c5(h_final))

    return h_final, float(c_final), int(h_final.size)
