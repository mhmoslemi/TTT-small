def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    mask = (h > 1e-10) & (h < 1.0 - 1e-10)
    if np.sum(mask) > 0:
        h[mask] += diff / np.sum(mask)
    h = np.clip(h, 0.0, 1.0)
    return h

        # For 0→1 flip at position p: delta_full[m] = (1-2*h[m-p]) if 0<=m-p<n

        for i, p in enumerate(zeros):
            start_m = max(0, p)
            end_m = min(nfull, p + n)
            seg = slice(start_m, end_m)
            idx = np.arange(start_m, end_m) - p
            D_plus[i, seg] = 1.0 - 2.0 * h[idx]
