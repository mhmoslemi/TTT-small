- If all lines are decreasing: minimum at t=1
- If all lines are increasing: minimum at t=0
- Otherwise: at the intersection of the line with the most negative slope and the line with the most positive slope (among those that are "active" at the optimum)

- t* where the slope changes from negative to positive

import numpy as np
import time
import math
from scipy.optimize import minimize

def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 1e6
    h = np.clip(h, 0.0, 1.0)
    conv = np.correlate(h, 1.0 - h, mode="full")
    c = float(np.max(conv) * (2.0 / n))
    return c if math.isfinite(c) else 1e6

def score_vector(h, n):
    n = int(n)
    total = float(np.sum(h))
    corr = np.correlate(h, h, mode="full")
    ac = corr[n - 1:n - 1 + n]
    prefix = np.zeros(n + 1)
    np.cumsum(h, out=prefix[1:])
    scores = np.zeros(2 * n - 1)
    scores[0] = total - ac[0]
    for d in range(1, n):
        scores[2 * d - 1] = prefix[n - d] - ac[d]
        scores[2 * d] = total - prefix[d] - ac[d]
    return scores

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(np.sum(h))
    if abs(diff) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(np.sum(h))
    if diff > 1e-12:
        mask = h < 1.0
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    elif diff < -1e-12:
        mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def coordinate_descent(h, n, max_passes=5000, grid_size=80, deadline=None):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    h = project_feasible(h, n)
    grid = np.linspace(0.0, 1.0, grid_size)
    best_c = compute_c5(h)
    
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if deadline is not None and time.time() > deadline:
                break
            old_val = h[i]
            A = np.zeros(2 * n - 1)
            B = np.zeros(2 * n - 1)
            C = np.zeros(2 * n - 1)
            A[0] = 1.0
            B[0] = -1.0
            C[0] = float(np.sum(h * (1.0 - h))) - h[i] * (1.0 - h[i])
            for d in range(1, n):
                p1 = 2 * d - 1
                p2 = 2 * d
                if d <= n - 1 - i:
                    A[p1] += (1.0 - h[i + d])
                    A[p2] += (-h[i + d])
                    C[p2] += h[i + d]
                if d <= i:
                    A[p1] += (-h[i - d])
                    C[p1] += h[i - d]
                    A[p2] += (1.0 - h[i - d])
            vals = A[None, :] * grid[:, None] + B[None, :] * (grid[:, None] ** 2) + C[None, :]
            obj = vals.max(axis=1)
            best_t = float(grid[obj.argmin()])
            if abs(best_t - old_val) > 1e-12:
                h[i] = best_t
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
    return h, best_c

def upsample_linear(parent, n):
    n0 = int(parent.size)
    if n == n0:
        return parent.copy()
    if n0 <= 0:
        return np.full(n, 0.5)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x, x0, parent.astype(float))

def perturbed_start(h, n, rng, sigma=0.1):
    h2 = h + sigma * rng.standard_normal(n)
    h2 = project_feasible(h2, n)
    return h2

def slsqp_obj(x, alpha):
    x = np.asarray(x, dtype=float)
    n = x.size
    if n == 0:
        return 1e6
    h = np.clip(x, 0.0, 1.0)
    scores = score_vector(h, n)
    m = float(np.max(scores))
    if alpha > 1e-14:
        w = np.exp((scores - m) / alpha)
        A = float(np.sum(w))
        if math.isfinite(A) and A > 0:
            m = m + alpha * math.log(A)
    return m * (2.0 / n)

def slsqp_grad(x, alpha):
    x = np.asarray(x, dtype=float)
    n = x.size
    if n == 0:
        return np.zeros(0)
    h = np.clip(x, 0.0, 1.0)
    scores = score_vector(h, n)
    m = float(np.max(scores))
    grad = np.zeros(n)
    if alpha <= 1e-14:
        tol = max(1e-8, 1e-7 * m)
        active = np.flatnonzero(scores >= m - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(scores))])
        coef = 1.0 / len(active)
    else:
        w = np.exp((scores - m) / alpha)
        A = float(np.sum(w))
        if not math.isfinite(A) or A <= 0:
            active = np.array([int(np.argmax(scores))])
            coef = 1.0
            A = 1.0
    for pi in (active if alpha <= 1e-14 else np.flatnonzero(w / A > 1e-10)):
        pi = int(pi)
        c = (w[pi] / A) if alpha > 1e-14 else 1.0 / len(active)
        if pi == 0:
            grad += c * (1.0 - 2.0 * h)
        elif pi % 2 == 1:
            d = (pi + 1) // 2
            m2 = n - d
            grad[:m2] += c * (1.0 - h[d:])
            grad[d:] += c * (-h[:m2])
        else:
            d = pi // 2
            m2 = n - d
            grad[:m2] += c * (-h[d:])
            grad[d:] += c * (1.0 - h[:m2])
    grad *= 2.0 / n
    return grad

def constraint_eq(x, *args):
    n = max(1, int(x.size))
    return (float(np.sum(x)) - 0.5 * float(n)) / float(n)

def constraint_jac(x, *args):
    n = max(1, int(x.size))
    return np.full((1, n), 1.0 / float(n))

def run_slsqp(h, n, deadline, time_limit):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": constraint_eq, "jac": constraint_jac}]
    best_h = h.copy()
    best_c = compute_c5(h)
    
    alpha_vals = [max(0.1, 0.003 * n), max(0.03, 0.001 * n), 0.0]
    total_time = min(time_limit, float(deadline - time.time())) if deadline else time_limit
    for idx, alpha in enumerate(alpha_vals):
        if deadline is not None and time.time() > deadline - 10:
            break
        rem = (deadline - time.time()) if deadline else total_time
        frac = 0.4 if idx < 2 else 0.6
        tl = min(rem / (len(alpha_vals) - idx), max(10.0, total_time * frac / 3))
        maxit = max(30, int(200.0 * (70.0 / n)))
        try:
            res = minimize(slsqp_obj, best_h, args=(alpha,), jac=slsqp_grad,
                           method="SLSQP", bounds=bounds, constraints=constraints,
                           options={"maxiter": maxit, "ftol": 1e-10, "disp": 0})
            if res is not None and hasattr(res, 'x') and res.x is not None:
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new)
                if c_new < best_c - 1e-12:
                    best_c = c_new
                    best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def binary_scores(h, n):
    n = int(n)
    scores = np.zeros(2 * n - 1)
    hf = h.astype(float)
    for d in range(1, n):
        scores[2 * d - 1] = float(np.dot(hf[:n - d], 1.0 - hf[d:]))
        scores[2 * d] = float(np.dot(hf[d:], 1.0 - hf[:n - d]))
    return scores

def binary_flip(scores, h, idx, new_val, n):
    i = int(idx)
    nv = int(new_val)
    for d in range(1, n):
        p1 = 2 * d - 1
        p2 = 2 * d
        if i - d >= 0:
            hj = int(h[i - d])
            if nv == 1:
                scores[p1] -= hj
                scores[p2] += 1.0 - hj
            else:
                scores[p1] += hj
                scores[p2] -= 1.0 - hj
        if i + d < n:
            hj = int(h[i + d])
            if nv == 1:
                scores[p1] += 1.0 - hj
                scores[p2] -= hj
            else:
                scores[p1] -= 1.0 - hj
                scores[p2] += hj
    h[i] = nv

def binary_swap(scores, h, i, j, n):
    hi = int(h[i])
    hj = int(h[j])
    if hi == hj:
        return False
    if hi == 0:
        binary_flip(scores, h, i, 1, n)
        binary_flip(scores, h, j, 0, n)
    else:
        binary_flip(scores, h, i, 0, n)
        binary_flip(scores, h, j, 1, n)
    return True

def binary_search(h_start, n, deadline, max_restarts=3, max_ops=50000):
    n = int(n)
    if n % 2 != 0:
        return None, 1e9
    rng = np.random.default_rng(12345)
    target = n // 2
    best_h = None
    best_c = 1e9
    stub_time = max(1.0, (deadline - time.time()) / max_restarts)
    
    for r in range(max_restarts):
        if deadline is not None and time.time() > deadline:
            break
        if r == 0 and h_start is not None:
            h = h_start.astype(int).copy()
        else:
            h = np.zeros(n, dtype=int)
            idx = rng.choice(n, size=target, replace=False)
            h[idx] = 1
        scores = binary_scores(h, n)
        cur = float(np.max(scores))
        local_best = cur
        local_best_h = h.copy()
        stall = 0
        ops = 0
        rd = time.time() + stub_time
        if deadline is not None:
            rd = min(rd, deadline)
        
        while ops < max_ops and time.time() < rd:
            ops += 1
            # Propose swap: target high-score lags
            active = np.flatnonzero(scores >= cur - 1e-9)
            if active.size > 0 and rng.random() < 0.7:
                pi = int(active[rng.integers(active.size)])
                if pi > 0:
                    if pi % 2 == 1:
                        d = (pi + 1) // 2
                        mask = (h[:n - d] == 1) & (h[d:] == 0)
                    else:
                        d = pi // 2
                        mask = (h[:n - d] == 0) & (h[d:] == 1)
                    cand = np.nonzero(mask)[0]
                    if cand.size > 0:
                        pos = int(cand[rng.integers(cand.size)])
                        i, j = pos, pos + d
                    else:
                        i = int(rng.integers(n))
                        if h[i] == 0:
                            j = int(np.nonzero(h == 1)[0][rng.integers(np.count_nonzero(h == 1))])
                        else:
                            j = int(np.nonzero(h == 0)[0][rng.integers(np.count_nonzero(h == 0))])
            else:
                zeros = np.nonzero(h == 0)[0]
                ones = np.nonzero(h == 1)[0]
                if zeros.size == 0 or ones.size == 0:
                    break
                i = int(zeros[rng.integers(zeros.size)])
                j = int(ones[rng.integers(ones.size)])
            
            if not binary_swap(scores, h, i, j, n):
                continue
            new_max = float(np.max(scores))
            if new_max + 1e-9 < cur:
                cur = new_max
                stall = 0
                if cur < local_best:
                    local_best = cur
                    local_best_h = h.copy()
            else:
                binary_swap(scores, h, i, j, n)
                stall += 1
            if stall > 5000:

        c = local_best * (2.0 / n)
        if c < best_c:
            best_c = c
            best_h = local_best_h.copy().astype(float)
    
    return best_h, best_c

    # Phase 2: Multi-start coordinate descent at n0

            dl = min(deadline - 30, time.time() + 20.0)

            update_best(h_ps, n0)

    resolutions = []
    for mult in [2, 3, 4]:
        nn = int(round(mult * n0 / 2.0)) * 2
        if nn >= 4 and nn <= 6 * n0 and nn not in resolutions:
            resolutions.append(nn)
    resolutions.sort()

        if time.time() > deadline - 80:

        h_up = project_feasible(h_up, nn)
        c_up = compute_c5(h_up)

            update_best(h_cd2, nn)

            if rem2 > 20:
                h_sl, c_sl = run_slsqp(best_h if best_n == nn else upsample_linear(best_h, nn), nn, deadline, min(rem2, 80.0))
                update_best(h_sl, nn)
    
    # Phase 4: SLSQP polish at best resolution
    if time.time() < deadline - 20:
        rem = deadline - time.time() - 5
        if rem > 20:
            h_sl, c_sl = run_slsqp(best_h, best_n, deadline, rem)

    # Final validation
    h_final = project_feasible(best_h, best_n)

        h_final = project_feasible(h0, n0)
        c_final = compute_c5(h_final)
        best_n = n0
    
    return h_final, float(c_final), int(best_n)
