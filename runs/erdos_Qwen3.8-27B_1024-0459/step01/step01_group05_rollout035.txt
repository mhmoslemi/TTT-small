Actually, looking at the problem again: C₅ = max_k ∫ h(x)(1-h(x+k)) dx. The integrand h(x)(1-h(x+k)) is always ≥ 0 since h ∈ [0,1]. So all correlations are non-negative. No absolute value needed.

Subgradient of f at h (for the k* that achieves max):
Let g_k(h) = sum_i h[i] * (1 - h[i+k]) (with validity conditions)
∂g_k/∂h[j] = (1 - h[j+k]) - h[j-k] (summing over valid terms)

So the subgradient of f = max_k g_k * dx is:
∂f/∂h[j] = dx * [(1 - h[j+k*]) - h[j-k*]] (for the active k*)

If multiple k are near the max, we can average their gradients (this is the "smoothed subgradient" approach).

The projected subgradient method:
h_{t+1} = proj_S(h_t - alpha_t * grad_t)

where S = {h : sum(h) = n/2, 0 ≤ h ≤ 1} and proj_S is the projection onto this set.

The projection onto S can be done by:
1. Compute h - alpha * grad
2. Water-fill to satisfy sum = n/2 while keeping in [0,1]

Let me implement this. The water-filling / projection:
- Start with x = h - alpha * grad
- Find lambda such that sum(clip(x + lambda, 0, 1)) = n/2
- Binary search on lambda (monotone in lambda)

This is efficient: O(n log(precision)) per iteration.

from scipy.optimize import linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def project_sum(h, target):
    """Project h onto {sum(h)=target, 0<=h<=1}."""
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    # Binary search on shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2, 0, 1)
    # Fix any residual
    s = np.sum(h)
    if abs(s - target) > 1e-12:
        diff = target - s
        free = (h > 1e-15) & (h < 1 - 1e-15)
        if np.any(free):
            h[free] += diff / np.sum(free)
        else:
            h[np.argmin(np.abs(h - 0.5))] += diff
        h = np.clip(h, 0, 1)
    return h

def subgradient_corr(h, k_lag):
    """Compute subgradient of corr[k_lag] w.r.t. h.
    k_lag is the index in the 'full' correlation output (0 to 2n-2)."""
    n = len(h)
    offset = k_lag - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        # h[i] appears as first factor with (1-h[i+offset])
        if 0 <= i + offset < n:
            grad[i] += (1 - h[i + offset])
        # h[i] appears as second factor: h[i-offset]*(1-h[i]), derivative wrt h[i] is -h[i-offset]
        if 0 <= i - offset < n:
            grad[i] -= h[i - offset]
    return grad

def subgrad_optimize(h, target, alpha0=0.01, decay=0.99, n_iter=5000, top_k=5):
    """Subgradient method with projection."""
    n = len(h)
    h = project_sum(h, target)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha = alpha0
    
    for it in range(n_iter):
        # Compute correlations
        comp = 1.0 - h
        corr = np.correlate(h, comp, mode='full')
        
        # Find top-k lags
        top_indices = np.argsort(corr)[-top_k:]
        max_val = corr[top_indices[0]]
        if max_val * dx < best_c5 - 1e-15:
            pass  # shouldn't happen
        
        # Weighted subgradient (weight by proximity to max)
        weights = np.exp(-5.0 * (max_val - corr[top_indices]))
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            grad += w * subgradient_corr(h, k)
        
        # Update
        h = h - alpha * grad * dx  # dx factor since we're maximizing corr*dx
        h = project_sum(h, target)
        
        # Check improvement
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
        
        alpha *= decay
    
    return best_h, best_c5

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def project_sum(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    s = np.sum(result)
    if abs(s - target) > 1e-13:
        diff = target - s
        free = (result > 1e-15) & (result < 1.0 - 1e-15)
        if np.any(free):
            result[free] += diff / np.sum(free)
        else:
            result[np.argmin(np.abs(result - 0.5))] += diff
        result = np.clip(result, 0.0, 1.0)
    return result

def smooth_max_obj(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -50.0, 50.0)
    return m + np.log(np.sum(np.exp(s))) / beta

def smooth_max_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -50.0, 50.0)
    exp_s = np.exp(s)
    w = exp_s / exp_s.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        offset = k - (n - 1)
        # d(corr[k])/d(h[i]) includes (1-h[i+offset]) if valid
        i_lo = max(0, -offset)
        i_hi = min(n, n - offset)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo + offset:i_hi + offset])
        # includes -h[i-offset] if valid
        i_lo2 = max(0, offset)
        i_hi2 = min(n, n + offset)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= wk * h[i_lo2 - offset:i_hi2 - offset]
    return grad

def obj_and_grad(h, beta):
    obj = smooth_max_obj(h, beta)
    grad = smooth_max_grad(h, beta)
    return obj, grad

def constraint_fun(h, target):
    return np.sum(h) - target

def constraint_jac(h, target):
    return np.ones(len(h))

def run_slsqp(h0, n, target, beta, maxiter=300, ftol=1e-12):
    h0 = project_sum(h0, target)
    res = minimize(
        lambda x: obj_and_grad(x, beta)[0],
        h0,
        jac=lambda x: obj_and_grad(x, beta)[1],
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': lambda x: constraint_fun(x, target),
                      'jac': lambda x: constraint_jac(x, target)}],
        options={'maxiter': maxiter, 'ftol': ftol}
    )
    if res.success or res.x is not None:
        return project_sum(res.x, target)
    return h0

def subgrad_refine(h, target, n_iter=3000, alpha0=0.05, decay=0.999):
    """Subgradient method to minimize max correlation."""
    n = len(h)
    h = project_sum(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha0
    
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = np.argmax(corr)
        # Use top-3 lags for smoothing
        top3 = np.argsort(corr)[-3:]
        weights = np.exp(-3.0 * (corr[k_max] - corr[top3]))

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    s = np.sum(result)
    if abs(s - target) > 1e-13:
        diff = target - s
        free = (result > 1e-14) & (result < 1.0 - 1e-14)
        if np.any(free):
            result[free] += diff / float(np.sum(free))
        else:
            idx = int(np.argmin(np.abs(result - 0.5)))
            result[idx] += diff
        result = np.clip(result, 0.0, 1.0)
    return result


def smooth_max_val(x, beta):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -50.0, 50.0)
    return float(m + np.log(np.sum(np.exp(s))) / beta)


def smooth_max_grad(x, beta):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -50.0, 50.0)
    exp_s = np.exp(s)
    w = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        offset = k - (n - 1)
        i_lo = max(0, -offset)
        i_hi = min(n, n - offset)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += wk * (1.0 - x[i_lo + offset:i_hi + offset])
        i_lo2 = max(0, offset)
        i_hi2 = min(n, n + offset)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] -= wk * x[i_lo2 - offset:i_hi2 - offset]
    return grad


def _obj_slsqp(x, beta):
    return smooth_max_val(x, beta)


def _grad_slsqp(x, beta):
    return smooth_max_grad(x, beta)


def _con_fun(x, target):
    return float(np.sum(x) - target)


def _con_jac(x, target):
    return np.ones(len(x))


def run_slsqp(h0, n, target, beta, maxiter, ftol):
    h0 = project_sum(h0, target)
    con = {'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (target,)}
    try:
        res = minimize(
            _obj_slsqp, h0, args=(beta,),
            jac=_grad_slsqp,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[con],
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        if res.x is not None:
            return project_sum(res.x, target)
    except Exception:
        pass
    return h0


def subgrad_refine(h, target, n_iter, alpha0, decay, top_k):
    n = len(h)
    h = project_sum(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha0
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = int(np.argmax(corr))
        top_idx = np.argsort(corr)[-top_k:]
        weights = np.exp(-4.0 * (corr[k_max] - corr[top_idx]))
        weights = weights / np.sum(weights)
        grad = np.zeros(n)
        for jj in range(len(top_idx)):
            wk = weights[jj]
            k = int(top_idx[jj])
            offset = k - (n - 1)
            i_lo = max(0, -offset)
            i_hi = min(n, n - offset)
            if i_lo < i_hi:
                grad[i_lo:i_hi] += wk * (1.0 - h[i_lo + offset:i_hi + offset])
            i_lo2 = max(0, offset)
            i_hi2 = min(n, n + offset)
            if i_lo2 < i_hi2:
                grad[i_lo2:i_hi2] -= wk * h[i_lo2 - offset:i_hi2 - offset]
        h = h - alpha * grad
        h = project_sum(h, target)
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
        alpha *= decay
    return best_h, best_c5


def pair_swap_search(h, target, n_trials, rng):
    n = len(h)
    h = project_sum(h, target)
    best_c5 = compute_c5(h)
    delta_base = 0.005
    for _ in range(n_trials):
        i, j = int(rng.integers(n)), int(rng.integers(n))
        if i == j:
            continue
        delta = delta_base * (1.0 if rng.random() < 0.5 else -1.0)
        h2 = h.copy()
        h2[i] += delta
        h2[j] -= delta
        h2 = np.clip(h2, 0.0, 1.0)
        h2 = project_sum(h2, target)
        c5_new = compute_c5(h2)
        if c5_new < best_c5:
            h = h2
            best_c5 = c5_new
    return h, best_c5


def make_sinusoidal_init(n, freq, amp, phase):
    x = np.arange(n) / n
    vals = 0.5 + amp * np.sin(2.0 * np.pi * freq * x + phase)
    return project_sum(np.clip(vals, 0.0, 1.0), n / 2.0)


def make_block_init(n, n_blocks, rng):
    blocks = np.zeros(n)
    for b in range(n_blocks):
        start = int(b * n / n_blocks)
        end = int((b + 1) * n / n_blocks)
        level = rng.uniform(0.1, 0.9)
        blocks[start:end] = level
    return project_sum(blocks, n / 2.0)


def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_sum(h_new, n_new / 2.0)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    def elapsed():
        return time.time() - t0

    def remaining_frac():
        return max(0.0, 1.0 - elapsed() / budget_s)

    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)
    c5_parent = compute_c5(h_parent)

    best_h = h_parent.copy()
    best_c5 = c5_parent
    best_n = n_parent

    # Phase 1: Progressive optimization at parent resolution
    n = n_parent
    target = n / 2.0
    h_cur = best_h.copy()

    if remaining_frac() > 0.15:
        h_cur = run_slsqp(h_cur, n, target, beta=5.0, maxiter=300, ftol=1e-12)
        c5_t = compute_c5(h_cur)
        if c5_t < best_c5:
            best_h, best_c5, best_n = h_cur.copy(), c5_t, n

    if remaining_frac() > 0.15:
        h_cur = run_slsqp(best_h, best_n, best_n / 2.0, beta=15.0, maxiter=400, ftol=1e-12)
        c5_t = compute_c5(h_cur)
        if c5_t < best_c5:
            best_h, best_c5, best_n = h_cur.copy(), c5_t, best_n

    if remaining_frac() > 0.15:
        h_cur = run_slsqp(best_h, best_n, best_n / 2.0, beta=50.0, maxiter=400, ftol=1e-13)
        c5_t = compute_c5(h_cur)
        if c5_t < best_c5:
            best_h, best_c5, best_n = h_cur.copy(), c5_t, best_n

    if remaining_frac() > 0.15:
        h_cur = run_slsqp(best_h, best_n, best_n / 2.0, beta=200.0, maxiter=300, ftol=1e-13)
        c5_t = compute_c5(h_cur)
        if c5_t < best_c5:
            best_h, best_c5, best_n = h_cur.copy(), c5_t, best_n

    # Phase 2: Subgradient refinement at parent resolution
    if remaining_frac() > 0.25:
        h_sg, c5_sg = subgrad_refine(best_h, best_n / 2.0, n_iter=3000, alpha0=0.04, decay=0.9995, top_k=3)
        if c5_sg < best_c5:
            best_h, best_c5, best_n = h_sg, c5_sg, best_n

    # Phase 3: Higher resolutions
    resolutions = [127, 150, 175, 200]
    for n_try in resolutions:
        if elapsed() > budget_s * 0.65:
            break
        h_interp = upsample(best_h, n_try)
        target_try = n_try / 2.0
        h_cur2 = h_interp

        if remaining_frac() > 0.1:
            h_cur2 = run_slsqp(h_cur2, n_try, target_try, beta=10.0, maxiter=250, ftol=1e-12)
            c5_t = compute_c5(h_cur2)
            if c5_t < best_c5:
                best_h, best_c5, best_n = h_cur2.copy(), c5_t, n_try

        if remaining_frac() > 0.1:
            h_cur2 = run_slsqp(best_h, best_n, best_n / 2.0, beta=50.0, maxiter=250, ftol=1e-12)
            c5_t = compute_c5(h_cur2)
            if c5_t < best_c5:
                best_h, best_c5, best_n = h_cur2.copy(), c5_t, best_n

        if remaining_frac() > 0.1:
            h_cur2 = run_slsqp(best_h, best_n, best_n / 2.0, beta=150.0, maxiter=200, ftol=1e-13)
            c5_t = compute_c5(h_cur2)
            if c5_t < best_c5:
                best_h, best_c5, best_n = h_cur2.copy(), c5_t, best_n

        if remaining_frac() > 0.1:
            h_sg2, c5_sg2 = subgrad_refine(best_h, best_n / 2.0, n_iter=1500, alpha0=0.02, decay=0.9997, top_k=3)
            if c5_sg2 < best_c5:
                best_h, best_c5, best_n = h_sg2, c5_sg2, best_n

    # Phase 4: Structured initializations at best resolution
    n_best = best_n
    tgt_best = n_best / 2.0
    structured_inits = [
        make_sinusoidal_init(n_best, 2, 0.3, 0.0),
        make_sinusoidal_init(n_best, 3, 0.25, 1.0),
        make_sinusoidal_init(n_best, 5, 0.2, 0.5),
        make_block_init(n_best, 6, rng),
        make_block_init(n_best, 10, rng),
    ]
    for h_init in structured_inits:
        if elapsed() > budget_s * 0.82:
            break
        h_s = run_slsqp(h_init, n_best, tgt_best, beta=20.0, maxiter=200, ftol=1e-12)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_h, best_c5, best_n = h_s, c5_s, n_best

    # Phase 5: Perturbation-based multi-start at best resolution
    n_best = best_n
    tgt_best = n_best / 2.0
    for attempt in range(8):
        if elapsed() > budget_s * 0.92:
            break
        noise_scale = 0.02 * (1.0 - attempt * 0.08)
        h_pert = best_h + rng.randn(n_best) * noise_scale
        h_pert = project_sum(h_pert, tgt_best)
        h_new = run_slsqp(h_pert, n_best, tgt_best, beta=40.0, maxiter=200, ftol=1e-12)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5, best_n = h_new, c5_new, n_best

    # Phase 6: Pair-swap local search
    if elapsed() > budget_s * 0.90:
        pass
    else:
        n_best = best_n
        tgt_best = n_best / 2.0
        n_trials = max(500, int((budget_s - elapsed()) * 80))
        h_ps, c5_ps = pair_swap_search(best_h, tgt_best, n_trials, rng)
        if c5_ps < best_c5:
            best_h, best_c5, best_n = h_ps, c5_ps, n_best

    # Phase 7: Final subgradient polish
    n_best = best_n
    tgt_best = n_best / 2.0
    if remaining_frac() > 0.02:
        n_iter_final = min(5000, int((budget_s - elapsed()) * 50))
        h_fg, c5_fg = subgrad_refine(best_h, tgt_best, n_iter_final, alpha0=0.01, decay=0.9999, top_k=3)
        if c5_fg < best_c5:
            best_h, best_c5, best_n = h_fg, c5_fg, n_best

    # Final: ensure exact constraint
    n_final = best_n
    tgt_final = n_final / 2.0
    best_h = project_sum(best_h, tgt_final)
    s_check = np.sum(best_h)
    if abs(s_check - tgt_final) > 1e-10:
        best_h[n_final // 2] += (tgt_final - s_check)
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    return (best_h, final_c5, n_final)
