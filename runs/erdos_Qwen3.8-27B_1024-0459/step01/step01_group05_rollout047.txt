def coordinate_descent_step(h, i):
    """Optimize coordinate i of h, keeping others fixed."""
    n = len(h)
    old_val = h[i]
    
    # Compute correlations with h[i] = 0 and h[i] = 1
    h0 = h.copy()
    h0[i] = 0.0
    h1 = h.copy()
    h1[i] = 1.0
    
    comp0 = 1.0 - h0
    comp1 = 1.0 - h1
    
    corr0 = np.correlate(h0, comp0, mode='full')
    corr1 = np.correlate(h1, comp1, mode='full')
    
    # c_k(t) = corr0[k] + t * (corr1[k] - corr0[k])
    a = corr0  # c_k(0)
    b = corr1 - corr0  # c_k(1) - c_k(0)
    
    # Find t in [0,1] minimizing max_k (a[k] + b[k]*t)
    # The minimum of a convex PL function in 1D
    # We can binary search on T: is there t in [0,1] with a[k]+b[k]*t <= T for all k?
    
    lo, hi = 0.0, 1.0
    # Binary search on T
    T_lo = np.max(a)  # t=0 gives max_k a[k]
    T_hi = np.max(a + b)  # t=1 gives max_k (a[k]+b[k])
    T_best = min(T_lo, T_hi)
    t_best = 0.0 if T_lo < T_hi else 1.0
    if T_lo == T_hi:
        t_best = 0.5  # doesn't matter
    
    # Actually, let's just do a fine search
    # The function is convex, so we can use ternary search or check breakpoints
    
    # Better: the minimum occurs at a breakpoint or at 0 or 1
    # A breakpoint of max_k(a_k + b_k*t) is where two lines that are both at the max intersect
    # Equivalently, it's where the derivative changes sign
    
    # Simplest: evaluate on a grid and refine
    t_grid = np.linspace(0, 1, 1001)
    for t in t_grid:
        val = np.max(a + t * b)
        if val < T_best - 1e-15:
            T_best = val
            t_best = t
    
    # Refine around t_best
    if 0 < t_best < 1:
        lo_t = max(0, t_best - 0.002)
        hi_t = min(1, t_best + 0.002)
        t_fine = np.linspace(lo_t, hi_t, 1001)
        for t in t_fine:
            val = np.max(a + t * b)
            if val < T_best - 1e-15:
                T_best = val
                t_best = t
    
    h[i] = t_best
    return h, T_best

            h_cur = run_slsqp(h_cur, n, target, beta, 200, 1e-14)
        try_update(h_cur, n)

    # 1b: Perturbation + reoptimize
    if elapsed() < budget_s * 0.30:
        for attempt in range(3):
            if elapsed() > budget_s * 0.32:
                break
            scale = 0.02 * (1.0 - attempt * 0.3)
            h_pert = best_h + np.random.randn(n) * scale
            h_pert = project_sum(h_pert, target)
            h_cur = run_slsqp(h_pert, n, target, 20.0, 200, 1e-14)
            try_update(h_cur, n)

    # Phase 2: Upsample to higher resolutions
    resolutions = [190, 285, 380]
    for n_try in resolutions:
        if elapsed() > budget_s * 0.75:
            break
        tgt_try = n_try / 2.0

        # Initialize by upsampling current best
        h_init = upsample(best_h, best_n, n_try)
        h_init = project_sum(h_init, tgt_try)

        # Optimize at this resolution
        for attempt in range(2):
            if elapsed() > budget_s * 0.78:
                break
            if attempt == 0:
                h_start = h_init
            else:
                h_start = h_init + np.random.randn(n_try) * 0.015
                h_start = project_sum(h_start, tgt_try)

            h_cur = h_start
            for beta in [5.0, 25.0]:
                if elapsed() > budget_s * 0.80:
                    break
                max_it = 250 if n_try <= 200 else 150
                h_cur = run_slsqp(h_cur, n_try, tgt_try, beta, max_it, 1e-13)
            try_update(h_cur, n_try)

            # Sharp polish
            if elapsed() < budget_s * 0.82:
                h_cur = run_slsqp(h_cur, n_try, tgt_try, 80.0, 150, 1e-14)
                try_update(h_cur, n_try)

    # Phase 3: Coordinate descent refinement on best solution
    if elapsed() < budget_s * 0.90:
        n_b = best_n
        tgt_b = n_b / 2.0
        h_cd = best_h.copy()
        n = n_b

        for sweep in range(3):
            if elapsed() > budget_s * 0.92:
                break
            for i in range(n - 1, -1, -1):
                old_val = h_cd[i]
                # Evaluate at t=0 and t=1 to get linear interpolation
                h_cd[i] = 0.0
                c0 = corr_full(h_cd)
                h_cd[i] = 1.0
                c1 = corr_full(h_cd)
                a = c0  # coefficients at t=0
                b = c1 - c0  # slope in t

                # Find t in [0,1] minimizing max_k(a[k] + b[k]*t)
                # Convex PL function: binary search on threshold
                # Feasible t for threshold T:
                #   t <= (T - a[k])/b[k] for b[k] > 0
                #   t >= (T - a[k])/b[k] for b[k] < 0
                #   a[k] <= T for b[k] == 0

                lo_T = max(np.max(a), np.max(a + b)) - 0.01
                hi_T = max(np.max(a), np.max(a + b)) + 0.01
                best_t = old_val
                best_T = np.max(a + old_val * b)

                # Fine grid search (fast for 1D convex)
                t_grid = np.linspace(0.0, 1.0, 501)
                vals = np.array([np.max(a + t * b) for t in t_grid])
                idx_best = np.argmin(vals)
                t_lo = t_grid[max(0, idx_best - 1)]
                t_hi = t_grid[min(len(t_grid) - 1, idx_best + 1)]
                t_fine = np.linspace(t_lo, t_hi, 201)
                vals_fine = np.array([np.max(a + t * b) for t in t_fine])
                idx_fine = np.argmin(vals_fine)
                best_t = t_fine[idx_fine]
                best_T = vals_fine[idx_fine]

                h_cd[i] = best_t

            # Project back to satisfy sum constraint
            h_cd = project_sum(h_cd, tgt_b)

        try_update(h_cd, n_b)

    # Phase 4: Final sharp SLSQP polish
    if elapsed() < budget_s * 0.97:
        n_b = best_n
        tgt_b = n_b / 2.0
        for beta in [100.0, 300.0]:
            if elapsed() > budget_s * 0.97:
                break
            h_cur = run_slsqp(best_h, n_b, tgt_b, beta, 200, 1e-15)
            try_update(h_cur, n_b)

    # Final: ensure exact constraint and compute
    n_final = best_n
    tgt_final = n_final / 2.0
    best_h = project_sum(best_h, tgt_final)

    # Verify and fix sum exactly
    s = np.sum(best_h)
    if abs(s - tgt_final) > 1e-10:
        if tgt_final - s > 0:
            idx = np.argmin(best_h)
        else:
            idx = np.argmax(best_h)
        best_h[idx] += (tgt_final - s)
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, best_n)
