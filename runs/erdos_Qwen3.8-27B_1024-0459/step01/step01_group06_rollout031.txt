        else:
            m2 = -lag
            m_len = n - m2
            if m_len > 0:
                grad[:m_len] += wk * (1.0 - h[m2:m2+m_len])
                grad[m2:m2+m_len] -= wk * h[:m_len]
    return grad * dx


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-10
        else:
            mask = h < 1.0 - 1e-10
        if mask.sum() == 0:
            break
        h[mask] -= diff / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = h.sum() - target
    if abs(diff) > 1e-15:
        if diff > 0:
            mask = h > 0
        else:
            mask = h < 1.0
        if mask.sum() > 0:
            h[mask] -= diff / mask.sum()
    return h


def upsample(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

    """FISTA optimization of soft-max C5."""
    h_t = h.copy()
    h_prev = h.copy()
    t_ = 1.0
    best_h = h_t.copy()
    best_c5 = compute_c5(h_t, n)
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        
        # Gradient
        g = soft_max_c5_grad(h_t, n, beta)
        
        # Step
        h_new = h_t - lr * g
        
        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_ * t_)) / 2.0

        # Project

        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h_t.copy()

  - If j is in {0, 1, ..., n-1-lag} (i.e., j = i for some valid i): contribution is -h[j+lag]
  - If j is in both sets: contribution is (1-h[j-lag]) - h[j+lag]

  - If j is in {0, 1, ..., n-1-m}: contribution is (1-h[j+m])
  - If j is in {m, m+1, ..., n-1}: contribution is -h[j-m]
  - If j is in both: (1-h[j+m]) - h[j-m]

import numpy as np
from scipy.optimize import minimize
import time


def compute_correlation(h, n):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h, n):
    c = compute_correlation(h, n)
    return float(np.max(c) * (2.0 / n))


def soft_max_c5(h, n, beta):
    c = compute_correlation(h, n)
    dx = 2.0 / n
    scaled = beta * c
    m = np.max(scaled)
    val = (m + np.log(np.sum(np.exp(scaled - m)))) / beta
    return float(val * dx)


def soft_max_c5_grad(h, n, beta):
    c = compute_correlation(h, n)
    dx = 2.0 / n
    scaled = beta * c
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            m_len = n - lag
            if m_len > 0:
                grad[lag:lag + m_len] += wk * (1.0 - h[:m_len])
                grad[:m_len] -= wk * h[lag:lag + m_len]
        else:
            m2 = -lag
            m_len = n - m2
            if m_len > 0:
                grad[:m_len] += wk * (1.0 - h[m2:m2 + m_len])
                grad[m2:m2 + m_len] -= wk * h[:m_len]
    return grad * dx


def hard_max_c5(h, n):
    return compute_c5(h, n)


def hard_max_c5_grad(h, n):
    c = compute_correlation(h, n)
    max_val = np.max(c)
    tolerance = max_val * 1e-4 + 1e-10
    close = c >= max_val - tolerance
    k_indices = np.where(close)[0]
    n_active = max(1, len(k_indices))
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_idx in k_indices:
        wk = 1.0 / n_active
        lag = k_idx - (n - 1)
        if lag >= 0:
            m_len = n - lag
            if m_len > 0:
                grad[lag:lag + m_len] += wk * (1.0 - h[:m_len])
                grad[:m_len] -= wk * h[lag:lag + m_len]
        else:
            m2 = -lag
            m_len = n - m2
            if m_len > 0:
                grad[:m_len] += wk * (1.0 - h[m2:m2 + m_len])
                grad[m2:m2 + m_len] -= wk * h[:m_len]
    return grad * dx


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-12
        else:
            mask = h < 1.0 - 1e-12
        if mask.sum() == 0:
            break
        h[mask] -= diff / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target
    if abs(diff) > 1e-15:
        if diff > 0:
            mask = h > 0
        else:
            mask = h < 1.0
        if mask.sum() > 0:
            h[mask] -= diff / mask.sum()
    return h


def upsample(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)


def fista_optimize(h_init, n, beta, max_iter, t0, time_limit):
    h_t = h_init.copy()
    h_prev = h_t.copy()
    t_val = 1.0
    alpha = 2.0 / (beta + 1.0)
    best_h = h_t.copy()
    best_c5 = compute_c5(h_t, n)
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        if it > 0:
            h_y = h_t + ((t_val - 1.0) / t_new) * (h_t - h_prev)
        else:
            h_y = h_t.copy()
        g = soft_max_c5_grad(h_y, n, beta)
        h_new = project_feasible(h_y - alpha * g, n)
        t_val = t_new
        h_prev = h_t
        h_t = h_new
        if it % 50 == 0:
            c5_now = compute_c5(h_t, n)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h_t.copy()
            alpha = max(0.1 * alpha, 2.0 / (beta + 1.0) * 0.5)
    c5_now = compute_c5(h_t, n)
    if c5_now < best_c5:
        best_c5 = c5_now
        best_h = h_t.copy()
    return best_h, best_c5


def sum_eq_fun(x, n):
    return float(np.sum(x) - n / 2.0)


def sum_eq_jac(x, n):
    return np.ones(n)


def slsqp_optimize(h_init, n, max_iter, t0, time_limit):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(
            fun=hard_max_c5,
            x0=h_init,
            jac=hard_max_c5_grad,
            args=(n,),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0}
        )
        h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n)
        c5_new = compute_c5(h_new, n)
        return h_new, c5_new
    except Exception:
        return h_init, compute_c5(h_init, n)
