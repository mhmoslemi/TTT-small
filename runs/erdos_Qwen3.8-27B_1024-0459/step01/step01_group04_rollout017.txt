    corr = np.correlate(h, h, mode="full")

    prefix = np.empty(n+1, dtype=float)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])
    total = prefix[n]
    scores = np.empty(2*n-1, dtype=float)
    scores[0] = total - ac[0]
    for d in range(1, n):
        scores[2*d-1] = prefix[n-d] - ac[d]
        scores[2*d] = total - prefix[d] - ac[d]
    return scores

        scores[2*d-1] = prefix[n-d] - ac[d]
        scores[2*d] = total - prefix[d] - ac[d]
    return scores

    e = np.exp(s)
    A = e.sum()
    if not math.isfinite(A) or A <= 0:
        return m * (2.0 / n)
    return (m + alpha * math.log(A)) * (2.0 / n)

        h_final = as_feasible(state.h, state.n)
        c_final = compute_c5(h_final)
        n_final = int(state.n)
    else:
        h_final = h0
        c_final = c0
        n_final = n0

    if not is_valid(h_final):
        h_final = as_feasible(h_final, n_final)
        c_final = compute_c5(h_final)

    return h_final, float(c_final), int(n_final)
