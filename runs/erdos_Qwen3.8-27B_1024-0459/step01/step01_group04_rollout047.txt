    active = np.flatnonzero(mask)
    return active, w[mask]

def score_grad_one(idx, h, n, out):
    if idx == 0:
        out += (1.0 - 2.0 * h)
        return
    if idx % 2 == 1:
        d = (idx + 1) // 2
        m = n - d
        if m <= 0:
            return
        out[:m] += (1.0 - h[d:])
        out[d:] += (-h[:m])
    else:
        d = idx // 2
        m = n - d
        if m <= 0:
            return
        out[:m] += (-h[d:])
        out[d:] += (1.0 - h[:m])

def obj_grad(x, alpha):
    check_deadline()
    n = x.size
    h = np.clip(x, 0.0, 1.0)
    scores = compute_scores(h, n)
    val = smooth_max_val(scores, alpha) * (2.0 / n)
    active, weights = smooth_max_weights(scores, alpha)
    grad = np.zeros(n, dtype=float)
    for idx, w_i in zip(active, weights):
        if w_i < 1e-14:
            continue
        score_grad_one(idx, h, n, grad)
    grad *= (2.0 / n)
    return val, grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * float(n)
    s = float(np.sum(h))
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s_mid = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s_mid < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 0.999999
            if np.any(mask):
                cnt = float(np.count_nonzero(mask))
                h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 0.000001
            if np.any(mask):
                cnt = float(np.count_nonzero(mask))
                h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)

def upsample_linear(parent, n):
    n0 = int(parent.size)
    if n == n0:
        return parent.copy()
    if n0 <= 0:
        return np.full(n, 0.5)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x, x0, parent).astype(float)

def lbfgsb_smooth_optimize(h_init, n, time_limit, alphas=None):
    h = project_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h)
    start_t = time.time()
    if alphas is None:
        alphas = [1.5, 0.8, 0.4, 0.2, 0.1, 0.05, 0.02, 0.01, 0.0]
    bounds = [(0.0, 1.0)] * n
    n_alphas = len(alphas)
    time_per_alpha = max(2.0, time_limit / n_alphas)
    
    for ai, alpha in enumerate(alphas):
        elapsed = time.time() - start_t
        remaining = time_limit - elapsed
        if remaining < 3.0:
            break
        t_alloc = min(time_per_alpha, remaining - 1.0)
        if t_alloc < 1.0:
            break
        stage_deadline = time.time() + t_alloc
        old_dl = _RUN_DEADLINE
        _RUN_DEADLINE = min(stage_deadline, _RUN_DEADLINE) if _RUN_DEADLINE else stage_deadline
        
        def _obj(x):
            check_deadline()
            hh = np.clip(x, 0.0, 1.0)
            sc = compute_scores(hh, n)
            return smooth_max_val(sc, alpha) * (2.0 / n) + 1e6 * (np.sum(hh) - 0.5*n)**2
        
        def _grad(x):
            check_deadline()
            hh = np.clip(x, 0.0, 1.0)
            sc = compute_scores(hh, n)
            act, wts = smooth_max_weights(sc, alpha)
            g = np.zeros(n, dtype=float)
            for idx, w_i in zip(act, wts):
                if w_i < 1e-14:
                    continue
                score_grad_one(idx, hh, n, g)
            g *= (2.0 / n)
            g += 2e6 * (np.sum(hh) - 0.5 * n)
            return g
        
        max_it = max(100, int(50 * t_alloc))
        try:
            res = minimize(_obj, h, jac=_grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': max_it, 'gtol': 1e-8, 'disp': 0})
            if res.x is not None:
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h_new
                h = h_new
        except TimeoutError:
            pass
        except Exception:
            pass
        finally:
            _RUN_DEADLINE = old_dl
        h = project_feasible(h, n)
    
    return best_h, best_c

def coordinate_descent(h, n, time_limit):
    h = project_feasible(h.copy(), n)
    cur_c = compute_c5(h)
    start_t = time.time()
    n_sq = n * n
    best_h = h.copy()
    best_c = cur_c
    no_improve = 0
    max_no_improve = 500
    
    while time.time() - start_t < time_limit:
        if no_improve > max_no_improve:
            break
        improved = False
        scores = compute_scores(h, n)
        cur_max = float(np.max(scores))
        for i in range(n):
            if time.time() - start_t > time_limit - 1.0:
                break
            orig = h[i]
            if orig <= 1e-9 or orig >= 1.0 - 1e-9:
                continue
            # Try small perturbations
            for delta in (-0.1, 0.1, -0.05, 0.05, -0.02, 0.02):
                new_val = orig + delta
                if new_val < 0 or new_val > 1:
                    continue
                # Adjust another coordinate to maintain sum
                # Find j != i with room
                h[i] = new_val
                adj = 0.5 * n - np.sum(h)
                if abs(adj) > 1e-10:
                    # Find best j to adjust
                    if adj > 0:
                        candidates = np.argsort(1.0 - h)
                    else:
                        candidates = np.argsort(h)
                    j = -1
                    for cj in candidates:
                        if int(cj) == i:
                            continue
                        if adj > 0 and h[int(cj)] < 1.0 - 1e-9:
                            j = int(cj)
                            break
                        if adj < 0 and h[int(cj)] > 1e-9:
                            j = int(cj)
                            break
                    if j < 0:
                        h[i] = orig
                        continue
                    h[j] = np.clip(h[j] + adj, 0.0, 1.0)
                    if abs(0.5*n - np.sum(h)) > 1e-8:
                        h[i] = orig
                        continue
                
                sc = compute_scores(h, n)
                new_max = float(np.max(sc))
                if new_max < cur_max - 1e-10:
                    cur_max = new_max
                    improved = True
                    if new_max * (2.0/n) < best_c:
                        best_c = new_max * (2.0/n)
                        best_h = h.copy()
                    no_improve = 0
                    orig = h[i]
                else:
                    h[i] = orig
            orig_val = h[i]
            # Try setting to 0 or 1
            for tv in (0.0, 1.0):
                if abs(tv - orig_val) < 1e-9:
                    continue
                h[i] = tv
                adj = 0.5 * n - np.sum(h)
                if abs(adj) > 1e-10:
                    if adj > 0:
                        j = int(np.argmax(1.0 - h))
                        if j == i:
                            for j2 in np.argsort(1.0 - h):
                                if int(j2) != i:
                                    j = int(j2)
                                    break
                        if h[j] >= 1.0 - 1e-9:
                            h[i] = orig_val
                            continue
                        h[j] = min(1.0, h[j] + adj)
                    else:
                        j = int(np.argmax(h))
                        if j == i:
                            for j2 in np.argsort(h):
                                if int(j2) != i:
                                    j = int(j2)
                                    break
                        if h[j] <= 1e-9:
                            h[i] = orig_val
                            continue
                        h[j] = max(0.0, h[j] + adj)
                    if abs(0.5*n - np.sum(h)) > 1e-8:
                        h[i] = orig_val
                        continue
                
                sc = compute_scores(h, n)
                new_max = float(np.max(sc))
                if new_max < cur_max - 1e-10:
                    cur_max = new_max
                    improved = True
                    if new_max * (2.0/n) < best_c:
                        best_c = new_max * (2.0/n)
                        best_h = h.copy()
                    no_improve = 0
                else:
                    h[i] = orig_val
        if not improved:
            no_improve += 1
    
    return best_h, best_c

def binary_scores_fast(h, n):
    hf = h.astype(float)
    scores = np.zeros(2 * n - 1, dtype=float)
    for d in range(1, n):
        m = n - d
        scores[2*d-1] = float(np.dot(hf[:m], 1.0 - hf[d:]))
        scores[2*d] = float(np.dot(hf[d:], 1.0 - hf[:m]))
    return scores

def binary_flip(scores, h, idx, new_val, n):
    i = idx
    hi = h[i]
    if hi == new_val:
        return
    h[i] = new_val
    for d in range(1, n):
        pidx = 2*d - 1
        midx = 2*d
        if i - d >= 0:
            hj = h[i-d]
            if new_val == 1:
                scores[pidx] -= hj
                scores[midx] += 1.0 - hj
            else:
                scores[pidx] += hj
                scores[midx] -= 1.0 - hj
        if i + d < n:
            hj = h[i+d]
            if new_val == 1:
                scores[pidx] += 1.0 - hj
                scores[midx] -= hj
            else:
                scores[pidx] -= 1.0 - hj
                scores[midx] += hj

def binary_local_search(h_init, n, time_limit, rng, state_update=None):
    n_half = n // 2
    h = h_init.astype(np.int8).copy()
    if np.sum(h) != n_half:
        order = np.argsort(h_init)
        h[:] = 0
        h[order[n_half:]] = 1
    scores = binary_scores_fast(h, n)
    cur = float(np.max(scores))
    best = cur
    best_h = h.copy()
    dx = 2.0 / n
    if state_update:
        state_update(h.astype(float), best * dx, n)
    
    start_t = time.time()
    stall = 0
    max_stall = 50000
    
    while time.time() - start_t < time_limit - 1.0:
        if stall > max_stall:
            break
        # Choose a swap
        r = rng.random()
        if r < 0.6:
            # Target active scores
            active = np.flatnonzero(scores >= cur - 1e-9)
            if len(active) > 0:
                idx = int(active[rng.integers(len(active))])
                if idx == 0:
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                else:
                    if idx % 2 == 1:
                        d = (idx + 1) // 2
                        mask = (h[:n-d] == 1) & (h[d:] == 0)
                    else:
                        d = idx // 2
                        mask = (h[:n-d] == 0) & (h[d:] == 1)
                    cand = np.nonzero(mask)[0]
                    if len(cand) > 0:
                        pos = int(cand[rng.integers(len(cand))])
                        i = pos
                        j = pos + d
                    else:
                        zeros = np.flatnonzero(h == 0)
                        ones = np.flatnonzero(h == 1)
                        if len(zeros) == 0 or len(ones) == 0:
                            break
                        i = int(zeros[rng.integers(len(zeros))])
                        j = int(ones[rng.integers(len(ones))])
            else:
                zeros = np.flatnonzero(h == 0)
                ones = np.flatnonzero(h == 1)
                i = int(zeros[rng.integers(len(zeros))])
                j = int(ones[rng.integers(len(ones))])
        else:
            zeros = np.flatnonzero(h == 0)
            ones = np.flatnonzero(h == 1)
            if len(zeros) == 0 or len(ones) == 0:
                break
            i = int(zeros[rng.integers(len(zeros))])
            j = int(ones[rng.integers(len(ones))])
        
        if i == j:
            continue
        if h[i] == h[j]:
            continue
        
        # Apply swap
        vi = int(h[i])
        vj = int(h[j])
        binary_flip(scores, h, i, vj, n)
        binary_flip(scores, h, j, vi, n)
        
        new_max = float(np.max(scores))
        if new_max < cur - 1e-9:
            cur = new_max
            stall = 0
            if cur < best:
                best = cur
                best_h = h.copy()
                if state_update:
                    state_update(h.astype(float), best * dx, n)
        elif new_max <= cur + 1e-9:
            stall += 1
            if stall > 100 and rng.random() < 0.05:
                # Keep the swap
                pass
            else:
                # Revert
                binary_flip(scores, h, i, vi, n)
                binary_flip(scores, h, j, vj, n)
                stall += 1
        else:
            # Revert
            binary_flip(scores, h, i, vi, n)
            binary_flip(scores, h, j, vj, n)
            stall += 1
    
    if state_update:
        state_update(best_h.astype(float), best * dx, n)
    return best_h.astype(float), best * dx

def make_starts(n, parent, rng):
    starts = []
    if parent is not None:
        hp = parent.copy()
        if hp.size != n:
            hp = upsample_linear(parent, n)
        hp = project_feasible(hp, n)
        starts.append(hp)
    
    # Add synthetic starts
    x = (np.arange(n) + 0.5) * (2.0 / n)
    for f in (1, 2, 3, 4, 5):
        for ph in (0.0, math.pi/2, math.pi, 1.5*math.pi):
            h = 0.5 + 0.35 * np.cos(math.pi * f * x + ph) + 0.1 * np.cos(2*math.pi*f*x)
            h = project_feasible(h, n)
            c = compute_c5(h)
            if c < 0.42:
                starts.append(h)
    
    # Random perturbation of parent
    if parent is not None:
        hp = upsample_linear(parent, n)
        h_noise = hp + 0.1 * rng.standard_normal(n)
        h_noise = project_feasible(h_noise, n)
        starts.append(h_noise)
    
    # Uniform
    starts.append(np.full(n, 0.5))
    
    return starts

def get_parent():
    try:
        p = np.asarray(initial_h_values, dtype=float)
        if p.ndim != 1 or p.size < 2:
            return None
        return p.copy()
    except Exception:
        return None

def run(seed=42, budget_s=1300, **kwargs):
    global _RUN_DEADLINE
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0:
        budget_s = 30.0
    reserve = 30.0 if budget_s > 100 else max(5.0, 0.1*budget_s)
    deadline = t0 + max(1.0, budget_s - reserve)
    _RUN_DEADLINE = deadline
    
    rng = np.random.default_rng(seed)
    parent = get_parent()
    
    best_h = None
    best_c = 1e9
    best_n = 0
    
    def state_update(h, c, n):
        nonlocal best_h, best_c, best_n
        if h is None or c >= best_c - 1e-15:
            return
        h = np.asarray(h, dtype=float)
        if h.size != n:
            return
        h = project_feasible(h, n)
        c_check = compute_c5(h)
        if c_check < best_c - 1e-15:
            best_h = h.copy()
            best_c = c_check
            best_n = n
    
    # Determine which resolutions to try
    trial_ns = []
    if parent is not None:
        n_par = int(parent.size)
        trial_ns = [n_par]
        for mult in (1.5, 2.0, 2.5, 3.0):
            nn = int(round(mult * n_par / 2.0)) * 2
            if nn >= 10 and nn <= 6*n_par and nn not in trial_ns:
                trial_ns.append(nn)
    else:
        trial_ns = [70, 100, 140]
    
    trial_ns.sort()
    
    elapsed_budget = budget_s - reserve
    n_trials = len(trial_ns)
    
    for ni, n in enumerate(trial_ns):
        elapsed = time.time() - t0
        remaining = deadline - time.time()
        if remaining < 20.0:
            break
        
        # Allocate time for this resolution
        frac = 1.0 / (n_trials - ni)
        alloc = min(remaining - 15.0, 0.6 * frac * elapsed_budget)
        if n == trial_ns[0] and budget_s > 300:
            alloc = max(alloc, min(remaining - 15.0, 0.35 * elapsed_budget))
        if alloc < 10.0:
            break
        
        # Split alloc: 60% continuous, 40% binary
        cont_time = 0.65 * alloc
        bin_time = 0.35 * alloc
        
        # Get starts
        starts = make_starts(n, parent, rng)
        # Limit starts
        if len(starts) > 5:
            starts = starts[:5]
        
        # Continuous optimization
        for si, h_start in enumerate(starts):
            rem = time.time() - t0
            if deadline - time.time() < 5.0:
                break
            t_alloc = cont_time / len(starts)
            if si > 0:
                t_alloc = min(t_alloc, (deadline - time.time()) - 5.0)
            if t_alloc < 3.0:
                break
            
            alphas = [1.5, 0.8, 0.4, 0.2, 0.1, 0.05, 0.02, 0.01, 0.0]
            if t_alloc < 30:
                alphas = [1.0, 0.5, 0.2, 0.1, 0.05, 0.0]
            if t_alloc < 15:
                alphas = [0.5, 0.2, 0.1, 0.05, 0.0]
            
            try:
                h_opt, c_opt = lbfgsb_smooth_optimize(h_start, n, t_alloc, alphas)
                state_update(h_opt, c_opt, n)
                
                # Coordinate descent polish
                cd_time = min(15.0, max(3.0, t_alloc * 0.15))
                if deadline - time.time() > cd_time + 2.0:
                    h_cd, c_cd = coordinate_descent(h_opt, n, cd_time)
                    state_update(h_cd, c_cd, n)
            except TimeoutError:
                pass
            except Exception:
                pass
        
        # Binary local search
        if deadline - time.time() > 10.0 and n % 2 == 0:
            n_restart = 3 if n <= 100 else 2
            bin_t_alloc = bin_time / n_restart
            for ri in range(n_restart):
                if deadline - time.time() < 5.0:
                    break
                if ri == 0 and best_h is not None and best_n == n:
                    h_bin_start = best_h
                elif ri == 1 and parent is not None:
                    h_bin_start = upsample_linear(parent, n)
                else:
                    h_bin_start = np.full(n, 0.5)
                    # Add some structure
                    for _ in range(n//4):
                        idx = rng.integers(n)
                        h_bin_start[idx] = 0.0
                    for _ in range(n//4):
                        idx = rng.integers(n)
                        h_bin_start[idx] = 1.0
                
                t_bl = min(bin_t_alloc, deadline - time.time() - 3.0)
                if t_bl < 3.0:
                    break
                try:
                    h_bp, c_bp = binary_local_search(h_bin_start, n, t_bl, rng, state_update)
                except Exception:
                    pass
    
    # Final: if we have a best, do one more polish
    if best_h is not None and deadline - time.time() > 10.0:
        n_f = best_n
        t_polish = min(30.0, deadline - time.time() - 5.0)
        try:
            h_cd, c_cd = coordinate_descent(best_h, n_f, t_polish)
            state_update(h_cd, c_cd, n_f)
        except Exception:
            pass
    
    if best_h is None:
        # Fallback
        n_fallback = 70
        best_h = np.full(n_fallback, 0.5)
        best_c = compute_c5(best_h)
        best_n = n_fallback
    elif parent is not None:
        # Also check parent
        hp = parent.copy()
        c_par = compute_c5(hp)
        if c_par < best_c:
            best_h = hp
            best_c = c_par
            best_n = int(hp.size)
    
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return best_h, float(best_c), int(best_n)
