```python
def solve_1d_max(min_t, max_t, a, b):
    """Minimize max_k (a[k]*t + b[k]) for t in [min_t, max_t]."""
    # The function is piecewise linear, convex (max of linear functions)
    # Find all breakpoints and evaluate
    m = len(a)
    # Breakpoints: where a[i]*t + b[i] = a[j]*t + b[j]
    # t = (b[j] - b[i]) / (a[i] - a[j]) for a[i] != a[j]
    candidates = [min_t, max_t]
    
    for i in range(m):
        for j in range(i+1, m):
            if abs(a[i] - a[j]) > 1e-14:
                t = (b[j] - b[i]) / (a[i] - a[j])
                if min_t <= t <= max_t:
                    candidates.append(t)
    
    best_t = min_t
    best_val = max(a * min_t + b)
    for t in candidates:
        val = max(a * t + b)
        if val < best_val - 1e-15:
            best_val = val
            best_t = t
    return best_t
```

```python
def solve_1d(c, b, lo=0.0, hi=1.0):
    m = len(c)
    best_v = max(np.maximum(c * lo + b, c * hi + b))
    best_t = lo if (c*lo+b).max() <= (c*hi+b).max() else hi
    
    # All pairwise intersections
    C = c[:, None] - c[None, :]  # m x m
    B = b[None, :] - b[:, None]  # m x m
    with np.errstate(divide='ignore', invalid='ignore'):
        T = B / C  # m x m, t = (b_j - b_i) / (c_i - c_j)
    
    # Mask valid intersections in [lo, hi]
    mask = (abs(C) > 1e-14) & (T >= lo) & (T <= hi) & (~np.eye(m, dtype=bool))
    T_valid = T[mask]
    if T_valid.size > 0:
        # For each valid T, compute v = c_i * T + b_i for all i
        # But we need to know which pair (i,j) each T came from
        # Simpler: just evaluate max(c*T + b) for each T
        idx = np.where(mask)
```

```python
            # j is right if max(0,-d) <= j-d <= min(n-1, n-1-d) and 0 <= j-d < n

                if i_min <= j <= i_max and 0 <= j+d < n:
```
