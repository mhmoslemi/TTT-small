    n = h.size

    x = np.clip(np.nan_to_num(x), 0.0, 1.0)

    """
    Coordinate descent on 0-1 variable.
    h: initial 0-1 vector with n/2 ones
    active_k: list of lag indices (offset from n-1 in full correlation)
    Returns: optimized h
    """
    target = 0.5 * n
    h = (h >= 0.5).astype(float)  # binarize
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h))
    if n_ones > target:
        # Remove excess ones (set to 0)
        idx = np.where(h == 1)[0]
        h[idx[-(n_ones-target):]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[:target-n_ones]] = 1
    
    best_h = h.copy()
    best_c = evaluate_erdos(h)
    k = len(active_k)
    
    for it in range(max_iter):
        if deadline is not None and time.perf_counter() >= deadline:
            break
        
        # Compute delta for each position (cost of setting h[i] to 1)
        delta = np.zeros(n)
        for t in active_k:
            lag = t - (n - 1)
            if lag == 0:
                # C(0) = sum h[i](1-h[i]), for 0-1 this is 0
                # delta contribution: 1-2h[i] = -1 for h[i]=1, +1 for h[i]=0
                # But for 0-1, C(0) is always 0, so skip
                continue
            elif lag > 0:
                # C(lag) = sum_{i=0}^{n-1-lag} h[i](1-h[i+lag])
                # delta for h[i]: (1-h[i+lag]) if i <= n-1-lag, minus h[i-lag] if i >= lag
                valid_first = np.arange(n - lag)  # i where i+lag < n
                valid_second = np.arange(lag, n)  # i where i-lag >= 0
                delta[valid_first] += (1.0 - h[valid_first + lag])
                delta[valid_second] -= h[valid_second - lag]
            else:
                # lag < 0, let lag_abs = -lag
                lag_abs = -lag
                # C(-lag_abs) = sum_{i=0}^{n-1-lag_abs} h[i+lag_abs](1-h[i])
                # delta for h[i]: (1-h[i-lag_abs]) if i >= lag_abs, minus h[i+lag_abs] if i < n-lag_abs
                valid_first = np.arange(lag_abs, n)  # i where i-lag_abs >= 0
                valid_second = np.arange(n - lag_abs)  # i where i+lag_abs < n
                delta[valid_first] += (1.0 - h[valid_first - lag_abs])
                delta[valid_second] -= h[valid_second + lag_abs]
        
        delta /= k
        
        # Set h[i]=1 for n/2 positions with smallest delta
        # Current ones should stay, check if any should be swapped
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Find the worst one (largest delta among ones) and best zero (smallest delta among zeros)
        if ones.size == 0 or zeros.size == 0:
            break
        
        worst_one_delta = delta[ones]
        best_zero_delta = delta[zeros]
        
        max_worst = np.max(worst_one_delta)
        min_best = np.min(best_zero_delta)
        
        if min_best >= max_worst - 1e-12:
            # No improvement possible
            break
        
        # Perform swaps: swap the worst on
        # Actually, let's find the optimal 0-1 assignment:
        # h[i]=1 for the n/2 indices with smallest delta
        sorted_idx = np.argsort(delta)
        new_ones = sorted_idx[:target]
        
        if np.array_equal(np.sort(ones), np.sort(new_ones)):
            break
        
        h_new = np.zeros(n)
        h_new[new_ones] = 1
        c_new = evaluate_erdos(h_new)
        
        if c_new < best_c - 1e-12:
            h = h_new
            ones = new_ones.copy()
            zeros = np.where(h == 0)[0]
            best_c = c_new
        else:
            break  # If full reassignment doesn't help, we're at a local min
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float('inf')
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if corr.size == 0:
        return float('inf')
    return float(np.max(corr) * dx)

def project_density(x, target):
    x = np.clip(np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-12:
        return x.copy()
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-13:
        free = np.where((h > 1e-15) & (h < 1.0 - 1e-15))[0]
        if free.size > 0:
            idx = free[np.argmax(1.0 - h[free])] if res > 0 else free[np.argmax(h[free])]
            h[idx] = np.clip(h[idx] + res, 0.0, 1.0)
    return h

def compute_deltas(h, active_lags, n):
    """Compute the coefficient (delta) for each h[i] in the active-set objective.
    active_lags: list of integer lags (positive or negative)
    Returns: delta array where delta[i] is the coefficient of h[i]
    """
    delta = np.zeros(n)
    for lag in active_lags:
        if lag == 0:
            continue  # C(0) is always 0 for 0-1
        elif lag > 0:
            # C(lag) = sum_{i=0}^{n-1-lag} h[i](1-h[i+lag])
            # Coeff of h[i]: (1-h[i+lag]) if i < n-lag, minus h[i-lag] if i >= lag
            if n - lag > 0:
                delta[:n-lag] += (1.0 - h[lag:n])
            if lag < n:
                delta[lag:] -= h[:n-lag]
        else:
            la = -lag  # lag_abs
            # C(-la) = sum_{i=0}^{n-1-la} h[i+la](1-h[i])
            # Coeff of h[i]: (1-h[i-la]) if i >= la, minus h[i+la] if i < n-la
            if la < n:
                delta[la:] += (1.0 - h[:n-la])
                delta[:n-la] -= h[la:n]
    return delta / float(len(active_lags))

def binarize_with_constraint(h, target):
    """Convert to 0-1 with exactly target ones."""
    n = h.size
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones > target:
        ones_idx = np.where(h_bin == 1)[0]
        # Remove the "worst" ones (highest h value... or just the last ones)
        h_bin[ones_idx[-(n_ones-target)]] = 0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        h_bin[zeros_idx[:target-n_ones]] = 1
    return h_bin

def mean_field_step(h, active_lags, n, target):
    """One mean-field update: reassign h[i]=1 to the target positions with smallest delta."""
    delta = compute_deltas(h, active_lags, n)
    sorted_idx = np.argsort(delta)
    new_ones = sorted_idx[:target]
    h_new = np.zeros(n)
    h_new[new_ones] = 1.0
    return h_new, delta

        # Mean-field step
        delta = compute_deltas(h, active_lags, n)
        sorted_idx = np.argsort(delta)
        new_ones = sorted_idx[:target]
