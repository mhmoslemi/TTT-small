def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def obj_soft(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * c
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / beta * dx)


def grad_soft(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * c
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-18:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            if n - 1 - lag >= 0:
                grad[:n - lag] += wk * (1.0 - h[lag:])
            if lag < n:
                grad[lag:] -= wk * h[:n - lag]
        else:
            m2 = -lag
            if n - 1 - m2 >= 0:
                grad[m2:] += wk * (1.0 - h[:n - m2])
            if m2 < n:
                grad[:n - m2] -= wk * h[m2:]
    return grad * dx

    return compute_c5(h, n)

        if diff > 0:

        else:
            mask = h < 1.0 - 1e-12
        if mask.sum() == 0:
            break
        adj = diff / mask.sum()

        else:

        h[mask] -= diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h
