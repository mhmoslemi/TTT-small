        target = n_best / 2.0
        time_remaining = budget_s * 0.96 - elapsed()

        # Multi-beta continuous refinement
        for beta_val in [20.0, 50.0, 150.0]:
            if elapsed() > budget_s * 0.96:
                break
            t_alloc = max(10.0, time_remaining / 3)
            h_ref = continuous_refine(best_h, n_best, target, t_alloc, beta=beta_val)
            try_update(h_ref, n_best)
            time_remaining = budget_s * 0.96 - elapsed()

        # Try L-BFGS-B style: penalty method
        if elapsed() < budget_s * 0.98:
            n_best = best_n
            target = n_best / 2.0
            h = project_to_sum(best_h, target)

            def penalty_obj(x, mu=100.0):
                hc = np.clip(x, 0.0, 1.0)
                penalty = mu * (np.sum(hc) - target) ** 2
                corr = np.correlate(hc, 1.0 - hc, mode='full')
                return smooth_max(corr, 80.0) + penalty

            bounds_lb = [(0.0, 1.0)] * n_best
            try:
                result = minimize(penalty_obj, h, method='L-BFGS-B',
                                 bounds=bounds_lb,
                                 options={'maxiter': 200, 'gtol': 1e-8})
                if result.x is not None:
                    h_pen = project_to_sum(result.x, target)
                    try_update(h_pen, n_best)
            except Exception:
                pass

    # Final: ensure validity
    target_final = best_n / 2.0
    best_h = project_to_sum(best_h, target_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)

    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, best_n)
