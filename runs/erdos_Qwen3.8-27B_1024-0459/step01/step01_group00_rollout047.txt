import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-15
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_vals(vals, beta):
    cmax = np.max(vals)
    return cmax + (1.0/beta) * np.log(np.sum(np.exp(beta*(vals - cmax))))

def c5_smooth(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max_vals(corr, beta)

def c5_direct(x, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return np.max(corr)

def subgradient_step(h, n, step_size):
    """Take a subgradient step on the C5 objective."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    # Subgradient of max_k corr[k] w.r.t. h
    # corr[max_idx] = sum_i h[i] * (1 - h[i + max_idx - (n-1)])
    # (depending on correlate convention)
    # For np.correlate(a, v, mode='full'):
    # c[k] = sum_n a[n+k] * v[n] for k in range(-(len(v)-1), len(a))
    # The full output has length len(a)+len(v)-1 = 2n-1
    # Index 0 corresponds to lag -(n-1), index n-1 corresponds to lag 0
    # So max_idx in the full output corresponds to lag max_idx - (n-1)
    
    lag = max_idx - (n - 1)
    # c[lag] = sum_i h[i] * (1-h[i+lag]) (approximately)
    # Actually for correlate(h, 1-h):
    # c[k] = sum_n h[n+k] * (1-h[n])
    # where n ranges over valid indices
    
    # Subgradient: dc/dh[i] for the max correlation term
    # c[lag] = sum_n h[n+lag] * (1-h[n])
    # dc/dh[i] = (1-h[i]) if i = n+lag for some valid n
    #            - h[i+lag] if i = n for some valid n
    #            + (1-h[i]) - h[i+lag] if both
    
    grad = np.zeros(n)
    for i in range(n):
        # Contribution when h[i] appears as the first argument: h[i] * (1-h[i-lag])
        # if 0 <= i - lag < n: grad[i] += (1 - h[i-lag])
        # Contribution when h[i] appears in (1-h[.]): -h[i+lag] * (1)... 
        # Actually c = sum_n h[n+lag] * (1-h[n])
        # dc/dh[i]: 
        #   if i = n+lag (i.e., n = i-lag, valid if 0<=i-lag<n): contribution (1-h[i-lag])
        #   if i = n (valid if 0<=i<n, always): contribution -h[i+lag] if 0<=i+lag<n
        gi = 0.0
        if 0 <= i - lag < n:
            gi += (1.0 - h[i - lag])
        if 0 <= i + lag < n:
            gi -= h[i + lag]
        grad[i] = gi
    
    # Project gradient step
    h_new = h - step_size * grad
    h_new = make_feasible(h_new, n)
    return h_new, max_idx

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def thue_morse(n):
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return tm

def generate_initializations(n, rng):
    inits = []
    
    # Thue-Morse
    tm = make_feasible(thue_morse(n), n)
    inits.append(tm)
    
    # Binary periodic patterns
    if n % 4 == 0:
        inits.append(make_feasible(np.tile([1,1,0,0], n//4), n))
        inits.append(make_feasible(np.tile([1,0,1,0], n//4), n))
    if n % 2 == 0:
        h = np.zeros(n)
        h[::2] = 1.0
        inits.append(make_feasible(h, n))
        h = np.zeros(n)

        g = i ^ (i >> 1)
        gray[i] = 1.0 if bin(g).count('1') % 2 == 0 else 0.0
    inits.append(make_feasible(gray, n))
    
    return inits

def optimize_slq_direct(n, h_init, max_iter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(c5_direct, h_init, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': ftol})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

    current = make_feasible(h_init.copy(), n)

    for beta in betas:
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            return smooth_max_vals(corr, beta)
        
        def jac(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            cmax = np.max(corr)
            weights = np.exp(beta * (corr - cmax))
            weights /= np.sum(weights)  # softmax weights
            # Gradient of smooth max
            # d/dh_i of corr[k] for each k, weighted by softmax
            grad = np.zeros(n)
            for k in range(len(corr)):
                if weights[k] < 1e-10:
                    continue
                lag = k - (n - 1)
                for i in range(n):
                    gi = 0.0
                    if 0 <= i - lag < n:
                        gi += (1.0 - x[i - lag])
                    if 0 <= i + lag < n:
                        gi -= x[i + lag]
                    grad[i] += weights[k] * gi
            return grad
        # Skip gradient for speed (it's O(n*2n) which is expensive)

def population_search(n, h_inits, n_pop=6, n_gen=30, time_limit=None, t_start=None):

        selected = [pop[i].copy() for i in sorted_idx[:n_pop//2]]

        # Elitism: keep best
        new_pop.append(pop[best_idx].copy())
        new_pop.append(best_h.copy())
        
        for _ in range(n_pop - 2):

            if i == j:

            h_child = 0.5 * selected[i] + 0.5 * selected[j]

                h_child[mi] += np.random.uniform(-0.1, 0.1)

        fitnesses = [compute_c5(h) for h in pop]
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_c5 = fitnesses[best_idx]
            best_h = pop[best_idx].copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-15
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_vals(vals, beta):
    cmax = np.max(vals)
    shifted = beta * (vals - cmax)
    shifted = np.clip(shifted, -500, 500)
    return cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))


def c5_smooth_obj(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max_vals(corr, beta)


def c5_direct_obj(x, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return np.max(corr)


def sum_constraint_fun(x, n):
    return np.sum(x) - n / 2.0


def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def thue_morse_seq(n):
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return tm


def generate_inits(n, rng, best_h, best_n):
    inits = []
    # Interpolated best
    h_interp = make_feasible(interpolate_to_n(best_h, n), n)
    inits.append(h_interp)
    # Thue-Morse
    inits.append(make_feasible(thue_morse_seq(n), n))
    # Binary periodic
    if n % 4 == 0:
        inits.append(make_feasible(np.tile([1, 1, 0, 0], n // 4).astype(float), n))
    if n % 2 == 0:
        h = np.zeros(n)
        h[:n // 4] = 1.0
        h[n // 2: 3 * n // 4] = 1.0
        inits.append(make_feasible(h, n))
    # Random structured
    freq = rng.randint(2, 8)
    phase = rng.uniform(0, 2 * np.pi)
    h_s = 0.5 + 0.45 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    inits.append(make_feasible(h_s, n))
    # Random binary
    h_r = (rng.rand(n) > 0.5).astype(float)
    inits.append(make_feasible(h_r, n))
    return inits


def optimize_smooth_anneal(n, h_init, betas, max_iter_per_beta=1500):
    current = make_feasible(h_init.copy(), n)
    best_c5 = compute_c5(current)
    best_h = current.copy()
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        def obj_func(x):
            return c5_smooth_obj(x, n, beta)
        constraints = [{'type': 'eq', 'fun': lambda x: sum_constraint_fun(x, n)}]
        try:
            res = minimize(obj_func, current, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': max_iter_per_beta, 'ftol': 1e-14})
            if res.fun < 1e10:
                current = make_feasible(res.x, n)
                c5 = compute_c5(current)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current.copy()
        except Exception:
            pass
    return best_h, best_c5


def optimize_slsq_direct(n, h_init, max_iter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: sum_constraint_fun(x, n)}]
    try:
        res = minimize(c5_direct_obj, h_init, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': max_iter, 'ftol': ftol})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None


def local_perturbation(h, n, n_trials, step_size, time_budget, t_start):

    best_c5 = compute_c5(h)
    best_h = h.copy()

        if time.time() - t_start > time_budget:
            break
        h_try = h.copy()
        n_mod = np.random.randint(1, max(2, n // 25))
        idxs = np.random.choice(n, n_mod, replace=False)

            h_try[idx] += np.random.uniform(-current_step, current_step)
        h_try = np.clip(h_try, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h_try)
        mask = (h_try > 0.005) & (h_try < 0.995)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h_try[mask] += diff / cnt

        if c5_try < best_c5 - 1e-14:
            h = h_try.copy()
            best_c5 = c5_try
            best_h = h.copy()
            current_step *= 0.93
            if current_step < 1e-5:
                current_step = 1e-5
    return best_h, best_c5


def subgradient_descent(h, n, n_iter, step_size, time_budget, t_start):
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    ss = step_size
    for _ in range(n_iter):
        if time.time() - t_start > time_budget:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            gi = 0.0
            if 0 <= i - lag < n:
                gi += (1.0 - h[i - lag])
            if 0 <= i + lag < n:
                gi -= h[i + lag]
            grad[i] = gi
        h_new = h - ss * grad
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            best_h = h.copy()
            ss *= 0.98
        else:
            ss *= 0.85
            if ss < 1e-7:
                break
    return best_h, best_c5


def binary_flip_search(h, n, n_trials, time_budget, t_start):
    """Search over binary-like configurations by flipping individual entries."""
    h = make_feasible(h.copy(), n)
    # Round to binary
    h_bin = (h > 0.5).astype(float)
    h_bin = make_feasible(h_bin, n)
    # If sum isn't exactly n/2, adjust
    s = np.sum(h_bin)
    if s != n / 2.0:
        idx_sorted = np.argsort(h)
        if s > n / 2.0:
            for i in idx_sorted[::-1]:
                if h_bin[i] == 1.0:
                    h_bin[i] = 0.0
                    break
        else:
            for i in idx_sorted:
                if h_bin[i] == 0.0:
                    h_bin[i] = 1.0
                    break
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    for _ in range(n_trials):
        if time.time() - t_start > time_budget:
            break
        # Flip one or two entries
        n_flips = np.random.randint(1, 4)
        h_try = h_bin.copy()
        flipped = []
        for _ in range(n_flips):
            idx = np.random.randint(n)
            h_try[idx] = 1.0 - h_try[idx]
            flipped.append(idx)
        # Fix constraint by flipping one more
        s = np.sum(h_try)
        if s > n / 2.0:
            ones = np.where(h_try == 1.0)[0]
            if len(ones) > 0:
                h_try[np.random.choice(ones)] = 0.0
        elif s < n / 2.0:
            zeros = np.where(h_try == 0.0)[0]
            if len(zeros) > 0:
                h_try[np.random.choice(zeros)] = 1.0
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()
    return best_h, best_c5


def coordinate_refine(h, n, max_rounds, time_budget, t_start):
    """Refine by trying pairs of coordinate changes (i, j) to maintain sum constraint."""
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    for r in range(max_rounds):
        if time.time() - t_start > time_budget:
            break
        improved = False
        n_pairs = min(2000, n * 2)
        for _ in range(n_pairs):
            if time.time() - t_start > time_budget:
                break
            i, j = np.random.choice(n, 2, replace=False)
            di = np.random.uniform(-0.05, 0.05)
            h_try = h.copy()
            h_try[i] += di
            h_try[j] -= di
            if 0 <= h_try[i] <= 1 and 0 <= h_try[j] <= 1:
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try.copy()
                    best_c5 = c5_try
                    improved = True
        if not improved:
            break
    return h, best_c5
