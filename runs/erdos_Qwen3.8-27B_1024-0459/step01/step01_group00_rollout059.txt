import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_value(corr_dx, beta):
    cmax = np.max(corr_dx)
    return cmax + (1.0 / beta) * np.log(np.sum(np.exp(beta * (corr_dx - cmax))))


def softmax_weights(corr_dx, beta):
    v = beta * corr_dx
    v = v - np.max(v)
    w = np.exp(v)
    s = np.sum(w)
    if s < 1e-300:
        return np.ones(len(values)) / len(values)
    return w / s


def compute_smooth_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr_dx = corr_raw * dx
    w = softmax_weights(corr_dx, beta)
    grad = np.zeros(n)
    for m in range(len(corr_dx)):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        if k >= 0:
            j1_lo = k
            j1_hi = n - 1
            j2_lo = 0
            j2_hi = n - 1 - k
        else:
            j1_lo = 0
            j1_hi = n - 1 + k
            j2_lo = -k
            j2_hi = n - 1
        if j1_lo <= j1_hi:
            idx = np.arange(j1_lo, j1_hi + 1)
            grad[idx] += w[m] * dx * (1.0 - h[idx - k])
        if j2_lo <= j2_hi:
            idx = np.arange(j2_lo, j2_hi + 1)
            grad[idx] -= w[m] * dx * h[idx + k]
    return grad


def gradient_descent_smooth(h, n, betas, iters_per_beta, alpha_init=0.05):
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    for beta in betas:
        alpha = alpha_init / np.sqrt(max(beta, 1.0))
        stagnation = 0
        for it in range(iters_per_beta):
            corr_dx = np.correlate(h, 1.0 - h, mode='full') * dx
            f_val = smooth_max_value(corr_dx, beta)
            grad = compute_smooth_gradient(h, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            accepted = False
            for bt in range(25):
                h_new = h - alpha * grad
                h_new = make_feasible(h_new, n)
                corr_new = np.correlate(h_new, 1.0 - h_new, mode='full') * dx
                f_new = smooth_max_value(corr_new, beta)
                if f_new < f_val - 1e-16:
                    accepted = True
                    break
                alpha *= 0.5
            if accepted:
                h = h_new
                c5_new = np.max(corr_new)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stagnation = 0
                else:
                    stagnation += 1
            else:
                alpha *= 0.1
                stagnation += 1
            if stagnation > iters_per_beta // 2:
                break
    return best_h, best_c5


def pair_swap_local_search(h, n, n_trials, rng):

    best_c5 = compute_c5(h)

                h_try[j] += delta

                    best_c5 = c5_try

                    h = h_try
                    best_c5 = c5_try
                    break

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)


def generate_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return make_feasible(h, n)


def generate_periodic(n, pattern):
    p = len(pattern)
    h = np.zeros(n)
    for i in range(n):
        h[i] = pattern[i % p]
    return make_feasible(h, n)


def generate_salt(n, rng):
    h = np.zeros(n)
    positions = rng.choice(n, n // 2, replace=False)
    h[positions] = 1.0
    return make_feasible(h, n)
