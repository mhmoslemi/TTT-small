            lo, hi = 0.0, 1.0

                else:

        c = compute_c5(h)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        if no_improve_count > 50:
            break
    
    return best_h, best_c

def compute_slope_bases(h, j, corr):
    n = len(h)
    slopes = np.zeros(2*n-1)
    bases = np.copy(corr)
    hj = h[j]
    for d in range(1, n):
        # Forward: p = n-1+d
        sp = 0.0
        if d <= j:
            sp += (1.0 - h[j-d])
        if d <= n-1-j:
            sp -= h[j+d]
        slopes[n-1+d] = sp
        bases[n-1+d] -= sp * hj
        
        # Backward: p = n-1-d
        sp = 0.0
        if d <= n-1-j:
            sp += (1.0 - h[j+d])
        if d <= j:

    return slopes, bases

def project_sum(h, target):
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s < target:
            hi = mid
        else:
            lo = mid
    h_new = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fix remaining error
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1 - h_new)[:max(1,int(abs(diff)))]
            # distribute
            per = diff / len(idx)

def smoothed_max_c5(h, alpha):
    n = h.size
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(conv))
    if alpha < 1e-14:
        return m * (2.0 / n)
    shifted = (conv - m) / alpha
    shifted = np.clip(shifted, -500, 500)
    A = float(np.sum(np.exp(shifted)))
    if not math.isfinite(A) or A <= 0:
        return m * (2.0 / n)
    return (m + alpha * math.log(A)) * (2.0 / n)

def slsqp_smoothed(h_init, n, time_limit):
    h = project_sum(h_init, 0.5 * n).copy()
    n = int(n)

        return smoothed_max_c5(np.clip(x, 0, 1), alpha)

        n2 = h.size

def _add_grad(grad, idx, coef, h, n):
    if coef == 0.0:
        return
    if idx == n - 1:
        grad += coef * (1.0 - 2.0 * h)
        return
    if idx % 2 == 1:
        d = (idx + 1) // 2
    else:
        d = idx // 2
    if d <= 0 or d >= n:
        return
    m = n - d
    if idx % 2 == 1:
        grad[:m] += coef * (1.0 - h[d:])
        grad[d:] += coef * (-h[:m])
    else:
        grad[:m] += coef * (-h[d:])
        grad[d:] += coef * (1.0 - h[:m])

        update_best(h0, c0, n0)
        
        # Coordinate descent on parent
        rem = max(5.0, float(_DEADLINE - time.time()) - 80.0)
        cd_time = min(200.0, rem * 0.4)
        if cd_time > 10:
            h_cd, c_cd = coord_descent(h0, n0, cd_time, rng)
            update_best(h_cd, c_cd, n0)
        
        # SLSQP polish
        rem = float(_DEADLINE - time.time()) - 60.0
        if rem > 30:
            best_local = best_h if best_n == n0 else h0
            slab = project_sum(slsqpolish(best_local, n0, min(60, rem)), 0.5 * n0)
            # Note: using slsqp_polish below
            pass
    
    # Phase 2: Coordinate descent at multiple resolutions
    resolutions = []
    if parent is not None:
        n0 = int(parent.size)
        resolutions = [n0, 2 * n0, 3 * n0]
        resolutions = [r for r in resolutions if 20 <= r <= 400]
    else:
        resolutions = [70, 140, 210]
    
    for n in resolutions:
        rem = float(_DEADLINE - time.time()) - 50.0
        if rem < 30:
            break
        # Build initial h for this resolution
        if parent is not None:
            h_init = project_sum(upsample_linear(parent, n), 0.5 * n)
        else:
            h_init = project_sum(synthetic_h(n, rng), 0.5 * n)
        
        # If we have a better solution at another resolution, try that too
        if best_h is not None and best_n != n:
            h_init2 = project_sum(upsample_linear(best_h, n), 0.5 * n)
            c_init2 = compute_c5(h_init2)
            c_init1 = compute_c5(h_init)
            if c_init2 < c_init1:
                h_init = h_init2
        
        c_init = compute_c5(h_init)
        update_best(h_init, c_init, n)
        
        # Coordinate descent
        cd_time = min(250.0, rem * 0.55)
        if cd_time > 15:
            h_cd, c_cd = coord_descent(h_init, n, cd_time, rng)
            update_best(h_cd, c_cd, n)
        
        # SLSQP polish on result
        rem = float(_DEADLINE - time.time()) - 40.0
        if rem > 30:
            h_pol = slsqp_polish(best_h if best_n == n else h_cd, n, min(80, rem))
            c_pol = compute_c5(h_pol)
            update_best(h_pol, c_pol, n)
    
    # Phase 3: Binary search at even resolutions
    bi_res = []
    if parent is not None:
        n0 = int(parent.size)
        bi_res = [2 * n0, 4 * n0] if 2 * n0 >= 20 else [n0, 2 * n0]
    else:
        bi_res = [140, 196]
    bi_res = [r for r in bi_res if r >= 20 and r <= 300 and r % 2 == 0]
    
    for n in bi_res:
        rem = float(_DEADLINE - time.time()) - 40.0
        if rem < 20:
            break
        # Get a good starting binary from best continuous
        if best_h is not None:
            h_cont = project_sum(upsample_linear(best_h, n), 0.5 * n)
            hb = np.zeros(n, dtype=np.int8)
            thresh = 0.5
            hb[h_cont >= thresh] = 1
            if hb.sum() != n // 2:
                # Adjust
                order = np.argsort(h_cont)
                hb = np.zeros(n, dtype=np.int8)
                hb[order[n//2:]] = 1
            hb = hb.astype(float)
        else:
            hb = make_binary(n, rng).astype(float)
        
        c_b = compute_c5(hb)
        update_best(hb, c_b, n)
        
        blim = min(120.0, rem * 0.6)
        if blim > 10:
            h_bin, c_bin = binary_local_search(hb, n, rng, time.time() + blim, max_ops=30000)
            update_best(h_bin, c_bin, n)
        
        # Upsample binary to continuous for further optimization
        if time.time() < _DEADLINE - 30:
            n2 = n
            h2 = project_sum(hb, 0.5 * n2)
            c2 = compute_c5(h2)
            cd_time2 = min(100.0, float(_DEADLINE - time.time()) - 30.0)
            if cd_time2 > 10:
                h2c, c2c = coord_descent(h2, n2, cd_time2, rng)
                update_best(h2c, c2c, n2)
    
    # Phase 4: Final SLSQP with smoothed max on best solution
    if best_h is not None:
        rem = float(_DEADLINE - time.time()) - 10.0
        if rem > 30:
            h_sm = slsqp_smoothed(best_h, best_n, rem)
            c_sm = compute_c5(h_sm)
            update_best(h_sm, c_sm, best_n)
        
        # One more coord descent at best resolution with random restart
        rem = float(_DEADLINE - time.time()) - 10.0
        if rem > 20:
            h_perturbed = best_h.copy()
            h_perturbed += rng.standard_normal(best_n) * 0.05
            h_perturbed = project_sum(h_perturbed, 0.5 * best_n)
            h_cd2, c_cd2 = coord_descent(h_perturbed, best_n, min(80, rem), rng)
            update_best(h_cd2, c_cd2, best_n)
    
    # Final check
    if best_h is None:
        n_fb = 70
        best_h = project_sum(np.full(n_fb, 0.5), 0.5 * n_fb)
        best_c = compute_c5(best_h)
        best_n = n_fb
    
    # Ensure validity
    best_h = np.clip(best_h, 0.0, 1.0)
    if abs(float(best_h.sum()) - 0.5 * best_n) > 1e-6:
        best_h = project_sum(best_h, 0.5 * best_n)
        best_c = compute_c5(best_h)
    
    return best_h, float(best_c), int(best_n)
