def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)


def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) <= 1e-14 * max(1, n):
        return x
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-15:
        free = np.where((h > 1e-14) & (h < 1.0 - 1e-14))[0]
        if free.size:
            if res > 0:
                h[free] += res / float(free.size)
            else:
                h[free] += res / float(free.size)
            h = np.clip(h, 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-15:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size:
            h[free[0]] += res
            h = np.clip(h, 0.0, 1.0)
    return h


def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    n = int(n)
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)


def compute_gradient_for_lag(h, s):
    """Compute gradient of c_s = sum_j h[j]*(1-h[j+s]) w.r.t. h."""
    n = h.size
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        # First term: (1-h[m+s]) for m=0..n-s-1
        g[0:n-s] += 1.0 - h[s:n]
        # Second term: -h[m-s] for m=s..n-1
        g[s:n] -= h[0:n-s]
    else:
        t = -s
        # First term: (1-h[m-t]) for m=t..n-1
        g[t:n] += 1.0 - h[0:n-t]
        # Second term: -h[m+t] for m=0..n-t-1
        g[0:n-t] -= h[t:n]
    return g


def get_active_lags(h, k=10, threshold_frac=0.0):
    """Get indices in correlate array corresponding to top-k overlaps."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    n = h.size
    center = n - 1
    idxs = np.argsort(corr)[-k:]
    # Convert to lag shifts
    lags = (idxs - center).astype(int)
    return lags, corr


def subgradient_step(h, lags, step_size):
    """Take a projected subgradient step."""
    n = h.size
    target = 0.5 * n
    g = np.zeros(n)
    for s in lags:
        g += compute_gradient_for_lag(h, int(s))
    g /= float(len(lags))
    h_new = h - step_size * g
    return project_density(h_new, target)


def frank_wolfe_dir(g, target):
    """Frank-Wolfe linear minimization: min g·y s.t. 0≤y≤1, sum(y)=target."""
    sorted_idx = np.argsort(g)
    n = g.size
    # Greedy: assign y=1 to smallest g, y=0 to largest, one fractional
    y = np.zeros(n)
    remaining = target
    for i in range(n):
        idx = sorted_idx[i]
        if remaining >= 1.0 - 1e-15:
            y[idx] = 1.0
            remaining -= 1.0
        elif remaining > 1e-15:
            y[idx] = remaining
            remaining = 0.0
        else:
            break
    return y

        g = np.zeros(n)

        g /= float(len(lags))
        y = frank_wolfe_dir(g, target)
        # Line search
        improved = False
        for gamma in (1.0 / (1.0 + 0.5 * cosh), 0.5 / (1.0 + 0.5 * cosh),

def smooth_max_objective(x, n, alpha, target):
    h = np.asarray(x, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode="full")

    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0 or not np.isfinite(z):
        return 1e100
    return float((2.0 / float(n)) * (m + np.log(z) / alpha))


def smooth_max_grad(x, n, alpha, target):

def make_binary_init(n, rng):
    """Create a random binary-like initial solution."""
    h = (rng.random(n) > 0.5).astype(float)
    # Fix to get exactly n/2 ones
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        one_idx = np.where(h == 1)[0]
        to_flip = rng.choice(one_idx, size=n_ones - target, replace=False)
        h[to_flip] = 0.0
    elif n_ones < target:
        zero_idx = np.where(h == 0)[0]
        to_flip = rng.choice(zero_idx, size=target - n_ones, replace=False)
        h[to_flip] = 1.0
    return h * 0.9 + 0.05  # slight interior


def make_periodic_init(n, rng, period_blocks=4):
    """Create periodic-block initial solution."""
    h = np.zeros(n)
    block = n // period_blocks
    if block < 2:
        return make_binary_init(n, rng)
    for i in range(period_blocks):
        val = 1.0 if rng.random() > 0.5 else 0.0
        start = i * block
        end = start + block if i < period_blocks - 1 else n
        h[start:end] = val
    h = project_density(h, 0.5 * n)
    # Add small noise
    h += rng.random(n) * 0.05
    return project_density(np.clip(h, 0, 1), 0.5 * n)

        pproj = project_density(parent, 0.5 * nparent)
        cp = evaluate_erdos(pproj)
        if cp < best_c:
            best_h = pproj.copy()
            best_c = cp
            best_n = nparent

    if best_h is None:
        n0 = 90
        best_h = make_binary_init(n0, rng)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    # Multi-resolution optimization
    resolutions = []
    base_n = best_n if best_n > 10 else 90
    for mult in (1, 2, 4):
        n = int(base_n * mult)
        if 40 <= n <= 1440:
            resolutions.append(n)
    for n in (90, 180, 360, 720):
        if n not in resolutions and 40 <= n <= 1440:
            resolutions.append(n)
    resolutions = sorted(set(resolutions))
    # Limit to manageable list
    resolutions = [n for n in resolutions if n <= 720]
    if len(resolutions) > 5:
        resolutions = resolutions[:5]

    for ni in resolutions:
        if deadline - time.perf_counter() < 20.0:
            break

        # Create starting points
        starts = []
        if best_h is not None:
            starts.append(upsample_h(best_h, ni))
        if parent is not None:
            starts.append(upsample_h(parent, ni))
        # Random inits
        for mm in range(2):
            starts.append(make_binary_init(ni, rng))
        starts.append(make_periodic_init(ni, rng, period_blocks=3))
        starts.append(make_periodic_init(ni, rng, period_blocks=7))

        # Deduplicate
        uniq = []
        for s in starts:
            s = project_density(s, 0.5 * ni)
            dup = False
            for u in uniq:
                if np.allclose(s, u, atol=1e-8):
                    dup = True
                    break
            if not dup:
                uniq.append(s)

        # Evaluate all starts
        best_start = None
        best_start_c = float("inf")
        for s in uniq:
            c = evaluate_erdos(s)
            if c < best_start_c:
                best_start_c = c
                best_start = s

        # Allocate time based on number of resolutions remaining
        n_res_left = len(resolutions) - resolutions.index(ni)
        total_time = deadline - time.perf_counter()
        if n_res_left <= 0:
            n_res_left = 1
        per_res = min(total_time / n_res_left, 300.0)

        if ni <= 90:
            per_start_time = min(per_res / max(1, len(uniq)), 40.0)
        elif ni <= 360:
            per_start_time = min(per_res / max(1, len(uniq)), 60.0)
        else:
            per_start_time = min(per_res / max(1, len(uniq)), 80.0)
        per_start_time = max(10.0, per_start_time)

        # Optimize from best start first, then others if time permits
        order = list(range(len(uniq)))
        # Put best start first
        for i, s in enumerate(uniq):
            if np.allclose(s, best_start, atol=1e-8):
                order.insert(0, order.pop(order.index(i)))
                break

        for idx in order:
            if deadline - time.perf_counter() < 8.0:
                break
            s = uniq[idx]
            c_s = evaluate_erdos(s)
            if c_s >= best_c + 0.01:
                continue  # Skip bad starts
            sub_dl = time.perf_counter() + per_start_time
            sub_dl = min(sub_dl, deadline)
            h_out, c_out = optimize_at_resolution(s, ni, sub_dl, rng)
            if c_out < best_c:
                best_h = h_out
                best_c = c_out
                best_n = ni

    # Final intensive polishing at best resolution
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 10.0:
        nbest = int(best_h.size)
        # More subgradient iterations
        h_p, c_p = subgrad_optimize(best_h, deadline, max_iter=500, k_active=6)
        if c_p < best_c:
            best_h, best_c = h_p, c_p
        # Local refinement
        if deadline - time.perf_counter() > 5.0:
            h_p2, c_p2 = local_pair_refine(best_h, deadline, max_iter=300)
            if c_p2 < best_c:
                best_h, best_c = h_p2, c_p
        # Final FW
        if deadline - time.perf_counter() > 5.0:
            h_p3, c_p3 = fw_optimize(best_h, deadline, max_iter=200, k_active=10)
            if c_p3 < best_c:
                best_h, best_c = h_p3, c_p3

    if best_h is None:
        n_fallback = 90
        best_h = make_binary_init(n_fallback, rng)
        best_c = evaluate_erdos(best_h)
        best_n = n_fallback

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
