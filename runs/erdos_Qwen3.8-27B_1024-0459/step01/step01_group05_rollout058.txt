    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

    return np.correlate(h, 1.0 - h, mode='full')

    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700, 700)
    return m + np.log(np.sum(np.exp(shifted))) / beta

    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700, 700)
    e = np.exp(shifted)
    return e / np.sum(e)

def obj_and_grad(h, beta):
    n = len(h)
    corr = compute_corr(h)  # length 2n-1
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700, 700)
    e = np.exp(shifted)
    w = e / np.sum(e)

    s = np.arange(-(n-1), n)  # shifts

    s_row = s[None, :]  # (1, 2n-1)
    
    js_m = j_idx - s_row  # j - s, shape (n, 2n-1)
    js_p = j_idx + s_row  # j + s, shape (n, 2n-1)

    valid_p = (js_p >= 0) & (js_p < n)

    idx_p = np.clip(js_p, 0, n-1)

    hp = np.take(h, idx_p, axis=0)  # (n, 2n-1): h[j+s]
    
    w_row = w[None, :]  # (1, 2n-1)
    
    term_m = np.where(valid_m, w_row * (1.0 - hm), 0.0)

    grad = np.sum(term_m, axis=1) - np.sum(term_p, axis=1)
    
    return obj, grad

    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    # Exact fix
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            free = (h < 1.0 - 1e-15)
        else:
            free = (h > 1e-15)
        if np.any(free):
            per = diff / np.sum(free)
            h[free] = np.clip(h[free] + per, 0.0, 1.0)
    return h

def run_slsqp(h0, n, target, beta, max_iter, ftol):
    h0 = project_sum(h0, target)
    
    def obj(h):
        return obj_and_grad(h, beta)[0]
    
    def obj_grad(h):
        return obj_and_grad(h, beta)
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
                    'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        result = minimize(obj_grad, h0, method='SLSQP', bounds=bounds,
                         constraints=constraints,
                         options={'maxiter': max_iter, 'ftol': ftol})
        if result.success or result.x is not None:
            return project_sum(result.x, target)
    except Exception:
        pass
    return h0

def projected_gradient(h, target, beta, n_iter, step_init=0.01):
    """Projected gradient descent with backtracking."""
    h = project_sum(h, target)
    step = step_init
    
    for it in range(n_iter):
        obj, grad = obj_and_grad(h, beta)

        for _ in range(30):
            h_new = project_sum(h - step * grad, target)
            obj_new = obj_and_grad(h_new, beta)[0]
            if obj_new < obj - 1e-12:
                break
            step *= 0.5
        else:
            break
        
        if obj_new <= obj:
            h = h_new
        
        # Reduce step if it was too small
        if step < 1e-10:
            break
        
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_val(corr, beta):

def obj_grad_smooth(h, beta):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700, 700)
    e = np.exp(shifted)
    w = e / np.sum(e)
    obj = m + np.log(np.sum(e)) / beta
    
    s_arr = np.arange(-(n-1), n)
    j_idx = np.arange(n)[:, None]
    s_row = s_arr[None, :]
    
    js_m = j_idx - s_row
    js_p = j_idx + s_row
    
    valid_m = (js_m >= 0) & (js_m < n)
    valid_p = (js_p >= 0) & (js_p < n)
    
    idx_m = np.clip(js_m, 0, n-1)
    idx_p = np.clip(js_p, 0, n-1)
    
    hm = np.take(h, idx_m.flatten()).reshape(n, -1)
    hp = np.take(h, idx_p.flatten()).reshape(n, -1)
    
    w_row = w[None, :]
    
    term_m = np.where(valid_m, w_row * (1.0 - hm), 0.0)
    term_p = np.where(valid_p, w_row * hp, 0.0)
    
    grad = np.sum(term_m, axis=1) - np.sum(term_p, axis=1)
    return obj, grad

def local_swap_search(h, target, n_trials=2000):
    """Local search by swapping pairs."""
    h = project_sum(h, target)
    n = len(h)
    c5_curr = compute_c5(h)
    rng = np.random.RandomState(12345)
    
    for _ in range(n_trials):
        i, j = rng.randint(0, n, 2)
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        h = project_sum(h, target)
        c5_new = compute_c5(h)
        if c5_new < c5_curr - 1e-14:
            c5_curr = c5_new
        else:
            h[i], h[j] = h[j], h[i]
            h = project_sum(h, target)
    return h
