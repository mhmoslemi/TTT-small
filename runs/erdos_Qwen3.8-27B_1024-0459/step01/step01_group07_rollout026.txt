import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def compute_conv(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_smooth_c5(h, n_points, T=20.0):
    dx = 2.0 / n_points
    conv = compute_conv(h)
    vals = conv * dx
    max_val = np.max(vals)
    # Numerical stability
    shifted = vals - max_val
    logsum = np.log(np.sum(np.exp(T * shifted))) / T + max_val
    return logsum

def compute_smooth_grad(h, n_points, T=20.0):
    n = n_points
    dx = 2.0 / n
    conv = compute_conv(h)
    vals = conv * dx
    max_val = np.max(vals)
    shifted = vals - max_val
    exp_vals = np.exp(T * shifted)
    p = exp_vals / np.sum(exp_vals)  # softmax weights
    
    # Compute gradient: dx * sum_k p_k * dconv[k]/dh[j]
    grad = np.zeros(n)
    
    # For each shift s (which corresponds to index k = s + n - 1 in conv):
    # If s >= 0: dconv[k]/dh[j] = (1-h[j-s]) * I[s<=j<=n-1] - h[j+s] * I[0<=j<=n-1-s]
    # If s < 0 (p=-s): dconv[k]/dh[j] = (1-h[j+p]) * I[0<=j<n-p] - h[j-p] * I[p<=j<n]
    
    for s in range(-(n-1), n):
        k = s + (n - 1)
        w = p[k] * dx
        if w < 1e-10:
            continue
        if s >= 0:
            # (1-h[j-s]) for j in [s, n-1]
            if s <= n - 1:
                grad[s:] += w * (1.0 - h[:n-s])
            # -h[j+s] for j in [0, n-1-s]
            if n - 1 - s >= 0:

        else:

            # -h[j-p] for j in [p, n-1]
            if p_val <= n - 1:
                grad[p_val:] += -w * h[:n-p_val]
    
    return grad

def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:

        if diff > 0:
            free = h < 1.0
            if not np.any(free):

            step = diff / np.sum(free)
            h[free] = np.minimum(h[free] + step, 1.0)
        else:

            if not np.any(free):

            step = diff / np.sum(free)

    h = np.clip(h, 0.0, 1.0)

    s = np.sum(h)
    if abs(s - target) > 1e-10:

def projected_grad_descent_smooth(h0, n_points, n_iter=200, T=20.0, lr_init=0.5):

            best_h = h_new.copy()

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx


def compute_conv(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_smooth_c5(h, n_points, T=20.0):
    dx = 2.0 / n_points
    conv = compute_conv(h)
    vals = conv * dx
    max_val = np.max(vals)
    shifted = vals - max_val
    return np.log(np.sum(np.exp(T * shifted))) / T + max_val


def compute_smooth_grad(h, n_points, T=20.0):
    n = n_points
    dx = 2.0 / n
    conv = compute_conv(h)
    vals = conv * dx
    max_val = np.max(vals)
    shifted = vals - max_val
    exp_vals = np.exp(T * shifted)
    p = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        k = s + (n - 1)
        w = p[k] * dx
        if w < 1e-12:
            continue
        if s >= 0:
            if s <= n - 1:
                grad[s:] += w * (1.0 - h[:n - s])
            if n - 1 - s >= 0:
                grad[:n - s] += -w * h[s:]
        else:
            pp = -s
            if n - 1 - pp >= 0:
                grad[:n - pp] += w * (1.0 - h[pp:])
            if pp <= n - 1:
                grad[pp:] += -w * h[:n - pp]
    return grad


def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            free = h < (1.0 - 1e-15)
            if not np.any(free):
                break
            h[free] = np.minimum(h[free] + diff / np.sum(free), 1.0)
        else:
            free = h > 1e-15
            if not np.any(free):
                break
            h[free] = np.maximum(h[free] + diff / np.sum(free), 0.0)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        h += (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h


def pgd_smooth(h0, n_points, n_iter=300, T=20.0, lr_init=1.0):
    n = n_points
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h, n)
    lr = lr_init
    for it in range(n_iter):
        grad = compute_smooth_grad(h, n, T)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        val_new = compute_c5(h_new, n)
        if val_new > best_val:
            accepted = False
            for frac in [0.5, 0.25, 0.1, 0.05]:
                h_trial = h - lr * frac * grad
                h_trial = project_feasible(h_trial, n)
                val_trial = compute_c5(h_trial, n)
                if val_trial < best_val:
                    h_new = h_trial
                    val_new = val_trial
                    accepted = True
                    break
            if not accepted:
                lr *= 0.5
                h_new = h.copy()
                val_new = best_val
        if val_new < best_val:
            best_val = val_new
            best_h = h_new.copy()
        h = h_new
        if it % 40 == 39:
            lr *= 0.92
    return best_h, best_val


def slsqp_smooth(h0, n_points, maxiter=300, ftol=1e-11, T=30.0):
    n = n_points
    target = n / 2.0

    def obj(x):
        return compute_smooth_c5(x, n, T)

    def jac(x):
        return compute_smooth_grad(x, n, T)
