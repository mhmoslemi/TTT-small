def project_box_sum(y, target):

    m = y.size

    if target <= 0.0:
        return np.zeros(m, dtype=float)
    if target >= m:
        return np.ones(m, dtype=float)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0

            hi = mid

def project_box_sum(y, target):
    y = np.asarray(y, dtype=float).copy()
    m = y.size
    if m == 0:
        return y
    target = float(target)
    if target <= 0.0:
        return np.zeros(m, dtype=float)
    if target >= float(m):
        return np.ones(m, dtype=float)

    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        vals = np.clip(y - mid, 0.0, 1.0)
        s = float(np.sum(vals))
        if s > target:
            lo = mid
        else:
            hi = mid

    alpha = 0.5 * (lo + hi)
    proj = np.clip(y - alpha, 0.0, 1.0)

    diff = target - float(np.sum(proj))
    for _ in range(25):
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            mask = proj < (1.0 - 1e-15)
        else:
            mask = proj > 1e-15
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        proj[mask] += diff / float(cnt)
        proj = np.clip(proj, 0.0, 1.0)
        diff = target - float(np.sum(proj))

    return proj


def make_feasible_full(h, target=None):

def full_from_v(v, n):
    m = n // 2
    v = np.asarray(v, dtype=float)
    if v.size < m:
        v = np.pad(v, (0, m - v.size), mode="edge")
    else:
        v = v[:m]
    return np.concatenate((v, v[::-1]))


def evaluate_full(h):
    h = np.asarray(h, dtype=float)
    if h.size < 2:
        return 1e100
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / float(h.size)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def interpolate_full(h_old, n_new):
    h_old = np.asarray(h_old, dtype=float)
    n_old = h_old.size
    if n_old == 0:
        h_old = np.ones(2) * 0.5

def weighted_gradient(h, weights):
    n = h.size
    g = np.zeros(n, dtype=float)
    if n == 0:
        return g
    L = 2 * n - 1
    for idx in range(L):
        wi = float(weights[idx])
        if wi <= 1e-16:
            continue
        k = idx - (n - 1)
        if k == 0:
            g += wi * (1.0 - 2.0 * h)
        elif k > 0:
            seg = n - k
            g[k:] += wi * (1.0 - h[:seg])
            g[:seg] -= wi * h[k:]
        else:
            r = -k
            seg = n - r
            g[:seg] += wi * (1.0 - h[r:])
            g[r:] -= wi * h[:seg]
    return g


def value_x(x, n, sym=False, beta=None, tol=1.0e-4):
    if sym:
        h = full_from_v(x, n)
    else:
        h = np.asarray(x, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = float(np.max(corr))

    if beta is None or np.isinf(beta):
        return cmax

    beta = float(beta)
    if beta <= 1e-8:
        return cmax

    z = beta * (corr - cmax)
    ez = np.exp(z)
    s = float(np.sum(ez))
    if s <= 0.0:
        ez = np.zeros_like(corr)
        ez[int(np.argmax(corr))] = 1.0
        s = 1.0
    return cmax + float(np.log(s)) / beta


def value_grad_x(x, n, sym=False, beta=None, tol=1.0e-4):
    if sym:
        h = full_from_v(x, n)
    else:
        h = np.asarray(x, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = float(np.max(corr))

    if beta is None or np.isinf(beta):
        tol = float(tol)
        if tol <= 0.0:
            tol = 1.0e-4
        z = (corr - cmax) / tol
        mask = z > -50.0
        ez = np.zeros_like(corr)
        ez[mask] = np.exp(z[mask])
        s = float(np.sum(ez))
        if s <= 0.0:
            ez[int(np.argmax(corr))] = 1.0
            s = 1.0
        w = ez / s
        val = cmax
    else:
        beta = float(beta)
        if beta <= 1e-8:
            w = np.full(corr.size, 1.0 / float(corr.size))
            val = cmax
        else:
            z = beta * (corr - cmax)
            ez = np.exp(z)
            s = float(np.sum(ez))
            if s <= 0.0:
                ez = np.zeros_like(corr)
                ez[int(np.argmax(corr))] = 1.0
                s = 1.0
            w = ez / s
            val = cmax + float(np.log(s)) / beta

    gh = weighted_gradient(h, w)
    if sym:
        m = n // 2
        gv = gh[:m] + np.flip(gh[m:])
    else:
        gv = gh
    return val, gv


def constraint_sum(x):
    return np.sum(x) - _TARGET


def obj_grad_current(x):

def slsqp_step(x, n, sym, beta, maxiter):
    global _N, _M, _TARGET, _SYM_MODE, _BETA

    _N = int(n)
    _M = int(n) // 2
    _SYM_MODE = bool(sym)
    _TARGET = float(n) / 4.0 if sym else float(n) / 2.0
    _BETA = float(beta)

    x0 = make_feasible_x(x, n, sym)
    bounds = [(0.0, 1.0)] * int(x0.size)
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    try:
        res = minimize(
            obj_grad_current,
            x0,
            jac=True,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': 1e-9, 'disp': False}
        )
    except Exception:
        return x0

    if res is not None:
        rx = np.asarray(res.x, dtype=float)
        if rx.size == x0.size and np.all(np.isfinite(rx)):
            return make_feasible_x(rx, n, sym)
    return x0


def projected_smooth_pg(x, n, sym, beta, max_iter, stop_time):
    target = float(n) / 4.0 if sym else float(n) / 2.0
    x = make_feasible_x(x, n, sym)
    best_x = x.copy()
    best_c = evaluate_x(x, n, sym)

    step = 0.1
    for _ in range(int(max_iter)):
        if time.time() > stop_time:
            break
        f0, g = value_grad_x(x, n, sym, beta)
        if not np.all(np.isfinite(g)):
            break
        if float(np.linalg.norm(g)) < 1e-12:
            break

        t = step
        accepted = False
        for _ in range(35):
            cand = project_box_sum(x - t * g, target)
            f1 = value_x(cand, n, sym, beta)
            if f1 <= f0 + 1e-10 * (1.0 + abs(f0)):
                accepted = True
                break
            t *= 0.5
            if t < 1e-10:
                break

        if not accepted:
            break

        x = cand
        c = evaluate_x(x, n, sym)
        if c < best_c:
            best_x = x.copy()
            best_c = c

        if t > 0.9 * step:
            step = min(0.8, step * 1.25)
        else:
            step = max(1e-4, t * 0.8)

    return best_x, best_c


def subgradient_polish(x, n, sym, stop_time):
    target = float(n) / 4.0 if sym else float(n) / 2.0
    x = make_feasible_x(x, n, sym)
    best_x = x.copy()
    best_c = evaluate_x(x, n, sym)

    for step in [1e-2, 5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4, 5e-5, 2e-5]:
        improved = True
        while improved and time.time() < stop_time:

def top_pairs_from_gradient(g, k=20, min_diff=1e-8):
    m = int(np.size(g))
    if m < 2 or k <= 0:
        return []

    if m <= 600:
        diff = g[:, None] - g[None, :]
        np.fill_diagonal(diff, np.inf)
        flat = diff.ravel()
        k2 = min(int(k), flat.size)
        if flat.size <= k2:
            small = np.argsort(flat)[:k2]
        else:
            small = np.argpartition(flat, k2)[:k2]
        small = small[np.argsort(flat[small])]

        pairs = []
        seen = set()
        for idx in small:
            d = float(flat[idx])
            if d > -float(min_diff):
                break
            i = int(idx // m)
            j = int(idx - i * m)
            if (i, j) not in seen:
                pairs.append((i, j))
                seen.add((i, j))
            if len(pairs) >= k:
                break
        return pairs

    order = np.argsort(g)
    pairs = []
    seen = set()
    for j in order[::-1]:
        for i in order:
            if i == j:
                continue
            d = g[i] - g[j]
            if d < -float(min_diff):
                if (int(i), int(j)) not in seen:
                    pairs.append((int(i), int(j)))
                    seen.add((int(i), int(j)))
                break
        if len(pairs) >= k:
            break
    return pairs


def pair_line_search(x, n, sym, p, q, base_c, prefer_positive=False):

    if p == q:

    xp = float(x[p])
    xq = float(x[q])
    tlo = max(-xp, xq - 1.0)
    thi = min(1.0 - xp, xq)
    if thi - tlo <= 1e-12:
        return x, base_c

    cand = set()
    cand.add(0.0)

    if prefer_positive:
        if thi > 0.0:
            for t in np.linspace(0.0, thi, 15):
                cand.add(float(t))
        if tlo < 0.0:
            for t in np.linspace(tlo, 0.0, 7):
                cand.add(float(t))
    else:
        for t in np.linspace(tlo, thi, 21):
            cand.add(float(t))

    local = [1e-5, 3e-5, 1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 1e-1, 0.25, 0.5]
    for d in local:
        for sgn in (-1.0, 1.0):
            t = float(sgn) * float(d)
            if tlo - 1e-14 <= t <= thi + 1e-14:

    best_un = float(base_c)
    best_x = x

    for t in cand:
        if not np.isfinite(t):
            continue
        tt = float(np.clip(t, tlo, thi))
        if abs(tt) <= 1e-13:
            continue
        xp2 = xp + tt

        if xp2 < 0.0 or xp2 > 1.0 or xq2 < 0.0 or xq2 > 1.0:
            continue
        x2 = x.copy()

        c2 = evaluate_x(x2, n, sym)

    if best_x is not x:

        cur_x = best_x.copy()
        cur_c = best_un
        if rr > rl:
            for t in np.linspace(rl, rr, 17):
                tt = float(np.clip(float(t), tlo, thi))
                xp2 = xp + tt
                xq2 = xq - tt
                if xp2 < 0.0 or xp2 > 1.0 or xq2 < 0.0 or xq2 > 1.0:
                    continue
                x2 = x.copy()
                x2[p] = xp2
                x2[q] = xq2
                c2 = evaluate_x(x2, n, sym)
                if c2 < cur_c - 1e-14:
                    cur_c = c2
                    cur_x = x2

        cur_x = make_feasible_x(cur_x, n, sym)
        cur_c = evaluate_x(cur_x, n, sym)
        if cur_c < float(base_c) - 1e-15:
            return cur_x, cur_c

    return x, base_c

                xp2 = xp + tt
                xq2 = xq - tt
                if xp2 < 0.0 or xp2 > 1.0 or xq2 < 0.0 or xq2 > 1.0:
                    continue
                x2 = x.copy()
                x2[p] = xp2
                x2[q] = xq2
                c2 = evaluate_x(x2, n, sym)
                if c2 < cur_c - 1e-14:
                    cur_c = c2
                    cur_x = x2

        cur_x = make_feasible_x(cur_x, n, sym)
        cur_c = evaluate_x(cur_x, n, sym)
        if cur_c < float(base_c) - 1e-15:
            return cur_x, cur_c

    return x, base_c


def pair_polish(x, n, sym, stop_time):
    x = make_feasible_x(x, n, sym)
    best_x = x.copy()
    best_c = evaluate_x(x, n, sym)

    max_main_epochs = 7 if x.size <= 120 else 4
    main_beta = 120.0 if x.size <= 120 else 60.0

    for _ in range(max_main_epochs):
        if time.time() > stop_time:
            break
        _, g = value_grad_x(best_x, n, sym, main_beta, 1.0e-4)
        k = 16 if g.size <= 120 else 8
        pairs = top_pairs_from_gradient(g, k=k, min_diff=1e-9)
        if len(pairs) == 0:
            break
        for p, q in pairs:
            if time.time() > stop_time:
                break
            x2, c2 = pair_line_search(best_x, n, sym, p, q, best_c, prefer_positive=True)
            if c2 < best_c - 1e-14:
                best_x = x2
                best_c = c2
                break

    no_improve = 0
    max_no_improve = 250
    while time.time() < stop_time and no_improve < max_no_improve:
        m = int(best_x.size)
        if m < 2:
            break
        p = int(np.random.randint(m))
        q = int(np.random.randint(m))
        if p == q:
            continue
        x2, c2 = pair_local_search(best_x, n, sym, p, q, best_c)
        if c2 < best_c - 1e-14:
            best_x = x2
            best_c = c2
            no_improve = 0
        else:
            no_improve += 1

    return best_x, best_c


def optimize_candidate(x, n, sym, max_seconds):
    max_seconds = float(max_seconds)
    if max_seconds <= 0.0:
        x0 = make_feasible_x(x, n, sym)
        return x0, evaluate_x(x0, n, sym)

    start = time.time()
    stop = start + max_seconds

    x0 = make_feasible_x(x, n, sym)
    best_x = x0.copy()
    best_c = evaluate_x(x0, n, sym)

    m = (n // 2) if sym else n

    if max_seconds >= 25.0 and m <= 600:
        if m <= 100:
            betas = [10.0, 50.0, 200.0, 800.0, float("inf")]
            iters = [150, 180, 220, 260, 180]
        elif m <= 200:
            betas = [20.0, 80.0, 300.0, 1000.0, float("inf")]
            iters = [100, 120, 150, 180, 120]
        elif m <= 400:
            betas = [20.0, 100.0, 500.0, float("inf")]
            iters = [70, 90, 110, 80]
        else:
            betas = [50.0, 300.0, float("inf")]
            iters = [50, 60, 50]

        if max_seconds < 80.0:
            betas = betas[-2:]
            iters = iters[-2:]

        slsq_stop = start + 0.55 * max_seconds
        xcur = best_x.copy()

        for idx, beta in enumerate(betas):
            rem = slsq_stop - time.time()
            if rem <= 3.0:
                break
            if m <= 100:
                per_sec = 30.0
            elif m <= 200:
                per_sec = 15.0
            else:
                per_sec = 8.0
            cap = int(rem * per_sec)
            it = min(int(iters[idx]), max(20, cap))
            x2 = slsqp_step(xcur, n, sym, beta, it)
            c2 = evaluate_x(x2, n, sym)
            if c2 < best_c:
                best_x = x2.copy()
                best_c = c2
            if c2 <= best_c + 0.02:
                xcur = x2
            else:
                xcur = best_x.copy()

    pg_stop = start + 0.78 * max_seconds
    if time.time() < pg_stop - 3.0:
        beta_pg = 120.0 if m <= 120 else 60.0
        x2, c2 = projected_smooth_pg(best_x, n, sym, beta_pg, 4000, pg_stop)
        if c2 < best_c:
            best_x = x2
            best_c = c2

    sub_stop = start + 0.90 * max_seconds
    if time.time() < sub_stop - 2.0:
        x2, c2 = subgradient_polish(best_x, n, sym, sub_stop)
        if c2 < best_c:
            best_x = x2
            best_c = c2

    if time.time() < stop - 2.0:
        x2, c2 = pair_polish(best_x, n, sym, stop)
        if c2 < best_c:
            best_x = x2
            best_c = c2

    best_x = make_feasible_x(best_x, n, sym)
    best_c = evaluate_x(best_x, n, sym)
    return best_x, best_c


def update_best_tuple(best_h, best_c, best_n, x_out, n_out, sym_out):
    if x_out is None:
        return best_h, best_c, best_n
    if sym_out:
        h_out = full_from_v(x_out, n_out)
    else:
        h_out = np.asarray(x_out, dtype=float).copy()
    h_out = make_feasible_full(h_out, float(n_out) / 2.0)
    c_out = evaluate_full(h_out)
    if np.isfinite(c_out) and c_out < best_c:
        return h_out, c_out, int(n_out)
    return best_h, best_c, best_n


def binarize_full(h, n):
    h = np.asarray(h, dtype=float)
    if h.size != n:
        h = interpolate_full(h, n)
    if n % 2 != 0:
        hb = np.zeros(n, dtype=float)
        target = n // 2
        order = np.argsort(-h)
        if target > 0:
            hb[order[:target]] = 1.0
        if target < n:
            hb[order[target]] = 0.5
        return make_feasible_full(hb, float(n) / 2.0)

    target = n // 2
    hb = np.zeros(n, dtype=float)
    order = np.argsort(-h)
    if target > 0:
        hb[order[:target]] = 1.0
    return hb


def binary_local_polish(h, n, stop_time):
    h = binarize_full(h, n)
    best = h.copy()
    best_c = evaluate_full(best)

    ones = np.nonzero(best > 0.5)[0]
    zeros = np.nonzero(best < 0.5)[0]

    no_improve = 0
    max_no_improve = 1200

    while time.time() < stop_time and no_improve < max_no_improve:
        if ones.size == 0 or zeros.size == 0:
            break
        i1 = int(np.random.randint(ones.size))
        i0 = int(np.random.randint(zeros.size))
        i = int(ones[i1])
        j = int(zeros[i0])
        if i == j:
            continue

        h2 = best.copy()
        h2[i] = 0.0
        h2[j] = 1.0
        c2 = evaluate_full(h2)

        if c2 < best_c - 1e-15:
            best = h2
            best_c = c2
            ones = np.nonzero(best > 0.5)[0]
            zeros = np.nonzero(best < 0.5)[0]
            no_improve = 0
        else:
            no_improve += 1

    return best, evaluate_full(best)


def run(seed=42, budget_s=1300, **kwargs):
    global _N, _M, _TARGET, _SYM_MODE, _BETA

    np.random.seed(int(seed))
    t0 = time.time()
    safe_budget = max(2.0, float(budget_s) - 5.0)
    deadline = t0 + safe_budget

    best_h = None
    best_n = 0
    best_c = 1e100

    parent = None
    try:
        parent = initial_h_values
    except Exception:
        parent = None

    if parent is not None:
        hpar = np.asarray(parent, dtype=float).ravel()
        if hpar.size >= 2:
            hpar = make_feasible_full(hpar, float(hpar.size) / 2.0)
            cpar = evaluate_full(hpar)
            if np.isfinite(cpar):
                best_h = hpar
                best_n = int(hpar.size)
                best_c = cpar

    if best_h is None:
        n0 = 160
        fb_h = None
        fb_c = 1e100
        for kind in range(5):
            htmp = structured_full(n0, kind)
            ctmp = evaluate_full(htmp)
            if np.isfinite(ctmp) and ctmp < fb_c:
                fb_c = ctmp
                fb_h = htmp
        htmp = structured_full(n0, 3)
        htmp += 0.25 * np.random.randn(n0)
        htmp = make_feasible_full(htmp, float(n0) / 2.0)
        ctmp = evaluate_full(htmp)
        if np.isfinite(ctmp) and ctmp < fb_c:
            fb_c = ctmp
            fb_h = htmp
        best_h = fb_h
        best_n = int(n0)
        best_c = fb_c

    candidates = []
    cid = 0

    def add_candidate(local_x, local_n, local_sym):
        nonlocal_cid = cid
        c0 = evaluate_x(local_x, local_n, local_sym)
        if not np.isfinite(c0):
            c0 = 1e100
        candidates.append((c0, nonlocal_cid, int(local_n), bool(local_sym), np.asarray(local_x, dtype=float)))
        return nonlocal_cid + 1

    cid = add_candidate(best_h, best_n, False)
    if best_n % 2 == 0 and best_n >= 4:
        vpar = make_sym_v(best_h, best_n)
        cid = add_candidate(vpar, best_n, True)

        hcomp = make_feasible_full(1.0 - best_h, float(best_n) / 2.0)
        cid = add_candidate(hcomp, best_n, False)
        vcomp = make_sym_v(hcomp, best_n)
        cid = add_candidate(vcomp, best_n, True)

    if best_n % 2 != 0 or best_n < 64:
        n_even = 160 if best_n < 80 else (best_n // 2) * 2
        if n_even < 24:
            n_even = 160
        h_even = interpolate_full(best_h, n_even)
        cid = add_candidate(h_even, n_even, False)
        if n_even % 2 == 0:
            v_even = make_sym_v(h_even, n_even)
            cid = add_candidate(v_even, n_even, True)

    if best_n > 256:
        n_small = 192 if best_n <= 384 else 160
        if n_small % 2 != 0:
            n_small += 1
        h_small = interpolate_full(best_h, n_small)
        cid = add_candidate(h_small, n_small, False)
        v_small = make_sym_v(h_small, n_small)
        cid = add_candidate(v_small, n_small, True)

    candidates.sort()

    processed = set()
    rank = 0

    for c0, idx, n, sym, x0 in candidates:
        remaining = deadline - time.time()
        if remaining < 80.0:
            break
        key = (n, sym)
        if key in processed:
            continue
        if not (24 <= n <= 480):
            continue

        if rank == 0:
            alloc = min(420.0, 0.45 * remaining)
        elif rank < 3:
            alloc = min(240.0, 0.35 * remaining)
        else:
            alloc = min(110.0, 0.22 * remaining)
        alloc = max(25.0, alloc)

        processed.add(key)
        x_opt, c_opt = optimize_candidate(x0, n, sym, alloc)
        best_h, best_c, best_n = update_best_tuple(best_h, best_c, best_n, x_opt, n, sym)
        rank += 1
        if rank >= 6:
            break

    extra_ns = []
    base_n = 160 if best_n <= 0 else best_n
    for ntry in [base_n, base_n // 2, base_n * 2, 96, 128, 160, 192, 240, 320, 480]:
        if ntry % 2 == 0 and 24 <= ntry <= 480 and ntry not in extra_ns:
            extra_ns.append(ntry)
    extra_ns.sort()

    extra_count = 0
    for ntry in extra_ns:
        remaining = deadline - time.time()
        if remaining < 120.0 or extra_count >= 4:
            break

        if (ntry, True) not in processed:
            h_init = interpolate_full(best_h, ntry)
            v_init = make_sym_v(h_init, ntry)
            if ntry <= 200:
                alloc = min(180.0, 0.35 * remaining)
            else:
                alloc = min(140.0, 0.30 * remaining)
            x_opt, c_opt = optimize_candidate(v_init, ntry, True, alloc)
            best_h, best_c, best_n = update_best_tuple(best_h, best_c, best_n, x_opt, ntry, True)
            processed.add((ntry, True))
            extra_count += 1

        remaining = deadline - time.time()
        if ntry <= 220 and (ntry, False) not in processed and remaining > 140.0 and extra_count < 5:
            h_init = interpolate_full(best_h, ntry)
            alloc = min(140.0, 0.30 * remaining)
            x_opt, c_opt = optimize_candidate(h_init, ntry, False, alloc)
            best_h, best_c, best_n = update_best_tuple(best_h, best_c, best_n, x_opt, ntry, False)
            processed.add((ntry, False))
            extra_count += 1

    remaining = deadline - time.time()
    if remaining > 180.0 and best_n >= 24 and best_n <= 280:
        for kind in [0, 1, 2, 3]:
            remaining = deadline - time.time()
            if remaining < 80.0:
                break
            h0 = structured_full(best_n, kind)
            alloc = min(55.0, 0.25 * remaining)
            x_opt, c_opt = optimize_candidate(h0, best_n, False, alloc)
            best_h, best_c, best_n = update_best_tuple(best_h, best_c, best_n, x_opt, best_n, False)

        remaining = deadline - time.time()
        if remaining > 100.0:
            h0 = make_feasible_full(2.0 * np.random.random(best_n), float(best_n) / 2.0)
            alloc = min(70.0, 0.35 * remaining)
            x_opt, c_opt = optimize_candidate(h0, best_n, False, alloc)
            best_h, best_c, best_n = update_best_tuple(best_h, best_c, best_n, x_opt, best_n, False)

    remaining = deadline - time.time()
    if remaining > 60.0:
        x_full = make_feasible_full(best_h, float(best_n) / 2.0)
        alloc = min(180.0, 0.55 * remaining)
        x_opt, c_opt = optimize_candidate(x_full, best_n, False, alloc)
        best_h, best_c, best_n = update_best_tuple(best_h, best_c, best_n, x_opt, best_n, False)

    remaining = deadline - time.time()
    if best_n % 2 == 0 and remaining > 50.0:
        v_full = make_sym_v(best_h, best_n)
        alloc = min(140.0, 0.55 * remaining)
        x_opt, c_opt = optimize_candidate(v_full, best_n, True, alloc)
        best_h, best_c, best_n = update_best_tuple(best_h, best_c, best_n, x_opt, best_n, True)

    remaining = deadline - time.time()
    if best_n % 2 == 0 and remaining > 60.0:
        stop_bin = t0 + safe_budget - max(2.0, 0.05 * (t0 + safe_budget - t0))
        stop_bin = min(deadline, time.time() + min(80.0, 0.45 * remaining))
        hbin = binarize_full(best_h, best_n)
        hbin, cbin = binary_local_polish(hbin, best_n, stop_bin)
        best_h, best_c, best_n = update_best_tuple(best_h, best_c, best_n, hbin, best_n, False)

    remaining = deadline - time.time()
    if remaining > 30.0:
        x_full = make_feasible_full(best_h, float(best_n) / 2.0)
        x_opt, c_opt = pair_polish(x_full, best_n, False, deadline)
        best_h, best_c, best_n = update_best_tuple(best_h, best_c, best_n, x_opt, best_n, False)

    remaining = deadline - time.time()
    if best_n % 2 == 0 and remaining > 25.0:
        v_full = make_sym_v(best_h, best_n)
        x_opt, c_opt = pair_polish(v_full, best_n, True, deadline)
        best_h, best_c, best_n = update_best_tuple(best_h, best_c, best_n, x_opt, best_n, True)

    best_h = make_feasible_full(best_h, float(best_n) / 2.0)
    best_c = evaluate_full(best_h)

    return (best_h, float(best_c), int(best_n))
