import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def compute_conv(h, n):
    """Return the full correlation vector (before dx scaling)."""
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max(values, T):
    """Log-sum-exp approximation of max."""
    v = values - np.max(values)
    return np.max(values) + T * np.log(np.sum(np.exp(v / T)))

def smooth_obj(h, n, T):
    conv = compute_conv(h, n)
    dx = 2.0 / n
    return smooth_max(conv * dx, T)

def smooth_grad(h, n, T):
    """Gradient of smooth_max(correlate(h, 1-h) * dx, T)"""
    conv = compute_conv(h, n)
    dx = 2.0 / n
    vals = conv * dx
    vm = vals - np.max(vals)
    weights = np.exp(vm / T) / np.sum(np.exp(vm / T))
    
    # Now conv[m] = sum_i h[i] * (1-h[i+k]) where k = m - (n-1)
    # d(conv[m])/d(h[i]) = (1 - h[i+k]) + h[i] * (-dh[i+k]/dh[i])
    # For the i-th element:
    #   In term h[i]*(1-h[i+k]): derivative = (1-h[i+k])
    #   In term h[j]*(1-h[j+k]) where j+k=i, i.e., j=i-k: derivative = -h[j] = -h[i-k]
    # So d(conv[m])/d(h[i]) = (1-h[i+k]) - h[i-k] (where valid)
    
    n_full = 2*n - 1
    grad = np.zeros(n)
    for i in range(n):
        for m in range(n_full):
            k = m - (n - 1)  # shift
            # h[i] appears in conv[m] as:
            # - h[i]*(1-h[i+k]) if i+k is valid: i+k in [0,n-1]
            # - h[i-k]*(1-h[i]) if i-k is valid: i-k in [0,n-1] → h[i]*(1-h[i]) term? No...
            
            # Actually conv[m] = sum_j h[j]*(1-h[j+k]) for valid j
            # d(conv[m])/d(h[i]) involves:
            #   j=i: h[i]*(1-h[i+k]) → contrib (1-h[i+k]) if i+k valid
            #   j+k=i (j=i-k): h[i-k]*(1-h[i]) → contrib h[i-k]*(-1) if i-k valid
            
            contrib = 0.0
            if 0 <= i + k <= n - 1:
                contrib += (1.0 - h[i + k])
            if 0 <= i - k <= n - 1:
                contrib += -h[i - k]
            grad[i] += weights[m] * contrib * dx
    
    return grad

    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
