def smooth_c5_objective_and_grad(h, n, temp, min_weight=1e-5):
    dx = 2.0/n
    q_raw = np.correlate(h, 1.0-h, mode='full')
    q = q_raw * dx
    m = np.max(q)
    # numerical: m might be 0
    shifted = q - m
    # to avoid under/overflow
    w = np.exp(shifted / temp)
    s = np.sum(w)
    obj = m + temp * np.log(s)  # Wait logsumexp: m + temp*log(sum exp((q-m)/temp)). Yes.
    w /= s
    # weights effective = w (since derivative of logsumexp) actually grad = sum exp((q-m)/temp)/sum * grad(q). Yes.
    grad = np.zeros(n)
    # For each index where w > threshold
    idxs = np.where(w > min_weight)[0]
    for idx in idxs:
       mass = w[idx]
       mshift = idx - (n-1)
       if mshift >=0:
           mm = mshift
           L = n-mm
           if L>0:
              grad[mm:] += mass * (1.0 - h[:L]) * dx
              grad[:L] += mass * (-h[mm:]) * dx
       else:
           p = -mshift
           L = n-p
           if L>0:
              grad[:L] += mass * (1.0 - h[p:]) * dx
              grad[p:] += mass * (-h[:L]) * dx
    return obj, grad

_OPT_TEMP = 0.01
_OPT_TARGET = 0.0

_PEN_N = 0
_PEN_TEMP = 0.01
_PEN_RHO = 80.0

_DE_N = 0
_DE_TARGET = 0.0
_DE_RHO = 50.0
_DE_TEMP = 0.02


def project_constrained(y, target):
    y = np.asarray(y, dtype=float).ravel()
    n = int(y.size)
    if n == 0:
        return np.array([], dtype=float)

    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    target = float(target)

    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        z = np.clip(y - mid, 0.0, 1.0)
        s = float(np.sum(z))
        if s > target:
            lo = mid
        else:
            hi = mid

    z = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    s = float(np.sum(z))
    diff = target - s

    if abs(diff) > 1e-12:
        interior = (z > 1e-12) & (z < 1.0 - 1e-12)
        cnt = float(np.sum(interior))
        if cnt > 0.0:
            z[interior] += diff / cnt
        else:
            z += diff / float(n)
        z = np.clip(z, 0.0, 1.0)

    s = float(np.sum(z))
    diff = target - s
    if abs(diff) > 1e-13:
        if diff > 0.0:
            idxs = np.argsort(z)
        else:
            idxs = np.argsort(-z)
        left = diff
        for idx in idxs:
            if abs(left) < 1e-15:
                break
            if left > 0.0:
                room = 1.0 - z[idx]
                if room <= 0.0:
                    continue
                add = min(room, left)
            else:
                room = z[idx]
                if room <= 0.0:
                    continue
                add = max(-room, left)
            z[idx] += add
            left -= add
        if abs(left) > 1e-12:
            z += left / float(n)
            z = np.clip(z, 0.0, 1.0)

    return z


def exact_c5(h, n):
    h = np.asarray(h, dtype=float)
    if h.size != n:
        n = int(h.size)
    if n == 0:
        return float("inf")
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / float(n)
    conv = np.correlate(h, 1.0 - h, mode="full")
    val = float(np.max(conv)) * dx
    if not math.isfinite(val):
        return float("inf")
    return val


def smooth_c5_objective(h, n, temp, min_w=1e-12):
    h = np.asarray(h, dtype=float)
    if h.size != n:
        n = int(h.size)
    if n == 0:
        return float("inf")

    dx = 2.0 / float(n)
    raw = np.correlate(h, 1.0 - h, mode="full")
    q = raw * dx
    m = float(np.max(q))
    if not math.isfinite(m):
        return float("inf")

    if temp <= 1e-14:
        return m

    z = q - m
    e = np.exp(z / temp)
    s = float(np.sum(e))
    if s <= 0.0 or not math.isfinite(s):
        return m
    return float(m + temp * math.log(s))


def smooth_c5_and_grad(h, n, temp, min_w=1e-12):
    h = np.asarray(h, dtype=float)
    if h.size != n:
        n = int(h.size)
    if n == 0:
        return float("inf"), np.array([], dtype=float)

    dx = 2.0 / float(n)
    raw = np.correlate(h, 1.0 - h, mode="full")
    q = raw * dx
    m = float(np.max(q))

    if not math.isfinite(m):
        return float("inf"), np.zeros(n, dtype=float)

    if temp <= 1e-14:
        active = (q > m - 1e-8)
        cnt = int(np.sum(active))
        if cnt <= 0:
            active = np.zeros(len(q), dtype=bool)
            active[int(np.argmax(q))] = True
            cnt = 1
        w = np.zeros(len(q), dtype=float)
        w[active] = 1.0 / float(cnt)
        obj = m
    else:
        z = q - m
        e = np.exp(z / temp)
        s = float(np.sum(e))
        if s <= 0.0:
            w = np.zeros_like(q)
            w[int(np.argmax(q))] = 1.0
            obj = m
        else:
            obj = float(m + temp * math.log(s))
            w = e / s

    grad = np.zeros(n, dtype=float)
    idxs = np.where(w > min_w)[0]
    for idx in idxs:
        mass = float(w[idx])
        if mass <= 0.0:
            continue
        msh = int(idx) - (n - 1)
        if msh >= 0:
            mm = msh
            L = n - mm
            if L > 0:
                grad[mm:] += mass * dx * (1.0 - h[:L])
                grad[:L] += mass * dx * (-h[mm:])
        else:
            p = -msh
            L = n - p
            if L > 0:
                grad[:L] += mass * dx * (1.0 - h[p:])
                grad[p:] += mass * dx * (-h[:L])

    return obj, grad


def _opt_eq(x):
    return float(np.sum(x) - _OPT_TARGET)


def _opt_eq_jac(x):
    return np.ones(_OPT_N, dtype=float)


def _opt_f(x):
    return smooth_c5_objective(x, _OPT_N, _OPT_TEMP, 1e-12)


def _opt_jac(x):
    _, g = smooth_c5_and_grad(x, _OPT_N, _OPT_TEMP, 1e-12)
    return g


def _pen_f(x):
    F = smooth_c5_objective(x, _PEN_N, _PEN_TEMP, 1e-12)
    d = float(np.sum(x)) - 0.5 * float(_PEN_N)
    return float(F + _PEN_RHO * d * d)


def _pen_jac(x):
    _, g = smooth_c5_and_grad(x, _PEN_N, _PEN_TEMP, 1e-12)
    d = float(np.sum(x)) - 0.5 * float(_PEN_N)
    g = g + (2.0 * _PEN_RHO * d) * np.ones(_PEN_N, dtype=float)
    return g


def _de_f(x):
    F = smooth_c5_objective(x, _DE_N, _DE_TEMP, 1e-12)
    d = float(np.sum(x)) - _DE_TARGET
    return float(F + _DE_RHO * d * d)


def slsqp_smooth_polish(h0, n, temps, maxiter, time_limit_s):
    global _OPT_N, _OPT_TEMP, _OPT_TARGET

    t0 = time.time()
    n = int(n)
    if n <= 0 or time_limit_s <= 5.0:
        h0 = project_constrained(h0, 0.5 * n)
        return h0, exact_c5(h0, n)

    h = project_constrained(h0, 0.5 * n)
    best_h = h.copy()
    best_c = exact_c5(h, n)
    current_c = best_c

    if time_limit_s < 20.0:
        temps_use = temps[:1]
        maxiter_use = min(int(maxiter), 18)
    elif time_limit_s < 45.0:
        temps_use = temps[:2]
        maxiter_use = min(int(maxiter), 40)
    elif time_limit_s < 90.0:
        temps_use = temps[:3]
        maxiter_use = min(int(maxiter), 80)
    else:
        temps_use = temps
        maxiter_use = int(maxiter)

    if n > 130:
        maxiter_use = min(maxiter_use, 35)
    elif n > 100:
        maxiter_use = min(maxiter_use, 60)
    elif n > 80:
        maxiter_use = min(maxiter_use, 90)

    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _opt_eq, "jac": _opt_eq_jac}]

    for temp in temps_use:
        if time.time() - t0 >= max(1.0, time_limit_s - 1.0):
            break

        _OPT_N = n
        _OPT_TEMP = float(temp)
        _OPT_TARGET = 0.5 * float(n)

        f_start = smooth_c5_objective(h, n, float(temp), 1e-12)
        ok = False
        h2 = h
        res_fun = f_start
        try:
            res = minimize(
                _opt_f,
                h.copy(),
                jac=_opt_jac,
                bounds=bounds,
                constraints=cons,
                method="SLSQP",
                options={"maxiter": maxiter_use, "ftol": 1e-11},
            )
            h2 = project_constrained(res.x, 0.5 * n)
            ok = bool(res.success)
            res_fun = float(res.fun)
        except Exception:
            h2 = h

        c2 = exact_c5(h2, n)
        if math.isfinite(c2) and c2 < best_c - 1e-12:
            best_h = h2.copy()
            best_c = c2

        if ok and (c2 < current_c - 1e-12 or (abs(c2 - current_c) <= 1e-9 and res_fun < f_start - 1e-10)):
            h = h2
            current_c = c2

    h = project_constrained(h, 0.5 * n)
    c = exact_c5(h, n)
    if c < best_c - 1e-12:
        best_h = h.copy()
        best_c = c

    return best_h, best_c


def lbfgs_smooth_polish(h0, n, temps, time_limit_s):
    global _PEN_N, _PEN_TEMP

    t0 = time.time()
    n = int(n)
    if n <= 0 or time_limit_s <= 6.0:
        h0 = project_constrained(h0, 0.5 * n)
        return h0, exact_c5(h0, n)

    _PEN_RHO_local = 80.0
    h = project_constrained(h0, 0.5 * n)
    best_h = h.copy()
    best_c = exact_c5(h, n)

    maxiter = 120
    if n > 130:
        maxiter = 70
    if time_limit_s < 25.0:
        temps_use = temps[:1]
        maxiter = min(maxiter, 40)
    elif time_limit_s < 60.0:
        temps_use = temps[:2]
        maxiter = min(maxiter, 80)
    else:
        temps_use = temps

    bounds = [(0.0, 1.0)] * n

    for temp in temps_use:
        if time.time() - t0 >= max(1.0, time_limit_s - 1.0):
            break

        _PEN_N = n
        _PEN_TEMP = float(temp)

        f_start = smooth_c5_objective(h, n, float(temp), 1e-12)
        try:
            d = float(np.sum(h)) - 0.5 * float(n)
            f_start += _PEN_RHO_local * d * d
        except Exception:
            pass

        try:
            res = minimize(
                _pen_f,
                h.copy(),
                jac=_pen_jac,
                bounds=bounds,
                method="L-BFGS-B",
                options={"maxiter": maxiter, "ftol": 1e-9},
            )
            h2 = project_constrained(res.x, 0.5 * n)
        except Exception:
            h2 = h

        c2 = exact_c5(h2, n)
        if math.isfinite(c2) and c2 < best_c - 1e-12:
            best_h = h2.copy()
            best_c = c2

        try:
            f2 = _pen_f(h2)
            if f2 < f_start - 1e-8:
                h = h2
        except Exception:
            h = h2

    h = project_constrained(h, 0.5 * n)
    c = exact_c5(h, n)
    if c < best_c - 1e-12:
        best_h = h.copy()
        best_c = c

    return best_h, best_c


def projected_gradient_polish(h0, n, time_limit_s, temps=None):
    if temps is None:
        temps = [0.005, 0.003, 0.002, 0.001, 0.0005, 0.0002]

    t0 = time.time()
    n = int(n)
    if n <= 0 or time_limit_s <= 3.0:
        h0 = project_constrained(h0, 0.5 * n)
        return h0, exact_c5(h0, n)

    h = project_constrained(h0, 0.5 * n)
    best_h = h.copy()
    best_c = exact_c5(h, n)

    for temp in temps:
        if time.time() - t0 >= max(1.0, time_limit_s - 1.0):
            break

        alpha = 0.35
        stagnant = 0
        it = 0
        max_iter = 220

        while it < max_iter and time.time() - t0 < max(1.0, time_limit_s - 0.5):
            F, grad = smooth_c5_and_grad(h, n, float(temp), 1e-9)
            if not math.isfinite(F):
                break
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-14:
                break

            exact_current = exact_c5(h, n)
            accepted = False
            step = alpha

            for _ in range(24):
                h_new = project_constrained(h - step * grad, 0.5 * n)
                F_new = smooth_c5_objective(h_new, n, float(temp), 1e-12)
                improved_smooth = math.isfinite(F_new) and F_new < F - 1e-13
                if improved_smooth:
                    c_new = exact_c5(h_new, n)
                    accepted = True
                else:
                    c_new = exact_c5(h_new, n)
                    if math.isfinite(c_new) and c_new < exact_current - 1e-12:
                        accepted = True
                        F_new = c_new

                if accepted:
                    if math.isfinite(c_new) and c_new < best_c - 1e-12:
                        best_h = h_new.copy()
                        best_c = c_new
                    h = h_new
                    break

                step *= 0.5

            if accepted:
                alpha = min(alpha * 1.08, 5.0)
                stagnant = 0
            else:
                alpha *= 0.5
                stagnant += 1
                if stagnant > 8:
                    break
            it += 1

    h = project_constrained(h, 0.5 * n)
    c = exact_c5(h, n)
    if c < best_c - 1e-12:
        best_h = h.copy()
        best_c = c

    return best_h, best_c


def polish_continuous_variants(h0, n, rng, time_limit_s):
    n = int(n)
    t0 = time.time()
    if n <= 0 or time_limit_s <= 4.0:
        h0 = project_constrained(h0, 0.5 * n)
        return h0, exact_c5(h0, n)

    h0 = project_constrained(h0, 0.5 * n)
    best_h = h0.copy()
    best_c = exact_c5(h0, n)

    variants = [h0]

    for scale in (0.02, 0.05, 0.10):
        hp = h0 + scale * rng.normal(0.0, 1.0, n)
        hp = project_constrained(hp, 0.5 * n)
        variants.append(hp)

    hc = project_constrained(1.0 - h0, 0.5 * n)
    variants.append(hc)

    hr = project_constrained(h0[::-1].copy(), 0.5 * n)
    variants.append(hr)

    for tau in (0.08, 0.18):
        hs = 1.0 / (1.0 + np.exp(-(h0 - 0.5) / tau))
        hs = project_constrained(hs, 0.5 * n)
        variants.append(hs)

    if time_limit_s < 25.0:
        num_variants = 1
    elif time_limit_s < 55.0:
        num_variants = 3
    elif time_limit_s < 110.0:
        num_variants = 5
    else:
        num_variants = len(variants)

    num_variants = min(len(variants), max(1, int(num_variants)))
    per_time = max(6.0, (time_limit_s - 8.0) / float(num_variants))

    if n <= 85:
        temps = [0.04, 0.02, 0.01, 0.005, 0.002]
        maxiter = 210
    elif n <= 110:
        temps = [0.05, 0.025, 0.012, 0.006, 0.003]
        maxiter = 130
    else:
        temps = [0.06, 0.03, 0.015, 0.007, 0.003]
        maxiter = 75

    for hvar in variants[:num_variants]:
        rem = time_limit_s - (time.time() - t0)
        if rem < 8.0:
            break
        hv, cv = slsqp_smooth_polish(hvar, n, temps, maxiter, min(per_time, rem - 4.0))
        if cv < best_c - 1e-12:
            best_h = hv.copy()
            best_c = cv

    rem = time_limit_s - (time.time() - t0)
    if rem > 25.0:
        hl, cl = lbfgs_smooth_polish(best_h, n, [0.03, 0.015, 0.007, 0.003], min(45.0, rem - 8.0))
        if cl < best_c - 1e-12:
            best_h = hl.copy()
            best_c = cl

    rem = time_limit_s - (time.time() - t0)
    if rem > 12.0:
        hg, cg = projected_gradient_polish(best_h, n, max(5.0, rem - 2.0))
        if cg < best_c - 1e-12:
            best_h = hg.copy()
            best_c = cg

    best_h = project_constrained(best_h, 0.5 * best_n if False else 0.5 * int(best_h.size))
    best_c = exact_c5(best_h, int(best_h.size))
    return best_h, best_c


def interpolate_to(h_coarse, n_coarse, n_fine):
    h_coarse = np.asarray(h_coarse, dtype=float).ravel()
    if int(n_coarse) != int(h_coarse.size):
        n_coarse = int(h_coarse.size)
    n_fine = int(n_fine)
    if n_fine == n_coarse:
        return h_coarse.copy()
    x = np.linspace(0.0, 2.0, n_coarse, endpoint=False)
    xf = np.linspace(0.0, 2.0, n_fine, endpoint=False)
    return np.interp(xf, x, h_coarse)


def topk_binary(scores, n):
    n = int(n)
    h = np.zeros(n, dtype=float)
    target = n // 2
    if target <= 0:
        return h
    if target >= n:
        h.fill(1.0)
        return h

    scores = np.asarray(scores, dtype=float).reshape(n)
    scores = np.nan_to_num(scores, nan=0.0, posinf=1.0, neginf=-1.0)
    if np.all(scores == scores[0]):
        scores = scores + np.linspace(0.0, 1e-12, n)

    idx = np.argsort(scores)[-target:]
    h[idx] = 1.0
    return h


def random_int_composition(total, parts, rng):
    total = int(total)
    parts = int(parts)
    if total <= 0:
        return []
    if parts <= 1:
        return [total]
    if parts > total:
        parts = total
    cuts = np.sort(rng.choice(np.arange(1, total), parts - 1, replace=False))
    out = []
    prev = 0
    for c in cuts:
        c = int(c)
        out.append(c - prev)
        prev = c
    out.append(total - prev)
    return out


def binary_blocks(n, k, rng, start=None):
    n = int(n)
    total = n // 2
    if total <= 0:
        return np.zeros(n, dtype=float)

    k = int(k)
    if k <= 0:
        k = 2
    if k > total:
        k = total

    one_len = random_int_composition(total, k, rng)
    zero_len = random_int_composition(total, k, rng)

    if start is None:
        start_one = bool(rng.random() < 0.5)
    else:
        start_one = bool(start)

    vals = []
    if start_one:
        for a, b in zip(one_len, zero_len):
            vals.extend([1.0] * int(a))
            vals.extend([0.0] * int(b))
    else:
        for a, b in zip(one_len, zero_len):
            vals.extend([0.0] * int(b))
            vals.extend([1.0] * int(a))

    if len(vals) != n:
        return topk_binary(rng.uniform(0.0, 1.0, n), n)
    return np.asarray(vals, dtype=float)


def binary_sinus(n, f, rng):
    n = int(n)
    x = np.arange(n, dtype=float)
    ph1 = float(rng.uniform(0.0, 2.0 * math.pi))
    ph2 = float(rng.uniform(0.0, 2.0 * math.pi))
    score = np.sin(2.0 * math.pi * float(f) * x / float(n) + ph1)
    score += 0.5 * np.sin(2.0 * math.pi * (2.0 * float(f)) * x / float(n) + ph2)
    score += 0.15 * rng.normal(0.0, 1.0, n)
    return topk_binary(score, n)


def make_binary_candidates(n, rng, ref1, ref2=None):
    n = int(n)
    cands = []

    for _ in range(3):
        cands.append(topk_binary(rng.uniform(0.0, 1.0, n), n))

    refs = [ref1]
    if ref2 is not None:
        refs.append(ref2)

    for ref in refs:
        if ref is None:
            continue
        ref = np.asarray(ref, dtype=float).ravel()
        if ref.size == 0:
            continue
        hp = interpolate_to(ref, len(ref), n)
        hp = project_constrained(hp, 0.5 * n)
        for scale in (0.0, 0.02, 0.05):
            sc = hp.copy()
            if scale > 0.0:
                sc = sc + scale * rng.normal(0.0, 1.0, n)
            cands.append(topk_binary(sc, n))
            cands.append(topk_binary(-sc, n))

    for k in (2, 3, 4, 6, 9, 14, 21):
        cands.append(binary_blocks(n, k, rng, True))
        cands.append(binary_blocks(n, k, rng, False))

    for f in (1, 2, 3, 5, 8):
        cands.append(binary_sinus(n, f, rng))

    return cands


def binary_eval(h, n):
    h = np.asarray(h, dtype=float)
    conv = np.correlate(h, 1.0 - h, mode="full")
    M = int(round(float(np.max(conv))))
    return M, conv


def random_active_pair(h, n, idx, rng):
    m = int(idx) - (n - 1)
    if m >= 0:
        L = n - m
        if L <= 0:
            return None
        for _ in range(18):
            i = int(rng.randint(L))
            if h[i] < 0.5 and h[i + m] > 0.5:
                return i, i + m
        cand = np.nonzero((h[:L] < 0.5) & (h[m:] > 0.5))[0]
        if len(cand) > 0:
            i = int(cand[int(rng.randint(len(cand)))])
            return i, i + m
        return None
    else:
        p = -m
        L = n - p
        if L <= 0:
            return None
        for _ in range(18):
            i = int(rng.randint(L))
            if h[i] > 0.5 and h[i + p] < 0.5:
                return i, i + p
        cand = np.nonzero((h[:L] > 0.5) & (h[p:] < 0.5))[0]
        if len(cand) > 0:
            i = int(cand[int(rng.randint(len(cand)))])
            return i, i + p
        return None


def binary_local_search(h, n, time_limit_s, rng):
    n = int(n)
    t0 = time.time()
    h = np.asarray(h, dtype=float).copy()
    if h.size != n:
        h = project_constrained(h, 0.5 * n)

    target = n // 2
    if int(round(float(np.sum(h)))) != target:
        h = topk_binary(h, n)

    M, conv = binary_eval(h, n)
    best_h = h.copy()
    best_C = M * (2.0 / float(n))
    sec = float(np.sum(conv * conv))
    best_score = M * 1000000.0 + sec
    no_improve = 0
    move = 0
    tabu = {}

    while time.time() - t0 < time_limit_s:
        move += 1

        use_reverse = (move % 32 == 0) or (no_improve > 9000 and rng.random() < 0.06)

        if use_reverse:
            L = int(rng.randint(2, max(2, n // 3)))
            a = int(rng.randint(0, max(1, n - L)))
            b = a + L
            old = h[a:b].copy()
            h[a:b] = old[::-1]

            Mnew, convnew = binary_eval(h, n)
            secnew = float(np.sum(convnew * convnew))
            scorenew = Mnew * 1000000.0 + secnew
            accept = False
            if Mnew < M:
                accept = True
            elif Mnew == M and secnew < sec:
                accept = True
            elif Mnew == M and abs(secnew - sec) < 1e-6 and rng.random() < 0.03:
                accept = True
            elif Mnew == M + 1 and no_improve > 6000 and rng.random() < 0.01:
                accept = True

            if accept:
                conv = convnew
                M = Mnew
                sec = secnew
                if scorenew < best_score:
                    best_score = scorenew
                    best_h = h.copy()
                    best_C = Mnew * (2.0 / float(n))
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[a:b] = old
                no_improve += 1
            continue

        pair = None
        guided_prob = 0.55 if no_improve < 3500 else 0.38
        if rng.random() < guided_prob:
            threshold = M - 0.5
            active = np.where(conv >= threshold)[0]
            if len(active) > 0:
                idx = int(rng.choice(active))
                pair = random_active_pair(h, n, idx, rng)

        if pair is None:
            ones = np.nonzero(h > 0.5)[0]
            zeros = np.nonzero(h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
        else:
            i, j = pair
            if i == j or h[i] == h[j]:
                ones = np.nonzero(h > 0.5)[0]
                zeros = np.nonzero(h < 0.5)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = int(rng.choice(ones))
                j = int(rng.choice(zeros))
                pair = None

        a, b = (i, j) if i < j else (j, i)
        if tabu.get((a, b), -10000) > move - 80:
            no_improve += 1
            continue

        h[i], h[j] = h[j], h[i]
        Mnew, convnew = binary_eval(h, n)
        secnew = float(np.sum(convnew * convnew))
        scorenew = Mnew * 1000000.0 + secnew

        accept = False
        if Mnew < M:
            accept = True
        elif Mnew == M:
            if secnew < sec:
                accept = True
            elif abs(secnew - sec) < 1e-6 and rng.random() < 0.10:
                accept = True
            elif no_improve > 5000 and rng.random() < 0.20:
                accept = True
        elif Mnew == M + 1 and no_improve > 9000 and rng.random() < 0.01:
            accept = True

        if accept:
            conv = convnew
            M = Mnew
            sec = secnew
            tabu[(a, b)] = move
            if scorenew < best_score:
                best_score = scorenew
                best_h = h.copy()
                best_C = Mnew * (2.0 / float(n))
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = h[j], h[i]
            no_improve += 1

        if move % 5000 == 0 and len(tabu) > 3500:
            tabu.clear()
        if no_improve > 32000:
            break

    return best_h, best_C


def binary_explore(n, time_limit_s, rng, ref1, ref2=None):
    n = int(n)
    t0 = time.time()

    try:
        cands = make_binary_candidates(n, rng, ref1, ref2)
    except Exception:
        cands = [topk_binary(rng.uniform(0.0, 1.0, n), n)]

    scored = []
    for h in cands:
        try:
            h = np.asarray(h, dtype=float)
            if h.size != n:
                h = topk_binary(h, n)
            M, conv = binary_eval(h, n)
            c = M * (2.0 / float(n))
            sec = float(np.sum(conv * conv))
            scored.append([c, sec, h])
        except Exception:
            continue

    if not scored:
        h = topk_binary(rng.uniform(0.0, 1.0, n), n)
        M, _ = binary_eval(h, n)
        return h, M * (2.0 / float(n))

    arr_c = np.array([item[0] for item in scored], dtype=float)
    arr_s = np.array([item[1] for item in scored], dtype=float)
    order = np.lexsort((arr_s, arr_c))
    scored_sorted = [scored[i] for i in order]

    best_c = scored_sorted[0][0]
    best_h = scored_sorted[0][2].copy()

    remaining = time_limit_s - (time.time() - t0)
    if remaining > 6.0:
        num_local = min(len(scored_sorted), max(1, int(remaining / 12.0)))
        num_local = min(num_local, 4)
        per_time = max(3.0, remaining / float(max(1, num_local)))

        for item in scored_sorted[:num_local]:
            rem = time_limit_s - (time.time() - t0)
            if rem < 4.0:
                break
            try:
                h2, c2 = binary_local_search(item[2], n, min(per_time, rem - 2.0), rng)
            except Exception:
                continue
            if math.isfinite(c2) and c2 < best_c - 1e-12:
                best_c = c2
                best_h = h2.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    M, _ = binary_eval(best_h, n)
    best_h = topk_binary(best_h, n)
    best_c = M * (2.0 / float(n))
    return best_h, best_c


def de_search(n, time_limit_s, rng):
    if differential_evolution is None:
        return None, float("inf")
    n = int(n)
    if n <= 0 or n > 120 or time_limit_s < 12.0:
        return None, float("inf")

    global _DE_N, _DE_TARGET, _DE_RHO, _DE_TEMP
    _DE_N = n
    _DE_TARGET = 0.5 * float(n)
    _DE_RHO = 50.0
    _DE_TEMP = 0.02

    h_seed = project_constrained(rng.uniform(0.0, 1.0, n), _DE_TARGET)
    bounds = [(0.0, 1.0)] * n
    seed = int(rng.randint(1, 2**31 - 1))

    res = None
    try:
        res = differential_evolution(
            _de_f,
            bounds,
            maxiter=22,
            tol=1e-5,
            popsize=9,
            seed=seed,
            init="latinhypercube",
            x0=h_seed,
            disp=False,
        )
    except TypeError:
        try:
            res = differential_evolution(
                _de_f,
                bounds,
                maxiter=22,
                tol=1e-5,
                popsize=9,
                seed=seed,
                init="latinhypercube",
                disp=False,
            )
        except Exception:
            return None, float("inf")
    except Exception:
        return None, float("inf")

    if res is None:
        return None, float("inf")

    h = project_constrained(res.x, _DE_TARGET)
    c = exact_c5(h, n)
    if not math.isfinite(c):
        return None, float("inf")
    return h, c


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    budget = float(budget_s)
    rng = np.random.RandomState(seed)

    parent = None
    try:
        parent = np.array(initial_h_values, dtype=float).ravel()
    except Exception:
        parent = None

    if parent is None or parent.size == 0:
        parent = binary_blocks(84, 6, rng, True)

    parent = project_constrained(parent, 0.5 * parent.size)
    best_h = parent.copy()
    best_n = int(parent.size)
    best_c = exact_c5(best_h, best_n)

    if (not math.isfinite(best_c)) or best_c > 1.0:
        parent = binary_blocks(84, 7, rng, False)
        parent = project_constrained(parent, 0.5 * parent.size)
        best_h = parent.copy()
        best_n = int(parent.size)
        best_c = exact_c5(best_h, best_n)

    rem = budget - (time.time() - t_start)
    if rem > 80.0:
        limit = min(120.0, max(12.0, rem - 60.0))
        try:
            h_c, c_c = polish_continuous_variants(best_h, best_n, rng, limit)
            if c_c < best_c - 1e-12:
                best_h = h_c.copy()
                best_c = c_c
                best_n = int(h_c.size)
        except Exception:
            pass

    n_candidates = []
    if best_n % 2 == 0:
        n_candidates.append(int(best_n))

    ordered_ns = []
    ordered_ns.extend([84, 86, 90, 100, 74, 78, 80, 94, 104, 112, 120])
    ordered_ns.extend([83, 85, 87, 89, 91, 95])

    for n in ordered_ns:
        if n not in n_candidates and 20 <= n <= 160:
            n_candidates.append(n)

    for idx, n in enumerate(n_candidates):
        rem = budget - (time.time() - t_start)
        if rem < 170.0:
            break

        n = int(n)
        if n <= 0:
            continue

        if idx < 5:
            binary_time = min(22.0, max(5.0, (rem - 180.0) * 0.18))
            cont_time = min(72.0, max(12.0, (rem - 150.0) * 0.45))
        else:
            binary_time = min(10.0, max(3.0, (rem - 160.0) * 0.08))
            cont_time = min(36.0, max(6.0, (rem - 140.0) * 0.25))

        if binary_time < 2.0:
            binary_time = 2.0
        if cont_time < 4.0:
            cont_time = 4.0

        if n % 2 == 0:
            h_b = None
            c_b = float("inf")
            try:
                h_b, c_b = binary_explore(n, binary_time, rng, parent, best_h)
            except Exception:
                pass

            if h_b is not None and math.isfinite(c_b):
                if c_b < best_c - 1e-12:
                    best_h = h_b.copy()
                    best_c = c_b
                    best_n = n

                if c_b < best_c + 0.02 and cont_time > 10.0:
                    try:
                        h_p, c_p = polish_continuous_variants(h_b, n, rng, cont_time)
                        if c_p < best_c - 1e-12:
                            best_h = h_p.copy()
                            best_c = c_p
                            best_n = int(h_p.size)
                    except Exception:
                        pass

        else:
            if cont_time > 18.0 and idx < 14:
                h_seed = interpolate_to(best_h, best_n, n)
                h_seed = project_constrained(h_seed, 0.5 * n)
                try:
                    h_p, c_p = polish_continuous_variants(h_seed, n, rng, cont_time)
                    if c_p < best_c - 1e-12:
                        best_h = h_p.copy()
                        best_c = c_p
                        best_n = int(h_p.size)
                except Exception:
                    pass

    rem = budget - (time.time() - t_start)
    if rem > 140.0:
        n2 = ((best_n // 2) + 3) * 2
        if n2 <= best_n:
            n2 = best_n + 2
        if n2 > 170:
            n2 = best_n
        if n2 != best_n:
            h_up = interpolate_to(best_h, best_n, n2)
            h_up = project_constrained(h_up, 0.5 * n2)
            try:
                h_p, c_p = polish_continuous_variants(h_up, n2, rng, min(100.0, rem - 100.0))
                if c_p < best_c - 1e-12:
                    best_h = h_p.copy()
                    best_c = c_p
                    best_n = int(h_p.size)
            except Exception:
                pass

        rem = budget - (time.time() - t_start)
        n2o = best_n + 5
        if rem > 80.0 and n2o != best_n and 20 <= n2o <= 150:
            h_up = interpolate_to(best_h, best_n, n2o)
            h_up = project_constrained(h_up, 0.5 * n2o)
            try:
                h_p, c_p = polish_continuous_variants(h_up, n2o, rng, min(55.0, rem - 70.0))
                if c_p < best_c - 1e-12:
                    best_h = h_p.copy()
                    best_c = c_p
                    best_n = int(h_p.size)
            except Exception:
                pass

    rem = budget - (time.time() - t_start)
    if rem > 150.0 and differential_evolution is not None:
        n_de = 100
        try:
            h_de, c_de = de_search(n_de, 45.0, rng)
        except Exception:
            h_de, c_de = None, float("inf")
        if h_de is not None and math.isfinite(c_de) and c_de < best_c + 0.03:
            try:
                h_p, c_p = polish_continuous_variants(h_de, n_de, rng, min(90.0, rem - 90.0))
                if c_p < best_c - 1e-12:
                    best_h = h_p.copy()
                    best_c = c_p
                    best_n = int(h_p.size)
            except Exception:
                pass

    rem = budget - (time.time() - t_start)
    if rem > 15.0:
        try:
            h_f, c_f = projected_gradient_polish(best_h, best_n, max(5.0, rem - 5.0))
            if c_f < best_c - 1e-12:
                best_h = h_f.copy()
                best_c = c_f
                best_n = int(h_f.size)
        except Exception:
            pass

    best_h = project_constrained(best_h, 0.5 * int(best_n))
    best_c = exact_c5(best_h, best_n)

    if (not math.isfinite(best_c)) or best_c > 1.0:
        fallback = binary_blocks(84, 6, rng, True)
        best_h = project_constrained(fallback, 0.5 * 84)
        best_n = 84
        best_c = exact_c5(best_h, best_n)

    if not math.isfinite(best_c):
        best_h = np.full(max(int(best_n), 2), 0.5, dtype=float)
        best_h = project_constrained(best_h, 0.5 * best_n)
        best_c = exact_c5(best_h, best_n)

    return best_h, float(best_c), int(best_n)
