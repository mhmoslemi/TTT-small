   - For each k, define f_k(h) = ∫ h(x)(1-h(x+k)) dx

        if h.size > n:
            h = h[:n]
        else:
            tmp = np.zeros(n)
            tmp[:h.size] = h
            h = tmp
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        sm = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if sm < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if np.any(mask):
                cnt = float(np.sum(mask))
                inc = min(diff / cnt, 1.0 - h[mask].min())
                if inc > 0:
                    h[mask] += inc
        else:
            mask = h > 1e-12
            if np.any(mask):
                cnt = float(np.sum(mask))
                dec = min(-diff / cnt, h[mask].min())
                if dec > 0:
                    h[mask] -= dec
    return h

def compute_scores(h):
    n = h.size
    ac = np.correlate(h, h, mode="full")[n-1:n-1+n]
    prefix = np.zeros(n + 1)
    np.cumsum(h, out=prefix[1:])
    total = prefix[n]
    scores = np.zeros(2 * n - 1)
    scores[0] = total - ac[0]
    for d in range(1, n):
        scores[2*d - 1] = prefix[n-d] - ac[d]
        scores[2*d] = total - prefix[d] - ac[d]
    return scores

def score_max_val(h):
    scores = compute_scores(h)
    return float(np.max(scores))

def objective_fn(x, alpha):
    check_deadline()
    n = x.size
    h = np.clip(x, 0.0, 1.0)
    scores = compute_scores(h)
    m = float(np.max(scores))
    if alpha < 1e-14:
        val = m
    else:
        w = np.exp((scores - m) / alpha)
        A = float(np.sum(w))
        if not math.isfinite(A) or A <= 0:
            val = m
        else:
            val = m + alpha * math.log(A)
    return val * (2.0 / n)

def gradient_fn(x, alpha):
    check_deadline()
    n = x.size
    h = np.clip(x, 0.0, 1.0)
    scores = compute_scores(h)
    m = float(np.max(scores))
    grad = np.zeros(n)
    if alpha < 1e-14:
        tol = max(1e-8, 1e-7 * m)
        active = np.flatnonzero(scores >= m - tol)
        if len(active) == 0:
            active = np.array([int(np.argmax(scores))])
        coef = 1.0 / float(len(active))
        for idx in active:
            _add_one_grad(grad, int(idx), coef, h, n)
    else:
        w = np.exp((scores - m) / alpha)
        A = float(np.sum(w))
        if not math.isfinite(A) or A <= 0:
            return grad
        thresh = 1e-10
        for idx in range(len(w)):
            c = w[idx] / A
            if c >= thresh:
                _add_one_grad(grad, idx, c, h, n)
    grad *= 2.0 / n
    return grad

def _add_one_grad(grad, idx, coef, h, n):
    if coef == 0.0:
        return
    if idx == 0:
        grad += coef * (1.0 - 2.0 * h)
        return
    if idx % 2 == 1:
        d = (idx + 1) // 2
        m = n - d
        if m <= 0:
            return
        grad[:m] += coef * (1.0 - h[d:])
        grad[d:] += coef * (-h[:m])
    else:
        d = idx // 2
        m = n - d
        if m <= 0:
            return
        grad[:m] += coef * (-h[d:])
        grad[d:] += coef * (1.0 - h[:m])

def projected_gradient_line(h, grad, alpha, max_iter=200):
    """Backtracking line search with projection."""
    n = h.size
    f0 = objective_fn(h, alpha)
    step = 1.0
    best_h = h.copy()
    best_f = f0
    for _ in range(50):
        h_new = project_feasible(h - step * grad, n)
        f_new = objective_fn(h_new, alpha)
        if f_new < f0 - 1e-4 * step * np.dot(grad, h - h_new):
            return h_new, f_new
        step *= 0.5
        if step < 1e-8:
            break
    # If line search fails, try smaller fixed steps
    for s in [0.1, 0.05, 0.01, 0.005, 0.001]:
        h_new = project_feasible(h - s * grad, n)
        f_new = objective_fn(h_new, alpha)
        if f_new < best_f:
            best_h = h_new
            best_f = f_new
    return best_h, best_f

def run_pg(h0, n, alpha0, alpha_end, max_iter, deadline):
    """Projected gradient with cooling alpha."""
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iter):
        check_deadline()
        if time.time() > deadline:
            break
        # Linearly interpolate alpha
        t = float(it) / max(1, max_iter - 1)
        alpha = alpha0 * (alpha_end / alpha0) ** t
        g = gradient_fn(h, alpha)
        if not np.all(np.isfinite(g)) or np.linalg.norm(g) < 1e-12:
            break
        h, f = projected_gradient_line(h, g, alpha)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if it % 100 == 0:
            pass  # Could log here
    return best_h, best_c

def slsqp_run(h0, n, time_limit, deadline):
    """Run SLSQP optimization."""
    d0 = time.time()
    if d0 + 10 > deadline:
        return None
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 0.5*n, 
                    'jac': lambda x: np.ones(n)/n}]
    
    alpha = max(0.1, 0.001 * n)
    maxiter = max(30, min(200, int(5000.0 / n)))
    
    res = None
    try:
        _RUN_DEADLINE = min(deadline, d0 + time_limit)
        res = minimize(
            lambda x: objective_fn(x, alpha),
            h0,
            jac=lambda x: gradient_fn(x, alpha),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False}
        )
    except BudgetError:
        res = None
    except Exception:
        res = None
    finally:
        _RUN_DEADLINE = deadline
    
    if res is not None and hasattr(res, 'x') and res.x is not None:
        h1 = project_feasible(res.x, n)
        c1 = compute_c5(h1)
        if math.isfinite(c1):
            return h1, c1
    return None

def upsample(h0, n):
    n0 = h0.size
    if n == n0:
        return h0.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x, x0, h0.astype(float))
