def sigmoid_stable(y):
    y = np.asarray(y, dtype=float)
    out = np.empty_like(y)
    pos = y >= 0.0
    out[pos] = 1.0 / (1.0 + np.exp(-y[pos]))

    out[~pos] = ey / (1.0 + ey)

def add_weighted_score_grad(gh, idx, coef, h, n):
    if coef == 0.0 or n == 0:
        return
    if idx == n - 1:
        gh += coef * (1.0 - 2.0 * h)
        return

        m = n - d
        if m <= 0:
            return
        gh[d:] += coef * (1.0 - h[:m])

def soft_objective_h(h, alpha_s):
    h = np.asarray(h, dtype=float)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)

        return 1e6, np.zeros(0, dtype=float)

    raw = np.correlate(h, 1.0 - h, mode="full")
    scores = np.maximum(raw, 0.0)
    m = float(np.max(scores))
    if not math.isfinite(m):
        return 1e6, np.zeros(n, dtype=float)

    if alpha_s <= 1e-8:
        tol = max(1e-9, 1e-8 * max(1.0, m))
        active = np.flatnonzero(scores >= m - tol)
        if active.size == 0:
            active = np.asarray([int(np.argmax(scores))], dtype=int)
        cnt = float(active.size)
        weights = np.zeros_like(scores)
        weights[active] = 1.0 / cnt
        val = m
    else:
        x = (scores - m) / float(alpha_s)
        x = np.clip(x, -745.0, 0.0)
        ex = np.exp(x)
        A = float(np.sum(ex))

                active = np.asarray([int(np.argmax(scores))], dtype=int)
            weights = np.zeros_like(scores)
            weights[active] = 1.0 / float(active.size)
            val = m
        else:
            weights = ex / A
            val = m + float(alpha_s) * math.log(A)

    gh = np.zeros(n, dtype=float)
    n_scores = len(scores)
    for i in range(n_scores):
        p = float(weights[i])
        if p > 1e-11:
            add_weighted_score_grad(gh, i, p, h, n)

    if not math.isfinite(val) or not np.all(np.isfinite(gh)):
        return 1e6, np.zeros(n, dtype=float)

    if not np.all(np.isfinite(out)):

    return out

        return h.copy()
    if n == 2:
        return np.array([0.75 * h[0] + 0.25 * h[1],
                         0.25 * h[0] + 0.75 * h[1]], dtype=float)
    out = h.copy()
    out[0] = 0.75 * h[0] + 0.25 * h[1]
    out[-1] = 0.25 * h[-2] + 0.75 * h[-1]
    out[1:-1] = 0.25 * h[:-2] + 0.5 * h[1:-1] + 0.25 * h[2:]
    return out


def sharpen_h(h):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return h
    if float(np.max(h)) - float(np.min(h)) < 1e-12:
        return np.full(h.shape, 0.5, dtype=float)
    return np.clip(0.5 + 1.6 * (h - 0.5), 0.0, 1.0)


def random_smooth_start(n, rng):
    n = int(n)
    x = np.arange(n, dtype=float)
    h = np.full(n, 0.5, dtype=float)
    for f in range(1, 7):
        amp = rng.uniform(0.04, 0.25) / math.sqrt(float(f + 1))
        phase = rng.uniform(0.0, 2.0 * math.pi)
        h += amp * np.sin(2.0 * math.pi * float(f) * x / float(max(1, n)) + phase)
    h += rng.normal(0.0, 0.02, n)
    return as_feasible(np.clip(h, 0.0, 1.0), n)


def initial_construct(n, rng):
    n = int(n)
    x = np.arange(n, dtype=float)
    best_h = as_feasible(np.full(n, 0.5), n)
    best_c = compute_c5(best_h)
    for f in (1, 2, 3, 4, 5):
        for ph in (0.0, 0.5 * math.pi):
            h = 0.5 + 0.45 * np.sin(2.0 * math.pi * float(f) * x / float(max(1, n)) + ph)
            h = as_feasible(np.clip(h, 0.0, 1.0), n)
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h
    for _ in range(3):
        h = random_smooth_start(n, rng)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h
    return best_h


def get_parent():
    try:
        p = np.asarray(initial_h_values, dtype=float)
        if p.ndim != 1 or p.size < 2:
            return None
        return p.copy()
    except Exception:
        return None


def is_same_candidate(a, b):
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    if a.size != b.size:
        return False
    return bool(np.allclose(a, b, rtol=1e-7, atol=1e-7))


def add_candidate(starts, h, n):
    if h is None:
        return False
    try:
        h2 = as_feasible(np.asarray(h, dtype=float), n)
        c2 = compute_c5(h2)
    except Exception:
        return False
    if not math.isfinite(c2):
        return False
    for hh in starts:
        if is_same_candidate(hh, h2):
            return False
    starts.append(h2)
    return True


def make_starts_for_n(n, parent, state, rng, limit=4):
    starts = []
    n = int(n)

    if parent is not None:
        ph = h_at_n_from(parent, n)
        add_candidate(starts, ph, n)
        if ph is not None:
            add_candidate(starts, smooth1(ph), n)
            add_candidate(starts, sharpen_h(ph), n)

    if state is not None and state.h is not None and state.n > 0:
        if state.n == n:
            add_candidate(starts, state.h, n)
        else:
            add_candidate(starts, upsample_linear(state.h, n), n)

    for _ in range(2):
        add_candidate(starts, random_smooth_start(n, rng), n)

    if len(starts) == 0:
        add_candidate(starts, np.full(n, 0.5, dtype=float), n)

    while len(starts) > limit:
        starts.pop()
    return starts


def run_logit_anneal(h0, n, state, time_limit):
    n = int(n)
    if h0 is None or n <= 1 or time_limit <= 20.0:
        return
    h = as_feasible(h0, n)
    c = compute_c5(h)
    state.update(h, c, n)
    deadline = time.time() + max(0.0, float(time_limit))

    base = max(0.05, 0.0025 * float(n))
    alphas = [base, 0.35 * base, 0.10 * base, 0.025 * base]
    maxiter = 120 if n <= 100 else (75 if n <= 200 else 45)
    stage_time = max(1.0, float(time_limit) / (len(alphas) + 0.5))

    for alpha in alphas:
        if time.time() > deadline - max(5.0, 0.4 * stage_time):
            break
        h = as_feasible(h, n)
        c = compute_c5(h)
        state.update(h, c, n)

        anchor, inds = get_anchor_inds(h)
        sig = adapt_sig_from_h(h, anchor)
        v0 = v_from_h(h, sig, anchor, inds)
        bounds = [(-ZMAX, ZMAX)] * int(inds.size)

        rem = max(0.0, deadline - time.time())
        it = int(max(20, min(maxiter + 20, int(maxiter * max(0.4, rem / stage_time)))))

        try:
            res = minimize(
                lbfgs_fun,
                v0,
                args=(sig, alpha, anchor, inds),
                method="L-BFGS-B",
                jac=lbfgs_grad,
                bounds=bounds,
                options={
                    "maxiter": it,
                    "ftol": 1e-9,
                    "gtol": 1e-6,
                    "maxcor": 10,
                    "maxls": 20,
                },
            )
        except Exception:
            res = None

        if res is not None and getattr(res, "x", None) is not None:
            try:
                h_new = h_from_v(res.x, sig, anchor, inds)
                h_new = as_feasible(h_new, n)
                c_new = compute_c5(h_new)
                if math.isfinite(c_new) and c_new < c - 1e-13:
                    h = h_new
                    c = c_new
                    state.update(h, c, n)
                elif math.isfinite(c_new):
                    h = h_new
                    c = c_new
            except Exception:
                pass

        if c <= SAFE_C5:
            return


def run_slsqp_polish(h0, n, state, time_limit):
    n = int(n)
    if h0 is None or n <= 1 or time_limit <= 12.0:
        return
    h = as_feasible(h0, n)
    c = compute_c5(h)
    state.update(h, c, n)
    deadline = time.time() + max(0.0, float(time_limit))

    base = max(0.05, 0.0025 * float(n))
    alphas = [0.18 * base, 0.05 * base, 0.012 * base, 1e-9]
    maxiter = 90 if n <= 100 else (55 if n <= 200 else 35)
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": equation_constraint, "jac": equation_jacobian}]

    for alpha in alphas:
        if time.time() > deadline - 5.0:
            break
        it = max(20, maxiter // (3 if alpha < 1e-8 else 1))
        try:
            res = minimize(
                slsqp_fun,
                h,
                args=(alpha,),
                method="SLSQP",
                jac=slsqp_grad,
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": it, "ftol": 1e-10, "disp": 0},
            )
        except Exception:
            res = None
        if res is not None and getattr(res, "x", None) is not None:
            h_new = as_feasible(res.x, n)
            c_new = compute_c5(h_new)
            if math.isfinite(c_new) and c_new < c - 1e-13:
                h = h_new
                c = c_new
                state.update(h, c, n)
            elif math.isfinite(c_new):
                h = h_new
                c = c_new


def lp_one_step(h0, n, state, max_change):
    n = int(n)
    h = as_feasible(h0, n)
    c = compute_c5(h)
    state.update(h, c, n)
    if n == 0 or max_change <= 0.0:
        return h, c, False

    raw = np.correlate(h, 1.0 - h, mode="full")
    scores = np.maximum(raw, 0.0)
    c_score = float(np.max(scores))
    if not math.isfinite(c_score):
        return h, c, False

    if n <= 300:
        use_idx = np.arange(len(scores), dtype=int)
    else:
        use_idx = np.flatnonzero(scores >= c_score - 1.0)
        if use_idx.size < 24 or use_idx.size > 140:
            order = np.argsort(scores)
            want = min(120, len(scores))
            use_idx = order[len(scores) - want + 1:]
        if use_idx.size == 0:
            use_idx = np.asarray([int(np.argmax(scores))], dtype=int)

    rows = []
    bubs = []
    for idx in use_idx:
        idx = int(idx)
        gh = np.zeros(n, dtype=float)
        add_weighted_score_grad(gh, idx, 1.0, h, n)
        row = np.empty(n + 1, dtype=float)
        row[:n] = gh
        row[n] = -1.0
        rows.append(row)
        bubs.append(-float(scores[idx]))

    if len(rows) == 0:
        return h, c, False

    A_ub = np.asarray(rows, dtype=float)
    b_ub = np.asarray(bubs, dtype=float)

    A_eq = np.ones(n + 1, dtype=float)
    A_eq[n] = 0.0
    b_eq = 0.0

    p_low = np.maximum(-h, -float(max_change))
    p_high = np.minimum(1.0 - h, float(max_change))
    t_low = c_score - (float(n) * float(max_change) + 1.0)
    t_high = c_score
    bounds = []
    for i in range(n):
        bounds.append((float(p_low[i]), float(p_high[i])))
        bounds.append((float(t_low), float(t_high)))

    cf = np.zeros(n + 1, dtype=float)
    cf[n] = 1.0

    res = None
    try:
        res = linprog(
            cf,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
        )
    except Exception:
        try:
            res = linprog(
                cf,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
            )
        except Exception:
            res = None

    if res is None or getattr(res, "status", 1) != 0 or getattr(res, "x", None) is None:
        return h, c, False

    p = np.asarray(res.x[:n], dtype=float)
    if not np.all(np.isfinite(p)):
        return h, c, False
    p = p - float(np.mean(p))

    best_h = h
    best_c = c
    alphas = [1.0, 0.8, 0.65, 0.5, 0.4, 0.33, 0.26, 0.2, 0.15, 0.11, 0.08, 0.06, 0.04, 0.025, 0.015, 0.008]
    for a in alphas:
        ht = np.clip(h + float(a) * p, 0.0, 1.0)
        ht = as_feasible(ht, n)
        ct = compute_c5(ht)
        if math.isfinite(ct) and ct < best_c - 1e-13:
            best_h = ht
            best_c = ct

    if best_c < c - 1e-12:
        state.update(best_h, best_c, n)
        return best_h, best_c, True
    return h, c, False


def lp_polish(h0, n, state, time_limit):
    n = int(n)
    if h0 is None or n <= 1 or time_limit <= 12.0:
        return
    h = as_feasible(h0, n)
    c = compute_c5(h)
    state.update(h, c, n)
    deadline = time.time() + max(0.0, float(time_limit))
    changes = [0.08, 0.05, 0.035, 0.025, 0.018, 0.012, 0.008, 0.005, 0.003, 0.0018, 0.001]
    idx = 0
    stuck = 0

    while time.time() < deadline - 2.0:
        if n > 300 and idx > 4:
            break
        mc = changes[idx] if idx < len(changes) else changes[-1] * 0.65
        if n > 220:
            mc = min(mc, 0.035)

        h, c, ok = lp_one_step(h, n, state, mc)
        if ok:
            stuck = 0
            if c <= SAFE_C5:
                break
            if idx > 4:
                idx -= 1
        else:
            stuck += 1
            if stuck >= 4:
                break
            idx += 1
            if idx >= len(changes):
                break


def random_local_polish(h0, n, state, time_limit, rng):
    n = int(n)
    if h0 is None or n <= 1 or time_limit <= 15.0 or rng is None:
        return
    h = as_feasible(h0, n)
    c = compute_c5(h)
    state.update(h, c, n)
    deadline = time.time() + max(0.0, float(time_limit))
    scales = [0.05, 0.035, 0.022, 0.013, 0.008]

    for scale in scales:
        for _ in range(3):
            if time.time() > deadline - 3.0:
                break
            try:
                pert = rng.normal(0.0, scale, n)
                hp = as_feasible(np.clip(h + pert, 0.0, 1.0), n)
                cp = compute_c5(hp)
                if math.isfinite(cp) and cp < c - 1e-12:
                    h = hp
                    c = cp
                    state.update(h, c, n)

                if c > 0.379:
                    if time.time() > deadline - 6.0:
                        break
                    h, c, _ = lp_one_step(h, n, state, 0.012)
                    if time.time() > deadline - 6.0:
                        break
                    h, c, _ = lp_one_step(h, n, state, 0.006)
            except Exception:
                pass


def finish_solution(state, fallback=None):
    if state is not None and state.h is not None and state.n > 0:
        h = np.array(state.h, dtype=float)
        n = int(state.n)
        c_state = float(state.c)
    elif fallback is not None:
        h = np.asarray(fallback, dtype=float)
        n = int(h.size)
        c_state = 1e6
    else:
        n = 70
        h = np.full(n, 0.5, dtype=float)
        c_state = compute_c5(h)

    h2 = None
    c2 = 1e6
    try:
        h2 = as_feasible(h, n)
        c2 = compute_c5(h2)
    except Exception:
        pass

    if (
        h2 is not None
        and math.isfinite(c2)
        and is_valid_candidate(h2)
        and (not math.isfinite(c_state) or c2 < c_state - 1e-12)
    ):
        return h2, float(c2), n

    if h2 is not None and math.isfinite(c2):
        return h2, float(c2), n

    if math.isfinite(c_state):
        return h, float(c_state), n

    c2 = compute_c5(h)
    return h, float(c2), n


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0.0:
        budget_s = 30.0

    reserve = 25.0 if budget_s > 90.0 else max(2.0, 0.08 * budget_s)
    deadline = t0 + max(1.0, budget_s - reserve)
    rng = np.random.default_rng(seed)

    parent = get_parent()
    if parent is not None and parent.size > 0:
        n0 = int(parent.size)
        h0 = as_feasible(parent, n0)
        c0 = compute_c5(h0)
    else:
        n0 = 70
        h0 = initial_construct(n0, rng)
        c0 = compute_c5(h0)

    state = OptimizerState(h0, c0, n0)

    if time.time() < deadline - 120.0:
        starts0 = make_starts_for_n(n0, parent, state, rng, limit=5)
        rem = deadline - time.time()
        if rem > 650.0:
            k = 3
        elif rem > 350.0:
            k = 2
        else:
            k = 1
        k = min(k, max(1, len(starts0)))
        per = min(170.0, max(25.0, float(rem - 220.0) / max(1, k)))
        for i in range(k):
            if time.time() > deadline - 90.0:
                break
            lt = min(per, max(25.0, deadline - time.time() - 90.0))
            if lt < 25.0:
                break
            try:
                run_logit_anneal(starts0[i], n0, state, lt)
            except Exception:
                pass

    if state.n == n0 and time.time() < deadline - 90.0:
        lt = min(90.0, max(20.0, deadline - time.time() - 90.0))
        if lt > 18.0:
            try:
                run_slsqp_polish(state.h, n0, state, lt)
            except Exception:
                pass

    if state.n == n0 and time.time() < deadline - 90.0:
        lt = min(210.0, max(20.0, deadline - time.time() - 90.0))
        if lt > 18.0:
            try:
                lp_polish(state.h, n0, state, lt)
            except Exception:
                pass

    if state.n == n0 and time.time() < deadline - 80.0:
        lt = min(80.0, max(20.0, deadline - time.time() - 80.0))
        if lt > 18.0:
            try:
                random_local_polish(state.h, n0, state, lt, rng)
            except Exception:
                pass

    if state.n == n0 and time.time() < deadline - 70.0:
        lt = min(150.0, max(20.0, deadline - time.time() - 70.0))
        if lt > 18.0:
            try:
                lp_polish(state.h, n0, state, lt)
            except Exception:
                pass

    if time.time() < deadline - 270.0:
        n1 = 2 * n0
        if 4 <= n1 <= 300:
            starts1 = make_starts_for_n(n1, parent, state, rng, limit=4)
            rem1 = deadline - time.time()
            if rem1 > 800.0:
                k1 = 2
            elif rem1 > 450.0:
                k1 = 2
            else:
                k1 = 1
            k1 = min(k1, max(1, len(starts1)))
            per1 = min(150.0, max(25.0, float(rem1 - 260.0) / max(1, k1)))
            for i in range(k1):
                if time.time() > deadline - 110.0:
                    break
                lt1 = min(per1, max(25.0, deadline - time.time() - 110.0))
                if lt1 < 25.0:
                    break
                try:
                    run_logit_anneal(starts1[i], n1, state, lt1)
                except Exception:
                    pass

        if state.n == n1 and time.time() < deadline - 80.0:
            lt1 = min(75.0, max(18.0, deadline - time.time() - 80.0))
            if lt1 > 18.0:
                try:
                    run_slsqp_polish(state.h, n1, state, lt1)
                except Exception:
                    pass

        if state.n == n1 and time.time() < deadline - 80.0:
            lt1 = min(170.0, max(18.0, deadline - time.time() - 80.0))
            if lt1 > 18.0:
                try:
                    lp_polish(state.h, n1, state, lt1)
                except Exception:
                    pass

        if state.n == n1 and time.time() < deadline - 70.0:
            lt1 = min(70.0, max(18.0, deadline - time.time() - 70.0))
            if lt1 > 18.0:
                try:
                    random_local_polish(state.h, n1, state, lt1, rng)
                except Exception:
                    pass

        if state.n == n1 and time.time() < deadline - 60.0:
            lt1 = min(130.0, max(18.0, deadline - time.time() - 60.0))
            if lt1 > 18.0:
                try:
                    lp_polish(state.h, n1, state, lt1)
                except Exception:
                    pass

    if time.time() < deadline - 60.0 and state.n is not None and state.n > 0:
        lt = min(110.0, max(15.0, deadline - time.time() - 45.0))
        if lt > 15.0:
            try:
                lp_polish(state.h, state.n, state, lt)
            except Exception:
                pass

    return finish_solution(state, fallback=h0)
