import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_grad_max(h, top_k=3):
    """Vectorized gradient of sum of top_k correlation values * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(-corr)[:top_k]
    grad = np.zeros(n)
    for k_idx in idx:
        lag = k_idx - (n - 1)
        if lag >= 0:
            m = lag
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] -= h[:n-m]
        else:
            m = -lag
            grad[m:] += (1.0 - h[:n-m])
            grad[:n-m] -= h[m:]
    return grad * dx

def project_feasible(y, n):
    """Project y onto {h: 0<=h_i<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Bisection for lambda
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_trial = np.clip(y - mid, 0.0, 1.0)
        s = h_trial.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Distribute among non-boundary elements
        free = np.where((h > 1e-10) & (h < 1 - 1e-10))[0]
        if len(free) > 0:
            h[free] += diff / len(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        for i in idx:
            h[i] = np.clip(h[i] + per, 0.0, 1.0)
    return h

def upsample_to(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def smooth_max_obj(x, n, dx, tau):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return float((m + tau * np.log(np.sum(np.exp((corr - m) / tau)))) * dx)

def proj_grad_method(h, n, max_iter=5000, top_k=3, verbose=False):
    """Projected gradient method for min max_k f_k(h)"""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.1
    dx = 2.0 / n
    
    for it in range(max_iter):
        if it % 100 == 0 and verbose:
            c5 = compute_c5(h)
            print(f"  PGM iter {it}: C5={c5:.6f}")
        
        # Compute gradient
        grad = compute_grad_max(h, top_k=top_k)
        
        # Adaptive step size
        # Try to choose alpha such that ||h - alpha*grad|| is feasible and improves
        h_trial = h - alpha * grad
        h_trial = project_feasible(h_trial, n)
        c5_trial = compute_c5(h_trial)
        
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            h = h_trial
            alpha = min(alpha * 1.05, 0.5)  # slightly increase
        else:
            alpha *= 0.8
            if alpha < 0.01:
                break
        
        # Check convergence
        if it > 100 and abs(compute_c5(h) - best_c5) < 1e-10:
            pass  # still continue but check
    
    return best_h, best_c5

                options={'maxiter': 2000, 'ftol': 1e-12, 'disp': False}
            )

            c5_new = compute_c5(h_new)

        except Exception:

import numpy as np
from scipy.optimize import minimize
import functools
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_grad_max(h, top_k=3):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(-corr)[:top_k]
    grad = np.zeros(n)
    for k_idx in idx:
        lag = k_idx - (n - 1)
        if lag >= 0:
            m = lag
            if m > 0:
                grad[:n - m] += (1.0 - h[m:])
                grad[m:] -= h[:n - m]
        else:
            m = -lag
            if m > 0:
                grad[m:] += (1.0 - h[:n - m])
                grad[:n - m] -= h[m:]
    return grad * dx


def project_feasible(y, n):
    target = n / 2.0
    lo = -1.5
    hi = 2.5
    for _ in range(70):
        mid = (lo + hi) / 2.0
        h_trial = np.clip(y - mid, 0.0, 1.0)
        s = h_trial.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(y - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if len(free) > 0:
            h[free] += diff / len(free)
            h = np.clip(h, 0.0, 1.0)
    return h


def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        for i in idx:
            h[i] = np.clip(h[i] + per, 0.0, 1.0)
    return h


def upsample_to(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def smooth_max_obj(x, n, dx, tau):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_args = np.exp((corr - m) / tau)
    log_sum = tau * np.log(np.sum(exp_args))
    return float((m + log_sum) * dx)


def eq_con_std(x, n):
    return np.array([x.sum() - n / 2.0])


def eq_jac_std(x, n):
    return np.ones((1, n))


def proj_grad_method(h, n, max_iter=5000, top_k=3, alpha_init=0.15):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha_init
    stuck_count = 0
    for it in range(max_iter):
        grad = compute_grad_max(h, top_k=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        h_trial = h - alpha * grad
        h_trial = project_feasible(h_trial, n)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5 - 1e-14:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            h = h_trial
            alpha = min(alpha * 1.08, 0.6)
            stuck_count = 0
        else:
            alpha *= 0.75
            stuck_count += 1
            if alpha < 0.005 or stuck_count > 200:
                break
    return best_h, best_c5


def slsqp_optimize(h0, n, maxiter=2000, ftol=1e-12):
    dx = 2.0 / n
    bounds_list = [(0.0, 1.0)] * n
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    tau_list = [0.15, 0.06, 0.02, 0.01, 0.004]
    for tau in tau_list:
        obj_p = functools.partial(smooth_max_obj, n=n, dx=dx, tau=tau)
        con_p = functools.partial(eq_con_std, n=n)
        jac_p = functools.partial(eq_jac_std, n=n)
        try:
            res = minimize(
                fun=obj_p, x0=best_h, method='SLSQP',
                bounds=bounds_list,
                constraints=[{'type': 'eq', 'fun': con_p, 'jac': jac_p}],
                options={'maxiter': maxiter, 'ftol': ftol, 'disp': False}
            )
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = fix_sum(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c5

        n_trials = 15 if n_best <= 360 else 8
        best_h, best_c5 = perturbation_search(best_h, n_best, rng, n_trials=n_trials, noise_scale=0.03)
        print(f"Phase 6 done: C5={best_c5:.6f}, t={time.time()-t0:.1f}s")

    # Phase 7: Final PGM polish
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.97:
        n_fin = len(best_h)
        h_fin, c5_fin = proj_grad_method(best_h, n_fin, max_iter=3000, top_k=2, alpha_init=0.08)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin
        print(f"Phase 7 done: C5={best_c5:.6f}, t={time.time()-t0:.1f}s")

    # Final verification and cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
