def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def compute_all_overlaps(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv * dx

def smooth_max_and_grad(h, n_points, tau):
    n = n_points
    dx = 2.0 / n
    f = compute_all_overlaps(h, n)  # length 2n-1
    max_f = np.max(f)
    logits = (f - max_f) / tau
    exp_logits = np.exp(logits)
    sum_exp = np.sum(exp_logits)
    weights = exp_logits / sum_exp
    val = max_f + tau * np.log(sum_exp)
    
    # Gradient
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        k = m - (n - 1)  # shift
        wm = weights[m] * dx
        if wm < 1e-14:
            continue
        if k == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif k > 0:
            if k < n:
                grad[0:n-k] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:n-k])
        else:
            p = -k
            if p < n:
                grad[p:n] += wm * (1.0 - h[0:n-p])
                grad[0:n-p] += wm * (-h[p:n])
    
    return val, grad

        if np.sum(free) > 0:
            h[free] += (target - s) / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

    lr = lr_init
    v = np.zeros(n) if momentum > 0 else None
    
    for it in range(n_iter):
        val, grad = smooth_max_and_grad(h_cur, n, tau)
        if v is not None:

            h_new = h_cur + v
        else:

        h_cur = h_new

        return smooth_max_value_only(x, n, tau)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx


def compute_all_overlaps(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv * dx


def smooth_max_value_only(h, n_points, tau):
    n = n_points
    f = compute_all_overlaps(h, n)
    max_f = np.max(f)
    logits = (f - max_f) / tau
    return max_f + tau * np.log(np.sum(np.exp(logits)))


def smooth_max_and_grad(h, n_points, tau):
    n = n_points
    dx = 2.0 / n
    f = compute_all_overlaps(h, n)
    max_f = np.max(f)
    logits = (f - max_f) / tau
    exp_logits = np.exp(logits)
    sum_exp = np.sum(exp_logits)
    weights = exp_logits / sum_exp
    val = max_f + tau * np.log(sum_exp)

    grad = np.zeros(n)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        wm = weights[m] * dx
        if wm < 1e-14:
            continue
        if k == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif k > 0:
            if k < n:
                grad[0:n - k] += wm * (1.0 - h[k:n])
                grad[k:n] += wm * (-h[0:n - k])
        else:
            p = -k
            if p < n:
                grad[p:n] += wm * (1.0 - h[0:n - p])
                grad[0:n - p] += wm * (-h[p:n])

    return val, grad

    target = n / 2.0

                break
            add = diff / cnt
            h[free] = np.minimum(h[free] + add, 1.0)
        else:
            free = h > 0.0
            cnt = np.sum(free)

                break
            add = diff / cnt
            h[free] = np.maximum(h[free] + add, 0.0)

        cnt = np.sum(free)
        if cnt > 0:
            h[free] += (target - s) / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

    n = n_points
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_val = compute_c5(h_cur, n)
    lr = lr_init

    for it in range(n_iter):

            best_val = hard_val

        h_cur = h_new
        lr *= lr_decay
