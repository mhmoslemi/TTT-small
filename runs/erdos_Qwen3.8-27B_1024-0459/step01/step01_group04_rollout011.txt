        while time.time() < _run_deadline:
            _check_time()
            obj, grad = soft_obj_grad(h, alpha)
            v = momentum * v - lr * grad
            h_new = h + v
            h_new = as_feasible(h_new, n)
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
                h_new = as_feasible(h - lr * grad * 0.5, n)
                c_try = compute_c5(h_new)
                if c_try < compute_c5(h):
                    h = h_new
                else:
                    h = as_feasible(h - lr * grad * 0.25, n)
            h = h_new
            if no_improve > 40:
                lr *= 0.7
                v *= 0.5
                no_improve = 0
                if lr < 1e-6:
                    break
            if alpha > 0.05:
                alpha *= 0.985
            elif alpha > 0.01:
                alpha *= 0.995
            else:
                alpha *= 0.999
            if alpha < 1e-6:
                alpha = 1e-6
    except _timeout_error:
        pass
    except Exception:
        pass
    finally:
        _run_deadline = old_dl
    c_final = compute_c5(h)
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c


def slsqp_polish(h0, n, time_limit, alpha=0.01):
    global _run_deadline
    x0 = as_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": eq_fun, "jac": eq_jac}]
    old_dl = _run_deadline
    _run_deadline = min(time_limit, _run_deadline) if _run_deadline is not None else time_limit
    try:
        res = minimize(
            slsqp_obj, x0, args=(alpha,),
            jac=slsqp_grad, method="SLSQP",
            bounds=bounds, constraints=constraints,
            options={"maxiter": 300, "ftol": 1e-10, "disp": 0}
        )
        if res is not None and res.x is not None:
            h = as_feasible(res.x, n)
            c = compute_c5(h)
            return h, c
    except _timeout_error:
        pass
    except Exception:
        pass
    finally:
        _run_deadline = old_dl
    return x0, compute_c5(x0)


def binary_near_opt(h, n):
    """Try rounding to binary and do local swaps."""
    h_f = as_feasible(h, n)
    target = n // 2
    order = np.argsort(h_f)[::-1]
    h_bin = np.zeros(n, dtype=float)
    h_bin[order[:target]] = 1.0
    c_bin = compute_c5(h_bin)
    scores = get_scores(h_bin)
    cur_max = float(scores.max())
    # Local swap improvement
    improved = True
    max_iter = 2000
    it = 0
    rng = np.random.default_rng(12345)
    while improved and it < max_iter:
        it += 1
        improved = False
        ones = np.flatnonzero(h_bin == 1)
        zeros = np.flatnonzero(h_bin == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Try random swaps
        for trial in range(5):
            i = int(ones[rng.integers(len(ones))])
            j = int(zeros[rng.integers(len(zeros))])
            h_bin[i], h_bin[j] = 0.0, 1.0
            new_max = float(get_scores(h_bin).max())
            ones = np.flatnonzero(h_bin == 1)
            zeros = np.flatnonzero(h_bin == 0)
            if new_max < cur_max - 1e-12:
                cur_max = new_max
                improved = True
            else:
                h_bin[i], h_bin[j] = 1.0, 0.0
    return h_bin, compute_c5(h_bin)


def upsample(parent, n):
    n0 = parent.size
    if n == n0:
        return parent.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x, x0, parent).astype(float)


def downsample(h, n):
    n0 = h.size
    if n == n0:
        return h.copy()
    if n0 % n == 0:
        r = n0 // n
        return h.reshape(n, r).mean(axis=1)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x, x0, h).astype(float)


def get_parent():
    try:
        p = np.asarray(initial_h_values, dtype=float)
        if p.ndim != 1 and p.size >= 2:
            return p.copy()
        return None
    except Exception:
        return None


def run(seed=42, budget_s=1300, **kwargs):
    global _run_deadline
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0:
        budget_s = 30.0
    reserve = 15.0
    deadline = t0 + max(1.0, budget_s - reserve)
    _run_deadline = deadline
    rng = np.random.default_rng(seed)

    parent = get_parent()
    if parent is not None and parent.size >= 10:
        n0 = parent.size
        h0 = parent.copy()
    else:
        n0 = 70
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h0 = 0.5 + 0.3 * np.cos(np.pi * x) + 0.1 * np.cos(2 * np.pi * x + 0.5)
        h0 = as_feasible(h0, n0)

    # State: best solution found
    best_h = as_feasible(h0, n0)
    best_c = compute_c5(best_h)
    best_n = n0

    def update_state(h, c, n):
        nonlocal best_h, best_c, best_n
        if c < best_c - 1e-14:
            best_h = as_feasible(h, n)
            best_c = float(compute_c5(best_h))
            best_n = int(n)

    # Phase 1: Projected gradient at native resolution
    rem = deadline - time.time()
    if rem > 30:
        t1 = time.time() + min(rem * 0.25, 120.0)
        h_pg, c_pg = projected_gradient(h0, n0, t1, initial_alpha=0.8)
        update_state(h_pg, c_pg, n0)

    # Phase 2: SLSQP polish at native resolution
    rem = deadline - time.time()
    if rem > 20:
        t2 = time.time() + min(rem * 0.3, 180.0)
        h_sp, c_sp = slsqp_polish(best_h, best_n, t2, alpha=0.02)
        update_state(h_sp, c_sp, best_n)

    # Phase 3: More projected gradient with smaller alpha
    rem = deadline - time.time()
    if rem > 30:
        t3 = time.time() + min(rem * 0.2, 120.0)
        h_pg2, c_pg2 = projected_gradient(best_h, best_n, t3, initial_alpha=0.05)
        update_state(h_pg2, c_pg2, best_n)

    # Phase 4: SLSQP with very small alpha (near-exact max)
    rem = deadline - time.time()
    if rem > 15:
        t4 = time.time() + min(rem * 0.25, 150.0)
        h_sp2, c_sp2 = slsqp_polish(best_h, best_n, t4, alpha=0.001)
        update_state(h_sp2, c_sp2, best_n)

    # Phase 5: Try nearby resolutions
    rem = deadline - time.time()
    for mult in [6.0/5.0, 7.0/5.0, 4.0/5.0, 5.0/7.0]:
        if rem < 40:
            break
        nn = int(round(n0 * mult / 2.0)) * 2
        if nn < 20 or nn > 200 or nn == n0:
            continue
        rem = deadline - time.time()
        if rem < 40:
            break
        h_up = as_feasible(upsample(best_h, nn), nn)
        c_up = compute_c5(h_up)
        update_state(h_up, c_up, nn)
        # Optimize at this resolution
        t_opt = time.time() + min(rem * 0.3, 100.0)
        rem = deadline - time.time()
        if rem > 30:
            h_opt, c_opt = projected_gradient(h_up, nn, t_opt, initial_alpha=0.3)
            update_state(h_opt, c_opt, nn)
            rem = deadline - time.time()
            if rem > 20:
                h_pol, c_pol = slsqp_polish(h_opt, nn, time.time() + min(rem*0.4, 80.0), alpha=0.01)
                update_state(h_pol, c_pol, nn)
        rem = deadline - time.time()

    # Phase 6: Binary check at native resolution
    rem = deadline - time.time()
    if rem > 10 and best_n % 2 == 0:
        h_bin, c_bin = binary_near_opt(best_h, best_n)
        if c_bin < best_c:
            update_state(h_bin, c_bin, best_n)

    # Phase 7: Final SLSQP polish
    rem = deadline - time.time()
    if rem > 10:
        t_final = time.time() + min(rem * 0.7, 100.0)
        h_f, c_f = slsqp_polish(best_h, best_n, t_final, alpha=0.005)
        update_state(h_f, c_f, best_n)

    # Final feasibility check
    best_h = as_feasible(best_h, best_n)
    best_c = float(compute_c5(best_h))
    _run_deadline = None
    return best_h, best_c, int(best_n)
