import numpy as np
from scipy.optimize import minimize
import time


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h, n):
    c = get_corr(h)
    return float(np.max(c) * (2.0 / n))


def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(2000):
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-14
        else:
            mask = h < 1.0 - 1e-14
        if not mask.any():
            break
        h[mask] -= diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    # Exact final correction
    diff = h.sum() - target
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.where(h > 1e-10)[0]
        else:
            idx = np.where(h < 1.0 - 1e-10)[0]
        if len(idx) > 0:
            h[idx[0]] -= diff
            h = np.clip(h, 0.0, 1.0)
    return h


def softmax_obj(h, n, beta):
    c = get_corr(h) * (2.0 / n)
    m = c.max()
    return float((m + np.log(np.exp(c - m).sum())) / beta)


def softmax_grad(h, n, beta):
    dx = 2.0 / n
    corr = get_corr(h)
    c = corr * dx
    scaled = beta * c
    m_val = scaled.max()
    w = np.exp(scaled - m_val)
    w /= w.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-16:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            grad[:n - lag] += wk * (1.0 - h[lag:n])
            grad[lag:] -= wk * h[:n - lag]
        else:
            m2 = -lag
            grad[m2:] += wk * (1.0 - h[:n - m2])
            grad[:n - m2] -= wk * h[m2:]
    return grad * dx


def hardmax_obj(h, n):
    c = get_corr(h) * (2.0 / n)
    return float(c.max())


def hardmax_grad(h, n):
    dx = 2.0 / n
    corr = get_corr(h)
    k_max = np.argmax(corr)
    lag = k_max - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        grad[:n - lag] += (1.0 - h[lag:n])
        grad[lag:] -= h[:n - lag]
    else:
        m2 = -lag
        grad[m2:] += (1.0 - h[:n - m2])
        grad[:n - m2] -= h[m2:]
    return grad * dx


def multimax_obj(h, n, m_top):
    c = get_corr(h) * (2.0 / n)
    c_sorted = np.sort(c)[::-1]
    top = c_sorted[:m_top]
    return float(np.max(top))


def subgradient_polish(h, n, n_steps, base_step, seed):
    rng = np.random.RandomState(seed)
    h = h.copy()
    for step in range(n_steps):
        frac = step / max(n_steps - 1, 1)
        lr = base_step * (1.0 - frac) ** 1.5
        if lr < 1e-12:
            break
        corr = get_corr(h)
        c = corr * (2.0 / n)
        # Use top-5 weights for a steeper direction
        c_sorted, c_idx = np.sort(c)[::-1], np.argsort(c)[::-1]
        m_top = min(5, len(c))
        weights = np.zeros(len(c))
        for j in range(m_top):
            weights[c_idx[j]] = (0.8 ** j)
        weights /= weights.sum()
        # Compute weighted gradient
        grad = np.zeros(n)
        for k_idx in range(2 * n - 1):
            wk = weights[k_idx]
            if wk < 1e-10:
                continue
            lag = k_idx - (n - 1)
            if lag >= 0:
                grad[:n - lag] += wk * (1.0 - h[lag:n])
                grad[lag:] -= wk * h[:n - lag]
            else:
                m2 = -lag
                grad[m2:] += wk * (1.0 - h[:n - m2])
                grad[:n - m2] -= wk * h[m2:]
        h = h - lr * grad
        h = project_feasible(h, n)
    return h


def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return project_feasible(h_new, n_new)


def sum_eq_fun(h, n):
    return np.sum(h) - n / 2.0


def sum_eq_jac(h, n):
    return np.ones(n)


def slsqp_softmax(h0, n, beta, maxiter, t0, budget_s, margin):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n,)}]
    if time.time() - t0 > budget_s - margin:
        return h0
    try:
        res = minimize(
            fun=softmax_obj,
            x0=h0,
            jac=softmax_grad,
            args=(n, beta),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return project_feasible(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        return h0


def slsqp_hardmax(h0, n, maxiter, t0, budget_s, margin):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n,)}]
    if time.time() - t0 > budget_s - margin:
        return h0
    try:
        res = minimize(
            fun=hardmax_obj,
            x0=h0,
            jac=hardmax_grad,
            args=(n,),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return project_feasible(np.clip(res.x, 0.0, 1.0), n)
    except Exception:
        return h0


def coordinate_pair_sweep(h, n, n_sweeps, t0, budget_s, margin):
    h = h.copy()
    best_c5 = compute_c5(h, n)
    for sweep in range(n_sweeps):
        if time.time() - t0 > budget_s - margin:
            break
        for i in range(0, n - 1, 2):
            j = (i + 1) % n
            h0_i, h0_j = h[i], h[j]
            best_delta = 0.0
            for delta in np.linspace(-0.05, 0.05, 21):
                if delta == 0:
                    continue
                ni, nj = h0_i + delta, h0_j - delta
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5_new = compute_c5(h, n)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_delta = delta
                h[i], h[j] = h0_i, h0_j
            h[i], h[j] = h0_i + best_delta, h0_j - best_delta
    return h, best_c5
