def project_feasible(h, n):

        h[mask] -= diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)

def subgradient_at(h, n, k_active):
    """Compute subgradient of f_{k_active} at h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    lag = k_active  # lag in [-（n-1), ..., n-1]
    # r[lag] = sum_i h[i]*(1-h[i+lag]) for valid i
    # dr[lag]/dh[j] = (1-h[j+lag])*I[j+lag valid] - h[j-lag]*I[j-lag valid]
    for j in range(n):
        coeff = 0.0
        jp = j + lag
        jm = j - lag
        if 0 <= jp < n:
            coeff += (1.0 - h[jp])
        if 0 <= jm < n:
            coeff -= h[jm]
        grad[j] = coeff * dx
    return grad

def subgradient_hard(h, n):
    """Subgradient of max_k f_k at h."""
    c = c5_values(h, n)
    k_max = np.argmax(c)
    lag = k_max - (n - 1)
    return subgradient_at(h, n, lag)

def bundle_subgradient(h, n, top_m=5):
    """Subgradient using convex combination of top-m active constraints."""
    c = c5_values(h, n)
    top_idx = np.argsort(c)[-top_m:]
    weights = c[top_idx]
    weights = np.exp(weights - np.max(weights))
    weights /= weights.sum()
    grad = np.zeros(n)

def subgradient_at(h, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag >= 0:
        # For j in [0, n-1-lag]: contributes (1-h[j+lag])
        grad[:n-lag] += (1.0 - h[lag:n])
        # For j in [lag, n-1]: contributes -h[j-lag]
        grad[lag:] -= h[:n-lag]
    else:
        l = -lag
        # For j in [l, n-1]: contributes (1-h[j-l])
        grad[l:] += (1.0 - h[:n-l])
        # For j in [0, n-1-l]: contributes -h[j+l]
        grad[:n-l] -= h[l:n]
    return grad * dx

    return c * (2.0 / n)

def compute_c5(h, n):
    return float(np.max(c5_values(h, n)))

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:

        adj = diff / mask.sum()
        h[mask] -= adj
        h = np.clip(h, 0.0, 1.0)

def subgradient_single(h, n, lag):
    """Subgradient of f_{lag} where lag = k - (n-1) for the k-th correlation entry."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[:m] += (1.0 - h[lag:])
            grad[lag:] -= h[:m]
    else:
        m = n + lag
        if m > 0:
            grad[-lag:] += (1.0 - h[:m])
            grad[:m] -= h[-lag:]
    return grad * dx

def bundle_gradient(h, n, top_m=10):

    for wi, ki in zip(w, idx):
        lag = ki - (n - 1)
        grad += wi * subgradient_single(h, n, lag)
    return grad

def nesterov_subgradient(h, n, t0, iters, lr, momentum=0.9, top_m=5, noise_every=500, noise_scale=0.05):
    """Nesterov-accelerated subgradient method."""
    h_curr = h.copy()
    h_nest = h.copy()
    v = np.zeros(n)
    best_local = compute_c5(h_curr, n)
    best_h = h_curr.copy()
    
    for t in range(iters):
        # Nesterov step
        x = h_nest + momentum * (h_nest - h_curr)
        x = project_feasible(x, n)
        
        # Subgradient at x
        g = bundle_gradient(x, n, top_m)
        
        # Update
        v = momentum * v + (1 - momentum) * g
        h_nest = x - lr / np.sqrt(t + 1) * v
        h_curr = project_feasible(h_nest, n)
        
        # Check improvement
        c5 = compute_c5(h_curr, n)
        if c5 < best_local:
            best_local = c5
            best_h = h_curr.copy()
        
        # Periodic noise
        if (t + 1) % noise_every == 0:
            rng = np.random.RandomState((t + 1) * 7919)
            h_nest += rng.randn(n) * noise_scale
            h_nest = project_feasible(h_nest, n)
            h_curr = project_feasible(h_nest, n)
    
    return best_h, best_local

    """Interpolate h from n points to n_new points."""
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2)  # Not modular, just linear
    # Actually, h is on [0,2] not periodic, so:
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

import numpy as np
from scipy.optimize import minimize
import time


def corr_full(h, n):
    return np.correlate(h, 1.0 - h, mode="full")


def c5_fvals(h, n):
    return corr_full(h, n) * (2.0 / n)


def c5_val(h, n):
    return float(np.max(c5_fvals(h, n)))


def project(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(500):
        d = h.sum() - tgt
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h > 1e-12
        else:
            m = h < 1.0 - 1e-12
        if not m.any():
            break
        h[m] -= d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_sing(h, n, lag):
    dx = 2.0 / n
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[:m] += (1.0 - h[lag:])
            g[lag:] -= h[:m]
    else:
        m = n + lag
        l = -lag
        if m > 0:
            g[l:] += (1.0 - h[:m])
            g[:m] -= h[l:]
    return g * dx


def grad_bundle(h, n, m, sharp):
    fv = c5_fvals(h, n)
    idx = np.argsort(fv)[-m:]
    w = np.exp(sharp * (fv[idx] - fv[idx].max()))
    w /= w.sum()
    g = np.zeros(n)
    for wi, ki in zip(w, idx):
        g += wi * grad_sing(h, n, ki - (n - 1))
    return g


def subgrad_opt(h, n, iters, lr, mom, mb, nseed):
    hc = h.copy()
    hn = h.copy()
    v = np.zeros(n)
    best = c5_val(hc, n)
    bh = hc.copy()
    rng = np.random.RandomState(nseed)
    noise_int = max(200, iters // 8)
    for t in range(iters):
        x = project(hn + mom * (hn - hc), n)
        g = grad_bundle(x, n, mb, 20.0)
        v = mom * v + (1.0 - mom) * g
        step = lr / (np.sqrt(float(t + 1)) + 0.5)
        hn = x - step * v
        hc = project(hn, n)
        cv = c5_val(hc, n)
        if cv < best:
            best = cv
            bh = hc.copy()
        if (t + 1) % noise_int == 0:
            hn += rng.randn(n) * 0.03
            hn = project(hn, n)
            hc = project(hn, n)
            best = c5_val(hc, n)
            bh = hc.copy()
    return bh, best


def interp_h(h, n, n2):
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, n2, endpoint=False)
    return project(np.interp(xn, xo, h), n2)


def sum_f(h, n):
    return h.sum() - n / 2.0


def sum_j(h, n):
    return np.ones(n)


def sobj_fun(x, n, beta):
    fv = c5_fvals(x, n)
    mx = np.max(fv)
    return mx + np.log(np.sum(np.exp(beta * (fv - mx)))) / beta


def sobj_jac(x, n, beta):
    fv = c5_fvals(x, n)
    mx = np.max(fv)
    w = np.exp(beta * (fv - mx))
    w /= w.sum()
    g = np.zeros(n)
    for ki in range(len(fv)):
        if w[ki] < 1e-14:
            continue
        g += w[ki] * grad_sing(x, n, ki - (n - 1))
    return g


def hobj_fun(x, n):
    return c5_val(x, n)


def hobj_jac(x, n):
    fv = c5_fvals(x, n)
    km = int(np.argmax(fv))
    return grad_sing(x, n, km - (n - 1))


def slsqp_run(h, n, objf, objg, maxit):
    b = [(0.0, 1.0)] * n
    c = [{'type': 'eq', 'fun': sum_f, 'jac': sum_j, 'args': (n,)}]
    try:
        r = minimize(objf, h, jac=objg, args=(n,), method='SLSQP',
                     bounds=b, constraints=c,
                     options={'maxiter': maxit, 'ftol': 1e-13})
        hn = project(np.clip(r.x, 0.0, 1.0), n)
        return hn, c5_val(hn, n)
    except Exception:
        return h, c5_val(h, n)
