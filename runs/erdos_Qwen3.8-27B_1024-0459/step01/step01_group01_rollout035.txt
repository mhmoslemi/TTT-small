import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        for i in idx:
            h[i] = np.clip(h[i] + per, 0.0, 1.0)
    return h

def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def smooth_max_val(h, tau):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    # Numerically stable soft max
    shifted = (corr - m) / tau
    shifted = np.clip(shifted, -50, 0)
    log_sum = m + tau * np.log(np.sum(np.exp(shifted)))
    return log_sum * dx

def smooth_max_obj(x, tau):
    h = np.clip(x, 0.0, 1.0)
    return smooth_max_val(h, tau)

def eq_con(x, n):
    return x.sum() - n / 2.0

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    # Get parent
    if 'initial_h_values' in dir() and initial_h_values is not None and np.size(initial_h_values) > 0:
        parent_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        parent_h = np.full(180, 0.5)
    
    best_h = parent_h.copy()
    best_c5 = compute_c5(best_h)
    
    # Try resolutions
    n_parent = len(parent_h)
    resolutions = sorted(set([n_parent, 2*n_parent, 4*n_parent]))
    resolutions = [r for r in resolutions if r <= 1440]
    
    for n in resolutions:
        elapsed = time.time() - t0
        if elapsed > budget_s * 0.7:
            break
        
        x0 = upsample(best_h, n)
        x0 = fix_sum(x0, n)
        bounds = [(0.0, 1.0)] * n
        
        # SLSQP with decreasing tau
        for tau in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.75:
                break
            maxiter = 2000 if n <= 360 else 1000
            res = minimize(
                fun=lambda x: smooth_max_obj(x, tau),
                x0=x0, method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': lambda x: eq_con(x, n)}],
                options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False}
            )
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = fix_sum(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
            x0 = h_new
        
        # Perturbation restarts
        for r in range(3):
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.8:

            noise = rng.normal(0, 0.03, n)
            x_pert = np.clip(x0 + noise, 0.0, 1.0)
            x_pert = fix_sum(x_pert, n)
            res = minimize(
                fun=lambda x: smooth_max_obj(x, 0.003),
                x0=x_pert, method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': lambda x: eq_con(x, n)}],
                options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}
            )
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = fix_sum(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:

                best_h = h_new.copy()

    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, len(best_h))

def subgradient_step(h, lr, threshold=0.01):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    cmax = np.max(corr)
    # Find active lags (within threshold of max)
    active = np.where(corr >= cmax - threshold)[0]
    w = np.zeros(2*n-1)
    w[active] = 1.0 / len(active)
    
    # Compute subgradient
    grad = np.zeros(n)
    center = n - 1
    for idx in active:
        lag = idx - center
        wi = w[idx]
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            # ∂c[lag]/∂h[j] = (1-h[j+lag]) for j ≤ n-1-lag, minus h[j-lag] for j ≥ lag
            l = lag
            for j in range(n):
                g = 0.0
                if j <= n - 1 - l:
                    g += wi * (1.0 - h[j + l])
                if j >= l and j - l <= n - 1 - l:
                    g -= wi * h[j - l]
                grad[j] += g
        else:
            l = -lag
            for j in range(n):
                g = 0.0
                if j >= l:
                    g += wi * (1.0 - h[j - l])
                if j <= n - 1 - l:
                    g -= wi * h[j + l]
                grad[j] += g
    
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = fix_sum(h_new, n)
    return h_new

import numpy as np
from scipy.optimize import minimize
import functools
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        for i in idx:
            h[i] = np.clip(h[i] + per, 0.0, 1.0)
    return h


def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def smooth_max_obj(x, n, dx, tau):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / tau
    shifted = np.clip(shifted, -80.0, 0.0)
    val = m + tau * np.log(np.sum(np.exp(shifted)))
    return float(val * dx)


def smooth_max_grad(x, n, dx, tau):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    shifted = (corr - cmax) / tau
    shifted = np.clip(shifted, -100.0, 0.0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    weights /= wsum

    grad = np.zeros(n)
    center = n - 1
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-12:
            continue
        lag = k - center
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            l = lag
            if l >= n:
                continue
            part1 = np.zeros(n)
            part1[:n - l] = 1.0 - h[l:]
            part2 = np.zeros(n)
            part2[l:] = h[:n - l]
            grad += wk * (part1 - part2)
        else:
            l = -lag
            if l >= n:
                continue
            part1 = np.zeros(n)
            part1[l:] = 1.0 - h[:n - l]
            part2 = np.zeros(n)
            part2[:n - l] = h[l:]
            grad += wk * (part1 - part2)
    return grad * dx


def eq_con(x, n):
    return float(x.sum() - n / 2.0)


def subgradient_step(h, lr, threshold_frac=0.02):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    thresh = threshold_frac * max(cmax * dx, 1e-6) / dx
    active = np.where(corr >= cmax - thresh)[0]
    if len(active) == 0:
        return h.copy()

    grad = np.zeros(n)
    center = n - 1
    for idx in active:
        lag = idx - center
        if lag == 0:
            grad += (1.0 - 2.0 * h) / len(active)
        elif lag > 0:
            l = lag
            if l >= n:
                continue
            part1 = np.zeros(n)
            part1[:n - l] = 1.0 - h[l:]
            part2 = np.zeros(n)
            part2[l:] = h[:n - l]
            grad += (part1 - part2) / len(active)
        else:
            l = -lag
            if l >= n:
                continue
            part1 = np.zeros(n)
            part1[l:] = 1.0 - h[:n - l]
            part2 = np.zeros(n)
            part2[:n - l] = h[l:]
            grad += (part1 - part2) / len(active)

    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = fix_sum(h_new, n)
    return h_new


def periodic_init(n, period_frac, duty_frac):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    period = 2.0 * period_frac
    h = np.zeros(n)
    for i in range(n):
        phase = (x[i] % period) / period
        h[i] = 1.0 if phase < duty_frac else 0.0
    h = fix_sum(h, n)
    return h

            jac_p = functools.partial(smooth_max_grad, n=n, dx=dx, tau=tau)
            con_p = functools.partial(eq_con, n=n)
            try:
                res = minimize(
                    fun=obj_p, jac=jac_p, x0=x0, method='SLSQP',
                    bounds=bounds_list,
                    constraints=[{'type': 'eq', 'fun': con_p}],
                    options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False}
