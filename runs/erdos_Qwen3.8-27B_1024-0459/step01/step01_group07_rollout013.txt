    # conv[j] = sum_i h[i]*(1-h[i-(j-(n-1))]) * dx for valid i
    # Let lag = j - (n-1)
    # conv[j] = sum_i h[i]*(1-h[i-lag]) * dx
    # d(conv[j])/d(h[k]) = (1-h[k-lag])*dx - h[k+lag]*dx (for valid indices)

    # For each lag j (centered at 0), the contribution to the gradient is:
    # weights[j] * [(1-h[k-lag])*dx - h[k+lag]*dx]

    #         = dx * sum_lag weights[lag+n-1] * [1 - h[k-lag] - h[k+lag]]

    # sum_lag weights[lag+n-1] * h[k+lag] = (weights_conv * h)[k] (flipped)

            # But weights[0] is at index n-1, not 0. Let me handle in the loop.
            pass

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-14
            if not np.any(free):
                break
            h[free] = np.minimum(h[free] + diff / free.sum(), 1.0)
        else:
            free = h > 1e-14
            if not np.any(free):
                break
            h[free] = np.maximum(h[free] + diff / free.sum(), 0.0)
    return np.clip(h, 0.0, 1.0)

def _penalty_obj(x, n, mu):
    target = n / 2.0
    val = compute_c5(x, n)
    pen = mu * (x.sum() - target) ** 2
    return val + pen

def slsqp_opt(h0, n, maxiter=500, mu=1e4):
    bounds = [(0.0, 1.0)] * n
    res = minimize(_penalty_obj, h0.copy(), args=(n, mu), method='SLSQP',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-13})
    h_opt = project_feasible(res.x, n)
    return h_opt, compute_c5(h_opt, n)
