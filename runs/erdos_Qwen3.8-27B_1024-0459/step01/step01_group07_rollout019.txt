import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def smooth_c5(h, n_points, temp=0.01):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(conv)
    # Numerical stability
    shifted = (conv - m) / temp
    log_sum = np.log(np.sum(np.exp(shifted))) - np.log(len(conv))
    return m + temp * log_sum

def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0
            cnt = np.sum(free)
            if cnt == 0:
                break
            step = diff / cnt
            h[free] = np.minimum(h[free] + step, 1.0)
        else:
            free = h > 0.0
            cnt = np.sum(free)
            if cnt == 0:
                break
            step = diff / cnt
            h[free] = np.maximum(h[free] + step, 0.0)
    return h

    """Optimize using L-BFGS-B with projection"""
    n_target = n / 2.0
    
    def obj(x):
        h = np.clip(x, 0, 1)
        return smooth_c5(h, n, temp)
    
    res = minimize(obj, h0.copy(), method='L-BFGS-B',
                   bounds=[(0,1)]*n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
    h_opt = project_feasible(res.x, n)
