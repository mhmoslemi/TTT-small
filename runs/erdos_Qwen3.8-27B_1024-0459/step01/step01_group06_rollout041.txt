        mask = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
        if mask.sum() > 0:
            h_proj[mask] -= diff / mask.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj


def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / n))


def c5_values(h, n):
    return np.correlate(h, 1.0 - h, mode="full")


def smooth_obj(h, n, beta):
    c = c5_values(h, n)
    scaled = beta * c
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / beta * (2.0 / n))


def smooth_grad(h, n, beta):
    c = c5_values(h, n)
    scaled = beta * c
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w /= w_sum
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-13:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] -= wk * h[lag:]
        else:
            d = -lag
            grad[:n - d] += wk * (1.0 - h[d:])
            grad[d:] -= wk * h[:n - d]
    return grad * dx


def hard_grad(h, n):
    c = c5_values(h, n)
    k_max = np.argmax(c)
    lag = k_max - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag >= 0:
        grad[lag:] += (1.0 - h[:n - lag])
        grad[:n - lag] -= h[lag:]
    else:
        d = -lag
        grad[:n - d] += (1.0 - h[d:])
        grad[d:] -= h[:n - d]
    return grad * dx


def pgd_smooth(h, n, beta, lr_init, max_iters, time_limit):
    t_start = time.time()
    h = h.copy()
    lr = lr_init
    best_local_c5 = compute_c5(h, n)
    h_best = h.copy()
    obj_prev = smooth_obj(h, n, beta)

    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        grad = smooth_grad(h, n, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        h_try = h
        obj_try = obj_prev
        for attempt in range(50):
            h_try = project_feasible(h - lr * grad, n)
            obj_try = smooth_obj(h_try, n, beta)
            if obj_try <= obj_prev + 1e-13:
                break
            lr *= 0.5
        else:
            break
        h = h_try
        obj_prev = obj_try
        lr = min(lr * 1.1, 2.0)

        c5_val = compute_c5(h, n)
        if c5_val < best_local_c5:
            best_local_c5 = c5_val
            h_best = h.copy()

    return h_best, best_local_c5


def slsqp_fun(x, n, beta):
    return smooth_obj(x, n, beta)


def slsqp_grad(x, n, beta):
    return smooth_grad(x, n, beta)


def sum_eq_fun(x, n):
    return np.sum(x) - n / 2.0


def sum_eq_jac(x, n):
    return np.ones(len(x))


def coordinate_descent(h, n, max_passes, time_limit):
    t_start = time.time()
    h = h.copy()
    best_c5 = compute_c5(h, n)
    dx = 2.0 / n
    step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002]

    for s in step_sizes:
        if time.time() - t_start > time_limit:
            break
        for p in range(max_passes):
            if time.time() - t_start > time_limit:
                break
            improved = False
            for i in range(n):
                for sgn in [1.0, -1.0]:
                    h_try = h.copy()
                    h_try[i] = h[i] + sgn * s
                    h_try = project_feasible(h_try, n)
                    c5_try = compute_c5(h_try, n)
                    if c5_try < best_c5 - 1e-12:
                        h = h_try
                        best_c5 = c5_try
                        improved = True
                        break
            if not improved and p > 0:
                break

    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float)
        n_init = len(h_init)
    except Exception:
        n_init = 100
        h_init = np.full(n_init, 0.5)

    n = 200

    if n_init != n:
        x_old = np.linspace(0.0, 2.0, n_init, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_old, h_init)
    else:
        h0 = h_init.copy()
    h0 = project_feasible(h0, n)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)

    # Stage 1: Smooth PGD with beta annealing
    for beta in [3.0, 10.0, 40.0, 150.0, 500.0]:
        if time.time() - t0 > budget_s - 120:
            break
        time_left = budget_s - 120 - (time.time() - t0)
        alloc = min(time_left * 0.2, 180.0)
        if alloc < 5.0:
            break
        h_new, c5_new = pgd_smooth(best_h, n, beta, lr_init=0.05, max_iters=50000, time_limit=alloc)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Stage 2: Hard max subgradient method
    if time.time() - t0 < budget_s - 80:
        time_left = budget_s - 80 - (time.time() - t0)
        alloc = min(time_left * 0.2, 120.0)
        h = best_h.copy()
        t_start = time.time()
        lr = 0.03
        for it in range(50000):
            if time.time() - t_start > alloc:
                break
            grad = hard_grad(h, n)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            h_try = project_feasible(h - lr * grad / gnorm, n)
            c5_try = compute_c5(h_try, n)
            c5_cur = compute_c5(h, n)
            if c5_try < c5_cur:
                h = h_try
                lr = min(lr * 1.03, 0.5)
            else:
                lr *= 0.7
            if c5_try < best_c5:
                best_h = h_try.copy()
                best_c5 = c5_try

    # Stage 3: Random restarts with PGD
    if time.time() - t0 < budget_s - 40:
        time_left = budget_s - 40 - (time.time() - t0)
        n_restarts = 80
        per_restart = time_left / n_restarts
        for i in range(n_restarts):
            if time.time() - t0 > budget_s - 20:
                break
            if per_restart < 0.5:
                break
            scale = 0.005 + 0.06 * rng.rand()
            h_pert = best_h + rng.randn(n) * scale
            h_pert = project_feasible(h_pert, n)
            h_new, c5_new = pgd_smooth(h_pert, n, 100.0, lr_init=0.03, max_iters=5000, time_limit=min(per_restart, 12.0))
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Stage 4: Coordinate descent polish
    if time.time() - t0 < budget_s - 30:
        time_left = budget_s - 30 - (time.time() - t0)
        alloc = min(time_left * 0.3, 60.0)
        h_cd, c5_cd = coordinate_descent(best_h, n, max_passes=3, time_limit=alloc)
        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd

    # Stage 5: SLSQP refinement
    if time.time() - t0 < budget_s - 15:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n,)}]
        beta_final = 500.0
        try:
            res = minimize(
                fun=slsqp_fun,
                x0=best_h,
                jac=slsqp_grad,
                args=(n, beta_final),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 400, 'ftol': 1e-14}
            )
            h_final = project_feasible(np.clip(res.x, 0.0, 1.0), n)
            c5_final = compute_c5(h_final, n)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
        except Exception:
            pass

    # Stage 6: Try higher resolution if time permits
    if time.time() - t0 < budget_s - 50:
        n_high = 300
        x_old = np.linspace(0.0, 2.0, n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_high, endpoint=False)
        h_high = np.interp(x_new, x_old, best_h)
        h_high = project_feasible(h_high, n_high)
        c5_high = compute_c5(h_high, n_high)

        if c5_high <= best_c5 + 0.008:
            for beta in [50.0, 200.0, 600.0]:
                if time.time() - t0 > budget_s - 25:
                    break
                time_left = budget_s - 25 - (time.time() - t0)
                alloc = min(time_left * 0.35, 80.0)
                if alloc < 5.0:
                    break
                h_high, c5_high = pgd_smooth(h_high, n_high, beta, lr_init=0.03, max_iters=30000, time_limit=alloc)
            if c5_high < best_c5:
                best_h = h_high
                best_c5 = c5_high
                n = n_high

    # Final coordinate descent polish
    if time.time() - t0 < budget_s - 10:
        h_cd, c5_cd = coordinate_descent(best_h, n, max_passes=2, time_limit=5.0)
        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)

    try:
        val = evaluate_erdos_solution(best_h, n)
        if val is not None and 0.0 < val < 1.0:
            best_c5 = min(best_c5, float(val))
    except Exception:
        pass

    return (best_h, best_c5, n)
