import time

def compute_c5(h):
    n = len(h)
    if n == 0:

    dx = 2.0 / n

def as_feasible(h, n):
    """Project h to the feasible set {0<=h<=1, sum(h)=n/2}"""
    h = np.asarray(h, dtype=float).copy()
    if h.size != n:
        if h.size > n:
            h = h[:n]
        else:
            tmp = np.zeros(n)
            m = min(n, h.size)
            if m > 0:
                tmp[:m] = h[:m]
            h = tmp
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    
    # Binary search for shift t
    if abs(h.sum() - target) < 1e-14:
        return h
    
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    
    # Fix residual
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def soft_max_obj(h, alpha, n):
    """Compute soft-max objective and return value"""
    scores = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(scores)
    if alpha <= 1e-14:
        return m
    w = np.exp((scores - m) / alpha)
    A = np.sum(w)
    if not np.isfinite(A) or A <= 0:
        return m
    return m + alpha * np.log(A)

def soft_max_grad(h, alpha, n):
    """Compute gradient of soft-max objective w.r.t. h"""
    scores = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(scores)
    
    if alpha <= 1e-14:
        # Subgradient of max: average of active
        active = (scores >= m - 1e-8)
        w = active.astype(float)
        w /= w.sum()
    else:
        w = np.exp((scores - m) / alpha)
        A = np.sum(w)
        w /= A
    
    # Now compute weighted sum of gradients
    # w is indexed 0 to 2n-2, corresponding to lags d = -(n-1) to n-1
    # Lag d corresponds to index d + n - 1

        if we <= 1e-15:

        # ∂s_e/∂h[k] = (1-h[k+e]) for k=0..n-1-e, minus h[k-e] for k=e..n-1
        grad[:n-e] += we * (1.0 - h[e:])
        grad[e:] -= we * h[:n-e]

import numpy as np
import time
import math
from scipy.optimize import minimize

# Global deadline tracking
class _Budget:
    deadline = None

def _check_time():
    if _Budget.deadline is not None and time.time() > _Budget.deadline:
        raise TimeoutError

def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 1e6
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def scores_of(h):
    return np.correlate(h, 1.0 - h, mode="full")

def softmax_value(scores, alpha):
    m = np.max(scores)
    if alpha < 1e-14:
        return m
    shifted = (scores - m) / alpha
    # For numerical stability
    max_shifted = np.max(shifted)  # should be 0
    log_sum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    return m + alpha * log_sum

def softmax_weights(scores, alpha):
    m = np.max(scores)
    if alpha < 1e-14:
        active = (scores >= m - 1e-9).astype(float)
        s = active.sum()
        if s == 0:
            active[int(np.argmax(scores))] = 1.0
            s = 1.0
        return active / s
    w = np.exp((scores - m) / alpha)
    s = w.sum()
    if s == 0 or not np.isfinite(s):
        w = np.zeros_like(scores)
        w[int(np.argmax(scores))] = 1.0
        return w
    return w / s

def obj_grad(h, alpha):
    """Compute (objective, gradient) of the soft-max C5 objective."""
    n = h.size
    scores = scores_of(h)
    m = np.max(scores)
    w = softmax_weights(scores, alpha)
    
    if alpha < 1e-14:
        val = m
    else:
        val = m + alpha * np.log(np.sum(w) + 1e-300)  # w already normalized, sum=1
        # Actually: softmax_value = m + alpha * log(sum(exp((s-m)/alpha)))
        # If w = exp((s-m)/alpha) / sum, then log(sum(exp)) = log(sum) 
        # sum(exp((s-m)/alpha)) = sum(w) * ... no.
        # w_i = exp((s_i-m)/alpha) / sum_j exp((s_j-m)/alpha)
        # So sum(exp((s_i-m)/alpha)) = sum_j exp((s_j-m)/alpha)
        # And log of that = log(sum(exp((s-m)/alpha)))
        # = log(sum(w) * sum(exp((s-m)/alpha)))... no that's circular.
        # Let me just recompute:
        exp_vals = np.exp((scores - m) / alpha)
        S = np.sum(exp_vals)
        val = m + alpha * np.log(S + 1e-300)
        w = exp_vals / (S + 1e-300)
    
    # Gradient computation
    grad = np.zeros(n, dtype=float)
    
    # Lag 0 (index n-1): s_0 = sum h[j](1-h[j])
    w0 = w[n-1]
    if w0 > 1e-16:
        grad += w0 * (1.0 - 2.0 * h)
    
    # Forward lags e=1..n-1 (index n-1+e)
    for e in range(1, n):
        we = w[n-1+e]
        if we < 1e-16:
            continue
        # d(s_e)/d(h[k]) for k=0..n-1-e: (1-h[k+e])
        # d(s_e)/d(h[k]) for k=e..n-1: -h[k-e]
        grad[:n-e] += we * (1.0 - h[e:])
        grad[e:] += we * (-h[:n-e])
    
    # Backward lags e=1..n-1 (index n-1-e, which is n-2-e... wait)
    # Lag -e corresponds to index (n-1) + (-e) = n-1-e
    for e in range(1, n):
        wb = w[n-1-e]
        if wb < 1e-16:
            continue
        # d(s_{-e})/d(h[k]):
        # for k=e..n-1: (1-h[k-e])
        # for k=0..n-1-e: -h[k+e]
        grad[e:] += wb * (1.0 - h[:n-e])
        grad[:n-e] += wb * (-h[e:])
    
    # Scale by dx = 2/n
    val *= (2.0 / n)
    grad *= (2.0 / n)
    
    return val, grad

def project_feasible(h, n):
    """Project h onto {h: 0<=h<=1, sum(h)=n/2}"""
    h = np.asarray(h, dtype=float).copy()
    if h.size < n:
        tmp = np.zeros(n)
        m = min(n, h.size)
        if m > 0:
            tmp[:m] = h[:m]
        h = tmp
    elif h.size > n:
        h = h[:n]
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    
    if abs(h.sum() - target) < 1e-13:
        return h
    
    # Binary search for additive shift
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    
    # Fix residual

    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    
    return h

def slsqp_optimize(h, n, time_limit, alpha=0.1):
    """SLSQP optimization with soft-max or exact max."""
    h = project_feasible(h, n).copy()
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        v, _ = obj_grad(x, alpha)
        return v
    
    def grad(x):
        _, g = obj_grad(x, alpha)
        return g
    
    def eq_fun(x):
        return np.sum(x) - 0.5 * n
    
    def eq_jac(x):
        return np.ones(n)
    
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    maxiter = max(50, int(500 * (70.0 / n)))
    maxiter = min(maxiter, 1000)
    
    t_start = time.time()
    try:
        res = minimize(
            obj, h, jac=grad, method='SLSQP',
            bounds=bounds, constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': 0}
        )
        if res.success or hasattr(res, 'x'):
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new)
            c_old = compute_c5(h)
            if c_new < c_old:
                return h_new
    except Exception:
        pass
    return h
