    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):

                no_improve = 0
        if not improved:
            no_improve += 1
            if no_improve >= patience:

        worst_shift = worst_idx - (n - 1)

        # Find positions contributing to worst shift
        # overlap at shift s: sum_i h[i]*(1-h[i+s]) for valid i
        s = worst_shift
        bad_positions = []
        for i in range(max(0, -s), min(n, n - s)):
            if h[i] == 1.0 and h[i + s] == 0.0:
                bad_positions.append(i)

        if len(bad_positions) > 0 and len(zeros) > 0:
            i = bad_positions[rng.randint(0, len(bad_positions))]
            # Find a zero to swap with, preferring one that won't create new bad pairs
            # Just pick a random zero
            j = zeros[rng.randint(0, len(zeros))]
        else:
            i = ones[rng.randint(0, len(ones))]
            j = zeros[rng.randint(0, len(zeros))]
        if i == j:
            continue

        h[i] = 0.0
        h[j] = 1.0
        c5_try = compute_c5(h)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            ones = list(np.where(h == 1.0)[0])
            zeros = list(np.where(h == 0.0)[0])
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    return h, best_c5

def final_slsqp_polish(h_init, n, max_iter=3000):
    global _N_OPT
    _N_OPT = n
    h0 = make_feasible(h_init.copy(), n)
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}
    try:
        res = minimize(_direct_obj, h0, method='SLSQP',
                      bounds=bounds, constraints=con,
                      options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return h_init, compute_c5(h_init)

def generate_inits(n, rng, h_parent=None):
    inits = []
    if h_parent is not None and len(h_parent) > 0:
        h_p = make_feasible(interp_h(h_parent, n), n)
        inits.append(h_p)
    inits.append(thue_morse_init(n))
    inits.append(antisymmetric_init(n))
    for _ in range(4):
        inits.append(random_binary(n, rng))
    # Phase-shifted Thue-Morse
    for shift in [1, 2, 3]:
        tm = np.zeros(n)
        for i in range(n):
            tm[i] = 1.0 if bin((i + shift) % n).count('1') % 2 == 0 else 0.0
        inits.append(make_feasible(tm, n))
    return inits

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    n = 256

    def remaining():
        return budget_s - (time.time() - start_time)

    h_parent = None
    try:
        h_parent_arr = np.array(initial_h_values, dtype=float)
        if h_parent_arr is not None and len(h_parent_arr) > 0:
            h_parent = h_parent_arr.copy()
    except Exception:
        h_parent = None

    best_h = thue_morse_init(n)
    best_c5 = compute_c5(best_h)

    inits = generate_inits(n, rng, h_parent)

    # Phase 1: Smooth max optimization for each init (coarse to fine on beta)
    for h0 in inits:
        if remaining() < 30:
            break
        h_opt, c5_val = smooth_max_optimize(h0, n, beta_start=15.0, beta_end=250.0,
                                            n_steps=5, max_iter=600)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()

    # Phase 2: Binary local search from best and from each init
    n_trials_bs = 80000
    timeout_bs = 50.0
    for h0 in [best_h] + inits[:4]:
        if remaining() < 60:
            break
        h_b, c5_b = binary_local_search(h0, n, rng, max_trials=n_trials_bs,
                                        stall_limit=15000, timeout=timeout_bs)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 3: Targeted binary search (focus on worst shift)
    for h0 in [best_h] + inits[:3]:
        if remaining() < 40:
            break
        h_t, c5_t = targeted_binary_search(h0, n, rng, max_trials=80000, timeout=45.0)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 4: More binary local search from the current best
    if remaining() > 60:
        n_trials_bs2 = int(min(200000, remaining() * 15000))
        h_b2, c5_b2 = binary_local_search(best_h, n, rng, max_trials=n_trials_bs2,
                                         stall_limit=20000, timeout=remaining() - 30)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

    # Phase 5: Targeted search again from best
    if remaining() > 40:
        h_t2, c5_t2 = targeted_binary_search(best_h, n, rng, max_trials=60000,
                                             timeout=remaining() - 20)
        if c5_t2 < best_c5:
            best_c5 = c5_t2
            best_h = h_t2.copy()

    # Phase 6: Continuous local search for fine tuning
    if remaining() > 50:
        h_c, c5_c = continuous_local_search(best_h, n, rng, max_trials=30000,
                                            step_sizes=[0.01, 0.03, 0.08],
                                            timeout=remaining() - 25)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()

    # Phase 7: Final SLSQP polish
    if remaining() > 20:
        h_f, c5_f = final_slsqp_polish(best_h, n, max_iter=3000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 8: One more round of binary search on the polished solution
    if remaining() > 30:
        h_b3, c5_b3 = binary_local_search(best_h, n, rng, max_trials=50000,
                                         stall_limit=10000, timeout=remaining() - 15)
        if c5_b3 < best_c5:
            best_c5 = c5_b3
            best_h = h_b3.copy()

    # Final feasibility
    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n)
