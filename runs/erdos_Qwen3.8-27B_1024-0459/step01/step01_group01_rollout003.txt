import numpy as np

def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n == 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def fix_sum(h, n):
    h = np.array(h, dtype=float).copy()
    target = 0.5 * float(n)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-13)[0]
        else:
            idx = np.where(h > 1e-13)[0]
        if idx.size == 0:
            break
        per = diff / float(idx.size)
        if per > 0:
            inc = np.minimum(1.0 - h[idx], per)
            h[idx] += inc
        else:
            dec = np.minimum(h[idx], -per)
            h[idx] -= dec
    h = np.clip(h, 0.0, 1.0)
    return h


def upsample_to(h_old, n_new):
    h_old = np.asarray(h_old, dtype=float)
    n_old = int(h_old.size)
    if n_old == 0:
        return np.zeros(n_new, dtype=float)
    if n_old == int(n_new):
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, int(n_new), endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new


def generate_markov(rng, n, flip_prob, max_run):
    n = int(n)
    z = np.empty(n, dtype=np.int64)
    sign = 1 if rng.random() < 0.5 else -1
    z[0] = sign
    run = 1
    for i in range(1, n):
        if run >= max_run:
            sign = -sign
        else:
            if rng.random() < flip_prob:
                sign = -sign
        z[i] = sign
        if z[i] == z[i - 1]:
            run += 1
        else:
            run = 1
    return z


def generate_ar_signs(rng, n, rho, max_run):
    n = int(n)
    z = np.empty(n, dtype=np.int64)
    y = 0.0
    run = 0
    for i in range(n):
        y = rho * y + rng.normal()
        val = 1 if y >= 0 else -1

            run = 1
        else:

            else:

    return z


def generate_run_sequence(rng, n, mean_run, max_run):
    n = int(n)
    z = np.zeros(n, dtype=np.int64)
    pos = 0
    sign = 1 if rng.random() < 0.5 else -1
    S = 0
    min_run = max(1, int(max_run * 0.25))
    while pos < n:
        rem = n - pos
        u = float(rng.random())
        if u < 1e-12:
            u = 1e-12
        lam = int(-math.log(u) / float(mean_run))
        if lam < min_run:
            lam = min_run
        if lam > max_run:
            lam = max_run

        if (sign == 1 and S > 0) or (sign == -1 and S < 0):
            lam = max(1, lam // 2)
        elif (sign == 1 and S < 0) or (sign == -1 and S > 0):
            lam = max(1, min(max_run, 2 * lam))

        lam = min(lam, rem)
        if lam <= 0:
            lam = 1
        z[pos:pos + lam] = sign
        pos += lam
        S += sign * lam
        sign = -sign
    return z


def balance_z(z, rng):
    z = np.array(z, dtype=np.int64).copy()
    n = int(z.size)
    if n == 0:
        return z
    target = n // 2
    plus = int(np.count_nonzero(z == 1))

    attempts = 0
    while plus < target and attempts < 1000:
        attempts += 1
        i = int(rng.integers(n))
        if z[i] == -1:
            z[i] = 1
            plus += 1

    attempts = 0
    while plus > target and attempts < 1000:
        attempts += 1
        i = int(rng.integers(n))
        if z[i] == 1:
            z[i] = -1
            plus -= 1

    if plus != target:
        if plus < target:
            idx = np.where(z == -1)[0]
            k = min(target - plus, idx.size)
            if k > 0:
                z[idx[:k]] = 1
        else:
            idx = np.where(z == 1)[0]
            k = min(plus - target, idx.size)
            if k > 0:
                z[idx[:k]] = -1
    return z


def binary_init_state(z):
    z = np.asarray(z, dtype=np.int64)
    n = int(z.size)
    if n == 1:
        R = np.zeros(1, dtype=np.int64)
        P = np.empty(2, dtype=np.int64)
        P[0] = 0
        P[1] = int(z[0])
        m = np.arange(1, n, dtype=np.int64)
        return R, P, m, 0

    full = np.correlate(z, z, mode="full")
    R = np.zeros(n, dtype=np.int64)
    R[0] = n
    R[1:] = full[n - 1:2 * n - 1]

    P = np.empty(n + 1, dtype=np.int64)
    P[0] = 0
    np.cumsum(z, out=P[1:])

    m = np.arange(1, n, dtype=np.int64)
    D = (n - m - R[1:]) // 2
    U = D + np.abs(P[1:n])
    return R, P, m, int(np.max(U)) if U.size else 0


def apply_swap_candidate(z, R, P, dr, i, j):
    n = int(z.size)
    dr.fill(0)
    si = int(z[i])
    sj = int(z[j])
    m0 = abs(int(i) - int(j))

    if i < n - 1:
        dr[1:n - i] -= 2 * si * z[i + 1:n]
    if i > 0:
        dr[1:i + 1] -= 2 * z[:i] * si

    if j < n - 1:
        dr[1:n - j] -= 2 * sj * z[j + 1:n]
    if j > 0:
        dr[1:j + 1] -= 2 * z[:j] * sj

    if m0 > 0:
        dr[m0] += 4 * si * sj

    if si == 1:
        a = i
        b = j
    else:
        a = j
        b = i

    if a < b:
        start = a + 1
        stop = b + 1
        inc = -2
    else:
        start = b + 1
        stop = a + 1
        inc = 2

    R[1:] += dr[1:]
    P[start:stop] += inc
    z[i], z[j] = z[j], z[i]
    return start, stop, inc


def choose_targeted_swap(z, n, lag, rng):
    lag = int(lag)
    if lag <= 0 or lag >= n:
        return None, None
    L = n - lag
    if L <= 0:
        return None, None

    for _ in range(40):
        i = int(rng.integers(L))
        j = i + lag
        if int(z[i]) != int(z[j]):
            need = -int(z[i])
            for __ in range(40):
                q = int(rng.integers(n))
                if int(z[q]) == need and q != i and q != j:
                    return i, q
            for __ in range(40):
                q = int(rng.integers(n))
                if int(z[q]) == need and q != i:
                    return i, q
    return None, None

def _cont_compute_vals(h):
    n = int(_cont_n)
    vals = np.empty(2 * n - 1, dtype=float)
    vals[0] = float(np.dot(h, 1.0 - h))
    idx = 1
    for m in range(1, n):
        L = n - m
        vals[idx] = float(np.dot(h[:L], 1.0 - h[m:]))
        idx += 1
        vals[idx] = float(np.dot(h[m:], 1.0 - h[:L]))
        idx += 1
    return vals


def cont_smooth_obj(x):
    x = np.asarray(x, dtype=float)
    h = np.clip(x, 0.0, 1.0)
    vals = _cont_compute_vals(h)
    maxv = float(np.max(vals))
    shifted = vals - maxv
    tau = float(_cont_tau)
    if tau <= 1e-12:
        return float(_cont_dx * maxv)
    s = np.sum(np.exp(shifted / tau))
    return float(_cont_dx * (maxv + tau * math.log(s)))


def cont_smooth_jac(x):
    x = np.asarray(x, dtype=float)
    h = np.clip(x, 0.0, 1.0)
    n = int(_cont_n)
    vals = _cont_compute_vals(h)
    maxv = float(np.max(vals))
    tau = float(_cont_tau)
    if tau <= 1e-12:
        grad = np.zeros(n, dtype=float)
        argmax = int(np.argmax(vals))
        if argmax == 0:
            grad[:] = _cont_dx * (1.0 - 2.0 * h)
        return grad

    expv = np.exp((vals - maxv) / tau)
    weights = expv / float(np.sum(expv))
    grad = np.zeros(n, dtype=float)
    dx = float(_cont_dx)

    w0 = float(weights[0])
    if w0 > 1e-6:
        grad += dx * w0 * (1.0 - 2.0 * h)

    idx = 1
    for m in range(1, n):
        wp = float(weights[idx])
        wn = float(weights[idx + 1])
        idx += 2
        if wp < 1e-6 and wn < 1e-6:
            continue
        L = n - m
        hL = h[:L]
        hm = h[m:]
        if wp > 1e-6:
            grad[:L] += dx * wp * (1.0 - hm)
            grad[m:] += -dx * wp * hL
        if wn > 1e-6:
            grad[m:] += dx * wn * (1.0 - hL)
            grad[:L] += -dx * wn * hm
    return grad


def cont_eq(x):
    x = np.asarray(x, dtype=float)
    return np.array([x.sum() - 0.5 * float(_cont_n)], dtype=float)


def cont_eq_jac(x):
    n = int(_cont_n)
    return np.ones((1, n), dtype=float)

        ph = parent_h
        if ph.size % 2 != 0:
            ph = upsample_to(ph, ph.size + 1)
        ph = fix_sum(ph, ph.size)
        best_h = ph.copy()
        best_c = float(compute_c5(ph))

    best_bin_z = None
    best_bin_U = None
    best_bin_n = None

    target = 0.38085

    if best_h is None:
        base_n = 180
        zb = generate_markov(rng, base_n, 0.04, 90)
        zb = balance_z(zb, rng)
        best_h = make_h_from_z(zb)
        best_c = float(compute_c5(best_h))

    start_list = []

    if parent_h is not None and np.size(parent_h) > 0:
        ph = parent_h
        if ph.size % 2 != 0:
            ph = upsample_to(ph, ph.size + 1)
        ph = fix_sum(ph, ph.size)
        zp = np.where(ph > 0.5, 1, -1).astype(np.int64)
        zp = balance_z(zp, rng)
        start_list.append((zp, int(ph.size), "parent_threshold"))

    n_values = [180, 176, 192, 160, 208]
    flip_probs = [0.05, 0.04, 0.03, 0.025, 0.02]
    run_means = [14, 18, 22, 26, 30]

    for n in n_values:
        if n < 120:
            n = 128
        for fp in flip_probs:
            zz = generate_markov(rng, n, fp, 90)
            zz = balance_z(zz, rng)
            start_list.append((zz, n, "markov"))
        for mr in run_means:
            zz = generate_run_sequence(rng, n, mr, 90)
            zz = balance_z(zz, rng)
            start_list.append((zz, n, "run"))
        zz = generate_ar_signs(rng, n, 0.94, 90)
        zz = balance_z(zz, rng)
        start_list.append((zz, n, "ar"))

    binary_deadline = t0 + max(60.0, min(720.0, total_budget * 0.55))

    for idx, (zz, n, tag) in enumerate(start_list):
        now = time.time()
        rem_total = hard_deadline - now
        if rem_total < 30.0:
            break
        if now >= binary_deadline:
            if best_c > target:
                cont_rem = hard_deadline - now
                if cont_rem < 120.0:
                    break
        remaining_binary = binary_deadline - now
        if remaining_binary < 8.0 and best_c < target:
            break

        if best_c <= 0.3770:
            per = 8.0
        elif best_c <= target:
            per = 14.0
        else:
            per = 22.0

        if idx == 0 and tag == "parent_threshold":
            per = max(per, 35.0)

        per = max(6.0, min(per, remaining_binary * 0.95, rem_total * 0.45))
        if per < 4.0:
            break

        targetU = int(math.floor(target * float(n) - 1e-12))
        if targetU < 0:
            targetU = 0

        zout, Uout = local_search_binary(zz, per, rng, targetU)
        if Uout < 0:
            Uout = 0

        if best_bin_z is None or Uout < best_bin_U or (
                Uout == best_bin_U and n < best_bin_n):
            best_bin_z = zout.copy()
            best_bin_U = int(Uout)
            best_bin_n = int(n)

        hb = make_h_from_z(zout)
        hb = fix_sum(hb, n)
        c_eval = float(compute_c5(hb))
        c_exact = float(Uout) / float(n)
        c_bound = min(c_eval, c_exact)
        if c_bound < best_c - 1e-12:
            best_h = hb.copy()
            best_c = c_bound

    if best_bin_z is not None and time.time() < hard_deadline - 120.0:
        cur_bin_c = float(best_bin_U) / float(best_bin_n)
        if cur_bin_c > target or cur_bin_c > best_c - 1e-12:
            extra_time = min(280.0, (hard_deadline - time.time()) * 0.35)
            if extra_time > 60.0:
                n_extra = best_bin_n
                targetU = int(math.floor(target * float(n_extra) - 1e-12))
                zout, Uout = local_search_binary(best_bin_z, extra_time, rng, targetU)
                if Uout < 0:
                    Uout = 0
                if Uout < best_bin_U:
                    best_bin_z = zout.copy()
                    best_bin_U = int(Uout)
                    best_bin_n = int(n_extra)
                    hb = make_h_from_z(zout)
                    hb = fix_sum(hb, n_extra)
                    c_eval = float(compute_c5(hb))
                    c_exact = float(Uout) / float(n_extra)
                    c_bound = min(c_eval, c_exact)
                    if c_bound < best_c - 1e-12:
                        best_h = hb.copy()
                        best_c = c_bound

    if time.time() < hard_deadline - 60.0 and best_h is not None:
        cont_starts = []
        bh = best_h
        if bh is not None and np.size(bh) > 0:
            cont_starts.append(bh.copy())
        if parent_h is not None and np.size(parent_h) > 0:
            ph = parent_h
            if ph.size % 2 != 0:
                ph = upsample_to(ph, ph.size + 1)
            ph = fix_sum(ph, ph.size)
            # Avoid exact duplicate length/value if present.
            if not (len(ph) == len(cont_starts[0]) and
                    np.allclose(ph, cont_starts[0], rtol=1e-12, atol=1e-12)):
                cont_starts.append(ph.copy())
        elif best_bin_z is not None:
            bhb = make_h_from_z(best_bin_z)
            if not (len(bhb) == len(cont_starts[0]) and
                    np.allclose(bhb, cont_starts[0], rtol=1e-12, atol=1e-12)):
                cont_starts.append(bhb.copy())

        for ci, cs in enumerate(cont_starts):
            now = time.time()
            rem = hard_deadline - now
            if rem < 80.0:
                break
            if ci == 0:
                limit = min(320.0, rem * 0.55)
            else:
                limit = min(220.0, rem * 0.45)
            limit = max(30.0, limit)
            try:
                hpol, cpol = polish_continuous(cs, limit, rng)
                if cpol < best_c - 1e-12:
                    best_h = hpol.copy()
                    best_c = float(cpol)
            except Exception:
                pass

    if best_h is None:
        zb = generate_markov(rng, 180, 0.04, 90)
        zb = balance_z(zb, rng)
        best_h = make_h_from_z(zb)
        best_c = float(compute_c5(best_h))

    best_h = np.asarray(best_h, dtype=float)
    if best_h.size % 2 != 0:
        best_h = upsample_to(best_h, best_h.size + 1)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, best_h.size)
    best_c = float(compute_c5(best_h))

    return best_h, best_c, int(best_h.size)
