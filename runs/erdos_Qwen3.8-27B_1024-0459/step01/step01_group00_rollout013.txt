        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(-h)
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def sa_optimize(h0, n, deadline, rng, T0=4.0, Tmin=0.005, reheat_threshold=3000):
    """
    Simulated annealing on binary sequence.
    Only swap moves (1->0 at i, 0->1 at j).
    """
    h = h0.copy()
    curr = corr_max_val(h)
    best = curr
    best_h = h.copy()
    ones = np.where(h == 1)[0].astype(np.int32)
    zeros = np.where(h == 0)[0].astype(np.int32)
    T = T0
    it = 0
    since_improve = 0
    
    while time.time() < deadline:
        # Pick random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Apply swap
        h[i] = 0.0
        h[j] = 1.0
        new = corr_max_val(h)
        delta = new - curr
        
        if delta <= 0 or rng.rand() < np.exp(-delta / T):
            # Accept
            curr = new
            # Update tracking
            ones = np.delete(ones, np.where(ones == i)[0][0])
            zeros = np.delete(zeros, np.where(zeros == j)[0][0])
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
            
            if new < best - 1e-12:
                best = new
                best_h = h.copy()
                since_improve = 0
                # Slight reheat on improvement
                T = min(T * 1.01, T0 * 0.5)
            else:
                since_improve += 1
        else:
            # Reject
            h[i] = 1.0
            h[j] = 0.0
            since_improve += 1
        
        # Cooling
        T = max(T * 0.99999, Tmin)
        
        # Reheat if stuck
        if since_improve > reheat_threshold:
            T = T0 * 0.3
            since_improve = 0
            # Also perturb current solution
            for _ in range(rng.randint(3, 10)):
                i2 = ones[rng.randint(len(ones))]
                j2 = zeros[rng.randint(len(zeros))]
                h[i2] = 0.0
                h[j2] = 1.0
                ones = np.delete(ones, np.where(ones == i2)[0][0])
                zeros = np.delete(zeros, np.where(zeros == j2)[0][0])
                ones = np.append(ones, j2)
                zeros = np.append(zeros, i2)
            curr = corr_max_val(h)
        
        it += 1
        # Periodic rebuild to prevent index drift
        if it % 50000 == 0:
            ones = np.where(h == 1)[0].astype(np.int32)
            zeros = np.where(h == 0)[0].astype(np.int32)
    
    return best_h, best

def local_search_binary(h0, n, rng, max_iter=100000, deadline=None):
    """Accept-only local search (hill climbing) on binary sequence."""
    h = h0.copy()
    curr = corr_max_val(h)
    best = curr
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    it = 0
    
    while it < max_iter:
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        # Try a batch of random swaps
        for _ in range(200):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new = corr_max_val(h)
            if new < curr:
                curr = new
                idx_o = np.where(ones == i)[0][0]
                idx_z = np.where(zeros == j)[0][0]
                ones = np.delete(ones, idx_o)
                zeros = np.delete(zeros, idx_z)
                ones = np.append(ones, j)
                zeros = np.append(zeros, i)
                if curr < best:
                    best = curr
