import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_grad(x, beta):
    m = np.max(x)
    shifted = beta * (x - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_s = np.exp(shifted)
    s = np.sum(exp_s)
    p = exp_s / s
    val = m + np.log(s) / beta
    return val, p

def compute_grad(h, p, dx):
    """Analytic gradient of smooth_max(corr*dx, beta) w.r.t. h."""
    n = len(h)
    m_vals = np.arange(2*n-1) - (n-1)
    
    js = np.arange(n)[:, None]
    ms = m_vals[None, :]
    
    idx1 = js + ms
    mask1 = (idx1 >= 0) & (idx1 < n)
    h1 = np.where(mask1, h[np.clip(idx1, 0, n-1)], 0.0)
    
    idx2 = js - ms
    mask2 = (idx2 >= 0) & (idx2 < n)
    h2 = np.where(mask2, h[np.clip(idx2, 0, n-1)], 0.0)
    
    p_row = p[None, :]
    
    s1 = np.sum(p_row * (1.0 - h1) * mask1, axis=1)
    s2 = np.sum(p_row * h2 * mask2, axis=1)
    
    return dx * (s1 - s2)

def project_box_sum(h, target):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target}."""
    n = len(h)
    # Find lambda: sum(clip(h + lambda, 0, 1)) = target
    lo = -max(1.0, h.max()) - 1.0
    hi = 1.0 + max(0.0, target/n - h.min()) + 1.0
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    return np.clip(h + lam, 0.0, 1.0)

            val, p = smooth_max_grad(corr * dx, beta)

                vel = 0.9 * vel
                if lr < 1e-10:

            # Update best

            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
    
    return best_h, best_c5

def smooth_max_vals(x, beta):

def grad_smooth_max(h, p, dx):

            h_init2 = proj(h_init2, tgt)
            t_end_pg2 = time.time() + min(40, (budget_s * 0.97 - elapsed()))
            h_opt2, _ = pg_optimize(h_init2, tgt, 600.0, 3000, 0.05, t_end_pg2)
            try_update(h_opt2, n_try)

    # Final polish: coordinate descent on the best
    if elapsed() < budget_s * 0.98:
        t_end_final = time.time() + (budget_s * 0.98 - elapsed())
        h_fin, c5_fin = coord_descent(best_h, best_n / 2.0, 5, t_end_final)
        best_h = proj(h_fin, best_n / 2.0)
        best_c5 = compute_c5(best_h)

    # Ensure exact constraint
    tgt_final = best_n / 2.0
    best_h = proj(best_h, tgt_final)
    s = np.sum(best_h)
    if abs(s - tgt_final) > 1e-12:
        best_h += (tgt_final - s) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = proj(best_h, tgt_final)

    final_c5 = compute_c5(best_h)

    return (best_h, final_c5, best_n)
