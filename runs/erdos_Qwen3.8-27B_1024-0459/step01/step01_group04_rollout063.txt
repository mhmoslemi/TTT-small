        # If j-k_m valid: t*(1-h[j-k_m])
        # If j+k_m valid: h[j+k_m]*(1-t)

        coef = float(weights[idx]) if alpha >= 1e-12 else (1.0 / len(np.where(scores >= m - max(1e-8, 1e-6*max(1.0,m)))[0]))
        k_shift = k - (n - 1)
        for j in range(n):
            v = 0.0
            jm = j - k_shift
            jp = j + k_shift
            if 0 <= jm < n:
                v += (1.0 - h[jm])
            if 0 <= jp < n:
                v -= h[jp]
            grad[j] += coef * v
    return grad * dx

def slsqp_constraints(x):
    n = x.size
    return (float(np.sum(x)) - 0.5 * n) / n

def slsqp_constraint_jac(x):
    n = max(1, x.size)
    return np.full(n, 1.0 / n)

def run_slsqp(h0, alpha, maxiter, deadline):
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': slsqp_constraints, 'jac': slsqp_constraint_jac}]
    
    old_dl = _RUN_DEADLINE
    _RUN_DEADLINE = deadline
    try:
        res = minimize(
            lse_objective, h0.copy(), args=(alpha,),
            jac=lse_gradient,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': 0}
        )
    except BudgetError:
        res = None
    except Exception:
        res = None
    finally:
        _RUN_DEADLINE = old_dl
    
    if res is not None and hasattr(res, 'x') and res.x is not None:
        return project(np.asarray(res.x, dtype=float), n)
    return project(h0.copy(), n)

def coord_descent_pass(h, n):
    h = project(h, n)
    scores = get_scores(h)
    
    for j in range(n):
        old_val = float(h[j])
        
        # Compute c_m for all m
        c = np.empty(2 * n - 1)
        for m in range(2 * n - 1):
            k_shift = m - (n - 1)
            cm = 0.0
            jm = j - k_shift
            jp = j + k_shift
            if 0 <= jm < n:
                cm += (1.0 - h[jm])
            if 0 <= jp < n:
                cm -= h[jp]
            c[m] = cm
        
        # Find best t in [0,1] minimizing max_m (scores[m] + (t-old_val)*c[m])
        cur_max = scores.max()
        # Active scores (within 1.0 of max for n=70 scale)
        active = np.where(scores >= cur_max - 1.0)[0]
        
        candidate_ts = {0.0, 1.0, old_val}
        for ii in range(len(active)):
            for jj in range(ii + 1, len(active)):
                i, j2 = int(active[ii]), int(active[jj])
                denom = c[j2] - c[i]
                if abs(denom) > 1e-12:
                    tc = old_val + (scores[i] - scores[j2]) / denom
                    if 0.0 <= tc <= 1.0:
                        candidate_ts.add(tc)
        
        best_t = old_val
        best_val = np.inf
        for t in candidate_ts:
            new_s = scores + (t - old_val) * c
            vm = float(new_s.max())
            if vm < best_val:
                best_val = vm
                best_t = t
        
        h[j] = best_t
    
    return project(h, n)

def upsample(parent, n):
    n0 = parent.size
    if n == n0:
        return parent.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x1, x0, parent)
