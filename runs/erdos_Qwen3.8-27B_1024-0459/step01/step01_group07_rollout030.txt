import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def binary_c5(bits, n):
    """bits is a boolean array of length n with exactly n//2 True values."""
    h = bits.astype(float)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def simulated_annealing_binary(n, n_ones, max_iter, t0, t_end, seed):
    rng = np.random.RandomState(seed)
    # Initialize with random binary
    idx = rng.choice(n, size=n_ones, replace=False)
    bits = np.zeros(n, dtype=bool)
    bits[idx] = True
    h = bits.astype(float)
    best_val = compute_c5(h, n)
    best_h = h.copy()
    cur_val = best_val
    T = 0.05
    ones_positions = np.where(bits)[0].tolist()
    zeros_positions = np.where(~bits)[0].tolist()
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        # Pick random 1 and 0 to swap
        i = ones_positions[rng.randint(len(ones_positions))]
        j = zeros_positions[rng.randint(len(zeros_positions))]
        
        # Perform swap
        h[i], h[j] = h[j], h[i]
        new_val = compute_c5(h, n)
        
        # Accept or reject
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
            cur_val = new_val
            # Update position lists
            ones_positions[ones_positions.index(i)] = j
            zeros_positions[zeros_positions.index(j)] = i
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # Revert
            h[i], h[j] = h[j], h[i]
        
        # Cool down
        if it % 1000 == 999:
            T *= 0.95
        
        # Check constraints
        assert abs(np.sum(h) - n_ones) < 1e-10
    
    return best_h, best_val

    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = np.where(h < 1.0 - 1e-12)[0]
            if len(free) == 0:
                break
            inc = min(diff / len(free), 1.0 - h[free]).max() if len(free) > 0 else 0
            h[free] = np.minimum(h[free] + diff / len(free), 1.0)
        else:
            free = np.where(h > 1e-12)[0]
            if len(free) == 0:
                break
            h[free] = np.maximum(h[free] + diff / len(free), 0.0)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        h += (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def sa_optimize(n, time_limit, rng, h_init=None, T_init=0.012, T_min=0.0003, cool_rate=0.9997):
    n_ones = n // 2
    if h_init is not None:
        h = h_init.copy()
    else:
        h = np.zeros(n)
        h[rng.choice(n, size=n_ones, replace=False)] = 1.0
    cur_val = compute_c5(h, n)
    best_val = cur_val
    best_h = h.copy()
    T = T_init
    ones_pos = list(np.where(h == 1.0)[0])
    zeros_pos = list(np.where(h == 0.0)[0])
    t_start = time.time()
    stuck_count = 0
    while time.time() - t_start < time_limit and T > T_min:
        i = ones_pos[rng.randint(len(ones_pos))]
        j = zeros_pos[rng.randint(len(zeros_pos))]
        h[i], h[j] = h[j], h[i]
        new_val = compute_c5(h, n)
        accept = (new_val <= cur_val) or (rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-8)))
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                stuck_count = 0
            else:
                stuck_count += 1
            ipos = ones_pos.index(i)
            ones_pos[ipos] = j
            jpos = zeros_pos.index(j)
            zeros_pos[jpos] = i
        else:
            h[i], h[j] = h[j], h[i]
            stuck_count += 1
        if stuck_count > 3000:
            for _ in range(5):
                ii = rng.choice(ones_pos)
                jj = rng.choice(zeros_pos)
                h[ii], h[jj] = h[jj], h[ii]
                ipos = list(ones_pos).index(ii)
                ones_pos[ipos] = jj
                jpos = list(zeros_pos).index(jj)
                zeros_pos[jpos] = ii
            ones_pos = list(np.where(h == 1.0)[0])
            zeros_pos = list(np.where(h == 0.0)[0])
            cur_val = compute_c5(h, n)
            stuck_count = 0
        T *= cool_rate
    return best_h, best_val

def greedy_improve(h, n, rng, max_sweeps=300):
    cur_val = compute_c5(h, n)
    h = h.copy()
    ones_pos = list(np.where(h == 1.0)[0])
    zeros_pos = list(np.where(h == 0.0)[0])
    for sw in range(max_sweeps):
        improved = False
        n_trials = min(500, len(ones_pos) * len(zeros_pos))
        for _ in range(n_trials):
            i = ones_pos[rng.randint(len(ones_pos))]
            j = zeros_pos[rng.randint(len(zeros_pos))]
            h[i], h[j] = h[j], h[i]
            v = compute_c5(h, n)
            if v < cur_val:
                cur_val = v
                ipos = ones_pos.index(i)
                ones_pos[ipos] = j
                jpos = zeros_pos.index(j)
                zeros_pos[jpos] = i
                improved = True
            else:
                h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return h, cur_val

def random_greedy_init(n, n_ones, rng):
    h = np.zeros(n)
    chosen = []
    for step in range(n_ones):
        best_pos = -1
        best_val = float('inf')
        candidates = [i for i in range(n) if i not in set(chosen)]
        rng.shuffle(candidates)
        for c in candidates:
            h[c] = 1.0
            v = compute_c5(h, n)
            h[c] = 0.0
            if v < best_val:
                best_val = v
                best_pos = c
        h[best_pos] = 1.0
        chosen.append(best_pos)
    return h

def slsqp_polish(h0, n, maxiter=200, ftol=1e-12):
    global _g_n
    _g_n = int(n)
    target = n / 2.0
    h0 = np.array(h0, dtype=float).copy()
    h0 = np.clip(h0, 0, 1)
    h0 += (target - np.sum(h0)) / n
    h0 = np.clip(h0, 0, 1)
    cons = [{'type': 'eq', 'fun': _g_cons_fun, 'jac': _g_cons_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_g_obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        x = np.clip(res.x, 0, 1)
        x += (target - np.sum(x)) / n
        x = np.clip(x, 0, 1)
        return x, compute_c5(x, n)
    except Exception:
        return h0, compute_c5(h0, n)

def thue_morse_init(n, n_ones, rng):
    tm = np.zeros(n, dtype=int)
    for k in range(1, n):
        tm[k] = tm[k >> 1] ^ (k & 1)
    h = (tm == 1).astype(float)
    if np.sum(h) != n_ones:
        h = project_binary(n, n_ones, h + rng.normal(0, 0.1, n))
    return h

def golden_ratio_init(n, n_ones, rng):
    golden = (1.0 + 5.0 ** 0.5) / 2.0
    h = np.zeros(n)
    for k in range(n_ones):
        pos = int(np.floor(((k + 0.5) * golden * n_ones) / n * n)) % n
        if h[pos] == 0:
            h[pos] = 1.0
    cnt = int(np.sum(h))
    if cnt < n_ones:
        remaining = [i for i in range(n) if h[i] == 0]
        extra = rng.choice(remaining, size=n_ones - cnt, replace=False)
        h[extra] = 1.0
    elif cnt > n_ones:
        ones = list(np.where(h == 1)[0])
        remove = rng.choice(ones, size=cnt - n_ones, replace=False)
        h[remove] = 0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    best_h = None
    best_val = float('inf')
    best_n = 0

    # Load parent
    try:
        parent = np.array(initial_h_values, dtype=float)
        best_n = len(parent)
        best_val = compute_c5(parent, best_n)
        best_h = parent.copy()
    except Exception:
        best_n = 70
        best_h = np.full(best_n, 0.5)
        best_val = compute_c5(best_h, best_n)

    n_main = best_n if best_n % 2 == 0 else best_n + 1
    n_ones_main = n_main // 2

    # Phase 1: Binary SA at main resolution (multiple restarts, diverse inits)
    n_sa = 10
    allocated = min(500, max(80, int(time_left() * 0.45)))
    sa_time_each = max(8, allocated // n_sa)

    inits = []
    inits.append(project_binary(n_main, n_ones_main, best_h))
    inits.append(golden_ratio_init(n_main, n_ones_main, rng))
    inits.append(thue_morse_init(n_main, n_ones_main, rng))
    rg = random_greedy_init(n_main, n_ones_main, rng)
    inits.append(rg)
    for _ in range(6):
        h_r = np.zeros(n_main)
        h_r[rng.choice(n_main, size=n_ones_main, replace=False)] = 1.0
        inits.append(h_r)

    for idx in range(len(inits)):
        if time_left() < 180:
            break
        tl = min(sa_time_each, max(5, time_left() - 180))
        if tl < 5:
            break
        h_try, v_try = sa_optimize(n_main, tl, rng, h_init=inits[idx].copy(),
                                    T_init=0.012, T_min=0.0003, cool_rate=0.9997)
        if v_try < best_val:
            best_val = v_try
            best_h = h_try.copy()
            best_n = n_main

    # Phase 2: Greedy improvement
    if time_left() > 50:
        h_try, v_try = greedy_improve(best_h, best_n, rng, max_sweeps=500)
        if v_try < best_val:
            best_val = v_try
            best_h = h_try

    # Phase 3: More SA with adaptive params
    if time_left() > 120:
        for attempt in range(6):
            if time_left() < 100:
                break
            h_pert = best_h + rng.normal(0, 0.15, best_n)
            h_pert = project_binary(best_n, best_n // 2, h_pert)
            tl = min(40, max(5, time_left() - 100))
            h_try, v_try = sa_optimize(best_n, tl, rng, h_init=h_pert,
                                        T_init=0.008, T_min=0.0002, cool_rate=0.9998)
            if v_try < best_val:
                best_val = v_try
                best_h = h_try.copy()
            h_try2, v_try2 = greedy_improve(h_try, best_n, rng, max_sweeps=200)
            if v_try2 < best_val:
                best_val = v_try2
                best_h = h_try2

    # Phase 4: Other resolutions
    other_ns = [n_main + 10, n_main - 10, n_main + 20, n_main + 40, 100, 120]
    other_ns = sorted(set([x for x in other_ns if x >= 40 and x % 2 == 0 and x != n_main]))
    for n_try in other_ns[:4]:
        if time_left() < 80:
            break
        n_t = n_try
        n_ones_t = n_t // 2
        x_best = np.linspace(0, 2, best_n, endpoint=False)
        x_try = np.linspace(0, 2, n_t, endpoint=False)
        h_interp = np.interp(x_try, x_best, best_h)
        h_init_t = project_binary(n_t, n_ones_t, h_interp)
        tl = min(50, max(10, time_left() - 60))
        h_try, v_try = sa_optimize(n_t, tl, rng, h_init=h_init_t,
                                    T_init=0.01, T_min=0.0003, cool_rate=0.9997)
        if v_try < best_val:
            best_val = v_try
            best_h = h_try.copy()
            best_n = n_t
        if time_left() > 40:
            h_rand = np.zeros(n_t)
            h_rand[rng.choice(n_t, size=n_ones_t, replace=False)] = 1.0
            h_try, v_try = sa_optimize(n_t, 25, rng, h_init=h_rand,
                                        T_init=0.01, T_min=0.0003, cool_rate=0.9997)
            if v_try < best_val:
                best_val = v_try
                best_h = h_try.copy()
                best_n = n_t

    # Phase 5: SLSQP polish (allow continuous values near binary)
    if time_left() > 40:
        h_cont = best_h.astype(float).copy()
        h_cont = np.clip(h_cont + rng.normal(0, 0.03, best_n), 0, 1)
        target = best_n / 2.0
        h_cont += (target - np.sum(h_cont)) / best_n
        h_cont = np.clip(h_cont, 0, 1)
        h_try, v_try = slsqp_polish(h_cont, best_n, maxiter=300, ftol=1e-13)
        if v_try < best_val:
            best_val = v_try
            best_h = h_try.copy()

        # Also polish the pure binary
        if time_left() > 30:
            h_bin = project_binary(best_n, best_n // 2, best_h)
            h_bin_f = h_bin.astype(float)
            h_bin_f += rng.normal(0, 0.01, best_n)
            h_bin_f = np.clip(h_bin_f, 0, 1)
            h_bin_f += (best_n / 2.0 - np.sum(h_bin_f)) / best_n
            h_bin_f = np.clip(h_bin_f, 0, 1)
            h_try, v_try = slsqp_polish(h_bin_f, best_n, maxiter=200, ftol=1e-13)
            if v_try < best_val:
                best_val = v_try
                best_h = h_try.copy()

    # Phase 6: Final binary polish
    if time_left() > 30:
        h_bin = project_binary(best_n, best_n // 2, best_h)
        h_try, v_try = sa_optimize(best_n, min(25, max(5, time_left() - 20)), rng,
                                    h_init=h_bin, T_init=0.005, T_min=0.0002, cool_rate=0.9998)
        if v_try < best_val:
            best_val = v_try
            best_h = h_try.copy()
        h_try, v_try = greedy_improve(best_h, best_n, rng, max_sweeps=300)
        if v_try < best_val:
            best_val = v_try
            best_h = h_try.copy()

    # Final feasibility check
    best_h = np.clip(np.array(best_h, dtype=float), 0.0, 1.0)
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-12:
        if diff > 0:
            free = np.where(best_h < 1.0 - 1e-10)[0]
            if len(free) > 0:
                best_h[free] += diff / len(free)
        else:
            free = np.where(best_h > 1e-10)[0]
            if len(free) > 0:
                best_h[free] += diff / len(free)
        best_h = np.clip(best_h, 0.0, 1.0)

    best_val = compute_c5(best_h, best_n)
    return (best_h, best_val, best_n)
