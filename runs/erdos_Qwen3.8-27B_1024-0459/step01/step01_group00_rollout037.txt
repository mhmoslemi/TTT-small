import numpy as np
from scipy.optimize import minimize

_opt_state = {'n': 0, 'beta': 10.0, 'penalty': 1000.0}

def _objective(x):
    n = _opt_state['n']
    beta = _opt_state['beta']
    penalty = _opt_state['penalty']
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    smax = cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))
    sd = np.sum(x) - n / 2.0
    return smax + penalty * sd * sd

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lbfqs_optimize(h, n, betas, penalty=5000.0, max_iter=2000):
    current = make_feasible(h.copy(), n)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        _opt_state['n'] = n

        _opt_state['penalty'] = penalty

            res = minimize(_objective, current, method='L-BFGS-B', bounds=bounds,
