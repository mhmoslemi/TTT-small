import time
import math
from scipy.optimize import minimize

_RUN_DEADLINE = None

class BudgetError(Exception):

def compute_c5(h):
    n = h.size
    h = np.clip(h, 0, 1)
    conv = np.correlate(h, 1-h, mode="full")
    return float(np.max(conv) * (2.0/n))

def compute_scores(h, n):
    """Compute all overlap scores efficiently."""
    scores = np.empty(2*n-1)
    # Zero shift
    scores[n-1] = float(np.sum(h * (1-h)))
    # Positive shifts d=1..n-1
    for d in range(1, n):
        m = n - d
        scores[n-1-d] = float(np.dot(h[:m], 1-h[d:]))  # shift +d
        scores[n-1+d] = float(np.dot(h[d:], 1-h[:m]))   # shift -d
    return scores

def smooth_max(scores, alpha):
    if alpha < 1e-10:
        return float(np.max(scores)), np.zeros_like(scores)
    s = scores - np.max(scores)
    e = np.exp(s / alpha)
    A = float(np.sum(e))
    val = float(np.max(scores)) + alpha * math.log(A)
    weights = e / A
    return val, weights

def smooth_max_grad(scores, weights, h, n):
    """Compute gradient of smooth max w.r.t. h."""
    grad = np.zeros(n)
    # Zero shift contribution
    idx0 = n - 1
    if weights[idx0] > 1e-15:
        grad += weights[idx0] * (1 - 2*h)
    # Cross shifts
    for d in range(1, n):
        i_pos = n - 1 - d  # index for +d shift
        i_neg = n - 1 + d  # index for -d shift
        wp = weights[i_pos]
        wn = weights[i_neg]
        m = n - d
        if wp > 1e-15:
            # ∂/∂h[i] of sum_{j=0}^{m-1} h[j]*(1-h[j+d])
            # = (1-h[i+d]) for i < m, and -h[i-d] for i >= d
            grad[:m] += wp * (1 - h[d:])
            grad[d:] += wp * (-h[:m])
        if wn > 1e-15:
            # ∂/∂h[i] of sum_{j=0}^{m-1} h[j+d]*(1-h[j])
            # = (1-h[i-d]) for i >= d, and -h[i+d] for i < m
            grad[d:] += wn * (1 - h[:m])
            grad[:m] += wn * (-h[d:])
    return grad

    """Project y onto {h: 0<=h<=1, sum(h)=n/2}."""
    target = 0.5 * n

    hi = max(1.0, 1.0 - y.min()) + 1.0

def _c5(h):
    n = h.size
    h = np.clip(h, 0.0, 1.0)
    cv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(cv) * (2.0 / n))


def _sc(h, n):
    s = np.empty(2 * n - 1, dtype=float)
    s[n - 1] = float(np.dot(h, 1.0 - h))
    for d in range(1, n):
        m = n - d
        s[n - 1 - d] = float(np.dot(h[:m], 1.0 - h[d:]))
        s[n - 1 + d] = float(np.dot(h[d:], 1.0 - h[:m]))
    return s


def _proj(y, n):

def _sgrad(s, w, h, n):
