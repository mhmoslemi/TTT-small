def compute_c5_exact(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n == 0:
        return 1e6
    h = np.clip(np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return 1e6
    v = float(np.max(corr) * (2.0 / float(n)))
    if not math.isfinite(v):
        return 1e6
    return v


def quick_c5(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n == 0:
        return 1e6
    h = np.clip(np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0), 0.0, 1.0)
    _, m = compute_scores(h, n)
    v = m * (2.0 / float(n))
    if not math.isfinite(v):
        return 1e6
    return float(v)


def is_valid_candidate(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n < 2:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if float(h.min()) < -1e-7 or float(h.max()) > 1.0 + 1e-7:
        return False
    if abs(float(h.sum()) - 0.5 * float(n)) > 1e-5:
        return False
    return True


def as_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    if h.size != n:
        tmp = np.zeros(n, dtype=float)
        m = min(n, h.size)
        if m > 0:
            tmp[:m] = h[:m]
        h = tmp
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    for _ in range(12):
        diff = target - float(h.sum())
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            mask = h < 1.0
            if not np.any(mask):
                break
            cnt = float(np.count_nonzero(mask))
            per = diff / cnt
            inc = min(per, float(1.0 - h[mask].min()))
            if inc <= 1e-15:
                break
            h[mask] += inc
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            cnt = float(np.count_nonzero(mask))
            per = -diff / cnt
            dec = min(per, float(h[mask].min()))
            if dec <= 1e-15:
                break
            h[mask] -= dec
    diff = target - float(h.sum())
    if abs(diff) > 1e-14:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return np.clip(h, 0.0, 1.0)


def upsample_linear(parent, n):
    parent = np.asarray(parent, dtype=float)
    n = int(n)
    n0 = int(parent.size)
    if n0 == 0:
        return np.full(n, 0.5, dtype=float)
    if n == n0:
        return parent.astype(float, copy=True)
    x0 = (np.arange(n0, dtype=float) + 0.5) * (2.0 / float(n0))
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    return np.interp(x, x0, parent.astype(float)).astype(float)


def smooth_downsample(h, n):
    parent = np.asarray(h, dtype=float)
    n = int(n)
    n0 = int(parent.size)
    if n0 == 0 or n == 0:
        return np.zeros(n, dtype=float)
    if n == n0:
        return parent.astype(float, copy=True)
    block = n0 // n
    if block <= 1:
        return upsample_linear(parent, n)
    if n0 % n != 0:
        p = np.zeros(n * block, dtype=float)
        p[:min(n0, n * block)] = parent[:min(n0, n * block)]
        parent = p
    return parent.reshape(n, block).mean(axis=1)


def compute_scores(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    if h.size != n:
        h = as_feasible(h, n)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    if n <= 0:
        return np.zeros(0, dtype=float), 1e6
    total = float(h.sum())
    scores = np.empty(max(1, 2 * n - 1), dtype=float)
    if n == 1:
        s0 = total - float(np.dot(h, h))
        scores[0] = s0
        m = float(s0)
        return scores, m
    scores[0] = total - float(np.dot(h, h))
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    if n <= 256:
        for d in range(1, n):
            prod = float(np.dot(h[:n - d], h[d:]))
            scores[2 * d - 1] = pref[n - d] - prod
            scores[2 * d] = total - pref[d] - prod
    else:
        L = 1 << (2 * n - 1).bit_length()
        H = np.fft.rfft(h, L)
        ac = np.fft.irfft(H * np.conj(H), L)[:n - 1]
        ac = np.maximum(ac, 0.0)
        for d in range(1, n):
            prod = float(ac[d - 1])
            scores[2 * d - 1] = pref[n - d] - prod
            scores[2 * d] = total - pref[d] - prod
    m = float(np.max(scores))
    if not math.isfinite(m):
        scores.fill(1e6)
        m = 1e6
    return scores, m


def compute_weights(scores, m, alpha, n):
    w = np.zeros(max(1, 2 * n - 1), dtype=float)
    if scores.size != w.size:
        return w
    if alpha <= 1e-12:
        tol = max(1e-10, 1e-6 * max(1.0, m))
        active = np.flatnonzero(scores >= m - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(scores))], dtype=int)
        w[active] = 1.0 / float(active.size)
        return w
    shifted = (scores - m) / alpha
    mask = shifted > -50.0
    if not np.any(mask):
        w[int(np.argmax(scores))] = 1.0
        return w
    e = np.exp(shifted[mask])
    A = float(e.sum())
    if not math.isfinite(A) or A <= 0.0:
        w[int(np.argmax(scores))] = 1.0
        return w
    w[mask] = e / A
    return w


def smooth_value_scaled(h, n, alpha):
    check_deadline()
    h = np.asarray(h, dtype=float)
    n = int(n)
    if h.size != n:
        h = as_feasible(h, n)
    h = np.clip(np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    scores, m = compute_scores(h, n)
    if m >= 1e5:
        return 1e6
    if alpha <= 1e-12:
        val = m
    else:
        shifted = (scores - m) / alpha
        mask = shifted > -50.0
        if not np.any(mask):
            val = m
        else:
            A = float(np.exp(shifted[mask]).sum())
            if not math.isfinite(A) or A <= 0.0:
                val = m
            else:
                val = m + alpha * math.log(A)
    if not math.isfinite(val):
        return 1e6
    return float(val * (2.0 / float(n)))


def weighted_gradient(h, n, weights):
    h = np.asarray(h, dtype=float)
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=float)
    if h.size != n:
        h = as_feasible(h, n)
    h = np.clip(np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    if weights.size != 2 * n - 1:
        return np.zeros(n, dtype=float)
    w0 = float(weights[0])
    a = weights[1:2 * n - 1:2]
    b = weights[2:2 * n:2]
    c = a + b

    PA = np.empty(n, dtype=float)
    PA[:n - 1] = np.cumsum(a)[::-1]
    PA[n - 1] = 0.0
    PB = np.empty(n, dtype=float)
    PB[0] = 0.0
    PB[1:] = np.cumsum(b)
    linear = PA + PB

    quad = np.zeros(n, dtype=float)
    nz = np.nonzero(c)[0] + 1
    if nz.size > 0:
        if nz.size <= max(10, n // 5):
            for d in nz:
                d = int(d)
                val = float(c[d - 1])
                if val == 0.0:
                    continue
                nd = n - d
                if nd <= 0:
                    continue
                quad[:nd] += val * h[d:]
                quad[d:] += val * h[:nd]
        else:
            L = 1 << (2 * n - 1).bit_length()
            G = np.zeros(L, dtype=float)
            darr = np.arange(1, n, dtype=int)
            G[darr] = c
            G[L - darr] = c
            H = np.fft.rfft(h, L)
            GF = np.fft.rfft(G, L)
            quad = np.real(np.fft.irfft(H * GF, L))[:n]

    grad = w0 * (1.0 - 2.0 * h) + linear - quad
    if not np.all(np.isfinite(grad)):
        return np.zeros(n, dtype=float)
    return grad


def smooth_gradient(h, n, alpha):
    check_deadline()
    h = np.asarray(h, dtype=float)
    n = int(n)
    if h.size != n:
        h = as_feasible(h, n)
    h = np.clip(np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    scores, m = compute_scores(h, n)
    if m >= 1e5:
        return np.zeros(n, dtype=float)
    w = compute_weights(scores, m, alpha, n)
    g = weighted_gradient(h, n, w)
    g *= (2.0 / float(n))
    if not np.all(np.isfinite(g)):
        return np.zeros(n, dtype=float)
    return g


def make_anti_h(g):
    g = np.asarray(g, dtype=float)
    m = int(g.size)
    if m <= 0:
        return np.zeros(0, dtype=float)
    g = np.clip(np.nan_to_num(g, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    h = np.empty(2 * m, dtype=float)
    h[:m] = g
    h[m:] = 1.0 - g
    return h


def anti_quick_c(g):
    h = make_anti_h(g)
    return quick_c5(h)


def project_to_anti_g(h, m):
    h = np.asarray(h, dtype=float)
    m = int(m)
    if h.size == 0 or m <= 0:
        return np.full(m, 0.5, dtype=float)
    H = upsample_linear(h, 2 * m)
    g = 0.5 * (H[:m] + 1.0 - H[m:])
    return np.clip(np.nan_to_num(g, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)


def anti_objective(g, alpha, m):
    check_deadline()
    g = np.asarray(g, dtype=float)
    if g.size != m:
        g = as_feasible(g, m)
    g = np.clip(np.nan_to_num(g, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    h = make_anti_h(g)
    return smooth_value_scaled(h, 2 * m, alpha)


def anti_gradient(g, alpha, m):
    check_deadline()
    g = np.asarray(g, dtype=float)
    if g.size != m:
        g = as_feasible(g, m)
    g = np.clip(np.nan_to_num(g, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    h = make_anti_h(g)
    gh = smooth_gradient(h, 2 * m, alpha)
    out = gh[:m] - gh[m:]
    if not np.all(np.isfinite(out)):
        return np.zeros(m, dtype=float)
    return out


def penalty_objective(h, alpha, lam, n):
    check_deadline()
    h = np.asarray(h, dtype=float)
    if h.size != n:
        h = as_feasible(h, n)
    h = np.clip(np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    val = smooth_value_scaled(h, n, alpha)
    err = float(h.sum()) - 0.5 * float(n)
    val += float(lam) * err * err
    if not math.isfinite(val):
        return 1e6
    return float(val)


def penalty_gradient(h, alpha, lam, n):
    check_deadline()
    h = np.asarray(h, dtype=float)
    if h.size != n:
        h = as_feasible(h, n)
    h = np.clip(np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    grad = smooth_gradient(h, n, alpha)
    err = float(h.sum()) - 0.5 * float(n)
    grad += (2.0 * float(lam) * err) * np.ones(n, dtype=float)
    if not np.all(np.isfinite(grad)):
        return np.zeros(n, dtype=float)
    return grad


def slsqp_objective(h, alpha, n):
    return smooth_value_scaled(h, n, alpha)


def slsqp_gradient(h, alpha, n):
    return smooth_gradient(h, n, alpha)


def slsqp_eq(x):
    n = max(1, int(x.size))
    return float(x.sum()) - 0.5 * float(n)


def slsqp_eq_jac(x):
    n = max(1, int(x.size))
    return np.ones(n, dtype=float)

    except Exception:

class OptimizerState:
    def __init__(self, h, c, n):
        self.h = np.asarray(h, dtype=float).copy()
        self.c = float(c)
        self.n = int(n)


def update_state(state, h):
    if h is None:
        return
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n < 2:
        return
    if not np.all(np.isfinite(h)):
        return
    hf = as_feasible(h, n)
    c = quick_c5(hf)
    if math.isfinite(c) and c < float(state.c) - 1e-14:
        state.h = hf
        state.c = float(c)
        state.n = n


def is_same_g(a, b):
    if a.size != b.size:
        return False
    return bool(np.allclose(a, b, rtol=1e-7, atol=1e-7))


def add_anti_start(g, cands):
    g = np.asarray(g, dtype=float)
    if g.ndim != 1 or g.size <= 1:
        return
    g = np.clip(np.nan_to_num(g, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    c = anti_quick_c(g)
    if not math.isfinite(c):
        return
    for _, gg in cands:
        if is_same_g(g, gg):
            return
    cands.append((float(c), g.astype(float, copy=True)))


def random_anti_g(m, rng):
    m = int(m)
    x = (np.arange(m, dtype=float) + 0.5) / float(m)
    g = np.full(m, 0.5, dtype=float)
    for k in range(1, 6):
        amp = 0.32 / float(k) * float(rng.uniform(0.15, 1.0))
        ph = float(rng.uniform(0.0, 2.0 * math.pi))
        g += amp * np.sin(2.0 * math.pi * float(k) * x + ph)
        if rng.random() < 0.35:
            g += amp * 0.5 * np.cos(2.0 * math.pi * float(k) * x + ph)
    return np.clip(np.nan_to_num(g, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)


def base_anti_g_list(m):
    m = int(m)
    x = (np.arange(m, dtype=float) + 0.5) / float(m)
    out = []
    out.append(1.0 - x)
    out.append(x)
    out.append(0.5 + 0.5 * np.cos(math.pi * x))
    out.append(0.5 - 0.5 * np.cos(math.pi * x))
    out.append(0.5 + 0.5 * np.sin(math.pi * x))
    out.append(0.5 - 0.5 * np.sin(math.pi * x))
    out.append(0.5 + 0.5 * np.sin(2.0 * math.pi * x - 0.5 * math.pi))
    out.append(0.5 + 0.5 * np.cos(2.0 * math.pi * x))
    for p in (0.6, 0.9, 1.4, 2.0):
        out.append(1.0 - np.power(x, p))
        out.append(np.power(1.0 - x, p))
    out.append(np.clip(1.0 - 3.0 * x, 0.0, 1.0))
    out.append(np.clip(3.0 * x, 0.0, 1.0))
    out.append(0.5 + 0.5 * np.tanh(3.0 * (0.5 - x)))
    out.append(0.5 + 0.5 * np.tanh(6.0 * (0.5 - x)))
    return out


def anti_starts(m, parent_h, state_h, best_anti_g, rng, limit=3):
    cands = []
    for g in base_anti_g_list(m):
        add_anti_start(g, cands)
    if parent_h is not None and parent_h.size > 0:
        add_anti_start(project_to_anti_g(parent_h, m), cands)
    if state_h is not None and state_h.size > 0:
        add_anti_start(project_to_anti_g(state_h, m), cands)
    if best_anti_g is not None and best_anti_g.size > 0:
        if best_anti_g.size == m:
            add_anti_start(best_anti_g, cands)
        else:
            add_anti_start(upsample_linear(best_anti_g, m), cands)
    for _ in range(20):
        add_anti_start(random_anti_g(m, rng), cands)
    try:
        cands.sort(key=lambda zz: zz[0])
    except Exception:
        pass
    if limit > 0 and len(cands) > limit:
        cands = cands[:limit]
    return [g for _, g in cands]


def run_anti_family(m, parent_h, state, best_anti_g, best_anti_c, rng, time_limit):
    global _RUN_DEADLINE
    if time_limit <= 10.0:
        return None, 1e9
    m = int(m)
    n = 2 * m
    m_start = time.time()
    m_deadline = m_start + float(time_limit)
    starts = anti_starts(m, parent_h, state.h, best_anti_g, rng,
                         limit=4 if m <= 80 else (2 if m <= 180 else 1))
    if not starts:
        return None, 1e9

    bounds = [(0.0, 1.0)] * m
    maxit_base = max(40, min(220, int(9000.0 / max(32.0, float(m)))))

    factors = (0.14, 0.05, 0.018)
    temps = [max(0.5, factor * float(n)) for factor in factors]
    temps.append(0.0)

    local_best_g = None
    local_best_c = 1e9

    for g0 in starts:
        if time.time() >= m_deadline - 3.0:
            break
        g = np.clip(np.asarray(g0, dtype=float).copy(), 0.0, 1.0)
        c0 = anti_quick_c(g)
        if math.isfinite(c0) and c0 < local_best_c:
            local_best_c = c0
            local_best_g = g.copy()
        if math.isfinite(c0) and c0 < best_anti_c:
            best_anti_c = c0
            best_anti_g = g.copy()
        if math.isfinite(c0):
            h = make_anti_h(g)
            update_state(state, h)

        no_improve_stages = 0
        for alpha in temps:
            if time.time() >= m_deadline - 3.0:
                break
            maxit = maxit_base
            if alpha == 0.0:
                maxit = max(20, int(0.65 * maxit))
            old_deadline = _RUN_DEADLINE
            _RUN_DEADLINE = min(_RUN_DEADLINE, m_deadline) if _RUN_DEADLINE is not None else m_deadline
            try:
                res = minimize(
                    anti_objective,
                    g,
                    args=(alpha, m),
                    jac=anti_gradient,
                    method="L-BFGS-B",
                    bounds=bounds,
                    options={"maxiter": int(maxit), "ftol": 1e-12, "disp": 0, "maxls": 24},
                )
            except Exception:
                res = None
            finally:
                _RUN_DEADLINE = old_deadline
            if res is not None and getattr(res, "x", None) is not None:
                g_new = np.clip(np.asarray(res.x, dtype=float), 0.0, 1.0)
                c_new = anti_quick_c(g_new)
                if math.isfinite(c_new):
                    if c_new < local_best_c - 1e-14:
                        local_best_c = c_new
                        local_best_g = g_new.copy()
                        no_improve_stages = 0
                    else:
                        no_improve_stages += 1
                    if c_new < best_anti_c:
                        best_anti_c = c_new
                        best_anti_g = g_new.copy()
                    h_new = make_anti_h(g_new)
                    update_state(state, h_new)
                    g = g_new
                else:
                    no_improve_stages += 2
            else:
                no_improve_stages += 2
            if no_improve_stages >= 3:
                break

    return local_best_g, local_best_c


def optimize_penalty(hstart, n, state, deadline, time_limit):
    global _RUN_DEADLINE
    if hstart is None or time_limit <= 15.0:
        return
    n = int(n)
    if n < 4:
        return
    d0 = time.time()
    d_end = min(float(deadline), d0 + float(time_limit))

    x0 = as_feasible(np.asarray(hstart, dtype=float), n)
    update_state(state, x0)
    x_best = state.h.copy()
    c_best = float(state.c)

    bounds = [(0.0, 1.0)] * n
    stages = [
        (max(0.5, 0.06 * float(n)), 0.7),
        (max(0.5, 0.022 * float(n)), 1.4),
        (max(0.3, 0.008 * float(n)), 3.0),
        (0.0, 6.0),
    ]
    maxit_base = max(35, min(160, int(9000.0 / max(48.0, float(n)))))

    for alpha, lam in stages:
        if time.time() >= d_end - 3.0:
            break
        maxit = maxit_base
        if alpha == 0.0:
            maxit = max(20, int(0.6 * maxit))
        old = _RUN_DEADLINE
        _RUN_DEADLINE = min(_RUN_DEADLINE, d_end) if _RUN_DEADLINE is not None else d_end
        try:
            res = minimize(
                penalty_objective,
                x0,
                args=(alpha, lam, n),
                jac=penalty_gradient,
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": int(maxit), "ftol": 1e-12, "disp": 0, "maxls": 24},
            )
        except Exception:
            res = None
        finally:
            _RUN_DEADLINE = old
        if res is not None and getattr(res, "x", None) is not None:
            xf = as_feasible(res.x, n)
            cx = quick_c5(xf)
            update_state(state, xf)
            if math.isfinite(cx) and cx < c_best - 1e-14:
                c_best = cx
                x_best = state.h.copy()
            x0 = x_best
        else:
            x0 = x_best


def optimize_slsqp(hstart, n, state, deadline, time_limit):
    global _RUN_DEADLINE
    if hstart is None or time_limit <= 35.0:
        return
    n = int(n)
    if n < 4 or n > 240:
        return
    d0 = time.time()
    d_end = min(float(deadline), d0 + float(time_limit))

    x0 = as_feasible(np.asarray(hstart, dtype=float), n)
    update_state(state, x0)
    x_best = state.h.copy()

    bounds = [(0.0, 1.0)] * n
    constraints = ({
        "type": "eq",
        "fun": slsqp_eq,
        "jac": slsqp_eq_jac,
    },)
    stages = [
        max(0.5, 0.05 * float(n)),
        max(0.3, 0.018 * float(n)),
        max(0.2, 0.006 * float(n)),
        0.0,
    ]
    maxit_base = max(30, min(150, int(7000.0 / max(48.0, float(n)))))

    for alpha in stages:
        if time.time() >= d_end - 3.0:
            break
        maxit = maxit_base
        if alpha == 0.0:
            maxit = max(15, int(0.55 * maxit))
        old = _RUN_DEADLINE
        _RUN_DEADLINE = min(_RUN_DEADLINE, d_end) if _RUN_DEADLINE is not None else d_end
        try:
            res = minimize(
                slsqp_objective,
                x_best,
                args=(alpha, n),
                jac=slsqp_gradient,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": int(maxit), "ftol": 1e-10, "disp": 0},
            )
        except Exception:
            res = None
        finally:
            _RUN_DEADLINE = old
        if res is not None and getattr(res, "x", None) is not None:
            xf = as_feasible(res.x, n)
            update_state(state, xf)
            x_best = state.h


def anti_coordinate_polish(g, deadline, time_limit, rng):
    global _RUN_DEADLINE
    if g is None or time_limit <= 5.0:
        return None, 1e9
    g = np.clip(np.asarray(g, dtype=float).copy(), 0.0, 1.0)
    m = int(g.size)
    d_end = min(float(deadline), time.time() + float(time_limit))
    c = anti_quick_c(g)
    best_g = g.copy()
    best_c = c
    step1 = 0.06
    step2 = 0.02
    old = _RUN_DEADLINE
    _RUN_DEADLINE = min(_RUN_DEADLINE, d_end) if _RUN_DEADLINE is not None else d_end
    try:
        order = np.arange(m, dtype=int)
        it = 0
        while time.time() < d_end - 1.0 and it < 2500:
            it += 1
            i = int(order[it % m])
            for delta in (-step1, -step2, step1, step2):
                g2 = g.copy()
                g2[i] = min(1.0, max(0.0, float(g2[i]) + delta))
                c2 = anti_quick_c(g2)
                if math.isfinite(c2) and c2 < best_c - 1e-13:
                    g = g2
                    best_c = c2
                    best_g = g2.copy()
                    break
            if it % 40 == 0:
                step1 = max(0.006, step1 * 0.82)
                step2 = max(0.0015, step2 * 0.82)
    finally:
        _RUN_DEADLINE = old
    return best_g, best_c


def run(seed=42, budget_s=1300, **kwargs):
    global _RUN_DEADLINE
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0.0:
        budget_s = 1300.0
    reserve = 30.0
    global_deadline = t0 + max(1.0, budget_s - reserve)
    _RUN_DEADLINE = global_deadline

    rng = np.random.default_rng(int(seed) if seed is not None else 42)

    parent = get_parent()
    if parent is not None and parent.size >= 2:
        n0 = int(parent.size)
        h0 = as_feasible(parent, n0)
    else:
        m0 = 70
        g0 = 1.0 - ((np.arange(m0, dtype=float) + 0.5) / float(m0))
        h0 = make_anti_h(g0)
        n0 = int(h0.size)

    c0 = quick_c5(h0)
    state = OptimizerState(h0.copy(), c0, n0)

    best_anti_g = project_to_anti_g(state.h, max(2, n0 // 2))
    best_anti_c = anti_quick_c(best_anti_g)
    if best_anti_c < state.c:
        update_state(state, make_anti_h(best_anti_g))

    base_m = max(2, state.n // 2)
    m_candidates = set()
    for m in (base_m // 2, base_m, 2 * base_m, 4 * base_m, 35, 70, 140, 280):
        try:
            mm = int(m)
        except Exception:
            continue
        if 18 <= mm <= 280:
            m_candidates.add(mm)
    m_list = sorted(m_candidates)
    if not m_list:
        m_list = [35, 70, 140, 280]

    m_weights = []
    for m in m_list:
        if m <= 45:
            m_weights.append(0.08)
        elif m <= 85:
            m_weights.append(0.24)
        elif m <= 170:
            m_weights.append(0.46)
        else:
            m_weights.append(0.22)
    total_w = float(sum(m_weights))
    if total_w <= 0.0:
        total_w = 1.0

    anti_budget = max(60.0, global_deadline - t0 - 300.0)
    for idx, m in enumerate(m_list):
        if time.time() >= global_deadline - 120.0:
            break
        frac = (m_weights[idx] / total_w)
        time_limit = frac * anti_budget
        time_limit = min(time_limit, max(10.0, global_deadline - time.time() - 210.0))
        if time_limit < 15.0:
            continue
        try:
            g_new, c_new = run_anti_family(
                m, parent, state, best_anti_g, best_anti_c, rng, float(time_limit)
            )
            if g_new is not None and math.isfinite(c_new) and c_new < best_anti_c:
                best_anti_g = g_new
                best_anti_c = c_new
        except Exception:
            pass

    gen_time = max(0.0, global_deadline - time.time() - 30.0)
    if gen_time > 80.0:
        n_general = int(state.n)
        if n_general < 120:
            n_general = 140
        elif n_general > 560:
            n_general = 560
        general_ns = [n_general]
        if n_general != 280:
            general_ns.append(280)
        if n_general != 140:
            general_ns.append(140)
        general_ns = sorted(set([nn for nn in general_ns if 4 <= nn <= 560]))

        starts = []
        for nn in general_ns[:2]:
            hst = upsample_linear(state.h, nn)
            starts.append((quick_c5(hst), hst, nn))
        if best_anti_g is not None and len(general_ns) > 0:
            nn = general_ns[0]
            g = best_anti_g if best_anti_g.size * 2 == nn else upsample_linear(best_anti_g, nn // 2)
            hst = make_anti_h(g)
            starts.append((quick_c5(hst), hst, nn))
        if parent is not None and len(general_ns) > 0:
            nn = general_ns[0]
            hst = upsample_linear(parent, nn)
            starts.append((quick_c5(hst), hst, nn))

        starts = [zz for zz in starts if zz[2] == zz[1].size]
        starts.sort(key=lambda yy: yy[0])

        per = gen_time / max(1.0, float(min(3, len(starts))))
        used = 0
        for _, hst, nn in starts[:3]:
            left = global_deadline - time.time() - 25.0
            if left <= 30.0:
                break
            used += per
            tlim = min(per, max(20.0, left - 25.0))
            if tlim <= 20.0:
                break
            try:
                optimize_penalty(hst, int(nn), state, global_deadline, float(tlim))
            except Exception:
                pass

    polish_time = max(0.0, global_deadline - time.time() - 25.0)
    if polish_time > 40.0 and best_anti_g is not None:
        tlim = min(90.0, polish_time - 15.0)
        try:
            pg, pc = anti_coordinate_polish(best_anti_g, global_deadline, float(tlim), rng)
            if pg is not None and math.isfinite(pc) and pc < best_anti_c:
                best_anti_g = pg
                best_anti_c = pc
            if pg is not None and math.isfinite(pc):
                update_state(state, make_anti_h(pg))
        except Exception:
            pass

    slsqp_time = max(0.0, global_deadline - time.time() - 25.0)
    if slsqp_time > 80.0:
        nsl = int(state.n)
        if nsl > 220:
            nsl = min(200, max(140, int(round(nsl / 3))))
        if nsl < 120:
            nsl = 140
        nsl = int(round(nsl / 2.0) * 2)
        hsl = upsample_linear(state.h, nsl)
        try:
            optimize_slsqp(hsl, nsl, state, global_deadline, min(140.0, slsqp_time - 15.0))
        except Exception:
            pass

    if state is not None and state.h is not None and state.h.size >= 2:
        h_final = as_feasible(state.h, state.n)
    else:
        h_final = as_feasible(h0, h0.size)

    if not is_valid_candidate(h_final):
        h_final = as_feasible(h_final, h_final.size)

    c_final = compute_c5_exact(h_final)
    if not math.isfinite(c_final):
        h_final = as_feasible(state.h if state is not None else h0, state.n if state is not None else h0.size)
        c_final = compute_c5_exact(h_final)

    return h_final, float(c_final), int(h_final.size)
