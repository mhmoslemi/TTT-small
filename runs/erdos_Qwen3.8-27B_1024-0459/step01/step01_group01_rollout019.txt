        used.add(pos)
    for pos in positions:
        h[pos] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    # Get initial values
    if 'initial_h_values' in dir() and initial_h_values is not None:
        try:
            parent_h = np.asarray(initial_h_values, dtype=float).copy()
        except Exception:
            parent_h = np.full(45, 0.5)
    else:
        parent_h = np.full(45, 0.5)
    
    n_parent = len(parent_h)
    best_h = parent_h.copy()
    best_c5 = compute_c5(fix_sum(best_h, n_parent))
    
    # Phase 1: Binary SA at parent resolution (n=180)
    n1 = n_parent
    deadline1 = t0 + budget_s * 0.35
    print(f"Phase 1: Binary SA at n={n1}, C5={best_c5:.6f}")
    
    # Start from thresholded parent
    h_bin1 = fix_binary(parent_h, n1)
    c5_bin1 = compute_c5(h_bin1)
    if c5_bin1 < best_c5:
        best_c5 = c5_bin1
        best_h = h_bin1.copy()
    
    h_res, c5_res = binary_sa(h_bin1, n1, rng, t0, deadline1, restarts=8)
    if c5_res < best_c5:
        best_c5 = c5_res
        best_h = h_res.copy()
    print(f"  After binary SA: C5={best_c5:.6f}")
    
    # Try different initial patterns for binary SA
    deadline1b = t0 + budget_s * 0.55
    alt_patterns = []
    # Periodic patterns
    for period in [1/3, 1/4, 1/5, 2/5, 3/7]:
        for duty in [0.4, 0.45, 0.5, 0.55]:
            p = periodic_pattern(n1, period, duty)
            p = fix_binary(p, n1)
            alt_patterns.append(p)
    # Wagner-like patterns
    for _ in range(5):
        w = wagner_sequence(n1, rng)
        alt_patterns.append(w)
    # Random patterns
    for _ in range(3):
        r = rng.random(n1)
        r = fix_binary(r, n1)
        alt_patterns.append(r)
    
    for idx, pat in enumerate(alt_patterns):
        if time.time() > deadline1b:
            break
        c5_pat = compute_c5(pat)
        if c5_pat < best_c5:
            best_c5 = c5_pat
            best_h = pat.copy()
        # Quick SA on each
        dl = min(deadline1b, time.time() + budget_s * 0.01)
        hr, cr = binary_sa(pat, n1, rng, t0, dl, restarts=2)
        if cr < best_c5:
            best_c5 = cr
            best_h = hr.copy()
    print(f"  After pattern exploration: C5={best_c5:.6f}")
    
    # Phase 2: Continuous L-BFGS-B at n=180
    n2 = n1
    deadline2 = t0 + budget_s * 0.75
    dx2 = 2.0 / n2
    
    # Start from best (possibly binary) solution, slightly smoothed
    h_cont = best_h.copy().astype(float)
    # Allow small fuzz around 0 and 1
    h_cont = np.clip(h_cont + rng.normal(0, 0.05, n2), 0.0, 1.0)
    h_cont = fix_sum(h_cont, n2)
    
    for tau in [0.05, 0.02, 0.01, 0.005]:
        if time.time() > deadline2:
            break
        maxiter = 2000
        h_new = lbfgs_smooth(h_cont, n2, dx2, tau, maxiter, t0, deadline2)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_cont = h_new
    print(f"  After L-BFGS-B: C5={best_c5:.6f}")
    
    # Phase 2b: Binary SA at higher resolution n=360
    n3 = 360
    deadline3 = t0 + budget_s * 0.88
    if best_c5 > 0.3805:
        h_up = upsample(best_h, n3)
        h_bin3 = fix_binary(h_up, n3)
        c5_3 = compute_c5(h_bin3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h_bin3.copy()
        h_res3, c5_res3 = binary_sa(h_bin3, n3, rng, t0, deadline3, restarts=5)
        if c5_res3 < best_c5:
            best_c5 = c5_res3
            best_h = h_res3.copy()
        print(f"  After n=360 binary SA: C5={best_c5:.6f}")
    
    # Phase 3: Fine-tuning
    deadline_final = t0 + budget_s * 0.97
    n_best = len(best_h)
    dx_best = 2.0 / n_best
    
    # Final binary polish
    if (best_h == 0).any() and (best_h == 1).any():  # It's binary
        h_pol, c5_pol = binary_sa(best_h, n_best, rng, t0, deadline_final, restarts=5)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    else:
        # Continuous polish with small tau
        h_cont2 = best_h.copy()
        h_cont2 = lbfgs_smooth(h_cont2, n_best, dx_best, 0.003, 1000, t0, deadline_final)
        c5_pol = compute_c5(h_cont2)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_cont2.copy()
        
        # Also try binary version
        h_bin_pol = fix_binary(h_cont2, n_best)
        c5_bp = compute_c5(h_bin_pol)
        if c5_bp < best_c5:
            best_c5 = c5_bp
            best_h = h_bin_pol.copy()
    
    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, len(best_h))
