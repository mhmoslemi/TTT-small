                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
        return project_feasible(r.x, n)
    except Exception:
        return project_feasible(h0, n)

def to_binary(h, n):
    n_ones = n // 2
    if n % 2 == 1:
        # For odd n, need sum = n/2 which is fractional
        # Just threshold at 0.5 and adjust
        h_bin = (h >= 0.5).astype(float)
        s = np.sum(h_bin)
        n_ones = int(n / 2)
        if s > n_ones:
            ones = np.where(h_bin == 1.0)[0]
            order = np.argsort(h[ones])[:int(s) - n_ones]
            h_bin[ones[order]] = 0.0
        elif s < n_ones:
            zeros = np.where(h_bin == 0.0)[0]
            order = np.argsort(-h[zeros])[:n_ones - int(s)]
            h_bin[zeros[order]] = 1.0
        return h_bin
    else:
        sorted_idx = np.argsort(-h)
        h_bin = np.zeros(n)
        h_bin[sorted_idx[:n_ones]] = 1.0
        return h_bin

def guided_binary_search(h_init, n, time_limit, rng):
    h = to_binary(h_init, n)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    dx = 2.0 / n
    start = time.time()
    stall = 0

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        max_idx = np.argmax(corr)
        # max_idx in full correlation: index n-1 corresponds to shift 0
        # shift = max_idx - (n-1)
        shift_k = max_idx - (n - 1)

        best_delta = 0.0
        best_pair = None
        n_trials = 60
        for _ in range(n_trials):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5(h)
            delta = best_c5 - new_c5
            if delta > best_delta:
                best_delta = delta
                best_pair = (i, j)
            else:
                h[j] = 0.0
                h[i] = 1.0

        if best_pair is not None and best_delta > 0:
            h[best_pair[0]] = 0.0
            h[best_pair[1]] = 1.0
            best_c5 += -best_delta
            best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            stall = 0
        else:
            stall += 1
            if stall > 200:
                # Perturb: random 5-swap
                for _ in range(5):
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i] = 0.0
                    h[j] = 1.0
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                stall = 0
                current_c5 = compute_c5(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()

    return best_h, best_c5

def simulated_annealing(h_init, n, time_limit, rng):
    h = to_binary(h_init, n)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    T = 0.01
    T_min = 0.0001
    cool_rate = 0.9995
    start = time.time()

    while time.time() - start < time_limit and T > T_min:
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[j] = 0.0
            h[i] = 1.0
        T *= cool_rate

    return best_h, best_c5

def coordinate_descent(h_init, n, time_limit, rng):
    h = project_feasible(h_init, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    epsilon = 0.05

    while time.time() - start < time_limit:
        improved = False
        for i in range(n):
            if time.time() - start > time_limit:
                break
            old_val = h[i]
            if old_val <= 0.01 or old_val >= 0.99:
                continue
            for delta in [-epsilon, epsilon]:
                new_val = old_val + delta
                if new_val < 0 or new_val > 1:
                    continue
                # Need to adjust another variable to maintain sum
                j = rng.randint(n)
                if j == i:
                    continue
                adj = -delta
                new_j = h[j] + adj
                if new_j < 0 or new_j > 1:
                    continue
                h[i] = new_val
                h[j] = new_j
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True
                else:
                    h[i] = old_val
                    h[j] = h[j] - adj
        if not improved:
            break
    return best_h, best_c5

def interpolate_to(h_src, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    xn = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    result = np.interp(xn, xp, h_src)
    return project_feasible(result, n_dst)

def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def make_periodic(n, freq, phase, rng):
    t = np.arange(n)
    h = 0.5 + 0.45 * np.cos(2.0 * np.pi * freq * t / n + phase)
    return project_feasible(np.clip(h, 0, 1), n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    parent_h = None
    parent_n = 0
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        if parent_n > 0:
            parent_c5 = compute_c5(parent_h)
            best_h = parent_h.copy()
            best_c5 = parent_c5
            best_n = parent_n
            print("Parent: n={}, C5={:.6f}".format(parent_n, parent_c5))
    except Exception:
        pass

    if best_h is None:
        n0 = 128
        best_h = to_binary(make_random_binary(n0, rng), n0)
        best_c5 = compute_c5(best_h)
        best_n = n0
        print("Random init: n={}, C5={:.6f}".format(n0, best_c5))

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time) - 10.0)

    resolutions = [96, 128, 160, 192, 256]
    # Sort by priority: try resolutions close to parent first, then others
    if parent_n > 0:
        resolutions.sort(key=lambda x: abs(x - parent_n))

    for n in resolutions:
        if remaining() < 50:
            break
        print("Processing n={}".format(n))

        inits = []
        # From parent
        if parent_h is not None and parent_n > 0:
            inits.append(interpolate_to(parent_h, parent_n, n))
        # From current best
        if best_h is not None and best_n != n:
            inits.append(interpolate_to(best_h, best_n, n))
        elif best_h is not None and best_n == n:
            inits.append(best_h.copy())
        # Random binary
        for _ in range(2):
            inits.append(make_random_binary(n, rng))
        # Periodic
        for freq in [2, 3, 5]:
            inits.append(make_periodic(n, freq, rng.rand() * 2 * np.pi, rng))

        for h0 in inits:
            if remaining() < 25:
                break
            h0 = project_feasible(h0, n)
            c5_0 = compute_c5(h0)

            # SLSQP smooth (beta=30)
            h1 = slsqp_phase(h0, n, 30.0, 200)
            c5_1 = compute_c5(h1)

            # SLSQP smooth (beta=100) from phase 1
            h2 = slsqp_phase(h1, n, 100.0, 150)
            c5_2 = compute_c5(h2)

            # SLSQP raw
            h3 = slsqp_raw(h2, n, 100)
            c5_3 = compute_c5(h3)

            # Binary projection
            h4 = to_binary(h3, n)
            c5_4 = compute_c5(h4)

            # Guided binary search from best continuous
            local_best = min(c5_0, c5_1, c5_2, c5_3, c5_4)
            local_h = h0
            if c5_1 < local_best:
                local_h, local_best = h1, c5_1
            if c5_2 < local_best:
                local_h, local_best = h2, c5_2
            if c5_3 < local_best:
                local_h, local_best = h3, c5_3
            if c5_4 < local_best:
                local_h, local_best = h4, c5_4

            # Binary local search
            if remaining() > 20:
                t_limit = min(25.0, remaining() * 0.3)
                h_bs, c5_bs = guided_binary_search(local_h, n, t_limit, rng)
                if c5_bs < local_best:
                    local_h, local_best = h_bs, c5_bs
                    if local_best < best_c5:
                        best_c5 = local_best
                        best_h = local_h.copy()
                        best_n = n
                        print("  Binary search improved: n={}, C5={:.6f}".format(n, best_c5))

            # Coordinate descent on continuous
            if remaining() > 15:
                t_limit = min(15.0, remaining() * 0.2)
                h_cd, c5_cd = coordinate_descent(local_h, n, t_limit, rng)
                if c5_cd < local_best:
                    local_h, local_best = h_cd, c5_cd
                    if local_best < best_c5:
                        best_c5 = local_best
                        best_h = local_h.copy()
                        best_n = n
                        print("  Coord descent improved: n={}, C5={:.6f}".format(n, best_c5))

            # Simulated annealing on binary
            if remaining() > 15:
                h_bin = to_binary(local_h, n)
                t_limit = min(20.0, remaining() * 0.2)
                h_sa, c5_sa = simulated_annealing(h_bin, n, t_limit, rng)
                if c5_sa < local_best:
                    local_best = c5_sa
                    if c5_sa < best_c5:
                        best_c5 = c5_sa
                        best_h = h_sa.copy()
                        best_n = n
                        print("  SA improved: n={}, C5={:.6f}".format(n, best_c5))

    # Final polish on best solution
    if remaining() > 30 and best_h is not None:
        n_final = best_n
        h_pol = project_feasible(best_h, n_final)
        # One more SLSQP round
        h_pol = slsqp_phase(h_pol, n_final, 200.0, 200)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol
        # Binary search one more time
        h_bin = to_binary(best_h, n_final)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin
            best_n = n_final
        if remaining() > 15:
            h_bs, c5_bs = guided_binary_search(best_h, best_n, min(20.0, remaining()), rng)
            if c5_bs < best_c5:
                best_c5 = c5_bs
                best_h = h_bs
                best_n = best_n

    # Final evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
