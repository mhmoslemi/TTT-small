import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def project_waterfill(h, target):
    """Project h onto {0<=h_i<=1, sum(h)=target} using water-filling."""
    h = h.astype(float).copy()
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target
    # f(lambda) = sum(clip(h_i - lambda, 0, 1)) is piecewise linear decreasing in lambda
    # f(-inf) = n, f(+inf) = 0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_obj_grad(h, beta):
    """Compute smooth max of correlation and its gradient."""
    n = len(h)
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    
    # Numerically stable log-sum-exp
    cmax = np.max(corr)
    exp_shifted = np.exp(np.clip(beta * (corr - cmax), -60, 60))
    sum_exp = np.sum(exp_shifted)
    f_val = cmax + np.log(sum_exp) / beta
    weights = exp_shifted / sum_exp
    
    # Gradient computation
    # corr[m] = sum_{n: valid} h[n+k] * (1-h[n]) where k = m-(n-1)
    # dcorr[m]/dh[i] = (1-h[i-k]) * I[valid for h term] + (-h[i+k]) * I[valid for (1-h) term]
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        k = m - (n - 1)
        n_lo = max(0, -k)
        n_hi = min(n - 1, n - 1 - k)
        for n_idx in range(n_lo, n_hi + 1):
            i_h = n_idx + k  # index in h for first factor
            i_c = n_idx       # index in h for second factor (1-h)
            if 0 <= i_h < n:
                grad[i_h] += w * (1.0 - h[i_c])
            if 0 <= i_c < n:
                grad[i_c] += w * (-h[i_h])
    
    return f_val, grad

def fista_solve(h0, n, target, betas, max_iter_per_beta, initial_lr=None):
    """FISTA optimization with smooth max objective."""
    h = project_waterfill(h0.copy(), target)
    h_prev = h.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    
    if initial_lr is None:
        initial_lr = 0.5
    
    for beta in betas:
        # Reset momentum
        h_prev = h.copy()
        t = 1.0
        lr = initial_lr / np.sqrt(beta)  # scale with beta
        
        for it in range(max_iter_per_beta):
            f_val, grad = smooth_max_obj_grad(h, beta)
            
            # Check convergence
            if it % 100 == 0:
                exact = compute_c5(h)
                if exact < best_val:
                    best_val = exact
                    best_h = h.copy()
            
            # FISTA step
            h_new = project_waterfill(h - lr * grad, target)
            
            # Momentum
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            h_corr = h_new + ((t - 1.0) / t_new) * (h_new - h_prev)
            h_prev = h.copy()
            h = project_waterfill(h_corr, target)
            t = t_new
            
            # Decay learning rate
            lr *= 0.999
            
            # Check for divergence
            if not np.isfinite(f_val):
                lr *= 0.5
                h = h_prev.copy()
    
    # Final exact check
    h = project_waterfill(h, target)
    exact = compute_c5(h)
    if exact < best_val:
        best_val = exact
        best_h = h.copy()
    
    return best_h, best_val

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def project_waterfill(h, target):
    h = h.astype(float).copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_obj_grad(h, beta):
    n = len(h)
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    
    cmax = np.max(corr)
    exp_shifted = np.exp(np.clip(beta * (corr - cmax), -60, 60))
    sum_exp = np.sum(exp_shifted)
    f_val = cmax + np.log(sum_exp) / beta
    W = exp_shifted / sum_exp
    
    # Gradient via correlation
    term1 = np.correlate(W, comp[::-1], mode='valid')
    term2 = np.correlate(W, h, mode='full')[n-1:2*n-1][::-1]
    grad = term1 - term2
    
    return f_val, grad

def fista_optimize(h0, n, target, beta_schedule, iters_per_beta, base_lr=0.1):
    h = project_waterfill(h0.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for beta in beta_schedule:
        h_prev = h.copy()
        t = 1.0
        lr = base_lr / np.sqrt(beta)
        
        for it in range(iters_per_beta):
            f_val, grad = smooth_max_obj_grad(h, beta)
            
            h_new = project_waterfill(h - lr * grad, target)
            
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            h_mom = h_new + ((t - 1.0) / t_new) * (h_new - h_prev)
            h_prev = h.copy()
            h = project_waterfill(h_mom, target)
            t = t_new
            
            # Periodic exact evaluation
            if (it + 1) % 500 == 0:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            
            # Adaptive lr
            lr *= 0.9997
        
        # End of beta stage
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

The W index is i+n-1-j. Let k = W_index - (n-1) = i+n-1-j-(n-1) = i-j.
So k = i-j.
j ranges from 0 to n-1, so k ranges from i to i-(n-1) = i-n+1.

For k>0 (i.e., j<i): valid when k<=min(i, n-1-i), i.e., i-j<=min(i, n-1-i).
- i-j<=i: always true (j>=0).
- i-j<=n-1-i: 2i<=n-1+j.

For k<=0 (i.e., j>=i): always valid.

So the issue is: for j<i, we need the additional constraint i-j<=n-1-i, i.e., j>=2i-n+1.

If 2i-n+1 > 0, then we need j >= 2i-n+1.
If 2i-n+1 <= 0, then all j>=0 are fine (and since j<i, we just need j in [0, i-1]).

So the correct term1[i] = sum_{j=max(0, 2i-n+1)}^{n-1} W[i+n-1-j] * comp[j]
= sum_{j=0}^{n-1} W[i+n-1-j] * comp[j] * I[j >= 2i-n+1]

This is NOT just np.correlate(W, comp[::-1], 'valid')[i]!

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def project_waterfill(h, target):
    h = h.astype(float).copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_obj_grad(h, beta):
    n = len(h)
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    
    cmax = np.max(corr)
    exp_shifted = np.exp(np.clip(beta * (corr - cmax), -60, 60))
    sum_exp = np.sum(exp_shifted)
    f_val = cmax + np.log(sum_exp) / beta
    W = exp_shifted / sum_exp
    
    # Gradient via loop (fast enough for n<=300)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = W[m]
        if w < 1e-15:
            continue
        k = m - (n - 1)
        n_lo = max(0, -k)
        n_hi = min(n - 1, n - 1 - k)
        if n_lo > n_hi:
            continue
        idx = np.arange(n_lo, n_hi + 1)
        grad[idx + k] += w * (1.0 - h[idx])
        grad[idx] -= w * h[idx + k]
    
    return f_val, grad

def slsqp_optimize(h0, n, target, beta, max_iter=200):
    """SLSQP optimization with analytical gradient for smooth max."""
    h0 = project_waterfill(h0, target)
    
    def obj(h):
        val, _ = smooth_max_obj_grad(h, beta)
        return val
    
    def grad(h):
        _, g = smooth_max_obj_grad(h, beta)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        result = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                         constraints=constraints,
                         options={'maxiter': max_iter, 'ftol': 1e-14})
        if result.x is not None:
            h_opt = project_waterfill(result.x, target)
            return h_opt
    except:
        pass
    return h0

def fista_optimize(h0, n, target, beta, max_iter, lr_init=1.0):
    """FISTA for smooth max objective."""
    h = project_waterfill(h0.copy(), target)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = 1.0
    lr = lr_init / np.sqrt(float(beta))
    
    for it in range(max_iter):
        _, grad = smooth_max_obj_grad(h, beta)
        h_new = project_waterfill(h - lr * grad, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        h_mom = h_new + ((t - 1.0) / t_new) * (h_new - h_prev)
        h_prev = h.copy()
        h = project_waterfill(h_mom, target)
        t = t_new
        lr *= 0.9995
        
        if (it + 1) % 2000 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    return best_h, best_c5

def interpolate_to_resolution(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_old)
    target = n_new / 2.0
    return project_waterfill(h_interp, target)
