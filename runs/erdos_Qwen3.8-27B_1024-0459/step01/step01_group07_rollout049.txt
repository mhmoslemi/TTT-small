import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def smooth_c5_obj(h, n_points, temp):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    scaled = conv * dx
    m = np.max(scaled)
    shifted = scaled - m
    return m + temp * np.log(np.sum(np.exp(shifted / temp)))

def smooth_c5_grad(h, n_points, temp):
    n = n_points
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    scaled = conv * dx
    m = np.max(scaled)
    shifted = scaled - m
    exp_s = np.exp(shifted / temp)
    w = exp_s / np.sum(exp_s)
    
    js = np.arange(n)
    ks = np.arange(2 * n - 1)
    J = js[:, None] + ks[None, :] - n + 1
    M = js[:, None] - ks[None, :] + n - 1
    
    valid_J = (J >= 0) & (J < n)
    valid_M = (M >= 0) & (M < n)
    
    h_J = np.zeros((n, 2 * n - 1))
    h_J[valid_J] = h[J[valid_J]]
    h_M = np.zeros((n, 2 * n - 1))
    h_M[valid_M] = h[M[valid_M]]
    
    g1 = np.sum(w[None, :] * (1.0 - h_J) * valid_J.astype(float), axis=1)
    g2 = np.sum(w[None, :] * h_M * valid_M.astype(float), axis=1)
    
    return dx * (g1 - g2)

def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0
            if not np.any(free):
                break
            h[free] = np.minimum(h[free] + diff / np.sum(free), 1.0)
        else:
            free = h > 0.0
            if not np.any(free):
                break
            h[free] = np.maximum(h[free] + diff / np.sum(free), 0.0)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        h += (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_smooth_optimize(h0, n_points, temp, maxiter=300, ftol=1e-12):
    n = n_points
    target = n / 2.0
    
    def obj(x):
        return smooth_c5_obj(x, n, temp)
    
    def grad(x):
        return smooth_c5_grad(x, n, temp)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x, res.fun

def projected_grad_descent(h0, n_points, temp, n_iter=200, lr_init=0.01):
    n = n_points
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h, n)
    lr = lr_init
    momentum = np.zeros(n)
    beta = 0.9
    
    for it in range(n_iter):
        g = smooth_c5_grad(h, n, temp)
        momentum = beta * momentum + (1 - beta) * g
        h_new = h - lr * momentum
        h_new = project_feasible(h_new, n)
        val = compute_c5(h_new, n)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
        if it % 50 == 49:
            lr *= 0.7
    
    return best_h, best_val

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.time() - t0)

            if v_try < best_val:

        except Exception:

    # Phase 2: Try multiple resolutions
    resolutions = [n0]

        if n_try != n0 and n_try > 60:

    resolutions = sorted(set(r for r in resolutions if r >= 60))
    
    for n_res in resolutions:

        if n_res == best_n and n_res == n0:
            continue  # Already did this
        
        # Interpolate best to this resolution
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_res, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2)
        h_interp = project_feasible(h_interp, n_res)
        
        v_interp = compute_c5(h_interp, n_res)
        
        # Optimize at this resolution

            try:

                h_try = project_feasible(h_try, n_res)

import numpy as np
from scipy.optimize import minimize
import time

_TEMP = 0.01


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx


def smooth_c5_obj(h, n_points, temp):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    scaled = conv * dx
    m = np.max(scaled)
    shifted = scaled - m
    return m + temp * np.log(np.sum(np.exp(shifted / temp)))


def smooth_c5_grad(h, n_points, temp):
    n = n_points
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    scaled = conv * dx
    m = np.max(scaled)
    shifted = scaled - m
    exp_s = np.exp(shifted / temp)
    w = exp_s / np.sum(exp_s)

    js = np.arange(n)
    ks = np.arange(2 * n - 1)
    J = js[:, None] + ks[None, :] - n + 1
    M = js[:, None] - ks[None, :] + n - 1

    valid_J = (J >= 0) & (J < n)
    valid_M = (M >= 0) & (M < n)

    h_J = np.zeros((n, 2 * n - 1))
    vc = valid_J
    h_J[vc] = h[J[vc]]
    h_M = np.zeros((n, 2 * n - 1))
    vc2 = valid_M
    h_M[vc2] = h[M[vc2]]

    w_row = w[None, :]
    g1 = np.sum(w_row * (1.0 - h_J) * valid_J.astype(np.float64), axis=1)
    g2 = np.sum(w_row * h_M * valid_M.astype(np.float64), axis=1)

    return dx * (g1 - g2)

            free = h < 1.0

                break

        else:
            free = h > 0.0

    s = np.sum(h)
    if abs(s - target) > 1e-10:
        h += (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h

                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
