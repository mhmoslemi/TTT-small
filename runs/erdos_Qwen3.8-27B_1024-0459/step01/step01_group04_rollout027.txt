        if time.time() > deadline - 5.0:
            break
        h_prev = h.copy()
        beta = 0.7 + 0.25 * (si / max(1, len(stages)))
        for it in range(max_iter):
            if time.time() > deadline - 5.0:
                break
            if it % 500 == 0:
                check_deadline()
            y = h.copy()
            if it > 0:
                y = h + beta * (h - h_prev)
                y = np.clip(y, 0.0, 1.0)
            g = compute_gradient(y, n, alpha)
            g_norm = float(np.linalg.norm(g))
            if g_norm < 1e-10:
                break
            lr = lr0 / (1.0 + 0.02 * it)
            h_new = y - lr * g
            h_new = as_feasible(h_new, n)
            h_prev = h
            h = h_new
            if (it + 1) % 200 == 0:
                c = compute_c5(h)
                if c < best_c - 1e-14:
                    best_c = c
                    best_h = h.copy()
        c = compute_c5(h)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
    return best_h, best_c

def upsample_linear(parent, n):
    n0 = int(parent.size)
    if n == n0:
        return parent.copy()
    if n0 <= 0:
        return np.full(n, 0.5)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x, x0, parent.astype(float)).astype(float)

def smooth1(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 2:
        return h.copy()
    out = h.copy()
    out[0] = 0.75 * h[0] + 0.25 * h[1]
    out[-1] = 0.25 * h[-2] + 0.75 * h[-1]
    out[1:-1] = 0.25 * h[:-2] + 0.5 * h[1:-1] + 0.25 * h[2:]
    return out

def binary_scores(h, n):
    n = int(n)
    scores = np.zeros(2 * n - 1, dtype=float)
    hf = h.astype(float)
    for d in range(1, n):
        scores[2 * d - 1] = float(np.dot(hf[:n - d], 1.0 - hf[d:]))
        scores[2 * d] = float(np.dot(hf[d:], 1.0 - hf[:n - d]))
    return scores

def binary_flip_update(scores, h, idx, new_val, n):
    i = int(idx)
    nv = int(new_val)
    for d in range(1, n):
        pidx = 2 * d - 1
        midx = 2 * d
        if i - d >= 0:
            hj = int(h[i - d])
            if nv == 1:
                scores[pidx] -= hj
                scores[midx] += 1.0 - hj
            else:
                scores[pidx] += hj
                scores[midx] -= 1.0 - hj
        if i + d < n:
            hj = int(h[i + d])
            if nv == 1:
                scores[pidx] += 1.0 - hj
                scores[midx] -= hj
            else:
                scores[pidx] -= 1.0 - hj
                scores[midx] += hj
    h[i] = nv

def binary_swap_apply(scores, h, i, j, n):
    i = int(i)
    j = int(j)
    if i == j:
        return False
    if int(h[i]) == int(h[j]):
        return False
    if h[i] == 1 and h[j] == 0:
        binary_flip_update(scores, h, i, 0, n)
        binary_flip_update(scores, h, j, 1, n)
    else:
        binary_flip_update(scores, h, i, 1, n)
        binary_flip_update(scores, h, j, 0, n)
    return True

def binary_local_search(start_h, n, deadline, max_ops, rng, state=None):
    n = int(n)
    target = n // 2
    h = np.zeros(n, dtype=np.int8)
    if start_h is not None and np.size(start_h) == n:
        vals = np.asarray(start_h, dtype=float).copy()
        vals = np.nan_to_num(vals, nan=0.0, posinf=1.0, neginf=0.0)
        vals = np.clip(vals, 0.0, 1.0)
        order = np.argsort(vals)
        h[order[target:]] = 1
    else:
        idx = rng.choice(n, size=target, replace=False)
        h[idx] = 1
    scores = binary_scores(h, n)
    cur = float(np.max(scores))
    best = cur
    best_h = h.copy()
    if state is not None:
        state.update(h.astype(float), best * (2.0 / n), n)
    ops = 0
    stall = 0
    temperature = 3.0
    while ops < max_ops and time.time() < deadline:
        ops += 1
        if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE:
            break
        if ops % 5000 == 0:
            temperature = max(0.3, temperature * 0.92)
        # Targeted move: try to fix the most violated constraint
        use_targeted = (rng.random() < 0.6)
        if use_targeted:
            max_idx = int(np.argmax(scores))
            if max_idx == 0:
                # shift 0: reduce sum h[i]*(1-h[i]) - not directly fixable by swap
                i_cand = int(rng.integers(n))
                j_cand = int(rng.integers(n))
                if h[i_cand] != h[j_cand]:
                    i, j = i_cand, j_cand
                else:
                    use_targeted = False
            else:
                if max_idx % 2 == 1:
                    d = (max_idx + 1) // 2
                    mask = (h[:n - d] == 1) & (h[d:] == 0)
                else:
                    d = max_idx // 2
                    mask = (h[d:] == 1) & (h[:n - d] == 0)
                    # For even index (negative shift): h[i]=1, h[i-d]=0 → swap i with i-d
                    mask = (h[:n - d] == 0) & (h[d:] == 1)
                cand = np.nonzero(mask)[0]
                if cand.size > 0:
                    if max_idx % 2 == 1:
                        pos = int(cand[int(rng.integers(cand.size))])
                        i, j = pos, pos + d
                    else:
                        pos = int(cand[int(rng.integers(cand.size))])
                        i, j = pos + d, pos
                else:
                    use_targeted = False
        if not use_targeted:
            zeros = np.flatnonzero(h == 0)
            ones = np.flatnonzero(h == 1)
            if zeros.size == 0 or ones.size == 0:
                break
            i = int(zeros[int(rng.integers(zeros.size))])
            j = int(ones[int(rng.integers(ones.size))])
        if not binary_swap_apply(scores, h, i, j, n):
            continue
        new_max = float(np.max(scores))
        delta = new_max - cur
        if delta < -1e-9:
            cur = new_max
            stall = 0
            if cur < best - 1e-12:
                best = cur
                best_h = h.copy()
                if state is not None:
                    state.update(h.astype(float), best * (2.0 / n), n)
        elif delta <= 1e-9:
            # Accept with small probability
            if rng.random() < 0.1:
                stall += 1
            else:
                binary_swap_apply(scores, h, i, j, n)
                stall += 1
        else:
            # Metropolis accept
            if math.exp(-delta / temperature) > rng.random():
                cur = new_max
                stall += 1
            else:
                binary_swap_apply(scores, h, i, j, n)
                stall += 1
        if stall > 5000:
            temperature = max(0.3, temperature * 0.8)
            stall = 0
    final_c = cur * (2.0 / n)
    if state is not None:
        state.update(h.astype(float), final_c, n)
    if final_c < best:
        best = final_c
        best_h = h.copy()
    return best_h.astype(float), float(best * (2.0 / n))

class OptimizerState:
    def __init__(self, h, c, n):
        self.h = np.array(h, dtype=float)
        self.c = float(c)
        self.n = int(n)
    def update(self, h, c, n):
        if h is None:
            return
        h = np.asarray(h, dtype=float)
        if h.ndim != 1 or h.size == 0:
            return
        if n is None:
            n = h.size
        n = int(n)
        if h.size != n:
            return
        try:
            h2 = as_feasible(h, n)
            c2 = compute_c5(h2)
        except Exception:
            return
        if math.isfinite(c2) and c2 < self.c - 1e-15:
            self.h = h2
            self.c = float(c2)
            self.n = n

def get_parent():
    try:
        p = np.asarray(initial_h_values, dtype=float)
        if p.ndim != 1 or p.size < 2:
            return None
        return p.copy()
    except Exception:
        return None

def synthetic_start(n, rng):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    best_h = as_feasible(np.full(n, 0.5), n)
    best_c = compute_c5(best_h)
    for f in (1, 2, 3, 5):
        for ph in (0.0, 0.5 * math.pi, math.pi):
            h_cand = 0.5 + 0.4 * np.cos(math.pi * f * x + ph)
            h_cand = as_feasible(h_cand, n)
            c_cand = compute_c5(h_cand)
            if c_cand < best_c:
                best_c = c_cand
                best_h = h_cand
    # Thue-Morse like
    tm = np.zeros(n)
    for i in range(n):
        cnt = 0
        v = i
        while v:
            cnt += v & 1
            v >>= 1
        tm[i] = 1.0 if cnt % 2 == 0 else 0.0
    h_tm = smooth1(tm)
    h_tm = as_feasible(h_tm, n)
    c_tm = compute_c5(h_tm)
    if c_tm < best_c:
        best_c = c_tm
        best_h = h_tm
    return best_h

def run(seed=42, budget_s=1300, **kwargs):
    global _RUN_DEADLINE
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0.0:
        budget_s = 30.0
    reserve = 20.0 if budget_s > 60.0 else max(2.0, 0.1 * budget_s)
    global_deadline = t0 + max(1.0, budget_s - reserve)
    _RUN_DEADLINE = global_deadline
    rng = np.random.default_rng(seed)
    parent = get_parent()
    if parent is not None and parent.size > 0:
        n0 = int(parent.size)
        h0 = parent.copy()
        c0 = compute_c5(h0)
    else:
        n0 = 70
        h0 = synthetic_start(n0, rng)
        c0 = compute_c5(h0)
    state = OptimizerState(h0.copy(), c0, n0)
    # Gradient descent at parent resolution
    if time.time() < global_deadline - 120.0:
        grad_deadline = min(global_deadline - 60.0, t0 + 200.0)
        try:
            gh, gc = nesterov_optimize(h0, n0, grad_deadline)
            state.update(gh, gc, n0)
        except Exception:
            pass
    # Binary local search at parent resolution
    if n0 % 2 == 0 and time.time() < global_deadline - 80.0:
        bin_deadline = min(global_deadline - 60.0, t0 + 350.0)
        try:
            best_start = state.h
            max_ops_n0 = 80000 if n0 <= 100 else 30000
            bh, bc = binary_local_search(best_start, n0, bin_deadline, max_ops_n0, rng, state)
        except Exception:
            pass
    # Upsample to higher resolution and optimize
    for mult in (2.0, 1.5, 3.0):
        nn = int(round(mult * n0 / 2.0)) * 2
        if nn < n0 or nn > 400:
            continue
        if time.time() >= global_deadline - 100.0:
            break
        h_up = upsample_linear(state.h, nn)
        h_up = as_feasible(h_up, nn)
        c_up = compute_c5(h_up)
        state.update(h_up, c_up, nn)
        # Also try binary at this resolution
        if nn % 2 == 0 and time.time() < global_deadline - 80.0:
            bin_dl = min(global_deadline - 60.0, time.time() + 60.0)
            try:
                bh2, bc2 = binary_local_search(h_up, nn, bin_dl, 40000, rng, state)
            except Exception:
                pass
        # Gradient at higher res
        if time.time() < global_deadline - 80.0:
            grad_dl = min(global_deadline - 60.0, time.time() + 120.0)
            try:
                gh2, gc2 = nesterov_optimize(state.h, nn, grad_dl)
                state.update(gh2, gc2, nn)
            except Exception:
                pass
    # Final SLSQP polish at best resolution
    if state is not None and state.h is not None and time.time() < global_deadline - 30.0:
        n_best = state.n
        h_best = state.h.copy()
        if time.time() < global_deadline - 35.0:
            def obj_fn(x, alpha):
                x = np.clip(x, 0.0, 1.0)
                scores = compute_scores(x, n_best)
                m = float(np.max(scores))
                if alpha > 1e-14:
                    w = np.exp((scores - m) / alpha)
                    A = float(np.sum(w))
                    if math.isfinite(A) and A > 0:
                        return (m + alpha * math.log(A)) * (2.0 / n_best)
                return m * (2.0 / n_best)
            def obj_grad(x, alpha):
                g = compute_gradient(np.clip(x, 0.0, 1.0), n_best, alpha)
                return g
            def con_eq(x):
                return (float(np.sum(x)) - 0.5 * float(n_best)) / float(n_best)
            def con_jac(x):
                return np.full((1, n_best), 1.0 / float(n_best))
            bounds = [(0.0, 1.0)] * n_best
            constraints = [{"type": "eq", "fun": con_eq, "jac": con_jac}]
            slsqp_dl = min(global_deadline - 10.0, time.time() + 50.0)
            try:
                res = minimize(
                    obj_fn, h_best, args=(0.05,),
                    jac=lambda x: obj_grad(x, 0.05),
                    method="SLSQP", bounds=bounds,
                    constraints=constraints,
                    options={"maxiter": 100, "ftol": 1e-10, "disp": 0}
                )
                if res is not None and res.x is not None:
                    h_polish = as_feasible(res.x, n_best)
                    c_polish = compute_c5(h_polish)
                    state.update(h_polish, c_polish, n_best)
            except Exception:
                pass
    if state is not None and state.h is not None:
        h_final = as_feasible(state.h, state.n)
        c_final = compute_c5(h_final)
        n_final = int(state.n)
    else:
        h_final = as_feasible(h0, n0)
        c_final = compute_c5(h_final)
        n_final = n0
    # Also keep the original parent as fallback
    c_parent = compute_c5(h0)
    if c_parent < c_final:
        h_final = h0
        c_final = c_parent
        n_final = n0
    if not np.all(np.isfinite(h_final)) or h_final.size == 0:
        h_final = np.full(n0, 0.5)
        c_final = compute_c5(h_final)
        n_final = n0
    h_final = as_feasible(h_final, h_final.size)
    c_final = compute_c5(h_final)
    return h_final, float(c_final), int(h_final.size)
