_ALPHA = 1.0

    n = h.size
    if n == 0:

    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_density(x, target):

    n = x.size

        s2 = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s2 < target:
            lo = mid
        else:
            hi = mid

    res = target - float(np.sum(h))

    alpha = _ALPHA
    h = np.asarray(x, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    y = alpha * (corr - m)

    z = float(np.sum(e))

    return float((2.0 / n) * (m + np.log(z) / alpha))

def softmax_jac(x):
    n = _NC
    alpha = _ALPHA
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return np.zeros(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))

    y = np.clip(y, -700.0, 0.0)

def corr_grad_for_lags(h, lags):
    """Compute average gradient of f_k over given lags."""
    n = len(h)
    g = np.zeros(n)
    center = n - 1
    for t in lags:
        k = t - center
        if k == 0:
            g += (1.0 - 2.0 * h)
        elif k > 0:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
        else:
            kk = -k
            g[:n-kk] += (1.0 - h[kk:])
            g[kk:] -= h[:n-kk]
    return g / float(len(lags))

def polish_subgrad(h, deadline, max_iter=2000, n_active=6):
    """Subgradient descent with Polyak step and iterate averaging."""
    n = len(h)
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    
    # Polyak step: need upper bound B on C5
    B = best_c + 0.05  # upper bound
    avg_h = np.zeros(n)
    count = 0
    cur_h = best.copy()
    cur_c = best_c
    stall = 0
    gamma = 1.0  # decaying step factor
    
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        active = np.argsort(corr)[-n_active:]
        g = corr_grad_for_lags(cur_h, active)
        
        # Normalize gradient
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        g = g / gn
        
        # Polyak step size: (B - cur_c) / ||g||_inf (using the unnormalized gradient)
        # Actually, for subgradient of max: step = (B - f(x)) / ||g||^2
        # Let's use eta = min(0.5, (B - cur_c + 1e-6) / max(gn*gn, 0.01))
        # But g is normalized, so ||g||=1. Step = (B - cur_c) / 1 = B - cur_c
        # That's too aggressive. Let's use:
        eta = min(0.05, 0.5 * gamma * (B - cur_c + 1e-8) / max(float(np.max(np.abs(g * n / 2.0))), 0.01))
        # Simpler: just use a fixed small step
        eta = 0.02 * gamma
        
        # Project onto feasible set
        new_h = project_density(cur_h - eta * g, target)
        new_c = evaluate_erdos(new_h)
        
        avg_h += new_h
        count += 1
        avg_h_norm = avg_h / count
        avg_h_norm = project_density(avg_h_norm, target)
        avg_c = evaluate_erdos(avg_h_norm)
        
        if new_c < cur_c - 1e-11:
            cur_h = new_h
            cur_c = new_c
            if new_c < best_c:
                best = new_h.copy()
                best_c = new_c
            B = min(B, cur_c + 0.03)
            stall = 0
        else:
            stall += 1
            gamma *= 0.98
            if stall > 100:
                # Try the averaged iterate
                if avg_c < best_c:
                    best = avg_h_norm.copy()
                    best_c = avg_c
                break
        
        if avg_c < best_c:
            best = avg_h_norm.copy()
            best_c = avg_c
    
    # Final check
    best = project_density(best, target)
    best_c = evaluate_erdos(best)
    return best, best_c

def coordinate_descent(h, deadline, target, max_rounds=50):
    """Coordinate descent: try adjusting each h[i] individually."""
    n = len(h)
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    
    for rnd in range(max_rounds):
        if time.perf_counter() >= deadline:
            break
        
        improved = False
        for i in range(n):
            if time.perf_counter() >= deadline:
                break
            for delta in (0.03, 0.1, -0.03, -0.1, 0.01, -0.01):
                cand = best.copy()
                cand[i] = np.clip(cand[i] + delta, 0.0, 1.0)
                cand = project_density(cand, target)
                c = evaluate_erdos(cand)
                if c < best_c - 1e-12:

                    best_c = c

def pair_search_full(h, deadline, target, n_active=6):
    """More thorough pair search."""
    n = len(h)
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    center = n - 1
    stall = 0
    
    while time.perf_counter() < deadline:
        corr = np.correlate(best, 1.0 - best, mode='full')
        active = np.argsort(corr)[-n_active:]
        g = corr_grad_for_lags(best, active)
        
        # Score indices
        neg_idx = np.argsort(g)  # most negative first (increase these)
        pos_idx = np.argsort(-g)  # most positive first (decrease these)
        
        improved = False
        for i in neg_idx[:min(10, n)]:
            if time.perf_counter() >= deadline:
                break
            for j in pos_idx[:min(10, n)]:
                if i == j:
                    continue
                # Compute max transfer
                if g[i] > g[j]:  # i should increase, j should decrease
                    cap = min(1.0 - best[i], best[j])
                else:
                    continue
                if cap < 1e-10:
                    continue
                
                d_opt = (g[j] - g[i]) / (2.0 * n_active)  # approximate
                d_opt = max(1e-4, min(d_opt, cap * 0.9))
                
                for d in (d_opt, 0.5 * cap, 0.2 * cap, cap * 0.99):
                    d = min(d, cap * 0.999)
                    if d < 1e-6:
                        continue
                    cand = best.copy()
                    cand[i] += d
                    cand[j] -= d
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-12:
                        best = cand
                        best_c = c
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try subgradient step
            gn = float(np.linalg.norm(g))
            if gn < 1e-12:
                break
            d = -g / gn
            d -= float(np.mean(d))
            for step in (0.01, 0.03, 0.07):
                cand = project_density(best + step * d, target)
                c = evaluate_erdos(cand)
                if c < best_c - 1e-12:
                    best = cand
                    best_c = c
                    improved = True
                    break
        
        if not improved:
            stall += 1
            if stall > 30:
                break
    
    best = project_density(best, target)
    best_c = evaluate_erdos(best)
    return best, best_c

def block_move_search(h, deadline, target, block_sizes=(2,3,4,5)):

def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    n = int(n)
    if n0 == n:
        return p.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.interp(x1, x0, p, left=p[0], right=p[-1])
    return project_density(h, 0.5 * n)

def make_structured_candidate(n, rng, mode):
    """Create different structured starting points."""
    n = int(n)
    if mode == 0:
        # Smooth sinusoidal
        x = (np.arange(n) + 0.5) * (2.0 / n)
        h = np.full(n, 0.5)
        for m in range(1, 8):
            amp = rng.uniform(0.03, 0.25) * (0.6 ** (m-1))
            freq = int(max(1, round(m * rng.uniform(0.5, 1.5))))
            phase = rng.uniform(0, 2*np.pi)
            h += amp * np.sin(2*np.pi*freq*x/2.0 + phase)
    elif mode == 1:

        h = np.zeros(n)

        pos = 0

            length = int(rng.integers(1, min(remaining+1, max(2, n//8))))
            val = float(rng.choice([0.0, 0.3, 0.5, 0.7, 1.0]))
            h[pos:pos+length] = val
            pos += length
            remaining = n - pos
    elif mode == 2:

        h = np.empty(n)

        for i in range(n):
            phase = (i // period) % 6
            h[i] = [0.0, 1.0, 0.5, 1.0, 0.0, 0.5][phase]

        # Fourier with specific frequencies
        x = (np.arange(n) + 0.5) * (2.0 / n)

            phase = rng.uniform(0, 2*np.pi)
            h += amp * np.cos(2*np.pi*f*x/2.0 + phase)
    else:
        # Random
        h = rng.random(n)
    
    return project_density(h, 0.5 * n)

import time
import numpy as np
from scipy.optimize import minimize

_NC = 1
_ALPHA = 1.0
_TARGET = 0.5

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_density(x, target):
    x = np.clip(np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    n = x.size
    s = float(np.sum(x))
    if abs(s - target) < 1e-13 * max(1, n):
        return x.copy()
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s2 = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s2 < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        free = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
        if free.size > 0:
            h[free] += res / float(free.size)
            h = np.clip(h, 0.0, 1.0)
            res2 = target - float(np.sum(h))
            if abs(res2) > 1e-13 and free.size > 0:
                h[free[0]] += res2
                h = np.clip(h, 0.0, 1.0)
    return h

def set_CTX(n, alpha, target):
    global _NC, _ALPHA, _TARGET
    _NC = int(n)
    _ALPHA = float(alpha)
    _TARGET = float(target)

def soft_obj(x):
    n = _NC
    alpha = _ALPHA
    h = np.asarray(x, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return 1e100
    return float((2.0 / n) * (m + np.log(z) / alpha))

def soft_jac(x):
    n = _NC
    alpha = _ALPHA
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return np.zeros(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return np.zeros(n)
    w = e / z
    g = np.zeros(n)
    center = n - 1
    for t in range(len(w)):
        if w[t] < 1e-15:
            continue
        p = float(w[t])
        k = t - center
        if k == 0:
            g += p * (1.0 - 2.0 * h)
        elif k > 0:
            g[k:] += p * (1.0 - h[:n - k])
            g[:n - k] -= p * h[k:]
        else:
            kk = -k
            g[:n - kk] += p * (1.0 - h[kk:])
            g[kk:] -= p * h[:n - kk]
    return (2.0 / n) * g

def sum_con(x):
    return float(np.sum(x) - _TARGET)

def sum_con_jac(x):
    return np.ones_like(x)

def corr_grad_lags(h, lag_indices):
    n = len(h)
    g = np.zeros(n)
    center = n - 1
    for t in lag_indices:
        t = int(t)
        k = t - center
        if k == 0:
            g += (1.0 - 2.0 * h)
        elif k > 0:
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] -= h[k:]
        else:
            kk = -k
            g[:n - kk] += (1.0 - h[kk:])
            g[kk:] -= h[:n - kk]
    return g / float(len(lag_indices))

def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    n = int(n)
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))

def intensive_pair_search(h, deadline, n_active=8, max_passes=200):

def gradient_polish(h, deadline, n_active=8, max_iter=3000):

        gn = float(np.linalg.norm(g_centered))
        if gn < 1e-14:

def optimize_resolution(h0, n, deadline):
