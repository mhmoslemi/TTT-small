    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target):
    """Project h onto {sum(h)=target, 0<=h<=1} via bisection on shift."""
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    shift = (lo + hi) / 2.0
    h_proj = np.clip(h - shift, 0.0, 1.0)
    # Fix tiny residual
    residual = target - np.sum(h_proj)
    if abs(residual) > 1e-13:
        if residual > 0:
            free = np.where(h_proj < 1.0 - 1e-12)[0]
        else:
            free = np.where(h_proj > 1e-12)[0]
        if len(free) > 0:
            h_proj[free] += residual / len(free)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def compute_corr(h):
    """Compute full correlation of h with 1-h."""
    comp = 1.0 - h
    return np.correlate(h, comp, mode='full')

def smooth_max_and_grad(h, beta):
    """Compute smooth max of correlations and gradient wrt h."""
    n = len(h)
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = beta * (corr - c_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    f_val = c_max + np.log(np.sum(exp_shifted)) / beta
    
    # Gradient
    a = 1.0 - 2.0 * h  # length n
    # g[j] = sum_{m=0}^{n-1} a[m] * w[j+m] for valid j+m < 2n-1
    # g = convolve(a_reversed, w, 'full')[n-1 : 2*n-1]
    a_rev = a[::-1]  # length n
    # convolve result length: n + (2n-1) - 1 = 3n-2
    conv = np.convolve(a_rev, w, 'full')
    g = conv[n-1:2*n-1]  # length n
    
    return f_val, g

    n = len(h0)
    h = project_feasible(h0.copy(), target)

        h_NEW = project_feasible(h - lr * g, target)

        h = project_feasible(h, target)
        t_k = t_new

    return best_h, best_f

            stuck = 0

Hmm, the swap local search is O(n) per swap (because of the correlation computation). For n=100, that's 100 operations per swap, and we can do maybe 100k swaps in the time budget. That's fine.

But wait, I can make the swap evaluation more efficient. When I swap h[i] from 1 to 0 and h[j] from 0 to 1:

The correlation c[k] = Σ_m h[m] * (1-h[m+k-n+1]) for valid m.

Change in c[k] due to h[i]: 1→0:
- Term where m=i: was h[i]*(1-h[i+k-n+1]) = 1*(1-h[i+k-n+1]), now 0. Change: -(1-h[i+k-n+1])
- Term where m+k-n+1=i, i.e., m = i-(k-n+1) = i+n-1-k: was h[m]*(1-h[i]) = h[m]*0 = 0. Now h[m]*1. Change: +h[m] (if m valid)

Change in c[k] due to h[j]: 0→1:
- Term where m=j: was 0, now 1*(1-h[j+k-n+1]). Change: +(1-h[j+k-n+1])
- Term where m = j+n-1-k: was h[m]*(1-h[j]) = h[m]*1, now h[m]*0. Change: -h[m]

So total delta c[k]:
= -(1-h[i+k-n+1]) + h[i+n-1-k] + (1-h[j+k-n+1]) - h[j+n-1-k]
  (all terms only if indices are valid: 0≤idx<n)

This is O(1) per k, and there are 2n-1 values of k. So O(n) per swap to evaluate all deltas. Then I take the max of (corr + delta).
