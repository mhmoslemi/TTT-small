    n = h.size

                g[lag:] -= p * h[:n - lag]
            else:

                g[:n - lag] -= p * h[lag:]
            if k > 0:
                g[:n - lag] -= p * h[lag:]
            else:
                g[lag:] -= p * h[:n - lag]
    return dx * g

def slsqp_solve(h0, n, deadline, alphas=None, maxiter=None):
    if alphas is None:
        if n <= 120:
            alphas = (50.0, 150.0, 400.0)
            maxiter = 200
        elif n <= 300:
            alphas = (50.0, 150.0, 400.0)
            maxiter = 150
        else:
            alphas = (40.0, 120.0, 350.0)
            maxiter = 100
    target = 0.5 * n
    h = project_density(h0, target)
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.perf_counter() >= deadline:
            break
        def obj(x):
            return smooth_objective(x, alpha)
        def grad(x):
            return smooth_grad(x, alpha)
        def con_eq(x):
            return float(np.sum(x) - target)
        def con_eq_jac(x):
            return np.ones(n, dtype=float)
        cons = ({"type": "eq", "fun": con_eq, "jac": con_eq_jac},)
        try:
            res = minimize(
                obj, h, method="SLSQP", jac=grad,
                bounds=bounds, constraints=cons,
                options={"maxiter": maxiter, "ftol": 1e-10, "disp": False}
            )
            xg = res.x
            if np.all(np.isfinite(xg)):
                cand = project_density(xg, target)
                c = evaluate_erdos(cand)
                if c < best_c:
                    best_h = cand
                    best_c = c
                h = cand if c <= best_c + 0.01 else best_h
        except Exception:
            h = best_h
    return best_h, best_c

def spectral_smooth(h, n_freqs=5):

        return h.copy()

    if n_freqs < H.size:
