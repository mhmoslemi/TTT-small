    g = np.zeros(n)
    for t in idxs:

        if k == 0:
            g += 1.0 - 2.0 * h

            if k > 0:
                # i-k valid when i >= k, so positions k..n-1 get (1-h[0..n-1-k])
                g[k:] += (1.0 - h[:n-k])
                # -h[i+k] for valid i+k
                # i+k valid when i < n-k, so positions 0..n-1-k get -h[k..n-1]

                # i-k = i+lag valid when i < n-lag
                g[:n-lag] += (1.0 - h[lag:])
                # i+k = i-lag valid when i >= lag
                g[lag:] -= h[:n-lag]

import time
import numpy as np
from scipy.optimize import minimize

_CTX = {}

def eval_c5(h):
    n = h.size
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(x, target):
    x = np.clip(np.nan_to_num(x), 0.0, 1.0)
    n = x.size
    target = float(target)
    s = float(x.sum())
    if abs(s - target) < 1e-13 * max(1, n):
        return x.copy()
    lo, hi = -1.5, 1.5
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x + mid, 0.0, 1.0).sum())
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(h.sum())
    if abs(res) > 1e-12:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            h[free] += res / float(free.size)
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    n = int(n)
    if n0 == n:
        return project(p.copy(), 0.5 * n)
    if n0 < 1:
        return project(np.full(max(1, n), 0.5), 0.5 * max(1, n))
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project(h, 0.5 * n)

def compute_subgrad_dir(h, topk=8):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idxs = np.argsort(corr)[-topk:]
    center = n - 1
    g = np.zeros(n)
    for t in idxs:
        k = int(t) - center
        if k == 0:
            g += 1.0 - 2.0 * h
        elif k > 0:
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] -= h[k:]
        else:
            lag = -k
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
    g /= float(len(idxs))
    return g, corr, idxs

def subgrad_optimize(h_in, deadline, topk=8, max_iter=8000):
    n = h_in.size
    target = 0.5 * n
    best = project(h_in, target)
    best_c = eval_c5(best)
    stall = 0
    adapt_step = 0.05

    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        if stall > 20:
            break

        g, corr, idxs = compute_subgrad_dir(best, topk)
        d = -g
        d -= d.mean()
        norm = np.linalg.norm(d)
        if norm < 1e-14:
            break
        d /= norm

        improved = False
        steps = (adapt_step, 0.5 * adapt_step, 2.0 * adapt_step,
                 0.01, 0.03, 0.07, 0.15, 0.3, 0.005)
        for step in steps:
            cand = project(best + step * d, target)
            c = eval_c5(cand)
            if c < best_c - 1e-12:
                best = cand
                best_c = c
                stall = 0
                improved = True
                if step != adapt_step and 0.001 < step < 1.0:
                    adapt_step = 0.8 * adapt_step + 0.2 * step
                    adapt_step = np.clip(adapt_step, 0.005, 0.5)
                break
        if not improved:
            stall += 1
            adapt_step *= 0.7

    return best, best_c

def pair_search(h_in, deadline, topk=6, max_iter=3000):

def _smooth_obj(x):
    n = _CTX["n"]
    alpha = _CTX["alpha"]
    target = _CTX["target"]
    h = x
    corr = np.correlate(h, 1.0 - h, mode="full")

    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(e.sum())
    if z <= 0 or not np.isfinite(z):
        return 1e100
    return float((2.0 / n) * (m + np.log(z) / alpha))

def _smooth_jac(x):
    n = _CTX["n"]
    alpha = _CTX["alpha"]
    h = x
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(e.sum())
    if z <= 0 or not np.isfinite(z):
        return np.zeros(n)
    w = e / z
    center = n - 1
    g = np.zeros(n)
    idx = np.where(w > 1e-10)[0]
    for t in idx:
        t = int(t)
        p = float(w[t])
        k = t - center

            g += p * (1.0 - 2.0 * h)

            g[k:] += p * (1.0 - h[:n-k])
            g[:n-k] -= p * h[k:]
        else:
            lag = -k
            g[:n-lag] += p * (1.0 - h[lag:])
            g[lag:] -= p * h[:n-lag]
    return (2.0 / n) * g

def make_fourier_start(n, rng, mode=0):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.full(n, 0.5)
    nterms = max(2, n // 15)
    for m in range(1, nterms + 1):
        amp = rng.uniform(0.03, 0.25) * (0.6 ** (m - 1))
        freq = max(1, int(round(m * rng.uniform(0.7, 1.4))))
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    if mode == 1:
        u = rng.random(n)
        thresh = np.median(u)
        h = 0.5 * h + 0.5 * (u > thresh).astype(float)
    elif mode == 2:
        levels = [0.0, 0.3, 0.5, 0.7, 1.0]
        h2 = np.empty(n)
        pos = 0
        while pos < n:
            rem = n - pos
            ln = int(rng.integers(1, max(2, rem // 3 + 1)))
            h2[pos:pos+ln] = float(rng.choice(levels))
            pos += ln
        h = 0.4 * h + 0.6 * h2
    return project(h, 0.5 * n)
