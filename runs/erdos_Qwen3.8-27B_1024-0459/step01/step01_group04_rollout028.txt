    guard = 0
    while abs(diff) > 1e-12 and guard <= n + 5:
        guard += 1
        if diff > 0.0:
            mask = h < 1.0
            if not np.any(mask):
                break
            cnt = int(np.count_nonzero(mask))
            per = diff / float(cnt)
            inc = min(per, float(1.0 - h[mask].min()))
            if inc <= 1e-15:
                break
            h[mask] += inc
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            cnt = int(np.count_nonzero(mask))
            per = -diff / float(cnt)
            dec = min(per, float(h[mask].min()))
            if dec <= 1e-15:
                break
            h[mask] -= dec
        diff = target - float(h.sum())
    if abs(diff) > 1e-13:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return np.clip(h, 0.0, 1.0)


def compute_scores(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    corr = np.correlate(h, h, mode="full")
    ac = corr[n - 1:n - 1 + n]
    prefix = np.empty(n + 1, dtype=float)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])
    total = prefix[n]
    scores = np.empty(2 * n - 1, dtype=float)
    scores[0] = total - ac[0]
    for d in range(1, n):
        scores[2 * d - 1] = prefix[n - d] - ac[d]
        scores[2 * d] = total - prefix[d] - ac[d]
    return scores


def score_and_grad(h, n, alpha):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    scores = compute_scores(h, n)
    m = float(np.max(scores))
    if alpha > 1e-14:
        shifted = (scores - m) / alpha
        max_shifted = float(np.max(shifted))
        shifted -= max_shifted
        w = np.exp(shifted)
        A = float(np.sum(w))
        if not math.isfinite(A) or A <= 0.0:
            val = m
            grad = np.zeros(n)
        else:
            val = m + alpha * (max_shifted + math.log(A))
            coeffs = w / A
            grad = np.zeros(n)
            for idx in range(2 * n - 1):
                c = coeffs[idx]
                if c < 1e-12:
                    continue
                _add_grad_term(grad, idx, c, h, n)
    else:
        val = m
        grad = np.zeros(n)
        tol = max(1e-8, 1e-7 * max(1.0, m))
        active = np.flatnonzero(scores >= m - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(scores))])
        c = 1.0 / float(active.size)
        for idx in active:
            _add_grad_term(grad, int(idx), c, h, n)
    grad *= 2.0 / n
    return float(val), grad


def _add_grad_term(grad, idx, coef, h, n):
    if coef == 0.0:
        return
    if idx == 0:
        grad += coef * (1.0 - 2.0 * h)
        return
    if idx % 2 == 1:
        d = (idx + 1) // 2
        m = n - d
        if m <= 0:
            return
        grad[:m] += coef * (1.0 - h[d:])
        grad[d:] += coef * (-h[:m])
    else:
        d = idx // 2
        m = n - d
        if m <= 0:
            return
        grad[:m] += coef * (-h[d:])
        grad[d:] += coef * (1.0 - h[:m])


def obj_lbfgsb(x, n, alpha, constraint_mu):
    check_deadline()
    h = np.clip(x, 0.0, 1.0)
    val, grad = score_and_grad(h, n, alpha)
    if constraint_mu > 0.0:
        s = float(np.sum(h)) - 0.5 * float(n)
        val += 0.5 * constraint_mu * s * s
        grad += constraint_mu * s * np.ones(n)
    if not math.isfinite(val):
        return 1e6, np.zeros(n)
    return val, grad


def gradient_lbfgsb(x, n, alpha, constraint_mu):
    check_deadline()
    h = np.clip(x, 0.0, 1.0)
    _, grad = score_and_grad(h, n, alpha)
    if constraint_mu > 0.0:
        s = float(np.sum(h)) - 0.5 * float(n)
        grad += constraint_mu * s * np.ones(n)
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(x.size)
    return grad


def upsample_linear(parent, n):
    n0 = int(parent.size)
    if n == n0:
        return parent.copy()
    if n0 <= 0:
        return np.full(n, 0.5)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x, x0, parent.astype(float))


def make_sawtooth(n, period, phase=0.0):
    x = np.arange(n, dtype=float)
    t = (x + phase) / float(period)
    h = np.abs(2.0 * (t - np.floor(t) - 0.5))
    h = as_feasible(h, n)
    return h


def make_cosine(n, freq, amp=0.5, phase=0.0):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + amp * np.cos(2.0 * math.pi * freq * x + phase)
    return as_feasible(h, n)


def get_parent():
    try:
        p = np.asarray(initial_h_values, dtype=float)
        if p.ndim != 1 or p.size < 2:
            return None
        return p.copy()
    except Exception:
        return None


class BestState:
    def __init__(self, h, c, n):
        self.h = np.array(h, dtype=float)
        self.c = float(c)
        self.n = int(n)

    def try_update(self, h, c, n):
        if h is None or c is None:
            return
        h = np.asarray(h, dtype=float)
        if h.ndim != 1 or h.size == 0:
            return
        n = int(n)
        if h.size != n:
            return
        if not math.isfinite(c):
            return
        h2 = as_feasible(h, n)
        c2 = compute_c5(h2)
        if math.isfinite(c2) and c2 < self.c - 1e-15:
            self.h = h2
            self.c = c2
            self.n = n


def run_continuous_lbfgsb(h, n, state, time_limit):
    n = int(n)
    if h is None or h.size != n:
        return
    if float(time_limit) < 10.0:
        return

    h_start = as_feasible(h, n)
    state.try_update(h_start, compute_c5(h_start), n)

    bounds = [(0.0, 1.0)] * n
    start_t = time.time()

    alpha_schedule = [max(0.5, 0.01 * n), max(0.2, 0.005 * n), max(0.05, 0.002 * n), 0.0]
    mu_schedule = [0.0, 0.0, 0.0, 0.0]

    total_alpha = time_limit
    fractions = [0.25, 0.25, 0.25, 0.25]
    cur_h = h_start.copy()

    for stage in range(len(alpha_schedule)):
        if time.time() > start_t + total_alpha - 5.0:
            break
        remaining = total_alpha - (time.time() - start_t)
        if remaining < 8.0:
            break
        alpha = alpha_schedule[stage]
        mu = mu_schedule[stage]
        maxiter = 500 if alpha > 0.01 * n else 1000
        maxiter = min(maxiter, int(30.0 * remaining / max(1.0, 2.0 / n * n)))
        maxiter = max(50, min(maxiter, 3000))

        res = None
        try:
            if alpha > 1e-14:
                res = minimize(
                    lambda x: obj_lbfgsb(x, n, alpha, mu),
                    cur_h,
                    jac=lambda x: gradient_lbfgsb(x, n, alpha, mu),
                    method="L-BFGS-B",
                    bounds=bounds,
                    options={"maxiter": maxiter, "ftol": 1e-14, "gtol": 1e-9, "disp": 0},
                )
            else:
                res = minimize(
                    lambda x: obj_lbfgsb(x, n, alpha, mu),
                    cur_h,
                    jac=lambda x: gradient_lbfgsb(x, n, alpha, mu),
                    method="L-BFGS-B",
                    bounds=bounds,
                    options={"maxiter": maxiter, "ftol": 1e-14, "gtol": 1e-9, "disp": 0},
                )
        except BudgetError:
            res = None
        except Exception:
            res = None

        if res is not None and hasattr(res, "x") and res.x is not None:
            h_new = as_feasible(res.x, n)
        else:
            h_new = cur_h

        state.try_update(h_new, compute_c5(h_new), n)
        cur_h = h_new.copy()

        if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE - 30:
            break


def run_slsqp_polish(h, n, state, time_limit):
    n = int(n)
    if h is None or h.size != n:
        return
    if float(time_limit) < 15.0:
        return

    h_start = as_feasible(h, n)
    state.try_update(h_start, compute_c5(h_start), n)

    bounds = [(0.0, 1.0)] * n
    target = 0.5 * float(n)

    def obj(x):
        check_deadline()
        h2 = np.clip(x, 0.0, 1.0)
        sc = compute_scores(h2, n)
        m = float(np.max(sc))
        alpha = 0.01 * n
        shifted = (sc - m) / alpha
        ms = float(np.max(shifted)) - ms
        w = np.exp((sc - m) / alpha)
        A = float(np.sum(w))
        if not math.isfinite(A) or A <= 0:
            return m * 2.0 / n
        return (m + alpha * math.log(A)) * 2.0 / n

    def con_fun(x):
        return float(np.sum(x)) - target

    def con_jac(x):
        return np.ones(n)

    constraints = [{"type": "eq", "fun": con_fun, "jac": con_jac}]
    maxiter = max(50, int(100.0 * time_limit / max(1.0, n / 30.0)))
    maxiter = min(maxiter, 500)

    res = None
    try:
        _RUN_DEADLINE_sub = _RUN_DEADLINE
        if _RUN_DEADLINE is not None:
            _RUN_DEADLINE = min(_RUN_DEADLINE_sub, time.time() + time_limit - 2)
        res = minimize(
            obj, h_start, method="SLSQP",
            bounds=bounds, constraints=constraints,
            options={"maxiter": maxiter, "ftol": 1e-10, "disp": 0},
        )
        _RUN_DEADLINE = _RUN_DEADLINE_sub
    except Exception:
        res = None
        if _RUN_DEADLINE is not None and _RUN_DEADLINE_sub is not None:
            _RUN_DEADLINE = _RUN_DEADLINE_sub

    if res is not None and hasattr(res, "x") and res.x is not None:
        h_new = as_feasible(res.x, n)
        state.try_update(h_new, compute_c5(h_new), n)


def binary_hill_climb(start_h, n, state, deadline, max_ops, stall_limit):
    n = int(n)
    if n % 2 != 0:
        return None, 1e9

    h = np.zeros(n, dtype=np.int8)
    target = n // 2
    if start_h is not None and np.size(start_h) == n:
        vals = np.nan_to_num(np.asarray(start_h, dtype=float), nan=0.5)
        vals = np.clip(vals, 0.0, 1.0)
        order = np.argsort(-vals)
        h[order[:target]] = 1
    else:
        indices = np.random.default_rng(0).choice(n, size=target, replace=False)
        h[indices] = 1

    # Compute initial scores
    hf = h.astype(float)
    scores = np.zeros(2 * n - 1, dtype=float)
    for d in range(1, n):
        scores[2 * d - 1] = float(np.dot(hf[:n-d], 1.0 - hf[d:]))
        scores[2 * d] = float(np.dot(hf[d:], 1.0 - hf[:n-d]))
    cur = float(np.max(scores))
    best = cur
    best_h = h.copy()
    state.try_update(h.astype(float), best * (2.0 / n), n)

    rng = np.random.default_rng(12345)
    ops = 0
    stall = 0

    while ops < max_ops and time.time() < deadline:
        ops += 1
        if ops % 2000 == 0:
            if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE:
                break

        # Propose swap
        zeros = np.flatnonzero(h == 0)
        ones = np.flatnonzero(h == 1)
        if zeros.size == 0 or ones.size == 0:
            break

        # Bias towards active constraints
        if rng.random() < 0.6:
            active = np.flatnonzero(scores >= cur - 1e-9)
            if active.size > 0:
                idx = int(active[int(rng.integers(active.size))])
                if idx != 0:
                    if idx % 2 == 1:
                        d = (idx + 1) // 2
                        mask = (h[:n-d] == 1) & (h[d:] == 0)
                    else:
                        d = idx // 2
                        mask = (h[:n-d] == 0) & (h[d:] == 1)
                    cand = np.nonzero(mask)[0]
                    if cand.size > 0:
                        i = int(cand[int(rng.integers(cand.size))])
                        j = i + d
                        _binary_swap(h, scores, i, j, n)
                        new_max = float(np.max(scores))
                        if new_max < cur - 1e-9:
                            cur = new_max
                            stall = 0
                            if cur < best:
                                best = cur
                                best_h = h.copy()
                                state.try_update(h.astype(float), best * (2.0 / n), n)
                        else:
                            _binary_swap(h, scores, i, j, n)
                            stall += 1
                        continue

        i = int(zeros[int(rng.integers(zeros.size))])
        j = int(ones[int(rng.integers(ones.size))])
        _binary_swap(h, scores, i, j, n)
        new_max = float(np.max(scores))
        if new_max < cur - 1e-9:
            cur = new_max
            stall = 0
            if cur < best:
                best = cur
                best_h = h.copy()
                state.try_update(h.astype(float), best * (2.0 / n), n)
        else:
            if rng.random() < 0.05:
                pass  # accept (annealing-like)
            else:
                _binary_swap(h, scores, i, j, n)
            stall += 1

        if stall > stall_limit:
            break

    state.try_update(best_h.astype(float), best * (2.0 / n), n)
    return best_h.astype(float), float(best * (2.0 / n))


def _binary_swap(h, scores, i, j, n):
    hi = int(h[i])
    hj = int(h[j])
    if hi == hj:
        return
    # Flip i and j
    for d in range(1, n):
        pidx = 2 * d - 1
        midx = 2 * d
        if i - d >= 0:
            hh = int(h[i - d])
            if hi == 1:
                scores[pidx] -= hh
                scores[midx] += 1.0 - hh
            else:
                scores[pidx] += hh
                scores[midx] -= 1.0 - hh
        if i + d < n:
            hh = int(h[i + d])
            if hi == 1:
                scores[pidx] += 1.0 - hh
                scores[midx] -= hh
            else:
                scores[pidx] -= 1.0 - hh
                scores[midx] += hh
        if j - d >= 0:
            hh = int(h[j - d])
            if hj == 1:
                scores[pidx] -= hh
                scores[midx] += 1.0 - hh
            else:
                scores[pidx] += hh
                scores[midx] -= 1.0 - hh
        if j + d < n:
            hh = int(h[j + d])
            if hj == 1:
                scores[pidx] += 1.0 - hh
                scores[midx] -= hh
            else:
                scores[pidx] -= 1.0 - hh
                scores[midx] += hh
    h[i] = hj
    h[j] = hi


def multi_start_lbfgsb(candidates, n, state, time_limit):
    n = int(n)
    if not candidates:
        return
    total_time = float(time_limit)
    per_candidate = max(15.0, total_time / len(candidates))
    deadline = time.time() + total_time

    for h_start in candidates:
        if time.time() > deadline - 10.0:
            break
        remaining = deadline - time.time()
        if remaining < 15.0:
            break
        alloc = min(per_candidate, remaining - 10.0)
        if alloc < 10.0:
            break
        try:
            run_continuous_lbfgsb(h_start, n, state, alloc)
        except Exception:
            pass


def run(seed=42, budget_s=1300, **kwargs):
    global _RUN_DEADLINE

    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0:
        budget_s = 30.0
    reserve = 20.0 if budget_s > 60.0 else max(2.0, 0.1 * budget_s)
    global_deadline = t0 + max(1.0, budget_s - reserve)
    _RUN_DEADLINE = global_deadline

    rng = np.random.default_rng(seed)

    # Get parent
    parent = get_parent()
    if parent is not None and parent.size > 0:
        n0 = int(parent.size)
        h0 = parent.copy()
        c0 = compute_c5(h0)
        h0 = as_feasible(h0, n0)
        c0 = compute_c5(h0)
    else:
        n0 = 70
        h0 = make_cosine(n0, freq=2, amp=0.4)
        c0 = compute_c5(h0)

    state = BestState(h0.copy(), c0, n0)
    best_initial = (h0.copy(), c0, n0)

    # Generate diverse candidates at n0
    candidates_n0 = [h0.copy()]
    for f in [1, 2, 3, 4]:
        for ph in [0.0, math.pi / 4, math.pi / 2]:
            hc = make_cosine(n0, freq=f, amp=0.45, phase=ph)
            candidates_n0.append(hc)
    for period in [5, 7, 10, 14]:
        ht = make_sawtooth(n0, period=period)
        candidates_n0.append(ht)

    # Phase 1: Binary hill-climbing at n0 (~60s)
    if n0 % 2 == 0 and _RUN_DEADLINE is not None and time.time() < _RUN_DEADLINE - 500.0:
        bin_time = min(60.0, max(10.0, 0.05 * budget_s))
        rh = h0.copy()
        for restart in range(3):
            if time.time() > t0 + bin_time:
                break
            if restart == 0:
                bh, bc = binary_hill_climb(rh, n0, state, time.time() + bin_time / 3.0, 50000, 5000)
            else:
                bh, bc = binary_hill_climb(None, n0, state, time.time() + bin_time / 3.0, 40000, 4000)
            if bh is not None:
                bhf = as_feasible(bh, n0)
                candidates_n0.append(bhf)

    # Phase 2: L-BFGS-B at n0 with multiple starts (~400s)
    if _RUN_DEADLINE is not None and time.time() < _RUN_DEADLINE - 200.0:
        remaining = _RUN_DEADLINE - time.time()
        n0_time = min(400.0, max(60.0, 0.35 * budget_s))
        n0_time = min(n0_time, remaining - 200.0)
        if n0_time > 30.0:
            # Take top candidates
            scored = []
            for hc in candidates_n0:
                cc = compute_c5(hc)
                scored.append((hc, cc))
            scored.sort(key=lambda x: x[1])
            top_cands = [sc for sc, _ in scored[:4]]
            multi_start_lbfgsb(top_cands, n0, state, n0_time)

    # Phase 3: Try finer resolution (upsample best to 2*n0 or 200)
    best_n_states = [(state.n, state.h.copy(), state.c)]
    best_n_states.append((n0, best_initial[0].copy(), best_initial[1]))

    fine_ns = []
    for mult in [1.5, 2.0, 2.5]:
        nn = int(round(mult * float(n0)))
        if nn % 2 != 0:
            nn += 1
        if nn >= n0 + 20 and nn <= 300:
            fine_ns.append(nn)
    fine_ns = sorted(set(fine_ns))

    for nn in fine_ns[:2]:  # At most 2 fine resolutions
        if _RUN_DEADLINE is None or time.time() >= _RUN_DEADLINE - 150.0:
            break
        remaining = _RUN_DEADLINE - time.time()
        if remaining < 100.0:
            break
        nn_time = min(300.0, remaining - 100.0)
        if nn_time < 60.0:
            break

        # Upsample from best
        h_fine = upsample_linear(state.h, nn)
        h_fine = as_feasible(h_fine, nn)

        # Also try from parent if different
        extra_cands = [h_fine]
        if parent is not None:
            hp = upsample_linear(parent, nn)
            hp = as_feasible(hp, nn)
            if np.max(np.abs(hp - h_fine)) > 0.01:
                extra_cands.append(hp)

        # Add a sawtooth at this resolution
        extra_cands.append(make_sawtooth(nn, period=max(5, nn // 10)))

        multi_start_lbfgsb(extra_cands[:3], nn, state, nn_time)

    # Phase 4: Polish best with SLSQP (~100s)
    if _RUN_DEADLINE is not None and time.time() < _RUN_DEADLINE - 30.0:
        remaining = _RUN_DEADLINE - time.time()
        polish_time = min(100.0, remaining - 20.0)
        if polish_time > 20.0:
            run_slsqp_polish(state.h, state.n, state, polish_time)

    # Phase 5: One more L-BFGS-B pass on the best with very small alpha
    if _RUN_DEADLINE is not None and time.time() < _RUN_DEADLINE - 10.0:
        remaining = _RUN_DEADLINE - time.time()
        if remaining > 20.0:
            n_best = state.n
            h_best = state.h.copy()
            bounds = [(0.0, 1.0)] * n_best
            alpha_final = 0.5
            maxiter = max(100, int(50.0 * remaining / 10.0))
            res = None
            try:
                res = minimize(
                    lambda x: obj_lbfgsb(x, n_best, alpha_final, 0.0),
                    h_best,
                    jac=lambda x: gradient_lbfgsb(x, n_best, alpha_final, 0.0),
                    method="L-BFGS-B",
                    bounds=bounds,
                    options={"maxiter": maxiter, "ftol": 1e-14, "gtol": 1e-10, "disp": 0},
                )
            except Exception:
                res = None
            if res is not None and hasattr(res, "x") and res.x is not None:
                h_polished = as_feasible(res.x, n_best)
                state.try_update(h_polished, compute_c5(h_polished), n_best)

    # Final
    h_final = as_feasible(state.h, state.n)
    c_final = compute_c5(h_final)
    n_final = int(state.n)

    if c_final < best_initial[1]:
        return h_final, float(c_final), n_final
    else:
        return best_initial[0], float(best_initial[1]), best_initial[2]
