def make_random_step(n, rng):
    """Create a random step function candidate."""
    h = np.empty(n, dtype=float)
    pos = 0
    while pos < n:
        rem = n - pos
        length = int(rng.integers(1, max(2, rem + 1)))
        val = float(rng.choice([0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0]))
        h[pos:pos + length] = val
        pos += length
    return project_density(h, 0.5 * n)

def make_peaked_candidate(n, rng):
    """Create a candidate with peaked structure."""
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.full(n, 0.5, dtype=float)
    for m in range(1, 8):
        amp = float(rng.uniform(0.1, 0.4)) * (0.6 ** (m - 1))
        freq = int(max(1, np.rint(m * float(rng.uniform(0.3, 1.2)))))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        h += amp * np.cos(2.0 * np.pi * freq * x / 2.0 + phase)
    return project_density(h, 0.5 * n)

def smooth_jacobian(x):

    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))

        return np.zeros(n, dtype=float)
    w = e / z

                best_n = int(best_h.size)

        # Gradient polish
        if deadline - time.perf_counter() > 5.0:
            for ak_p in (8, 12, 16, 24):
                if time.perf_counter() >= deadline - 1.0:
                    break
                h_gp, c_gp = gradient_polish(best_h, deadline, max_iter=400, active_k=ak_p)
                if c_gp < best_c:
                    best_h = h_gp
                    best_c = c_gp
                    best_n = int(best_h.size)

    if best_h is None:
        fallback = project_density(np.full(45, 0.5, dtype=float), 22.5)
        return fallback, evaluate_erdos(fallback), 45

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
