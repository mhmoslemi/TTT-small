def project_density(x, target):
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-13:
        return x
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s_mid = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s_mid < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-14:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:

    t2 = deadline - 5.0
    if time.perf_counter() < t2:
        h2, c2 = optimize_coordinate(best, t2, max_iter=10000)
        if c2 < best_c:
            best, best_c = h2, c2

    # Phase 3: short SLSQP
    t3 = deadline - 2.0
    if time.perf_counter() < t3 and n <= 200:
        h3, c3 = optimize_slsqp(best, t3, n)
        if c3 < best_c:
            best, best_c = h3, c3

    # Phase 4: binary local search
    if time.perf_counter() < deadline - 1.0:
        h4, c4 = binary_local_search(best, deadline - 0.5, max_iter=10000)
        if c4 < best_c:
            best, best_c = h4, c4

    # Final: coordinate + active set one more round
    if time.perf_counter() < deadline - 0.5:
        h5, c5 = optimize_coordinate(best, deadline - 0.1, max_iter=5000)
        if c5 < best_c:
            best, best_c = h5, c5

    best = project_density(best, target)
    best_c = evaluate_erdos(best)
    return best, best_c


def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 15.0)

    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    # Load parent
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        nparent = parent.size
        target_p = 0.5 * nparent
        pp = project_density(parent, target_p)
        cp = evaluate_erdos(pp)
        best_h = pp.copy()
        best_c = cp
        best_n = nparent

    if best_h is None:
        n0 = 90
        best_h = make_binary_random(n0, rng)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    # Resolutions to try
    base_n = best_n if best_n > 1 else 90
    res_list = []
    for mult in (1, 2, 3, 4, 6, 8):
        nn = int(base_n * mult)
        if 4 <= nn <= 540 and nn not in res_list:
            res_list.append(nn)
    for nn in (90, 180, 270, 360):
        if nn not in res_list and 4 <= nn <= 540:
            res_list.append(nn)
    res_list = sorted(set(res_list))

    # Allocate time to resolutions
    # Give more time to promising resolutions
    n_res = len(res_list)
    # Weight: prefer n around 180-360
    weights = []
    for nn in res_list:
        if 120 <= nn <= 360:
            weights.append(3.0)
        elif nn <= 90:
            weights.append(2.0)
        else:
            weights.append(1.5)
    total_w = sum(weights)
    per_res_time = [(budget - 80.0) * w / total_w for w in weights]
    # Cap individual
    per_res_time = [min(t, 400.0) for t in per_res_time]
    per_res_time = [max(t, 20.0) for t in per_res_time]

    for ri, n in enumerate(res_list):
        if deadline - time.perf_counter() < 25.0:
            break

        res_deadline = time.perf_counter() + min(per_res_time[ri], deadline - time.perf_counter() - 30.0)
        if res_deadline > deadline:
            res_deadline = deadline - 30.0
        if res_deadline - time.perf_counter() < 15.0:
            continue

        # Generate multiple initializations for this resolution
        inits = []

        # 1. Upsample best
        inits.append(upsample_h(best_h, n))

        # 2. Upsample parent
        if parent is not None:
            inits.append(upsample_h(parent, n))

        # 3. Quantized upsample
        if parent is not None:
            inits.append(make_quantized(upsample_h(parent, n), n))

        # 4. Random binary
        inits.append(make_binary_random(n, rng))

        # 5. Another random binary
        if n <= 200:
            inits.append(make_binary_random(n, rng))

        # 6. Smooth random
        x = (np.arange(n) + 0.5) * (2.0 / float(n))
        h_smooth = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2 + rng.uniform(0, 2 * np.pi))
        h_smooth += 0.15 * np.sin(2 * np.pi * 7 * x / 2 + rng.uniform(0, 2 * np.pi))
        inits.append(project_density(h_smooth, 0.5 * n))

        # Deduplicate
        uniq_inits = []
        for s in inits:
            s = project_density(s, 0.5 * n)
            dup = False
            for u in uniq_inits:
                if np.allclose(s, u, atol=1e-6):
                    dup = True
                    break
            if not dup:
                uniq_inits.append(s)

        # Allocate time per initialization
        n_init = len(uniq_inits)
        per_init_time = max(8.0, (res_deadline - time.perf_counter()) / n_init)

        for ii, init_h in enumerate(uniq_inits):
            if deadline - time.perf_counter() < 12.0:
                break
            sub_deadline = time.perf_counter() + min(per_init_time, res_deadline - time.perf_counter())
            if sub_deadline - time.perf_counter() < 8.0:
                continue

            # Evaluate initial
            c_init = evaluate_erdos(init_h)

            # Choose optimization strategy based on remaining time and n
            phase_deadline = sub_deadline - 5.0

            # Phase A: active momentum
            if time.perf_counter() < phase_deadline - 10.0:
                h_a, c_a = optimize_active_momentum(init_h, min(phase_deadline - 10.0, sub_deadline - 15.0), top_k=10, max_iter=4000)
                if c_a < c_init:
                    init_h, c_init = h_a, c_a

            # Phase B: coordinate descent
            if time.perf_counter() < phase_deadline - 5.0:
                h_b, c_b = optimize_coordinate(init_h, phase_deadline - 5.0, max_iter=15000)
                if c_b < c_init:
                    init_h, c_init = h_b, c_b

            # Phase C: SLSQP (only for small n)
            if n <= 180 and time.perf_counter() < sub_deadline - 3.0:
                h_c, c_c = optimize_slsqp(init_h, sub_deadline - 1.0, n)
                if c_c < c_init:
                    init_h, c_init = h_c, c_c

            # Final projection
            init_h = project_density(init_h, 0.5 * n)
            c_init = evaluate_erdos(init_h)

            if c_init < best_c:
                best_h = init_h.copy()
                best_c = c_init
                best_n = n

    # Final intensive polish on the best solution
    if best_h is not None and deadline - time.perf_counter() > 20.0:
        h_final, c_final = polish_final(best_h, deadline - 2.0)
        if c_final < best_c:
            best_h, best_c = h_final, c_final

    # Verify and return
    if best_h is None:
        fallback = project_density(np.full(90, 0.5, dtype=float), 45.0)
        return fallback, evaluate_erdos(fallback), 90

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
