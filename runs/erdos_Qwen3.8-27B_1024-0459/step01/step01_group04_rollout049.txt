def score_vec(h, n):
    """Compute all 2n-1 overlap scores (before multiplying by dx)."""
    h = np.asarray(h, dtype=float)
    corr = np.correlate(h, h, mode="full")
    ac = corr[n-1:n-1+n]  # autocorrelation at lags 0..n-1
    prefix = np.empty(n+1, dtype=float)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])
    total = prefix[n]
    scores = np.empty(2*n-1, dtype=float)

        scores[n-1+d] = prefix[n-d] - ac[d]  # positive lag d
        scores[n-1-d] = total - prefix[d] - ac[d]  # negative lag d
    return scores
