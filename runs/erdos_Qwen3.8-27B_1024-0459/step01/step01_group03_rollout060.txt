import time
import numpy as np
from scipy.optimize import minimize

# Global for optimization callbacks
_NC = 1
_ALPHA = 1.0
_TARGET = 0.5

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)

def project_density(x, target):
    """Project x onto {0<=x<=1, sum(x)=target}"""
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    
    s = float(np.sum(x))
    if abs(s - target) < 1e-12:
        return x
    
    # Bisection for the water-filling level
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    
    # Fix residual
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            h[free] += res / free.size
            h = np.clip(h, 0.0, 1.0)
            # Re-fix if needed
            res2 = target - float(np.sum(h))
            if abs(res2) > 1e-10:
                free = np.where((h > 0.0) & (h < 1.0))[0]
                if free.size > 0:
                    idx = free[0]
                    h[idx] += res2
                    h = np.clip(h, 0.0, 1.0)
    
    return h

def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)

def smooth_objective(x):
    n = _NC
    alpha = _ALPHA
    if n <= 1:
        return 0.0
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return 1e100
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0 or not np.all(np.isfinite(corr)):
        return 1e100
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return 1e100
    return float((2.0 / float(n)) * (m + np.log(z) / alpha))

def smooth_jacobian(x):
    n = _NC
    alpha = _ALPHA
    if n <= 1:
        return np.zeros(max(1, n), dtype=float)
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return np.zeros(n, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0 or not np.all(np.isfinite(corr)):
        return np.zeros(n, dtype=float)
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return np.zeros(n, dtype=float)
    w = e / z
    # Only process significant weights
    idx = np.where(w > 1e-10)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(w))], dtype=int)
    g = np.zeros(n, dtype=float)
    center = n - 1
    for t in idx:
        t = int(t)
        p = float(w[t])
        if p <= 0.0:
            continue
        k = t - center
        if k == 0:
            g += p * (1.0 - 2.0 * h)
        else:
            lag = abs(k)
            if k > 0:
                g[lag:] += p
            else:
                g[:n - lag] += p
            g[:n - lag] -= p * h[lag:]
            g[lag:] -= p * h[:n - lag]
    return (2.0 / float(n)) * g

def sum_constraint(x):
    return float(np.sum(x) - _TARGET)

def sum_constraint_jac(x):
    return np.ones_like(np.asarray(x, dtype=float), dtype=float)

def corr_grad_for_lag(h, k):
    """Gradient of corr at lag k (where k = index - (n-1))"""
    n = h.size
    g = np.zeros(n, dtype=float)
    if k == 0:
        g = 1.0 - 2.0 * h
    else:
        lag = abs(k)
        if k > 0:
            g[lag:] += 1.0
        else:
            g[:n - lag] += 1.0
        g[:n - lag] -= h[lag:]
        g[lag:] -= h[:n - lag]
    return g

def active_lag_gradient(h, frac=0.1, min_k=3):
    """Weighted gradient focusing on top lags"""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    # Get top lags
    k_count = max(min_k, int(np.ceil(len(corr) * frac)))
    top_idx = np.argsort(corr)[-k_count:]
    # Inverse-temperature weighting
    top_vals = corr[top_idx]
    shifted = top_vals - top_vals.min()
    weights = np.exp(shifted / max(1e-10, shifted.max() + 1e-10))
    weights /= weights.sum()
    g = np.zeros(n, dtype=float)
    for i, t in enumerate(top_idx):
        k = int(t) - center
        g += weights[i] * corr_grad_for_lag(h, k)
    return g

def polish_active(h, deadline, max_iter=200, active_frac=0.05, min_k=4):
    """Gradient descent on active lags with projection"""
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    if n < 2:
        return best, best_c
    stall = 0
    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        g = active_lag_gradient(best, active_frac, min_k)
        d = -g
        d -= np.mean(d)
        norm = np.linalg.norm(d)
        if norm < 1e-14:
            break
        d /= norm
        improved = False
        for step in (0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5):
            cand = project_density(best + step * d, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-12:
                best = cand
                best_c = c
                improved = True
                break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break
    return project_density(best, target), evaluate_erdos(best)

    n = h.size

    best = project_density(h, target)
    best_c = evaluate_erdos(best)

        return best, best_c

                method="SLSQP", jac=smooth_jacobian,
                bounds=bounds, constraints=constraints,
                options={"maxiter": maxiter, "ftol": 1e-11, "disp": False}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                cand = project_density(res.x, target)
                c = evaluate_erdos(cand)
                if c < best_c:
                    best_h = cand
                    best_c = c
                h = cand if c < best_c + 0.03 else best_h
        except Exception:
            pass

    # Polish phases
    rem = deadline - time.perf_counter()
    if rem > 8.0:
        h2, c2 = pair_swap_aggressive(best_h, time.perf_counter() + rem * 0.35, max_iter=600, top_frac=0.05, min_k=6, n_pos=8, n_neg=8)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    rem = deadline - time.perf_counter()
    if rem > 5.0:
        h3, c3 = polish_gradient_active(best_h, time.perf_counter() + rem * 0.6, max_iter=300, top_frac=0.05, min_k=6)
        if c3 < best_c:
            best_h = h3
            best_c = c3

    rem = deadline - time.perf_counter()
    if rem > 5.0 and n <= 200:
        h4, c4 = triple_swap_search(best_h, time.perf_counter() + rem * 0.7, max_iter=300)
        if c4 < best_c:
            best_h = h4
            best_c = c4

    rem = deadline - time.perf_counter()
    if rem > 3.0:
        h5, c5 = polish_gradient_active(best_h, time.perf_counter() + rem * 0.9, max_iter=200, top_frac=0.04, min_k=8)
        if c5 < best_c:
            best_h = h5
            best_c = c5

    return project_density(best_h, target), evaluate_erdos(best_h)


def make_diverse_start(n, rng, mode):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    if mode == 0:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * x / 2 + rng.uniform(0, 6.28))
        h += 0.15 * np.sin(2 * np.pi * 5 * x / 2 + rng.uniform(0, 6.28))
    elif mode == 1:
        h = np.empty(n, dtype=float)
        seg = int(n // 8) + 1
        for i in range(0, n, seg):
            end = min(i + seg, n)
            h[i:end] = float(rng.uniform(0.1, 0.9))
    elif mode == 2:
        u = rng.random(n)
        med = float(np.median(u))
        h = np.where(u > med, 0.75, 0.25).astype(float)
        h += rng.normal(0, 0.05, n)
    else:
        h = 0.5 + 0.4 * np.sin(2 * np.pi * int(rng.integers(2, 7)) * x / 2 + rng.uniform(0, 6.28))
    return project_density(h, 0.5 * n)


def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 18.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        np_ = int(parent.size)
        pt = 0.5 * np_
        pp = project_density(parent, pt)
        cp = evaluate_erdos(pp)
        best_h = pp.copy()
        best_c = cp
        best_n = np_

    if best_h is None:
        n0 = 90
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2) + 0.15 * np.sin(2 * np.pi * 5 * x / 2 + 1.0)
        best_h = project_density(h_init, 0.5 * n0)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    base_n = best_n if best_n > 1 else 90

    # Define resolutions to try, prioritizing n=180 and n=360
    n_list = []
    for target_n in [base_n, 2 * base_n, 4 * base_n]:
        if 2 <= target_n <= 500:
            n_list.append(target_n)
    for extra_n in [90, 120, 180, 240, 360, 450]:
        if extra_n not in n_list and 2 <= extra_n <= 500:
            n_list.append(extra_n)
    n_list = sorted(set(n_list))

    # Time allocation: more time to higher resolutions (they have more expressiveness)
    total_time = deadline - time.perf_counter()
    # Weight: base_n gets less, 2*base_n gets more, 4*base_n gets most
    weights = []
    for n in n_list:
        if n <= base_n:
            weights.append(1.0)
        elif n <= 2 * base_n:
            weights.append(2.5)
        else:
            weights.append(3.0)
    wsum = sum(weights)
    time_alloc = {}
    for n, w in zip(n_list, weights):
        time_alloc[n] = max(25.0, total_time * w / wsum)

    for ni in n_list:
        rem = deadline - time.perf_counter()
        if rem < 25.0:
            break

        alloc = time_alloc.get(ni, max(25.0, rem / 3))
        alloc = min(alloc, rem - 5.0)

        # Build starting points
        starts = []
        starts.append(upsample_h(best_h, ni))
        if parent is not None:
            starts.append(upsample_h(parent, ni))

        n_rand = 2 if ni <= 120 else (1 if ni <= 240 else 1)
        for mm in range(n_rand):
            starts.append(make_diverse_start(ni, rng, mm % 4))

        # Deduplicate
        uniq = []
        for s in starts:
            s = project_density(s, 0.5 * ni)
            dup = False
            for u in uniq:
                if np.allclose(s, u, atol=1e-6):
                    dup = True
                    break
            if not dup:
                uniq.append(s)

        per_start = alloc / len(uniq)
        per_start = max(10.0, per_start)

        for s in uniq:
            rem2 = deadline - time.perf_counter()
            if rem2 < 12.0:
                break
            sub_dl = time.perf_counter() + min(per_start, rem2 - 3.0)
            h_out, c_out = optimize_resolution(ni, s, sub_dl, seed)
            if h_out is not None:
                h_out = project_density(h_out, 0.5 * ni)
                c_out = evaluate_erdos(h_out)
                if c_out < best_c:
                    best_h = h_out.copy()
                    best_c = c_out
                    best_n = ni

    # Final intensive polish on the best solution
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 12.0:
        nbest = int(best_h.size)
        target = 0.5 * nbest
        h_fin = project_density(best_h, target)

        # Binarize push then re-optimize
        if nbest <= 200:
            h_bin = binarize_push(h_fin, target, thresh=0.12)
            c_bin = evaluate_erdos(h_bin)
            if c_bin < best_c:
                h_fin = h_bin
                best_c = c_bin
                best_h = h_bin.copy()

        # Extended pair swap
        if rem > 15.0 and nbest <= 220:
            h2, c2 = pair_swap_aggressive(h_fin, time.perf_counter() + rem * 0.35,
                                          max_iter=3000, top_frac=0.04, min_k=8, n_pos=12, n_neg=12)
            if c2 < best_c:
                h_fin = h2
                best_c = c2
                best_h = h2.copy()
                best_n = nbest

        # Final gradient polish
        rem2 = deadline - time.perf_counter()
        if rem2 > 6.0:
            h3, c3 = polish_gradient_active(h_fin, time.perf_counter() + rem2 * 0.85,
                                            max_iter=800, top_frac=0.03, min_k=10)
            if c3 < best_c:
                h_fin = h3
                best_c = c3
                best_h = h3.copy()
                best_n = nbest

        # One more pair pass
        rem3 = deadline - time.perf_counter()
        if rem3 > 5.0 and nbest <= 200:
            h4, c4 = pair_swap_aggressive(h_fin, time.perf_counter() + rem3 * 0.8,
                                          max_iter=1000, top_frac=0.03, min_k=8, n_pos=6, n_neg=6)
            if c4 < best_c:
                h_fin = h4
                best_c = c4
                best_h = h4.copy()
                best_n = nbest

    if best_h is None:
        fallback = project_density(np.full(45, 0.5), 22.5)
        return fallback, evaluate_erdos(fallback), 45

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
