import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent_step(h, i, n):
    """Update coordinate i to minimize C5."""
    h_old = h[i]
    h[i] = 0.0
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = 1.0
    corr1 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = h_old
    
    A = corr0  # value at t=0
    D = corr1 - corr0  # change per unit t
    
    # Minimize max_l (A[l] + D[l]*t) for t in [0,1]
    # This is a 1D convex minimization
    best_t = h_old
    best_val = np.max(A + h_old * D)
    
    # Coarse grid
    ts = np.linspace(0, 1, 21)
    vals = np.array([np.max(A + t * D) for t in ts])
    best_idx = np.argmin(vals)
    best_val = vals[best_idx]
    best_t = ts[best_idx]
    
    # Refine if not at boundary
    if best_t > 0 and best_t < 1:
        lo = ts[max(0, best_idx - 1)]
        hi = ts[min(20, best_idx + 1)]
        ts_fine = np.linspace(lo, hi, 50)
        vals_fine = np.array([np.max(A + t * D) for t in ts_fine])
        best_idx_f = np.argmin(vals_fine)
        if vals_fine[best_idx_f] < best_val:
            best_val = vals_fine[best_idx_f]
            best_t = ts_fine[best_idx_f]
    
    h[i] = best_t
    return best_t

def coord_descent(h, n, n_sweeps, rng, verbose=False):
    h = h.copy()
    order = rng.permutation(n)
    best_c5 = compute_c5(h)
    
    for sweep in range(n_sweeps):
        for i in order:
            coord_descent_step(h, i, n)
        
        # Project to feasible set
        h = make_feasible(h, n)
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
        elif verbose and sweep % 20 == 0:
            pass
    
    return h, best_c5

def slsqp_optimize(h, n, max_iter=3000, beta=100):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        cmax = np.max(corr)
        return cmax + (1.0/beta) * np.log(np.sum(np.exp(beta * (corr - cmax))))
    
    def obj_grad(x):
        # Numerical gradient
        eps = 1e-7
        g = np.zeros_like(x)
        f0 = obj(x)
        for i in range(len(x)):
            x[i] += eps
            f1 = obj(x)
            x[i] -= eps
            g[i] = (f1 - f0) / eps
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': 1e-14})
        h_out = make_feasible(res.x, n)
        return h_out, compute_c5(h_out)
    except Exception:
        return None, None

def bashou(h, n, rng, budget_s, cooldown=0.98, max_perturb=0.15):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current = h.copy()
    t_start = time.time()
    
    while time.time() - t_start < budget_s - 30:
        # Coordinate descent
        h_cd, c5_cd = coord_descent(current, n, n_sweeps=30, rng=rng)
        if c5_cd < best_c5 - 1e-12:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        current = h_cd.copy()
        
        # SLSQP polish
        h_sl, c5_sl = slsqp_optimize(current, n, max_iter=1000, beta=100)
        if h_sl is not None and c5_sl < best_c5 - 1e-12:
            best_c5 = c5_sl
            best_h = h_sl.copy()
            current = h_sl.copy()
        
        # Perturb
        n_pert = max(3, int(0.05 * n))
        idxs = rng.choice(n, size=n_pert, replace=False)
        current = current.copy()
        for idx in idxs:
            current[idx] += rng.uniform(-max_perturb, max_perturb)
        current = np.clip(current, 0, 1)
        current = make_feasible(current, n)
        
        max_perturb *= cooldown
    
    return best_h, best_c5

def interpolate_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0, 2, n_src, endpoint=False) + 1.0/n_src
    x_dst = np.linspace(0, 2, n_dst, endpoint=False) + 1.0/n_dst
    return np.interp(x_dst, x_src, h_src)
