        return smooth_max_grad_only(x, n, tau)

    def eq_con(x):
        return x.sum() - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    try:
        res = minimize(
            s_obj, h0.copy(), jac=s_jac, method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': eq_con, 'jac': eq_jac}],
            options={'maxiter': max_iter, 'ftol': 1e-14, 'disp': False}
        )
        h_new = np.clip(res.x, 0.0, 1.0)
        h_new = fix_sum(h_new, n)
    except Exception:
        h_new = h0.copy()
    return h_new


def thue_morse_init(n):
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = (tm[i >> 1] ^ (i & 1)) & 1
    h = tm.astype(float)
    h = fix_sum(h, n)
    return h


def modular_init(n, period, on_frac):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.zeros(n)
    for i in range(n):
        phase = (x[i] / period) % 1.0
        if phase < on_frac:
            h[i] = 1.0
    h = fix_sum(h, n)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None and np.size(initial_h_values) > 0:
        best_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(180, 0.5)

    best_c5 = compute_c5(best_h)
    n_parent = len(best_h)

    resolutions = []
    for r in [n_parent, 2 * n_parent, 4 * n_parent]:
        if r >= 40 and r <= 2000:
            resolutions.append(r)
    # Add some standard resolutions
    for r in [180, 360, 720]:
        if r not in resolutions and 40 <= r <= 2000:
            resolutions.append(r)
    resolutions = sorted(set(resolutions))

    # Phase 1: Optimize at parent resolution with L-BFGS-B
    elapsed = time.time() - t0
    h_work = fix_sum(best_h, n_parent)
    c5_work = compute_c5(h_work)
    if c5_work < best_c5:
        best_c5 = c5_work
        best_h = h_work.copy()

    for tau in [0.1, 0.05, 0.02, 0.01, 0.005]:
        if time.time() - t0 > budget_s * 0.25:
            break
        lam = 500.0
        h_work = run_lbfgs(h_work, n_parent, tau, lam, 3000, budget_s * 0.25)
        c5_work = compute_c5(h_work)
        if c5_work < best_c5 - 1e-12:
            best_c5 = c5_work
            best_h = h_work.copy()

    # SLSQP polish at parent resolution
    if time.time() - t0 < budget_s * 0.30:
        h_work = slsqp_refine(h_work, n_parent, 0.003, 1000, t0, budget_s * 0.30)
        c5_work = compute_c5(h_work)
        if c5_work < best_c5 - 1e-12:
            best_c5 = c5_work
            best_h = h_work.copy()

    # Phase 2: Higher resolution optimization
    for n in resolutions:
        if n <= n_parent:
            continue
        elapsed = time.time() - t0
        if elapsed > budget_s * 0.75:
            break

        # Upsample best solution
        h_hi = upsample_to(best_h, n)
        h_hi = fix_sum(h_hi, n)
        c5_hi = compute_c5(h_hi)

        # L-BFGS-B optimization at higher resolution
        for tau in [0.08, 0.04, 0.02, 0.01, 0.005, 0.002]:
            if time.time() - t0 > budget_s * 0.78:
                break
            lam = 1000.0
            h_hi = run_lbfgs(h_hi, n, tau, lam, 2000, budget_s * 0.78)
            c5_hi = compute_c5(h_hi)
            if c5_hi < best_c5 - 1e-12:
                best_c5 = c5_hi
                best_h = h_hi.copy()

        # Projected gradient for additional refinement
        if time.time() - t0 < budget_s * 0.80:
            for tau in [0.01, 0.005]:
                h_hi = projected_gradient(h_hi, n, tau, 500, budget_s * 0.80, t0)
                c5_hi = compute_c5(h_hi)
                if c5_hi < best_c5 - 1e-12:
                    best_c5 = c5_hi
                    best_h = h_hi.copy()

        # SLSQP final polish at this resolution
        if time.time() - t0 < budget_s * 0.82:
            h_hi = slsqp_refine(h_hi, n, 0.002, 800, t0, budget_s * 0.82)
            c5_hi = compute_c5(h_hi)
            if c5_hi < best_c5 - 1e-12:
                best_c5 = c5_hi
                best_h = h_hi.copy()

    # Phase 3: Diverse initializations at best resolution
    n_best = len(best_h)
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.90:
        # Try Thue-Morse
        h_try = thue_morse_init(n_best)
        h_try = run_lbfgs(h_try, n_best, 0.02, 1000.0, 1000, budget_s * 0.90)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Try modular patterns
    if time.time() - t0 < budget_s * 0.92:
        for period_frac in [0.25, 1.0/3.0, 0.2, 0.4, 0.125]:
            h_try = modular_init(n_best, 2.0 * period_frac, 0.5)
            h_try = run_lbfgs(h_try, n_best, 0.02, 1000.0, 500, budget_s * 0.92)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
            if time.time() - t0 > budget_s * 0.92:
                break

    # Phase 4: Random restarts from best
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.96:
        n_restarts = 8
        for r in range(n_restarts):
            if time.time() - t0 > budget_s * 0.96:
                break
            noise = rng.normal(0, 0.05, n_best)
            h_pert = np.clip(best_h + noise, 0.0, 1.0)
            h_pert = fix_sum(h_pert, n_best)
            h_pert = run_lbfgs(h_pert, n_best, 0.01, 1000.0, 1000, budget_s * 0.96)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5 - 1e-12:
                best_c5 = c5_pert
                best_h = h_pert.copy()

    # Phase 5: Final SLSQP polish
    if time.time() - t0 < budget_s * 0.99:
        h_final = slsqp_refine(best_h, n_best, 0.001, 2000, t0, budget_s * 0.99)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
