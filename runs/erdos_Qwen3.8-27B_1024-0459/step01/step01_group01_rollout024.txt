def smooth_max_grad(corr, tau):
    """Returns gradient of smooth_max w.r.t. corr (i.e., softmax weights)"""
    m = corr.max()
    weights = np.exp((corr - m) / tau)
    weights /= weights.sum()
    return weights  # These are the softmax probabilities

def corr_grad(h, n, weights, dx):
    """Compute gradient of weighted sum of correlations w.r.t. h"""
    grad = np.zeros(n)
    # weights indexed as [0, ..., 2n-2] corresponding to lags [-(n-1), ..., n-1]
    for lag_idx in range(2*n-1):
        w = weights[lag_idx]
        if w < 1e-12:
            continue
        m = lag_idx - (n-1)  # actual lag
        # ∂corr_m/∂h[j] = (1-h[j+m]) for 0≤j+m<n, minus h[j-m] for 0≤j-m<n
        j = np.arange(n)

        jm2 = j - m
        mask2 = (jm2 >= 0) & (jm2 < n)
        grad[mask2] -= w * h[jm2[mask2]]

from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        per = diff / cnt
        h[mask] = np.clip(h[mask] + per, 0.0, 1.0)
    return h

def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def smooth_max_and_weights(corr, tau):
    m = corr.max()
    exps = np.exp((corr - m) / tau)
    s = exps.sum()
    val = (m + tau * np.log(s))
    weights = exps / s
    return val, weights

def corr_gradient(h, n, weights):
    """Compute sum_k weights[k] * d(corr_k)/d(h), shape (n,)"""
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-15:
            continue
        m = idx - (n - 1)
        if m >= 0:
            # (1-h[j+m]) for j=0..n-1-m
            if n - m > 0:
                grad[:n-m] += w * (1.0 - h[m:])
            # -h[j-m] for j=m..n-1
            if m > 0:
                grad[m:] -= w * h[:n-m]
        else:
            p = -m
            # (1-h[j-p]) for j=p..n-1
            grad[p:] += w * (1.0 - h[:n-p])
            # -h[j+p] for j=0..n-1-p
            if n - p > 0:
                grad[:n-p] -= w * h[p:]
    return grad

def fista_subgradient(h0, n, t0, time_limit_frac, budget_s, max_iters=50000):
    """FISTA with decreasing tau for smooth max of correlations."""
    dx = 2.0 / n
    h = h0.copy()
    h = fix_sum(h, n)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Tau schedule: start large, decrease
    tau_start = 5.0
    tau_end = 0.05
    n_tau_switches = 5
    tau_per_phase = max_iters // n_tau_switches
    
    for t in range(max_iters):
        if time.time() - t0 > budget_s * time_limit_frac:
            break
        
        # Update tau
        phase = min(t // tau_per_phase, n_tau_switches - 1)
        tau = tau_start * (tau_end / tau_start) ** (phase / n_tau_switches)
        
        # Compute correlation
        corr = np.correlate(h, 1.0 - h, mode="full")
        sm_val, weights = smooth_max_and_weights(corr, tau)
        
        # Compute gradient
        grad = corr_gradient(h, n, weights) * dx
        
        # FISTA update
        t_fista = t + 1
        y = h + ((t_fista - 1) / (t_fista + 2)) * (h - h_prev) if t > 0 else h.copy()
        y = np.clip(y, 0, 1)
        
        # Recompute gradient at y for FISTA (or use h's gradient - simpler)
        # For simplicity, use h's gradient (standard proximal gradient)
        step = 1.0 / (np.linalg.norm(grad, ord=np.inf) + 1e-10)
        # Actually, the Lipschitz constant of the smooth max gradient is bounded by 2*n/dx * (1/tau)
        # Let's use a fixed step based on problem scale

        step = min(step, 0.5)
        step = max(step, 0.001)
        
        h_new = h - step * grad
        h_new = fix_sum(h_new, n)
        h_prev = h
        h = h_new
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        per = diff / cnt
        h[mask] = np.clip(h[mask] + per, 0.0, 1.0)
    return h

def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def smooth_max_val_weights(corr, tau):
    m = corr.max()
    shifted = (corr - m) / tau
    max_shifted = shifted.max()
    exps = np.exp(shifted - max_shifted)
    s = exps.sum()
    val = m + tau * (max_shifted + np.log(s))
    weights = exps / s
    return val, weights

def weighted_corr_grad(h, n, weights):
    n_idx = 2 * n - 1
    grad = np.zeros(n)
    for idx in range(n_idx):
        w = weights[idx]
        if w < 1e-15:
            continue
        m = idx - (n - 1)
        if m >= 0:
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
            if m > 0:
                grad[m:] -= w * h[:n - m]
        else:
            p = -m
            grad[p:] += w * (1.0 - h[:n - p])
            if n - p > 0:
                grad[:n - p] -= w * h[p:]
    return grad

        sm_val, weights = smooth_max_val_weights(corr, tau)
        grad = weighted_corr_grad(h, n, weights) * dx
        
        # Momentum

def slsqp_optimize(h0, n, t0, deadline, tau=0.1):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    constraint = {'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}
    
    def obj(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = corr.max()
        return (m + tau * np.log(np.sum(np.exp((corr - m) / tau)))) * dx
    
    try:

                       constraints=[constraint],
