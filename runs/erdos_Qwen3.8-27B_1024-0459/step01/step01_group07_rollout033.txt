def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = (h < 1.0 - 1e-15)
            if not np.any(free):
                break
            add = diff / np.sum(free)
            h[free] = np.minimum(h[free] + add, 1.0)
        else:
            free = (h > 1e-15)
            if not np.any(free):
                break
            add = diff / np.sum(free)
            h[free] = np.maximum(h[free] + add, 0.0)
    h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    s = np.sum(h)
    if abs(s - target) > 1e-12:
        # Distribute remaining error
        diff = target - s
        free = (h > 1e-8) & (h < 1 - 1e-8)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0, 1)
    return h

def slsqp_optimize(h0, n_points, maxiter=300, ftol=1e-10):
    n = n_points
    target = n / 2.0
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return compute_c5(x, n)
    
    def jac(x):
        dx = 2.0 / n
        conv = np.correlate(x, 1.0 - x, mode='full')
        k_max = int(np.argmax(conv))
        ka = k_max - (n - 1)
        g = np.zeros(n)
        lo = max(0, ka)
        hi = min(n, n + ka)
        if lo < hi:
            g[lo:hi] += (1.0 - x[lo-ka:hi-ka])
        lo2 = max(0, -ka)
        hi2 = min(n, n - ka)
        if lo2 < hi2:
            g[lo2:hi2] += (-x[lo2+ka:hi2+ka])
        return g * dx
    
    try:
        res = minimize(obj, h0.copy(), jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x, compute_c5(res.x, n)
    except Exception:
        return h0.copy(), compute_c5(h0, n)

def binary_local_search(h_init, n, n_iter=5000, t_init=0.1, t_final=0.001, seed=42):
    """Simulated annealing on binary assignments"""
    rng = np.random.RandomState(seed)
    target = n // 2
    # Start from binary version of h_init
    h_b = np.zeros(n)
    idx = np.argsort(-h_init)[:target]
    h_b[idx] = 1.0
    
    cur_val = compute_c5(h_b, n)
    best_h = h_b.copy()
    best_val = cur_val
    
    t = t_init
    cooling = (t_final / t_init) ** (1.0 / n_iter)
    
    for it in range(n_iter):
        t *= cooling
        # Flip a random pair (one 1 -> 0, one 0 -> 1)
        ones = np.where(h_b == 1.0)[0]
        zeros = np.where(h_b == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h_b[i] = 0.0
        h_b[j] = 1.0
        new_val = compute_c5(h_b, n)
        
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_b.copy()
        else:
            h_b[i] = 1.0
            h_b[j] = 0.0
    
    return best_h, best_val

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx


def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = (h < 1.0 - 1e-14)
            if not np.any(free):
                break
            add = diff / np.sum(free)
            h[free] = np.minimum(h[free] + add, 1.0)
        else:
            free = (h > 1e-14)
            if not np.any(free):
                break
            add = diff / np.sum(free)
            h[free] = np.maximum(h[free] + add, 0.0)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-12:
        diff = target - s
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0, 1)
    return h


def hard_max_grad(h, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(conv))
    ka = k_max - (n - 1)
    grad = np.zeros(n)
    if ka >= 0:
        if ka < n:
            grad[ka:] += (1.0 - h[:n - ka])
        if n - ka > 0:
            grad[:n - ka] += (-h[ka:])
    else:
        p = -ka
        if n - p > 0:
            grad[:n - p] += (1.0 - h[p:])
        if p < n:
            grad[p:] += (-h[:n - p])
    return grad * dx


def smooth_max_obj(h, n, temp):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(conv)
    if temp < 1e-10:
        return m
    shifted = conv - m
    return m + temp * np.log(np.sum(np.exp(shifted / temp)))


def smooth_max_grad(h, n, temp):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(conv)
    
    if temp < 1e-10:
        return hard_max_grad(h, n)
    
    shifted = conv - m
    w = np.exp(shifted / temp)
    w /= w.sum()
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-12:
            continue
        ka = idx - (n - 1)
        if ka >= 0:
            if ka < n:
                grad[ka:] += wi * (1.0 - h[:n - ka])
            if n - ka > 0:
                grad[:n - ka] += wi * (-h[ka:])
        else:
            p = -ka
            if n - p > 0:
                grad[:n - p] += wi * (1.0 - h[p:])
            if p < n:
                grad[p:] += wi * (-h[:n - p])
    return grad * dx


def proj_grad_descent_smooth(h0, n, temp, n_iter=1000, lr_init=0.03, time_limit=None, t_start=None):

    best_h = h.copy()
    best_val = compute_c5(h, n)
    lr = lr_init
    prev_obj = smooth_max_obj(h, n, temp)

def slsqp_optimize(h0, n_points, maxiter=300, ftol=1e-10):
    n = n_points
    target = n / 2.0
    
    def obj(x):
        return compute_c5(x, n)
    
    def jac(x):
        return hard_max_grad(x, n)
    
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0.copy(), jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_out = project_feasible(res.x, n)
        return h_out, compute_c5(h_out, n)
    except Exception:
        return h0.copy(), compute_c5(h0, n)


def binary_sa(h_init, n, n_iter=4000, t_init=0.08, t_final=1e-5, seed=42, time_limit=None, t_start=None):
    rng = np.random.RandomState(seed)
    target = n // 2
    h_b = np.zeros(n)
    idx = np.argsort(-h_init)[:target]
    h_b[idx] = 1.0
    
    cur_val = compute_c5(h_b, n)
    best_h = h_b.copy()
    best_val = cur_val
    
    if t_final / t_init > 0:
        cooling = (t_final / t_init) ** (1.0 / max(n_iter, 1))
    else:
        cooling = 0.999
    
    t = t_init
    for it in range(n_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        t *= cooling
        
        ones = np.where(h_b == 1.0)[0]
        zeros = np.where(h_b == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h_b[i] = 0.0
        h_b[j] = 1.0
        new_val = compute_c5(h_b, n)
        
        delta = new_val - cur_val
        if delta < 0 or (t > 0 and rng.random() < np.exp(-delta / max(t, 1e-12))):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_b.copy()
        else:
            h_b[i] = 1.0
            h_b[j] = 0.0
    
    return best_h, best_val


def structured_init(n, rng):
    """Generate diverse structured initializations"""
    strat = rng.randint(8)
    h = np.zeros(n)
    if strat == 0:
        h = rng.uniform(0, 1, n)
    elif strat == 1:
        h = rng.binomial(1, 0.5, n).astype(float)
    elif strat == 2:
        # Alternating blocks
        blk = rng.randint(2, 8)
        on = 0
        for i in range(n):
            pos_in_block = i % (2 * blk)
            if pos_in_block < blk:
                h[i] = 1.0
            else:
                h[i] = 0.0
        h += rng.normal(0, 0.15, n)
    elif strat == 3:
        # Sine-based
        freq = rng.uniform(1.5, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.3 * np.sin(np.pi * freq * np.arange(n) / n + phase)
    elif strat == 4:
        # Quadratic pattern
        x = np.linspace(0, 1, n)
        h = (x < 0.5) * 1.0
        h = np.roll(h, rng.randint(0, n))
        h += rng.normal(0, 0.2, n)
    elif strat == 5:
        # Multi-frequency
        h = 0.5
        for _ in range(rng.randint(2, 5)):
            f = rng.uniform(1, 8)
            p = rng.uniform(0, 2 * np.pi)
            h += rng.uniform(-0.2, 0.2) * np.cos(np.pi * f * np.arange(n) / n + p)
    elif strat == 6:
        # Random permutation of half-ones
        perm = rng.permutation(n)
        h[perm[:n // 2]] = 1.0
        h += rng.normal(0, 0.1, n)
    else:
        # Triangular wave
        x = np.linspace(0, 1, n)
        h = 1.0 - np.abs(2 * x - 1)
        h = (h > 0.5).astype(float)
        h += rng.normal(0, 0.1, n)
    
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)
