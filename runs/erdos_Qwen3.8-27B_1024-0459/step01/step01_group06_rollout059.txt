    """Project v onto {0 ≤ x ≤ 1, sum(x) = target_sum} via bisection on lambda"""
    lo = -1.0
    hi = 1.0
    # Actually λ can be larger depending on v
    lo = -1.0 - np.max(v) + target_sum/n
    hi = 1.0 - np.min(v) + target_sum/n
    # Simpler: just use wide bounds
    lo = -max(v) + target_sum/n - 1
    hi = -min(v) + target_sum/n + 1

        mid = (lo + hi) / 2
        h = np.clip(v + mid, 0, 1)
        s = h.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    
    return np.clip(v + (lo + hi) / 2, 0.0, 1.0)

def project_box_sum(v, target):
    """Project v onto {0 <= x_i <= 1, sum(x) = target} using bisection."""
    lo = -float(np.max(v)) - target / len(v) - 1.0
    hi = float(-np.min(v)) + target / len(v) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(v + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)


def grad_soft_max(h, n, beta):
    c = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * c
    mx = np.max(scaled)
    w = np.exp(scaled - mx)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(c)] = 1.0
    else:
        w /= ws
    dx = 2.0 / n

Actually, one more important thing: the Nesterov momentum formula. The standard Nesterov accelerated gradient is:

x_{k+1} = project(y_k - step * grad(y_k))

Or equivalently:
