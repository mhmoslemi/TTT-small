def binary_sa(h_start, n, state, time_limit, rng):
    """Simulated annealing for binary optimization."""
    # Initialize binary
    h = make_feasible(h_start, n)
    # Round to binary with projection
    h_sorted = np.sort(h)[::-1]
    target = n // 2
    h_bin = np.zeros(n, dtype=np.int8)
    h_bin[:target] = 1
    # Assign based on ranking
    order = np.argsort(h)[::-1]
    h_bin[order[:target]] = 1
    
    # Compute scores
    def bin_scores(hb):
        s = np.zeros(n, dtype=float)
        hf = hb.astype(float)
        for d in range(1, n):
            s[d] = np.dot(hf[:n-d], 1.0-hf[d:]) + np.dot(hf[d:], 1.0-hf[:n-d])
        return s, float(np.max(s))
    
    scores, cur_max = bin_scores(h_bin)
    best_max = cur_max
    best_h = h_bin.copy()
    
    if state is not None:
        state.update(h_bin.astype(float), best_max * 2.0/n, n)
    
    t0 = time.time()
    t_end = t0 + time_limit
    if _DEADLINE:
        t_end = min(t_end, _DEADLINE - 5)
    
    T0 = 3.0
    T_end = 0.1
    n_iters = 0
    max_iters = 500000
    
    stall = 0
    stall_limit = 5000
    
    while n_iters < max_iters and time.time() < t_end:
        check_time()
        n_iters += 1
        # Temperature
        frac = max(0, 1.0 - n_iters / max_iters)
        T = T0 * (T_end/T0)**frac
        
        # Pick a random swap: one 1 and one 0
        ones = np.flatnonzero(h_bin == 1)
        zeros = np.flatnonzero(h_bin == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        # Try swap
        h_bin[i] = 0
        h_bin[j] = 1
        
        new_scores, new_max = bin_scores(h_bin)
        delta = new_max - cur_max
        
        if delta <= 0 or rng.random() < math.exp(-delta / T):
            cur_max = new_max
            scores = new_scores
            if cur_max < best_max:
                best_max = cur_max
                best_h = h_bin.copy()
                stall = 0
                if state is not None:
                    state.update(h_bin.astype(float), best_max * 2.0/n, n)
            else:
                stall += 1
        else:
            h_bin[i] = 1
            h_bin[j] = 0
            stall += 1
        
        if stall > stall_limit:
            # Restart with perturbation
            for _ in range(10):
                ones = np.flatnonzero(h_bin == 1)
                zeros = np.flatnonzero(h_bin == 0)
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
                h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            scores, cur_max = bin_scores(h_bin)
            stall = 0
    
    final_c = best_max * 2.0 / n
    if state is not None:
        state.update(best_h.astype(float), final_c, n)
    return best_h.astype(float), float(final_c)

def upsample(h, n_new):
    """Upsample h to length n_new using interpolation."""
    n_old = h.size
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h)

def make_bin_from_cont(h_cont, n):

    h = np.asarray(h_cont, dtype=float)
    target = n // 2
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin
